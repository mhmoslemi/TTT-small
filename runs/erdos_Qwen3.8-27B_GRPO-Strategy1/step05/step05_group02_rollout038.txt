    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s

            return h

    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')

    h_test[i] = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_grad_one_lag(h, n, l):
    g = np.zeros(n)
    if l >= 0:
        cnt = n - l
        g[l:] += (1.0 - h[:cnt])
        g[:cnt] -= h[l:]
    else:
        lp = -l
        cnt = n - lp
        g[lp:] += (1.0 - h[:cnt])
        g[:cnt] -= h[lp:]
    return g

    return best_h, best_c5

def coord_single_fast(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    h1 = h0.copy()
    h1[i] = 1.0
    c1 = np.correlate(h1, 1.0 - h1, mode='full')
    coefs = c1 - c0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(c0 + c * coefs)
    fd = np.max(c0 + d * coefs)
    for _ in range(45):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a); fc = np.max(c0 + c * coefs)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a); fd = np.max(c0 + d * coefs)
    return (a + b) / 2.0

def coord_descent(h, n, t_end, max_sweeps=5000, rng=None):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    if rng is None:
        rng = np.random.RandomState(0)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                imp += 1
        if imp == 0:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    return best_h, best_c5

def make_perturbation(n, cool, rng, intensity=1.0):
    mod = rng.randint(0, 6)
    if mod == 0:
        bs = rng.randint(6, max(7, n // 3))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.05 + 0.25 * cool) * intensity * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 8))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.04 + 0.12 * cool) * intensity * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(6, max(7, n // 3))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.08 + 0.3 * cool) * intensity * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.02 + 0.08 * cool) * intensity * rng.standard_normal(n)
    elif mod == 4:
        half = n // 2
        p = np.zeros(n)
        p[:half] = (0.06 + 0.15 * cool) * intensity * rng.standard_normal(half)
        p[half:] = -(0.06 + 0.15 * cool) * intensity * rng.standard_normal(n - half)
    else:
        p = np.zeros(n)
        for _ in range(3):
            bs = rng.randint(4, max(5, n // 5))
            st = rng.randint(0, n - bs)
            p[st:st + bs] += (0.05 + 0.2 * cool) * intensity * rng.standard_normal(bs)
    return p

def try_flipped(h, n, target):
    h_f = project_h(1.0 - h, target)
    c5_f = compute_c5(h_f)
    c5_orig = compute_c5(h)
    if c5_f < c5_orig:
        return h_f, c5_f
    return h, c5_orig

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    target = n / 2.0
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    best_h, best_c5 = try_flipped(best_h, n, target)

    # Phase 1: Coordinate descent + subgradient hybrid (35% budget)
    t_end1 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t_end1, max_sweeps=2000, rng=rng)
    best_h, best_c5 = try_flipped(best_h, n, target)
    # Subgradient to escape any remaining flat spots
    best_h, best_c5 = subgrad_descent(best_h, n, t_end1, max_iter=50000)
    best_h, best_c5 = try_flipped(best_h, n, target)

    # Phase 2: Basin hopping (40% budget)
    t_end2 = t0 + budget_s * 0.75
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    bh_rng = np.random.RandomState(seed + 1)
    for it in range(n_bh):
        if elapsed() > budget_s * 0.75:
            break
        cool = 1.0 - 0.85 * (it / n_bh)
        int_fac = 1.0 + 0.5 * (1.0 - cool)
        pert = make_perturbation(n, cool, bh_rng, int_fac)
        h_pert = project_h(work_h + pert, target)
        time_per_bh = max(0.5, (t_end2 - t0 - elapsed()) / max(n_bh - it, 1))
        t_local = t0 + elapsed() + time_per_bh * 0.7
        hc, c5v = coord_descent(h_pert, n, t_local, max_sweeps=300, rng=bh_rng)
        # Also try flipped version
        hc, c5v = try_flipped(hc, n, target)
        # Occasional subgradient boost
        if it % 8 == 7 and elapsed() < budget_s * 0.70:
            hc, c5v = subgrad_descent(hc, n, min(t0 + elapsed() + time_per_bh * 0.2, t_end2), max_iter=20000)
            hc, c5v = try_flipped(hc, n, target)
        temp = 0.0015 * cool + 0.0002
        if c5v < work_c5 or (c5v - work_c5 < temp * 3 and bh_rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9))):
            work_h = hc
            work_c5 = c5v
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
    best_h, best_c5 = try_flipped(best_h, n, target)

    # Phase 3: Multi-start from different initial structures (10% budget)
    t_end3 = t0 + budget_s * 0.85
    ms_rng = np.random.RandomState(seed + 2)
    n_starts = 5
    for s in range(n_starts):
        if elapsed() > budget_s * 0.85:
            break
        if s == 0:
            start = np.concatenate([np.ones(n // 4), np.zeros(n - n // 4)])
        elif s == 1:
            start = (np.sin(np.pi * np.arange(n) / n) ** 2).astype(float)
        elif s == 2:
            start = ms_rng.rand(n)
        elif s == 3:
            start = np.zeros(n)
            start[n//4:3*n//4] = 1.5
        else:
            start = project_h(best_h + 0.3 * ms_rng.standard_normal(n), target)
        start = project_h(start, target)
        t_soc = t0 + elapsed() + max(0.5, (t_end3 - t0 - elapsed()) / max(n_starts - s, 1)) * 0.6
        hc, c5v = coord_descent(start, n, t_soc, max_sweeps=300, rng=ms_rng)
        hc, c5v = try_flipped(hc, n, target)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Final intensive coordinate descent (12% budget)
    t_end4 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t_end4, max_sweeps=3000, rng=np.random.RandomState(seed + 3))
    best_h, best_c5 = try_flipped(best_h, n, target)

    # Phase 5: SLSQP polish with smooth max
    if elapsed() < budget_s * 0.99:
        def smooth_obj(x):
            dx_v = 2.0 / n
            corr_v = np.correlate(x, 1.0 - x, mode='full')
            T_s = 50000.0
            m_v = np.max(corr_v)
            sh = np.clip(T_s * (corr_v - m_v), -50.0, 50.0)
            return (m_v + np.log(np.sum(np.exp(sh))) / T_s) * dx_v

        def eq_con(x):
            return x.sum() - target

        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': eq_con}],
                          options={'maxiter': 300, 'ftol': 1e-18})
            hc = project_h(np.clip(res.x, 0, 1), target)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
        # Final coord descent
        if elapsed() < budget_s * 0.995:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.998, max_sweeps=200, rng=np.random.RandomState(seed+4))
            best_h, best_c5 = try_flipped(best_h, n, target)

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
