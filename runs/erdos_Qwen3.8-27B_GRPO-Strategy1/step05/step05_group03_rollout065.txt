For a fixed pair (a,b), the problematic lag is d = b-a (or a-b). There's at most 2 such lags per pair.

For those specific (a,b,k) triples, the first-order formula is off. But since there are only ~2 entries per pair affected, and we're taking the MAX over all k, the error is small and unlikely to affect the argmin.

Let me just go with the first-order approximation and do exact verification only for the top few candidates. This should be fast enough.

def swap_round(h, n):
    """One round of swap annealing: find best swap, return new h and c5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = corr.max() * dx
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    m = len(ones)  # should be n//2
    z = len(zeros)
    n_corr = 2 * n - 1
    k_range = np.arange(n_corr)
    
    # Compute flip effects for ones (1→0)
    a_idx = ones[:, None]  # (m, 1)
    # d = k - (n-1)
    d = k_range[None, :] - (n - 1)  # (1, n_corr)
    
    # Effect on corr[k] from flipping h[a] 1→0:
    # -(1-h[a+d]) * [0<=a+d<n] + h[a-d] * [0<=a-d<n], with d=0 entry = 0
    idx1 = a_idx + d  # (m, n_corr): a + d = a + k - (n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    val1 = np.where(valid1, -(1.0 - h[np.clip(idx1, 0, n-1)]), 0.0)
    
    idx2 = a_idx - d  # (m, n_corr): a - d = a - k + (n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    val2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    v = val1 + val2  # (m, n_corr)
    v[:, n-1] = 0.0  # fix d=0
    
    # Effect on corr[k] from flipping h[b] 0→1:
    b_idx = zeros[:, None]  # (z, 1)
    idx3 = b_idx + d
    valid3 = (idx3 >= 0) & (idx3 < n)
    val3 = np.where(valid3, (1.0 - h[np.clip(idx3, 0, n-1)]), 0.0)
    
    idx4 = b_idx - d
    valid4 = (idx4 >= 0) & (idx4 < n)
    val4 = np.where(valid4, -h[np.clip(idx4, 0, n-1)], 0.0)
    
    w = val3 + val4  # (z, n_corr)
    w[:, n-1] = 0.0
    
    # For swap (ia, jb): new_corr ≈ corr + v[ia] + w[jb]
    # Want: min_{ia,jb} max_k (corr[k] + v[ia,k] + w[jb,k])
    base = corr[None, :] + v  # (m, n_corr)
    # new_vals[ia, jb, k] = base[ia, k] + w[jb, k]
    # This is (m, z, n_corr) = (80, 80, 319) = 2M entries, 16MB
    new_vals = base[:, None, :] + w[None, :, :]
    new_c5s = new_vals.max(axis=2)  # (m, z)
    
    flat = new_c5s.argmin()
    ia = flat // z
    jb = flat % z
    a_pos = ones[ia]
    b_pos = zeros[jb]
    approx_c5 = new_c5s[ia, jb] * dx
    
    return a_pos, b_pos, approx_c5


def swap_anneal(h, n, t_end):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    
    T0 = 0.004
    T_min = 0.00005
    it = 0
    patience = 0
    max_pat = 3000
    
    while time.time() < t_end:
        it += 1
        frac = min(1.0, it / 5000.0)
        T = T_min + (T0 - T_min) * (1 - frac) ** 1.5
        
        a_pos, b_pos, approx_c5 = swap_round(cur_h, n)
        
        # Exact evaluation
        cur_h[a_pos] = 0.0
        cur_h[b_pos] = 1.0
        new_c5 = compute_c5(cur_h)
        
        accept = new_c5 <= cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10))
        
        if not accept:
            cur_h[a_pos] = 1.0
            cur_h[b_pos] = 0.0
            patience += 1
        else:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                patience = 0
            else:
                patience += 1
        
        # Kick
        if patience > max_pat:
            patience = 0
            for _ in range(3):
                i = np.random.choice(np.where(cur_h == 1)[0])
                j = np.random.choice(np.where(cur_h == 0)[0])
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            cur_c5 = compute_c5(cur_h)
    
    return best_h, best_c5

For the final coordinate descent, I'll use a simpler version than the previous code's expensive per-coordinate optimization. Instead, I'll do:
- For each coordinate, compute the optimal value given all others fixed (this is a 1D concave maximization → actually it's a minimization of a piecewise-linear convex function, so the optimum is at a vertex or where the max is flat)
- Actually, for continuous h, fixing all other coordinates, the objective as a function of h[i] is:
  max_k (corr_k + h[i] * coef_k)
  where coef_k depends on the position i.
  This is the max of linear functions → convex. Minimum of a convex function on [0,1] is at a point where the subgradient contains 0, or at a boundary.

This is actually a nice structure! The optimal h[i] (given others fixed) is where the max of the active lags is minimized. This can be found by binary search or by finding the point where two or more lags achieve the same max value.

def coord_opt(h, n, i):
    """Find optimal h[i] given all others fixed."""
    # corr_k(h) = corr_k_base + h[i] * coef_k for each k
    # where corr_k_base = corr_k(h) with h[i] set to 0
    # and coef_k = contribution of h[i] to corr_k
    
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    
    # coef[k] = contribution of h[i] to corr[k]
    # corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
    # h[i] contributes:
    # as h[j] when j=i: h[i]*(1-h[i+k-(n-1)]) → coef from this term: (1-h[i+k-(n-1)])
    # as (1-h[j]) when j+k-(n-1)=i, j=i-k+(n-1): -h[i-k+(n-1)] * h[i] → wait
    # Actually: h[j]*(1-h[j+d]) where d=k-(n-1)
    # h[i] appears as:
    #   first factor: j=i, term = h[i]*(1-h[i+d]) → linear in h[i] with coef (1-h[i+d])
    #   second factor: j+d=i, j=i-d, term = h[i-d]*(1-h[i]) → linear in h[i] with coef -h[i-d]
    #   d=0: both refer to same term h[i]*(1-h[i]) → quadratic! Coef is (1-h[i]) - h[i]... 
    #   Actually for d=0: h[i]*(1-h[i]), which is NOT linear in h[i]. It's h[i]-h[i]².
    #   So the objective for d=0 is NOT linear in h[i].
    
    # Hmm, this complicates things. Let me just use minimize_scalar.
    
    def f(hi):
        h[i] = hi
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = corr.max()
        h[i] = old
        return val
    
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 50})
    new_val = f(res.x)
    old_val = f(old)
    h[i] = res.x if new_val < old_val else old
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    m = n // 2
    idx = np.argpartition(h, -m)[-m:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def slsqp_smooth(h, T, n, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, t_end):
    """Continuous coordinate descent."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    max_sweeps = 200
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]

                best_h[i] = old
