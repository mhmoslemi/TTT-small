        g[:N - L] -= h[L:N]

            ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0.0
        elif n1 < nh:
            ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1.0
        sd = min(deadline, time.time() + min(40, max(15, 12000.0 / Nt)))
        hr, cr = sa_binary(ht, Nt, sd, T0=0.025, use_grad=True, double_frac=0.3)
        c5r = compute_c5(hr, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_b = np.linspace(0, 2, Nt, endpoint=False)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: L-BFGS from perturbed start (120s)
    if time.time() < dl - 400:
        d1 = min(dl - 350, t0 + 120)
        for r in range(3):
            if time.time() > d1:
                break
            hr = proj01(best_h + np.random.randn(N0) * (0.10 - 0.02 * r), N0 / 2.0)
            hl, cl = lbfgs_run(hr, N0, min(d1, time.time() + 40), T=[25.0, 40.0, 55.0][r], max_outer=2)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 2: SA binary, gradient-guided (350s)
    if time.time() < dl - 250:
        d2 = min(dl - 200, time.time() + 350)
        hb, c2 = sa_binary(best_h, N0, d2, T0=0.03, use_grad=True, double_frac=0.25)
        if c2 < best_c5:
            best_c5, best_h = c2, hb.copy()

    # Phase 3: L-BFGS refinement (100s)
    if time.time() < dl - 150:
        d3 = min(dl - 100, time.time() + 100)
        h3 = proj01(best_h + np.random.randn(N0) * 0.05, N0 / 2.0)
        hl3, cl3 = lbfgs_run(h3, N0, d3, T=40.0, max_outer=3)
        if cl3 < best_c5:
            best_c5, best_h = cl3, hl3.copy()

    # Phase 4: Multi-resolution SA (200s)
    if time.time() < dl - 80:
        best_h, best_c5 = try_resolutions(best_h, best_c5, min(dl - 40, time.time() + 200), seed + 7)

    # Phase 5: Final SA polish (remaining time)
    if time.time() < dl - 5:
        d5 = dl - 2.0
        Nf = len(best_h)
        hb5, c5f = sa_binary(best_h, Nf, d5, T0=0.015, use_grad=True, double_frac=0.35)
        if c5f < best_c5:
            best_c5, best_h = c5f, hb5.copy()

    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, len(best_h))
    fc5 = compute_c5(best_h, len(best_h))
    if fc5b < fc5:
        best_h, best_c5 = fb, fc5b
    return best_h, best_c5, len(best_h)
