def fft_corr(h):
    N = len(h)

    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N, rng):
    hb = (h > 0.5).astype(float)
    nt = N // 2
    n1 = int(hb.sum())
    if n1 == nt:
        return hb
    if n1 > nt:
        ones = np.where(hb == 1)[0]
        hb[rng.choice(ones, n1 - nt, replace=False)] = 0
    else:
        zeros = np.where(hb == 0)[0]
        hb[rng.choice(zeros, nt - n1, replace=False)] = 1
    return hb

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, rng=None):
    h = make_binary(h, N, rng if rng else np.random.RandomState(0))
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_c = P_cur
    best_h = h.copy()
    if rng is None:
        rng = np.random.RandomState(7)
    temp = 0.01
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp * 0.996)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        n_s = max(20, int(len(ones) * sample_frac))
        p_list = rng.choice(ones, min(n_s, len(ones)), replace=False)
        best_new_P = P_cur
        best_move = None
        best_2p = None
        for p in p_list:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_move = (int(p), int(zeros[mi]))
        if len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.3:
            n_try = min(80, max(30, N // 5))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur - D[p1] - D[p2] + D[q1] + D[q2]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_move = None
                    best_2p = (p1, p2, q1, q2)
        if best_2p is not None:
            p1, p2, q1, q2 = best_2p
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
        elif best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
        else:
            if stall > 30 and rng.random() < temp:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 12:
                break
            else:
                stall += 1
                continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-14:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=80, lr0=0.01, beta0=0.9, smax_temp=0.006, perturb_interval=300):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    center = N - 1
    rng = np.random.RandomState(99)
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top_idx = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top_idx].max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        top_vals = corr[top_idx]
        tw = np.exp((top_vals - top_vals.max()) / smax_temp)
        tw /= tw.sum()
        grad = np.zeros(N)
        for i in range(k):
            w = tw[i]
            L = int(top_idx[i] - center)
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w * (1.0 - y[:N - L])
                grad[:N - L] -= w * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - y[M:])
                    grad[M:] -= w * y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.975
            beta = min(0.97, 0.84 + it * 0.00015)
        if it % perturb_interval == 0:
            h += rng.randn(N) * (0.008 + 0.012 * lr / lr0)
            h = project_sum(h, target)
            m *= 0.5
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    xo = np.linspace(0, 2, N_old, endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def downsample(h, N_old, N_new):
    xo = np.linspace(0, 2, N_old, endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, xo, h)
