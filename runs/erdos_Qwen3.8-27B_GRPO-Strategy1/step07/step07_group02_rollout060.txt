def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on shift"""
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-12:

            lo = mid
        else:
            hi = mid

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    target = N // 2
    h = (h > 0.5).astype(float)
    s = int(h.sum())
    while s > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx)] = 0
        s -= 1
    while s < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx)] = 1
        s += 1
    return h

def compute_C(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2*N-1)[None, :] - (N-1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    cmk = np.clip(i_mk, 0, N-1)
    cpk = np.clip(i_pk, 0, N-1)
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[cmk], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[cpk], 0.0)
    return A - B

def c_single(h, N, idx):
    k_vals = np.arange(2*N-1) - (N-1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def local_search(h, N, deadline, max_stuck=150, samp=200, div_per=80, div_n=4):
    h = make_binary(h, N)
    C = compute_C(h, N)
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stuck = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), samp)
        nz = min(len(zeros), samp)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        # Batch evaluation
        best_mv = best_c + 1e-10
        best_i, best_j = -1, -1
        bs = 40
        for i0 in range(0, ns, bs):
            i1 = min(i0+bs, ns)
            vals = corr[None,None,:] + C[os_[i0:i1]][:,None,:] - C[zs][None,:,:]
            nm = vals.max(axis=2)
            fm = nm.min()
            if fm < best_mv:
                best_mv = float(fm)
                mi = np.unravel_index(nm.argmin(), nm.shape)
                best_i, best_j = i0+mi[0], mi[1]
        if best_mv < best_c - 1e-10:
            a = os_[best_i]
            b = zs[best_j]
            h[a] = 0
            h[b] = 1
            C[a,:] = c_single(h, N, a)
            C[b,:] = c_single(h, N, b)
            corr = fft_corr(h)
            if best_mv < best_c - 1e-14:
                best_c = best_mv
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
        if stuck >= div_per:
