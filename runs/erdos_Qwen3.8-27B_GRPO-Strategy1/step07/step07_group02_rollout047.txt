def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def compute_C_row(h, N, idx):
    """Compute the row of the C matrix for position idx."""
    k_vals = np.arange(2*N-1) - (N-1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def local_search(h, N, deadline, max_iters=50000, sample_sz=200):

    h = h.astype(float).copy()

        h[i] = 1
