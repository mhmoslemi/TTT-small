def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def fix_sum_binary(h, target):
    h = h.copy()
    n1 = int(np.round(h.sum()))
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        np.random.shuffle(ones)
        h[ones[:n1-target]] = 0
    elif n1 < target:
        zeros = np.where(h <= 0.5)[0]
        np.random.shuffle(zeros)
        h[zeros[:target-n1]] = 1
    return h

def directed_local_search(h, N, deadline, max_iters=100000):
    """Local search focusing on worst lags."""
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    ones = set(int(x) for x in np.where(h == 1)[0])
    zeros = set(int(x) for x in np.where(h == 0)[0])
    
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters and no_improve < 5000:
        it += 1
        # Find worst lag
        peak_idx = int(np.argmax(corr))
        peak_val = corr[peak_idx]
        
        # Try to find a move that reduces the peak
        # Random pair swap
        if ones and zeros:
            oi = int(np.random.choice(list(ones)))
            zi = int(np.random.choice(list(zeros)))
            # Apply swap
            h[oi] = 0; h[zi] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                ones.discard(oi); ones.add(zi)
                zeros.add(oi); zeros.discard(zi)
                if cur_max * 2.0/N < best_c:
                    best_c = cur_max * 2.0/N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Revert
                h[oi] = 1; h[zi] = 0
                no_improve += 1
    
    return best_h, best_c

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, target):
    h = h.copy()
    n1 = int(np.round(h.sum()))
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        np.random.shuffle(ones)
        h[ones[:n1 - target]] = 0
    elif n1 < target:
        zeros = np.where(h <= 0.5)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - n1]] = 1
    return h

def local_descent(h, N, deadline, max_iters=200000, use_blocks=True):
    h = h.astype(float).copy()

    h = fix_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    it = 0
    no_improv = 0
    stall = 3000
    while time.time() < deadline and it < max_iters:
        it += 1
        if no_improv > stall:

            for _ in range(n_pert):
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi], h[zi] = h[zi], h[oi]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            no_improv = 0
            stall = min(stall * 2, 20000)
            continue
        
        # Try random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        oi = int(ones[np.random.randint(len(ones))])
        zi = int(zeros[np.random.randint(len(zeros))])
        
        h[oi] = 0
        h[zi] = 1
        new_corr = fft_corr(h)
        new_max = float(np.max(new_corr))
        
        if new_max < cur_max - 1e-12:
            corr = new_corr
            cur_max = new_max
            if cur_max * 2.0 / N < best_c:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
            no_improv = 0
        else:
            h[oi] = 1
            h[zi] = 0
            no_improv += 1
        
        # Occasionally try block move
        if use_blocks and it % 5 == 0:
            L = int(np.random.choice([2, 3, 4]))
            # Find blocks
            one_blocks = [i for i in range(N - L + 1) if np.all(h[i:i+L] == 1)]
            zero_blocks = [i for i in range(N - L + 1) if np.all(h[i:i+L] == 0)]
            if one_blocks and zero_blocks:
                ob = one_blocks[np.random.randint(len(one_blocks))]
                zb = zero_blocks[np.random.randint(len(zero_blocks))]
                if abs(ob - zb) >= L:

                            best_h = h.copy()
                        no_improv = 0
                    else:
                        h[ob:ob+L] = 1
                        h[zb:zb+L] = 0
                        no_improv += 1
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def c5_val(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def bin_proj(h, N):
    h = np.asarray(h, dtype=float)
    nh = N // 2
    order = np.argsort(-h)
    out = np.zeros(N)
    out[order[:nh]] = 1.0
    return out

def fix_bin(h, target):
    h = h.copy()
    n1 = int(np.round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h <= 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h

def binary_sa(h, N, deadline, max_it=300000, T0=0.06, T1=1e-5):
    h = h.astype(float).copy()
    h = fix_bin(h, N // 2)
    corr = fft_corr(h)
    cur = float(np.max(corr))
    best_c = cur * 2.0 / N
    best_h = h.copy()
    n_steps = max(1, max_it)
    cool = (T1 / T0) ** (1.0 / n_steps)
    T = T0
    it = 0
    stall = 0
    while time.time() < deadline and it < max_it:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        roi = int(ones[np.random.randint(len(ones))])
        rzi = int(zeros[np.random.randint(len(zeros))])
        h[roi] = 0; h[rzi] = 1
        nc = fft_corr(h)
        nm = float(np.max(nc))
        dE = nm - cur
        if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
            corr = nc; cur = nm
            if cur * 2.0 / N < best_c - 1e-13:
                best_c = cur * 2.0 / N; best_h = h.copy(); stall = 0
            else:
                stall += 1

            corr = fft_corr(h)
            cur = float(np.max(corr))
            T = max(T, T0 * 0.3)
            stall = 0
        T *= cool
    return best_h, best_c

def anti_period_fix(h, N, kshift):

        j = i + kshift
        if h[i] == h[j]:
            if h[i] == 1:
                h[i] = 0
            else:
                h[j] = 1

    return h

    h = np.zeros(N)
    i = 0
    while i < N:
        h[i] = 1.0 if (i + 1) * alpha - i * alpha >= 0 else 0.0
        i += 1
    h = (np.floor(np.arange(N + 1) * alpha) - np.floor(np.arange(N) * alpha)).astype(float)
    h = fix_bin(h, N // 2)
    return h

def freq_init(N, seed):

        v += np.sin(2 * np.pi * f * np.arange(N) / N + p)
