    #   corr[shift] changes by: -(1 - h[p + shift]) for all valid shifts where p+shift in range
    #   Plus: for shifts where i = p - shift (i.e., h[i] multiplies (1-h[p])), 
    #         corr[shift] changes by: -h[p - shift] * (-1) = +h[p-shift]*0...

    pos = np.arange(N)[:, None]  # (N, 1)
    sv = shifts[None, :]  # (1, 2N-1)
    
    # For corr at shift L: sum over x of h[x] * (1 - h[x+L]) where both indices valid
    # When h[p] changes by delta (1->0 means delta=-1):
    # corr changes by: delta * (1 - h[p+L])  [from the h[x] term where x=p]
    #                + h[p-L] * (-delta)    [from the (1-h[x+L]) term where x+L=p, so x=p-L]
    # For delta = -1 (1->0):
    # corr changes by: -(1 - h[p+L]) + h[p-L]
    
    pm = pos - sv  # p - L: index for h[p-L]
    pp = pos + sv  # p + L: index for h[p+L]
    
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)

    hp = np.where(vp, h[pp_c], 0.0)  # h[p+L] or 0
    
    # D[p, s] = change in corr[s] when h[p] goes from 1 to 0
    # = -(1-h[p+L])*vp + h[p-L]*vm
    D = -(1.0 - hp) * vp.astype(float) + hm * vm.astype(float)
    return D

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

    h = h.astype(float).copy()

                grad += (1.0 - 2.0 * y)
            elif L > 0:

        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, 0.9 + it * 0.00005)
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_binary(h, N):
    h = (h >= 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 == nh:
        return h
    if n1 > nh:
        ones = np.where(h == 1)[0]
        idx = np.random.choice(ones, n1 - nh, replace=False)
        h[idx] = 0
    else:
        zeros = np.where(h == 0)[0]
        idx = np.random.choice(zeros, nh - n1, replace=False)
        h[idx] = 1
    return h

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hp) * vp.astype(float) + hm * vm.astype(float)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def greedy_ls(h, N, deadline, max_sweeps=300, sf=1.0):
    h = h.copy()
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 35:
                break
    return best_h, best_c * (2.0 / N)

def perturb(h, N, k):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    k = min(k, len(ones), len(zeros))
    if k == 0:
        return h
    p_idx = np.random.choice(ones, k, replace=False)
    q_idx = np.random.choice(zeros, k, replace=False)
    h[p_idx] = 0.0
    h[q_idx] = 1.0
    return h

def ils_search(h0, N, deadline, k_start=10, k_min=3, mem=15):
    h = h0.copy()
    best_c = float(np.max(fft_corr(h))) * (2.0 / N)
    best_h = h.copy()
    it = 0
    no_improve = 0
    while time.time() < deadline:
        it += 1
        sub_dl = deadline - 0.5
        h_ls = greedy_ls(h, N, sub_dl, max_sweeps=60, sf=0.65)
        c_now = float(np.max(fft_corr(h_ls))) * (2.0 / N)
        if c_now < best_c - 1e-13:
            best_c = c_now
            best_h = h_ls.copy()
            no_improve = 0
            k = max(k_min, k_start - it * (k_start - k_min) // 80)
            h = perturb(best_h, N, k)
        else:
            no_improve += 1
            k = max(k_min, int(k_start * (1.0 + 0.3 * no_improve)))
            h = perturb(best_h, N, min(k, N // 4))
            if no_improve > mem:
                h = perturb(best_h, N, max(10, k_start * 2))
                no_improve = 0
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=40, lr0=0.015):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.993
            beta = min(0.96, 0.88 + it * 0.00006)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    h_bin = ensure_sum_binary(h0, N)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Greedy from initial
    d1 = min(deadline - 850, t0 + 30)
    h1, c5_1 = greedy_ls(h_bin, N, d1, max_sweeps=250, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: ILS main (moderate kicks)
    d2 = min(deadline - 650, t0 + 160)
    h2, c5_2 = ils_search(best_h, N, d2, k_start=10, k_min=3, mem=12)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: ILS from random start
    if time.time() < deadline - 500:
        h_rand = np.zeros(N)
        h_rand[np.random.choice(N, N // 2, replace=False)] = 1.0
        d3 = min(deadline - 480, t0 + 260)
        h3, c5_3 = ils_search(h_rand, N, d3, k_start=14, k_min=5, mem=20)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Diversified ILS restarts from best
    if time.time() < deadline - 350:
        d4 = min(deadline - 340, t0 + 400)
        for r in range(6):
            if time.time() > d4:
                break
            k_p = 5 + r * 3
            h_r = perturb(best_h, N, k_p)
            sub = min(d4, time.time() + max(5, (d4 - time.time()) / (6 - r)))
            h_r, c5_r = ils_search(h_r, N, sub, k_start=k_p + 4, k_min=3, mem=10)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r.copy()

    # Phase 5: ILS with very large perturbation (escape basin)
    if time.time() < deadline - 200:
        h_big = perturb(best_h, N, N // 6)
        d5 = min(deadline - 190, t0 + 500)
        h5, c5_5 = ils_search(h_big, N, d5, k_start=12, k_min=4, mem=18)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Nesterov continuous
    if time.time() < deadline - 120:
        hc = ensure_sum_binary(best_h, N).astype(float)
        hc += np.random.randn(N) * 0.07
        hc = project_sum(hc, N / 2.0)
        d6 = min(deadline - 110, t0 + 620)
        h6, c5_6 = nesterov_opt(hc, N, d6, k=45, lr0=0.018)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Second ILS from Nesterov result (if improved)
    if time.time() < deadline - 80:
        h7 = ensure_sum_binary(best_h, N)
        h7 = perturb(h7, N, 7)
        d7 = min(deadline - 70, t0 + 720)
        h7, c5_7 = ils_search(h7, N, d7, k_start=8, k_min=3, mem=10)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final greedy polish
    if time.time() < deadline - 1:
        hF = ensure_sum_binary(best_h, len(best_h))
        sf_fin = 1.0 if len(hF) <= 400 else 0.85
        hF2, c5F = greedy_ls(hF, len(hF), deadline - 0.5, max_sweeps=200, sf=sf_fin)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
