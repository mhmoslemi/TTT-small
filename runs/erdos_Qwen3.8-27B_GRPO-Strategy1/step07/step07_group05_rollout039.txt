∂corr[k]/∂h[j]: corr[k] involves products h[i]*(1-h[m]), so:
∂corr[k]/∂h[j] = sum of terms where j appears

For the "distance d" formulation:
A[d] = sum_{i=0}^{N-1-d} h[i]*(1-h[i+d])
∂A[d]/∂h[j] = (1-h[j+d]) if j+d < N [h[j] appears as first factor]

- h[i]*(-1) if i+d = j, i.e., i = j-d

So ∂A[d]/∂h[j] = (1-h[j+d])*1_{j+d<N} - h[j-d]*1_{j-d≥0}

Similarly for B[d] = sum_{i=0}^{N-1-d} h[i+d]*(1-h[i]):
∂B[d]/∂h[j] = (1-h[j-d])*1_{j-d≥0} - h[j+d]*1_{j+d<N}

And the full correlation has both A[d] and B[d] components.

The total objective: F(h) = (1/β) log(sum_d exp(β*A[d]) + sum_d exp(β*B[d]))
(neglecting the zero at d=0)

∂F/∂h[j] = (1/sum) * sum_d (exp(β*A[d])*β*∂A[d]/∂h[j] + exp(β*B[d])*β*∂B[d]/∂h[j])

def sliding_moves(h, N):
    """Find all possible sliding moves: shift a run of 1s by ±1"""
    moves = []
    for i in range(N):
        if h[i] == 1:
            # Find the run containing i
            start = i
            while start > 0 and h[start-1] == 1:
                start -= 1
            # Try sliding right: h[start:start+run_len] -> 0s, h[end:end+1] needs to be 0
            end = i
            while end < N-1 and h[end+1] == 1:
                end += 1
            run_len = end - start + 1

            if end + 1 < N and h[end+1] == 0:

            if start > 0 and h[start-1] == 0:
                moves.append(('slide_l', start, end, start-1))
            i = end  # skip rest of run

    """Compute gradient of log-sum-exp approximation of max correlation"""
    corr = full_corr(h)  # length 2N-1
    # Weights
    corr_max = np.max(corr)
    w = np.exp(beta * (corr - corr_max))
    w_sum = w.sum()
    w /= w_sum
    
    grad = np.zeros(N)
    center = N - 1

    # corr[k] = sum_n h[n]*(1-h[n-k+N-1])
    # ∂corr[k]/∂h[j] = (1-h[j+k-center]) if valid - h[j-center-k] if valid...

    # For k = center + d (d >= 0): B[d] = sum_{i=0}^{N-1-d} h[i+d]*(1-h[i])
    
    # ∂A[d]/∂h[j] = (1-h[j+d])*1{j+d<N} - h[j-d]*1{j-d>=0}
    # ∂B[d]/∂h[j] = (1-h[j-d])*1{j-d>=0} - h[j+d]*1{j+d<N}
    
    # Combine: for each d, the weight is w[center-d] + w[center+d] (approximately)
    # Actually they have different weights

    for k in range(2*N-1):
        if w[k] < 1e-6:
            continue

        # Valid n: max(0, k-N+1) <= n <= min(N-1, k)
        n_lo = max(0, k - N + 1)
        n_hi = min(N - 1, k)

        # ∂/∂h[j]: 
        #   if j = n: contribution = (1-h[n-k+N-1])
        #   if j = n-k+N-1: contribution = -h[n]

        # Second term: for j where j = n-k+N-1 for some valid n, contribution -= h[j+k-N+1] * wk

def cont_refine(h, N, deadline, beta=8.0, lr=0.02):
    """Continuous refinement using peak-tracking gradient"""
    h = h.astype(float).copy()
    best_c = c5_of(h)
    best_h = h.copy()
    target = N / 2.0
    step = lr
    
    while time.time() < deadline - 1:
        corr = full_corr(h)
        k_star = int(np.argmax(corr))
        
        # Gradient of corr[k_star] w.r.t. h
        # corr[k] = sum_n h[n]*(1-h[n-k+N-1])
        # ∂corr[k*]/∂h[j] = (1-h[j-k*+N-1])*1{valid} - h[j+k*-N+1]*1{valid}
        grad = np.zeros(N)
        k = k_star
        # First part: j appears as first factor h[j]
        # n = j, m = j - k + N - 1, valid: 0<=n<N, 0<=m<N
        n_lo = max(0, k - N + 1)
        n_hi = min(N - 1, k)
        # j from n_lo to n_hi, m = j - k + N - 1
        m_lo = n_lo - k + N - 1
        m_hi = n_hi - k + N - 1
        if m_lo >= 0 and m_hi < N:
            grad[n_lo:n_hi+1] += (1.0 - h[m_lo:m_hi+1])
        
        # Second part: j appears as second factor (1-h[j])
        # m = j, n = j + k - N + 1, valid: 0<=n<N
        # j from 0 to N-1, n = j+k-N+1 must be in [0,N-1]
        # j >= N-1-k, j <= N-1-k+N... let's redo
        # n = m + k - (N-1) where m = j (second factor index)
        # n must be in [0, N-1], so 0 <= m+k-N+1 <= N-1
        # 1-N-k <= m <= N-1-k... wait
        # m + k - N + 1 >= 0 → m >= N-1-k
        # m + k - N + 1 <= N-1 → m <= 2N-2-k
        # Also 0 <= m <= N-1
        m_lo2 = max(0, N - 1 - k)
        m_hi2 = min(N - 1, 2*N - 2 - k)
        if m_lo2 <= m_hi2:
            n_vals = m_lo2 + k - N + 1
            grad[m_lo2:m_hi2+1] -= h[n_vals:n_vals + (m_hi2 - m_lo2 + 1)]
        
        # Gradient step
        h_new = h - step * grad
        # Project to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Project to sum = target
        diff = h_new.sum() - target
        if abs(diff) > 1e-10:
            # Binary search for shift
            lo, hi = -2.0, 2.0
            for _ in range(40):
                mid = (lo + hi) / 2
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        
        c_new = c5_of(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        step *= 0.999
    
    return best_h, best_c
