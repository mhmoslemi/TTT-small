    elif s < nh:
        zeros = np.where(h == 0)[0]
        h[np.random.choice(zeros, nh - s, replace=False)] = 1
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    # Precompute C rows
    C = np.zeros((N, 2 * N - 1))
    for i in range(N):
        C[i] = compute_C_row(h, N, i)
    temp = temp0
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        pk = float(np.max(corr))
        if pk < best_c - 1e-14:
            best_c = pk
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        # Approximate: C[i] gives change for 1->0 at i; -C[j] gives change for 0->1 at j
        Co = C[os_]  # shape (ns, L)
        Cz = -C[zs]  # shape (nz, L), sign for 0->1
        new_approx = corr[None, None, :] + Co[None, :, :] + Cz[:, None, :]
        approx_max = new_approx.max(axis=2)
        # Get top-K smallest approximate max
        k_verify = min(verify_top, approx_max.size)
        flat_idx = np.argpartition(approx_max.ravel(), k_verify - 1)[:k_verify]
        best_move = None
        best_val = pk
        for fi in flat_idx:
            a, b = divmod(fi, nz)  # a indexes os_, b indexes zs_
            # Verify exactly
            h2 = h.copy()
            h2[os_[a]] = 0.0
            h2[zs[b]] = 1.0
            c2 = fft_corr(h2)
            m2 = float(np.max(c2))
            if m2 < best_val - 1e-15:
                best_val = m2
                best_move = (os_[a], zs[b], c2)
        # Acceptance
        accepted = False
        if best_move is not None:
            i, j, new_corr = best_move
            if no_improve < 200 or np.random.random() < 0.1:
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                C[i] = compute_C_row(h, N, i)
                C[j] = compute_C_row(h, N, j)
                accepted = True
        elif temp > 0 and np.random.random() < temp:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h2 = h.copy()
            h2[p] = 0.0
            h2[q] = 1.0
            c2 = fft_corr(h2)
            if c2.max() < best_c + 2.0:
                h = h2
                corr = c2
                C[p] = compute_C_row(h, N, p)
                C[q] = compute_C_row(h, N, q)
                accepted = True
        if not accepted and no_improve >= kick_every:
            kk = min(kick_sz, len(ones), len(zeros))
            for _ in range(kk):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0.0
                h[q] = 1.0
                C[p] = compute_C_row(h, N, p)
                C[q] = compute_C_row(h, N, q)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            no_improve = 0
        temp = temp0 * (cool ** it)
    return best_h, best_c

def subgradient_opt(h, N, deadline, k_active=30, lr0=0.008, decay=0.995):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    h = project_01_sum(h, target)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h))) * (2.0 / N)
    yr = np.zeros_like(h)
    lr = lr0
    while time.time() < deadline:
        y = h + 0.5 * yr
        corr = fft_corr(y)
        idx_top = np.argpartition(corr, -k_active)[-k_active:]
        # Compute gradient of sum of top-k corr
        grad = np.zeros(N)
        center = N - 1
        for ai in idx_top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_active
        h_new = project_01_sum(y - lr * grad, target)
        yr = 0.5 * (h_new - yr + h_new - h)
        h = h_new
        cur_c = float(np.max(fft_corr(h))) * (2.0 / N)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        lr *= decay
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=np.float64)
    N0 = len(h0)
    # Ensure binary and correct sum
    hb = (h0 > 0.5).astype(np.float64)
    s = int(hb.sum())
    if s != N0 // 2:
        if s > N0 // 2:
            idx = np.where(hb == 1)[0]
            hb[np.random.choice(idx, s - N0 // 2, replace=False)] = 0
        else:
            idx = np.where(hb == 0)[0]
            hb[np.random.choice(idx, N0 // 2 - s, replace=False)] = 1
    best_h = hb.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Strong binary local search (main effort)
    d1 = min(deadline - 300, t0 + 350)
    h1, c1 = binary_search_exact(hb, N0, d1, max_iters=20000, sample_sz=120,
                                  temp0=0.0, verify_top=10, kick_every=100, kick_sz=3)
    c5_1 = c1 * (2.0 / N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Simulated annealing binary search
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 200)
        h2, c2 = binary_search_exact(best_h, N0, d2, max_iters=15000, sample_sz=100,
                                      temp0=0.3, cool=0.997, verify_top=8,
                                      kick_every=60, kick_sz=5)
        c5_2 = c2 * (2.0 / N0)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Continuous subgradient polish
    if time.time() < deadline - 180:
        d3 = min(deadline - 160, time.time() + 120)
        h3 = best_h.copy()
        h3 += np.random.randn(N0) * 0.05
        h3 = project_01_sum(h3, N0 / 2.0)
        h3r, c5_3 = subgradient_opt(h3, N0, d3, k_active=40, lr0=0.006, decay=0.996)
        # Also try binary version
        h3b = (h3r > 0.5).astype(np.float64)
        s3 = int(h3b.sum())
        if s3 > N0 // 2:
            idx = np.where(h3b == 1)[0]
            h3b[np.random.choice(idx, s3 - N0 // 2, replace=False)] = 0
        elif s3 < N0 // 2:
            idx = np.where(h3b == 0)[0]
            h3b[np.random.choice(idx, N0 // 2 - s3, replace=False)] = 1
        c5_3b = compute_c5(h3b, N0)
        c5_3c = compute_c5(h3r, N0)
        if c5_3b < best_c5:
            best_c5, best_h = c5_3b, h3b.copy()
        elif c5_3c < best_c5:
            best_c5, best_h = c5_3c, h3r.copy()

    # Phase 4: Focused binary search on best
    if time.time() < deadline - 100:
        d4 = min(deadline - 80, time.time() + 90)
        h4 = (best_h > 0.5).astype(np.float64)
        s4 = int(h4.sum())
        if s4 != N0 // 2:
            if s4 > N0 // 2:
                idx = np.where(h4 == 1)[0]
                h4[np.random.choice(idx, s4 - N0 // 2, replace=False)] = 0
            elif s4 < N0 // 2:
                idx = np.where(h4 == 0)[0]
                h4[np.random.choice(idx, N0 // 2 - s4, replace=False)] = 1
        h4r, c4 = binary_search_exact(h4, N0, d4, max_iters=10000, sample_sz=150,
                                       temp0=0.0, verify_top=12, kick_every=80, kick_sz=3)
        c5_4 = c4 * (2.0 / N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4r.copy()

    # Phase 5: Try smaller N for potentially better ratio
    for Ntry in [300, 260, 220]:
        if time.time() > deadline - 50:
            break
        x_full = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Ntry, endpoint=False)
        h_int = np.interp(x_new, x_full, best_h)
        h_bt = (h_int > 0.5).astype(np.float64)
        st = int(h_bt.sum())
        if st > Ntry // 2:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, st - Ntry // 2, replace=False)] = 0
        elif st < Ntry // 2:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, Ntry // 2 - st, replace=False)] = 1
        d5 = min(deadline - 15, time.time() + 40)
        h5, c5_raw = binary_search_exact(h_bt, Ntry, d5, max_iters=5000,
                                          sample_sz=Ntry // 2, temp0=0.0,
                                          verify_top=6, kick_every=50, kick_sz=3)
        c5_5 = c5_raw * (2.0 / Ntry)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final polish
    if time.time() < deadline - 3:
        hF = (best_h > 0.5).astype(np.float64)
        NF = len(hF)
        sF = int(hF.sum())
        if sF != NF // 2:
            if sF > NF // 2:
                idx = np.where(hF == 1)[0]
                hF[np.random.choice(idx, sF - NF // 2, replace=False)] = 0
            elif sF < NF // 2:
                idx = np.where(hF == 0)[0]
                hF[np.random.choice(idx, NF // 2 - sF, replace=False)] = 1
        hFr, cF = binary_search_exact(hF, NF, deadline - 0.5, max_iters=2000,
                                       sample_sz=min(NF, 120), temp0=0.0,
                                       verify_top=10, kick_every=60, kick_sz=3)
        c5F = cF * (2.0 / NF)
        if c5F < best_c5:
            best_c5, best_h = c5F, hFr.copy()

    # Final: compare continuous vs binary
    final_bin = (best_h > 0.5).astype(np.float64)
    NF = len(best_h)
    sfb = int(final_bin.sum())
    if sfb != NF // 2:
        if sfb > NF // 2:
            idx = np.where(final_bin == 1)[0]
            final_bin[np.random.choice(idx, sfb - NF // 2, replace=False)] = 0
        elif sfb < NF // 2:
            idx = np.where(final_bin == 0)[0]
            final_bin[np.random.choice(idx, NF // 2 - sfb, replace=False)] = 1
    c5_bin = compute_c5(final_bin, NF)
    c5_cont = compute_c5(best_h, NF)
    if c5_bin <= c5_cont:
        return final_bin, c5_bin, NF
    else:
        return best_h, c5_cont, NF
