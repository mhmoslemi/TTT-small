    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary polish at N0
    h_bin = ensure_binary(h0, N0, rng)
    d1 = min(deadline - 550, t0 + budget_s * 0.12)
    h1, c5_1 = binary_anneal(h_bin, N0, d1, 600, 0.45, 150, rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Upscale to N1=2*N0, Nesterov continuous
    N1 = 2 * N0
    if time.time() < deadline - 400:
        h_up = upsample(best_h, N0, N1)
        h_up = project_sum(h_up, N1 / 2.0)
        d2 = min(deadline - 350, time.time() + budget_s * 0.16)
        h2, c5_2 = nesterov_refine(h_up, N1, d2, k_top=60, lr0=0.010, beta0=0.88)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Binary anneal at N1 with aggressive 2-swaps
    if time.time() < deadline - 250:
        h_b1 = ensure_binary(best_h, N1, rng)
        d3 = min(deadline - 200, time.time() + budget_s * 0.20)
        h3, c5_3 = binary_anneal(h_b1, N1, d3, 500, 0.4, 200, rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Nesterov at N1, different params
    if time.time() < deadline - 150:
        hc = best_h + rng.randn(N1) * 0.025
        hc = project_sum(hc, N1 / 2.0)
        d4 = min(deadline - 100, time.time() + budget_s * 0.12)
        h4, c5_4 = nesterov_refine(hc, N1, d4, k_top=100, lr0=0.007, beta0=0.91)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Another binary restart at N1
    if time.time() < deadline - 80:
        rng2 = np.random.RandomState(seed + 77)
        h_b2 = ensure_binary(best_h, N1, rng2)
        ones_t = np.where(h_b2 == 1)[0]
        zeros_t = np.where(h_b2 == 0)[0]
        for _ in range(6):
            h_b2[ones_t[rng2.randint(len(ones_t))]] = 0.0
            h_b2[zeros_t[rng2.randint(len(zeros_t))]] = 1.0
        d5 = min(deadline - 40, time.time() + budget_s * 0.08)
        h5, c5_5 = binary_anneal(h_b2, N1, d5, 300, 0.35, 180, rng2)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final Nesterov polish at best resolution
    N_fin = len(best_h)
    if time.time() < deadline - 5:
        d6 = deadline - 2
        h6, c5_6 = nesterov_refine(best_h.copy(), N_fin, d6, k_top=40, lr0=0.005, beta0=0.95)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
