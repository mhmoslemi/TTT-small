    L = 2*N-1
    return np.real(np.fft.irfft(np.fft.rfft(h,L)*np.fft.rfft(np.flip(1.0-h),L),L))

def c5_of(h):
    return float(fft_corr(h).max())*(2.0/len(h))

def binarize(h,N):
    h=np.clip(h.astype(float),0,1)
    t=N//2; n1=int(np.round(h.sum()))
    if n1>t:
        idx=np.flatnonzero(h>0.5)
        h[idx[np.argsort(-h[idx])[:n1-t]]]=0
    elif n1<t:
        idx=np.flatnonzero(h<0.5)
        h[idx[np.argsort(h[idx])[:t-n1]]]=1
    return(h>0.5).astype(float)

def mk_deltas(h):
    N=len(h); s=2.0*h-1.0
    D=np.zeros((N,2*N-1))
    for i in range(N): D[i,i:i+N]=s
    return s,D

def rdelta(D,s,i):
    N=D.shape[0]; D[i]=0.0; D[i,i:i+N]=s

def sa(h0,dl,t0=0.04,ck=0.9992,npk=6,npp=18,tm=0.001,ppi=5000,ppn=5,nmx=12000):
    N=len(h0); h=h0.copy(); c=fft_corr(h).copy()
    bc,bh=float(c.max()),h.copy(); s,D=mk_deltas(h)
    T,it,ni=t0,0,0
    while time.time()<dl:
        it+=1
        if it%1000==0: T=max(T*ck,tm)
        o=np.flatnonzero(h==1); z=np.flatnonzero(h==0)
        if len(o)<3 or len(z)<3: break
        pk=np.argpartition(c,-npk)[-npk:]
        cp,cq=[],[]
        for p_ in pk:
            k=int(p_)-(N-1)
            if k==0: continue
            if k>0:
                r=np.arange(N-k); m=(h[r]==1)&(h[r+k]==0)
            else:
                k2=-k; r=np.arange(k2,N); m=(h[r]==1)&(h[r-k2]==0)
            pp,qq=r[m],(r+k if k>0 else r-k2)[m]
            if pp.size:
                ns=min(npp,pp.size); sl=np.random.choice(pp.size,ns,replace=False)
                cp.extend(pp[sl]); cq.extend(qq[sl])
        nr=max(4,N//40)
        cp.extend(np.random.choice(o,min(nr,len(o)),replace=False).tolist())
        cq.extend(np.random.choice(z,min(nr,len(z)),replace=False).tolist())
        cp=np.array(sorted(set(cp)),dtype=int)[:120]
        cq=np.array(sorted(set(cq)),dtype=int)[:120]
        if not cp.size or not cq.size: break
        bv,bi,bj=1e10,0,0
        for ci in range(0,cp.size,8):
            ce=min(ci+8,cp.size)
            v=c[None,None]+D[cp[ci:ce]][:,None,:]-D[cq][None,:,:]
            nm=v.max(2); fi=int(nm.argmin())
            ri,rj=fi//cq.size,fi%cq.size
            if nm[ri,rj]<bv: bv,bi,bj=float(nm[ri,rj]),ci+ri,rj
        p,q=int(cp[bi]),int(cq[bj])
        dc=bv-bc
        if dc<-1e-12 or(T>tm and np.random.random()<np.exp(-dc/max(T,1e-9))):
            h[p],h[q]=0.,1.
            c+=D[p].copy()-D[q].copy()
            s[p],s[q]=-s[p],-s[q]
            rdelta(D,s,p); rdelta(D,s,q)
            nv=float(c.max())
            if nv<bc-1e-12: bc,bh,ni=nv,h.copy(),0
            else: ni+=1
        else: ni+=1
        if ni>=ppi:
            for _ in range(ppn):
                r1,r2=np.random.choice(o),np.random.choice(z)
                if h[r1]==1 and h[r2]==0:
                    h[r1],h[r2]=0.,1.
                    c+=D[r1].copy()-D[r2].copy()
                    s[r1],s[r2]=-s[r1],-s[r2]
                    rdelta(D,s,r1); rdelta(D,s,r2)
            ni=0; T=max(T,t0*0.6)
        if ni>nmx: break
    return bh,bc

def greedy(h0,dl):
    N=len(h0); h=h0.copy(); c=fft_corr(h).copy()
    s,D=mk_deltas(h)
    improved=True; guard=0
    while improved and time.time()<dl and guard<50:
        improved=False; guard+=1
        o=np.flatnonzero(h==1); z=np.flatnonzero(h==0)
        if len(o)<2 or len(z)<2: break
        bm=1e-10; bp,bq=-1,-1
        cs=30
        for ci in range(0,len(o),cs):
            ce=min(ci+cs,len(o))
            v=(c[None,None,:]+D[o[ci:ce]][:,None,:]-D[z][None,:,:]).max(axis=2)
            fi=int(v.argmin()); ri,ri2=fi//len(z),fi%len(z)
            if v[ri,ri2]<bm:
                bm=float(v[ri,ri2]); bp,bq=int(o[ci+ri]),int(z[ri2])
        if bp>=0 and bm<bm:
            bv=float(c.max())+bm
            c_now=float(c.max())
            if bv<c_now-1e-10:
                h[bp],h[bq]=0.,1.
                c+=D[bp].copy()-D[bq].copy()
                s[bp],s[bq]=-s[bp],-s[bq]
                rdelta(D,s,bp); rdelta(D,s,bq)
                improved=True
    return h

def nesterov(h0,dl,lr=0.01,kt=35):
    N=len(h0); h=h0.astype(float).copy(); m=np.zeros(N)
    spk=float(fft_corr(h).max()); best=h.copy()
    beta,it=0.8,0
    while time.time()<dl:
        y=np.clip(h+m,0,1); c=fft_corr(y)
        pk=float(c.max())
        if pk<spk: spk,best=pk,y.copy()
        tp=np.argpartition(c,-kt)[-kt:]
        g=np.zeros(N)
        for t_ in tp:
            L=int(t_-(N-1))
            if L==0: g+=1.0-2.0*y
            elif L>0:
                g[L:]+=1.0-y[:N-L]; g[:N-L]-=y[L:]
            else:
                M=-L; g[:N-M]+=1.0-y[M:]; g[M:]-=y[:N-M]
        g/=len(tp)
        h=np.clip(y-lr*g,0,1)
        h=np.clip(h-(h.sum()-N/2)/N,0,1)
        m=beta*m+(h-y)
        it+=1
        if it%400==0: lr*=0.97; beta=min(0.93,0.78+it*0.0003)
    return best,spk*(2.0/N)

def run(seed=42,budget_s=1100,**kwargs):

    # Phase 2: Greedy descent
    if time.time()<dl-200:
        hg=greedy(best_h,dl-180)
        cg=c5_of(hg)
        if cg<best_c5: best_c5,best_h=cg,hg.copy()

        hc=best_h.astype(float)+np.random.randn(N0)*0.03

        hn,cn=nesterov(hc,d4,lr=0.008,kt=30)

        sub=min(dl-40,time.time()+50)

        cr=c5_of(hr)
        if cr<best_c5: best_c5,best_h=cr,hr.copy()
