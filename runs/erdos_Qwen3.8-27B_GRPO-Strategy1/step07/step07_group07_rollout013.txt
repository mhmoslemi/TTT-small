    seq = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        seq[i] = c % 2
    return seq

        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def thue_morse(n):
    seq = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        seq[i] = c % 2
    return seq

def cont_opt(h, deadline, k=50, lr0=0.02, mom=0.85):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 100:
            lr0 *= 0.8
            stale = 0
        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            L = int(m) - center
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[L:] += (1.0 - h[:N-L])
                grad[:N-L] -= h[L:]
            elif -N < L < 0:
                grad[:N+L] += (1.0 - h[-L:])
                grad[-L:] -= h[:N+L]
        grad /= k
        h_new = project(h + mom * v - lr0 * grad, tgt)
        v = mom * v + (h_new - h)
        h = h_new
        lr0 = max(lr0 * 0.9995, 0.002)
    return best_h, best_c
