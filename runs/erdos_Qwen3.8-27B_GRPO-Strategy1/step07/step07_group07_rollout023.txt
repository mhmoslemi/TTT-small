def compute_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

But if h is binary (0 or 1), then h[i]*(1-h[i+k]) is just 1 if h[i]=1 and h[i+k]=0, else 0. So for binary h, corr_k is just a count, which IS linear in h (it's a sum of products of binary variables, which for binary variables equals the minimum, which is... no, it's not linear either).

Actually for binary h: h[i]*(1-h[i+k]) = h[i] - h[i]*h[i+k]. Still quadratic.

So it's not a simple LP. The continuous relaxation (0 <= h <= 1) makes it a quadratic programming problem.

        best_c, best_h = c1, h1.copy()
    # Phase 2: Binary SA - multiple diverse starts (350s)
    starts = [fix_binary(best_h), fix_binary(thue_morse(N0)), fix_binary(np.random.rand(N0))]
    remaining = deadline - 450
    per_start = max(60, remaining / len(starts))
    for start in starts:
        if time.time() > deadline - 420:
            break
        d2 = min(time.time() + per_start, deadline - 400)
        h2, c2 = binary_sa(start, d2, T0=0.005, cool=0.9996, max_no_imp=500, guided=0.35)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()
    # Phase 3: Greedy refinement (150s)
    if time.time() < deadline - 250:
        d3 = min(time.time() + 120, deadline - 240)
        h3, c3 = binary_greedy(best_h, d3, n_crit=35, s_sz=50)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()
    # Phase 4: Low-temp SA polish (150s)
    if time.time() < deadline - 120:
        for temp, mx in [(0.001, 250), (0.0004, 200)]:
            if time.time() > deadline - 50:
                break
            d4 = min(time.time() + 60, deadline - 45)
            h4, c4 = binary_sa(best_h, d4, T0=temp, cool=0.999, max_no_imp=mx, guided=0.4)
            if c4 < best_c:
                best_c, best_h = c4, h4.copy()
    # Phase 5: Try nearby resolutions (100s)
    for N_try in [380, 400, 420, 440]:
        if time.time() > deadline - 60:
            break
        if N_try == N0:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d5 = min(time.time() + 40, deadline - 30)
        h_bt, c5 = binary_sa(h_bt, d5, T0=0.002, cool=0.999, max_no_imp=200, guided=0.3)
        if c5 < best_c:
            best_c, best_h = c5, h_bt.copy()
    # Phase 6: Final greedy polish
    if time.time() < deadline - 8:
        d6 = min(time.time() + 25, deadline - 2)
        h6, c6 = binary_greedy(best_h, d6, n_crit=40, s_sz=55)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
