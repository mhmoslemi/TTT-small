def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):

                M = -L
                grad[M:] += wt * (1.0 - h[:N - M])
                grad[:N - M] -= wt * h[M:]
        h = h - lr * grad
        h = fix_sum_cont(h, target)
        c = compute_c5(h, N)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
        if it % 400 == 0:
            lr *= 0.93
            beta *= 0.82
    return best_h, best_c

def binary_anneal(h, N, deadline, max_iters=60000, T0=0.03, T1=1e-7):
    h = h.astype(float).copy()
    nh = N // 2
    h = binary_fix(h, nh)
    cur_max = float(np.max(fft_corr(h)))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = T0
    cool = (T1 / T0) ** (1.0 / max(1, max_iters))
    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        r = np.random.random()
        if r < 0.55:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0.0; h[zi] = 1.0
            nm = float(np.max(fft_corr(h)))
            dE = nm - cur_max
            if dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
                cur_max = nm
                if cur_max * 2.0 / N < best_c:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[oi] = 1.0; h[zi] = 0.0
        elif r < 0.8:
            L = int(np.random.choice([3, 4, 5, 7, 8, 12, 16, 24]))
            if L >= N:
                continue
            a = int(np.random.randint(0, N - L + 1))
            h[a:a + L] = h[a:a + L][::-1]
            nm = float(np.max(fft_corr(h)))
            dE = nm - cur_max
            if dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
                cur_max = nm
                if cur_max * 2.0 / N < best_c:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[a:a + L] = h[a:a + L][::-1]
        else:
            L = int(np.random.choice([2, 3, 4, 6, 8, 12]))
            if 2 * L >= N:
                continue
            a = int(np.random.randint(0, N - 2 * L))
            b = a + L + int(np.random.randint(0, N - 2 * L - a + 1))
            tmp = h[a:a + L].copy()
            h[a:a + L] = h[b:b + L]
            h[b:b + L] = tmp
            nm = float(np.max(fft_corr(h)))
            dE = nm - cur_max
            if dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
                cur_max = nm
                if cur_max * 2.0 / N < best_c:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[b:b + L] = h[a:a + L].copy()
                h[a:a + L] = tmp
        T *= cool
        if no_improve > 4000:
            for _ in range(np.random.randint(3, 9)):
                o = int(np.random.choice(np.where(h == 1)[0]))
                z = int(np.random.choice(np.where(h == 0)[0]))
                h[o], h[z] = h[z], h[o]
            cur_max = float(np.max(fft_corr(h)))
            T = max(T, 0.012)
            no_improve = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: Binary annealing from initial
    h1 = (h0 > 0.5).astype(float)
    d1 = min(deadline - 400, t0 + 200)
    h1, c1 = binary_anneal(h1, N0, d1, max_iters=100000, T0=0.05, T1=1e-5)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()
    # Phase 2: Continuous LSE refinement (corrected gradient)
    if time.time() < deadline - 300:
        hc = best_h.astype(float) + np.random.randn(N0) * 0.015
        hc = np.clip(hc, 0.0, 1.0)
        hc = fix_sum_cont(hc, N0 / 2.0)
        d2 = min(deadline - 240, time.time() + 200)
        hc, c2 = lse_refine(hc, N0, d2, lr_init=0.004, beta_init=5.0, max_iters=8000)
        if c2 < best_c5:
            best_c5, best_h = c2, hc.copy()
    # Phase 3: Binary anneal from best
    if time.time() < deadline - 200:
        h3 = binary_fix(best_h.copy(), N0 // 2)
        d3 = min(deadline - 140, time.time() + 160)
        h3, c3 = binary_anneal(h3, N0, d3, max_iters=120000, T0=0.02, T1=1e-8)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
    # Phase 4: Second continuous pass with lower temperature
    if time.time() < deadline - 100:
        hc4 = best_h.astype(float) + np.random.randn(N0) * 0.008
        hc4 = np.clip(hc4, 0.0, 1.0)
        hc4 = fix_sum_cont(hc4, N0 / 2.0)
        d4 = min(deadline - 40, time.time() + 80)
        hc4, c4 = lse_refine(hc4, N0, d4, lr_init=0.0015, beta_init=2.5, max_iters=5000)
        if c4 < best_c5:
            best_c5, best_h = c4, hc4.copy()
    # Phase 5: Final short binary polish
    if time.time() < deadline - 10:
        hF = binary_fix(best_h.copy(), N0 // 2)
        d5 = deadline - 1
        hF, c5F = binary_anneal(hF, N0, d5, max_iters=40000, T0=0.005, T1=1e-10)
        c5F = compute_c5(hF, N0)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, N0)
    return best_h, best_c5, N0
