def fft_corr(h):

def compute_c5(h):

def project_sum(h, target):

        return h
    lo, hi = -1.0, 1.0

        else:

def compute_D(h, N):

    beta = 0.9

        y = h + m

        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()

        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_c5_val(h):
    return float(np.max(fft_corr(h)))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    t0 = time.time()

    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    h_bin = ensure_sum(h0, N)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy (12s)
    d1 = min(t0 + 12, deadline - 800)
    h1, c1 = greedy_fast(best_h, N, d1, max_sweeps=200, sf=0.7)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: SA main exploration (180s)
    d2 = min(t0 + 195, deadline - 600)
    h2, c2 = sa_search(best_h, N, d2, T0=4.5, Tcool=0.993, T_min=0.02, sf=0.7, kick_every=120, kick_size=5)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 3: ILS-style restart with larger perturbation (150s)
    d3 = min(t0 + 345, deadline - 450)
    h3 = perturb(best_h, N, 18)
    h3, c3 = sa_search(h3, N, d3, T0=5.0, Tcool=0.994, T_min=0.01, sf=0.65, kick_every=100, kick_size=6)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: Low-temp SA polish (130s)
    d4 = min(t0 + 475, deadline - 320)
    h4, c4 = sa_search(best_h, N, d4, T0=2.5, Tcool=0.992, T_min=0.008, sf=0.8, kick_every=250, kick_size=3)
    if c4 < best_c5:
        best_c5, best_h = c4, h4.copy()

    # Phase 5: Nesterov continuous (120s)
    d5 = min(t0 + 595, deadline - 200)
    hc = project_sum(best_h + np.random.randn(N) * 0.05, N / 2.0)
    h5, c5 = nesterov_opt(hc, N, d5, k=70, lr0=0.018)
    if c5 < best_c5:
        best_c5, best_h = c5, h5.copy()

    # Phase 6: SA after Nesterov perturbation (110s)
    d6 = min(t0 + 705, deadline - 100)
    h6 = perturb(ensure_sum(best_h, N), N, 10)
    h6, c6 = sa_search(h6, N, d6, T0=3.5, Tcool=0.993, T_min=0.006, sf=0.75, kick_every=110, kick_size=4)
    if c6 < best_c5:
        best_c5, best_h = c6, h6.copy()

    # Phase 7: Very low temp SA (100s)
    d7 = min(t0 + 805, deadline - 40)
    h7, c7 = sa_search(best_h, N, d7, T0=1.5, Tcool=0.991, T_min=0.003, sf=0.85, kick_every=350, kick_size=3)
    if c7 < best_c5:
        best_c5, best_h = c7, h7.copy()

    # Phase 8: Final greedy polish (remaining time)
    if time.time() < deadline - 2:
        hF = ensure_sum(best_h, N)
        hF2, cF = greedy_fast(hF, N, deadline - 0.5, max_sweeps=400, sf=0.8)
        if cF < best_c5:
            best_c5, best_h = cF, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N
