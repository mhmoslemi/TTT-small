import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def compute_delta_matrix(h, crit_lags):
    N = len(h)
    K = len(crit_lags)
    R = np.zeros((N, K))
    idx = np.arange(N)
    for li in range(K):
        L = int(crit_lags[li])
        if L == 0:
            continue
        elif abs(L) < N:
            if L > 0:
                m1 = idx >= L
                m2 = idx <= N - 1 - L
                R[m1, li] += -(1.0 - h[idx[m1] - L])
                R[m2, li] += h[idx[m2] + L]
            else:
                Lp = -L
                m1 = idx <= N - 1 - Lp
                m2 = idx >= Lp
                R[m1, li] += -(1.0 - h[idx[m1] + Lp])
                R[m2, li] += h[idx[m2] - Lp]
    return R

        nz = min(len(zeros), s_sz)

        R_o = R_full[os_, :]
        R_z = R_full[zs, :]
