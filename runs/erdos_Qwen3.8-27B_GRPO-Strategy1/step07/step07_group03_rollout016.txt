import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def ensure_sum_binary(h, N):
    nh = N // 2
    h = (h > 0.5).astype(float).copy()
    n1 = int(h.sum())
    if n1 == nh:
        return h
    rng = np.random.RandomState(99)
    if n1 > nh:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k=80, lr0=0.015):

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()

    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.96, 0.88 + it * 0.00005)
    return best_h, best_pk * dx
