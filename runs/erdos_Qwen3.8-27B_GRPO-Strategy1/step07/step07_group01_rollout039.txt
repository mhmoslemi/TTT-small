        h_res, c5_res = local_search(h_cur, N1, time.time() + rem * 0.93, seed=seed + restart * 31)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 2: Local search at N0=1140 from best
    N_best = len(best_h)
    if N_best != N0:
        h_up = to_binary(np.interp(np.arange(N0), np.linspace(0, 1, N_best, endpoint=False), best_h), N0, rng)
    else:
        h_up = best_h.copy()
    if time.time() < deadline - 120:
        rem = min(deadline - 120 - time.time(), 350)
        h_res, c5_res = local_search(h_up, N0, time.time() + rem * 0.95, seed=seed + 500)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 3: Second N=1140 pass from perturbed best
    if time.time() < deadline - 80:
        h_pert = best_h.astype(float) + rng.randn(len(best_h)) * 0.3
        h_pert = to_binary(h_pert, len(best_h), rng)
        rem = min(deadline - 80 - time.time(), 150)
        h_res2, c5_res2 = local_search(h_pert, len(best_h), time.time() + rem * 0.9, seed=seed + 900)
        if c5_res2 < best_c5:
            best_c5 = c5_res2
            best_h = h_res2.copy()

    # Phase 4: Continuous refinement
    if time.time() < deadline - 20:
        Nc = len(best_h)
        rem = min(deadline - 20 - time.time(), 100)
        h_res3, c5_res3 = continuous_refine(best_h, Nc, time.time() + rem * 0.9, seed=seed + 1200)
        if c5_res3 < best_c5:
            best_c5 = c5_res3
            best_h = h_res3.copy()

    # Phase 5: Final binary polish on the very best
    if time.time() < deadline - 8:
        Nf = len(best_h)
        rem = min(deadline - 8 - time.time(), 60)
        h_res4, c5_res4 = local_search(best_h, Nf, time.time() + rem * 0.9, seed=seed + 2000)
        if c5_res4 < best_c5:
            best_c5 = c5_res4
            best_h = h_res4.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
