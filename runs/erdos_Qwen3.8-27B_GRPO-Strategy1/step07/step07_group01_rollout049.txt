    N = len(h)
    L = 2 * N - 1

    return np.fft.irfft(H * G, L)

def grad_max_corr(corr, h, N, k_top, T):

    idx = np.argpartition(corr, -k_top)[-k_top:]

    w = np.exp(T * (c_vals - c_vals.max()))
    w /= w.sum()

    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
