This is exactly what D computes (the previous D matrix formula). So D[j, k] IS the gradient of C(k) with respect to h[j] (treating all h as continuous).

So for the continuous problem, the gradient of max_k C(k) is well-approximated by the average of D[j, k] over the top-k peaks.

The previous Nesterov code does something similar but a bit differently. Let me improve it:

```python
def nesterov_continuous(h, N, deadline, k_top=20, lr0=0.05):
    h = np.asarray(h, dtype=float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(fft_corr(h).max())
    m = np.zeros(N)
    T = 0.9  # momentum
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        C = float(c.max())
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
        # Compute top-k indices
        if k_top >= len(c):
            top = np.arange(len(c))
        else:
            top = np.argpartition(c, -k_top)[-k_top:]
        # Gradient: average of D[:, top]
        # But D depends on h, so compute it
        # Actually, let's compute the gradient directly
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            shift = int(ai - center)
            pos = np.arange(N)
            pm = pos - shift
            pp = pos + shift
            vm = (pm >= 0) & (pm < N)
            vp = (pp >= 0) & (pp < N)
            pm_c = np.clip(pm, 0, N - 1)
            pp_c = np.clip(pp, 0, N - 1)
            term1 = (1.0 - y[pm_c]) * vm  # (1 - h[j+shift]) for valid j
            term2 = y[pp_c] * vp  # h[j-shift] for valid j
            grad += term1 - term2
        grad /= len(top)
        # Nesterov step
        h_new = project_sum(y - lr * grad, target)
        m = T * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.95
            T = min(0.98, 0.85 + it * 0.0001)
    return best_h, best_c * dx
```

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = np.asarray(h, dtype=float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def sa_single(h, N, deadline, T0=4.0, Tcool=0.993, T_min=0.02, kick_every=120, kick_size=6):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    stall = 0
    while time.time() < deadline:
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall >= kick_every:
            for _ in range(kick_size):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) < 1 or len(zeros) < 1:
                    break
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.4
            c = fft_corr(h)
            P = float(c.max())
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        h[p], h[q] = 0.0, 1.0
        c = fft_corr(h)
        P_new = float(c.max())
        delta = P_new - P
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            P = P_new
        else:
            h[p], h[q] = 1.0, 0.0
            c = fft_corr(h)
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def sa_double(h, N, deadline, T0=4.0, Tcool=0.9994, T_min=0.015, kick_every=800, kick_size=8):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    stall = 0
    while time.time() < deadline:
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p1 = ones[np.random.randint(len(ones))]
                p2 = ones[np.random.randint(len(ones))]
                q1 = zeros[np.random.randint(len(zeros))]
                q2 = zeros[np.random.randint(len(zeros))]
                if p1 != p2 and q1 != q2:
                    h[p1] = 0.0; h[p2] = 0.0; h[q1] = 1.0; h[q2] = 1.0
            stall = 0
            T = T0 * 0.5
            c = fft_corr(h)
            P = float(c.max())
            continue
        p1 = ones[np.random.randint(len(ones))]
        p2 = ones[np.random.randint(len(ones))]
        if p2 == p1:
            p2 = ones[(np.random.randint(len(ones)) + 1) % len(ones)]
        q1 = zeros[np.random.randint(len(zeros))]
        q2 = zeros[np.random.randint(len(zeros))]
        if q2 == q1:
            q2 = zeros[(np.random.randint(len(zeros)) + 1) % len(zeros)]
        h[p1] = 0.0; h[p2] = 0.0; h[q1] = 1.0; h[q2] = 1.0
        c = fft_corr(h)
        P_new = float(c.max())
        delta = P_new - P
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            P = P_new
        else:
            h[p1] = 1.0; h[p2] = 1.0; h[q1] = 0.0; h[q2] = 0.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def nesterov_cont(h, N, deadline, k_top=30, lr0=0.04):
    h = np.asarray(h, dtype=float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(fft_corr(h).max())
    m = np.zeros(N)
    T = 0.88
    lr = lr0
    it = 0
    center = N - 1
    pos = np.arange(N)
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        C = float(c.max())
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
        kk = min(k_top, len(c))
        top = np.argpartition(c, -kk)[-kk:]
        grad = np.zeros(N)
        for ai in top:
            sh = int(ai - center)
            pm = pos - sh
            pp = pos + sh
            vm = (pm >= 0) & (pm < N)
            vp = (pp >= 0) & (pp < N)
            pm_c = np.clip(pm, 0, N - 1)
            pp_c = np.clip(pp, 0, N - 1)
            grad += (1.0 - y[pm_c]) * vm - y[pp_c] * vp
        grad /= kk
        h_new = project_sum(y - lr * grad, target)
        m = T * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.96
            T = min(0.98, 0.85 + it * 0.00005)
    return best_h, best_c * dx

def greedy_final(h, N, deadline, sf=1.0):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    center = N - 1
    while time.time() < deadline:
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            sh_m = p
            sh_p = -p
            pm = np.arange(2 * N - 1) - center
            pm_m = pm + sh_m
            pm_p = pm - sh_p
            vm_m = (pm_m >= 0) & (pm_m < N)
            vm_p = (pm_p >= 0) & (pm_p < N)
            pm_m_c = np.clip(pm_m, 0, N - 1)
            pm_p_c = np.clip(pm_p, 0, N - 1)
            Dp = -(1.0 - h[pm_m_c]) * vm_m + h[pm_p_c] * vm_p
            base = c - Dp
            Dz = np.zeros(2 * N - 1)
            for q in zeros:
                pm_q = pm + q
                pp_q = pm - q
                vq_m = (pm_q >= 0) & (pm_q < N)
                vq_p = (pp_q >= 0) & (pp_q < N)
                pm_q_c = np.clip(pm_q, 0, N - 1)
                pp_q_c = np.clip(pp_q, 0, N - 1)
                Dz_q = (1.0 - h[pm_q_c]) * vq_m - h[pp_q_c] * vq_p
                cand = base - Dz_q
                cmax = float(cand.max())
                if cmax < best_new_P:
                    best_new_P = cmax
                    best_pq = (int(p), int(q))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.asarray(initial_h_values, dtype=float)
    N0 = len(h0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
