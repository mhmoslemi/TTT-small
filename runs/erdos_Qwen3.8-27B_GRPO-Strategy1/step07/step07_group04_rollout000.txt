        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        delta = best_new_P - P
        if best_pq is not None and (delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            if do_2swap and np.random.random() < 0.4:
                ones2 = np.where(h == 1)[0]
                zeros2 = np.where(h == 0)[0]
                if len(ones2) > 1 and len(zeros2) > 1:
                    p2 = ones2[np.random.randint(len(ones2))]
                    q2 = zeros2[np.random.randint(len(zeros2))]
                    h[p2], h[q2] = 0.0, 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def smooth_max_grad(h, N, deadline, T_smooth=50.0, k=40, lr0=0.015, max_iter=8000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while it < max_iter and time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        weights = np.exp((corr[top] - corr[top].max()) * T_smooth)
        weights /= weights.sum()
        grad = np.zeros(N)
        center = N - 1
        for j, ai in enumerate(top):
            L = int(ai - center)
            w = weights[j] * T_smooth
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w * (1.0 - y[:N - L])
                grad[:N - L] -= w * y[L:]
            else:
                M = -L
                grad[:N - M] += w * (1.0 - y[M:])
                grad[M:] -= w * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            T_smooth = max(5.0, T_smooth * 0.85)
            lr = lr * 0.985
            beta = min(0.97, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

def block_sa(h, N, deadline, max_sweeps=3000, T0=4.0, Tcool=0.994, T_min=0.04, block_sizes=None):
    if block_sizes is None:
        block_sizes = [1, 2, 3]
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall > 100:
            for _ in range(3):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0
            continue
        bs = block_sizes[np.random.randint(len(block_sizes))]
        if bs == 1:
            D = compute_D(h, N)
            Dz = D[zeros, :]
            best_new_P = P + 1.0
            best_pq = None
            p_list = ones if len(ones) <= 400 else np.random.choice(ones, 400, replace=False)
            for p in p_list:
                base = c - D[p, :]
                cand = base[None, :] - Dz
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
                if npk[mi] < best_new_P:
                    best_new_P = float(npk[mi])
                    best_pq = (int(p), int(zeros[mi]))
            delta = best_new_P - P
            if best_pq and (delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))):
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
            else:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            for b in range(bs):
                pp = p + b if p + b < N else None
                qq = q + b if q + b < N else None
                if pp is None or qq is None:
                    pp2 = p - b if p - b >= 0 else None
                    qq2 = q - b if q - b >= 0 else None
                    if pp2 is None or qq2 is None:
                        break
                    p, q = pp2, qq2
                else:
                    p, q = pp, qq
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy + short SA at N0
    h_bin = ensure_sum(h0, N0)
    d1 = t0 + 20
    h1, c5_1 = greedy_sa(h_bin, N0, d1, 200, 3.0, 0.993, 0.05, 1.0, 150, 4, False)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Main SA at N0 with 2-swaps
    d2 = t0 + 220
    h2, c5_2 = greedy_sa(ensure_sum(best_h, N0), N0, d2, 12000, 4.5, 0.9935, 0.03, 1.0, 120, 5, True)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Second SA with block moves
    d3 = t0 + 400
    h3, c5_3 = block_sa(ensure_sum(best_h, N0), N0, d3, 8000, 5.0, 0.993, 0.03, [1, 2, 3])
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Smoothed-max continuous at N0
    d4 = t0 + 580
    hc = project_sum(best_h.copy(), N0 / 2.0)
    hc = project_sum(hc + np.random.randn(N0) * 0.04, N0 / 2.0)
    h4, c5_4 = smooth_max_grad(hc, N0, d4, T_smooth=60.0, k=50, lr0=0.018, max_iter=10000)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Upsample to 2*N0, SA
    N1 = 2 * N0
    if time.time() < deadline - 250:
        h_up = ensure_sum(upsample(best_h, N1), N1)
        d5 = t0 + 800
        h5, c5_5 = greedy_sa(h_up, N1, d5, 6000, 3.5, 0.994, 0.04, 0.6, 180, 4, False)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Back to N0, diverse restart SA
    if time.time() < deadline - 100:
        h_SA = ensure_sum(upsample(best_h, N0), N0)
        for _ in range(15):
            p = np.where(h_SA == 1)[0]
            q = np.where(h_SA == 0)[0]
            if len(p) > 0 and len(q) > 0:
                h_SA[p[np.random.randint(len(p))]], h_SA[q[np.random.randint(len(q))]] = 0.0, 1.0
        d6 = deadline - 60
        h6, c5_6 = greedy_sa(h_SA, N0, d6, 15000, 5.5, 0.9925, 0.02, 1.0, 80, 7, True)
        if c5_6 < best_c5:
            best_c5, best_h = c6_6 if False else c5_6, h6.copy()

    # Phase 7: Final smooth-max polish
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hf = project_sum(best_h.astype(float).copy(), Nf / 2.0)
        h7, c5_7 = smooth_max_grad(hf, Nf, deadline - 1, T_smooth=80.0, k=60, lr0=0.008, max_iter=5000)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Final: binarize if needed and recompute
    N_fin = len(best_h)
    h_final = ensure_sum(best_h, N_fin)
    c5_bin = compute_c5(h_final)
    c5_cont = compute_c5(best_h)
    if c5_bin < c5_cont:
        best_h, best_c5 = h_final, c5_bin
    else:
        best_c5 = c5_cont

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
