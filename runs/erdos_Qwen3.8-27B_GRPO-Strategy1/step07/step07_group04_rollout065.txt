        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def fista_smooth(h_in, N, deadline, tau=80, k_top=60, lr_init=0.3):
    target = N / 2.0
    x = project_box_sum(h_in, target)
    y = x.copy()
    t = 1.0
    lr = lr_init
    best_x = x.copy()
    best_val = float(np.max(fft_corr(x)))
    center = N - 1
    it = 0
    while time.time() < deadline:
        corr_y = fft_corr(y)
        cmax = float(corr_y.max())
        kk = min(k_top, 2 * N - 1)
        top = np.argpartition(corr_y, -kk)[-kk:]
        w = np.exp(tau * (corr_y[top] - cmax))
        ws = w.sum()
        w = w / ws if ws > 1e-30 else np.ones(len(top)) / len(top)
        g = np.zeros(N)
        for i in range(len(top)):
            r = center - int(top[i])
            if r >= 0 and r < N:
                g[r:] += w[i] * (1.0 - y[:N - r])
                g[:N - r] -= w[i] * y[r:]
            elif r < 0 and r > -N:
                rp = -r
                g[:N - rp] += w[i] * (1.0 - y[rp:])
                g[rp:] -= w[i] * y[:N - rp]
        x_new = project_box_sum(y - lr * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = project_box_sum(y_new, target)
        x, y, t = x_new, y_new, t_new
        val = float(np.max(fft_corr(x)))
        if val < best_val - 1e-14:
            best_val = val
            best_x = x.copy()
        it += 1
        if it % 300 == 0:
            lr *= 0.97
    return best_x, best_val * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy at N0
    h_bin = ensure_bin(h0, N0)
    d1 = min(deadline - 900, t0 + 25)
    h1, c5_1 = greedy(h_bin, N0, d1, max_sweeps=500, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: SA at N0 (long exploration)
    d2 = min(deadline - 750, t0 + 180)
    h2, c5_2 = sa_binary(h_bin, N0, d2, T0=5.0, Tcool=0.993, T_min=0.025,
                         sf=1.0, kick_every=100, kick_size=6, max_sweeps=10000)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: FISTA continuous at N0
    if time.time() < deadline - 600:
        hc = project_box_sum(best_h + np.random.randn(N0) * 0.08, N0 / 2.0)
        d3 = min(deadline - 550, t0 + 320)
        h3, c5_3 = fista_smooth(hc, N0, d3, tau=70, k_top=55, lr_init=0.25)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Greedy + SA at N0 from FISTA result
    if time.time() < deadline - 450:
        h4b = ensure_bin(best_h, N0)
        d4a = min(deadline - 400, t0 + 380)
        h4, c5_4 = greedy(h4b, N0, d4a, max_sweeps=600, sf=1.0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
        d4b = min(deadline - 350, t0 + 450)
        h4s, c5_4s = sa_binary(h4, N0, d4b, T0=3.5, Tcool=0.994, T_min=0.03,
                               sf=0.8, kick_every=150, kick_size=4, max_sweeps=5000)
        if c5_4s < best_c5:
            best_c5, best_h = c5_4s, h4s.copy()

    # Phase 5: Coarse resolution N=380 SA (faster per sweep)
    if time.time() < deadline - 300:
        Nc = N0 // 2
        hc_coarse = upsample(best_h, Nc)
        hc_coarse = ensure_bin(hc_coarse, Nc)
        d5 = min(deadline - 250, t0 + 620)
        h5, c5_5 = sa_binary(hc_coarse, Nc, d5, T0=4.0, Tcool=0.994, T_min=0.03,
                             sf=1.0, kick_every=80, kick_size=4, max_sweeps=8000)
        h5_up = upsample(h5, N0)
        h5_up = ensure_bin(h5_up, N0)
        c5_5u = compute_c5(h5_up)
        if c5_5u < best_c5:
            best_c5, best_h = c5_5u, h5_up.copy()
        elif c5_5 < compute_c5(h5):
            pass  # keep best at fine resolution

    # Phase 6: FISTA at N0 with perturbation from best
    if time.time() < deadline - 180:
        hc2 = best_h + np.random.randn(N0) * 0.05
        hc2 = project_box_sum(hc2, N0 / 2.0)
        d6 = min(deadline - 150, t0 + 820)
        h6, c5_6 = fista_smooth(hc2, N0, d6, tau=100, k_top=40, lr_init=0.15)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Greedy + SA polish from FISTA binary
    if time.time() < deadline - 100:
        h7b = ensure_bin(best_h, N0)
        d7a = min(deadline - 80, t0 + 900)
        h7, c5_7 = greedy(h7b, N0, d7a, max_sweeps=800, sf=1.0)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()
        d7b = min(deadline - 50, t0 + 950)
        h7s, c5_7s = sa_binary(h7, N0, d7b, T0=2.5, Tcool=0.995, T_min=0.02,
                               sf=1.0, kick_every=200, kick_size=3, max_sweeps=4000)
        if c5_7s < best_c5:
            best_c5, best_h = c5_7s, h7s.copy()

    # Phase 8: Final greedy sweep
    N_fin = len(best_h)
    d8 = deadline - 1
    sf_fin = 1.0 if N_fin <= 500 else 0.7
    hF = ensure_bin(best_h, N_fin)
    hF2, c5F = greedy(hF, N_fin, d8, max_sweeps=2000, sf=sf_fin)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    return best_h, compute_c5(best_h), len(best_h)
