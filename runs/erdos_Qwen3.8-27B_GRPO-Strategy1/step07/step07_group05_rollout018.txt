def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        to_remove = np.random.choice(idx, n1 - target, replace=False)
        h[to_remove] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        to_add = np.random.choice(idx, target - n1, replace=False)
        h[to_add] = 1
    return h

def soft_max_grad(corr, T):
    shifted = corr / T
    m = np.max(shifted)
    exps = np.exp(shifted - m)
    w = exps / exps.sum()
    return w  # weights over shifts

def continuous_step(h, N, T_smooth, lr):
    corr = fft_corr(h)
    w = soft_max_grad(corr, T_smooth)
    # Gradient of corr[k] w.r.t. h[j]
    # corr has length 2N-1, center at N-1
    grad = np.zeros(N)
    center = N - 1
    # Only process significant shifts
    sig = np.where(w > 1e-4)[0]
    for ki in sig:
        L = int(ki - center)  # shift
        wt = w[ki]
        if L == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif L > 0:
            if L < N:
                grad[L:] += wt * (1.0 - h[:N-L])
                grad[:N-L] -= wt * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N-M]

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy()
    n1 = int(np.round(h.sum()))
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def fix_sum_continuous(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = (-2.0, 0.0) if s > target else (0.0, 2.0)
    for _ in range(40):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def continuous_refine(h, N, deadline, lr0=0.003, T0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    T = T0
    it = 0
    while time.time() < deadline - 1:
        it += 1
        corr = fft_corr(h)
        shifted = corr / T
        m = np.max(shifted)
        exps = np.exp(shifted - m)
        w = exps / exps.sum()
        grad = np.zeros(N)
        center = N - 1
        sig = np.where(w > 1e-4)[0]
        for ki in sig:
            L = int(ki - center)
            wt = w[ki]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[L:] += wt * (1.0 - h[:N - L])
                grad[:N - L] -= wt * h[L:]
            elif -N < L < 0:
                M = -L
                grad[:N - M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N - M]
        h = fix_sum_continuous(np.clip(h - lr * grad, 0, 1), target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 200 == 0:
            lr *= 0.96
            T *= 0.93
    return best_h, best_c

def anneal_binary(h, N, deadline, max_iters=50000, T0=0.05, T1=1e-5,
                  two_swap_prob=0.2):
    h = fix_sum_binary(h.astype(float).copy(), N // 2)
    cur_max = float(np.max(fft_corr(h)))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = T0
    cool = (T1 / T0) ** (1.0 / max(1, max_iters))
    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if np.random.random() < two_swap_prob:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            o1, o2 = np.random.choice(ones, 2, replace=False)
            z1, z2 = np.random.choice(zeros, 2, replace=False)
            h[o1], h[o2], h[z1], h[z2] = 0, 0, 1, 1
            new_max = float(np.max(fft_corr(h)))
            dE = new_max - cur_max
            if dE < -1e-14 or (T > 1e-7 and np.random.random() < np.exp(-dE / T)):
                cur_max = new_max
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c, best_h, no_improve = cur_max * 2.0 / N, h.copy(), 0
                else:
                    no_improve += 1
            else:
                h[o1], h[o2], h[z1], h[z2] = 1, 1, 0, 0
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi], h[zi] = 0, 1
            new_max = float(np.max(fft_corr(h)))
            dE = new_max - cur_max
            if dE < -1e-14 or (T > 1e-7 and np.random.random() < np.exp(-dE / T)):
                cur_max = new_max
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c, best_h, no_improve = cur_max * 2.0 / N, h.copy(), 0
                else:
                    no_improve += 1
            else:
                h[oi], h[zi] = 1, 0
        T *= cool
        if no_improve > 4000:
            for _ in range(np.random.randint(3, 8)):
                oi = int(np.random.choice(np.where(h == 1)[0]))
                zi = int(np.random.choice(np.where(h == 0)[0]))
                h[oi], h[zi] = h[zi], h[oi]
            cur_max = float(np.max(fft_corr(h)))
            T = max(T, 0.025)
            no_improve = 0
    return best_h, best_c

def structured_init(N, variant):
    np.random.seed(8888 + variant * 619)
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[np.random.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        x = np.arange(N)
        v = np.sin(2 * np.pi * 7 * x / N) + 0.8 * np.sin(2 * np.pi * 13 * x / N)
        h = (v > np.median(v)).astype(float)
    else:
        x = np.arange(N)
        v = sum(np.sin(2 * np.pi * f * x / N + f * 0.4) for f in [5, 11, 19, 29])
        h = (v > np.median(v)).astype(float)
    return fix_sum_binary(h, nh)
