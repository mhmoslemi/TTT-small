def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def project_sum(h, target):

            lo = mid
        else:
            hi = mid

def compute_D(h, N):
    """Compute the change in corr when setting h[i]=0 for each i and each shift."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def subgradient_opt(h, N, deadline, k_active=30, rng=None):
    """Subgradient method with active set for the max."""
    if rng is None:
        rng = np.random.RandomState(1)
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    lr = 0.02
    it = 0
    
    while time.time() < deadline:
        corr = fft_corr(h)
        # Find active set
        top_idx = np.argpartition(corr, -k_active)[-k_active:]
        pk = float(corr[top_idx].max())
        
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        # Compute subgradient: average of gradients over active set
        # For lag L: grad[m] += (1-h[m+L])*1[m<N-L] - h[m-L]*1[m>=L]
        center = N - 1
        grad = np.zeros(N)
        for ai in top_idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[:N-L] += (1.0 - h[L:])
                grad[L:] -= h[:N-L]
            elif L < 0:
                M = -L
                if M < N:
                    grad[M:] += (1.0 - h[:N-M])
                    grad[:N-M] -= h[M:]
        
        grad /= k_active
        h_new = project_sum(h - lr * grad, target)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.98

            if c_bin < pk:
                h = h_bin
            # Small perturbation
            h = project_sum(h + rng.randn(N) * 0.02, target)
    
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum_binary(h, N, rng_seed=42):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    r = np.random.RandomState(rng_seed)
    if n1 > nh:
        idx = r.choice(np.where(h_bin == 1)[0], n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        idx = r.choice(np.where(h_bin == 0)[0], nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, rng, max_iter=2000, n_sample=50, do_3swap=0.15, n_2swap=80):
    h = ensure_sum_binary(h, N, seed + 7)
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())

    best_h = h.copy()
    D = compute_D(h, N)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        n_s = min(n_sample, len(ones))
        p_list = rng.choice(ones, n_s, replace=False) if n_s < len(ones) else ones
        D_z = D[zeros, :]
        best_1P = P_cur
        best_1m = None
        for p in p_list:
            base = c_cur + D[p]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_1P:
                best_1P = float(npk[mi])
                best_1m = (int(p), int(zeros[mi]))
        best_2P = best_1P
        best_2m = None
        if len(ones) >= 2 and len(zeros) >= 2:

def subgrad_opt(h, N, deadline, k_act=35, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    lr = 0.025
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        top = np.argpartition(corr, -k_act)[-k_act:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[:N - L] += (1.0 - h[L:])
                grad[L:] -= h[:N - L]
            elif -N < L < 0:
                M = -L
                grad[M:] += (1.0 - h[:N - M])
                grad[:N - M] -= h[M:]
        grad /= k_act
        h = project_sum(h - lr * grad, target)
        it += 1
        if it % 80 == 0:
            lr *= 0.985
        if it % 400 == 0:
            hb = ensure_sum_binary(h, N, rng.randint(10000))
            cb = float(np.max(fft_corr(hb)))
            if cb < pk:
                h = hb.copy()
            h = project_sum(h + rng.randn(N) * 0.025, target)
    return best_h, best_pk * dx

def upsample_to(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
