def fft_corr(h):

    target = N // 2
    h = h.copy()
    h_bin = (h >= 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:

    rng = np.random.RandomState(42)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        # Remove random ones
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):

    pos = np.arange(N)[:,None]
    sv = shifts[None,:]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)

    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):

            break
        P = float(corr.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break

def nesterov_refine(h, N, deadline, k=40, lr0=0.015, n_iters=None):

    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        # Top-k gradient
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0*y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]

        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.985
    return best_h, best_pk * dx

def _project(h, target):
    h = np.clip(h, 0, 1)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def upsample(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

    k_top = np.argpartition(corr, -top_k)[-top_k:]
    Dz_top = D[zeros, k_top]  # (n0, top_k)
    c_top = corr[k_top]  # (top_k,)
    
    n_s = len(ones) if sf >= 1.0 else max(50, int(len(ones)*sf))
    p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
    
    best_val = P + 1
    best_pq = None
    for p in p_list:
        Dp_top = D[p, k_top]
        cand = c_top[None,:] + Dp_top[None,:] - Dz_top  # (n0, top_k)
        npk = cand.max(axis=1)  # (n0,)
        mi = int(np.argmin(npk))
        if npk[mi] < best_val:
            best_val = float(npk[mi])
            best_pq = (int(p), int(zeros[mi]))

    if best_pq is not None:
        p, q = best_pq
        full_new = corr + D[p,:] - D[q,:]
        full_max = float(full_new.max())
        # If full max is worse than top-k estimate, still return it
        # but the caller should use full_max
        return best_pq, full_max, full_new
    
    return None, P, corr

def quick_swap(corr, D, ones, zeros, topk=30):
    """Fast swap using top-k screen. Returns (p, q, new_corr or None)."""
    P = float(corr.max())
    kt = np.argpartition(corr, -topk)[-topk:]
    ck = corr[kt]
    Dzk = D[zeros, kt]
    best_v = P + 1

        cand = ck[None,:] + Dpk[None,:] - Dzk
        nk = cand.max(axis=1)
        mi = int(np.argmin(nk))

            best_v = float(nk[mi])
            best_pq = (int(p), int(zeros[mi]))

    target, dx = N/2.0, 2.0/N
    corr = fft_corr(h)
    bp, bh = float(corr.max()), h.copy()
    m = np.zeros(N)
    lr, beta = lr0, 0.9
    it = 0

        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < bp - 1e-14: bp, bh = pk, h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        c = N-1
        for ai in top:
            L = int(ai-c)
            if L==0: grad += 1-2*y
            elif L>0:
                grad[L:] += 1-y[:N-L]
                grad[:N-L] -= y[L:]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    K = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * K, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def ensure_bin(h, N):
    target = N // 2
    h = h.astype(float).copy()
    hb = (h >= 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == target: return hb
    rng = np.random.RandomState(99)
    if n1 > target:
        idx = rng.choice(np.where(hb==1)[0], n1-target, replace=False)
        hb[idx] = 0
    else:
        idx = rng.choice(np.where(hb==0)[0], target-n1, replace=False)
        hb[idx] = 1
    return hb

def comp_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:,None]
    sv = shifts[None,:]
    pm, pp = pos-sv, pos+sv
    vm, vp = (pm>=0)&(pm<N), (pp>=0)&(pp<N)
    hm = np.where(vm, h[np.clip(pm,0,N-1)], 0.0)
    hp = np.where(vp, h[np.clip(pp,0,N-1)], 0.0)
    return -(1.0-hm)*vm + hp*vp

def quick_swap(corr, D, ones, zeros, topk=30):
    P = float(corr.max())
    kt = np.argpartition(corr, -topk)[-topk:]
    ck = corr[kt]
    Dzk = D[zeros, kt]
    best_v = P + 1
    best_pq = None
    for p in ones:
        Dpk = D[p, kt]
        cand = ck + Dpk - Dzk
        nk = cand.max(axis=1)
        mi = int(np.argmin(nk))
        if nk[mi] < best_v:
            best_v = float(nk[mi])
            best_pq = (int(p), int(zeros[mi]))
    if best_pq is None: return None
    p, q = best_pq
    nc = corr + D[p,:] - D[q,:]
    return p, q, nc

def sa_loop(h, N, deadline, T0=4.0, Tcool=0.996, Tmin=0.01,
            kick_every=80, kick_n=4, topk=30):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    bc, bh = float(corr.max()), h.copy()
    T, stall = T0, 0
    while time.time() < deadline:
        P = float(corr.max())
        if P < bc - 1e-12: bc, bh, stall = P, h.copy(), 0
        else: stall += 1
        ones = np.where(h>0.5)[0]
        zeros = np.where(h<=0.5)[0]
        if len(ones)<2 or len(zeros)<2: break
        if stall >= kick_every:
            for _ in range(kick_n):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p],h[q] = 0.,1.
            stall, T = 0, T0*0.5
            corr = fft_corr(h)
            continue
        D = comp_D(h, N)
        res = quick_swap(corr, D, ones, zeros, topk)
        if res is None: break
        p,q,nc = res
        nP = float(nc.max())
        d = nP - P
        if d < -1e-14 or (T>Tmin and np.random.random()<np.exp(-max(0,d)/max(T,1e-10))):
            h[p],h[q] = 0.,1.
            corr = nc
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p],h[q] = 0.,1.
            corr = fft_corr(h)
        T = max(Tmin, T*Tcool)
    return bh, bc*(2.0/N)

def greedy_loop(h, N, deadline, topk=30):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    bc, bh = float(corr.max()), h.copy()
    stall = 0
    while time.time() < deadline:
        P = float(corr.max())
        if P < bc - 1e-12: bc, bh, stall = P, h.copy(), 0
        else:
            stall += 1
            if stall > 30: break
        ones = np.where(h>0.5)[0]
        zeros = np.where(h<=0.5)[0]
        if len(ones)<2 or len(zeros)<2: break
        D = comp_D(h, N)
        res = quick_swap(corr, D, ones, zeros, topk)
        if res is None: break
        p,q,nc = res
        if float(nc.max()) >= P - 1e-13: break
        h[p],h[q] = 0.,1.
        corr = nc
    return bh, bc*(2.0/N)

def nesterov(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N/2.0
    dx = 2.0/N
    corr = fft_corr(h)
    bp, bh = float(corr.max()), h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < bp - 1e-14: bp, bh = pk, h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        c = N-1
        for ai in top:
            L = int(ai-c)
            if L==0:
                grad += 1-2*y
            elif L>0:
                grad[L:] += 1-y[:N-L]
                grad[:N-L] -= y[L:]
            else:
                M = -L
                if M<N:
                    grad[:N-M] += 1-y[M:]
                    grad[M:] -= y[:N-M]
        grad /= k
        hn = _proj(y-lr*grad, target)
        m = beta*(hn-h)
        h = hn
        it += 1
        if it%200==0: lr *= 0.99
    return bh, bp*dx
