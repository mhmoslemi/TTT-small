def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_c5_corr(h, N):
    # Use np.correlate for exact evaluation
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    shifted = T * (c - mx)
    # Numerical stability
    max_shifted = shifted.max()
    e = np.exp(shifted - max_shifted)
    w = e / e.sum()
    val = mx + (np.log(e.sum()) + max_shifted) / T
    g = np.zeros(N)
    ctr = N - 1
    active = np.where(w > 1e-8)[0]
    for i in active:
        g += w[i] * lin_grad(int(i - ctr), h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_run(h, N, deadline, T, target, max_outer, maxiter=200, maxfun=600):
    h = h.astype(float).copy()

                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-15, 'gtol': 1e-10})
        h = proj01(res.x, target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lam *= 2.0
    return best_h, best_c

def subgrad_fista(h, N, deadline, alpha0=0.04, atol=1e-4, max_iter=5000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    y = h.copy()
    t_k = 1.0
    m = np.zeros(N)
    alpha = alpha0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = np.clip(y, 0, 1)
        corr = fft_corr(y)

        # Score for flipping 0→1 at position j: reduces corr by -grad[j]
        # Total improvement: grad[i] + (-grad[j]) = grad[i] - grad[j]

        si = grad[ones]  # want large

        # Best: maximize si - sj
        # Top candidates
        top_i = np.argsort(si)[-10:]
        top_j = np.argsort(sj)[:10]

    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - tgt) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in np.where(w > 1e-7)[0]:
        g += w[i] * lgrad(int(i - ctr), h, N)
    s = h.sum() - tgt
    val += 0.5 * lam * s * s
    g[:] += lam * s
    return val, g

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - tgt) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def og(h, T, tgt, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in np.where(w > 1e-7)[0]:
        g += w[i] * lgrad(int(i - ctr), h, N)
    s = h.sum() - tgt
    val += 0.5 * lam * s * s
    g[:] += lam * s
    return val, g

def lbfgs(h, N, dl, T, tgt, outer=3, mi=250, mf=800):
    bh, bc = h.copy(), c5v(h)
    lam = 5.0
    for _ in range(outer):
        if time.time() > dl:
            break
        try:
            r = minimize(og, h, args=(T, tgt, lam), jac=True,
                         method='L-BFGS-B', bounds=[(0, 1)] * N,
                         options={'maxiter': mi, 'maxfun': mf, 'ftol': 1e-15, 'gtol': 1e-10})
            h = proj(r.x, tgt)
        except Exception:
            break
        c = c5v(h)
        if c < bc:
            bc, bh = c, h.copy()
        lam *= 2.0
    return bh, bc

def fista(h, N, dl, a0=0.03, atol=1e-4, mxit=5000):
    tgt = N / 2.0
    bh, bpk = h.copy(), np.max(fft_corr(h))
    y, tk, alpha = h.copy(), 1.0, a0
    for it in range(mxit):
        if time.time() > dl:
            break
        y = np.clip(y, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < bpk:
            bpk, bh = pk, h.copy()
        act = np.where(corr >= pk - atol)[0]
        g = np.zeros(N)
        ctr = N - 1
        for ai in act:
            g += lgrad(int(ai - ctr), y, N)
        g /= len(act)
        hn = proj(y - alpha * g, tgt)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        y = np.clip(hn + (tk - 1.0) / tn * (hn - h), 0, 1)
        h, tk = hn, tn
        if (it + 1) % 500 == 0:
            alpha *= 0.95
    return bh, bpk * (2.0 / N)

def bsearch(h, N, dl, mxi=300, ntr=20, ann=0.0, rng=None):
    if rng is None:
        rng = np.random
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        for j in rng.choice(np.where(h == 1)[0], n1 - nh, replace=False):
            h[j] = 0
    elif n1 < nh:
        for j in rng.choice(np.where(h == 0)[0], nh - n1, replace=False):
            h[j] = 1
    bc, bh = float(np.max(fft_corr(h))), h.copy()
    stall, temp = 0, ann
    for it in range(mxi):
        if time.time() > dl or stall > 200:
            break
        P = float(np.max(fft_corr(h)))
        imp = P < bc - 1e-14
        if imp:
            bc, bh = P, h.copy()
        ones, zeros = np.where(h == 1)[0], np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bd, bi, bj = 0.0, -1, -1
        for _ in range(ntr):
            oi = int(ones[rng.randint(len(ones))])
            zi = int(zeros[rng.randint(len(zeros))])
            h[oi], h[zi] = 0.0, 1.0
            d = P - float(np.max(fft_corr(h)))
            if d > bd + temp:
                bd, bi, bj = d, oi, zi
            h[oi], h[zi] = 1.0, 0.0
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            imp = True
        stall = 0 if imp else stall + 1
        temp = max(0.0, ann * (1.0 - it / mxi))
    return bh, bc

def mres(bh, bc, dl, rng):
    Nc = len(bh)
    xb = np.linspace(0, 2, Nc, endpoint=False)
    for Nt in [240, 280, 300, 320, 340, 360, 400, 440, 480, 520, 600, 700, 800]:
        if time.time() > dl or Nt % 2 or Nt == Nc:
            continue
        ht = (np.interp(np.linspace(0, 2, Nt, endpoint=False), xb, bh) > 0.5).astype(float)
        nh, n1 = Nt // 2, int(ht.sum())
        if n1 > nh:
            for j in rng.choice(np.where(ht == 1)[0], n1 - nh, replace=False):
                ht[j] = 0
        elif n1 < nh:
            for j in rng.choice(np.where(ht == 0)[0], nh - n1, replace=False):
                ht[j] = 1
        hr, _ = bsearch(ht, Nt, min(dl, time.time() + 20), 120, 14, 0.0, rng)
        cr = c5v(hr)
        if cr < bc:
            bc, bh = cr, hr.copy()
            xb = np.linspace(0, 2, Nt, endpoint=False)
    return bh, bc
