import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    return h

def corr_gradient(h, N, weights):
    """Compute gradient of weighted sum of correlations."""
    grad = np.zeros(N)
    for k_idx, w in enumerate(weights):
        if w < 1e-12:
            continue
        L = k_idx - (N - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] += w * (1.0 - h[L:])
            grad[L:] += -w * h[:N - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:N - M])
            grad[:N - M] += -w * h[M:]
    return grad

def continuous_opt(h_init, N, deadline, tau_init=0.3, n_restarts=3):
    """Softmax-weighted gradient descent with ADAM."""
    t0 = time.time()
    per_restart = (deadline - t0) / max(n_restarts, 1)
    best_h = h_init.copy()
    best_pk = compute_c5(h_init, N)
    
    for r in range(n_restarts):
        rem = deadline - time.time()
        if rem < 10:
            break
        if r > 0:
            h = best_h.copy() + np.random.randn(N) * 0.1
            h = np.clip(h, 0, 1)
            h = np.clip(h - (h.sum() - N/2.0)/N, 0, 1)
        else:
            h = h_init.copy()
        
        m_adam = np.zeros(N)
        v_adam = np.zeros(N)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        lr = 0.05
        tau = tau_init
        it = 0
        
        while time.time() < t0 + (r+1) * per_restart - 2:
            corr = fft_corr(h)
            pk = float(corr.max())
            if pk < best_pk:
                best_pk = pk
                best_h = h.copy()
            
            # Softmax weights
            shift = corr - corr.max()
            w_raw = np.exp(shift / tau)
            w_sum = w_raw.sum()
            if w_sum > 0:
                weights = w_raw / w_sum
            else:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            
            grad = corr_gradient(h, N, weights)
            
            # ADAM update
            it += 1
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * grad**2
            m_hat = m_adam / (1 - beta1**it)
            v_hat = v_adam / (1 - beta2**it)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = np.clip(h_new, 0, 1)
            h_new = np.clip(h_new - (h_new.sum() - N/2.0)/N, 0, 1)
            h = h_new
            
            if it % 100 == 0:
                lr *= 0.97
                tau *= 0.95
    
    return best_h, best_pk * (2.0/N)

        hc2 = np.clip(hc2 - (hc2.sum() - N0 / 2.0) / N0, 0, 1)
        hn2, c4 = adam_smooth(hc2, N0, d4, tau=0.08, lr_init=0.015)
        if c4 < best_c5:
            best_c5, best_h = c4, hn2.copy()

    # Phase 5: Final SA polish (very low temp, many peaks)
    if time.time() < deadline - 5:
        h5 = make_binary(best_h, N0)
        d5 = deadline - 3
        h5b, c5 = sa_search(h5, d5, n_peaks=10, n_per_peak=30,
                            cool_rate=0.9998, t_start=0.012)
        if c5 < best_c5:
            best_c5, best_h = c5, h5b.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
