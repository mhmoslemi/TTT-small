```python
def compute_swap_delta(h, a, b, N):
    s = np.arange(-(N-1), N)  # 759 values for N=380
    delta = np.zeros(2*N-1)
    
    # Term 1: -(1-h[a+s]) for valid a+s
    valid1 = (a+s >= 0) & (a+s < N)
    delta[valid1] -= (1 - h[a+s[valid1]])
    
    # Term 2: +(1-h[b+s]) for valid b+s
    valid2 = (b+s >= 0) & (b+s < N)
    delta[valid2] += (1 - h[b+s[valid2]])
    
    # Term 3: +h[a-s] for valid a-s and a-s≠b
    valid3 = (a-s >= 0) & (a-s < N) & (a-s != b)
    delta[valid3] += h[a-s[valid3]]
    
    # Term 4: -h[b-s] for valid b-s and b-s≠a
    valid4 = (b-s >= 0) & (b-s < N) & (b-s != a)
    delta[valid4] -= h[b-s[valid4]]
    
    return delta
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def init_corr(h):
    N = len(h)
    corr = np.zeros(2 * N - 1)
    for s in range(-(N - 1), N):
        lo = max(0, -s)
        hi = min(N, N - s)
        if lo < hi:
            corr[s + N - 1] = np.dot(h[lo:hi], 1.0 - h[lo + s:hi + s])
    return corr

def swap_delta(h, a, b, N):
    s = np.arange(-(N - 1), N)
    delta = np.zeros(2 * N - 1)
    v1 = (a + s >= 0) & (a + s < N)
    delta[v1] -= (1.0 - h[a + s[v1]])
    v2 = (b + s >= 0) & (b + s < N)
    delta[v2] += (1.0 - h[b + s[v2]])
    v3 = (a - s >= 0) & (a - s < N) & (a - s != b)
    delta[v3] += h[a - s[v3]]
    v4 = (b - s >= 0) & (b - s < N) & (b - s != a)
    delta[v4] -= h[b - s[v4]]
    return delta

def greedy_binary(h, N, deadline, max_rounds=5000, ntry=300):
    corr = init_corr(h)
    cur_max = float(np.max(corr))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_h = h.copy()
    best_max = cur_max
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        worst_idx = np.argmax(corr)
        s_star = int(worst_idx - (N - 1))
        improved = False
        best_d = 0.0
        best_pa, best_pb = -1, -1
        for _ in range(ntry):
            ai = int(np.random.randint(len(ones)))
            bi = int(np.random.randint(len(zeros)))
            a, b = int(ones[ai]), int(zeros[bi])
            d = swap_delta(h, a, b, N)
            nm = float(np.max(corr + d))
            if nm < cur_max - 1e-12:
                if nm < best_d:
                    best_d, best_pa, best_pb = nm, a, b
                d2 = 0.0
                d2 -= (1.0 - (h[a + s_star] if 0 <= a + s_star < N else 0))
                d2 += (1.0 - (h[b + s_star] if 0 <= b + s_star < N else 0))
                if 0 <= a - s_star < N and a - s_star != b:
                    d2 += h[a - s_star]

                    d2 -= h[b - s_star]

            corr += d
            h[best_pa], h[best_pb] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_max = float(np.max(corr))
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            improved = True
        if not improved:

    return best_h, best_max

    n1 = int(ht.sum())
    if n1 > nh:
        ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1
    sd1 = min(deadline, time.time() + 40)
    ht, _ = greedy_binary(ht, N_t, sd1, max_rounds=3000, ntry=150)
    c = float(np.max(fft_corr(ht))) * (2.0 / N_t)
    if c < c5_best:
        sd2 = min(deadline, time.time() + 30)
        ht, _ = anneal_binary(ht, N_t, sd2, t0_val=0.15, t_end=0.005, max_iter=3000)
        c = float(np.max(fft_corr(ht))) * (2.0 / N_t)
    return ht, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h = (h0 > 0.5).astype(float)
    target = N0 // 2
    s = int(h.sum())
    if s > target:
        h[np.random.choice(np.where(h == 1)[0], s - target, replace=False)] = 0
    elif s < target:
        h[np.random.choice(np.where(h == 0)[0], target - s, replace=False)] = 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: targeted sweep (fast, deterministic)
    d1 = min(dl - 500, t0 + 80)
    hb, mb = targeted_sweep(h, N0, d1, max_rounds=3000)
    c1 = float(np.max(fft_corr(hb))) * (2.0 / N0)
    if c1 < best_c5:
        best_c5, best_h = c1, hb.copy()

    # Phase 2: greedy random
    d2 = min(dl - 400, time.time() + 200)
    hb2, mb2 = greedy_binary(best_h, N0, d2, max_rounds=6000, ntry=250)
    c2 = float(np.max(fft_corr(hb2))) * (2.0 / N0)
    if c2 < best_c5:
        best_c5, best_h = c2, hb2.copy()

    # Phase 3: simulated annealing
    if time.time() < dl - 300:
        d3 = min(dl - 250, time.time() + 200)
        hb3, mb3 = anneal_binary(best_h, N0, d3, t0_val=0.25, t_end=0.002, max_iter=5000)
        c3 = float(np.max(fft_corr(hb3))) * (2.0 / N0)
        if c3 < best_c5:
            best_c5, best_h = c3, hb3.copy()

    # Phase 4: another targeted sweep
    if time.time() < dl - 200:
        d4 = min(dl - 150, time.time() + 120)
        hb4, mb4 = targeted_sweep(best_h, N0, d4, max_rounds=3000)
        c4 = float(np.max(fft_corr(hb4))) * (2.0 / N0)
        if c4 < best_c5:
            best_c5, best_h = c4, hb4.copy()

    # Phase 5: multi-resolution
    for Nt in [200, 260, 300, 340, 420, 460, 520, 600]:
        if time.time() > dl - 100:
            break
        if Nt == N0:
            continue
        ht, ct = try_resolution(best_h, best_c5, Nt, min(dl - 60, time.time() + 50))
        if ct < best_c5:
            best_c5, best_h = ct, ht.copy()

    # Phase 6: final targeted sweep on best
    if time.time() < dl - 40:
        d6 = min(dl - 20, time.time() + 80)
        hb6, _ = targeted_sweep(best_h, len(best_h), d6, max_rounds=2000)
        c6 = float(np.max(fft_corr(hb6))) * (2.0 / len(best_h))
        if c6 < best_c5:
            best_c5, best_h = c6, hb6.copy()

    # Phase 7: final greedy
    if time.time() < dl - 10:
        d7 = min(dl - 5, time.time() + 50)
        hb7, _ = greedy_binary(best_h, len(best_h), d7, max_rounds=3000, ntry=200)
        c7 = float(np.max(fft_corr(hb7))) * (2.0 / len(best_h))
        if c7 < best_c5:
            best_c5, best_h = c7, hb7.copy()

    return best_h, best_c5, len(best_h)
```
