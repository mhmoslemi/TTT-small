def ensure_binary(h, N, seed=77):

    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin

    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def thue_morse_binary(N, seed=0):
    """Generate Thue-Morse-like binary sequence with exactly N/2 ones."""
    # Thue-Morse: t(n) = popcount(n) mod 2
    t = np.array([bin(i).count('1') % 2 for i in range(N)], dtype=float)
    n1 = int(t.sum())
    target = N // 2
    if n1 == target:
        return t
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(t == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        t[idx] = 0
    else:
        zeros = np.where(t == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        t[idx] = 1
    return t

def periodic_binary(N, period=16, phase=0, seed=0):
    """Generate periodic binary pattern."""
    # Use a pattern that avoids long runs
    base = np.zeros(N)
    # Simple: alternating with occasional breaks
    for i in range(N):
        base[i] = ((i // period + phase) % 2)
    # Add some structure
    n1 = int(base.sum())
    target = N // 2
    if n1 != target:
        rng = np.random.RandomState(seed)
        if n1 > target:
            ones = np.where(base == 1)[0]
            idx = rng.choice(ones, n1 - target, replace=False)
            base[idx] = 0
        else:
            zeros = np.where(base == 0)[0]
            idx = rng.choice(zeros, target - n1, replace=False)
            base[idx] = 1
    return base

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def tm_binary(N, seed=0):
    t = np.array([bin(i).count('1') % 2 for i in range(N)], dtype=float)
    n1, target = int(t.sum()), N // 2
    if n1 == target:
        return t
    rng = np.random.RandomState(seed)
    if n1 > target:
        t[rng.choice(np.where(t==1)[0], n1-target, replace=False)] = 0
    else:
        t[rng.choice(np.where(t==0)[0], target-n1, replace=False)] = 1
    return t

def binary_local_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, pert_every=25, two_swap=False):
    h = ensure_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(456)
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    vm_f = vm.astype(float)
    vp_f = vp.astype(float)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c, best_h, stall = P, h.copy(), 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm_f + hp * vp_f
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            p_list = rng.choice(ones, max(30, int(len(ones)*sample_frac)), replace=False)
        else:
            p_list = ones
        best_np, best_pq = P, None
        for p in p_list:
            cand = (c + D[p,:])[None,:] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_np - 1e-14:
                best_np, best_pq = float(npk[mi]), (int(p), int(zeros[mi]))
        did2 = False
        if two_swap and stall > 6 and best_pq is None:
            nos, nzs = min(12, len(ones)), min(12, len(zeros))
            os_e = rng.choice(ones, nos, replace=False)
            zs_e = rng.randrange(nos, nzs) if nzs > 1 else 1
            zs_e = rng.choice(zeros, nzs, replace=False)
            for a in range(nos):
                for b in range(a+1, nos):
                    d12 = c + D[int(os_e[a]),:] + D[int(os_e[b]),:]
                    for cc in range(nzs):
                        d12q = d12 - D[int(zs_e[cc]),:]
                        if nzs - cc - 1 > 0:
                            cands = d12q[None,:] - D[zs_e[cc+1:]]
                            npk = cands.max(axis=1)
                            mi2 = int(np.argmin(npk))
                            if npk[mi2] < best_np - 1e-14:
                                best_np = float(npk[mi2])
                                best_pq = (int(os_e[a]), int(os_e[b]), int(zs_e[cc]), int(zs_e[cc+1+mi2]))
                        did2 = True
            if best_pq is not None and len(best_pq) == 4:
                h[best_pq[0]]=0; h[best_pq[1]]=0; h[best_pq[2]]=1; h[best_pq[3]]=1
        if best_pq is not None and len(best_pq) == 2:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        elif not did2:
            if stall > pert_every:
                nr = min(3 + stall // 8, 8)
                for _ in range(nr):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > pert_every + 15:
                bl = rng.randint(4, min(15, N//12))
                st = rng.randint(0, N - bl)
                h[st:st+bl] = 1.0 - h[st:st+bl]
                h = ensure_binary(h, N, seed=int(rng.randint(10000)))
                stall = 0
            else:
                break
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h, best_c5 = h0.copy(), compute_c5(h0, N0)
    # Phase 1: Continuous Nesterov from initial
    d1 = min(deadline - 520, time.time() + 140)
    hc = h0 + np.random.randn(N0) * 0.05
    h_n, c5_n = nesterov_opt(project_sum(hc, N0/2.0), N0, d1, k=40, T=80.0, lr0=0.018)
    if c5_n < best_c5:
        best_c5, best_h = c5_n, h_n.copy()
    # Phase 2: Primary binary local search at N0
    if time.time() < deadline - 420:
        d2 = min(deadline - 380, time.time() + 190)
        h_br, c5_br = binary_local_search(ensure_binary(best_h, N0), N0, d2, max_sweeps=900, sample_frac=1.0, pert_every=22, two_swap=True)
        if c5_br < best_c5:
            best_c5, best_h = c5_br, h_br.copy()
    # Phase 3: Multi-start at N_sm
    N_sm = N0 // 2
    if time.time() < deadline - 260:
        h_tm = tm_binary(N_sm, seed)
        d3a = min(deadline - 220, time.time() + 55)
        h_tm_o, _ = nesterov_opt(h_tm, N_sm, d3a, k=25, T=50.0, lr0=0.02)
        if time.time() < deadline - 180:
            h_tm_b, c5_tm = binary_local_search(ensure_binary(h_tm_o, N_sm), N_sm, min(deadline-170, time.time()+50), max_sweeps=250, sample_frac=0.7, pert_every=18, two_swap=True)
            h_up = project_sum(upsample(h_tm_b, N_sm, N0), N0/2.0)
            c5_up = compute_c5(h_up, N0)
            if c5_up < best_c5:
                best_c5, best_h = c5_up, h_up.copy()
    # Phase 4: Second continuous pass
    if time.time() < deadline - 160:
        hc2 = best_h + np.random.randn(len(best_h)) * 0.03
        d4 = min(deadline - 130, time.time() + 75)
        h4, c5_4 = nesterov_opt(project_sum(hc2, len(best_h)/2.0), len(best_h), d4, k=50, T=130.0, lr0=0.01)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
    # Phase 5: Final binary polish
    N_fin = len(best_h)
    if time.time() < deadline - 40:
        sf = 1.0 if N_fin <= 600 else 0.65
        d5 = min(deadline - 5, time.time() + 95)
        hF, c5F = binary_local_search(ensure_binary(best_h, N_fin, seed=99), N_fin, d5, max_sweeps=600, sample_frac=sf, pert_every=18, two_swap=True)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    # Phase 6: Micro continuous
    if time.time() < deadline - 4:
        hFc = best_h + np.random.randn(len(best_h)) * 0.012
        d6 = min(deadline - 2, time.time() + 22)
        h6, c5_6 = nesterov_opt(project_sum(hFc, len(best_h)/2.0), len(best_h), d6, k=50, T=200.0, lr0=0.005)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
