            old_obj = float(corr.max())
        dc = new_obj - old_obj
        if dc < -1e-12 or (T > 0.0008 and np.random.random() < np.exp(-dc / (T + 1e-10))):
            h[p], h[q] = 0.0, 1.0
            corr = new_corr
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p + N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q + N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1

        if no_impr > 6000:
            if len(ones) > 8 and len(zeros) > 8:
                nr2 = min(4, len(ones), len(zeros))
                rp = np.random.choice(ones, nr2, replace=False)
                rq = np.random.choice(zeros, nr2, replace=False)
                h[rp] = 0.0
                h[rq] = 1.0
                corr = fft_corr(h)
                s, deltas = build_deltas(h)
            T = max(T * 1.8, 0.015)
            no_impr = 0

    return best_h, best_c

def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=35):
    h = h_in.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta_m = 0.84
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta_m * (h_new - h)
        h = h_new
        it += 1
        if it % 350 == 0:
            lr *= 0.992
            beta_m = min(0.94, 0.80 + it * 0.00015)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Soft-max SA (explore broad landscape)
    d1 = min(deadline - 450, t0 + 280)
    h1, c1 = sa_hybrid(h_bin, d1, mode="soft", n_peaks=7, n_per_peak=22,
                       cool_rate=0.999, t_start=0.05, max_iters=200000, beta=40.0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Nesterov continuous polish
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 110)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, c2 = nesterov_polish(hc, N0, d2, lr=0.009, k_top=45)
        if c2 < best_c5:
            best_c5, best_h = c2, hn.copy()

    # Phase 3: Hard-max SA (exploit)
    if time.time() < deadline - 200:
        h3 = make_binary(best_h, N0)
        d3 = min(deadline - 150, time.time() + 220)
        h3, c3 = sa_hybrid(h3, d3, mode="hard", n_peaks=8, n_per_peak=28,
                           cool_rate=0.9994, t_start=0.025, max_iters=180000)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Multi-resolution exploration
    for N_try in [390, 400, 420, 460, 500]:
        if time.time() > deadline - 100:
            break
        if N_try % 2 != 0:
            N_try += 1
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, N_try)
        sub_dl = min(deadline - 50, time.time() + 45)
        hr, cr = sa_hybrid(hb, sub_dl, mode="soft", n_peaks=5, n_per_peak=15,
                           cool_rate=0.999, t_start=0.035, max_iters=60000, beta=60.0)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 5: Final targeted SA at best resolution
    if time.time() < deadline - 5:
        Ny = len(best_h)
        h5 = make_binary(best_h, Ny)
        h5, c5f = sa_hybrid(h5, deadline - 2, mode="hard", n_peaks=10, n_per_peak=32,
                            cool_rate=0.9997, t_start=0.012, max_iters=120000)
        if c5f < best_c5:
            best_c5, best_h = c5f, h5

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
