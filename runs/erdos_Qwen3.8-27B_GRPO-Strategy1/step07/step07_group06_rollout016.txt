    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_fg(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_chain(h_init, N, deadline, T_list, max_iter_each=120, lam=50.0):
    h = h_init.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    for T in T_list:
        if time.time() > deadline:
            break
        res = sp_min(smooth_fg, h, args=(T, target, lam), jac=True,
                     method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': max_iter_each, 'maxfun': 300, 'ftol': 1e-15})
        h = proj01(res.x, target)
        c5 = compute_c5(h, N)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def bin_search(h, N, deadline, max_iters=3000, ntri=25):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline or stall > 400:
            if stall > 400 and time.time() < deadline:
                for _ in range(min(5, nh // 20)):
                    oi = int(np.random.choice(np.where(h == 1)[0]))
                    zi = int(np.random.choice(np.where(h == 0)[0]))
                    h[oi], h[zi] = 0.0, 1.0
                stall = 0
                continue
            break
        corr = fft_ctr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c, best_h = P, h.copy()
            stall = 0
            continue
        ctr = N - 1
        peaks = np.where(corr >= P - 1.0)[0]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        for _ in range(ntri):
            if len(peaks) > 0 and np.random.rand() < 0.55:
                pk = int(np.random.choice(peaks))
                L = pk - ctr
                if L >= 0:
                    cands = np.where((h[:N-L] == 1) & (h[L:] == 0))[0]
                else:
                    cands = np.where((h[-L:] == 1) & (h[:N+L] == 0))[0]
                oi = int(np.random.choice(cands)) if len(cands) else int(np.random.choice(ones))
            else:
                oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd:
                bd, bi, bj = d, oi, zi
            h[oi], h[zi] = 1.0, 0.0
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
        else:
            stall += 1
    return best_h, best_c

def fft_ctr(h):
    return fft_corr(h)

def try_resolutions(best_h, best_c5, deadline, N_list):
    x_b = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in N_list:
        if time.time() > deadline:
            break
        if Nt % 2:
            Nt += 1
        if Nt == len(best_h):
            continue
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_b, best_h)
        ht = (hi > 0.5).astype(float)
        nh = Nt // 2
        n1 = int(ht.sum())
        if n1 > nh:
            ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0.0
        elif n1 < nh:
            ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1.0
        sd = min(deadline, time.time() + 30)
        hr, _ = bin_search(ht, Nt, sd, max_iters=150, ntri=18)
        c5r = compute_c5(hr, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 6.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Quick binary search
    d1 = min(dl - 500, t0 + 40)
    hb, _ = bin_search(best_h, N0, d1, max_iters=200, ntri=20)
    c = compute_c5(hb, N0)
    if c < best_c5:
        best_c5, best_h = c, hb.copy()

    # Phase 2: L-BFGS chain (T=15→40→100) from initial
    if time.time() < dl - 400:
        d2 = min(dl - 350, time.time() + 180)
        hl, cl = lbfgs_chain(best_h, N0, d2, T_list=[15, 40, 100], max_iter_each=150, lam=30.0)
        if cl < best_c5:
            best_c5, best_h = cl, hl.copy()

    # Phase 3: L-BFGS with random restarts
    if time.time() < dl - 300:
        d3 = min(dl - 250, time.time() + 200)
        for r in range(5):
            if time.time() > d3:
                break
            noise = 0.12 - 0.015 * r
            hr = best_h + np.random.randn(N0) * noise
            hr = proj01(hr, N0 / 2.0)
            T_chain = [20, 60, 150] if r < 3 else [40, 100]
            hl3, cl3 = lbfgs_chain(hr, N0, min(d3, time.time() + 35), T_list=T_chain, max_iter_each=100, lam=25.0)
            if cl3 < best_c5:
                best_c5, best_h = cl3, hl3.copy()

    # Phase 4: Guided binary search (long)
    if time.time() < dl - 200:
        d4 = min(dl - 150, time.time() + 280)
        hb4, c4r = bin_search(best_h, N0, d4, max_iters=5000, ntri=28)
        c4 = compute_c5(hb4, N0)
        if c4 < best_c5:
            best_c5, best_h = c4, hb4.copy()

    # Phase 5: Multi-resolution
    if time.time() < dl - 120:
        N_list = [260, 300, 340, 420, 460, 520, 600]
        best_h, best_c5 = try_resolutions(best_h, best_c5, min(dl - 80, time.time() + 150), N_list)

    # Phase 6: Final L-BFGS polish on best
    Nc = len(best_h)
    if time.time() < dl - 40:
        d6 = min(dl - 20, time.time() + 70)
        h6 = best_h + np.random.randn(Nc) * 0.04
        h6 = proj01(h6, Nc / 2.0)
        hl6, cl6 = lbfgs_chain(h6, Nc, d6, T_list=[50, 120, 250], max_iter_each=80, lam=20.0)
        if cl6 < best_c5:
            best_c5, best_h = cl6, hl6.copy()

    # Phase 7: Final binary polish
    if time.time() < dl - 10:
        hf = (best_h > 0.5).astype(float)
        hf, _ = bin_search(hf, len(best_h), dl - 2.0, max_iters=500, ntri=24)
        cf = compute_c5(hf, len(best_h))
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Compare continuous vs binary
    Nf = len(best_h)
    fc5 = compute_c5(best_h, Nf)
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, Nf)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, Nf
