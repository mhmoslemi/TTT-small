np.correlate index 0 → cross-corr index (0-2) mod 5 = 3: cross-corr[3]=0 ✓
np.correlate index 1 → cross-corr index (1-2) mod 5 = 4: cross-corr[4]=1 ✓
np.correlate index 2 → cross-corr index (2-2) mod 5 = 0: cross-corr[0]=0 ✓
np.correlate index 3 → cross-corr index (3-2) mod 5 = 1: cross-corr[1]=1 ✓
np.correlate index 4 → cross-corr index (4-2) mod 5 = 2: cross-corr[2]=0 ✓

So np.correlate(h, 1-h, 'full') = ifft(conj(FFT(h)) * FFT(1-h)) with a specific indexing (fftshift by N-1).

```python
def corr_full(h):

    c = np.fft.irfft(np.conj(H) * GH, L)
    # c[k] = cross-corr(h, 1-h)[k] for k=0..L-1
    # np.correlate convention: index n corresponds to k = (n-(N-1)) mod L
    # So to match np.correlate: result[n] = c[(n-(N-1)) % L]
    # Or equivalently: result = np.roll(c, N-1)
    return np.roll(c, N-1)
```

```python
def grad_step(h, N, lr, top_k=10):
    corr = corr_full(h)
    peak = np.max(corr)
    
    # Get top K lags
    top_idx = np.argsort(-corr)[:top_k]
    
    # Weighted gradient (weighted by how close to max each lag is)
    weights = np.exp(-(corr[peak] - corr[top_idx]) / 0.1)  # soft weighting
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for wi, ki in zip(weights, top_idx):
        k = ki - (N-1)  # actual lag
        if k >= 0:
            # corr[k] = sum_{i=0}^{N-1-k} h[i]*(1-h[i+k])
            # d/dh[j] = (1-h[j+k]) for j=0..N-1-k, minus h[j-k] for j=k..N-1
            grad[:N-k] += wi * (1 - h[k:N])
            grad[k:] -= wi * h[:N-k]
        else:
            m = -k
            # corr[-m] = sum_{i=0}^{N-1-m} h[i+m]*(1-h[i])
            # d/dh[j]: for j=m..N-1: (1-h[j-m]); for j=0..N-1-m: -h[j+m]
            grad[m:N] += wi * (1 - h[:N-m])
            grad[:N-m] -= wi * h[m:N]
    
    # Step
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    
    # Fix sum constraint
    target = N / 2.0
    diff = target - h_new.sum()
    if diff > 0:
        marg = 1 - h_new
        if marg.sum() > 0:
            h_new += diff / marg.sum() * marg
    elif diff < 0:
        marg = h_new
        if marg.sum() > 0:
            h_new += diff / marg.sum() * marg
    h_new = np.clip(h_new, 0, 1)
    
    return h_new
```

```python
import numpy as np
import time

def corr_full(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    GH = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(np.conj(H) * GH, L)
    return np.roll(c, N - 1)

def compute_c5(h):
    N = len(h)
    return float(np.max(corr_full(h))) * 2.0 / N

def fix_sum(h, target):
    h = h.copy()
    diff = target - h.sum()
    if diff > 0:
        marg = 1.0 - h
        if marg.sum() > 1e-12:
            h += diff / marg.sum() * marg
    elif diff < 0:
        marg = h
        if marg.sum() > 1e-12:
            h += diff / marg.sum() * marg
    return np.clip(h, 0.0, 1.0)

def gradient_optimize(h, N, deadline, lr0=0.005, top_k=15, cooling=0.9995):
    h = fix_sum(h.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    corr = corr_full(h)
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        peak = np.max(corr)
        top_idx = np.argsort(-corr)[:top_k]
        weights = np.exp(-(peak - corr[top_idx]) / 0.08)
        weights /= weights.sum()
        grad = np.zeros(N)
        for wi, ki in zip(weights, top_idx):
            k = int(ki) - (N - 1)
            if k >= 0:
                grad[:N - k] += wi * (1.0 - h[k:N])
                grad[k:] -= wi * h[:N - k]
            else:
                m = -k
                grad[m:N] += wi * (1.0 - h[:N - m])
                grad[:N - m] -= wi * h[m:N]
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = fix_sum(h_new, N / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        corr = corr_full(h)
        if it % 200 == 0:
            lr *= cooling
        if stall > 2000:
            h += np.random.randn(N) * 0.01
            h = fix_sum(np.clip(h, 0, 1), N / 2.0)
            corr = corr_full(h)
            lr = max(lr, 0.003)
            stall = 0
    return best_h, best_c5

def binary_steepest(h_bin, N, deadline, max_iter=20000):
    h = h_bin.astype(float).copy()
    corr = corr_full(h)
    cur_peak = np.max(corr)
    best_peak = cur_peak
    best_h = h.copy()
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        n_cand = min(300, len(ones) * len(zeros))
        best_delta = 0
        best_i, best_j = -1, -1
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        tried = set()
        for _ in range(n_cand):
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
            if i == j or (i, j) in tried:
                continue
            tried.add((i, j))
            h[i] = 0
            h[j] = 1
            new_peak = np.max(corr_full(h))
            delta = cur_peak - new_peak
            if delta > best_delta:
                best_delta = delta
                best_i, best_j = i, j
            h[i] = 1
            h[j] = 0
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = corr_full(h)
            cur_peak = np.max(corr)
            if cur_peak < best_peak - 1e-12:
                best_peak = cur_peak
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            for _ in range(3):
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                if oi != zi:
                    h[oi], h[zi] = 1.0, 0.0
                    h[zi], h[oi] = 0.0, 1.0
            corr = corr_full(h)
            cur_peak = np.max(corr)
        if stall > 500:
            break
    return best_h, best_peak * 2.0 / N

def block_steepest(h_bin, N, deadline, max_iter=5000):
    h = h_bin.astype(float).copy()
    corr = corr_full(h)
    cur_peak = np.max(corr)
    best_peak = cur_peak
    best_h = h.copy()
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        best_delta = 0
        best_move = None
        for _ in range(100):
            L = int(np.random.choice([2, 3, 4, 5]))
            if L >= len(ones) or L >= len(zeros):
                L = min(L, len(ones) - 1, len(zeros) - 1)
            if L < 2:
                continue
            oi = int(np.random.choice(ones[:max(1, len(ones) - L)]))
            zi = int(np.random.choice(zeros[:max(1, len(zeros) - L)]))
            if abs(oi - zi) < L:
                continue
            if h[oi:oi + L].sum() != L or h[zi:zi + L].sum() != 0:
                continue
            h[oi:oi + L] = 0
            h[zi:zi + L] = 1
            new_peak = np.max(corr_full(h))
            delta = cur_peak - new_peak
            if delta > best_delta:
                best_delta = delta
                best_move = (oi, zi, L)
            h[oi:oi + L] = 1
            h[zi:zi + L] = 0
        if best_move is not None:
            oi, zi, L = best_move
            h[oi:oi + L] = 0
            h[zi:zi + L] = 1
            corr = corr_full(h)
            cur_peak = np.max(corr)
            if cur_peak < best_peak - 1e-12:
                best_peak = cur_peak
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi], h[zi] = 0.0, 1.0
            corr = corr_full(h)
            cur_peak = np.max(corr)
        if stall > 300:
            break
    return best_h, best_peak * 2.0 / N

def structured_init(N, variant, rng):
    nh = N // 2
    if variant == 0:
        return (rng.random(N) < 0.5).astype(float)
    elif variant == 1:
        x = np.arange(N)
        v = np.sin(2 * np.pi * 5 * x / N) + 0.7 * np.sin(2 * np.pi * 11 * x / N) + 0.5 * np.cos(2 * np.pi * 3 * x / N)
        h = (v > np.median(v)).astype(float)
        return fix_sum(h, nh)
    elif variant == 2:
        x = np.arange(N)
        h = np.zeros(N)
        for f in [3, 7, 13]:
            h += np.sin(2 * np.pi * f * x / N + 0.5 * f)
        h = (h > np.median(h)).astype(float)
        return fix_sum(h, nh)
    else:
        h = ((np.arange(N) * 7 + 3) % 13 < 7).astype(float)
        return fix_sum(h, nh)
```
