    delta = np.zeros(2 * N - 1)
    # First index contribution: term at i=pos
    if a == 1.0:
        # Remove: 1*(1-h[pos-k+N-1]) for valid k
        k_lo = max(0, pos - N + 1)
        k_hi = min(pos + 0, 2 * N - 2)
        idx = np.arange(k_lo, k_hi + 1)
        si = pos - idx + N - 1
        delta[idx] += -(1.0 - h[si])
    else:
        k_lo = max(0, pos - N + 1)
        k_hi = min(pos + 0, 2 * N - 2)
        idx = np.arange(k_lo, k_hi + 1)
        si = pos - idx + N - 1
        delta[idx] += (1.0 - h[si])
    # Second index contribution: term where j-k+N-1 = pos -> j = k-pos+N-1
    k_lo2 = max(0, pos)
    k_hi2 = min(pos + N - 1, 2 * N - 2)
    idx2 = np.arange(k_lo2, k_hi2 + 1)
    sj = idx2 - pos + N - 1
    mask = sj != pos
    if a == 1.0:
        delta[idx2[mask]] += h[sj[mask]]
    else:
        delta[idx2[mask]] += -h[sj[mask]]
    return delta

def greedy_search(h, N, deadline, max_iter=100000):
    C = fft_corr(h)
    best_val = float(C.max())
    best_h = h.copy()
    dx = 2.0 / N
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Compute flip effects for all ones and zeros
        # For swap (p in ones -> 0, q in zeros -> 1):
        # new_C = C - flip[p] + flip[q] (approximately; ignore O(1) interaction)
        n_o, n_z = len(ones), len(zeros)
        # Vectorized: compute delta for all swaps
        # flip_one[p_idx, k] = -flip_effect(h, N, ones[p_idx])[k]
        # flip_zero[q_idx, k] = flip_effect(h, N, zeros[q_idx])[k]
        # This is O((n_o+n_z)*N) which is ~N^2. Too slow per iteration.
        
        # Alternative: compute D_add[i] = flip_effect of setting h[i] to 1 (from current state)
        # Then for p in ones: delta_p = -flip_add_at_p (since we're going 1->0)
        # For q in zeros: delta_q = +flip_add_at_q
        # We need flip_add for all positions. O(N^2) to compute.
        
        # Faster: use the structure. The flip effect for position i has a specific form.
        # Let me compute it more efficiently.
        # flip(i) for h[i]=1 (removing): 
        #   at k in [i-N+1, i]: -(1-h[i-k+N-1])
        #   at k in [i, i+N-1], k!=i+N-1: +h[k-i+N-1]
        # flip(i) for h[i]=0 (adding):
        #   at k in [i-N+1, i]: +(1-h[i-k+N-1])
        #   at k in [i, i+N-1], k!=i+N-1: -h[k-i+N-1]
        
        # Let me compute the "add" effect for all positions at once using convolution.
        # add_effect[i, k] for k where i-k+N-1 is valid: (1-h[i-k+N-1])
        # This is a "correlation" of h with all-ones, shifted.
        
        # Actually, let me just do O(N^2) computation but with numpy vectorization.
        # For each position i, the non-zero entries of flip_add[i] are at specific k values.
        
        # Simpler approach: just evaluate a random sample of swaps
        n_sample = min(500, n_o * n_z)
        p_idx = np.random.randint(0, n_o, n_sample)
        q_idx = np.random.randint(0, n_z, n_sample)
        ps = ones[p_idx]
        qs = zeros[q_idx]
        
        best_new_max = best_val
        best_pq = None
        for ii in range(n_sample):
            p, q = ps[ii], qs[ii]
            # Compute new correlation approximately
            fp = flip_effect(h, N, p)
            fq = flip_effect(h, N, q)
            new_C = C - fp + fq
            nm = float(new_C.max())
            if nm < best_new_max:
                best_new_max = nm
                best_pq = (p, q)
        
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            C = fft_corr(h)
            if C.max() < best_val:
                best_val = float(C.max())
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best_h, best_val * dx

def anneal_search(h, N, deadline, T0=5.0, T1=0.1, alpha=0.997, max_iter=200000):
    C = fft_corr(h)
    best_val = float(C.max())
    best_h = h.copy()
    dx = 2.0 / N
    T = T0
    rng = np.random.RandomState(123)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Single swap
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        h[p], h[q] = 0.0, 1.0
        C_new = fft_corr(h)
        new_max = float(C_new.max())
        delta = new_max - float(C.max())
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            C = C_new
            if new_max < best_val:
                best_val = new_max
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
        T *= alpha
        if T < T1:
            T = T1
    return best_h, best_val * dx

def multi_swap_search(h, N, deadline, max_iter=50000):
    C = fft_corr(h)
    best_val = float(C.max())
    best_h = h.copy()
    dx = 2.0 / N
    rng = np.random.RandomState(77)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # 2-swap: move two 1s to two 0 positions
        p1, p2 = rng.choice(ones, 2, replace=False)
        q1, q2 = rng.choice(zeros, 2, replace=False)
        h[p1] = h[p2] = 0.0
        h[q1] = h[q2] = 1.0
        C_new = fft_corr(h)
        new_max = float(C_new.max())
        if new_max < float(C.max()):
            C = C_new
            if new_max < best_val:
                best_val = new_max
                best_h = h.copy()
            stall = 0
        else:
            h[p1] = h[p2] = 1.0
            h[q1] = h[q2] = 0.0
            stall += 1
            if stall > 200:
                break
    return best_h, best_val * dx

def perturbation_restart(h, N, n_perturb, seed=0):
    h2 = h.copy()
    rng = np.random.RandomState(seed)
    ones = np.where(h2 == 1)[0]
    zeros = np.where(h2 == 0)[0]
    n = min(n_perturb, len(ones) // 2, len(zeros) // 2)
    if n < 1:
        return h2
    ps = rng.choice(ones, n, replace=False)
    qs = rng.choice(zeros, n, replace=False)
    h2[ps] = 0.0
    h2[qs] = 1.0
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    h = ensure_binary(h0, N, seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Phase 1: Intensive greedy search
    d1 = min(deadline - 600, time.time() + 180)
    h1, c5_1 = greedy_search(h, N, d1, max_iter=50000)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    
    # Phase 2: Annealing from best
    if time.time() < deadline - 400:
        d2 = min(deadline - 380, time.time() + 200)
        h2, c5_2 = anneal_search(best_h, N, d2, T0=8.0, T1=0.05, alpha=0.9985)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
    
    # Phase 3: More greedy
    if time.time() < deadline - 300:
        d3 = min(deadline - 280, time.time() + 150)
        h3, c5_3 = greedy_search(best_h, N, d3, max_iter=40000)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
    
    # Phase 4: Multi-swap search
    if time.time() < deadline - 250:
        d4 = min(deadline - 230, time.time() + 120)
        h4, c5_4 = multi_swap_search(best_h, N, d4, max_iter=30000)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
    
    # Phase 5: Perturbation restarts
    for restart in range(5):
        if time.time() > deadline - 120:
            break
        h_pert = perturbation_restart(best_h, N, 4 + restart * 2, seed=100 + restart * 37)
        d5 = min(deadline - 100, time.time() + 50)
        h5, c5_5 = greedy_search(h_pert, N, d5, max_iter=15000)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
    
    # Phase 6: Final anneal + greedy
    if time.time() < deadline - 80:
        d6 = min(deadline - 60, time.time() + 60)
        h6, c5_6 = anneal_search(best_h, N, d6, T0=3.0, T1=0.02, alpha=0.999)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()
    
    if time.time() < deadline - 20:
        d7 = min(deadline - 10, time.time() + 30)
        h7, c5_7 = greedy_search(best_h, N, d7, max_iter=20000)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()
    
    # Phase 7: Final micro greedy
    if time.time() < deadline - 3:
        d8 = deadline - 2
        h8, c5_8 = greedy_search(best_h, N, d8, max_iter=10000)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N
