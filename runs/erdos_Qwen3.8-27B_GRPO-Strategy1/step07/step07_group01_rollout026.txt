import numpy as np
import time

def compute_corr(h, N):
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def binarize(h, N):
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    target = N // 2
    if n1 == target:
        return h_b
    rng = np.random.RandomState(123)
    if n1 > target:
        ones = np.where(h_b == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_b[idx] = 0
    else:
        zeros = np.where(h_b == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_b[idx] = 1
    return h_b
