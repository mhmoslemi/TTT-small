        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        P_curr = float(np.max(np.correlate(h_bin, 1 - h_bin, 'full')))
        best_val = P_curr
        best_move = None
        n_try = min(150, max(60, len(ones) * 2))
        for _ in range(n_try):
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h_bin[p] = 0.0
            h_bin[q] = 1.0
            nv = float(np.max(np.correlate(h_bin, 1 - h_bin, 'full')))
            if nv < best_val:
                best_val = nv
                best_move = ('1', p, q)
            h_bin[p] = 1.0
            h_bin[q] = 0.0
        if rng.random() < 0.35 and len(ones) >= 4 and len(zeros) >= 4:
            for _ in range(40):
                i1, i2 = int(rng.randint(len(ones))), int(rng.randint(len(ones)))
                j1, j2 = int(rng.randint(len(zeros))), int(rng.randint(len(zeros)))
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                h_bin[p1] = 0.0; h_bin[p2] = 0.0
                h_bin[q1] = 1.0; h_bin[q2] = 1.0
                nv = float(np.max(np.correlate(h_bin, 1 - h_bin, 'full')))
                if nv < best_val:
                    best_val = nv
                    best_move = ('2', p1, p2, q1, q2)
                h_bin[p1] = 1.0; h_bin[p2] = 1.0
                h_bin[q1] = 0.0; h_bin[q2] = 0.0
        if best_move is not None and best_val < P_curr - 1e-12:
            if best_move[0] == '1':
                h_bin[best_move[1]] = 0.0
                h_bin[best_move[2]] = 1.0
            else:
                h_bin[best_move[1]] = 0.0; h_bin[best_move[2]] = 0.0
                h_bin[best_move[3]] = 1.0; h_bin[best_move[4]] = 1.0
            if best_val < best_c - 1e-13:
                best_c = best_val
                best_h = h_bin.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                for _ in range(2 + rng.randint(3)):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h_bin[p], h_bin[q] = 0.0, 1.0
                stall = 0
            elif stall > 400:
                break
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Nesterov continuous optimization (40% time)
    d1 = t0 + budget_s * 0.38
    rng = np.random.RandomState(seed)
    hc = best_h + rng.randn(N0) * 0.02
    hc = project_sum(hc, N0 / 2.0)
    h1, c5_1 = nesterov_opt(hc, N0, min(d1, deadline - 300), k=130, lr0=0.007, beta0=0.84)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Binary polish (25% time)
    d2 = t0 + budget_s * 0.62
    h2, c5_2 = binary_polish(best_h, N0, min(d2, deadline - 200), seed=seed)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Second Nesterov from binary + noise (20% time)
    if time.time() < deadline - 120:
        hc2 = best_h + rng.randn(N0) * 0.03
        hc2 = project_sum(hc2, N0 / 2.0)
        d3 = min(t0 + budget_s * 0.82, deadline - 100)
        h3, c5_3 = nesterov_opt(hc2, N0, d3, k=100, lr0=0.005, beta0=0.86)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upscale to 2N, Nesterov, downsample (10% time)
    if time.time() < deadline - 60:
        N1 = 2 * N0
        h_up = upsample(best_h, N0, N1)
        h_up = project_sum(h_up, N1 / 2.0)
        d4 = min(t0 + budget_s * 0.92, deadline - 40)
        h4, c5_4 = nesterov_opt(h_up, N1, d4, k=150, lr0=0.004, beta0=0.83)
        if c5_4 < best_c5:
            h_down = upsample(h4, N1, N0)
            h_down = project_sum(h_down, N0 / 2.0)
            c5_dn = compute_c5(h_down, N0)
            if c5_dn < best_c5:
                best_c5, best_h = c5_dn, h_down.copy()
            else:
                h4b, c5_4b = binary_polish(h4, N1, d4 + (deadline - d4) * 0.5, seed=seed + 1)
                if c5_4b < best_c5:
                    best_c5, best_h = c5_4b, h4b.copy()

    # Phase 5: Final binary polish
    N_fin = len(best_h)
    d5 = deadline - 1
    h5, c5_5 = binary_polish(best_h, N_fin, d5, seed=seed + 7)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
