Let me now think about the Nesterov part. The gradient computation for the continuous case:

For the correlation at shift L (in the range -(N-1) to N-1):
If L >= 0: c_L = sum_{i=0}^{N-1-L} h[i] * (1-h[i+L])
  d/dh[j] for j < N-L: (1-h[j+L])
  d/dh[j] for j >= L: -h[j-L]
  
If L < 0 (let L' = -L > 0): c_L = sum_{i=L'}^{N-1} h[i] * (1-h[i-L'])
  = sum_{i=0}^{N-1-L'} h[i+L'] * (1-h[i])
  d/dh[j] for j >= L': (1-h[j-L'])  [as the h[i+L'] term where i+L'=j]
  d/dh[j] for j < N-L': -h[j+L']  [as the (1-h[i]) term where i=j]

Wait, for L < 0, c_L = sum_{i=-L}^{N-1} h[i]*(1-h[i+L]) where L is negative.
Let L' = -L > 0. Then c_{-L'} = sum_{i=L'}^{N-1} h[i]*(1-h[i-L'])
For h[j]: appears as h[i] when i=j (requires L' <= j <= N-1): contribution factor is (1-h[j-L'])
           appears as (1-h[i]) where i-L'=j, i=j+L' (requires j+L' <= N-1): contribution factor is -h[j+L']

So:
- For L > 0:
  grad[j] += (1-h[j+L]) for j = 0..N-1-L
  grad[j] -= h[j-L] for j = L..N-1

- For L < 0 (L' = -L):
  grad[j] += (1-h[j-L']) for j = L'..N-1
  grad[j] -= h[j+L'] for j = 0..N-1-L'

    h = h.astype(float).copy()
    fix_binary(h, N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max()) * (2.0 / N)
    best_h = h.copy()
    for _ in range(rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        Co = c_matrix_positions(h, N, ones)
        Cz = c_matrix_positions(h, N, zeros)
        crit = np.argwhere(corr > float(corr.max()) * 0.98 + 1e-6)
        cr = crit[0] if len(crit) else (corr > float(corr.max()) * 0.95 + 1e-6)
        splash = np.abs(corr - corr.max()) < 0.002
        if notch.empty():
            splash = np.ones(2 * N - 1, dtype=bool)
        Co_s = Co * splash[None, :]
        Cz_s = Cz * splash[None, :]
        new = corr[None, None, :] + Co_s[None, :, :] + Cz_s[:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        P = float(corr.max())
        if mv < P - 1e-14:
            h[ones[mi[1]]] = 0
            h[zeros[mi[0]]] = 1
            corr = fft_corr(h)
            c5 = float(corr.max()) * (2.0 / N)
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
            break
        else:
            break
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=80, lr0=0.01, beta0=0.85):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N - L] += (1.0 - y[L:])
                grad[L:] -= y[:N - L]
            else:
                M = -L
                grad[M:] += (1.0 - y[:N - M])
                grad[:N - M] -= y[M:]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.985
            beta = min(0.97, 0.85 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 > 0.5).astype(float)
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: aggressive local search with blocks
    d1 = min(deadline - 380, t0 + 280)
    hs, cs = local_search(best_h, N0, d1, sample_size=130, block_prob=0.2, kick_thresh=60, kick_size=8)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()
    # Phase 2: Nesterov continuous
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.08
        hc = project_01(hc, N0 / 2.0)
        hn, cn = nesterov_opt(hc, N0, d2, k=70, lr0=0.008, beta0=0.82)
        if cn < best_c5:
            best_c5, best_h = cn, hn.copy()
    # Phase 3: greedy sweep
    if time.time() < deadline - 180:
        hg, cg = greedy_sweep(best_h, N0, time.time() + 40, rounds=10)
        if cg < best_c5:
            best_c5, best_h = cg, hg.copy()
    # Phase 4: local search with annealing
    if time.time() < deadline - 150:
        d4 = min(deadline - 130, time.time() + 180)
        h4 = (best_h > 0.5).astype(float)
        hs4, cs4 = local_search(h4, N0, d4, sample_size=140, block_prob=0.18, kick_thresh=50, kick_size=10)
        if cs4 < best_c5:
            best_c5, best_h = cs4, hs4.copy()
    # Phase 5: multi-resolution
    for Nt in [360, 400, 420, 440]:
        if time.time() > deadline - 80:
            break
        if Nt == N0:
            continue
        x_b = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = (h_int > 0.5).astype(float)
        fix_binary(h_bt, Nt // 2)
        sub_dl = min(deadline - 30, time.time() + 50)
        h_rt, c_rt = local_search(h_bt, Nt, sub_dl, sample_size=Nt // 2, block_prob=0.15, kick_thresh=40, kick_size=6)
        if c_rt < best_c5:
            best_c5, best_h = c_rt, h_rt.copy()
    # Phase 6: final Nesterov polish
    if time.time() < deadline - 25:
        hF = best_h + np.random.randn(len(best_h)) * 0.03
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 10, k=90, lr0=0.005, beta0=0.80)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    # Phase 7: final greedy + binary cleanup
    if time.time() < deadline - 5:
        best_h_g, best_cg = greedy_sweep(best_h, len(best_h), deadline - 1, rounds=8)
        if best_cg < best_c5:
            best_c5, best_h = best_cg, best_h_g.copy()
    hb = (best_h > 0.5).astype(float)
    cb = compute_c5(hb, len(hb))
    if cb < best_c5:
        best_h, best_c5 = hb, cb
    return best_h, best_c5, len(best_h)
