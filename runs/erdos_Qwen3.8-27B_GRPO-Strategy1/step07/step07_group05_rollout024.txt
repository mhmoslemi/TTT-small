def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)

    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 0.5:
        return h
    lo, hi = -2.0, 2.0

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def soft_max_grad(h, corr, temp):
    N = len(h)
    peak = np.max(corr)
    w = np.exp((corr - peak) / temp)
    ws = w.sum()
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    sig = np.where(w > np.max(w) * 1e-4)[0]
    for ki in sig:
        wt = w[ki]
        r = ki - center
        if r == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif r > 0:
            n_valid = N - r
            grad[:n_valid] += wt * (1.0 - h[r:r+n_valid])
            grad[r:] -= wt * h[:n_valid]
        else:
            M = -r
            n_valid = N - M
            grad[M:] += wt * (1.0 - h[:n_valid])
            grad[:n_valid] -= wt * h[M:]
    return grad

Now let me also think about whether the continuous optimization will actually help. The key question: is the optimal solution binary or non-binary?

At the optimal, we have K+ = {k: C[k] = max}. For a small perturbation δh with sum(δh) = 0:
d(max C[k])/dt = max_{k in K+} dC[k]/dt

For the maximum to not increase, we need dC[k]/dt <= 0 for all k in K+. This is a system of linear inequalities in δh. If the only feasible δh = 0, then the current point is a local optimum.

For binary h, δh is constrained to swap 1s and 0s (for maintaining sum). For continuous h, δh can be any vector with sum 0 and |h[i] + δh[i]| ≤ 1.

The continuous relaxation has more degrees of freedom, so it's easier to satisfy the KKT conditions. This means the continuous optimum should be at least as good as the binary optimum.

However, in practice, the continuous landscape might have more local optima, and the gradient descent might get stuck. That's why I'm interleaving binary and continuous phases.

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 0.01:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def soft_max_grad(h, corr, temp):
    N = len(h)
    peak = np.max(corr)
    w = np.exp((corr - peak) / temp)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N)
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = np.max(w) * 1e-4
    sig = np.where(w > thresh)[0]
    for ki in sig:
        wt = w[ki]
        r = ki - center
        if r == 0:

        elif r > 0:

            grad[:nv] += wt * (1.0 - h[r:r + nv])

        else:
            M = -r
            nv = N - M
            grad[M:] += wt * (1.0 - h[:nv])
            grad[:nv] -= wt * h[M:]
    return grad
