import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(999 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def block_swap(h, p, q, block_len, N):
    """Swap block starting at p (ones) with block starting at q (zeros), length block_len."""
    # Check that the block at p is all 1s and block at q is all 0s
    p_end = min(p + block_len, N)
    q_end = min(q + block_len, N)
    if p_end > N or q_end > N:
        return False
    if not np.all(h[p:p_end] == 1) or not np.all(h[q:q_end] == 0):
        return False
    h[p:p_end] = 0
    h[q:q_end] = 1
    return True

def greedy_binary(h, N, deadline, max_sweeps=3000, sf=1.0):
    """Greedy single-swap binary optimization."""
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Compute D matrix
        shifts = np.arange(2 * N - 1) - (N - 1)
        pos = np.arange(N)[:, None]
        sv = shifts[None, :]
        pm = pos - sv
        pp = pos + sv
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm + hp * vp
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def nesterov_lse(h, N, deadline, beta=50.0, lr0=0.03, k_top=80):
    """Nesterov with log-sum-exp smoothing."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta_mom = 0.9
    it = 0
    center = N - 1
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        # Log-sum-exp gradient
        corr_shifted = corr - corr.max()  # for numerical stability
        weights = np.exp(beta * corr_shifted)
        weights /= weights.sum()
        
        # Gradient of LSE w.r.t. y
        # corr[k] = sum_i y[i] * (1-y[i-s]) for appropriate s
        # d(corr[k])/d(y[i]) depends on i and k
        # Use the same approach: for each k, the gradient contribution is
        # But with LSE weights, gradient = sum_k w_k * d(corr[k])/d(y)
        
        # Efficient computation: top-k approximation
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for ai in top_idx:
            s = int(ai - center)
            w_k = weights[ai] / beta  # approx weight
            if s == 0:
                grad += (1.0 - 2.0 * y)
            elif s > 0:
                grad[s:] += (1.0 - y[:N - s])
                grad[:N - s] -= y[s:]
            else:
                M = -s
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        
        # Actually the proper LSE gradient:
        # f = (1/beta) log(sum exp(beta * corr[k]))
        # df/dy[i] = sum_k p_k * d(corr[k])/dy[i] where p_k = exp(beta*corr[k])/sum
        # So gradient = sum_k p_k/beta * ... 
        # But for Nesterov, we just need the direction. Let's use the weighted sum.
        
        h_new = project_sum(y - lr * grad, target)
        m = beta_mom * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
            beta_mom = min(0.97, 0.9 + it * 0.00005)
            beta = beta * 1.02  # slowly increase smoothing parameter
    
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(777 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.permutation(ones)[:n1 - nh]
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.permutation(zeros)[:nh - n1]
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def nesterov_anneal(h, N, deadline, beta_start=20.0, beta_end=200.0, lr0=0.03, k_top=100):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta_mom = 0.9
    center = N - 1
    it_total = 0
    
    while time.time() < deadline:
        # Anneal beta
        frac = min(1.0, it_total / 2000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        # Top-k gradient
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for ai in top_idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * y)
            elif s > 0:
                grad[s:] += (1.0 - y[:N - s])
                grad[:N - s] -= y[s:]
            else:
                M = -s
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]

        h = h_new
