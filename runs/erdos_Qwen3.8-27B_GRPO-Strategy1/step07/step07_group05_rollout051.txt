        # Bisection on shift
        lo_b, hi_b = -10.0, 10.0
        for _ in range(50):
            mid = (lo_b + hi_b) / 2
            s_val = np.clip(h_new - mid, 0, 1).sum()
            if s_val > target:
                lo_b = mid
            else:
                hi_b = mid
        h_new = np.clip(h_new - (lo_b + hi_b) / 2, 0, 1)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.9995
        if int(time.time() * 10) % 100 < 10:
            eps *= 0.98
    
    return best_h, best_c

def structured_init(N, variant, seed_val=12345):
    np.random.seed(seed_val + variant * 7919)
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[np.random.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        x = np.arange(N)
        h = ((x * 7 + 3) % 13 < 7).astype(float)
        h = fix_sum(h, nh)
    elif variant == 2:
        x = np.arange(N, dtype=float)
        v = np.sin(2*np.pi*3*x/N) + 0.8*np.sin(2*np.pi*7*x/N + 0.5) + 0.6*np.cos(2*np.pi*11*x/N + 1.2)
        h = (v > np.median(v)).astype(float)
        h = fix_sum(h, nh)
    elif variant == 3:
        # Thue-Morse based
        x = np.arange(N)
        tm = np.zeros(N)
        for i in range(N):
            tm[i] = bin(i).count('1') % 2
        h = (tm == 0).astype(float)
        h = fix_sum(h, nh)
    else:
        x = np.arange(N, dtype=float)
        v = np.sin(2*np.pi*5*x/N + 0.3) + np.sin(2*np.pi*13*x/N + 1.1) + 0.5*np.cos(2*np.pi*19*x/N)
        h = (v > np.median(v)).astype(float)
        h = fix_sum(h, nh)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SA on binary N=380 (60% of time)
    h_b = (h0 > 0.5).astype(float)
    d1 = t_start + (budget_s - 10) * 0.60
    h1, c1 = sa_binary(h_b, N0, min(deadline, d1), T0=0.15)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()
    
    # Phase 2: Continuous refinement (15%)
    if time.time() < deadline:
        d2 = t_start + (budget_s - 10) * 0.75
        hc = best_h + np.random.randn(N0) * 0.01
        hc = np.clip(hc, 0, 1)
        # Fix sum
        lo_b, hi_b = -5.0, 5.0
        for _ in range(50):
            mid = (lo_b + hi_b) / 2
            if np.clip(hc - mid, 0, 1).sum() > N0 / 2.0:
                lo_b = mid
            else:
                hi_b = mid
        hc = np.clip(hc - (lo_b + hi_b) / 2, 0, 1)
        hc, c2 = cont_optimize(hc, N0, min(deadline, d2))
        if c2 < best_c5:
            best_c5, best_h = c2, hc.copy()
    
    # Phase 3: Multi-start at other N values (20%)
    if time.time() < deadline:
        for N_try in [760, 400, 500, 600]:
            if time.time() > deadline - 30:
                break
            if N_try == N0:
                continue
            if N_try % 2 != 0:
                N_try += 1
            best_loc_c = np.inf
            best_loc_h = None
            n_starts = 3 if N_try > 600 else 5
            for vi in range(n_starts):
                if time.time() > deadline - 15:
                    break
                hv = structured_init(N_try, vi, seed_val=seed + 500)
