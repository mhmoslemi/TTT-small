    h = h.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    alpha = alpha0
    best_c = float(np.max(fft_corr(h)))

    while time.time() < deadline:
        t += 1
        # Compute gradient
        grad = lse_gradient(h, alpha)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)

            best_h = h_new.copy()
        
        h = h_new
        
        # Anneal alpha

            lr *= 0.995

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s-target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-0.5*(lo+hi), 0, 1)

def lse_grad(h, alpha):
    N = len(h)
    corr = fft_corr(h)
    cs = alpha*(corr - corr.max())
    p = np.exp(cs)
    p /= p.sum()
    w = p
    L = 3*N
    Wf = np.fft.rfft(w, L)
    conv = np.fft.irfft(Wf*np.fft.rfft(1.0-h, L), L)
    first = conv[N-1:2*N-1]
    cross = np.fft.irfft(np.conj(Wf)*np.fft.rfft(h, L), L)
    second = cross[N-1::-1]
    return first - second

    idx = np.argpartition(corr, -k)[-k:]

            g2[:N+t] = h[-t:]  # h[-t:] = h[|t|:]
