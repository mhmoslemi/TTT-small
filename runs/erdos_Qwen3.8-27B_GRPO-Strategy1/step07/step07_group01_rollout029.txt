import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > max_stall:

        if stall > max_stall * 2:
            T = max(T, 0.1)
    
    return best_h, best_c * dx

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=50, T=80.0, lr0=0.018):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.995
            beta = min(0.98, 0.82 + it * 0.0002)
        if it % 200 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(20.0, min(400.0, T * (0.85 if gap < 0.003 else 1.1)))
    return best_h, best_pk * dx

def ensure_binary(h, N, rng):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_sa(h_init, N, deadline, rng, n_sample=300):
    h = ensure_binary(h_init, N, rng).copy()
    dx = 2.0 / N
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    T = 2.5
    T_min = 0.06
    cool = 0.9965
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        shifts = np.arange(2 * N - 1) - (N - 1)
        p_all = np.arange(N)[:, None]
        s_all = shifts[None, :]
        pm = p_all - s_all
        pp = p_all + s_all
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
        Dz = D[zeros, :]
        if len(ones) > n_sample:
            p_list = rng.choice(ones, n_sample, replace=False)
        else:
            p_list = ones
        best_delta = np.inf
        best_pq = None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi] - P)
            if d < best_delta:
                best_delta = d
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        if best_delta < -1e-12:
            accept = True
        elif T > T_min:
            accept = rng.random() < np.exp(-best_delta / T)
        else:
            accept = False
        if accept:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_c - 1e-13:
                best_c = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T = max(T_min, T * cool)
        if stall > 120:
            n_r = min(3 + stall // 35, 12)
            ones_now = np.where(h == 1)[0]
            zeros_now = np.where(h == 0)[0]
            for _ in range(n_r):
                pi = int(ones_now[rng.randint(len(ones_now))])
                qi = int(zeros_now[rng.randint(len(zeros_now))])
                h[pi], h[qi] = 0.0, 1.0
            c = fft_corr(h)
            P = float(c.max())
            stall = 0
            T = max(T, 1.2)
        if stall > 350:
            stall = 0
            T = max(T, 2.0)
    return best_h, best_c * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)

    best_c5 = compute_c5(best_h, N0)

        hF, c5F = binary_sa(best_h, Nf, d6, rng, n_sample=sf)
        if c5F < best_c5:

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
