A faster approach: Don't enumerate all pairs. Instead:

2. For the swap to reduce R(k*), we need h[a+k*]+h[a-k*] - h[b+k*]-h[b-k*] < 0

3. So prefer a where h[a±k*] are both 0, and b where h[b±k*] are both 1
4. Among such candidates, check the full max

This narrows down the candidates significantly.

def compute_grad_hlxp(h, N, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*R(k))) w.r.t. h."""
    corr = fft_corr(h)  # length 2N-1
    R = corr / N  # normalize
    # Softmax weights
    R_shifted = R - R.max()
    w = np.exp(beta * R_shifted)
    w /= w.sum()
    
    grad = np.zeros(N)
    for k_idx in range(2*N-1):
        if w[k_idx] < 1e-10:
            continue
        k = k_idx - (N-1)  # actual lag
        wt = w[k_idx]
        if k == 0:
            grad += wt * (1.0 - 2.0*h)
        elif k > 0:

                grad[i+k] -= wt * h[i]
        else:
            kk = -k
            for i in range(kk, N):
                grad[i] += wt * (1.0 - h[i-kk])
                grad[i-kk] -= wt * h[i]

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft((1.0-h)[::-1], L), L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def grad_corr_hlxp(h, N, beta):
    corr = fft_corr(h)
    R = corr / N
    Rm = R - R.max()
    w = np.exp(beta * Rm)
    w /= w.sum()
    grad = np.zeros(N)
    c = N - 1
    for ki in range(2*N-1):
        if w[ki] < 1e-13:
            continue
        k = ki - c
        wt = w[ki]
        if k == 0:
            grad += wt*(1.0 - 2.0*h)
        elif k > 0:
            grad[:N-k] += wt*(1.0 - h[k:])
            grad[k:] -= wt*h[:N-k]
        else:
            M = -k
            grad[M:] += wt*(1.0 - h[:N-M])
            grad[:N-M] -= wt*h[M:]
    return grad

def project_box_sum(h, S):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)/2.0
        if np.clip(h-mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2.0, 0, 1)

        h = h_new
