import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_gradient(corr, beta):
    """Compute gradient of log-sum-exp of corr w.r.t. logits, approximated."""
    # Soft max weights
    max_c = corr.max()
    exp_c = np.exp(beta * (corr - max_c))
    weights = exp_c / exp_c.sum()

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_nesterov(h, N, deadline, beta_init=20.0, beta_max=80.0, lr0=0.02, 
                  mom0=0.85, eval_every=5):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    mom = mom0
    beta = beta_init
    L2 = 4 * N  # FFT length for convolution
    it = 0
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        max_c = float(corr.max())
        
        # Soft-max weights
        exp_c = np.exp(beta * (corr - max_c))
        w_sum = float(exp_c.sum())
        if w_sum < 1e-300:
            w = np.zeros(len(corr))
            w[int(np.argmax(corr))] = 1.0
        else:
            w = exp_c / w_sum
        w_rev = w[::-1]
        
        # Gradient: g[i] = 1 - (y*w)[i+N-1] - (y*w_rev)[i+N-1]
        Y = np.fft.rfft(y, L2)
        conv_w = np.real(np.fft.irfft(Y * np.fft.rfft(w, L2), L2))
        conv_wr = np.real(np.fft.irfft(Y * np.fft.rfft(w_rev, L2), L2))
        
        idx = N - 1
        grad = 1.0 - conv_w[idx:idx+N] - conv_wr[idx:idx+N]
        
        # Projected gradient step
        h_new = project_sum(y - lr * grad, target)
        m = mom * (h_new - h)
        h = h_new
        it += 1
        
        # Track best (evaluate every few steps)
        if it % eval_every == 0:
            pk = float(np.max(fft_corr(h)))
            if pk < best_pk - 1e-11:
                best_pk = pk
                best_h = h.copy()
        
        # Schedule
        if it % 500 == 0:
            lr *= 0.96
            mom = min(0.95, mom + 0.002)
            beta = min(beta_max, beta * 1.08)
    
    return best_h, best_pk * dx

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(999 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def fft_corr(h):

    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_nesterov(h, N, deadline, beta_init=15.0, beta_max=80.0, lr0=0.015, mom0=0.85, eval_every=4):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    mom = mom0
    beta = beta_init
    L2 = 4 * N
    it = 0
    stall = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        max_c = float(corr.max())
        exp_c = np.exp(beta * (corr - max_c))
        w_sum = float(exp_c.sum())
        if w_sum < 1e-300:
            w = np.zeros(len(corr)); w[int(np.argmax(corr))] = 1.0
        else:
            w = exp_c / w_sum
        w_rev = w[::-1]
        Y = np.fft.rfft(y, L2)
        conv_w = np.real(np.fft.irfft(Y * np.fft.rfft(w, L2), L2))
        conv_wr = np.real(np.fft.irfft(Y * np.fft.rfft(w_rev, L2), L2))
        idx = N - 1
        grad = 1.0 - conv_w[idx:idx+N] - conv_wr[idx:idx+N]
        h_new = project_sum(y - lr * grad, target)
        m = mom * (h_new - h)
        h = h_new
        it += 1
        if it % eval_every == 0:
            pk = float(np.max(fft_corr(h)))
            if pk < best_pk - 1e-11:
                best_pk = pk
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 400:
                h = best_h + np.random.RandomState(it * 7).randn(N) * 0.03
                h = project_sum(h, target)
                m = np.zeros(N)
                stall = 0
        if it % 400 == 0:
            lr = max(0.001, lr * 0.95)
            mom = min(0.96, mom + 0.003)
            beta = min(beta_max, beta * 1.1)
    return best_h, best_pk * dx

def ensure_sum(h, N):
    nh = N // 2
    h = np.asarray(h, dtype=float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(777 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def binary_anneal(h, N, deadline, rng, two_swap_prob=0.35, max_sweeps=1000, sample_frac=1.0):
    h = ensure_sum(h, N).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    temp = 0.008
    stall = 0
    n_corr = 2 * N - 1
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp * 0.994)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones, n_zeros = len(ones), len(zeros)
        if n_ones < 2 or n_zeros < 2:
            break
        m_idx = np.arange(n_corr)
        A = np.zeros((n_ones, n_corr))
        for i, p in enumerate(ones):
            i1 = p + N - 1 - m_idx
            v1 = (i1 >= 0) & (i1 < N)
            A[i, v1] = h[i1[v1]]
            i2 = m_idx - N + 1 + p
            v2 = (i2 >= 0) & (i2 < N)
            A[i, v2] += h[i2[v2]]
        B = np.zeros((n_zeros, n_corr))
        for i, q in enumerate(zeros):
            i1 = q + N - 1 - m_idx
            v1 = (i1 >= 0) & (i1 < N)
            B[i, v1] = h[i1[v1]]
            i2 = m_idx - N + 1 + q
            v2 = (i2 >= 0) & (i2 < N)
            B[i, v2] += h[i2[v2]]
        best_new_P = P_cur
        best_pq = None
        if sample_frac < 1.0 and n_ones > 30:
            n_s = max(30, int(n_ones * sample_frac))
            p_ids = rng.choice(n_ones, n_s, replace=False)
        else:
            p_ids = range(n_ones)
        for i in p_ids:
            base = c_cur + A[i, :]
            npk = (base[None, :] - B).max(axis=1)
            mi = int(np.argmin(npk))
            if float(npk[mi]) < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(ones[i]), int(zeros[mi]))
        best_2p = None
        if rng.random() < two_swap_prob and n_ones >= 2 and n_zeros >= 2:
            n_try = min(60, max(20, N // 5))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, n_ones, 2)
                j1, j2 = rng.randint(0, n_zeros, 2)
                if i1 == i2 or j1 == j2:
                    continue
                delta = A[i1,:] + A[i2,:] - B[j1,:] - B[j2,:]
                new_P = float((c_cur + delta).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = (int(ones[i1]), int(ones[i2]), int(zeros[j1]), int(zeros[j2]))
        if best_2p is not None:
            p1, p2, q1, q2 = best_2p
            h[p1] = h[p2] = 0.0; h[q1] = h[q2] = 1.0
        elif best_pq is not None:
            h[best_pq[0]] = 0.0; h[best_pq[1]] = 1.0
        else:
            if stall > 35:
                for _ in range(2):
                    p = int(ones[rng.randint(n_ones)]); q = int(zeros[rng.randint(n_zeros)])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 12:
                break
            else:
                stall += 1; continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-12:
            best_c = P_cur; best_h = h.copy(); stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def upsample_linear(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: LSE Nesterov continuous at N0
    d1 = min(deadline - 500, t0 + budget_s * 0.28)
    h1, c5_1 = lse_nesterov(best_h, N0, d1, beta_init=12.0, beta_max=60.0, lr0=0.014, mom0=0.84)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    # Phase 2: Binary anneal at N0
    if time.time() < deadline - 380:
        h_bin = ensure_sum(best_h, N0)
        d2 = min(deadline - 320, time.time() + budget_s * 0.22)
        h2, c5_2 = binary_anneal(h_bin, N0, d2, rng, two_swap_prob=0.4, max_sweeps=900, sample_frac=1.0)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
    # Phase 3: Upscale, LSE Nesterov at 2*N0
    N1 = 2 * N0
    if time.time() < deadline - 250:
        h_up = project_sum(upsample_linear(best_h, N0, N1), N1 / 2.0)
        d3 = min(deadline - 200, time.time() + budget_s * 0.2)
        h3, c5_3 = lse_nesterov(h_up, N1, d3, beta_init=18.0, beta_max=70.0, lr0=0.008, mom0=0.88)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
    # Phase 4: Binary anneal at N1 (sampled)
    if time.time() < deadline - 130:
        h_bin1 = ensure_sum(best_h, N1)
        d4 = min(deadline - 100, time.time() + budget_s * 0.12)
        h4, c5_4 = binary_anneal(h_bin1, N1, d4, rng, two_swap_prob=0.3, max_sweeps=350, sample_frac=0.4)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
    # Phase 5: Back to N0, LSE Nesterov with high beta
    if time.time() < deadline - 80:
        h_back = project_sum(upsample_linear(best_h, len(best_h), N0), N0 / 2.0)
        h_back += rng.randn(N0) * 0.015
        h_back = project_sum(h_back, N0 / 2.0)
        d5 = min(deadline - 50, time.time() + budget_s * 0.1)
        h5, c5_5 = lse_nesterov(h_back, N0, d5, beta_init=50.0, beta_max=120.0, lr0=0.006, mom0=0.92)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
    # Phase 6: Final binary polish at N0
    Nfin = N0
    h_pol = ensure_sum(upsample_linear(best_h, len(best_h), Nfin), Nfin)
    h_pol2, c5_pol = binary_anneal(h_pol, Nfin, deadline - 1, rng, two_swap_prob=0.3, max_sweeps=400, sample_frac=1.0)
    if c5_pol < best_c5:
        best_c5, best_h = c5_pol, h_pol2.copy()
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
