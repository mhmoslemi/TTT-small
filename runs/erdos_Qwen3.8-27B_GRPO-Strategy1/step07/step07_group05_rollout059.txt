    delta = np.zeros(2*N-1)
    one_minus_h = 1.0 - h
    
    # h[a] goes 1->0: removes contribution (1-h[a+k-(N-1)]) for valid k
    # a+k-(N-1) in [0, N-1] => k in [N-1-a, N-1-a+N-1] = [N-1-a, 2N-2-a]

    # delta at k: -(1-h[a+k-(N-1)]) = -one_minus_h[:,:, a + (k - (N-1))]

    idx = np.arange(a, a + (k_end - k_start))
    delta[k_start:k_end] -= one_minus_h[idx]
    
    # (1-h[a]) goes 0->1: adds contribution h[(N-1)-k+a] for valid k

One more consideration: after each swap, the PreA and PreB matrices change. I need to recompute them.
Computing PreA: for each position a in [0,N), set PreA[a, N-1-a : 2N-1-a] = -(1-h[0:N]) and PreA[a, a:a+N] += h[N-1::-1].
That's 2N-1 assignments per position, N positions total: O(N*(2N)) = O(N²) ≈ 300K ops. Fast.

Similarly for PreB.

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    idx = np.argsort(-h)
    out = np.zeros(len(h))
    out[idx[:target_ones]] = 1.0
    return out

def compute_precomputed(h, N):
    """Compute PreA and PreB matrices."""
    L = 2 * N - 1
    PreA = np.zeros((N, L))
    PreB = np.zeros((N, L))
    omh = 1.0 - h
    h_rev = h[::-1]
    
    for a in range(N):
        # Term 1: -(1-h[0:N]) at k = N-1-a to 2N-2-a
        PreA[a, N-1-a : 2*N-1-a] = -omh
        # Term 2: +h[N-1::-1] at k = a to a+N-1
        PreA[a, a : a+N] += h_rev
    
    for b in range(N):
        # Term 3: +(1-h[0:N]) at k = N-1-b to 2N-2-b
        PreB[b, N-1-b : 2*N-1-b] = omh
        # Term 4: -h[N-1::-1] at k = b to b+N-1
        PreB[b, b : b+N] += -h_rev
    
    return PreA, PreB
```

At k=a+1: h[N-1-a+a+1] = h[N].

h[i] = h[N-1], h[N-2], ..., h[0].
That's h_rev (h in reverse order) placed at positions a, a+1, ..., a+N-1.

So PreA[a, a:a+N] += h_rev is correct! h_rev = [h[N-1], h[N-2], ..., h[0]].
At position a: h_rev[0] = h[N-1]. ✓
At position a+1: h_rev[1] = h[N-2]. ✓
...
At position a+N-1: h_rev[N-1] = h[0]. ✓

def greedy_local_search(h, N, max_iters=200, deadline=None):
    h = h.astype(float).copy()
    nh = N // 2
    L = 2*N - 1
    PreA, PreB = compute_precomputed(h, N)
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        R = fft_corr(h)
        cur_max = np.max(R)
        
        PrefA = PreA[ones_idx]  # (n_ones, L)
        PrefB = PreB[zeros_idx]  # (n_zeros, L)
        
        # Find best swap
        best_val = cur_max
        best_a_local = -1
        best_b_local = -1
        
        # Batch: process 20 a's at a time
        batch_size = 20
        for start in range(0, len(ones_idx), batch_size):
            end = min(start + batch_size, len(ones_idx))
            batch_A = PrefA[start:end]  # (batch, L)
            # new_maxs[i, j] = max_k(R[k] + batch_A[i, k] + PrefB[j, k])
            # = max_k(R[None,:,k] + batch_A[i, :]) + ... no, doesn't factor.
            
            # (batch, n_zeros, L)
            # new_maxs_batch = np.max(R[None,None,:] + batch_A[:,None,:] + PrefB[None,:,:], axis=2)
            # This is (20, 210, 759) = 3.2M values = 25MB. OK.
            
            new_maxs_batch = np.max(R[None, None, :] + batch_A[:, None, :] + PrefB[None, :, :], axis=2)
            # Shape: (batch, n_zeros)
            
            local_best = np.min(new_maxs_batch)
            if local_best < best_val - 1e-10:
                best_val = local_best
                li, lj = np.unravel_index(np.argmin(new_maxs_batch), new_maxs_batch.shape)
                best_a_local = ones_idx[start + li]
                best_b_local = zeros_idx[lj]
        
        if best_a_local < 0:
            break  # Local optimum
        
        h[best_a_local] = 0
        h[best_b_local] = 1
        PreA, PreB = compute_precomputed(h, N)
    
    return h
