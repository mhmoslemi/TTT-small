import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    e /= e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if e[i] > 1e-8:
            g += e[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_run(h, N, deadline, T_seq, lam_start=5.0):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    for T in T_seq:
        if time.time() > deadline:
            break
        lam = lam_start
        for _ in range(3):
            if time.time() > deadline:
                break
            res = sp_min(smooth_max_grad, h, args=(T, target, lam),
                        jac=True, method='L-BFGS-B', 
                        bounds=[(0,1)]*N,
                        options={'maxiter':100, 'maxfun':300, 'ftol':1e-13})
            h = proj01(res.x, target)
            c = compute_c5(h, N)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lam *= 3.0
    return best_h, best_c

def subgrad(h, N, deadline, alpha0=0.03, beta=0.93, atol=1e-4):

    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    alpha = alpha0
    for it in range(6000):
        if time.time() > deadline:
            break
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 500 == 0:
            alpha *= 0.92
            beta = min(0.97, 0.90 + 0.0001 * it)
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    e /= e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if e[i] > 1e-8:
            g += e[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_phase(h, N, deadline, T_seq, lam_start=4.0, max_outer=3):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    for T in T_seq:
        if time.time() > deadline:
            break
        lam = lam_start
        for _ in range(max_outer):
            if time.time() > deadline:
                break
            res = sp_min(smooth_obj, h, args=(T, target, lam),
                        jac=True, method='L-BFGS-B',
                        bounds=[(0, 1)] * N,
                        options={'maxiter': 120, 'maxfun': 350, 'ftol': 1e-14})
            h = proj01(res.x, target)
            c = compute_c5(h, N)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lam *= 2.5
    return best_h, best_c

def bin_local(h, N, deadline, max_iters=3000, ntri=20, anneal=0.0):

    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[np.random.choice(len(idx), n1 - nh, replace=False)]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[np.random.choice(len(idx), nh - n1, replace=False)]] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    temp = anneal
    recent = set()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bd = -1e10
        bm = None

        temp = max(0.0, anneal * max(0.0, 1.0 - it / float(max_iters)))
    return best_h, best_c

def subgrad_ne(h, N, deadline, alpha0=0.03, beta=0.94, atol=5e-5):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    alpha = alpha0
    for it in range(8000):
        if time.time() > deadline:
            break
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 500 == 0:
            alpha *= 0.93
            beta = min(0.97, 0.91 + 0.0001 * it)
    return best_h, best_pk * (2.0 / N)

def try_res(h_best, c5_best, Nt, deadline):
    N = len(h_best)
    if N == Nt or Nt % 2:
        return h_best, c5_best
    x_b = np.linspace(0, 2, N, endpoint=False)
    xn = np.linspace(0, 2, Nt, endpoint=False)
    hi = np.interp(xn, x_b, h_best)
    ht = (hi > 0.5).astype(float)
    nh = Nt // 2
    n1 = int(ht.sum())
    if n1 > nh:
        idx = np.where(ht == 1)[0]
        ht[idx[np.random.choice(len(idx), n1 - nh, replace=False)]] = 0
    elif n1 < nh:
        idx = np.where(ht == 0)[0]
        ht[idx[np.random.choice(len(idx), nh - n1, replace=False)]] = 1
    hr, _ = bin_local(ht, Nt, min(deadline, time.time() + 30), max_iters=250, ntri=18)
    c5r = compute_c5(hr, Nt)
    if c5r < c5_best:
        return hr, c5r
    return h_best, c5_best
