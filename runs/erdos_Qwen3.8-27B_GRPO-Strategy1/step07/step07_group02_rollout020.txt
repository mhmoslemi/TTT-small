import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def nesterov_opt(h, N, deadline, k_top=60, lr0=0.01, beta0=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h))) + 1e-15
    v = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    stall = 0
    while time.time() < deadline:
        y = h + v
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        h_new = project_01(h + v - lr * grad, target)
        v = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.97
            beta = min(0.98, beta0 + it * 0.0003)

def sa_search(h, N, deadline, T0=0.3, T_min=0.01, cool=0.999):

def greedy_construct(N):
    h = np.zeros(N)
    h[0] = 1
    count = 1
    while count < N // 2:
        best_pos = 0
        best_val = np.inf
        for pos in range(N):
            if h[pos] > 0:
                continue
            h[pos] = 1
            val = float(np.max(fft_corr(h)))
            h[pos] = 0
            if val < best_val:
                best_val = val
                best_pos = pos
        h[best_pos] = 1
        count += 1
    return h

    elif s < target:

        h_bt[np.random.choice(idx, target - s, replace=False)] = 1
    return h_bt

Wait, I can be smarter. Instead of computing the full FFT for each trial, I can compute the CHANGE in correlation due to a single swap analytically.

When we flip h[p] from 1 to 0 and h[q] from 0 to 1:
- The change in h is: Δh = -1 at p, +1 at q
- The change in (1-h) is: +1 at p, -1 at q
- The change in corr = h*(1-h_shifted) is:
  Δcorr[k] = sum_i Δh[i] * (1-h[i+k]) + h[i] * Δ(1-h)[i+k]
  = sum_i Δh[i] * (1-h[i+k]) - sum_i h[i] * Δh[i+k]

def swap_delta(h, N, p, q):

    Compute the change in corr array when h[p]:1->0 and h[q]:0->1.
    p is currently 1, q is currently 0.
    Returns the change in the full correlation array.
    """

    # Δ(1-h): +1 at p, -1 at q

    # = sum_i h[i] * (1-h[i + (N-1-m)])

    # corr[m] = sum_i h[i] * (1-h[i + (N-1-m)])
