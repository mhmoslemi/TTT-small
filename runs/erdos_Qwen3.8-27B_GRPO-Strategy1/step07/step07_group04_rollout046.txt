        # Greedy polish at N1

        h3g, c3g = greedy_pass(h3, N1, gdl3, max_iter=100, top_k=5, sf=0.4)
        c5_3g = c3g * (2.0 / N1)
        if c5_3g < best_c5:
            best_c5, best_h = c5_3g, h3g

    # Phase 4: Downsample back to N0 and refine
    N_fin = len(best_h)
    if N_fin != N0:
        h_ds = ensure_sum(upsample(best_h, N_fin, N0), N0)
        d4 = min(deadline - 60, time.time() + 60)
        h4, c4 = sa_epoch(h_ds, N0, d4, 2.5, 0.996, 0.01,
                          max_sweeps=4000, top_k=4, sf=0.7,
                          kick_every=100, kick_size=3)
        c5_4 = c4 * (2.0 / N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Phase 5: Final intensive greedy polish
    N_fin = len(best_h)
    d5 = deadline - 2
    sf_fin = 1.0 if N_fin <= 760 else 0.5
    hF, cF = greedy_pass(best_h, N_fin, d5, max_iter=500, top_k=3, sf=sf_fin)
    c5F = cF * (2.0 / N_fin)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
