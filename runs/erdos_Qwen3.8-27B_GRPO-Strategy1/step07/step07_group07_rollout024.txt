```python
def smooth_grad(h, corr, eps=30, K=100):
    N = len(h)
    center = N - 1
    c = corr * (2.0 / N)
    # Softmax weights for top-K
    top_k_idx = np.argsort(c)[-K:]
    top_k_vals = c[top_k_idx]
    M = top_k_vals[-1]
    weights = np.exp(eps * (top_k_vals - M))
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for i, (idx, w) in enumerate(zip(top_k_idx, weights)):
        L = int(idx) - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:N-L]
        else:
            L2 = -L
            grad[L2:] += w * (1.0 - h[:N-L2])
            grad[:N-L2] -= w * h[L2:]
    return grad * (2.0 / N)
```

```python
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    eps = eps_start
    stale = 0
    
    while time.time() < deadline:
        t += 1
        lr = lr0 / np.sqrt(1 + t * 0.01)
        
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1

            lr *= 0.8
            stale = 0

        v = beta2 * v + (1 - beta2) * g * g

        vhat = v / (1 - beta2 ** t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
```

The challenge is step 2 efficiently. For a swap at (i,j):

- We want to minimize the new max

- Look at the top-K current lags
- For each candidate swap, compute what the new values would be for those K lags
- Pick the swap that minimizes the max of those K values

```python
def binary_best_improve(h, deadline, K=20, max_rounds=500):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    c = float(np.max(corr)) * (2.0 / N)
    best_c = c
    best_h = h.copy()
    center = N - 1
    
    round_ = 0
    while time.time() < deadline and round_ < max_rounds:
        # Get top K lags
        top_k_idx = np.argsort(corr)[-K:]
        top_k_lags = (top_k_idx - center).astype(int)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_move = None
        best_new_max = float(np.max(corr))
        
        for oi in ones:
            for zi in zeros:
                # Compute new top-K values after swap
                new_max = 0.0
                for L in top_k_lags:
                    idx = center + L
                    delta = 0.0
                    # Effect of removing 1 at oi, adding 1 at zi on corr[idx]
                    # corr[idx] = sum_m h[m] * (1-h[m-L]) (adjusted for sign)
                    # After swap: h[oi] 1->0, h[zi] 0->1
                    # Change in corr at lag L:
                    #   - contribution of h[oi]: was (1-h[oi-L]) if oi+L valid, etc.
                    #   + contribution of h[zi]: now (1-h[zi-L]) if zi+L valid, etc.
                    #   - also affects the (1-h[.]) part for other positions
                    if L > 0:
                        # corr[L] = sum_{m=0}^{N-L-1} h[m]*(1-h[m+L])
                        # h[oi] 1->0: remove (1-h[oi+L]) if 0<=oi<N-L; add h[oi-L] if L<=oi<N
                        # h[zi] 0->1: add (1-h[zi+L]) if 0<=zi<N-L; remove h[zi-L] if L<=zi<N
                        if 0 <= oi < N - L:
                            delta -= (1 - h[oi + L])
                        if L <= oi < N:
                            delta += h[oi - L]  # h[oi-L]*(1-h[oi]) was contributing, now h[oi]=0
                        # Wait, h[oi-L]*(1-h[oi]) : when h[oi] was 1, this was 0. Now h[oi]=0, so it's h[oi-L]*1 = h[oi-L]
                        # So the change for this term is +h[oi-L] (it goes from 0 to h[oi-L])
                        # And h[oi]*(1-h[oi+L]) : this was (1-h[oi+L]), now it's 0. Change: -(1-h[oi+L])
                        if 0 <= zi < N - L:
                            delta += (1 - h[zi + L])
                        if L <= zi < N:
                            delta -= h[zi - L]  # h[zi-L]*(1-h[zi]) was h[zi-L]*0=0, now h[zi-L]*1... 
                            # Wait, h[zi] was 0, so h[zi-L]*(1-h[zi]) = h[zi-L]*1 = h[zi-L]
                            # Now h[zi] = 1, so h[zi-L]*(1-h[zi]) = 0. Change: -h[zi-L]
                    elif L < 0:
                        L2 = -L
                        # corr[-L2] = sum_{m=L2}^{N-1} h[m]*(1-h[m-L2])
                        if L2 <= oi < N:
                            delta -= (1 - h[oi - L2])
                        if 0 <= oi < N - L2:
                            delta += h[oi + L2]
                        if L2 <= zi < N:
                            delta += (1 - h[zi - L2])
                        if 0 <= zi < N - L2:
                            delta -= h[zi + L2]
                    else:
                        delta = 0  # L=0: h[m]*(1-h[m]) = 0 for binary
                    
                    new_val = corr[idx] + delta
                    if new_val > new_max:
                        new_max = new_val
                
                if new_max < best_new_max - 1e-10:
                    best_new_max = new_max
                    best_move = (oi, zi)
        
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
            corr = compute_corr(h)
            c = float(np.max(corr)) * (2.0 / N)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            break  # No improving move found
        
        round_ += 1
    
    return best_h, best_c
```
