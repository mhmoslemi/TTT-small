    temp = init_temp
    stall = 0

def nesterov(h_init, N, deadline, k_start=200, k_end=40, lr0=0.02):
    h = h_init.astype(float).copy()
    h = project_sum(h, N / 2.0)
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        k_cur = max(k_end, int(k_start * max(0, 1.0 - it / 2000.0)))
        k_cur = min(k_cur, 2 * N - 1)
        top = np.argpartition(corr, -k_cur)[-k_cur:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_cur
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.97
            beta = min(0.98, 0.88 + it * 0.00005)
    
    return best_h, best_pk * dx

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    vp = (pp >= 0) & (pp < N)

    hp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def nesterov(hi, N, deadline, ks=250, ke=50, lr0=0.015):
    h = project_sum(hi.copy(), N / 2.0)
    dx = 2.0 / N
    bh = h.copy()
    bp = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    ctr = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        kc = max(ke, min(ks, int(ks * max(0.15, 1.0 - it / 3000.0))))
        top = np.argpartition(corr, -kc)[-kc:]
        pk = float(corr[top].max())
        if pk < bp - 1e-14:
            bp = pk
            bh = h.copy()
        g = np.zeros(N)
        for ai in top:
            L = int(ai - ctr)
            if L == 0:
                g += 1.0 - 2.0 * y
            elif L > 0:
                g[L:] += 1.0 - y[:N - L]
                g[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    g[:N - M] += 1.0 - y[M:]
                    g[M:] -= y[:N - M]
        g /= kc
        hn = project_sum(y - lr * g, N / 2.0)
        m = beta * (hn - h)
        h = hn
        it += 1
        if it % 150 == 0:
            lr *= 0.96
            beta = min(0.98, 0.89 + it * 4e-5)
    return bh, bp * dx

def upsample(ho, No, Nn):
    xo = np.linspace(0, 2, No, endpoint=False)
    xn = np.linspace(0, 2, Nn, endpoint=False)
    return np.interp(xn, xo, ho)
