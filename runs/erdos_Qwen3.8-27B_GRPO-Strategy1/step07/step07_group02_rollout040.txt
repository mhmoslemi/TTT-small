                best_i = ones[a_start + local_best_idx[0]]
                best_j = zeros[local_best_idx[1]]
        
        if best_i == -1:
            break  # Local optimum
        
        # Apply swap
        h[best_i] = 0
        h[best_j] = 1
    
    return h, compute_c5(h)
