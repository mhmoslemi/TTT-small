    valid_p_minus = (p - lam >= 0) & (p - lam < N)
    idx_pp = np.clip(p + lam, 0, N-1)
    idx_pm = np.clip(p - lam, 0, N-1)
    delta[valid_p_plus] -= (1 - h[idx_pp[valid_p_plus]])
    delta[valid_p_minus] += h[idx_pm[valid_p_minus]]
    # Effect of q going 0->1
    valid_q_plus = (q + lam >= 0) & (q + lam < N)
    valid_q_minus = (q - lam >= 0) & (q - lam < N)
    idx_qp = np.clip(q + lam, 0, N-1)
    idx_qm = np.clip(q - lam, 0, N-1)
    delta[valid_q_plus] += (1 - h[idx_qp[valid_q_plus]])
    delta[valid_q_minus] -= h[idx_qm[valid_q_minus]]
    return delta
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N, seed=77):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_E(h, N):
    L = 2 * N - 1
    E = np.zeros((N, L))
    o_h = 1.0 - h
    h_rev = h[::-1]
    for i in range(N):
        E[i, N - 1 - i : 2 * N - 1 - i] += o_h[i:]
        E[i, i : N + i] -= h_rev
    return E

def greedy_binary(h, N, deadline, max_sweeps=3000, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = fft_corr(h)
    best_P = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        E = compute_E(h, N)
        P = float(c.max())
        if P < best_P - 1e-12:
            best_P = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_po = min(150, len(ones))
        n_zo = min(150, len(zeros))
        p_list = rng.choice(ones, n_po, replace=False)
        q_list = rng.choice(zeros, n_zo, replace=False)
        best_new = P - 1e-15
        best_pq = None
        for p in p_list:
            base = c - E[p, :]
            for q in q_list:
                m = float(np.max(base + E[q, :]))
                if m < best_new:
                    best_new = m
                    best_pq = (int(p), int(q))
        if best_pq is None:
            n_r = min(2 + stall // 15, 6)
            for _ in range(n_r):
                pp = int(ones[rng.randint(len(ones))])
                qq = int(zeros[rng.randint(len(zeros))])
                h[pp], h[qq] = 0.0, 1.0
            c = fft_corr(h)
            stall = 0
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
    return best_h, best_P * (2.0 / N)

def sa_binary(h, N, deadline, seed=42, T0=8.0, T1=0.05):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = fft_corr(h)
    P = float(c.max())
    best_P = P
    best_h = h.copy()
    n_swaps = max(5000, int((deadline - time.time()) * 300))
    for it in range(n_swaps):
        if time.time() > deadline:
            break
        frac = (it + 1) / n_swaps
        T = T0 * (T1 / T0) ** frac
        p = int(rng.randint(N))
        q = int(rng.randint(N))
        if p == q or h[p] == h[q]:
            continue
        if h[p] == 0:
            p, q = q, p
        E = compute_E(h, N)
        new_c = c - E[p, :] + E[q, :]
        new_P = float(np.max(new_c))
        delta = new_P - P
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-6)):
            h[p], h[q] = 0.0, 1.0
            c = new_c
            P = new_P
            if P < best_P - 1e-12:
                best_P = P
                best_h = h.copy()
    return best_h, best_P * (2.0 / N)

def soft_max_nesterov(h_init, N, deadline, k=50, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.994
            beta = min(0.98, 0.85 + it * 0.0002)
        if it % 300 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(15.0, min(400.0, T * (0.85 if gap < 0.003 else 1.1)))
    return best_h, best_pk * dx

def upsample_h(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old
