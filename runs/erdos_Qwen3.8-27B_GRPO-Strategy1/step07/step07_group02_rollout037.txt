```python
def C_row(h, N, i):
    """Compute C[i, :] for a single position i"""
    lags = np.arange(2*N-1) - (N-1)
    im = i - lags
    ip = i + lags
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N-1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N-1)], 0.0)
    return A - B
```

For δ=-1, h=1: 1*0 + (-1) - 2*1*(-1) - 1 = 0 - 1 + 2 - 1 = 0. ✓

```python
import numpy as np
import time

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj_val(h, N):
    return float(np.max(corr(h))) * 2.0 / N

def C_row(h, N, i):
    lags = np.arange(2 * N - 1) - (N - 1)
    im = i - lags
    ip = i + lags
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    C = A - B
    C[N - 1] = 0.0
    return C

def fix_binary(h, target):
    n1 = int(np.round(h).sum())
    r = np.round(h)
    if n1 > target:
        idx = np.where(r == 1)[0]
        sel = np.random.choice(idx, n1 - target, replace=False)
        r[sel] = 0
    elif n1 < target:
        idx = np.where(r == 0)[0]
        sel = np.random.choice(idx, target - n1, replace=False)
        r[sel] = 1
    return r.astype(float)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def tabu_search(h0, N, deadline, tabu_len=25, sample_sz=100, kick_thresh=80,
                kick_sz=4, anneal_rate=0.008, max_kick_streak=6):
    h = fix_binary(h0.copy(), N // 2)
    c = corr(h)
    best_val = float(np.max(c))
    best_h = h.copy()
    tabu_until = np.zeros(N, dtype=np.int32)
    it = 0
    stuck = 0
    kick_streak = 0

    while time.time() < deadline:
        it += 1
        P = float(np.max(c))
        if P < best_val - 1e-12:
            best_val = P
            best_h = h.copy()
            stuck = 0
            kick_streak = 0
        else:
            stuck += 1

        if stuck > kick_thresh:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if kick_streak >= max_kick_streak or len(ones) < 3 or len(zeros) < 3:
                if time.time() > deadline - 2:
                    break
                kick_streak += 1
                stuck = 0
                continue
            kn = min(kick_sz + kick_streak, len(ones), len(zeros))
            ko = np.random.choice(ones, kn, replace=False)
            kz = np.random.choice(zeros, kn, replace=False)
            for i in range(kn):
                h[ko[i]] = 0
                h[kz[i]] = 1
            c = corr(h)
            tabu_until[ko] = it + tabu_len
            tabu_until[kz] = it + tabu_len
            stuck = 0
            kick_streak += 1
            continue

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)

        C_o = np.stack([C_row(h, N, int(i)) for i in os_])
        C_z = np.stack([C_row(h, N, int(i)) for i in zs])
        newP = (c[None, None, :] - C_o[None, :, :] + C_z[:, None, :]).max(axis=2)

        o_tab = tabu_until[os_] > it
        z_tab = tabu_until[zs] > it
        for i in range(ns):
            if o_tab[i]:
                newP[:, i] = 1e9
        for j in range(nz):
            if z_tab[j]:
                newP[j, :] = 1e9

        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if np.isinf(mv) or mv > 1e8:
            break

        accept = mv < P - 1e-12
        if not accept and stuck > 20:
            accept = np.random.random() < np.exp(-stuck * anneal_rate)

        if not accept:
            if stuck > 200:
                ones2 = np.where(h == 1)[0]
                zeros2 = np.where(h == 0)[0]
                p = int(np.random.choice(ones2))
                q = int(np.random.choice(zeros2))
                h[p] = 0
                h[q] = 1
                c = corr(h)
                tabu_until[p] = it + tabu_len
                tabu_until[q] = it + tabu_len
                stuck = 0
            else:
                break
        else:
            p = int(os_[mi[1]])
            q = int(zs[mi[0]])
            h[p] = 0
            h[q] = 1
            c = c - C_o[mi[1], :] + C_z[mi[0], :]
            c[N - 1] = float(np.sum(h * (1 - h)))
            tabu_until[p] = it + tabu_len
            tabu_until[q] = it + tabu_len

    return best_h, best_val * 2.0 / N

def subgradient_opt(h0, N, deadline, k_active=50, lr0=0.03, decay=0.9995):
    h = h0.astype(float).copy()
    target = N / 2.0
    h = project_01(h, target)
    best_h = h.copy()
    c = corr(h)
    best_val = float(np.max(c))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    stuck = 0

    while time.time() < deadline:
        it += 1
        y = np.clip(h + m, 0, 1)
        c = corr(h)
        P = float(np.max(c))
        if P < best_val - 1e-10:
            best_val = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

        top = np.argpartition(c, -k_active)[-k_active:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            k = int(ai - center)
            if k == 0:
                grad += (1.0 - 2.0 * h)
            elif k > 0:
                if k < N:
                    grad[k:] += (1.0 - h[:N - k])
                    grad[:N - k] -= h[k:]
            else:
                M = -k
                if M < N:
                    grad[:N - M] += (1.0 - h[M:])
                    grad[M:] -= h[:N - M]
        grad /= k_active

        h_new = project_01(h - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new

        if it % 100 == 0:
            lr *= decay
            beta = min(0.95, 0.85 + it * 0.0003)
            if stuck > 150:
                h = project_01(h + np.random.randn(N) * 0.05, target)
                m = np.zeros(N)
                stuck = 0
                lr = max(lr, 0.01)

    return best_h, best_val * 2.0 / N

def multi_swap_refine(h0, N, deadline, pair_sz=2, sample_sz=40):
    h = fix_binary(h0.copy(), N // 2)
    c = corr(h)
    best_val = float(np.max(c))
    best_h = h.copy()
    stuck = 0
    it = 0

    while time.time() < deadline:
        it += 1
        P = float(np.max(c))
        if P < best_val - 1e-12:
            best_val = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 60:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < pair_sz or len(zeros) < pair_sz:
            break

        from itertools import combinations
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)

        best_mv = P
        best_pair = None
        C_os = {int(i): C_row(h, N, int(i)) for i in os_}
        C_zs = {int(j): C_row(h, N, int(j)) for j in zs}

        if pair_sz == 1:
            for i in range(ns):
                for j in range(nz):
                    mv = (c - C_os[int(os_[i])] + C_zs[int(zs[j])]).max()
                    if mv < best_mv:
                        best_mv = float(mv)
                        best_pair = (int(os_[i]), int(zs[j]))
        else:
            o_combos = list(combinations(range(ns), pair_sz))
            z_combos = list(combinations(range(nz), pair_sz))
            for co in o_combos:
                delta_o = np.zeros(2 * N - 1)
                for ii in co:
                    delta_o -= C_os[int(os_[ii])]
                for cz in z_combos:
                    delta = delta_o.copy()
                    for jj in cz:
                        delta += C_zs[int(zs[jj])]
                    mv = (c + delta).max()
                    if mv < best_mv:
                        best_mv = float(mv)
                        best_pair = (tuple(int(os_[ii]) for ii in co),
                                     tuple(int(zs[jj]) for jj in cz))

        if best_pair is None or best_mv >= P - 1e-12:
            if stuck > 30:
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1
                c = corr(h)
            else:
                break
        else:
            po, qz = best_pair
            for p in po:
                h[p] = 0
            for q in qz:
                h[q] = 1
            c = corr(h)

    return best_h, best_val * 2.0 / N

    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)

    N0 = len(h0)
    best_h = h0.copy()

    if obj_val(h_bin, N0) < best_c5:

    h_t, c_t = tabu_search(h_bin, N0, d1, tabu_len=30, sample_sz=120, kick_thresh=60, kick_sz=4)
    if c_t < best_c5:
        best_c5, best_h = c_t, h_t.copy()

    # Phase 2: Subgradient continuous (25%)
    if time.time() < deadline - 0.25 * budget_s:
        d2 = min(deadline - 0.25 * budget_s, time.time() + 0.27 * budget_s)
        hc = best_h + np.random.randn(N0) * 0.15
        hc = project_01(hc, N0 / 2.0)
        hs, cs = subgradient_opt(hc, N0, d2, k_active=60, lr0=0.025)
        if cs < best_c5:
            best_c5, best_h = cs, hs.copy()
        h_bin2 = fix_binary(hs.copy(), N0 // 2)
        cb = obj_val(h_bin2, N0)
        if cb < best_c5:
            best_c5, best_h = cb, h_bin2

    # Phase 3: Tabu from best (25%)
    if time.time() < deadline - 0.15 * budget_s:
        h3 = fix_binary(best_h.copy(), N0 // 2)
        d3 = min(deadline - 0.15 * budget_s, time.time() + 0.25 * budget_s)
        h_t3, c_t3 = tabu_search(h3, N0, d3, tabu_len=20, sample_sz=140, kick_thresh=50,
                                  kick_sz=5, anneal_rate=0.012)
        if c_t3 < best_c5:
            best_c5, best_h = c_t3, h_t3.copy()

    # Phase 4: Multi-swap refine (10%)
    if time.time() < deadline - 60:
        h4 = fix_binary(best_h.copy(), N0 // 2)
        d4 = min(deadline - 60, time.time() + 0.1 * budget_s)
        h_m, c_m = multi_swap_refine(h4, N0, d4, pair_sz=2, sample_sz=30)
        if c_m < best_c5:
            best_c5, best_h = c_m, h_m.copy()

    # Phase 5: Try different N values
    for N_try in [N0 - 10, N0 + 10, N0 + 20]:
        if time.time() > deadline - 50:
            break
        if N_try < 100 or N_try % 2 != 0:
            continue
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = fix_binary(h_int, N_try // 2)
        sub_dl = min(deadline - 15, time.time() + 40)
        h_rt, c_rt = tabu_search(h_bt, N_try, sub_dl, tabu_len=20, sample_sz=80,
                                  kick_thresh=40, kick_sz=3)
        if c_rt < best_c5:
            best_c5, best_h = c_rt, h_rt

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        hF = fix_binary(best_h.copy(), len(best_h) // 2)
        hF, cF = tabu_search(hF, len(best_h), deadline - 2, tabu_len=15,
                              sample_sz=150, kick_thresh=40, kick_sz=3, anneal_rate=0.015)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    final_c5 = obj_val(best_h, len(best_h))
    h_rounded = fix_binary(best_h, len(best_h) // 2)
    cr = obj_val(h_rounded, len(best_h))
    if cr <= final_c5:
        final_c5 = cr
        best_h = h_rounded

    return best_h, final_c5, len(best_h)
```
