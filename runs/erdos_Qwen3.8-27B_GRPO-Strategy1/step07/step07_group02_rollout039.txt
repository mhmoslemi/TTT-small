import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = h.sum() - target
        if abs(d) < 1e-15:
            break
        h = np.clip(h - d / len(h), 0.0, 1.0)
    return h

def softmax_grad(h, beta):
    N = len(h)
    L = 2 * N - 1
    corr = fft_corr(h)
    e = beta * (corr - corr.max())
    e = np.minimum(e, 0)
    p = np.exp(e)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(L)
        p[N-1] = 1.0
    else:
        p /= ps
    grad = np.zeros(N)
    for s in range(-(N-1), N):
        w = p[s + (N-1)]
        if w < 1e-14:
            continue
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            nv = N - s
            grad[:nv] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:nv]
        else:
            m = -s
            nv = N - m
            grad[m:] += w * (1.0 - h[:nv])
            grad[:nv] -= w * h[m:]
    return grad, float(np.max(corr))

def continuous_opt(h, N, deadline, beta0=50.0, lr0=0.03, max_iters=5000):
    """Softmax-guided gradient descent."""
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = float('inf')
    lr = lr0
    beta = beta0
    target = N / 2.0
    vel = np.zeros(N)
    momentum = 0.9
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, c_raw = softmax_grad(h, beta)
        # Project gradient to zero-sum subspace
        grad -= grad.mean()
        vel = momentum * vel - lr * grad
        h = project_box_sum(h + vel, target)
        c = c_raw * (2.0 / N)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        # Anneal
        if it % 50 == 0:
            lr *= 0.985
            beta *= 1.003
    return best_h, best_c

        C[i, :] = compute_C_row(h, N, i)

            break

        # For position i with h[i]=1: contributes (1-h[i+shift]) to corr[k_abs]

        # (1-h)[i+shift] is 1 when h[i+shift]=0

        # As h=1: o contributes (1-h[o+shift]) to corr[shift] for valid o+shift

        # But also: o appears in OTHER positions' contributions:
        # corr[shift] also has h[j]*h[j+shift] terms; if j+shift=o, then h[j]*h[o] is affected
        # h[o] going 1->0 removes h[j]*1 from those terms: corr increases by h[j] for j where j+shift=o
        # i.e., j = o-shift, valid j
        
        # This is what the C matrix stores. Let me define:
        # C[i][k_abs] = the amount corr[k_abs] decreases when h[i] goes from 1 to 0
        # = (1-h[i+shift]) + h[i-shift] (for valid indices)
        #   [from o's own contribution] + [from other positions that had h[o] as second factor]
        
        # Actually: corr[k] = sum_j h[j]*(1-h[j+s]) where s = shift
        # d(corr[k])/d(h[i]) = (1-h[i+s]) - h[i-s] (for valid indices, 0 otherwise)
        # This is the change when h[i] increases by 1 (from 0 to 1)
        # So flipping 1->0 changes corr by: -(1-h[i+s]) + h[i-s] = -C[i]
        # Flipping 0->1 changes corr by: +(1-h[i+s]) - h[i-s] = +C[i]
        
        # After flipping o: 1->0 and z: 0->1:
        # new_corr = corr - C[o] + C[z] (approximately, ignoring second-order effects)
        # The second-order effect: C[o] was computed with h[z]=0, but after flip h[z]=1
        # And C[z] was computed with h[o]=1, but after flip h[o]=0
        # Second order: the interaction between o and z
        # corr depends on h[o]*h[z] terms if o and z are related by a shift
        # This is O(1) effect, small enough to ignore for the search
        
        Co = C[oi]  # shape (ns, L)
        Cz = C[zi]  # shape (nz, L)
        L = 2*N-1
        newP = (corr[None, None, :] - Co[None, :, :] + Cz[:, None, :]).max(axis=2)

            C[zi[mi[0]], :] = compute_C_row(h, N, zi[mi[0]])
        else:
            break
    
    return best_h, best_c * (2.0 / N)

def compute_C_row(h, N, i):
    """Compute C[i]: the gradient of corr w.r.t. h[i] (increase from 0 to 1)."""
    L = 2 * N - 1
    C = np.zeros(L)
    s = i  # as first factor: h[i]*(1-h[i+shift])
    # (1-h[i+shift]) for valid shift: shift in [-i, N-1-i]
    lo = max(0, -i)  # shift >= -i means i+shift >= 0
    hi = min(2*N-2, N-1+i)  # shift <= N-1-i means i+shift <= N-1, so shift+k_index <= N-1-i, k_index = shift+N-1 <= 2N-2-i+... 
    # k_index = shift + (N-1), shift in [-(N-1), N-1], k_index in [0, 2N-2]
    # valid: 0 <= i+shift < N, i.e., -i <= shift <= N-1-i
    # k_index range: [N-1-i, N-1+N-1-i] = [N-1-i, 2N-2-i]
    lo_k = max(0, N-1-i)
    hi_k = min(2*N-2, 2*N-2-i)  # wait: shift <= N-1-i, k_index = shift+N-1 <= N-1-i+N-1 = 2N-2-i
    # and shift >= -i, k_index >= -i+N-1 = N-1-i
    # So lo_k = N-1-i, hi_k = 2N-2-i (but also must be in [0, 2N-2])
    lo_k = max(0, N-1-i)
    hi_k = min(2*N-2, 2*N-2-i)
    if lo_k <= hi_k:
        shifts = np.arange(lo_k, hi_k+1) - (N-1)
        C[lo_k:hi_k+1] += (1.0 - h[i + shifts])
    
    # As second factor: h[j]*(1-h[j+shift]) where j+shift = i, j = i-shift
    # shift in [i-N+1, i], k_index in [i, i+N-1]... 
    # j = i-shift >= 0 => shift <= i
    # j+shift = i < N (always true since i < N)
    # j < N => i-shift < N => shift > i-N => shift >= i-N+1
    # k_index = shift + N-1, range: [i-N+1+N-1, i+N-1] = [i, i+N-1]
    lo_k2 = max(0, i)
    hi_k2 = min(2*N-2, i+N-1)
    if lo_k2 <= hi_k2:
        shifts2 = np.arange(lo_k2, hi_k2+1) - (N-1)
        C[lo_k2:hi_k2+1] -= h[i - shifts2]
    
    return C

def fix_sum(h, target_sum):
    """Fix sum of binary h to target_sum."""
    s = int(h.sum())
    if s > target_sum:
        idx = np.where(h == 1)[0]
        n_remove = s - target_sum
        if n_remove > 0 and len(idx) >= n_remove:
            h[np.random.choice(idx, n_remove, replace=False)] = 0
    elif s < target_sum:
        idx = np.where(h == 0)[0]
        n_add = target_sum - s
        if n_add > 0 and len(idx) >= n_add:
            h[np.random.choice(idx, n_add, replace=False)] = 1
    return h

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        d = h.sum() - target
        if abs(d) < 1e-15:
            break
        h = np.clip(h - d / len(h), 0.0, 1.0)
    return h

def softmax_grad(h, beta):
    N = len(h)
    L = 2 * N - 1
    corr = fft_corr(h)
    e = beta * (corr - corr.max())
    e = np.minimum(e, 0.0)
    p = np.exp(e)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(L); p[N - 1] = 1.0
    else:
        p /= ps
    grad = np.zeros(N)
    for s in range(-(N - 1), N):
        w = p[s + (N - 1)]
        if w < 1e-14:
            continue
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            nv = N - s
            grad[:nv] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:nv]
        else:
            m = -s
            nv = N - m
            grad[m:] += w * (1.0 - h[:nv])
            grad[:nv] -= w * h[m:]
    return grad, float(np.max(corr))
