import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            perm = np.random.permutation(len(idx))[:n1 - nh]
            h[idx[perm]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        if len(idx) >= nh - n1:
            perm = np.random.permutation(len(idx))[:nh - n1]
            h[idx[perm]] = 1
    return h

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def cmat_search(h, N, deadline, max_iters=20000, sample_sz=200,
                temp=0.0, kick_thresh=60, kick_sz=5, refresh_every=200):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    C = np.array([compute_C_single(h, N, i) for i in range(N)])
    stuck = 0
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        if it % refresh_every == 0:
            C = np.array([compute_C_single(h, N, i) for i in range(N)])
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

            break

        nz = min(len(zeros), sample_sz)

        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        # Use top-3 correlation peaks for evaluation
        top_k = np.argpartition(corr, -3)[-3:]
        newP = np.maximum.reduce([
            (corr[top_k[0]] + Co[:, top_k[0]] + Cz[:, top_k[0]]).reshape(ns, nz) if False else
            np.max(corr[None, None, :] + Co[None, :, :] + Cz[:, None, :], axis=2)
        ])
        mi = np.unravel_index(np.argmin(newP), newP.shape)

        if mv < P - 1e-14:

            h[p] = 0
            h[q] = 1
            C[p] = compute_C_single(h, N, p)
            C[q] = compute_C_single(h, N, q)
            corr = fft_corr(h)
            P = float(np.max(corr))
            if P < best_c - 1e-14:

            else:

        elif temp > 0 and np.random.random() < temp * np.exp(-stuck * 0.004):

            C[p] = compute_C_single(h, N, p)
            C[q] = compute_C_single(h, N, q)
            corr = fft_corr(h); P = float(np.max(corr))
        elif stuck >= kick_thresh:

            C = np.array([compute_C_single(h, N, i) for i in range(N)])
            corr = fft_corr(h); P = float(np.max(corr))

        else:

    return best_h, best_c

def nesterov_opt(h, N, deadline, k=80, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.96, 0.88 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

corr[top5][None, :] + C[ones][:, top5] has shape (190, 5).

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            perm = np.random.permutation(len(idx))[:n1 - nh]
            h[idx[perm]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        if len(idx) >= nh - n1:
            perm = np.random.permutation(len(idx))[:nh - n1]
            h[idx[perm]] = 1
    return h

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def build_C(h, N):
    return np.array([compute_C_single(h, N, i) for i in range(N)])

def greedy_search(h, N, deadline, max_iters=100000, top_k=5, refresh_every=80,
                  kick_thresh=40, kick_sz=4, temp=0.0):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    C = build_C(h, N)
    stuck = 0
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        if it % refresh_every == 0:
            C = build_C(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        tk = np.argpartition(corr, -top_k)[-top_k:]
        corr_tk = corr[tk]
        score_o = np.max(corr_tk[None, :] + C[ones][:, tk], axis=1)
        io = int(np.argmin(score_o))
        base = (corr_tk + C[ones[io]][tk])[None, :]
        score_z = np.max(base + C[zeros][:, tk], axis=1)
        iz = int(np.argmin(score_z))
        newP = float(score_z[iz])
        if newP < P - 1e-14:
            p, q = int(ones[io]), int(zeros[iz])
            h[p] = 0
            h[q] = 1
            C[p] = compute_C_single(h, N, p)
            C[q] = compute_C_single(h, N, q)
            corr = fft_corr(h)
            P = float(np.max(corr))
            if P < best_c - 1e-14:
                best_c, best_h, stuck = P, h.copy(), 0
            else:
                stuck += 1
        else:
            stuck += 1
            if temp > 0 and np.random.random() < temp * np.exp(-stuck * 0.005):
                p, q = int(np.random.choice(ones)), int(np.random.choice(zeros))
                h[p] = 0; h[q] = 1
                C[p] = compute_C_single(h, N, p)
                C[q] = compute_C_single(h, N, q)
                corr = fft_corr(h); P = float(np.max(corr))
            elif stuck >= kick_thresh:
                for _ in range(kick_sz):
                    p, q = int(np.random.choice(ones)), int(np.random.choice(zeros))
                    h[p] = 0; h[q] = 1
                    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                C = build_C(h, N)
                corr = fft_corr(h); P = float(np.max(corr))
                stuck = 0
            else:
                p, q = int(np.random.choice(ones)), int(np.random.choice(zeros))
                h[p] = 0; h[q] = 1
                C[p] = compute_C_single(h, N, p)
                C[q] = compute_C_single(h, N, q)
                corr = fft_corr(h); P = float(np.max(corr))
    return best_h, best_c

def random_sample_search(h, N, deadline, max_iters=10000, sample_sz=180,
                         temp=0.0, kick_thresh=50, kick_sz=5):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    C = build_C(h, N)
    stuck = 0
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        new = corr[None, None, :] + C[os_][None, :, :] + C[zs][:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-14:
            p, q = int(os_[mi[1]]), int(zs[mi[0]])
            h[p] = 0; h[q] = 1
            C[p] = compute_C_single(h, N, p)
            C[q] = compute_C_single(h, N, q)
            corr = fft_corr(h); P = float(np.max(corr))
            if P < best_c - 1e-14:
                best_c, best_h, stuck = P, h.copy(), 0
            else:
                stuck += 1
        elif temp > 0 and np.random.random() < temp * np.exp(-stuck * 0.004):
            p, q = int(np.random.choice(ones)), int(np.random.choice(zeros))
            h[p] = 0; h[q] = 1
            C[p] = compute_C_single(h, N, p)
            C[q] = compute_C_single(h, N, q)
            corr = fft_corr(h); P = float(np.max(corr))
        elif stuck >= kick_thresh:

            C = build_C(h, N)

        else:
            break
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=70, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.994
            beta = min(0.96, 0.88 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
