    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def greedy_step(h, N, c):
    """One greedy swap step. Returns (new_h, new_max, p, q)."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) < 2 or len(zeros) < 2:
        return h, float(c.max()), None, None
    D = compute_D(h, N)
    Dz = D[zeros, :]
    best_new_P = float(c.max()) - 1e-14
    best_pq = None
    for p in ones:
        base = c - D[p, :]
        cand = base[None, :] - Dz
        npk = cand.max(axis=1)
        mi = int(np.argmin(npk))
        if npk[mi] < best_new_P:
            best_new_P = float(npk[mi])
            best_pq = (int(p), int(zeros[mi]))
    if best_pq is None:
        return h, float(c.max()), None, None
    new_h = h.copy()
    new_h[best_pq[0]] = 0.0
    new_h[best_pq[1]] = 1.0
    return new_h, best_new_P, best_pq[0], best_pq[1]

def sa_opt(h, N, deadline, T0=3.0, Tcool=0.996, T_min=0.03, sf=1.0, 
           kick_every=200, kick_size=5, block_prob=0.15, max_sweeps=20000):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Kick when stalled
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = max(T, T0 * 0.5)
            continue
        
        D = compute_D(h, N)
        Dz = D[zeros, :]
        
        if sf < 1.0:
            n_s = max(30, int(len(ones) * sf))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        
        # Choose move type
        use_block = (rng.random() < block_prob) and len(ones) >= 4 and len(zeros) >= 4
        
        if use_block:
            # Block move: find block of 2-4 consecutive ones and move to zeros
            bm = rng.randint(2, min(5, len(ones)//10 + 2))
            # Find start of a block of bm consecutive ones
            valid_starts = []
            for i in range(len(ones) - bm + 1):
                if np.all(ones[i:i+bm] == np.arange(ones[i], ones[i]+bm)):
                    valid_starts.append(int(ones[i]))
            if not valid_starts:
                valid_starts = [int(ones[rng.randint(len(ones)-1)])]
                bm = 1
            else:
                p_start = valid_starts[rng.randint(len(valid_starts))]
            
            # Find block of bm consecutive zeros
            valid_zeros = []
            for i in range(len(zeros) - bm + 1):
                if np.all(zeros[i:i+bm] == np.arange(zeros[i], zeros[i]+bm)):
                    valid_zeros.append(int(zeros[i]))
            if not valid_zeros:
                # Fall back to pair swap
                p = p_list[rng.randint(len(p_list))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
                T = max(T_min, T * Tcool)
                continue
            else:
                q_start = valid_zeros[rng.randint(len(valid_zeros))]
            
            # Evaluate block move
            delta_corr = -D[p_start:p_start+bm, :].sum(axis=0) + D[q_start:q_start+bm, :].sum(axis=0)
            new_P = float((c + delta_corr).max())
            delta = new_P - P
            if delta < -1e-14 or rng.random() < np.exp(-max(0, delta) / max(T, 1e-10)):
                h[p_start:p_start+bm] = 0.0
                h[q_start:q_start+bm] = 1.0
        else:
            # Pair swap (standard)
            best_new_P = P + 1.0
            best_pq = None
            for p in p_list:
                base = c - D[p, :]
                cand = base[None, :] - Dz
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
                if npk[mi] < best_new_P:
                    best_new_P = float(npk[mi])
                    best_pq = (int(p), int(zeros[mi]))
            if best_pq is None:
                break
            delta = best_new_P - P
            if delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
            else:
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
        
        T = max(T_min, T * Tcool)
    
    return best_h, best_c * (2.0 / N)

def adam_opt(h, N, deadline, lr=0.03, k_top=30, beta1=0.9, beta2=0.999, eps=1e-8):
    """Continuous Adam optimization of the max correlation."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)  # Adam momentum
    v = np.zeros(N)  # Adam velocity
    t = 0
    dx = 2.0 / N
    
    while time.time() < deadline:
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        # Get top-k shifts
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        
        # Compute gradient
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            L_shift = int(ai - center)
            if L_shift == 0:
                grad += (1.0 - 2.0 * h)
            elif L_shift > 0:
                grad[L_shift:] += (1.0 - h[:N - L_shift])
                grad[:N - L_shift] -= h[L_shift:]
            else:
                M = -L_shift
                if M < N:
                    grad[:N - M] += (1.0 - h[M:])
                    grad[M:] -= h[:N - M]
        grad = grad * (2.0 / k_top)  # Scale gradient
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        
        if t % 500 == 0:
            lr *= 0.95
    
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h.astype(float), L)
    G = np.fft.fft(np.flip(1.0 - h.astype(float)), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        p_list = ones if sf >= 1.0 else np.random.choice(ones, max(30, int(len(ones)*sf)), replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def sa_improved(h, N, deadline, T0=4.0, Tcool=0.9965, T_min=0.015, sf=1.0,
                kick_every=300, kick_size=5, double_prob=0.3, max_sweeps=50000):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    rng = np.random.RandomState(42)
    tabu = set()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                tabu.discard((p, q))
                tabu.discard((q, p))
            stall = 0
            T = max(T, T0 * 0.6)
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        p_list = ones if sf >= 1.0 else rng.choice(ones, max(40, int(len(ones)*sf)), replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            key = (int(p), int(zeros[mi]))
            if key in tabu:
                continue
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            p = int(p_list[rng.randint(len(p_list))])
            q = int(zeros[rng.randint(len(zeros))])
            delta = 0.0
        else:
            delta = best_new_P - P
        accept = delta < -1e-14
        if not accept and T > T_min:
            accept = rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))
        if accept and best_pq is not None:
            p, q = best_pq
            h[p], h[q] = 0.0, 1.0
            tabu.add((p, q))
            tabu.add((q, p))
        else:
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            tabu.add((p, q))
        if len(tabu) > 5000:
            tabu = set(list(tabu)[-2000:])
        if rng.random() < double_prob and len(ones) > 5 and len(zeros) > 5:
            ones2 = np.where(h == 1)[0]
            zeros2 = np.where(h == 0)[0]
            if len(ones2) >= 2 and len(zeros2) >= 2:
                p2 = int(ones2[rng.randint(len(ones2))])
                q2 = int(zeros2[rng.randint(len(zeros2))])
                h[p2], h[q2] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def adam_continuous(h, N, deadline, lr=0.02, k_top=20, beta1=0.85, beta2=0.999, eps=1e-7):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    dx = 2.0 / N
    while time.time() < deadline:
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            Ls = int(ai - center)
            if Ls == 0:
                grad += (1.0 - 2.0 * h)
            elif Ls > 0:
                grad[Ls:] += (1.0 - h[:N - Ls])
                grad[:N - Ls] -= h[Ls:]
            else:
                M = -Ls
                if M < N:
                    grad[:N - M] += (1.0 - h[M:])
                    grad[M:] -= h[:N - M]
        grad *= 2.0 / k_top
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        if t % 800 == 0:
            lr *= 0.97
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    h_bin = ensure_sum(h0, N0)
    # Phase 1: Quick greedy
    d1 = min(deadline - 700, t0 + 35)
    h1, c5_1 = greedy_loops(h_bin, N0, d1, max_sweeps=150, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    # Phase 2: Main SA with tabu + double moves
    d2 = min(deadline - 500, t0 + 280)
    h2, c5_2 = sa_improved(h_bin, N0, d2, T0=4.0, Tcool=0.9965, T_min=0.015,
                           sf=1.0, kick_every=350, kick_size=5, double_prob=0.25, max_sweeps=100000)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()
    # Phase 3: Adam continuous
    if time.time() < deadline - 400:
        hc = best_h.copy() if len(best_h) == N0 else upsample(best_h, len(best_h), N0)
        hc = project_sum(hc + np.random.randn(N0) * 0.04, N0 / 2.0)
        d3 = min(deadline - 350, t0 + 380)
        h3, c5_3 = adam_continuous(hc, N0, d3, lr=0.018, k_top=18)
        c5_3b = compute_c5(ensure_sum(h3, N0))
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
        if c5_3b < best_c5:
            best_c5, best_h = c5_3b, ensure_sum(h3, N0).copy()
    # Phase 4: Second SA round with perturbation
    if time.time() < deadline - 250:
        h4s = ensure_sum(best_h if len(best_h) == N0 else upsample(best_h, len(best_h), N0), N0)
        for _ in range(12):
            p = np.where(h4s == 1)[0]
            q = np.where(h4s == 0)[0]
            if len(p) > 0 and len(q) > 0:
                h4s[p[np.random.randint(len(p))]] = 0.0
                h4s[q[np.random.randint(len(q))]] = 1.0
        d4 = min(deadline - 200, t0 + 460)
        h4, c5_4 = sa_improved(h4s, N0, d4, T0=5.5, Tcool=0.9955, T_min=0.02,
                               sf=1.0, kick_every=250, kick_size=6, double_prob=0.3, max_sweeps=80000)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
    # Phase 5: Upsample to 2N0
    N1 = 2 * N0
    if time.time() < deadline - 180:
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d5a = min(deadline - 130, t0 + 540)
        h5a, c5_5a = greedy_loops(h_up, N1, d5a, max_sweeps=150, sf=0.55)
        if c5_5a < best_c5:
            best_c5, best_h = c5_5a, h5a.copy()
    if time.time() < deadline - 60:
        h_up2 = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d5b = min(deadline - 40, t0 + 620)
        h5b, c5_5b = sa_improved(h_up2, N1, d5b, T0=3.5, Tcool=0.996, T_min=0.02,
                                 sf=0.45, kick_every=400, kick_size=4, double_prob=0.2, max_sweeps=50000)
        if c5_5b < best_c5:
            best_c5, best_h = c5_5b, h5b.copy()
    # Phase 6: Final greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 2:
        hF = ensure_sum(best_h, N_fin)
        sf_f = 1.0 if N_fin <= 500 else 0.5
        hF2, c5F = greedy_loops(hF, N_fin, deadline - 0.5, max_sweeps=800, sf=sf_f)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
