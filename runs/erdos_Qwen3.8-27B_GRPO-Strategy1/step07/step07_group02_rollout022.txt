def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_row(h, N, idx):
    """Compute C[idx, :] - the effect of h[idx] going 0->1 on all correlations."""
    m = np.arange(2 * N - 1) - (N - 1)
    im = idx - m
    ip = idx + m
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N-1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N-1)], 0.0)
    return A - B
