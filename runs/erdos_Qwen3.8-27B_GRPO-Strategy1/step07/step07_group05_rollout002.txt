def continuous_refine(h, N, deadline, lr0=0.005):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    it = 0
    while time.time() < deadline - 5:
        it += 1
        corr = fft_corr(h)
        peak_val = np.max(corr)
        # Softmax weights over correlation values
        t = 0.05
        w = np.exp((corr - peak_val) / t)
        w /= w.sum()
        
        # Gradient of weighted sum of C(s)
        grad = np.zeros(N)
        center = N - 1
        for ki in range(len(corr)):
            if w[ki] < 1e-5:
                continue
            s = ki - center
            wt = w[ki]
            if s == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wt * (1.0 - h[:N-s])
                grad[:N-s] -= wt * h[s:]
            else:
                m = -s
                grad[:N-m] += wt * (1.0 - h[m:])
                grad[m:] -= wt * h[:N-m]
        
        # Update with projection
        h_new = np.clip(h - lr * grad, 0, 1)
        # Bisection for sum constraint
        lo, hi = -5.0, 5.0

            best_c = c_new

    return best_h, best_c
