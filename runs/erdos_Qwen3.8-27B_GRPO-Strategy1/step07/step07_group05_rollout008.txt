```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        n_move = n1 - target
        perm = np.random.permutation(len(idx))[:n_move]
        h[idx[perm]] = 0

        idx = np.where(h == 0)[0]

        w = np.zeros(len(corr))
        w[top_idx] = 1.0
        w /= w.sum()
        
        # Compute gradient

        center = N - 1
        for ki in top_idx:

            elif L > 0 and L < N:

                grad[:N-M] += (1.0 - h[M:])

        # Update

        # Project to sum = target

        # Simple projection: adjust uniformly
```

```python
def subgradient_step(h, N, lr, top_m=20):
    corr = fft_corr(h)  # length 2N-1
    peak = corr.max()
    # Softmax weights over top lags
    logits = (corr - peak) / 0.01  # temperature
    logits = np.clip(logits, -10, 0)
    w = np.exp(logits)
    w /= w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    # For each lag k, the gradient of corr[k] w.r.t. h is:
    # d/dh[i] of sum_j h[j]*(1-h[j+k]) 
    # = (1-h[i+k]) - h[i-k] (with boundary handling)
    # But this is O(N) per lag, and we have O(N) lags.
    # Instead, use the FFT structure.
    
    # Actually, let's compute it directly for the significant lags.
    significant = np.where(w > 0.001)[0]
    for ki in significant:
        L = int(ki - center)
        wt = w[ki]
        if L == 0:
            grad += wt * (1 - 2*h)
        elif 0 < L < N:
            # corr[k] = sum_{i=0}^{N-1-L} h[i] * (1-h[i+L])
            # d/dh[i] = (1-h[i+L]) for i < N-L, and -h[i-L] for i >= L
            g = np.zeros(N)
            g[:N-L] += (1 - h[L:])
            g[L:] -= h[:N-L]
            grad += wt * g
        elif -N < L < 0:
            M = -L
            g = np.zeros(N)
            g[M:] += (1 - h[:N-M])
            g[:N-M] -= h[M:]
            grad += wt * g
    
    return grad
```

```python
            zeros = np.where(h == 0)[0]
            if len(ones) < 3 or len(zeros) < 3:
                continue
            oi = np.random.choice(ones, 3, replace=False)
            zi = np.random.choice(zeros, 3, replace=False)
            h[oi] = 0; h[zi] = 1
            nc = fft_corr(h)
            nm = float(np.max(nc))
            dE = nm - cur_max
            if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE/T)):

                c = cur_max * 2.0 / N
                if c < best_c:

def continuous_opt(h, N, deadline, lr0=0.02):
    """Subgradient-based continuous optimization."""
    h = project_continuous(h, N)
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        corr = fft_corr(h)
        peak = float(np.max(corr))
        c = peak * 2.0 / N
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Subgradient
        temp = max(0.5, 0.5 * np.exp(-it / 5000.0))
        logits = (corr - peak) / temp
        logits = np.clip(logits, -20, 0)
        w = np.exp(logits)
        ws = w.sum()
        if ws < 1e-10:
            break
        w /= ws
        
        grad = np.zeros(N)
        center = N - 1
        sig = np.where(w > 0.005)[0]
        for ki in sig:
            L = int(ki - center)
            wt = w[ki]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[:N-L] += wt * (1.0 - h[L:])
                grad[L:] -= wt * h[:N-L]
            elif -N < L < 0:
                M = -L
                grad[M:] += wt * (1.0 - h[:N-M])
                grad[:N-M] -= wt * h[M:]
        
        # Step
        h = h - lr * grad
        h = project_continuous(h, N)
        
        if it % 2000 == 0:
            lr *= 0.95
    return best_h, best_c
```
