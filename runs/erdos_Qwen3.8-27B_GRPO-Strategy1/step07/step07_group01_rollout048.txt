    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, N, rng):
    """Make h binary with exactly N//2 ones."""
    target = N // 2
    h_bin = (h >= 0.5).astype(np.float64)
    n1 = int(h_bin.sum())
    if n1 == target:

    if n1 > target:

        # Prefer to remove ones with lowest h value
        order = np.argsort(h[ones_idx])
        to_remove = ones_idx[order[:n1-target]]
        h_bin[to_remove] = 0

        order = np.argsort(-h[zeros_idx])
        to_add = zeros_idx[order[:target-n1]]
        h_bin[to_add] = 1

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, N):
    target = N // 2
    h_bin = np.zeros(N)
    order = np.argsort(-h)
    h_bin[order[:target]] = 1.0
    return h_bin

def greedy_search(h, N, deadline, rng, max_iters=200000, target_frac=0.4):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = fft_corr(h)
    P = float(corr.max())
    best_P = P
    best_h = h.copy()
    it = 0
    stall = 0
    while time.time() < deadline and it < max_iters:
        s_star = int(np.argmax(corr))
        shift = s_star - (N - 1)
        if shift > 0:
            valid_i = np.arange(N - shift)
        elif shift < 0:
            valid_i = np.arange(-shift, N)
        else:
            valid_i = np.arange(N)
        waste = valid_i[(h[valid_i] == 1) & (h[np.clip(valid_i + shift, 0, N-1)] == 0)]
        if len(waste) == 0:
            waste = ones
        if rng.random() < target_frac and len(waste) > 0:
            p = waste[rng.randint(len(waste))]
        else:
            p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        h[p] = 0.0
        h[q] = 1.0
        corr = fft_corr(h)
        P_new = float(corr.max())
        if P_new < P:
            P = P_new
            if P < best_P:
                best_P = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            p_pos = np.where(ones == p)[0]
            if len(p_pos) > 0:
                ones = np.delete(ones, p_pos)
            ones = np.append(ones, q)
            q_pos = np.where(zeros == q)[0]
            if len(q_pos) > 0:
                zeros = np.delete(zeros, q_pos)
            zeros = np.append(zeros, p)
        else:
            h[p] = 1.0
            h[q] = 0.0
            stall += 1
            if stall > 3000:
                break
        it += 1
    return best_h, best_P * (2.0 / N)

    corr = fft_corr(h)

    best_P = P
    best_h = h.copy()
    if T_init is None:
        T_init = max(5.0, P * 0.15)
    T = T_init
    T_min = max(1.0, P * 0.005)
    it = 0
    while time.time() < deadline:
        if it % 1500 == 0 and it > 0:
            T = max(T_min, T * 0.998)
        use_2swap = (rng.random() < 0.25 and len(ones_list) >= 2 and len(zeros_list) >= 2)
        if use_2swap:
            p1 = ones_list[rng.randint(len(ones_list))]
            p2 = ones_list[rng.randint(len(ones_list))]
            if p1 == p2:
                it += 1
                continue
            q1 = zeros_list[rng.randint(len(zeros_list))]
            q2 = zeros_list[rng.randint(len(zeros_list))]
            if q1 == q2:
                it += 1

            h[p1] = 0; h[p2] = 0

            corr = fft_corr(h)

            delta = P_new - P

    # Phase 2: SA with multiple restarts
    n_sa = 4
    sa_time_each = min(120.0, (deadline - t0 - 80) / (n_sa + 2))
    for i in range(n_sa):

        # Perturb for diversity
        n_flip = rng.randint(5, 20)
        ones = np.where(h_start == 1)[0]
        zeros = np.where(h_start == 0)[0]
        for _ in range(n_flip):
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h_start[p], h_start[q] = 0, 1
        d_sa = min(time.time() + sa_time_each, deadline - 100)
        T_val = best_c5 * N0 / 2 * 0.1 * (0.5 + 0.3 * i)

        if c5_sa < best_c5:

        if c5_g2 < best_c5:
            best_c5, best_h = c5_g2, h_g2.copy()

    # Phase 4: Try smaller N (downsample)
    N_small = N0 // 2
    if N_small >= 300 and time.time() < deadline - 50:
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_small, endpoint=False)
        h_small = np.interp(x_new, x_old, best_h)
        h_small_bin = project_binary(h_small, N_small)
        d4 = min(time.time() + 40, deadline - 30)
        h_s, c5_s = sa_search(h_small_bin, N_small, d4, rng, T_init=0.8)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

            best_c5, best_h = c5_f, h_f.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
