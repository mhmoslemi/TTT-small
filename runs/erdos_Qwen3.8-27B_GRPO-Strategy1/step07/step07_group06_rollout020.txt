    target = N // 2
    assert int(h.sum()) == target
    corr = fft_corr(h)
    best_c = np.max(corr)
    best_h = h.copy()
    stall = 0
    temp = 0.0

        if time.time() > deadline:
            break
        if stall > 200:
            break
        P = np.max(corr)
        improved = False
        if P < best_c:
            best_c = P
            best_h = h.copy()
            improved = True
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

            h[i], h[j] = 0, 1

        if not improved:

            temp = min(0.01, temp + 0.0001)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, N*2)  # pad to power of 2 for speed
    G = np.fft.fft(np.flip(1.0 - h), N*2)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def to_binary(h, N):
    h = (h > 0.5).astype(np.float64)
    target = N // 2
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            one_idx = np.where(h == 1)[0]
            rem = np.random.choice(one_idx, n1 - target, replace=False)
            h[rem] = 0
        else:
            zero_idx = np.where(h == 0)[0]
            add = np.random.choice(zero_idx, target - n1, replace=False)
            h[add] = 1
    return h

def binary_optimize(h, N, deadline, max_stall=300, n_trials=20, temp_scale=0.003):
    """Local search on binary h with simulated annealing."""
    h = h.copy()
    target = N // 2
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_P = P
    best_h = h.copy()
    stall = 0
    it = 0
    temp = temp_scale
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline and stall < max_stall:
        it += 1
        P = float(np.max(corr))
        if P < best_P:
            best_P = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                temp *= 0.9
                if temp < 1e-5:
                    temp = 0.0
        
        best_delta = 0.0
        bi, bj = -1, -1
        best_new_corr = None
        
        for _ in range(n_trials):
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            h[i], h[j] = 0.0, 1.0
            nc = fft_corr(h)
            nP = float(np.max(nc))
            delta = P - nP
            if delta > best_delta or (temp > 0 and delta > -temp * 0.5 and np.random.random() < np.exp(delta / temp)):
                best_delta = delta
                bi, bj = i, j
                best_new_corr = nc
            h[i], h[j] = 1.0, 0.0
        
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            corr = best_new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        
    return best_h, best_P

```python
def fast_swap_eval(h, i, j, corr, N):
    """Evaluate the change in max corr after swapping h[i]:1->0, h[j]:0->1.
    Returns (new_max, new_corr) or just the delta info."""
    # We modify h in-place: h[i] goes 1->0, h[j] goes 0->1
    # But we haven't done it yet. We compute delta analytically.
    # delta for shift +k:
    #   -(1-h[i+k]) + h[i-k] + (1-h[j+k]) - h[j-k]
    # delta for shift -k:
    #   -(1-h[i-k]) + h[i+k] + (1-h[j-k]) - h[j+k]
    
    # Vectorized:
    dk = np.arange(1, N)
    
    # For +k shifts:
    # term1: 1-h[i+k] for i+k < N
    t1 = np.zeros(N)
    valid1 = i + dk < N
    t1[valid1] = 1.0 - h[i + dk[valid1]]
    # term2: h[i-k] for i-k >= 0
    valid2 = i - dk >= 0
    t2 = np.zeros(N)
    t2[valid2] = h[i - dk[valid2]]
    # term3: 1-h[j+k] for j+k < N
    t3 = np.zeros(N)
    valid3 = j + dk < N
    t3[valid3] = 1.0 - h[j + dk[valid3]]
    # term4: h[j-k] for j-k >= 0
    t4 = np.zeros(N)
    valid4 = j - dk >= 0
    t4[valid4] = h[j - dk[valid4]]
    
    delta_plus = -t1 + t2 + t3 - t4  # delta for corr[N-1+k], k=1..N-1
    delta_minus = -t4 + t3 + t2 - t1  # Wait, let me recompute
    
    # For -k shifts:
    #   -(1-h[i-k]) + h[i+k] + (1-h[j-k]) - h[j+k]
    # term1': 1-h[i-k] for i-k >= 0
    t1p = np.zeros(N)
    t1p[valid2] = 1.0 - h[i - dk[valid2]]
    # term2': h[i+k] for i+k < N
    t2p = np.zeros(N)
    t2p[valid1] = h[i + dk[valid1]]
    # term3': 1-h[j-k] for j-k >= 0
    t3p = np.zeros(N)
    t3p[valid4] = 1.0 - h[j - dk[valid4]]
    # term4': h[j+k] for j+k < N
    t4p = np.zeros(N)
    t4p[valid3] = h[j + dk[valid3]]
    
    delta_minus = -t1p + t2p + t3p - t4p  # delta for corr[N-1-k], k=1..N-1
    
    # Apply to corr
    new_corr = corr.copy()
    new_corr[N:] += delta_plus  # corr[N], corr[N+1], ..., corr[2N-2]
    new_corr[:N-1] += delta_minus[::-1]  # corr[N-2], corr[N-3], ..., corr[0]
    
    return new_corr, float(np.max(new_corr))
```
