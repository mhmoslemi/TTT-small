Let me try using a **smooth max** (log-sum-exp) instead of the hard max. This gives a differentiable approximation to the max:

smooth_max(x, τ) = τ * log(sum(exp(x_i/τ)))

For τ small, this approximates the max. The gradient is:
∂/∂x_i [τ * log(sum(exp(x_j/τ)))] = exp(x_i/τ) / sum(exp(x_j/τ))

This is a softmax! So the gradient of the smooth max w.r.t. the correlation at lag k is proportional to exp(corr[k]/τ) / sum(exp(corr[j]/τ)).

Then the gradient w.r.t. h[0..N-1] is: sum_k softmax_k * ∂corr[k]/∂h

This is a weighted average of the individual lag gradients, where the weights are the softmax probabilities. This is much smoother than just using the top-k lags.

    """Compute gradient of smooth max of corr w.r.t. h."""
    N = len(h)
    center = N - 1
    # Softmax weights for all lags (or just a subset for speed)
    # Use all lags for accuracy
    vals = corr - np.max(corr)  # for numerical stability
    weights = np.exp(vals / tau)
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for k in range(len(corr)):
        if weights[k] < 1e-8:
            continue
        L = abs(int(k) - center)
        if L == 0:
            grad += weights[k] * (1.0 - 2.0 * h)
        elif L < N:

            lo = np.arange(0, N - L)
            grad[hi] += weights[k] * (1.0 - h[hi - L])
            grad[lo] -= weights[k] * h[lo + L]
    return grad

import numpy as np

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

        hb, c2 = binary_fast_il(hb, d2, max_iters=8000, n_crit=5, n_sample=55)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    if time.time() < deadline - 280:
        d3 = min(deadline - 240, time.time() + 100)
        hsa = fix_binary(best_h)
        hsa, c3 = binary_sa(hsa, d3, max_iters=4000, T0=0.035, T1=0.0005)
        if c3 < best_c:
            best_c, best_h = c3, hsa.copy()
        if c3 < best_c or True:
            d3b = min(deadline - 220, time.time() + 60)
            hb3 = fix_binary(hsa)
            hb3, c3b = binary_fast_il(hb3, d3b, max_iters=4000, n_crit=5, n_sample=50)
            if c3b < best_c:
                best_c, best_h = c3b, hb3.copy()

    if time.time() < deadline - 200:
        d4 = min(deadline - 160, time.time() + 80)
        h4 = project(best_h + np.random.randn(N0) * 0.03, N0 / 2.0)
        h4, c4 = cont_smooth(h4, d4, tau0=0.8, lr0=0.008, mom=0.88)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    for N_try in [400, 440, 500]:
        if time.time() > deadline - 130:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d5 = min(deadline - 60, time.time() + 50)
        h_bt, c5v = binary_fast_il(h_bt, d5, max_iters=3000, n_crit=5, n_sample=45)
        if time.time() < deadline - 40:
            hc = project(h_bt + np.random.randn(N_try) * 0.03, N_try / 2.0)
            d6 = min(deadline - 30, time.time() + 25)
            hc, c6 = cont_smooth(hc, d6, tau0=1.0, lr0=0.01, mom=0.85)
            h6b = fix_binary(hc)
            c6b = compute_c5(h6b)
            if c6 < c5v:
                c5v, h_bt = c6, fix_binary(hc)
            elif c6b < c5v:
                c5v, h_bt = c6b, h6b
        if c5v < best_c:
            best_c, best_h = c5v, h_bt.copy()

    if time.time() < deadline - 5:
        hB = fix_binary(best_h)
        d7 = min(deadline - 2, time.time() + 40)
        hB, c7 = binary_fast_il(hB, d7, max_iters=3000, n_crit=4, n_sample=60)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()

    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
