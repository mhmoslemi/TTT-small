import numpy as np
import time

def full_corr(h):
    """Compute full cross-correlation of h with (1-h) using FFT."""
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * np.conj(G), L))

def get_c5(h, N):
    return float(np.max(full_corr(h))) * (2.0 / N)

def binarize(h, N):
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        rem = np.random.choice(idx, n1 - target, replace=False)
        h[rem] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        rem = np.random.choice(idx, target - n1, replace=False)
        h[rem] = 1
    return (h > 0.5).astype(float)

def overlap_at_lag(h, L):
    """Compute overlap at lag L (can be negative). N = len(h)."""
    N = len(h)
    if L >= 0:
        if L >= N:
            return 0
        return np.dot(h[:N-L], 1.0 - h[L:])
    else:
        M = -L
        if M >= N:
            return 0
        return np.dot(h[M:], 1.0 - h[:N-M])

def gradient_softmax(h, corr, alpha, k_top):
    """Compute gradient of soft-max objective."""
    N = len(h)
    # Get top k lags
    idx = np.argpartition(corr, -k_top)[-k_top:]
    # Softmax weights
    exp_c = alpha * corr[idx]
    exp_c -= exp_c.max()
    w = np.exp(exp_c)
    w /= w.sum()
    
    grad = np.zeros(N)
    for j, kk in enumerate(idx):
        L = int(kk) - (N - 1)  # physical lag
        M = abs(L)
        if M >= N:
            continue
        sgn = 1 if L >= 0 else -1
        # d(overlap)/dh[i] = (1-h[i+L])*[valid] - h[i-L]*[valid]
        if L >= 0:
            grad[:N-L] += w[j] * (1.0 - h[L:])
            grad[M:] -= w[j] * h[:N-M]
        else:
            grad[M:] += w[j] * (1.0 - h[:N-M])
            grad[:N-M] -= w[j] * h[M:]
    return grad

def softmax_descent(h0, N, deadline, alpha_start=15, k_top=30, lr=0.05):

    best_h = h.copy()

        c = float(corr.max())

            best_c = c
            best_h = h.copy()
        grad = gradient_softmax(h, corr, alpha_start, k_top)
        h_new = h - lr_cur * grad
        # Project: clip to [0,1], adjust sum to N/2
        h_new = np.clip(h_new, 0, 1)
        for _ in range(3):
            adj = (h_new.sum() - N/2.0) / N
            h_new = np.clip(h_new - adj, 0, 1)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr_cur *= 0.97
    return best_h, best_c * (2.0/N)

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2*N - 1))
    for i in range(N):
        deltas[i, i:i+N] = s
    return s, deltas

def targeted_sa(h_bin, deadline, n_peaks=8, n_per_peak=20, cool_rate=0.9993, t_start=0.035, max_iters=100000, use_blocks=True):
    N = len(h_bin)
    h = h_bin.copy()
    corr = full_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.001)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Pick candidate swaps
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp = vi[mask]
                bq = (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp = vi[mask]
                bq = (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel])
                cand_q.extend(bq[sel])
        
        # Random candidates
        nr = min(10, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False))
        cand_q.extend(np.random.choice(zeros, nr, replace=False))
        
        cand_p = np.array(sorted(set(np.array(cand_p, dtype=int))))
        cand_q = np.array(sorted(set(np.array(cand_q, dtype=int))))
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        
        # Evaluate candidates
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 8
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.005 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 1.0 - h[p], 1.0 - h[q]
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p+N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q+N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
            if no_impr > 300:
                # Big kick: flip several random pairs
                nk = min(5, len(ones)//2, len(zeros)//2)
                if nk > 0:
                    kp = np.random.choice(np.where(h==1)[0], nk, replace=False)
                    kq = np.random.choice(np.where(h==0)[0], nk, replace=False)
                    for a, b in zip(kp, kq):
                        h[a], h[b] = 1.0-h[a], 1.0-h[b]
                        corr += deltas[a].copy() - deltas[b].copy()
                        s[a], s[b] = -s[a], -s[b]
                        deltas[a, :] = 0.0; deltas[a, a:a+N] = s
                        deltas[b, :] = 0.0; deltas[b, b:b+N] = s
                    best_c = float(corr.max())
                    best_h = h.copy()
                    no_impr = 0
                    T = max(T * 1.5, 0.01)
        else:
            no_impr += 1
        
        if no_impr > 5000:
            break
    
    return best_h, best_c
