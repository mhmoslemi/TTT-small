def fft_corr(h):

    H = np.fft.rfft(h, L)

def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=50):
    h = h_in.astype(float).copy()
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.8
    it = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        pk = float(corr_y.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr_y, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.93, 0.78 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel])
                cand_q.extend(bq[sel])
        nr = min(10, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False))
        cand_q.extend(np.random.choice(zeros, nr, replace=False))
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        best_new = 1e10
        bi, bj = 0, 0
        chunk = 8
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            flip_pair(h, p, q, corr, s, deltas, N)
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        if no_impr > 12000:
            break
    return best_h, best_c

def tabu_search(h_bin, deadline, tabu_len=180, n_peaks=10, max_iters=60000):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    tabu = np.zeros(N, dtype=int)
    it = 0
    no_impr = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p, cand_q = [], []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp, bq = vi[mask], (vi - M)[mask]
            if len(bp) > 0:
                ns = min(20, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel])
                cand_q.extend(bq[sel])
        nr = min(12, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False))
        cand_q.extend(np.random.choice(zeros, nr, replace=False))
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        free_p = cand_p[tabu[cand_p] < it]
        free_q = cand_q[tabu[cand_q] < it]
        if len(free_p) < 3:
            free_p = cand_p
        if len(free_q) < 3:
            free_q = cand_q
        dp = deltas[free_p]
        dq = deltas[free_q]
        best_new = 1e10
        bi, bj = 0, 0
        chunk = 8
        for ci in range(0, len(free_p), chunk):
            ce = min(ci + chunk, len(free_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        p, q = int(free_p[bi]), int(free_q[bi]) if bi == bj else int(free_q[bj])
        p = int(free_p[bi])
        q = int(free_q[bj])
        flip_pair(h, p, q, corr, s, deltas, N)
        tabu[p] = it + tabu_len
        tabu[q] = it + tabu_len
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_impr = 0
        else:
            no_impr += 1
        if no_impr > 8000:
            break
    return best_h, best_c

def nesterov_polish(h_in, N, deadline, lr=0.008, k_top=60):
    h = h_in.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.80
    it = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        pk = float(corr_y.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr_y, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.93, 0.78 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Main SA with kick mechanism
    d1 = min(deadline - 350, t0 + 220)
    h1, _ = sa_main(h_bin, d1, n_peaks=6, n_per_peak=18, cool_rate=0.9991, t_start=0.05)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Tabu search
    if time.time() < deadline - 280:
        d2 = min(deadline - 220, time.time() + 140)
        h2, _ = tabu_search(best_h, d2, tabu_len=160, n_peaks=10)
        c2 = compute_c5(h2, N0)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Nesterov continuous polish
    if time.time() < deadline - 180:
        d3 = min(deadline - 130, time.time() + 90)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, c3 = nesterov_polish(hc, N0, d3, lr=0.008, k_top=60)
        if c3 < best_c5:
            best_c5, best_h = c3, hn.copy()

    # Phase 4: SA from perturbed start (escape)
    if time.time() < deadline - 120:
        h4 = make_binary(best_h, N0)
        n_flip = 8
        fidx = np.random.choice(N0, n_flip, replace=False)
        for idx in fidx:
            h4[idx] = 1.0 - h4[idx]
        h4 = make_binary(h4, N0)
        d4 = min(deadline - 80, time.time() + 80)
        h4, _ = sa_main(h4, d4, n_peaks=8, n_per_peak=22, cool_rate=0.9994, t_start=0.03)
        c4 = compute_c5(h4, N0)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Final SA (long, very low temp, many peaks)
    if time.time() < deadline - 5:
        h5 = make_binary(best_h, N0)
        d5 = deadline - 2
        h5, _ = sa_main(h5, d5, n_peaks=12, n_per_peak=35, cool_rate=0.9999, t_start=0.012, max_iters=300000)
        c5 = compute_c5(h5, N0)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
