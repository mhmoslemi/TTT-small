import numpy as np
import time

def fft_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_row(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def compute_C_mat(h, N):
    return np.vstack([compute_C_row(h, N, i) for i in range(N)])

def fix_sum(h, nh):
    n1 = int(np.sum(h == 1))
    if n1 > nh:
        ones = np.where(h == 1)[0]
        h[np.random.choice(ones, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        zeros = np.where(h == 0)[0]
        h[np.random.choice(zeros, nh - n1, replace=False)] = 1
    return h

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def local_search(h, N, deadline, sample_sz=None, kick_thresh=50, kick_sz=4, max_kicks=60):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    corr0 = fft_corr(h)
    best_c = float(np.max(corr0))
    best_h = h.copy()
    C = compute_C_mat(h, N)
    if sample_sz is None:
        sample_sz = min(N // 2 - 1, 200)
    stuck = 0
    kicks_used = 0
    while time.time() < deadline:
        P = float(np.max(corr0))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        new = corr0[None, None, :] + Co[None, :, :] + Cz[:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-14:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
            C[os_[mi[1]]] = compute_C_row(h, N, os_[mi[1]])
            C[zs[mi[0]]] = compute_C_row(h, N, zs[mi[0]])
            corr0 = fft_corr(h)
        elif stuck >= kick_thresh and kicks_used < max_kicks:
            for _ in range(kick_sz):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0; h[q] = 1
                C[p] = compute_C_row(h, N, p)
                C[q] = compute_C_row(h, N, q)
            corr0 = fft_corr(h)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stuck = 0
            kicks_used += 1
        else:

    return best_h, best_c * (2.0 / N)

def two_swap_search(h, N, deadline, sz=12, max_rounds=3000):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    corr0 = fft_corr(h)
    best_c = float(np.max(corr0))
    best_h = h.copy()
    C = compute_C_mat(h, N)
    L = 2 * N - 1
    stuck = 0
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        P = float(np.max(corr0))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 80:
                break
        ones = np.where(h == 1)[0]

        if len(ones) < 2 or len(zeros) < 2:

        ns = min(len(ones), sz)
        nz = min(len(zeros), sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        best_mv = P
        best_idx = None
        for i in range(ns):
            for j in range(i + 1, ns):
                Co2 = Co[i] + Co[j]
                for a in range(nz):

                        nv = np.max(corr0 + Co2 + Cz[a] + Cz[b])
                        if nv < best_mv - 1e-14:
                            best_mv = nv
                            best_idx = (i, j, a, b)

            i, j, a, b = best_idx
            h[os_[i]] = 0; h[os_[j]] = 0
            h[zs[a]] = 1; h[zs[b]] = 1
            C[os_[i]] = compute_C_row(h, N, os_[i])
            C[os_[j]] = compute_C_row(h, N, os_[j])
            C[zs[a]] = compute_C_row(h, N, zs[a])
            C[zs[b]] = compute_C_row(h, N, zs[b])
            corr0 = fft_corr(h)

def subgradient_opt(h_init, N, deadline, lr0=0.005, topk=50):
    h = h_init.astype(float).copy()
    fix_sum(h, N // 2)
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    beta = 0.85
    lr = lr0
    it = 0
    h_avg = h.copy()
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -topk)[-topk:]
        w = corr[top]
        w = w / max(w.sum(), 1e-12)
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            wt = float(w[np.searchsorted(top, ai)])
            if L == 0:
                grad += wt * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wt * (1.0 - y[:N - L])
                grad[:N - L] -= wt * y[L:]
            else:
                M = -L
                grad[:N - M] += wt * (1.0 - y[M:])
                grad[M:] -= wt * y[:N - M]
        h_new = project_01(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        h_avg += h
        it += 1
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.95, 0.82 + it * 0.0003)
            c_test = compute_c5((h_avg / it), N)
            c_bin = compute_c5((h_avg / it > 0.5).astype(float), N)
            if min(c_test, c_bin) < best_c:
                best_c = min(c_test, c_bin)
                best_h = np.clip(h_avg / it, 0, 1).copy()
    c1 = compute_c5(np.clip(h_avg / it, 0, 1), N)
    c2 = compute_c5((h_avg / it > 0.5).astype(float), N)
    if c1 < best_c:
        best_c, best_h = c1, (h_avg / it).copy()
    if c2 < best_c:
        best_c, best_h = c2, ((h_avg / it > 0.5).astype(float)).copy()
    return best_h, best_c

def try_resolution(N, h_ref, deadline):

    return h_ls, c5_ls
