def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)

    return np.real(np.fft.irfft(H * G, L))

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)

def fix_binary(h):

def compute_deltas(h, N):
    """Compute 1->0 and 0->1 delta vectors for all positions."""
    k = np.arange(-(N-1), N)  # (2N-1,)
    n_all = 2 * N - 1
    
    # For 1->0 at position p: delta[k] = h[p+k](valid) - (1-h[p-k])(valid)
    # For 0->1 at position p: delta[k] = -h[p+k](valid) + (1-h[p-k])(valid)

    P = np.arange(N)[:, None]  # (N, 1)
    K = k[None, :]  # (1, 2N-1)

    h2 = np.where(valid2, 1.0 - h[np.clip(idx2, 0, N-1)], 0.0)

    delta_0to1 = -h1 + h2  # (N, 2N-1)

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0].copy()
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0].copy()
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def compute_deltas(h, N):
    k = np.arange(-(N - 1), N)
    P = np.arange(N)[:, None]
    K = k[None, :]
    idx1 = P + K
    idx2 = P - K
    v1 = (idx1 >= 0) & (idx1 < N)
    v2 = (idx2 >= 0) & (idx2 < N)
    ci1 = np.clip(idx1, 0, N - 1)
    ci2 = np.clip(idx2, 0, N - 1)
    h1 = np.where(v1, h[ci1], 0.0)
    h2 = np.where(v2, 1.0 - h[ci2], 0.0)
    return h1 - h2, -h1 + h2

def binary_opt(h, deadline, top_k=100, sa_temp=0.003, sa_cool=0.9965, max_no_imp=400):
    N = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c = float(corr.max() * (2.0 / N))
    best_h = h.copy()
    temp = sa_temp
    no_imp = 0
    while time.time() < deadline and no_imp < max_no_imp:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        if n1 == 0 or n0 == 0:
            break
        D1, D0 = compute_deltas(h, N)
        d1 = D1[ones]
        d0 = D0[zeros]

        d1f = d1[:, tl]
        d0f = d0[:, tl]

        npk = nv.max(axis=2)
        fi = int(npk.argmin())

            zp = zeros[best_zi]
            h[op] = 0

            new_corr = compute_corr(h)
