            d = cur_max - nP
            if d > best_d:
                best_d, ba, bb = d, a, b
        if ba >= 0:
            accept = best_d > 0 or (temp > 1e-9 and np.random.random() < np.exp(max(best_d, -10) / temp))
            if accept:
                h[ba], h[bb] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if best_d > 0:
                    nm = cur_max - best_d
                    if nm < best_c - 1e-14:
                        best_c = nm
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        temp *= 0.9997
    return best_h, best_c * (2.0 / N)

def _block_move(h, N, deadline, max_iters=2000):
    h = (h > 0.5).astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        cur_max = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        best_d = -np.inf
        best_move = None
        for _ in range(20):
            blen = int(np.random.choice([2, 3, 4]))
            start = int(ones[np.random.randint(max(1, len(ones) - blen))])
            block = list(range(start, start + blen))
            if not all(h[i] == 1 for i in block):
                continue
            dest = int(zeros[np.random.randint(len(zeros))])
            new_block = list(range(dest, dest + blen))
            if any(h[i] == 1 for i in new_block) or new_block[-1] >= N or new_block[0] < 0:
                continue
            if any(i == j for i in block for j in new_block):
                continue
            for i in block:
                h[i] = 0.0
            for i in new_block:
                h[i] = 1.0
            nP = float(np.max(fft_corr(h)))
            for i in new_block:
                h[i] = 0.0
            for i in block:
                h[i] = 1.0
            d = cur_max - nP
            if d > best_d:
                best_d = d
                best_move = (block, new_block)
        if best_move and best_d > 0:
            for i in best_move[0]:
                h[i] = 0.0
            for i in best_move[1]:
                h[i] = 1.0
            nm = cur_max - best_d
            if nm < best_c - 1e-14:
                best_c = nm
                best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def _subgrad(h, N, deadline, alpha0=0.04, beta=0.93):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    alpha = alpha0
    for it in range(6000):
        if time.time() > deadline:
            break
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - 1e-5)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 300 == 0:
            alpha *= 0.95
            beta = min(0.98, 0.90 + 0.0001 * it)
    return best_h, best_pk * (2.0 / N)

def _multi_res(best_h, best_c5, deadline):
    x_b = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [300, 340, 420, 460, 520, 620, 760, 380]:
        if time.time() > deadline:
            break
        if Nt % 2:
            Nt += 1
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_b, best_h)
        ht = (hi > 0.5).astype(float)
        nh = Nt // 2
        n1 = int(ht.sum())
        if n1 > nh:
            ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0
        elif n1 < nh:
            ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1
        sd = min(deadline, time.time() + 30)
        hr, c5r = _binary_local(ht, Nt, sd, ntri=25, temp0=0.003)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_b = np.linspace(0, 2, Nt, endpoint=False)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 4.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Quick binary search
    d1 = min(dl - 400, t0 + 50)
    hb, c1 = _binary_local(best_h, N0, d1, ntri=25, temp0=0.004)
    if c1 < best_c5:
        best_c5, best_h = c1, hb

    # Phase 2: L-BFGS continuous with multiple T
    if time.time() < dl - 300:
        d2 = min(dl - 250, time.time() + 130)
        for T in [25.0, 40.0, 55.0]:
            if time.time() > d2:
                break
            hr = best_h + np.random.randn(N0) * 0.15
            hr = proj01(hr, N0 / 2.0)
            hl, cl = _lbfgs(hr, N0, min(d2, time.time() + 40), T, 3, 180)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 3: Aggressive binary + block moves
    if time.time() < dl - 200:
        d3a = min(dl - 160, time.time() + 120)
        hb3, c3 = _binary_local(best_h, N0, d3a, ntri=35, temp0=0.003)
        if c3 < best_c5:
            best_c5, best_h = c3, hb3
    if time.time() < dl - 100:
        d3b = min(dl - 80, time.time() + 50)
        hb4, c4 = _block_move(best_h, N0, d3b, max_iters=1500)
        if c4 < best_c5:
            best_c5, best_h = c4, hb4

    # Phase 4: Subgradient
    if time.time() < dl - 100:
        d4 = min(dl - 70, time.time() + 60)
        hs = best_h + np.random.randn(N0) * 0.06
        hs = proj01(hs, N0 / 2.0)
        hs7, cs7 = _subgrad(hs, N0, d4)
        if cs7 < best_c5:
            best_c5, best_h = cs7, hs7.copy()

    # Phase 5: Multi-resolution
    if time.time() < dl - 60:
        d5 = dl - 30
        best_h, best_c5 = _multi_res(best_h, best_c5, d5)

    # Phase 6: Final polish
    Nf = len(best_h)
    if time.time() < dl - 15:
        d6 = min(dl - 8, time.time() + 30)
        h6 = best_h + np.random.randn(Nf) * 0.05
        h6 = proj01(h6, Nf / 2.0)
        hl6, cl6 = _lbfgs(h6, Nf, d6, 35.0, 2, 150)
        if cl6 < best_c5:
            best_c5, best_h = cl6, hl6

    if time.time() < dl - 1.0:
        hb7, c7 = _binary_local(best_h, Nf, dl - 0.5, ntri=30, temp0=0.002)
        if c7 < best_c5:
            best_c5, best_h = c7, hb7

    # Final comparison: continuous vs binary
    fc = compute_c5(best_h, Nf)
    fb = (best_h > 0.5).astype(float)
    fc_b = compute_c5(fb, Nf)
    if fc_b < fc:
        best_h, fc = fb, fc_b
    return best_h, fc, Nf
