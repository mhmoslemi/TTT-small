import numpy as np
import time

def fft_corr(h):
    N = len(h)
    Lp = 1 << (2*N - 1).bit_length()
    H = np.fft.rfft(h, Lp)
    F = np.fft.rfft(1.0 - h, Lp)
    return np.fft.irfft(H * np.conj(F), Lp)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def full_local_search(h, deadline, max_rounds=200):
    """Complete single-swap local search"""
    N = len(h)
    h = h.copy()
    best_c = np.max(fft_corr(h))
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_d = 0
        best_pair = None
        # Try all single swaps
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > best_d:
                    best_d = d
                    best_pair = (i, j)
                h[i] = 1
                h[j] = 0
        if best_pair is None or best_d <= 1e-12:
            break
        h[best_pair[0]] = 0
        h[best_pair[1]] = 1
    return h, best_c * (2.0 / N)

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    Lp = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, Lp)
    F = np.fft.rfft(1.0 - h, Lp)
    return np.fft.irfft(H * np.conj(F), Lp)

def c5_of(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(s, h, N):
    """Gradient of corr at shift s w.r.t. h"""
    g = np.zeros(N)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += 1.0 - h[:N - s]
        g[:N - s] -= h[s:N]
    else:
        M = -s
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj_simplex(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def smooth_fn(h, T, target):
    """Smoothed max objective + value and gradient"""
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    Z = e.sum()
    w = e / Z
    val = mx + np.log(Z) / T
    g = np.zeros(N)
    ctr = N - 1
    for k in range(len(c)):
        if w[k] > 1e-10:
            g += w[k] * lin_grad(k - ctr, h, N)
    # Constraint penalty
    s = h.sum() - target
    val += 0.5 * s * s
    g += s * np.ones(N)
    return val, g

def lbfgs_opt(h, deadline, T=30.0, max_iter=200):
    """L-BFGS on smoothed objective"""
    N = len(h)
    target = float(N) // 2
    h = h.copy()
    res = sp_min(smooth_fn, h, args=(T, target), jac=True,
                 method='L-BFGS-B', bounds=[(0, 1)] * N,
                 options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    h = proj_simplex(res.x, target)
    return h, c5_of(h)

def full_ls(h, deadline, max_rounds=30):
    """Complete binary single-swap local search"""
    N = len(h)
    h = h.copy()
    b = c5_of(h)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        P = float(np.max(fft_corr(h)))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bi, bj, bd = -1, -1, 0.0
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                nP = float(np.max(fft_corr(h)))
                if P - nP > bd:
                    bd, bi, bj = P - nP, int(i), int(j)
                h[j] = 0.0
            h[i] = 1.0
        if bi < 0:
            break
        h[bi] = 0.0
        h[bj] = 1.0
        nc = c5_of(h)
        if nc < b:
            b = nc
    return h, b

def subgrad_opt(h, deadline, n_iter=3000, alpha0=0.05):
    """Nesterov subgradient for continuous h"""
    N = len(h)
    target = float(N) // 2
    h = h.copy()
    m = np.zeros(N)
    best_h = h.copy()
    best_c = c5_of(h)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        alpha = alpha0 / np.sqrt(it + 1)
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = corr.max()
        active = np.where(corr >= pk - 1e-4)[0]
        g = np.zeros(N)
        ctr = N - 1
        for k in active:
            g += lin_grad(k - ctr, y, N)
        g /= len(active)
        # Add constraint gradient
        s = y.sum() - target
        g += s
        h_new = proj_simplex(y - alpha * g, target)
        m = 0.92 * (h_new - h)
        h = h_new
        c = c5_of(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def to_bin(h, N):
    """Convert to binary with N/2 ones"""
    nb = (h > 0.5).astype(float)
    t = N // 2
    s = int(nb.sum())
    if s > t:
        o = np.where(nb == 1)[0]
        np.random.shuffle(o)
        nb[o[t:]] = 0
    elif s < t:
        z = np.where(nb == 0)[0]
        np.random.shuffle(z)
        nb[z[:t - s]] = 1
    return nb
