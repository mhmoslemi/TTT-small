import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, target):
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    n1 = len(ones)
    if n1 > target:
        np.random.shuffle(ones)
        h[ones[:n1-target]] = 0.0
    elif n1 < target:
        np.random.shuffle(zeros)
        h[zeros[:target-n1]] = 1.0
    return h

def struct_init(N, variant, seed):
    rng = np.random.RandomState(seed)
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[rng.choice(N, nh, replace=False)] = 1.0
    elif variant == 1:
        h = (np.arange(N) % 3 < 2).astype(float)  # 2/3 duty
        # Adjust to exactly N/2
        ones = np.where(h == 1)[0]
        if len(ones) > nh:
            np.random.shuffle(ones)
            h[ones[nh:]] = 0.0
        elif len(ones) < nh:
            zeros = np.where(h == 0)[0]
            np.random.shuffle(zeros)

        thresh = np.median(v)

            new_max = float(np.max(new_corr))
            dE = new_max - cur_max

    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, target):
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    n1 = len(ones)
    if n1 > target:
        idx = np.random.permutation(n1)[:n1 - target]
        h[ones[idx]] = 0.0
    elif n1 < target:
        idx = np.random.permutation(len(zeros))[:target - n1]
        h[zeros[idx]] = 1.0
    return h

def struct_init(N, variant, seed):
    rng = np.random.RandomState(seed)
    nh = N // 2
    x = np.arange(N)

             + 0.5*np.cos(2*np.pi*11*x/N+1.0) + 0.3*np.sin(2*np.pi*17*x/N+2.0))
        h = (v > np.median(v)).astype(float)
        h = fix_binary(h, nh)
    elif variant == 2:
        phi = (1.0 + np.sqrt(5.0)) / 2.0

def cont_refine(h, N, deadline, lr0=0.004):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    temp = 0.012
    it = 0
    while time.time() < deadline - 1:
        it += 1
        corr = fft_corr(h)
        peak = np.max(corr)
        w = np.exp((corr - peak) / temp)
        w /= w.sum()
        grad = np.zeros(N)
        for ki in range(N - 1, 2 * N - 1):
            wt = w[ki]
            if wt < 0.003:
                continue
            L = ki - (N - 1)
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[L:] += wt * (1.0 - h[:N-L])
                grad[:N-L] -= wt * h[L:]
            elif -N < L < 0:
                M = -L
                grad[:N-M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N-M]
        h_new = np.clip(h - lr * grad, 0, 1)
        lo, hi = -5.0, 5.0
        for _ in range(38):
            mid = (lo + hi) / 2.0
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2.0, 0, 1)
        c_new = compute_c5(h_new, N)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 120 == 0:
            lr *= 0.993
            temp *= 0.93
    return best_h, best_c
