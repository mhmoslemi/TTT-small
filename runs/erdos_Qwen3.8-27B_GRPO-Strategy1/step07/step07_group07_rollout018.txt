import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def soft_max_grad(h, corr, temp=0.01):
    """Gradient of soft-max of corr w.r.t. h, scaled by 2/N"""
    N = len(h)
    center = N - 1
    # Soft max weights
    mc = np.max(corr)
    w = np.exp((corr - mc) / temp)
    w /= w.sum()
    # Gradient of corr[k] w.r.t. h[i]
    # corr[k] = sum_j h[j] * (1-h[j+k-center]) for valid j
    # d/dh[i] corr[k] = (1-h[i+k-center]) - h[i-k+center] (when in range)
    grad = np.zeros(N)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-10:
            continue
        k = m - center
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif k > 0:
            # corr[m] = sum_{j=0}^{N-1-k} h[j] * (1-h[j+k])
            # d/dh[i] = (1-h[i+k]) if i < N-k  (from first factor)
            #           - h[i-k]    if i >= k   (from second factor)
            hi_idx = np.arange(0, N - k)
            lo_idx = np.arange(k, N)
            grad[hi_idx] += wm * (1.0 - h[hi_idx + k])
            grad[lo_idx] -= wm * h[lo_idx - k]
        else:
            k = -k
            # corr[m] = sum_{j=k}^{N-1} h[j] * (1-h[j-k])
            hi_idx = np.arange(k, N)
            lo_idx = np.arange(0, N - k)
            grad[hi_idx] += wm * (1.0 - h[hi_idx - k])
            grad[lo_idx] -= wm * h[lo_idx + k]
    return grad

    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0

    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                temp_cur *= 1.05  # anneal: make soft-max sharper
        
        # Soft-max gradient
        mc = np.max(corr)
        w = np.exp((corr - mc) / temp_cur)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        
        grad = np.zeros(N)
        center = N - 1
        # Vectorized gradient computation
        active = w > 1e-8
        active_idx = np.where(active)[0]
        for m in active_idx:

            k = int(m - center)
            if k == 0:
                grad += wm * (1.0 - 2.0 * h)
            elif k > 0:

        v = 0.85 * v + 0.15 * (h_new - h)

            best_c, best_h = c2, hb.copy()

    if time.time() < deadline - 250:
        d3 = min(deadline - 250, time.time() + 90)
        hf = freq_smooth(best_h, 0.65)
        h3 = project(hf + np.random.randn(N0) * 0.04, N0 // 2)
        h3, c3 = cont_smax(h3, d3, temp_init=1.5, lr0=0.008, mom=0.92)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    if time.time() < deadline - 180:
        d4 = min(deadline - 180, time.time() + 100)
        hb4 = fix_binary(best_h)
        hb4, c4 = binary_il(hb4, d4, top_k=60, sample_sz=80, restart_every=180)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()

    for N_try in [570, 760]:
        if time.time() > deadline - 90:
            break
        h_int = interp_h(best_h, N0, N_try)
        h_int = project(h_int, N_try // 2)
        d5 = min(deadline - 70, time.time() + 45)
        h5, c5v = cont_smax(h_int, d5, temp_init=3.0, lr0=0.01, mom=0.9)
        if time.time() < deadline - 35:
            hb5 = fix_binary(h5)
            d6 = min(deadline - 25, time.time() + 22)
            hb5, c6 = binary_il(hb5, d6, top_k=50, sample_sz=60, restart_every=150)
            if c6 < c5v:
                c5v = c6
                h5 = hb5
        if c5v < best_c:
            best_c, best_h = c5v, h5.copy()

    if time.time() < deadline - 8:
        hB = fix_binary(best_h)
        d7 = deadline - 2
        hB, c7 = binary_il(hB, d7, top_k=65, sample_sz=85, restart_every=120)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()

    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    elif best_c < final_c:
        final_c = best_c

    Nf = len(best_h)
    if abs(best_h.sum() - Nf / 2.0) > 1e-6:
        best_h = project(best_h, Nf / 2.0)
        final_c = compute_c5(best_h)
    return best_h, final_c, Nf
