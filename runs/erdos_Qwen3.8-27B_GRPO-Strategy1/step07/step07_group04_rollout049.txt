import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.fft.irfft(np.fft.rfft(h, L) * np.conj(np.fft.rfft(1.0 - h, L)))

def build_Dmat(h):
    N = len(h)
    v = 2.0 * h - 1.0  # shape (N,), values in {-1, 1} for binary
    D = np.zeros((N, 2 * N - 1))
    idx = np.arange(N)
    D[idx[:, None], idx[:, None] + np.arange(N)[None, :]] = v[None, :]
    return D

def greedy_local_search(h, deadline, max_iters=50000):
    N = len(h)
    h = h.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = fft_corr(h)
    D_mat = build_Dmat(h)
    P = float(corr.max())
    best_h = h.copy()
    best_P = P
    
    it = 0
    while time.time() < deadline:
        it += 1
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Find best swap
        Dz = D_mat[zeros]  # shape (len(zeros), 2N-1)
        best_new_P = P
        best_pq = None
        
        for p in ones:
            base = corr + D_mat[p]
            new_maxes = (base[None, :] - Dz).max(axis=1)
            mi = int(new_maxes.argmin())
            if new_maxes[mi] < best_new_P - 1e-10:
                best_new_P = float(new_maxes[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            break
        
        p, q = best_pq
        # Apply swap
        h[p] = 0.0
        h[q] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Update corr
        corr = corr + D_mat[p] - D_mat[q]
        
        # Update D_mat
        i_idx = np.arange(N)
        D_mat[i_idx, i_idx + q] += 2.0
        D_mat[i_idx, i_idx + p] -= 2.0
        
        P = best_new_P
        if P < best_P:
            best_P = P
            best_h = h.copy()
        
        if it % 500 == 0:
            # Verify
            corr_check = fft_corr(h)
            if abs(corr_check.max() - P) > 1e-6:
                corr = corr_check
                P = float(corr.max())
    
    return best_h, best_P * (2.0 / N)

import numpy as np
import time
import random

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.fft.irfft(np.fft.rfft(h, L) * np.conj(np.fft.rfft(1.0 - h, L)))

def build_Dmat(h):
    N = len(h)
    v = 2.0 * h - 1.0
    D = np.zeros((N, 2 * N - 1), dtype=np.float64)
    idx = np.arange(N)
    D[idx[:, None], idx[:, None] + np.arange(N)[None, :]] = v[None, :]
    return D

def upsample_fn(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def fix_sum_binary(h, N):
    h = h.copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        oi = np.where(h == 1)[0]
        sel = np.random.choice(oi, n1 - target, replace=False)
        h[sel] = 0.0
    elif n1 < target:
        zi = np.where(h == 0)[0]
        sel = np.random.choice(zi, target - n1, replace=False)
        h[sel] = 1.0
    return h

def greedy_ls(h, deadline, max_iters=80000, verify_every=2000):

    h = h.astype(float).copy()
    corr = fft_corr(h)
    D_mat = build_Dmat(h)
    P = float(corr.max())
    best_h = h.copy()
    best_P = P
    it = 0
    stall = 0

    while time.time() < deadline:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        Dz = D_mat[zeros]
        best_new_P = P - 1e-14
        best_pq = None
        n_zero = len(zeros)

        for p in ones:
            base = corr + D_mat[p]
            new_maxes = (base[None, :] - Dz).max(axis=1)
            mi = int(new_maxes.argmin())
            if new_maxes[mi] < best_new_P:
                best_new_P = float(new_maxes[mi])
                best_pq = (int(p), int(zeros[mi]))

        if best_pq is None:
            break

        p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        corr = corr + D_mat[p] - D_mat[q]
        i_idx = np.arange(N)
        D_mat[i_idx, i_idx + p] -= 2.0
        D_mat[i_idx, i_idx + q] += 2.0
        P = best_new_P

        if P < best_P - 1e-12:
            best_P = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        if it % verify_every == 0:
            c_check = fft_corr(h)
            p_check = float(c_check.max())
            if abs(p_check - P) > 0.01:
                corr = c_check
                D_mat = build_Dmat(h)
                P = p_check

    return best_h, best_P * (2.0 / N)

def ils_search(h_init, deadline, max_kick=12):
    N = len(h_init)
    best_h = h_init.copy()
    best_c5 = float(np.max(fft_corr(h_init))) * (2.0 / N)

        else:
            no_improve += 1
