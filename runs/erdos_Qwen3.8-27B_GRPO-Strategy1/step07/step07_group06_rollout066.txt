import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = max(1, 2 * N - 1)
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def c5_val(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:N-L] += 1.0 - h[L:N]
        g[L:] -= h[:N-L]
    else:
        M = -L
        g[M:] += 1.0 - h[:N-M]
        g[:N-M] -= h[M:]
    return g

def soft_obj(h, T, target, lam):
    c = fft_corr(h)
    cm = c.max()
    e = np.exp(T * (c - cm))
    val = cm + np.log(e.sum()) / T
    w = e / e.sum()

    ctr = len(h) - 1
    nw = np.where(w > 1e-10)[0]
    for i in nw:
        g += w[i] * lin_grad(i - ctr, h, len(h))
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g = g + lam * s
    return val, g

            hl, cl = do_lbfgs(hr, N0, d2, T, 2)
            if cl < best_c5:
                best_c5, best_h = cl, hl

    if time.time() < dl - 500:
        d3 = min(dl - 450, time.time() + 55)
        hs = best_h + np.random.randn(N0) * 0.06
        hs = proj01(hs, N0 / 2.0)
        hsn, cn = do_subgrad(hs, N0, d3, 10, 0.022)
        if cn < best_c5:
            best_c5, best_h = cn, hsn

    if time.time() < dl - 320:
        d4 = min(dl - 280, time.time() + 140)
        hb4, cb4 = do_bin(best_h, N0, d4, 16, 0.014, 300)
        cb45 = cb4 * (2.0 / N0)
        if cb45 < best_c5:
            best_c5, best_h = cb45, hb4

    if time.time() < dl - 180:
        best_h, best_c5 = try_res(best_h, best_c5, min(dl - 140, time.time() + 250))
        N0 = len(best_h)

    if time.time() < dl - 60:
        d6 = min(dl - 30, time.time() + 55)
        for T in [60, 100]:
            if time.time() > d6:
                break
            h6 = best_h + np.random.randn(N0) * 0.04
            h6 = proj01(h6, N0 / 2.0)
            hl6, cl6 = do_lbfgs(h6, N0, d6, T, 2)
            if cl6 < best_c5:
                best_c5, best_h = cl6, hl6

    if time.time() < dl - 25:
        d7 = min(dl - 8, time.time() + 25)
        h7 = best_h + np.random.randn(N0) * 0.02
        h7 = proj01(h7, N0 / 2.0)
        hs7, cs7 = do_subgrad(h7, N0, d7, 14, 0.012)
        if cs7 < best_c5:
            best_c5, best_h = cs7, hs7

    if time.time() < dl - 1.5:
        hf = (best_h > 0.5).astype(float)
        hf, cf = do_bin(hf, N0, dl - 0.5, 15, 0.01, 100)
        cf5 = cf * (2.0 / N0)
        if cf5 < best_c5:
            best_c5, best_h = cf5, hf

    fc5 = c5(best_h, N0)
    fb = (best_h > 0.5).astype(float)
    fb5 = c5(fb, N0)
    if fb5 < fc5:
        best_h, fc5 = fb, fb5
    return best_h, fc5, N0
