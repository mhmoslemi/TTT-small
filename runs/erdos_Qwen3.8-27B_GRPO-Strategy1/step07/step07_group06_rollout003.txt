    G = np.fft.fft(np.flip(1.0 - h), L)

def lin_grad(L, h, N):

    if L == 0:

    return best_h, best_v * (2.0 / N)

def grad_bin_search(h, N, deadline, max_iters=500, ntrial=25):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        corr = fft_corr(h)
        mk = float(np.max(corr))
        improved = mk < best_c - 1e-14
        if improved:
            best_c = mk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        T = 40.0
        e = np.exp(np.clip(T * (corr - mk), -30, 30))
        w = e / e.sum()
        ctr = N - 1
        grad = np.zeros(N)
        for i in range(len(corr)):
            if w[i] > 1e-7:
                grad += w[i] * lin_grad(i - ctr, h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Gradient-guided: prefer removing 1s where grad is high (contributes to max)
        # and adding 1s where grad is low
        rem_score = -grad[ones] + 1.0
        add_score = grad[zeros] + 1.0
        rem_score = np.clip(rem_score, 0.1, None)
        add_score = np.clip(add_score, 0.1, None)
        p_rem = rem_score / rem_score.sum()
        p_add = add_score / add_score.sum()
        bd, bi, bj = -1e9, -1, -1
        for _ in range(ntrial):
            i = np.random.choice(len(ones), p=p_rem)
            j = np.random.choice(len(zeros), p=p_add)
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = mk - nP
            h[oi], h[zi] = 1.0, 0.0
            if d > bd:
                bd, bi, bj = d, oi, zi
        # Also try a few random swaps
        for _ in range(3):
            i = int(np.random.randint(len(ones)))
            j = int(np.random.randint(len(zeros)))
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = mk - nP
            h[oi], h[zi] = 1.0, 0.0
            if d > bd:
                bd, bi, bj = d, oi, zi
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            stall = 0
    return best_h, best_c

    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < 5000:

        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()

        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            alpha *= 0.97
    return best_h, best_pk * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

            g += w[i] * lin_grad(i - ctr, h, N)
