def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)

def compute_c5(h, N):

    return float(np.max(corr)) * (2.0 / N)

def safe_get(h, idx, N):
    """Get h[idx] with OOB returning 0"""
    if 0 <= idx < N:
        return h[idx]
    return 0.0

def greedy_step(h, corr, N, top_m=5):

    center = N - 1
    # Get top m lags

    hot = sorted_idx[:top_m]

    zeros = np.where(h == 0)[0]
    if len(ones) < 1 or len(zeros) < 1:
        return None

    best_new_max = float('inf')

    candidates = set()
    for idx in hot:

        scores_a = np.array([safe_get(h, a-k, N) + safe_get(h, a+k, N) for a in ones])
        # Score b's: maximize h[b-k] + h[b+k]

                candidates.add((int(ones[ai]), int(zeros[bi])))

    for a, b in candidates:
        if a == b:
            continue
        # Compute delta for all lags
        # Δ_c[lag] = h[a-lag] + h[a+lag] - h[b-lag] - h[b+lag]

        delta = np.zeros(2*N-1)
        for pos in lags:

        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        if new_max < best_new_max:
            best_new_max = new_max
            best_pair = (a, b, new_corr)
    
    if best_pair is not None and best_new_max < float(np.max(corr)) - 1e-12:

    return None

import numpy as np
import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(get_corr(h))) * (2.0 / N)

def compute_delta(h, a, b, N):
    lags = np.arange(-(N-1), N)
    ia_m = a - lags; ia_p = a + lags
    ib_m = b - lags; ib_p = b + lags
    h_am = np.where((ia_m>=0)&(ia_m<N), h[ia_m], 0.0)
    h_ap = np.where((ia_p>=0)&(ia_p<N), h[ia_p], 0.0)
    h_bm = np.where((ib_m>=0)&(ib_m<N), h[ib_m], 0.0)
    h_bp = np.where((ib_p>=0)&(ib_p<N), h[ib_p], 0.0)
    return h_am + h_ap - h_bm - h_bp

def score_pos(pos, k, h, N):
    pm = pos - k; pp = pos + k
    sm = (pm>=0)&(pm<N); sp = (pp>=0)&(pp<N)
    return np.where(sm, h[pm], 0.0) + np.where(sp, h[pp], 0.0)

def fix_sum(h, target):
    h = h.copy()
    ones = np.where(h==1)[0]
    if len(ones) > target:
        h[np.random.choice(ones, len(ones)-target, replace=False)] = 0
    elif len(ones) < target:
        zeros = np.where(h==0)[0]
        h[np.random.choice(zeros, target-len(ones), replace=False)] = 1
    return h

def guided_search(h, N, deadline, max_iters=200000, T0=0.06):

    corr = get_corr(h)
    best_c = float(np.max(corr))*2.0/N
    best_h = h.copy()
    T = T0
    cool = (1e-8/T)**(1.0/max(1,50000))
    no_imp = 0
    center = N-1
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        cur_max = float(np.max(corr))
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)<1 or len(zeros)<1:
            break
        accepted = False
        r = np.random.random()
        if r < 0.75:
            sorted_idx = np.argsort(-corr)
            hot = [int(idx-center) for idx in sorted_idx[:4]]
            best_nm = cur_max; best_pair = None; best_nc = None

                sa = score_pos(ones, k, h, N)
                sb = score_pos(zeros, k, h, N)

            if best_pair is not None:
