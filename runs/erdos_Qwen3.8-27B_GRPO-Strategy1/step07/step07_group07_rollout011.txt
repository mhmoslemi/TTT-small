        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Continuous optimization with corrected gradient
    d1 = min(deadline - 750, t0 + 130)
    h1 = project(h0 + np.random.randn(N0) * 0.05, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, beta=5.0, lr0=0.018, mom=0.90, frac=0.8)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Binary SA
    if time.time() < deadline - 550:
        d2 = min(deadline - 520, time.time() + 200)
        hb = fix_binary(best_h)
        hb, c2 = binary_sa(hb, d2, max_iters=8000, T0=0.002, Tdec=0.9985, n_crit=25)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    # Phase 3: Fine continuous refinement
    if time.time() < deadline - 350:
        d3 = min(deadline - 320, time.time() + 120)
        h3 = project(best_h + np.random.randn(N0) * 0.025, N0 / 2.0)
        h3, c3 = cont_opt(h3, d3, beta=8.0, lr0=0.008, mom=0.92, frac=0.9)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Binary SA round 2 (cooler)
    if time.time() < deadline - 200:
        d4 = min(deadline - 170, time.time() + 150)
        hb4 = fix_binary(best_h)
        hb4, c4 = binary_sa(hb4, d4, max_iters=6000, T0=0.0008, Tdec=0.9992, n_crit=25)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()

    # Phase 5: Try alternative N values
    for N_try in [400, 360, 440]:
        if time.time() > deadline - 150:
            break
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d5 = min(deadline - 70, time.time() + 70)
        h_bt, c5v = binary_sa(h_bt, d5, max_iters=4000, T0=0.0015, Tdec=0.998, n_crit=25)
        if c5v < best_c:
            best_c, best_h = c5v, h_bt.copy()

    # Phase 6: Final polish - continuous then binary
    if time.time() < deadline - 50:
        d6 = min(deadline - 30, time.time() + 30)
        h6 = project(best_h + np.random.randn(len(best_h)) * 0.02, len(best_h) / 2.0)
        h6, c6 = cont_opt(h6, d6, beta=10.0, lr0=0.005, mom=0.93, frac=0.92)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    if time.time() < deadline - 10:
        hB = fix_binary(best_h)
        d7 = min(deadline - 3, time.time() + 40)
        hB, c7 = binary_sa(hB, d7, max_iters=3000, T0=0.0005, Tdec=0.9995, n_crit=30)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()

    # Final check
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
