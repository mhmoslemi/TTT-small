```python
def fft_corr(h):

    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_onto_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    rng = np.random.RandomState(42)

    else:
        zeros = np.where(h_bin == 0)[0]

    #   -(1-h[p+d]) for p+d valid (losing the h term)
    #   + h[p-d] for p-d valid (gaining the 1-h term, since 1-h[p] goes from 0 to 1)
    
    # In the c array indexing: c[m] = sum_i h[i]*(1-h[i+m-N+1])
    # d = m - N + 1
    # For position p in "h": contributes to c[m] where p+m-N+1 is valid
    #   i.e., 0 <= p+m-N+1 < N, so 1-N-p <= m-N+1 < N-p, so p <= m < N+p
    #   d ranges from 0 to N-1-p
    # For position p in "1-h" (as (1-h[p])): contributes to c[m] where i+m-N+1=p
    #   i = p-m+N-1, valid when 0 <= p-m+N-1 < N, so p-N+1 <= m-N+1 < p+1
    #   d ranges from -(N-1-p) to 0
    #   Contribution: h[i] = h[p-d] where d = m-N+1, so i = p-d
    
    # OK let me just compute it directly for each p:
    D = np.zeros((N, 2*N-1))
    for p in range(N):
        # Remove h=1 contribution: for d = 0, 1, ..., N-1-p
        # c index = N-1+d
        for d in range(N - p):
            cidx = N - 1 + d
            D[p, cidx] -= (1.0 - h[p + d])
        # Add 1-h contribution: for d = 0, 1, ..., N-1-p
        # c index = N-1-d
        for d in range(1, N - p):
            cidx = N - 1 - d
            D[p, cidx] += h[p - d]
        # Also d=0 for the 1-h part: cidx = N-1, contribution h[p-0] = h[p] = 1
        # But wait, at d=0, both effects apply:
        # Remove: -(1-h[p]) = -(1-1) = 0
        # Add: +h[p] = 1
        # So D[p, N-1] += 1 ... but we already subtracted (1-h[p])=0 for d=0
        # And we need to add h[p] = 1 for the 1-h part at d=0
        D[p, N-1] += h[p]  # = 1
    return D
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def ensure_binary(h, N, seed=42):
    target = N // 2
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == target:
        return hb
    rng = np.random.RandomState(seed + N)
    if n1 > target:
        idx = rng.choice(np.where(hb==1)[0], n1-target, replace=False)
        hb[idx] = 0
    else:
        idx = rng.choice(np.where(hb==0)[0], target-n1, replace=False)
        hb[idx] = 1
    return hb

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0-hm)*vm + hp*vp

def greedy_best_swap(h, N, c, D, ones, zeros, sf=1.0):
    """Find best single swap. Returns (p, q) or None."""
    P = float(c.max())
    Dz = D[zeros, :]
    n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones)*sf))
    p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
    best_np = P
    best_pq = None
    for p in p_list:
        base = c - D[p, :]
        cand = base[None, :] - Dz
        npk = cand.max(axis=1)
        mi = int(np.argmin(npk))
        if npk[mi] < best_np:
            best_np = float(npk[mi])
            best_pq = (int(p), int(zeros[mi]))
    return best_pq

def greedy_search(h, N, deadline, max_sweeps=5000, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        pq = greedy_best_swap(h, N, c, D, ones, zeros, sf)
        if pq is None:
            break
        h[pq[0]] = 0.0
        h[pq[1]] = 1.0
    return best_h, best_c * (2.0/N)

def sa_search(h, N, deadline, max_sweeps=10000, T0=3.0, Tcool=0.995, T_min=0.05, sf=1.0):

def continuous_opt(h, N, deadline, k_top=30, lr0=0.03):
    h = h.astype(float).copy()
    target = N/2.0
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    lr = lr0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max()) * (2.0/N)
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
        # Top-k softmax
        idx = np.argpartition(c, -k_top)[-k_top:]
        vals = c[idx]
        T = 20.0 + 40.0 * min(1.0, t/300)
        exps = np.exp(T*(vals - vals.max()))
        w = exps / exps.sum()
        # Gradient
        grad = np.zeros(N)
        center = N - 1
        for ki in range(k_top):
            ai = idx[ki]
            d = int(ai - center)
            wi = w[ki]
            if d == 0:
                grad += wi * (1.0 - 2.0*h)
            elif d > 0:
                grad[d:] += wi * (1.0 - h[:N-d])
                grad[:N-d] -= wi * h[d:]
            else:
                M = -d
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad*grad
        mhat = m/(1-beta1**t)
        vhat = v/(1-beta2**t)
        h_new = h - lr * mhat/(np.sqrt(vhat)+eps)
        h = project_box_sum(h_new, target)
        if t % 200 == 0:
            lr *= 0.97
    return best_h, best_c
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def ensure_binary(h, N, seed=42):
    target = N // 2
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == target:
        return hb
    rng = np.random.RandomState(seed + N)
    if n1 > target:
        idx = rng.choice(np.where(hb==1)[0], n1-target, replace=False)
        hb[idx] = 0
    else:
        idx = rng.choice(np.where(hb==0)[0], target-n1, replace=False)
        hb[idx] = 1
    return hb

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0-hm)*vm + hp*vp

def greedy_search(h, N, deadline, max_sweeps=5000, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones)*sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_np = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_np:
                best_np = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None or best_np >= P - 1e-14:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0/N)

def sa_search(h, N, deadline, max_sweeps=20000, T0=3.5, Tcool=0.996, T_min=0.03, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:

        if stall > 120:
            for _ in range(4):

                    h[q[np.random.randint(len(q))]] = 1.0
            stall = 0
            T = max(T0*0.7, T)
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:

        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones)*sf))

        best_np = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_np:
                best_np = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_np - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.,delta)/max(T,1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0/N)

def continuous_opt(h, N, deadline, k_top=35, lr0=0.025, T_start=15.0, T_end=80.0, restart_every=400):
    h = h.astype(float).copy()
    target = N/2.0
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    lr = lr0
    no_improve = 0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max()) * (2.0/N)
        if P < best_c - 1e-11:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > restart_every:
            h = best_h + np.random.randn(N)*0.02
            h = project_box_sum(h, target)
            m = np.zeros(N)
            v = np.zeros(N)
            lr = lr0 * 0.7
            t = 0
            no_improve = 0
            continue
        idx = np.argpartition(c, -k_top)[-k_top:]
        vals = c[idx]
        T = T_start + (T_end - T_start) * min(1.0, t/400.0)
        exps = np.exp(T*(vals - vals.max()))
        w = exps / exps.sum()
        grad = np.zeros(N)
        center = N - 1
        for ki in range(len(idx)):
            d = int(idx[ki] - center)
            wi = w[ki]
            if d == 0:
                grad += wi * (1.0 - 2.0*h)
            elif d > 0:
                grad[d:] += wi * (1.0 - h[:N-d])
                grad[:N-d] -= wi * h[d:]
            else:
                M = -d
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad*grad
        mhat = m/(1-beta1**t)
        vhat = v/(1-beta2**t)
        h = project_box_sum(h - lr*mhat/(np.sqrt(vhat)+eps), target)
        if t % 300 == 0:
            lr *= 0.96
    return best_h, best_c
```
