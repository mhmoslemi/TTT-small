from scipy.optimize import minimize

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def compute_C_mat(h, N):
    C = np.zeros((N, 2*N-1))
    for i in range(N):
        C[i] = compute_C_single(h, N, i)
    return C

def fix_sum(h, nh):
    s = int(np.round(h.sum()))

        nf = s - nh
        if 0 < nf <= len(idx):
            h[np.random.choice(idx, nf, replace=False)] = 0

        idx = np.where(h == 0)[0]
        nf = nh - s
        if 0 < nf <= len(idx):
            h[np.random.choice(idx, nf, replace=False)] = 1
    return h

def nesterov_opt(h, N, deadline, k=80, lr0=0.012):

            best_h = h.copy()
