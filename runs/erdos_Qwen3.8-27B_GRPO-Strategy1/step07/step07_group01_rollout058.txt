    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return hm * vm.astype(float) - (1.0 - hp) * vp.astype(float)

def binary_ls(h_init, N, deadline, max_sweeps=6000, topk=3):
    h = h_init.astype(float).copy()
    dx = 2.0 / N
    shifts = np.arange(2 * N - 1) - (N - 1)
    c = fft_corr(h)
    best_max = float(c.max())
    best_h = h.copy()
    stall = 0
    T_anneal = 1.5
    cg = np.random.RandomState(7)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        cur_max = float(c.max())
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        if n1 < 1 or n0 < 1:
            break
        D = make_D(h, N, shifts)
        D_o = D[ones, :]
        D_z = D[zeros, :]
        neg_Dz = -D_z
        best_nm = cur_max + 1e9
        best_pq = None
        best_t3 = float(np.sort(c)[-topk:].sum())
        for pi in range(n1):
            v = c + D_o[pi, :]
            cm = (v[None, :] + neg_Dz).max(axis=1)
            mi = int(np.argmin(cm))
            nm = float(cm[mi])
            if nm < best_nm - 1e-12:
                nc = v - D_z[mi]
                t3 = float(np.sort(nc)[-topk:].sum())
                best_nm = nm
                best_pq = (int(ones[pi]), int(zeros[mi]))
                best_t3 = t3
            elif abs(nm - best_nm) < 1e-12 and nm < cur_max:
                nc = v - D_z[mi]
                t3 = float(np.sort(nc)[-topk:].sum())
                if t3 < best_t3:
                    best_pq = (int(ones[pi]), int(zeros[mi]))
                    best_t3 = t3
        if best_pq is not None and best_nm < cur_max - 1e-12:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            if stall > 15 and T_anneal > 0.05:
                accepted = False
                cur_t3 = float(np.sort(c)[-topk:].sum())
                n_try = min(30, n1 * 2)
                for _ in range(n_try):
                    pi = int(cg.randint(n1))
                    qi = int(cg.randint(n0))
                    pr, qr = int(ones[pi]), int(zeros[qi])
                    nc = c + D[pr] - D[qr]
                    nm = float(nc.max())
                    t3 = float(np.sort(nc)[-topk:].sum())
                    if nm <= cur_max + T_anneal and t3 < cur_t3 - 0.5:
                        h[pr] = 0.0
                        h[qr] = 1.0
                        accepted = True
                        break
                if not accepted:
                    T_anneal *= 0.98
            if stall > 50:
                ns = min(4, n1, n0)
                for _ in range(ns):
                    pi = int(cg.randint(len(ones)))
                    qi = int(cg.randint(len(zeros)))
                    h[int(ones[pi])] = 0.0
                    h[int(zeros[qi])] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    n1, n0 = len(ones), len(zeros)
                stall = 0
                T_anneal = 1.5
    return best_h, best_max * dx

def nesterov_step(h_init, N, deadline, k=40, T=80.0, lr0=0.018):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        center = N - 1
        grad = np.zeros(N)
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.995
            beta = min(0.985, 0.82 + it * 0.0003)
        if it % 300 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(20.0, min(400.0, T * (0.9 if gap < 0.01 else 1.08)))
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = to_binary(h0, N0, seed=seed)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Intensive binary local search
    d1 = min(deadline - 380, t0 + 420)
    h1, c1 = binary_ls(best_h, N0, d1, max_sweeps=8000, topk=3)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Continuous Nesterov refinement
    if time.time() < deadline - 300:
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 260, time.time() + 100)
        h2, c2 = nesterov_step(hc, N0, d2, k=35, T=100.0, lr0=0.015)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Binary LS from best (or continuous result)
    if time.time() < deadline - 200:
        hb = to_binary(best_h, N0, seed=seed + 1)
        d3 = min(deadline - 160, time.time() + 200)
        h3, c3 = binary_ls(hb, N0, d3, max_sweeps=8000, topk=3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Second continuous pass
    if time.time() < deadline - 120:
        hc4 = best_h + np.random.randn(N0) * 0.03
        hc4 = project_sum(hc4, N0 / 2.0)
        d4 = min(deadline - 80, time.time() + 60)
        h4, c4 = nesterov_step(hc4, N0, d4, k=50, T=150.0, lr0=0.012)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Final binary polish
    if time.time() < deadline - 20:
        hb5 = to_binary(best_h, N0, seed=seed + 2)
        d5 = min(deadline - 3, time.time() + 50)
        h5, c5 = binary_ls(hb5, N0, d5, max_sweeps=5000, topk=2)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Final: ensure sum constraint and correct C5
    best_h = project_sum(best_h, N0 / 2.0)
    final_c5 = compute_c5(best_h, N0)
    if final_c5 > best_c5:
        best_h = to_binary(best_h, N0, seed=seed)
        final_c5 = best_c5
    return best_h, final_c5, N0
