    idx = np.arange(N)[:, None]  # (N, 1)
    k_vals = (np.arange(2*N-1) - (N-1))[None, :]  # (1, 2N-1)
    i_mk = idx - k_vals  # (N, 2N-1)

    B = np.where(valid_pk, h[h_idk], 0.0)

In the continuous case: minimize max_k corr_k subject to 0≤h≤1, sum(h)=N/2.

This is a linear programming problem! The variables are h[0],...,h[N-1] and the constraint is that each corr[k] ≤ C for all k, minimize C.

corr[k] = Σ_i h[i+k](1-h[i]) = Σ_i h[i+k] - h[i+k]h[i]

The h[i+k]h[i] term makes it non-linear (bilinear). So it's NOT a linear program.

However, for binary h, h[i+k]h[i] is 1 iff both are 1. So corr[k] = (number of i where h[i+k]=1) - (number of i where both h[i+k]=1 and h[i]=1) = |P shifted| - |P ∩ (P-k)|.

This is still not linear in the position variables.

But in the continuous case, if we relax h to [0,1], the problem becomes:
minimize max_k (Σ_i h[i+k] - h[i]h[i+k])
subject to 0≤h≤1, Σh = N/2

The objective is not convex (due to the product term). But we can use subgradient methods.

The subgradient of corr[k] w.r.t. h[j]:
- h[j] appears in h[i+k] for i=j-k: contribution (1-h[j-k])
- h[j] appears in h[i] for i=j: contribution -(h[j+k]) [from -h[i]h[i+k] = -h[j]h[j+k]]
- h[j] appears in h[i] for the linear part Σh[i+k]: no, wait.
  corr[k] = Σ_i h[i+k] - Σ_i h[i]h[i+k]
  d/dh[j]: appears in first sum when i+k=j, i.e., i=j-k: contribution 1
  appears in second sum when i=j: contribution -h[j+k]
  appears in second sum when i+k=j, i=j-k: contribution -h[j-k]

def continuous_polish(h, N, deadline, k_top=30, lr0=0.008):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.92
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        y_new = y - lr * grad
        hn = np.clip(y_new, 0, 1)
        s = hn.sum()

        it += 1
