import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D_mat(h, N):
    shifts = np.arange(-(N - 1), N)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    return -np.where(vm, 1.0 - h[pm_c], 0.0) + np.where(vp, h[pp_c], 0.0)

def greedy(h, N, deadline, max_sweeps=2000, sf=1.0, stall_limit=60):
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    best_h = h.copy()
    best_P = P
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D_mat(h, N)
        Dz = D[zeros, :]
        if sf < 1.0:
            n_s = max(15, int(len(ones) * sf))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            cand = (c + D[p])[None, :] - Dz
            peaks = cand.max(axis=1)
            mi = int(np.argmin(peaks))
            if peaks[mi] < best_new_P - 1e-14:
                best_new_P = float(peaks[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:

        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)

    return best_h, best_P * (2.0 / N)

def upsample(h, old_n, new_n):
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

    h = h.copy()
    n_ones = N // 2
    n1 = int(np.round(h.sum()))
    h_bin = (h >= 0.5).astype(float)

        perm = rng.choice(idx, n1 - n_ones, replace=False)

        idx = np.flatnonzero(h_bin == 0)
        perm = rng.choice(idx, n_ones - n1, replace=False)
