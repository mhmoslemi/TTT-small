def swap_delta_f(h, p, q, f, N):
    """
    Compute change in f(k) for all k when h[p]: 1->0 and h[q]: 0->1.
    Returns delta_f array of length N.
    """
    # h[p] changes by -1 (was 1, now 0)
    # Contribution of p to f(k):
    #   As h[i+k] (the "1" position): k in [1, p], contribution = (1-h[p-k])
    #   As h[i] (the "0" position): k in [1, N-1-p], contribution = h[p+k] * (1-h[p]) = 0 since h[p]=1
    # Wait, if h[p]=1, the (1-h[p]) term is 0, so p only matters as the "1" in h[i+k]
    # After flip, h[p]=0, so p contributes as the "0" in (1-h[i]): h[p+k] for k in [1, N-1-p]
    
    # Old contribution of p to f(k) (h[p]=1):
    #   k in [1, p]: (1 - h[p-k])
    #   k in [1, N-1-p]: 0 (since (1-h[p])=0)
    
    # New contribution of p to f(k) (h[p]=0):
    #   k in [1, p]: 0 (since h[p]=0)
    #   k in [1, N-1-p]: h[p+k]
    
    # Delta from p:
    #   k in [1, min(p, N-1-p)]: -(1-h[p-k]) + h[p+k]
    #   k in [min(p,N-1-p)+1, p]: -(1-h[p-k])  [if p > N-1-p]
    #   k in [min(p,N-1-p)+1, N-1-p]: h[p+k]  [if N-1-p > p]
    
    # Similarly for q (h[q]: 0->1):
    # Old contribution (h[q]=0): k in [1, N-1-q]: h[q+k]
    # New contribution (h[q]=1): k in [1, q]: (1-h[q-k])
    # Delta from q:
    #   k in [1, min(q, N-1-q)]: (1-h[q-k]) - h[q+k]
    #   k in [min(q,N-1-q)+1, q]: (1-h[q-k])  [if q > N-1-q]
    #   k in [min(q,N-1-q)+1, N-1-q]: -h[q+k]  [if N-1-q > q]
    
    delta = np.zeros(N)
    # From p (1->0):
    # Remove old contribution: -(1-h[p-k]) for k in 1..p
    # Add new contribution: h[p+k] for k in 1..(N-1-p)
    k_range_p_left = min(p, N-1)  # k goes from 1 to p
    if p >= 1:
        ks = np.arange(1, p+1)
        delta[ks] -= (1.0 - h[p - ks])
    k_range_p_right = min(N-1-p, N-1)
    if N-1-p >= 1:
        ks = np.arange(1, N-p)

def compute_f(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[N-1:].copy()

def swap_delta(h, p, q, N):
    d = np.zeros(N)
    if p >= 1:
        d[1:p+1] -= 1.0 - h[p-1::-1]
    if N-1-p >= 1:
        d[1:N-p] += h[p+1:N]
    if N-1-q >= 1:
        d[1:N-q] -= h[q+1:N]
    if q >= 1:
        d[1:q+1] += 1.0 - h[q-1::-1]
    return d

    h = h.astype(np.float64)
    nh = N // 2
    order = np.argsort(-h)
    h = np.zeros(N)
    h[order[:nh]] = 1.0
    return h
