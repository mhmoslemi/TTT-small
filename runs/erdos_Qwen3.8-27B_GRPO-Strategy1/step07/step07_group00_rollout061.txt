import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def smooth_max_gd(h_in, N, deadline, beta0=30.0, beta_max=300.0):
    h = h_in.astype(float).copy()
    hx = h.copy()
    corr = fft_corr(hx)
    s, deltas = build_deltas(hx)
    # deltas shape: (N, 2N-1), deltas[j, k] = contribution of position j to corr at offset k
    # grad[j] = sum_k w[k] * deltas[j, k] where w = softmax(beta * corr) / beta
    
    best_c5 = float(corr.max()) * (2.0 / N)
    best_h = h.copy()
    beta = beta0
    lr = 0.02
    m = np.zeros(N)
    beta_seq = 0.85
    it = 0
    
    while time.time() < deadline:
        it += 1
        y = h + m
        
        # Compute smooth gradient
        corr = fft_corr(y)
        cc = corr - corr.max()  # shift for numerical stability
        w = np.exp(beta * cc) / beta  # shape (2N-1,)
        # grad = deltas^T @ w  (shape N,)
        grad = deltas.T @ w  # (N, 2N-1)^T @ (2N-1,) -> (N,)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad = grad / gnorm

        h_new = y - lr * grad
        # Project: clip and fix sum
        h_new = np.clip(h_new, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        
        m = beta_seq * (h_new - h)
        h = h_new
        
        # Update best
        corr = fft_corr(h)
        c5 = float(corr.max()) * (2.0 / N)
        if c5 < best_c5:

            s, deltas = build_deltas(h)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(s, N):
    deltas = np.zeros((N, 2 * N - 1))

def compute_grad_topk(h, corr, top_lags, N):
    """Compute gradient of top-k correlation terms."""
    grad = np.zeros(N)
    for t in top_lags:
        d = int(t) - (N - 1)
        if d == 0:
            # corr[0_lag] = sum h[i]*(1-h[i]) = 0 for binary, skip
            continue
        if d > 0:
            # corr = sum_{i=0}^{N-1-d} h[i]*(1-h[i+d])
            # d/dh[j] = (1-h[j+d]) for j in [0, N-1-d]

            idx1 = np.arange(0, N - d)
            grad[idx1] += (1.0 - h[idx1 + d])
            idx2 = np.arange(d, N)
            grad[idx2] -= h[idx2 - d]
        else:

            grad[idx1] += (1.0 - h[idx1 - M])
            idx2 = np.arange(0, N - M)
            grad[idx2] -= h[idx2 + M]
    return grad / len(top_lags)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(s, N):
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return deltas

def sa_focused(h_bin, deadline, n_peaks=2, n_per_peak=18, cool_rate=0.9993, t_start=0.035):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s = 2.0 * h - 1.0
    deltas = build_deltas(s, N)
    T = t_start
    it = 0
    no_impr = 0
    while time.time() < deadline and it < 200000:
        it += 1
        if it % 600 == 0:
            T = max(T * cool_rate, 0.001)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p, cand_q = [], []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off == 0:
                continue
            if k_off > 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp = vi[mask]; bq = (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp = vi[mask]; bq = (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = max(2, len(ones) // 40)
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), 10):

            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)

                bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0

            s[p], s[q] = -s[p], -s[q]

        else:
            no_impr += 1
        if no_impr > 6000:

    return best_h, best_c

def compute_grad_lag(h, d, N):
    grad = np.zeros(N)
    if d == 0:
        return grad
    if d > 0:
        idx1 = np.arange(0, N - d)
        grad[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, N)
        grad[idx2] -= h[idx2 - d]
    else:
        M = -d
        idx1 = np.arange(M, N)
        grad[idx1] += (1.0 - h[idx1 - M])
        idx2 = np.arange(0, N - M)
        grad[idx2] -= h[idx2 + M]
    return grad

def smooth_gd(h_in, N, deadline, n_top=3, lr_init=0.01, beta_init=40.0, beta_rate=1.008):
    h = h_in.astype(float).copy()
    lr = lr_init
    beta = beta_init
    m = np.zeros(N)
    beta_mom = 0.8
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        y = h + m
        corr = fft_corr(y)
        peak_idx = np.argsort(-corr)[:n_top]
        peak_idx = [p for p in peak_idx if int(p) != N - 1]
        if len(peak_idx) == 0:
            break
        grad = np.zeros(N)
        for p in peak_idx:
            d = int(p) - (N - 1)
            grad += compute_grad_lag(y, d, N)
        grad /= len(peak_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            grad /= gnorm
        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta_mom * (h_new - h)
        h = h_new
        c5 = compute_c5(h, N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 400 == 0:
            lr *= 0.975
            beta *= beta_rate
            beta_mom = min(0.95, beta_mom + 0.003)
        if lr < 0.0005:
            break
    return best_h, best_c5

def micro_adjust(h_in, N, deadline):
    h = h_in.astype(float).copy()
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    lr = 0.002
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        corr = fft_corr(h)
        pk = int(np.argmax(corr))
        d = pk - (N - 1)
        if d == 0:
            break
        grad = compute_grad_lag(h, d, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad /= gnorm
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        c5_new = compute_c5(h_new, N)
        if c5_new < best_c5 - 1e-7:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        elif c5_new < best_c5 + 1e-6:
            h = h_new
        else:
            lr *= 0.9
        if lr < 1e-6:
            break
    return best_h, best_c5
