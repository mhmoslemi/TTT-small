import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pnorm_obj_grad(h, N, p, target, lam):
    c = fft_corr(h)
    ac = np.abs(c)
    cp_sum = np.sum(ac ** p)
    val_base = cp_sum ** (1.0 / p)
    s = h.sum() - target
    val = val_base
    if lam > 0:
        val += 0.5 * lam * s * s
    g = np.zeros(N)
    if cp_sum > 1e-15:
        scale = cp_sum ** (-1.0 / p)
        ctr = N - 1
        for i in range(len(c)):
            w_i = ac[i] ** (p - 1)
            if w_i > 1e-10:
                g += scale * w_i * lin_grad(i - ctr, h, N)
    if lam > 0:

            break
        try:

                         jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                         options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-15})
            h = proj01(res.x, target)
        except Exception:
            break
        v = float(np.max(fft_corr(h)))
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.0
    return best_h, best_v * (2.0 / N)

    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    alpha = alpha0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = proj01(h + m, target)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:

            best_h = h.copy()
        idx = np.argsort(corr)[::-1][:topk]

        m = beta * (h_new - h)
        h = h_new
        if it % 600 == 0:

    return best_h, best_pk * (2.0 / N)

        hf, _ = bin_search_sys(hf, len(best_h), dl - 1.0, max_iters=200, win=4, nrand=30)
        cf = compute_c5(hf, len(best_h))
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Final: compare continuous vs binary
    fc5 = compute_c5(best_h, len(best_h))
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, len(best_h))
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, len(best_h)
