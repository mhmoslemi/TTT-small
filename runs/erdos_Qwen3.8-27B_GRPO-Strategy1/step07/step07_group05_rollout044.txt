    N_corr = len(corr)
    center = N - 1
    
    while time.time() < deadline and it < 200000:
        it += 1
        r = np.random.random()
        
        if r < 0.4:
            # Single swap guided by peak
            peak_lags = np.where(corr > cur_max * 0.92)[0]
            if len(peak_lags) > 0:
                pk = peak_lags[np.random.randint(len(peak_lags))]
            else:
                pk = np.random.randint(N_corr)
            L = int(pk - center)
            
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            
            if L >= 0 and L < N:
                # h[i]=1, h[i+L]=0 contributes
                valid = []
                for i in range(N - L):
                    if h[i] == 1 and h[i+L] == 0:
                        valid.append(i)
                if valid:
                    i = valid[np.random.randint(len(valid))]
                    j = zeros[np.random.randint(len(zeros))]
                else:
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
            elif L < 0 and -L < N:
                M = -L
                valid = []
                for i in range(N - M):
                    if h[i] == 0 and h[i+M] == 1:
                        valid.append(i)
                if valid:
                    j = valid[np.random.randint(len(valid))]
                    i = ones[np.random.randint(len(ones))]
                else:
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
            else:
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
            
            h[i] = 0
            h[j] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
                cur_max = new_max
                corr = new_corr
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 1
                h[j] = 0
                
        elif r < 0.7:
            # Block swap: swap a block of 1s with a block of 0s
            L_block = np.random.choice([1, 2, 3, 4, 5, 8])
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < L_block or len(zeros) < L_block:
                continue
            # Find a contiguous block of 1s of length >= L_block
            one_positions = []
            i = 0
            while i < N:
                if h[i] == 1:
                    j = i
                    while j < N and h[j] == 1:
                        j += 1
                    if j - i >= L_block:
                        one_positions.append(i)
                    i = j
                else:
                    i += 1
            zero_positions = []
            i = 0
            while i < N:
                if h[i] == 0:
                    j = i
                    while j < N and h[j] == 0:
                        j += 1
                    if j - i >= L_block:
                        zero_positions.append(i)
                    i = j
                else:
                    i += 1
            
            if not one_positions or not zero_positions:
                continue
            ob = one_positions[np.random.randint(len(one_positions))]
            zb = zero_positions[np.random.randint(len(zero_positions))]
            if abs(ob - zb) < L_block:
                continue
            h[ob:ob+L_block] = 0
            h[zb:zb+L_block] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
                cur_max = new_max
                corr = new_corr
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[ob:ob+L_block] = 1
                h[zb:zb+L_block] = 0
        else:
            # 4-point move: swap two 1s with two 0s
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = np.random.choice(ones, 2, replace=False)
            j1, j2 = np.random.choice(zeros, 2, replace=False)
            h[i1] = h[i2] = 0
            h[j1] = h[j2] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
                cur_max = new_max
                corr = new_corr
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i1] = h[i2] = 1
                h[j1] = h[j2] = 0
        
        T *= 0.99998
        if no_improve > 3000:
            stall_reset += 1
            # Perturb
            n_pert = min(5, stall_reset * 2)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(n_pert):
                if len(ones) < 1 or len(zeros) < 1:
                    break
                oi = ones[np.random.randint(len(ones))]
                zi = zeros[np.random.randint(len(zeros))]
                h[oi], h[zi] = h[zi], h[oi]
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, 0.02)
            no_improve = 0
            if stall_reset > 10:
                break
    
    return best_h, best_c
