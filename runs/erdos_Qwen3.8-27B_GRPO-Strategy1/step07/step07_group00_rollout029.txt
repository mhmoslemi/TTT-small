def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))

    return float(np.max(fft_corr(h))) * (2.0 / N)

    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]

OK, I'm getting bogged down in sign conventions. Let me take a step back and use a cleaner approach.

The key insight is: I should just compute the effect of a flip by:
1. Precomputing corr via FFT
2. For a proposed flip of position i: the change in corr is (1-2*h[i]) * (1-h[partner]) where partner depends on the lag.

But actually, the cleanest approach for correctness is:
- Precompute the "influence vector" for each position i: infl[i] = the vector of changes to corr[k] when h[i] is toggled.
- This can be computed once and updated incrementally.

Let me use a different, cleaner formulation:

For `np.correlate(h, v, 'full')` where v = 1-h:
result[k] = sum_n h[n] * v[n + k - (Nv - 1)] for valid indices
With Nv = N: result[k] = sum_n h[n] * v[n + k - N + 1]

So result[k] = sum_n h[n] * (1 - h[n + k - N + 1])

For a given position i, it appears in result[k] when n=i, so:
contribution of h[i] to result[k] = h[i] * (1 - h[i + k - N + 1])
valid when 0 <= i + k - N + 1 <= N-1, i.e., N-1-i <= k <= 2N-2-i

If we flip h[i] (0→1 or 1→0), the change in result[k] for valid k is:
(1 - 2*h[i]) * (1 - h[i + k - N + 1])

So the influence vector infl[i] has:
infl[i][k] = (1-2*h[i]) * (1-h[i+k-N+1]) for N-1-i <= k <= 2N-2-i, else 0

When h[i]=1: infl[i][k] = -(1-h[...])  (negative)
When h[i]=0: infl[i][k] = +(1-h[...])  (positive)

After flipping h[i]: new_result = result + infl[i]
And infl[i] becomes -infl[i] (since (1-2*(1-h[i])) = -(1-2*h[i]))

For two flips (p: 1→0, q: 0→1):
new_result ≈ result + infl[p] + infl[q] (ignoring the small interaction where p is a partner of q)

The interaction: if i+k-N+1 = q for some k, then flipping q changes the partner value seen from p's perspective. But this is at most one term, so for a heuristic it's fine to ignore.

OK so let me just use this convention:
- infl[i][k] = (1-2*h[i]) * (1-h[i+k-N+1]) for valid k
- Flipping i: corr_new = corr + infl[i]
- infl[i] flips sign after the flip

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        np.random.shuffle(ones)
        h[ones[:n1 - target]] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - n1]] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_influence(h):
    """infl[i][k] = change in corr[k] when h[i] is toggled."""
    N = len(h)
    L = 2 * N - 1
    s = 1.0 - 2.0 * h  # +1 if h=0, -1 if h=1
    infl = np.zeros((N, L))
    for i in range(N):
        k_lo = max(0, N - 1 - i)
        k_hi = min(L - 1, 2 * N - 2 - i)
        if k_lo <= k_hi:
            partners = np.arange(k_lo, k_hi + 1) - i + N - 1
            infl[i, k_lo:k_hi+1] = s[i] * (1.0 - h[partners])
    return infl

def greedy_local_search(h, deadline, max_rounds=20):
    """Greedy: repeatedly find best swap and apply."""
    N = len(h)
    L = 2 * N - 1
    h = h.copy()
    corr = fft_corr(h)
    infl = build_influence(h)
    best_c = float(corr.max())
    best_h = h.copy()
    rounds = 0
    
    while time.time() < deadline and rounds < max_rounds:
        rounds += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Evaluate all pairs (chunked to avoid memory issues)
        best_delta = 0
        best_p, best_q = -1, -1
        chunk = 32
        for ci in range(0, len(ones), chunk):
            ce = min(ci + chunk, len(ones))
            # For each p in ones[ci:ce], for each q in zeros:
            # new_max = (corr + infl[p] + infl[q]).max()
            for cj in range(0, len(zeros), 64):
                ce2 = min(cj + 64, len(zeros))
                # Shape: (ce-ci, ce2-cj, L)
                vals = corr[None, None, :] + infl[ones[ci:ce]][:, None, :] + infl[zeros[cj:ce2]][None, :, :]
                mx = vals.max(axis=2)
                idx = np.unravel_index(mx.argmin(), mx.shape)
                if mx[idx] < best_delta:
                    best_delta = float(mx[idx])
                    best_p = int(ones[ci + idx[0]])
                    best_q = int(zeros[cj + idx[1]])
        
        if best_p < 0:
            break  # no improving move
        
        # Apply swap
        h[best_p], h[best_q] = 0.0, 1.0
        corr = corr + infl[best_p] + infl[best_q]
        infl[best_p] = -infl[best_p]
        infl[best_q] = -infl[best_q]
        c = float(corr.max())
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

    N = len(h)
    L = 2 * N - 1
    h = h.copy()
    corr = fft_corr(h)
    infl = build_influence(h)
    best_c = float(corr.max())
    best_h = h.copy()
    T = T_start
    no_impr = 0
    it = 0
    
    while time.time() < deadline and no_impr < 8000:
        it += 1
        if it % 600 == 0:
            T = max(T * cool, 0.001)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Generate candidates biased toward peak lags
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cands_p, cands_q = [], []
        for pk in peak_lags:
            k = int(pk)
            # For lag k, the "bad" pairs are h[i]=1 and h[i+k-N+1]=0
            k_shift = k - (N - 1)

            ce = min(ci + chunk, len(cands_p))
            vals = corr[None, None, :] + infl[cands_p[ci:ce]][:, None, :] + infl[cands_q][None, :, :]
            mx = vals.max(axis=2)

        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-max(dc, 0) / T)):
            h[p], h[q] = 0.0, 1.0
            corr = corr + infl[p] + infl[q]
            infl[p] = -infl[p]
            infl[q] = -infl[q]
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:

        else:
            no_impr += 1
    
    return best_h, best_c

def subgrad_polish(h_in, N, deadline, lr=0.008, n_top=30):
    """Sub-gradient descent for continuous relaxation."""
    h = h_in.astype(float).copy()
    m = np.zeros(N)
    beta = 0.8
    best_c = float(fft_corr(h).max()) * (2.0/N)
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        c = float(corr.max()) * (2.0/N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        top = np.argpartition(corr, -n_top)[-n_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t) - (N - 1)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += (1.0 - h[:N-L])
                grad[:N-L] -= h[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - h[M:])
                grad[M:] -= h[:N-M]
        grad /= len(top)
        
        h_new = h - lr * grad
        # Project onto {0<=h<=1, sum=N/2}
        h_new = np.clip(h_new, 0, 1)
        adj = (h_new.sum() - N/2.0) / N
        h_new = np.clip(h_new - adj, 0, 1)
        m = beta * (h_new - h) + m  # Actually standard Nesterov: m = beta*m + (1-beta)*(h_new-h)
        # Let me use simpler momentum
        h = h_new
        it += 1
        if it % 500 == 0:
            lr *= 0.99
            beta = min(0.95, 0.8 + 0.0001 * it)
    return best_h, best_c
