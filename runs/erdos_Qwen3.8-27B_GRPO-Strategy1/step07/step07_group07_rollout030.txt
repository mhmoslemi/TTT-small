import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def cont_opt(h, deadline, k=50, lr0=0.012, mom=0.9):

            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 40:
            lr0 = max(lr0 * 0.82, 0.0003)
            stale = 0
        max_val = float(np.max(corr))
        thresh = max_val - max(max_val * 0.04, 0.3)
        near_max = np.where(corr >= thresh)[0]
        if len(near_max) > k:

        grad = np.zeros(N)

            if M == 0:

                grad[lo_idx] -= h[lo_idx + M]

        v = 0.85 * v + 0.25 * (h_new - h)
        h = h_new
    return best_h, best_c

        best_move = None
        min_d = float(np.min(deltas))
        if min_d < -1e-8:
            bi, bj = np.unravel_index(np.argmin(deltas), deltas.shape)
            best_move = (os_[bi], zs[bj])
        elif T > 1e-10 and no_imp > 30:
            prob = np.exp(-max(min_d * 2.0 / N, -10.0) / T)
            if np.random.random() < prob:
                bi, bj = np.unravel_index(np.argmin(deltas), deltas.shape)
                best_move = (os_[bi], zs[bj])
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
            corr = compute_corr(h)
        elif no_imp > 300 and np.random.random() < 0.3:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
            corr = compute_corr(h)
        else:
            if no_imp > 500:
                break
        T *= 0.9997
    return best_h, best_c

def resample(h_old, N_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: Continuous opt on parent
    d1 = min(t0 + 90, deadline - 500)
    h1 = project(best_h + np.random.randn(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=50, lr0=0.015, mom=0.9)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    # Phase 2: Binary SA at parent resolution
    d2 = min(t0 + 280, deadline - 350)
    hb = fix_binary(best_h)
    hb, c2 = binary_sa(hb, d2, n_crit=30, s_sz=55, T0=0.004)
    if c2 < best_c:
        best_c, best_h = c2, hb.copy()
    # Phase 3: Try N=400
    if time.time() < deadline - 280:
        h400 = resample(best_h, 400)
        h400 = project(h400 + np.random.randn(400) * 0.03, 200.0)
        d3a = min(time.time() + 50, deadline - 250)
        h400, c3a = cont_opt(h400, d3a, k=50, lr0=0.01, mom=0.9)
        hb400 = fix_binary(h400)
        d3b = min(time.time() + 90, deadline - 220)
        hb400, c3b = binary_sa(hb400, d3b, n_crit=30, s_sz=55, T0=0.004)
        if c3b < best_c:
            best_c, best_h = c3b, hb400.copy()
    # Phase 4: Try N=450
    if time.time() < deadline - 220:
        h450 = resample(best_h, 450)
        h450 = project(h450 + np.random.randn(450) * 0.03, 225.0)
        d4a = min(time.time() + 40, deadline - 190)
        h450, c4a = cont_opt(h450, d4a, k=50, lr0=0.01, mom=0.9)
        hb450 = fix_binary(h450)
        d4b = min(time.time() + 80, deadline - 160)
        hb450, c4b = binary_sa(hb450, d4b, n_crit=30, s_sz=55, T0=0.004)
        if c4b < best_c:
            best_c, best_h = c4b, hb450.copy()
    # Phase 5: Final binary SA at best resolution
    d5 = deadline - 5
    hB = fix_binary(best_h)
    hB, cB = binary_sa(hB, d5, n_crit=35, s_sz=60, T0=0.003)
    if cB < best_c:
        best_c, best_h = cB, hB.copy()
    # Final check
    final_c = compute_c5(best_h)
    hB2 = fix_binary(best_h)
    cB2 = compute_c5(hB2)
    if cB2 < final_c:
        best_h, final_c = hB2, cB2
    return best_h, final_c, len(best_h)
