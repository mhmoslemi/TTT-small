            lr0 *= 0.75
            stale = 0
            if lr0 < 1e-5:
                break
    return best_h, best_c

def interp_solution(h_old, N_new):
    N_old = len(h_old)
    if N_old == N_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    d1 = min(deadline - 400, t0 + 250)
    h1 = fix_binary(best_h)
    h1, c1 = biased_sa_binary(h1, d1, n_lags=12, use_double=True)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    if time.time() < deadline - 300:
        d2 = min(deadline - 280, time.time() + 130)
        h2 = project(best_h + np.random.randn(N0) * 0.04, N0 / 2.0)
        h2, c2 = cont_opt(h2, d2, k=25, lr0=0.014, mom=0.89)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    if time.time() < deadline - 250:
        d3 = min(deadline - 230, time.time() + 150)
        h3 = fix_binary(best_h)
        h3, c3 = biased_sa_binary(h3, d3, n_lags=15, use_double=True)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    for N_try in [382, 440, 480, 520]:
        if time.time() > deadline - 80:
            break
        h_int = interp_solution(best_h, N_try)
        h_bt = fix_binary(h_int)
        d4 = min(deadline - 50, time.time() + 60)
        h_bt, c4 = biased_sa_binary(h_bt, d4, n_lags=12, use_double=True)
        if time.time() < deadline - 20:
            hc = project(h_bt + np.random.randn(N_try) * 0.03, N_try / 2.0)
            d4b = min(deadline - 10, time.time() + 15)
            hc, c4b = cont_opt(hc, d4b, k=20, lr0=0.01, mom=0.9)
            if c4b < c4:
                c4, h_bt = c4b, hc
        if c4 < best_c:
            best_c, best_h = c4, h_bt.copy()

    if time.time() < deadline - 5:
        hB = fix_binary(best_h)
        d5 = min(deadline - 3, time.time() + 30)
        hB, c5 = biased_sa_binary(hB, d5, n_lags=10, use_double=False)
        if c5 < best_c:
            best_c, best_h = c5, hB.copy()

    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
