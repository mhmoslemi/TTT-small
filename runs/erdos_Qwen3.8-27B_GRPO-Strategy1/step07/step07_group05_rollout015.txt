        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        
        best_delta_max = np.max(corr)  # current max
        best_a, best_b = -1, -1

            b = zeros[np.random.randint(len(zeros))]

            delta = compute_delta(h, a, b, N)
            new_corr = corr + delta

                best_a, best_b = a, b

                if h[a] == 1 and h[b] == 0:
                    h[a], h[b] = h[b], h[a]
