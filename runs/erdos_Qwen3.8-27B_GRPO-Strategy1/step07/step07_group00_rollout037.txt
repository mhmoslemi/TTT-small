def smooth_max_and_grad(corr, h, N, tau):
    """Return (smooth_max_value, gradient w.r.t. h)."""

    log_Z = np.log(np.exp((corr - c_max) / tau).sum())
    w = np.exp((corr - c_max - log_Z) / tau)
    grad = np.zeros(N)
    active = np.where(w > 1e-8)[0]

        k_off = k - (N - 1)
        wk = w[k]

            grad[:nv] += wk * (1.0 - h[k_off:k_off+nv])
            grad[k_off:] -= wk * h[:nv]

            nv = N - M
            grad[M:] += wk * (1.0 - h[:nv])

    if time.time() < deadline - 150:
        d3 = min(deadline - 120, time.time() + 120)
        hc2 = best_h.astype(float) * (1.0 - 0.02) + 0.02 * np.random.randn(N0)
        hc2 = project_box_sum(np.clip(hc2, 0, 1), N0)
        hf2, cf2 = fista_optimize(hc2, N0, d3, tau_start=1.5, tau_end=0.15)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2.copy()
    # Phase 4: SA from multiple perturbations
    for trial in range(3):
        if time.time() > deadline - 80:
            break
        hp = make_binary(best_h, N0)
        n_swap = 5 + trial * 3
        ones_i = np.where(hp == 1)[0]
        zeros_i = np.where(hp == 0)[0]
        for _ in range(n_swap):
            pi = np.random.choice(len(ones_i))
            qi = np.random.choice(len(zeros_i))
            hp[ones_i[pi]], hp[zeros_i[qi]] = 0.0, 1.0
            ones_i = np.where(hp == 1)[0]
            zeros_i = np.where(hp == 0)[0]
        d4 = min(deadline - 50, time.time() + 100)
        hs2, cs2 = binary_sa(hp, d4, t_start=0.03, cool_rate=0.9996)
        c5s2 = cs2 * (2.0 / N0)
        if c5s2 < best_c5:
            best_c5, best_h = c5s2, hs2.copy()
    # Phase 5: Final FISTA fine-tune
    if time.time() < deadline - 5:
        hc3 = project_box_sum(best_h.astype(float) + np.random.randn(N0) * 0.01, N0)
        hf3, cf3 = fista_optimize(hc3, N0, deadline - 2, tau_start=0.8, tau_end=0.1)
        if cf3 < best_c5:
            best_c5, best_h = cf3, hf3.copy()
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
