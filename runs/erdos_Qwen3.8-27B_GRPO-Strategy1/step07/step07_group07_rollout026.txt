c[2] = sum_n a_{n+2} * conj(b_n) = a_2*b_0 = 1*0 = 0
c[-1] = sum_n a_{n-1} * conj(b_n) = a_0*b_1 = 1*1 = 1
c[-2] = sum_n a_{n-2} * conj(b_n) = a_0*b_2 = 1*0 = 0

So the full output (index 0 to 2N-2) corresponds to k = -(N-1) to N-1:
full[0] = c[-2] = 0
full[1] = c[-1] = 1
full[2] = c[0] = 0
full[3] = c[1] = 1
full[4] = c[2] = 0

So full = [0, 1, 0, 1, 0]. ✓ (matches my earlier calculation)

Now, the FFT approach in the code:
H = fft(h, L), G = fft(flip(1-h), L)
result = ifft(H * G)

flip(1-h) = [1-h[N-1], ..., 1-h[0]]

For h=[1,0,1]: 1-h=[0,1,0], flip=[0,1,0] (palindrome)

IFFT(FFT([1,0,1,0,0]) * FFT([0,1,0,0,0]))

This computes the linear convolution:
conv[n] = sum_k h_pad[k] * flip(1-h)_pad[n-k]

conv[0] = h[0]*f[0] = 1*0 = 0
conv[1] = h[0]*f[1] + h[1]*f[0] = 1*1 + 0*0 = 1
conv[2] = h[0]*f[2] + h[1]*f[1] + h[2]*f[0] = 0 + 0 + 0 = 0
conv[3] = h[1]*f[2] + h[2]*f[1] = 0 + 1 = 1
conv[4] = h[2]*f[2] = 0

So conv = [0, 1, 0, 1, 0]. ✓

This matches np.correlate(h, 1-h, "full") = [0, 1, 0, 1, 0]. 

But wait, is the FFT convolution the same as np.correlate in general?

conv[n] = sum_k a[k] * b[n-k] (this is convolution)
np.correlate(a,b)[k] = sum_n a[n+k] * conj(b[n]) (this is cross-correlation)

conv = a * b (convolution)
correlate(a,b)[k] = conj(a) * b [k + (N_a - 1)] ... hmm this is getting confusing.

Actually: correlate(a, b) = convolve(a, flip(conj(b)))

For real sequences: correlate(a, b) = convolve(a, flip(b))

In our case: correlate(h, 1-h) = convolve(h, flip(1-h))

And the FFT computes: ifft(fft(h) * fft(flip(1-h))) = convolve(h, flip(1-h)) (linear, with zero padding)

So yes, the code is correct! ✓

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_ls(h_init, deadline, n_crit=40, max_iters=200000):
    N = len(h_init)
    h = h_init.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    no_imp = 0
    last_swap = None

    while time.time() < deadline and it < max_iters:
        corr = compute_corr(h)
        c_raw = float(np.max(corr))
        c = c_raw * (2.0 / N)

        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5000:
                break

        top_lags = np.argsort(corr)[-n_crit:]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        O = len(ones)
        Z = len(zeros)
        if O < 1 or Z < 1:
            break

        tl = top_lags.astype(np.int64)
        a_off = tl[None, :] - ones[:, None]
        vo = (a_off >= 0) & (a_off < N)
        do = np.where(vo, -1.0 + 2.0 * h[np.clip(a_off, 0, N - 1)], 0.0)

        b_off = tl[None, :] - zeros[:, None]
        vz = (b_off >= 0) & (b_off < N)
        dz = np.where(vz, 1.0 - 2.0 * h[np.clip(b_off, 0, N - 1)], 0.0)

        ct = corr[tl]
        scores = (do[:, None, :] + dz[None, :, :] + ct[None, None, :]).max(axis=2)

        if last_swap is not None:
            la, lb = last_swap
            ia = np.searchsorted(ones, la)
            ib = np.searchsorted(zeros, lb)
            if ia < O and ones[ia] == la and ib < Z and zeros[ib] == lb:
                scores[ia, ib] = np.inf

        idx_min = np.argmin(scores)
        ii = int(idx_min // Z)
        jj = int(idx_min % Z)
        a_pos = int(ones[ii])
        b_pos = int(zeros[jj])

        if scores[ii, jj] >= c_raw + 1e-10:
            if no_imp % 500 == 499 and no_imp > 500:
                a_pos = int(np.random.choice(ones))
                b_pos = int(np.random.choice(zeros))
            else:
                it += 1
                continue

        h[a_pos] = 0.0
        h[b_pos] = 1.0
        last_swap = (a_pos, b_pos)
        it += 1

    return best_h, best_c

def cont_opt(h_init, deadline, k=40, lr0=0.02, mom=0.85):
    N = len(h_init)
    tgt = N / 2.0
    h = h_init.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    lr = lr0
    center = N - 1

    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 50:
                lr *= 0.82
                stale = 0

        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            M = abs(int(m) - center)
            if M == 0:
                grad += (1.0 - 2.0 * h)
            elif M < N:
                hi_idx = np.arange(M, N)
                lo_idx = np.arange(0, N - M)
                grad[hi_idx] += (1.0 - h[hi_idx - M])
                grad[lo_idx] -= h[lo_idx + M]
        grad /= k

        h_new = project(h + mom * v - lr * grad, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new

    return best_h, best_c

def expand_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)
