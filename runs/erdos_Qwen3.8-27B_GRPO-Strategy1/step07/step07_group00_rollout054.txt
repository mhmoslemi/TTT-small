            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        for _ in range(3):
            h_new = np.clip(h_new - adj, 0.0, 1.0)
            adj = (h_new.sum() - N / 2.0) / N
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.975
            beta = min(0.94, 0.80 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Continuous optimization from parent
    d1 = min(deadline - 400, t0 + 120)
    hc, c1 = cont_optimize(h0, N0, d1, k_top=50, lr0=0.02, alpha=10.0)
    if c1 < best_c5:
        best_c5, best_h = c1, hc.copy()

    # Phase 2: Binarize + SA
    if time.time() < deadline - 250:
        hb = make_binary(best_h, N0)
        d2 = min(deadline - 200, time.time() + 180)
        h1, _ = targeted_sa(hb, d2, n_peaks=6, n_per_peak=20, cool_rate=0.999, t_start=0.05, max_stall=8000)
        c2 = compute_c5(h1, N0)
        if c2 < best_c5:
            best_c5, best_h = c2, h1.copy()

    # Phase 3: Nesterov polish on continuous
    if time.time() < deadline - 180:
        d3 = min(deadline - 130, time.time() + 100)
        hn, c3 = nesterov_polish(best_h.astype(float), N0, d3, lr=0.006, k_top=60)
        if c3 < best_c5:
            best_c5, best_h = c3, hn.copy()

    # Phase 4: SA with different parameters (more peaks, slower cooling)
    if time.time() < deadline - 100:
        h4 = make_binary(best_h, N0)
        d4 = min(deadline - 50, time.time() + 200)
        h4, _ = targeted_sa(h4, d4, n_peaks=8, n_per_peak=25, cool_rate=0.9995, t_start=0.03, max_stall=10000)
        c4 = compute_c5(h4, N0)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Try nearby resolutions
    for N_try in [360, 400, 440]:
        if time.time() > deadline - 60:
            break
        if N_try % 2 != 0:
            N_try += 1
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, N_try)
        sub_dl = min(deadline - 25, time.time() + 45)
        hr, _ = targeted_sa(hb, sub_dl, n_peaks=5, n_per_peak=12, cool_rate=0.999, t_start=0.04)
        cr = compute_c5(hr, N_try)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 6: Final fine SA + continuous touch
    if time.time() < deadline - 3:
        Ny = len(best_h)
        h6 = make_binary(best_h, Ny)
        h6, _ = targeted_sa(h6, deadline - 2, n_peaks=10, n_per_peak=35, cool_rate=0.9998, t_start=0.012, max_stall=5000)
        c6 = compute_c5(h6, Ny)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    return best_h, best_c5, len(best_h)
