  So changing h[p] from 1 to 0 changes corr[k] by -(1-h[p-(k-(N-1))]) for each valid k.
  
  Also, h[p] appears in (1-h[n-s_k]) when n-s_k = p, i.e., n = p+s_k.
  The term is h[p+s_k] * (1-h[p]). Changing h[p] from 1 to 0 changes (1-h[p]) from 0 to 1.
  So corr[k] changes by +h[p+s_k] for each valid k where p+s_k is in range.

Total change to corr[k] from flipping h[p] 1→0:
Δ_k = -(1-h[p-s_k]) + h[p+s_k] (with validity checks)
     = -1 + h[p-s_k] + h[p+s_k]

For the 2-swap (p1,p2 → 0, q1,q2 → 1):
Total Δ_k = sum of individual flips.

For a 1→0 flip at position p: Δ_k = -1 + h[p-s_k] + h[p+s_k]
For a 0→1 flip at position q: Δ_k = +(1-h[q-s_k]) - h[q+s_k]

So for the 2-swap:
Δ_k = [-1+h[p1-s]+h[p1+s]] + [-1+h[p2-s]+h[p2+s]] + [(1-h[q1-s])-h[q1+s]] + [(1-h[q2-s])-h[q2+s]]

where s = k-(N-1), all with validity (0≤index<N).

The new max is max(old_corr + Δ).

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N
    while L < 2 * N:
        L *= 2
    H = np.fft.rfft(h, L)
    V = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * V, L)[:2 * N - 1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_spectral(corr, h, N, T):
    c_max = corr.max()
    w = np.exp(T * (corr - c_max))
    w /= w.sum()
    L = 1
    while L < 4 * N:
        L *= 2
    w_c = np.zeros(L)
    w_c[N-1:3*N-2] = w
    v_c = np.zeros(L)
    v_c[:N] = 1.0 - h
    h_c = np.zeros(L)
    h_c[:N] = h
    Wf = np.fft.rfft(w_c)
    Vf = np.fft.rfft(v_c)
    Hf = np.fft.rfft(h_c)
    partA = np.fft.irfft(Wf * Vf)[N-1:2*N-1]
    w_fl = w[::-1]
    w_fl_c = np.zeros(L)
    w_fl_c[:2*N-1] = w_fl
    WFl = np.fft.rfft(w_fl_c)
    partB = np.fft.irfft(WFl * Hf)[2*N-2:3*N-2]
    return T * (partA - partB)

def nesterov_opt(h_init, N, deadline, T=80.0, lr0=0.03):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    T_cur = T
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
        grad = grad_spectral(corr, y, N, T_cur)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.994
            beta = min(0.97, 0.82 + it * 0.0002)
        if it % 200 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(50, len(cs)-1)]
            T_cur = max(10.0, min(500.0, T_cur * (0.9 if gap < 0.01 else 1.08)))
    return best_h, best_pk * (2.0 / N)

def binarize(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin
