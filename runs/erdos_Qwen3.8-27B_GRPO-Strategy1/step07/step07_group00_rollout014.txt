import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def adam_polish(h_in, N, deadline, lr=0.05, k_top=60, iters_per_check=200):
    h = h_in.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    
    while time.time() < deadline:
        t += 1
        y = h.copy()
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient of max correlation (subgradient)
        top = np.argpartition(corr, -k_top)[-k_top:]
        # Weight by how much each is above the second max
        top_vals = corr[top]
        max_val = top_vals.max()
        weights = np.exp(5.0 * (top_vals - max_val))
        weights /= weights.sum()
        
        grad = np.zeros(N)
        for wi, tidx in zip(weights, top):
            L = int(tidx - (N - 1))
            if L == 0:
                g = 1.0 - 2.0 * y
            elif L > 0:
                g = np.zeros(N)
                g[L:] += (1.0 - y[:N-L])
                g[:N-L] -= y[L:]
            else:
                M = -L
                g = np.zeros(N)
                g[:N-M] += (1.0 - y[M:])
                g[M:] -= y[:N-M]
            grad += wi * g
        
        grad /= k_top
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to [0,1] and fix sum
        h_new = np.clip(h_new, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        h = h_new
        
        # Decay lr
        if t % 500 == 0:
            lr *= 0.95
    
    return best_h, best_pk * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):

    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):

        w /= w.sum()
