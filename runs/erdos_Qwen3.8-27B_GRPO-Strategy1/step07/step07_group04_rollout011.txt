    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def lse_grad(h, T, top_m=80):
    N = len(h)
    corr = fft_corr(h)
    mc = float(np.max(corr))
    exps = np.exp((corr - mc) / T)
    total = float(np.sum(exps))
    lse_val = mc + T * np.log(total)
    w = exps / total
    top_idx = np.argpartition(w, -top_m)[-top_m:]
    grad = np.zeros(N)
    center = N - 1
    for ai in top_idx:
        wi = w[ai]
        if wi < 1e-8:
            continue
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] += wi * (1.0 - h[L:])
            grad[N - L:] -= wi * h[:N - L]
        else:
            M = -L
            grad[M:] += wi * (1.0 - h[:N - M])
            grad[:N - M] -= wi * h[M:]
    return lse_val, grad, corr

def continuous_lse(h, N, deadline, T_start=0.5, T_end=0.02, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m_vec = np.zeros(N)
    beta = 0.85
    lr = lr0
    it = 0
    T = T_start
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 800.0)
        T = T_start * (T_end / T_start) ** frac
        top_m = max(30, int(100 * (1.0 - 0.5 * frac)))
        lse_v, grad, corr = lse_grad(h, T, top_m)
        pk = float(np.max(corr))
        if pk < best_c - 1e-14:
            best_c = pk
            best_h = h.copy()
        y = h + m_vec
        g = lse_grad(y, T, top_m)[1]
        h_new = project_sum(y - lr * g, target)
        m_vec = beta * (h_new - h)
        h = h_new
        if it % 300 == 0:
            lr *= 0.92
            beta = min(0.96, 0.82 + 0.0001 * it)
    return best_h, best_c * dx

def binary_sa(h, N, deadline, max_sweeps=8000, T0=3.5, Tcool=0.994, T_min=0.04,
              sf=1.0, kick_every=140, kick_size=4, two_swap=False):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            nk = kick_size
            if two_swap:
                nk = min(nk, 2)
            for _ in range(nk):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            stall = 0
            T = max(T0 * 0.5, T_min * 2)
            c = fft_corr(h)
            P = float(c.max())
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def greedy_binary(h, N, deadline, max_sweeps=1500, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 900, t0 + 25)
    h1, c5_1 = greedy_binary(h_bin, N0, d1, max_sweeps=200, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Main SA at N0 (long, two_swap kicks)
    d2 = min(deadline - 750, t0 + 280)
    sf2 = 1.0 if N0 <= 800 else 0.7
    h2, c5_2 = binary_sa(h_bin, N0, d2, max_sweeps=12000, T0=4.0, Tcool=0.9945,
                         T_min=0.03, sf=sf2, kick_every=160, kick_size=5, two_swap=True)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Continuous log-sum-exp at N0
    if time.time() < deadline - 550:
        hc = best_h.copy() if len(best_h) == N0 else upsample(best_h, len(best_h), N0)
        hc = project_sum(hc + np.random.randn(N0) * 0.04, N0 / 2.0)
        d3 = min(deadline - 500, t0 + 430)
        h3, c5_3 = continuous_lse(hc, N0, d3, T_start=0.4, T_end=0.015, lr0=0.025)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: SA at N0 after continuous (refined starting point)
    if time.time() < deadline - 350:
        h4s = ensure_sum(upsample(best_h, len(best_h), N0), N0)
        d4 = min(deadline - 300, t0 + 580)
        h4, c5_4 = binary_sa(h4s, N0, d4, max_sweeps=8000, T0=3.0, Tcool=0.994,
                             T_min=0.03, sf=1.0, kick_every=120, kick_size=4, two_swap=True)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Upsample to 2*N0, continuous + SA
    N1 = 2 * N0
    if time.time() < deadline - 200:
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        h_upc = project_sum(h_up + np.random.randn(N1) * 0.03, N1 / 2.0)
        d5a = min(deadline - 160, t0 + 680)
        h5a, c5_5a = continuous_lse(h_upc, N1, d5a, T_start=0.3, T_end=0.01, lr0=0.015)
        if c5_5a < best_c5:
            best_c5, best_h = c5_5a, h5a.copy()
        if time.time() < deadline - 50:
            h5b = ensure_sum(best_h, N1)
            d5b = min(deadline - 30, t0 + 780)
            h5b2, c5_5b = binary_sa(h5b, N1, d5b, max_sweeps=3000, T0=2.5,
                                    Tcool=0.995, T_min=0.04, sf=0.5,
                                    kick_every=150, kick_size=3, two_swap=True)
            if c5_5b < best_c5:
                best_c5, best_h = c5_5b, h5b2.copy()

    # Phase 6: Final greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 2:
        hF = ensure_sum(best_h, N_fin)
        sf_fin = 1.0 if N_fin <= 400 else (0.7 if N_fin <= 800 else 0.5)
        d6 = max(deadline - 0.5, t0 + 820)
        hF2, c5F = greedy_binary(hF, N_fin, d6, max_sweeps=400, sf=sf_fin)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
