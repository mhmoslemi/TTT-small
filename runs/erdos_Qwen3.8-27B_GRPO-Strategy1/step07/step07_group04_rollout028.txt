import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    res = target - s
    if abs(res) < 1e-12:
        return h
    # Binary search for the offset
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h + mid, 0.0, 1.0).sum()) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)

def nesterov_smooth(h0, N, deadline, max_iter=5000):
    """Nesterov with smooth-max gradient."""
    h = h0.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_P = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = 0.03
    beta = 0.9
    T_smooth = 0.5  # temperature for softmax weights
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        y = h + m
        corr = fft_corr(y)
        P = float(corr.max())
        if P < best_P - 1e-14:
            best_P = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        
        # Smooth gradient: weighted sum of D columns
        weights = np.exp(corr / T_smooth)
        weights /= weights.sum()
        D = compute_D(y, N)
        grad = D.T @ weights  # shape (N,)

        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0001)
            T_smooth = max(0.1, T_smooth * 0.99)
    return best_h, best_P * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h + mid, 0.0, 1.0).sum()) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum_binary(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())

        return h_bin
    rng = np.random.RandomState(42 + N)

        ones = np.where(h_bin == 1)[0]

        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def binary_greedy(h, N, deadline, max_sweeps=2000, sf=1.0):
    h = h.astype(float).copy()

    best_h = h.copy()

        c = fft_corr(h)
        P = float(c.max())

            best_P = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
