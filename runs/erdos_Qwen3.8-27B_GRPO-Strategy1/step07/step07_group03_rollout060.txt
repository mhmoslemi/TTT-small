import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0-h), L)))

def c5_of(h):
    return float(fft_corr(h).max()) * (2.0/len(h))

def proj(h, tgt):
    h = np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        if np.clip(h-mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def binsum(h, N):
    nh = N//2
    h = np.round(h).astype(float)
    h = np.clip(h, 0, 1)
    n1 = int(h.sum())
    if n1 == nh: return h
    r = np.random.RandomState(7+N)
    if n1 > nh:
        h[r.choice(np.where(h==1)[0], n1-nh, replace=False)] = 0
    else:
        h[r.choice(np.where(h==0)[0], nh-n1, replace=False)] = 1
    return h

def Dmat(h):
    N = len(h)
    sh = np.arange(2*N-1) - (N-1)
    p = np.arange(N)[:,None]
    sv = sh[None,:]
    D = -(1-h[np.clip(p-sv,0,N-1)]) * ((p-sv>=0)&(p-sv<N)) + h[np.clip(p+sv,0,N-1)] * ((p+sv>=0)&(p+sv<N))
    return D

def grad_topk(h, tk=80):

    c = fft_corr(h)
    top = np.argpartition(c, -tk)[-tk:]
    # Weight by proximity to max
    mx = c.max()
    mn = c[top].min()
    w = (c[top] - mn) / (mx - mn + 1e-10)
    w = w / w.sum()
    g = np.zeros(N)
    ctr = N-1
    for i in range(len(top)):

            g[:N-M] -= wi * h[M:]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def c5_of(h):
    return float(fft_corr(h).max()) * (2.0 / len(h))

def proj(h, tgt):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - tgt) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binsum(h, N):
    nh = N // 2
    h = np.clip(np.round(h), 0, 1).astype(float)
    n1 = int(h.sum())
    if n1 == nh:
        return h
    r = np.random.RandomState(7 + N)
    if n1 > nh:
        h[r.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    else:
        h[r.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    return h

def Dmat(h):
    N = len(h)
    sh = np.arange(2 * N - 1) - (N - 1)
    p = np.arange(N)[:, None]
    sv = sh[None, :]
    pm = p - sv
    pp = p + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[np.clip(pm, 0, N-1)], 0.0)
    hp = np.where(vp, h[np.clip(pp, 0, N-1)], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def grad_weighted(h, tk=100):
    N = len(h)
    c = fft_corr(h)
    top = np.argpartition(c, -tk)[-tk:]
    mx = c.max()
    mn_c = c[top].min()
    w = (c[top] - mn_c) / max(mx - mn_c, 1e-10)
    w = w / w.sum()
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(top)):
        L = int(top[i] - ctr)
        wi = w[i]
        if L == 0:
            g += wi * (1.0 - 2.0 * h)
        elif L > 0:
            g[:N-L] += wi * (1.0 - h[L:])
            g[L:] -= wi * h[:N-L]
        else:
            M = -L
            g[M:] += wi * (1.0 - h[:N-M])
            g[:N-M] -= wi * h[M:]
    return g, float(c.max())

def nesterov(h, N, dl, tk=100, lr0=0.006, beta0=0.88):

    h = proj(h, N / 2.0)
    m = np.zeros(N)
    best_h = h.copy()
    best_c = float(fft_corr(h).max())
    lr = lr0
    beta = beta0
    stall = 0
    for it in range(500000):

            break

        g, _ = grad_weighted(y, tk)

        m = beta * (h_new - h)
        h = h_new
        c = float(fft_corr(h).max())

    x0 = np.linspace(0, 2, len(h), endpoint=False)
