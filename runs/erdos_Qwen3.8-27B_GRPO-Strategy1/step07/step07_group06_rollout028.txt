def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = np.max(c)
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

                best_h = h.copy()

    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = np.max(c)
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_run(h, N, deadline, T_schedule):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h, N)
    lam = 4.0
    for T in T_schedule:
        if time.time() > deadline:
            break
        for it in range(3):
            if time.time() > deadline:
                break
            res = sp_min(smooth_obj, h, args=(T, target, lam),
                         jac=True, method='L-BFGS-B',
                         bounds=[(0, 1)] * N,
                         options={'maxiter': 80, 'maxfun': 250, 'ftol': 1e-15})
            h = proj01(res.x, target)
            c = compute_c5(h, N)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        lam = min(lam * 2.5, 200.0)
    return best_h, best_c

def binary_search(h, N, deadline, max_iters=300, ntri=20, two_swap=0.1):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        bd = 0.0
        if np.random.rand() < two_swap and len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(ntri // 3):
                i1 = np.random.randint(len(ones))
                i2 = np.random.randint(len(ones))
                j1 = np.random.randint(len(zeros))
                j2 = np.random.randint(len(zeros))
                if i1 == i2 or j1 == j2:
                    continue
                oi, oj, zj, zk = int(ones[i1]), int(ones[i2]), int(zeros[j1]), int(zeros[j2])
                h[oi], h[oj], h[zj], h[zk] = 0.0, 0.0, 1.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > bd:
                    bd = d
                    bi1, bi2, bj1, bj2 = oi, oj, zj, zk
                h[oi], h[oj], h[zj], h[zk] = 1.0, 1.0, 0.0, 0.0
            if bd > 0:
                h[bi1], h[bi2], h[bj1], h[bj2] = 0.0, 0.0, 1.0, 1.0
                improved = True
        else:
            bi, bj = -1, -1
            for _ in range(ntri):
                i = np.random.randint(len(ones))
                j = np.random.randint(len(zeros))
                oi, zi = int(ones[i]), int(zeros[j])
                h[oi], h[zi] = 0.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > bd:
                    bd, bi, bj = d, oi, zi
                h[oi], h[zi] = 1.0, 0.0
            if bi >= 0:
                h[bi], h[bj] = 0.0, 1.0
                improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c

def resample_binary(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    hi = np.interp(x_new, x_old, h_old, period=2.0)
    ht = (hi > 0.5).astype(float)
    nh = N_new // 2
    n1 = int(ht.sum())
    if n1 > nh:
        idx = np.where(ht == 1)[0]
        np.random.shuffle(idx)
        ht[idx[:n1 - nh]] = 0.0
    elif n1 < nh:
        idx = np.where(ht == 0)[0]
        np.random.shuffle(idx)
        ht[idx[:nh - n1]] = 1.0
    return ht
