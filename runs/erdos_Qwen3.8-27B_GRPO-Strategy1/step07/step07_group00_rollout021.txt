import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(1.0-h, L), L))

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0/N)

def make_binary(h, N):
    h = np.array(h, dtype=float).copy()

    elif n1 < target:
        idx = np.argsort(h)[:target-n1]

    deltas = np.zeros((N, 2*N-1))
    for i in range(N):

def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=30):
    h = np.array(h_in, dtype=float).copy()
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.8
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N-1))
            if L == 0:
                grad += 1.0 - 2.0*y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top)
        h_new = np.clip(y - lr*grad, 0.0, 1.0)
        adj = (h_new.sum() - N/2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.92, 0.78 + it*0.00015)

def fft_corr(h):
    N = len(h)

    return float(fft_corr(h).max()) * (2.0 / N)

def make_binary(h, N):
    h = np.array(h, dtype=float).copy()
    target = N // 2
    n1 = int(round(h.sum()))
    if n1 > target:
        idx = np.argsort(-h)[:n1 - target]
        h[idx] = 0.0
    elif n1 < target:
        idx = np.argsort(h)[:target - n1]
        h[idx] = 1.0

    h[h <= 0.5] = 0.0

    N = len(h)
    s = 2.0 * h - 1.0

    y = h.copy()
    corr = fft_corr(y)
    best_pk = float(corr.max())
    best_h = h.copy()
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.988
            beta = min(0.93, 0.78 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        N0 = 380
        h0 = np.zeros(N0)
        for i in range(N0 // 2):
            h0[2 * i] = 1.0
        h0 = make_binary(h0, N0)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Targeted SA
    d1 = min(deadline - 350, t0 + 220)
    h1, c1 = targeted_sa(h_bin, d1, n_peaks=7, n_per_peak=20, cool_rate=0.9991, t_start=0.055)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Exhaustive local search
    if time.time() < deadline - 200:
        h2 = make_binary(best_h, N0)
        d2 = min(deadline - 180, time.time() + 100)
        while time.time() < d2:
            h2, c2 = exhaustive_ls(h2, max_sweeps=15)
            if c2 >= best_c5 - 1e-14:
                break
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Nesterov continuous polish
    if time.time() < deadline - 160:
        d3 = min(deadline - 120, time.time() + 110)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.035
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        wn, c3 = nesterov_polish(hc, N0, d3, lr=0.011, k_top=40)
        if c3 < best_c5:
            best_c5, best_h = c3, wn.copy()

    # Phase 4: Targeted SA with lower temp
    if time.time() < deadline - 120:
        h4 = make_binary(best_h, N0)
        d4 = min(deadline - 70, time.time() + 200)
        h4, c4 = targeted_sa(h4, d4, n_peaks=10, n_per_peak=28, cool_rate=0.9996, t_start=0.022)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Exhaustive LS final
    if time.time() < deadline - 30:
        h5 = make_binary(best_h, N0)
        h5, c5 = exhaustive_ls(h5, max_sweeps=25)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Try alternate resolutions
    for N_try in [360, 400, 440, 480]:
        if time.time() > deadline - 50:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == N0:
            continue
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, N_try)
        sub_dl = min(deadline - 25, time.time() + 40)
        hr, cr = targeted_sa(hb, sub_dl, n_peaks=6, n_per_peak=14, cool_rate=0.999, t_start=0.045)
        hr, cr = exhaustive_ls(hr, max_sweeps=5)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 7: Final fine SA + LS
    if time.time() < deadline - 5:
        Ny = len(best_h)
        hf = make_binary(best_h, Ny)
        hf, _ = targeted_sa(hf, deadline - 3, n_peaks=12, n_per_peak=30, cool_rate=0.9997, t_start=0.012)
        hf, cf = exhaustive_ls(hf, max_sweeps=10)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
