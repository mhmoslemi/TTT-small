def project_box_sum(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target}."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Bisection for the shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_max_corr(h, k_active=None):
    """Compute gradient of max_k R_k w.r.t. h.
    R_k = sum_i h[i] * (1-h[i+k]) with valid indices.
    
    Returns gradient vector of length N.
    """
    N = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    if k_active is None:
        k_active = 1
    # Get top k indices
    top_idx = np.argpartition(c, -k_active)[-k_active:]
    # Softmax weights
    top_c = c[top_idx]
    weights = np.exp(top_c - top_c.max())
    weights /= weights.sum()
    
    grad = np.zeros(N)
    one_minus_h = 1.0 - h
    for w, ai in zip(weights, top_idx):
        k = int(ai - center)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            # R_k = sum_{i=0}^{N-1-k} h[i] * (1-h[i+k])
            # d/dh[j]: if j < N-k: (1-h[j+k]) * w
            #           if j >= k: -h[j-k] * w
            # Combined: (1-h[j+k]) for j in [0, N-k), -h[j-k] for j in [k, N)
            # For j in [0, k): only (1-h[j+k])
            # For j in [k, N-k): (1-h[j+k]) - h[j-k]
            # For j in [N-k, N): only -h[j-k]
            g = np.zeros(N)
            g[:N-k] += one_minus_h[k:]  # (1-h[j+k]) for j=0..N-k-1
            g[k:] -= h[:N-k]            # -h[j-k] for j=k..N-1
            grad += w * g
        else:
            k_abs = -k
            # R_{-k_abs} = sum_{i=k_abs}^{N-1} h[i] * (1-h[i-k_abs])
            # d/dh[j]: if j >= k_abs: (1-h[j-k_abs])
            #           if j < N-k_abs: -h[j+k_abs]
            g = np.zeros(N)
            g[k_abs:] += one_minus_h[:N-k_abs]
            g[:N-k_abs] -= h[k_abs:]
            grad += w * g
    return grad

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_hinge(h, k_top=30):
    """Subgradient of max_k R_k using top-k hinge."""
    N = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    idx = np.argpartition(c, -k_top)[-k_top:]
    top_c = c[idx]
    # Hinge: weight = max(0, c_i - c_max + margin)
    margin = 0.01
    w = np.maximum(0, top_c - top_c.max() + margin)
    if w.sum() < 1e-12:
        w = np.zeros(k_top); w[np.argmax(top_c)] = 1
    else:
        w /= w.sum()
    
    grad = np.zeros(N)
    for wi, ai in zip(w, idx):
        k = int(ai - center)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            g = np.zeros(N)
            g[:N-k] += (1.0 - h[k:])
            g[k:] -= h[:N-k]
            grad += wi * g
        else:
            ka = -k
            g = np.zeros(N)
            g[ka:] += (1.0 - h[:N-ka])
            g[:N-ka] -= h[ka:]
            grad += wi * g
    return grad

def swap_delta(h, N):
    """Compute D[p, k] = change in corr[k] when h[p] flips 1->0."""
    # corr index: 0 to 2N-2, shift = corr_idx - (N-1)
    shifts = np.arange(2*N - 1) - (N-1)  # -(N-1) to (N-1)
    D = np.zeros((N, 2*N-1))
    for p in range(N):
        for j, s in enumerate(shifts):
            pm = p - s
            pp = p + s
            val = 0.0
            if 0 <= pm < N:
                val += -(1.0 - h[pm])
            if 0 <= pp < N:
                val += h[pp]
            D[p, j] = val
    return D

def swap_delta(h, N):
    shifts = np.arange(2*N - 1) - (N-1)  # shape (2N-1,)
    pos = np.arange(N)[:, None]  # shape (N, 1)
    sv = shifts[None, :]  # shape (1, 2N-1)
    pm = pos - sv  # p - s
    pp = pos + sv  # p + s
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

    return grad

def nesterov_run(h0, N, deadline, lr0=0.012, k0=35):
    target = N / 2.0
    dx = 2.0 / N
    h = project_box_sum(h0.astype(float).copy(), target)
    c = np.correlate(h, 1.0 - h, mode='full')
    best_pk = float(np.max(c))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    k = k0
    it = 0
    stall = 0
    while time.time() < deadline:
        y = h + m
        g = grad_hinge(y, k_top=k)
        h_new = project_box_sum(y - lr * g, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 15 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
            pk = float(np.max(c))
            if pk < best_pk - 1e-14:
                best_pk = pk
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall >= 50:
                m = np.zeros(N)
                stall = 0
                k = max(10, k // 2)
                lr = lr0 * (0.6 + 0.8 * np.random.random())
            if it % 150 == 0:
                lr *= 0.97
                beta = min(0.97, 0.88 + 0.09 * min(1.0, it / 3000.0))
                k = min(N // 3, k + 8)
    return best_h, best_pk * dx

def swap_delta(h, N):
    shifts = np.arange(2*N - 1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_anneal(h, N, deadline, max_sweeps=400):
    h = h.astype(float).copy()
    target = N // 2
    h = (h >= 0.5).astype(float)
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            ones = np.where(h == 1)[0]
            h[np.random.choice(ones, n1 - target, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[np.random.choice(zeros, target - n1, replace=False)] = 1
    best_c = float(np.max(np.correlate(h, 1 - h, mode='full')))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_cur = np.correlate(h, 1 - h, mode='full')
        P_cur = float(np.max(c_cur))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        D = swap_delta(h, N)
        Dz = D[zeros, :]
        best_new_P = P_cur
        best_swap = None
        for p in ones:
            base = c_cur - D[p, :]
            cands = base[None, :] - Dz
            npk = cands.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_swap = (int(p), int(zeros[mi]))
        best_2swap = None
        n_try = min(100, max(30, N // 4))
        for _ in range(n_try):
            i1, i2 = np.random.randint(0, len(ones), 2)
            j1, j2 = np.random.randint(0, len(zeros), 2)
            if i1 == i2 or j1 == j2:
                continue
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            new_P = float((c_cur - D[p1] - D[p2] + D[q1] + D[q2]).max())
            if new_P < best_new_P:
                best_new_P = new_P
                best_swap = None
                best_2swap = (p1, p2, q1, q2)
        if best_2swap:
            p1, p2, q1, q2 = best_2swap
            h[p1] = h[p2] = 0.0; h[q1] = h[q2] = 1.0
        elif best_swap:
            h[best_swap[0]] = 0.0; h[best_swap[1]] = 1.0
        else:
            stall += 1
            if stall > 18:
                for _ in range(3):
                    p = int(ones[np.random.randint(len(ones))])
                    q = int(zeros[np.random.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 6:
                break
            else:
                continue
        c_new = float(np.max(np.correlate(h, 1 - h, mode='full')))
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def upsample(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(np.correlate(best_h, 1 - best_h, mode='full'))) * (2.0 / N0)

    # Phase 1: Binary polish at N0
    d1 = min(deadline - 550, t0 + 160)
    h1, c5_1 = binary_anneal(h0, N0, d1, max_sweeps=350)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Continuous Nesterov at N0 (primary)
    if time.time() < deadline - 350:
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_box_sum(hc, N0 / 2.0)
        d2 = min(deadline - 300, time.time() + 180)
        h2, c5_2 = nesterov_run(hc, N0, d2, lr0=0.012, k0=40)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Upscale to 2*N0, continuous optimize
    N1 = 2 * N0
    if time.time() < deadline - 200:
        h_up = upsample(best_h, N1)
        h_up = project_box_sum(h_up, N1 / 2.0)
        d3 = min(deadline - 160, time.time() + 170)
        h3, c5_3 = nesterov_run(h_up, N1, d3, lr0=0.010, k0=50)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Second continuous run at N0 with different perturbation
    if time.time() < deadline - 120:
        hc2 = best_h.copy()
        if len(hc2) != N0:
            x_old = np.linspace(0, 2, len(hc2), endpoint=False)
            x_new = np.linspace(0, 2, N0, endpoint=False)
            hc2 = np.interp(x_new, x_old, hc2)
        hc2 += np.random.randn(N0) * 0.08
        hc2 = project_box_sum(hc2, N0 / 2.0)
        d4 = min(deadline - 90, time.time() + 100)
        h4, c5_4 = nesterov_run(hc2, N0, d4, lr0=0.018, k0=25)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Binary polish at best resolution
    N_fin = len(best_h)
    if time.time() < deadline - 10:
        d5 = deadline - 2
        h5, c5_5 = binary_anneal(best_h, N_fin, d5, max_sweeps=150)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Final check
    final_c5 = float(np.max(np.correlate(best_h, 1 - best_h, mode='full'))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
