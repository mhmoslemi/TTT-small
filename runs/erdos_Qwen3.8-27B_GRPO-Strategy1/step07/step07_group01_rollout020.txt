def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0

        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)

            grad += wi * (1.0 - 2.0 * h)

        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]

def compute_D(h, N):
    """Precompute the delta matrix for single-flip changes."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all  # p - s: neighbor below
    pp = p_all + s_all  # p + s: neighbor above
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
    return D, shifts

def thue_morse(N):
    """Generate Thue-Morse sequence of length N."""
    seq = np.zeros(N, dtype=np.uint8)
    for i in range(1, N):
        # t(2n) = t(n), t(2n+1) = 1 - t(n)
        p = i
        cnt = 0
        while p:
            cnt ^= (p & 1)
            p >>= 1
        seq[i] = cnt
    return seq.astype(float)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def D_at_shifts(h, positions, shifts_int, N):

        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argpartition(c, -K)[-K:]
        s_int = offsets[top_idx]  # actual integer shifts
        c_top = c[top_idx]
        D_ones = D_at_shifts(h, ones, s_int, N)
        D_zeros = D_at_shifts(h, zeros, s_int, N)
        best_new_P = P
        best_pq = None
        for pi in range(len(ones)):
            base = c_top + D_ones[pi, :]
            cand = base[None, :] - D_zeros
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(ones[pi]), int(zeros[mi]))
        if best_pq is not None and best_new_P < P - 1e-12:
            p, q = best_pq
            h[p], h[q] = 0.0, 1.0
            c_check = fft_corr(h)
            if float(c_check.max()) > P + 1e-6:
                h[p], h[q] = 1.0, 0.0
                stall += 1
        else:
            if stall > perturb_every:
                n_r = min(2 + stall // 12, 6)
                for _ in range(n_r):
                    p = int(ones[np.random.randint(len(ones))])
                    q = int(zeros[np.random.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Continuous Nesterov (two variants, ~160s)
    starts = [h0 + np.random.randn(N0) * 0.05]
    sq = np.linspace(0, 1, N0)
    starts.append(project_sum(np.sin(np.pi * sq) ** 2, N0 / 2.0) * 0.5 + 0.5 * h0)
    for si, hc in enumerate(starts):
        if time.time() > deadline - 650:
            break
        for T_val in [70.0, 180.0]:
            if time.time() > deadline - 600:
                break
            hc2 = project_sum(hc + np.random.randn(N0) * 0.03, N0 / 2.0)
            d = min(deadline - 550, time.time() + 65)
            h_n, c5_n = nesterov_opt(hc2, N0, d, k=40, T=T_val, lr0=0.015)
            if c5_n < best_c5:
                best_c5, best_h = c5_n, h_n.copy()

    # Phase 2: Main binary search (~550s)
    if time.time() < deadline - 250:
        h_bin = ensure_binary(best_h, N0)
        d_b = min(deadline - 200, time.time() + 350)
        h_br, c5_br = binary_search_opt(h_bin, N0, d_b, K=250, max_sweeps=10000, perturb_every=18)
        if c5_br < best_c5:
            best_c5, best_h = c5_br, h_br.copy()

    # Phase 3: Continuous re-entry (~80s)
    if time.time() < deadline - 150:
        hc = best_h + np.random.randn(N0) * 0.025
        hc = project_sum(hc, N0 / 2.0)
        d3 = min(deadline - 120, time.time() + 60)
        h3, c5_3 = nesterov_opt(hc, N0, d3, k=50, T=120.0, lr0=0.009)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Second binary pass with perturbation (~100s)
    if time.time() < deadline - 60:
        h_b2 = ensure_binary(best_h, N0)
        for _ in range(8):
            p = int(np.random.randint(N0))
            if h_b2[p] == 1:
                q = int(np.random.randint(N0))
                if h_b2[q] == 0:
                    h_b2[p], h_b2[q] = 0.0, 1.0
        d4 = min(deadline - 30, time.time() + 70)
        h4, c5_4 = binary_search_opt(h_b2, N0, d4, K=300, max_sweeps=8000, perturb_every=15)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Final micro-polish
    if time.time() < deadline - 10:
        hFc = best_h + np.random.randn(N0) * 0.01
        hFc = project_sum(hFc, N0 / 2.0)
        d5 = min(deadline - 3, time.time() + 20)
        h5, c5_5 = nesterov_opt(hFc, N0, d5, k=50, T=200.0, lr0=0.005)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
        hFb = ensure_binary(best_h, N0)
        c5Fb = compute_c5(hFb, N0)
        if c5Fb < best_c5:
            best_c5, best_h = c5Fb, hFb

    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
