import time

def fft_corr(h):

    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)

    return float(np.max(fft_corr(h))) * dx

            g[L:] -= h[:N-L]
        else:

            g[M:] += (1.0 - h[:N-M])
            g[:N-M] -= h[M:]
    g /= len(k_indices)
    return g

    target = N / 2.0
    dx = 2.0 / N
    h = h_init.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    k_cur = k_start
    it = 0

    while time.time() < deadline:
        N_total = 2 * N - 1
        y = h + beta * m
        y = project_sum(y, target)
        corr = fft_corr(y)
        top_k = np.argpartition(corr, -k_cur)[-k_cur:]
        pk = float(corr[top_k].max())

        g = grad_max(h, N, top_k)
        # Add L2 smoothing

            pk_new = float(corr_new.max())
            if pk_new < pk - 1e-10:

            lr_try *= 0.5

            lr_try = lr * 0.1  # minimum step
        
        m = h_new - h
        h = h_new

            best_val = pk_new
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Progressive constraint activation

            k_cur = min(k_max, k_cur + 10)
            stall_count = 0
            lr *= 0.95
            beta = min(0.95, beta + 0.01)
        
        it += 1
        if it % 500 == 0:

def binary_hill_climb(h_init, N, deadline, n_iterations=50000):
    target = N // 2
    h = h_init.astype(float).copy()
    # Ensure binary
    h[(h > 0.5)] = 1.0
    h[(h <= 0.5)] = 0.0
    # Fix sum
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(42)
        to_remove = rng.choice(ones, n1 - target, replace=False)
        h[to_remove] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(42)
        to_add = rng.choice(zeros, target - n1, replace=False)
        h[to_add] = 1
    
    corr = fft_corr(h)
    best = float(corr.max())
    best_h = h.copy()
    rng = np.random.RandomState(123)
    
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try 1-swap
        if len(ones) > 0 and len(zeros) > 0:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
            val = float(corr.max())
            if val < best - 1e-12:
                best = val
                best_h = h.copy()
            else:
                h[p], h[q] = 1.0, 0.0  # revert
        
        # Try 2-swap
        if len(ones) > 1 and len(zeros) > 1:
            p1, p2 = rng.choice(ones, 2, replace=False)
            q1, q2 = rng.choice(zeros, 2, replace=False)
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
            corr = fft_corr(h)
            val = float(corr.max())
            if val < best - 1e-12:
                best = val
                best_h = h.copy()
            else:
                h[p1] = h[p2] = 1.0
                h[q1] = h[q2] = 0.0
    
    return best_h, best * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0

            lo = mid
        else:

    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k_idx):

            g += 1.0 - 2.0 * h

        else:
            M = -L
            g[M:] += (1.0 - h[:N-M])
            g[:N-M] -= h[M:]

    return g

        h3 = best_h + rng.randn(N0) * 0.04
        h3, c5_3 = nesterov_cont(h3, N0, t3, k0=50, kmax=200, lr0=0.004, seed=seed + 200)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
        if time.time() < deadline - 100:
            hb3 = to_binary(best_h, N0)
            hb3, c5b3 = binary_greedy(hb3, N0, deadline - 80, iters_per_batch=300, n_restart=3, seed=seed + 500)
            if c5b3 < best_c5:
                best_c5, best_h = c5b3, hb3.copy()

    # Phase 4: Upscale to 2*N0, continuous refine
    if time.time() < deadline - 100:
        N1 = 2 * N0
        h_up = upsample(best_h, N1)
        h_up = project_sum(h_up, N1 / 2.0)
        t4 = min(deadline - 60, time.time() + budget_s * 0.07)
        h4, c5_4 = nesterov_cont(h_up, N1, t4, k0=40, kmax=150, lr0=0.004, seed=seed + 300)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
        if time.time() < deadline - 30:
            hb4 = to_binary(best_h, N1)
            hb4, c5b4 = binary_greedy(hb4, N1, deadline - 10, iters_per_batch=200, n_restart=3, seed=seed + 600)
            if c5b4 < best_c5:
                best_c5, best_h = c5b4, hb4.copy()

    # Phase 5: Final polish at original N
    if time.time() < deadline - 15 and len(best_h) != N0:
        h_back = upsample(best_h, N0)
        h_back = project_sum(h_back, N0 / 2.0)
        hbF = to_binary(h_back, N0)
        hbF, c5F = binary_greedy(hbF, N0, deadline - 3, iters_per_batch=600, n_restart=4, seed=seed + 900)
        if c5F < best_c5:
            best_c5, best_h = c5F, hbF.copy()
    elif time.time() < deadline - 15:
        hbF = to_binary(best_h, len(best_h))
        Nf = len(best_h)
        hbF, c5F = binary_greedy(hbF, Nf, deadline - 3, iters_per_batch=600, n_restart=4, seed=seed + 900)
        if c5F < best_c5:
            best_c5, best_h = c5F, hbF.copy()

    # Ensure sum constraint is tight
    Nf = len(best_h)
    best_h = project_sum(best_h, Nf / 2.0)
    final_c5 = compute_c5(best_h, Nf)
    return best_h, final_c5, Nf
