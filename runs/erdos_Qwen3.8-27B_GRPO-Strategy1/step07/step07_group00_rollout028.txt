import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N=None):
    if N is None:
        N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def weighted_obj(corr, k_top=5, temperature=1.0):
    """Weighted objective: emphasizes top peaks"""
    top = np.sort(corr)[-k_top:]
    weights = np.exp(np.arange(k_top) / temperature)
    return float(np.sum(weights * top) / np.sum(weights))

def exhaustive_local(h, deadline):
    N = len(h)
    h = h.copy()
    corr = fft_corr(h)
    s, deltas = build_deltas(h)
    best_c = corr.max()
    best_h = h.copy()
    
    while time.time() < deadline:
        # Get active peaks
        top_k = 5
        peak_lags = np.argpartition(corr, -top_k)[-top_k:]
        
        # Generate candidates near peaks
        cand_p = set()
        cand_q = set()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                for i in range(N - k_off):
                    if h[i] == 1 and h[i + k_off] == 0:
                        cand_p.add(i)
                        cand_q.add(i + k_off)
            else:
                M = -k_off
                for i in range(M, N):
                    if h[i] == 1 and h[i - M] == 0:
                        cand_p.add(i)
                        cand_q.add(i - M)
        
        # Limit candidates
        cand_p = np.array(list(cand_p)[:200])
        cand_q = np.array(list(cand_q)[:200])
        
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        
        # Evaluate all pairs
        dp = deltas[cand_p]  # (P, 2N-1)
        dq = deltas[cand_q]  # (Q, 2N-1)
        
        # Find best swap: minimize max(corr + dp[i] - dq[j])
        # Process in chunks to avoid memory issues
        chunk = 20
        best_new = 1e10
        best_p, best_q = -1, -1
        
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            # (chunk, Q, 2N-1)
            vals = corr[None, None, :] + dp[ci:ce, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)  # (chunk, Q)
            mi = np.unravel_index(nm.argmin(), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                best_p, best_q = mi[0] + ci, mi[1]
        
        if best_new >= best_c - 1e-12:
            break  # No improvement found
        
        p = int(cand_p[best_p])
        q = int(cand_q[best_q])
        h[p], h[q] = 0.0, 1.0
        corr += deltas[p].copy() - deltas[q].copy()
        s[p], s[q] = -s[p], -s[q]
        deltas[p, :] = 0.0
        deltas[p, p:p+N] = s
        deltas[q, :] = 0.0
        deltas[q, q:q+N] = s
        best_c = best_new
        best_h = h.copy()
    
    return best_h, best_c * (2.0/N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N=None):
    if N is None:
        N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas
