O(1) delta: When I swap h[a]: 1→0 and h[b]: 0→1, the change in R[k] for a specific k is:
- The pair (a, a+k): was h[a]*(1-h[a+k]) = 1*(1-h[a+k]), now 0*(1-h[a+k]) = 0. Delta: -(1-h[a+k]) = h[a+k]-1
- The pair (k, a) i.e. (a-k, a): was h[a-k]*(1-h[a]) = h[a-k]*0 = 0, now h[a-k]*(1-0) = h[a-k]. Delta: +h[a-k]
- Similarly for b:
- The pair (b, b+k): was 0*(1-h[b+k]) = 0, now 1*(1-h[b+k]). Delta: +(1-h[b+k])
- The pair (b-k, b): was h[b-k]*(1-0) = h[b-k], now h[b-k]*(1-1) = 0. Delta: -h[b-k]

So ΔR[k] = (h[a+k]-1) + h[a-k] + (1-h[b+k]) - h[b-k]

This is O(1) per k! But we need to check the max over all k, which is O(N). So the total delta computation is O(N) per swap.

For N=380, that's 380 operations per swap. Compared to FFT which is ~380*9 = 3420 operations. So the O(N) delta is actually faster than FFT for checking a single swap!

But we only need to check if the NEW max is less than the OLD max. We can do:
1. Compute the delta vector ΔR (length 2N-1) in O(N)
2. New R = old R + ΔR
3. new_max = max(new R) = max(old R + ΔR)

The max can be computed in O(N). So total O(N) per swap evaluation, which is much faster than O(N log N) for FFT.

Let me implement this! This will be the key speedup.

```python
def swap_delta(h, a, b, N):
    """Compute the change in correlation when h[a]:1->0 and h[b]:0->1."""
    corr = np.zeros(2*N-1)
    mid = N - 1
    
    # Effect of h[a]: 1->0
    # At lag k: delta from pair (a, a+k) is -(1-h[a+k]) for k where 0<=a+k<N
    #            delta from pair (a-k, a) is +h[a-k] for k where 0<=a-k<N
    # In the correlation array, lag k is at index mid+k
    for k in range(-(N-1), N):
        # Pair (a, a+k): contributes to R[k] when 0<=a and 0<=a+k<N
        if 0 <= a + k < N:
            corr[mid + k] -= (1 - h[a + k])
        # Pair (a-k, a): contributes to R[k] when 0<=a-k<N and 0<=a<N
        if 0 <= a - k < N:
            corr[mid + k] += h[a - k]
    
    # Effect of h[b]: 0->1
    for k in range(-(N-1), N):
        if 0 <= b + k < N:
            corr[mid + k] += (1 - h[b + k])
        if 0 <= b - k < N:
            corr[mid + k] -= h[b - k]
    
    return corr
```

```python
def swap_delta_vec(h, a, b, N):
    """Vectorized: change in correlation when h[a]:1->0, h[b]:0->1."""
    L = 2*N - 1
    mid = N - 1
    d = np.zeros(L)
    
    # h[a] goes 1->0:  
    # For each k, the term h[a]*(1-h[a+k]) in R[k] vanishes: delta = -(1-h[a+k])
    # The term h[a-k]*(1-h[a]) in R[k] changes from 0 to h[a-k]: delta = +h[a-k]
    # 
    # R[k] at index mid+k for k in [-(N-1), N-1]
    # 
    # Term 1: -(1-h[a+k]) for k where 0<=a+k<N, i.e., -a<=k<N-a
    k_min1 = max(-(N-1), -a)
    k_max1 = min(N-1, N-1-a)
    for k in range(k_min1, k_max1+1):

    k_max2 = min(N-1, a)

    k_max3 = min(N-1, N-1-b)

    k_max4 = min(N-1, b)
```

```python
def swap_delta_vec(h, a, b, N):
    L = 2*N - 1
    mid = N - 1
    d = np.zeros(L)
    
    # h[a]: 1->0
    # Term 1: -(1-h[a+k]) at index mid+k, for k where 0<=a+k<N
    #   k ranges: max(-mid, -a) to min(mid, N-1-a)
    #   a+k ranges: max(0, a-a) to ... = max(0,0)=0 to min(N-1, N-1)
    #   Actually: 0 <= a+k < N means -a <= k < N-a, clipped to [-mid, mid]
    ks = np.arange(max(-mid, -a), min(mid, N-1-a) + 1)
    d[mid + ks] -= (1.0 - h[a + ks])
    
    # Term 2: +h[a-k] at index mid+k, for k where 0<=a-k<N
    #   0 <= a-k < N means a-N < k <= a, clipped to [-mid, mid]
    ks = np.arange(max(-mid, -(N-1-a)), min(mid, a) + 1)
    d[mid + ks] += h[a - ks]
    
    # h[b]: 0->1
    # Term 3: +(1-h[b+k]) at index mid+k, for k where 0<=b+k<N
    ks = np.arange(max(-mid, -b), min(mid, N-1-b) + 1)
    d[mid + ks] += (1.0 - h[b + ks])
    
    # Term 4: -h[b-k] at index mid+k, for k where 0<=b-k<N
    ks = np.arange(max(-mid, -(N-1-b)), min(mid, b) + 1)
    d[mid + ks] -= h[b - ks]
    
    return d
```

```python
def best_swap_search(h, N, corr, n_trials=200):
    """Try n_trials random swaps, return the best one."""

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_val = cur_max
    best_a, best_b = -1, -1
    
    for _ in range(n_trials):
        a = int(ones[np.random.randint(len(ones))])  # 1 -> 0
        b = int(zeros[np.random.randint(len(zeros))])  # 0 -> 1

        new_max = float(np.max(corr + d))

            best_val = new_max
            best_a, best_b = a, b
    
    return best_a, best_b, best_val
```

Actually, for a 2-swap (a1→0, a2→0, b1→1, b2→1), the delta is the sum of two single-swap deltas:

The cross terms arise because the two swaps interact. For example, if b1 = a2+lag, then the pair (a2, b1) is affected by both changes. This makes 2-swap deltas more complex.

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(np.conj(H) * G, L))

def c5_from_corr(corr, N):
    return float(np.max(corr)) * (2.0 / N)

def fix_sum(h, target):
    h = h.copy()
    s = int(np.round(h.sum()))
    diff = target - s
    if diff > 0:
        idx = np.where(h < 1)[0]
        h[np.random.choice(idx, diff, replace=False)] += 1
    elif diff < 0:
        idx = np.where(h > 0)[0]
        h[np.random.choice(idx, -diff, replace=False)] -= 1
    return h

def swap_delta(h, a, b, N):
    L = 2 * N - 1
    mid = N - 1
    d = np.zeros(L)
    # h[a]: 1->0
    ks1 = np.arange(max(-mid, -a), min(mid, N - 1 - a) + 1)
    d[mid + ks1] -= (1.0 - h[a + ks1])
    ks2 = np.arange(max(-mid, a - (N-1)), min(mid, a) + 1)
    d[mid + ks2] += h[a - ks2]
    # h[b]: 0->1
    ks3 = np.arange(max(-mid, -b), min(mid, N - 1 - b) + 1)
    d[mid + ks3] += (1.0 - h[b + ks3])
    ks4 = np.arange(max(-mid, b - (N-1)), min(mid, b) + 1)
    d[mid + ks4] -= h[b - ks4]
    return d

def targeted_search(h, N, deadline, top_lags=5, per_lag=50, stall_limit=500):
    nh = N // 2
    h = h.astype(float).copy()
    h = fix_sum(h, nh)
    corr = fft_corr(h)
    cur = float(np.max(corr))
    best_h = h.copy()
    best_c = cur
    stall = 0
    mid = N - 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline:
        if stall > stall_limit:
            # Perturb
            for _ in range(np.random.randint(3, 8)):
                oa = int(np.random.choice(ones))
                zb = int(np.random.choice(zeros))
                h[oa], h[zb] = h[zb], h[oa]
            corr = fft_corr(h)
            cur = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
            continue
        
        worst = np.argsort(-corr)[:top_lags]
        improved = False
        
        for wi in worst:
            k = int(wi)
            lag = k - mid
            if abs(lag) >= N:
                continue
            if lag >= 0:
                rng = N - lag
                bad = np.where((h[:rng] == 1) & (h[lag:lag+rng] == 0))[0]
            else:
                L = -lag
                rng = N - L
                bad = np.where((h[L:L+rng] == 1) & (h[:rng] == 0))[0] + L
            
            if len(bad) == 0:
                continue
            
            n_try = min(per_lag, len(bad))
            for _ in range(n_try):
                i = int(bad[np.random.randint(len(bad))])
                fp = i + lag  # position to set to 1 (currently 0)
                if fp < 0 or fp >= N or h[fp] != 0:
                    continue
                j = int(ones[np.random.randint(len(ones))])
                if j == fp:
                    continue
                
                d = swap_delta(h, j, fp, N)  # j: 1->0, fp: 0->1
                nm = float(np.max(corr + d))
                if nm < cur - 1e-12:
                    h[j] = 0
                    h[fp] = 1
                    corr += d
                    cur = nm
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    stall = 0
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
    
    return best_h, best_c
```
