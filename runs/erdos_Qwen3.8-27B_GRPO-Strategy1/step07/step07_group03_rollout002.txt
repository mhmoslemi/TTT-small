    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    if rng is None:
        rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def binary_anneal(h, N, deadline, max_sweeps=600, two_swap_frac=0.45, n_two_try=120, rng=None):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N, rng)
    if rng is None:
        rng = np.random.RandomState(7)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    temp = 0.015
    stall = 0
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp * 0.996)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        best_new_P = P_cur
        best_pq = None
        for p in ones:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        best_2p = None
        if len(ones) >= 4 and len(zeros) >= 4 and rng.random() < two_swap_frac:
            n_try = min(n_two_try, max(40, N // 3))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = (p1, p2, q1, q2)
        if best_2p is not None:
            p1, p2, q1, q2 = best_2p
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
        elif best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            if stall > 20:
                for _ in range(4):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 12:
                break
            else:
                stall += 1
                continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-13:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k_init=50, k_max=120, lr0=0.018, lr_decay=0.99, beta0=0.88):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    k = k_init
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr.topk if hasattr(corr, 'topk') else corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= lr_decay
            beta = min(0.96, beta0 + it * 0.0002)
            k = min(k_max, k + 2)
    return best_h, best_pk * dx

def coord_descent(h, N, deadline, n_sweeps=200):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(N):
            old_val = h[i]
            lo = 0.0
            hi = 1.0
            best_v = old_val
            best_obj = float(np.max(fft_corr(h)))
            for _ in range(25):
                mid = (lo + hi) * 0.5
                if abs(mid - best_v) < 1e-8:
                    break
                h[i] = mid
                obj = float(np.max(fft_corr(h)))
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_v = mid
                    hi = mid
                else:
                    if mid > old_val:
                        lo = mid
                    else:
                        hi = mid if mid > lo else lo
                    lo = lo if mid > lo else mid
            h[i] = best_v
        h = project_sum(h, target)
        c = float(np.max(fft_corr(h)))
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            improved = True
        if not improved and c >= best_c - 1e-14:
            break
    return best_h, best_c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Aggressive binary anneal at N0
    h_bin = ensure_sum(h0, N0, rng)
    d1 = min(deadline - 500, t0 + budget_s * 0.33)
    h1, c5_1 = binary_anneal(h_bin, N0, d1, max_sweeps=800, two_swap_frac=0.5, n_two_try=150, rng=rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Nesterov continuous at 3x resolution
    N_up = 3 * N0
    h_up = upsample(best_h, N0, N_up)
    h_up = project_sum(h_up, N_up / 2.0)
    d2 = min(deadline - 350, time.time() + budget_s * 0.22)
    h2, c5_2 = nesterov_opt(h_up, N_up, d2, k_init=60, k_max=150, lr0=0.02, lr_decay=0.992)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Binarize and binary anneal again at N0
    if time.time() < deadline - 250:
        h_b3 = ensure_sum(upsample(best_h, len(best_h), N0), N0, rng)
        # Perturb slightly
        ones_t = np.where(h_b3 == 1)[0]
        zeros_t = np.where(h_b3 == 0)[0]
        for _ in range(3):
            h_b3[ones_t[rng.randint(len(ones_t))]] = 0.0
            h_b3[zeros_t[rng.randint(len(zeros_t))]] = 1.0
        h_b3 = ensure_sum(h_b3, N0, rng)
        d3 = min(deadline - 200, time.time() + budget_s * 0.15)
        h3, c5_3 = binary_anneal(h_b3, N0, d3, max_sweeps=500, two_swap_frac=0.5, n_two_try=130, rng=rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Nesterov at N0 with different params
    if time.time() < deadline - 120:
        hc4 = upsample(best_h, len(best_h), N0) + rng.randn(N0) * 0.06
        hc4 = project_sum(hc4, N0 / 2.0)
        d4 = min(deadline - 80, time.time() + budget_s * 0.10)
        h4, c5_4 = nesterov_opt(hc4, N0, d4, k_init=40, k_max=100, lr0=0.025, lr_decay=0.988)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Coordinate descent at N0
    if time.time() < deadline - 60:
        hc5 = upsample(best_h, len(best_h), N0)
        hc5 = project_sum(hc5, N0 / 2.0)
        d5 = min(deadline - 30, time.time() + budget_s * 0.06)
        h5, c5_5 = coord_descent(hc5, N0, d5, n_sweeps=80)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final binary polish
    N_fin = len(best_h)
    hF = ensure_sum(best_h, N_fin, rng)
    sf = 0.5 if N_fin > 500 else 1.0
    d6 = deadline - 2
    hF2, c5F = binary_anneal(hF, N_fin, d6, max_sweeps=300, two_swap_frac=0.4, n_two_try=100, rng=rng)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    # Final evaluation with exact method
    final_c5 = compute_c5_exact(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure sum constraint is met
    target_sum = len(best_h) / 2.0
    best_h = project_sum(best_h, target_sum)
    final_c5 = compute_c5_exact(best_h)
    return best_h, final_c5, len(best_h)
