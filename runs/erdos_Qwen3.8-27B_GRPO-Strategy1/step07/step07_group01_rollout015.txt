```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(99)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1-target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target-n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def nesterov_opt(h_init, N, deadline, k=40, T=60.0, lr0=0.02):
    h = h_init.astype(float).copy()
    target = N/2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T*(c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            sv = int(idx[ii] - center)
            wi = w[ii]
            if sv == 0:
                grad += wi*(1.0 - 2.0*h)
            elif sv > 0:
                grad[sv:] += wi*(1.0 - h[:N-sv])
                grad[:N-sv] -= wi*h[sv:]
            else:
                M = -sv
                grad[:N-M] += wi*(1.0 - h[M:])
                grad[M:] -= wi*h[:N-M]
        grad /= k
        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.996
            beta = min(0.98, 0.85 + it*0.00015)
        if it % 400 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs)-1)]
            T = max(15.0, min(300.0, T*(0.9 if gap < 0.005 else 1.05)))
    return best_h, best_pk*(2.0/N)

def sa_search(h, N, deadline, n_max, T0, T1, rng):
    h = h.copy()
    c = fft_corr(h)
    cur_max = float(c.max())
    best_c = cur_max
    best_h = h.copy()
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    s_arr = np.arange(2*N-1) - (N-1)
    stall = 0
    for it in range(n_max):
        if time.time() > deadline:
            break
        frac = it / n_max
        T = max(T1, T0 * ((T1/T0) ** frac))
        pi = rng.randint(len(ones_list))
        qi = rng.randint(len(zeros_list))
        p = ones_list[pi]
        q = zeros_list[qi]
        ps = p - s_arr
        psp = p + s_arr
        qs = q - s_arr
        qsp = q + s_arr
        m1 = (ps >= 0) & (ps < N)
        m2 = (psp >= 0) & (psp < N)
        m3 = (qs >= 0) & (qs < N)
        m4 = (qsp >= 0) & (qsp < N)
        delta = np.zeros(2*N-1)
        if m1.any():
            delta[m1] -= (1.0 - h[ps[m1]])
        if m2.any():
            delta[m2] += h[psp[m2]]
        if m3.any():
            delta[m3] += (1.0 - h[qs[m3]])
        if m4.any():
            delta[m4] -= h[qsp[m4]]
        new_max = float(np.max(c + delta))
        if new_max < cur_max:
            accept = True
        elif T > 1e-10:
            accept = rng.random() < np.exp(-(new_max - cur_max)/T)
        else:
            accept = False
        if accept:
            h[p] = 0.0
            h[q] = 1.0
            ones_list[pi] = q
            zeros_list[qi] = p
            c += delta
            cur_max = new_max
            stall = 0
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
            stall += 1
        else:
            stall += 1
        if stall > 300000:
            # perturb: do 5 random swaps
            for _ in range(5):
                pp = ones_list[rng.randint(len(ones_list))]
                qq = zeros_list[rng.randint(len(zeros_list))]
                h[pp], h[qq] = 0.0, 1.0
                # update lists

                ones_list[oi] = qq
                zeros_list[zi] = pp
            c = fft_corr(h)

def greedy_polish(h, N, deadline, n_max=200000):
    h = h.copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    rng = np.random.RandomState(55)
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    s_arr = np.arange(2*N-1) - (N-1)
    for it in range(n_max):
        if time.time() > deadline:
            break
        pi = rng.randint(len(ones_list))
        qi = rng.randint(len(zeros_list))
        p = ones_list[pi]
        q = zeros_list[qi]
        ps = p - s_arr; psp = p + s_arr
        qs = q - s_arr; qsp = q + s_arr
        m1 = (ps>=0)&(ps<N); m2 = (psp>=0)&(psp<N)
        m3 = (qs>=0)&(qs<N); m4 = (qsp>=0)&(qsp<N)
        delta = np.zeros(2*N-1)
        if m1.any(): delta[m1] -= (1.0-h[ps[m1]])
        if m2.any(): delta[m2] += h[psp[m2]]
        if m3.any(): delta[m3] += (1.0-h[qs[m3]])
        if m4.any(): delta[m4] -= h[qsp[m4]]
        new_max = float(np.max(c+delta))
        if new_max < best_c - 1e-15:
            h[p]=0; h[q]=1
            ones_list[pi]=q; zeros_list[qi]=p
            c += delta
            best_c = new_max
            best_h = h.copy()
    return best_h, best_c*(2.0/N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
