    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def ensure_sum(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[target:]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h

def focused_anneal(h, N, deadline, max_iters=50000, T0=0.04, T1=1e-5):
    h = ensure_sum(h.astype(float).copy(), N // 2)
    corr = fft_corr(h)
    cur_c = float(np.max(corr)) * 2.0 / N
    best_c = cur_c
    best_h = h.copy()
    T = T0
    cool = (T1 / T0) ** (1.0 / max(1, max_iters))
    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        r = np.random.random()
        if r < 0.7:
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if len(ones_idx) < 2 or len(zeros_idx) < 2:
                break
            oi = int(np.random.choice(ones_idx))
            zi = int(np.random.choice(zeros_idx))
            if oi == zi:
                continue
            h[oj] = h[zi] if False else 0
            h[oi] = 0
            h[zi] = 1
            nc = fft_corr(h)
            new_c = float(np.max(nc)) * 2.0 / N
            dE = new_c - cur_c
            if dE < -1e-12 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
                cur_c = new_c
                corr = nc
                if cur_c < best_c - 1e-12:
                    best_c, best_h, no_improve = cur_c, h.copy(), 0
                else:
                    no_improve += 1
            else:
                h[oi], h[zi] = 1, 0
        else:
            L = int(np.random.choice([2, 3, 4, 5]))
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if len(ones_idx) < L + 1 or len(zeros_idx) < L + 1:
                continue
            ob = int(np.random.choice(ones_idx[:N - L]))
            zb = int(np.random.choice(zeros_idx[:N - L]))
            if zb <= ob < zb + L or ob <= zb < ob + L:
                continue
            h[ob:ob + L] = 0
            h[zb:zb + L] = 1
            nc = fft_corr(h)
            new_c = float(np.max(nc)) * 2.0 / N
            dE = new_c - cur_c
            if dE < -1e-12 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
                cur_c = new_c
                corr = nc
                if cur_c < best_c - 1e-12:
                    best_c, best_h, no_improve = cur_c, h.copy(), 0
                else:
                    no_improve += 1
            else:
                h[ob:ob + L] = 1
                h[zb:zb + L] = 0
        T *= cool
        if no_improve > 4000:
            for _ in range(3 + int(np.random.random() * 4)):
                a = int(np.random.choice(np.where(h == 1)[0]))
                b = int(np.random.choice(np.where(h == 0)[0]))
                h[a], h[b] = h[b], h[a]
            corr = fft_corr(h)
            cur_c = float(np.max(corr)) * 2.0 / N
            T = max(T, 0.015)
            no_improve = 0
    return best_h, best_c

def grad_softmax(h, N, soft_temp):
    corr = fft_corr(h)
    peak = np.max(corr)
    w = np.exp((corr - peak) / soft_temp)
    w /= w.sum()
    center = N - 1
    grad = np.zeros(N)
    sig = np.where(w > 5e-4)[0]
    for mi in sig:
        s = int(mi - center)
        wi = w[mi]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            fac = np.zeros(N)
            fac[:N - s] += (1.0 - h[s:])
            fac[s:] += (-h[:N - s])
            grad += wi * fac
        else:
            s2 = -s
            fac = np.zeros(N)
            fac[s2:] += (1.0 - h[:N - s2])
            fac[:N - s2] += (-h[s2:])
            grad += wi * fac
    return grad

def continuous_refine(h, N, deadline, lr0=0.002, st0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h, best_c = h.copy(), compute_c5(h, N)
    lr, st = lr0, st0
    for _ in range(5000):
        if time.time() > deadline - 0.5:
            break
        grad = grad_softmax(h, N, st)
        h_new = np.clip(h - lr * (grad - grad.mean()), 0, 1)
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        c_new = compute_c5(h_new, N)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
        lr *= 0.995
        st *= 0.98
    return best_h, best_c

def quasiperiodic_init(N, slope):
    x = np.arange(N) * slope
    frac = x - np.floor(x)
    h = (frac < 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        order = np.argsort(np.abs(frac[idx] - 0.5))
        h[idx[order[:n1 - nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.argsort(np.abs(frac[idx] - 0.5))
        h[idx[order[:nh - n1]]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h, best_c5 = h0.copy(), compute_c5(h0, N0)

    # Phase 1: Binary anneal initial (160s)
    h_b = (h0 > 0.5).astype(float)
    d1 = min(deadline - 550, t0 + 160)
    hs, c1 = focused_anneal(h_b, N0, d1, max_iters=50000, T0=0.035, T1=1e-5)
    if c1 < best_c5:
        best_c5, best_h = c1, hs.copy()

    # Phase 2: Continuous refine (130s)
    if time.time() < deadline - 380:
        hc = best_h + np.random.randn(N0) * 0.008
        hc = np.clip(hc, 0, 1)
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(hc - mid, 0, 1).sum() > N0 / 2.0:
                lo = mid
            else:
                hi = mid
        hc = np.clip(hc - (lo + hi) / 2, 0, 1)
        d2 = min(deadline - 340, time.time() + 130)
        hc, c2 = continuous_refine(hc, N0, d2, lr0=0.0015, st0=0.01)
        if c2 < best_c5:
            best_c5, best_h = c2, hc.copy()

    # Phase 3: Quasiperiodic multi-start (200s)
    slopes = [0.309, 0.382, 0.236, 0.618, 0.724]
    N_trials = [300, 340, 360, 400, 440, 480]
    t_per = 180.0 / (len(N_trials) * 2)
    for Nt in N_trials:
        if time.time() > deadline - 130:
            break
        if Nt % 2:
            Nt += 1
        if Nt == N0:
            continue
        for sl in slopes[:2]:
            if time.time() > deadline - 100:
                break
            hv = quasiperiodic_init(Nt, sl)
            ds = min(deadline - 100, time.time() + t_per)
            hr, cr = focused_anneal(hv, Nt, ds, max_iters=12000, T0=0.04, T1=1e-5)
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()

    # Phase 4: Deep binary search on best (remaining)
    if time.time() < deadline - 15:
        h4 = project_binary(best_h, len(best_h) // 2)
        d4 = deadline - 1
        h4, c4 = focused_anneal(h4, len(h4), d4, max_iters=60000, T0=0.012, T1=1e-7)
        c4 = compute_c5(h4, len(h4))
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    hF = project_binary(best_h, len(best_h) // 2)
    cF = compute_c5(hF, len(hF))
    if cF < best_c5:
        best_c5, best_h = cF, hF
    return best_h, best_c5, len(best_h)
