    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_binary(h, target):
    h = h.copy()
    s = int(h.sum())
    if s > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - s]] = 1
    return h

def build_C(h, N):
    return np.array([compute_C_single(h, N, i) for i in range(N)])

def full_steepest(h, N, deadline, max_iters=800, chunk=50):
    h = h.copy()
    best_c = np.inf
    best_h = h.copy()
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if len(ones) < 1 or len(zeros) < 1:
            break
        Co = np.array([compute_C_single(h, N, p) for p in ones])
        Cz = np.array([compute_C_single(h, N, q) for q in zeros])
        best_newP = P + 1e-12
        best_pi = -1
        best_qi = -1
        for cs in range(0, len(ones), chunk):
            ce = min(cs + chunk, len(ones))
            new = corr[None, None, :] - Co[cs:ce, None, :] + Cz[None, :, :]
            newP = new.max(axis=2)
            fi = np.argmin(newP)
            pi_l = int(fi // len(zeros)) + cs
            qi_l = int(fi % len(zeros))
            val = float(newP[pi_l - cs, qi_l])
            if val < best_newP:
                best_newP = val
                best_pi = pi_l
                best_qi = qi_l
        if best_newP < P - 1e-14:
            p = int(ones[best_pi])
            q = int(zeros[best_qi])
            h[p] = 0
            h[q] = 1
            corr = fft_corr(h)
            P = best_newP
            if P < best_c - 1e-15:
                best_c = P
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            break
    return best_h, best_c

def anneal(h, N, deadline, C=None, T0=0.0004, cool=0.9997, sample_sz=130, restart_T=0.0003):
    h = h.copy()
    best_h = h.copy()
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    T = T0
    stuck = 0
    total_stuck = 0
    if C is None:
        C = build_C(h, N)
    else:
        C = C.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    while time.time() < deadline:
        it += 1
        if stuck > 120:
            kick_sz = 3 + (stuck // 60) % 4
            for _ in range(kick_sz):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1
                C[p] = compute_C_single(h, N, p)
                C[q] = compute_C_single(h, N, q)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            P = float(np.max(corr))
            stuck = 0
            T = restart_T
            total_stuck = 0
            continue
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        new = corr[None, None, :] - Co[:, None, :] + Cz[None, :, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        delta = mv - P
        accept = False
        if delta < -1e-14:
            accept = True
        elif T > 1e-12 and np.random.random() < np.exp(-delta / T):
            accept = True
        if accept:
            p = int(os_[mi[1]])
            q = int(zs[mi[0]])
            h[p] = 0
            h[q] = 1
            C[p] = compute_C_single(h, N, p)
            C[q] = compute_C_single(h, N, q)
            corr = fft_corr(h)
            P = mv
            if P < best_c - 1e-15:
                best_c = P
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            total_stuck += 1
        else:
            stuck += 1
        T *= cool
        if T < 1e-12:
            T = restart_T * 0.9
    return best_h, best_c, C

def nesterov_polish(h, N, deadline, k=60, lr0=0.006):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            Lv = int(ai - center)
            if Lv == 0:
                grad += (1.0 - 2.0 * y)
            elif Lv > 0:
                grad[Lv:] += (1.0 - y[:N - Lv])
                grad[:N - Lv] -= y[Lv:]
            else:
                M = -Lv
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.985
            beta = min(0.97, 0.87 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    target = N0 // 2
    hb = (h0 > 0.5).astype(float)
    hb = fix_binary(hb, target)
    best_h = hb.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Full steepest descent (exact local optimum)
    d1 = min(deadline - 500, t0 + 150)
    h1, c1 = full_steepest(hb, N0, d1, max_iters=600, chunk=50)
    if c1 * (2.0 / N0) < best_c5:
        best_c5 = c1 * (2.0 / N0)
        best_h = h1.copy()

    # Phase 2: Annealing from steepest result
    if time.time() < deadline - 350:
        d2 = min(deadline - 340, time.time() + 200)
        C2 = build_C(best_h, N0)
        h2, c2, _ = anneal(best_h, N0, d2, C=C2, T0=0.0004, cool=0.9996, sample_sz=140, restart_T=0.0003)
        if c2 * (2.0 / N0) < best_c5:
            best_c5 = c2 * (2.0 / N0)
            best_h = h2.copy()

    # Phase 3: Multi-restart perturbation + steepest
    for restart in range(4):
        if time.time() > deadline - 250:
            break
        h_r = best_h.copy()
        ones_r = np.where(h_r == 1)[0]
        zeros_r = np.where(h_r == 0)[0]
        n_kick = 4 + restart * 2
        for _ in range(n_kick):
            p = int(np.random.choice(ones_r))
            q = int(np.random.choice(zeros_r))
            h_r[p] = 0
            h_r[q] = 1
            ones_r = np.where(h_r == 1)[0]
            zeros_r = np.where(h_r == 0)[0]
        d3 = min(deadline - 240, time.time() + 45)
        h3, c3 = full_steepest(h_r, N0, d3, max_iters=300, chunk=60)
        if c3 * (2.0 / N0) < best_c5:
            best_c5 = c3 * (2.0 / N0)
            best_h = h3.copy()

    # Phase 4: Multi-resolution
    for N_try in [360, 400, 440]:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = (h_int > 0.5).astype(float)
        h_bt = fix_binary(h_bt, N_try // 2)
        d4 = min(deadline - 110, time.time() + 50)
        h4, c4 = full_steepest(h_bt, N_try, d4, max_iters=200, chunk=50)
        c4v = c4 * (2.0 / N_try)
        if c4v < best_c5:
            best_c5 = c4v
            best_h = h4.copy()
            N0 = N_try

    # Phase 5: Extended annealing at best resolution
    if time.time() < deadline - 60:
        d5 = min(deadline - 50, time.time() + 150)
        C5 = build_C(best_h, N0)
        h5, c5v, _ = anneal(best_h, N0, d5, C=C5, T0=0.0003, cool=0.9995, sample_sz=130, restart_T=0.0002)
        if c5v * (2.0 / N0) < best_c5:
            best_c5 = c5v * (2.0 / N0)
            best_h = h5.copy()

    # Phase 6: Nesterov + final steepest
    if time.time() < deadline - 25:
        hF = best_h.astype(float) + np.random.randn(N0) * 0.03
        hF = project_01(hF, N0 / 2.0)
        d6 = min(deadline - 15, time.time() + 40)
        hF, cF = nesterov_polish(hF, N0, d6, k=70, lr0=0.004)
        hFb = (hF > 0.5).astype(float)
        hFb = fix_binary(hFb, N0 // 2)
        cFb = compute_c5(hFb, N0)
        if cFb < best_c5:
            best_c5 = cFb
            best_h = hFb.copy()

    if time.time() < deadline - 2:
        d7 = deadline - 1
        h7, c7 = full_steepest(best_h, N0, d7, max_iters=50, chunk=50)
        if c7 * (2.0 / N0) < best_c5:
            best_c5 = c7 * (2.0 / N0)
            best_h = h7.copy()

    # Final: check binary vs continuous
    hB = (best_h > 0.5).astype(float)
    hB = fix_binary(hB, N0 // 2)
    cB = compute_c5(hB, N0)
    cC = compute_c5(best_h, N0)
    if cB < cC:
        best_h = hB
        best_c5 = cB
    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
