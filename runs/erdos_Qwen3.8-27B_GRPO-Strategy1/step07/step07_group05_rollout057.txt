import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    Lp = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, Lp)
    G = np.fft.rfft((1.0 - h)[::-1], Lp)
    return np.real(np.fft.irfft(H * G, Lp))[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum(h, target):
    h = h.copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = np.maximum(1.0 - h, 0)
        tr = room.sum()
        if tr > 1e-12:
            h += diff * room / tr
    else:
        ta = h.sum()
        if ta > 1e-12:
            h += diff * h / ta
    return np.clip(h, 0.0, 1.0)

def fix_sum_bin(h, target):
    h = h.copy()
    n1 = h.sum()
    if int(n1) > target:
        idx = np.where(h > 0.5)[0]
        h[np.random.choice(idx, int(n1) - target, replace=False)] = 0
    elif int(n1) < target:
        idx = np.where(h < 0.5)[0]
        h[np.random.choice(idx, target - int(n1), replace=False)] = 1
    return h

def weighted_grad(h, w, N):
    center = N - 1
    grad = w[center] * (1.0 - 2.0 * h)
    for k in range(1, N):
        wp = w[center + k]
        wm = w[center - k] if center - k >= 0 else 0.0
        if wp > 1e-12:
            grad[k:] += wp * (1.0 - h[:N-k])
            grad[:N-k] -= wp * h[k:]
        if wm > 1e-12:
            grad[:N-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:N-k]
    return grad

def continuous_opt(h, N, deadline, lr0=0.008, t_soft0=0.04):
    h = np.array(h, dtype=float).copy()
    nh = N / 2.0
    h = fix_sum(h, nh)
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    t_soft = t_soft0
    while time.time() < deadline - 1:
        it += 1
        corr = fft_corr(h)
        peak = corr.max()
        w = np.exp((corr - peak) / t_soft)
        w /= w.sum()
        grad = weighted_grad(h, w, N)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        bc1 = 1 - 0.9**it
        bc2 = 1 - 0.999**it
        h = h - lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)
        h = np.clip(h, 0, 1)
        h = fix_sum(h, nh)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 200 == 0:
            lr *= 0.99
            t_soft = max(0.005, t_soft * 0.95)
        if it % 2000 == 0:
            lr *= 0.92
            t_soft = max(0.005, t_soft * 0.85)
    return best_h, best_c

def bin_anneal(h, N, deadline, T0=0.06, max_it=40000):
    h = fix_sum_bin(h, N // 2)
    corr = fft_corr(h)
    cur_max = corr.max()
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    nh = N // 2
    T = T0
    cool = (1e-5 / T0) ** (1.0 / max_it)
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        r = np.random.random()
        if r < 0.7:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            oi = np.random.randint(len(ones))
            zi = np.random.randint(len(zeros))
            oi, zi = ones[oi], zeros[zi]
            h[oi], h[zi] = 0, 1
            nm = fft_corr(h).max()
            if nm < cur_max or (T > 1e-6 and np.random.random() < np.exp(-(nm - cur_max) / T)):
                cur_max = nm
                if nm * 2.0 / N < best_c - 1e-14:
                    best_c = nm * 2.0 / N
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[oi], h[zi] = 1, 0
        else:
            # Block swap
            L = int(np.random.choice([2, 3, 3, 4, 5]))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < L + 1 or len(zeros) < L + 1:
                T *= cool
                continue
            ob = ones[np.random.randint(0, len(ones) - L + 1)]
            zb = zeros[np.random.randint(0, len(zeros) - L + 1)]
            if zb - ob >= L or ob - zb >= L:
                tmp = h[ob:ob+L].copy()
                h[ob:ob+L] = h[zb:zb+L]
                h[zb:zb+L] = tmp
                nm = fft_corr(h).max()
                if nm < cur_max or (T > 1e-6 and np.random.random() < np.exp(-(nm - cur_max) / T)):
                    cur_max = nm
                    if nm * 2.0 / N < best_c - 1e-14:
                        best_c = nm * 2.0 / N
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    h[ob:ob+L], h[zb:zb+L] = tmp, h[ob:ob+L].copy()
        T *= cool
        if stall > 3000:
            for _ in range(8):
                oi = np.random.choice(np.where(h == 1)[0])
                zi = np.random.choice(np.where(h == 0)[0])
                h[oi], h[zi] = h[zi], h[oi]
            corr = fft_corr(h)
            cur_max = corr.max()
            T = max(T, 0.02)
            stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, N)

    hb = (h0 > 0.5).astype(float)
    hb, c1 = bin_anneal(hb, N, d1)
    if c1 < best_c:
        best_c, best_h = c1, hb.copy()
    
    # Phase 2: Continuous optimization (350s)
    d2 = min(deadline - 200, time.time() + 350)
    hc = best_h + np.random.randn(N) * 0.01
    hc = np.clip(hc, 0, 1)
    hc = fix_sum(hc, N / 2.0)
    hc, c2 = continuous_opt(hc, N, d2)
    if c2 < best_c:

    # Phase 3: Diversify - perturb and re-optimize (150s)

        # Invert a random segment
        s = np.random.randint(0, N - 30)
        e = s + np.random.randint(20, 60)
        e = min(e, N)
        hp[s:e] = 1.0 - hp[s:e]

        if c3 < best_c:

        d4 = min(deadline - 40, time.time() + 60)

        h5, c5 = continuous_opt(h5, N, d5, lr0=0.003, t_soft0=0.02)

    # Validate

    best_h = fix_sum(best_h, N / 2.0)
    best_c = compute_c5(best_h, N)
    return best_h, best_c, N

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    Lp = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, Lp)
    G = np.fft.rfft((1.0 - h)[::-1], Lp)
    return np.real(np.fft.irfft(H * G, Lp))[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum(h, target):
    h = h.copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = np.maximum(1.0 - h, 0)
        tr = room.sum()
        if tr > 1e-14:
            h += diff * room / tr
    else:
        ta = h.sum()
        if ta > 1e-14:
            h += diff * h / ta
    return np.clip(h, 0.0, 1.0)

def fix_sum_bin(h, target):
    h = h.copy()
    n1 = int(h.sum() + 0.5)
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def weighted_grad(h, w, N):
    center = N - 1
    grad = w[center] * (1.0 - 2.0 * h)
    for k in range(1, N):
        wp = w[center + k]
        wm = w[center - k] if center - k >= 0 else 0.0
        if wp > 1e-14:
            grad[k:] += wp * (1.0 - h[:N - k])
            grad[:N - k] -= wp * h[k:]
        if wm > 1e-14:
            grad[:N - k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:N - k]
    return grad

def continuous_opt(h, N, deadline, lr0=0.007, t_soft0=0.035):
    h = np.array(h, dtype=float).copy()
    nh = N / 2.0
    h = fix_sum(h, nh)
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    t_soft = t_soft0
    while time.time() < deadline - 1:
        it += 1
        corr = fft_corr(h)
        peak = corr.max()
        w = np.exp((corr - peak) / t_soft)
        w /= w.sum()
        grad = weighted_grad(h, w, N)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        bc1 = 1.0 - 0.9 ** it
        bc2 = 1.0 - 0.999 ** it
        h = h - lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum(h, nh)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 300 == 0:
            lr *= 0.985
            t_soft = max(0.004, t_soft * 0.94)
        if it % 3000 == 0:
            lr *= 0.9
            t_soft = max(0.004, t_soft * 0.8)
    return best_h, best_c

        T *= cool

                h[oi], h[zi] = h[zi], h[oi]
            cur_max = float(fft_corr(h).max())
            T = max(T, 0.015)
            stall = 0
