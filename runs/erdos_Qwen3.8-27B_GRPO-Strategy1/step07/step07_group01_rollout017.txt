```python
            best_h = h.copy()
            best_N = N

    # Phase 1: Binarize and search initial_h_values
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = ensure_binary(h0, N0, seed)
    c5_bin = compute_c5(h_bin, N0)
    update(h_bin, c5_bin, N0)

    d1 = min(deadline - 300, t0 + 130)
    h_g, c5_g = greedy_search(h_bin, N0, d1, max_sweeps=500)
    update(h_g, c5_g, N0)

    # Phase 2: ILS from best
    if time.time() < deadline - 200:
        d2 = min(deadline - 150, time.time() + 90)
        h_i, c5_i = ils_search(best_h, N0, d2, max_restarts=45, base_time=10)
        update(h_i, c5_i, N0)

    # Phase 3: Try structured patterns at N0
    if time.time() < deadline - 150:
        for period in [8, 12, 16, 24]:
            if time.time() > deadline - 100:
                break
            h_pat = make_periodic(N0, period, 0, seed + period)
            d3 = min(deadline - 80, time.time() + 15)
            h_pr, c5_pr = greedy_search(h_pat, N0, d3, max_sweeps=40)
            update(h_pr, c5_pr, N0)

    # Phase 4: Try N=570
    N_sm = 570
    if time.time() < deadline - 100:
        h_sm = np.interp(np.linspace(0, 2, N_sm, endpoint=False),
                         np.linspace(0, 2, best_N, endpoint=False), best_h)
        h_sm = ensure_binary(h_sm, N_sm, seed + 10)
        d4 = min(deadline - 70, time.time() + 55)
        h_sm2, c5_sm = greedy_search(h_sm, N_sm, d4, max_sweeps=400)
        update(h_sm2, c5_sm, N_sm)
        if time.time() < deadline - 50:
            d4b = min(deadline - 40, time.time() + 30)
            h_sm3, c5_sm3 = ils_search(best_h if best_N == N_sm else h_sm2, N_sm, d4b, max_restarts=20, base_time=8)
            update(h_sm3, c5_sm3, N_sm)

    # Phase 5: Try N=990
    N_md = 990
    if time.time() < deadline - 70:
        h_md = np.interp(np.linspace(0, 2, N_md, endpoint=False),
                         np.linspace(0, 2, best_N, endpoint=False), best_h)
        h_md = ensure_binary(h_md, N_md, seed + 20)
        d5 = min(deadline - 50, time.time() + 40)
        h_md2, c5_md = greedy_search(h_md, N_md, d5, max_sweeps=250)
        update(h_md2, c5_md, N_md)

    # Phase 6: Random restarts at best_N
    if time.time() < deadline - 30:
        for trial in range(5):
            if time.time() > deadline - 10:
                break
            h_r = random_binary(best_N, seed + 200 + trial * 7)
            d6 = min(deadline - 10, time.time() + 18)
            h_r2, c5_r = greedy_search(h_r, best_N, d6, max_sweeps=60)
            update(h_r2, c5_r, best_N)

    # Phase 7: Final intensive ILS
    if time.time() < deadline - 5:
        d7 = deadline - 2
        h_f, c5_f = ils_search(best_h, best_N, d7, max_restarts=30, base_time=5)
        update(h_f, c5_f, best_N)

    final_c5 = compute_c5(best_h, best_N)
    return best_h, final_c5, best_N
```
