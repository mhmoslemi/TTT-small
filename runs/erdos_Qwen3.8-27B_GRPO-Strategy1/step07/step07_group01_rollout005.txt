    target = N // 2

    zeros = np.where(h == 0)[0].astype(int)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N, seed=77):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def downsample(h, N_new):
    N_old = len(h)
    if N_new >= N_old:
        return h.copy()
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def upsample(h, N_new):
    N_old = len(h)
    if N_new <= N_old:
        return h.copy()
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def build_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

    P = float(c.max())
    best_P = P
    best_h = h.copy()
    ones = np.where(h == 1)[0].astype(int)
    zeros = np.where(h == 0)[0].astype(int)
    total_moves = moves_per_sweep * max_sweeps
    moves_done = 0
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        D = build_D(h, N)
        for m in range(moves_per_sweep):
            if time.time() > deadline:
                break

            prog = min(1.0, moves_done / total_moves)
            Tc = T0 * (T1 / T0) ** prog

            qi = np.random.randint(len(zeros))

                c = new_c

                h[q] = 1.0

                c = new_c
