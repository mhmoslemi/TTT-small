def project(h, target_sum):

    if abs(s - target_sum) < 1e-10:

        if np.clip(h - mid, 0, 1).sum() >= target_sum:
            lo = mid
        else:

def fix_binary(h, rng):

import numpy as np
import time

def compute_c5_full(h):
    N = len(h)
    L = 2 * N - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h, L) * np.conj(np.fft.fft(1.0 - h, L))))
    return float(np.max(corr)) * (2.0 / N), corr

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() >= target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h, rng):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def subgrad(h, corr, center, n_lags):
    N = len(h)
    top = np.argsort(corr)[-n_lags:]
    g = np.zeros(N)
    for m in top:
        L = int(m) - center
        if L == 0:
            g += 1.0 - 2.0 * h
        elif 0 < L < N:
            g[:N - L] += 1.0 - h[L:]
            g[L:] -= h[:N - L]
        elif -N < L < 0:
            Lp = -L
            g[Lp:] += 1.0 - h[:N - Lp]
            g[:N - Lp] -= h[Lp:]
    return g / n_lags

    h = h.astype(float).copy()

    best_h = h.copy()

        h = h_new
        t_k = t_new

    best_c = compute_c5_full(h)[0]
    center = N - 1
    Lt = 2 * N - 1
    no_impr = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = np.real(np.fft.ifft(np.fft.fft(h, Lt) * np.conj(np.fft.fft(1.0 - h, Lt))))
        cur_max = float(np.max(corr))
        c = cur_max * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_impr = 0
        else:
            no_impr += 1
        if no_impr > 250:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argsort(corr)[-n_top:]
        corr_top = corr[top_idx].copy()
        top_lags = (top_idx - center).astype(int)
        eff = compute_effects(h, top_lags, N)
        eff_o = eff[ones]
        eff_z = eff[zeros]
        n_o, n_z = len(oner_list) if False else len(ones), len(zeros)
        best_val = cur_max + 1.0
        best_i, best_j = -1, -1
        chunk = 40
        for ist in range(0, n_o, chunk):
            ied = min(ist + chunk, n_o)
            vals = corr_top[None, None, :] + eff_o[ist:ied, None, :] - eff_z[None, :, :]
            mx = vals.max(axis=2)
            fi = int(np.argmin(mx))
            ci2, cj2 = fi // n_z, fi % n_z
            if mx[ci2, cj2] < best_val:
                best_val = mx[ci2, cj2]
                best_i = ist + ci2
                best_j = cj2
        if best_val < cur_max - 1e-8 and best_i >= 0:
            h[ones[best_i]] = 0
            h[zeros[best_j]] = 1
        elif rng.random() < 0.25:
            ri = int(rng.integers(n_o))
            rj = int(rng.integers(n_z))
            h[ones[ri]] = 0
            h[zeros[rj]] = 1
    return best_h, best_c

def binary_2swap(h, deadline, rng, n_top=40, n_trials=200):
    N = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5_full(h)[0]
    center = N - 1
    Lt = 2 * N - 1
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        i1, i2 = rng.choice(ones, 2, replace=False)
        j1, j2 = rng.choice(zeros, 2, replace=False)
        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
        corr = np.real(np.fft.ifft(np.fft.fft(h, Lt) * np.conj(np.fft.fft(1.0 - h, Lt))))
        c = float(np.max(corr)) * (2.0 / N)
        if c > best_c:
            h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
        else:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5_full(best_h)[0]

    d1 = min(deadline - 550, t0 + 130)
    h1 = project(h0 + rng.standard_normal(N0) * 0.04, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, rng, n_lags=70, lr0=0.010)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    if time.time() < deadline - 400:
        d2 = min(deadline - 380, time.time() + 200)
        hb = fix_binary(best_h, rng)
        hb, c2 = binary_fast(hb, d2, rng, n_top=50, max_sweeps=6000)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    if time.time() < deadline - 280:
        d3 = min(deadline - 260, time.time() + 80)
        h3 = project(best_h + rng.standard_normal(N0) * 0.02, N0 / 2.0)
        h3, c3 = cont_opt(h3, d3, rng, n_lags=90, lr0=0.006)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    if time.time() < deadline - 150:
        d4 = min(deadline - 130, time.time() + 100)
        hb4 = fix_binary(best_h, rng)
        hb4, c4 = binary_fast(hb4, d4, rng, n_top=60, max_sweeps=4000)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()

    if time.time() < deadline - 80:
        d5 = min(deadline - 60, time.time() + 40)
        hb2 = binary_2swap(best_h, d5, rng, n_top=40, n_trials=300)
        if hb2[1] < best_c:
            best_c, best_h = hb2[1], hb2[0].copy()

    if time.time() < deadline - 60:
        for Nt in [420, 460, 500]:
            if time.time() > deadline - 50:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, Nt, endpoint=False)
            h_int = np.interp(x_new, x_old, best_h)
            h_bt = fix_binary(h_int, rng)
            d6 = min(deadline - 30, time.time() + 50)
            h_bt, c6 = binary_fast(h_bt, d6, rng, n_top=45, max_sweeps=2000)
            if c6 < best_c:
                best_c, best_h = c6, h_bt.copy()

    if time.time() < deadline - 5:
        hB = fix_binary(best_h, rng)
        d7 = min(deadline - 2, time.time() + 30)
        hB, c7 = binary_fast(hB, d7, rng, n_top=55, max_sweeps=3000)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()

    final_c = compute_c5_full(best_h)[0]
    hB = fix_binary(best_h, rng)
    cB = compute_c5_full(hB)[0]
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
