For 2-swap (p1,p2 → 0, q1,q2 → 1):
- Change: D[p1] + D[p2] - D[q1] - D[q2]
- New max: max(corr + D[p1] + D[p2] - D[q1] - D[q2])

The previous code computes D once per sweep (O(N^2)) and then evaluates swaps. This is O(N^2) per sweep for D computation, then O(N) per single swap eval, O(N) per 2-swap eval.

For N=380: D computation is O(380 * 759) ≈ 288K ops. Per sweep with 190 single swaps evaluated: 190 * 759 ≈ 144K. 2-swap with 100 tries: 100 * 759 ≈ 76K. Total per sweep: ~500K ops. In 1000 sweeps: 500M ops. At ~100M ops/s in numpy: ~5 seconds. Fine.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0 - h), L)))

def compute_c5(h):
    return float(np.max(fft_corr(h)) * (2.0 / len(h)))

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    s = float(h.sum())
    if abs(s - target) < 1e-12: return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0, 1).sum()) > target: lo = mid
        else: hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def ensure_binary(h, N):
    target = N // 2
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == target: return hb
    rng = np.random.RandomState(42)
    if n1 > target:
        idx = np.where(hb==1)[0]
        idx = idx[np.argsort(h[idx])[::-1]]  # remove smallest values first
        for k in range(n1-target): hb[idx[k]] = 0
    else:
        idx = np.where(hb==0)[0]
        idx = idx[np.argsort(1-h[idx])[::-1]]  # add largest values first
        for k in range(target-n1): hb[idx[k]] = 1
    return hb

def bin_anneal(h, N, deadline, rng, max_sweeps=500):
    hb = ensure_binary(h, N).copy()
    corr0 = fft_corr(hb)
    best_c = float(corr0.max())
    best_hb = hb.copy()
    c0 = corr0.copy()
    temp = 0.01
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > deadline: break
        temp = max(0.002, temp * 0.998)
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        if len(ones) < 2 or len(zeros) < 2: break
        # Compute D
        D = compute_D(hb, N)
        P = float(c0.max())
        best_improve = P
        best_move = None
        # Single swaps
        for p in ones:
            base = c0 - D[p]
            cands = base[None,:] - D[zeros]
            mval = cands.max(axis=1)
            mi = int(mval.argmin())
            if mval[mi] < best_improve:
                best_improve = float(mval[mi])
                best_move = ('1', p, int(zeros[mi]))
        # 2-swaps
        n_try = min(200, N)
        for _ in range(n_try):
            i1, i2 = rng.randint(0, len(ones), 2)
            j1, j2 = rng.randint(0, len(zeros), 2)
            if i1==i2 or j1==j2: continue
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            nv = float((c0 + D[p1] + D[p2] - D[q1] - D[q2]).max())
            if nv < best_improve:
                best_improve = nv
                best_move = ('2', p1, p2, q1, q2)
        if best_move is not None:
            if best_move[0] == '1':
                hb[best_move[1]] = 0; hb[best_move[2]] = 1
            else:
                hb[best_move[1]] = 0; hb[best_move[2]] = 0
                hb[best_move[3]] = 1; hb[best_move[4]] = 1
            c0 = fft_corr(hb)
            P = float(c0.max())
            if P < best_c - 1e-14:
                best_c = P; best_hb = hb.copy(); stall = 0
            else: stall += 1
        else:

                    ones = np.where(hb==1)[0]; zeros = np.where(hb==0)[0]
                c0 = fft_corr(hb); stall = 0
            elif stall > 8: break
    return best_hb, best_c * (2.0/N)

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:,None]
    pm = pos - shifts[None,:]
    pp = pos + shifts[None,:]
    vm = (pm>=0)&(pm<N); vp = (pp>=0)&(pp<N)
    pmc = np.clip(pm,0,N-1); ppc = np.clip(pp,0,N-1)
    return -(1-h[pmc])*vm + h[ppc]*vp

def nesterov_soft(h, N, deadline, alpha_start=50, alpha_end=500, n_iter=None):
    """Nesterov with soft-max gradient."""
    h = h.astype(float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    beta = 0.9
    it = 0
    lr = 0.008
    if n_iter is None:
        n_iter = 10**9
    while it < n_iter and time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-13:
            best_pk = pk; best_h = y.copy()
        # Soft-max gradient
        a = alpha_start + (alpha_end - alpha_start) * min(1.0, it/300.0)
        sh = corr - corr.max()
        w = np.exp(a * sh)
        w /= w.sum()
        # Compute gradient via weighted sum of per-shift gradients
        # grad[j] = sum_m w[m] * [(1-y[j+L_m])*1[valid] - y[j-L_m]*1[valid]]
        # = sum_m w[m]*(1-y[j+L_m])*1[valid] - sum_m w[m]*y[j-L_m]*1[valid]
        # First term: sum_m w[m]*1[valid] - sum_m w[m]*y[j+L_m]*1[valid]
        # Second term: sum_m w[m]*y[j-L_m]*1[valid]
        # Can compute via convolution/FFT
        
        # Simpler: just do it as a loop over significant weights
        center = N - 1
        grad = np.zeros(N)
        for m in range(len(corr)):
            if w[m] < 1e-10: continue
            L = m - center
            if L == 0:
                grad += w[m] * (1.0 - 2.0*y)
            elif L > 0:
                nv = N - L
                grad[:nv] += w[m] * (1.0 - y[L:])
                grad[L:] -= w[m] * y[:nv]
            else:
                M = -L
                nv = N - M
                grad[:nv] += w[m] * (1.0 - y[M:])
                grad[M:] -= w[m] * y[:nv]
        grad /= a  # scale by 1/alpha for the log-sum-exp
        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.97
            beta = min(0.95, 0.85 + it*0.0002)
    return best_h, best_pk*dx
