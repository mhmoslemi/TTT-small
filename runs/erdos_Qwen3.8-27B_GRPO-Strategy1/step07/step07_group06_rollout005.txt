def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):

        else:

    e = np.exp(T * (c - mx))

    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def bin_search(h, N, deadline, max_iters=500, ntri=25, anneal=0.0, kswap=1):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        for i in np.random.choice(idx, n1 - nh, replace=False):
            h[i] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        for i in np.random.choice(idx, nh - n1, replace=False):
            h[i] = 1
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stall = 0
    temp = anneal
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, o_move, z_move = 0.0, None, None
        for _ in range(ntri):
            ks = min(kswap, len(ones), len(zeros))
            so = np.random.choice(len(ones), ks, replace=False)
            sz = np.random.choice(len(zeros), ks, replace=False)
            o_list = [int(ones[x]) for x in so]
            z_list = [int(zeros[x]) for x in sz]
            for p in o_list:
                h[p] = 0.0
            for p in z_list:
                h[p] = 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd = d
                o_move, z_move = o_list[:], z_list[:]
            for p in o_list:
                h[p] = 1.0
            for p in z_list:
                h[p] = 0.0
        if o_move is not None:
            for p in o_move:
                h[p] = 0.0
            for p in z_move:
                h[p] = 1.0
            corr = fft_corr(h)
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))
    return best_h, best_c

def subgrad_topk(h, N, deadline, K=5, alpha0=0.03, beta=0.92):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    ctr = N - 1
    while time.time() < deadline and it < 6000:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(N)
        for ai in idx:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= K
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            alpha *= 0.95
            beta = min(0.98, 0.90 + 0.0003 * it)
    return best_h, best_pk * (2.0 / N)

def lbfgs_run(h, N, deadline, T=30.0, max_outer=4):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_and_grad(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_and_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 120, 'maxfun': 300, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj_and_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.5
    return best_h, best_v * (2.0 / N)

def try_resolutions(best_h, best_c5, deadline):
    x_b = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [180, 220, 260, 300, 340, 420, 460, 520, 620, 760, 880]:
        if time.time() > deadline:
            break
        if Nt % 2:
            Nt += 1
        if Nt == len(best_h):
            continue
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_b, best_h)
        ht = (hi > 0.5).astype(float)
        nh = Nt // 2
        n1 = int(ht.sum())
        if n1 > nh:
            idx = np.where(ht == 1)[0]
            for i in np.random.choice(idx, n1 - nh, replace=False):
                ht[i] = 0
        elif n1 < nh:
            idx = np.where(ht == 0)[0]
            for i in np.random.choice(idx, nh - n1, replace=False):
                ht[i] = 1
        sd = min(deadline, time.time() + 22)
        hr, _ = bin_search(ht, Nt, sd, max_iters=120, ntri=16, kswap=1)
        c5r = compute_c5(hr, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_b = np.linspace(0, 2, Nt, endpoint=False)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Quick 1-swap
    d1 = min(dl - 500, t0 + 40)
    hb, _ = bin_search(best_h, N0, d1, max_iters=100, ntri=20, kswap=1)
    c = compute_c5(hb, N0)
    if c < best_c5:
        best_c5, best_h = c, hb.copy()

    # Phase 2: 2-swap (key new move)
    if time.time() < dl - 420:
        d2 = min(dl - 370, time.time() + 110)
        hb2, _ = bin_search(best_h, N0, d2, max_iters=250, ntri=14, kswap=2)
        c2 = compute_c5(hb2, N0)
        if c2 < best_c5:
            best_c5, best_h = c2, hb2.copy()

    # Phase 3: L-BFGS with varying T
    if time.time() < dl - 300:
        d3 = min(dl - 250, time.time() + 120)
        for r in range(3):
            if time.time() > d3:
                break
            hr = best_h + np.random.randn(N0) * (0.15 - 0.03 * r)
            hr = proj01(hr, N0 / 2.0)
            hl, cl = lbfgs_run(hr, N0, d3, T=20.0 + 15.0 * r, max_outer=3)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 4: Top-K subgradient
    if time.time() < dl - 200:
        d4 = min(dl - 160, time.time() + 80)
        hs = best_h + np.random.randn(N0) * 0.06
        hs = proj01(hs, N0 / 2.0)
        h4, c4 = subgrad_topk(hs, N0, d4, K=5, alpha0=0.025, beta=0.93)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: 3-swap annealed
    if time.time() < dl - 140:
        d5 = min(dl - 100, time.time() + 60)
        h5, _ = bin_search(best_h, N0, d5, max_iters=150, ntri=10, kswap=3, anneal=0.003)
        c5v = compute_c5(h5, N0)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()

    # Phase 6: Multi-resolution
    if time.time() < dl - 80:
        best_h, best_c5 = try_resolutions(best_h, best_c5, min(dl - 40, time.time() + 130))

    # Phase 7: Final 1-swap polish
    if time.time() < dl - 25:
        d7 = min(dl - 12, time.time() + 55)
        h7, _ = bin_search(best_h, len(best_h), d7, max_iters=300, ntri=25, kswap=1)
        c7 = compute_c5(h7, len(best_h))
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Final: compare continuous vs binary
    fc5 = compute_c5(best_h, len(best_h))
    fb = (best_h > 0.5).astype(float)
    nf = int(fb.sum())
    tf = len(fb) // 2
    if nf > tf:
        idx = np.where(fb == 1)[0]
        for i in np.random.choice(idx, nf - tf, replace=False):
            fb[i] = 0
    elif nf < tf:
        idx = np.where(fb == 0)[0]
        for i in np.random.choice(idx, tf - nf, replace=False):
            fb[i] = 1
    fc5b = compute_c5(fb, len(fb))
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    if time.time() < dl:
        hf, _ = bin_search(best_h, len(best_h), dl, max_iters=40, ntri=20, kswap=1)
        cf = compute_c5(hf, len(best_h))
        if cf < fc5:
            best_h, fc5 = hf, cf
    return best_h, fc5, len(best_h)
