import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_single(h, N, idx):
    L = 2 * N - 1
    k_vals = np.arange(L) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def compute_C_mat(h, N):
    C = np.empty((N, 2*N-1))
    for i in range(N):
        C[i] = compute_C_single(h, N, i)
    return C

def fix_sum_binary(h, target_sum):
    ones = int(h.sum())
    if ones > target_sum:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:ones-target_sum]] = 0
    elif ones < target_sum:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target_sum-ones]] = 1
    return h
