import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_smooth(h, N, deadline, T=30.0, max_outer=4):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_and_grad(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_and_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj_and_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.0
    return best_h, best_v * (2.0 / N)

def bin_search_v2(h, N, deadline, max_iters=600, ntri=30, anneal=0.002, multi=2):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    rng = np.random.default_rng(42)
    if n1 > nh:
        idx = rng.choice(np.where(h == 1)[0], n1 - nh, replace=False)
        h[idx] = 0
    elif n1 < nh:
        idx = rng.choice(np.where(h == 0)[0], nh - n1, replace=False)
        h[idx] = 1
    
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    temp = anneal
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
            
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

            j = int(rng.integers(len(zeros)))

            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:

                best_move = (oi, zi, None, None)
            h[oi], h[zi] = 1.0, 0.0

        if multi >= 2:
            for _ in range(ntri // 2):
                i1, i2 = int(rng.integers(len(ones))), int(rng.integers(len(ones)))
                j1, j2 = int(rng.integers(len(zeros))), int(rng.integers(len(zeros)))
                if i1 == i2 or j1 == j2:
                    continue
                oi1, oi2 = int(ones[i1]), int(ones[i2])
                zj1, zj2 = int(zeros[j1]), int(zeros[j2])
                h[oi1], h[oi2], h[zj1], h[zj2] = 0.0, 0.0, 1.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > bd + temp:
                    bd = d
                    best_move = (oi1, oi2, zj1, zj2)
                h[oi1], h[oi2], h[zj1], h[zj2] = 1.0, 1.0, 0.0, 0.0
        
        if best_move is not None:
            m1, m2, m3, m4 = best_move
            if m3 is None:
                h[m1], h[m2] = 0.0, 1.0
            else:
                h[m1], h[m2], h[m3], h[m4] = 0.0, 0.0, 1.0, 1.0
            improved = True
        
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))
    
    return best_h, best_c

def grad_binary(h, N, deadline, max_iters=3000):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    rng = np.random.default_rng(123)
    if n1 > nh:
        idx = rng.choice(np.where(h == 1)[0], n1 - nh, replace=False)
        h[idx] = 0
    elif n1 < nh:
        idx = rng.choice(np.where(h == 0)[0], nh - n1, replace=False)
        h[idx] = 1
    
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Find active lags
        active = np.where(corr >= P - 1e-4)[0]
        
        # Compute gradient
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), h, N)
        
        # Find best swap guided by gradient
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Sort ones by gradient (most negative = most to flip)
        # Sort zeros by gradient (most positive = most to flip)
        one_grads = grad[ones]
        zero_grads = grad[zeros]
        
        candidates = 20
        top_ones = ones[np.argsort(one_grads)[:candidates]]
        top_zeros = zeros[np.argsort(zero_grads)[::-1][:candidates]]
        
        bd, best_move = 0.0, None
        for i in top_ones[:5]:
            for j in top_zeros[:5]:
                h[i], h[j] = 0.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > bd:
                    bd = d
                    best_move = (int(i), int(j))
                h[i], h[j] = 1.0, 0.0
        
        if best_move is not None and bd > 0:
            h[best_move[0]], h[best_move[1]] = 0.0, 1.0
        else:
            # Random perturbation
            i = int(rng.integers(len(ones)))
            j = int(rng.integers(len(zeros)))
            h[int(ones[i])], h[int(zeros[j])] = 0.0, 1.0
    
    return best_h, best_c

        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(1.0 / T * (c - mx))
    w = e / e.sum()
    val = mx + T * np.log(e.sum())
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_smooth(h, N, deadline, T_scale=1.0, max_outer=4):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_and_grad(h, 1.0 / T_scale, target, 0.0)[0]
    lam = 3.0
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        T = T_scale * (1.0 + 0.5 * outer)
        res = sp_min(smooth_obj_and_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 250, 'maxfun': 600, 'ftol': 1e-15})
        h = proj01(res.x, target)
        v = smooth_obj_and_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 1.8
    return best_h, best_v * (2.0 / N)

def grad_binary(h, N, deadline, max_iters=4000, rng_seed=7):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    rng = np.random.default_rng(rng_seed)
    if n1 > nh:
        idx = rng.choice(np.where(h == 1)[0], n1 - nh, replace=False)
        h[idx] = 0
    elif n1 < nh:
        idx = rng.choice(np.where(h == 0)[0], nh - n1, replace=False)
        h[idx] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        active = np.where(corr >= P - 2e-4)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        og = grad[ones]
        zg = grad[zeros]
        n1c = min(8, len(ones))
        n0c = min(8, len(zeros))
        ti = ones[np.argsort(og)[:n1c]]
        tj = zeros[np.argsort(zg)[::-1][:n0c]]
        bd, bm = 0.0, None
        for i in ti:
            for j in tj:
                h[i], h[j] = 0.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > bd:
                    bd, bm = d, (int(i), int(j))
                h[i], h[j] = 1.0, 0.0
        if bm is not None and bd > 0:
            h[bm[0]], h[bm[1]] = 0.0, 1.0
            improved = True
        else:
            i = int(rng.integers(len(ones)))
            j = int(rng.integers(len(zeros)))
            h[int(ones[i])], h[int(zeros[j])] = 0.0, 1.0
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c

def multi_swap_binary(h, N, deadline, max_iters=400, ntri=20, anneal=0.003, rng_seed=99):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    rng = np.random.default_rng(rng_seed)
    if n1 > nh:
        idx = rng.choice(np.where(h == 1)[0], n1 - nh, replace=False)
        h[idx] = 0
    elif n1 < nh:
        idx = rng.choice(np.where(h == 0)[0], nh - n1, replace=False)
        h[idx] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    temp = anneal
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 180:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        bd, bm = 0.0, None
        for _ in range(ntri):
            i1, j1 = int(rng.integers(len(ones))), int(rng.integers(len(zeros)))
            oi, zi = int(ones[i1]), int(zeros[j1])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd, bm = d, (oi, zi, -1, -1)
            h[oi], h[zi] = 1.0, 0.0
        for _ in range(ntri // 2):
            i1, i2 = int(rng.integers(len(ones))), int(rng.integers(len(ones)))
            j1, j2 = int(rng.integers(len(zeros))), int(rng.integers(len(zeros)))
            if i1 == i2 or j1 == j2:
                continue
            oi1, oi2 = int(ones[i1]), int(ones[i2])
            zj1, zj2 = int(zeros[j1]), int(zeros[j2])
            h[oi1], h[oi2], h[zj1], h[zj2] = 0.0, 0.0, 1.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd, bm = d, (oi1, oi2, zj1, zj2)
            h[oi1], h[oi2], h[zj1], h[zj2] = 1.0, 1.0, 0.0, 0.0
        if bm is not None:
            if bm[2] == -1:
                h[bm[0]], h[bm[1]] = 0.0, 1.0
            else:
                h[bm[0]], h[bm[1]], h[bm[2]], h[bm[3]] = 0.0, 0.0, 1.0, 1.0
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))
    return best_h, best_c

def subgrad_ne(h, N, deadline, alpha0=0.03, beta=0.93, atol=1e-5):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    alpha = alpha0
    for it in range(6000):
        if time.time() > deadline:
            break
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= max(1, len(active))
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 300 == 0:
            alpha *= 0.95
            beta = min(0.97, 0.88 + 0.0003 * it)
    return best_h, best_pk * (2.0 / N)

def try_resolutions(best_h, best_c5, deadline, rng):
    x_b = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [340, 360, 400, 420, 460, 500, 520, 560, 600, 640, 700, 800]:
        if time.time() > deadline:
            break
        if Nt % 2:
            Nt += 1
        if Nt == len(best_h):
            continue
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_b, best_h)
        ht = (hi > 0.5).astype(float)
        nh = Nt // 2
        n1 = int(ht.sum())
        if n1 > nh:
            ht[rng.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0
        elif n1 < nh:
            ht[rng.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1
        sd = min(deadline, time.time() + 25)
        hr, _ = multi_swap_binary(ht, Nt, sd, max_iters=100, ntri=15, rng_seed=Nt)
        c5r = compute_c5(hr, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_b = xn
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 4.0
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: L-BFGS smooth on initial
    d1 = min(dl - 500, t0 + 70)
    h1, c1 = lbfgs_smooth(best_h, N0, d1, T_scale=35.0, max_outer=3)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Gradient-guided binary
    d2 = min(dl - 400, time.time() + 120)
    hb, cb = grad_binary(best_h, N0, d2, max_iters=3000, rng_seed=7)
    cb5 = cb * (2.0 / N0)
    if cb5 < best_c5:
        best_c5, best_h = cb5, hb.copy()

    # Phase 3: L-BFGS from binary (key: continuous refinement of best binary)
    d3 = min(dl - 300, time.time() + 150)
    h3 = hb.astype(float).copy()
    h3, c3 = lbfgs_smooth(h3, N0, d3, T_scale=50.0, max_outer=3)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: Multi-start L-BFGS
    d4 = min(dl - 250, time.time() + 120)
    for r in range(3):
        if time.time() > d4:
            break
        hr = best_h + np.random.randn(N0) * (0.12 - 0.03 * r)
        hr = proj01(hr, N0 / 2.0)
        hl, cl = lbfgs_smooth(hr, N0, d4, T_scale=40.0 + 10.0 * r, max_outer=2)
        if cl < best_c5:
            best_c5, best_h = cl, hl.copy()

    # Phase 5: Subgradient Nesterov
    if time.time() < dl - 180:
        d5 = min(dl - 140, time.time() + 90)
        hs = best_h + np.random.randn(N0) * 0.06
        hs = proj01(hs, N0 / 2.0)
        hsn, cn = subgrad_ne(hs, N0, d5, alpha0=0.025, beta=0.94, atol=5e-5)
        if cn < best_c5:
            best_c5, best_h = cn, hsn.copy()

    # Phase 6: Multi-swap binary
    if time.time() < dl - 120:
        d6 = min(dl - 80, time.time() + 80)
        hm, cm = multi_swap_binary(best_h, N0, d6, max_iters=300, ntri=20, anneal=0.002, rng_seed=111)
        cm5 = cm * (2.0 / N0)
        if cm5 < best_c5:
            best_c5, best_h = cm5, hm.copy()

    # Phase 7: Resolution changes
    if time.time() < dl - 60:
        best_h, best_c5 = try_resolutions(best_h, best_c5, min(dl - 30, time.time() + 100), rng)

    # Phase 8: Final L-BFGS polish
    if time.time() < dl - 20:
        d8 = min(dl - 10, time.time() + 50)
        h8 = best_h + np.random.randn(len(best_h)) * 0.04
        h8 = proj01(h8, len(best_h) / 2.0)
        hl8, cl8 = lbfgs_smooth(h8, len(best_h), d8, T_scale=60.0, max_outer=2)
        if cl8 < best_c5:
            best_c5, best_h = cl8, hl8.copy()

    # Final: compare continuous vs binary at the current N
    Nf = len(best_h)
    fc5 = compute_c5(best_h, Nf)
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, Nf)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    # Ensure constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    if best_h.sum() != Nf / 2.0:
        best_h = proj01(best_h, Nf / 2.0)
        fc5 = compute_c5(best_h, Nf)
    return best_h, fc5, Nf
