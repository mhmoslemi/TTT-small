import numpy as np

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_lse(h, corr, beta, k=60):
    N = len(h)
    center = N - 1
    max_c = np.max(corr)
    exp_c = np.exp(np.clip(beta * (corr - max_c), -400, 400))
    w = exp_c / exp_c.sum()
    top_idx = np.argsort(w)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        wi = w[m]
        L = int(m) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            hi = np.arange(L, N)
            lo = np.arange(0, N - L)
            grad[hi] += wi * (1.0 - h[hi - L])
            grad[lo] -= wi * h[lo + L]
        elif -N < L < 0:
            L2 = -L
            hi = np.arange(L2, N)
            lo = np.arange(0, N - L2)
            grad[hi] += wi * (1.0 - h[hi - L2])
            grad[lo] -= wi * h[lo + L2]
    return grad / beta

    tgt = N / 2.0
    h = h.astype(float).copy()

    best_h = h.copy()
    best_c = compute_c5(h)
    beta = beta0
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 40:
            beta = min(beta * 1.2, beta_max)
            lr0 *= 0.88
            stale = 0
        g = grad_lse(h, corr, beta)
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-12:
            g = g / gnorm * 0.4
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.75 * v + 0.35 * (h_new - h)
        h = h_new
    return best_h, best_c

def binary_il(h, deadline, max_iters=5000, n_crit=30, s_sz=45):
