So: d/dh[i] corr[k] = (1-h[i+k]) * 1[i+k in range] + h[i-k] * 1[i-k in range]

If k* = argmax corr, then grad[i] = (1-h[i+k*]) * 1[i+k* valid] + h[i-k*] * 1[i-k* valid]

def grad_c5(h, N, topk=30):
    c = fft_corr(h)
    L = 2*N - 1
    center = N - 1
    top = np.argpartition(c, -topk)[-topk:]
    grad = np.zeros(N)
    for ai in top:
        k = int(ai - center)
        # d/dh[i] corr[k] = (1-h[i+k])*[i+k in [0,N)) + h[i-k]*[i-k in [0,N))
        if k == 0:
            grad += (1.0 - 2.0*h)
        elif k > 0:
            grad[:N-k] += (1.0 - h[k:])  # (1-h[i+k]) for i in [0, N-k)
            grad[k:] += h[:N-k]  # h[i-k] for i in [k, N)
        else:
            m = -k
            grad[:N-m] += (1.0 - h[m:])  # (1-h[i+k]) where i+k=i-m for i in [0,N-m)
            grad[m:] += h[:N-m]  # h[i-k]=h[i+m] for i in [m, N)
    grad /= len(top)
    return grad, c.max()

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))[:L]

def c5_val(h):

        else:
