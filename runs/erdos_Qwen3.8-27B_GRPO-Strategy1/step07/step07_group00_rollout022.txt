def lse_smooth(corr, temp):
    """Log-sum-exp smoothed max."""
    corr_shifted = corr - corr.max()
    return corr.max() + temp * np.log(np.sum(np.exp(corr_shifted / temp)))

def lse_smooth_grad(h, N, temp, k_top=50):

    corr = fft_corr(h)
    smooothed_max = lse_smooth(corr, temp)
