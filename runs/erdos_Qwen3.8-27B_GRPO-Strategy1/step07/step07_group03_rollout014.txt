Let me also think about a completely different angle:

The problem is essentially: find a binary sequence of length N with N/2 ones such that the cross-correlation with (1-sequence) is minimized in the max sense.

For the cross-correlation at shift s:
corr(s) = sum_i h[i] * (1-h[i+s])

For binary h with N/2 ones:
At shift 0: corr(0) = 0 (since h[i]*(1-h[i]) = 0)
At shift s ≠ 0: corr(s) = (N/2 - |mask|) - (number of 1-1 overlaps at shift s)
where |mask| is the size of the overlap region.

Wait, more precisely:
corr(s) = sum_{i: i,i+s both in [0,N)} h[i] * (1-h[i+s])
= sum_{i: i,i+s both in [0,N), h[i]=1} 1 - sum_{i: i,i+s both in [0,N), h[i]=1, h[i+s]=1} 1
= (number of 1s in the "valid i" range) - (number of 1-1 pairs at shift s)

The "valid i" range for shift s > 0 is i in [0, N-s), size N-s.
For shift s < 0, it's i in [-s, N), size N+s = N-|s|.

So corr(s) = (number of 1s in valid range) - autocorr(s)

To minimize max corr(s), we want:
- The number of 1s in each "valid range" to be as balanced as possible (≈ (N-s)/2 for shift s)
- The autocorrelation to be as large as possible at each shift (maximize 1-1 overlaps)

These two goals are somewhat in tension.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N, rng=None):
    nh = N // 2
    h = np.asarray(h, dtype=float)
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    if rng is None:
        rng = np.random.RandomState(42)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_delta(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    return (1.0 - h[pp_c]) * vp - h[pm_c] * vm

def local_search(h, N, deadline, rng, max_sweeps=2000):

    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_c = P_cur
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        delta = compute_delta(h, N)
        D_one = delta[ones, :]
        D_zero = delta[zeros, :]
        N1 = len(ones)
        N0 = len(zeros)
        best_new_P = P_cur
        best_pq = None
        n_try = min(N1, 100)

        for pi in p_idx:
            base = c_cur - D_one[pi]
            cand = (base[None, :] + D_zero).max(axis=1)
            mi = int(np.argmin(cand))
            if cand[mi] < best_new_P:
                best_new_P = float(cand[mi])
                best_pq = (int(ones[pi]), int(zeros[mi]))
        best_2swap = None
        if N1 >= 2 and N0 >= 2 and rng.random() < 0.4:

def nesterov_opt(h, N, deadline, k=60, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
            beta = min(0.97, 0.85 + it * 0.00008)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def downsample(h_old, N_old, N_new):

    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
