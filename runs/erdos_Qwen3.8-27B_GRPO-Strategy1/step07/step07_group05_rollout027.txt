                h[oi], h[zi] = 1, 0
        else:
            one_starts = np.where(h[:-L] & (1 - h[L:]))[0]
            zero_starts = np.where((1 - h[:-L]) & h[L:])[0]
            if len(one_starts) == 0 or len(zero_starts) == 0:
                T *= cool
                continue
            os_ = int(np.random.choice(one_starts))
            zs = int(np.random.choice(zero_starts))
            h[os_:os_ + L] = 0
            h[zs:zs + L] = 1
            nm = float(fft_corr(h).max())
            dE = nm - cur_max
            if dE < -1e-14 or np.random.random() < np.exp(-max(dE, 0) / max(T, 1e-10)):
                cur_max = nm
                if nm < best_c:
                    best_c = nm
                    best_h = h.copy()
            else:
                h[os_:os_ + L] = 1
                h[zs:zs + L] = 0
        T *= cool
    return best_h, best_c * (2.0 / N)

def fine_tune(h, N, deadline):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_v = np.max(fft_corr(h))
    lr = 0.001
    for _ in range(5000):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        active = np.where(corr > corr.max() - 2.0)[0]
        if len(active) == 0:
            break
        w = np.zeros(len(corr))
        w[active] = 1.0 / len(active)
        grad = compute_grad(h, N, active, w)
        h_new = project_sum(h - lr * grad, N / 2.0)
        v_new = np.max(fft_corr(h_new))
        if v_new < best_v:
            best_v = v_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.995
    return project_sum(best_h, N / 2.0), best_v * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    target = N / 2.0
    c0 = compute_c5(h0, N)
    best_h = h0.copy()
    best_c5 = c0

    # Phase 1: Binary SA on initial
    h_bin = (h0 > 0.5).astype(float)
    dl1 = min(deadline - 600, t0 + 150)
    h_b, c_b = binary_sa(h_bin, N, dl1, T0=0.07, max_iters=60000)
    if c_b < best_c5:
        best_c5 = c_b
        best_h = h_b.copy()

    # Phase 2: Continuous descent (primary)
    dl2 = min(deadline - 250, t0 + 450)
    h_c, c_c = continuous_descent(best_h, N, dl2, beta_init=50, lr_init=0.01, lr_decay=0.993, n_top=25)
    if c_c < best_c5:
        best_c5 = c_c
        best_h = h_c.copy()

    # Phase 3: Second continuous pass with perturbation
    if time.time() < deadline - 150:
        h_p = project_sum(best_h + np.random.randn(N) * 0.015, target)
        dl3 = min(deadline - 120, time.time() + 180)
        h_c2, c_c2 = continuous_descent(h_p, N, dl3, beta_init=100, lr_init=0.004, lr_decay=0.995, n_top=15)
        if c_c2 < best_c5:
            best_c5 = c_c2
            best_h = h_c2.copy()

    # Phase 4: Fine-tune the best
    if time.time() < deadline - 60:
        dl4 = min(deadline - 30, time.time() + 80)
        h_ft, c_ft = fine_tune(best_h, N, dl4)
        if c_ft < best_c5:
            best_c5 = c_ft
            best_h = h_ft.copy()

    # Phase 5: Final binary check
    if time.time() < deadline - 10:
        h_bf = project_sum(best_h, target)
        h_bin2 = (h_bf > 0.5).astype(float)
        h_f, c_f = binary_sa(h_bin2, N, deadline - 1, T0=0.01, max_iters=30000, block_sizes=(1, 1, 1, 2, 2))
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    # Validate output
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target) > 0.01:
        best_h = project_sum(best_h, target)
    final_c5 = compute_c5(best_h, N)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, N
