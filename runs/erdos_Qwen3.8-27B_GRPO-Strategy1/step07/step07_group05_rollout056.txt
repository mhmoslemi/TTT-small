    """Compute change in all lags when h[i]: 1→0 and h[j]: 0→1"""
    delta = np.zeros(2*N - 1)
    center = N - 1

        ik = i + L
        if 0 <= ik < N:

    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def project_binary(h, target_ones):
    idx = np.argsort(-h)
    out = np.zeros(len(h))
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy()
    n1 = int(round(h.sum()))
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def swap_delta(h, i, j, N):
    Ls = np.arange(-N + 1, N)
    d = np.zeros(2 * N - 1)
    ik = i + Ls
    m = (ik >= 0) & (ik < N)
    d[m] -= (1.0 - h[ik[m]])
    im = i - Ls
    m = (im >= 0) & (im < N)
    d[m] += h[im[m]]
    jk = j + Ls
    m = (jk >= 0) & (jk < N)
    d[m] += (1.0 - h[jk[m]])
    jm = j - Ls
    m = (jm >= 0) & (jm < N)
    d[m] -= h[jm[m]]
    return d

def directed_binary_search(h, N, deadline, max_rounds=8000):
    h = h.astype(float).copy()
    h = fix_sum_binary(h, N // 2)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    c = N - 1
    no_improve = 0
    stall_limit = 150
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        worst_idx = int(np.argmax(corr))
        k = worst_idx - c
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        ci = [i for i in ones if 0 <= i + k < N and h[i + k] == 0]
        cj = [j for j in zeros if 0 <= j + k < N and h[j + k] == 1]
        
        best_improve = cur_max
        best_swap = None
        
        if ci and cj:
            ci_s = ci[:40] if len(ci) > 40 else ci
            cj_s = cj[:40] if len(cj) > 40 else cj
            for i in ci_s:
                for j in cj_s:
                    d = swap_delta(h, i, j, N)
                    nm = float(np.max(corr + d))
                    if nm < best_improve - 1e-12:
                        best_improve = nm
                        best_swap = (i, j)
        
        n_rand = 10 + min(20, no_improve // 10)
        for _ in range(n_rand):
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
            if i == j:
                continue
            d = swap_delta(h, i, j, N)
            nm = float(np.max(corr + d))
            if nm < best_improve - 1e-12:
                best_improve = nm
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            d = swap_delta(h, i, j, N)
            h[i] = 0.0
            h[j] = 1.0
            corr += d
            cur_max = float(np.max(corr))
            if cur_max * 2.0 / N < best_c - 1e-14:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > stall_limit:
                n_pert = 5 + no_improve // 50
                for _ in range(n_pert):
                    i = int(np.random.choice(ones))
                    j = int(np.random.choice(zeros))
                    h[i], h[j] = h[j], h[i]
                corr = fft_corr(h)
                cur_max = float(np.max(corr))
                no_improve = 0
    
    return best_h, best_c

def continuous_refine(h, N, deadline, lr0=0.008):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_c = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    temp = 0.025
    c = N - 1
    it = 0
    
    while time.time() < deadline:
        it += 1
        corr = fft_corr(h)
        peak = float(np.max(corr))
        w = np.exp((corr - peak) / temp)
        w /= w.sum()
        
        grad = np.zeros(N)
        active = np.where(w > 0.002)[0]
        for L in active:
            lag = L - c
            wt = w[L]
            ik = np.arange(N) + lag
            im = np.arange(N) - lag
            vmk = (ik >= 0) & (ik < N)
            vmm = (im >= 0) & (im < N)
            g = np.zeros(N)
            g[vmk] += (1.0 - h[ik[vmk]])
            g[vmm] -= h[im[vmm]]
            grad += wt * g
        
        h_new = project_sum(h - lr * grad, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 150 == 0:
            lr *= 0.975
            temp *= 0.94
        if it % 500 == 0 and temp < 0.005:
            temp = 0.015
            lr = min(lr * 1.5, lr0)
    
    return best_h, best_c

def structured_init(N, variant, rng):
    nh = N // 2
    x = np.arange(N)
    if variant == 0:
        h = np.zeros(N)
        h[rng.choice(N, nh, replace=False)] = 1.0
    elif variant == 1:
        v = np.sin(2*np.pi*x*7.0/N) + 0.6*np.sin(2*np.pi*x*13.0/N+0.5) + 0.4*np.cos(2*np.pi*x*3.0/N+1.0)
        h = (v > np.median(v)).astype(float)
        h = fix_sum_binary(h, nh)
    elif variant == 2:
        v = 0.0
        for f in [3, 7, 11, 19]:
            v += np.sin(2*np.pi*f*x/N + 0.15*f)
        h = (v > np.median(v)).astype(float)
        h = fix_sum_binary(h, nh)
    elif variant == 3:
        h = ((x * 7 + 3) % 13 < 6).astype(float)
        h = fix_sum_binary(h, nh)
    else:
        v = np.sin(2*np.pi*x*5.0/N + 0.3) * 1.2 + np.sin(2*np.pi*x*17.0/N) * 0.8
        h = (v > np.median(v)).astype(float)
        h = fix_sum_binary(h, nh)
    return h
