    corr = compute_corr(h)
    return float(np.max(corr)) * (2.0 / N), corr

def project_onto_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5

        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

Wait, I realize the gradient computation in the continuous case is slow because of the Python loop. Let me vectorize it:

For a given L:
d_corr[m]/d_h[i] = -h[(i+L) % N] (if 0≤i+L<N) + (1-h[(i-L) % N]) (if 0≤i-L<N)

More precisely:
- Term 1: -h[i+L] for i where 0 ≤ i+L < N, i.e., 0 ≤ i < N-L (if L > 0) or -L ≤ i < N (if L < 0)
- Term 2: (1-h[i-L]) for i where 0 ≤ i-L < N, i.e., L ≤ i < N+L (if L > 0) or 0 ≤ i < N+L (if L < 0)

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        for i in range(n1 - nh):
            idx[i] = idx[np.random.randint(len(idx) - i - 1)]
            idx[i], idx[-1-i] = idx[-1-i], idx[i]
        h[idx[:n1-nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        for i in range(nh - n1):
            idx[i] = idx[np.random.randint(len(idx) - i - 1)]
            idx[i], idx[-1-i] = idx[-1-i], idx[i]
        h[idx[:nh-n1]] = 1
    return h

def binary_il(h, deadline, n_crit=30, s_sz=50, max_iters=10000, temp0=0.003):
    N = len(h)
    center = N - 1
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    best_c = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    temp = temp0
    no_imp = 0
    it = 0
    dx = 2.0 / N
    
    while time.time() < deadline and it < max_iters:
        top_idx = np.argsort(corr)[-n_crit:]
        top_lags = (top_idx - center).astype(int)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        
        n_l = len(top_lags)
        delta_rem = np.zeros((ns, n_l))
        delta_add = np.zeros((nz, n_l))
        
        for li, L in enumerate(top_lags):
            # Remove 1 at os_: Δ = h[os_-L] - (1-h[os_+L]) with bounds
            pm = os_ - L
            pp = os_ + L
            m1 = (pm >= 0) & (pm < N)
            m2 = (pp >= 0) & (pp < N)
            delta_rem[m1, li] += h[pm[m1]]
            delta_rem[m2, li] -= (1.0 - h[pp[m2]])
            
            # Add 1 at zs_: Δ = (1-h[zs_+L]) - h[zs_-L] with bounds
            qm = zs - L
            qp = zs + L
            m3 = (qm >= 0) & (qm < N)
            m4 = (qp >= 0) & (qp < N)
            delta_add[m4, li] += (1.0 - h[qp[m4]])
            delta_add[m3, li] -= (h[qm[m3]])
        
        cur_corr_top = corr[center + top_lags]
        total = cur_corr_top[None, None, :] + delta_rem[:, None, :] + delta_add[None, :, :]
        peaks = total.max(axis=2)
        
        min_pk = peaks.min()
        mi = np.unravel_index(peaks.argmin(), peaks.shape)
        cur_peak = float(np.max(corr))
        
        accept = False
        if min_pk < cur_peak - 1e-10:
            accept = True
        elif temp > 0.0001 and np.random.random() < np.exp(-min(50, (min_pk - cur_peak) / temp)):
            accept = True
        
        if accept:
            h[os_[mi[0]]] = 0
            h[zs[mi[1]]] = 1
            corr = compute_corr(h)
            c_new = float(np.max(corr)) * dx
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
            if no_imp > 150:
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi] = 0
                h[zi] = 1
                corr = compute_corr(h)
        
        temp *= 0.9995
        if no_imp > 300:
            no_imp = 0
            temp = max(temp0 * 0.6, temp)
        
        it += 1
    
    return best_h, best_c

Let me check the gradient formula once more. 

corr[center+L] = Σ_j h[j+L] * (1-h[j]) for j where 0≤j<N and 0≤j+L<N

d/dh[i] corr[center+L]:
- h[i] appears in the "h[j+L]" slot when j+L = i, i.e., j = i-L. Contribution: (1-h[i-L]) if 0≤i-L<N
- h[i] appears in the "(1-h[j])" slot when j = i. Contribution: -h[i+L] if 0≤i+L<N

So d/dh[i] = (1-h[i-L]) [if 0≤i-L<N] + (-h[i+L]) [if 0≤i+L<N]

In my code:
- i1: indices where 0≤i+L<N, i.e., -L≤i<N-L, intersected with [0,N): [max(0,-L), min(N, N-L))
  For these: g[i] += -h[i+L]
- i2: indices where 0≤i-L<N, i.e., L≤i<N+L, intersected with [0,N): [max(0,L), min(N, N+L))
  For these: g[i] += 1-h[i-L]

def block_move(h, direction=1):
    N = len(h)
    blen = np.random.randint(3, 15)
    start = np.random.randint(0, max(1, N - blen))
    ones_in = h[start:start+blen].copy()
    zeros_total = N - int(h.sum())
    ones_total = int(h.sum())
    if direction == 1 and ones_in.sum() > 0 and zeros_total > 0:
        h[start:start+blen] = np.where(ones_in > 0, 0, 0)
        h[start:start+blen][ones_in > 0] = 0
        extra = int(ones_in.sum())
        zidx = np.where(h == 0)[0]
        pick = np.random.choice(zidx, min(extra, len(zidx)), replace=False)
        h[pick] = 1
    return h
