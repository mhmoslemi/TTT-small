```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0-h)[::-1], L)
    return np.real(np.fft.irfft(H*G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def grad_smooth_max(h, N, T):
    g = 1.0 - h
    corr = fft_corr(h)
    peak = np.max(corr)
    w = np.exp((corr - peak) / T)
    w /= w.sum()
    L = 2*N - 1
    W = np.fft.rfft(w, L)
    G = np.fft.rfft(g[::-1], L)
    first_term = np.real(np.fft.irfft(W * G, L))[:N]
    wc = np.cumsum(np.insert(w, 0, 0.0))
    second_term = h * (wc[N:N+N] - wc[:N])
    return first_term - second_term

def project_binary(h, n_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:n_ones]] = 1.0
    return out

def continuous_refine(h, N, deadline, T0=0.03, lr0=0.01):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    T = T0
    it = 0
    no_improve = 0
    while time.time() < deadline - 2:
        it += 1
        grad = grad_smooth_max(h, N, T)
        # Subtract mean to preserve sum
        grad -= np.mean(grad)
        h_new = np.clip(h - lr * grad, 0, 1)
        # Project to sum = target
        # Binary search for threshold
        sorted_h = np.sort(h_new)[::-1]
        # Find threshold where sum of clipped values = target
        lo, hi = -1.0, 2.0
        for _ in range(30):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo+hi)/2, 0, 1)
        c_new = compute_c5(h_new, N)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if it % 100 == 0:
            lr *= 0.98
            T *= 0.97
        if no_improve > 2000:
            break
    return best_h, best_c

def binary_anneal(h, N, deadline, T0=0.05, block_sizes=(1,1,1,2,2,3,5)):
    h = h.astype(float).copy()
    nh = N // 2
    # Ensure sum
    if h.sum() != nh:
        h = project_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = T0
    T_end = 1e-6
    no_improve = 0
    it = 0
    while time.time() < deadline and it < 200000:
        it += 1
        L = int(np.random.choice(block_sizes))
        if L == 1:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0; h[zi] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < 0 or (T > 1e-8 and np.random.random() < np.exp(-dE/T)):
                cur_max = new_max
                corr = new_corr
                if cur_max * 2/N < best_c:
                    best_c = cur_max * 2/N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[oi] = 1; h[zi] = 0
        else:
            # Block move: find random positions for L-blocks
            ob = int(np.random.randint(0, N-L+1))
            zb = int(np.random.randint(0, N-L+1))
            if abs(ob - zb) < L:
                continue

    h = np.array(seq[:N], dtype=float)

    return h
```
