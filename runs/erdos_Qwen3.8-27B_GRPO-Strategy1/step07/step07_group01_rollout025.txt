        if it % 350 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k - 1, len(cs) - 1)]
            T = max(8.0, min(500.0, T * (0.85 if gap < 0.003 else 1.07)))
    return best_h, best_pk * (2.0 / N)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(77)
    if n1 > target:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, max_sweeps=800, perturb_every=22):
    h = ensure_binary(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    dx = 2.0 / N
    center = N - 1
    shifts = np.arange(2 * N - 1) - center
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        pa = np.arange(N)[:, None]
        sa = shifts[None, :]
        pm, pp = pa - sa, pa + sa
        vm, vp = (pm >= 0) & (pm < N), (pp >= 0) & (pp < N)
        pmc, ppc = np.clip(pm, 0, N - 1), np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pmc], 0.0)
        hp = np.where(vp, h[ppc], 0.0)
        D1 = -(1.0 - hp) * vp.astype(float) + hm * vm.astype(float)
        D0 = (1.0 - hp) * vp.astype(float) - hm * vm.astype(float)
        D1o = D1[ones, :]
        D0z = D0[zeros, :]
        best_new_P = P
        best_mv = None
        n_try = min(len(ones), max(70, int(len(ones) * 0.65)))
        rng_s = np.random.RandomState(sweep * 31 + 13)
        ps = rng_s.choice(ones, n_try, replace=False) if n_try < len(ones) else ones
        for pi in range(len(ps)):
            base = c + D1o[pi, :]
            cand = (base[None, :] + D0z).max(axis=1)
            mi = int(np.argmin(cand))
            if cand[mi] < best_new_P - 1e-12:
                best_new_P = float(cand[mi])
                best_mv = (1, pi, mi)
        # Double swap on stall
        if best_mv is None and stall > perturb_every:
            rng_d = np.random.RandomState(sweep * 47 + 7)
            nc = min(4, len(ones))
            pc = rng_d.choice(ones, nc, replace=False)
            for i2 in range(nc):
                for j2 in range(i2 + 1, nc):
                    b2 = c + D1[pc[i2], :] + D1[pc[j2], :]
                    cq1 = (b2[None, :] + D0z).max(axis=1)
                    q1i = int(np.argmin(cq1))
                    b3 = b2 + D0z[q1i, :]
                    cq2 = (b3[None, :] + D0z).max(axis=1)
                    cq2[q1i] = np.inf
                    q2i = int(np.argmin(cq2))
                    if cq2[q2i] < best_new_P - 1e-12:
                        best_new_P = float(cq2[q2i])
                        best_mv = (2, pc[i2], pc[j2], zeros[q1i], zeros[q2i])
                        break
                if best_mv is not None:
                    break
        if best_mv is None:
            n_r = min(2 + stall // 8, 6)
            rng_p = np.random.RandomState(sweep * 67 + 3)
            for _ in range(n_r):
                p = int(ones[rng_p.randint(len(ones))])
                q = int(zeros[rng_p.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            stall = 0
        else:
            if best_mv[0] == 1:
                h[ps[best_mv[1]]] = 0.0
                h[zeros[best_mv[2]]] = 1.0
            else:
                h[int(best_mv[1])] = 0.0
                h[int(best_mv[2])] = 0.0
                h[int(best_mv[3])] = 1.0
                h[int(best_mv[4])] = 1.0
            c = fft_corr(h)
    return best_h, best_c * dx

def upsample(ho, No, Nn):
    xo = np.linspace(0, 2, No, endpoint=False)
    xn = np.linspace(0, 2, Nn, endpoint=False)
    return np.interp(xn, xo, ho)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Nesterov multi-T
    for T_val in [60.0, 140.0, 300.0]:
        if time.time() > dl - 520:
            break
        hc = h0.copy() + np.random.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        d = min(dl - 420, time.time() + 110)
        h_n, c5_n = nesterov_opt(hc, N0, d, k=45, T=T_val, lr0=0.014)
        if c5_n < best_c5:
            best_c5, best_h = c5_n, h_n.copy()

    # Phase 2: Binary search
    if time.time() < dl - 380:
        hb = ensure_binary(best_h, N0)
        d_b = min(dl - 320, time.time() + 160)
        h_br, c5_br = binary_search(hb, N0, d_b, max_sweeps=700, perturb_every=18)
        if c5_br < best_c5:
            best_c5, best_h = c5_br, h_br.copy()

    # Phase 3: Try N0//2
    N2 = N0 // 2
    if N2 % 2 == 0 and N2 >= 200 and time.time() < dl - 280:
        hdn = np.zeros(N2)
        for i in range(N2):
            hdn[i] = best_h[2 * i:2 * i + 2].mean()
        hdn = project_sum(hdn, N2 / 2.0)
        d3 = min(dl - 220, time.time() + 70)
        h3, c5_3 = nesterov_opt(hdn, N2, d3, k=30, T=100.0, lr0=0.012)
        h3b = ensure_binary(h3, N2)
        if time.time() < dl - 160:
            d3b = min(dl - 140, time.time() + 50)
            h3br, c5_3b = binary_search(h3b, N2, d3b, max_sweeps=400, perturb_every=18)
            if c5_3b < best_c5:
                best_c5, best_h = c5_3b, h3br.copy()

    # Phase 4: Nesterov refinement at N0
    if time.time() < dl - 180:
        hc2 = best_h + np.random.randn(N0) * 0.03
        hc2 = project_sum(hc2, N0 / 2.0)
        d4 = min(dl - 140, time.time() + 75)
        h4, c5_4 = nesterov_opt(hc2, N0, d4, k=55, T=180.0, lr0=0.009)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Multi-start binary
    if time.time() < dl - 160:
        for trial in range(4):
            if time.time() > dl - 60:
                break
            p = best_h.copy()
            nf = np.random.randint(12, 35)
            pp = np.random.choice(N0, nf, replace=False)
            p[pp] = 1.0 - p[pp]
            pb = ensure_binary(p, N0)
            d5 = min(dl - 50, time.time() + 55)
            h5, c5_5 = binary_search(pb, N0, d5, max_sweeps=350, perturb_every=15)
            if c5_5 < best_c5:
                best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final binary polish
    Nf = len(best_h)
    if time.time() < dl - 35:
        hF = ensure_binary(best_h, Nf)
        d6 = min(dl - 8, time.time() + 50)
        hF2, c5F = binary_search(hF, Nf, d6, max_sweeps=450, perturb_every=16)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Phase 7: Micro Nesterov
    if time.time() < dl - 12:
        hFc = best_h + np.random.randn(len(best_h)) * 0.012
        hFc = project_sum(hFc, len(best_h) / 2.0)
        d7 = min(dl - 3, time.time() + 20)
        h7, c5_7 = nesterov_opt(hFc, len(best_h), d7, k=55, T=250.0, lr0=0.005)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()
        # Try binary from continuous result
        hFb = ensure_binary(h7 if c5_7 < best_c5 else best_h, len(best_h))
        cb = float(np.max(fft_corr(hFb))) * (2.0 / len(best_h))
        if cb < best_c5:
            best_c5, best_h = cb, hFb

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
