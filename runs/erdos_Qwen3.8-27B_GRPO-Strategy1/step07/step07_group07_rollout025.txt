        L_arr = top_lags.astype(int)
        for oi in os_:
            if tc - tabu_oi[oi] < tabu_sz:
                continue
            d_oi = np.zeros(len(L_arr))
            for li in range(len(L_arr)):
                L = L_arr[li]
                d = 0.0
                if L <= oi:
                    d -= (1.0 - h[oi - L])
                if oi + L < N:
                    d += h[oi + L]
                d_oi[li] = d
            for zi in zs:
                if tc - tabu_zi[zi] < tabu_sz:
                    continue
                new_pk = 0.0
                for li in range(len(L_arr)):
                    L = L_arr[li]
                    d = 0.0
                    if L <= zi:
                        d += (1.0 - h[zi - L])
                    if zi + L < N:
                        d -= h[zi + L]
                    val = corr[center + L] + d_oi[li] + d
                    if val > new_pk:
                        new_pk = val
                accept = new_pk < best_new - 1e-10
                if not accept and T > 0.0005:
                    if new_pk < cur_peak and np.random.random() < np.exp(-(new_pk - cur_peak) / T):
                        accept = True
                if accept:
                    best_new = new_pk
                    best_move = (oi, zi)
        # Pairs move
        if use_pairs and ns >= 4 and nz >= 4:
            for _ in range(min(3, ns // 2)):
                oi1 = int(np.random.choice(os_))
                oi2 = int(np.random.choice(os_))
                zi1 = int(np.random.choice(zs))
                zi2 = int(np.random.choice(zs))
                if oi1 == oi2 or zi1 == zi2:
                    continue
                if tc - tabu_oi[oi1] < tabu_sz or tc - tabu_oi[oi2] < tabu_sz:
                    continue
                if tc - tabu_zi[zi1] < tabu_sz or tc - tabu_zi[zi2] < tabu_sz:
                    continue
                new_pk = 0.0
                for li in range(len(L_arr)):
                    L = L_arr[li]
                    d = 0.0
                    for oi in (oi1, oi2):
                        if L <= oi:
                            d -= (1.0 - h[oi - L])
                        if oi + L < N:
                            d += h[oi + L]
                    for zi in (zi1, zi2):
                        if L <= zi:
                            d += (1.0 - h[zi - L])
                        if zi + L < N:
                            d -= h[zi + L]
                    val = corr[center + L] + d
                    if val > new_pk:
                        new_pk = val
                if new_pk < best_new - 1e-10:
                    best_new = new_pk
                    best_move = (oi1, oi2, zi1, zi2)
        if best_move is not None:
            if len(best_move) == 2:
                h[best_move[0]] = 0
                h[best_move[1]] = 1
                tabu_oi[best_move[0]] = tc
                tabu_zi[best_move[1]] = tc
            else:
                h[best_move[0]] = 0
                h[best_move[1]] = 0
                h[best_move[2]] = 1
                h[best_move[3]] = 1
                tabu_oi[best_move[0]] = tc
                tabu_oi[best_move[1]] = tc
                tabu_zi[best_move[2]] = tc
                tabu_zi[best_move[3]] = tc
        it += 1
        if it % 30 == 0:
            corr = compute_corr(h)
    return best_h, best_c

def cont_opt(h, deadline, k=40, lr0=0.02, mom=0.88):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 60:
            lr0 *= 0.8
            if lr0 < 0.001:
                break
            stale = 0
        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            M = abs(int(m) - center)
            if M == 0:
                grad += (1.0 - 2.0 * h)
            elif M < N:
                hi_idx = np.arange(M, N)
                lo_idx = np.arange(0, N - M)
                grad[hi_idx] += (1.0 - h[hi_idx - M])
                grad[lo_idx] -= h[lo_idx + M]
        g = grad / k
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: Quick continuous polish
    d1 = min(deadline - 600, t0 + 60)
    h1 = project(best_h + np.random.randn(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=35, lr0=0.015, mom=0.9)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    # Phase 2: Main binary SA (long)
    d2 = min(deadline - 250, time.time() + 400)
    hb = fix_binary(best_h)
    hb, c2 = binary_sa(hb, d2, n_crit=40, s_sz=60, tabu_sz=100, max_iters=12000, use_pairs=True)
    if c2 < best_c:
        best_c, best_h = c2, hb.copy()
    # Phase 3: Continuous relaxation to escape
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 80)
        h3 = project(best_h + np.random.randn(N0) * 0.05, N0 / 2.0)
        h3, c3 = cont_opt(h3, d3, k=30, lr0=0.012, mom=0.92)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()
    # Phase 4: Another binary SA round with different params
    if time.time() < deadline - 100:
        d4 = min(deadline - 80, time.time() + 100)
        hb4 = fix_binary(best_h)
        hb4, c4 = binary_sa(hb4, d4, n_crit=50, s_sz=70, tabu_sz=60, max_iters=8000, use_pairs=True)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()
    # Phase 5: Try slightly different N
    for N_try in [360, 350, 370]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d5 = min(deadline - 30, time.time() + 35)
        h_bt, c5v = binary_sa(h_bt, d5, n_crit=35, s_sz=50, tabu_sz=80, max_iters=3000, use_pairs=False)
        if c5v < best_c:
            best_c, best_h = c5v, h_bt.copy()
    # Phase 6: Final binary polish
    if time.time() < deadline - 5:
        hB = fix_binary(best_h)
        d7 = min(deadline - 2, time.time() + 30)
        hB, c7 = binary_sa(hB, d7, n_crit=45, s_sz=80, tabu_sz=50, max_iters=3000, use_pairs=True)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
