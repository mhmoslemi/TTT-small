```python
def compute_D(h, N):
    D = np.zeros((N, 2*N-1))
    for p in range(N):
        s_range = np.arange(-(N-1), N)  # lags
        # h[p+s] for valid s: 0 ≤ p+s < N → -p ≤ s < N-p
        idx1 = p + s_range
        valid1 = (idx1 >= 0) & (idx1 < N)
        # 1-h[p-s] for valid s: 0 ≤ p-s < N → p-N < s ≤ p
        idx2 = p - s_range
        valid2 = (idx2 >= 0) & (idx2 < N)
        D[p, valid1] += h[idx1[valid1]]
        D[p, valid2] -= (1.0 - h[idx2[valid2]])
    return D
```

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    R = np.fft.irfft(H * np.conj(G), L)
    return np.concatenate([R[L - (N - 1):L], R[0:N]])

def compute_D(h, N):
    D = np.zeros((N, 2 * N - 1))
    s_range = np.arange(-(N - 1), N)
    for p in range(N):
        idx1 = p + s_range
        v1 = (idx1 >= 0) & (idx1 < N)
        idx2 = p - s_range
        v2 = (idx2 >= 0) & (idx2 < N)
        D[p, v1] += h[idx1[v1]]
        D[p, v2] -= (1.0 - h[idx2[v2]])
    return D

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def greedy_binary_opt(h, N, deadline, max_sweeps=20000, perturb_after=150):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    D = compute_D(h, N)
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D_o = D[ones]
        D_z = D[zeros]
        best_val = np.inf
        best_oi, best_zi = -1, -1
        for i in range(len(ones)):
            M = (corr + D_o[i])[None, :] - D_z
            maxes = M.max(axis=1)
            idx = int(maxes.argmin())
            if maxes[idx] < best_val:
                best_val = maxes[idx]
                best_oi, best_zi = i, idx
        if best_oi < 0 or best_val >= best_c - 1e-12:
            stall += 1
            if stall > perturb_after:
                if stall > perturb_after * 3:
                    break
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
            else:
                p = ones[best_oi]
                q = zeros[best_zi]
        else:
            stall = 0
            p = ones[best_oi]
            q = zeros[best_zi]
        corr = corr + D[p] - D[q]
        h[p], h[q] = 0.0, 1.0
        D = compute_D(h, N)
        c = float(corr.max())
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    w = e / e.sum()
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * lin_grad(int(i - ctr), h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_opt(h0, N, deadline, T=30.0):
    target = N / 2.0
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lam = 5.0
    for outer in range(4):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj, h, args=(T, target, lam), jac=True,
                     method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 120, 'maxfun': 300, 'ftol': 1e-14})
        h = proj01(res.x, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lam *= 2.5
    return best_h, best_c
