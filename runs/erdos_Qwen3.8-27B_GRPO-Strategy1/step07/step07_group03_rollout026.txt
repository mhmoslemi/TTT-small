```python
def fft_corr(h):
    N = len(h)
```

```python
def compute_D_correct(h, N):
    L = 2 * N - 1
    pos = np.arange(N)[:, None]  # (N, 1)
    shifts = (np.arange(L) - (N - 1))[None, :]  # (1, L)
    pm = pos - shifts  # (N, L) = i - s
    pp = pos + shifts  # (N, L) = i + s
    vm = (pm >= 0) & (pm < N)  # (N, L)
    vp = (pp >= 0) & (pp < N)  # (N, L)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[:, None][pm_c], 0.0)  # h[pm] where valid
    hp = np.where(vp, h[:, None][pp_c], 0.0)  # h[pp] where valid
    val = hm * vm - (1.0 - hp) * vp
    sign = (2.0 * h - 1.0)[:, None]
    return sign * val
```
