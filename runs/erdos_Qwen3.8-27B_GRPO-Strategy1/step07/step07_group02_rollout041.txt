    best_h = h.copy()
    best_c = compute_c5(h)
    eta = 1.0  # Will decay
    
    for it in range(max_iter):
        if time.time() > deadline - 1:
            break
        corr = fft_corr(h)
        c = float(np.max(corr))
        if c * (2.0/N) < best_c:
            best_c = c * (2.0/N)
            best_h = h.copy()
        
        # Compute subgradient of max(corr)
        center = N - 1
        # Use top-K shifts for smoother gradient
        top_k = min(20, N)
        idx = np.argpartition(corr, -top_k)[-top_k:]

                grad[L:] -= h[:v]
            elif L < 0:

                    grad[:v] -= h[M:]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def C_row(h, N, idx):
    """Contribution of position idx to all shifts."""
    N2 = 2 * N - 1
    shifts = np.arange(N2) - (N - 1)  # L values
    ip = idx + shifts
    im = idx - shifts
    valid_p = (ip >= 0) & (ip < N)
    valid_m = (im >= 0) & (im < N)
    A = np.where(valid_p, 1.0 - h[np.clip(ip, 0, N-1)], 0.0)
    B = np.where(valid_m, h[np.clip(im, 0, N-1)], 0.0)
    return A - B

def swap_search(h, N, deadline, max_iters=50000, sz=80):
    """Local binary swap search focused on reducing max correlation."""
    h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    
    # Precompute C matrix (expensive but reusable)
    Cm = np.array([C_row(h, N, i) for i in range(N)])
    
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        s = min(len(ones), sz)
        z = min(len(zeros), sz)
        if stuck > 30:
            s = min(len(ones), sz + 60)
            z = min(len(zeros), sz + 60)
        oi = np.random.choice(ones, s, replace=False)
        zi = np.random.choice(zeros, z, replace=False)
        
        # Chunked evaluation to avoid huge 3D arrays
        best_mv, best_oi, best_zi = P, -1, -1
        ch = 40
        for a0 in range(0, s, ch):
            a1 = min(a0 + ch, s)
            for b0 in range(0, z, ch):
                b1 = min(b0 + ch, z)
                sub = corr[None, None, :] - Cm[oi[a0:a1]][:, None, :] + Cm[zi[b0:b1]][None, :, :]
                nm = sub.max(axis=2)
                mi = np.unravel_index(np.argmin(nm), nm.shape)
                mv = float(nm[mi])
                if mv < best_mv:
                    best_mv = mv
                    best_oi = a0 + mi[0]
                    best_zi = b0 + mi[1]

            pa = oi[best_oi]
            pb = zi[best_zi]
            h[pa] = 0
            h[pb] = 1
            Cm[pa] = C_row(h, N, pa)
            Cm[pb] = C_row(h, N, pb)

                h[q] = 1

                Cm[q] = C_row(h, N, q)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stuck = 0
        elif stuck > 200:
            break
    
    return best_h, best_c

def fw_opt(h, N, deadline, max_iter=3000):
    """Frank-Wolfe optimization for continuous relaxation."""
    target = N / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        c = float(np.max(corr))
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        center = N - 1
        top_k = 25
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            L = int(ai - center)
            if L > 0:
                v = N - L
                grad[:v] += (1.0 - h[L:]) * 0.5
                grad[L:] -= h[:v] * 0.5
            elif L < 0:
                M = -L
                v = N - M
                if v > 0:
                    grad[M:] += (1.0 - h[:v]) * 0.5
                    grad[:v] -= h[M:] * 0.5
            else:
                grad += (1.0 - 2.0 * h) * 0.5
        grad /= top_k
        
        # Linear minimization over extreme points
        order = np.argsort(grad)
        h_lin = np.zeros(N)
        h_lin[order[:int(target)]] = 1.0
        
        # Step size
        if it < 200:
            eta = 2.0 / (it + 2.0)
        else:
            eta = 1.0 / (it - 198)
        h_new = (1.0 - eta) * h + eta * h_lin
        h = project_01(h_new, target)
    
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, top_k=40, lr0=0.015):
    """Nesterov accelerated projected gradient."""
    target = N / 2.0
    h = h.astype(float).copy()
    m = np.zeros(N)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    lr = lr0
    beta = 0.9
    center = N - 1
    
    for it in range(50000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        c = float(np.max(corr))
        if c * (2.0/N) < best_c:
            best_c = c * (2.0/N)
            best_h = h.copy()
        
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            L = int(ai - center)
            if L > 0:
                v = N - L
                grad[:v] += (1.0 - y[L:])
                grad[L:] -= y[:v]
            elif L < 0:
                M = -L
                v = N - M
                if v > 0:
                    grad[M:] += (1.0 - y[:v])
                    grad[:v] -= y[M:]
            else:
                grad += (1.0 - 2.0 * y)
        grad /= top_k
        
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 200 == 0:
            lr *= 0.98
            beta = min(0.97, 0.88 + it * 0.0003)
    
    return best_h, best_c

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    out = np.fft.irfft(H * G, L)
    return out[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_bin_sum(h, n_target):
    s = int(np.round(h.sum()))
    if s > n_target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, s - n_target, replace=False)] = 0
    elif s < n_target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, n_target - s, replace=False)] = 1
    return h

def C_all(h, N):
    """Compute C matrix: C[p, m] = (1-h[p+m-(N-1)]) - h[p-m+(N-1)] for valid indices."""
    N2 = 2 * N - 1
    shifts = np.arange(N2) - (N - 1)

    # Vectorized: create index arrays
    pos = np.arange(N)[:, None]  # (N, 1)
    ip = pos + shifts[None, :]  # (N, N2): p + L for all p, L
    im = pos - shifts[None, :]  # (N, N2): p - L for all p, L
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    A = np.where(vp, 1.0 - h[np.clip(ip, 0, N-1)], 0.0)
    B = np.where(vm, h[np.clip(im, 0, N-1)], 0.0)
    return A - B
