        it += 1

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    return np.where(h > 0.5, 1.0, 0.0)

def init_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s[i]
    return s, deltas

def upd_delta(s, deltas, pos):
    N = len(s)
    deltas[pos, :] = 0.0
    deltas[pos, pos:pos + N] = s[pos]

        cp = np.array(sorted(set(cp)), dtype=int)

    bp = float(corr.max())
    bh = h.copy()

        it += 1
        if it % 500 == 0:
            lr *= 0.99
            beta = min(0.95, 0.83 + it * 0.0002)
    return bh, bp * (2.0/N)

    elif n1 < t:

        h[np.random.choice(idx, t - n1, replace=False)] = 1
    return np.where(h > 0.5, 1.0, 0.0)

def idl(h):
    N = len(h)
    s = 2.0 * h - 1.0
    dl = np.zeros((N, 2*N-1))
    for i in range(N):
        dl[i, i:i+N] = s[i]
    return s, dl

def udl(s, dl, p):
    N = len(s)
    dl[p, :] = 0.0
    dl[p, p:p+N] = s[p]

def sa(h0, dl_t, npk, npp, cr, T0, mx, bfreq, blen):
    """Simulated annealing with single and block swaps."""
    N = len(h0)
    h = h0.copy()
    corr = fft_corr(h)
    bc, bh = float(corr.max()), h.copy()
    s, dl = idl(h)
    T, it, ni = T0, 0, 0
    while time.time() < dl_t and it < mx:
        it += 1
        if it % 1200 == 0:
            T = max(T * cr, 0.0008)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Peak-directed candidates
        pk = np.argpartition(corr, -npk)[-npk:]
        cp, cq = [], []
        for p in pk:
            ko = int(p) - (N-1)
            if ko >= 0:
                vi = np.arange(0, N-ko)
                m = (h[vi]==1) & (h[vi+ko]==0)
                bp, bq = vi[m], (vi+ko)[m]
            else:
                M = -ko
                vi = np.arange(M, N)
                m = (h[vi]==1) & (h[vi-M]==0)
                bp, bq = vi[m], (vi-M)[m]
            if len(bp) > 0:
                ns = min(npp, len(bp))
                sl = np.random.choice(len(bp), ns, replace=False)
                cp.extend(bp[sl].tolist())
                cq.extend(bq[sl].tolist())
        nr = min(18, len(ones), len(zeros))
        cp.extend(np.random.choice(ones, nr, replace=False).tolist())
        cq.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cp = np.array(sorted(set(cp)), dtype=int)
        cq = np.array(sorted(set(cq)), dtype=int)
        if len(cp)==0 or len(cq)==0:
            break
        # Best single swap
        dp, dq = dl[cp], dl[cq]
        bn, bi, bj = 1e10, 0, 0
        for ci in range(0, len(cp), 12):
            ce = min(ci+12, len(cp))
            v = corr[None,None,:] + dp[ci:ce,None,:] - dq[None,:,:]
            nm = v.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < bn:
                bn, bi, bj = float(nm[mi]), mi[0]+ci, mi[1]
        p, q = int(cp[bi]), int(cq[bj])
        dc = bn - bc
        acc = dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc/(T+1e-9)))
        if acc:
            h[p], h[q] = 0.0, 1.0
            corr += dl[p].copy() - dl[q].copy()
            s[p], s[q] = -s[p], -s[q]
            udl(s, dl, p)
            udl(s, dl, q)
            c = float(corr.max())
            if c < bc - 1e-12:
                bc, bh, ni = c, h.copy(), 0
            else:
                ni += 1
        else:
            ni += 1
        # Block swap
        if bfreq > 0 and it % bfreq == 0 and len(ones) > blen and len(zeros) > blen:
            bl = np.random.randint(1, blen+1)
            p2 = int(np.random.choice(ones))
            q2 = int(np.random.choice(zeros))
            p2 = (p2 - bl + 1) % N
            q2 = (q2 - bl + 1) % N
            pp = (np.arange(p2, p2+bl)) % N
            qq = (np.arange(q2, q2+bl)) % N
            if np.all(h[pp]==1) and np.all(h[qq]==0):
                oh = h.copy()
                h[pp], h[qq] = 0.0, 1.0
                nc = fft_corr(h)
                cm = float(nc.max())
                dc2 = cm - bc
                if dc2 < -1e-12 or (T > 0.002 and np.random.random() < np.exp(-dc2/(T+1e-9))):
                    corr = nc
                    s = 2.0*h - 1.0
                    for idx in list(set(pp.tolist()) | set(qq.tolist())):
                        udl(s, dl, idx)
                    if cm < bc:
                        bc, bh, ni = cm, h.copy(), 0
                    else:
                        ni += 1
                else:
                    h[:] = oh
                    ni += 1
        if ni > 12000:
            break
    return bh, bc
