import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:n1 - target]] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - n1]] = 1
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def targeted_sa(h_bin, deadline, n_peaks=6, n_per_peak=18, cool_rate=0.9993, t_start=0.045, max_iters=70000):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.001)
        if no_impr > 2500 and T > 0.004:
            T = min(T * 1.6, 0.06)
            no_impr = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp = vi[mask]
                bq = (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp = vi[mask]
                bq = (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = min(12, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 8
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p + N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q + N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        if no_impr > 6000:
            break
    return best_h, best_c

def subgrad_polish(h_in, N, deadline, lr_init=0.006, act_frac=0.93):
    h = h_in.astype(float).copy()
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.87
    lr = lr_init
    it = 0
    while time.time() < deadline:
        h = np.clip(h, 0, 1)
        adj = (h.sum() - N / 2.0) / N
        h = np.clip(h - adj, 0, 1)
        corr = fft_corr(h)
        c5 = float(corr.max()) * (2.0 / N)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        thr = corr.max() * act_frac
        active = np.where(corr >= thr)[0]
        grad = np.zeros(N)
        for t in active:
            L = int(t) - (N - 1)
            if L == 0:
                grad += 1.0 - 2.0 * h
            elif L > 0:
                grad[:N - L] += (1.0 - h[L:])
                grad[L:] -= h[:N - L]
            else:
                M = -L
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
        grad /= len(active)
        m = beta * m + grad
        h_new = h - lr * m
        h_new = np.clip(h_new, 0, 1)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0, 1)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.991
            beta = min(0.96, 0.85 + it * 0.0004)
            act_frac = min(0.99, act_frac + 0.003)
    return best_h, best_c5
