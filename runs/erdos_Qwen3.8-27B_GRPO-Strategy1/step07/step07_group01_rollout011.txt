Actually, for the binary case, I can compute the correlation change exactly and efficiently:

For a swap (p: 1→0, q: 0→1), the new correlation at shift s is:
c_new(s) = c(s) + Δ(s)

where Δ(s) depends on the positions p, q relative to s.

For s > 0:
- p contributes: -(1-h[p+s]) if p+s < N, and -h[p-s] if p-s >= 0
- q contributes: +(1-h[q+s]) if q+s < N, and -h[q-s] if q-s >= 0

For s < 0 (let M = -s):
- p contributes: -(1-h[p-M]) if p-M >= 0, and -h[p+M] if p+M < N
- q contributes: +(1-h[q-M]) if q-M >= 0, and -h[q+M] if q+M < N

So the total change is:
Δ(s) = f_p(s) + f_q(s)

where f_p is the effect of turning off p, and f_q is the effect of turning on q.

The key insight: f_p(s) is non-zero only for s ∈ {s : p+s ∈ [0,N) or p-s ∈ [0,N)} = O(N) values of s.
Similarly for f_q.

But we need the max over ALL s, not just the top-k. So we can't just look at top-k shifts.

However, we can use the following trick: after a swap, the new max can only be at one of:
- The old top shifts (which might have decreased)
- Shifts where the max was previously just below the old max (which might have increased)

So if we track the top-50 shifts, and the new max is at a position outside this set, it would need to have increased by more than the gap to the 50th largest. This is unlikely for small perturbations.

So the approximation is: only check the top-50 shifts after each swap. This reduces the verification cost from O(N) to O(50).

But the search for the best swap still requires checking the effect on all top-50 shifts for all (p, q) pairs. That's 570 × 570 × 50 = 16M operations. At numpy speed, maybe 0.1s per sweep. That's much better!

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def nesterov_opt(h_init, N, deadline, k=50, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(np.max(corr))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_topk = corr[idx]
        w = np.exp(T * (c_topk - c_topk.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.97
            beta = min(0.98, 0.85 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def ensure_binary(h, N, rng):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

        h_try = upsample(best_h, len(best_h), N_try)
        h_b4 = ensure_binary(h_try, N_try, rng)
        d4 = min(deadline - 50, time.time() + 45)
        h4, c5_4 = binary_local_search(h_b4, N_try, d4, perturb_every=15, max_perturb=8, k_focus=50)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
    if time.time() < deadline - 25:
        hFc = best_h.copy() + rng.randn(len(best_h)) * 0.012
        hFc = project_sum(hFc, len(best_h) / 2.0)
        d5 = min(deadline - 12, time.time() + 22)
        h5, c5_5 = nesterov_opt(hFc, len(best_h), d5, k=60, T=200.0, lr0=0.006)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
        hFb = ensure_binary(best_h, len(best_h), rng)
        d5b = min(deadline - 4, time.time() + 18)
        hFb2, c5Fb = binary_local_search(hFb, len(hFb), d5b, perturb_every=10, max_perturb=5, k_focus=50)
        if c5Fb < best_c5:
            best_c5, best_h = c5Fb, hFb2.copy()
    if time.time() < deadline - 2:
        hP = best_h.copy().astype(float)
        dP = min(deadline - 1, time.time() + 15)
        hP2, c5P = binary_local_search(hP, len(hP), dP, perturb_every=8, max_perturb=4, k_focus=50)
        if c5P < best_c5:
            best_c5, best_h = c5P, hP2.copy()
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
