import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N, seed=77):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def build_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def greedy_search(h, N, deadline, max_sweeps, rng, T, stall, T_floor=0.008):
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    sweep = 0
    while sweep < max_sweeps:
        if time.time() > deadline:
            break
        sweep += 1
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        T = max(T_floor, T * 0.9965)
        if stall > 40 and T < 0.2:
            T = 0.35
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = build_D(h, N)
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        n_try = min(len(ones), 400)
        p_list = rng.choice(ones, n_try, replace=False) if len(ones) > n_try else ones
        n_worse_accepted = 0
        for p in p_list:
            npk = (c[None, :] + D[p][None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            val = float(npk[mi])
            if val < best_new_P:
                best_new_P = val
                best_pq = (int(p), int(zeros[mi]))
            elif T > T_floor * 4 and n_worse_accepted < 2 and (val - P) < 0.02:
                if rng.random() < np.exp(-(val - P) / (T * 0.04)):
                    n_worse_accepted += 1
                    if val < best_new_P:
                        best_new_P = val
                        best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            n_r = min(1 + stall // 20, 6)
            for _ in range(n_r):
                p2 = int(ones[rng.randint(len(ones))])
                q2 = int(zeros[rng.randint(len(zeros))])
                h[p2], h[q2] = 0.0, 1.0
            stall = 0
            T = max(T, 0.3)
    return best_h, best_c * (2.0 / N), sweep, T, stall

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = y.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
