```python
def greedy_improve(h, N, deadline, max_rounds=200):
    """Systematically try single flips and keep the best one each round."""
    h = h.astype(float).copy()
    best_c = np.max(fft_corr(h))
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        max_c = corr.max()
        # Try each position to flip
        best_delta = 0
        best_idx = -1
        for i in range(N):
            old = h[i]
            h[i] = 1.0 - old
            new_corr = fft_corr(h)
            new_max = new_corr.max()
            delta = max_c - new_max
            if delta > best_delta:
                best_delta = delta
                best_idx = i
            h[i] = old
        if best_idx >= 0:
            h[best_idx] = 1.0 - h[best_idx]
            best_c = best_c - best_delta
        else:
            break
    return h, best_c
```

```python
    #   Change in (1-h): Δ(1-h) = -(1-2*val) = 2*val - 1

    #   Contribution from h[i] as the "h" part: (1-val)*(1-h)[i+p-(N-1)] - val*(1-h)[i+p-(N-1)]
    #     = (1-2*val) * (1-h)[i+p-(N-1)]

    #   h[i] changes by dh = 1 - 2*h[i]  (from h[i] to 1-h[i])
    #   (1-h)[i] changes by -(dh)
```
