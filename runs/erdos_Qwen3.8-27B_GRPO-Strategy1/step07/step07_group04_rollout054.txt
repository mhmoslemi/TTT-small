import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    """Project h onto {0 <= h_i <= 1, sum(h) = target}"""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, c_full, kk):
    """Compute subgradient of max at top-k positions"""
    N = len(h)
    top = np.argpartition(c_full, -kk)[-kk:]
    grad = np.zeros(N)
    for ai in top:
        m = N - 1 - int(ai)
        j_lo = max(0, m - N + 1)
        j_hi = min(N - 1, m)
        t = m - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi+1] += 1.0 - 2.0 * h[t]
    return grad / kk

def adam_continuous(h_init, N, deadline, kk=30, lr0=0.01, iters_max=50000):
    """Adam optimizer for continuous h"""
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    
    m_adam = np.zeros(N)
    v_adam = np.zeros(N)
    beta1, beta2 = 0.8, 0.999
    eps = 1e-8
    t_adam = 0
    lr = lr0
    stall = 0
    stall_patience = 500
    
    it = 0
    while time.time() < deadline and it < iters_max:
        c = fft_corr(h)
        pk = float(np.max(c))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_patience:
                # Adaptive: reduce lr, or break
                lr *= 0.7
                stall = 0
                if lr < 1e-6:
                    break
        
        grad = grad_topk(h, c, kk)
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad**2
        m_hat = m_adam / (1 - beta1**t_adam)
        v_hat = v_adam / (1 - beta2**t_adam)
        h_new = project_box_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        h = h_new
        it += 1
        
        if it % 1000 == 0:
            lr *= 0.95
    return best_h, best_pk * dx

def ensure_binary_sum(h, N):

def compute_D_swap(h, N):
    """For binary h, compute the change in correlation matrix for swapping p->0, q->1"""
    # When h[p] goes from 1 to 0 and h[q] goes from 0 to 1:
    # Effect on c[k]: the contribution of h[p]=1 is removed, contribution of h[q]=0 (i.e., 1-h[q]=1) is removed
    # New: h[p]=0 (no contribution as 1), 1-h[p]=1 (contributes as (1-h))
    #       h[q]=1 (contributes as 1), 1-h[q]=0 (no contribution as (1-h))
    # 
    # Original c[k] = sum_i h[i]*(1-h[i+k_stuff])
    # The p position contributes: 1*(1-h[m-p]) where m = N-1-k (from earlier analysis)
    # After change: p contributes 0*(1-h[...]) - h[m-p]*1 = -h[m-p]... 
    
    # This is getting complex. Let me just use the direct formula.
    # c_full[k] = sum_i h[i]*(1-h[N-1-k-i])
    # Delta from p: was (1-h[N-1-k-p]), now 0. Change: -(1-h[N-1-k-p])
    # Also, p appears in h[N-1-k-i] when N-1-k-i=p, i.e., i=N-1-k-p. 
    #   Was h[N-1-k-p]*(1-h[p]) but h[p] was 1, so was 0. Now h[p]=0, so h[N-1-k-p]*(1-0)=h[N-1-k-p]
    #   Change: +h[N-1-k-p]
    # Delta from q: was 0*(1-h[N-1-k-q]) = 0, now 1*(1-h[N-1-k-q]). Change: +(1-h[N-1-k-q])
    # Also q appears in h[N-1-k-i] when i=N-1-k-q. Was h[N-1-k-q]*(1-0)=h[N-1-k-q], now h[N-1-k-q]*(1-1)=0
    #   Change: -h[N-1-k-q]
    #
    # Total delta[k] = -(1-h[m-p]) + h[m-p] + (1-h[m-q]) - h[m-q] where m=N-1-k
    # = -1 + 2*h[m-p] + 1 - 2*h[m-q]
    # = 2*(h[m-p] - h[m-q])
    
    # For binary h: delta[k] = 2*(h[m-p] - h[m-q]) where m = N-1-k
    # So delta is determined by the values at positions m-p and m-q
    
    # This means for each k, the change is 2*(h[N-1-k-p] - h[N-1-k-q]) (with bounds checking)
    # This is O(N) per (p,q) pair, or O(N^2) for all pairs.

    # = 2*h[m-p] - 1 for valid m-p, 0 otherwise
    
    N2 = 2 * N - 1
    shifts = np.arange(N2)
    m_arr = N - 1 - shifts  # m for each k
    
    # For each position p, the effect of setting h[p] from 1 to 0:
    # delta_p[k] = -(1 - h[m-p]) + h[m-p] if 0<=m-p<N else 0
    # = 2*h[m-p] - 1 if 0<=m-p<N else 0
    
    # For setting h[q] from 0 to 1:
    # delta_q[k] = (1-h[m-q]) - h[m-q] if 0<=m-q<N else 0
    # = 1 - 2*h[m-q] if 0<=m-q<N else 0
    
    # Total: delta[p,q][k] = delta_p[k] + delta_q[k]
    # New c[k] = c[k] + delta[p,q][k]
    # We want to minimize max_k (c[k] + delta[p,q][k])
    
    return m_arr

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, c_full, kk):
    N = len(h)
    top = np.argpartition(c_full, -kk)[-kk:]
    grad = np.zeros(N)
    for ai in top:
        m = N - 1 - int(ai)
        j_lo = max(0, m - N + 1)
        j_hi = min(N - 1, m)
        t = m - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi+1] += 1.0 - 2.0 * h[t]
    return grad / float(kk)

def adam_opt(h_init, deadline, kk_start=50, kk_end=5, lr0=0.015, total_iters=30000):

    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    
    m_ad = np.zeros(N)
    v_ad = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t_ad = 0
    lr = lr0
    stall = 0
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        # Adaptive kk
        frac = it / float(total_iters)
        kk = max(kk_end, int(kk_start * (1 - frac) + kk_end * frac))
        
        c = fft_corr(h)
        pk = float(np.max(c))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                lr *= 0.5
                stall = 0
                if lr < 1e-7:
                    break
        
        grad = grad_topk(h, c, kk)
        t_ad += 1
        m_ad = b1 * m_ad + (1 - b1) * grad
        v_ad = b2 * v_ad + (1 - b2) * grad * grad
        m_hat = m_ad / (1 - b1 ** t_ad)
        v_hat = v_ad / (1 - b2 ** t_ad)
        h = project_box_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        if it % 5000 == 4999:
            lr *= 0.85
    
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, c_full, kk):
    N = len(h)
    top = np.argpartition(c_full, -kk)[-kk:]
    grad = np.zeros(N)
    for ai in top:
        m = N - 1 - int(ai)
        j_lo = max(0, m - N + 1)
        j_hi = min(N - 1, m)
        t = m - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += 1.0 - 2.0 * h[t]
    return grad / float(kk)

def adam_opt(h_init, deadline, kk_start=60, kk_end=3, lr0=0.018, total_iters=60000):
    N = len(h_init)
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m_ad = np.zeros(N)
    v_ad = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t_ad = 0
    lr = lr0
    stall = 0
    for it in range(total_iters):
        if time.time() > deadline:
            break
        frac = min(1.0, it / float(total_iters))
        kk = max(kk_end, int(kk_start * (1.0 - frac) + kk_end * frac))
        c = fft_corr(h)
        pk = float(np.max(c))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                lr *= 0.5
                stall = 0
                if lr < 1e-7:
                    break
        grad = grad_topk(h, c, kk)
        t_ad += 1
        m_ad = b1 * m_ad + (1.0 - b1) * grad
        v_ad = b2 * v_ad + (1.0 - b2) * grad * grad
        m_hat = m_ad / (1.0 - b1 ** t_ad)
        v_hat = v_ad / (1.0 - b2 ** t_ad)
        h = project_box_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        if it % 8000 == 7999:
            lr *= 0.82
    return best_h, best_pk * dx

def binary_greedy(h, deadline, max_sweeps=5000, max_pq=250):
    N = len(h)
    h = h.astype(float).copy()
    dx = 2.0 / N
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    N2 = 2 * N - 1
    m_arr = N - 1 - np.arange(N2)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        n_p = min(len(ones), max_pq)
        n_q = min(len(zeros), max_pq)
        p_sub = ones[:n_p]
        q_sub = zeros[:n_q]
        mp = m_arr[:, None] - p_sub[None, :]
        vp = (mp >= 0) & (mp < N)
        mpc = np.clip(mp, 0, N - 1)
        dp = np.where(vp, 2.0 * h[mpc] - 1.0, 0.0)
        mq = m_arr[:, None] - q_sub[None, :]
        vq = (mq >= 0) & (mq < N)
        mqc = np.clip(mq, 0, N - 1)
        dq = np.where(vq, 1.0 - 2.0 * h[mqc], 0.0)
        base_p = c[None, :] + dp.T
        chunk = max(1, 15000000 // (N2 * n_q))
        best_val = P + 1.0
        best_iq = None
        for is_ in range(0, n_p, chunk):
            ie = min(is_ + chunk, n_p)
            temp = base_p[is_:ie, None, :] + dq.T[None, :, :]
            mx = temp.max(axis=2)
            gi, gj = np.unravel_index(np.argmin(mx), mx.shape)
            val = float(mx[gi, gj])
            if val < best_val:
                best_val = val
                best_iq = (is_ + gi, gj)
        if best_iq is None or best_val >= P - 1e-14:
            break
        p, q = int(p_sub[best_iq[0]]), int(q_sub[best_iq[1]])
        h[p] = 0.0
        h[q] = 1.0
    return best_h, best_c * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def ensure_binary_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def sa_binary(h, deadline, T0=4.0, Tcool=0.994, T_min=0.04, max_sweeps=8000,

        n_p = int(len(ones) * sf) if sf < 1.0 else len(ones)

        if accept:
