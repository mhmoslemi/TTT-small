New corr at lag L: corr[L] + Δcorr[L]
New max: max over L of (corr[L] + Δcorr[L])

But we can't just look at top_lags! A 2-swap might reduce the current max lag but increase a previously lower lag. However, for efficiency, we'll only check the top n_crit lags (plus maybe a few more margin).

Actually, for safety, let's check top_lags plus a margin of 10 more lags below the max.

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_softmax(h, corr, beta, top_k):
    N = len(h)
    center = N - 1
    cm = np.max(corr)
    w = np.exp(beta * (corr - cm) / (10.0 * max(cm, 1.0)))
    w /= np.sum(w)
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(N)
    for m in top_idx:
        wi = w[m]
        M = abs(int(m) - center)
        if M == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += wi * (1.0 - h[hi_idx - M])
            grad[lo_idx] -= wi * h[lo_idx + M]
    return grad

def cont_opt(h, deadline, beta, top_k, lr0, mom):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 50:
            lr0 *= 0.93
            stale = 0
        g = grad_softmax(h, corr, beta, top_k)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def delta_10(h, p, lags):
    """Delta corr when h[p] goes 1->0, for given lags. Returns array."""
    N = len(h)
    out = np.zeros(len(lags))
    for i, L in enumerate(lags):
        d = 0.0
        if 0 <= p + L < N:
            d -= (1.0 - h[p + L])
        if 0 <= p - L < N:
            d -= h[p - L]
        out[i] = d
    return out

def delta_01(h, q, lags):
    """Delta corr when h[q] goes 0->1, for given lags. Returns array."""
    N = len(h)
    out = np.zeros(len(lags))
    for i, L in enumerate(lags):
        d = 0.0
        if 0 <= q + L < N:
            d += (1.0 - h[q + L])
        if 0 <= q - L < N:
            d -= h[q - L]
        out[i] = d
    return out

def binary_sa(h, deadline, n_crit, s_sz, T0, use_2swap):
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    T = T0
    no_imp = 0
    it = 0
    center = N - 1
    cool_rate = 0.9995
    while time.time() < deadline and no_imp < 600:
        corr = compute_corr(h)
        c_raw = float(np.max(corr))
        c = c_raw * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        check_n = min(n_crit + 10, 2 * N - 1)
        top_lags = (np.argsort(corr)[-check_n:] - center).astype(int)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        d_arr_1 = np.array([delta_10(h, oi, top_lags) for oi in os_])
        d_arr_0 = np.array([delta_01(h, zi, top_lags) for zi in zs])
        best_move = None
        best_new_raw = c_raw + 1e-9
        for ii in range(ns):
            for jj in range(nz):
                new_vals = corr[center + top_lags] + d_arr_1[ii] + d_arr_0[jj]
                npk = np.max(new_vals)
                if npk < best_new_raw:
                    best_new_raw = npk
                    best_move = (os_[ii], zs[jj])
        if use_2swap and len(ones) >= 4 and len(zeros) >= 4 and it % 3 == 0:
            o2 = np.random.choice(ones, 4, replace=False)
            z2 = np.random.choice(zeros, 4, replace=False)
            d2_1 = np.array([delta_10(h, oi, top_lags) for oi in o2])
            d2_0 = np.array([delta_01(h, zi, top_lags) for zi in z2])
            for a in range(4):
                for b in range(a + 1, 4):
                    for c2 in range(4):
                        for d2 in range(c2 + 1, 4):
                            new_vals = corr[center + top_lags] + d2_1[a] + d2_1[b] + d2_0[c2] + d2_0[d2]
                            npk = np.max(new_vals)
                            if npk < best_new_raw:
                                best_new_raw = npk
                                best_move = (o2[a], o2[b], z2[c2], z2[d2])
        T *= cool_rate
        if best_move is not None:
            delta = best_new_raw - c_raw
            if delta < 0 or (T > 0.0005 and np.random.random() < np.exp(-abs(delta) / T)):
                if len(best_move) == 2:
                    h[best_move[0]] = 0
                    h[best_move[1]] = 1
                else:
                    h[best_move[0]] = 0
                    h[best_move[1]] = 0
                    h[best_move[2]] = 1
                    h[best_move[3]] = 1
        elif no_imp > 200 and np.random.random() < 0.25:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        it += 1
    return best_h, best_c
