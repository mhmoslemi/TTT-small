The bottleneck is probably the candidate generation (finding bad positions). Let me optimize that.

1. Find top n_peaks lags
2. For each peak lag, find positions where h=1 and h+k=0

This is O(N * n_peaks) per iteration, which for N=380, n_peaks=6 is ~2300 operations. That's fine.

The inner loop of SA is:

2. Best candidate selection: O(|cand_p| * |cand_q| * 1) using the delta matrix (just look up values)
3. Update: O(N) for corr update

So each iteration is roughly O(N * n_peaks + N) ≈ O(N) for small n_peaks.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def make_binary(h, N, rng):
    h = np.asarray(h, dtype=float)
    if len(h) != N:
        x = np.linspace(0, 2, len(h), endpoint=False)
        xn = np.linspace(0, 2, N, endpoint=False)
        h = np.interp(xn, x, h)
    order = np.argsort(-h)
    h2 = np.zeros(N)
    h2[order[:N//2]] = 1.0
    return h2

def build_deltas(h):
    N = len(h)
    s = 2.0*h - 1.0
    deltas = np.zeros((N, 2*N-1))
    for i in range(N):
        deltas[i, i:i+N] = s
    return s, deltas

def sa_search(h_bin, deadline, seed_offset=0, n_peaks=6, n_per_peak=15,
              cool_rate=0.9992, t_start=0.04, max_iters=100000,
              use_2swap=False, restart_stuck=4000):
    rng = np.random.RandomState(seed_offset)
    N = len(h_bin)
    h = h_bin.astype(float).copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 1000 == 0:
            T = max(T * cool_rate, 0.001)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Generate candidates weighted by peak lags
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N-1)
            if k_off >= 0:
                vi = np.arange(0, N-k_off)
                mask = (h[vi] == 1) & (h[vi+k_off] == 0)
                bp = vi[mask]
                bq = (vi+k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi-M] == 0)
                bp = vi[mask]
                bq = (vi-M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = rng.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        
        # Add random candidates
        nr = min(10, len(ones), len(zeros))
        rp = rng.choice(ones, nr, replace=False)
        rq = rng.choice(zeros, nr, replace=False)
        cand_p.extend(rp.tolist())
        cand_q.extend(rq.tolist())
        
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        
        # Evaluate all (p, q) pairs using delta matrix
        dp = deltas[cand_p]  # shape (|cp|, L)
        dq = deltas[cand_q]  # shape (|cq|, L)
        # We want to minimize max(corr + dp - dq)
        # For each (i, j): new_corr = corr + dp[i] - dq[j]
        # new_max = max(corr + dp[i] - dq[j])
        
        # Chunk evaluation
        chunk = 8
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci+chunk, len(cand_p))
            # vals[i, j, :] = corr + dp[ci+i] - dq[j]
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)  # shape (chunk_size, |cq|)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0]+ci, mi[1]
        
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        
        # Also try 2-swap if enabled
        accept = False
        if dc < -1e-12 or (T > 0.001 and rng.random() < np.exp(-dc/(T+1e-9))):
            accept = True
        elif use_2swap and rng.random() < 0.15:
            # 2-swap: pick another (p2, q2) from remaining candidates
            mask_p = np.ones(len(cand_p), dtype=bool)
            mask_q = np.ones(len(cand_q), dtype=bool)
            mask_p[bi] = False
            mask_q[bj] = False
            if mask_p.any() and mask_q.any():
                p2_idx = rng.choice(np.where(mask_p)[0])
                q2_idx = rng.choice(np.where(mask_q)[0])
                p2, q2 = int(cand_p[p2_idx]), int(cand_q[q2_idx])
                if p2 != p and q2 != q:
                    # Approximate 2-swap change
                    new_corr_approx = corr + dp[bi] - dq[bj] + deltas[p2] - deltas[q2]
                    dc2 = float(new_corr_approx.max()) - best_c
                    if dc2 < -1e-12 or (T > 0.001 and rng.random() < np.exp(-dc2/(T+1e-9))):
                        h[p], h[q] = 0.0, 1.0
                        h[p2], h[q2] = 0.0, 1.0
                        corr = fft_corr(h)  # full recompute for 2-swap
                        s, deltas = build_deltas(h)
                        c = float(corr.max())
                        if c < best_c - 1e-12:
                            best_c = c
                            best_h = h.copy()
                            no_impr = 0
                        else:
                            no_impr += 1
                        continue
                    else:
                        no_impr += 1
                        continue
        
        if accept:
            h[p], h[q] = 0.0, 1.0
            corr = corr + deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p+N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q+N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        
        # Restart mechanism
        if no_impr > restart_stuck:
            # Perturb best_h and restart
            h = best_h.copy()
            n_kick = rng.randint(2, 6)
            o = np.where(h == 1)[0]
            z = np.where(h == 0)[0]
            ko = rng.choice(o, min(n_kick, len(o)), replace=False)
            kz = rng.choice(z, min(n_kick, len(z)), replace=False)
            h[ko] = 0.0
            h[kz[:len(ko)]] = 1.0
            corr = fft_corr(h)
            best_c_cur = float(corr.max())
            s, deltas = build_deltas(h)
            no_impr = 0
            T = t_start * 0.6
        
    return best_h, best_c

def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=40):
    h = h_in.astype(float).copy()
    m = np.zeros(N)
    beta = 0.85
    best_pk = float('inf')
    best_h = h.copy()
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N-1))
            if L == 0:
                grad += 1.0 - 2.0*y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top)
        
        h_new = np.clip(y - lr*grad, 0.0, 1.0)
        adj = (h_new.sum() - N/2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        
        if (int(time.time()*10) % 7) == 0:  # roughly every 0.7s, decay lr
            lr *= 0.998
            beta = min(0.92, beta + 0.0001)
    
    return best_h, best_pk * (2.0/N)
