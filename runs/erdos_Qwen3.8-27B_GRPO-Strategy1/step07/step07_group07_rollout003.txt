        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        top_lags = top_lags[top_lags != 0]
        if len(top_lags) == 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), pool_sz)
        nz = min(len(zeros), pool_sz)
        os_ = np.random.choice(ones, ns, replace=False) if ns < len(ones) else ones.copy()
        zs = np.random.choice(zeros, nz, replace=False) if nz < len(zeros) else zeros.copy()
        nL = len(top_lags)
        delta_o = np.zeros((ns, nL))
        delta_z = np.zeros((nz, nL))
        for li, L in enumerate(top_lags):
            vp = (os_ + L >= 0) & (os_ + L < N)
            vm = (os_ - L >= 0) & (os_ - L < N)
            dp = np.zeros(ns)
            dm = np.zeros(ns)
            if np.any(vp):
                dp[vp] = 1.0 - h[os_[vp] + L]
            if np.any(vm):
                dm[vm] = h[os_[vm] - L]
            delta_o[:, li] = -dp + dm
            vpz = (zs + L >= 0) & (zs + L < N)
            vmz = (zs - L >= 0) & (zs - L < N)
            dpz = np.zeros(nz)
            dmz = np.zeros(nz)
            if np.any(vpz):
                dpz[vpz] = 1.0 - h[zs[vpz] + L]
            if np.any(vmz):
                dmz[vmz] = h[zs[vmz] - L]
            delta_z[:, li] = dpz - dmz
        corr_top = corr[center + top_lags]
        new_val = corr_top[None, None, :] + delta_o[:, None, :] + delta_z[None, :, :]
        new_peak = new_val.max(axis=2)
        cur_peak = float(np.max(corr))
        flat_idx = int(np.argmin(new_peak))
        b_oi, b_zi = flat_idx // nz, flat_idx % nz
        if new_peak[b_oi, b_zi] < cur_peak - 1e-10:
            h[os_[b_oi]] = 0
            h[zs[b_zi]] = 1
        elif no_imp > 80 and np.random.random() < 0.35:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        it += 1
    return best_h, best_c

def structured_start(N, seed_val):
    np.random.seed(seed_val)
    x = np.arange(N)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / N + np.random.uniform(0, 2*np.pi))
    h += 0.15 * np.sin(2 * np.pi * 7 * x / N + np.random.uniform(0, 2*np.pi))
    return project(h, N / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: Continuous from initial
    d1 = min(deadline - 520, t0 + 140)
    h1 = project(h0 + np.random.randn(N0) * 0.05, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, temp0=0.05, lr0=0.015, mom=0.88, n_lags=90)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    # Phase 2: Binary IL
    if time.time() < deadline - 400:
        d2 = min(deadline - 380, time.time() + 200)
        hb = fix_binary(best_h)
        hb, c2 = binary_il_fast(hb, d2, n_crit=35, pool_sz=80, max_no_imp=400)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()
    # Phase 3: Continuous refine
    if time.time() < deadline - 250:
        d3 = min(deadline - 230, time.time() + 100)
        h3 = project(best_h + np.random.randn(N0) * 0.02, N0 / 2.0)
        h3, c3 = cont_opt(h3, d3, temp0=0.02, lr0=0.006, mom=0.9, n_lags=120)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()
    # Phase 4: Binary IL again
    if time.time() < deadline - 160:
        d4 = min(deadline - 140, time.time() + 130)
        hb4 = fix_binary(best_h)
        hb4, c4 = binary_il_fast(hb4, d4, n_crit=40, pool_sz=90, max_no_imp=350)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()
    # Phase 5: Try structured start at same N
    if time.time() < deadline - 200:
        hs = structured_start(N0, seed + 7)
        d5 = min(deadline - 180, time.time() + 60)
        hs, c5v = cont_opt(hs, d5, temp0=0.04, lr0=0.01, mom=0.85, n_lags=80)
        if c5v < best_c:
            best_c, best_h = c5v, hs.copy()
    # Phase 6: Try different N
    for N_try in [400, 440, 500]:
        if time.time() > deadline - 90:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d6 = min(deadline - 70, time.time() + 40)
        h_bt, c6 = binary_il_fast(h_bt, d6, n_crit=30, pool_sz=70, max_no_imp=200)
        if c6 < best_c:
            best_c, best_h = c6, h_bt.copy()
    # Final polish
    if time.time() < deadline - 5:
        hB = fix_binary(best_h)
        d7 = min(deadline - 3, time.time() + 30)
        hB, c7 = binary_il_fast(hB, d7, n_crit=45, pool_sz=100, max_no_imp=200)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
