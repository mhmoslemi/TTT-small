import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    if n1 > target:

        idx = np.random.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = np.random.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def make_D(h, N):
    s = np.arange(2*N-1) - (N-1)
    ps = np.arange(N)[:, None] + s[None, :]
    pm = np.arange(N)[:, None] - s[None, :]
    vp = (ps >= 0) & (ps < N)
    vm = (pm >= 0) & (pm < N)
    psc = np.clip(ps, 0, N-1)
    pmc = np.clip(pm, 0, N-1)
    D = np.where(vp, 1.0 - h[psc], 0.0) - np.where(vm, h[pmc], 0.0)
    return D

    best_h = h.copy()
    m = np.zeros(N)

    it = 0

            best_h = h.copy()

        h = h_new
        if it % 200 == 0 and lr < 1e-4:

    return best_h, best_pk * (2.0 / N)
