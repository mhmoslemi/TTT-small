            lo = mid
        else:
            hi = mid

def compute_grad(h, T, k):

    N = len(h)
    c = fft_lin_corr(h)
    # Pick top k values
    idx = np.argpartition(c, -k)[-k:]

def fft_lin_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    circ = np.fft.irfft(H * np.conj(G), L)
    return np.roll(circ, N - 1)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_lin_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(h, T, k):
    N = len(h)
    c = fft_lin_corr(h)
    idx = np.argpartition(c, -k)[-k:]
    c_top = c[idx]
    c_max = c_top.max()
    w = np.exp(T * (c_top - c_max))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1

        l = m - center
        wi = w[ii]
        if l == 0:

        h_br, c5_br = binary_local_search(h_bin, d_b, max_sweeps=500, perturb_every=25)
        if c5_br < best_c5:
            best_c5, best_h = c5_br, h_bin.copy() if c5_br < best_c5 else h_br.copy()
            best_h = h_br.copy()

    # Phase 3: Try N2 = N0//2 (downsample)
    N2 = N0 // 2
    if time.time() < deadline - 250:
        h_d = upsample(best_h, len(best_h), N2)
        h_d = project_sum(h_d, N2 / 2.0)
        d3 = min(deadline - 200, time.time() + 80)
        h3, c5_3 = nesterov_opt(h_d, d3, k=30, T=70.0, lr0=0.016)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
        if time.time() < deadline - 150:
            h_b3 = ensure_binary(best_h, len(best_h))
            d3b = min(deadline - 120, time.time() + 50)
            h3b, c5_3b = binary_local_search(h_b3, d3b, max_sweeps=200, perturb_every=20)
            if c5_3b < best_c5:
                best_c5, best_h = c5_3b, h3b.copy()

    # Phase 4: Second continuous pass at N0 from best
    if time.time() < deadline - 200:
        h_res = best_h.copy() if len(best_h) == N0 else upsample(best_h, len(best_h), N0)
        hc2 = h_res + np.random.randn(N0) * 0.03
        hc2 = project_sum(hc2, N0 / 2.0)
        d4 = min(deadline - 150, time.time() + 80)
        h4, c5_4 = nesterov_opt(hc2, d4, k=50, T=150.0, lr0=0.010)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Try N3 = N0 * 7 // 6 or nearby (slightly different resolution)
    N3 = 720
    if time.time() < deadline - 150 and N3 != N0:
        h_up3 = upsample(best_h, len(best_h), N3)
        h_up3 = project_sum(h_up3, N3 / 2.0)
        d5 = min(deadline - 100, time.time() + 60)
        h5, c5_5 = nesterov_opt(h_up3, d5, k=40, T=90.0, lr0=0.012)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
        if time.time() < deadline - 80:
            h_b5 = ensure_binary(best_h, len(best_h))
            d5b = min(deadline - 50, time.time() + 30)
            h5b, c5_5b = binary_local_search(h_b5, d5b, max_sweeps=100, perturb_every=20)
            if c5_5b < best_c5:
                best_c5, best_h = c5_5b, h5b.copy()

    # Phase 6: Final binary polish at best N
    if time.time() < deadline - 40:
        Nf = len(best_h)
        hF = ensure_binary(best_h, Nf)
        d6 = min(deadline - 10, time.time() + 40)
        hF2, c5F = binary_local_search(hF, d6, max_sweeps=200, perturb_every=20)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Phase 7: Final continuous micro-adjustment
    if time.time() < deadline - 5:
        hFc = best_h.copy() + np.random.randn(len(best_h)) * 0.015
        hFc = project_sum(hFc, len(best_h) / 2.0)
        d7 = min(deadline - 2, time.time() + 25)
        h7, c5_7 = nesterov_opt(hFc, d7, k=50, T=200.0, lr0=0.006)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
