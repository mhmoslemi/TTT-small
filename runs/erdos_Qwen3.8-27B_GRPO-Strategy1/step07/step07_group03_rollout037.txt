Actually, the most efficient binary local search computes the "delta" for each possible move in O(N) time per candidate, and with vectorization, we can compute all deltas at once.

For a binary h, when we flip h[p] from 1 to 0:
- For each shift L, the correlation changes by: -(1 - h[p-L]) for the term where i=p, and -h[p+L] for the term where i-k=p (i.e., i=p+L)

So delta_flip1[p][L] = -(1-h[p-L]) - h[p+L] (with boundary handling)

When we flip h[q] from 0 to 1:
delta_flip0[q][L] = (1-h[q-L]) + h[q+L]

The net change for a swap (p:1→0, q:0→1) is:
delta_swap[p,q][L] = delta_flip1[p][L] + delta_flip0[q][L]

And the new max is max(c_cur + delta_swap[p,q]).

We can precompute delta_flip1 and delta_flip0 as matrices of shape (N, 2N-1), then for any (p,q) pair, the new correlation is c_cur + D1[p] + D0[q].

To find the best swap: minimize max(c_cur + D1[p] + D0[q]) over all valid (p,q).

This is a "minimax" problem. A common approach:
- For each p, find the q that minimizes max(c_cur + D1[p] + D0[q])
- This can be done by: for fixed p, the function to minimize is max_L (c_cur[L] + D1[p][L] + D0[q][L])
- Which is max_L (base_p[L] + D0[q][L]) where base_p = c_cur + D1[p]
- For fixed base_p, the best q minimizes the pointwise max with D0[q]

This is still O(N^2 * N) in the worst case, but with N=380, that's about 55 million operations per sweep, which is fast.

Actually, let me precompute D1 and D0 matrices:

D1[p, L] = -(1-h[p-L]) - h[p+L] for valid indices (where p-L is in [0,N) or p+L is in [0,N))
D0[q, L] = (1-h[q-L]) + h[q+L]

For binary h, D1[p, L] + D0[q, L] gives the change in correlation at shift L.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        idx = np.where(h_bin == 1)[0]
        h_bin[rng.choice(idx, n1 - nh, replace=False)] = 0
    else:
        idx = np.where(h_bin == 0)[0]
        h_bin[rng.choice(idx, nh - n1, replace=False)] = 1
    return h_bin

def compute_D(h, N):
    """Compute D1 (flip 1→0) and D0 (flip 0→1) matrices for binary h"""
    L_arr = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = L_arr[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hv_m = np.where(vm, h[pm_c], 0.0)
    hv_p = np.where(vp, h[pp_c], 0.0)
    D1 = -(1.0 - hv_m) * vm - hv_p * vp  # flip 1→0
    D0 = (1.0 - hv_m) * vm + hv_p * vp  # flip 0→1
    return D1, D0

def binary_local_search(h, N, deadline, rng, max_sweeps=500, sa_temp=0.005):
    """Best-improvement binary local search with 1-swap, 2-swap, and SA"""
    h = h.astype(float).copy()
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_c = P_cur
    best_h = h.copy()
    temp = sa_temp
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.0005, temp * 0.997)
        D1, D0 = compute_D(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_new_P = P_cur + 1.0  # target to beat
        best_move = None
        # Best 1-swap
        D1_ones = D1[ones, :]  # (n1, 2N-1)
        D0_zeros = D0[zeros, :]  # (n0, 2N-1)
        # For each p in ones, find best q in zeros
        for pi in range(len(ones)):
            base = c_cur + D1_ones[pi]
            vals = (base[None, :] + D0_zeros).max(axis=1)
            mi = int(np.argmin(vals))
            if vals[mi] < best_new_P:
                best_new_P = float(vals[mi])
                best_move = (int(ones[pi]), int(zeros[mi]), 0)
        # 2-swap: try a random subset
        n_2swap = min(80, len(ones) // 2)
        for _ in range(n_2swap):
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            new_P = float((c_cur + D1[p1] + D1[p2] + D0[q1] + D0[q2]).max())
            if new_P < best_new_P:
                best_new_P = new_P
                best_move = (p1, p2, q1, q2, 1)
        # Apply best move or SA move
        if best_move is not None and best_new_P < P_cur - 1e-14:
            if best_move[2] == 0:
                p, q = best_move[0], best_move[1]
                h[p] = 0.0
                h[q] = 1.0
            else:
                p1, p2, q1, q2 = best_move[0], best_move[1], best_move[2], best_move[3]
                h[p1] = h[p2] = 0.0
                h[q1] = h[q2] = 1.0
            c_cur = fft_corr(h)
            P_cur = float(c_cur.max())
            if P_cur < best_c - 1e-14:
                best_c = P_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # SA: try a random move
            if stall > 5:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                new_P = float((c_cur + D1[p] + D0[q]).max())
                delta = new_P - P_cur
                if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
                    h[p] = 0.0
                    h[q] = 1.0
                    c_cur = fft_corr(h)
                    P_cur = float(c_cur.max())
                    if P_cur < best_c - 1e-14:
                        best_c = P_cur
                        best_h = h.copy()
                        stall = 0
            stall += 1
            if stall > 50:
                break
    return best_h, best_c * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        idx = np.where(h_bin == 1)[0]
        h_bin[rng.choice(idx, n1 - nh, replace=False)] = 0
    else:
        idx = np.where(h_bin == 0)[0]
        h_bin[rng.choice(idx, nh - n1, replace=False)] = 1
    return h_bin

def compute_D(h, N):
    L_arr = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = L_arr[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hv_m = np.where(vm, h[pm_c], 0.0)
    hv_p = np.where(vp, h[pp_c], 0.0)
    D1 = -(1.0 - hv_m) * vm + hv_p * vp
    # Fix L=0 (center) for binary: exact change is 0
    center = N - 1
    D1[:, center] = 0.0
    D0 = (1.0 - hv_m) * vm - hv_p * vp
    D0[:, center] = 0.0
    return D1, D0

def binary_local_search(h, N, deadline, rng, max_sweeps=600, sa_temp=0.006):
    h = h.astype(float).copy()
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_c = P_cur
    best_h = h.copy()
    temp = sa_temp
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.0005, temp * 0.996)
        D1, D0 = compute_D(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_new_P = P_cur + 1.0
        best_move = None
        # Best 1-swap: for each p in ones, find best q in zeros
        D1_ones = D1[ones, :]
        D0_zeros = D0[zeros, :]
        base_all = c_cur[None, :] + D1_ones  # (n1, 2N-1)
        vals = (base_all[:, :, None] + D0_zeros[None, :]).max(axis=1)
        # vals[pi, qi] = max_L (c_cur + D1[ones[pi]] + D0[zeros[qi]])
        flat_idx = int(np.argmin(vals))
        pi_best, qi_best = divmod(flat_idx, len(zeros))
        best_new_P = float(vals[pi_best, qi_best])
        best_move = (int(ones[pi_best]), int(zeros[qi_best]), 0)
        # 2-swap
        n_2swap = min(60, len(ones) // 3)
        for _ in range(n_2swap):

            q1, q2 = int(zeros[j1]), int(zeros[j2])
            new_P = float((c_cur + D1[p1] + D1[p2] - D0[q1] - D0[q2]).max())
            if new_P < best_new_P:

                best_move = (p1, p2, q1, q2, 1)
        if best_move is not None and best_new_P < P_cur - 1e-14:

    return best_h, best_c * (2.0 / N)

def nesterov_lse(h, N, deadline, beta_lse=40.0, lr0=0.008, k_top=30):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    m = np.zeros(N)
    lr = lr0
    beta_n = 0.9
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        c_max = corr[top].max()
        w = np.exp(beta_lse * (corr[top] - c_max))
        w = w / w.sum()
        grad = np.zeros(N)
        for idx, ai in enumerate(top):
            L = int(ai - center)
            wi = w[idx]
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N-L])
                grad[:N-L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N-M]
        h_new = project_sum(y - lr * grad, target)
        m = beta_n * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.97
            beta_lse = min(200.0, beta_lse * 1.6)
            beta_n = min(0.97, 0.9 + it * 0.00008)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
