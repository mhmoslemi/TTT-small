import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5_fft(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N, seed=77):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())

    if n1 == target:
        return h_bin
    if n1 > target:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], target - n1, replace=False)

    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top_idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[top_idx]
        w = np.exp(50.0 * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(k):
            s = int(top_idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k

        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, 0.85 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

            h8, c5_8 = binary_sa(hF, N0, d8, seed=999, ntries=100, T_init=0.015)
            if c5_8 < best_c5:
                best_c5, best_h = c5_8, h8.copy()

    N_fin = len(best_h)
    dx = 2.0 / N_fin
    corr_exact = np.correlate(best_h, 1.0 - best_h, mode="full")
    final_c5 = float(np.max(corr_exact)) * dx
    return best_h, final_c5, N_fin
