import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=40, T=60.0, lr0=0.02):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.996
            beta = min(0.98, 0.85 + it * 0.00015)

            T = max(15.0, min(300.0, T * (0.9 if gap < 0.005 else 1.05)))
    return best_h, best_pk * dx

def compute_swap_deltas(h, N):
    """Precompute the effect of swapping each (one, zero) pair."""
    c = fft_corr(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    shifts = np.arange(2 * N - 1) - (N - 1)
    # For a swap of one at p to 0 and zero at q to 1:
    # delta at shift s: -h[p-s] + (1-h[q-s]) ... 
    # Actually let's use the formula: when h[p] goes 1->0 and h[q] goes 0->1
    # corr_new[s] = corr[s] + delta
    # The change in corr at shift s is:
    #   new_h[p] = 0, new_h[q] = 1 (was 0)
    #   corr(s) = sum_x h[x]*(1-h[x+s])
    #   Change: at x=p: was 1*(1-h[p+s]), now 0*(1-h[p+s]) => change: -(1-h[p+s])
    #           at x=q: was 0*(1-h[q+s]), now 1*(1-h[q+s]) => change: +(1-h[q+s])
    #   But also if x+s=p (i.e., x=p-s): was h[p-s]*(1-1)=0, now h[p-s]*(1-0)=h[p-s] => change: +h[p-s]
    #           if x+s=q (i.e., x=q-s): was h[q-s]*(1-0)=h[q-s], now h[q-s]*(1-1)=0 => change: -h[q-s]
    # So total delta at shift s: -(1-h[p+s]) + (1-h[q+s]) + h[p-s] - h[q-s]
    # where h[x] is the ORIGINAL h, valid only if 0<=x<N
    return c, ones, zeros, shifts

                best_ij = (i0 + mi, mj)
        if best_ij is not None and best_val < P - 1e-12:
            p_i = int(ones[best_ij[0]])
            q_j = int(zeros[best_ij[1]])
            h[p_i] = 0.0
            h[q_j] = 1.0
            P_full = float(np.max(fft_corr(h)))
            if P_full >= P - 1e-12:
                h[p_i] = 1.0
                h[q_j] = 0.0
                stall += 1
        else:
            if 12 <= stall <= 60:
                n_pert = 1 + min(stall // 12, 3)
                for _ in range(n_pert):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 90:
                break
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Nesterov to sharpen
    for T_val, lr_v in [(60.0, 0.02), (150.0, 0.015)]:
        if time.time() > deadline - 450:
            break
        rng_p = np.random.RandomState(seed + 100 + int(T_val))
        hc = h0.copy() + rng_p.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        d = min(deadline - 380, time.time() + 75)
        h_n, c5_n = nesterov_opt(hc, N0, d, k=45, T=T_val, lr0=lr_v)
        if c5_n < best_c5:
            best_c5, best_h = c5_n, h_n.copy()

    # Phase 2: Main binary local search at N0
    if time.time() < deadline - 250:
        h_bin = ensure_binary(best_h, N0)
        d_b = min(deadline - 120, time.time() + 400)
        h_br, c5_br = binary_ls(h_bin, N0, d_b, K=320, max_sweeps=3000)
        if c5_br < best_c5:
            best_c5, best_h = c5_br, h_br.copy()

    # Phase 3: Try smaller N for easier optimization
    N_small = max(500, N0 // 2)
    if time.time() < deadline - 150:
        h_s = upsample(best_h, len(best_h), N_small)
        h_s = project_sum(h_s, N_small / 2.0)
        d_s = min(deadline - 100, time.time() + 40)
        h_sn, c5_sn = nesterov_opt(h_s, N_small, d_s, k=30, T=80.0, lr0=0.015)
        if c5_sn < best_c5:
            best_c5, best_h = c5_sn, h_sn.copy()
        if time.time() < deadline - 80:
            h_sb = ensure_binary(best_h, N_small)
            d_sb = min(deadline - 60, time.time() + 40)
            h_sbr, c5_sbr = binary_ls(h_sb, N_small, d_sb, K=180, max_sweeps=800)
            if c5_sbr < best_c5:
                best_c5, best_h = c5_sbr, h_sbr.copy()

    # Phase 4: Second binary LS with different random seed for perturbation
    if time.time() < deadline - 80:
        h4 = ensure_binary(best_h, N0, rng=np.random.RandomState(seed + 500))
        # Perturb a few bits from current best
        rng4 = np.random.RandomState(seed + 600)
        ones4 = np.where(h4 == 1)[0]
        zeros4 = np.where(h4 == 0)[0]
        for _ in range(3):
            p = int(ones4[rng4.randint(len(ones4))])
            q = int(zeros4[rng4.randint(len(zeros4))])
            h4[p], h4[q] = 0.0, 1.0
        d4 = min(deadline - 40, time.time() + 80)
        h4r, c5_4r = binary_ls(h4, N0, d4, K=320, max_sweeps=1500)
        if c5_4r < best_c5:
            best_c5, best_h = c5_4r, h4r.copy()

    # Phase 5: Final Nesterov micro-tuning on best
    if time.time() < deadline - 30:
        hF = best_h.astype(float).copy()
        rng_f = np.random.RandomState(seed + 900)
        hF = hF + rng_f.randn(len(hF)) * 0.015
        hF = project_sum(hF, len(hF) / 2.0)
        d5 = min(deadline - 10, time.time() + 35)
        h5, c5_5 = nesterov_opt(hF, len(hF), d5, k=50, T=200.0, lr0=0.006)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final binary polish
    if time.time() < deadline - 5:
        Nfin = len(best_h)
        hF2 = ensure_binary(best_h, Nfin)
        d6 = max(time.time() + 0.5, deadline - 3)
        hF3, c5F3 = binary_ls(hF2, Nfin, d6, K=min(350, 2 * Nfin - 1), max_sweeps=5000)
        if c5F3 < best_c5:
            best_c5, best_h = c5F3, hF3.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
