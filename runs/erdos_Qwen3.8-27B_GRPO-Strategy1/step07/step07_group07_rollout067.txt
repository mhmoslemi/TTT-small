    """Vectorized swap search. Returns (h, new_c5, improved)."""
    N = len(h)
    center = N - 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, compute_c5(h), False
    
    if len(ones) > n_oi:
        oi_pool = np.random.choice(ones, n_oi, replace=False)
    else:
        oi_pool = ones
    
    L_arr = np.arange(1, N, dtype=np.int64)
    pos_corr = corr[center + 1: 2 * N - 1].astype(float)
    neg_corr = corr[:center][::-1].astype(float)
    cur_peak = float(np.max(corr))
    
    best_new_peak = cur_peak
    best_move = None
    
    for oi in oi_pool:
        oi = int(oi)
        # Compute delta_oi for positive lags
        dp = np.zeros(N - 1)
        m1 = min(N - 1 - oi, N - 1)
        if m1 >= 1:
            dp[:m1] -= (1.0 - h[oi + 1: oi + m1 + 1])
        m2 = min(oi, N - 1)
        if m2 >= 1:
            dp[:m2] += h[oi: oi - m2: -1]
        
        # Compute delta_oi for negative lags
        dn = np.zeros(N - 1)
        if m2 >= 1:
            dn[:m2] -= (1.0 - h[oi: oi - m2: -1])
        m3 = min(N - 1 - oi, N - 1)
        if m3 >= 1:
            dn[:m3] += h[oi + 1: oi + m3 + 1]
        
        # Vectorize over zeros
        zi_L = zeros[:, None] + L_arr[None, :]  # (nz, N-1)
        mask_pos = zi_L < N
        delta_zp = np.where(mask_pos, 1.0 - h[zi_L], 0.0)
        
        zi_mL = zeros[:, None] - L_arr[None, :]
        mask_neg = zi_mL >= 0
        delta_zp -= np.where(mask_neg, h[zi_mL], 0.0)
        
        new_pos_peak = (pos_corr[None, :] + dp[None, :] + delta_zp).max(axis=1)
        
        delta_zn = np.where(mask_neg, 1.0 - h[zi_mL], 0.0)
        delta_zn -= np.where(mask_pos, h[zi_L], 0.0)
        
        new_neg_peak = (neg_corr[None, :] + dn[None, :] + delta_zn).max(axis=1)
        
        new_peaks = np.maximum(new_pos_peak, new_neg_peak)
        new_peaks = np.maximum(new_peaks, 0.0)
        
        imin = int(np.argmin(new_peaks))
        if new_peaks[imin] < best_new_peak - 1e-12:
            best_new_peak = float(new_peaks[imin])
            best_move = (oi, int(zeros[imin]))
    
    if best_move is not None:
        h[best_move[0]] = 0.0
        h[best_move[1]] = 1.0
        return h, best_new_peak * (2.0 / N), True
    return h, cur_peak * (2.0 / N), False
