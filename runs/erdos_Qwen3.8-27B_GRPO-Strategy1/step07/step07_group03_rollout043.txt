    return grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(1.0 - h, L)))

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_binary(h, N):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 == nh:
        return h
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx, n1 - nh, replace=False)] = 0.0
    else:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, nh - n1, replace=False)] = 1.0
    return h

def delta_add(h, p, corr, N):
    """Compute change in corr when adding 1 at position p (0->1 flip)."""
    shifts = np.arange(2*N-1) - (N-1)
    pm = p - shifts
    pp = p + shifts
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    d = np.zeros(2*N-1)
    d[vm] += (1.0 - h[pm[vm]])
    d[vp] -= h[pp[vp]]
    return d

def binary_local_search(h, N, deadline, rng, max_sweeps=2000, two_swap_prob=0.3, three_swap_prob=0.05):
    h = h.astype(float).copy()
    if int(h.sum()) != N//2:
        h = ensure_binary(h, N)
    corr = fft_corr(h)
    P = float(corr.max())
    best_h = h.copy()
    best_P = P
    temp = 0.015
    stall = 0
    max_stall = 300
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp * 0.9985)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Try 1-swap: for each one p, find best zero q
        best_delta = 0  # we want to decrease P
        best_move = None
        best_type = 0
        
        # 1-swap: try all p, for each find best q
        # Use D matrix approach
        # D[p, s] = change in corr[s] when flipping h[p] 0->1
        # For 1-swap p->q (p is 1, q is 0): corr_new = corr - D[p] + D[q]
        # We want to minimize max(corr_new)
        
        # Precompute D for all positions (O(N^2))
        D = np.zeros((N, 2*N-1))
        for p in range(N):
            D[p] = delta_add(h, p, corr, N)
        
        # 1-swap: for p in ones, q in zeros: new_P = max(corr - D[p] + D[q])
        # For efficiency: for each p, the best q minimizes max(corr - D[p] + D[q])
        # = min_q max_s (corr[s] - D[p,s] + D[q,s])
        # For each p: base = corr - D[p], then find q minimizing max(base + D[q])
        base_all = corr[None, :] - D[ones, :]  # shape (n_ones, 2N-1)
        # For each row, we want min over q of max(base_row + D[q])
        # This is O(n_ones * n_zeros * (2N-1)) which is O(N^3) for N=380: ~55M
        # That's too slow per sweep. Let's sample.
        
        # Alternative: for each p, greedily pick q that minimizes the max
        # Even better: just evaluate a subset
        n_sample_p = min(len(ones), 380)
        p_sample = ones if n_sample_p >= len(ones) else rng.choice(ones, n_sample_p, replace=False)
        
        best_1swap_P = P
        for p in p_sample:
            base = corr - D[p]
            # For each q in zeros: new_P = max(base + D[q])
            # Vectorized: (n_zeros, 2N-1) matrix, take max along axis 1
            if len(zeros) <= 200:
                cand = base[None, :] + D[zeros, :]  # (n_zeros, 2N-1)
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
            else:
                # Sample zeros
                zs = rng.choice(zeros, 200, replace=False)
                cand = base[None, :] + D[zs, :]
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
                zs = zeros  # keep track
            if npk[mi] < best_1swap_P:
                best_1swap_P = float(npk[mi])
                best_move = (int(p), int(zeros[mi]))
                best_type = 1
        
        # 2-swap: sample random pairs
        best_2swap_P = P
        if rng.random() < two_swap_prob and len(ones) >= 4 and len(zeros) >= 4:
            n_try = min(80, max(20, N//5))
            for _ in range(n_try):
                i1, i2 = rng.randint(len(ones)), rng.randint(len(ones))
                j1, j2 = rng.randint(len(zeros)), rng.randint(len(zeros))
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_corr = corr - D[p1] - D[p2] + D[q1] + D[q2]
                new_P = float(new_corr.max())
                if new_P < best_2swap_P:
                    best_2swap_P = new_P
                    best_move = (p1, p2, q1, q2)
                    best_type = 2
        
        # 3-swap: rare
        if rng.random() < three_swap_prob and len(ones) >= 6 and len(zeros) >= 6:
            for _ in range(15):

                idx_q = rng.choice(len(zeros), 3, replace=False)
                p1,p2,p3 = int(ones[idx_p[0]]), int(ones[idx_p[1]]), int(ones[idx_p[2]])
                q1,q2,q3 = int(zeros[idx_q[0]]), int(zeros[idx_q[1]]), int(zeros[idx_q[2]])
                new_corr = corr - D[p1] - D[p2] - D[p3] + D[q1] + D[q2] + D[q3]
                new_P = float(new_corr.max())
                if new_P < best_2swap_P:
                    best_2swap_P = new_P
                    best_move = (p1,p2,p3,q1,q2,q3)
                    best_type = 3
        
        # Choose best move (with some randomness for escaping)
        if best_type > 0 and (best_move[1] < P or rng.random() < temp):
            if best_type == 1:
                h[best_move[0]] = 0.0
                h[best_move[1]] = 1.0
            elif best_type == 2:
                h[best_move[0]] = h[best_move[1]] = 0.0
                h[best_move[2]] = h[best_move[3]] = 1.0
            elif best_type == 3:
                h[best_move[0]] = h[best_move[1]] = h[best_move[2]] = 0.0
                h[best_move[3]] = h[best_move[4]] = h[best_move[5]] = 1.0
            corr = fft_corr(h)
            P = float(corr.max())
            if P < best_P - 1e-13:
                best_P = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > max_stall:
                break
            # Random perturbation
            if stall % 50 == 0:
                for _ in range(rng.randint(1,4)):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 1.0 - h[p], 1.0 - h[q]
                corr = fft_corr(h)
                P = float(corr.max())
    
    return best_h, best_P * (2.0/N)

def adam_continuous(h, N, deadline, beta=5.0, lr0=0.02, target=None):
    """Adam-based continuous optimization with softmax-smoothed max."""
    if target is None:
        target = N / 2.0
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_P = float(fft_corr(h).max())
    
    m_hat = np.zeros(N)
    v_hat = np.zeros(N)
    t = 0
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    shifts = np.arange(2*N-1) - (N-1)
    
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        # Softmax weights
        c_shifted = corr - corr.max()
        w = np.exp(beta * c_shifted)
        w /= w.sum()
        
        # Gradient: for each p, grad[p] = sum_s w[s]*(1-h[p-s])*[valid] - sum_s w[s]*h[p+s]*[valid]
        grad = np.zeros(N)
        for si in range(2*N-1):
            ws = w[si]
            if ws < 1e-10:
                continue
            s = shifts[si]
            # (1-h[p-s]) for valid p-s
            lo1 = max(0, s)
            hi1 = min(N, s + N)
            if lo1 < hi1:
                grad[lo1:hi1] += ws * (1.0 - h[lo1-s:hi1-s])
            # -h[p+s] for valid p+s
            lo2 = max(0, -s)
            hi2 = min(N, N - s)
            if lo2 < hi2:
                grad[lo2:hi2] -= ws * h[lo2+s:hi2+s]
        
        # Adam update
        m_hat = b1*m_hat + (1-b1)*grad
        v_hat = b2*v_hat + (1-b2)*grad**2
        m_hat_c = m_hat / (1 - b1**t)
        v_hat_c = v_hat / (1 - b2**t)
        h = project_sum(h - lr * m_hat_c / (np.sqrt(v_hat_c) + eps), target)
        
        P = float(fft_corr(h).max())
        if P < best_P - 1e-12:
            best_P = P
            best_h = h.copy()
        
        if t % 200 == 0:
            lr *= 0.97
            beta = min(20.0, beta * 1.1)
    
    return best_h, best_P * (2.0/N)

    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
