def solve_lp_for_h(v, N):
    """Given v (length N), find h to minimize max correlation."""
    # c[N-1+s] = sum_{n valid} h[n] * v[n+s]
    # Build constraint matrix A for: A @ [h; t] <= b
    # For each shift s: sum_n A_s[n] * h[n] - t <= 0
    # A_s[n] = v[n+s] if 0<=n+s<N, else 0
    
    L = 2*N - 1  # number of shifts
    # c_coeffs: L x N matrix where c_coeffs[m, n] = v[n + (m - N + 1)] if valid
    
    # For shift m (0 to L-1), s = m - (N-1)
    # c[m] = sum_n h[n] * v[n+s] where 0<=n<N and 0<=n+s<N
    
    # Build A_eq, b_eq for the LP
    # Minimize [0, 0, ..., 0, 1] @ [h; t]
    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0  # minimize t
    
    # Inequality constraints: for each m, sum_n v[n+s_m]*h[n] - t <= 0
    A_ub = np.zeros((L, N + 1))
    for m in range(L):

        for n in range(N):

            if 0 <= ns < N:

        A_ub[m, N] = -1.0
    b_ub = np.zeros(L)
    
    # Bounds: 0 <= h[n] <= 1, t free (or t >= 0)
    bounds = [(0, 1)] * N + [(0, None)]
    
    # Equality: sum h = N/2
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([N / 2.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    return h_init, None

import numpy as np
import time
from scipy.optimize import linprog

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft((1.0 - h).astype(float), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def build_A(h, N):
    """Build the L×N matrix where A[m,n] gives coefficient of h[n] (or v[n]) in c[m].
    For c[m] = sum_n h[n]*v[n+s]: A[m,n] = v[n+s] if valid.
    For c[m] = sum_n h[n+s]*v[n]: A[m,n] = h[n+s] if valid.
    """
    L = 2*N - 1
    s_vals = np.arange(L) - (N-1)
    n_vals = np.arange(N)
    idx = n_vals[None, :] + s_vals[:, None]
    valid = (idx >= 0) & (idx < N)
    idx_c = np.clip(idx, 0, N-1)
    A = np.where(valid, h[idx_c], 0.0)
    return A

def solve_lp(A, N, target):
    """Solve min t s.t. A@x <= t, 0<=x<=1, sum(x)=target. Returns x, t."""
    L = A.shape[0]
    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0
    A_ub = np.column_stack([A, -np.ones(L)])
    b_ub = np.zeros(L)
    bounds = [(0.0, 1.0)] * N + [(0.0, None)]
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([target])
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:N].copy(), float(res.x[N])
    return None, None

```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft((1.0 - h).astype(float), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def build_A(v, N, mode='h'):
    L = 2 * N - 1
    s = np.arange(L) - (N - 1)
    n = np.arange(N)
    if mode == 'h':
        idx = n[None, :] + s[:, None]
    else:
        idx = n[None, :] - s[:, None]
    valid = (idx >= 0) & (idx < N)
    idx_c = np.clip(idx, 0, N - 1)
    return np.where(valid, v[idx_c], 0.0)

def solve_lp(A, N, target):
    L = A.shape[0]
    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0
    A_ub = np.column_stack([A, -np.ones(L)])
    b_ub = np.zeros(L)
    bounds = [(0.0, 1.0)] * N + [(0.0, None)]
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([target])
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:N].copy(), float(res.x[N])
    return None, None

def alternating_lp(h0, N, deadline, max_iters=30):
    target = N / 2.0
    h = np.clip(h0, 0, 1).astype(float).copy()
    # Project to satisfy sum constraint
    s = h.sum()
    if abs(s - target) > 1e-10:
        # Simple projection
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if diff > 0:
            mask = h < 1.0
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        else:
            mask = h > 0.0
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    prev_c = best_c
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Solve for h (v = 1-h fixed)
        v = 1.0 - h
        A_h = build_A(v, N, 'h')
        h_new, _ = solve_lp(A_h, N, target)
        if h_new is not None:
            h = h_new
        # Solve for v (h fixed)
        A_v = build_A(h, N, 'v')
        v_new, _ = solve_lp(A_v, N, target)
        if v_new is not None:
            h = 1.0 - v_new
        
        c_now = float(np.max(fft_corr(h)))
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
        if abs(c_now - prev_c) < 1e-10:
            break
        prev_c = c_now
    
    return best_h, best_c

def bin_local_search(h, N, deadline, max_iters=10000, ntri=25, anneal=0.005):

    nh = N // 2
    n1 = int(h.sum())

            h[idx[np.random.randint(len(idx))]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        for i in range(nh - n1):
            h[idx[np.random.randint(len(idx))]] = 1
    
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stall = 0
    temp = anneal
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            improved = True
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd = 0.0
        best_move = None
        
        for _ in range(ntri):
            i = np.random.randint(len(ones))
            j = np.random.randint(len(zeros))
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:

            h[oi], h[zi] = 1.0, 0.0
        
        if best_move:
            h[best_move[0]], h[best_move[1]] = 0.0, 1.0
            improved = True
        
        if not improved:
            stall += 1
        else:
            stall = 0
        
        corr = fft_corr(h)
        temp = max(0.0, anneal * 0.9995 ** it)
    
    return best_h, best_c

def make_binary(h, N):

    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        order = np.argsort(h[idx])  # flip smallest 1s
        for i in range(n1 - nh):
            h[idx[order[i]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.argsort(-h[idx])  # flip largest 0s
        for i in range(nh - n1):
            h[idx[order[i]]] = 1
    return h
