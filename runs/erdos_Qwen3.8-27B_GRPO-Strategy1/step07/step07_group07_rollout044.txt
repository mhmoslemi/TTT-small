                    best_new_peak = new_peak

                c = float(np.max(corr)) * (2.0 / N)

        it += 1
    
    return best_h, best_c

def vec_swap_eval(h, corr, os_, zs, top_lags, N):
    ns = len(os_)
    nz = len(zs)
    nc = len(top_lags)
    
    # Compute vec_oi: shape (ns, nc)
    vec_oi = np.zeros((ns, nc))
    for i, oi in enumerate(os_):
        for j, L in enumerate(top_lags):
            v = 0.0
            if 0 <= oi + L < N:
                v += h[oi + L]
            if 0 <= oi - L < N:
                v -= h[oi - L]
            vec_oi[i, j] = v
    
    # Compute vec_zi: shape (nz, nc)
    vec_zi = np.zeros((nz, nc))
    for j, zi in enumerate(zs):

            v = 0.0
            if 0 <= zi + L < N:
                v += h[zi + L]
            if 0 <= zi - L < N:
                v -= h[zi - L]

        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 60:
            lr0 *= 0.82
            stale = 0
        g = grad_corr_top(h, corr, k)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def vec_lag_effect(h, positions, top_lags, N):
    ns = len(positions)
    nc = len(top_lags)
    out = np.zeros((ns, nc))
    for i in range(ns):
        p = positions[i]
        for j, L in enumerate(top_lags):
            v = 0.0
            if 0 <= p + L < N:
                v += h[p + L]
            if 0 <= p - L < N:
                v -= h[p - L]
            out[i, j] = v
    return out

def binary_sa(h, deadline, n_crit=50, s_sz=60, T0=0.8, cool_rate=0.9995, max_iters=100000):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    c = float(np.max(corr)) * (2.0 / N)
    best_c = c
    best_h = h.copy()
    center = N - 1
    T = T0
    it = 0
    no_imp = 0
    top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
    while time.time() < deadline and it < max_iters:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
            top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        else:
            no_imp += 1
        if no_imp > 500:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(3):
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi], h[zi] = h[zi], h[oi]
            no_imp = 0
            it += 1
            T *= cool_rate
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        vec_oi = vec_lag_effect(h, os_, top_lags, N)
        vec_zi = vec_lag_effect(h, zs, top_lags, N)
        base = corr[center + top_lags]
        delta = vec_oi[:, np.newaxis, :] - vec_zi[np.newaxis, :, :]
        new_corrs = base[np.newaxis, np.newaxis, :] + delta
        new_peaks = np.max(new_corrs, axis=2)
        best_idx = np.unravel_index(np.argmin(new_peaks), new_peaks.shape)
        best_new_peak = float(new_peaks[best_idx]) * (2.0 / N)
        cur_peak = float(c)
        accept = False
        if best_new_peak < cur_peak:
            accept = True
        else:
            diff = (best_new_peak - cur_peak) / max(T, 1e-10)
            if diff < 50 and np.random.random() < np.exp(-diff):
                accept = True
        if accept:
            oi = int(os_[best_idx[0]])
            zi = int(zs[best_idx[1]])
            h[oi] = 0
            h[zi] = 1
        T *= cool_rate
        it += 1
    return best_h, best_c

def double_swap_perturb(h, n_swaps=2):
    N = len(h)
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_swaps):
        if len(ones) < 1 or len(zeros) < 1:
            break
        oi = int(np.random.choice(ones))
        zi = int(np.random.choice(zeros))
        h[oi] = 0
        h[zi] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Quick continuous optimization (150s)
    d1 = min(t0 + 150, deadline - 600)
    h1 = project(best_h + np.random.randn(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=50, lr0=0.018, mom=0.90)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Binary SA with vectorized eval (300s)
    if time.time() < deadline - 350:
        d2 = min(time.time() + 300, deadline - 300)
        hb = fix_binary(best_h)
        hb, c2 = binary_sa(hb, d2, n_crit=60, s_sz=70, T0=1.0, cool_rate=0.9997, max_iters=200000)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    # Phase 3: Continuous refinement from binary (120s)
    if time.time() < deadline - 200:
        d3 = min(time.time() + 120, deadline - 180)
        h3 = project(best_h + np.random.randn(N0) * 0.03, N0 / 2.0)
        h3, c3 = cont_opt(h3, d3, k=60, lr0=0.01, mom=0.92)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Try different N values
    for N_try in [400, 440, 500, 360]:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2.0, N0, endpoint=False)
        x_new = np.linspace(0, 2.0, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        rem = deadline - 100 - time.time()
        if rem < 30:
            break
        d4 = time.time() + rem * 0.45
        h_bt, c4 = binary_sa(h_bt, d4, n_crit=50, s_sz=55, T0=0.6, cool_rate=0.9996, max_iters=80000)
        rem2 = deadline - 80 - time.time()
        if rem2 > 25:
            hc = project(h_bt + np.random.randn(N_try) * 0.04, N_try / 2.0)
            d5 = time.time() + rem2 * 0.6
            hc, c5v = cont_opt(hc, d5, k=50, lr0=0.012, mom=0.90)
            if c5v < c4:
                c4, h_bt = c5v, hc
        if c4 < best_c:
            best_c, best_h = c4, h_bt.copy()
            N0 = N_try

    # Phase 5: Final binary polish (remaining time)
    if time.time() < deadline - 5:
        rem = deadline - 5 - time.time()
        Nf = len(best_h)
        hf = fix_binary(best_h)
        hf, cf = binary_sa(hf, deadline - 2, n_crit=70, s_sz=80, T0=0.4, cool_rate=0.9999, max_iters=300000)
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    # Final: try both binary and continuous, return best
    final_h = best_h.copy()
    final_c = compute_c5(final_h)
    hB = fix_binary(final_h)
    cB = compute_c5(hB)
    if cB < final_c:
        final_h, final_c = hB, cB

    # Quick continuous nudge
    if time.time() < deadline - 1:
        hf2 = project(final_h + np.random.randn(len(final_h)) * 0.02, len(final_h) / 2.0)
        hf2, cf2 = cont_opt(hf2, deadline - 1, k=60, lr0=0.008, mom=0.93)
        if cf2 < final_c:
            final_h, final_c = hf2, cf2

    return final_h, compute_c5(final_h), len(final_h)
