def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def sa_core(h_bin, deadline, n_peaks, n_per_peak, cool_rate, t_start, max_iters, reheat_every):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.002)
        if it % reheat_every == 0 and no_impr > reheat_every // 2:
            T = max(T * 2.5, 0.012)
            no_impr = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp = vi[mask]
                bq = (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp = vi[mask]
                bq = (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = min(8, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 10
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.002 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            dpc = deltas[p].copy()
            corr += dpc - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p + N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q + N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        if no_impr > 5000:
            break
    return best_h, best_c

def jam_lag(h, lag, N, n_iters=500):
    """Focus SA moves on reducing a specific lag."""
    h = h.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    for _ in range(n_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        k_off = int(lag) - (N - 1)
        if k_off >= 0:
            vi = np.arange(0, N - k_off)
            mask = (h[vi] == 1) & (h[vi + k_off] == 0)
            bp = vi[mask]
            bq = (vi + k_off)[mask]
        else:
            M = -k_off
            vi = np.arange(M, N)
            mask = (h[vi] == 1) & (h[vi - M] == 0)
            bp = vi[mask]
            bq = (vi - M)[mask]
        if len(bp) == 0:
            break
        idx = np.random.randint(len(bp))
        p, q = int(bp[idx]), int(bq[idx])
        new_corr = corr + deltas[p] - deltas[q]
        nc = float(new_corr.max())
        if nc < best_c - 1e-12:
            h[p], h[q] = 0.0, 1.0
            corr = new_corr
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p + N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q + N] = s
            best_c = nc
            best_h = h.copy()
        else:
            if np.random.random() < 0.02:
                h2 = best_h.copy()
                p2, q2 = np.random.randint(N//2), np.random.randint(N//2)
                ones_i = np.where(h2 == 1)[0]
                zeros_i = np.where(h2 == 0)[0]
                pp, qq = int(np.random.choice(ones_i)), int(np.random.choice(zeros_i))
                h2[pp], h2[qq] = 0.0, 1.0
                c2 = float(fft_corr(h2).max())
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
                    h = h2
                    corr = fft_corr(h)
                    s, deltas = build_deltas(h)
    return best_h, best_c

def grad_polish(h_in, N, deadline, k_top=30, lr_init=0.006):
    h = h_in.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.80
    lr = lr_init
    it = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        pk = float(corr_y.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr_y, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.96
            beta = min(0.93, 0.78 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)
