import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_corr_top(h, corr, k=50):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def cont_opt(h, deadline, k=50, lr0=0.02, mom=0.90):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    lr = lr0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale % 40 == 40:
            lr *= 0.88
        if stale > 250:
            break
        g = grad_corr_top(h, corr, k)
        h_new = project(h + mom * v - lr * g, tgt)
        v = mom * v + 0.1 * (h_new - h)
        h = h_new
    return best_h, best_c

def to_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_vec_search(h, deadline, n_crit=50, s_sz=50, max_iters=200000):
    N = len(h)
    h = to_binary(h.copy())
    corr = compute_corr(h)
    cur_peak = float(np.max(corr))
    best_c = cur_peak * (2.0 / N)
    best_h = h.copy()
    center = N - 1
    it = 0
    no_imp = 0
    temp = 5.0

    ones_list = np.where(h == 1)[0]
    zeros_list = np.where(h == 0)[0]

    while time.time() < deadline and it < max_iters:
        c_val = cur_peak * (2.0 / N)
        if c_val < best_c - 1e-14:
            best_c = c_val
            best_h = h.copy()
            no_imp = 0

            break

        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        corr_top = corr[center + top_lags]

        ns = min(len(ones_list), s_sz)
        nz = min(len(zeros_list), s_sz)
        os_ = np.random.choice(ones_list, ns, replace=False)

        D_oi = np.zeros((ns, n_crit))
        for i in range(ns):
            oi = os_[i]
            for j in range(n_crit):
                L = top_lags[j]

                if 0 <= oi - L < N:
                    d += h[oi - L]

        D_zi = np.zeros((nz, n_crit))
        for i in range(nz):
            zi = zs[i]

                L = top_lags[j]

                if 0 <= zi - L < N:
                    d += h[zi - L]
                D_zi[i, j] = d

        new_peaks = np.max(corr_top[None, None, :] - D_oi[:, None, :] - D_zi[None, :, :], axis=2)
        min_flat = int(np.argmin(new_peaks))
        min_i, min_j = divmod(min_flat, nz)
        best_new = float(new_peaks[min_i, min_j])

        if best_new < cur_peak - 1e-10:
            h[os_[min_i]] = 0
            h[zs[min_j]] = 1
            ones_list = np.where(h == 1)[0]
            zeros_list = np.where(h == 0)[0]
            corr = compute_corr(h)
            cur_peak = float(np.max(corr))
        elif np.random.random() < np.exp(-1.0 / max(temp, 0.3)):
            oi = int(np.random.choice(ones_list))
            zi = int(np.random.choice(zeros_list))
            h[oi] = 0
            h[zi] = 1
            ones_list = np.where(h == 1)[0]
            zeros_list = np.where(h == 0)[0]
            corr = compute_corr(h)
            cur_peak = float(np.max(corr))

        it += 1

    return best_h, best_c
