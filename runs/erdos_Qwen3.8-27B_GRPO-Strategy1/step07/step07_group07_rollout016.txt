        # h_minus[p, j] = h[p - crit_lags[j]] if 0 <= p - crit_lags[j] < N else 0
        
        h_plus = np.zeros((N, n_crit))
        h_minus = np.zeros((N, n_crit))
        for j, L in enumerate(crit_lags):
            if L >= 0:
                start, end = 0, N - L
                h_plus[:end, j] = h[L:]  # h[0+L], h[1+L], ..., h[N-1-L+L] = h[L:N]
                # Actually: for p in 0..N-1-L: h[p+L] = h[L:p+1]
                # h_plus[p, j] = h[p+L] for p = 0..N-1-L
                # That's h[L:N] placed in h_plus[0:N-L, j]
                h_plus[:N-L, j] = h[L:]
            else:  # L < 0, so p+L >= 0 means p >= -L
                start = -L
                h_plus[start:, j] = h[:N+L]
                # h_plus[p, j] = h[p+L] for p = -L..N-1
                # That's h[0:N+L] placed in h_plus[-L:N, j]
            
            # h_minus[p, j] = h[p-L] if 0 <= p-L < N
            if L >= 0:
                # p-L >= 0 means p >= L
                h_minus[L:, j] = h[:N-L]
            else:
                h_minus[:N+L, j] = h[-L:]
        
        # Now for oi and zi:
        # delta[oi, zi] = h_plus[oi] + h_minus[oi] - h_plus[zi] - h_minus[zi]
        # (dot with what? No, it's element-wise over lags)
        
        # h_plus[oi] has shape (n_crit,)
        # We want: for each (oi, zi): new_vals = corr_crit + h_plus[oi] + h_minus[oi] - h_plus[zi] - h_minus[zi]
        
        # For all oi in os_: h_plus_os = h_plus[os_] shape (ns, n_crit)
        # For all zi in zs: h_plus_zs = h_plus[zs] shape (nz, n_crit)
        
        h_plus_os = h_plus[os_]  # (ns, n_crit)
        h_minus_os = h_minus[os_]  # (ns, n_crit)
        h_plus_zs = h_plus[zs]  # (nz, n_crit)
        h_minus_zs = h_minus[zs]  # (nz, n_crit)
        
        # delta[oi_i, zi_j] = h_plus_os[i] + h_minus_os[i] - h_plus_zs[j] - h_minus_zs[j]
        # new_vals[oi_i, zi_j, k] = corr_crit[k] + delta[oi_i, zi_j, k]
        # peak[oi_i, zi_j] = max_k new_vals[oi_i, zi_j, k]
        
        # Using broadcasting:
        # lhs = h_plus_os + h_minus_os  # (ns, n_crit)
        # rhs = h_plus_zs + h_minus_zs  # (nz, n_crit)
        # delta_all = lhs[:, np.newaxis, :] - rhs[np.newaxis, :, :]  # (ns, nz, n_crit)
        # new_all = corr_crit[np.newaxis, np.newaxis, :] + delta_all  # (ns, nz, n_crit)
        # peaks = new_all.max(axis=2)  # (ns, nz)
        
        lhs = h_plus_os + h_minus_os  # (ns, n_crit)
        rhs = h_plus_zs + h_minus_zs  # (nz, n_crit)
        delta_all = lhs[:, np.newaxis, :] - rhs[np.newaxis, :, :]  # (ns, nz, n_crit)
        new_all = corr_crit[np.newaxis, np.newaxis, :] + delta_all
        peaks = new_all.max(axis=2)  # (ns, nz)
        
        flat_idx = np.argmin(peaks)
        i_best, j_best = np.unravel_index(flat_idx, peaks.shape)
        
        if peaks[i_best, j_best] < cur_peak - 1e-8:
            oi = os_[i_best]
            zi = zs[j_best]
            h[oi] = 0
            h[zi] = 1
            corr = compute_corr(h)
            cur_peak = float(np.max(corr))
            if cur_peak * (2.0/N) < best_c - 1e-12:
                best_c = cur_peak * (2.0/N)
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
            # Random perturbation
            if no_imp > 30 and np.random.random() < 0.3:
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi] = 0
                h[zi] = 1
                corr = compute_corr(h)
                cur_peak = float(np.max(corr))
    
    return best_h, best_c

def compute_corr(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def build_lookup(h, crit_lags):
    N = len(h)
    nc = len(crit_lags)
    hp = np.zeros((N, nc))
    hm = np.zeros((N, nc))
    for j, L in enumerate(crit_lags):
        if L >= 0:
            hp[:N - L, j] = h[L:]
            hm[L:, j] = h[:N - L]
        else:
            hp[-L:, j] = h[:N + L]
            hm[:N + L, j] = h[-L:]
    return hp, hm

            break
        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            L = int(m) - center
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif abs(L) < N:
                if L > 0:
                    grad[:N - L] += (1.0 - h[L:])
                    grad[L:] -= h[:N - L]

        h_new = project(h + mom * v - lr0 * grad, tgt)

        h = h_new
    return best_h, best_c
