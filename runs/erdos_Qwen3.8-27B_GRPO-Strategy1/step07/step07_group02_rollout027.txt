    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def make_binary(h, n_ones):
    h = np.round(h).clip(0, 1).astype(float)
    s = int(h.sum())
    if s > n_ones:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, s - n_ones, replace=False)] = 0
    elif s < n_ones:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, n_ones - s, replace=False)] = 1
    return h

        h = h_new

            lr *= 0.97
        
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    
    # Phase 1: Binary search on initial (N=380)
    d1 = min(deadline - 300, t0 + 250)
    h1 = (h0 > 0.5).astype(float)
    h1 = make_binary(h1, N0 // 2)
    h1, c1 = binary_search_v2(h1, N0, d1, sample_sz=200, max_stall=50, kick_size=5)
    c5_1 = c1 * (2.0 / N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    
    # Phase 2: Continuous refinement
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 100)
        hc = best_h + np.random.randn(N0) * 0.05
        hc = project_01_sum(hc, N0 / 2.0)
        hc, c5c = continuous_refine(hc, N0, d2, init_lr=0.01)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc.copy()
    
    # Phase 3: Binary search again from perturbed continuous
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 150)
        h3 = (best_h > 0.5).astype(float)
        h3 = make_binary(h3, N0 // 2)
        h3, c3 = binary_search_v2(h3, N0, d3, sample_sz=250, max_stall=40, kick_size=6)
        c5_3 = c3 * (2.0 / N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
    
    # Phase 4: Try N=400
    N_alt = 400
    if time.time() < deadline - 150:
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_alt, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_alt = make_binary(h_int, N_alt // 2)
        d4 = min(deadline - 100, time.time() + 120)
        h4, c4 = binary_search_v2(h_alt, N_alt, d4, sample_sz=200, max_stall=40, kick_size=5)
        c5_4 = c4 * (2.0 / N_alt)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
    
    # Phase 5: Final binary polish
    if time.time() < deadline - 10:
        hF = (best_h > 0.5).astype(float)
        NF = len(hF)
        hF = make_binary(hF, NF // 2)
        d5 = deadline - 2
        hF, cF = binary_search_v2(hF, NF, d5, sample_sz=200, max_stall=30, kick_size=3)
        c5F = cF * (2.0 / NF)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    
    # Final check: compare continuous vs binary
    final_c5 = compute_c5(best_h, len(best_h))
    h_bin = (best_h > 0.5).astype(float)
