def fft_corr(h):
    N = len(h)

def corr_k(h, k, N):
    """Compute overlap at lag k (k can be negative or positive)"""
    if k >= 0:
        return np.dot(h[k:N], 1.0 - h[:N-k])
    else:
        M = -k
        return np.dot(h[:N-M], 1.0 - h[M:N])

def lin_grad_at_k(k, h, N):
    """Gradient of corr_k with respect to h"""
    g = np.zeros(N)
    if k == 0:

    Nc = max(30, N0 // 4)
    x0 = np.linspace(0, 2, N0, endpoint=False)
    xc = np.linspace(0, 2, Nc, endpoint=False)
    hc = ensure_sum(np.interp(xc, x0, h_start), Nc)
    d1 = min(deadline, time.time() + 50)
    hc, _ = block_flip_search(hc, Nc, d1, max_iters=250, block_sizes=(1,2,3))
    xf = np.linspace(0, 2, N_t, endpoint=False)
    hf = ensure_sum(np.interp(xf, xc, hc), N_t)
    d2 = min(deadline, time.time() + 80)
    hf, _ = block_flip_search(hf, N_t, d2, max_iters=400, block_sizes=(1,2,3,4))
    return hf

def targeted_search(h, N, deadline, max_iters=5000):
    h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline or stall > 200:
            break
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_c - 1e-14:
            best_c = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ctr = N - 1
        worst = np.argsort(corr)[::-1][:6]
        improved = False
        for wk in worst:
            k = int(wk - ctr)
            if k > 0 and k < N:
                mask = (h == 1) & (np.roll(h, -k) == 0)
                mask[np.arange(N) + k >= N] = False
                if mask.any():
                    ci = int(np.random.choice(np.where(mask)[0]))
                    h[ci], h[ci+k] = 0.0, 1.0
                    if float(np.max(fft_corr(h))) <= pk:
                        improved = True
                    else:
                        h[ci], h[ci+k] = 1.0, 0.0
            elif k < 0 and -k < N:
                M = -k
                mask = (h[M:] == 1) & (np.roll(h[M:], M) == 0)
                if mask.any():
                    ci = int(M + np.random.choice(np.where(mask)[0]))
                    h[ci], h[ci-M] = 0.0, 1.0
                    if float(np.max(fft_corr(h))) <= pk:
                        improved = True
                    else:
                        h[ci], h[ci-M] = 1.0, 0.0
        if not improved:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            h[int(ones[np.random.randint(len(ones))])], h[int(zeros[np.random.randint(len(zeros))])] = 0.0, 1.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = ensure_sum(h0, N0)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Block flip search
    d1 = min(dl - 600, t0 + 70)
    hb, _ = block_flip_search(best_h, N0, d1, max_iters=300, block_sizes=(1,2,3,4,5))
    c1 = compute_c5(hb, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, hb.copy()

    # Phase 2: Coarse-to-fine
    if time.time() < dl - 500:
        d2 = min(dl - 450, time.time() + 130)
        hcf = coarse_to_fine(best_h, N0, d2)
        c2 = compute_c5(hcf, N0)
        if c2 < best_c5:
            best_c5, best_h = c2, hcf.copy()

    # Phase 3: L-BFGS top-k (multiple k, multiple restarts)
    if time.time() < dl - 350:
        d3 = min(dl - 300, time.time() + 180)
        target = float(N0) / 2.0
        for ktop in [5, 12, 25]:
            if time.time() > d3:
                break
            for tr in range(3):
                if time.time() > d3:
                    break
                hr = best_h + np.random.randn(N0) * (0.14 - 0.03 * tr)
                hr = proj01(hr, target)
                lam = 4.0
                for _ in range(5):
                    if time.time() > d3:
                        break
                    res = sp_min(topk_obj, hr, args=(28.0, ktop, target, lam),
                                 jac=True, method='L-BFGS-B',
                                 bounds=[(0, 1)] * N0,
                                 options={'maxiter': 120, 'maxfun': 350, 'ftol': 1e-15})
                    hr = proj01(res.x, target)
                    lam *= 3.0
                c5t = compute_c5(hr, N0)
                if c5t < best_c5:
                    best_c5, best_h = c5t, hr.copy()

    # Phase 4: Targeted local search
    if time.time() < dl - 250:
        d4 = min(dl - 200, time.time() + 150)
        ht, _ = targeted_search(best_h, N0, d4, max_iters=8000)
        c4 = compute_c5(ht, N0)
        if c4 < best_c5:
            best_c5, best_h = c4, ht.copy()

    # Phase 5: More aggressive block search with larger blocks
    if time.time() < dl - 180:
        d5 = min(dl - 140, time.time() + 120)
        h5, _ = block_flip_search(best_h, N0, d5, max_iters=600, block_sizes=(1,2,3,4,6,8), ntries=35)
        c5v = compute_c5(h5, N0)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()

    # Phase 6: Try different resolutions
    if time.time() < dl - 120:
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        for Nt in [300, 340, 360, 420]:
            if time.time() > dl - 60:
                break
            if Nt == len(best_h) or Nt % 2:
                continue
            xn = np.linspace(0, 2, Nt, endpoint=False)
            hi = ensure_sum(np.interp(xn, x_b, best_h), Nt)
            sd = min(dl - 50, time.time() + 40)
            hr, _ = block_flip_search(hi, Nt, sd, max_iters=250, block_sizes=(1,2,3,4))
            c5r = compute_c5(hr, Nt)
            if c5r < best_c5:
                best_c5, best_h = c5r, hr.copy()
                x_b = xn

    # Phase 7: Final targeted search
    if time.time() < dl - 50:
        d7 = min(dl - 30, time.time() + 60)
        h7, _ = targeted_search(best_h, N0, d7, max_iters=6000)
        c7 = compute_c5(h7, N0)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Phase 8: Final block polish
    if time.time() < dl - 5:
        hf = ensure_sum(best_h, N0)
        hf, _ = block_flip_search(hf, N0, dl - 2, max_iters=250, block_sizes=(1,2,3))
        cf = compute_c5(hf, N0)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Final: compare continuous vs binary, ensure constraint
    fc5 = compute_c5(best_h, N0)
    fb = ensure_sum(best_h, N0)
    fc5b = compute_c5(fb, N0)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    else:
        best_h = ensure_sum(best_h, N0)
        fc5 = compute_c5(best_h, N0)
    return best_h, fc5, len(best_h)
