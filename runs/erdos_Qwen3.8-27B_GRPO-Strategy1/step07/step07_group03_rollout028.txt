    target = N / 2.0
    best_h = h.copy()
    best_val = float(fft_corr(h).max())
    m = np.zeros(N)  # momentum
    beta = 0.9
    lr = lr0
    
    for it in range(10000):
        if time.time() > deadline:
            break
        
        corr = fft_corr(h)
        cur_max = float(corr.max())
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_h = h.copy()
        
        # Get top-k active constraints
        top_k = np.argpartition(corr, -k_active)[-k_active:]
        
        # Compute subgradient
        grad = np.zeros(N)
        center = N - 1
        for m_idx in top_k:
            L = int(m_idx - center)

            # (1-h[j+L]) for valid j+L

                grad[:N-L] += (1.0 - h[L:])
                # -h[j-L] valid when j >= L
                grad[L:] -= h[:N-L]
            else:

                # j+L = j-M valid when j >= M
                grad[M:] += (1.0 - h[:N-M])
                # -h[j-L] = -h[j+M] valid when j+M < N, i.e., j < N-M
                grad[:N-M] -= h[M:]

        h_new = project_sum(h - lr * m, target)

            lr *= 0.99
        if it % 200 == 0:
            beta = min(0.95, 0.9 + 0.0001 * it)
        
        h = h_new

    Returns V (for 1→0) and W (for 0→1) matrices of shape (N, 2N-1)."""
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)
    # V[i] = effect of h[i] going from 1 to 0 on corr
    # V[i, s] = -(1-h[i+shift[s]]) + h[i-shift[s]]
    idx_plus = np.arange(N)[:, None] + shifts[None, :]
    valid_plus = (idx_plus >= 0) & (idx_plus < N)
    A = np.where(valid_plus, 1.0 - h[np.clip(idx_plus, 0, N-1)], 0.0)
    idx_minus = np.arange(N)[:, None] - shifts[None, :]
    valid_minus = (idx_minus >= 0) & (idx_minus < N)
    B = np.where(valid_minus, h[np.clip(idx_minus, 0, N-1)], 0.0)
    V = -A + B  # effect of removing a 1
    W = A - B   # effect of adding a 1
    return V, W

- In 300 seconds: 6000-9000 iterations

That's very good! But the issue is that after each improvement, we need to recompute D (effect matrix) because h changed. Actually, for a single swap only one D row changes... but it's easier to just recompute.

Wait, actually for a single swap p→q:
- D[p] changes because h[p] changed from 1 to 0
- D[q] changes because h[q] changed from 0 to 1
- Other D rows might change too! Because A[i] depends on h[i+shift] and B[i] depends on h[i-shift].
- If h[p] changes, then D[i] changes for all i where i+shift = p or i-shift = p, i.e., for all i.

So we need to recompute D after every swap. That's O(N^2) = 144K for N=380. Negligible compared to the search.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum_bin(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def compute_D(h, N):
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)
    idx_p = np.arange(N)[:, None] + shifts[None, :]
    vp = (idx_p >= 0) & (idx_p < N)
    A = np.where(vp, 1.0 - h[np.clip(idx_p, 0, N - 1)], 0.0)
    idx_m = np.arange(N)[:, None] - shifts[None, :]
    vm = (idx_m >= 0) & (idx_m < N)
    B = np.where(vm, h[np.clip(idx_m, 0, N - 1)], 0.0)
    return A - B

def binary_local_search(h, N, deadline, rng, max_iters=10000, batch_sz=30):
    h = ensure_sum_bin(h, N)
    c_cur = fft_corr(h)
    best_c = float(c_cur.max())
    best_h = h.copy()
    stall = 0
    L = 2 * N - 1

    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, nz = len(ones), len(zeros)
        if n1 < 1 or nz < 1:
            break

        D = compute_D(h, N)
        D_ones = D[ones]
        D_zeros = D[zeros]
        improved = False

        best_val = best_c
        best_pq = None
        bs = min(batch_sz, n1)
        for start in range(0, n1, bs):
            end = min(start + bs, n1)
            batch = D_ones[start:end]
            arr = c_cur[None, None, :] - batch[:, None, :] + D_zeros[None, :, :]
            mx = arr.max(axis=2)
            fi, fj = np.unravel_index(np.argmin(mx), mx.shape)
            if float(mx[fi, fj]) < best_val - 1e-12:
                best_val = float(mx[fi, fj])
                best_pq = (int(ones[start + fi]), int(zeros[fj]))

        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c_cur = fft_corr(h)
            improved = True

        n_2swap = min(300, max(50, N // 2))
        for _ in range(n_2swap):
            if n1 < 2 or nz < 2:
                break
            i1, i2 = rng.randint(0, n1, 2)
            j1, j2 = rng.randint(0, nz, 2)
            if i1 == i2 or j1 == j2:
                continue
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            nc = c_cur - D[p1] - D[p2] + D[q1] + D[q2]
            nm = float(nc.max())
            if nm < best_c - 1e-12:
                h[p1] = h[p2] = 0.0
                h[q1] = h[q2] = 1.0
                c_cur = nc
                improved = True
                break

        if improved:
            cur = float(c_cur.max())
            if cur < best_c - 1e-13:
                best_c = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                for _ in range(4):
                    p = int(ones[rng.randint(n1)])
                    q = int(zeros[rng.randint(nz)])
                    h[p], h[q] = 0.0, 1.0
                c_cur = fft_corr(h)
                stall = 0
            elif stall > 80:
                break

    return best_h, best_c * (2.0 / N)

def subgradient_opt(h, N, deadline, k_lo=15, k_hi=80, lr0=0.008):
    h = h.astype(float).copy()
    dx = 2.0 / N
    target = N / 2.0
    best_h = h.copy()
    best_val = float(fft_corr(h).max())
    m = np.zeros(N)
    beta = 0.88
    lr = lr0
    it = 0

    while time.time() < deadline:
        corr = fft_corr(h)
        cur_max = float(corr.max())
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_h = h.copy()

        frac = min(1.0, it / 600.0)
        k = max(5, int(k_lo + frac * (k_hi - k_lo)))
        top_k = np.argpartition(corr, -k)[-k:]

        grad = np.zeros(N)
        center = N - 1
        for m_idx in top_k:
            s = int(m_idx - center)
            if s >= 0:
                if N - s > 0:
                    grad[:N - s] += (1.0 - h[s:])
                    grad[s:] -= h[:N - s]
            else:
                M = -s
                if M < N:
                    grad[M:] += (1.0 - h[:N - M])
                    grad[:N - M] -= h[M:]
        grad /= k

        m = beta * m + grad
        h = project_sum(h - lr * m, target)
        it += 1
        if it % 100 == 0:
            lr *= 0.985
        if it % 400 == 0:
            beta = min(0.95, 0.88 + 0.00008 * it)

    return best_h, best_val * dx

def upsample_fn(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)

    # Phase 2: Subgradient continuous at N0
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + budget_s * 0.14)
        hc = best_h + rng.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        h2, c2 = subgradient_opt(hc, N0, d2, k_lo=12, k_hi=50, lr0=0.007)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Upscale to 2*N0, binary search
    N1 = 2 * N0
    if time.time() < deadline - 150:
        h_up = ensure_sum_bin(upsample_fn(best_h, len(best_h), N1), N1)
        d3 = min(deadline - 120, time.time() + budget_s * 0.18)

    if c5 < best_c5:
        best_c5, best_h = c5, h5.copy()

    return best_h, final_c5, len(best_h)
