def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft(np.flip(1.0 - h.astype(float)), L)
    return np.fft.irfft(H * G, L)[:2*N-1]

def compute_c5(h, N):

    return float(np.max(corr)) * (2.0 / N)

def project_box_sum(h, target):

def continuous_refine(h_init, N, deadline, lr0=0.05, subdecay=0.999):
    """Subgradient method on continuous relaxation"""
    h = h_init.astype(float).copy()
    h = project_box_sum(h, N/2.0)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    lr = lr0
    momentum = np.zeros(N)
    beta = 0.85
    
    while time.time() < deadline:
        y = h + momentum
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        
        # Subgradient: use top-k lags with softmax weighting
        k_top = min(30, len(corr))
        top = np.argpartition(corr, -k_top)[-k_top:]
        temp_w = 0.02
        w = np.exp((corr[top] - corr[top].max()) / temp_w)
        w /= w.sum()
        
        grad = np.zeros(N)
        center = N - 1
        for idx, wt in zip(top, w):
            L = int(idx - center)
            if L == 0:
                grad += wt * (1.0 - 2.0 * y)
            elif L > 0 and L < N:
                grad[L:] += wt * (1.0 - y[:N-L])
                grad[:N-L] -= wt * y[L:]
            elif -L < N:
                M = -L
                grad[:N-M] += wt * (1.0 - y[M:])
                grad[M:] -= wt * y[:N-M]
        
        h_new = project_box_sum(y - lr * grad, N/2.0)
        momentum = beta * (h_new - h)
        h = h_new
        lr *= subdecay
    
    return best_h, best_pk

import numpy as np
import time

def compute_c5(h, N):
    dh = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dh)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def delta_vec_positions(h, positions, flip_type, N):
    """Compute delta correlation vectors for flipping positions.
    flip_type=1: 1->0, flip_type=0: 0->1
    Returns array of shape (K, 2N-1)
    """
    K = len(positions)
    idx = np.arange(2 * N - 1)  # 0..2N-2
    left_idx = positions[:, None] + (N - 1) - idx[None, :]
    right_idx = positions[:, None] + idx[None, :] - (N - 1)
    vl = (left_idx >= 0) & (left_idx < N)
    vr = (right_idx >= 0) & (right_idx < N)
    h_left = np.where(vl, h[np.clip(left_idx, 0, N - 1)], 0.0)
    h_right = np.where(vr, h[np.clip(right_idx, 0, N - 1)], 0.0)
    if flip_type == 1:  # 1->0
        return h_left - (1.0 - h_right) * vr
    else:  # 0->1
        return (1.0 - h_right) * vr - h_left

def local_search(h, N, deadline, top_k=80, temp=0.0, kick_thresh=60, kick_sz=4, max_iters=100000):
    h = h.astype(float).copy()
    target = N // 2
    while h.sum() > target:
        h[np.random.choice(np.where(h == 1)[0])] = 0
    while h.sum() < target:
        h[np.random.choice(np.where(h == 0)[0])] = 1
    best_c = compute_c5(h, N)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 5 or len(zeros) < 5:
            break
        ns = min(len(ones), top_k)
        nz = min(len(zeros), top_k)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        d1 = delta_vec_positions(h, os_, 1, N)  # (ns, 2N-1)
        d0 = delta_vec_positions(h, zs, 0, N)  # (nz, 2N-1)
        new_c = corr[None, None, :] + d1[:, None, :] + d0[None, :, :]
        new_max = new_c.max(axis=2)  # (ns, nz)
        curr_max = corr.max()
        mi = np.unravel_index(new_max.argmin(), new_max.shape)
        mv = new_max[mi]
        accept = False
        if mv < curr_max - 1e-12:
            accept = True
        elif temp > 0 and np.random.random() < temp * np.exp(-stuck * 0.008):
            accept = True
        if accept:
            h[os_[mi[0]]] = 0
            h[zs[mi[1]]] = 1
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
            c = float(corr.max()) * (2.0 / N)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1

                h[q] = 1
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
            stuck = 0
        else:

def subgradient_refine(h_init, N, deadline, lr0=0.06, decay=0.9995):
    h = h_init.astype(float).copy()
    h = project_box_sum(h, N / 2.0)
    best_h = h.copy()
    best_pk = float(np.max(np.correlate(h, 1 - h, mode="full")))
    lr = lr0
    mom = np.zeros(N)
    beta = 0.85
    while time.time() < deadline:
        y = h + mom
        corr = np.correlate(y, 1 - y, mode="full").astype(float)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        k_top = min(25, len(corr))
        top = np.argpartition(corr, -k_top)[-k_top:]
        w = np.exp((corr[top] - corr[top].max()) / 0.015)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for idx_c, wt in zip(top, w):
            d = int(idx_c - center)
            if d == 0:
                grad += wt * (1.0 - 2.0 * y)
            elif d > 0 and d < N:
                grad[d:] += wt * (1.0 - y[:N - d])
                grad[:N - d] -= wt * y[d:]
            else:
                m = -d
                if m < N:
                    grad[:N - m] += wt * (1.0 - y[m:])
                    grad[m:] -= wt * y[:N - m]
        h_new = project_box_sum(y - lr * grad, N / 2.0)
        mom = beta * (h_new - h)
        h = h_new
        lr *= decay
    return best_h, best_pk * (2.0 / N)

def block_swap_search(h, N, deadline, block_size=4, max_iters=5000, top_k=30):

        p1 = np.random.randint(0, N - block_size + 1)
        # Find a block with same number of 1s
        c1 = h[p1:p1 + block_size].sum()
        found = False
        for attempt in range(50):
            p2 = np.random.randint(0, N - block_size + 1)
            if p2 < p1 + block_size and p2 + block_size > p1:
                continue
            c2 = h[p2:p2 + block_size].sum()
            if c2 == target - c1:  # After swap: block1 has target-c1 ones, block2 has c1 ones
                # Actually we need: after swap, total ones unchanged
                # Swap: h[p1:p1+bs] gets h[p2:p2+bs] and vice versa
                # Total ones: same. Good, any two blocks work for sum preservation.
                found = True
                break
        if not found:
            # Just do a random two-block swap (sum preserving)
            p2 = np.random.randint(0, N - block_size + 1)
            if abs(p2 - p1) < block_size:
                p2 = (p2 + block_size + 1) % max(N - block_size, 1)
        temp_a = h[p1:p1 + block_size].copy()
        h[p1:p1 + block_size] = h[p2:p2 + block_size]
        h[p2:p2 + block_size] = temp_a
        c = compute_c5(h, N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stuck = 0
        else:
            # Revert

            h[p1:p1 + block_size] = temp
            stuck += 1
            if stuck > 200:
                stuck = 0
    return best_h, best_c
