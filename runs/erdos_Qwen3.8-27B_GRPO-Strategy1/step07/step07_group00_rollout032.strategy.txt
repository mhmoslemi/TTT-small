```python
def repair(h, target):
    d = sum(h) - target
    if abs(d) < 1e-3:
        return h
    flip_indices = np.argsort(np.abs(np.random.randn(len(h)))[:abs(d)])
    if d > 0:  # too many 1's
        h[flip_indices[:d]] = False
    else:    # too few 1's
        h[flip_indices[:abs(d)]] = True
    return h
```
