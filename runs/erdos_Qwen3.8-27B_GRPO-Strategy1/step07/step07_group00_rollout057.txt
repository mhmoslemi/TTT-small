def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        if len(idx) < n1 - target:
            idx = np.argsort(-h)[:n1 - target]
        h[np.random.choice(idx, min(n1 - target, len(idx)), replace=False)] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        if len(idx) < target - n1:
            idx = np.argsort(h)[:target - n1]
        h[np.random.choice(idx, min(target - n1, len(idx)), replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def project_constrained(h, target_sum):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def targeted_sa(h_bin, deadline, n_peaks=8, n_per_peak=20, cool_rate=0.999, t_start=0.05, max_iters=100000):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 300 == 0:
            T = max(T * cool_rate, 0.001)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        mx = corr.max()
        active = np.where(corr > 0.88 * mx)[0]
        if len(active) > n_peaks * 2:
            active = np.argpartition(corr, -n_peaks * 2)[-n_peaks * 2:]
        cand_p, cand_q = [], []
        for pk in active:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp, bq = vi[mask], (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = min(6, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp, dq = deltas[cand_p], deltas[cand_q]
        chunk = 10
        best_new, bi, bj = 1e10, 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[None, None, :] + dp[ci:ce, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi]); bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0; deltas[p, p:p+N] = s
            deltas[q, :] = 0.0; deltas[q, q:q+N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c; best_h = h.copy(); no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        if no_impr > 5000:
            break
    return best_h, best_c

def steepest_binary(h_bin, deadline, max_esc=5, esc_n=3):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    esc_left = max_esc
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        dp, dq = deltas[ones], deltas[zeros]
        best_new, best_pi, best_qj = best_c + 1e-9, -1, -1
        chunk = 20
        for ci in range(0, len(ones), chunk):
            ce = min(ci + chunk, len(ones))
            vals = corr[None, None, :] + dp[ci:ce, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi]); best_pi, best_qj = mi[0] + ci, mi[1]
        if best_pi < 0:
            if esc_left > 0:
                esc_left -= 1
                for _ in range(esc_n):
                    p = int(np.random.choice(ones)); q = int(np.random.choice(zeros))
                    h[p], h[q] = 0.0, 1.0
                    corr += deltas[p].copy() - deltas[q].copy()
                    s[p], s[q] = -s[p], -s[q]
                    deltas[p, :] = 0.0; deltas[p, p:p+N] = s
                    deltas[q, :] = 0.0; deltas[q, q:q+N] = s
                ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
            else:
                break
        p = int(ones[best_pi]); q = int(zeros[best_qj])
        h[p], h[q] = 0.0, 1.0
        corr += deltas[p].copy() - deltas[q].copy()
        s[p], s[q] = -s[p], -s[q]
        deltas[p, :] = 0.0; deltas[p, p:p+N] = s
        deltas[q, :] = 0.0; deltas[q, q:q+N] = s
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c; best_h = h.copy()
    return best_h, best_c

def grad_descent_cont(h_in, N, deadline, lr=0.003, k_top=30):
    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_constrained(h, target)
    m = np.zeros(N)
    beta = 0.8
    best_h = h.copy()
    best_c = float(fft_corr(h).max())
    it = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        top = np.argpartition(corr_y, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t) - (N - 1)
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L]); grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:]); grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_constrained(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 30 == 0:
            c = float(fft_corr(h).max())
            if c < best_c:
                best_c = c; best_h = h.copy()
            lr *= 0.9995
            beta = min(0.94, 0.80 + it * 0.0001)
    c = float(fft_corr(h).max())
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    # P1: SA
    d1 = min(deadline - 500, t0 + 200)
    h1, _ = targeted_sa(h_bin, d1, n_peaks=8, n_per_peak=20, cool_rate=0.999, t_start=0.05)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5: best_c5, best_h = c1, h1.copy()

    # P2: Steepest descent
    if time.time() < deadline - 400:
        d2 = min(deadline - 350, time.time() + 120)
        h2, _ = steepest_binary(best_h, d2, max_esc=5, esc_n=3)
        c2 = compute_c5(h2, N0)
        if c2 < best_c5: best_c5, best_h = c2, h2.copy()

    # P3: Continuous gradient descent
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 180)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = np.clip(hc, 0, 1)
        hc = project_constrained(hc, N0 / 2.0)
        hn, c3 = grad_descent_cont(hc, N0, d3, lr=0.003, k_top=30)
        if c3 < best_c5: best_c5, best_h = c3, hn.copy()

    # P4: SA from rounded continuous
    if time.time() < deadline - 130:
        h4 = make_binary(best_h, N0)
        d4 = min(deadline - 100, time.time() + 100)
        h4, _ = targeted_sa(h4, d4, n_peaks=10, n_per_peak=25, cool_rate=0.9993, t_start=0.03)
        c4 = compute_c5(h4, N0)
        if c4 < best_c5: best_c5, best_h = c4, h4.copy()

    # P5: Steepest descent
    if time.time() < deadline - 60:
        d5 = min(deadline - 30, time.time() + 50)
        h5, _ = steepest_binary(best_h, d5, max_esc=3, esc_n=2)
        c5 = compute_c5(h5, N0)
        if c5 < best_c5: best_c5, best_h = c5, h5.copy()

    # P6: Multi-resolution
    for Nt in [400, 440, 500]:
        if time.time() > deadline - 30: break
        Nt -= Nt % 2
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, Nt)
        dl = min(deadline - 15, time.time() + 30)
        hr, _ = targeted_sa(hb, dl, n_peaks=6, n_per_peak=15, cool_rate=0.999, t_start=0.04)
        cr = compute_c5(hr, Nt)
        if cr < best_c5: best_c5, best_h = cr, hr.copy()

    # P7: Final continuous GD
    if time.time() < deadline - 3:
        Ny = len(best_h)
        hy, cy = grad_descent_cont(best_h, Ny, deadline - 2, lr=0.002, k_top=20)
        if cy < best_c5: best_c5, best_h = cy, hy

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
