```python
    # Project h to {x: sum(x)=target, 0<=x<=1}
    for _ in range(50):
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # Distribute diff to non-saturated variables
        if diff > 0:
            mask = h < 1 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return h
```

```python
import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_at_peak(h, k, N):
    # k is the shift (0-indexed from center), i.e., correlation index N-1+k
    # R[N-1+k] = sum_i h[i+k]*(1-h[i]) for valid i
    grad = np.zeros(N)
    if k >= 0:
        # h[i+k] appears for i in [0, N-1-k]
        # (1-h[i]) appears for i in [0, N-1-k]
        for i in range(N - k):
            grad[i + k] += (1.0 - h[i])  # from h[i+k] term
            grad[i] -= h[i + k]           # from (1-h[i]) term
    else:
        kk = -k
        for i in range(N - kk):
            grad[i] += (1.0 - h[i + kk])
            grad[i + kk] -= h[i]
```

```python
def project_sum(h, target):
    h = h.copy()

        h = np.clip(h, 0.0, 1.0)
```
```python
        d = target - h.sum()
        if abs(d) < 1e-14:
```

```python
def softmax_gradient(h, N, top_m, tau):
    corr = compute_corr(h)
    center = N - 1
    top_idx = np.argsort(corr)[-top_m:]
    vals = corr[top_idx]
    w = np.exp((vals - vals.max()) / tau)
    w /= w.sum()
    grad = np.zeros(N)

        best_c, best_h = c1, h1.copy()

    # Phase 2: Binary SA
    d2 = min(deadline - 300, time.time() + 220)
    hb, cb = sa_binary(best_h, N0, d2, T0=0.06, T1=1e-7)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 3: Targeted binary search
    d3 = min(deadline - 200, time.time() + 180)
    ht, ct = targeted_sa(best_h, N0, d3, max_iters=150000)
    if ct < best_c:
        best_c, best_h = ct, ht.copy()

    # Phase 4: Fine gradient from best
    h4 = best_h.astype(float) + np.random.randn(len(best_h)) * 0.04
    h4 = project_sum(np.clip(h4, 0, 1), len(best_h) / 2.0)
    d4 = min(deadline - 100, time.time() + 160)
    h4, c4 = grad_descent(h4, N0, d4, lr0=0.3, top_m=5, tau0=3.0)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: Try other N
    for Nt in [360, 400, 420, 480, 500, 540]:
        if time.time() > deadline - 50:
            break
        hn = interp_periodic(best_h, Nt)
        dn = min(deadline - 30, time.time() + 45)
        hn, cn = grad_descent(hn, Nt, dn, lr0=0.5, top_m=8, tau0=7.0)
        if cn < best_c:
            best_c, best_h = cn, hn.copy()

    # Phase 6: Final polish
    hF = project_sum(best_h, len(best_h) / 2.0)
    dF = deadline - 0.5
    hF, cF = grad_descent(hF, len(hF), dF, lr0=0.15, top_m=4, tau0=2.0)
    if cF < best_c:
        best_c, best_h = cF, hF

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
