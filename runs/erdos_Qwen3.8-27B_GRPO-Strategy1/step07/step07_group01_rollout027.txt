import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(77)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def sa_binary(h, N, deadline, rng, T0=2.0, T_end=0.05, multi_frac=0.08):
    h = h.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    P = best_c
    it = 0
    while time.time() < deadline:
        T = T0 * (T_end / T0) ** (min(it, 2000000) / 2000000.0)
        if it % 5000 == 0:
            ones_list = np.where(h == 1)[0]
            zeros_list = np.where(h == 0)[0]
        if rng.random() < multi_frac:
            if len(ones_list) < 2 or len(zeros_list) < 2:
                it += 1
                continue
            i1 = rng.randint(len(ones_list))
            i2 = rng.randint(len(ones_list))
            if i1 == i2:
                it += 1
                continue
            j1 = rng.randint(len(zeros_list))
            j2 = rng.randint(len(zeros_list))
            if j1 == j2:
                it += 1
                continue
            p1, p2 = ones_list[i1], ones_list[i2]
            q1, q2 = zeros_list[j1], zeros_list[j2]
            h[p1] = 0.0; h[p2] = 0.0; h[q1] = 1.0; h[q2] = 1.0
            new_corr = fft_corr(h)
            new_P = float(new_corr.max())
            if new_P < P or (T > 1e-9 and rng.random() < np.exp(-(new_P - P) / T)):
                P = new_P; corr = new_corr
                if P < best_c - 1e-10:
                    best_c = P; best_h = h.copy()
            else:
                h[p1] = 1.0; h[p2] = 1.0; h[q1] = 0.0; h[q2] = 0.0
        else:
            p = ones_list[rng.randint(len(ones_list))]
            q = zeros_list[rng.randint(len(zeros_list))]
            h[p] = 0.0; h[q] = 1.0
            new_corr = fft_corr(h)
            new_P = float(new_corr.max())
            if new_P < P or (T > 1e-9 and rng.random() < np.exp(-(new_P - P) / T)):
                P = new_P; corr = new_corr
                if P < best_c - 1e-10:
                    best_c = P; best_h = h.copy()
            else:
                h[p] = 1.0; h[q] = 0.0
        it += 1
    return best_h, best_c * (2.0 / N)

def greedy_local_search(h, N, deadline, sample_frac=1.0):
    h = ensure_binary(h, N)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    P = best_c
    Nsz = N
    shifts = np.arange(2 * Nsz - 1) - (Nsz - 1)
    p_all = np.arange(Nsz)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < Nsz)
    vp = (pp >= 0) & (pp < Nsz)
    pm_c = np.clip(pm, 0, Nsz - 1)
    pp_c = np.clip(pp, 0, Nsz - 1)
    while time.time() < deadline:
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        n_check = len(ones) if sample_frac >= 1.0 else max(30, int(len(ones) * sample_frac))
        np.random.shuffle(ones)
        ones_check = ones[:n_check]
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        for p in ones_check:
            base = corr + D[p, :]
            npk = (base[None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-12:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        pv, qv = best_pq
        h[pv] = 0.0; h[qv] = 1.0
        corr += D[pv, :] - D[qv, :]
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P; best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def perturb_binary(h, N, n_swaps, rng):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_swaps):
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        h[p] = 0.0; h[q] = 1.0
    return h

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = ensure_binary(h0, N0)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: SA at N0
    d1 = min(deadline - 350, t0 + 450)
    h1, c5_1 = sa_binary(best_h, N0, d1, rng, T0=2.5, T_end=0.05, multi_frac=0.08)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Greedy polish
    d2 = min(deadline - 250, time.time() + 100)
    h2, c5_2 = greedy_local_search(best_h, N0, d2, sample_frac=1.0)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Perturb + SA (restart from perturbed best)
    h_p = perturb_binary(best_h, N0, 8, rng)
    d3 = min(deadline - 180, time.time() + 250)
    h3, c5_3 = sa_binary(h_p, N0, d3, rng, T0=1.5, T_end=0.03, multi_frac=0.1)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Greedy polish
    d4 = min(deadline - 120, time.time() + 80)
    h4, c5_4 = greedy_local_search(best_h, N0, d4, sample_frac=1.0)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Another perturb + shorter SA
    h_p2 = perturb_binary(best_h, N0, 5, rng)
    d5 = min(deadline - 80, time.time() + 150)
    h5, c5_5 = sa_binary(h_p2, N0, d5, rng, T0=1.0, T_end=0.03, multi_frac=0.06)

        best_c5, best_h = c5_5, h5

    d6 = min(deadline - 30, time.time() + 60)

    if c5_6 < best_c5:

    # Phase 7: Try N=1000 downsample
    N_alt = 1000
    if time.time() < deadline - 60 and N_alt != N0:

        h_down = ensure_binary(h_down, N_alt)
        c5_down = compute_c5(h_down, N_alt)

        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7
        elif c5_down < best_c5:
            best_c5, best_h = c5_down, h_down

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
