import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)

    lo, hi = -3.0, 3.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def cG(h, N):
    M = 2 * N - 1
    G = np.zeros((N, M))
    for m in range(M):
        lag = m - (N - 1)
        if lag == 0:
            G[:, m] = 1.0 - 2.0 * h
        elif lag > 0:
            G[:N - lag, m] = 1.0 - h[lag:]
            G[lag:, m] -= h[:N - lag]
        else:
            L2 = -lag
            G[L2:, m] = 1.0 - h[:N - L2]
            G[:N - L2, m] -= h[L2:]
    return G
