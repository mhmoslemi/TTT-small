def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.sum(h >= 0.5))
    if n1 > target:
        # flip some 1s to 0s
        ones = np.where(h >= 0.5)[0]
        np.random.shuffle(ones)
        h[ones[:n1 - target]] = 0.0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - n1]] = 1.0
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    assert np.sum(h) == target
    return h

def gradient_of_corr_max(h, N, k_top=50):
    """Compute subgradient of max_k corr_k w.r.t. h."""
    corr = fft_corr(h)
    # Focus on top k_top correlations
    top = np.argpartition(corr, -k_top)[-k_top:]
    grad = np.zeros(N)
    for t in top:
        L = int(t - (N - 1))  # actual lag
        if L == 0:
            grad += 1.0 - 2.0 * h
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            grad[:N - M] += (1.0 - h[M:])
            grad[M:] -= h[:N - M]
    grad /= len(top)
    return grad, corr, float(corr.max())

    h_new = h - lr * grad
    # Project onto [0,1]
