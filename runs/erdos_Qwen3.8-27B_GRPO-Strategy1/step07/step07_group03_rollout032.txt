```python
def lse_smooth_gradient(h, beta):
    N = len(h)
    corr = fft_corr(h)  # length 2N-1
    m = corr.max()
    exp_bc = np.exp(beta * (corr - m))
    w = exp_bc / exp_bc.sum()  # length 2N-1
    
    # For each j, compute:
    # T1[j] = sum over s: w[s+N-1] * (1-h[j+s]) for 0<=j+s<N
    # T2[j] = sum over s: w[s+N-1] * h[j-s] for 0<=j-s<N
    
    # Vectorized: create index matrices
    j = np.arange(N)[:, None]  # N x 1
    s = np.arange(-(N-1), N) [None, :]  # 1 x (2N-1)
    # Valid ranges:
    # For T1: 0 <= j+s < N, i.e., -j <= s <= N-1-j
    # For T2: 0 <= j-s < N, i.e., j-N+1 <= s <= j
    
    valid1 = (j + s >= 0) & (j + s < N)
    valid2 = (j - s >= 0) & (j - s < N)
    
    # For T1: (1-h[j+s]) where valid
    j_plus_s = np.clip(j + s, 0, N-1)
    T1 = np.where(valid1, w[None, :] * (1.0 - h[j_plus_s]), 0.0).sum(axis=1)
    
    # For T2: h[j-s] where valid
    j_minus_s = np.clip(j - s, 0, N-1)
    T2 = np.where(valid2, w[None, :] * h[j_minus_s], 0.0).sum(axis=1)
    
    grad = (T1 - T2) / beta
    return grad
```

```python
def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
```

```python
    s = np.arange(-(N - 1), N)[None, :]
    v1 = (j + s >= 0) & (j + s < N)
    p1 = np.clip(j + s, 0, N - 1)
    T1 = np.sum(np.where(v1, w[None, :] * (1.0 - h[p1]), 0.0), axis=1)
    v2 = (j - s >= 0) & (j - s < N)
    p2 = np.clip(j - s, 0, N - 1)
    T2 = np.sum(np.where(v2, w[None, :] * h[p2], 0.0), axis=1)
    return T1 - T2

def smooth_opt(h, N, deadline, beta_lo=5.0, beta_hi=60.0, n_iter=400, lr=0.4, mom=0.82):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best = h.copy()
    best_v = float(fft_corr(h).max())
    m = np.zeros(N)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter - 1, 1)
        b = beta_lo * (beta_hi / beta_lo) ** frac
        pk = float(fft_corr(h).max())
        if pk < best_v - 1e-12:
            best_v = pk
            best = h.copy()
        g = lse_grad(h, b)
        lr_t = lr / (1.0 + 0.008 * it)
        m = mom * m - lr_t * g
        h = project_sum(h + m, target)
    return best, best_v * dx

def ensure_sum(h, N):
    nh = N // 2
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    if n1 == nh:
        return h_b
    rng = np.random.RandomState(999 + N)
    if n1 > nh:
        idx = np.where(h_b == 1)[0]
        h_b[rng.choice(idx, n1 - nh, replace=False)] = 0
    else:
        idx = np.where(h_b == 0)[0]
        h_b[rng.choice(idx, nh - n1, replace=False)] = 1
    return h_b

def compute_D(h, N):
    j = np.arange(N)[:, None]
    s = np.arange(-(N - 1), N)[None, :]
    v1 = (j + s >= 0) & (j + s < N)
    p1 = np.clip(j + s, 0, N - 1)
    T1 = np.sum(np.where(v1, 1.0 - h[p1], 0.0), axis=1)
    v2 = (j - s >= 0) & (j - s < N)
    p2 = np.clip(j - s, 0, N - 1)
    T2 = np.sum(np.where(v2, h[p2], 0.0), axis=1)
    return T1 - T2

def binary_anneal(h, N, deadline, max_sweeps=500, two_swap=0.35, rng=None):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    if rng is None:
        rng = np.random.RandomState(7)
    best_c = float(fft_corr(h).max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        D = compute_D(h, N)
        best_new_P = P_cur
        best_pq = None
        best_2p = None
        n_try = min(300, max(80, N))
        for _ in range(n_try):
            i1 = rng.randint(len(ones))
            j1 = rng.randint(len(zeros))
            p, q = int(ones[i1]), int(zeros[j1])
            new_P = float((c_cur - D[p] + D[q]).max())
            if new_P < best_new_P - 1e-14:
                best_new_P = new_P
                best_pq = (p, q)
        if len(ones) >= 4 and len(zeros) >= 4 and rng.random() < two_swap:
            n_try2 = min(150, N // 2)
            for _ in range(n_try2):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur - D[p1] - D[p2] + D[q1] + D[q2]).max())
                if new_P < best_new_P - 1e-14:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = (p1, p2, q1, q2)
        if best_2p is not None:
            h[best_2p[0]] = h[best_2p[1]] = 0.0
            h[best_2p[2]] = h[best_2p[3]] = 1.0
        elif best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            stall += 1
            if stall > 15:
                n_esc = min(4, len(ones), len(zeros))
                for _ in range(n_esc):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 5:
                continue
            continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-13:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def downsample(h_old, N_new):
    return upsample(h_old, N_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary anneal at N0
    h_bin = ensure_sum(h0, N0)
    d1 = t0 + budget_s * 0.22
    h1, c51 = binary_anneal(h_bin, N0, d1, max_sweeps=600, two_swap=0.35, rng=rng)
    if c51 < best_c5:
        best_c5, best_h = c51, h1.copy()

    # Phase 2: LSE smooth GD at N0 (main continuous refinement)
    if time.time() < deadline - 350:
        hc = ensure_sum(best_h, N0) + rng.randn(N0) * 0.08
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 300, t0 + budget_s * 0.45)
        h2, c52 = smooth_opt(hc, N0, d2, beta_lo=4.0, beta_hi=50.0,
                             n_iter=500, lr=0.35, mom=0.80)
        if c52 < best_c5:
            best_c5, best_h = c52, h2.copy()

    # Phase 3: Upsample to 2*N0, LSE smooth GD
    N1 = 2 * N0
    if time.time() < deadline - 250:
        h_up = project_sum(upsample(ensure_sum(best_h, N0), N1), N1 / 2.0)
        d3 = min(deadline - 200, t0 + budget_s * 0.65)
        h3, c53 = smooth_opt(h_up, N1, d3, beta_lo=4.0, beta_hi=40.0,
                             n_iter=250, lr=0.3, mom=0.78)
        if c53 < best_c5:
            best_c5, best_h = c53, h3.copy()

    # Phase 4: Downsample to N0, second LSE round
    h_ds = project_sum(downsample(best_h, N0), N0 / 2.0)
    if time.time() < deadline - 150:
        hc4 = h_ds + rng.randn(N0) * 0.04
        hc4 = project_sum(hc4, N0 / 2.0)
        d4 = min(deadline - 100, t0 + budget_s * 0.80)
        h4, c54 = smooth_opt(hc4, N0, d4, beta_lo=15.0, beta_hi=150.0,
                             n_iter=300, lr=0.2, mom=0.85)
        if c54 < best_c5:
            best_c5, best_h = c54, h4.copy()

    # Phase 5: Binary anneal from best continuous
    N_fin = len(best_h)
    if N_fin != N0:
        best_h = downsample(best_h, N0)
        N_fin = N0
    h_b5 = ensure_sum(best_h, N_fin)
    if time.time() < deadline - 50:
        d5 = min(deadline - 30, t0 + budget_s * 0.92)
        h5, c55 = binary_anneal(h_b5, N_fin, d5, max_sweeps=400, two_swap=0.35, rng=rng)
        if c55 < best_c5:
            best_c5, best_h = c55, h5.copy()

    # Phase 6: Short LSE polish on the binary result
    if time.time() < deadline - 10:
        hc6 = ensure_sum(best_h, len(best_h)) + rng.randn(len(best_h)) * 0.03
        hc6 = project_sum(hc6, len(best_h) / 2.0)
        d6 = deadline - 2
        h6, c56 = smooth_opt(hc6, len(best_h), d6, beta_lo=30.0, beta_hi=200.0,
                             n_iter=100, lr=0.15, mom=0.85)
        if c56 < best_c5:
            best_c5, best_h = c56, h6.copy()

    # Final binary snap and verify
    h_final = ensure_sum(best_h, len(best_h))
    c5_final = compute_c5(h_final, len(h_final))
    c5_best = compute_c5(best_h, len(best_h))
    if c5_final < c5_best:
        best_h, best_c5 = h_final, c5_final
    else:
        # Try the continuous version too
        best_c5 = min(best_c5, c5_best)

    return best_h, best_c5, len(best_h)
```
