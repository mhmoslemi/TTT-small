def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

Actually wait - the problem says "Find a step function h: [0, 2] → [0, 1]". A step function with values in [0,1] doesn't have to be binary! It can take any value in [0,1] at each step. So continuous values are allowed!

But the evaluation uses np.correlate(h, 1-h, mode="full") * dx, and the constraint is sum(h)*dx = 1.

So we're looking for a vector h in [0,1]^N with sum(h) = N/2 that minimizes max(corr(h, 1-h) * dx).

This is a continuous optimization problem! The binary restriction is just a heuristic.

Let me focus on continuous optimization with better methods:
1. L-BFGS-B (scipy) with the smoothed max objective
2. Projected gradient with momentum
3. Coordinate descent

Actually, the objective max_k corr[k] is not smooth. We need to smooth it. The soft-max (log-sum-exp) is a good smooth approximation:
f_T(h) = (1/T) * log(sum_k exp(T * corr[k]))

As T→∞, this approaches max_k corr[k]. The gradient is:
∇f_T = sum_k softmax(T*corr) * ∇corr[k]

This is what the previous code does. Let me improve the optimization:

1. Use Adam optimizer
2. Start from multiple initial points
3. Use a smarter T schedule (start low, increase)
4. Periodically project to the constraint manifold

Let me also try a different approach: since this is a max-min type problem, we can use a bisection approach:
- Binary search on C: is there h with max_k corr[k] <= C?
- This is a solvability question that might be easier

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def corr_grad(h, N, shifts, weights):
    """Compute weighted sum of correlation gradients."""
    grad = np.zeros(N)
    center = N - 1
    for s, w in zip(shifts, weights):
        s = int(s)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += w * (1.0 - h[s:])
            grad[s:] += -w * h[:N-s]
        else:
            M = -s
            grad[M:] += w * (1.0 - h[:N-M])
            grad[:N-M] += -w * h[M:]
    return grad

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def adam_smoothed(h0, N, deadline, k=50, T_init=50.0, T_max=500.0, lr=0.01):
    """Adam optimizer for smoothed max correlation."""
    h = h0.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h))) * dx
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    T = T_init
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        # Softmax weights on top-k
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        
        shifts = idx - (N - 1)
        weights = w / k
        grad = corr_grad(h, N, shifts, weights)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = project_box_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        
        val = float(np.max(fft_corr(h))) * dx
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        # Adapt T
        if t % 50 == 0:
            T = min(T_max, T * 1.08)
            lr *= 0.999
    return best_h, best_val
