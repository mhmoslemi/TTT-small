import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_to_binary(h, n_ones):
    # Project to binary with exactly n_ones ones
    threshold = np.sort(h)[::-1][n_ones - 1] if n_ones > 0 else 1.0
    # Handle ties
    above = h >= threshold
    n_above = above.sum()
    if n_above > n_ones:
        # Randomly drop some
        above_idx = np.where(above)[0]
        to_remove = np.random.choice(above_idx, n_above - n_ones, replace=False)
        above[to_remove] = False
    elif n_above < n_ones:
        below_idx = np.where(~above)[0]
        to_add = np.random.choice(below_idx, n_ones - n_above, replace=False)
        above[to_add] = True
    return above.astype(float)

def nesterov_rnd(h, N, deadline, k_top=50, lr=0.01):
    h = h.astype(float).copy()
    t = np.zeros(N)
    best_h = h.copy(); best_c = float(np.max(fft_corr(h)))
    beta = 0.9
    while time.time() < deadline:
        y = h + t
        corr = fft_corr(y)
        c = float(np.max(corr))
        if c < best_c: best_c = c; best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        g = np.zeros(N)
        ctr = N-1
        for ai in top:
            L = int(ai - ctr)
            if L == 0: g += (1-2*y)
            elif L > 0:
                g[L:] += (1-y[:N-L]); g[:N-L] -= y[L:]
            else:
                M = -L; g[:N-M] += (1-y[M:]); g[M:] -= y[:N-M]
        g /= k_top
        h_new = np.clip(y - lr*g, 0, 1)
        h_new *= N/2.0 / max(1e-8, h_new.sum())

        # Project to sum = N/2

            h_new = np.clip(h_new, 0, 1)
        t = beta*(h_new - h)
        h = h_new
        lr *= 0.9995
        if lr < 0.001: lr = 0.001
    return best_h, best_c * 2.0/N
