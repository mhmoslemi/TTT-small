def subgrad_polish(h_init, N, deadline, lr=0.015, k_top=30, beta_start=0.8,

    best_h = h.copy()

    for restart in range(n_restart):
        if time.time() > deadline - 30:
            break
        m = np.zeros(N)
        beta = beta_start
        lr_r = lr * (0.8 ** restart)
        for it in range(5000):
            if time.time() > deadline - 30:
                break
            y = h + m
            # Clip and normalize

            y = np.clip(y - (y.sum() - N/2.0) / N, 0, 1)

                if L == 0:
                    grad += 1.0 - 2.0 * y
                elif L > 0:
                    grad[L:] += (1.0 - y[:N-L])
                    grad[:N-L] -= y[L:]
                else:

                    grad[:N-M] += (1.0 - y[M:])

            m = beta * (h_new - h)
            h = h_new
            
            if it % 500 == 0:
                lr_r *= 0.97
                beta = min(beta_end, beta + 0.0002)
    
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(np.flip(1.0 - h), L), L))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def make_binary(h, rng):
    N = len(h)
    h = np.asarray(h, dtype=float).copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    tgt = N // 2
    s = int(h.sum())
    if s > tgt:
        h[rng.choice(np.where(h == 1)[0], s - tgt, replace=False)] = 0
    elif s < tgt:
        h[rng.choice(np.where(h == 0)[0], tgt - s, replace=False)] = 1
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    d = np.zeros((N, 2 * N - 1))
    for i in range(N):
        d[i, i:i+N] = s
    return s, d

def get_cands(corr, h, npk, npp, rng):
    N = len(h)
    top = np.argpartition(corr, -npk)[-npk:]
    c1, c0 = [], []
    for pk in top:
        k = int(pk) - (N - 1)
        if k == 0 or abs(k) >= N:
            continue
        if k > 0:
            vi = np.arange(N - k)
            m = (h[vi] == 1) & (h[vi + k] == 0)
            c1.extend(vi[m].tolist()); c0.extend((vi + k)[m].tolist())
        else:
            M = -k
            vi = np.arange(M, N)
            m = (h[vi] == 1) & (h[vi - M] == 0)
            c1.extend(vi[m].tolist()); c0.extend((vi - M)[m].tolist())
    c1 = np.array(sorted(set(c1)), int) if c1 else np.array([], int)
    c0 = np.array(sorted(set(c0)), int) if c0 else np.array([], int)
    if len(c1) > npp: c1 = rng.choice(c1, npp, replace=False)
    if len(c0) > npp: c0 = rng.choice(c0, npp, replace=False)
    on = np.where(h == 1)[0]; zo = np.where(h == 0)[0]
    nr = min(8, len(on), len(zo))
    if nr > 0:
        c1 = np.concatenate([c1, rng.choice(on, nr, replace=False)])
        c0 = np.concatenate([c0, rng.choice(zo, nr, replace=False)])
    c1 = np.array(sorted(set(c1.tolist())), int)
    c0 = np.array(sorted(set(c0.tolist())), int)
    return c1, c0

def binary_sa(hb, dl, rng, npk=8, npp=25, ts=0.04, cl=0.9992,
              mx=200000, p2=0.35, n2=25):
    N = len(hb)
    h = hb.copy()
    corr = fft_corr(h)
    bc = float(corr.max()); bh = h.copy()
    s, dl_t = build_deltas(h)
    T = ts; it = 0; ni = 0
    while time.time() < dl and it < mx:
        it += 1
        if it % 400 == 0: T = max(T * cl, 0.001)
        c1, c0 = get_cands(corr, h, npk, npp, rng)
        if len(c1) < 1 or len(c0) < 1: break
        d1, d0 = dl_t[c1], dl_t[c0]
        bnm = 1e10; bmv = None
        if len(c1) >= 2 and len(c0) >= 2 and rng.random() < p2:
            n1, n0 = len(c1), len(c0)
            for _ in range(n2):
                i1, i2 = rng.choice(n1, 2, replace=False)
                j1, j2 = rng.choice(n0, 2, replace=False)
                m = float(np.max(corr + d1[i1] + d1[i2] - d0[j1] - d0[j2]))
                if m < bnm: bnm, bmv = m, (2, i1, i2, j1, j2)
        else:
            ch = 5
            for ci in range(0, len(c1), ch):
                ce = min(ci + ch, len(c1))
                vals = corr[None,None,:] + d1[ci:ce,None,:] - d0[None,:,:]
                nm = vals.max(2)
                mi = np.unravel_index(int(nm.argmin()), nm.shape)
                if nm[mi] < bnm: bnm, bmv = float(nm[mi]), (1, mi[0]+ci, mi[1], -1, -1)
        if bmv is None: ni += 1; continue
        tp, a, b, c, dd = bmv
        dc = bnm - bc
        if dc < -1e-12 or (T > 0.001 and rng.random() < np.exp(-dc/(T+1e-9))):
            if tp == 1:
                p, q = int(c1[a]), int(c0[b])
                h[p], h[q] = 0.0, 1.0
                s[p], s[q] = -s[p], -s[q]
                corr += d1[a] - d0[b]
                dl_t[p,:]=0; dl_t[p,p:p+N]=s
                dl_t[q,:]=0; dl_t[q,q:q+N]=s
            else:
                p1,p2 = int(c1[a]),int(c1[b]); q1,q2 = int(c0[c]),int(c0[dd])
                h[p1],h[p2],h[q1],h[q2] = 0,0,1,1
                s[p1],s[p2],s[q1],s[q2] = -s[p1],-s[p2],-s[q1],-s[q2]
                corr = fft_corr(h)
                for idx in [p1,p2,q1,q2]:
                    dl_t[idx,:]=0; dl_t[idx,idx:idx+N]=s
        c = float(corr.max())
        if c < bc - 1e-12: bc, bh, ni = c, h.copy(), 0
        else: ni += 1
        if ni > 5000: break
    return bh, bc

def greedy_climb(hb, dl, rng, npk=10, npp=30, mx=300):
    N = len(hb)
    h = hb.copy(); corr = fft_corr(h)
    bc = float(corr.max()); bh = h.copy()
    s, dl_t = build_deltas(h)
    for _ in range(mx):
        if time.time() > dl: break
        c1, c0 = get_cands(corr, h, npk, npp, rng)
        if len(c1) == 0 or len(c0) == 0: break
        d1, d0 = dl_t[c1], dl_t[c0]
        ch = 10; bnm = bc + 1e-9; ba, bb = -1, -1
        for ci in range(0, len(c1), ch):
            ce = min(ci+ch, len(c1))
            vals = corr[None,None,:] + d1[ci:ce,None,:] - d0[None,:,:]
            nm = vals.max(2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < bnm: bnm, ba, bb = float(nm[mi]), mi[0]+ci, mi[1]
        if ba < 0: break
        p, q = int(c1[ba]), int(c0[bb])
        h[p], h[q] = 0.0, 1.0
        s[p], s[q] = -s[p], -s[q]
        corr += d1[ba] - d0[bb]
        dl_t[p,:]=0; dl_t[p,p:p+N]=s
        dl_t[q,:]=0; dl_t[q,q:q+N]=s
        bc = float(corr.max()); bh = h.copy()
    return bh, bc

def subgrad(h_init, N, dl, lr=0.012, kt=40, nr=2):
    h = h_init.astype(float).copy(); bt = 0.85
    bc = 1e10; bh = h.copy()
    for r in range(nr):
        if time.time() > dl - 20: break
        m = np.zeros(N); beta = bt; lrr = lr * (0.85 ** r)
        for it in range(4000):
            if time.time() > dl - 20: break
            y = np.clip(h + m, 0, 1)
            y = np.clip(y - (y.sum() - N/2)/N, 0, 1)
            corr = fft_corr(y)
            pk = float(corr.max()) * (2.0/N)
            if pk < bc: bc, bh = pk, h.copy()
            top = np.argpartition(corr, -kt)[-kt:]
            gr = np.zeros(N)
            for t in top:
                L = int(t) - (N-1)
                if L == 0: gr += 1.0 - 2.0*y
                elif L > 0:
                    gr[L:] += (1.0 - y[:N-L]); gr[:N-L] -= y[L:]
                else:
                    M = -L; gr[:N-M] += (1.0 - y[M:]); gr[M:] -= y[:N-M]
            gr /= len(top)
            hn = np.clip(y - lrr*gr, 0, 1)
            hn = np.clip(hn - (hn.sum() - N/2)/N, 0, 1)
            m = beta*(hn - h); h = hn
            if it % 500 == 0: lrr *= 0.97; beta = min(0.94, beta + 0.0002)
    return bh, bc

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(np.flip(1.0 - h), L), L))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def make_binary(h, rng):
    N = len(h)
    h = np.asarray(h, dtype=float).copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    tgt = N // 2
    s = int(h.sum())
    if s > tgt:
        h[rng.choice(np.where(h == 1)[0], s - tgt, replace=False)] = 0
    elif s < tgt:
        h[rng.choice(np.where(h == 0)[0], tgt - s, replace=False)] = 1
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    d = np.zeros((N, 2 * N - 1))
    for i in range(N):
        d[i, i:i + N] = s
    return s, d

def get_cands(corr, h, npk, npp, rng):
    N = len(h)
    top = np.argpartition(corr, -npk)[-npk:]
    c1, c0 = [], []
    for pk in top:
        k = int(pk) - (N - 1)
        if k == 0 or abs(k) >= N:
            continue
        if k > 0:
            vi = np.arange(N - k)
            m = (h[vi] == 1) & (h[vi + k] == 0)
            c1.extend(vi[m].tolist())
            c0.extend((vi + k)[m].tolist())
        else:
            M = -k
            vi = np.arange(M, N)
            m = (h[vi] == 1) & (h[vi - M] == 0)
            c1.extend(vi[m].tolist())
            c0.extend((vi - M)[m].tolist())
    c1 = np.array(sorted(set(c1)), dtype=int) if c1 else np.array([], dtype=int)
    c0 = np.array(sorted(set(c0)), dtype=int) if c0 else np.array([], dtype=int)
    if len(c1) > npp:
        c1 = np.array(rng.choice(c1, npp, replace=False), dtype=int)
    if len(c0) > npp:
        c0 = np.array(rng.choice(c0, npp, replace=False), dtype=int)
    on = np.where(h == 1)[0]
    zo = np.where(h == 0)[0]
    nr = min(8, len(on), len(zo))
    if nr > 0:
        c1 = np.concatenate([c1, rng.choice(on, nr, replace=False)])
        c0 = np.concatenate([c0, rng.choice(zo, nr, replace=False)])
    c1 = np.array(sorted(set(c1.tolist())), dtype=int)
    c0 = np.array(sorted(set(c0.tolist())), dtype=int)
    return c1, c0

    corr = fft_corr(h)
    bc = float(corr.max())
    bh = h.copy()
    s, dl_t = build_deltas(h)
    T = ts
    it = 0
    ni = 0
    while time.time() < dl and it < mx:
        it += 1
        if it % 400 == 0:
            T = max(T * cl, 0.001)
        c1, c0 = get_cands(corr, h, npk, npp, rng)
        if len(c1) < 1 or len(c0) < 1:
            break
        d1, d0 = dl_t[c1], dl_t[c0]
        bnm = 1e10

def subgrad(h_init, N, dl, lr=0.01, kt=40, nr=3):
    h = h_init.astype(float).copy()
    bc = 1e10
    bh = h.copy()
    for r in range(nr):
        if time.time() > dl - 20:
            break
        m = np.zeros(N)
        beta = 0.83
        lrr = lr * (0.82 ** r)
        for it in range(4000):
            if time.time() > dl - 20:
                break
            y = np.clip(h + m, 0, 1)
            y = np.clip(y - (y.sum() - N / 2.0) / N, 0, 1)
            corr = fft_corr(y)
            pk = float(corr.max()) * (2.0 / N)
            if pk < bc:
                bc, bh = pk, h.copy()
            top = np.argpartition(corr, -kt)[-kt:]
            gr = np.zeros(N)
            for t in top:
                L = int(t) - (N - 1)
                if L == 0:
                    gr += 1.0 - 2.0 * y
                elif L > 0:
                    gr[L:] += (1.0 - y[:N - L])
                    gr[:N - L] -= y[L:]
                else:
                    M = -L
                    gr[:N - M] += (1.0 - y[M:])
                    gr[M:] -= y[:N - M]
            gr /= len(top)
            hn = np.clip(y - lrr * gr, 0, 1)
            hn = np.clip(hn - (hn.sum() - N / 2.0) / N, 0, 1)
            m = beta * (hn - h)
            h = hn
            if it % 500 == 0:
                lrr *= 0.97
                beta = min(0.94, beta + 0.0002)
    return bh, bc
