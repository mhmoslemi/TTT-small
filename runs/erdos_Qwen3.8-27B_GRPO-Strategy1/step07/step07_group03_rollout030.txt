import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def binary_anneal(h, N, deadline, max_sweeps=800, two_frac=0.4, n_2try=250, sample_frac=1.0, rng=None):

    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    if rng is None:

    c_cur = fft_corr(h)
    best_c = float(c_cur.max())
    best_h = h.copy()
    temp = 0.014
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.001, temp * 0.9985)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        p_list = ones if sample_frac >= 1.0 or len(ones) <= 60 else \
            rng.choice(ones, max(40, int(len(ones) * sample_frac)), replace=False)
        best_new_P = float(c_cur.max())
        best_pq = None
        for p in p_list:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        best_2p = None
        if len(ones) >= 4 and len(zeros) >= 4 and rng.random() < two_frac:
            for _ in range(n_2try):

def nesterov_opt(h, N, deadline, k=70, lr0=0.01):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        y_new = y - lr * grad
        y_new = np.clip(y_new, 0.0, 1.0)
        s = float(y_new.sum())
        if abs(s - target) > 1e-8:
            lo, hi = -1.0, 1.0
            for _ in range(40):
                mid = (lo + hi) * 0.5

                    lo = mid
                else:
                    hi = mid
            y_new = np.clip(y_new - (lo + hi) * 0.5, 0.0, 1.0)
        m = beta * (y_new - h)
        h = y_new
        it += 1
        if it % 150 == 0:
            lr *= 0.982
            beta = min(0.97, 0.86 + it * 0.0002)
    return best_h, best_pk * dx

    x_new = np.linspace(0, 2, N_new, endpoint=False)
