    score_p[valid_ms] += h[np.clip(ms[valid_ms],0,N-1)]
    # Score for adding q: h[q-s] + h[q+s] (want high)
    qs = zeros + s
    mz = zeros - s
    score_q = np.zeros(len(zeros))
    valid_qs = (qs>=0)&(qs<N)
    valid_mz = (mz>=0)&(mz<N)
    score_q[valid_qs] += h[np.clip(qs[valid_qs],0,N-1)]
    score_q[valid_mz] += h[np.clip(mz[valid_mz],0,N-1)]
    # Best: minimize score_p - score_q
    # Try top few candidates
    p_order = np.argsort(score_p)[:min(20, len(ones))]
    q_order = np.argsort(-score_q)[:min(20, len(zeros))]
    best_diff = 999
    best_pq = None
    for pi in p_order:
        for qi in q_order:
            d = score_p[pi] - score_q[qi]
            if d < best_diff:
                best_diff = d
                best_pq = (int(ones[pi]), int(zeros[qi]))
    return best_pq

The objective is max_k corr_k / (N/2). For a good solution, the maximum is achieved at a few shifts, not all. If I can identify the "active" shifts (those near the max), I can focus optimization on them.

**Alternative approach: Minimax via cutting plane**

1. Start with a set of "important" shifts S (initially {k*})
2. Solve: min max_{k∈S} corr_k (this is a linear program in the continuous case!)
3. Add the new argmax to S
4. Repeat until convergence
5. Round and do binary local search

The LP for step 2:
min t
s.t. corr_k ≤ t for k ∈ S
     0 ≤ h_i ≤ 1
     Σ h_i = N/2

Where corr_k = Σ_i h[i] * (1-h[i+k]) - this is NOT linear in h!

So it's not a simple LP. It's a quadratic program (or we can linearize with h[i]*(1-h[i+k]) ≤ ...).

Actually, for the continuous case, corr_k(h) = Σ_i h[i] - Σ_i h[i]*h[i+k], which is quadratic. The max of quadratics is not convex.

But we can use a **Dinkelbach/Freudenthal-type iteration**:
- Fix the active set S
- For h in [0,1]^N with sum constraint, minimize Σ_{k∈S} w_k * corr_k(h)
- This subgradient approach is what the Nesterov code was doing

Let me try a different angle. What if I use **coordinate descent** on the continuous relaxation?

minimize max_k corr_k(h)
subject to 0 ≤ h_i ≤ 1, Σ h_i = N/2

Subgradient of corr_k at h:
∂corr_k/∂h[i] = (1-h[i+k]) - h[i-k] (with h[j]=0 for j∉[0,N-1])

For the max, the subgradient is the subgradient of the active (max) component.

Using projected subgradient:
h_new = project(h - lr * (1-h[·+k*]) + h[·-k*], sum=N/2, [0,1])

Where k* is the current argmax.

This is O(N) per step (in addition to O(N log N) for the FFT to find k*). Very fast!

The convergence of subgradient is slow (O(1/√t)), but with Nesterov acceleration it's O(1/t).

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    V = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * np.conj(V)))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, N, rng):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    tgt = N // 2
    if n1 > tgt:
        idx = rng.choice(np.where(hb == 1)[0], n1 - tgt, replace=False)
        hb[idx] = 0
    elif n1 < tgt:
        idx = rng.choice(np.where(hb == 0)[0], tgt - n1, replace=False)
        hb[idx] = 1
    return hb

def targeted_swap(h, N, corr, top_n=15):
    k_star = int(np.argmax(corr))
    s = k_star - (N - 1)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None
    ps = ones + s
    ms = ones - s
    sp = np.zeros(len(ones))
    vps = (ps >= 0) & (ps < N)
    vms = (ms >= 0) & (ms < N)
    sp[vps] += h[np.clip(ps[vps], 0, N - 1)]
    sp[vms] += h[np.clip(ms[vms], 0, N - 1)]
    qs = zeros + s
    mz = zeros - s
    sq = np.zeros(len(zeros))
    vqs = (qs >= 0) & (qs < N)
    vmz = (mz >= 0) & (mz < N)
    sq[vqs] += h[np.clip(qs[vqs], 0, N - 1)]
    sq[vmz] += h[np.clip(mz[vmz], 0, N - 1)]
    p_ord = np.argsort(sp)[:min(top_n, len(ones))]
    q_ord = np.argsort(-sq)[:min(top_n, len(zeros))]
    best_d = 999.0
    best_pq = None
    for pi in p_ord:
        for qi in q_ord:
            d = sp[pi] - sq[qi]
            if d < best_d:
                best_d = d
                best_pq = (int(ones[pi]), int(zeros[qi]))
    return best_pq

def full_search_swap(h, N, corr, ones, zeros, sf=0.4, rng=None):
    L = 2 * N - 1
    n_s = max(30, int(len(ones) * sf))
    p_list = ones if sf >= 1.0 else ones[rng.choice(len(ones), n_s, replace=False)]
    shifts = np.arange(L) - (N - 1)
    best_new_P = float(corr.max()) + 1.0
    best_pq = None
    z_s = zeros[:, None] + shifts[None, :]
    z_m = zeros[:, None] - shifts[None, :]
    z_valid_s = (z_s >= 0) & (z_s < N)
    z_valid_m = (z_m >= 0) & (z_m < N)
    z_sc = np.zeros_like(z_s, dtype=float)
    z_sc[z_valid_s] = h[np.clip(z_s[z_valid_s], 0, N - 1)]
    z_sc += np.where(z_valid_m, h[np.clip(z_m[z_valid_m], 0, N - 1)], 0.0)
    for p in p_list:
        p_s = p + shifts
        p_m = p - shifts
        p_valid_s = (p_s >= 0) & (p_s < N)
        p_valid_m = (p_m >= 0) & (p_m < N)
        p_sc = np.zeros(L)
        p_sc[p_valid_s] = h[np.clip(p_s[p_valid_s], 0, N - 1)]
        p_sc[p_valid_m] += h[np.clip(p_m[p_valid_m], 0, N - 1)]
        base = corr + p_sc[None, :]
        cand = base - z_sc
        nm = cand.max(axis=1)
        mi = int(np.argmin(nm))
        if nm[mi] < best_new_P:
            best_new_P = float(nm[mi])
            best_pq = (int(p), int(zeros[mi]))
    return best_pq

def nesterov_opt(h, N, deadline, k=40, lr0=0.025):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        center = N - 1
        grad = np.zeros(N)
        weights = np.zeros(len(top))
        for ii, ai in enumerate(top):
            L_shift = int(ai - center)
            w = 1.0 + 0.5 * (corr[ai] - corr[-1])
            weights[ii] = max(0.1, w)
        wsum = weights.sum()
        for ii, ai in enumerate(top):
            L_shift = int(ai - center)
            w = weights[ii] / wsum
            if L_shift == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L_shift > 0:
                grad[L_shift:] += w * (1.0 - y[:N - L_shift])
                grad[:N - L_shift] -= w * y[L_shift:]
            else:
                M = -L_shift
                if M < N:
                    grad[:N - M] += w * (1.0 - y[M:])
                    grad[M:] -= w * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.97
            beta = min(0.98, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def greedy_polish(h, N, deadline, max_sweeps=500, sf=0.7):

        if result is None:

        p, q = result
        h[p] = 0.0
        h[q] = 1.0
    return best_h, best_c * (2.0 / N)
