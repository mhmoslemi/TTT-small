def flip_update(corr, h, p, N):
    """Compute the change in full correlation when h[p] flips from 0 to 1."""
    m = np.arange(2*N-1) - (N-1)
    # First term: (1-h[p+m]) for p+m in [0,N-1]
    idx1 = (p + m >= 0) & (p + m < N)
    term1 = np.zeros(2*N-1)
    term1[idx1] = 1 - h[p + m[idx1]]
    # Second term: -h[p-m] for p-m in [0,N-1]
    idx2 = (p - m >= 0) & (p - m < N)
    term2 = np.zeros(2*N-1)
    term2[idx2] = -h[p - m[idx2]]
    return term1 + term2

import numpy as np

def fft_corr_full(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr_full(h)
    return float(np.max(corr)) * (2.0 / N)

def incremental_flip(corr, h, p, flip_to_one, N):
    """Apply a flip and return updated corr. flip_to_one: 0->1 or 1->0."""
    m = np.arange(2*N - 1) - (N - 1)
    if flip_to_one:
        # h[p]: 0 -> 1
        idx1 = (p + m >= 0) & (p + m < N)
        idx2 = (p - m >= 0) & (p - m < N)
        corr[idx1] += (1.0 - h[p + m[idx1]])
        corr[idx2] -= h[p - m[idx2]]
        h[p] = 1.0
    else:
        # h[p]: 1 -> 0
        idx1 = (p + m >= 0) & (p + m < N)
        idx2 = (p - m >= 0) & (p - m < N)
        corr[idx1] -= (1.0 - h[p + m[idx1]])
        corr[idx2] += h[p - m[idx2]]
        h[p] = 0.0
    return corr, h

def fix_binary_sum(h, target_ones, rng):
    h = h.copy()

        to_add = rng.choice(idx, target_ones - n1, replace=False)
        h[to_add] = 1
    return h

import numpy as np
import time

def fft_corr_full(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr_full(h))) * (2.0 / N)

def fix_sum(h, target, rng):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        rem = rng.choice(idx, n1 - target, replace=False)
        h[rem] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        add = rng.choice(idx, target - n1, replace=False)
        h[add] = 1
    return h

def make_delta(oi, zi, h, N, m_arr):
    L = 2 * N - 1
    delta = np.zeros(L)
    ia = (oi + m_arr >= 0) & (oi + m_arr < N)
    ib = (oi - m_arr >= 0) & (oi - m_arr < N)
    delta[ia] -= (1.0 - h[oi + m_arr[ia]])
    delta[ib] += h[oi - m_arr[ib]]
    ic = (zi + m_arr >= 0) & (zi + m_arr < N)
    id_ = (zi - m_arr >= 0) & (zi - m_arr < N)
    delta[ic] += (1.0 - h[zi + m_arr[ic]])
    delta[id_] -= h[zi - m_arr[id_]]
    delta[N - 1] -= 2.0
    d = zi - oi
    if 0 < d < N:
        delta[d + N - 1] += 1.0
    elif -N < d < 0:
        delta[d + N - 1] += 1.0
    return delta
