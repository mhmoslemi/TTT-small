    L = 2 * N - 1
    H = np.fft.fft(h, L)

        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    
    return best_h, best_c

        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def sa_binary(h, deadline, n_iter_max=800000, t_init=0.04, t_min=5e-6, two_p=0.2, kick_every=8000):
    N = len(h)
    h = h.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max
    best_h = h.copy()
    n_iters = 0
    t = t_init
    no_improve = 0
    while time.time() < deadline and n_iters < n_iter_max:
        batch = 500
        for _ in range(batch):
            if n_iters >= n_iter_max:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            use_two = np.random.random() < two_p and len(ones) >= 2 and len(zeros) >= 2
            if use_two:
                o1 = ones[np.random.randint(len(ones))]
                o2 = ones[np.random.randint(len(ones))]
                while o2 == o1:
                    o2 = ones[np.random.randint(len(ones))]
                z1 = zeros[np.random.randint(len(zeros))]
                z2 = zeros[np.random.randint(len(zeros))]
                while z2 == z1:
                    z2 = zeros[np.random.randint(len(zeros))]
                h[o1] = 0; h[o2] = 0; h[z1] = 1; h[z2] = 1
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max
                if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):
                    cur_max = new_max; corr = new_corr
                    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                    if cur_max < best_c:
                        best_c = cur_max; best_h = h.copy(); no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[o1] = 1; h[o2] = 1; h[z1] = 0; h[z2] = 0
            else:
                o = ones[np.random.randint(len(ones))]
                z = zeros[np.random.randint(len(zeros))]
                h[o] = 0; h[z] = 1
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max
                if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):
                    cur_max = new_max; corr = new_corr
                    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                    if cur_max < best_c:
                        best_c = cur_max; best_h = h.copy(); no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[o] = 1; h[z] = 0
            n_iters += 1
        t = max(t * 0.96, t_min)
        if no_improve > kick_every:
            nk = min(8, len(ones) // 3, len(zeros) // 3)
            if nk >= 2:
                ko = np.random.choice(np.where(h == 1)[0], nk, replace=False)
                kz = np.random.choice(np.where(h == 0)[0], nk, replace=False)
                h[ko] = 0; h[kz] = 1
                ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                t = max(t * 2.0, 0.02)
            no_improve = 0
    return best_h, best_c * (2.0 / N)

def greedy_binary(h, deadline, n_crit=30, sz=55):
    N = len(h)
    h = h.astype(float).copy()
    corr = compute_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    center = N - 1
    no_imp = 0
    while time.time() < deadline and no_imp < 300:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        cur_peak = float(np.max(corr))
        best_move = None
        best_new = cur_peak
        ns = min(len(ones), sz)
        nz = min(len(zeros), sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        for oi in os_:
            d_rem = np.zeros(2 * N - 1)
            for L in top_lags:
                idx = center + L
                d = 0.0
                if 0 <= oi + L < N:
                    d += h[oi + L]
                if 0 <= oi - L < N:
                    d += h[oi - L]
                d_rem[idx] = d
            for zi in zs:
                new_peak = 0.0
                for L in top_lags:
                    idx = center + L
                    d = 0.0
                    if 0 <= zi + L < N:
                        d -= h[zi + L]
                    if 0 <= zi - L < N:
                        d -= h[zi - L]
                    val = corr[idx] + d_rem[idx] + d
                    if val > new_peak:
                        new_peak = val
                if new_peak < best_new - 1e-10:
                    best_new = new_peak
                    best_move = (oi, zi)
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
            corr = compute_corr(h)
            cp = float(np.max(corr))
            no_imp = 0
            if cp < best_c:
                best_c = cp
                best_h = h.copy()
        else:
            if no_imp > 80 and np.random.random() < 0.3:
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi] = 0; h[zi] = 1
                corr = compute_corr(h)
                cp = float(np.max(corr))
                if cp < best_c:
                    best_c = cp; best_h = h.copy()
            no_imp += 1
    return best_h, best_c * (2.0 / N)

def cont_opt(h, deadline, k=30, lr0=0.015, mom=0.90):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c; best_h = h.copy(); stale = 0
        else:
            stale += 1
        if stale > 50:
            lr0 *= 0.75; stale = 0
            if lr0 < 1e-5:
                break
        top_idx = np.argsort(corr)[-k:]
        wts = np.sort(corr)[-k:]
        wts = (wts - wts[0] + 1e-5) ** 2
        wts /= wts.sum()
        grad = np.zeros(N)
        for w, m in zip(wts, top_idx):
            M = abs(int(m) - center)
            if M == 0:
                grad += w * (1.0 - 2.0 * h)
            elif M < N:
                hi = np.arange(M, N)
                lo = np.arange(0, N - M)
                grad[hi] += w * (1.0 - h[hi - M])
                grad[lo] -= w * h[lo + M]
        h_new = project(h + mom * v - lr0 * grad, tgt)
        v = 0.82 * v + 0.28 * (h_new - h)
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    d1 = min(deadline - 550, t0 + 90)
    h1 = project(best_h + np.random.randn(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=30, lr0=0.012, mom=0.91)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    if time.time() < deadline - 350:
        hb = fix_binary(best_h)
        d2 = min(deadline - 330, time.time() + 250)
        hb, c2 = sa_binary(hb, d2, n_iter_max=600000, t_init=0.035, t_min=3e-6, two_p=0.2, kick_every=6000)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    if time.time() < deadline - 120:
        hb3 = fix_binary(best_h)
        d3 = min(deadline - 100, time.time() + 80)
        hb3, c3 = greedy_binary(hb3, d3, n_crit=30, sz=55)
        if c3 < best_c:
            best_c, best_h = c3, hb3.copy()

    for N_try in [400, 440, 500]:
        if time.time() > deadline - 90:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d4 = min(deadline - 40, time.time() + 50)
        h_bt, c4 = sa_binary(h_bt, d4, n_iter_max=200000, t_init=0.025, t_min=5e-6, two_p=0.2, kick_every=5000)
        if time.time() < deadline - 20:
            d5 = min(deadline - 15, time.time() + 18)
            h_bt, c5v = greedy_binary(h_bt, d5, n_crit=25, sz=50)
        else:
            c5v = c4
        if c5v < best_c:
            best_c, best_h = c5v, h_bt.copy()

    if time.time() < deadline - 5:
        hB = fix_binary(best_h)
        d6 = min(deadline - 2, time.time() + 35)
        hB, c6 = sa_binary(hB, d6, n_iter_max=200000, t_init=0.012, t_min=2e-6, two_p=0.25, kick_every=5000)
        if c6 < best_c:
            best_c, best_h = c6, hB.copy()

    final_h = fix_binary(best_h)
    final_c = compute_c5(final_h)
    cont_c = compute_c5(best_h)
    if cont_c < final_c:
        final_h, final_c = best_h, cont_c
    return final_h, final_c, len(final_h)
