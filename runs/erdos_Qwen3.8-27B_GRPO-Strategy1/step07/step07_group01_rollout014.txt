            h[best_pq[1]] = 1.0
    return best_h, best_P * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of(best_h)
    
    # Phase 1: Nesterov at N0 with various params
    configs = [(35, 50, 0.02, 0.85), (50, 120, 0.015, 0.90), (45, 80, 0.018, 0.87)]
    for i, (k_t, T_v, lr_v, b_v) in enumerate(configs):
        if time.time() > deadline - 400:
            break
        noise = 0.05 + 0.02 * i
        hc = best_h.copy() + np.random.randn(N0) * noise
        hc = project_sum(hc, N0 / 2.0)
        d = min(deadline - 350, time.time() + 100)
        h_n, c5_n = nesterov_opt(hc, N0, d, k_top=k_t, T=T_v, lr0=lr_v, beta0=b_v)
        if c5_n < best_c5:
            best_c5, best_h = c5_n, h_n.copy()
    
    # Phase 2: Binary search at N0
    if time.time() < deadline - 350:
        h_bin = to_binary(best_h, N0, 11)
        d = min(deadline - 300, time.time() + 140)
        h_br, c5_br = binary_search_h(h_bin, N0, d, seed_val=11)
        if c5_br < best_c5:
            best_c5, best_h = c5_br, h_br.copy()
    
    # Phase 3: Try coarser N for a different landscape
    N_coarse = max(400, N0 // 2)
    if time.time() < deadline - 280:
        h_c = upsample(best_h, N0, N_coarse)
        h_c = project_sum(h_c, N_coarse / 2.0)
        d = min(deadline - 250, time.time() + 80)
        h_cn, c5_cn = nesterov_opt(h_c, N_coarse, d, k_top=30, T=60, lr0=0.02, beta0=0.85)
        if c5_cn < best_c5:
            best_c5 = c5_cn
            best_h_coarse = h_cn.copy()
            # Upsample back and use as new starting point
            h_up = upsample(best_h_coarse, N_coarse, N0)
            if time.time() < deadline - 200:
                d2 = min(deadline - 170, time.time() + 60)
                h_br2, c5_br2 = binary_search_h(to_binary(h_up, N0, 22), N0, d2, seed_val=22)
                if c5_br2 < best_c5:
                    best_c5, best_h = c5_br2, h_br2.copy()
    
    # Phase 4: Upsample to finer resolution
    N_fine = min(1500, 2 * N0)
    if N_fine > N0 and time.time() < deadline - 220:
        h_f = upsample(best_h, len(best_h), N_fine)
        h_f = project_sum(h_f, N_fine / 2.0)
        d = min(deadline - 180, time.time() + 80)
        h_fn, c5_fn = nesterov_opt(h_f, N_fine, d, k_top=55, T=70, lr0=0.014, beta0=0.88)
        if c5_fn < best_c5:
            best_c5, best_h = c5_fn, h_fn.copy()
        if time.time() < deadline - 140:
            h_fb = to_binary(best_h, len(best_h), 33)
            d2 = min(deadline - 110, time.time() + 60)
            h_fbr, c5_fbr = binary_search_h(h_fb, len(best_h), d2, seed_val=33)
            if c5_fbr < best_c5:
                best_c5, best_h = c5_fbr, h_fbr.copy()
    
    # Phase 5: Second round of binary search from best
    N_cur = len(best_h)
    if time.time() < deadline - 120:
        # Add small perturbation to escape current basin
        h_pert = best_h.copy()
        rng2 = np.random.RandomState(seed + 100)
        n_pert = max(5, N_cur // 50)
        flip_idx = rng2.choice(N_cur, n_pert, replace=False)
        for fi in flip_idx:
            h_pert[fi] = 1.0 - h_pert[fi]
        h_pert = to_binary(h_pert, N_cur, seed + 200)
        d = min(deadline - 80, time.time() + 90)
        h_pr, c5_pr = binary_search_h(h_pert, N_cur, d, seed_val=seed + 200)
        if c5_pr < best_c5:
            best_c5, best_h = c5_pr, h_pr.copy()
    
    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 50:
        h_fin = best_h.copy() + np.random.randn(len(best_h)) * 0.03
        h_fin = project_sum(h_fin, len(best_h) / 2.0)
        d = min(deadline - 15, time.time() + 40)
        h_fin2, c5_fin = nesterov_opt(h_fin, len(best_h), d, k_top=60, T=200, lr0=0.008, beta0=0.92)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin2.copy()
    
    # Phase 7: Final binary polish
    if time.time() < deadline - 10:
        hF = to_binary(best_h, len(best_h), seed + 999)
        sf = 1.0 if len(best_h) <= 600 else 0.6
        d = min(deadline - 2, time.time() + 40)
        hF2, c5F = binary_search_h(hF, len(best_h), d, seed_val=seed + 888)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()
    
    final_c5 = c5_of(best_h)
    return best_h, final_c5, len(best_h)
