import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G)[:L].real

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_smooth_max(h, corr, alpha=20.0, top_k=30):
    """Gradient using smooth max approximation."""
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-top_k:]
    max_val = corr[top_idx[0]]
    # Soft max weights
    weights = np.exp(alpha * (corr[top_idx] - max_val))
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for w, m in zip(weights, top_idx):
        M = int(m) - center
        if M == 0:
            grad += w * (1.0 - 2.0 * h)
        elif abs(M) < N:
            if M > 0:
                hi_idx = np.arange(M, N)
                lo_idx = np.arange(0, N - M)
                grad[hi_idx] += w * (1.0 - h[hi_idx - M])
                grad[lo_idx] -= w * h[lo_idx + M]
            else:
                M = -M
                hi_idx = np.arange(M, N)
                lo_idx = np.arange(0, N - M)
                grad[hi_idx] += w * (1.0 - h[hi_idx - M])
                grad[lo_idx] -= w * h[lo_idx + M]
    return grad

    tgt = N / 2.0
    h = h.astype(float).copy()

    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr0
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 100:
                lr *= 0.7
                stale = 0
                v *= 0.1
        g = grad_smooth_max(h, corr, alpha=alpha, top_k=k)
        h_new = project(h + mom * v - lr * g, tgt)
        v = 0.9 * v + 0.1 * (h_new - h)
        h = h_new
    return best_h, best_c

def binary_sa(h, deadline, t_start=5.0, t_end=0.01, restarts=5):
    """Simulated annealing for binary optimization."""
    N = len(h)
    h = fix_binary(h.copy())
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    center = N - 1
    
    total_time_avail = deadline - time.time()
    est_iters = int(total_time_avail * 3000)  # rough estimate
    t = t_start
    cool_rate = (t_end / t_start) ** (1.0 / max(est_iters, 1))
    
    no_improve = 0
    it = 0
    while time.time() < deadline:
        it += 1
        # Periodic restart
        if no_improve > est_iters // restarts:
            h = best_h.copy()
            cur_c = best_c
            no_improve = 0
            t = t_start * 0.5
            continue
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Try swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        
        # SA acceptance
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):
            cur_c = new_c
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1
            h[j] = 0
            no_improve += 1
        
        t *= cool_rate
        if t < t_end:
            t = t_end
    
    return best_h, best_c

def binary_greedy(h, deadline, n_crit=15, s_sz=25, max_rounds=200):
    """Greedy local search: only accept improving swaps, try structured neighborhood."""
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    center = N - 1
    rounds = 0
    while time.time() < deadline and rounds < max_rounds:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        top_lags = np.argsort(corr)[-n_crit:]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        # Try swaps focused on top lags
        made_move = False
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        # Priority: ones/zeros that contribute to top lags
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        for oi in os_:
            if made_move:
                break
            for zi in zs:
                h[oi] = 0
                h[zi] = 1
                new_c = compute_c5(h)
                if new_c < c - 1e-10:
                    c = new_c
                    made_move = True
                    break
                else:
                    h[oi] = 1
                    h[zi] = 0
        if not made_move:
            # Random perturbation
            oi = ones[np.random.randint(len(ones))]
            zi = zeros[np.random.randint(len(zeros))]
            h[oi] = 0
            h[zi] = 1
        rounds += 1
    return best_h, compute_c5(best_h)

    w = np.exp(alpha * (corr[top_idx] - mx))
    w /= w.sum()
    grad = np.zeros(N)
    for wi, m in zip(w, top_idx):
        M = int(m) - center
        if M == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < M < N:
            grad[M:] += wi * (1.0 - h[:-M])
            grad[:-M] -= wi * h[M:]
        elif -N < M < 0:
            M = -M
            grad[M:] += wi * (1.0 - h[:-M])
            grad[:-M] -= wi * h[M:]
    return grad

def cont_opt(h, deadline, lr0=0.02, mom=0.88, alpha=30.0, top_k=30):
    N = len(h)
    tgt = float(N) / 2.0
    h = h.copy().astype(np.float64)
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr0
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 80:
                lr *= 0.7
                stale = 0
                v *= 0.2
        g = grad_smooth(h, corr, alpha, top_k)
        h_new = project(h + mom * v - lr * g, tgt)
        v = 0.88 * v + 0.12 * (h_new - h)
        h = h_new
    return best_h, best_c

def binary_sa(h, deadline, t_init=5.0, t_min=0.002, n_restarts=5, rng=None):
    N = len(h)
    h = fix_binary(h, rng).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    t = t_init
    total_est = max(10, int((deadline - time.time()) * 40000))
    cool = (t_min / t_init) ** (1.0 / total_est)
    no_imp = 0
    restart_len = total_est // max(n_restarts, 1)
    while time.time() < deadline:
        if no_imp > restart_len:
            h = best_h.copy()
            cur_c = best_c
            t = t_init * 0.4
            no_imp = 0
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < -1e-14 or (t > t_min and rng.random() < np.exp(-delta / t)):
            cur_c = new_c
            if cur_c < best_c - 1e-13:
                best_c = cur_c
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h[i] = 1
            h[j] = 0
            no_imp += 1
        t = max(t * cool, t_min)
    return best_h, best_c

def binary_greedy(h, deadline, n_crit=20, s_sz=30, rng=None):
    N = len(h)
    h = fix_binary(h, rng).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    stall = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 200:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        made = False
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = rng.choice(ones, ns, replace=False)
        zs = rng.choice(zeros, nz, replace=False)
        for oi in os_:
            if made:
                break
            for zi in zs:
                h[oi] = 0
                h[zi] = 1
                nc = compute_c5(h)
                if nc < c - 1e-11:
                    c = nc
                    made = True
                    break
                h[oi] = 1
                h[zi] = 0
        if not made:
            oi = ones[rng.integers(len(ones))]
            zi = zeros[rng.integers(len(zeros))]
            h[oi] = 0
            h[zi] = 1
            stall += 5
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=np.float64)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Continuous optimization (100s)
    d1 = min(deadline - 550, t0 + 100)
    h1 = project(h0 + rng.standard_normal(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, lr0=0.018, mom=0.9, alpha=35.0, top_k=28)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Binary SA - main search (400s)
    if time.time() < deadline - 300:
        d2 = min(deadline - 290, time.time() + 400)
        hb = fix_binary(best_h, rng)
        hb, c2 = binary_sa(hb, d2, t_init=4.0, t_min=0.001, n_restarts=5, rng=rng)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    # Phase 3: Greedy refinement (130s)
    if time.time() < deadline - 110:
        d3 = min(deadline - 100, time.time() + 130)
        hb3 = fix_binary(best_h, rng)
        hb3, c3 = binary_greedy(hb3, d3, n_crit=22, s_sz=35, rng=rng)
        if c3 < best_c:
            best_c, best_h = c3, hb3.copy()

    # Phase 4: Try alternate resolutions
    for N_try in [390, 420]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int, rng)
        d5 = min(deadline - 60, time.time() + 40)
        h_bt, c5v = binary_sa(h_bt, d5, t_init=3.0, t_min=0.002, n_restarts=3, rng=rng)
        if c5v < best_c:
            best_c, best_h = c5v, h_bt.copy()

    # Phase 5: Another SA burst on best (60s)
    if time.time() < deadline - 15:
        d6 = min(deadline - 5, time.time() + 60)
        hb6 = fix_binary(best_h, rng)
        hb6, c6 = binary_sa(hb6, d6, t_init=2.0, t_min=0.0005, n_restarts=4, rng=rng)
        if c6 < best_c:
            best_c, best_h = c6, hb6.copy()

    # Phase 6: Final greedy polish
    if time.time() < deadline - 2:
        d7 = deadline - 2
        hB = fix_binary(best_h, rng)
        hB, c7 = binary_greedy(hB, d7, n_crit=25, s_sz=45, rng=rng)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()

    final_c = compute_c5(best_h)
    hB = fix_binary(best_h, rng)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
