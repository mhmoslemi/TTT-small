def fft_corr(h):
    """Compute full linear cross-correlation of h and (1-h) using FFT."""
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    """Project to binary with exactly target_ones ones."""
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def golden_ratio_init(N):
    """Initialize using golden ratio distribution for even spread."""
    nh = N // 2
    phi = (1 + np.sqrt(5)) / 2
    positions = np.floor(np.arange(nh) * phi % 1 * N).astype(int)
    h = np.zeros(N)

    # Fix if any duplicates

        # Fall back to uniform
        positions = np.linspace(0, N-1, nh).round().astype(int)
        h = np.zeros(N)

    return h

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def golden_init(N):
    nh = N // 2
    inv_phi = 1.0 / ((1.0 + np.sqrt(5.0)) / 2.0)
    positions = np.floor(np.arange(nh) * inv_phi * N).astype(int) % N
    h = np.zeros(N)
    # Handle duplicates
    seen = set()
    count = 0
    for p in positions:
        if p not in seen:
            h[p] = 1.0
            seen.add(p)
            count += 1
    if count < nh:
        # Fill remaining
        empty = np.where(h == 0)[0]
        extra = np.random.choice(empty, nh - count, replace=False)
        h[extra] = 1.0
    return h

def interleaved_init(N, freq1, freq2):
    nh = N // 2
    x = np.arange(N)
    v = np.sin(2*np.pi*freq1*x/N) + 0.7*np.sin(2*np.pi*freq2*x/N)
    h = (v > np.median(v)).astype(float)
    return project_binary(h, nh)

def stepl_search(h, N, deadline, batch=300, max_iters=100000):
    """Steepest descent: evaluate batch of random swaps, pick best."""
    h = h.astype(float).copy()
    nh = N // 2
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Sample batch of swaps
        oi_b = ones[np.random.randint(len(ones), size=batch)]
        zi_b = zeros[np.random.randint(len(zeros), size=batch)]
        
        best_dE = 1e-14  # threshold: must improve

        for idx in range(batch):
            oi, zi = oi_b[idx], zi_b[idx]
            if oi == zi:
                continue
            h[oi] = 0
            h[zi] = 1
            nm = float(np.max(fft_corr(h)))
            dE = nm - cur_max
            h[oi] = 1
            h[zi] = 0
            if dE < best_dE:
                best_dE = dE
                best_oi, best_zi = oi, zi
        
        if best_oi >= 0:
            h[best_oi] = 0
            h[best_zi] = 1
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c_now = cur_max * 2.0 / N
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Perturb to escape
            if no_improve > 50:
                for _ in range(np.random.randint(2, 8)):
                    a = ones[np.random.randint(len(ones))]
                    b = zeros[np.random.randint(len(zeros))]
                    h[a], h[b] = h[b], h[a]
                corr = fft_corr(h)
                cur_max = float(np.max(corr))
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                no_improve = 0
    
    return best_h, best_c

def anneal_search(h, N, deadline, T0=0.05, cool_factor=0.9995, max_iters=200000):

    cur_max = float(np.max(corr))

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

            h[zi] = 0

                a = ones[np.random.randint(len(ones))]
                b = zeros[np.random.randint(len(zeros))]

            cur_max = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            T = max(T, 0.03)
            no_improve = 0
    
    return best_h, best_c

    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        oi = ones[np.random.randint(len(ones))]
        zi = zeros[np.random.randint(len(zeros))]
        if oi == zi:
            continue
        h[oi] = 0.0
        h[zi] = 1.0
        nm = float(np.max(fft_corr(h)))
        dE = nm - cur_max
        if dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
            cur_max = nm
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max * 2.0 / N < best_c:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[oi] = 1.0
            h[zi] = 0.0
        T *= cool_factor
        if no_improve > 4000:
            for _ in range(5):
                a = ones[np.random.randint(len(ones))]
                b = zeros[np.random.randint(len(zeros))]
                h[a], h[b] = h[b], h[a]
            cur_max = float(np.max(fft_corr(h)))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            T = max(T, 0.025)
            no_improve = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 > 0.5).astype(float)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Heavy steepest descent on the given construction
    d1 = min(deadline - 500, t0 + 350)
    hs, c1 = stepl_search(best_h, N0, d1, batch=400, max_iters=300000)
    if c1 < best_c5:
        best_c5, best_h = c1, hs.copy()

    # Phase 2: Annealing to explore neighborhood
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 200)
        h2, c2 = anneal_search(best_h, N0, d2, T0=0.03, cool_factor=0.99998)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Another steepest pass (finer convergence)
    if time.time() < deadline - 180:
        d3 = min(deadline - 130, time.time() + 150)
        h3, c3 = stepl_search(best_h, N0, d3, batch=600, max_iters=200000)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Try different N values with structured initializations
    n_list = [200, 240, 280, 320, 360, 400, 440]
    for N_try in n_list:
        if time.time() > deadline - 60:
            break
        if N_try == N0:
            continue
        if N_try % 2 != 0:
            N_try += 1
        best_loc_c = np.inf
        best_loc_h = None
        inits = [golden_init(N_try), sine_init(N_try, 3, 7),
                 sine_init(N_try, 5, 11)]
        for vi, hv in enumerate(inits):
            if time.time() > deadline - 40:
                break
            d4 = min(deadline - 30, time.time() + 28)
            hr, cr = stepl_search(hv, N_try, d4, batch=300, max_iters=80000)
            if cr < best_loc_c:
                best_loc_c, best_loc_h = cr, hr.copy()
        if best_loc_h is not None and best_loc_c < best_c5:
            best_c5, best_h = best_loc_c, best_loc_h

    # Phase 5: Final polish with high batch steepest
    hF = (best_h > 0.5).astype(float)
    d5 = deadline - 1
    hF, c5F = stepl_search(hF, len(hF), d5, batch=600, max_iters=200000)
    c5F = compute_c5(hF, len(hF))
    if c5F < best_c5:
        best_c5, best_h = c5F, hF

    # Ensure valid binary output
    hB = (best_h > 0.5).astype(float)
    nh = len(hB) // 2
    if int(round(hB.sum())) != nh:
        hB = project_binary(hB, nh)
    cB = compute_c5(hB, len(hB))
    if cB < best_c5:
        best_c5, best_h = cB, hB

    return best_h, best_c5, len(best_h)
