def compute_D(h, N):
    """Compute D[p, k] = change in corr[k] if h[p] goes from 1 to 0."""
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv  # p - shift
    pp = pos + sv  # p + shift
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    # When h[p] changes by delta=1 (0->1):
    # corr changes by: (1-h[p+L]) for valid p+L, minus h[p-L] for valid p-L
    # When h[p] changes by delta=-1 (1->0):
    # corr changes by: -(1-h[p+L]) for valid p+L, plus h[p-L] for valid p-L
    # D here: change when h[p] goes 1->0
    return -(1.0 - hp) * vp + hm * vm

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary_sum(h, N, seed=42):
    h = h.astype(float).copy()
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - target, replace=False)
        h_bin[idx] = 0.0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], target - n1, replace=False)
        h_bin[idx] = 1.0
    return h_bin

def random_binary(N, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(N)
    h[rng.choice(N, N // 2, replace=False)] = 1.0
    return h

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hp) * vp + hm * vm

def sa_binary(h, N, deadline, T0=3.0, Tcool=0.999, T_min=0.02, sf=1.0, kick_every=60, kick_size=3):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    stall = 0
    rng = np.random.RandomState(999)
    while time.time() < deadline:
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = max(T0 * 0.4, T_min * 3)
            c = fft_corr(h)
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
        best_nm = P + 1.0
        best_p = int(p_list[0])
        best_qi = 0
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            nms = cand.max(axis=1)
            mi = int(np.argmin(nms))
            if nms[mi] < best_nm:
                best_nm = float(nms[mi])
                best_p = int(p)
                best_qi = mi
        delta = best_nm - P
        if delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_p] = 0.0
            h[int(zeros[best_qi])] = 1.0
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        c = fft_corr(h)
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    best_h = None
    best_c5 = 1.0

    # Phase 1: SA at N=200 (very fast, easy to optimize)
    N1 = 200
    d1 = min(deadline - 800, t0 + 80)
    for rng_seed in [seed, seed + 7, seed + 13]:
        if time.time() > d1:
            break
        h1 = random_binary(N1, seed=rng_seed)
        h1b, c5_1 = sa_binary(h1, N1, d1, T0=2.0, Tcool=0.9995, T_min=0.01, sf=1.0, kick_every=40, kick_size=3)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1b.copy()

    # Phase 2: SA at N=300
    N2 = 300
    d2 = min(deadline - 600, t0 + 220)
    for rng_seed in [seed + 1, seed + 20]:
        if time.time() > d2:
            break
        h2 = random_binary(N2, seed=rng_seed)
        h2b, c5_2 = sa_binary(h2, N2, d2, T0=3.0, Tcool=0.999, T_min=0.02, sf=1.0, kick_every=50, kick_size=3)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2b.copy()

    # Phase 3: SA at N=400
    N3 = 400
    if time.time() < deadline - 400:
        d3 = min(deadline - 350, t0 + 380)
        h3 = random_binary(N3, seed=seed + 33)
        h3b, c5_3 = sa_binary(h3, N3, d3, T0=3.5, Tcool=0.999, T_min=0.02, sf=0.7, kick_every=70, kick_size=4)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3b.copy()

    # Phase 4: SA at N=760 from initial_h_values
    try:
        N4 = len(initial_h_values)
        if time.time() < deadline - 200:
            h4 = ensure_binary_sum(np.array(initial_h_values, dtype=float), N4)
            d4 = min(deadline - 150, t0 + 550)
            h4b, c5_4 = sa_binary(h4, N4, d4, T0=4.0, Tcool=0.999, T_min=0.03, sf=0.5, kick_every=80, kick_size=4)
            if c5_4 < best_c5:
                best_c5, best_h = c5_4, h4b.copy()
    except Exception:
        pass

    # Phase 5: Upsample best small-N solution and SA at larger N
    if time.time() < deadline - 150:
        N_small = len(best_h)
        N_up = 2 * N_small if N_small <= 300 else 4 * N_small
        N_up = min(N_up, 800)
        h_up = ensure_binary_sum(upsample(best_h, N_small, N_up), N_up)
        d5 = min(deadline - 80, t0 + 700)
        sf_up = 1.0 if N_up <= 300 else (0.6 if N_up <= 500 else 0.35)
        h_upb, c5_up = sa_binary(h_up, N_up, d5, T0=3.0, Tcool=0.999, T_min=0.02, sf=sf_up, kick_every=60, kick_size=3)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_upb.copy()

    # Phase 6: Final refinement at the best N
    if time.time() < deadline - 5:
        N_fin = len(best_h)
        h_fin = best_h.copy()
        d6 = deadline - 2
        sf_fin = 1.0 if N_fin <= 300 else (0.6 if N_fin <= 500 else 0.3)
        h_finb, c5_fin = sa_binary(h_fin, N_fin, d6, T0=2.0, Tcool=0.9995, T_min=0.01, sf=sf_fin, kick_every=50, kick_size=2)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_finb.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
