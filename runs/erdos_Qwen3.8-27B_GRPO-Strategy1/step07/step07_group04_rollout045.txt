import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N, seed=42):
    nh = N // 2
    h = np.asarray(h, dtype=float)
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    rng = np.random.RandomState(seed + N)
    if n1 > nh:
        ones = np.flatnonzero(hb == 1)
        hb[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.flatnonzero(hb == 0)
        hb[rng.choice(zeros, nh - n1, replace=False)] = 1
    return hb

def compute_Dmat(h, N):
    Nf = 2 * N - 1
    shifts = np.arange(Nf) - (N - 1)
    i = np.arange(N)[:, None]
    pm = i - shifts[None, :]
    pp = i + shifts[None, :]
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pmc = np.clip(pm, 0, N - 1)
    ppc = np.clip(pp, 0, N - 1)
    D_rem = -(1.0 - h[ppc]) * vp + h[pmc] * vm
    D_add = (1.0 - h[ppc]) * vp - h[pmc] * vm
    return D_rem, D_add

def local_search(h, N, deadline, max_sweeps=2000, sample_frac=1.0, stall_limit=45):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) < 1 or len(zeros) < 1:
            break
        D_rem, D_add = compute_Dmat(h, N)
        n_s = max(15, int(len(ones) * sample_frac))
        p_s = np.random.choice(ones, min(n_s, len(ones)), replace=False)
        best_new = P + 1.0
        best_pq = None
        for p in p_s:
            base = c + D_rem[p]
            cand_max = (D_add[zeros] + base[None, :]).max(axis=1)
            mi = int(np.argmin(cand_max))
            if cand_max[mi] < best_new:
                best_new = float(cand_max[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def continuous_refine(h0, N, deadline, lr0=0.03, n_iter=2000):
    h = h0.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    lr = lr0
    beta = 0.9
    m = np.zeros(N)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        y = h + m
        c = fft_corr(y)
        pk = float(np.max(c))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        k_top = min(40, len(c))
        top = np.argpartition(-c, k_top - 1)[:k_top]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        y_new = y - lr * grad
        y_new = np.clip(y_new, 0.0, 1.0)
        s = y_new.sum()
        if abs(s - target) > 1e-10:
            lo, hi = -1.0, 1.0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                if np.clip(y_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            y_new = np.clip(y_new - (lo + hi) * 0.5, 0.0, 1.0)
        h_new = y_new
        m = beta * (h_new - h)
        h = h_new
        if it % 300 == 0:
            lr *= 0.97
            beta = min(0.97, 0.85 + it * 0.0002)
    return best_h, best_pk * dx
