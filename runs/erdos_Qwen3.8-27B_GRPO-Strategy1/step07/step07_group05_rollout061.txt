def fft_linear_corr(h):
    """Compute linear correlation using FFT (zero-padded)"""
    N = len(h)
    L = 2 * N - 1
    # Zero-pad
    hp = np.zeros(L)
    hp[N-1:] = h  # place at end
    bp = np.zeros(L)
    bp[N-1:] = 1.0 - h
    
    H = np.fft.rfft(hp)
    B = np.fft.rfft(bp)
    circ = np.real(np.fft.irfft(H * np.conj(B), L))
    
    # The circular correlation of zero-padded signals gives linear correlation
    # circ[k] = sum_i hp[i] * bp[(i-k) mod L]
    # For k in [0, L-1], this is the cross-correlation
    # We want correlate(h, 1-h, 'full') which has 2N-1 = L elements
    # correlate(a,b,'full')[k] = sum_n a[n] * b[n-(k-(N-1))]  (0-indexed)
    # = sum_n a[n] * conj(b)[n-k+N-1]
    
    # Actually, let me just use the standard formula:
    # np.correlate(a, v, 'full')[k] = sum_n a[n+k-N+1] * conj(v[n])  for k=0,...,2N-2
    # With a=h, v=1-h:
    # result[k] = sum_n h[n+k-N+1] * (1-h[n])
    
    # The FFT approach: convolve h with reversed(1-h)
    rev_b = (1.0 - h)[::-1]

    # = sum_i h[i] * (1-h[N-1-k+i])

    # = sum_j h[j-N+1+k] * (1-h[j])
    # This is correlate(h, 1-h, 'full')[k + N - 1] for valid k
    # Actually conv has length L = 2N-1, and:
    # conv[k] = correlate(h, 1-h, 'full')[N-1+k] for k=-(N-1),...,N-1
    # So full_corr = conv reversed appropriately
    
    # Let me verify: for N=3, h=[a,b,c], 1-h=[A,B,C]

def fast_c5(h):
    """Fast C5 computation"""
    N = len(h)
    corr = fft_linear_corr(h)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy()
    n1 = int(np.sum(h > 0.5))
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target-n1]] = 1
    return h

import numpy as np
import time

def fft_linear_corr(h):
    N = len(h)
    L = 2 * N - 1
    rev_b = (1.0 - h)[::-1]
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(rev_b, L), L))

def fast_c5(h):
    N = len(h)
    corr = fft_linear_corr(h)
    return float(np.max(corr)) * (2.0 / N), corr

def project_binary(h, target_ones):
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(np.sum(h == 1))
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target-n1]] = 1
    return h

def anneal(h, N, deadline, T0=0.05, T1=1e-5, iters=30000,

    T, cool = T0, (T1/T0)**(1.0/max(1,iters))

        it += 1

            oi, zi = int(np.random.choice(ones)), int(np.random.choice(zeros))

            nc, _ = fast_c5(h)
            dE = nc - cur_c
            if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE/T)):
                cur_c = nc

                while j < N and h[j] == h[i]: j += 1
                if j - i >= B:
                    if h[i] == 1: os_.append(i)
                    else: zs.append(i)
                i = j
            if not os_ or not zs:

import numpy as np
import time

def fft_linear_corr(h):
    N = len(h)
    L = 2 * N - 1
    rev_b = (1.0 - h)[::-1]
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(rev_b, L), L))

def fast_c5(h):
    N = len(h)
    corr = fft_linear_corr(h)
    return float(np.max(corr)) * (2.0 / N), corr

def project_binary(h, target_ones):
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy().astype(float)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(np.sum(h == 1))
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h
