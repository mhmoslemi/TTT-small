    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

    return float(np.max(fft_corr(h))) * (2.0 / N)

        return h
    lo, hi = -1.5, 1.5

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def smooth_obj(h, T, target, lam, N):
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    es = e.sum()
    w = e / es
    val = mx + np.log(es) / T
    g = np.zeros(N)
    ctr = N - 1
    active = np.where(w > 1e-8)[0]
    for ai in active:
        g += w[ai] * lin_grad(int(ai - ctr), h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, N, deadline, T=30.0, max_outer=4):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lam = 5.0

        c = compute_c5(h, N)

                jj2 = int(zeros[np.random.randint(len(zeros))])
                h[ii2], h[jj2] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                corr = fft_corr(h)
                pk = float(np.max(corr))
    corr2 = fft_corr(best_h)
    best_pk = float(np.max(corr2))
    return best_h, best_pk * (2.0 / N)

def binary_l2(g, N, deadline, max_iters=2000):
    """Minimize sum of squares of overlaps via swap search"""
    h = (g > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        p = np.where(h == 1)[0]
        h[p[np.random.choice(n1, n1 - nh, replace=False)]] = 0
    elif n1 < nh:
        p = np.where(h == 0)[0]
        h[p[np.random.choice(n1, nh - n1, replace=False)]] = 1
    corr = fft_corr(h)
    ctr = N - 1
    valid = np.arange(1, N)
    ov = np.zeros(2 * N - 1)
    for dd in valid:
        ov[ctr + dd] = corr[ctr + dd]
        ov[ctr - dd] = corr[ctr - dd]
    l2 = float(np.sum(ov ** 2))
    best_l2 = l2
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        best_imp = 0.0
        best_ii = best_jj = -1
        for t in range(25):
            ii = int(ones[np.random.randint(len(ones))])
            jj = int(zeros[np.random.randint(len(zeros))])
            d_ov = np.zeros(2 * N - 1)
            for dd in np.random.choice(valid, min(len(valid), 80), replace=False):
                d = int(dd)
                delta = 0.0
                im, ip = ii - d, ii + d
                jm, jp = jj - d, jj + d
                if 0 <= im < N:
                    delta += h[im]
                if 0 <= ip < N:
                    delta += h[ip]
                if 0 <= jm < N:
                    delta -= h[jm]
                if 0 <= jp < N:
                    delta -= h[jp]
                d_ov[ctr + int(dd)] = delta
                d_ov[ctr - int(dd)] = -delta
            new_ov = ov + d_ov
            new_l2 = float(np.sum(new_ov ** 2))
            imp = l2 - new_l2
            if imp > best_imp:
                best_imp = imp
                best_ii, best_jj = ii, jj
        if best_ii >= 0:
            h[best_ii], h[best_jj] = 0.0, 1.0
            corr = fft_corr(h)
            for dd in valid:
                ov[ctr + dd] = corr[ctr + dd]
                ov[ctr - dd] = corr[ctr - dd]
            l2 = float(np.sum(ov ** 2))
            if l2 < best_l2:
                best_l2 = l2
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            stall += 1
    c5 = compute_c5(best_h, N)
    return best_h, c5

def multistart_binary(start, N, deadline, n_starts=4):
    best_h = start.copy()
    best_c5 = compute_c5(start, N)
    per = max(5.0, (deadline - time.time()) / n_starts)
    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        sd = time.time() + per
        h_s = start.copy()
        if s == 1:
            h_s = np.roll(h_s, np.random.randint(1, N))
        elif s == 2:
            h_s = h_s[::-1]
        elif s == 3:
            h_s = np.roll(h_s, N // 3)
        h_s = (h_s + np.random.randn(N) * 0.05)
        h_s = proj_sum(h_s, N / 2.0)
        h_r, c_r = binary_smart(h_s, N, sd)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 4.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: L-BFGS continuous (multiple T)
    d1 = min(dl - 500, t0 + 120)
    for T_val in [25.0, 45.0]:
        if time.time() > d1:
            break
        h_p = best_h + np.random.randn(N0) * 0.15
        h_p = proj_sum(h_p, N0 / 2.0)
        h_r, c_r = lbfgs_opt(h_p, N0, min(time.time() + 55, d1), T=T_val, max_outer=4)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r

    # Phase 2: Smart binary search on best
    if time.time() < dl - 400:
        d2 = min(dl - 350, time.time() + 250)
        h_b, c_b = binary_smart(best_h, N0, d2, max_iters=5000, ntrial=40, gap=1.0)
        if c_b < best_c5:
            best_c5, best_h = c_b, h_b

    # Phase 3: L2-based binary (different landscape)
    if time.time() < dl - 300:
        d3 = min(dl - 250, time.time() + 120)
        h_l, c_l = binary_l2(best_h, N0, d3, max_iters=1500)
        if c_l < best_c5:
            best_c5, best_h = c_l, h_l

    # Phase 4: Multi-start binary with structured perturbations
    if time.time() < dl - 200:
        d4 = min(dl - 150, time.time() + 200)
        h_m, c_m = multistart_binary(best_h, N0, d4, n_starts=4)
        if c_m < best_c5:
            best_c5, best_h = c_m, h_m

    # Phase 5: Final L-BFGS polish
    if time.time() < dl - 80:
        d5 = min(dl - 40, time.time() + 60)
        h_f = best_h + np.random.randn(N0) * 0.04
        h_f = proj_sum(h_f, N0 / 2.0)
        h_r, c_r = lbfgs_opt(h_f, N0, d5, T=40.0, max_outer=3)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r

    # Phase 6: Final binary polish (tight gap)
    if time.time() < dl - 10:
        d6 = min(dl - 3, time.time() + 60)
        h_f2, c_f2 = binary_smart(best_h, N0, d6, max_iters=3000, ntrial=50, gap=0.5)
        if c_f2 < best_c5:
            best_c5, best_h = c_f2, h_f2

    # Ensure binary
    h_bin = (best_h > 0.5).astype(float)
    nh = N0 // 2
    n1 = int(h_bin.sum())
    if n1 > nh:
        p = np.where(h_bin == 1)[0]
        h_bin[p[np.random.choice(n1, n1 - nh, replace=False)]] = 0
    elif n1 < nh:
        p = np.where(h_bin == 0)[0]
        h_bin[p[np.random.choice(n1, nh - n1, replace=False)]] = 1
    c5_bin = compute_c5(h_bin, N0)
    if c5_bin <= best_c5:
        best_h, best_c5 = h_bin, c5_bin
    else:
        best_h = proj_sum(best_h, N0 / 2.0)
        best_c5 = compute_c5(best_h, N0)
    return best_h, best_c5, N0
