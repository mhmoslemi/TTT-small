    """One step of softmax-approximated gradient descent."""
    N = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    
    # Softmax weights
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / tau)
    w /= w.sum()
    
    # Gradient: sum_k w[k] * grad_corr_k
    grad = np.zeros(N)
    center = N - 1
    for k in range(2*N-1):
        w_k = w[k]
        if w_k < 1e-7:
            continue

        if lag >= 0:

import time

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    N = len(h)
    return float(np.max(full_corr(h))) * (2.0 / N)

def to_binary(h, n_ones):
    idx = np.argsort(-h)
    out = np.zeros_like(h, dtype=float)
    out[idx[:n_ones]] = 1.0
    return out

def project_sum(h, target):
    """Project h to [0,1]^N with sum = target using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_grad(h, w):
    """Compute gradient of weighted sum of correlations w.r.t. h."""
    N = len(h)
    g = 1.0 - h
    center = N - 1
    grad = np.zeros(N)
    
    # Zero lag
    w0 = w[center]
    if w0 > 1e-9:
        grad += w0 * (1.0 - 2.0 * h)
    
    # Positive lags k = 1, ..., N-1
    for k in range(1, N):
        wk = w[center + k]
        if wk < 1e-9:
            continue
        grad[k:] += wk * g[:N-k]
        grad[:N-k] -= wk * h[k:]
    
    # Negative lags k = 1, ..., N-1 (w[center-k])
    for k in range(1, N):
        wk = w[center - k]
        if wk < 1e-9:
            continue
        grad[:N-k] += wk * g[k:]
        grad[k:] -= wk * h[:N-k]
    
    return grad

def continuous_opt(m_h, N, deadline, tau0=0.15, tau_min=0.005, lr0=0.003, lr_decay=0.9999):
    """Gradient-based continuous optimization."""
    h = m_h.astype(float).copy()
    h = project_sum(h, N / 2.0)
    best_h = h.copy()
    best_c = c5_of(h)
    tau = tau0
    lr = lr0
    it = 0
    
    while time.time() < deadline and tau > tau_min:
        it += 1
        corr = full_corr(h)
        max_c = np.max(corr)
        # Softmax weights
        w = np.exp((corr - max_c) / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        
        grad = compute_grad(h, w)
        h_new = h - lr * grad
        h_new = project_sum(h_new, N / 2.0)
        
        c_new = c5_of(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive learning rate
        if it % 50 == 0:
            lr *= lr_decay
        if it % 200 == 0:
            tau *= 0.95
    
    return best_h, best_c

def binary_local_search(h, N, deadline, T0=0.03, T1=1e-6):
    """Simulated annealing with swap moves on binary h."""
    h = to_binary(h, N // 2)
    nh = N // 2
    corr = full_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = T0
    stall = 0
    stall_limit = 3000
    it = 0
    max_iters = 10**9
    
    while time.time() < deadline and it < max_iters:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Targeted move: bias toward peak
        peak_idx = int(np.argmax(corr))
        lag = peak_idx - (N - 1)
        
        if np.random.random() < 0.6:
            # Targeted: pick a 1-position that contributes to peak
            if lag > 0:
                cands = [n + lag for n in range(N - lag) if h[n] == 0 and h[n + lag] == 1]
                if cands:
                    oi = cands[np.random.randint(len(cands))]
                else:
                    oi = int(np.random.choice(ones))
            elif lag < 0:
                a = -lag
                cands = [m for m in range(N - a) if h[m] == 1 and h[m + a] == 0]
                if cands:
                    oi = cands[np.random.randint(len(cands))]
                else:
                    oi = int(np.random.choice(ones))
            else:
                oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
        else:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
        
        if oi == zi:
            continue
        
        h[oi], h[zi] = 0.0, 1.0
        new_corr = full_corr(h)
        new_max = float(np.max(new_corr))
        dE = new_max - cur_max
        
        if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
            cur_max = new_max
            corr = new_corr
            if cur_max * 2.0 / N < best_c - 1e-12:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[oi], h[zi] = 1.0, 0.0
        
        T *= 0.99997
        if stall > stall_limit:
            for _ in range(8):
                o = int(np.random.choice(ones))
                z = int(np.random.choice(zeros))
                h[o], h[z] = h[z], h[o]
            corr = full_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, 0.005)
            stall = 0
    
    return best_h, best_c

def structured_init(N, variant):
    np.random.seed(42 + variant * 997)
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[np.random.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        t = np.arange(N)
        v = np.sin(2*np.pi*7*t/N) + 0.6*np.sin(2*np.pi*13*t/N + 0.5) + 0.4*np.cos(2*np.pi*5*t/N)
        h = (v > np.median(v)).astype(float)
        h = to_binary(h, nh)
    elif variant == 2:
        h = ((np.arange(N) * 7) % 11 < 5).astype(float)
        h = to_binary(h, nh)
    else:
        t = np.arange(N)
        v = np.sum(np.sin(2*np.pi*f*t/N + 0.1*f) for f in [3, 7, 13, 21, 37])
        h = (v > np.median(v)).astype(float)
        h = to_binary(h, nh)
    return h
