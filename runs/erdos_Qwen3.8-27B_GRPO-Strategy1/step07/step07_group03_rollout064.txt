The gradient of corr[m] w.r.t. h[i]:
In the FFT formulation, corr = IFFT(FFT(h) * FFT(flip(1-h)))

The first term is sum(h) = N/2 (constant).
The second term is the autocorrelation of h at lag (m-(N-1)).

Where autocorr_h[lag] = sum_i h[i]*h[i+lag]

This is much simpler! We just need to minimize:

Subject to h ∈ [0,1]^N, sum(h) = N/2.

Wait, that's not quite right. We want to minimize max_lag (N/2 - autocorr[lag]) = N/2 - min_lag autocorr[lag].

We want all lags to have as many overlapping 1's as possible. This is like designing a binary sequence with high minimum autocorrelation!

        hb[rng.choice(idx, n1-nh, replace=False)] = 0
    else:
        idx = np.where(hb==0)[0]
        hb[rng.choice(idx, nh-n1, replace=False)] = 1

def compute_D(h, N):
    m = np.arange(2*N-1) - (N-1)
    i = np.arange(N)[:,None]
    j = i + m[None,:]
    valid = (j>=0) & (j<N)
    jc = np.clip(j, 0, N-1)
    return np.where(valid, 1.0-h[jc], 0.0)

def binary_anneal(h, N, deadline, rng, max_sweeps=600):
    h = ensure_binary(h, N, rng).copy()
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_h = h.copy()
    best_c = P_cur
    stall = 0
    temp = 0.008
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp * 0.996)
        
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        no, nz = len(ones), len(zeros)
        if no < 2 or nz < 2:
            break
        
        D = compute_D(h, N)
        n_m = 2*N - 1
        
        best_P = P_cur
        best_move = None
        
        # Full search for best single swap
        Dz = D[zeros]  # (nz, n_m)
        for p in ones:
            base = c_cur - D[p]  # (n_m,)
            cand = base[None,:] + Dz  # (nz, n_m)
            vals = cand.max(axis=1)  # (nz,)
            mi = int(np.argmin(vals))
            if vals[mi] < best_P:
                best_P = float(vals[mi])
                best_move = ('1s', int(p), int(zeros[mi]))
        
        # Two-swap
        if no >= 4 and nz >= 4:
            n_try = min(100, max(40, N//4))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, no, 2)
                j1, j2 = rng.randint(0, nz, 2)
                if i1==i2 or j1==j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                np_ = float((c_cur - D[p1] - D[p2] + D[q1] + D[q2]).max())
                if np_ < best_P:
                    best_P = np_
                    best_move = ('2s', p1, p2, q1, q2)
        
        if best_move is not None:
            if best_move[0] == '2s':
                h[best_move[2]] = h[best_move[3]] = 0
                h[best_move[4]] = h[best_move[5]] = 1
            else:
                h[best_move[1]] = 0
                h[best_move[2]] = 1
            c_cur = fft_corr(h)
            P_cur = float(c_cur.max())
            if P_cur < best_c - 1e-13:
                best_c = P_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Simulated annealing: accept random move with prob exp(-Δ/T)
            if stall > 15:
                for _ in range(5):
                    p = int(ones[rng.randint(no)])
                    q = int(zeros[rng.randint(nz)])
                    h[p], h[q] = 0.0, 1.0
                c_cur = fft_corr(h)
                P_cur = float(c_cur.max())
                stall = 0
            elif stall > 5 and rng.random() < temp * 100:
                p = int(ones[rng.randint(no)])
                q = int(zeros[rng.randint(nz)])
                h[p], h[q] = 0.0, 1.0
                c_cur = fft_corr(h)
                P_cur = float(c_cur.max())
                if P_cur < best_c - 1e-13:
                    best_c = P_cur
                    best_h = h.copy()
                stall = 0
            elif stall > 8:
                break
            else:
                stall += 1
    
    return best_h, best_c * (2.0/N)

def nesterov_refine(h, N, deadline, k=40, lr0=0.006):
    h = np.array(h, dtype=float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    
    for it in range(200000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0*y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= k
        
        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        
        if it % 300 == 0:
            lr *= 0.98
            beta = min(0.93, 0.85 + it*0.00003)
    
    return best_h, best_pk * dx

def final_grad_descent(h, N, deadline, n_iter=5000):
    """Very fine gradient descent to shave off tiny amounts."""
    h = np.array(h, dtype=float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_v = float(np.max(fft_corr(h)))
    lr = 0.001
    
    for it in range(n_iter):
        if time.time() > deadline or lr < 1e-8:
            break
        corr = fft_corr(h)
        kmax = int(np.argmax(corr))
        pk = float(corr[kmax])
        if pk < best_v - 1e-14:
            best_v = pk
            best_h = h.copy()
        
        center = N - 1
        L = kmax - center
        grad = np.zeros(N)
        if L == 0:
            grad = 1.0 - 2.0*h
        elif L > 0:
            grad[L:] += (1.0 - h[:N-L])
            grad[:N-L] -= h[L:]
        else:
            M = -L
            grad[:N-M] += (1.0 - h[M:])
            grad[M:] -= h[:N-M]
        
        h_new = project_sum(h - lr*grad, target)
        h = h_new
        lr *= 0.999
    
    return best_h, best_v * dx

def upsample(h_old, N_old, N_new):
    xo = np.linspace(0, 2, N_old, endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, xo, h_old)

                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= k
        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        if it % 400 == 0:
            lr *= 0.98
            beta = min(0.93, 0.85 + it*0.00002)
    return best_h, best_pk * dx

def fine_grad(h, N, deadline, topk=4):
    h = np.array(h, dtype=float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_v = float(np.max(fft_corr(h)))
    lr = 0.0008
    center = N - 1
    for it in range(50000):
        if time.time() > deadline or lr < 1e-10:
            break
        corr = fft_corr(h)
        top = np.argpartition(corr, -topk)[-topk:]
        pk = float(corr[top].max())
        if pk < best_v - 1e-14:
            best_v = pk
            best_h = h.copy()
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0*h)
            elif L > 0:
                grad[L:] += (1.0 - h[:N-L])
                grad[:N-L] -= h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - h[M:])
                    grad[M:] -= h[:N-M]
        grad /= topk
        h = project_sum(h - lr*grad, target)
        lr *= 0.9997
    return best_h, best_v * dx

def upsample(h_old, N_old, N_new):
    xo = np.linspace(0, 2, N_old, endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, N0)

    # Phase 1: Main binary annealing at N0
    d1 = min(deadline - 280, t0 + budget_s*0.40)
    h1, c5_1 = binary_anneal(best_h, N0, d1, rng, max_sweeps=1200)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Nesterov continuous at N0
    if time.time() < deadline - 200:
        hc = best_h + rng.randn(N0)*0.04
        hc = project_sum(hc, N0/2.0)
        d2 = min(deadline - 180, time.time() + budget_s*0.12)
        h2, c5_2 = nesterov_refine(hc, N0, d2, k=40, lr0=0.005)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    # Phase 3: Fine gradient descent
    if time.time() < deadline - 140:
        d3 = min(deadline - 130, time.time() + budget_s*0.05)
        h3, c5_3 = fine_grad(best_h, N0, d3, topk=4)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 4: Binary restart with perturbation
    if time.time() < deadline - 120:
        rng4 = np.random.RandomState(seed+555)
        hb = ensure_binary(best_h, N0, rng4)
        ot = np.where(hb==1)[0]
        zt = np.where(hb==0)[0]
        for _ in range(12):
            hb[ot[rng4.randint(len(ot))]] = 0
            hb[zt[rng4.randint(len(zt))]] = 1
        d4 = min(deadline - 100, time.time() + budget_s*0.12)
        h4, c5_4 = binary_anneal(hb, N0, d4, rng4, max_sweeps=500)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Phase 5: Try 2x resolution binary
    N1 = 2*N0
    if time.time() < deadline - 80:
        h_up = upsample(ensure_binary(best_h, len(best_h), rng), len(best_h), N1)
        h_up = ensure_binary(h_up, N1, rng)
        d5 = min(deadline - 60, time.time() + budget_s*0.10)
        h5, c5_5 = binary_anneal(h_up, N1, d5, rng, max_sweeps=180)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5

    # Phase 6: Final continuous polish
    if time.time() < deadline - 15:
        Nf = len(best_h)
        hf = best_h + rng.randn(Nf)*0.02
        hf = project_sum(hf, Nf/2.0)
        d6 = min(deadline - 10, time.time() + budget_s*0.04)
        h6, c5_6 = nesterov_refine(hf, Nf, d6, k=30, lr0=0.003)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6
        if time.time() < deadline - 4:
            h7, c5_7 = fine_grad(best_h, len(best_h), deadline-2, topk=3)
            if c5_7 < best_c5:
                best_c5, best_h = c5_7, h7

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
