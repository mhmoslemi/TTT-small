def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

    N = len(h)
    c = fft_corr(h)
    center = N - 1
    if k_top is None or k_top >= 2*N-1:
        top = np.arange(2*N-1)
    else:
        top = np.argpartition(c, -k_top)[-k_top:]
    
    grad = np.zeros(N)

        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv
    pp = pos - sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, 1.0 - h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return hm - hp
