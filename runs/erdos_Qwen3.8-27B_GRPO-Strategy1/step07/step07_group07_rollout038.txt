import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0/N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo+hi)/2.0
        if np.clip(h-mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2.0, 0, 1)

def soft_grad(h, corr, k=50, temp=2.0):
    N = len(h)
    center = N-1
    idx = np.argsort(corr)[-k:]
    w = np.exp((corr[idx]-corr[idx[-1]])/temp)
    w /= w.sum()
    grad = np.zeros(N)
    for m, wt in zip(idx, w):
        M = abs(int(m)-center)
        if M == 0:
            grad += wt*(1.0-2.0*h)
        elif M < N:
            hi_i = np.arange(M, N)
            lo_i = np.arange(0, N-M)
            grad[hi_i] += wt*(1.0-h[hi_i-M])
            grad[lo_i] -= wt*h[lo_i+M]
    return grad

def cont_opt(h, deadline, k=50, lr0=0.02, mom=0.85, temp=2.0):
    N = len(h)
    tgt = N/2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    lr = lr0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr))*(2.0/N)
        if c < best_c - 1e-10:
            best_c, best_h, stale = c, h.copy(), 0
        else:
            stale += 1
        if stale > 60:
            lr *= 0.85
            stale = 0
        g = soft_grad(h, corr, k, temp)
        h_new = project(h + mom*v - lr*g, tgt)
        v = 0.85*v + 0.15*(h_new-h)
        h = h_new
    return best_h, best_c

    N = len(h)
    nh = N//2

    elif n1 < nh:
        idx = np.where(h==0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1
    return h

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def soft_grad(h, corr, k=50, temp=2.0):
    N = len(h)
    center = N - 1
    idx = np.argsort(corr)[-k:]
    w = np.exp((corr[idx] - corr[idx[-1]]) / temp)
    w /= w.sum()
    grad = np.zeros(N)
    for m, wt in zip(idx, w):
        M = abs(int(m) - center)
        if M == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif M < N:
            hi_i = np.arange(M, N)
            lo_i = np.arange(0, N - M)
            grad[hi_i] += wt * (1.0 - h[hi_i - M])
            grad[lo_i] -= wt * h[lo_i + M]
    return grad

def cont_opt(h, deadline, k=50, lr0=0.018, mom=0.87, temp=1.5):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    lr = lr0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c, best_h, stale = c, h.copy(), 0
        else:
            stale += 1
        if stale > 50:
            lr *= 0.85
            stale = 0
        g = soft_grad(h, corr, k, temp)
        h_new = project(h + mom * v - lr * g, tgt)
        v = 0.85 * v + 0.15 * (h_new - h)
        h = h_new
    return best_h, best_c

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_il(h, deadline, max_iters=5000, nlags=6, s_one=90, s_zero=65):
    N = len(h)

    best_c = compute_c5(h)

    no_imp = 0

        if c < best_c - 1e-12:
            best_c, best_h, no_imp = c, h.copy(), 0
        else:
            no_imp += 1
        if no_imp > 500:
