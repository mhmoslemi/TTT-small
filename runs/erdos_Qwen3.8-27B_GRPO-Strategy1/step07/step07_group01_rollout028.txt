def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = h[pm_c] * vm.astype(float)
    hp = h[pp_c] * vp.astype(float)
    D = hp - (1.0 - hm) * vm.astype(float)
    return D

def binary_greedy(h_init, N, deadline, rng, max_sweeps=800, sample_size=None):
    h = h_init.copy()
    c = fft_corr(h)
    P = float(c.max())
    best_h = h.copy()
    best_c = P
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 25:
            break
        D = build_D(h)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ss = sample_size if sample_size else len(ones)
        ss = min(ss, len(ones))
        p_list = rng.choice(ones, ss, replace=False)
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
        P = float(c.max())
    return best_h, best_c * (2.0 / N)

def binary_sa(h_init, N, deadline, rng, T0=4.0, T1=0.02, cooling=0.9997, max_swaps=3):
    h = h_init.copy()
    c = fft_corr(h)
    P = float(c.max())
    best_h = h.copy()
    best_P = P
    T = T0
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        r = rng.random()
        n_sw = 1 if r < 0.45 else (2 if r < 0.82 else max_swaps)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < n_sw or len(zeros) < n_sw:
            break
        p_sw = rng.choice(ones, n_sw, replace=False)
        q_sw = rng.choice(zeros, n_sw, replace=False)
        h[p_sw] = 0.0
        h[q_sw] = 1.0
        new_c = fft_corr(h)
        new_P = float(new_c.max())
        delta = new_P - P
        accept = delta <= 0 or (delta < T * 5 and rng.random() < np.exp(-delta / max(T, 1e-10)))
        if accept:
            c = new_c
            P = new_P
            if P < best_P:
                best_P = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[p_sw] = 1.0
            h[q_sw] = 0.0
            stall += 1
        T *= cooling
        if T < T1:
            T = T1
        if stall > 3000 and it > 10000:
            n_r = min(5, N // 200)
            for _ in range(n_r):
                p = int(rng.randint(N))
                if h[p] == 1:
                    q = int(rng.choice(zeros))
                    h[p], h[q] = 0.0, 1.0
                else:
                    q = int(rng.choice(ones))
                    h[p], h[q] = 1.0, 0.0
            c = fft_corr(h)
            P = float(c.max())
            T = max(T, 1.5)
            stall = 0
    return best_h, best_P * (2.0 / N)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=45, T=70.0, lr0=0.018):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 50 == 0:
            lr *= 0.994
            beta = min(0.98, 0.84 + it * 0.00025)
            T = max(20.0, min(300.0, T * (0.96 if it % 200 == 0 else 1.0)))
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0, rng)
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)

    # Phase 1: Quick greedy convergence
    d1 = min(deadline - 800, t0 + 80)
    h1, c5_1 = binary_greedy(h_bin, N0, d1, rng, max_sweeps=600, sample_size=200)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: SA with multi-swap (exploration)
    d2 = min(deadline - 500, t0 + 320)
    h2, c5_2 = binary_sa(best_h, N0, d2, rng, T0=5.0, T1=0.03, cooling=0.9996, max_swaps=3)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Greedy polish after SA
    if time.time() < deadline - 350:
        d3 = min(deadline - 300, time.time() + 80)
        h3, c5_3 = binary_greedy(best_h, N0, d3, rng, max_sweeps=500, sample_size=300)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 4: Second SA from different perturbation
    if time.time() < deadline - 280:
        h_pert = best_h.copy()
        for _ in range(15):
            p = int(rng.randint(N0))
            o_i = np.where(h_pert == 1)[0]
            z_i = np.where(h_pert == 0)[0]
            if h_pert[p] == 1 and len(z_i) > 0:
                q = int(rng.choice(z_i))
                h_pert[p], h_pert[q] = 0.0, 1.0
            elif h_pert[p] == 0 and len(o_i) > 0:
                q = int(rng.choice(o_i))
                h_pert[p], h_pert[q] = 1.0, 0.0
        d4 = min(deadline - 220, time.time() + 80)
        h4s, c5_4s = binary_sa(h_pert, N0, d4, rng, T0=4.0, T1=0.02, cooling=0.9997, max_swaps=2)
        if c5_4s < best_c5:
            best_c5, best_h = c5_4s, h4s
        if c5_4s < best_c5 or time.time() < deadline - 200:
            d4b = min(deadline - 180, time.time() + 40)
            h4g, c5_4g = binary_greedy(best_h, N0, d4b, rng, max_sweeps=300, sample_size=250)
            if c5_4g < best_c5:
                best_c5, best_h = c5_4g, h4g

    # Phase 5: Continuous optimization
    if time.time() < deadline - 150:
        hc = best_h.astype(float) + rng.randn(N0) * 0.025
        hc = project_sum(hc, N0 / 2.0)
        d5 = min(deadline - 110, time.time() + 80)
        h5, c5_5 = nesterov_opt(hc, N0, d5, k=50, T=90.0, lr0=0.014)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5
        h5b = make_binary(h5, N0, rng) if c5_5 < best_c5 else make_binary(best_h, N0, rng)
        if time.time() < deadline - 90:
            d5b = min(deadline - 70, time.time() + 40)
            h5b_r, c5_5b = binary_greedy(h5b, N0, d5b, rng, max_sweeps=200, sample_size=200)
            if c5_5b < best_c5:
                best_c5, best_h = c5_5b, h5b_r

    # Phase 6: Try N=570 (downsample)
    N6 = N0 // 2
    if N6 >= 200 and time.time() < deadline - 100:
        h_half = (best_h[:N6] + best_h[N6:]) / 2.0
        h_half = project_sum(h_half, N6 / 2.0)
        h6_bin = make_binary(h_half, N6, rng)
        d6a = min(deadline - 70, time.time() + 35)
        h6a, c5_6a = binary_greedy(h6_bin, N6, d6a, rng, max_sweeps=400, sample_size=150)
        if c5_6a < best_c5:
            best_c5, best_h = c5_6a, h6a
        if time.time() < deadline - 40:
            d6b = min(deadline - 25, time.time() + 30)
            h6b, c5_6b = binary_sa(h6a, N6, d6b, rng, T0=2.5, T1=0.02, cooling=0.9997, max_swaps=3)
            if c5_6b < best_c5:
                best_c5, best_h = c5_6b, h6b

    # Phase 7: Final greedy + micro SA
    N_fin = len(best_h)
    if time.time() < deadline - 15:
        d7 = min(deadline - 5, time.time() + 12)
        h7, c5_7 = binary_greedy(best_h, N_fin, d7, rng, max_sweeps=200, sample_size=None)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7
    if time.time() < deadline - 2:
        d8 = min(deadline - 1, time.time() + 8)
        h8, c5_8 = binary_sa(best_h, N_fin, d8, rng, T0=1.0, T1=0.01, cooling=0.999, max_swaps=2)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
