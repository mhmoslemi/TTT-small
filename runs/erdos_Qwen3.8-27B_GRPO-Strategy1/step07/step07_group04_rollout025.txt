        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def grad_lsmax(h, corr, beta, top_k=80):
    """Gradient of log-sum-exp smooth max."""
    N = len(h)
    max_c = corr.max()
    w = np.exp(beta * (corr - max_c))
    if len(w) > top_k:
        idx = np.argpartition(w, -top_k)[-top_k:]
    else:
        idx = np.arange(len(w))
    w_top = w[idx]
    sw = w_top.sum()
    if sw < 1e-300:
        return np.zeros(N)
    w_top /= sw
    grad = np.zeros(N)
    center = N - 1
    for t in range(len(idx)):
        wk = w_top[t]
        L = int(idx[t] - center)
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            grad[:N - M] -= wk * h[M:]
            grad[M:] += wk * (1.0 - h[:N - M])
    return grad

def smooth_opt(h, N, deadline, beta_end=60.0, lr0=0.04, top_k=70):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    n_it = 0
    lr = lr0
    while time.time() < deadline:
        beta = min(beta_end, 3.0 + 0.15 * n_it)
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = grad_lsmax(h, corr, beta, top_k)
        h = project_sum(h - lr * grad, target)
        if n_it % 200 == 0:
            lr *= 0.97
        if n_it > 300 and n_it % 100 == 0 and np.random.random() < 0.15:
            h = project_sum(h + np.random.randn(N) * 0.03, target)
        n_it += 1
    return best_h, best_pk * dx

def binary_sa(h, N, deadline, T0=5.0, Tcool=0.993, T_min=0.03, sf=1.0, kick_every=120, kick_size=5, max_sweeps=10000):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            c = fft_corr(h)
            P = float(c.max())
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def greedy_binary(h, N, deadline, max_sweeps=2000, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)
