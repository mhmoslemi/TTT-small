    # Phase 1: Primary search with full-search every 30 iters
    d1 = min(deadline - 480, t0 + 420)
    hs, c5s = search(h0, N0, d1, sz=150, full_every=30, kick_thresh=50, kick_sz=4)
    if c5s < best_c5:
        best_c5, best_h = c5s, hs.copy()

    # Phase 2: Second search, larger samples, less frequent full search
    if time.time() < deadline - 300:
        d2 = min(deadline - 280, time.time() + 200)
        h2 = (best_h > 0.5).astype(float)
        ones2 = np.where(h2 == 1)[0]
        zeros2 = np.where(h2 == 0)[0]
        for _ in range(6):
            h2[np.random.choice(ones2)] = 0
            h2[np.random.choice(zeros2)] = 1
            ones2 = np.where(h2 == 1)[0]
            zeros2 = np.where(h2 == 0)[0]
        hs2, c5s2 = search(h2, N0, d2, sz=N0 // 2, full_every=15, kick_thresh=40, kick_sz=5)
        if c5s2 < best_c5:
            best_c5, best_h = c5s2, hs2.copy()

    # Phase 3: Try N=370
    if time.time() < deadline - 180:
        d3 = min(deadline - 160, time.time() + 90)
        h3 = interp_binarize(best_h, 370)
        hs3, c5s3 = search(h3, 370, d3, sz=120, full_every=25, kick_thresh=35, kick_sz=4)
        if c5s3 < best_c5:
            best_c5, best_h = c5s3, hs3.copy()

    # Phase 4: Try N=390
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 70)
        h4 = interp_binarize(best_h, 390)
        hs4, c5s4 = search(h4, 390, d4, sz=130, full_every=25, kick_thresh=35, kick_sz=4)
        if c5s4 < best_c5:
            best_c5, best_h = c5s4, hs4.copy()

    # Phase 5: Try N=400
    if time.time() < deadline - 80:
        d5 = min(deadline - 60, time.time() + 60)
        h5 = interp_binarize(best_h, 400)
        hs5, c5s5 = search(h5, 400, d5, sz=140, full_every=25, kick_thresh=35, kick_sz=4)
        if c5s5 < best_c5:
            best_c5, best_h = c5s5, hs5.copy()

    # Phase 6: Final intensive search on best N
    NB = len(best_h)
    if time.time() < deadline - 10:
        d6 = deadline - 5
        h6 = (best_h > 0.5).astype(float)
        hs6, c5s6 = search(h6, NB, d6, sz=NB // 2, full_every=10, kick_thresh=30, kick_sz=3)
        if c5s6 < best_c5:
            best_c5, best_h = c5s6, hs6.copy()

    # Final: check binary rounding
    hB = (best_h > 0.5).astype(float)
    fix_bin(hB, len(best_h) // 2)
    c5B = compute_c5(hB, len(best_h))
    if c5B < best_c5:
        best_c5, best_h = c5B, hB

    return best_h, best_c5, len(best_h)
