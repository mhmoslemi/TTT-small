def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(12345)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(12345)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_sa_search(h_init, N, deadline, T0=0.002, k_top=60, max_sweeps=100000):
    h = h_init.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    dx = 2.0 / N
    best_c5 = P
    best_h = h.copy()
    T = T0
    stall = 0
    rng = np.random.RandomState(42 + int(N))
    
    # Precompute structure
    center = N - 1
    p_all = np.arange(N)[:, None]
    s_all = np.arange(2*N-1) - center
    s_all = s_all[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Recompute D
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nz = len(zeros)
        no = len(ones)
        
        if no < 1 or nz < 1:
            break
        
        # Top-k
        kk = min(k_top, 2*N-1)
        top_k = np.argpartition(-c, kk)[:kk]
        ck = c[top_k]
        D_k = D[:, top_k]  # (N, kk)
        Dz_k = D_k[zeros]  # (nz, kk)
        
        # Find best 1-swap using top-k
        best_np = P + 0.001
        best_pq = None
        # Track top few for compound moves
        top_moves = []
        
        for p in ones:
            base_k = ck + D_k[p]  # (kk,)
            cand_k = base_k[None, :] - Dz_k  # (nz, kk)
            npk = cand_k.max(axis=1)  # (nz,)
            mi = int(np.argmin(npk))
            val = float(npk[mi])
            if val < best_np:
                if best_pq is not None:
                    top_moves.append((best_np, best_pq))
                best_np = val
                best_pq = (int(p), int(zeros[mi]))
                if len(top_moves) == 5:
                    top_moves = top_moves[-5:]
        
        # Try the best move
        accepted = False
        if best_pq is not None:
            p, q = best_pq
            h[p] = 0.0
            h[q] = 1.0
            c = fft_corr(h)
            P_new = float(c.max())
            if P_new < P:
                P = P_new
                if P < best_c5:
                    best_c5 = P
                    best_h = h.copy()
                stall = 0
                accepted = True
            elif rng.random() < np.exp(-((P_new - P) * N / 2) / T):
                # SA accept (scale by N to make T meaningful)
                P = P_new
                stall += 1
                accepted = True
            else:
                # Reject
                h[p] = 1.0
                h[q] = 0.0
                stall += 1
        
        # Try compound 2-swap from top moves
        if not accepted and len(top_moves) >= 1:
            # Try applying best_pq + second best
            if best_pq is not None:
                p1, q1 = best_pq
                for np2, (p2_, q2_) in top_moves:
                    if p2_ == p1 and q2_ == q1:
                        continue
                    h[p1] = 0.0
                    h[q1] = 1.0
                    h[p2_] = 0.0
                    h[q2_] = 1.0
                    c2 = fft_corr(h)
                    P2 = float(c2.max())
                    if P2 < P:
                        P = P2
                        c = c2
                        if P < best_c5:
                            best_c5 = P
                            best_h = h.copy()
                        stall = 0
                        accepted = True
                        break
                    else:
                        # Undo
                        h[p1] = 1.0
                        h[q1] = 0.0
                        h[p2_] = 1.0
                        h[q2_] = 0.0
        
        if not accepted:
            stall += 1
        
        # Adaptation
        if stall > 25:
            T = min(T0, T * 1.1)
            n_r = min(5, 1 + stall // 25)
            for _ in range(n_r):
                p = int(ones[rng.randint(no)])
                q = int(zeros[rng.randint(nz)])
                h[p], h[q] = 0.0, 1.0
            stall = 0
            c = fft_corr(h)
            P = float(c.max())
        
        if sweep % 100 == 99:
            T = max(0.00001, T * 0.97)
    
    return best_h, best_c5 * dx

def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        # Soft-max gradient
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N-s])
                grad[:N-s] -= wi * h[s:]
            else:
                M = -s
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.997
            beta = min(0.98, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

        h3, c5_3 = binary_sa_search(h_b3, N0, d3, T0=0.0015, k_top=80)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
    # Phase 4: Second continuous pass with different params
    if time.time() < deadline - 100:
        hc4 = best_h.astype(float).copy() + np.random.randn(N0) * 0.015
        hc4 = project_sum(hc4, N0 / 2.0)
        d4 = min(time.time() + 70, deadline - 50)
        h4, c5_4 = nesterov_opt(hc4, N0, d4, k=50, T=180.0, lr0=0.007)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
    # Phase 5: Final binary polish with sharper top-k
    if time.time() < deadline - 15:
        hF = ensure_binary(best_h, N0)
        d5 = min(time.time() + 45, deadline - 3)
        h5, c5_5 = binary_sa_search(hF, N0, d5, T0=0.0005, k_top=120)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
