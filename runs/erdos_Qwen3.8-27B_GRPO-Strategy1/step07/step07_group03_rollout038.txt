    return np.real(np.fft.ifft(H * G))

def compute_D(h, N):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    target_n = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target_n:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target_n:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - target_n, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], target_n - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_anneal(h, N, deadline, max_sweeps=500, do_2swap=True, do_3swap=False, rng=None):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    if rng is None:
        rng = np.random.RandomState(7)
    c_cur = fft_corr(h)
    best_c = float(c_cur.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        D = compute_D(h, N)
        cur_P = float(c_cur.max())
        best_new_P = cur_P
        best_action = None  # (type, indices)
        
        # 1-swap
        D_z = D[zeros, :]
        for p in ones:
            base = c_cur + D[p, :]
            cands = base[None, :] - D_z
            maxes = cands.max(axis=1)
            mi = int(np.argmin(maxes))
            if maxes[mi] < best_new_P:
                best_new_P = float(maxes[mi])
                best_action = ('1', int(p), int(zeros[mi]))
        
        # 2-swap
        if do_2swap and len(ones) >= 4:
            n_try = min(60, N // 6)
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:

                q1, q2 = int(zeros[j1]), int(zeros[j2])

                if np_ < best_new_P:
                    best_new_P = np_
                    best_action = ('2', p1, p2, q1, q2)
        
        # 3-swap
        if do_3swap and len(ones) >= 6:

                h[best_action[2]] = 1.0

    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=60, lr0=0.015, beta0=0.9):
    h = np.clip(h.astype(float), 0, 1).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        top = np.argpartition(corr_y, -k)[-k:]
        
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N-L] += (1.0 - y[L:])
                grad[L:] -= y[:N-L]
            else:
                M_ = -L
                if M_ < N:
                    grad[:N-M_] += (1.0 - y[M_:])
                    grad[M_:] -= y[:N-M_]
        grad /= k
        
        h_new = project_sum(h - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        
        if it % 50 == 0:
            corr_h = fft_corr(h)
            pk = float(corr_h.max())
            if pk < best_pk - 1e-12:
                best_pk = pk
                best_h = h.copy()
        
        if it % 200 == 0:
            lr *= 0.96
            beta = min(0.97, beta0 + it * 0.0002)
    
    # Final check
    corr_final = fft_corr(h)
    if float(corr_final.max()) < best_pk:
        best_pk = float(corr_final.max())
        best_h = h.copy()
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    target_n = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target_n:
        return h_bin
    rng = np.random.RandomState(777)
    if n1 > target_n:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - target_n, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], target_n - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_anneal(h, N, deadline, max_sweeps=500, do_2sw=True, do_3sw=False, rng=None):

    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    if rng is None:
        rng = np.random.RandomState(7)
    c_cur = fft_corr(h)
    best_c = float(c_cur.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        D = compute_D(h, N)
        cur_P = float(c_cur.max())
        best_new_P = cur_P
        best_act = None
        D_z = D[zeros, :]
        for p in ones:
            base = c_cur + D[p, :]
            maxes = (base[None, :] - D_z).max(axis=1)
            mi = int(np.argmin(maxes))
            if maxes[mi] < best_new_P:
                best_new_P = float(maxes[mi])
                best_act = (1, int(p), int(zeros[mi]))
        if do_2sw and len(ones) >= 4:

                if nv < best_new_P:

            n_try = min(30, N // 10)

                ia = rng.choice(len(ones), 3, replace=False)
                ja = rng.choice(len(zeros), 3, replace=False)
                ps = tuple(int(ones[x]) for x in ia)
                qs = tuple(int(zeros[x]) for x in ja)
                nv = float((c_cur + D[ps[0]] + D[ps[1]] + D[ps[2]] - D[qs[0]] - D[qs[1]] - D[qs[2]]).max())

def nesterov_opt(h, N, deadline, k=70, lr0=0.018, beta0=0.92):
    h = np.clip(h.astype(float), 0, 1).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        top = np.argpartition(corr_y, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N-L] += (1.0 - y[L:])
                grad[L:] -= y[:N-L]
            else:
                M_ = -L
                if M_ < N:
                    grad[:N-M_] += (1.0 - y[M_:])
                    grad[M_:] -= y[:N-M_]
        grad /= k
        h_new = project_sum(h - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 40 == 0:
            pk = float(np.max(fft_corr(h)))
            if pk < best_pk - 1e-13:
                best_pk = pk; best_h = h.copy()
        if it % 150 == 0:
            lr *= 0.96
            beta = min(0.97, beta0 + it * 0.0002)
    pk = float(np.max(fft_corr(h)))
    if pk < best_pk:
        best_pk = pk; best_h = h.copy()
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary steepest descent + 2-swap at N0
    h_bin = ensure_sum(h0, N0)
    d1 = t0 + budget_s * 0.26
    h1, c5_1 = binary_anneal(h_bin, N0, d1, max_sweeps=700, do_2sw=True, do_3sw=False, rng=rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Nesterov continuous at N0
    if time.time() < deadline - 260:
        hc = best_h + rng.randn(N0) * 0.055
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 220, time.time() + budget_s * 0.11)
        h2, c5_2 = nesterov_opt(hc, N0, d2, k=80, lr0=0.022, beta0=0.93)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Binary refine at N0 after continuous (with 3-swap)
    if time.time() < deadline - 180:
        h_t = ensure_sum(best_h, N0)
        d3 = min(deadline - 140, time.time() + budget_s * 0.13)
        h3, c5_3 = binary_anneal(h_t, N0, d3, max_sweeps=500, do_2sw=True, do_3sw=True, rng=rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upscale to 2*N0, binary anneal
    N1 = 2 * N0
    if time.time() < deadline - 130:
        h_up = upsample(best_h, len(best_h), N1)
        h_up_b = ensure_sum(h_up, N1)
        d4 = min(deadline - 90, time.time() + budget_s * 0.12)
        h4, c5_4 = binary_anneal(h_up_b, N1, d4, max_sweeps=180, do_2sw=True, do_3sw=False, rng=rng)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Nesterov at current resolution
    if time.time() < deadline - 70:
        Nc = len(best_h)
        hc2 = best_h + rng.randn(Nc) * 0.035
        hc2 = project_sum(hc2, Nc / 2.0)
        d5 = min(deadline - 40, time.time() + budget_s * 0.06)
        h5, c5_5 = nesterov_opt(hc2, Nc, d5, k=60, lr0=0.012, beta0=0.9)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Restart with perturbation at N0
    if time.time() < deadline - 30:
        rng2 = np.random.RandomState(seed + 314)
        h_r = ensure_sum(best_h[:N0] if len(best_h) >= N0 else upsample(best_h, len(best_h), N0), N0)
        taken = set()
        while len(taken) < 6:
            idx = rng2.randint(N0)
            if idx not in taken:
                taken.add(idx)
        for idx in taken:
            h_r[idx] = 1.0 - h_r[idx]
        h_r = ensure_sum(h_r, N0)
        d6 = min(deadline - 10, time.time() + budget_s * 0.04)
        h6, c5_6 = binary_anneal(h_r, N0, d6, max_sweeps=200, do_2sw=True, do_3sw=True, rng=rng2)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final binary polish
    N_fin = len(best_h)
    hF = ensure_sum(best_h, N_fin)
    d7 = deadline - 1
    hF2, c5F = binary_anneal(hF, N_fin, d7, max_sweeps=120, do_2sw=True, do_3sw=False, rng=rng)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
