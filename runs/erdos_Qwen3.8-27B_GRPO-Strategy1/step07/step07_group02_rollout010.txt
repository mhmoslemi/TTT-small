def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def compute_C_mat(h, N):
    C = np.zeros((N, 2 * N - 1))
    for i in range(N):
        C[i] = compute_C_single(h, N, i)
    return C

def fix_sum_binary(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[:n1 - nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[:nh - n1]]] = 1
    return h

def opt_binary(h, N, deadline, sample_sz=200, kick_thresh=80, kick_sz=5, C_rescan=60):
    h = h.astype(float).copy()
    fix_sum_binary(h, N // 2)
    C = compute_C_mat(h, N)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    stuck = 0
    it = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if it % C_rescan == 0 and it > 0:
            C = compute_C_mat(h, N)
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        newP = np.zeros((ns, nz))
        for a in range(ns):
            newP[a] = np.max((corr + Co[a])[None, :] + Cz, axis=1)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = newP[mi]
        if mv < P - 1e-14:
            p = int(os_[mi[1]])
            q = int(zs[mi[0]])
            h[p] = 0
            h[q] = 1
            C[p] = compute_C_single(h, N, p)
            C[q] = compute_C_single(h, N, q)
            corr = fft_corr(h)
            P = float(np.max(corr))
            stuck = 0
            if P < best_c:
                best_c = P
                best_h = h.copy()
        else:
            stuck += 1
            if stuck >= kick_thresh:
                for _ in range(kick_sz):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0
                    h[q] = 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                C = compute_C_mat(h, N)
                corr = fft_corr(h)
                P = float(np.max(corr))
                stuck = 0
        it += 1
    return best_h, best_c

def full_scan_pass(h, N, C, corr, P, deadline):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_imp = 0
    order = np.random.permutation(len(ones))
    for oi in order:
        if time.time() > deadline:
            break
        p = int(ones[oi])
        D = corr + C[p]
        nm = np.max(D[None, :] + C[zeros], axis=1)
        bi = int(np.argmin(nm))
        if nm[bi] < P - 1e-14:
            q = int(zeros[bi])
            h[p] = 0
            h[q] = 1
            C = compute_C_mat(h, N)
            corr = fft_corr(h)
            P = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_imp += 1
    return h, C, corr, P, n_imp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 > 0.5).astype(float)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Initial binary optimization (150s)
    d1 = min(deadline - 500, t0 + 150)
    h1, _ = opt_binary(h0, N0, d1, sample_sz=180, kick_thresh=60, kick_sz=5)
    c5_1 = compute_c5(h1, N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Multi-resolution exploration (200s)
    if time.time() < deadline - 350:
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        N_list = [360, 370, 390, 400, 410, 420, 440, 460]
        for N_try in N_list:
            if time.time() > deadline - 350:
                break
            x_new = np.linspace(0, 2, N_try, endpoint=False)
            h_int = np.interp(x_new, x_old, best_h)
            h_bt = (h_int > 0.5).astype(float)
            fix_sum_binary(h_bt, N_try // 2)
            sub_dl = min(deadline - 350, time.time() + 22)
            h_rt, _ = opt_binary(h_bt, N_try, sub_dl, sample_sz=min(N_try//2, 150), kick_thresh=50)
            c5_rt = compute_c5(h_rt, N_try)
            if c5_rt < best_c5:
                best_c5, best_h = c5_rt, h_rt
                x_old = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 3: Multi-start diverse perturbations (300s)
    if time.time() < deadline - 100:
        N_best = len(best_h)
        h_base = (best_h > 0.5).astype(float)
        n_restarts = 10
        for r in range(n_restarts):
            if time.time() > deadline - 100:
                break
            sub_dl = min(deadline - 100, time.time() + 28)
            h_rs = h_base.copy()
            if r == 0:
                pass
            elif r == 1:
                h_rs = np.roll(h_base, 3).copy()
                fix_sum_binary(h_rs, N_best // 2)
            elif r == 2:
                h_rs = np.roll(h_base, -4).copy()
                fix_sum_binary(h_rs, N_best // 2)
            elif r == 3:
                mid = N_best // 2
                blk = max(3, N_best // 12)
                h_rs[mid - blk:mid + blk] = 1 - h_base[mid - blk:mid + blk]
                fix_sum_binary(h_rs, N_best // 2)
            elif r == 4:
                third = N_best // 3
                h_rs[third:2*third] = 1 - h_base[third:2*third]
                fix_sum_binary(h_rs, N_best // 2)
            else:
                s = 3 + r * 4
                ones_r = np.where(h_rs == 1)[0]
                zeros_r = np.where(h_rs == 0)[0]
                for _ in range(s):
                    p = int(np.random.choice(ones_r))
                    q = int(np.random.choice(zeros_r))
                    h_rs[p] = 0
                    h_rs[q] = 1
                    ones_r = np.where(h_rs == 1)[0]
                    zeros_r = np.where(h_rs == 0)[0]
            h_rs, _ = opt_binary(h_rs, N_best, sub_dl, sample_sz=190, kick_thresh=50, kick_sz=4)
            c5_rs = compute_c5(h_rs, N_best)
            if c5_rs < best_c5:
                best_c5, best_h = c5_rs, h_rs

    # Phase 4: Long intensive optimization with full enumeration (remaining)
    N_f = len(best_h)
    h_f = (best_h > 0.5).astype(float)
    h_f, _ = opt_binary(h_f, N_f, deadline - 5,
                        sample_sz=N_f // 2, kick_thresh=70, kick_sz=5, C_rescan=40)
    c5_f = compute_c5(h_f, N_f)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f

    # Final binary threshold check
    final_c5 = compute_c5(best_h, len(best_h))
    best_bin = (best_h > 0.5).astype(float)
    final_c5_bin = compute_c5(best_bin, len(best_bin))
    if final_c5_bin < final_c5:
        best_h = best_bin
        final_c5 = final_c5_bin

    return best_h, final_c5, len(best_h)
