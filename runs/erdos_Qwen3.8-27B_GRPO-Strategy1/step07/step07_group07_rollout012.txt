import numpy as np

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_corr_top(h, corr, k=50):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def soft_max_opt(h, deadline, k=50, lr0=0.015, mom=0.88, beta=50):

    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()

            stale = 0
        if lr < 1e-6:
            break
        
        # Soft max gradient
        max_c = np.max(corr)
        exp_c = np.exp(beta * (corr - max_c))
        weights = exp_c / exp_c.sum()
        
        # Gradient of soft-max objective
        grad = np.zeros(N)
        center = N - 1
        for m in range(2 * N - 1):
            if weights[m] < 1e-4:
                continue
            M = abs(int(m) - center)
            w = weights[m]
            if M == 0:
                grad += w * (1.0 - 2.0 * h)
            elif M < N:
                hi_idx = np.arange(M, N)

                grad[hi_idx] += w * (1.0 - h[hi_idx - M])
                grad[lo_idx] -= w * h[lo_idx + M]
        
        h_new = project(h + mom * v - lr * grad, tgt)
        v = mom * v + (h_new - h)
        h = h_new
    
    return best_h, best_c

def compute_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

# Gradient for top-k lags
def grad_topk(h, corr, k=50):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

    # Final evaluation
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
