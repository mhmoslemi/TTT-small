                best_new_c = c
                best_flip = i
            h[i] = 1.0 - h[i]
        if best_flip >= 0:
            h[best_flip] = 1.0 - h[best_flip]
            best_c = best_new_c
        else:
            break
    return h, best_c

def focused_descent(h, N, deadline, max_rounds=300):
    h = (h > 0.5).astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        mx = float(corr.max())
        if mx < best_c - 1e-14:
            best_c = mx
            best_h = h.copy()
        tol = 0.08
        active = np.where(corr >= mx - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        improved = False
        ctr = N - 1
        for ai in active[:8]:
            k = int(ai - ctr)
            if k >= 0:
                prod = h[k:] * (1.0 - h[:N - k])
                positions_h = np.arange(k, N) - k
            else:
                M = -k
                prod = h[:N - M] * (1.0 - h[M:])
                positions_h = np.arange(N - M)
            if len(prod) == 0:
                continue
            top_n = min(4, len(prod))
            top_idx = np.argsort(prod)[-top_n:]
            for ti in top_idx:
                pos = int(positions_h[ti])
                h[pos] = 1.0 - h[pos]
                nm = float(np.max(fft_corr(h)))
                if nm < mx - 1e-12:
                    mx = nm
                    improved = True
                    if mx < best_c - 1e-14:
                        best_c = mx
                        best_h = h.copy()
                    break
                else:
                    h[pos] = 1.0 - h[pos]
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def try_resolutions(best_h, best_c5, deadline):
    x_b = np.linspace(0, 2, len(best_h), endpoint=False)
    resolutions = [200, 220, 240, 260, 280, 300, 320, 340, 360, 400, 440, 480, 520]
    for Nt in resolutions:
        if time.time() > deadline:
            break
        if Nt % 2:
            Nt += 1
        if Nt == len(best_h):
            continue
        if time.time() > deadline - 40:
            break
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_b, best_h)
        ht = (hi > 0.5).astype(float)
        nh = Nt // 2
        n1 = int(ht.sum())
        if n1 > nh:
            ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0.0
        elif n1 < nh:
            ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1.0
        sd = min(deadline, time.time() + 40)
        hr, _ = bin_search(ht, Nt, sd, max_iters=300, ntri=22)
        hb, _ = block_swap_search(hr, Nt, min(deadline, time.time() + 25), max_iters=200, ntri=18)
        hr, _ = single_bit_hill(hb, Nt, min(deadline, time.time() + 15), max_rounds=6)
        c5r = float(np.max(fft_corr(hr))) * (2.0 / Nt)
        if c5r < best_c5 - 1e-6:
            best_c5 = c5r
            best_h = hr.copy()
            x_b = np.linspace(0, 2, Nt, endpoint=False)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(fft_corr(best_h))) * (2.0 / N0)

    # Phase 1: binary search
    d1 = min(dl - 500, t0 + 60)
    hb, _ = bin_search(best_h, N0, d1, max_iters=400, ntri=25, anneal=0.0005)
    c = float(np.max(fft_corr(hb))) * (2.0 / N0)
    if c < best_c5:
        best_c5, best_h = c, hb.copy()

    # Phase 2: L-BFGS smoothed (multiple restarts)
    if time.time() < dl - 400:
        d2 = min(dl - 350, time.time() + 180)
        for r in range(5):
            if time.time() > d2:
                break
            noise = 0.20 - 0.03 * r
            hr = best_h + np.random.randn(N0) * noise
            hr = proj01(hr, N0 / 2.0)
            T_val = 25.0 + 10.0 * r
            hl, cl = lbfgs_smooth(hr, N0, min(d2, time.time() + 35), T=T_val, max_outer=4)
            c5l = float(np.max(fft_corr(hl))) * (2.0 / N0)
            if c5l < best_c5:
                best_c5, best_h = c5l, hl.copy()

    # Phase 3: block swap search
    if time.time() < dl - 300:
        d3 = min(dl - 260, time.time() + 100)
        hb3, _ = block_swap_search(best_h, N0, d3, max_iters=500, ntri=22)
        c3 = float(np.max(fft_corr(hb3))) * (2.0 / N0)
        if c3 < best_c5:
            best_c5, best_h = c3, hb3.copy()

    # Phase 4: focused descent on active lags
    if time.time() < dl - 220:
        d4 = min(dl - 180, time.time() + 80)
        hf4, cf4 = focused_descent(best_h, N0, d4, max_rounds=400)
        c4 = float(np.max(fft_corr(hf4))) * (2.0 / N0)
        if c4 < best_c5:
            best_c5, best_h = c4, hf4.copy()

    # Phase 5: binary polish
    if time.time() < dl - 160:
        d5 = min(dl - 120, time.time() + 80)
        hb5, _ = bin_search(best_h, N0, d5, max_iters=500, ntri=30, anneal=0.0002)
        c5v = float(np.max(fft_corr(hb5))) * (2.0 / N0)
        if c5v < best_c5:
            best_c5, best_h = c5v, hb5.copy()

    # Phase 6: multi-resolution
    if time.time() < dl - 100:
        best_h, best_c5 = try_resolutions(best_h, best_c5, min(dl - 50, time.time() + 200))
        N0 = len(best_h)

    # Phase 7: single-bit hill climb
    if time.time() < dl - 40:
        d7 = min(dl - 20, time.time() + 30)
        hs7, cs7 = single_bit_hill(best_h, N0, d7, max_rounds=20)
        c7 = float(np.max(fft_corr(hs7))) * (2.0 / N0)
        if c7 < best_c5:
            best_c5, best_h = c7, hs7.copy()

    # Phase 8: final focused descent + binary
    if time.time() < dl - 20:
        d8a = min(dl - 10, time.time() + 15)
        hf8, _ = focused_descent(best_h, N0, d8a, max_rounds=200)
        c8 = float(np.max(fft_corr(hf8))) * (2.0 / N0)
        if c8 < best_c5:
            best_c5, best_h = c8, hf8.copy()
    if time.time() < dl - 5:
        d8b = min(dl - 2, time.time() + 10)
        hb8, _ = bin_search(best_h, N0, d8b, max_iters=200, ntri=20)
        c8b = float(np.max(fft_corr(hb8))) * (2.0 / N0)
        if c8b < best_c5:
            best_c5, best_h = c8b, hb8.copy()

    # Final: ensure binary, compute with np.correlate
    best_h = (best_h > 0.5).astype(float)
    nh = len(best_h) // 2
    n1 = int(best_h.sum())
    if n1 > nh:
        best_h[np.random.choice(np.where(best_h == 1)[0], n1 - nh, replace=False)] = 0.0
    elif n1 < nh:
        best_h[np.random.choice(np.where(best_h == 0)[0], nh - n1, replace=False)] = 1.0
    fc5 = eval_c5(best_h)
    return best_h, fc5, len(best_h)
