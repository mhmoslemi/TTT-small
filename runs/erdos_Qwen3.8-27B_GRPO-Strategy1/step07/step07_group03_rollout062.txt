def compute_deriv(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    # For each shift k, D[:, k] is:
    # (1-h[i+k]) for valid i+k, minus h[i-k] for valid i-k
    # This is a shifted version of (1-h) minus a shifted version of h
    D = np.zeros((N, 2*N-1))
    for idx, k in enumerate(shifts):
        # (1-h[i+k]): valid for i in [max(0,-k), min(N, N-k))
        if k >= 0:
            # i+k ranges from k to N-1, so i from 0 to N-1-k
            D[:N-k, idx] = (1.0 - h[k:N])  # h[i+k] for i=0..N-1-k
        else:
            # i+k ranges from 0 to N-1+k, so i from -k to N-1
            D[-k:, idx] = (1.0 - h[:N+k])  # h[i+k] for i=-k..N-1, h[0..N-1+k]
        
        # h[i-k]: valid for i in [max(0,k), min(N, N+k))
        if k >= 0:
            # i-k ranges from 0 to N-1-k... wait
            # i-k >= 0 means i >= k. i-k < N means i < N+k. But i < N always.
            # So i from k to N-1.
            D[k:, idx] -= h[:N-k]  # h[i-k] for i=k..N-1, h[0..N-1-k]
        else:
            # i-k >= 0 means i >= k (negative, always true for i>=0). i-k < N means i < N+k.
            # So i from 0 to N-1+k.
            D[:N+k, idx] -= h[-k:]  # h[i-k] for i=0..N-1+k, h[-k..N-1]
