    return np.real(np.fft.ifft(H * G))

def softmax_grad(h, N, top_k=50, beta=200.0):
    """Compute gradient of log-sum-exp of top shifts."""
    corr = fft_corr(h)
    center = N - 1
    # Get top k shifts
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[top_idx]
    # Softmax weights
    top_max = top_vals.max()
    weights = np.exp(beta * (top_vals - top_max))
    weights /= weights.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for ai, w in zip(top_idx, weights):
        L = int(ai - center)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += w * (1.0 - h[M:])
                grad[M:] -= w * h[:N - M]
    return grad, float(top_vals.max())

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(99 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def nesterov_focus(h, N, deadline, top_k=40, beta=300.0, lr0=0.005, lr_decay=0.9995):
    """Nesterov with adaptive softmax focus on worst shifts."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta_mom = 0.82
    lr = lr0
    it = 0
    
    while time.time() < deadline:
        y = h + beta_mom * m
        corr_y = fft_corr(y)
        # Top-k selection
        idx = np.argpartition(corr_y, -top_k)[-top_k:]
        top_v = corr_y[idx]
        # Adaptive beta: increase when top values are close
        spread = top_v.max() - top_v.min()
        if spread < 0.1:
            beta = beta * 1.05
        elif spread > 2.0:
            beta = max(50.0, beta * 0.95)
        
        tmax = top_v.max()
        w = np.exp((top_v - tmax) * beta)
        w /= w.sum()
        
        grad = np.zeros(N)
        center = N - 1
        for ai, wi in zip(idx, w):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * h[:N - M]
        
        h_new = project_sum(y - lr * grad, target)
        m = beta_mom * (h_new - h)
        h = h_new
        it += 1
        
        if it % 10 == 0:
            c = float(fft_corr(h).max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
        
        lr *= lr_decay
        if it % 500 == 0:
            beta_mom = min(0.92, 0.80 + it * 0.00001)
    
    final_c = float(fft_corr(h).max())
    if final_c < best_c:
        best_c, best_h = final_c, h.copy()
    return best_h, best_c * dx

def binary_local(h, N, deadline, max_iter=3000, rng=None):
    """Fast binary local search with smart 2-swaps."""
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    if rng is None:
        rng = np.random.RandomState(42)
    
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_c = P_cur
    best_h = h.copy()
    stall = 0
    D = compute_D(h, N)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        best_new_P = P_cur
        best_mv = None
        
        # Try all single swaps (fast: precomputed D)
        D_z = D[zeros, :]
        # Sample a subset of ones for speed
        n_sample = min(len(ones), max(30, N // 6))
        p_idx = rng.choice(len(ones), n_sample, replace=False) if len(ones) > n_sample else np.arange(len(ones))
        
        for pi in p_idx:
            p = int(ones[pi])
            base = c_cur - D[p, :]
            cand_max = (base[None, :] - D_z).max(axis=1)
            mi = int(np.argmin(cand_max))
            if cand_max[mi] < best_new_P - 1e-14:
                best_new_P = float(cand_max[mi])
                best_mv = (p, int(zeros[mi]), 0)
        
        # 2-swaps: random but targeted
        n_2swap = 40
        for _ in range(n_2swap):
            i1, i2 = rng.randint(0, len(ones), 2)
            j1, j2 = rng.randint(0, len(zeros), 2)
            if i1 == i2 or j1 == j2:
                continue
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
            if new_P < best_new_P - 1e-14:
                best_new_P = new_P
                best_mv = (p1, p2, 1, q1, q2)
        
        if best_mv is not None:
            if best_mv[2] == 0:
                h[best_mv[0]] = 0.0
                h[best_mv[1]] = 1.0
            else:
                p1, p2, _, q1, q2 = best_mv
                h[p1] = h[p2] = 0.0
                h[q1] = h[q2] = 1.0
            c_cur = fft_corr(h)
            P_cur = float(c_cur.max())
            D = compute_D(h, N)
            if P_cur < best_c - 1e-13:
                best_c = P_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 50:
                # Random restart perturbation
                for _ in range(2):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                c_cur = fft_corr(h)
                P_cur = float(c_cur.max())
                D = compute_D(h, N)
                stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_h, best_c * (2.0 / N)

def upsample_h(h_old, N_old, N_new):

    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
