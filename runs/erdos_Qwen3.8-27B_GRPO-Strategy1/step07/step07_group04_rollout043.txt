```python
def compute_gradient(h, N, w):
    """w is the softmax weights over shifts, indexed 0..2N-2 (center at N-1)"""
    g = np.zeros(N)
    for k in range(-N+1, N):
        wk = w[k + N - 1]
        if abs(wk) < 1e-15:
            continue
        # shift k: c_k involves h[i+k] and h[i] for valid i
        if k >= 0:
            # i from 0 to N-1-k
            for i in range(N - k):
                # h[i+k] * (1-h[i])
                g[i+k] += wk * (1 - h[i])
                g[i] -= wk * h[i+k]
        else:
            kk = -k
            # i from kk to N-1
            for i in range(kk, N):
                g[i+k] += wk * (1 - h[i])
                g[i] -= wk * h[i+k]
    return g
```

```python
def compute_gradient_vec(h, N, w):
    g = np.zeros(N)
    for k in range(-N+1, N):
        wk = w[k + N - 1]
        if abs(wk) < 1e-15:
            continue
        if k >= 0:
            idx1 = np.arange(k, N)
            idx2 = np.arange(0, N-k)
            g[idx1] += wk * (1 - h[idx2])
            g[idx2] -= wk * h[idx1]
        else:
            kk = -k
            idx1 = np.arange(0, N-kk)
            idx2 = np.arange(kk, N)
            g[idx1] += wk * (1 - h[idx2])
            g[idx2] -= wk * h[idx1]
    return g
```

This is still O(N) Python loops (over k) each doing O(N) numpy ops. For N=760, that's 1519 iterations. Each takes maybe 10μs for the numpy ops. Total: ~15ms per gradient. That's acceptable for ~70 iterations per second.

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def greedy_binary(h, N, deadline, max_sweeps=2000, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def binary_sa(h, N, deadline, max_sweeps=20000, T0=3.0, Tcool=0.995, T_min=0.03, sf=1.0, kick_every=200, kick_size=4):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            # Structured kick: pick a shift and swap along it
            k = np.random.randint(1, N)
            valid_pairs = []
            for i in range(N - k if k > 0 else k, N - (N-k) if k > 0 else 0):
                pass
            # Simpler: just do random swaps
            for _ in range(kick_size):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def logsumexp_subgrad(h, N, deadline, tau0, lr0, n_iter=300):
    h = h.astype(float).copy()
    h = project_sum(h, N / 2.0)
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    tau = tau0
    lr = lr0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        c_shifted = c / tau
        cmax = c_shifted.max()
        w = np.exp(c_shifted - cmax)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        # Compute gradient
        g = np.zeros(N)
        for k_idx in range(2 * N - 1):
            wk = w[k_idx]
            if wk < 1e-10:
                continue
            k = k_idx - (N - 1)
            if k >= 0:
                if k < N:
                    idx_h = np.arange(k, N)
                    idx_s = np.arange(0, N - k)
                    g[idx_h] += wk * (1.0 - h[idx_s])
                    g[idx_s] -= wk * h[idx_h]
            else:
                kk = -k
                if kk < N:
                    idx_h = np.arange(0, N - kk)
                    idx_s = np.arange(kk, N)
                    g[idx_h] += wk * (1.0 - h[idx_s])
                    g[idx_s] -= wk * h[idx_h]
        # Gradient step with projection
        h = project_sum(h - lr * g, N / 2.0)
        # Track best
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        # Schedule
        lr *= 0.998
        tau = max(tau * 0.999, 0.3 / N)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(99991 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0.0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1.0
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

    best_h = h.copy()

            break

        else:
            stall += 1
            if stall > 40: break

        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))

        if best_pq is None: break

    return best_h, best_c * (2.0 / N)

def lse_subgrad(h, N, deadline, tau=30.0, lr=0.03, top_k=40):

    h = project_sum(h, N / 2.0)
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    t_cur = tau
    lr_cur = lr
    while time.time() < deadline:
        c = fft_corr(h)
        c_s = c / t_cur
        c_max = c_s.max()
        w = np.exp(c_s - c_max)
        w_sum = w.sum()
        if w_sum < 1e-30: break
        w = w / w_sum
        # Only use top-k weights for gradient (saves time)
        if top_k < len(w):
            top_idx = np.argpartition(w, -top_k)[-top_k:]
            w_use = np.zeros_like(w)
            w_use[top_idx] = w[top_idx]
            w_use /= w_use.sum()
        else:
            w_use = w
        # Gradient
        g = np.zeros(N)
        center = N - 1
        for ki in range(2 * N - 1):
            wk = w_use[ki]
            if wk < 1e-7: continue
            k = ki - center
            if k >= 0 and k < N:
                g[k:] += wk * (1.0 - h[:N-k])
                g[:N-k] -= wk * h[k:]
            elif k < 0 and -k < N:
                kk = -k
                g[:N-kk] += wk * (1.0 - h[kk:])
                g[kk:] -= wk * h[:N-kk]
        # Update
        h = project_sum(h - lr_cur * g, N / 2.0)
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        # Schedule
        lr_cur *= 0.999
        t_cur = max(t_cur * 0.9995, 2.0 / N)
    return best_h, best_pk * (2.0 / N)
```

```python
        if best_pq is None: break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def sa_optimize(h, N, deadline, T0=3.0, Tcool=0.995, T_min=0.02, sf=1.0, kick_every=150, kick_size=4):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P; best_h = h.copy(); stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2: break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.4
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None: break
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0; h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def lse_subgrad(h, N, deadline, tau0=35.0, lr0=0.025, top_k=35):
    h = h.astype(float).copy()
    h = project_sum(h, N / 2.0)
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    tau = tau0
    lr = lr0
    it = 0
    while time.time() < deadline:
        c = fft_corr(h)
        c_s = c / tau
        c_max = c_s.max()
        w = np.exp(c_s - c_max)
        w_sum = w.sum()
        if w_sum < 1e-30: break
        w = w / w_sum
        active = np.where(w > 1e-8)[0]
        if len(active) > top_k:
            top_idx = np.argsort(w)[-top_k:]
            w_use = np.zeros_like(w)
            w_use[top_idx] = w[top_idx]
            w_use /= w_use.sum()
            active = top_idx
        else:
            w_use = w
        g = np.zeros(N)
        center = N - 1
        for ki in active:
            wk = w_use[ki]
            if wk < 1e-10: continue
            k = ki - center
            if k >= 0 and k < N:
                g[k:] += wk * (1.0 - h[:N-k])
                g[:N-k] -= wk * h[k:]
            elif k < 0 and -k < N:
                kk = -k
                g[:N-kk] += wk * (1.0 - h[kk:])
                g[kk:] -= wk * h[:N-kk]
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-14:
            h = project_sum(h - lr * g, N / 2.0)
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        it += 1
        if it % 100 == 0:
            lr *= 0.97
            tau = max(tau * 0.96, 1.5 / N)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick binary greedy
    h_bin = ensure_sum(h0, N)
    d1 = min(deadline - 650, t0 + 15)
    h1, c5_1 = greedy_binary(h_bin, N, d1, max_sweeps=120, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: LSE subgradient (continuous, smooth start)
    d2 = min(deadline - 500, t0 + 120)
    h2, c5_2 = lse_subgrad(best_h, N, d2, tau0=45.0, lr0=0.02, top_k=30)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Binary round + greedy polish
    h3b = ensure_sum(best_h, N)
    d3 = min(deadline - 430, t0 + 155)
    h3, c5_3 = greedy_binary(h3b, N, d3, max_sweeps=200, sf=0.9)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: SA for binary exploration
    h_sa = ensure_sum(best_h, N)
    d4 = min(deadline - 280, t0 + 330)
    h4, c5_4 = sa_optimize(h_sa, N, d4, T0=4.0, Tcool=0.994, T_min=0.03, sf=1.0, kick_every=130, kick_size=4)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Second LSE round (sharper)
    if time.time() < deadline - 130:
        h5i = best_h.astype(float).copy()
        h5i += np.random.randn(N) * 0.03
        h5i = project_sum(h5i, N / 2.0)
        d5 = min(deadline - 80, t0 + 490)
        h5, c5_5 = lse_subgrad(h5i, N, d5, tau0=25.0, lr0=0.018, top_k=28)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: SA with diverse perturbation
    if time.time() < deadline - 70:
        h6s = ensure_sum(best_h, N)
        for _ in range(8):
            p = np.where(h6s == 1)[0]
            q = np.where(h6s == 0)[0]
            if len(p) > 0 and len(q) > 0:
                h6s[p[np.random.randint(len(p))]], h6s[q[np.random.randint(len(q))]] = 0.0, 1.0
        d6 = min(deadline - 40, t0 + 640)
        h6, c5_6 = sa_optimize(h6s, N, d6, T0=5.0, Tcool=0.993, T_min=0.02, sf=0.8, kick_every=100, kick_size=5)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final LSE fine polish
    if time.time() < deadline - 25:
        h7i = best_h.astype(float).copy()
        h7i += np.random.randn(N) * 0.015
        h7i = project_sum(h7i, N / 2.0)
        d7 = min(deadline - 12, t0 + 1050)
        h7, c5_7 = lse_subgrad(h7i, N, d7, tau0=15.0, lr0=0.012, top_k=22)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final binary greedy micro-polish
    if time.time() < deadline - 2:
        h8b = ensure_sum(best_h, N)
        h8, c5_8 = greedy_binary(h8b, N, deadline - 0.5, max_sweeps=400, sf=0.7)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N
```
