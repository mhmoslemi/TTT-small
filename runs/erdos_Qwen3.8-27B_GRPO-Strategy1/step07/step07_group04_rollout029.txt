```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_gradient(h, beta, dx):
    N = len(h)
    corr = fft_corr(h)
    g = corr * dx
    g_max = g.max()
    shifted = beta * (g - g_max)
    # For numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    w_vec = exp_shifted / S  # this is the softmax weight
    
    # Gradient via convolution
    first = np.convolve(1.0 - h, w_vec, 'full')[N-1 : 2*N-1]
    second = np.convolve(h, w_vec, 'full')[N-1 : 2*N-1]
    grad = first - second
    return grad

def adam_smooth(h, N, deadline, beta=80, lr=0.03, beta1=0.9, beta2=0.999):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    m = np.zeros(N)
    v = np.zeros(N)
    t = 1
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h))) * dx
    stall = 0
    while time.time() < deadline:
        grad = smooth_gradient(h, beta, dx)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        h_new = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), target)
        h = h_new
        t += 1
        c = float(np.max(fft_corr(h))) * dx
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            break
        if t % 1000 == 0:
            lr *= 0.95
            beta *= 1.02
    return best_h, best_c
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, beta):
    N = len(h)
    corr = fft_corr(h)
    g = corr
    g_max = g.max()
    shifted = np.clip(beta * (g - g_max), -60, 60)
    w = np.exp(shifted)
    w /= w.sum()
    first = np.convolve(1.0 - h, w, 'full')[N-1:2*N-1]
    second = np.convolve(h, w, 'full')[N-1:2*N-1]
    return first - second, float(g_max)

def adam_opt(h, N, deadline, beta_init=40, lr=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    v = np.zeros(N)
    t = 1
    beta = beta_init
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h))) * (2.0/N)
    stall = 0
    while time.time() < deadline:
        grad, _ = smooth_grad(h, beta)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1.0 - 0.9**t)
        v_hat = v / (1.0 - 0.999**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-7)
        h = project_sum(h - step, target)
        t += 1
        c = float(np.max(fft_corr(h))) * (2.0/N)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if t % 500 == 0:
            beta = min(beta * 1.1, 500.0)
            lr *= 0.97
        if stall > 3000:
            # Perturb and reset
            h = project_sum(h + np.random.randn(N) * 0.02, target)
            m *= 0.1
            v *= 0.1
            stall = 0
            beta = beta_init
            lr = 0.04
    return best_h, best_c

def ensure_binary_sum(h, N):
    h_bin = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_hc(h, N, deadline, max_iter=20000):
    """Binary hill climbing with random restarts"""
    h = ensure_binary_sum(h, N).copy()
    dx = 2.0 / N
    best_c = float(np.max(fft_corr(h))) * dx
    best_h = h.copy()
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        one_positions = np.where(h == 1)[0]
        zero_positions = np.where(h == 0)[0]
        if len(one_positions) < 2 or len(zero_positions) < 2:
            break
        p = one_positions[np.random.randint(len(one_positions))]
        q = zero_positions[np.random.randint(len(zero_positions))]
        h[p], h[q] = 0.0, 1.0
        c_new = float(np.max(fft_corr(h))) * dx
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h.copy()
            stall = 0
        else:
            h[p], h[q] = 1.0, 0.0
            stall += 1
        if stall > 200:
            # Restart with perturbation
            for _ in range(8):
                pp = one_positions[np.random.randint(len(one_positions))]
                qq = zero_positions[np.random.randint(len(zero_positions))]
                h[pp], h[qq] = 1.0 - h[pp], 1.0 - h[qq]
            stall = 0
    return best_h, best_c
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, beta):
    N = len(h)
    corr = fft_corr(h)
    g = corr - corr.max()
    shifted = np.clip(beta * g, -60, 60)
    w = np.exp(shifted)
    w /= w.sum()
    w_flip = w[::-1]
    first = np.convolve(1.0 - h, w_flip, 'full')[N - 1:2 * N - 1]
    second = np.convolve(h, w, 'full')[N - 1:2 * N - 1]
    return first - second

def adam_smooth(h_init, N, deadline, beta0=30, lr0=0.04, beta1=0.9, beta2=0.999):

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)

            row = new_maxes[p, zeros].copy()
            mi = int(np.argmin(row))
            if row[mi] * dx < P * dx - 1e-14:
                delta = row[mi] - float(np.max(corr))
                if delta < best_delta:
                    best_delta = delta
                    best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            stall += 1
            if stall > 30:
                for _ in range(6):
                    pp = ones[np.random.randint(len(ones))]
                    qq = zeros[np.random.randint(len(zeros))]
                    h[pp], h[qq] = 0.0, 1.0
                corr = fft_corr(h)
                P = float(np.max(corr)) * dx
                stall = 0
            else:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
                corr = fft_corr(h)
                P = float(np.max(corr)) * dx
            if P > best_c:
                h = best_h.copy()
                corr = fft_corr(h)
                P = float(np.max(corr))
            continue
        p, q = best_pq
        h[p], h[q] = 0.0, 1.0
        corr = fft_corr(h)
        P = float(np.max(corr)) * dx
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            h = best_h.copy()
            corr = fft_corr(h)
            P = float(np.max(corr))
    return best_h, best_c

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, kind='linear')

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth Adam at N0 (main exploration)
    d1 = min(deadline - 350, t0 + 250)
    h1 = project_sum(h0, N0 / 2.0)
    h1c, c1 = adam_smooth(h1, N0, d1, beta0=25, lr0=0.05)
    if c1 < best_c5:
        best_c5, best_h = c1, h1c.copy()

    # Phase 2: Binary sweep at N0
    if time.time() < deadline - 250:
        h2_init = best_h if len(best_h) == N0 else h0
        h2_init = project_sum(h2_init, N0 / 2.0)
        d2 = min(deadline - 200, t0 + 400)
        h2c, c2 = binary_hc(h2_init, N0, d2, max_sweeps=150)
        if c2 < best_c5:
            best_c5, best_h = c2, h2c.copy()

    # Phase 3: Smooth Adam with higher beta (refinement)
    if time.time() < deadline - 180:
        h3 = project_sum(best_h + np.random.randn(N0) * 0.008, N0 / 2.0)
        d3 = min(deadline - 130, t0 + 550)
        h3c, c3 = adam_smooth(h3, N0, d3, beta0=80, lr0=0.02)
        if c3 < best_c5:
            best_c5, best_h = c3, h3c.copy()

    # Phase 4: Try N=1000
    N1 = 1000
    if time.time() < deadline - 130:
        h_up = upsample(best_h, len(best_h), N1)
        h_up = project_sum(h_up, N1 / 2.0)
        d4 = min(deadline - 80, t0 + 750)
        h4c, c4 = adam_smooth(h_up, N1, d4, beta0=40, lr0=0.03)
        if c4 < best_c5:
            best_c5, best_h = c4, h4c.copy()

    # Phase 5: Binary at N=1000
    if time.time() < deadline - 60 and len(best_h) == N1:
        d5 = min(deadline - 30, t0 + 900)
        h5c, c5 = binary_hc(best_h, N1, d5, max_sweeps=100)
        if c5 < best_c5:
            best_c5, best_h = c5, h5c.copy()

    # Phase 6: Final polish - smooth Adam at best N
    if time.time() < deadline - 2:
        Nf = len(best_h)
        hf = project_sum(best_h + np.random.randn(Nf) * 0.004, Nf / 2.0)
        d6 = deadline - 1
        h6c, c6 = adam_smooth(hf, Nf, d6, beta0=150, lr0=0.008)
        if c6 < best_c5:
            best_c5, best_h = c6, h6c.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
