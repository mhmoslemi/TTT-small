def fft_corr(h):

    L = 2*N-1
    return np.real(np.fft.irfft(np.fft.rfft(h,L)*np.fft.rfft(np.flip(1.0-h),L),L))

            if vl.any(): delta[ii[vl], Lp[vl]] += 1.0
            # Adjust: delta[p] was set to 0, but the above might have added to it
            delta[p,:] = 0.0
            # Update all deltas affected by h[q] change (0->1): delta[i][N-1+q-i] -= 1
            Lq2 = N-1+q-ii
            vl2 = (Lq2>=0)&(Lq2<2*N-1)
            if vl2.any(): delta[ii[vl2], Lq2[vl2]] -= 1.0
            # Fix delta[q]: it should be the new contribution (not the old minus 1)
            Ls2=max(0,N-1-q); Le2=min(2*N-1,N-1+q+1)
            if Le2>Ls2: delta[q,Ls2:Le2] = 1.0 - h[q+np.arange(Ls2,Le2)-(N-1)]
            c = float(corr.max())
            if c<bc: bc=c; bh=h.copy(); no_imp=0
            else: no_imp+=1
        else: no_imp+=1
        if no_imp>3000: break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = make_binary(h0, N0)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Continuous LSE optimization
    d1 = min(deadline-500, t0+280)
    hc = best_h.astype(float) + np.random.randn(N0)*0.03
    hc = np.clip(hc,0,1); hc -= (hc.sum()-N0/2)/N0; hc = np.clip(hc,0,1)
    h1, c1 = cont_opt(hc, N0, d1, a0=50, amax=400, lr0=0.05)
    if c1 < best_c5: best_c5, best_h = c1, h1.copy()
    
    # Phase 2: Binary SA from best continuous
    if time.time() < deadline-300:
        d2 = min(deadline-250, time.time()+250)
        h2, c2 = bin_sa(make_binary(best_h, N0), d2, npeak=6, npp=14, T0=0.05, cr=0.9992)
        if c2 < best_c5: best_c5, best_h = c2, h2.copy()
    
    # Phase 3: Another continuous pass from improved binary
    if time.time() < deadline-200:
        d3 = min(deadline-150, time.time()+120)
        hc3 = best_h.astype(float) + np.random.randn(N0)*0.02
        hc3 = np.clip(hc3,0,1); hc3 -= (hc3.sum()-N0/2)/N0; hc3 = np.clip(hc3,0,1)
        h3, c3 = cont_opt(hc3, N0, d3, a0=100, amax=500, lr0=0.03)
        if c3 < best_c5: best_c5, best_h = c3, h3.copy()
    
    # Phase 4: Binary SA with lower temp, more peaks
    if time.time() < deadline-100:
        d4 = min(deadline-50, time.time()+180)
        h4, c4 = bin_sa(make_binary(best_h, len(best_h)), d4, npeak=8, npp=20, T0=0.02, cr=0.9996)
        if c4 < best_c5: best_c5, best_h = c4, h4.copy()
    
    # Phase 5: Try different resolution
    if time.time() < deadline-60:
        for Nt in [400, 440, 500]:
            if time.time() > deadline-40: break
            xo = np.linspace(0,2,len(best_h),endpoint=False)
            xn = np.linspace(0,2,Nt,endpoint=False)
            hi = np.interp(xn, xo, best_h)
            hb = make_binary(hi, Nt)
            sub_dl = min(deadline-20, time.time()+35)
            hr, cr = bin_sa(hb, sub_dl, npeak=5, npp=10, T0=0.04, cr=0.999)
            if cr < best_c5: best_c5, best_h = cr, hr.copy()
    
    # Phase 6: Final SA polish
    if time.time() < deadline-3:
        Ny = len(best_h)
        h6, c6 = bin_sa(make_binary(best_h, Ny), deadline-2, npeak=10, npp=25, T0=0.012, cr=0.9998)
        if c6 < best_c5: best_c5, best_h = c6, h6.copy()
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
