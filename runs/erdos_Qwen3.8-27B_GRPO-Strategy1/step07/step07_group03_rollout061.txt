def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N, seed=42):
    nh = N // 2
    h_b = (h >= 0.5).astype(float)
    n1 = int(h_b.sum())
    if n1 == nh:
        return h_b
    rng = np.random.RandomState(seed)
    if n1 > nh:
        idx = np.where(h_b == 1)[0]
        h_b[idx[rng.permutation(len(idx))[:n1-nh]]] = 0
    else:
        idx = np.where(h_b == 0)[0]
        h_b[idx[rng.permutation(len(idx))[:nh-n1]]] = 1
    return h_b

def upsample(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def compute_dgrad(h):
    N = len(h)
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    ip = pos + sv
    im = pos - sv
    vp = ((ip >= 0) & (ip < N)).astype(float)
    vm = ((im >= 0) & (im < N)).astype(float)
    ip_c = np.clip(ip, 0, N-1)
    im_c = np.clip(im, 0, N-1)
    return (1.0 - h[ip_c]) * vp - h[im_c] * vm

    for sweep in range(max_sweeps):

            break
        temp *= 0.999
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

            break

            h[best_move[1]] = 0.0
            h[best_move[2]] = 1.0
        elif best_move[0] == 2:
            h[best_move[1]] = h[best_move[2]] = 0.0

            best_h = h.copy()

def fista_smooth(h, N, deadline, beta=50, k_iter=2000, lr0=0.005):
    """FISTA with log-sum-exp smooth approximation of max correlation"""
    h = h.astype(float).copy()
    target = N / 2.0
    y = h.copy()
    m = np.zeros(N)
    t = 1.0
    lr = lr0
    best_h = h.copy()
    best_PK = float(np.max(fft_corr(h)))
    
    for it in range(k_iter):
        if time.time() > deadline:
            break
        corr = fft_corr(y)
        # Smooth max approximation
        mc = corr.max()
        e = np.exp(beta * (corr - mc))
        w = e / e.sum()
        smooth_val = (1.0/beta) * np.log(e.sum()) + mc
        
        # Gradient of smooth max wrt h
        # grad[i] = sum_k w[k] * dgrad[i,k]
        dgrad = compute_dgrad(y)
        grad = dgrad.T @ w  # (N,)
        
        # FISTA update
        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to sum constraint
        diff = target - h_new.sum()
        h_new += diff / N
        h_new = np.clip(h_new, 0.0, 1.0)
        # Re-project more carefully
        for _ in range(10):
            diff = target - h_new.sum()
            if abs(diff) < 1e-12:
                break
            h_new += diff / N
            h_new = np.clip(h_new, 0.0, 1.0)
        
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        m = (h_new - y) * ((t - 1) / t_new)
        y = h_new + m
        t = t_new
        
        # Evaluate actual objective
        PK = float(np.max(fft_corr(h_new)))
        if PK < best_PK - 1e-13:
            best_PK = PK
            best_h = h_new.copy()
        
        lr *= 0.9999
        if it % 500 == 0 and it > 0:
            lr *= 0.95
            beta *= 1.2
    
    return best_h, best_PK * (2.0 / N)

∂/∂h_i [ (1/β) log(Σ_k exp(β * corr_k)) ] = Σ_k w_k * (∂corr_k/∂h_i)

And ∂corr_k/∂h_i = dgrad[i, k] (as I computed). So grad = dgrad.T @ w is correct.

But wait, I need to verify my dgrad formula. Let me re-derive:

corr_k where k indexes the FFT output position. The shift is s = k - (N-1).

corr_k = Σ_j h[j] * (1 - h[j+s]) for valid j, j+s in [0, N-1]

∂corr_k / ∂h[i]:
- If i is a "j": contribution is (1 - h[i+s]) if i+s is valid
- If i is a "j+s", i.e., j = i-s: contribution is h[i-s] * (-1) if i-s is valid

So: ∂corr_k/∂h[i] = (1-h[i+s])*[0≤i+s<N] - h[i-s]*[0≤i-s<N]

This matches my dgrad formula. Good.

def fista_continuous(h, N, deadline, beta=60, lr0=0.003):
    h = h.astype(float).copy()
    target = N / 2.0
    y = h.copy()
    m = np.zeros(N)
    t_k = 1.0
    lr = lr0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    
    while time.time() < deadline:
        corr = fft_corr(y)
        mc = corr.max()
        e = np.exp(beta * (corr - mc))
        w = e / e.sum()
        
        dgrad = compute_dgrad(y)
        grad = dgrad.T @ w
        
        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        for _ in range(10):
            h_new += (target - h_new.sum()) / N
            h_new = np.clip(h_new, 0.0, 1.0)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t_k*t_k)) / 2.0
        m = (h_new - y) * ((t_k - 1.0) / t_new)
        y = h_new + m
        y = np.clip(y, 0.0, 1.0)
        t_k = t_new
        
        pk = float(np.max(fft_corr(h_new)))

            best_h = h_new.copy()
        
        lr *= 0.9998
        if int(t_k) % 50 == 0:
            lr *= 0.97
            beta = min(beta * 1.15, 500.0)
    
    return best_h, best_pk * (2.0/N)
