            h_old = h.copy()
            i_idx = rng.choice(ones, n_swap, replace=False)
            j_idx = rng.choice(zeros, n_swap, replace=False)
            h[i_idx] = 0
            h[j_idx] = 1
            c = fft_corr(h)
            new_pk = float(c.max())
            if new_pk < curr_pk or rng.random() < np.exp(-(new_pk - curr_pk) / max(T, 1e-10)):
                if new_pk < best_pk:
                    best_pk = new_pk
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                curr_pk = new_pk
            else:
                h[:] = h_old
                stall += 1
        else:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            c = fft_corr(h)
            new_pk = float(c.max())
            if new_pk < curr_pk or rng.random() < np.exp(-(new_pk - curr_pk) / max(T, 1e-10)):
                if new_pk < best_pk:
                    best_pk = new_pk
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                curr_pk = new_pk
            else:
                h[i] = 1.0
                h[j] = 0.0
                stall += 1
        it += 1
        if stall > restart_stall:
            h = best_h.copy()
            n_p = rng.randint(2, 5)
            ones_b = np.where(h == 1)[0]
            zeros_b = np.where(h == 0)[0]
            for _ in range(n_p):
                h[ones_b[rng.randint(len(ones_b))]] = 0.0
                h[zeros_b[rng.randint(len(zeros_b))]] = 1.0
            c = fft_corr(h)
            curr_pk = float(c.max())
            stall = 0
    return best_h, best_pk * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Nesterov warmup at N0 (80s)
    d1 = min(deadline - 620, time.time() + 80)
    hc = h0.copy() + rng.randn(N0) * 0.04
    hc = project_sum(hc, N0 / 2.0)
    h_n, c5_n = nesterov_opt(hc, N0, d1, k=50, T=100.0, lr0=0.014)
    if c5_n < best_c5:
        best_c5, best_h = c5_n, h_n.copy()

    # Phase 2: Binary SA - main search (600s)
    d2 = min(deadline - 100, time.time() + 600)
    h_bin = ensure_binary(best_h, N0, rng)
    h_sa, c5_sa = binary_sa(h_bin, N0, d2, rng, T0=0.6, T1=0.003, multi_every=4000, restart_stall=5000)
    if c5_sa < best_c5:
        best_c5, best_h = c5_sa, h_sa.copy()

    # Phase 3: Nesterov from best continuous (55s)
    if time.time() < deadline - 120:
        NFin = len(best_h)
        hc2 = best_h + rng.randn(NFin) * 0.03
        hc2 = project_sum(hc2, NFin / 2.0)
        d3 = min(deadline - 100, time.time() + 50)
        h3, c5_3 = nesterov_opt(hc2, NFin, d3, k=60, T=150.0, lr0=0.008)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
            best_c5 = min(best_c5, c5_3)

    # Phase 4: Second binary SA from refreshed start (70s)
    if time.time() < deadline - 80:
        NFin = len(best_h)
        h_b2 = ensure_binary(best_h, NFin, rng)
        d4 = min(deadline - 60, time.time() + 65)
        h4, c5_4 = binary_sa(h_b2, NFin, d4, rng, T0=0.4, T1=0.005, multi_every=3500, restart_stall=4000)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Try half resolution
    N_half = N0 // 2
    if N_half >= 300 and time.time() < deadline - 100:
        h_half = best_h[:N_half * 2].reshape(N_half, 2).mean(axis=1)
        h_half = project_sum(h_half, N_half / 2.0)
        d5 = min(deadline - 80, time.time() + 50)
        h5, c5_5 = nesterov_opt(h_half, N_half, d5, k=40, T=80.0, lr0=0.012)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
        if time.time() < deadline - 40:
            h5b = ensure_binary(h5, N_half, rng)
            d5b = min(deadline - 30, time.time() + 30)
            h5b, c5_5b = binary_sa(h5b, N_half, d5b, rng, T0=0.4, T1=0.01, multi_every=3000, restart_stall=3000)
            if c5_5b < best_c5:
                best_c5, best_h = c5_5b, h5b.copy()

    # Phase 6: Final binary polish (30s)
    NFin = len(best_h)
    if time.time() < deadline - 10:
        hF = ensure_binary(best_h, NFin, rng)
        d6 = min(deadline - 2, time.time() + 28)
        hF, c5F = binary_sa(hF, NFin, d6, rng, T0=0.2, T1=0.004, multi_every=2500, restart_stall=3000)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Final micro Nesterov
    if time.time() < deadline - 3:
        NFin = len(best_h)
        hFc = best_h + rng.randn(NFin) * 0.015
        hFc = project_sum(hFc, NFin / 2.0)
        d7 = min(deadline - 1, time.time() + 15)
        h7, c5_7 = nesterov_opt(hFc, NFin, d7, k=50, T=120.0, lr0=0.006)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
