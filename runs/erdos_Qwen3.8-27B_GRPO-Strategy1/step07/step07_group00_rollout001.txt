    # Phase 1: Continuous Nesterov (find better basin)
    if time.time() < deadline - 500:
        d1 = min(deadline - 420, time.time() + 130)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, c1 = nesterov_polish(hc, N0, d1, lr=0.014, k_top=50, mom=0.84)
        if c1 < best_c5:
            best_c5, best_h = c1, hn.copy()

    # Phase 2: Aggressive binary SA
    if time.time() < deadline - 350:
        h2 = make_binary(best_h, N0)
        d2 = min(deadline - 280, time.time() + 180)
        h2, c2 = targeted_sa(h2, d2, n_peaks=10, n_per_peak=22, cool_rate=0.999, t_start=0.05, chunk=12)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Continuous polish (finer)
    if time.time() < deadline - 220:
        d3 = min(deadline - 170, time.time() + 90)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.02
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, c3 = nesterov_polish(hc, N0, d3, lr=0.007, k_top=80, mom=0.89)
        if c3 < best_c5:
            best_c5, best_h = c3, hn.copy()

    # Phase 4: Multi-resolution cycle
    if time.time() < deadline - 160:
        Ns = max(80, N0 // 3)
        if Ns % 2 != 0:
            Ns += 1
        xo = np.linspace(0, 2, N0, endpoint=False)
        xn = np.linspace(0, 2, Ns, endpoint=False)
        hs = np.interp(xn, xo, best_h)
        hs_bin = make_binary(hs, Ns)
        d4a = min(deadline - 80, time.time() + 55)
        hs_bin, _ = targeted_sa(hs_bin, d4a, n_peaks=6, n_per_peak=14, cool_rate=0.999, t_start=0.035)
        # Upscale and refine
        xo2 = np.linspace(0, 2, Ns, endpoint=False)
        xn2 = np.linspace(0, 2, N0, endpoint=False)
        hu = np.interp(xn2, xo2, hs_bin)
        hu_bin = make_binary(hu, N0)
        d4b = min(deadline - 40, time.time() + 65)
        hu_bin, c4 = targeted_sa(hu_bin, d4b, n_peaks=12, n_per_peak=25, cool_rate=0.9994, t_start=0.03, chunk=12)
        if c4 < best_c5:
            best_c5, best_h = c4, hu_bin.copy()

    # Phase 5: Multiple short SA restarts with perturbation
    if time.time() < deadline - 30:
        for attempt in range(4):
            if time.time() > deadline - 8:
                break
            h5 = make_binary(best_h, N0)
            n_flip = max(3, N0 // 80)
            ones = np.where(h5 == 1)[0]
            zeros = np.where(h5 == 0)[0]
            fp = np.random.choice(ones, min(n_flip, len(ones)), replace=False)
            fq = np.random.choice(zeros, min(n_flip, len(zeros)), replace=False)
            h5[fp] = 0.0
            h5[fq] = 1.0
            d5 = min(deadline - 3, time.time() + 35)
            h5, c5 = targeted_sa(h5, d5, n_peaks=12, n_per_peak=28, cool_rate=0.9996, t_start=0.018, chunk=10)
            if c5 < best_c5:
                best_c5, best_h = c5, h5.copy()

    # Phase 6: Micro continuous polish
    if time.time() < deadline - 1:
        hf, cm = micro_polish(best_h, N0, n_steps=600)
        if cm < best_c5:
            best_c5, best_h = cm, hf.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
