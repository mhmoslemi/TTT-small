    T = T0
    stall = 0
    rng = np.random.RandomState(int(time.time()*1000) % 2**31)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % 3 == 0:
            corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Kick mechanism
        if stall >= kick_every:
            n_kick = kick_size
            for _ in range(n_kick):
                if len(ones) > 1 and len(zeros) > 1:
                    p = ones[rng.integers(len(ones))]
                    q = zeros[rng.integers(len(zeros))]
                    if p != q:
                        h[p], h[q] = 0.0, 1.0
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
            stall = 0
            T = T0 * 0.5
            corr = fft_corr(h)
            continue
        # Try 2-swap move with some probability
        if len(ones) >= 2 and len(zeros) >= 2 and rng.random() < two_swap_prob:
            p1 = ones[rng.integers(len(ones))]
            p2_idx = (rng.integers(len(ones)) + 1) % len(ones)
            p2 = ones[p2_idx]
            q1 = zeros[rng.integers(len(zeros))]
            q2 = zeros[rng.integers(len(zeros))]
            # Check no overlap
            if len({p1, p2, q1, q2}) == 4:
                h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                new_corr = fft_corr(h)
                new_P = float(new_corr.max())
                delta = new_P - P
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    if new_P < best_c:
                        best_c = new_P
                        best_h = h.copy()
                    corr = new_corr
                else:
                    h[p1] = 1; h[p2] = 1; h[q1] = 0; h[q2] = 0
                T = max(T_min, T * Tcool)
                continue
        # Standard single swap with best-move selection
        A, B = compute_D(h, N)
        Bz = B[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = corr + A[p, :]
            cand = base[None, :] + Bz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[rng.integers(len(ones))]
            q = zeros[rng.integers(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
        if sweep % 3 == 2:
            corr = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def smooth_max_grad(h, N, tau=50.0, top_k=50):
    """Compute gradient of smooth max (log-sum-exp / tau) of correlation."""
    corr = fft_corr(h)
    top = np.argpartition(corr, -top_k)[-top_k:]
    sc = corr[top] - corr[top].max()
    w = np.exp(tau * sc)
    w /= w.sum()
    t = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)
    pp = pos[:,None] + t[None,:]
    pm = pos[:,None] - t[None,:]
    hp_p = (pp >= 0) & (pp < N)
    hm_p = (pm >= 0) & (pm < N)
    hp_c = np.clip(pp, 0, N-1)
    hm_c = np.clip(pm, 0, N-1)
    hp_val = np.where(hp_p, h[hp_c], 0.0)
    hm_val = np.where(hm_p, h[hm_c], 0.0)
    # grad[i] = sum_k w_k * [(1-h[i-t_k]) - h[i+t_k]]
    # = sum_k w_k - sum_k w_k*h[i-t_k] - sum_k w_k*h[i+t_k]
    wt = w  # weights for top_k shifts only
    # For efficiency, compute using matrix ops
    # grad = sum of wt[j] * ((1-h[i-t_j]) - h[i+t_j]) for j in top_k
    # Use subset of shifts
    t_sub = t[top]
    pp_sub = pos[:,None] + t_sub[None,:]
    pm_sub = pos[:,None] - t_sub[None,:]
    hp_sub = (pp_sub >= 0) & (pp_sub < N)
    hm_sub = (pm_sub >= 0) & (pm_sub < N)
    hp_s = np.where(hp_sub, h[np.clip(pp_sub, 0, N-1)], 0.0)
    hm_s = np.where(hm_sub, h[np.clip(pm_sub, 0, N-1)], 0.0)
    grad = np.zeros(N)
    for j in range(len(top)):
        grad += wt[j] * (1.0 - hm_s[:,j]) - wt[j] * hp_s[:,j]
    return grad, float(corr.max())

def nesterov_smooth(h, N, deadline, tau=80.0, top_k=60, lr0=0.015, k_its=2000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    for it in range(k_its):
        if time.time() > deadline:
            break
        y = h + m
        grad, pk = smooth_max_grad(y, N, tau, top_k)
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
        h_new = np.clip(y - lr * grad, 0, 1)
        # Project to target sum
        s = h_new.sum()
        if abs(s - target) > 1e-6:
            # Simple bisection projection
            lo, hi = -1.0, 1.0
            for _ in range(40):
                mid = (lo + hi) * 0.5
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo+hi)*0.5, 0, 1)
        m = beta * (h_new - h)
        h = h_new
        if it % 300 == 0:
            lr *= 0.97
            tau *= 1.05
            beta = min(0.97, 0.9 + it * 0.0001)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy
    h_bin = ensure_binary(h0, N, seed=42)
    d1 = min(deadline - 600, t0 + 40)
    h1, c5_1 = greedy_binary(h_bin, N, d1, max_sweeps=400, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Main SA (aggressive, long)
    d2 = min(deadline - 400, t0 + 200)
    h2, c5_2 = sa_optimize(h1, N, d2, T0=5.0, Tcool=0.993, T_min=0.03,
                           max_sweeps=15000, sf=1.0, kick_every=150, kick_size=5,
                           two_swap_prob=0.2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Smooth-max Nesterov (continuous)
    if time.time() < deadline - 200:
        hc = best_h.copy()
        hc = np.clip(hc + np.random.randn(N) * 0.08, 0, 1)
        s = hc.sum()
        if abs(s - N/2) > 0.1:
            hc = np.clip(hc * (N/2) / s, 0, 1)
        d3 = min(deadline - 180, t0 + 320)
        h3, c5_3 = nesterov_smooth(hc, N, d3, tau=60.0, top_k=70, lr0=0.018, k_its=3000)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary SA with different params (more exploration)
    if time.time() < deadline - 150:
        h4_start = ensure_binary(upsample(best_h, N), N, seed=123)
        for _ in range(8):
            p = np.where(h4_start == 1)[0]
            q = np.where(h4_start == 0)[0]
            if len(p) > 0 and len(q) > 0:
                h4_start[p[np.random.randint(len(p))]], h4_start[q[np.random.randint(len(q))]] = 0.0, 1.0
        d4 = min(deadline - 120, t0 + 440)
        h4, c5_4 = sa_optimize(h4_start, N, d4, T0=6.0, Tcool=0.992, T_min=0.02,
                               max_sweeps=12000, sf=0.8, kick_every=100, kick_size=6,
                               two_swap_prob=0.25)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Second SA round from best (more exploitation)
    if time.time() < deadline - 80:
        h5_start = ensure_binary(best_h, N, seed=777)
        d5 = min(deadline - 50, t0 + 560)
        h5, c5_5 = sa_optimize(h5_start, N, d5, T0=3.0, Tcool=0.995, T_min=0.01,
                               max_sweeps=15000, sf=1.0, kick_every=200, kick_size=3,
                               two_swap_prob=0.1)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Nesterov polish
    if time.time() < deadline - 30:
        hc2 = np.clip(best_h + np.random.randn(N) * 0.03, 0, 1)
        s2 = hc2.sum()
        if abs(s2 - N/2) > 0.1:
            hc2 = np.clip(hc2 * (N/2) / s2, 0, 1)
        d6 = min(deadline - 15, t0 + 650)
        h6, c5_6 = nesterov_smooth(hc2, N, d6, tau=100.0, top_k=40, lr0=0.008, k_its=2000)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 2:
        hF = ensure_binary(best_h, N_fin, seed=999)
        sf_fin = 1.0 if N_fin <= 800 else 0.7
        hF2, c5F = greedy_binary(hF, N_fin, deadline - 0.5, max_sweeps=500, sf=sf_fin)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
