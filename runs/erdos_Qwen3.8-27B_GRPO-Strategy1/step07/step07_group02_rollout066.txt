```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def project_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def C_row(h, N, idx):
    k = np.arange(2 * N - 1) - (N - 1)
    ip = idx + k
    im = idx - k
    Ap = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    Bp = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    return Ap - Bp

def binary_search(h, N, deadline, max_iters=10000, sample_sz=200, temp=0.0, tabu_sz=200):
    h = h.astype(float).copy()
    best_c = np.inf
    best_h = h.copy()
    C = np.array([C_row(h, N, i) for i in range(N)])
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    stuck = 0
    tabu = {}
    it = 0
    while it < max_iters and time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        if temp > 0:
            ns2 = min(len(ones), max(10, sample_sz // 4))
            nz2 = min(len(zeros), max(10, sample_sz // 4))
            os2 = np.random.choice(ones, ns2, replace=False)
            zs2 = np.random.choice(zeros, nz2, replace=False)
        new = corr[None, None, :] + C[os_][None, :, :] + C[zs][:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        p, q = int(os_[mi[1]]), int(zs[mi[0]])
        do_move = False
        if mv < P - 1e-12:
            do_move = True
        elif temp > 0 and stuck > 20:
            if np.random.random() < temp * np.exp(-stuck * 0.003):
                do_move = True
            elif stuck > 80:
                p = int(os2[np.random.randint(len(os2))])
                q = int(zs2[np.random.randint(len(zs2))])
                do_move = True
        if do_move:
            h[p] = 0.0
            h[q] = 1.0
            C[p] = C_row(h, N, p)
            C[q] = C_row(h, N, q)
            corr = fft_corr(h)
            P = float(np.max(corr))
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            if stuck >= 100:
                # Kick: random moves
                for _ in range(3):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0.0
                    h[q] = 1.0
                    C[p] = C_row(h, N, p)
                    C[q] = C_row(h, N, q)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = fft_corr(h)
                P = float(np.max(corr))
                best_c = min(best_c, P)
                stuck = 0
            else:
                break
        it += 1
    return best_h, best_c

def nesterov(h, N, deadline, topk=80, lr0=0.015):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -topk)[-topk:]
        grad = np.zeros(N)
        center = N - 1
        for ai in idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= topk
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.89 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary search (aggressive, 300s)
    d1 = min(t0 + 300, deadline - 400)
    hb = (h0 > 0.5).astype(float)
    hb, _ = binary_search(hb, N0, d1, max_iters=15000, sample_sz=200, temp=0.0)
    c5b = compute_c5(hb)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()
    
    # Phase 2: Continuous polish (150s)
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 150)
        hc = best_h + np.random.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov(hc, N0, d2, topk=60, lr0=0.01)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    
    # Phase 3: Binary with annealing from best (200s)
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 200)
        h3 = (best_h > 0.5).astype(float)
        h3, _ = binary_search(h3, N0, d3, max_iters=12000, sample_sz=200, temp=0.2)
        c53 = compute_c5(h3)
        if c53 < best_c5:
            best_c5, best_h = c53, h3.copy()
    
    # Phase 4: Try different N values
    for N_try in [N0 - 2, N0 + 2, N0 + 4]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        ht = (hi > 0.5).astype(float)
        ht = project_sum(ht, N_try / 2.0)
        ht = (ht > 0.25).astype(float)  # re-threshold
        # Fix count
        n1 = int(ht.sum())
        target = N_try // 2
        if n1 != target:
            if n1 > target:
                idx = np.where(ht == 1)[0]
                ht[np.random.choice(idx, n1 - target, replace=False)] = 0
            else:
                idx = np.where(ht == 0)[0]
                ht[np.random.choice(idx, target - n1, replace=False)] = 1
        sub_dl = min(deadline - 30, time.time() + 60)
        ht, _ = binary_search(ht, N_try, sub_dl, max_iters=5000, sample_sz=180, temp=0.0)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht.copy()
    
    # Phase 5: Continuous final polish
    if time.time() < deadline - 20:
        hF = best_h + np.random.randn(len(best_h)) * 0.03
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov(hF, len(best_h), deadline - 5, topk=100, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    
    # Final: try both binary and continuous
    hB = (best_h > 0.5).astype(float)
    # Fix count
    n1 = int(hB.sum())
    t2 = len(hB) // 2
    if n1 != t2:
        if n1 > t2:
            idx = np.where(hB == 1)[0]
            hB[np.random.choice(idx, n1 - t2, replace=False)] = 0
        else:
            idx = np.where(hB == 0)[0]
            hB[np.random.choice(idx, t2 - n1, replace=False)] = 1
    c5B = compute_c5(hB)
    c5C = compute_c5(best_h)
    if c5B < best_c5:
        best_c5, best_h = c5B, hB
    elif c5C < best_c5:
        best_c5 = c5C
    
    return best_h, best_c5, len(best_h)
```

```python
    im = idx - k
    Ap = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    Bp = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    return Ap - Bp

def fix_binary(h, target):
    h = h.copy()
    n1 = int(h.sum() + 0.5)
    if n1 > target:
        idx = np.where(h == 1)[0]
        to_fix = min(len(idx), n1 - target)
        h[np.random.choice(idx, to_fix, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        to_fix = min(len(idx), target - n1)
        h[np.random.choice(idx, to_fix, replace=False)] = 1
    return h

def binary_search(h, N, deadline, max_iters=10000, sample_sz=180, temp=0.0, kick_thresh=60, kick_sz=4):
    h = h.astype(float).copy()
    best_c = np.inf
    best_h = h.copy()
    C = np.zeros((N, 2 * N - 1))
    for i in range(N):
        C[i] = C_row(h, N, i)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    stuck = 0
    it = 0
    while it < max_iters and time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        new = corr[None, None, :] + C[os_][None, :, :] + C[zs][:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        p, q = int(os_[mi[1]]), int(zs[mi[0]])
        do_move = False
        if mv < P - 1e-12:
            do_move = True
        elif temp > 0 and stuck > 15:
            prob = temp * np.exp(-stuck * 0.004)
            if np.random.random() < prob:
                do_move = True
        elif stuck >= kick_thresh:
            for _ in range(kick_sz):
                pp = int(np.random.choice(ones))
                qq = int(np.random.choice(zeros))
                h[pp] = 0.0
                h[qq] = 1.0
                C[pp] = C_row(h, N, pp)
                C[qq] = C_row(h, N, qq)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            P = float(np.max(corr))
            best_c = min(best_c, P)
            stuck = 0
            it += 1
            continue
        if do_move:
            h[p] = 0.0
            h[q] = 1.0
            C[p] = C_row(h, N, p)
            C[q] = C_row(h, N, q)
            corr = fft_corr(h)
            P = float(np.max(corr))
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            break
        it += 1
    return best_h, best_c

def nesterov_opt(h, N, deadline, topk=80, lr0=0.012):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -topk)[-topk:]
        grad = np.zeros(N)
        center = N - 1
        for ai in idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= topk
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 180 == 0:
            lr *= 0.992
            beta = min(0.97, 0.88 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive binary search
    d1 = min(t0 + 290, deadline - 380)
    hb = (h0 > 0.5).astype(float)
    hb = fix_binary(hb, N0 // 2)
    hb, _ = binary_search(hb, N0, d1, max_iters=14000, sample_sz=180, temp=0.0, kick_thresh=55, kick_sz=5)
    c5b = compute_c5(hb)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 2: Nesterov continuous polish
    if time.time() < deadline - 280:
        d2 = min(deadline - 260, time.time() + 110)
        hc = best_h + np.random.randn(N0) * 0.06
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, topk=70, lr0=0.01)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search with annealing
    if time.time() < deadline - 220:
        d3 = min(deadline - 200, time.time() + 200)
        h3 = (best_h > 0.5).astype(float)
        h3 = fix_binary(h3, N0 // 2)
        h3, _ = binary_search(h3, N0, d3, max_iters=12000, sample_sz=180, temp=0.25, kick_thresh=45, kick_sz=5)
        c53 = compute_c5(h3)
        if c53 < best_c5:
            best_c5, best_h = c53, h3.copy()

    # Phase 4: Multi-resolution (try nearby N values)
    for N_try in [N0 - 2, N0 + 2, N0 + 4, N0 + 6]:
        if time.time() > deadline - 90:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        ht = (hi > 0.5).astype(float)
        ht = fix_binary(ht, N_try // 2)
        sub_dl = min(deadline - 40, time.time() + 55)
        ht, _ = binary_search(ht, N_try, sub_dl, max_iters=4000, sample_sz=160, temp=0.1, kick_thresh=35, kick_sz=4)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht.copy()

    # Phase 5: Continuous final polish on best
    if time.time() < deadline - 25:
        hF = best_h + np.random.randn(len(best_h)) * 0.04
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 8, topk=100, lr0=0.007)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 6: Final binary cleanup
    if time.time() < deadline - 1:
        hB = (best_h > 0.5).astype(float)
        hB = fix_binary(hB, len(best_h) // 2)
        hB, _ = binary_search(hB, len(best_h), deadline - 0.5, max_iters=500, sample_sz=160, temp=0.0, kick_thresh=40, kick_sz=3)
        c5B = compute_c5(hB)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB
    # Also check continuous version
    c5C = compute_c5(best_h)
    if c5C < best_c5:
        best_c5 = c5C
    return best_h, best_c5, len(best_h)
```
