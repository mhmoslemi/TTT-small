```python
def lbfgs_smooth(h, N, deadline, T, max_outer):
    """Smoothed L-BFGS optimization"""
    def lin_grad(L_idx, h, N):
        g = np.zeros(N)
        if L_idx == 0:
            g[:] = 1.0 - 2.0 * h
        elif L_idx > 0:
            g[L_idx:] += 1.0 - h[:N - L_idx]
            g[:N - L_idx] -= h[L_idx:N]
        else:
            M = -L_idx
            g[:N - M] += 1.0 - h[M:]
            g[M:] -= h[:N - M]
        return g

        e = np.exp(T * (c - mx))

        val = mx + np.log(e.sum()) / T

        val += 0.5 * lam * s * s
        g += lam * s
```

```python
def full_delta(h, a, b, d, N):
    """Compute delta in corr[d] for swap (a:1->0, b:0->1), d>0."""
    # Affected t positions: a, b, a-d, b-d
    affected = set()
    for t in [a, b]:
        if 0 <= t <= N-1-d:
            affected.add(t)
    for t in [a-d, b-d]:
        if 0 <= t <= N-1-d:
            affected.add(t)
    
    delta = 0.0
    for t in affected:
        # orig term: h[t+d] * (1-h[t])
        old = h[t+d] * (1 - h[t])
        # new term: h_new[t+d] * (1-h_new[t])
        ht_d = h[t+d]
        if t + d == a: ht_d = 0.0
        elif t + d == b: ht_d = 1.0
        ht = h[t]
        if t == a: ht = 0.0
        elif t == b: ht = 1.0
        new = ht_d * (1 - ht)
        delta += new - old
    return delta
```
