    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

        h_bin3 = fix_binary(hn, N0 // 2)
        c5_b3 = compute_c5(h_bin3, N0)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
        if c5_b3 < best_c5:
            best_c5, best_h = c5_b3, h_bin3.copy()

    # Phase 4: More perturbed restarts (larger jumps)
    for r in range(5):
        if time.time() > deadline - 80:
            break
        h_r = best_h.copy()
        ones_r = np.where(h_r == 1)[0]
        zeros_r = np.where(h_r == 0)[0]
        for _ in range(10 + r * 4):
            p = np.random.choice(ones_r)
            q = np.random.choice(zeros_r)
            h_r[p] = 0
            h_r[q] = 1
            ones_r = np.where(h_r == 1)[0]
            zeros_r = np.where(h_r == 0)[0]
        d_r = min(deadline - 50, time.time() + 45)
        h_sr, _ = swap_optimize(h_r, N0, d_r, max_iters=4000, sample=200, temp=0.06, kick_at=25, kick_n=4)
        c5_r = compute_c5(h_sr, N0)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_sr.copy()

    # Phase 5: Multi-resolution
    for Nt in [350, 400, 420]:
        if time.time() > deadline - 40:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int, Nt // 2)
        d5 = min(deadline - 15, time.time() + 28)
        h_rt, _ = swap_optimize(h_bt, Nt, d5, max_iters=3000, sample=Nt // 2, kick_at=25, kick_n=3, temp=0.03)
        # Also exhaustive sweeps
        if time.time() < deadline - 8:
            h_rt = exhaustive_sweeps(h_rt, Nt, deadline - 5, n_sweeps=5)
        c5_rt = compute_c5(h_rt, Nt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Exhaustive sweeps on best solution
    h_best = fix_binary(best_h, len(best_h) // 2)
    if time.time() < deadline - 2:
        h_best = exhaustive_sweeps(h_best, len(best_h), deadline - 1, n_sweeps=15)
        c5_ex = compute_c5(h_best, len(best_h))
        if c5_ex < best_c5:
            best_c5, best_h = c5_ex, h_best.copy()

    # Final: one more directed pass
    if time.time() < deadline - 0.5:
        hF = fix_binary(best_h, len(best_h) // 2)
        hF, _ = swap_optimize(hF, len(best_h), deadline - 0.3, max_iters=2000, sample=200, kick_at=20, kick_n=3)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final binarization check
    h_bin_final = fix_binary(best_h, len(best_h) // 2)
    c5_bin = compute_c5(h_bin_final, len(best_h))
    if c5_bin < best_c5:
        best_h = h_bin_final
        best_c5 = c5_bin

    return best_h, best_c5, len(best_h)
