def nesterov_opt(h_in, N, deadline, lr=0.015, k_top=60, beta_init=0.85):
    h = h_in.astype(float).copy()
    hx = h.copy()
    best_pk = float(fft_corr(hx).max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = beta_init
    it = 0
    while time.time() < deadline:
        y = h + m
        y = np.clip(y, 0, 1)
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        lr_cur = lr * (1.0 / (1.0 + 0.0001 * it))
        h_new = np.clip(y - lr_cur * grad, 0.0, 1.0)
        # Project onto sum = N/2 via bisection
        for _ in range(20):
            adj = (h_new.sum() - N / 2.0) / N
            h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.95, beta_init + it * 0.00015)

    return best_h, best_pk * (2.0 / N)

    return best_h, final_c5, len(best_h)
