        h = h_new
    return best, best_c

def binary_sa(h_init, deadline, T0=3.0, T1=0.03, n_iter=100000, seed=42):
    N = len(h_init)
    h = h_init.copy()
    dh = 2.0 / N
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * dh
    best_h = h.copy()
    n_corr = 2 * N - 1
    ks = np.arange(-(N - 1), N)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    rng = np.random.default_rng(seed)
    delta = np.zeros(n_corr)
    buf = np.zeros(n_corr)
    it = 0
    no_imp = 0
    while it < n_iter and time.time() < deadline:
        frac = min(it / n_iter, 1.0)
        T = T0 * (T1 / T0) ** frac
        if no_imp > 8000:
            T = max(T, 0.5)
            no_imp = 0
        pi = int(rng.integers(len(ones_list)))
        qi = int(rng.integers(len(zeros_list)))
        p = ones_list[pi]
        q = zeros_list[qi]
        delta[:] = 0.0
        pk = p + ks
        m = (pk >= 0) & (pk < N)
        delta[m] += h[pk[m]]
        qk = q + ks
        m = (qk >= 0) & (qk < N)
        delta[m] -= h[qk[m]]
        pm = p - ks
        m = (pm >= 0) & (pm < N)
        delta[m] += h[pm[m]]
        qm = q - ks
        m = (qm >= 0) & (qm < N)
        delta[m] -= h[qm[m]]
        np.add(corr, delta, out=buf)
        new_max = float(np.max(buf))
        d_energy = new_max - cur_max
        if d_energy <= 0 or rng.random() < np.exp(-d_energy / max(T, 1e-10)):
            h[p] = 0.0
            h[q] = 1.0
            ones_list[pi] = q
            zeros_list[qi] = p
            corr, buf = buf, corr
            cur_max = new_max
            if cur_max * dh < best_c - 1e-12:
                best_c = cur_max * dh
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        it += 1
    return best_h, best_c

def two_swap_sa(h_init, deadline, T0=4.0, T1=0.1, n_iter=30000, seed=7):
    N = len(h_init)
    h = h_init.copy()
    dh = 2.0 / N
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * dh
    best_h = h.copy()
    n_corr = 2 * N - 1
    ks = np.arange(-(N - 1), N)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    rng = np.random.default_rng(seed)
    delta = np.zeros(n_corr)
    buf = np.zeros(n_corr)
    it = 0
    while it < n_iter and time.time() < deadline:
        T = T0 * (T1 / T0) ** (it / n_iter)
        if len(ones_list) < 2 or len(zeros_list) < 2:
            break
        idx1 = rng.choice(len(ones_list), 2, replace=False)
        idx2 = rng.choice(len(zeros_list), 2, replace=False)
        p1, p2 = ones_list[idx1[0]], ones_list[idx1[1]]
        q1, q2 = zeros_list[idx2[0]], zeros_list[idx2[1]]
        delta[:] = 0.0
        for pp in (p1, p2):
            pk = pp + ks
            m = (pk >= 0) & (pk < N)
            delta[m] += h[pk[m]]
            pm = pp - ks
            m = (pm >= 0) & (pm < N)
            delta[m] += h[pm[m]]
        for qq in (q1, q2):
            qk = qq + ks
            m = (qk >= 0) & (qk < N)
            delta[m] -= h[qk[m]]
            qm = qq - ks
            m = (qm >= 0) & (qm < N)
            delta[m] -= h[qm[m]]
        np.add(corr, delta, out=buf)
        new_max = float(np.max(buf))
        d_e = new_max - cur_max
        if d_e <= 0 or rng.random() < np.exp(-d_e / max(T, 1e-10)):
            h[p1] = 0.0; h[p2] = 0.0; h[q1] = 1.0; h[q2] = 1.0
            ones_list[idx1[0]] = q1; ones_list[idx1[1]] = q2
            zeros_list[idx2[0]] = p1; zeros_list[idx2[1]] = p2
            corr, buf = buf, corr
            cur_max = new_max
            if cur_max * dh < best_c - 1e-12:
                best_c = cur_max * dh
                best_h = h.copy()
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Continuous optimization (~100s)
    d1 = min(deadline - 600, t0 + 100)
    h1 = project(h0 + np.random.randn(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=30, lr=0.018)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Intensive binary SA at N0 (~350s)
    hb = to_binary(best_h, rng)
    d2 = min(deadline - 250, time.time() + 350)
    hb, c2 = binary_sa(hb, d2, T0=3.0, T1=0.03, n_iter=200000, seed=42)
    if c2 < best_c:
        best_c, best_h = c2, hb.copy()

    # Phase 3: Two-swap SA (~120s)
    if time.time() < deadline - 150:
        d3 = min(deadline - 130, time.time() + 120)
        h3, c3 = two_swap_sa(best_h, d3, T0=4.0, T1=0.1, n_iter=40000, seed=7)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Try alternate resolution
    if time.time() < deadline - 200:
        N_alt = N0 + 20
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_alt, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_alt = to_binary(h_int, rng)
        d4 = min(deadline - 180, time.time() + 100)
        h_alt, c4 = binary_sa(h_alt, d4, T0=2.5, T1=0.05, n_iter=80000, seed=99)
        if c4 < best_c:
            best_c, best_h = c4, h_alt.copy()
            N0 = N_alt

    # Phase 5: Refine at best resolution with fresh SA
    if time.time() < deadline - 120:
        hb5 = to_binary(best_h, rng)
        d5 = min(deadline - 100, time.time() + 100)
        hb5, c5 = binary_sa(hb5, d5, T0=2.0, T1=0.02, n_iter=150000, seed=123)
        if c5 < best_c:
            best_c, best_h = c5, hb5.copy()

    # Phase 6: Two-swap refinement
    if time.time() < deadline - 60:
        d6 = min(deadline - 50, time.time() + 55)
        h6, c6 = two_swap_sa(best_h, d6, T0=3.0, T1=0.05, n_iter=20000, seed=55)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    # Phase 7: Final SA polish
    if time.time() < deadline - 10:
        hb7 = to_binary(best_h, rng)
        d7 = min(deadline - 2, time.time() + 30)
        hb7, c7 = binary_sa(hb7, d7, T0=1.5, T1=0.01, n_iter=60000, seed=200)
        if c7 < best_c:
            best_c, best_h = c7, hb7.copy()

    # Final validation
    final_c = compute_c5(best_h)
    hB = to_binary(best_h, rng)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    if final_c > best_c:
        final_c = best_c
    return best_h, final_c, len(best_h)
