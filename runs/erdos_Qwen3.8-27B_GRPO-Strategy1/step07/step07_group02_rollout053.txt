import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def c_row(h, N, idx):
    k = np.arange(2 * N - 1) - (N - 1)
    im = idx - k
    ip = idx + k
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    return A - B

def proj(h, t):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - t) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        if len(idx) >= nh - n1:
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def bin_search(h, N, dl, max_it=5000, samp=120, kick_t=40, kick_n=4, two_swap_prob=0.3):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    best_c = np.inf
    best_h = h.copy()
    Cm = np.zeros((N, 2 * N - 1))
    for i in range(N):
        Cm[i] = c_row(h, N, i)
    stuck = 0
    for it in range(max_it):
        if time.time() > dl:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), samp)
        nz = min(len(zeros), samp)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        # 1-swap
        new = corr[None, None, :] + Cm[os_][None, :, :] + Cm[zs][:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        # 2-swap
        do_two = (np.random.random() < two_swap_prob) or (stuck > kick_t * 2)
        mv2 = P + 1.0
        mi2 = None
        if do_two and ns >= 2 and nz >= 2:
            # Sample pairs
            ntr = min(30, ns * (ns - 1) // 2, nz * (nz - 1) // 2)
            for _ in range(ntr):
                a, b = np.random.choice(ns, 2, replace=False)
                c2, d2 = np.random.choice(nz, 2, replace=False)
                delta = Cm[os_[a]] + Cm[os_[b]] + Cm[zs[c2]] + Cm[zs[d2]]
                val = float(np.max(corr + delta))
                if val < mv2:
                    mv2 = val
                    mi2 = (a, b, c2, d2)
        if mv2 < mv and mi2 is not None:
            a, b, c2, d2 = mi2
            h[os_[a]] = 0; h[os_[b]] = 0
            h[zs[c2]] = 1; h[zs[d2]] = 1
            Cm[os_[a]] = c_row(h, N, os_[a])
            Cm[os_[b]] = c_row(h, N, os_[b])
            Cm[zs[c2]] = c_row(h, N, zs[c2])
            Cm[zs[d2]] = c_row(h, N, zs[d2])
        elif mv < P - 1e-14:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
            Cm[os_[mi[1]]] = c_row(h, N, os_[mi[1]])
            Cm[zs[mi[0]]] = c_row(h, N, zs[mi[0]])
        elif stuck >= kick_t:
            for _ in range(kick_n):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0; h[q] = 1
                Cm[p] = c_row(h, N, p)
                Cm[q] = c_row(h, N, q)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stuck = 0
        else:
            break
    return best_h, best_c

def grad_opt(h, N, dl, k=60, lr0=0.01):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h))) * (2.0 / N)
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < dl:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr)) * (2.0 / N)
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        # Top-k gradient
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                if L < N:
                    grad[L:] += (1.0 - y[:N - L])
                    grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = proj(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0003)
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def c_row(h, N, idx):
    k = np.arange(2 * N - 1) - (N - 1)
    im = idx - k
    ip = idx + k
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    return A - B

def proj(h, t):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - t) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        if len(idx) >= nh - n1:
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def bin_search(h, N, dl, max_it=5000, samp=120, kick_t=40, kick_n=4, two_prob=0.25):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    best_c = np.inf
    best_h = h.copy()
    Cm = np.zeros((N, 2 * N - 1))
    for i in range(N):
        Cm[i] = c_row(h, N, i)
    stuck = 0
    L = 2 * N - 1
    for it in range(max_it):
        if time.time() > dl:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), samp)
        nz = min(len(zeros), samp)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        new = corr[None, None, :] + Cm[os_][None, :, :] + Cm[zs][:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        mv2 = P + 1.0
        mi2 = None
        do_two = (np.random.random() < two_prob) or (stuck > kick_t * 2)
        if do_two and ns >= 2 and nz >= 2:
            ntr = min(20, ns * (ns - 1) // 2, nz * (nz - 1) // 2)
            for _ in range(ntr):
                a, b = np.random.choice(ns, 2, replace=False)
                c2, d2 = np.random.choice(nz, 2, replace=False)
                delta = Cm[os_[a]] + Cm[os_[b]] + Cm[zs[c2]] + Cm[zs[d2]]
                val = float(np.max(corr + delta))
                if val < mv2:
                    mv2 = val
                    mi2 = (a, b, c2, d2)
        if mv2 < mv and mi2 is not None:
            a, b, c2, d2 = mi2
            h[os_[a]] = 0; h[os_[b]] = 0
            h[zs[c2]] = 1; h[zs[d2]] = 1
            Cm[os_[a]] = c_row(h, N, os_[a])
            Cm[os_[b]] = c_row(h, N, os_[b])
            Cm[zs[c2]] = c_row(h, N, zs[c2])
            Cm[zs[d2]] = c_row(h, N, zs[d2])
        elif mv < P - 1e-14:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
            Cm[os_[mi[1]]] = c_row(h, N, os_[mi[1]])
            Cm[zs[mi[0]]] = c_row(h, N, zs[mi[0]])
        elif stuck >= kick_t:
            for _ in range(kick_n):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0; h[q] = 1
                Cm[p] = c_row(h, N, p)
                Cm[q] = c_row(h, N, q)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stuck = 0
        else:
            break
    return best_h, best_c

def grad_opt(h, N, dl, k=50, lr0=0.008, beta_s=40.0):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h))) * (2.0 / N)
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    while time.time() < dl:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr)) * (2.0 / N)
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        # Soft-max gradient
        exp_c = np.exp(beta_s * (corr - np.max(corr)))
        w = exp_c / exp_c.sum()
        grad = np.zeros(N)
        for ai in range(len(corr)):
            wi = w[ai]
            if wi < 1e-8:
                continue
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                if L < N:
                    grad[L:] += wi * (1.0 - y[:N - L])
                    grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        h_new = proj(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.985
            beta = min(0.97, 0.88 + it * 0.0003)
            beta_s *= 0.95
    return best_h, best_c
