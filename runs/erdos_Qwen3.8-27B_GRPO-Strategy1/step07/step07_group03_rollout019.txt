def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    delta = np.zeros(2*N - 1)
    # For shift s, delta[s] = -(1-h[p+s])*(valid) + h[p-s]*(valid) + (1-h[q+s])*(valid) - h[q-s]*(valid)
    # shift index: L = s + N - 1
    # p+s valid: 0 <= p+s < N -> -p <= s < N-p -> N-1-p <= L < N+N-1-p -> L in [N-1-p, 2N-2-p]
    # p-s valid: 0 <= p-s < N -> p-N+1 <= s <= p -> N-1+p-N+1 <= L <= N-1+p -> L in [p-N+1+N-1, N-1+p] = [p, N-1+p]

    s = np.arange(-(N-1), N)  # shifts
    # p+s in [0,N): indices where 0 <= p+s < N
    valid_ps = (s >= -p) & (s < N - p)
    valid_pms = (s <= p) & (s >= p - N + 1)
    valid_qs = (s >= -q) & (s < N - q)
    valid_qms = (s <= q) & (s >= q - N + 1)
    
    # h[p+s] for valid: s maps to p+s
    delta[valid_ps] -= (1 - h[p + s[valid_ps]])
    delta[valid_pms] += h[p - s[valid_pms]]
    delta[valid_qs] += (1 - h[q + s[valid_qs]])
    delta[valid_qms] -= h[q - s[valid_qms]]
    return delta

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(777 + N)
    if n1 > nh:
        idx = np.where(h_bin == 1)[0]
        h_bin[rng.choice(idx, n1 - nh, replace=False)] = 0
    else:
        idx = np.where(h_bin == 0)[0]
        h_bin[rng.choice(idx, nh - n1, replace=False)] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm, pp = pos - sv, pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c, pp_c = np.clip(pm, 0, N - 1), np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_anneal(h, N, deadline, rng, max_sweeps=3000, sample_frac=1.0):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    corr = fft_corr(h)
    best_h = h.copy()
    best_c = float(corr.max())
    temp = 0.014
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp * 0.9955)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        P_cur = float(corr.max())
        n_ones_s = max(40, int(len(ones) * sample_frac))
        p_list = rng.choice(ones, min(n_ones_s, len(ones)), replace=False)
        best_new_P = P_cur
        best_move = None
        for p in p_list:
            base = corr - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_move = (int(p), int(zeros[mi]))
        # Structured 2-swap: sequential refinement
        if best_move is not None and len(ones) > 4:
            p1, q1 = best_move
            q1_idx = int(np.where(zeros == q1)[0][0])
            corr2 = corr - D[p1, :] - D_z[q1_idx, :]
            n_try2 = min(30, max(10, len(p_list) // 4))
            p2_list = rng.choice(ones, min(n_try2, len(ones)), replace=False)
            for p2 in p2_list:
                if p2 == p1:
                    continue
                base2 = corr2 - D[p2, :]
                cand2 = base2[None, :] - D_z
                mask = np.ones(len(zeros), dtype=bool)
                mask[q1_idx] = False
                if mask.sum() < 2:
                    continue
                npk2 = cand2[mask, :].max(axis=1)
                mi2 = int(np.argmin(npk2))
                if npk2[mi2] < best_new_P:
                    best_new_P = float(npk2[mi2])
                    best_move = (p1, q1, int(p2), int(zeros[mi2]))
        # Diversity jump when stalled
        if best_move is None or (stall > 25 and rng.random() < 0.4):
            n_div = 2 + rng.randint(0, 3)
            if len(ones) > n_div and len(zeros) > n_div:
                p_div = rng.choice(ones, min(n_div, len(ones)-1), replace=False)
                q_div = rng.choice(zeros, min(n_div, len(zeros)-1), replace=False)
                h_save = h.copy()
                h[p_div] = 0.0
                h[q_div] = 1.0
                corr_new = fft_corr(h)
                c_new = float(corr_new.max())

                    corr = corr_new
                    best_move = None
                else:
                    h = h_save
                    corr = fft_corr(h)
            if best_move is None:
                stall += 5
                if stall > 80:
                    break
                continue
        # Apply move
        if len(best_move) == 2:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
        elif len(best_move) == 4:
            h[best_move[0]] = 0.0
            h[best_move[2]] = 0.0
            h[best_move[1]] = 1.0
            h[best_move[3]] = 1.0
        corr = fft_corr(h)
        P_cur = float(corr.max())
        if P_cur < best_c - 1e-13:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def nesterov_refine(h, N, deadline, k=50, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.97
            beta = min(0.96, 0.85 + it * 0.0002)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def perturb_binary(h, N, n_flip, rng):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_flip):
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        h[p] = 0.0
        h[q] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h
