import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_topk(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.02, beta0=0.9):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        grad = grad_topk(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.996
            beta = min(0.98, 0.84 + it * 0.00014)
        if it % 200 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(10.0, min(500.0, T * (0.8 if gap < 0.003 else 1.12)))
    return best_h, best_pk * (2.0 / N)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(99)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_sa(h_init, N, deadline, n_shifts=180, n_sample=250, t_start=0.6, t_end=0.002):
    h = ensure_binary(h_init, N)
    rng = np.random.RandomState(7)
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]

    def calc_D():
        pm = p_all - s_all
        pp = p_all + s_all
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

    D = calc_D()
    sweep = 0
    while time.time() < deadline:
        if sweep % 15 == 0:
            D = calc_D()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        top_idx = np.argpartition(c, -n_shifts)[-n_shifts:]
        c_t = c[top_idx]
        D_t = D[:, top_idx]
        if len(ones) > n_sample:
            p_list = rng.choice(ones, n_sample, replace=False)
        else:
            p_list = ones
        Dz_t = D_t[zeros, :]
        temp = t_start * (t_end / t_start) ** (float(sweep) / 8000.0)
        best_nP = np.inf
        best_pq = None
        for p in p_list:
            base = c_t + D_t[p, :]
            cand = base[None, :] - Dz_t
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP:
                best_nP = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        c = fft_corr(h)
        nP = float(c.max())
        delta = nP - P
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-9)):
            P = nP
            if P < best_c - 1e-12:
                best_c = P
                best_h = h.copy()
        else:
            h[p] = 1.0
            h[q] = 0.0
        sweep += 1
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
