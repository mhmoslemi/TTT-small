def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def compute_D(h, N):
    N2 = 2 * N - 1
    center = N - 1
    D = np.zeros((N, N2))
    for m in range(N2):
        shift = m - center
        if shift > 0:
            if N - shift > 0:
                D[:N - shift, m] += 1.0 - h[shift:]
            D[shift:, m] -= h[:N - shift]
        elif shift < 0:
            s = -shift
            D[s:, m] += 1.0 - h[:N - s]
            D[:N - s, m] -= h[s:]
        else:
            D[:, m] = 1.0 - 2.0 * h
    return D

def lse_nesterov(h, N, deadline, beta_start=15, beta_end=250, top_k=35, lr0=0.007):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = beta_start
    lr = lr0
    it = 0
    stall = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        top = np.argpartition(corr, -top_k)[-top_k:]
        scaled = beta * corr[top]
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        grad = np.zeros(N)
        for idx_m, wi in zip(top, w):
            shift = int(idx_m - center)
            if shift > 0:
                if N - shift > 0:
                    grad[:N - shift] += wi * (1.0 - y[shift:])
                grad[shift:] -= wi * y[:N - shift]
            elif shift < 0:
                s = -shift
                grad[s:] += wi * (1.0 - y[:N - s])
                grad[:N - s] -= wi * y[s:]
            else:
                grad += wi * (1.0 - 2.0 * y)
        h_new = project_sum(y - lr * grad, target)
        m = 0.9 * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            beta = min(beta_end, beta * 1.18)
            lr *= 0.994
        elif it % 500 == 0:
            lr = max(0.001, lr * 0.9)
    return best_h, best_pk * dx

def binary_anneal(h, N, deadline, max_sweeps=600, sample_frac=1.0, two_swap=0.3, rng=None):
    h = ensure_sum(h, N)
    if rng is None:
        rng = np.random.RandomState(7)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P_cur
        best_pq = None
        for p in p_list:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        best_2p = None
        if len(ones) >= 4 and len(zeros) >= 4 and rng.random() < two_swap:
            n_try = min(300, max(60, N // 2))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = (p1, p2, q1, q2)
        if best_2p is not None:
            h[best_2p[0]] = h[best_2p[1]] = 0.0
            h[best_2p[2]] = h[best_2p[3]] = 1.0
        elif best_pq is not None and best_new_P < P_cur - 1e-12:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            if stall > 35:
                n_esc = 4
                for _ in range(n_esc):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 12:
                break
            else:
                stall += 1
                continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-13:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def block_swap_explore(h, N, rng, n_blocks=30):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) < 4 or len(zeros) < 4:
        return h
    for _ in range(n_blocks):
        bs = rng.randint(2, 5)
        if len(ones) < bs * 2 or len(zeros) < bs * 2:
            break
        oi = rng.choice(len(ones) - bs + 1)
        zi = rng.choice(len(zeros) - bs + 1)
        block_o = set(ones[oi:oi + bs].tolist())
        block_z = set(zeros[zi:zi + bs].tolist())
        if block_o & block_z:
            continue
        for p in block_o:
            h[p] = 0.0
        for q in block_z:
            h[q] = 1.0
    return h
