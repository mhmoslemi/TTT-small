import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def binarize_sum(h, N):
    nh = N // 2
    idx = np.argsort(-h)
    out = np.zeros(N)
    out[idx[:nh]] = 1.0
    return out

def grad_top(h, top_k=25):
    N = len(h)
    c = fft_corr(h)
    top = np.argpartition(c, -top_k)[-top_k:]
    mc = c[top].max()
    w = np.exp((c[top] - mc) / 0.06)
    w /= w.sum()
    center = N - 1
    g = np.zeros(N)
    for i, k in enumerate(top):
        L = int(k - center)
        if L == 0:
            g += w[i] * (1.0 - 2.0 * h)
        elif L > 0:
            g[:N-L] += w[i] * (1.0 - h[L:])
            g[L:] -= w[i] * h[:N-L]
        else:
            M = -L
            g[M:] += w[i] * (1.0 - h[:N-M])
            g[:N-M] -= w[i] * h[M:]
    return g, c

def adam_run(h, N, deadline, top_k=30, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N); v = np.zeros(N)
    t = 0
    while time.time() < deadline:
        t += 1
        g, c = grad_top(h, top_k)
        pk = float(c.max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        m = 0.9*m + 0.1*g
        v = 0.999*v + 0.001*g*g
        mhat = m / (1 - 0.9**t)
        vhat = v / (1 - 0.999**t)
        lr = lr0 / (1.0 + 0.0003 * t)
        h = project_sum(h - lr * mhat / (np.sqrt(vhat) + 1e-8), target)
    return best_h, best_pk * dx

def sa_run(h, N, deadline, T0=3.0, Tcool=0.993, T_min=0.015,
           max_sweeps=20000, mult_prob=0.25, kick_every=200, kick_size=6,
           n_local=12, seed=42):

    rng = np.random.RandomState(seed)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.35
            continue
        r = rng.random()
        if r < mult_prob and len(ones) >= 6 and len(zeros) >= 6:
            # 2-for-2 move
            ps = rng.choice(ones, 2, replace=False)
            qs = rng.choice(zeros, 2, replace=False)
            h[ps] = 0.0; h[qs] = 1.0
            Pn = float(fft_corr(h).max())
            d = Pn - P
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                P = Pn
            else:

        else:

            if best_d < 0 or rng.random() < np.exp(-max(0.0,best_d) / max(T, 1e-10)):
                P += best_d
            else:
                h[p] = 1.0; h[best_q] = 0.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def block_shift(h, N, rng, n_shifts=3, block_len=4):
    """Shift random blocks of ones."""
    h = h.copy()
    for _ in range(n_shifts):
        ones = np.where(h > 0.5)[0]
        if len(ones) < block_len:
            break
        start = rng.randint(len(ones))
        # Find contiguous ones
        block = [ones[start]]
        for j in range(1, min(block_len, len(ones)-start)):
            if ones[start+j] == ones[start] + 1:
                block.append(ones[start+j])
        if len(block) >= 2:
            shift = rng.randint(1, 3) * (1 if rng.random() < 0.5 else -1)
            new_pos = [x + shift for x in block]
            if all(0 <= x < N for x in new_pos) and all(h[x] < 0.5 for x in new_pos):
                h[block] = 0.0
                h[new_pos] = 1.0
    return h

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0/len(h))

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s-target) < 1e-10: return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def upsample(h, No, Nn):
    return np.interp(np.linspace(0,2,Nn,endpoint=False), np.linspace(0,2,No,endpoint=False), h)

def binarize(h, N):
    idx = np.argsort(-h)
    out = np.zeros(N); out[idx[:N//2]] = 1.0
    return out

def grad_top(h, tk=25):
    N = len(h); c = fft_corr(h)
    top = np.argpartition(c, -tk)[-tk:]
    mc = c[top].max()
    w = np.exp((c[top]-mc)/0.06); w /= w.sum()
    ctr = N-1; g = np.zeros(N)
    for i,k in enumerate(top):
        L = int(k-ctr)
        if L==0: g += w[i]*(1-2*h)
        elif L>0:
            g[:N-L] += w[i]*(1-h[L:]); g[L:] -= w[i]*h[:N-L]
        else:
            M=-L; g[M:] += w[i]*(1-h[:N-M]); g[:N-M] -= w[i]*h[M:]
    return g, c

def adam(h, N, dl, tk=30, lr0=0.012):
    h = h.astype(float).copy(); tgt=N/2.0; dx=2.0/N
    bh=h.copy(); bp=float(np.max(fft_corr(h)))
    m=np.zeros(N); v=np.zeros(N); t=0
    while time.time()<dl:
        t+=1; g,c=grad_top(h,tk); pk=float(c.max())
        if pk<bp-1e-13: bp=pk; bh=h.copy()
        m=.9*m+.1*g; v=.999*v+.001*g*g
        lr=lr0/(1+3e-4*t)
        h=project_sum(h-lr*m/(1-.9**t)/(np.sqrt(v/(1-.999**t))+1e-8), tgt)
    return bh, bp*dx

def compute_D(h,N):
    sh=np.arange(2*N-1)-(N-1)
    pos=np.arange(N)[:,None]; sv=sh[None,:]
    pm=pos-sv; pp=pos+sv
    vm=(pm>=0)&(pm<N); vp=(pp>=0)&(pp<N)
    hm=np.where(vm,h[np.clip(pm,0,N-1)],0)
    hp=np.where(vp,h[np.clip(pp,0,N-1)],0)
    return -(1-hm)*vm+hp*vp

def sa_d(h,N,dl,T0=3,Tc=.993,Tm=.01,ms=30000,mp=.22,ke=180,ks=5,sf=1,seed=42):
    h=h.astype(float).copy(); rng=np.random.RandomState(seed)
    bc=float(np.max(fft_corr(h))); bh=h.copy(); T=T0; st=0
    for sw in range(ms):

        c=fft_corr(h); P=float(c.max())
        if P<bc-1e-13: bc=P; bh=h.copy(); st=0

        ones=np.where(h>.5)[0]; zeros=np.where(h<.5)[0]

        if st>=ke:

        if rng.random()<mp and len(ones)>=6:
            # 2-for-2
            ps=rng.choice(ones,2,replace=False); qs=rng.choice(zeros,2,replace=False)
            h[ps]=0; h[qs]=1
            Pn=float(fft_corr(h).max()); d=Pn-P
            if d<0 or rng.random()<np.exp(-d/max(T,1e-10)): P=Pn
            else: h[ps]=1; h[qs]=0
        else:
            D=compute_D(h,N); Dz=D[zeros,:]
            n_s=len(ones) if sf>=1 else max(30,int(len(ones)*sf))
            pl=ones if sf>=1 else rng.choice(ones,n_s,replace=False)
            bnp=P+1; bpq=None
            for p in pl:

                cand=base[None,:]-Dz
                npk=cand.max(axis=1)
                mi=int(np.argmin(npk))
                if npk[mi]<bnp: bnp=float(npk[mi]); bpq=(int(p),int(zeros[mi]))
            if bpq is None: break
            d=bnp-P
            if d<-1e-14 or (T>Tm and rng.random()<np.exp(-max(0,d)/max(T,1e-10))):
                h[bpq[0]]=0; h[bpq[1]]=1
            else:
                p=ones[rng.randint(len(ones))]; q=zeros[rng.randint(len(zeros))]
                h[p],h[q]=0,1
        T=max(Tm,T*Tc)
    return bh, bc*(2.0/N)

def perturb(h,N,rng,n_pts=8):
    h=h.copy()
    p=np.where(h>.5)[0]; q=np.where(h<.5)[0]
    for _ in range(n_pts):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def upsample(h, No, Nn):
    return np.interp(np.linspace(0, 2, Nn, endpoint=False),
                     np.linspace(0, 2, No, endpoint=False), h)

def binarize(h, N):
    idx = np.argsort(-h)
    out = np.zeros(N)
    out[idx[:N // 2]] = 1.0
    return out

def grad_top(h, tk=25):
    N = len(h)
    c = fft_corr(h)
    top = np.argpartition(c, -tk)[-tk:]
    mc = c[top].max()
    w = np.exp((c[top] - mc) / 0.06)
    w /= w.sum()
    ctr = N - 1
    g = np.zeros(N)
    for i, k in enumerate(top):
        L = int(k - ctr)
        if L == 0:
            g += w[i] * (1.0 - 2.0 * h)
        elif L > 0:
            g[:N - L] += w[i] * (1.0 - h[L:])
            g[L:] -= w[i] * h[:N - L]
        else:
            M = -L
            g[M:] += w[i] * (1.0 - h[:N - M])
            g[:N - M] -= w[i] * h[M:]
    return g, c

def adam_run(h, N, dl, tk=30, lr0=0.013):
    h = h.astype(float).copy()
    tgt = N / 2.0
    dx = 2.0 / N
    bh = h.copy()
    bp = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    while time.time() < dl:
        t += 1
        g, c = grad_top(h, tk)
        pk = float(c.max())
        if pk < bp - 1e-13:
            bp = pk
            bh = h.copy()
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        lr = lr0 / (1.0 + 3e-4 * t)
        h = project_sum(h - lr * m / (1 - 0.9**t) / (np.sqrt(v / (1 - 0.999**t)) + 1e-8), tgt)
    return bh, bp * dx

def compute_D(h, N):
    sh = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = sh[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    hp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def sa_d(h, N, dl, T0=3.0, Tc=0.993, Tm=0.01, ms=30000,
         mp=0.22, ke=170, ks=5, sf=1.0, seed=42):
    h = h.astype(float).copy()
    rng = np.random.RandomState(seed)
    bc = float(np.max(fft_corr(h)))
    bh = h.copy()
    T = T0
    st = 0
    for sw in range(ms):
        if time.time() > dl:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < bc - 1e-13:
            bc = P
            bh = h.copy()
            st = 0
        else:
            st += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        if st >= ke:
            for _ in range(ks):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            st = 0
            T = T0 * 0.35
            c = fft_corr(h)
            P = float(c.max())
            continue
        if rng.random() < mp and len(ones) >= 6:
            ps = rng.choice(ones, 2, replace=False)
            qs = rng.choice(zeros, 2, replace=False)
            h[ps] = 0.0
            h[qs] = 1.0
            Pn = float(fft_corr(h).max())
            d = Pn - P
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                P = Pn
            else:
                h[ps] = 1.0
                h[qs] = 0.0
        else:
            D = compute_D(h, N)
            Dz = D[zeros, :]
            n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))
            pl = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
            bnp = P + 1.0
            bpq = None
            for p in pl:
                base = c - D[p, :]
                cand = base[None, :] - Dz
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
                if npk[mi] < bnp:
                    bnp = float(npk[mi])
                    bpq = (int(p), int(zeros[mi]))
            if bpq is None:
                break
            d = bnp - P
            if d < -1e-14 or (T > Tm and rng.random() < np.exp(-max(0.0, d) / max(T, 1e-10))):
                h[bpq[0]] = 0.0
                h[bpq[1]] = 1.0
            else:
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
        T = max(Tm, T * Tc)
    return bh, bc * (2.0 / N)

def perturb(h, N, rng, n_pts=8):
    h = h.copy()
    p = np.where(h > 0.5)[0]
    q = np.where(h < 0.5)[0]
    for _ in range(n_pts):
        if len(p) < 2 or len(q) < 2:
            break
        h[p[rng.randint(len(p))]], h[q[rng.randint(len(q))]] = 0.0, 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
