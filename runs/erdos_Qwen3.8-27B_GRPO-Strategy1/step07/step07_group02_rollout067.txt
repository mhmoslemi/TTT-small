import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D1(h, N):
    k_vals = np.arange(-(N-1), N)
    D1 = np.zeros((N, 2*N-1))
    i_idx = np.arange(N)
    for i in range(N):
        ik = i + k_vals
        v1 = (ik >= 0) & (ik < N)
        D1[i, v1] -= (1.0 - h[ik[v1]])
        ik2 = i - k_vals
        v2 = (ik2 >= 0) & (ik2 < N)
        D1[i, v2] += h[ik2[v2]]
    return D1

def update_D1(D1, h, p, dv, N):
    """Update D1 after h[p] changes by dv (dv = -1 for 1→0, +1 for 0→1)."""
    # D1[i,k] has term -(1-h[i+k]) when i+k=p, k=p-i
    # and term +h[i-k] when i-k=p, k=i-p
    # For h[p] changing by dv:
    # Term -(1-h[i+k]) = -1+h[i+k]: change = +dv when i+k=p
    # Term +h[i-k]: change = +dv when i-k=p
    
    # i+k = p => k = p-i, valid when 0<=p-i<2N-1 (i.e., N-1 <= i-p <= N-1... let me just check 0<=k_idx<2N-1)
    # k_idx = (p-i) + (N-1)
    i_range = np.arange(N)
    k1 = p - i_range  # k values for i+k=p
    k1_idx = k1 + (N-1)
    valid1 = (k1_idx >= 0) & (k1_idx < 2*N-1)
    D1[i_range[valid1], k1_idx[valid1]] += dv  # because -(1-h[i+k]) changes by +dv when h[p] changes by dv
    
    k2 = i_range - p  # k values for i-k=p

def fft_corr(h):

    L = 2 * N - 1

    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def compute_D1(h, N):

            break
        ob = one_blocks[np.random.randint(len(one_blocks))]
        zb = zero_blocks[np.random.randint(len(zero_blocks))]
        # Compute delta for swapping blocks
        delta = np.zeros(2 * N - 1)
        for p in ob:
            delta += D1[p]
        for q in zb:
            delta -= D1[q]
        newP = float(np.max(corr + delta))
        d = newP - P
        if d < -1e-10 or (temp > 1e-6 and np.random.random() < np.exp(-max(0.0, -d) / max(temp, 1e-8))):
            for p in ob:
                h[p] = 0
            for q in zb:
                h[q] = 1
            corr += delta
            P = float(np.max(corr))
            for p in ob:
                update_D1(D1, p, -1, N)
            for q in zb:
                update_D1(D1, q, 1, N)
            if P < best_c - 1e-10:
                best_c = P
                best_h = h.copy()
        temp *= cooling
    return best_h, best_c

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    best_h = (h0 > 0.5).astype(float)
    target = N0 // 2

        best_h[np.random.choice(idx, target - cur_sum, replace=False)] = 1
    best_c5 = compute_c5(best_h, N0)

    d1 = min(deadline - 450, t0 + 300)

    if time.time() < deadline - 350:

    if time.time() < deadline - 200:
        h5 = best_h.copy()
        ones5 = np.where(h5 == 1)[0]
        zeros5 = np.where(h5 == 0)[0]
        for _ in range(20):
            h5[int(np.random.choice(ones5))] = 0
            h5[int(np.random.choice(zeros5))] = 1

        d5 = min(deadline - 180, time.time() + 180)
