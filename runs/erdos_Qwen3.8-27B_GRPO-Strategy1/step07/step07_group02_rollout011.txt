    # Log-sum-exp for numerical stability
    c_max = np.max(corr)
    log_sum = c_max + np.log(np.sum(np.exp(T * (corr - c_max))))
    val = log_sum / T  # This is the soft max
    # Gradient: softmax-weighted average of individual gradients
    weights = np.exp(T * (corr - c_max))

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def soft_max_grad(h, N, T):
    """Compute soft-max objective and its gradient."""
    corr = fft_corr(h)
    c_max = float(np.max(corr))
    shifted = T * (corr - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    val = c_max + np.log(exps.sum()) / T
    grad = np.zeros(N)
    center = N - 1
    thresh = max(1e-10, 1.0 / (N * 10))
    active = np.where(weights > thresh)[0]
    for k_idx in active:
        w = weights[k_idx]
        lag = k_idx - center
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:N - lag] += w * (1.0 - h[lag:])
            grad[lag:] += -w * h[:N - lag]
        else:
            m = -lag
            grad[m:] += w * (1.0 - h[:N - m])
            grad[:N - m] += -w * h[m:]
    return val, grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def soft_max_obj_grad(h, N, T):
    corr = fft_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(T * (corr - cmax), -30, 30)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    val = cmax + np.log(exps.sum()) / T
    grad = np.zeros(N)
    center = N - 1
    thresh = 1.0 / (N * 5)
    for ki in range(len(corr)):
        if w[ki] < thresh:
            continue
        wi = w[ki]
        lag = ki - center
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:N-lag] += wi * (1.0 - h[lag:])
            grad[lag:] += -wi * h[:N-lag]
        else:
            m = -lag
            grad[m:] += wi * (1.0 - h[:N-m])
            grad[:N-m] += -wi * h[m:]
    return val, grad

def adam_optim(h, N, deadline, T_start=20, T_end=500, lr_init=0.005):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    t = 0
    b1, b2 = 0.9, 0.999
    eps = 1e-7
    while time.time() < deadline:
        T = T_start + (T_end - T_start) * min(1.0, t / 3000)
        _, grad = soft_max_obj_grad(h, N, T)
        t += 1
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * grad * grad
        m1h = m1 / (1 - b1**t)
        m2h = m2 / (1 - b2**t)
        h_new = project_01(h - lr_init * m1h / (np.sqrt(m2h) + eps), target)
        h = h_new
        if t % 10 == 0:

            if c < best_c5:

                best_h = h.copy()
        if t % 1000 == 0:
            lr_init *= 0.9
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5

def binary_local(h, N, deadline, sample_sz=80, max_stuck=40):
