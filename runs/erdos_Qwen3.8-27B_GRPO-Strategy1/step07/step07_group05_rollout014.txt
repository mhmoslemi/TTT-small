        # Softmax weights

        grad = np.zeros(N)

                grad[:N-m] -= wm * h[m:]
        
        # Update

        target = N / 2.0

            best_h = h.copy()

        lr *= 0.999
        alpha *= 0.997

def compute_R(h):
    N = len(h)
    R = np.empty(N)
    for m in range(N):
        R[m] = np.dot(h[m:], h[:N - m])
    return R

def compute_P(h):
    N = len(h)
    P = np.zeros((N, N))
    P[:, 0] = 2.0 * h
    for m in range(1, N):
        P[m:, m] = h[:N - m]
        P[:N - m, m] += h[m:]
    return P

def c5_from_R(R, N):
    return float(np.max(N / 2.0 - R[1:])) * 2.0 / N

def make_binary(h, N):
    h = np.asarray(h, dtype=float)
    nh = N // 2
    idx = np.argsort(-h)
    out = np.zeros(N)
    out[idx[:nh]] = 1.0
    return out

def greedy_local_search(h, N, deadline, max_steps=100000, stall_reset=300):
    nh = N // 2
    h = h.copy()
    R = compute_R(h)
    P = compute_P(h)
    cur_c5 = c5_from_R(R, N)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    stall = stall_reset
    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        Q_max = np.full((len(ones), len(zeros)), -np.inf)
        for m in range(1, N):
            A = N / 2.0 - R[m] + P[ones, m]
            B = P[zeros, m]
            np.maximum(Q_max, A[:, None] - B[None, :], out=Q_max)
        flat = int(np.argmin(Q_max))
        ii, jj = np.unravel_index(flat, Q_max.shape)
        new_val = float(Q_max[flat]) * 2.0 / N
        bi = int(ones[ii])
        bj = int(zeros[jj])
        h[bi] = 0.0
        h[bj] = 1.0
        R = compute_R(h)
        P = compute_P(h)
        cur_c5 = c5_from_R(R, N)
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > stall:
            for _ in range(8):
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi], h[zi] = h[zi], h[oi]
            R = compute_R(h)
            P = compute_P(h)
            cur_c5 = c5_from_R(R, N)
            no_improve = 0
            stall = min(stall + 200, 3000)
    return best_h, best_c5

def continuous_refine(h, N, deadline, lr0=0.008):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = c5_from_R(compute_R(h), N)
    lr = lr0
    alpha = 0.05
    it = 0
    while time.time() < deadline:
        it += 1
        R = compute_R(h)
        vals = N / 2.0 - R[1:]
        peak = vals.max()
        w = np.exp((vals - peak) / alpha)
        w /= w.sum()
        grad = np.zeros(N)
        for idx in range(1, N):
            wm = w[idx - 1]
            if wm < 1e-5:
                continue
            m = idx
            grad[m:] += wm * (1.0 - h[:N - m])
            grad[:N - m] -= wm * h[m:]
        h = np.clip(h - lr * grad, 0.0, 1.0)
        target = N / 2.0
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = (lo + hi) / 2.0
            if np.clip(h - mid, 0.0, 1.0).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
        R = compute_R(h)
        c5 = c5_from_R(R, N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.998
        alpha *= 0.995
    return best_h, best_c5

def structured_init(N, variant, seed):
    rng = np.random.RandomState(seed)
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[rng.choice(N, nh, replace=False)] = 1.0
    elif variant == 1:
        x = np.arange(N)
        v = np.sin(2 * np.pi * x * 7.0 / N) + 0.6 * np.cos(2 * np.pi * x * 13.0 / N)
        h = (v > np.median(v)).astype(float)
        h = make_binary(h, N)
    elif variant == 2:
        x = np.arange(N)
        v = np.sin(2 * np.pi * x * 3.0 / N) + np.sin(2 * np.pi * x * 11.0 / N + 0.5)
        h = (v > np.median(v)).astype(float)
        h = make_binary(h, N)
    else:
        h = (np.arange(N) % 3 != 0).astype(float)
        h = make_binary(h, N)
    return h

        hc = best_h.astype(float) + np.random.randn(N0) * 0.01
        hc = np.clip(hc, 0, 1)
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = (lo + hi) / 2.0
            if np.clip(hc - mid, 0, 1).sum() > N0 / 2.0:
                lo = mid
            else:
                hi = mid
        hc = np.clip(hc - (lo + hi) / 2.0, 0, 1)
        d2 = min(deadline - 200, time.time() + 120)
        hc, c2 = continuous_refine(hc, N0, d2)
        if c2 < best_c5:
            best_c5, best_h = c2, hc.copy()

    # Phase 3: Another binary greedy round from best
    if time.time() < deadline - 150:
        h3 = make_binary(best_h, len(best_h))
        d3 = min(deadline - 100, time.time() + 200)
        h3, c3 = greedy_local_search(h3, len(best_h), d3, stall_reset=150)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Multi-N exploration
    for N_try in [190, 240, 320, 400]:
        if time.time() > deadline - 60:
            break
        if N_try == N0:
            continue
        if N_try % 2 != 0:
            N_try += 1
        best_loc_c = np.inf
        best_loc_h = None
        for vi in range(3):
            if time.time() > deadline - 30:
                break
            hv = structured_init(N_try, vi, seed + vi * 100 + N_try)
            d_sub = min(deadline - 20, time.time() + 50)
            hr, cr = greedy_local_search(hv, N_try, d_sub, stall_reset=100)
            if cr < best_loc_c:
                best_loc_c, best_loc_h = cr, hr.copy()
        if best_loc_h is not None and best_loc_c < best_c5:
            best_c5, best_h = best_loc_c, best_loc_h

    # Phase 5: Final greedy polish
    hF = make_binary(best_h, len(best_h))
    d5 = deadline - 2
    hF, c5F = greedy_local_search(hF, len(hF), d5, stall_reset=100)
    c5F = c5_from_R(compute_R(hF), len(hF))
    if c5F < best_c5:
        best_c5, best_h = c5F, hF

    # Ensure valid output
    hB = make_binary(best_h, len(best_h) // 2 * 2)
    cB = c5_from_R(compute_R(hB), len(hB))
    if cB < best_c5:
        best_c5, best_h = cB, hB

    return best_h, best_c5, len(best_h)
