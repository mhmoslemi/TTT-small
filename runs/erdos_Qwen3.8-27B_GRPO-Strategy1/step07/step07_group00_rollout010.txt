def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=30, beta_init=0.8):
    h = h_in.astype(float).copy()
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = beta_init
    it = 0
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:

            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[:N-L] += (1.0 - y[L:])
                grad[L:] += (-y[:N-L])
            else:
                M = -L
                grad[M:] += (1.0 - y[:N-M])
                grad[:N-M] += (-y[M:])

        if it % 300 == 0:

            beta = min(0.93, beta_init + it * 0.0001)
    
    return best_h, best_pk * (2.0 / N)
