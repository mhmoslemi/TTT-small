import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def compute_flip_matrix(h):
    N = len(h)
    L_total = 2 * N - 1
    center = N - 1
    D = np.zeros((N, L_total))
    L_vals = np.arange(L_total) - center
    for i in range(N):
        valid1 = (i - L_vals >= 0) & (i - L_vals < N)
        valid2 = (i + L_vals >= 0) & (i + L_vals < N)
        # (1-h[i-L]) where valid
        hl = np.where(valid1, i - L_vals, 0)
        hr = np.where(valid2, i + L_vals, 0)
        D[i] = valid1.astype(float) * (1.0 - h[hl]) - valid2.astype(float) * h[hr]
    return D

def full_grad(h, corr, n_top=60):
    N = len(h)
    center = N - 1
    max_corr = float(np.max(corr))
    top_idx = np.argsort(corr)[-n_top:]
    weights = np.zeros(len(corr))
    for m in top_idx:
        weights[m] = 1.0 / (1.0 + (max_corr - corr[m]) * 50.0)
    grad = np.zeros(N)
    L_vals = np.arange(2*N-1) - center
    for m in top_idx:
        w = weights[m]
        L = m - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L < N:
            hi_idx = np.arange(L, N)
            lo_idx = np.arange(0, N - L)
            grad[hi_idx] += w * (1.0 - h[hi_idx - L])
            grad[lo_idx] -= w * h[lo_idx + L]
    return grad
