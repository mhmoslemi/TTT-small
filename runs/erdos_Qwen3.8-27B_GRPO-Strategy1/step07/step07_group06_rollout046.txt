    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 500:
            # Perturb
            for _ in range(5):
                i = np.random.randint(N)
                if h[i] == 1:
                    j = np.random.randint(N)
                    if h[j] == 0:
                        h[i], h[j] = 0, 1
            stall = 0
            continue
        
        corr = fft_corr(h)
        mx = float(corr.max())
        if mx < best_c - 1e-12:
            best_c = mx
            best_h = h.copy()
        
        # Active set: lags where corr is within 0.5 of max
        active = np.where(corr >= mx - 0.5)[0]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute "damage" of moving each 1 to 0: sum over active lags of contribution
        # For position i going 1→0:
        #   It was contributing h[i+k]=(1) at lag -k (if i+k in range) to corr[N-1-k]
        #   And (1-h[i])=0 at lag +k contributions
        # Actually, let me just compute the gradient numerically for speed
        
        # Compute gradient of max (approximated by active set average)
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            k = int(ai) - ctr
            grad += lin_grad(k, h, N)
        grad /= len(active)
        
        # Best swap: minimize grad[i] - grad[j] for i in ones, j in zeros
        # i.e., maximize grad[j] - grad[i]
        gi = grad[ones]
        gj = grad[zeros]
        
        # Top candidates
        k_o = min(ntry, len(gi))
        k_z = min(ntry, len(gj))
        top_o = np.argsort(gi)[:k_o]  # ones with lowest gradient (best to remove)
        top_z = np.argsort(gj)[-k_z:]  # zeros with highest gradient (best to add)
        
        bd = -np.inf
        bi, bj = -1, -1
        for oi in top_o:
            for zi in top_z:
                oi2 = int(ones[oi])
                zj = int(zeros[zi])
                h[oi2], h[zj] = 0.0, 1.0
                nc = float(np.max(fft_corr(h)))
                h[oi2], h[zj] = 1.0, 0.0
                d = mx - nc
                if d > bd:
                    bd, bi, bj = d, oi2, zj
        
        # Also try a few random
        for _ in range(3):
            oi = int(np.random.randint(len(ones)))
            zi = int(np.random.randint(len(zeros)))
            oi2, zj = int(ones[oi]), int(zeros[zi])
            if oi2 == bj and zj == bi:
                continue
            h[oi2], h[zj] = 0.0, 1.0
            nc = float(np.max(fft_corr(h)))
            h[oi2], h[zj] = 1.0, 0.0
            d = mx - nc
            if d > bd:
                bd, bi, bj = d, oi2, zj
        
        if bi >= 0 and bd > 0:
            h[bi], h[bj] = 0.0, 1.0
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_opt(h, N, deadline, T=40.0, max_outer=3, lam0=5.0):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lam = lam0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 250, 'maxfun': 600, 'ftol': 1e-15})
        h = proj01(res.x, target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lam *= 3.0
    return best_h, best_c

def grad_bin_search(h, N, deadline, max_iters=4000, ntry=20):
    h = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0].copy()
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0.0
    elif n1 < target:
        idx = np.where(h == 0)[0].copy()
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1.0
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 400:
            for _ in range(5):
                i = int(np.random.randint(N))
                j = int(np.random.randint(N))
                if h[i] == 1.0 and h[j] == 0.0:
                    h[i], h[j] = 0.0, 1.0
            stall = 0
            continue
        corr = fft_corr(h)
        mx = float(corr.max())
        if mx < best_c - 1e-13:
            best_c = mx
            best_h = h.copy()
        active = np.where(corr >= mx - 0.3)[0]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai) - ctr, h, N)
        grad /= len(active)
        gi = grad[ones]
        gj = grad[zeros]
        kn = min(ntry, len(gi))
        kn2 = min(ntry, len(gj))
        top_o = np.argsort(gi)[:kn]
        top_z = np.argsort(gj)[-kn2:]
        bd = -np.inf
        bi, bj = -1, -1
        for oi in top_o:
            for zi in top_z:
                oi2, zj = int(ones[oi]), int(zeros[zi])
                h[oi2], h[zj] = 0.0, 1.0
                nc = float(np.max(fft_corr(h)))
                h[oi2], h[zj] = 1.0, 0.0
                d = mx - nc

        for _ in range(4):

            stall += 1

def subgrad_ne(h, N, deadline, alpha0=0.035, beta=0.92, atol=1e-5, max_it=6000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < max_it:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai) - ctr, y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            alpha *= 0.94
            beta = min(0.98, 0.88 + 0.0002 * it)
    return best_h, best_pk * (2.0 / N)
