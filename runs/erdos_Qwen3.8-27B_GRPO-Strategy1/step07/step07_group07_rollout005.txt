import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

            no_imp = 0
        else:

        temp = max(0.001, 0.02 * (0.998 ** no_imp))
        
        ones = np.where(h == 1)[0]

        top_lags = (top_idx - center).astype(int)

        os_ = np.random.choice(ones, ns, replace=False)

    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()

            stale = 0
        center = N - 1

        grad = np.zeros(N)
        for m in top_idx:
            M = abs(int(m) - center)
            if M == 0:

        h_new = project(h + mom * v - lr0 * grad, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_il_vec(h, deadline, n_crit=30, s_sz=50, max_iters=10000):
    N = len(h)
    center = N - 1
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    no_imp = 0
    it = 0
    while time.time() < deadline and it < max_iters and no_imp < 800:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        temp = max(0.0005, 0.015 * (0.997 ** no_imp))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argsort(corr)[-n_crit:]
        top_lags = (top_idx - center).astype(int)
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        a_pL = os_[:, None] + top_lags[None, :]
        a_mL = os_[:, None] - top_lags[None, :]
        b_pL = zs[:, None] + top_lags[None, :]
        b_mL = zs[:, None] - top_lags[None, :]
        vp_a = (a_pL >= 0) & (a_pL < N)
        vm_a = (a_mL >= 0) & (a_mL < N)
        vp_b = (b_pL >= 0) & (b_pL < N)
        vm_b = (b_mL >= 0) & (b_mL < N)
        sp_a = np.clip(a_pL, 0, N - 1)
        sm_a = np.clip(a_mL, 0, N - 1)
        sp_b = np.clip(b_pL, 0, N - 1)
        sm_b = np.clip(b_mL, 0, N - 1)
        eff_a = -((1.0 - h[sp_a]) * vp_a).astype(float) + (h[sm_a] * vm_a).astype(float)
        eff_b = ((1.0 - h[sp_b]) * vp_b).astype(float) - (h[sm_b] * vm_b).astype(float)
        cur_corr = corr[top_idx]
        new_peak = np.max(cur_corr[None, None, :] + eff_a[:, None, :] + eff_b[None, :, :], axis=2)
        best_pair = np.unravel_index(np.argmin(new_peak), new_peak.shape)

            h[zs[best_pair[1]]] = 1
        elif best_new_c < c + temp and np.random.random() < np.exp(-(best_new_c - c) / max(temp, 1e-8)):
            h[os_[best_pair[0]]] = 0
            h[zs[best_pair[1]]] = 1
        it += 1
    return best_h, best_c

        sp_a = np.clip(a_pL, 0, N - 1)

def cont_opt(h, deadline, k=40, lr0=0.02, mom=0.85):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 50:
            lr0 *= 0.8
            stale = 0
        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            M = abs(int(m) - center)
            if M == 0:
                grad += (1.0 - 2.0 * h)
            elif M < N:
                hi_idx = np.arange(M, N)
                lo_idx = np.arange(0, N - M)
                grad[hi_idx] += (1.0 - h[hi_idx - M])
                grad[lo_idx] -= h[lo_idx + M]
        grad /= k
        h_new = project(h + mom * v - lr0 * grad, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c
