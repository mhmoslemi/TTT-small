            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        hb[rng.choice(idx, n1 - nh, replace=False)] = 0
    else:
        idx = np.where(hb == 0)[0]
        hb[rng.choice(idx, nh - n1, replace=False)] = 1
    return hb

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def bin_anneal(h, N, deadline, rng):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    temp = 0.009
    stall = 0
    for sweep in range(200000):
        if time.time() > deadline:
            break
        temp = max(0.0008, temp * 0.9987)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        best_new_P = P_cur
        best_move = None
        for p in ones:
            base = c_cur + D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_move = (0, int(p), int(zeros[mi]))
        if len(ones) >= 4 and len(zeros) >= 4:
            n_try = 100
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                nc = c_cur + D[p1] + D[p2] - D[q1] - D[q2]
                nP = float(nc.max())
                if nP < best_new_P:
                    best_new_P = nP
                    best_move = (1, p1, p2, q1, q2)
        if len(ones) >= 6 and len(zeros) >= 6 and rng.random() < 0.12:
            for _ in range(30):
                ip = rng.randint(0, len(ones), 3)
                iq = rng.randint(0, len(zeros), 3)
                if len(set(ip.tolist())) < 3 or len(set(iq.tolist())) < 3:
                    continue
                p1, p2, p3 = int(ones[ip[0]]), int(ones[ip[1]]), int(ones[ip[2]])
                q1, q2, q3 = int(zeros[iq[0]]), int(zeros[iq[1]]), int(zeros[iq[2]])
                nc = c_cur + D[p1] + D[p2] + D[p3] - D[q1] - D[q2] - D[q3]
                nP = float(nc.max())
                if nP < best_new_P:
                    best_new_P = nP
                    best_move = (2, p1, p2, p3, q1, q2, q3)
        if best_move is not None:
            mt = best_move[0]
            if mt == 0:
                h[best_move[1]] = 0.0
                h[best_move[2]] = 1.0
            elif mt == 1:
                h[best_move[1]] = h[best_move[2]] = 0.0
                h[best_move[3]] = h[best_move[4]] = 1.0
            else:
                h[best_move[1]] = h[best_move[2]] = h[best_move[3]] = 0.0
                h[best_move[4]] = h[best_move[5]] = h[best_move[6]] = 1.0
            c_cur = fft_corr(h)
            P_cur = float(c_cur.max())
            if P_cur < best_c - 1e-13:
                best_c = P_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            if stall > 25:
                nf = min(6, len(ones), len(zeros))
                for _ in range(nf):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                c_cur = fft_corr(h)
                P_cur = float(c_cur.max())
                stall = 0
            elif stall > 8:
                break
            else:
                stall += 1
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, rng):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = 0.014
    beta = 0.88
    it = 0
    k_top = 80
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k_top)[-k_top:]
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0 and L < N:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            elif L < 0:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        # Binary push: encourage 0/1 values
        grad += 3.0 * y * (y - 1.0)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.97
            beta = min(0.96, 0.85 + it * 0.0002)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def downsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Thorough binary anneal at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 500, t0 + budget_s * 0.30)
    h1, c5_1 = bin_anneal(h_bin, N0, d1, rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1b: Restart with perturbation from best
    if time.time() < deadline - 420:
        rng2 = np.random.RandomState(seed + 77)
        h_pert = best_h.copy()
        ones_t = np.where(h_pert == 1)[0]
        zeros_t = np.where(h_pert == 0)[0]
        for _ in range(8):
            h_pert[ones_t[rng2.randint(len(ones_t))]] = 0.0
            h_pert[zeros_t[rng2.randint(len(zeros_t))]] = 1.0
        d1b = min(deadline - 380, time.time() + budget_s * 0.06)
        h1b, c5_1b = bin_anneal(h_pert, N0, d1b, rng2)
        if c5_1b < best_c5:
            best_c5, best_h = c5_1b, h1b.copy()

    # Phase 2: Nesterov continuous refinement at N0
    if time.time() < deadline - 300:
        hc = best_h + rng.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 260, time.time() + budget_s * 0.12)
        h2, c5_2 = nesterov_opt(hc, N0, d2, rng)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Upscale to 2*N0, binary anneal
    N1 = 2 * N0
    if time.time() < deadline - 180:
        h_up = ensure_sum(upsample(best_h, N0, N1), N1)
        d3 = min(deadline - 150, time.time() + budget_s * 0.16)
        h3, c5_3 = bin_anneal(h_up, N1, d3, rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Downsample back, Nesterov at N0
    if time.time() < deadline - 120:
        h_ds = downsample(best_h, len(best_h), N0)
        hc2 = h_ds + rng.randn(N0) * 0.05
        hc2 = project_sum(hc2, N0 / 2.0)
        d4 = min(deadline - 80, time.time() + budget_s * 0.08)
        h4, c5_4 = nesterov_opt(hc2, N0, d4, rng)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Final binary polish
    N_fin = len(best_h)
    hF = ensure_sum(best_h, N_fin)
    rng_fin = np.random.RandomState(seed + 200)
    hF2, c5F = bin_anneal(hF, N_fin, deadline - 1, rng_fin)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
