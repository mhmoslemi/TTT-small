def _c5_smooth(h, dx, t=50.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # scaled for numerical stability
    corr_shifted = corr - corr.max()
    exp_c = np.exp(t * corr_shifted)
    log_sum = np.log(exp_c.sum()) + corr.max()
    f = log_sum / t * dx
    return f, exp_c / exp_c.sum()  # return value and probabilities

def _c5_smooth_grad(h, dx, t=50.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_shifted = corr - corr.max()
    exp_c = np.exp(t * corr_shifted)
    p = exp_c / exp_c.sum()  # probability weights
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-15:
            continue
        lag = k - (n - 1)
        local_grad = np.zeros(n)
        if lag > 0:
            idx1 = np.arange(lag, n)
            local_grad[idx1] += (1.0 - h[idx1 - lag])
            idx2 = np.arange(0, n - lag)
            local_grad[idx2] -= h[idx2 + lag]
        elif lag < 0:
            L = -lag
            idx1 = np.arange(0, n - L)
            local_grad[idx1] += (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            local_grad[idx2] -= h[idx2 - L]
        else:
            local_grad[:] = 1.0 - 2.0 * h
        grad += p[k] * local_grad
    
    return grad * dx
