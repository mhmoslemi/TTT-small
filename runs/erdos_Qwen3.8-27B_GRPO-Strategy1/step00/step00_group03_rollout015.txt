import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def objective_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        return c_max * dx
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def sum_constraint(h, n):
    return np.sum(h) - n * 0.5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n * 0.5
    s = np.sum(h_new)
    if s > 1e-15:
        h_new = h_new * (target / s)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-14:
        interior = (h_new > 1e-6) & (h_new < 1.0 - 1e-6)
        n_int = np.sum(interior)
        if n_int > 0:
            h_new[interior] += diff / n_int
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
