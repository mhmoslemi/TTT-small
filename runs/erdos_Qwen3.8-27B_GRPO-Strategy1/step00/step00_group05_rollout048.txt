import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

_target_sum = 35.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * corr * dx
    return float(logsumexp(vals) / beta)

def eq_constraint(h):
    return np.sum(h) - _target_sum

def eq_jac(h):
    return np.ones(len(h))

def clamp_fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, factor):
    return np.repeat(h, factor)

def downsample(h, target_n):
    if len(h) == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, h)
