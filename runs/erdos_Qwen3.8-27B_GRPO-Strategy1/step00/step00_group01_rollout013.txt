def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    val = (max_c + np.log(sum_exp) / beta) * dx
    weights = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:

            if c5_i < best_c5:
                best_h = h_interp
                best_c5 = c5_i
            for beta in [32, 64, 128]:
                if tl() < 150:
                    break
                h_try = lbbfs_opt(best_h, beta, 50.0, 300)
                h_try = proj_sum(np.clip(h_try, 0.0, 1.0), 0.5 * len(h_try))
                c5 = compute_c5(h_try)
                if c5 < best_c5:
                    best_h = h_try
                    best_c5 = c5
            if tl() < 100:
                break

    # Final FISTA polish
    if tl() > 50:
        h_p = fista_polish(best_h.copy(), 512, 1000, 0.5 / len(best_h))
        h_p = proj_sum(h_p, 0.5 * len(h_p))
        c5 = compute_c5(h_p)
        if c5 < best_c5:
            best_h = h_p
            best_c5 = c5

    best_h = proj_sum(np.clip(best_h, 0.0, 1.0), 0.5 * len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
