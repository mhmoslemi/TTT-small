import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_obj_grad(h):
    """Returns (objective, gradient) for C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    obj = float(corr[max_idx]) * dx
    grad = np.zeros(n)
    if s > 0:
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
    elif s < 0:
        s2 = -s
        grad[:n-s2] += (1.0 - h[s2:])
        grad[s2:] -= h[:n-s2]
    else:
        grad = 1.0 - 2.0 * h
    return obj, grad * dx

def slsqp_optimize(h0, max_iter=500):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        v, _ = c5_obj_grad(x)
        return v
    def grad(x):
        _, g = c5_obj_grad(x)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
    # Adjust sum to be exact
    h = res.x.copy()
    h = np.clip(h, 0, 1)
    # Fix sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute diff evenly
        h += diff / n
        h = np.clip(h, 0, 1)
        # Re-fix
        diff = target - np.sum(h)
        h += diff / n
    return h

    """Local search on binary h. h_bin is boolean array with sum=n/2."""
    rng = np.random.RandomState(seed)
    n = len(h_bin)

    best_c5 = compute_c5(h)
    current_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < current_c5 - 1e-12:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
            # Update ones/zeros (just keep them, they're indices)
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return h, best_c5

    """Upsample by repeating each element `factor` times."""
    return np.repeat(h, factor)

def upsample_smart(h, factor=2):
    """Upsample by splitting each element into half-value segments."""
    n = len(h)
    out = np.zeros(n * factor)
    for i in range(n):
        out[i*factor] = h[i]
        out[i*factor+1] = h[i]
    return out  # same as np.repeat for factor=2
