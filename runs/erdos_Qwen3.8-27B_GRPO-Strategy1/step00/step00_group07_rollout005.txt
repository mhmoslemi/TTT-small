import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def fix_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(free) == 0:
            free = np.where(h > 0.001)[0] if diff > 0 else np.where(h < 0.999)[0]
        if len(free) == 0:
            break
        per = diff / len(free)
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def sa_optimize(h_init, n, dx, target_sum, rng, t_deadline, best_ref):
    h = h_init.copy()
    T = 0.003
    c5_cur = compute_c5(h, dx)
    iters = 0
    while time.time() < t_deadline:
        r = rng.random()
        if r < 0.55:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                j = (j + 1) % n
            if rng.random() < 0.6:
                delta = rng.uniform(-0.02, 0.02)
            else:
                delta = rng.uniform(-0.08, 0.08)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        elif r < 0.8:
            blk_len = rng.randint(2, min(8, n // 4 + 1))
            start = rng.randint(0, n - blk_len + 1)
            end = start + blk_len
            old_sum = np.sum(h[start:end])
            w = np.abs(rng.randn(blk_len)) + 0.1
            new_block = w / w.sum() * old_sum
            new_block = np.clip(new_block, 0.0, 1.0)
            new_block = fix_sum(new_block, old_sum, rng)
            new_h = h.copy()
            new_h[start:end] = new_block
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                j = (j + 1) % n
            new_h = h.copy()
            new_h[i] = 1.0 - h[i]
            diff = new_h[i] - h[i]
            new_h[j] -= diff
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = fix_sum(new_h, target_sum, rng)
        c5_new = compute_c5(new_h, dx)
        if c5_new <= c5_cur or (c5_new - c5_cur) / max(T, 1e-10) < np.log(rng.random()) + 1e-10:
            h = new_h
            c5_cur = c5_new
            if c5_new < best_ref[0]:
                best_ref[0] = c5_new
                best_ref[1] = h.copy()
        T *= 0.99997
        iters += 1
        if iters % 50000 == 0 and T > 0.0001:
            T = max(T * 0.9, 0.0001)
    return h, c5_cur

def local_refine(h, n, dx, target_sum, rng, t_deadline, best_ref):
    h = h.copy()
    c5_cur = compute_c5(h, dx)
    steps = [0.005, 0.01, 0.02, 0.03, 0.05, -0.005, -0.01, -0.02, -0.03, -0.05]
    while time.time() < t_deadline:
        improved = False
        for _ in range(500):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                j = (j + 1) % n
            for d in steps:
                new_h = h.copy()
                new_h[i] += d
                new_h[j] -= d
                if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
                    continue
                new_h = fix_sum(new_h, target_sum, rng)
                c5_t = compute_c5(new_h, dx)
                if c5_t < c5_cur - 1e-10:
                    h = new_h
                    c5_cur = c5_t
                    if c5_cur < best_ref[0]:
                        best_ref[0] = c5_cur
                        best_ref[1] = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            h = h + rng.randn(n) * 0.005
            h = np.clip(h, 0.0, 1.0)
            h = fix_sum(h, target_sum, rng)
            c5_cur = compute_c5(h, dx)
    return h, c5_cur
