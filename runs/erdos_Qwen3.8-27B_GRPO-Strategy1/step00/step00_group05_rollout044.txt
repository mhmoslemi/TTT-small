        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(parent, new_n):
    old_n = len(parent)
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, parent)
    return make_feasible(h_new, new_n)

def obj_func(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def constraint_sum(x, target):
    return float(np.sum(x) - target)

def optimize_local(h0, n, maxiter, ftol):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (target_sum,)}]
    try:
        result = minimize(
            obj_func, h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_opt = make_feasible(result.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary_random(n, rng):
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def make_structured(n, rng, pattern='two_block'):
    h = np.zeros(n)
    if pattern == 'two_block':
        w = n // 4
        gap = rng.integers(n // 8, n // 2)
        h[:w] = 1.0
        h[gap:gap + w] = 1.0
        h = make_feasible(h, n)
    elif pattern == 'three_block':
        w = n // 6
        offsets = rng.permutation(n)[:3]
        positions = sorted(offsets * (n // 6) % (n - w))
        for p in positions:
            h[p:p + w] = 1.0
        h = make_feasible(h, n)
    else:
        h = rng.random(n)
        h = make_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    parent = np.array(initial_h_values, dtype=np.float64)
    best_h = parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Optimize at parent resolution
    if elapsed() < budget_s - 30:
        h_opt, c5 = optimize_local(best_h, best_n, 300, 1e-10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = len(h_opt)

    # Phase 2: Coarse-to-fine optimization
    resolutions = [140, 280, 560]
    for n in resolutions:
        if elapsed() > budget_s - 80:
            break

        # Build candidates
        candidates = []
        candidates.append(upsample(best_h, n))
        candidates.append(upsample(parent, n))

        # Binary random starts
        for _ in range(2):
            candidates.append(make_binary_random(n, rng))

        # Structured candidates
        for pat in ['two_block', 'three_block']:
            candidates.append(make_structured(n, rng, pattern=pat))

        # Perturbed upsample
        up = upsample(best_h, n)
        pert = up + rng.normal(0, 0.05, n)
        candidates.append(make_feasible(pert, n))

        n_candidates = len(candidates)
        maxiter = max(100, min(400, 60000 // n))

        for i, h0 in enumerate(candidates):
            remaining = budget_s - 60 - elapsed()
            if remaining < 5:
                break
            ftol = 1e-9 if i < 3 else 1e-7
            h_opt, c5 = optimize_local(h0, n, maxiter, ftol)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
                # Re-optimize from improved solution
                if elapsed() < budget_s - 60:
                    h_opt2, c52 = optimize_local(h_opt, n, maxiter, 1e-10)
                    if c52 < best_c5:
                        best_c5 = c52
                        best_h = h_opt2

    # Phase 3: Try more random restarts at best resolution if time allows
    if elapsed() < budget_s - 40:
        n = best_n
        attempts = min(8, int((budget_s - 40 - elapsed()) / max(5, 100 * n // 70)))
        for _ in range(attempts):
            if elapsed() > budget_s - 20:
                break
            h_r = make_binary_random(n, rng)
            h_opt, c5 = optimize_local(h_r, n, 200, 1e-8)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    # Phase 4: Final high-precision polish
    if elapsed() < budget_s - 15:
        n = best_n
        h_opt, c5 = optimize_local(best_h, n, 500, 1e-12)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    return best_h, best_c5, best_n
