import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def objective(params, n):
    """Objective: max of correlation * dx"""
    h = params
    corr = np.correlate(h, 1 - h, mode="full") * (2.0/n)
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    best_h = None
    best_c5 = float('inf')
    
    if 'initial_h_values' in dir() or True:
        try:
            init = np.array(initial_h_values, dtype=float)
            best_c5_init = compute_c5(init)
            best_h = init.copy()
            best_c5 = best_c5_init
        except:
            pass
    
    # Try different resolutions
    for n in [42, 84, 126, 200, 300, 500, 800, 1200, 2000]:
        if time.time() - start_time > budget_s * 0.85:
            break
        
        # Create initial guess by upsampling
        if best_h is not None and best_h != init:
            h0 = np.interp(np.linspace(0, 2, n, endpoint=False), 
                          np.linspace(0, 2, len(best_h), endpoint=False), 
                          best_h)
        else:
            h0 = np.full(n, 0.5)
        
        # Enforce constraint
        h0 = np.clip(h0, 0, 1)
        # Adjust sum to n/2
        diff = n/2 - np.sum(h0)
        h0 += diff / n
        h0 = np.clip(h0, 0, 1)
        
        # Optimize
        try:
            result = minimize(
                objective, h0, args=(n,),
                method='SLSQP',
                bounds=[(0,1)]*n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}],
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            
            h_opt = np.clip(result.x, 0, 1)
            # Fix sum
            h_opt *= (n/2) / np.sum(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            
            c5_new = compute_c5(h_opt)
            if c5_new < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_new
        except:
            pass
    
    # Final evaluation
    n = len(best_h)
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, n)

1. Start with h - α*∇f
2. Find α (scalar) such that the result has sum n/2 and is in [0,1]

1. Compute g = ∇f(h)
2. For step size s: h_new[i] = h[i] - s*g[i]
3. Project: h_proj = clip(h_new, 0, 1)
4. Adjust sum: h_proj += (n/2 - sum(h_proj))/n (if this keeps all in [0,1])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def get_correlation(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n * 0.5 - np.sum(h)
    if diff > 0:
        mask = h < 1.0
        if mask.any():
            h[mask] += diff / mask.sum()
    elif diff < 0:
        mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def gradient_softmax(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlation(h)
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / temp)
    weights = exp_c / exp_c.sum()
    
    # Gradient computation (O(n^2) but vectorized per lag)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        wk = weights[k] * dx
        if wk < 1e-16:
            continue
        if m >= 0:
            # corr[k] = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
            # d/dh[j]: (1-h[j-m]) for j in [m, n-1] (from h[i+m] term, i=j-m)
            #          -h[j+m] for j in [0, n-1-m] (from (1-h[i]) term, i=j)
            if m < n:
                grad[m:n] += wk * (1.0 - h[0:n-m])
                grad[0:n-m] -= wk * h[m:n]
        else:
            p = -m  # p > 0
            # corr[k] = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
            # d/dh[j]: (1-h[j-p]) for j in [p, n-1] (from h[i] term, i=j)
            #          -h[j+p] for j in [0, n-1-p] (from (1-h[i-p]) term, i=j+p)
            if p < n:
                grad[p:n] += wk * (1.0 - h[0:n-p])
                grad[0:n-p] -= wk * h[p:n]
    return grad, np.max(corr)

def pgd_optimize(h, n, n_iter, temp, time_limit):
    h = h.copy()
    step = 1.0 / (n * 0.1)  # initial step size
    best_local = compute_c5(h)
    
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        grad, c5_max = gradient_softmax(h, temp)
        # Projected gradient step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_local:
            best_local = c5_new
            h = h_new
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.8
            if step < 1e-6:
                break
    
    return h, best_local

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_exact(h, target_sum):
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def softmax_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    ec = np.exp((corr - mc) / temp)
    w = ec / ec.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        if m >= 0:
            grad[m:n] += wk * (1.0 - h[0:n - m])
            grad[0:n - m] -= wk * h[m:n]
        else:
            p = -m
            grad[p:n] += wk * (1.0 - h[0:n - p])
            grad[0:n - p] -= wk * h[p:n]
    return grad, mc

def pgd_optimize(h, n, max_iter, time_limit):
    h = h.copy()
    step = max(1.0, n / 15.0)
    best_c5 = compute_c5(h)
    stag = 0
    temps = [0.08, 0.04, 0.02, 0.01]
    ti = 0
    temp = temps[0]
    it = 0
    while it < max_iter:
        if time.time() > time_limit:
            break
        grad, _ = softmax_gradient(h, temp)
        h_new = h - step * grad
        h_new = project_exact(h_new, n * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            h = h_new
            stag = 0
            step = min(step * 1.1, n * 0.5)
        else:
            stag += 1
            step *= 0.85
            if stag > 30:
                stag = 0
                if ti + 1 < len(temps):
                    ti += 1
                    temp = temps[ti]
                step = max(1.0, n / 30.0)
        it += 1
    return h, best_c5

def slsqp_refine(h, n, time_limit):
    if n > 300 or time.time() > time_limit:
        return h, compute_c5(h)
    try:
        res = minimize(
            compute_c5, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
            options={'maxiter': 150, 'ftol': 1e-12}
        )
        hx = project_exact(np.clip(res.x, 0, 1), n / 2.0)
        return hx, compute_c5(hx)
    except Exception:
        return h, compute_c5(h)

def make_periodic_init(n, freq):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    return project_exact(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    deadline = t0 + budget_s * 0.92

    except Exception:

    best_c5 = compute_c5(best_h)

    cur = best_h.copy()
    seen_n = set()

    resolutions = [42, 63, 84, 126, 168, 252, 378, 504, 756, 1008, 1512, 2016, 3024]

    for nt in resolutions:

        if nt in seen_n:
            continue
        seen_n.add(nt)
        if nt < len(cur):
            nt = len(cur)

        n = len(cur)

        cur, c5 = pgd_optimize(cur, n, n_iter, time.time() + remaining * 0.7)

        if n <= 200:

        if c5 < best_c5:
            best_c5 = c5
            best_h = cur.copy()

    # Random restarts at best resolution
    n_best = len(best_h)
    n_restarts = 5
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if n_best > 600:
            h_r = upsample(best_h, 300)
        else:
            h_r = best_h + 0.04 * np.random.randn(n_best)
            h_r = project_exact(h_r, n_best / 2.0)
        nr = len(h_r)
        h_r, c5r = pgd_optimize(h_r, nr, 300, min(time.time() + 30, deadline))
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_r.copy()
            n_best = nr

    # Try periodic-inited restart

            if time.time() > deadline:

            h_p = make_periodic_init(n_best, freq)

                best_h = h_p.copy()

    # Final polish with SLSQP if small enough
    if len(best_h) <= 150 and time.time() < deadline:

    best_h = project_exact(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
