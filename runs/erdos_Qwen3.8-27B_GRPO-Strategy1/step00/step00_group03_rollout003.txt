    return np.max(corr) * dx

def project_onto_simplex(h, n, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Bisection on lambda: h_proj = clip(h - lambda, 0, 1), find lambda s.t. sum = target
    lo, hi = -1.0, 1.0

        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
