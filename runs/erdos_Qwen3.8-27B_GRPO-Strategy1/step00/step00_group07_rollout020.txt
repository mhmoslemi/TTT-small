```python
                    best_h = h_new
                    best_c5 = c5_new
                    h_cur = h_new.copy()
        except Exception:
            pass

        if elapsed() > budget_s - 30:
            break

        # Coordinate descent refinement
        if elapsed() < budget_s - 20:
            h_cd, c5_cd = coordinate_descent(h_cur, t_soft, steps=80)
            if c5_cd < best_c5:
                best_h = h_cd
                best_c5 = c5_cd

        # Increase soft-max sharpness over time
        t_soft *= 1.1

    # Final coordinate descent on best
    if elapsed() < budget_s - 10:
        h_final, c5_final = coordinate_descent(best_h, t_soft, steps=300)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    return best_h, best_c5, n
```
