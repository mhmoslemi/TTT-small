import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            h[mask] -= diff / mask.sum()
        else:
            mask = h < 1
            h[mask] += (-diff) / mask.sum()
    return np.clip(h, 0.0, 1.0)
