import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full") * dx
    return corr

def grad_corr_weighted(h, weights):
    """Gradient of sum_k weights[k]*corr[k] w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(weights)):
        if abs(weights[k]) < 1e-12:
            continue
        p = k - (n - 1)
        w = weights[k]
        # d/dh[m] corr[k] = (1-h[m+p])*I[0<=m+p<=n-1] - h[m-p]*I[0<=m-p<=n-1]
        # First term: (1-h[m+p]) for m where m+p in [0,n-1], i.e., m in [-p, n-1-p]
        lo1 = max(0, -p)
        hi1 = min(n-1, n-1-p)
        if lo1 <= hi1:
            m_range = np.arange(lo1, hi1+1)
            g[lo1:hi1+1] += w * (1 - h[m_range + p])
        
        # Second term: -h[m-p] for m where m-p in [0,n-1], i.e., m in [p, n-1+p]
        lo2 = max(0, p)
        hi2 = min(n-1, n-1+p)
        if lo2 <= hi2:
            m_range = np.arange(lo2, hi2+1)
            g[lo2:hi2+1] -= w * h[m_range - p]
    
    g *= dx
    return g

def project_to_feasible(v, n):
    """Project onto {x: 0<=x_i<=1, sum(x)=n/2}."""
    target = n / 2.0
    # Bisection on Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Final check
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Adjust a few elements
        idx = np.argsort(result)[::-1] if diff > 0 else np.argsort(result)
        for i in idx:
            adj = min(abs(diff), abs(1-result[i]) if diff > 0 else result[i])
            result[i] += adj if diff > 0 else -adj
            diff -= adj * np.sign(diff)
            if abs(diff) < 1e-12:
                break
    return result

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full") * dx
    return corr


def grad_corr_weighted(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        p = k - (n - 1)
        lo1 = max(0, -p)
        hi1 = min(n - 1, n - 1 - p)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += w * (1.0 - h[lo1 + p:hi1 + 1 + p])
        lo2 = max(0, p)
        hi2 = min(n - 1, n - 1 + p)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= w * h[lo2 - p:hi2 + 1 - p]
    g *= dx
    return g


def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return result

            break
        corr = compute_corr(h)
        c5 = np.max(corr)
        if c5 < best_local_c5 - 1e-12:
            best_local_c5 = c5

            stall_count = 0
        else:
            stall_count += 1

        T = max(0.003, 0.08 * (1.0 - 0.995 * it / max(n_iters, 1)))
        m = c5
        w = np.exp((corr - m) / T)
        w_sum = w.sum()
        w = w / w_sum

        g_centered = g - np.mean(g)
        g_norm = np.linalg.norm(g_centered)

        lr = base_lr / (1.0 + 0.005 * it)

        h = h_new
