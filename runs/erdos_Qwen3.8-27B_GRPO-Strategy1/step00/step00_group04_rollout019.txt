def pair_flip_search(h_start, n, budget_fn):
    best_c5 = compute_c5(h_start, n)
    h_work = h_start.copy()
    improved = True
    max_sweeps = 5
    for sweep in range(max_sweeps):
        if budget_fn() < 20 or not improved:
            break
        improved = False
        for i in range(n):
            if budget_fn() < 20:
                break
            for tv in [0.0, 1.0]:
                delta = tv - h_work[i]
                if abs(delta) < 1e-5:
                    continue
                if delta > 0:
                    cands = np.where(h_work > abs(delta) + 1e-6)[0]
                else:
                    cands = np.where(h_work < 1.0 - abs(delta) - 1e-6)[0]
                for j in cands:
                    if j == i:
                        continue
                    h_try = h_work.copy()
                    h_try[i] = tv
                    h_try[j] = h_work[j] - delta
                    if h_try[j] < -1e-9 or h_try[j] > 1.0 + 1e-9:
                        continue
                    h_try = np.clip(h_try, 0.0, 1.0)
                    if abs(np.sum(h_try) - n / 2.0) < 1e-6:
                        c = compute_c5(h_try, n)
                        if c < best_c5 - 1e-12:
                            best_c5 = c
                            h_work = h_try.copy()
                            improved = True
                            break
                if improved:
                    break
    return h_work, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = 0.5 * np.ones(42)
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)

    best_c5 = compute_c5(h_init, n_init)
    best_h = h_init.copy()
    best_n = n_init

    # Multi-resolution SLSQP
    resolutions = [84, 126, 196, 280, 400]
    cur_h = h_init.copy()
    cur_n = n_init

    for n in resolutions:
        if rem() < 80:
            break
        _N = n
        h0 = upsample_linear(cur_h, cur_n, n)
        h0 = project_feasible(h0, n)

        trial_ps = [0.0, 0.06, 0.12, 0.25]
        for tp in trial_ps:
            if rem() < 40:
                break
            if tp == 0.0:
                x0 = h0.copy()
            else:
                x0 = project_feasible(h0 + tp * rng.randn(n), n)

            try:
                mi = 200 if n <= 200 else 120
                h_res, c5_res = slsqp_optimize(x0, n, maxiter=mi)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
                    best_n = n
            except Exception:
                pass

        # Update cur for next scale
        cur_h = upsample_linear(best_h, best_n, n)
        cur_n = n

    # Local refinement at best resolution
    if rem() > 40:
        n_work = best_n
        h_work, c5_ls = local_search(best_h, n_work, rem)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_work.copy()

    # Pair-flip refinement
    if rem() > 50:
        h_wf, c5_pf = pair_flip_search(best_h, best_n, rem)
        if c5_pf < best_c5:
            best_c5 = c5_pf
            best_h = h_wf.copy()

    # Final SLSQP polish from best point
    if rem() > 60:
        n_fin = best_n
        try:
            h_fin, c5_fin = slsqp_optimize(best_h, n_fin, maxiter=150)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        except Exception:
            pass

    # Final local search after polish
    if rem() > 30:
        h_ls, c5_ls2 = local_search(best_h, best_n, rem)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
