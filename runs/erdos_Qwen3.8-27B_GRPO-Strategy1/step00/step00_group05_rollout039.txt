def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(logsumexp(temperature * corr) / temperature)

def fix_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-9) & (h < 1.0 - 1e-9)
    if diff > 0:
        cap = 1.0 - h[mask]
        tot = np.sum(cap)
        if tot > 0:
            h[mask] += cap * min(1.0, diff / tot)
    else:
        cap = h[mask]
        tot = np.sum(cap)
        if tot > 0:

        idx = np.argmax(h) if residual < 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
