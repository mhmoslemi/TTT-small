import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def local_search(h, dx, rng, time_limit):
    n = len(h)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h, dx)
    no_improve_count = 0
    
    while time.time() - start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_pair = None
        best_val = best_c5
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h, dx)
                if c5 < best_val - 1e-15:
                    best_val = c5
                    best_pair = (int(i), int(j))
                h[i] = 1.0
                h[j] = 0.0

            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            best_c5 = best_val
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 3:
                break
            # Random perturbation
            n_swap = min(no_improve_count + 1, 5)

            zeros_list = zeros.tolist()
            for _ in range(n_swap):
                if len(ones_list) < 2 or len(zeros_list) < 2:
                    break
                ii = rng.randint(len(ones_list))
                jj = rng.randint(len(zeros_list))

                oz = zeros_list[jj]
                h[oi], h[oz] = h[oz], h[oi]
                ones_list = [x for x in ones_list if x != oi]
                zeros_list = [x for x in zeros_list if x != oz]
                zeros_list.append(oi)
                ones_list.append(oz)
            best_c5 = compute_c5(h, dx)
