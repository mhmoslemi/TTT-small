        else:
            m = -lag
            g[:n - m] += wi * (1.0 - u[m:])
            g[m:] -= wi * u[:n - m]
    g += 2.0 * mu * (u.sum() - n / 2.0)
    return g

def _obj(u):
    n = _PARAMS['n']
    dx = _PARAMS['dx']
    temp = _PARAMS['temp']
    mu = _PARAMS['mu']
    corr = np.correlate(u, 1.0 - u, mode="full") * dx
    val = smooth_max(corr, temp)
    val += mu * (u.sum() - n / 2.0) ** 2
    return val

def _grad(u):
    n = _PARAMS['n']
    dx = _PARAMS['dx']
    temp = _PARAMS['temp']
    mu = _PARAMS['mu']
    return _compute_grad(u, n, dx, temp, mu)

def _con_fun(u):
    n = _PARAMS['n']
    return u.sum() - n / 2.0

def _con_jac(u):
    n = _PARAMS['n']
    return np.ones(n)

def _slsqp_obj(u):
    dx = _PARAMS['dx']
    temp = _PARAMS['temp']
    corr = np.correlate(u, 1.0 - u, mode="full") * dx
    return smooth_max(corr, temp)

def _interp(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x0 = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    f = interp1d(x0, h_src, kind='cubic', fill_value='extrapolate')
    return np.clip(f(x1), 0.0, 1.0)

def _run_lbfgsb(h0, n, dx, temps, max_iter_base):
    target = n / 2.0
    for temp in temps:
        if time.time() > _PARAMS['deadline'] - 15:
            break
        _PARAMS['n'] = n
        _PARAMS['dx'] = dx
        _PARAMS['temp'] = temp
        _PARAMS['mu'] = 500.0
        max_it = max(50, max_iter_base - n)
        try:
            res = minimize(
                _obj, h0.copy(), jac=_grad,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': max_it, 'gtol': 1e-7, 'maxfun': max_it * 4}
            )
            h0 = normalize_h(res.x, n)
        except Exception:
            pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s
    _PARAMS['deadline'] = deadline

    best_h = None
    best_c5 = 999.0
    best_n = 0

    h_init = np.asarray(initial_h_values, dtype=np.float64)
    n_init = len(h_init)

    def remaining():
        return deadline - time.time()

    # Phase 1: SLSQP at initial resolution
    if remaining() > 50:
        n = n_init
        dx = 2.0 / n
        _PARAMS['n'] = n
        _PARAMS['dx'] = dx
        _PARAMS['temp'] = 0.001
        h0 = normalize_h(h_init, n)
        constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(
                _slsqp_obj, h0.copy(), method='SLSQP',
                bounds=[(0.0, 1.0)] * n, constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-9}
            )
            h0 = normalize_h(res.x, n)
        except Exception:
            pass
        c5 = c5_of(h0, dx)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h0.copy(), n

    # Phase 2: Coarse-to-fine L-BFGS-B
    resolutions = [n_init]
    cn = n_init
    for _ in range(5):
        cn *= 2
        if cn > 2048:
            break
        resolutions.append(cn)

    for n in resolutions:
        if remaining() < 40:
            break
        dx = 2.0 / n

        if n == n_init and best_h is not None:
            h0 = best_h.copy()
        else:
            src_h = best_h if best_h is not None else h_init
            src_n = best_n if best_n > 0 else n_init
            h0 = _interp(src_h, src_n, n)
            h0 = normalize_h(h0, n)

        c5 = c5_of(h0, dx)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h0.copy(), n

        temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        h0 = _run_lbfgsb(h0, n, dx, temps, 250)
        c5 = c5_of(h0, dx)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h0.copy(), n

    # Phase 3: Perturbation restarts
    if remaining() > 30:
        n = best_n
        dx = 2.0 / n
        for attempt in range(8):
            if remaining() < 12:
                break
            h0 = best_h.copy()
            noise_scale = 0.05 + 0.1 * attempt
            h0 = np.clip(h0 + noise_scale * np.random.randn(n), 0.0, 1.0)
            h0 = normalize_h(h0, n)
            temps = [0.02, 0.01, 0.005]
            h0 = _run_lbfgsb(h0, n, dx, temps, 120)
            c5 = c5_of(h0, dx)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h0.copy(), n

    # Phase 4: Try a different starting point (uniform-ish with structure)
    if remaining() > 40:
        n = max(best_n, 168)
        dx = 2.0 / n
        # Create a sinusoidal starting point
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = 0.5 + 0.3 * np.sin(np.pi * x)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = normalize_h(h0, n)
        temps = [0.05, 0.02, 0.01, 0.005, 0.002]
        h0 = _run_lbfgsb(h0, n, dx, temps, 200)
        c5 = c5_of(h0, dx)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h0.copy(), n

    # Final: try the evaluator
    if best_h is None:
        best_h = h_init.copy()
        best_n = n_init
        best_c5 = c5_of(best_h, 2.0 / n_init)

    try:
        val = evaluate_erdos_solution(best_h)
        if val is not None and float(val) < best_c5:
            best_c5 = float(val)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
