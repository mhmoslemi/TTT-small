        else:
            max_d = min(1.0 - h[i], h[j])

        if max_d < 1e-12:
            T *= cool_rate
            continue

        # Adaptive step size
        scale = min(max_d, 0.001 + 0.05 * (T / T0 + 0.1))
        d = rng.uniform(0, scale)
        if rng.random() < 0.05:
            d = rng.uniform(0.5 * max_d, max_d)

        delta = d if flip else -d
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        if new_h[i] < -1e-14 or new_h[i] > 1 + 1e-14 or new_h[j] < -1e-14 or new_h[j] > 1 + 1e-14:
            T *= cool_rate
            continue

        new_c5 = compute_c5(new_h, dx)
        dc = new_c5 - current_c5

        if dc <= 0 or rng.random() < np.exp(-dc / T):
            h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cool_rate

    return best_h, best_c5


def binary_snap(h, n, dx, rng, n_trials=2000):
    """Try to improve by snapping values and doing binary swaps."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    for _ in range(n_trials):
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            h[i], h[j] = h[j], h[i]  # revert
            h[i], h[j] = h[j], h[i]
            # actually just swap back
            h[i], h[j] = h[j], h[i]

    return best_h, best_c5


def local_refine(h, n, dx, time_limit, rng):
    """Coordinate descent: try small adjustments to each element."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step = 0.005

    while time.time() < time_limit:
        improved = False
        for i in range(n):
            for direction in [1, -1]:
                d = direction * step
                j = (i + 1) % n
                new_h = h.copy()
                new_h[i] += d
                new_h[j] -= d
                if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
                    continue
                c5 = compute_c5(new_h, dx)
                if c5 < best_c5 - 1e-10:
                    h = new_h
                    best_h = h.copy()
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 10

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    dx = 2.0 / n
    h = make_valid(h, n)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Phase 1: Main SA (60% of time)
    t1 = start_time + budget_s * 0.55
    h, c5 = sa_optimize(h, n, dx, 0.01, 1e-8, 0.9997, t1, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Finer SA (20% of time)
    t2 = start_time + budget_s * 0.78
    h2 = best_h.copy()
    h2_, c5_ = sa_optimize(h2, n, dx, 0.003, 1e-9, 0.99995, t2, rng)
    if c5_ < best_c5:
        best_c5 = c5_
        best_h = h2_.copy()

    # Phase 3: Local refinement (15% of time)
    t3 = start_time + budget_s * 0.95
    h3 = best_h.copy()
    h3, c5_ = local_refine(h3, n, dx, t3, rng)
    if c5_ < best_c5:
        best_c5 = c5_
        best_h = h3.copy()

    # Final verification
    best_h = make_valid(best_h, n)
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
