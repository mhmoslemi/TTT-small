import numpy as np
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] += diff / count
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, n, active):
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s == 0:
            grad += 1.0 - 2.0 * h
        elif s > 0:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] -= h[:n - s]
        else:
            sp = -s
            grad[sp:] += 1.0 - h[:n - sp]
            grad[:n - sp] -= h[sp:]
    return grad * dx

def optimize_from(h, n, t_start, budget_s, iters, alpha_init):
    """Projected gradient descent with backtracking."""
    h = project_constraint(h, n).copy()
    best_h = h.copy()
    best_score = compute_score(h)
    alpha = alpha_init
    
    for it in range(iters):
        if time.time() - t_start > budget_s - 20:
            break
        
        score_c = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(score_c)
        threshold = max_c - max(0.01, 0.002 * max_c)
        active = np.where(score_c >= threshold)[0]
        
        grad = compute_subgradient(h, n, active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking
        step = alpha
        improved = False
        for _ in range(20):
            h_new = h - step * grad
            h_new = project_constraint(h_new, n)
            ns = compute_score(h_new)
            if ns < best_score - 1e-15:
                h = h_new
                best_h = h.copy()
                best_score = ns
                improved = True
                break
            step *= 0.5
            if step < 1e-10:
                break
        
        if improved:
            alpha = min(alpha * 1.2, 0.5)
        else:
            alpha *= 0.7
    
    if compute_score(best_h) < best_score:
        best_score = compute_score(best_h)
    return best_h, best_score

    """Gradient of the k-th element of correlate(h, 1-h, 'full') w.r.t. h."""
    grad = np.zeros(n)

    lo1 = max(0, k)

    # Actually: j-k >= 0 → j >= k (if k > 0) or j >= 0 (if k <= 0)

import numpy as np

def project_h(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                h.fill(0.5)
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                h.fill(0.5)
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    h += (target - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def pgd_step(h, n, t_start, budget, max_iters=50000, alpha0=0.05):
    """Projected gradient descent."""
    h = project_h(h, n).copy()
    best_h = h.copy()

    alpha = alpha0
    stall = 0
    
    for it in range(max_iters):

            break
        
        cv = c5_vec(h)
        mx = np.max(cv)
        # Active set: top values within 1% of max
        thresh = mx - max(0.005, 0.005 * mx)
        active = np.where(cv >= thresh)[0]
        
        # Average gradient over active set
        grad = np.zeros(n)
        for k in active:
            grad += num_grad(h, n, k)
        grad /= len(active)

        if gnorm < 1e-12:

        alpha_try = alpha
        improved = False
        for _ in range(20):

            ns = c5_score(h_new)
            if ns < best_s - 1e-14:

                best_h = h.copy()
                best_s = ns
                improved = True
                stall = 0
                break
            alpha_try *= 0.5
            if alpha_try < 1e-12:

        else:

            stall += 1

            h = project_h(h + np.random.randn(n) * 0.1, n)
            stall = 0
            alpha = alpha0
