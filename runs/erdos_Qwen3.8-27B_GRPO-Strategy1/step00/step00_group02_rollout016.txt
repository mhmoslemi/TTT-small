    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    return np.clip(h - alpha, 0.0, 1.0)

def obj_func(x, n, beta, penalty):

    x = h + rng.randn(n) * scale
    return project_feasible(np.clip(x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize
import time


def smooth_max(corr, beta):
    M = np.max(corr)
    shifted = corr - M
    return M + np.log(np.sum(np.exp(beta * shifted))) / beta


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    return np.clip(h - alpha, 0.0, 1.0)


def objective(x, n, beta, penalty):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(corr, beta)
    dx = 2.0 / n
    pen = penalty * (np.sum(h) - n / 2.0) ** 2
    return sm * dx + pen


def make_perturbed(h, rng, scale):
    n = len(h)
    x = h + rng.randn(n) * scale
    return project_feasible(np.clip(x, 0.0, 1.0), n)


def make_structured(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, max(4, n // 8))
    edges = np.sort(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    edges = np.concatenate([[0], edges, [n]])
    for i in range(len(edges) - 1):
        val = rng.choice([0.0, 1.0, 0.2, 0.4, 0.6, 0.8])
        h[edges[i]:edges[i + 1]] = val
    return project_feasible(h, n)


def make_symmetric(n, rng, base):
    x = base.copy()
    half = n // 2
    pert = rng.randn(half) * 0.15
    x[:half] = np.clip(x[:half] + pert, 0, 1)
    if n % 2 == 0:
        x[half:] = x[:half][::-1]
    else:
        x[half + 1:] = x[half - 1:0:-1]
        x[half] = min(1.0, max(0.0, x[half]))
    return project_feasible(x, n)


def local_polish(h, n, rng, n_iters):
    best = h.copy()
    best_val = eval_c5(best)
    for _ in range(n_iters):
        x = make_perturbed(best, rng, 0.05)
        try:
            res = minimize(
                objective, x, args=(n, 80.0, 5000.0),
                method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 100, 'ftol': 1e-12, 'maxfun': 2000}
            )
            h_opt = project_feasible(res.x, n)
            v = eval_c5(h_opt)
            if v < best_val:
                best_val = v
                best = h_opt
        except Exception:
            pass
    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()
    deadline = t_start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.full(45, 0.5)

    n = len(h_init)
    rng = np.random.RandomState(seed)
    bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(h_init, n)
    best_c5 = eval_c5(best_h)

    # Phase 1: Main optimization loop with diverse starts
    iteration = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 20:
            break

        # Choose starting point strategy
        r = iteration % 6
        if r == 0:
            x0 = best_h.copy()
        elif r == 1:
            x0 = make_perturbed(best_h, rng, 0.1)
        elif r == 2:
            x0 = make_perturbed(best_h, rng, 0.25)
        elif r == 3:
            x0 = make_structured(n, rng)
        elif r == 4:
            x0 = make_symmetric(n, rng, best_h)
        else:
            # Gradient-free nudge: flip a random subset
            x0 = best_h.copy()
            mask = rng.rand(n) < 0.3
            x0[mask] = 1.0 - x0[mask]
            x0 = project_feasible(x0, n)

        beta_use = 40.0 + 20.0 * min(iteration / 50.0, 1.0)
        try:
            res = minimize(
                objective, x0, args=(n, beta_use, 2000.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 400, 'ftol': 1e-11, 'gtol': 1e-8, 'maxfun': 6000}
            )
            h_opt = project_feasible(res.x, n)
            c5_val = eval_c5(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
        except Exception:
            pass
        iteration += 1

    # Phase 2: Local polish around best solution
    time_left = deadline - time.time()
    if time_left > 30:
        n_polish = max(1, int(time_left / 35))
        best_h, best_c5 = local_polish(best_h, n, rng, n_polish)

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, n
