import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_subgrad(h, n):
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    max_val = np.max(corr)
    active = corr >= max_val - 1e-9
    n_act = max(np.sum(active), 1)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if not active[j]:
            continue
        lag = j - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            m = np.arange(lo1, hi1 + 1)
            grad[m] += (1.0 - h[m - lag])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            m = np.arange(lo2, hi2 + 1)
            grad[m] -= h[m + lag]
    grad /= n_act
    grad *= dx
    return grad, max_val * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
    return h

def slsqp_optimize(h0, n, max_iter=150, smooth_eps=0.0):
    h0 = project_feasible(h0.copy(), n)
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        if smooth_eps > 0:
            scaled = corr * dx / smooth_eps
            m = np.max(scaled)
            return smooth_eps * (m + np.log(np.sum(np.exp(scaled - m))))
        return np.max(corr) * dx
    def grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        if smooth_eps > 0:
            scaled = corr / smooth_eps
            m = np.max(scaled)
            weights = np.exp(scaled - m)
            weights /= np.sum(weights)
            g = np.zeros(n)
            for j in range(2 * n - 1):
                if weights[j] < 1e-12:
                    continue
                lag = j - (n - 1)
                lo1 = max(0, lag)
                hi1 = min(n - 1, n - 1 + lag)
                if lo1 <= hi1:
                    mm = np.arange(lo1, hi1 + 1)
                    g[mm] += (1.0 - x[mm - lag]) * weights[j]
                lo2 = max(0, -lag)
                hi2 = min(n - 1, n - 1 - lag)
                if lo2 <= hi2:
                    mm = np.arange(lo2, hi2 + 1)
                    g[mm] -= x[mm + lag] * weights[j]
            return g
        g, _ = compute_subgrad(x, n)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-12})
    return project_feasible(res.x, n)

def thue_morse_sequence(n):
    seq = np.zeros(n)
    for i in range(n):
        seq[i] = bin(i).count('1') % 2
    return seq - 0.5  # center around 0, scale to [0,1] later

def golden_ratio_sequence(n):
    seq = np.zeros(n)
    for i in range(n):
        seq[i] = (i * np.sqrt(5)) % 1.0
    return (seq < 0.5).astype(float)
