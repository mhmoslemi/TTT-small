        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 80
    ts = n / 2.0

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).flatten()
    else:
        h0 = np.full(n, 0.5)
    if h0.size != n:
        h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, h0.size), h0)
    h0 = repair_sum(h0, ts)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Also try binary projection as secondary start
    h_bin = (h0 >= 0.5).astype(np.float64)
    n_ones = int(h_bin.sum())
    if n_ones != n // 2:
        if n_ones > n // 2:
            ones = np.where(h_bin == 1)[0]
            rm = np.random.choice(ones, n_ones - n // 2, replace=False)
            h_bin[rm] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            add = np.random.choice(zeros, n // 2 - n_ones, replace=False)
            h_bin[add] = 1
    h_bin = repair_sum(h_bin, ts)
    bin_c5 = compute_c5(h_bin)
    if bin_c5 < best_c5:
        best_c5 = bin_c5
        best_h = h_bin.copy()

    # SA with mixed moves
    T = 0.005
    T_floor = 1e-9
    h = best_h.copy()
    c5_h = best_c5
    stall = 0
    stall_limit = 5000

    while time.time() - t0 < budget_s - 10:
        r = np.random.random()

        if r < 0.35:
            # Random pair move
            i = int(np.random.randint(n))
            j = int(np.random.randint(n))
            if i == j:
                continue
            s = 0.005 + 0.1 * min(1.0, T / 0.005)
            d = np.random.uniform(-s, s)
            h_new = h.copy()
            h_new[i] += d
            h_new[j] -= d
            if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                continue

        elif r < 0.70:
            # Informed: target peak correlation lag
            corr = np.correlate(h, 1.0 - h, mode='full')
            pk = int(np.argmax(corr))
            lag = pk - (n - 1)
            if lag >= 0:
                c = h[:n - lag] * (1.0 - h[lag:])
                top = int(np.argmax(c))
                i, j = top, top + lag
            else:
                kl = -lag
                c = h[kl:] * (1.0 - h[:n - kl])
                top = int(np.argmax(c))
                i, j = top + kl, top
            d = np.random.uniform(0.02, 0.2)
            if np.random.random() < 0.5:
                d = -d
            h_new = h.copy()
            h_new[i] += d
            h_new[j] -= d
            if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                continue

        elif r < 0.85:
            # Block move: shift a small block
            bl = int(np.random.randint(1, 4))
            i = int(np.random.randint(n - bl))
            j = int(np.random.randint(n - bl))
            if abs(i - j) < bl:
                continue
            d = np.random.uniform(-0.08, 0.08)
            h_new = h.copy()
            h_new[i:i + bl] += d
            h_new[j:j + bl] -= d
            if h_new[i:i+bl].min() < 0 or h_new[i:i+bl].max() > 1 or h_new[j:j+bl].min() < 0 or h_new[j:j+bl].max() > 1:
                continue

        else:
            # Swap-like: transfer mass from higher to lower
            i = int(np.random.randint(n))
            j = int(np.random.randint(n))
            if i == j:
                continue
            mx = min(h[i], 1.0 - h[j])
            d = mx * np.random.uniform(0.2, 0.7)
            if h[i] < 0.5:
                d = -d
                mx = min(1.0 - h[i], h[j])
                d = -mx * np.random.uniform(0.2, 0.7)
            h_new = h.copy()
            h_new[i] += d
            h_new[j] -= d
            if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                continue

        new_c5 = compute_c5(h_new)
        delta = new_c5 - c5_h
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, T_floor)):
            h = h_new
            c5_h = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            T = max(T_floor, 0.99995 * T)
        else:
            stall += 1

        if stall > stall_limit:
            # Perturbation restart
            kcnt = n // 3
            kick = np.random.choice(n, kcnt, replace=False)
            h[kick] += np.random.randn(kcnt) * 0.07
            h = repair_sum(h, ts)
            c5_h = compute_c5(h)
            T = 0.005
            stall = 0
            stall_limit = max(3000, stall_limit + 500)

    # Final polish: very small random moves
    time_remaining = budget_s - 10 - (time.time() - t0)
    if time_remaining > 5:
        T = 0.0005
        h = best_h.copy()
        c5_h = best_c5
        while time.time() - t0 < budget_s - 5:
            i = int(np.random.randint(n))
            j = int(np.random.randint(n))
            if i == j:
                continue
            d = np.random.uniform(-0.01, 0.01)
            h_new = h.copy()
            h_new[i] += d
            h_new[j] -= d
            if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                continue
            new_c5 = compute_c5(h_new)
            if new_c5 < c5_h:
                h = h_new
                c5_h = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()

    best_h = repair_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
