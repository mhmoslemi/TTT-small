def compute_c5(h, n_points):

    corr = np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, target):
    """Fix sum(h) to exactly target, preserving [0,1] bounds."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Spread adjustment across all elements proportionally to available capacity
    if diff > 0:
        capacity = 1.0 - h
        total_cap = capacity.sum()
        if total_cap < 1e-12:
            return h
        h += capacity * (diff / total_cap)
    else:
        capacity = h
        total_cap = capacity.sum()
        if total_cap < 1e-12:
            return h
        h -= capacity * (-diff / total_cap)
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):

    corr = np.correlate(h, 1.0 - h, mode="full")

def fix_sum_exact(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-14
        if not mask.any():
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 1e-14
        if not mask.any():
            return h
        h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + residual, 0.0, 1.0))
