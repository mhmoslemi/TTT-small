import numpy as np
from scipy.optimize import minimize
import time

_curr_dx = 2.0
_curr_tsum = 1.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_func(h):
    return compute_c5(h, _curr_dx)

def sum_fun(h):
    return np.sum(h) - _curr_tsum

def project_h(h, ts):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-10:
        h = h * (ts / s)
    return np.clip(h, 0.0, 1.0)

            i, j = rng.randint(n), rng.randint(n)

    if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
