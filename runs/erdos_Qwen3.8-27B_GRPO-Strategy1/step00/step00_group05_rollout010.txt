    # np.correlate(a, v, 'full')[n] = sum_i a[i] * v[i - n + len(v) - 1]

    # corrs[k] = sum_i h[i] * (1-h)[i - k + n - 1]
    #        = sum_i h[i] * (1 - h[i - k + n - 1])

    lag = max_idx - (n - 1)
    
    grad = np.zeros(n)
    # c_k = sum_i h[i]*(1-h[i+lag]) for valid i, i+lag
    # dc_k/dh[j] = (1-h[j+lag]) if j+lag in [0,n-1]
    #              + (-h[j-lag]) if j-lag in [0,n-1]  (from h[j] being in the second term)

    #           + sum_i h[i]*(-1)*delta(i+lag=j) = -h[j-lag] (if j-lag valid)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def c5_grad(h, n):
    dx = 2.0 / n
    om = 1.0 - h
    corrs = np.correlate(h, om, mode='full')
    mi = int(np.argmax(corrs))
    c5 = float(corrs[mi] * dx)
    lag = mi - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, lag)
    hi1 = min(n, lag + n)
    if hi1 > lo1:
        g[lo1:hi1] += 1.0 - h[lo1 - lag:hi1 - lag]
    lo2 = max(0, -lag)
    hi2 = min(n, n - lag)
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 + lag:hi2 + lag]
    g *= dx
    return c5, g
