    if elapsed() < budget_s * 0.7:
        h_cd = coord_descent(best_h, n0, alpha=100.0, n_sweeps=5)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd

    # Phase 4: Try upscaling to n_new and optimize
    if elapsed() < budget_s * 0.85:
        for scale in [2, 3]:
            if elapsed() > budget_s * 0.85:
                break
            n_new = n0 * scale
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n_new, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_feasible(h_up, n_new)

            alpha_up = 50.0
            constraints = [{'type': 'eq',
                            'fun': lambda x: np.sum(x) - n_new / 2.0,
                            'jac': lambda x: np.ones(n_new)}]
            bounds = [(0.0, 1.0)] * n_new
            try:
                res = minimize(obj_func, h_up, args=(n_new, alpha_up), method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 150, 'ftol': 1e-10})
                h_res = project_feasible(res.x, n_new)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5_res
            except Exception:
                pass

    # Phase 5: Final refinement at best resolution
    if elapsed() < budget_s * 0.95:
        nf = len(best_h)
        for alpha in [80.0, 150.0, 300.0]:
            if elapsed() > budget_s * 0.95:
                break
            constraints = [{'type': 'eq',
                            'fun': lambda x: np.sum(x) - nf / 2.0,
                            'jac': lambda x: np.ones(nf)}]
            bounds = [(0.0, 1.0)] * nf
            try:
                res = minimize(obj_func, best_h, args=(nf, alpha), method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 200, 'ftol': 1e-12})
                h_res = project_feasible(res.x, nf)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5_res
            except Exception:
                pass

        # One more coordinate descent pass
        if elapsed() < budget_s * 0.95:
            h_cd = coord_descent(best_h, nf, alpha=200.0, n_sweeps=3)
            c5_cd = compute_c5(h_cd)
            if c5_cd < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_cd

    # Final projection and verification
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
