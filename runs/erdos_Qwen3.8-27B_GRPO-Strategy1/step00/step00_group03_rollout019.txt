def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # log-sum-exp for numerical stability
    c = corr - corr.max()
    exp_c = np.exp(c / alpha)
    weights = exp_c / exp_c.sum()
    
    sg = np.zeros(n)
    shifts = np.arange(len(corr)) - (n - 1)  # d values
    
    for idx, d in enumerate(shifts):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        # (1-h[j-d]) for valid j
        lo1 = max(0, d)
        hi1 = min(n, n + d)
        if lo1 < hi1:
            sg[lo1:hi1] += w * (1.0 - h[lo1 - d:hi1 - d])
        # -h[j+d] for valid j
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            sg[lo2:hi2] -= w * h[lo2 + d:hi2 + d]
    
    return sg
