    m = np.max(vals)
    if m < 0:
        return 0.0
    v = vals - m
    return m + np.log(np.sum(np.exp(beta * np.clip(v, -20, 20)))) / beta

            mask = h < 1.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()

        except Exception:
            pass

    # Phase 3: Higher resolution optimization
    for n_target in [140, 280]:
        if time.time() - t0 > budget_s * 0.80:
            break
        h_up = np.interp(
            np.linspace(0, n0 - 1, n_target),
            np.arange(n0), best_h)
        h_up = fix_sum(h_up, n_target)

        for beta in [50.0, 200.0, 600.0]:
            if time.time() - t0 > budget_s * 0.80:
                break
            try:
                h_opt = _slsqp_optimize(h_up, n_target, beta)
                h_try = fix_sum(h_opt, n_target)
                c5_try = eval_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    h_up = h_try.copy()
            except Exception:
                pass

        deadline3 = t0 + budget_s * 0.80
        h_up, c5_local = _swap_search(h_up, n_target, rng, 8000, t0, deadline3)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_up.copy()

    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.90:
        n_final = len(best_h)
        try:
            h_opt = _slsqp_optimize(best_h, n_final, 1000.0)
            h_try = fix_sum(h_opt, n_final)
            c5_try = eval_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
