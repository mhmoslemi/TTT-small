import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full") * dx
    return float(np.max(corr))

def c5_grad(h, t=50.0):
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1-h, mode="full") * dx
    max_f = np.max(f)
    shifted = t * (f - max_f)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= w.sum()
    val = max_f + np.log(np.sum(np.exp(t * (f - max_f)))) / t
    
    u = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for k in range(n):
        end = min(n, 2*n-1-k)
        if end > 0:
            grad[k] = np.dot(w[k:k+end], u[:end])
    grad *= dx
    return val, grad

def project_capacity(h, target_sum):
    n = len(h)
    target_sum = float(np.clip(target_sum, 0.0, float(n)))
    eta_lo = h.min() - 1.0
    eta_hi = h.max()
    for _ in range(80):
        eta = (eta_lo + eta_hi) / 2.0
        x = np.clip(h - eta, 0.0, 1.0)
        if x.sum() > target_sum:
            eta_lo = eta
        else:
            eta_hi = eta
    return np.clip(h - (eta_lo + eta_hi)/2.0, 0.0, 1.0)

    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    
    def time_left():
        return budget_s - (time.time() - t0) - 10
    
    def try_optimize(h0, n, t_param=100.0, max_iter=300):
        nonlocal best_h, best_c5
        if time_left() < 20:
            return
        dx = 2.0 / n
        target_sum = n / 2.0
        h0 = project_capacity(h0.copy(), target_sum)
        
        # SLSQP optimization
        def obj(h):
            v, _ = c5_grad(h, t_param)
            return v
        def obj_grad(h):
            _, g = c5_grad(h, t_param)
            return g
        def cons_fn(h):
            return np.sum(h) - target_sum
        def cons_jac(h):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}]
        
        try:
            res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-12})
            h_opt = project_capacity(res.x, target_sum)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        except Exception:
            pass
        
        # Also try projected gradient
        if time_left() < 10:
            return
        h_pg = h0.copy()
        alpha = 1.0 / (n * dx)  # step size
        for it in range(500):
            if time_left() < 10:
                break
            v, g = c5_grad(h_pg, t_param)
            h_new = h_pg - alpha * g
            h_new = project_capacity(h_new, target_sum)

            if v_new < v:

        except:

    n_candidates = [47, 49, 51, 55, 60, 70, 80, 90, 100]

    if best_h is None:
        best_h = np.full(47, 0.5)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
