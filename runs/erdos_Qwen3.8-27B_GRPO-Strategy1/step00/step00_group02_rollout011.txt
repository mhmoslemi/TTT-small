import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair_h(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14 * n:
            break
        frac = diff / n
        if h.min() + frac >= 0 and h.max() + frac <= 1:
            h += frac
        else:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h, n, lam):
    h_c = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    c5 = np.max(corr) * 2.0 / n
    viol = np.sum(h_c) - n / 2.0
    return c5 + lam * viol * viol

def _optimize_level(h, n, rng, t0, budget_s, max_tries=3):
    h_best = h.copy()
    c5_best = _compute_c5(h_best)
    time_avail = budget_s * 0.85 - (time.time() - t0)
    max_tries = min(max_tries, max(1, int(time_avail / 40)))
    for r in range(max_tries):
        if time.time() - t0 > budget_s * 0.85:
            break
        if r == 0:
            x0 = h_best.copy()
        elif r == 1:
            x0 = h_best + rng.normal(0, 0.05, n)
            x0 = np.clip(x0, 0, 1)
        else:
            x0 = h_best + rng.normal(0, 0.12, n)
            x0 = np.clip(x0, 0, 1)
        res = minimize(
            _obj, x0, args=(n, 8.0),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        h_try = _repair_h(res.x, n)
        c5_try = _compute_c5(h_try)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n_cur = len(h0)
    h_cur = _repair_h(h0, n_cur)
    best_h = h_cur.copy()
    best_c5 = _compute_c5(best_h)

    max_n = 360
    while n_cur < max_n:
        if time.time() - t0 > budget_s * 0.75:
            break
        h_cur = np.repeat(h_cur, 2)
        n_cur = len(h_cur)
        h_cur = _repair_h(h_cur, n_cur)
        h_cur, c5_now = _optimize_level(h_cur, n_cur, rng, t0, budget_s, max_tries=3)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h_cur.copy()

    # Additional restarts at the finest resolution we reached
    if time.time() - t0 < budget_s * 0.85:
        n_final = len(best_h)
        for r in range(3):
            if time.time() - t0 > budget_s * 0.9:
                break
            x0 = best_h + rng.normal(0, 0.08 + 0.04 * r, n_final)
            x0 = np.clip(x0, 0, 1)
            res = minimize(
                _obj, x0, args=(n_final, 8.0),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n_final,
                options={'maxiter': 600, 'ftol': 1e-12, 'gtol': 1e-9}
            )
            h_try = _repair_h(res.x, n_final)
            c5_try = _compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    best_h = _repair_h(best_h, len(best_h))
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, len(best_h)
