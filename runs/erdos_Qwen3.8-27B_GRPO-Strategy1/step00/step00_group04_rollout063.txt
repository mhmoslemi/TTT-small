def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def smooth_max_val(corr, tau):
    c = corr / tau
    c -= c.max()
    e = np.exp(c)
    return tau * np.log(e.sum()), e / e.sum()

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def pgd_optimize(h, n, max_iter, tau_start, tau_end, lr_start, lr_end, time_budget):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(max_iter):
        if time.time() - start > time_budget:
            break
        t = i / max(max_iter - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        
        # Compute gradient
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        _, weights = smooth_max_val(corr, tau)
        v = 1.0 - 2.0 * h
        
        grad = np.zeros(n)
        for j in range(n):
            s = n - 1 - j
            grad[j] = dx * np.dot(weights[s:s+n], v)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new

For S1: let m = j + k - (n-1). As k ranges 0..2n-2, m ranges j-(n-1) .. j+n-1. But we need m in [0,n-1], so m ranges max(0, j-(n-1)) .. min(n-1, j+n-1).

w_k where k = m - j + (n-1), so w_{m-j+n-1}.

S1[j] = sum_{m=max(0,j-n+1)}^{min(n-1,j+n-1)} w_{m-j+n-1} * (1-h[m])

For S2: let m = j - (k-(n-1)) = j + n - 1 - k. As k ranges 0..2n-2, m ranges j+n-1 .. j-n+1. We need m in [0,n-1].

w_k where k = j + n - 1 - m, so w_{j+n-1-m}.

S2[j] = sum_{m=max(0,j-n+1)}^{min(n-1,j+n-1)} w_{j+n-1-m} * h[m]

So: grad[j] = dx * (S1[j] - S2[j])

Now, S1[j] = sum_m w_{m-j+n-1} * (1-h[m]) for m in valid range
S2[j] = sum_m w_{j+n-1-m} * h[m] for m in valid range

Let me define vectors:
For S1: it's a weighted sum of (1-h) with weights that depend on j. The weight for component m is w_{m-j+n-1}. As j varies, this is a sliding window of w.

For S2: weighted sum of h with weights w_{j+n-1-m}. Similarly a sliding window.

    # A is Toeplitz with first row [w[n-1], w[n-2], ..., w[0]] and first col [w[n-1], w[n], ..., w[2n-2]]

    # A[0,j] = w[0-j+n-1] = w[n-1-j] for j=0..n-1 → w[n-1], w[n-2], ..., w[0]

    S2 = A @ h
    grad = dx * (S1 - S2)
