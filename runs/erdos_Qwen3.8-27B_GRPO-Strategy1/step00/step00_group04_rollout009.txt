def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1-h.astype(float), mode='full')
    return float(np.max(corr)) * dx

def scale_to_n(h_old, new_n, rng):
    old_n = len(h_old)
    if new_n == old_n:
        return h_old.astype(float).copy()
    # Linear interpolation then threshold
    x_old = np.arange(old_n, dtype=float)
    x_new = rng.integers(0, old_n*new_n//old_n, new_n) / (new_n//old_n) if new_n > old_n else ...
    # Simpler: use np.interp
    x_old = np.arange(old_n)
    x_new = np.arange(new_n) * old_n / new_n
    h_interp = np.interp(x_new, x_old, h_old.astype(float))
    h_bin = (h_interp >= 0.5).astype(float)
    # Fix sum
    target = new_n // 2
    s = int(h_bin.sum())
    if s < target:
        idx = np.where(h_bin == 0)[0]
        rng.shuffle(idx)
        h_bin[idx[:target-s]] = 1
    elif s > target:
        idx = np.where(h_bin == 1)[0]
        rng.shuffle(idx)
        h_bin[idx[:s-target]] = 0
    return h_bin

        h, c5 = local_search(best_h, min(remaining * 0.7, 200.0), rng,
                             stall_threshold=8000, max_resets=10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    n_pts = len(best_h)
    dx = 2.0 / n_pts
    assert abs(best_h.sum() * dx - 1.0) < 1e-9, f"Sum constraint violated: {best_h.sum() * dx}"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Range constraint violated"

    return best_h, best_c5, n_pts
