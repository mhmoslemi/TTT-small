import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, T):
    m = np.max(vals)
    if T < 1e-8:
        return float(m)
    return float(m + np.log(np.sum(np.exp((vals - m) * T))) / T)

class _Obj:
    def __init__(self, T):
        self.T = T
    def __call__(self, x):
        dx = 2.0 / len(x)
        return smooth_max(np.correlate(x, 1 - x, mode='full') * dx, self.T)

class _Con:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return float(np.sum(x) - self.n * 0.5)

def _fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = n * 0.5 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] += diff / m.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = n * 0.5 - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmin(h))
        h[i] = float(np.clip(h[i] + diff, 0.0, 1.0))
    return h

def _upscale(h_c, n_f):
    n_c = len(h_c)
    out = np.empty(n_f)
    for i in range(n_f):
        p = i * n_c / n_f
        i0 = int(p) % n_c
        f = p - int(p)
        out[i] = (1.0 - f) * h_c[i0] + f * h_c[(i0 + 1) % n_c]
    return _fix_sum(out, n_f)

def _slsqp(h, n, temps, t0, budget):
    bh = h.copy()
    bc = compute_c5(h)
    h_c = h.copy()
    bnds = [(0.0, 1.0)] * n
    cf = _Con(n)
    cons = [{'type': 'eq', 'fun': cf}]
    for T in temps:
        if budget - (time.time() - t0) < 12:
            break
        of = _Obj(T)
        r = minimize(of, h_c, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': 5000, 'ftol': 1e-15})
        if r.x is not None and not np.any(np.isnan(r.x)):
            hn = _fix_sum(r.x, n)
            c = compute_c5(hn)
            if c < bc:
                bc = c
                bh = hn
            h_c = hn
    return bh, bc

def _perturb(h, n, t0, budget, nmax=2000):
    bh = h.copy()
    bc = compute_c5(h)
    h_c = h.copy()
    sig = 0.06
    rng = np.random.default_rng(99)
    for it in range(nmax):
        if budget - (time.time() - t0) < 12:
            break
        np_ = int(rng.integers(1, 4))
        ix = rng.choice(n, np_, replace=False)
        ht = h_c.copy()
        ht[ix] += rng.normal(0.0, sig, np_)
        ht = _fix_sum(ht, n)
        c = compute_c5(ht)
        if c < bc - 1e-12:
            bc = c
            bh = ht
            h_c = ht
            sig = min(sig * 1.1, 0.2)
        elif it % 80 == 0 and sig > 0.01:
            sig *= 0.9
    return bh, bc

def _binary_round(h, n):
    n_ones = int(round(n * 0.5))
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n_ones]] = 1.0
    return h_bin, compute_c5(h_bin)
