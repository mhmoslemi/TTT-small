    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: SA + gradient on original resolution (45% of time)
    p1_end = start_time + time_limit * 0.45
    T_init = 0.08 * best_c5
    for restart in range(30):
        if time.time() > p1_end:
            break
        h_s, c5_s = sa_search(best_h.copy(), n, dx, T_init, p1_end, start_time, rng)
        if c5_s < best_c5:
            best_h = h_s.copy()
            best_c5 = c5_s
        h_g, c5_g = gradient_descent(best_h.copy(), n, dx, p1_end, start_time)
        if c5_g < best_c5:
            best_h = h_g.copy()
            best_c5 = c5_g
        T_init = max(0.01 * best_c5, 0.002)

    # Phase 2: Coordinate refinement (5%)
    p2_end = start_time + time_limit * 0.52
    h_cr, c5_cr = coordinate_refine(best_h.copy(), n, dx, p2_end, start_time)
    if c5_cr < best_c5:
        best_h = h_cr
        best_c5 = c5_cr

    # Phase 3: Alternative starts (13%)
    p3_end = start_time + time_limit * 0.65
    for alt in range(4):
        if time.time() > p3_end:
            break
        rng2 = np.random.RandomState(seed + 7777 + alt * 131)
        h_alt = np.zeros(n)
        ones_idx = rng2.choice(n, size=n // 2, replace=False)
        h_alt[ones_idx] = 1.0
        c5_alt = compute_c5(h_alt, dx)
        if c5_alt < best_c5 * 1.15:
            h_s, c5_s = sa_search(h_alt, n, dx, 0.04 * c5_alt, p3_end, start_time, rng2)
            h_g, c5_g = gradient_descent(h_s, n, dx, p3_end, start_time)
            if c5_g < best_c5:
                best_h = h_g
                best_c5 = c5_g

    # Phase 4: Upscale and refine (30%)
    p4_end = start_time + time_limit * 0.95
    for scale in [2, 4]:
        n_new = n * scale
        if n_new > 2000 or time.time() > p4_end:
            break
        dx_new = 2.0 / n_new
        h_up = np.interp(
            np.linspace(0.0, 2.0, n_new, endpoint=False),
            np.linspace(0.0, 2.0, n, endpoint=False),
            best_h
        )
        target_new = n_new / 2.0
        h_up += (target_new - np.sum(h_up)) / n_new
        h_up = np.clip(h_up, 0.0, 1.0)
        c5_up = compute_c5(h_up, dx_new)
        h_s, c5_s = sa_search(h_up, n_new, dx_new, 0.015 * c5_up, p4_end, start_time, rng)
        h_g, c5_g = gradient_descent(h_s, n_new, dx_new, p4_end, start_time)
        if c5_g < best_c5:
            best_h = h_g
            best_c5 = c5_g
            best_n = n_new
            n = n_new

    # Final polish with remaining time
    if time.time() < start_time + time_limit * 0.98:
        h_f, c5_f = gradient_descent(best_h.copy(), best_n, 2.0 / best_n,
                                     start_time + time_limit * 0.98, start_time)
        if c5_f < best_c5:
            best_h = h_f
            best_c5 = c5_f

    final_c5 = compute_c5(best_h, 2.0 / best_n)
    # Ensure sum constraint is met
    target_final = best_n / 2.0
    diff = target_final - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, final_c5, best_n
