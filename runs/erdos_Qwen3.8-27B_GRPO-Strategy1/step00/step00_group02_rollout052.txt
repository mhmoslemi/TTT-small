        if free.sum() > 0:
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_c5(h):
    n = len(h)
    corr = get_corr(h)
    return np.max(corr) * 2.0 / n

def get_subgrad(h, corr, top_frac=0.05):
    n = len(h)
    dx = 2.0 / n
    max_c = np.max(corr)
    if max_c < 1e-15:
        return np.zeros(n)
    thresh = max_c - top_frac * max_c
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        # For each m: grad[m] += (1-h[m-k])*[0<=m-k<n] - h[m+k]*[0<=m+k<n]
        # Vectorized:
        # term1: (1-h[m-k]) for m where 0<=m-k<n, i.e., k<=m<n+k (if k>=0) or 0<=m<n-k (if k<0)
        # term2: -h[m+k] for m where 0<=m+k<n, i.e., -k<=m<n-k (if k>=0) or 0<=m<n+k (if k<0)
        if k >= 0:
            # term1: m from k to n-1, (1-h[m-k])
            mask1 = np.zeros(n, dtype=bool)
            mask1[k:] = True
            grad[mask1] += (1.0 - h[:n-k])
            # term2: m from 0 to n-1-k, -h[m+k]
            mask2 = np.zeros(n, dtype=bool)
            mask2[:n-k] = True
            grad[mask2] -= h[k:]
        else:
            kk = -k
            # term1: m from 0 to n-1-kk, (1-h[m+kk]) = (1-h[m-k])
            mask1 = np.zeros(n, dtype=bool)
            mask1[:n-kk] = True
            grad[mask1] += (1.0 - h[kk:])
            # term2: m from kk to n-1, -h[m-kk] = -h[m+k]
            mask2 = np.zeros(n, dtype=bool)
            mask2[kk:] = True
            grad[mask2] -= h[:n-kk]
    grad *= dx / len(active)
    return grad

def smooth_obj_and_grad(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    scaled = T * corr * dx
    max_s = np.max(scaled)
    expscaled = np.exp(scaled - max_s)
    w = expscaled / expscaled.sum()  # weights
    obj = (max_s + np.log(expscaled.sum())) / T
    # Gradient: sum_k w[k] * dx * ∂c[k]/∂h[m] for each m
    # (the T and dx cancel in the weight normalization)
    # Actually: ∂f/∂h[m] = sum_k w[k] * dx * ∂c[k]/∂h[m]
    # because f = (1/T)*log(sum exp(T*c[k]*dx)) 
    # ∂f/∂h[m] = (1/T) * sum_k [T*dx*∂c[k]/∂h[m] * exp(T*c[k]*dx)] / sum_j exp(T*c[j]*dx)
    # = dx * sum_k w[k] * ∂c[k]/∂h[m]
    
    grad = np.zeros(n)
    # Only use non-negligible weights for efficiency

            kk = -k
            grad[:n-kk] += w[idx] * (1.0 - h[kk:])

        t_bin = min(remaining * 0.4, 100)
        h_b, c5_b = proj_grad_descent(h_bin, max_time=t_bin, seed=seed + 50)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b
    
    # Phase 4: try larger resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        for scale in [2, 3]:
            n_new = n0 * scale
            if n_new > 200 or remaining < 40:
                break
            x_old = np.linspace(0, 2, n0)
            x_new = np.linspace(0, 2, n_new)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_onto_feasible(h_up, n_new / 2.0)
            t_up = min(remaining * 0.5, 150)
            h_up, c5_up = proj_grad_descent(h_up, max_time=t_up, seed=seed + 200)
            if c5_up < best_c5:
                best_h, best_c5 = h_up, c5_up
            remaining = budget_s - (time.time() - t0)
    
    # Final: ensure feasibility and return
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_feasible(best_h, n_final / 2.0)
    final_c5 = get_c5(best_h)
    
    return (best_h, final_c5, n_final)
