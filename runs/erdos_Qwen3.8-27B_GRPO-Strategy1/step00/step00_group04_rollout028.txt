    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return h

        idx = np.argsort(-h)
        rem = diff
        for i in idx:
            take = min(h[i], rem)
            h[i] -= take
            rem -= take
            if rem < 1e-14:

    else:

        rem = -diff

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.argsort(-h)
        rem = diff
        for i in idx:
            take = min(h[i], rem)
            h[i] -= take
            rem -= take
            if rem < 1e-14:
                break
    else:
        idx = np.argsort(h)
        rem = -diff
        for i in idx:
            take = min(1.0 - h[i], rem)
            h[i] += take
            rem -= take
            if rem < 1e-14:
                break
    return h

def smax(corr, T):
    c = corr.max()
    sh = np.clip((corr - c) * T, -50, 0)
    return c + np.log(np.sum(np.exp(sh))) / T

def smax_grad(corr, h, n, dx, T):
    c = corr.max()
    sh = np.clip((corr - c) * T, -50, 0)
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                grad[d:] += wk * (1.0 - h[:n - d])
            if d == 0:
                grad[:] += -wk * h
            else:
                grad[:n - d] += -wk * h[d:]
        else:
            dd = -d
            if dd < n:
                grad[:n - dd] += wk * (1.0 - h[dd:])
            grad[dd:] += -wk * h[:n - dd]
    return grad

def make_obj_jac(nt, T, lam):
    def obj(x):
        h = np.clip(x, 0, 1)
        dx = 2.0 / nt
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return smax(corr, T) + lam * (h.sum() - nt / 2.0) ** 2
    def jac(x):
        h = np.clip(x, 0, 1)
        dx = 2.0 / nt
        corr = np.correlate(h, 1 - h, mode='full') * dx
        g = smax_grad(corr, h, nt, dx, T)
        g += 2.0 * lam * (h.sum() - nt / 2.0)
        return g
    return obj, jac

def optimize_level(h_start, nt, t0, budget_s, lam=500.0):
    h = project_sum(h_start, nt)
    for T in [25, 80, 250, 800]:
        if time.time() - t0 > budget_s * 0.78:

        obj, jac = make_obj_jac(nt, T, lam)
        try:

                           bounds=[(0.0, 1.0)] * nt,
                           options={'maxiter': 250, 'gtol': 1e-8, 'maxfun': 4000})
            h2 = project_sum(res.x, nt)
            c5_2 = compute_c5(h2)
            if c5_2 <= compute_c5(h) + 1e-10:

def coord_search(h, nt, t0, budget_s, n_rounds=3):
    for _ in range(n_rounds):
        if time.time() - t0 > budget_s * 0.8:
            break
        improved = False
        for i in range(nt):

                h[i + 1 if i + 1 < nt else i - 1] += (h_save - trial)
