import time

def compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_grad_single(h, s):
    """Gradient of c[s] (the s-th correlation scaled by dx) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j - s and j - s < n:
            g += (1.0 - h[j - s])
        if 0 <= j + s and j + s < n:
            g -= h[j + s]
        grad[j] = dx * g
    return grad

def smooth_obj_grad(h, tau):
    n = len(h)
    c = compute_c(h)
    c_max = c.max()
    if tau < 1e8:
        shifted = tau * (c - c_max)
        exps = np.exp(shifted)
        p = exps / exps.sum()
        obj = c_max + np.log(exps.sum()) / tau
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] > 1e-10:
                s = k - (n - 1)
                grad += p[k] * corr_grad_single(h, s)
        return obj, grad
    else:
        k_star = c.argmax()
        s = k_star - (n - 1)
        return c_max, corr_grad_single(h, s)

def project_constraint(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(200):

        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def all_correlation_grads(h):
    """Returns grad_matrix[k, j] = dc[k]/dh[j] for all k, j"""
    n = len(h)
    dx = 2.0 / n
    # For shift s = k - (n-1):
    # grad[k, j] = dx * [(1-h[j-s]) * I[0<=j-s<n] - h[j+s] * I[0<=j+s<n]]
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        # First term: valid when 0 <= j-s < n, i.e., s <= j < s+n
        lo1 = max(0, s)
        hi1 = min(n, n + s)  # j < n+s, j < n
        if lo1 < hi1:
            j_idx = np.arange(lo1, hi1)
            grads[k, j_idx] += dx * (1.0 - h[j_idx - s])
        # Second term: valid when 0 <= j+s < n, i.e., -s <= j < n-s
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            j_idx = np.arange(lo2, hi2)
            grads[k, j_idx] -= dx * h[j_idx + s]
    return grads
