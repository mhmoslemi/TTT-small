import numpy as np
import time
import math


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)


def binarize_top(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=np.float64)
    hb[idx[:target]] = 1.0
    return hb


def fix_binary_sum(h, rng):
    n = len(h)
    target = n // 2
    s = int(np.round(h.sum()))
    diff = target - s
    if diff > 0:
        z = np.where(h == 0)[0]
        if len(z) >= diff:
            h[rng.choice(z, diff, replace=False)] = 1
    elif diff < 0:
        o = np.where(h == 1)[0]
        if len(o) >= -diff:
            h[rng.choice(o, -diff, replace=False)] = 0
    return h


def local_search(h, max_time, rng, exhaustive_threshold=25000):
    n = len(h)
    h = h.astype(np.float64).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = fix_binary_sum(h, rng)
    c5 = compute_c5(h)

    while time.time() < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_pairs = len(ones) * len(zeros)
        improved = False

        if n_pairs <= exhaustive_threshold:
            for i in ones:
                if time.time() >= max_time:
                    break
                for j in zeros:
                    h[i], h[j] = 0.0, 1.0
                    nc = compute_c5(h)
                    if nc < c5 - 1e-15:
                        c5 = nc
                        improved = True
                        break
                    h[i], h[j] = 1.0, 0.0
                if improved:
                    break
        else:
            n_tries = min(30000, max(3000, 4 * n))
            for _ in range(n_tries):
                if time.time() >= max_time:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = 0.0, 1.0
                nc = compute_c5(h)
                if nc < c5 - 1e-15:
                    c5 = nc
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[i], h[j] = 1.0, 0.0

        if not improved:
            break

    return h, c5


def anneal(h, max_time, rng):
    n = len(h)
    h = h.astype(np.float64).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = fix_binary_sum(h, rng)
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    T = 0.006
    T_min = 0.0002
    stall = 0

    while T > T_min and time.time() < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc = compute_c5(h)
        d = nc - c5
        if d < 0 or (T > 1e-9 and rng.random() < math.exp(-d / T)):
            c5 = nc
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T *= 0.9993
        if stall > 3000:
            T = max(T * 0.85, T_min)
            stall = 0

    return best_h, best_c5


def upscale_interp(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    return binarize_top(h_interp)


def upscale_repeat(h_old, n_new):
    factor = n_new // len(h_old)
    h_rep = np.repeat(h_old, factor).astype(np.float64)
    remainder = n_new - len(h_rep)
    if remainder > 0:
        h_rep = np.concatenate([h_rep, h_old[:remainder]])
    return binarize_top(h_rep)
