    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    diff = target - h.sum()

def binary_project(h, n):
    """Try to make h close to binary"""
    h = h.copy().astype(float)
    threshold = 0.5
    h_bin = (h > threshold).astype(float)
    # Adjust to get correct sum
    target = n / 2.0
    current_sum = h_bin.sum()
    if abs(target - current_sum) < 1e-10:
        return h_bin
    # Need to adjust
    if target > current_sum:
        # Need more 1s: change some 0s to 1s
        zeros = np.where(h_bin == 0)[0]
        # Pick the ones with highest h values
        order = zeros[np.argsort(-h[zeros])]
        for idx in order:
            if h_bin.sum() >= target - 1e-10:
                break
            h_bin[idx] = 1.0
    else:
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        for idx in order:
            if h_bin.sum() <= target + 1e-10:
                break
            h_bin[idx] = 0.0
    
    # If sum is not exact, use fractional values
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        # Find a value to make fractional
        if diff > 0:
            candidates = np.where(h_bin == 0)[0]
        else:
            candidates = np.where(h_bin == 1)[0]
        if len(candidates) > 0:
            h_bin[candidates[0]] += diff
    return h_bin
