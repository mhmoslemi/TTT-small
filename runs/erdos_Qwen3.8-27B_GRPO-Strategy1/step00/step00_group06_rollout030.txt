        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5_new = compute_c5(h_cur)
        if c5_new <= c5_cur:
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_cur.copy()
        else:
            if rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-8)):
                c5_cur = c5_new
            else:
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        T *= 0.9997
        if T < 5e-6:
            T = 5e-6
    return h_best, c5_best


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 10

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Try parent
    try:
        parent = np.array(initial_h_values, dtype=float)
        n = len(parent)
        h = to_binary(parent, n, rng)
        dx = 2.0 / n
        if abs(h.sum() * dx - 1.0) > 0.05:
            h = np.clip(h, 0, 1)
            h = to_binary(h, n, rng)
        h, c5 = greedy_binary(h, n, rng, max_passes=40, time_end=t_end)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        # SA from parent
        if time.time() < t_end - 30:
            t_sa = min(t_end - 20, time.time() + 90)
            h2, c52 = sa_binary(best_h, best_n, 0.015, t_sa, rng)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2
                best_n = best_n
            h3, c53 = greedy_binary(best_h, best_n, rng, max_passes=30, time_end=t_end)
            if c53 < best_c5:
                best_c5 = c53
                best_h = h3
    except Exception:
        pass

    # Try various even n with random starts
    n_list = [96, 100, 128, 160, 200, 256]
    for n in n_list:
        if time.time() >= t_end - 40:
            break
        for trial in range(3):
            if time.time() >= t_end - 40:
                break
            h = np.zeros(n)
            ones_idx = rng.choice(n, size=n // 2, replace=False)
            h[ones_idx] = 1.0
            h, c5 = greedy_binary(h, n, rng, max_passes=30, time_end=t_end)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n
                # SA improvement
                if time.time() < t_end - 25:
                    t_sa = min(t_end - 15, time.time() + 50)
                    h2, c52 = sa_binary(best_h, best_n, 0.01, t_sa, rng)
                    if c52 < best_c5:
                        best_c5 = c52
                        best_h = h2
                    h3, c53 = greedy_binary(best_h, best_n, rng, max_passes=20, time_end=t_end)
                    if c53 < best_c5:
                        best_c5 = c53
                        best_h = h3

    # Final SA on global best
    if best_h is not None and time.time() < t_end - 5:
        t_sa = min(t_end - 2, time.time() + 60)
        h2, c52 = sa_binary(best_h, best_n, 0.02, t_sa, rng)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
        h3, c53 = greedy_binary(best_h, best_n, rng, max_passes=20, time_end=t_end)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3

    # Final verification
    if best_h is None:
        n = 96
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
