import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_to_simplex_box(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = np.sum(h) - target
    if diff > 0:
        # Need to decrease
        # Decrease the largest values first
        order = np.argsort(-h)
        remaining = diff
        for i in order:
            if h[i] > 0:
                dec = min(h[i], remaining)
                h[i] -= dec
                remaining -= dec
                if remaining < 1e-10:
                    break
    else:
        # Need to increase
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            if h[i] < 1:
                inc = min(1 - h[i], remaining)
                h[i] += inc
                remaining -= inc
                if remaining < 1e-10:
                    break
    return h

        m = np.max(corr)
        return m + np.log(np.sum(np.exp(t * (corr - m)))) / t

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_constraint(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def smooth_max_obj(h, tau):
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

def compute_grad(h, w, n):
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        m = idx - (n - 1)
        if m >= 0:
            if n - m > 0:
                grad[m:n] += wi * (1.0 - h[0:n - m])
                grad[0:n - m] -= wi * h[m:n]
        else:
            l = -m
            if n - l > 0:
                grad[0:n - l] += wi * (1.0 - h[l:n])
                grad[l:n] -= wi * h[0:n - l]
    return grad * dx_val

    t0 = time.time()

    def remaining():

    # Phase 1: SLSQP with smooth max, increasing temperature
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]

    for tau in [3, 5, 10, 20, 40, 80, 160]:
        if remaining() < 120:
            break
        for trial in range(6):
            if remaining() < 60:

            h_start = best_h.copy()

                h_start = project_constraint(h_start, n)
            try:
                res = minimize(lambda h: smooth_max_obj(h, tau), h_start,
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 400, 'ftol': 1e-12})
                h_new = project_constraint(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:

            pass

    return best_h, best_c5, n
