import numpy as np
import time

def project_feasible(h, target_sum):
    """Project h onto {sum(h)=target_sum, 0<=h<=1} via bisection on Lagrange mult."""
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, n):
    """Compute C5 = max_k correlate(h, 1-h, k) * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def nelder_mead(h0, n, target_sum, max_iter=300):
    """Nelder-Mead with box-plane projection."""
    h = project_feasible(h0, target_sum)
    f0 = compute_c5(h, n)
    simplex = [h.copy()]
    fvals = [f0]
    for i in range(n):
        v = h.copy()
        step = 0.08 if v[i] < 0.95 else -0.08
        v[i] += step
        v = project_feasible(v, target_sum)
        simplex.append(v.copy())
        fvals.append(compute_c5(v, n))
    for it in range(max_iter):
        idx = np.argsort(fvals)
        simplex = [simplex[j].copy() for j in idx]
        fvals = [fvals[j] for j in idx]
        if fvals[0] < 0.375:
            break
        diam = np.max(np.abs(simplex[-1] - simplex[0]))
        if diam < 1e-9:
            break
        centroid = np.mean(simplex[:-1], axis=0)
        centroid = project_feasible(centroid, target_sum)
        xr = centroid + (centroid - simplex[-1])
        xr = project_feasible(xr, target_sum)
        fr = compute_c5(xr, n)
        if fr < fvals[-2]:
            if fr < fvals[0]:
                xe = centroid + 2.0 * (centroid - simplex[-1])
                xe = project_feasible(xe, target_sum)
                fe = compute_c5(xe, n)
                if fe < fr:
                    simplex[-1] = xe
                    fvals[-1] = fe
                else:
                    simplex[-1] = xr
                    fvals[-1] = fr
            else:
                simplex[-1] = xr
                fvals[-1] = fr
        else:
            xc = centroid + 0.5 * (simplex[-1] - centroid)
            xc = project_feasible(xc, target_sum)
            fc = compute_c5(xc, n)
            if fc < fvals[-1]:
                simplex[-1] = xc
                fvals[-1] = fc
            else:
                for i in range(1, n + 1):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_feasible(simplex[i], target_sum)
                    fvals[i] = compute_c5(simplex[i], n)
    return simplex[0].copy(), fvals[0]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(45, 0.5)

    n = len(h_best)
    target_sum = n / 2.0
    h_best = project_feasible(h_best, target_sum)
    best_c5 = compute_c5(h_best, n)

    # Phase 1: Nelder-Mead from initial
    h_opt, c5_opt = nelder_mead(h_best, n, target_sum, 400)
    if c5_opt < best_c5:
        best_c5, h_best = c5_opt, h_opt.copy()

    # Phase 2: Adaptive random restarts
    sigma = 0.03
    count = 0
    while time.time() - t0 < budget_s - 20:
        count += 1
        # Small local perturbation
        h_pert = project_feasible(h_best + rng.normal(0, sigma, n), target_sum)
        h_opt, c5_opt = nelder_mead(h_pert, n, target_sum, 150)
        if c5_opt < best_c5:
            best_c5, h_best = c5_opt, h_opt.copy()
            sigma = max(0.005, sigma * 0.7)
            # Extra polish
            h_opt2, c5_opt2 = nelder_mead(h_best, n, target_sum, 300)
            if c5_opt2 < best_c5:
                best_c5, h_best = c5_opt2, h_opt2.copy()
        else:
            sigma = min(0.12, sigma * 1.08)

        # Occasional block perturbation
        if count % 7 == 0:
            blk = rng.randint(2, 8)
            start = rng.randint(0, n - blk)
            h_block = h_best.copy()
            h_block[start:start + blk] = np.clip(
                h_block[start:start + blk] + rng.normal(0, 0.1, blk), 0, 1)
            h_block = project_feasible(h_block, target_sum)
            h_opt, c5_opt = nelder_mead(h_block, n, target_sum, 150)
            if c5_opt < best_c5:

        # Rare large restart

            h_rand = project_feasible(h_rand, target_sum)

    if c5_opt < best_c5:
        best_c5, h_best = c5_opt, h_opt.copy()

    h_best = project_feasible(h_best, target_sum)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
