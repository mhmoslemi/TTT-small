import numpy as np
from scipy.optimize import minimize

_g_n = 45
_g_target = 22.5


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        if s < target:
            room = 1.0 - h
            rs = np.sum(room)
            if rs > 1e-15:
                h += (target - s) * room / rs
            else:
                h[:] = 0.5
        else:
            h -= (s - target) * h / max(s, 1e-15)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj(x):
    return compute_c5(x, _g_n)


def _con_eq(x):
    return np.sum(x) - _g_target


def _obj_pen(x):
    hc = np.clip(x, 0.0, 1.0)
    return compute_c5(hc, _g_n) + 4.0 * abs(np.sum(hc) - _g_target) / _g_target


def upsample(h, n_old, n_new):
    t_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    t_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(t_new, t_old, h)


def cobyla_run(h0, n, maxfev):
    global _g_n, _g_target
    _g_n = n
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq}]
    try:
        res = minimize(_obj, h0, method='COBYLA', bounds=bounds,
                       constraints=cons,
                       options={'maxfev': maxfev, 'rhobnd': 1e-4})
        return res
    except Exception:
        return None
