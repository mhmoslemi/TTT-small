                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break

    def eval_c5(h_arr):
        return float(np.correlate(h_arr, 1.0 - h_arr, mode='full').max() * dx)

    best_c = eval_c5(h)
    best_h = h.copy()
    cur_c = best_c

    T0 = 0.004
    T_min = 1e-8
    sa_end = budget_s * 0.65
    polish_end = budget_s * 0.95

    # Phase 1: Simulated Annealing
    iters = 0
    stagnant = 0
    while time.time() - start_time < sa_end:
        elapsed = time.time() - start_time
        frac = elapsed / sa_end
        T = max(T_min, T0 * (1.0 - frac) ** 2.5)

        # 90% pair move, 10% triple move
        if random.random() < 0.9:
            i = random.randrange(n)
            j = random.randrange(n)
            if i == j:
                continue
            scale = 0.02 + 0.18 * (T / T0)
            delta = random.uniform(-scale, scale)
            max_d = min(h[i], 1.0 - h[j])
            min_d = -min(h[j], 1.0 - h[i])
            if delta > max_d:
                delta = max_d * 0.5 if max_d > 0 else 0.0
            elif delta < min_d:
                delta = min_d * 0.5 if min_d < 0 else 0.0
            if abs(delta) < 1e-14:
                continue
            h[i] -= delta
            h[j] += delta
            new_c = eval_c5(h)
            if new_c <= cur_c:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                stagnant += 1
                if T > T_min:
                    log_r = math.log(random.random())
                    if log_r > -(new_c - cur_c) / T:
                        cur_c = new_c
                    else:
                        h[i] += delta
                        h[j] -= delta
                else:
                    h[i] += delta
                    h[j] -= delta
        else:
            # Triple move: pick 3 indices, redistribute
            idx = random.sample(range(n), 3)
            a, b, c2 = idx
            d1 = random.uniform(-0.05, 0.05)
            d2 = random.uniform(-0.05, 0.05)
            d3 = -(d1 + d2)
            changes = [(a, -d1), (b, -d2), (c2, -d3)]
            valid = True
            for idx_c, dc in changes:
                nv = h[idx_c] + dc
                if nv < -1e-12 or nv > 1 + 1e-12:
                    valid = False
                    break
            if not valid:
                continue
            for idx_c, dc in changes:
                h[idx_c] += dc
            new_c = eval_c5(h)
            if new_c <= cur_c:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                stagnant += 1
                if T > T_min:
                    log_r = math.log(random.random())
                    if log_r > -(new_c - cur_c) / T:
                        cur_c = new_c
                    else:
                        for idx_c, dc in changes:
                            h[idx_c] -= dc
                else:
                    for idx_c, dc in changes:
                        h[idx_c] -= dc

        iters += 1
        # Occasional reheat if very stagnant
        if stagnant > 80000 and T > T_min * 100:
            h = best_h.copy()
            cur_c = best_c
            stagnant = 0

    # Phase 2: Greedy local polish
    h = best_h.copy()
    cur_c = best_c
    polish_rounds = 0
    while time.time() - start_time < polish_end:
        improved = False
        for _ in range(500):
            i = random.randrange(n)
            j = random.randrange(n)
            if i == j:
                continue
            for trial in range(3):
                if trial == 0:
                    delta = random.uniform(-0.005, 0.005)
                elif trial == 1:
                    delta = random.uniform(-0.02, 0.02)
                else:
                    # Try opposite direction of current best perturbation
                    delta = random.choice([-1, 1]) * random.uniform(0.001, 0.01)
                max_d = min(h[i], 1.0 - h[j])
                min_d = -min(h[j], 1.0 - h[i])
                if delta > max_d:
                    delta = max_d * 0.5 if max_d > 0 else 0.0
                elif delta < min_d:
                    delta = min_d * 0.5 if min_d < 0 else 0.0
                if abs(delta) < 1e-14:
                    continue
                h[i] -= delta
                h[j] += delta
                new_c = eval_c5(h)
                if new_c < cur_c:
                    cur_c = new_c
                    improved = True
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        polish_rounds += 1
        if not improved and polish_rounds > 10:
            break

    # Phase 3: Final fine-tuning with very small steps
    h = best_h.copy()
    cur_c = best_c
    while time.time() - start_time < budget_s - 5:
        improved = False
        for _ in range(200):
            i = random.randrange(n)
            j = random.randrange(n)
            if i == j:
                continue
            delta = random.uniform(-0.002, 0.002)
            max_d = min(h[i], 1.0 - h[j])
            min_d = -min(h[j], 1.0 - h[i])
            if delta > max_d:
                delta = max_d * 0.5 if max_d > 0 else 0.0
            elif delta < min_d:
                delta = min_d * 0.5 if min_d < 0 else 0.0
            if abs(delta) < 1e-15:
                continue
            h[i] -= delta
            h[j] += delta
            new_c = eval_c5(h)
            if new_c < cur_c:
                cur_c = new_c
                improved = True
            else:
                h[i] += delta
                h[j] -= delta
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            break

    best_c = eval_c5(best_h)
    # Ensure validity
    best_h = np.clip(best_h, 0.0, 1.0)
    # Final sum fix
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-14:
        for i in range(n):
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
                best_h[i] += add
                diff -= add
            else:
                sub = min(-diff, best_h[i])
                best_h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    best_c = eval_c5(best_h)
    return (best_h, best_c, n)
