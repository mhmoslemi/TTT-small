import time

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h):
    return np.max(_corr_vals(h))

def _smooth_obj(h, temp):
    vals = _corr_vals(h)
    m = np.max(vals)
    return m + temp * np.log(np.sum(np.exp((vals - m) / temp)))

def _smooth_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    vals = _corr_vals(h)
    m = np.max(vals)
    e = np.exp((vals - m) / temp)
    w = e / np.sum(e)
    
    grad = np.zeros(n)
    o_h = 1.0 - h
    for j in range(2 * n - 1):
        if w[j] < 1e-12:

        k = j - (n - 1)
        wk = w[j] * dx
        if k >= 0:
            # m-k valid: m in [k, n-1], o_h index m-k in [0, n-1-k]
            grad[k:] += wk * o_h[:n-k]

            # m+k = m-kp valid: m in [kp, n-1], h index m-kp in [0, n-1-kp]
            grad[kp:] -= wk * h[:n-kp]

                continue
            j = int(r[np.argmax(h[r])])
            h[j] = min(1.0, h[j] + min(-delta, 1.0))
        h = _proj(h, n)
        c = _c5(h)
        if c > best_c + 1e-6:
            h[i] = old
            h = _proj(h, n)
        else:
            best_c = c
    return h


def _slsqp(h0, n, t, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(_sobj, h0, jac=lambda x: _sgrad(x, t),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return _proj(res.x, n)


def _local_search(h, n, time_limit, rng):
    best_c = _c5(h)
    h = h.copy()
    deadline = time.time() + time_limit
    eps_base = 0.02
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        eps = eps_base * (0.1 + 0.9 * rng.rand())
        for sign in (1, -1):
            h2 = h.copy()
            h2[i] += sign * eps
            h2[j] -= sign * eps
            if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                c2 = _c5(h2)
                if c2 < best_c - 1e-12:
                    h = h2
                    best_c = c2
        if rng.rand() < 0.01:
            eps_base = max(0.001, eps_base * 0.9)
    return h, best_c


def _subgrad_search(h, n, iters=3000):
    h = h.copy()
    best_c = _c5(h)
    lr = 0.02
    for it in range(iters):
        v = _cv(h)
        idx = int(np.argmax(v))
        k = idx - (n - 1)
        g = np.zeros(n)
        oh = 1.0 - h
        if k >= 0:
            g[k:] += oh[:n - k]
            g[:n - k] -= h[k:]
        else:
            kp = -k
            g[:n - kp] += oh[kp:]
            g[kp:] -= h[:n - kp]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h2 = h - lr * g / gn
        h2 = _proj(h2, n)
        c2 = _c5(h2)
        if c2 < best_c:
            h = h2
            best_c = c2
        else:
            lr *= 0.95
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        h = (np.arange(70) % 2).astype(np.float64)
    n = len(h)
    best = h.copy()
    bc = _c5(best)

    def ok(frac):
        return (time.time() - t0) < budget_s * frac

    # Phase 1: SLSQP with decreasing temperature
    for t in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if not ok(0.30):
            break
        h2 = _slsqp(best.copy(), n, t, 200)
        c = _c5(h2)
        if c < bc:
            best, bc = h2, c

    # Phase 2: Binary push
    if ok(0.40):
        hb = _binpush(best, 400)
        c = _c5(hb)
        if c < bc:
            best, bc = hb.copy(), c

    # Phase 3: Subgradient local search
    if ok(0.45):
        h2, c = _subgrad_search(best, n, 3000)
        if c < bc:
            best, bc = h2.copy(), c

    # Phase 4: Tighter SLSQP
    if ok(0.55):
        for t in [0.003, 0.001]:
            if not ok(0.62):
                break
            h2 = _slsqp(best.copy(), n, t, 400)
            c = _c5(h2)
            if c < bc:
                best, bc = h2, c

    # Phase 5: Binary push again
    if ok(0.65):
        hb = _binpush(best, 600)
        c = _c5(hb)
        if c < bc:
            best, bc = hb.copy(), c

    # Phase 6: Local pairwise search
    if ok(0.70):
        remaining = budget_s * 0.80 - (time.time() - t0)
        if remaining > 5:
            h2, c = _local_search(best, n, min(remaining, 60), rng)
            if c < bc:
                best, bc = h2.copy(), c

    # Phase 7: Random restarts
    if ok(0.75):
        for _ in range(25):
            if not ok(0.88):
                break
            h2 = best.copy()
            mask = rng.random(n) < 0.12
            if mask.sum() > 0:
                h2[mask] = np.clip(h2[mask] + 0.04 * rng.randn(int(mask.sum())), 0, 1)
                h2 = _proj(h2, n)
                h2 = _slsqp(h2, n, 0.01, 100)
                h2 = _binpush(h2, 100)
                c = _c5(h2)
                if c < bc:
                    best, bc = h2.copy(), c

    # Phase 8: Final local polish
    if ok(0.90):
        remaining = budget_s * 0.95 - (time.time() - t0)
        if remaining > 3:
            h2, c = _local_search(best, n, min(remaining, 30), rng)
            if c < bc:
                best, bc = h2.copy(), c

    # Final validation
    best = np.clip(best, 0.0, 1.0)
    bc = _c5(best)
    return best, bc, n
