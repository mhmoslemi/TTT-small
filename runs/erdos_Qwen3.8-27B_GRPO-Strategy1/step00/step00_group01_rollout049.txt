    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = _adjust_sum(h_init, len(h_init) / 2.0)
        c5_init = _compute_c5(h_init)
        best_c5 = c5_init
        best_h = h_init.copy()

    n_list = [48, 50, 52, 54, 56, 60, 64, 72, 80]

    for n in n_list:
        phase1_deadline = t0 + budget_s * 0.45
        if time.time() > phase1_deadline:
            break
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (target,)}]
        x_new = np.linspace(0.0, 2.0, n)

        for attempt in range(3):
            if time.time() > phase1_deadline:
                break
            if attempt == 0 and h_init is not None:
                x_old = np.linspace(0.0, 2.0, len(h_init))
                h0 = np.interp(x_new, x_old, h_init)
            elif attempt == 1:
                h0 = 0.5 + 0.35 * np.sin(2.0 * np.pi * rng.randint(2, 5) * x_new / 2.0)
                h0 = np.clip(h0, 0.0, 1.0)
            else:
                h0 = rng.rand(n) * 0.3 + 0.35
            h0 = _adjust_sum(h0, target)

            for T in [20.0, 60.0, 120.0]:
                if time.time() > phase1_deadline:
                    break
                res = minimize(_smooth_obj, h0, args=(dx, T),
                             method='L-BFGS-B', bounds=bounds,
                             constraints=cons,
                             options={'maxiter': 300, 'ftol': 1e-13})
                h0 = _adjust_sum(res.x, target)
                c5_try = _compute_c5(h0)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h0.copy()

        # Binary SA from continuous projection
        phase2_deadline = t0 + budget_s * 0.7
        if time.time() > phase2_deadline:
            break
        h_bin = _to_binary(h0)
        n_sa = 4000
        h_bin, c5_bin = _sa_search(h_bin, rng, n_sa, phase2_deadline)
        h_bin, c5_bin = _greedy_improve(h_bin, rng, 2000, phase2_deadline)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        # Also try structured/random binary starts
        for pat in range(4):
            if time.time() > phase2_deadline:
                break
            h_start = _structured_binary(n, rng, pat)
            h_start, c5_s = _sa_search(h_start, rng, 2000, phase2_deadline)
            h_start, c5_s = _greedy_improve(h_start, rng, 1000, phase2_deadline)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()

    # Final extended SA on best
    final_deadline = t0 + budget_s * 0.95
    if best_h is not None and time.time() < final_deadline:
        n = len(best_h)
        if n % 2 == 0:
            h_bin = _to_binary(best_h)
            for restart in range(5):
                if time.time() > final_deadline:
                    break
                if restart > 0:
                    # Perturb best
                    h_bin = best_h.copy()
                    for _ in range(10):
                        a = rng.randint(n)
                        b = rng.randint(n)
                        if h_bin[a] != h_bin[b]:
                            h_bin[a], h_bin[b] = h_bin[b], h_bin[a]
                h_bin, c5_r = _sa_search(h_bin, rng, 30000, final_deadline)
                h_bin, c5_r = _greedy_improve(h_bin, rng, 5000, final_deadline)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_bin.copy()

    # Final validation and return
    if best_h is not None:
        np.clip(best_h, 0.0, 1.0, out=best_h)
        best_h = _adjust_sum(best_h, len(best_h) / 2.0)
        best_c5 = _compute_c5(best_h)
    else:
        n = 48
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = _compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
