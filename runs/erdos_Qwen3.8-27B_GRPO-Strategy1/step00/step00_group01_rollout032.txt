import numpy as np
from scipy.optimize import minimize
import time

_global_target_sum = 23.5

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad_k(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] = 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        ss = -s
        g[:n - ss] = 1.0 - h[ss:]
        g[ss:] -= h[:n - ss]
    return g

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            hi = mid
        else:
            lo = mid
    p = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(p)
    if abs(diff) > 1e-13:
        if diff > 0:
            room = np.where(p < 0.9999, 1.0 - p, 0.0)
        else:
            room = np.where(p > 1e-6, p, 0.0)
        tot = np.sum(room)
        if tot > 1e-15:
            p += (diff / tot) * room
    return p

def smooth_obj_val(h, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(t * (corr - m)))) / t

def smooth_grad_vec(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exponents = t * (corr - m)
    exponents -= np.max(exponents)
    w = np.exp(exponents)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-9:
            grad += w[k] * compute_grad_k(h, k)
    return grad

def exact_subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    return compute_grad_k(h, k_max)

def _constraint_fun(h):
    return np.sum(h) - _global_target_sum

def _constraint_jac(h):
    return np.ones(len(h))

def _smoob_wrapper(h):
    return smooth_obj_val(h, _smooth_t)

def _sgrad_wrapper(h):
    return smooth_grad_vec(h, _smooth_t)

_smooth_t = 100.0

def pgd_optimize(h_init, target_sum, max_iter, t0, lr0, time_limit):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    t_cur = t0

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        t_cur = max(5.0, t0 * (0.97 ** (it // 200)))

        if t_cur > 1.0:
            grad = smooth_grad_vec(h, t_cur) * dx
        else:
            grad = exact_subgrad(h) * dx

        lr = lr0 * (0.9995 ** (it // 100))
        accepted = False
        for bt in range(40):
            h_new = project(h - lr * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new <= best_c5 + 1e-12:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            pass
    return best_h, best_c5

def slsqp_optimize(h_init, target_sum, t_smooth, max_iter=300):
    global _smooth_t, _global_target_sum
    _smooth_t = t_smooth
    _global_target_sum = target_sum
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]

    result = minimize(_smoob_wrapper, h_init, jac=_sgrad_wrapper, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
    h_out = np.clip(result.x, 0.0, 1.0)
    h_out = project(h_out, target_sum)
    c5_out = compute_c5(h_out)
    return h_out, c5_out
