def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h, n):
    """Repair h to have sum exactly n/2 while staying in [0,1]"""
    target = n / 2.0
    current = np.sum(h)
    if abs(current - target) < 1e-10:
        return h
    diff = target - current
    # Distribute the difference
    if diff > 0:
        # Need to increase sum: add to elements < 1
        capacity = 1.0 - h
        total_cap = np.sum(capacity)
        if total_cap > 0:
            frac = min(diff / total_cap, 1.0)
            h = h + capacity * frac
    else:
        # Need to decrease sum: subtract from elements > 0
        capacity = h
        total_cap = np.sum(capacity)
        if total_cap > 0:
            frac = min(-diff / total_cap, 1.0)
            h = h - capacity * frac
    # Final clip and fix
    h = np.clip(h, 0, 1)
    # If still off, fix with a single element
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + residual)
    return h

def random_h(n, rng):

    h = rng.random(n)
    h = h / np.sum(h) * (n / 2.0)
    h = np.clip(h, 0, 1)
    h = repair_sum(h, n)
    return h

def block_h(n, frac=0.5, rng=None):
    """Block construction: 1s in the middle, 0s on edges"""
    h = np.zeros(n)
    n1 = int(n * frac)
    start = (n - n1) // 2
    h[start:start+n1] = 1.0
    # Fix sum
    needed = n/2 - n1
    if abs(needed) > 1e-10:
        if needed > 0:
            h[start-1] = needed  # if start > 0
        else:
            h[start+n1-1] = 1 + needed

def optimize_pair_swaps(h, n, n_iterations=1000, rng=None, T=0.01):
    """Coordinate descent with pair swaps"""
    best_c5 = compute_c5(h, n)
    h = h.copy()
    
    for it in range(n_iterations):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2 * T  # in [-T, T]
        new_i = h[i] + delta
        new_j = h[j] - delta
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        h_old = np.array([h[i], h[j]])
        h[i] = new_i
        h[j] = new_j
        new_c5 = compute_c5(h, n)
        if new_c5 < best_c5 or (rng.random() < np.exp(-(new_c5 - best_c5) / max(T, 1e-6))):
            best_c5 = min(best_c5, new_c5)
        else:
            # Revert
            h[i], h[j] = h_old
        T *= 0.999  # Cool
    return h, best_c5

def optimize_directional(h, n, n_iterations=200, rng=None):

    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Proportional redistribution
    if diff > 0:
        # Add to positions with room
        room = 1.0 - h
        total_room = room.sum()
        if total_room < diff:
            # Can't fit, set some to 1
            n_full = int(diff)
            idxs = np.argsort(-h)[:n_full]
            h[idxs] = 1.0
            diff -= n_full
            room = 1.0 - h
            total_room = room.sum()
        if total_room > 0:
            h += room * (diff / total_room)
    else:
        space = h

        if total_space < -diff:
            n_zero = int(-diff)
            idxs = np.argsort(h)[:n_zero]
            h[idxs] = 0.0

            space = h
            total_space = space.sum()

One effective approach: for a small number of "active" lags (those near the maximum), compute the gradient and do a projected gradient step.

The gradient of S_k = sum_i h[i+k]*(1-h[i]) with respect to h[j]:
- For the term where j = i+k (i.e., i=j-k): contributes (1-h[j-k])
- For the term where j = i (i.e., the (1-h[i]) term): contributes -h[j+k]
- (These are valid only for appropriate indices)

So ∂S_k/∂h[j] = (1-h[j-k]) * [j-k valid] - h[j+k] * [j+k valid]

The subgradient of max_k S_k is the average of gradients over active lags (or just the gradient of the max lag for a simpler approach).

def subgradient_step(h, n, lr=0.1):
    dx = 2.0 / n
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Find active lags (near max)
    max_corr = np.max(corr)
    active = np.where(corr > 0.9 * max_corr)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # actual lag
        # For this lag, ∂/∂h[j] of sum_i h[i+|k|]*(1-h[i])
        # k > 0: sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # k < 0: sum_{i=-k}^{n-1} h[i+k]*(1-h[i]) = sum_{i=0}^{n-1+k} h[i]*(1-h[i-k]) (by reindexing)
        
        if k >= 0:
            for i in range(n - k):
                # h[i+k]*(1-h[i])
                # d/dh[i+k] = (1-h[i])
                # d/dh[i] = -h[i+k]
                grad[i+k] += (1-h[i])
                grad[i] -= h[i+k]
        else:
            k_abs = -k
            for i in range(k_abs, n):
                # h[i+k_abs... wait, let me reindex
                # For lag k<0: sum_{i=-k}^{n-1} h[i+k]*(1-h[i])
                # Let j = i+k, then i = j-k: sum_{j=0}^{n-1-k} h[j]*(1-h[j-k])
                for j in range(n + k):  # n-1-k+1 = n+k range... 
                    # h[j]*(1-h[j-k])
                    # d/dh[j] = (1-h[j-k])
                    # d/dh[j-k] = -h[j]
                    grad[j] += (1 - h[j-k])
                    grad[j-k] -= h[j]
    
    # Scale gradient
    grad = grad * lr / max(len(active), 1)
    
    # Projected gradient step (maintain sum constraint)
    h_new = h - grad
    h_new = repair_sum(h_new, n)
    return h_new

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(3):  # iterate a few times
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 0:
                h += room * min(diff / tr, 1)
        else:
            space = h.copy()
            ts = space.sum()
            if ts > 0:
                h -= space * min(-diff / ts, 1)
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
    # Final fix
    if abs(diff) > 1e-12:
        idx = np.argmax(1-h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
        # If still not exact, distribute
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            others = np.where(np.arange(n) != idx)[0]
            h[others] += diff2 / len(others)
            h = np.clip(h, 0, 1)
    return h

def hammersley_start(n, rng):
    """Create a start based on low-discrepancy sequence"""
    # Use a simple approach: place mass at quasi-random positions
    h = np.zeros(n)
    n1 = n // 2
    # Place 1s at positions based on golden ratio sequence
    positions = set()
    for i in range(n1):
        pos = int((i * 0.618033988749895) % 1 * n)
        positions.add(pos % n)
    # Fill remaining
    while len(positions) < n1:
        positions.add(rng.integers(n))
    for p in positions:
        h[p] = 1.0
    # Fix sum (need n/2 total)
    h = repair_sum(h, n)
    return h
