import numpy as np
import time
from scipy.signal import correlate

def _project_to_manifold(h, target, n):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(20):
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h += diff / n
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def _c5_value(h, dx):
    return correlate(h, 1.0 - h, mode='full').max() * dx

def _grad_at_max(h, dx, n):
    c = correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(c)
    grad = np.zeros(n)
    eps = 1e-7
    for j in range(n):
        step = min(eps, 1.0 - h[j], h[j])
        if step < 1e-14:
            continue
        hp = h.copy()
        hp[j] += step
        cm = h.copy()
        cm[j] -= step
        cp = correlate(hp, 1.0 - hp, mode='full')[k_max]
        cq = correlate(cm, 1.0 - cm, mode='full')[k_max]
        grad[j] = (cp - cq) / (2.0 * step)
    return grad * dx

def _try_binary_local(h, dx, n, target, rng, max_iter):
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    best = h_bin.copy()
    best_c5 = _c5_value(best, dx)
    for it in range(max_iter):
        i = rng.randint(n)
        if best[i] == 1.0:
            candidates = np.where(best == 0.0)[0]
        else:
            candidates = np.where(best == 1.0)[0]
        if len(candidates) == 0:
            break
        j = candidates[rng.randint(len(candidates))]
        trial = best.copy()
        trial[i] = 1.0 - trial[i]
        trial[j] = 1.0 - trial[j]
        c5_new = _c5_value(trial, dx)
        if c5_new < best_c5 - 1e-12:
            best = trial
            best_c5 = c5_new
    return best, best_c5
