from scipy.optimize import minimize

_N = 45

def _obj(x):
    h = np.clip(x, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

        else:

        if k == 0:

        np.clip(h, 0.0, 1.0, out=h)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    if n0 == n:
