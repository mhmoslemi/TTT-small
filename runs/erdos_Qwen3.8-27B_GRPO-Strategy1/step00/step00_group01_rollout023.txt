import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode='full'))

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        if mask.any():
            room = 1.0 - h[mask]
            s = room.sum()
            if s > 1e-14:
                h[mask] += diff * room / s
    else:
        mask = h > 0.0
        if mask.any():
            space = h[mask]
            s = space.sum()
            if s > 1e-14:
                h[mask] += diff * space / s
    return np.clip(h, 0.0, 1.0)

def local_search(h, n, deadline):
    best = h.copy()
    best_c5 = compute_c5(best)
    steps = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001]
    no_improve = 0
    while time.time() < deadline:
        improved = False
        for delta in steps:
            for i in range(n):
                for j in range(i + 1, n):
                    if best[i] >= delta - 1e-12 and best[j] <= 1.0 - delta + 1e-12:
                        best[i] -= delta
                        best[j] += delta
                        c5 = compute_c5(best)
                        if c5 < best_c5 - 1e-14:
                            best_c5 = c5
                            improved = True
                        else:
                            best[i] += delta
                            best[j] -= delta
                    if best[j] >= delta - 1e-12 and best[i] <= 1.0 - delta + 1e-12:
                        best[i] += delta
                        best[j] -= delta
                        c5 = compute_c5(best)
                        if c5 < best_c5 - 1e-14:
                            best_c5 = c5
                            improved = True
                        else:
                            best[i] -= delta
                            best[j] += delta
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return best, best_c5

def anneal(h, n, deadline, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    T = 0.002
    for it in range(800000):
        if time.time() > deadline:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        d = rng.uniform(-0.5, 0.5)
        cur[i] += d
        cur[j] -= d
        if cur[i] < -1e-12 or cur[i] > 1.0 + 1e-12 or cur[j] < -1e-12 or cur[j] > 1.0 + 1e-12:
            cur[i] -= d
            cur[j] += d
            continue
        new_c5 = compute_c5(cur)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
        else:
            cur[i] -= d
            cur[j] += d
        if it % 200 == 0:
            T *= 0.998
            if T < 1e-7:
                T = 1e-7
    return best, best_c5
