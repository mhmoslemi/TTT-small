            print(f"  Polish error: {e}")

    # Final validation
    final_c5 = evaluate_c5(best_h)
    final_sum = np.sum(best_h)
    print(f"Best: n={best_n}, C5={final_c5:.6f}, sum={final_sum:.6f}, target={best_n/2.0}")
    return (best_h, final_c5, best_n)
