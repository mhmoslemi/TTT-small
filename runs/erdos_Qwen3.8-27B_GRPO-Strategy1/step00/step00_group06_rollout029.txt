import time
from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def make_binary(h, target_sum):
    """Round h to binary, fixing the sum."""
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    diff = target_sum - np.sum(h_bin)
    if diff > 0:
        # Need to change some 0s to 1s
        zeros = np.where(h_bin == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for i in range(int(diff)):
            h_bin[zeros[i]] = 1
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for i in range(int(-diff)):
            h_bin[ones[i]] = 0
    return h_bin

def local_search(h, max_sweeps=2000, time_limit=None):
    """Steepest descent local search with all swaps."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:

        best_swap_c5 = best_c5
        best_i, best_j = -1, -1
        for i in zeros:
            for j in ones:
                h[i], h[j] = 1.0, 0.0
                c5 = compute_c5(h)
                if c5 < best_swap_c5:
                    best_swap_c5 = c5
                    best_i, best_j = i, j
                h[i], h[j] = 0.0, 1.0
        if best_i >= 0:
            h[best_i], h[best_j] = 1.0, 0.0
            best_c5 = best_swap_c5
        else:
            break
    return h, best_c5
