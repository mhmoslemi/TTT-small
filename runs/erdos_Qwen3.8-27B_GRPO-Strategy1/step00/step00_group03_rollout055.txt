import time
import scipy

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find active set (with small tolerance)
    tol = 1e-10
    active = corr >= max_val - tol
    k_offsets = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        k = int(k_offsets[idx])
        g = np.zeros(n)
        if k >= 0:
            if k < n:
                g[k:] += 1.0 - h[:n-k]
                g[:n-k] -= h[k:]
        else:
            kp = -k
            if kp < n:
                g[:n-kp] += 1.0 - h[kp:]
                g[kp:] -= h[:n-kp]
        grad += g
    grad *= dx / max(1, np.sum(active))
    return max_val * dx, grad

def project_box_sum(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 1.0  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def upsample(h, factor):
    """Upsample h by integer factor using linear interpolation"""
    n = len(h)
    new_n = n * factor
    old_x = np.linspace(0, 2, n, endpoint=False)  # or endpoint=True?
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    # Renormalize
    new_h = np.clip(new_h, 0, 1)
    new_h = project_box_sum(new_h, new_n / 2.0)
    return new_h

    h = h0.copy().astype(float)
    n = len(h)
    best_c = compute_c5(h)
    best_h = h.copy()
    target_sum = n / 2.0

        if c < best_c:
            best_c = c
            best_h = h.copy()

        h_new = project_box_sum(h_new, target_sum)
        
        c_new = compute_c5(h_new)
        if c_new < c:
            h = h_new
            alpha = min(1.0, alpha * 1.05)
        else:
            alpha *= 0.7
        
        if alpha < 1e-8:

    return best_h, best_c

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = (corr - mx) > -1e-9
    if not np.any(active):
        active = np.zeros(2*n-1, dtype=bool)
        active[np.argmax(corr)] = True
    ko = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        k = int(ko[idx])
        if 0 <= k < n:
            grad[k:] += 1.0 - h[:n-k]
            grad[:n-k] -= h[k:]
        elif -n < k <= 0:
            kp = -k
            grad[:n-kp] += 1.0 - h[kp:]
            grad[kp:] -= h[:n-kp]
    return mx * dx, grad * dx / float(np.sum(active))

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
