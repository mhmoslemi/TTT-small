These are different! For k>0, we're looking at 1s in h that are preceded by 0 (at distance s). For k<0, we're looking at 0s in h that are preceded by 1 (at distance s).

For a binary sequence, let me denote:
P_s = #{(i, i+s) : h[i]=0, h[i+s]=1, both valid} (number of 0→1 transitions at distance s)
Q_s = #{(i, i+s) : h[i]=1, h[i+s]=0, both valid} (number of 1→0 transitions at distance s)

Then:
corr at k=s = P_s (for s>0)
corr at k=-s = Q_s (for s>0)

And C₅ = max(0, max_s P_s, max_s Q_s) * dx

Note that P_s + Q_s = (n-s) - #{(i,i+s) : h[i]=h[i+s]} = (n-s) - #{agree at distance s}
Also, P_s = Q_s + (number of 1s in [s,n-1] - number of 1s in [0, n-1-s])

    no_improve_count = 0

    deadline1 = t0 + 0.55 * budget_s
    deadline2 = t0 + 0.85 * budget_s
    deadline = t0 + budget_s - 5

    while time.time() < deadline:
        elapsed = time.time() - t0
        if elapsed < deadline1 - t0:
            r = rng.random()
            if r < 0.45:
                new_h = make_pair_move(h, n, target_sum, rng)
            elif r < 0.70:
                new_h = make_block_move(h, n, target_sum, rng)
            elif r < 0.85:
                new_h = make_swap_move(h, n, rng)
            else:
                # Larger perturbation
                new_h = h + rng.standard_normal(n) * 0.08
                new_h = repair(new_h, target_sum)
                if abs(new_h.sum() - target_sum) > 1e-9:
                    continue
        elif elapsed < deadline2 - t0:
            # Targeted: focus on worst lags
            r = rng.random()
            if r < 0.6:
                new_h = make_pair_move(h, n, target_sum, rng)
            elif r < 0.85:
                new_h = make_block_move(h, n, target_sum, rng, max_len=4)
            else:
                new_h = make_swap_move(h, n, rng)
        else:
            # Fine-tuning
            new_h = make_pair_move(h, n, target_sum, rng)

        if new_h is None:
            continue

        new_c5 = compute_c5_fast(new_h, dx)
        dc = new_c5 - h_c5

        if dc < -1e-13 or (T > 1e-10 and rng.random() < np.exp(-dc / T)):
            h = new_h
            h_c5 = new_c5
            if h_c5 < best_c5 - 1e-10:
                best_h = h.copy()
                best_c5 = h_c5
                no_improve_count = 0
                T = max(1e-10, T * 0.995)
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1

        if no_improve_count > 5000:
            no_improve_count = 0
            T = min(T * 1.5, 0.005)
            # Nudge to escape
            h = h + rng.standard_normal(n) * 0.03
            h = repair(h, target_sum)
            if abs(h.sum() - target_sum) < 1e-9:
                h_c5 = compute_c5_fast(h, dx)

    # Final verification
    best_c5 = compute_c5_full(best_h)

    return best_h, best_c5, n
