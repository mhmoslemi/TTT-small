import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(vals)
    e = np.exp(alpha * (vals - m))
    sm = m + np.log(np.sum(e)) / alpha
    s = h.sum()
    return sm + mu * (s - target) ** 2

def smooth_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(vals)
    e = np.exp(alpha * (vals - m))
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        wk = w[k]
        if shift >= 0:
            a = np.arange(0, n - shift)
            grad[a] -= wk * h[a + shift]
            b = np.arange(shift, n)
            grad[b] += wk * (1.0 - h[b - shift])
        else:
            s2 = -shift
            a = np.arange(0, n - s2)
            grad[a] += wk * (1.0 - h[a + s2])
            b = np.arange(s2, n)
            grad[b] -= wk * h[b - s2]
    grad *= dx
    s = h.sum()
    grad += 2.0 * mu * (s - target)
    return grad

def project(h, n):
    target = n * 0.5
    h = h.copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def coord_polish(h, n, max_rounds=15, rng=np.random.default_rng(42)):
    target = n * 0.5
    best = compute_c5(h)
    deltas = [0.02, -0.02, 0.05, -0.05, 0.01, -0.01]
    for _ in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for j in order:
            for d in deltas:
                nv = h[j] + d
                if nv < 0 or nv > 1:
                    continue
                old_j = h[j]
                h[j] = nv
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    if diff > 0:
                        cand = np.where((h < 1.0) & (np.arange(n) != j))[0]
                    else:
                        cand = np.where((h > 0.0) & (np.arange(n) != j))[0]
                    if len(cand) == 0:
                        h[j] = old_j
                        continue
                    j2 = cand[0]
                    h[j2] += diff
                    if h[j2] < -1e-9 or h[j2] > 1.0 + 1e-9:
                        h[j] = old_j
                        continue
                    h = np.clip(h, 0.0, 1.0)
                c5v = compute_c5(h)
                if c5v < best - 1e-10:
                    best = c5v
                    improved = True
                    break
                else:
                    h[j] = old_j
            if improved:
                break
        if not improved:
            break
    h = project(h, n)
    return h, compute_c5(h)

def try_binary(h, n):
    if n % 2 != 0:
        return None, None
    t = n // 2
    h_bin = (h > 0.5).astype(float)
    ones = np.where(h_bin == 1)[0]
    if len(ones) > t:
        h_bin[ones[t:]] = 0.0
    elif len(ones) < t:
        zeros = np.where(h_bin == 0)[0]
        h_bin[zeros[:t - len(ones)]] = 1.0
    c5b = compute_c5(h_bin)
    return h_bin, c5b

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.zeros(96)
        h_start[:48] = 1.0

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    # Build list of resolutions to try
    n_base = len(h_start)
    resolutions = sorted(set([n_base, 64, 80, 96, 112, 128, 144, 160, 192, 224, 256, 320, 384]))

    for n in resolutions:
        if time.time() - t0 > budget_s - 60:
            break

        h = interpolate_h(best_h, n)
        h = project(h, n)
        c5v = compute_c5(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

        # L-BFGS-B with increasing alpha
        for alpha in [20, 50, 100, 200, 400]:
            if time.time() - t0 > budget_s - 30:
                break
            mu = 3000.0
            bounds = [(0.0, 1.0)] * n
            try:
                res = optimize.minimize(
                    smooth_obj, h,
                    jac=smooth_grad,
                    args=(alpha, mu),
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10}
                )
                h_opt = np.clip(res.x, 0.0, 1.0)
                h_opt = project(h_opt, n)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h = h_opt.copy()
            except Exception:
                pass

        # Coordinate polish
        if time.time() - t0 > budget_s - 25:
            break
        h_cs, c5_cs = coord_polish(h, n, max_rounds=10, rng=rng)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs

    # Try binary snap on best solution

        nb = len(best_h)
        if nb % 2 == 0:

                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t

    # Final validation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
