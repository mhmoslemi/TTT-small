import numpy as np
from scipy.optimize import minimize
import time

def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    j_max = int(np.argmax(corr))
    L = j_max - (n - 1)
    n_min = max(0, -L)
    n_max = min(n - 1, n - 1 - L)
    grad = np.zeros(n)
    for m in range(n):
        if n_min <= m - L <= n_max:
            grad[m] += (1.0 - h[m - L])
        if n_min <= m <= n_max:
            grad[m] += (-h[m + L])
    return grad * dx

def _make_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h += diff * room / room.sum()
        else:
            mass = h.copy()
            if mass.sum() < 1e-15:
                break
            h += diff * mass / mass.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_optimize(h0, target_sum, maxiter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
             'jac': lambda h: np.ones(n)}]
    try:
        res = minimize(lambda h: _c5_value(h), h0.copy(), jac=_c5_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = _make_feasible(res.x, target_sum)
        return h_out, _c5_value(h_out)
    except Exception:
        return h0, _c5_value(h0)

def _trustconstr_optimize(h0, target_sum, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    A = np.ones((1, n))
    from scipy.optimize import LinearConstraint
    lc = LinearConstraint(A, target_sum, target_sum)
    try:
        res = minimize(lambda h: _c5_value(h), h0.copy(), jac=_c5_grad,
                       method='trust-constr', bounds=bounds, constraints=[lc],
                       options={'maxiter': maxiter, 'gtol': 1e-11, 'xtol': 1e-12})
        h_out = _make_feasible(res.x, target_sum)
        return h_out, _c5_value(h_out)
    except Exception:
        return h0, _c5_value(h0)

def _generate_starts(n, rng):
    starts = []
    xg = np.linspace(0.0, 1.0, n)
    ts = n / 2.0
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.2, 0.4, 0.49]:
            h_s = 0.5 + amp * np.sin(2.0 * np.pi * freq * xg)
            starts.append(_make_feasible(h_s, ts))
            h_c = 0.5 + amp * np.cos(2.0 * np.pi * freq * xg)
            starts.append(_make_feasible(h_c, ts))
    for freq in [1, 2, 3]:
        h_sq = 0.5 + 0.49 * np.sign(np.sin(2.0 * np.pi * freq * xg))
        starts.append(_make_feasible(h_sq, ts))
    for _ in range(4):
        h_r = rng.uniform(0.0, 1.0, n)
        starts.append(_make_feasible(h_r, ts))
    return starts
