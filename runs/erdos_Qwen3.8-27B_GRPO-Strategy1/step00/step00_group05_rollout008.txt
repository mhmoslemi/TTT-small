            break
        n_target = n0 * mult
        h_up = upsample(h_cur, n_target)

        # Binary swap search (larger budget for higher res)
        n_iter = 6000
        h_up, c5 = binary_swap_search(h_up, n_iter, seed + 7777 * mult,
                                      tabu_len=30,
                                      time_limit=budget_s * 0.45, t0=t0)

        # Smooth perturbation
        if time.time() - t0 < budget_s * 0.55:
            h_up, c5 = smooth_perturb_search(h_up, 3000, seed + 9999 * mult,
                                             alpha=0.08,
                                             time_limit=budget_s * 0.6, t0=t0)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_up.copy()

        # Gradient refinement
        if time.time() - t0 < budget_s * 0.5:
            rem = (budget_s * 0.65) - (time.time() - t0)
            if rem > 5:
                h_g, c5_g = gradient_refine(h_up, 200, tau=0.04,
                                            time_limit=budget_s * 0.65, t0=t0)
                if c5_g < c5:
                    h_up = h_g
                    c5 = c5_g
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_up.copy()

        h_cur = h_up.copy()

    # Phase 3: try a couple random restarts at best resolution if time remains
    if time.time() - t0 < budget_s * 0.5:
        n_best = len(best_h)
        for rs in range(3):
            if time.time() - t0 > budget_s * 0.75:
                break
            h_rand = random_binary_init(n_best, seed + 12345 + rs * 31)
            h_r, c5_r = binary_swap_search(h_rand, 3000, seed + 55555 + rs * 97,
                                           tabu_len=40,
                                           time_limit=budget_s * 0.75, t0=t0)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final verification
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
