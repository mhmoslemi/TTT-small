        return smooth_max_grad(x, t)

        return np.ones(n)

    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new)
