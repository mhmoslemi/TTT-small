    current_sum = np.sum(h)
    diff = current_sum - target
    if abs(diff) > 1e-10:
        # Redistribute
        if diff > 0:
            # Need to remove mass from entries > 0

import numpy as np

import time

def compute_lags(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_c5(lags):
    return float(np.max(lags))

def subgrad_max_lags(h, n, lags, m=3):
    """Approximate subgradient of max of lags using top m"""
    top = np.argsort(-lags)[:m]
    wl = lags[top]
    w = np.exp(wl - wl[0])
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top):
        shift = k - (n - 1)
        for j in range(n):
            g = 1.0
            if 0 <= j + shift < n:
                g -= h[j + shift]
            if 0 <= j - shift < n:
                g -= h[j - shift]
            grad[j] += wi * g
    return grad

def repair(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if diff <= 0: break
            rem = min(h[i], diff)
            h[i] -= rem; diff -= rem
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= 0: break
            add = min(1.0 - h[i], -diff)
            h[i] += add; diff += add
    return np.clip(h, 0, 1)

def project_and_fix(h, n):
    """Project to simplex-like constraint: 0<=h<=1, sum=h_target"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(10):
        diff = h.sum() - target
        if abs(diff) < 1e-10: break
        mask_high = h < 1.0
        mask_low = h > 0.0
        if diff > 0:
            idx = np.where(mask_high)[0]
            if len(idx) == 0: break
            per = min(diff / len(idx), np.min(h))
            h[idx] -= per
        else:
            idx = np.where(mask_low)[0]
            if len(idx) == 0: break
            per = min(-diff / len(idx), 1.0 - np.max(h))
            h[idx] += per
    return np.clip(h, 0, 1)

    best = get_c5(compute_lags(h, dx))
    lags = compute_lags(h, dx)
    
    for it in range(max_iter):
        improved = False
        
        if use_grad:
            # Gradient step
            g = subgrad_max_lags(h, n, lags, m=3)
            alpha = 0.01
            for _ in range(3):  # Line search
                h_new = h - alpha * g
                h_new = project_and_fix(h_new, n)
                lags_new = compute_lags(h_new, dx)
                c5_new = get_c5(lags_new)
                if c5_new < best - 1e-10:
                    h = h_new
                    lags = lags_new
                    best = c5_new
                    improved = True
                    break
                alpha *= 0.5

        for _ in range(40):

            j = int(rng.integers(0, n))
            if i == j: continue
            d = rng.uniform(-0.1, 0.1)
            if not (0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1): continue
            h[i] += d; h[j] -= d
            lags_new = compute_lags(h, dx)
            c5_new = get_c5(lags_new)
            if c5_new < best - 1e-10:
                best = c5_new
                lags = lags_new
                improved = True
            else:
                h[i] -= d; h[j] += d
        
        if not improved and it > 100:

    return h, best

import time

def compute_lags(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(30):
        d = h.sum() - target
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h > 0
            cnt = m.sum()
            if cnt == 0: break
            step = min(d / cnt, h[m].min() * 0.999)
            h[m] -= step
        else:
            m = h < 1
            cnt = m.sum()
            if cnt == 0: break
            step = min(-d / cnt, (1.0 - h[m].max()) * 0.999)
            h[m] += step
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, shift):
    g = np.zeros(n)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
    elif shift > 0:
        g[:n - shift] += 1.0 - h[shift:]
        g[shift:] += -h[:n - shift]
    else:
        s = -shift
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] += -h[s:]
    return g

def local_search(h, n, dx, max_iter=500, seed=0):
    rng = np.random.default_rng(seed)
    lags = compute_lags(h, dx)
    best = float(lags.max())
    stall = 0

    for it in range(max_iter):
        improved = False
        # Gradient step on top lags
        top_m = min(4, len(lags))
        ti = np.argsort(-lags)[:top_m]
        w = np.exp((lags[ti] - lags.max()) * 15.0)
        w /= w.sum()
        g = np.zeros(n)
        for wj, k in zip(w, ti):
            g += wj * grad_lag(h, n, k - (n - 1))
        alpha = 0.06
        for _ in range(6):
            hn = fix_sum(h - alpha * g, n)
            ln = compute_lags(hn, dx)
            c5 = float(ln.max())
            if c5 < best - 1e-12:
                h, lags, best, improved = hn, ln, c5, True
                break
            alpha *= 0.45

        # Random transfers
        for _ in range(55):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j: continue
            d = rng.uniform(-0.12, 0.12)
            if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1: continue
            h[i] += d; h[j] -= d
            ln = compute_lags(h, dx)
            c5 = float(ln.max())
            if c5 < best - 1e-12:
                lags, best, improved = ln, c5, True
            else:
                h[i] -= d; h[j] += d

        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 35: break
    return h, best

def make_cosine(n, freq=2):
    return 0.5 + 0.5 * np.cos(2.0 * math.pi * freq * np.arange(n) / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # Get initial
    if 'initial_h_values' in globals():
        h0 = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h0 = make_cosine(80, 2)
    n = len(h0)
    dx = 2.0 / n
    h0 = fix_sum(h0.copy(), n)

    # Phase 1: optimize initial
    best_h, best_c5 = local_search(h0, n, dx, 1000, seed)

    # Try a few alternative starts
    alts = [make_cosine(n, 1), make_cosine(n, 3), 0.5 * np.ones(n)]
    for ai, alt in enumerate(alts):
        if time.time() - t0 > budget_s - 200: break
        alt = fix_sum(alt + 0.01 * rng.standard_normal(n), n)
        ha, ca = local_search(alt, n, dx, 400, seed + 1000 + ai)
        if ca < best_c5:
            best_h, best_c5 = ha, ca

    # Phase 2: Basin hopping
    temp = 0.05
    it = 0
    while time.time() - t0 < budget_s - 80:
        it += 1
        scale = temp * (0.04 + 0.015 * math.sqrt(1.0 + 0.02 * it))
        h_p = best_h + scale * rng.standard_normal(n)

        if it % 7 == 0:
            h_p = np.roll(best_h, int(rng.integers(-4, 5)))
            h_p += 0.03 * rng.standard_normal(n)
        elif it % 11 == 0:
            # locale: swap a block
            bl = int(rng.integers(5, n // 3))
            st = int(rng.integers(0, n - bl))
            h_p = best_h.copy()
            h_p[st:st+bl], h_p[n-bl:n] = h_p[n-bl:n].copy(), h_p[st:st+bl].copy()
            h_p += 0.02 * rng.standard_normal(n)

        h_p = fix_sum(h_p, n)

        mi = max(50, min(400, 500 - it // 3))
        h_new, c5_new = local_search(h_p, n, dx, mi, seed + it * 17)

        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            temp = max(0.006, temp * 0.97)
        elif rng.random() < math.exp(-(c5_new - best_c5) / max(temp, 1e-10)):
            best_h = h_new.copy()
            temp = min(0.18, temp * 1.04)
        if it % 10 == 0:
            temp = max(0.004, temp * 0.88)

    # Phase 3: final polish with fine steps
    if time.time() - t0 < budget_s - 10:
        best_h, best_c5 = local_search(best_h, n, dx, 3000, seed + 99991)

    return best_h, best_c5, n
