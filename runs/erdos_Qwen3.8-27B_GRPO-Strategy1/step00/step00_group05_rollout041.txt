import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def golden_init(n):
    h = np.zeros(n)
    n_ones = n // 2
    alpha = (np.sqrt(5.0) - 1.0) / 2.0
    used = set()
    for i in range(n_ones):
        p = int(((i + 0.5) * alpha) % 1.0 * n)
        while p in used and len(used) < n:
            p = (p + 1) % n
        h[p] = 1.0
        used.add(p)
    return h

def random_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def local_search_1opt(h, max_passes, t_limit):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    for _ in range(max_passes):
        if time.time() - t0 > t_limit:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        for i1 in range(len(ones)):
            for i0 in range(len(zeros)):
                p1, p0 = int(ones[i1]), int(zeros[i0])
                h[p1], h[p0] = 0.0, 1.0
                nc = compute_c5(h)
                if nc < best_c5 - 1e-12:
                    best_c5 = nc
                    improved = True
                else:
                    h[p1], h[p0] = 1.0, 0.0
        if not improved:
            break
    return h, best_c5

def sa_01(h_init, t_allot, rng, T0=None, cool_rate=0.9997):

        T0 = best_c5 * 0.012
    T = T0
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    stale = 0
    restart_count = 0
    while (time.time() - t0) < t_allot:
        batch_size = 100 if T > best_c5 * 1e-4 else 1
        for _ in range(batch_size):
            i1 = rng.randint(len(ones))
            i0 = rng.randint(len(zeros))
            p1 = int(ones[i1])
            p0 = int(zeros[i0])
            h[p1], h[p0] = 0.0, 1.0
            ones[i1], zeros[i0] = p0, p1
            nc = compute_c5(h)
            d = nc - cur_c5
            if d < 0 or (T > 1e-12 and rng.random() < np.exp(-d / T)):
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                    stale = 0
                else:
                    stale += batch_size
            else:
                h[p1], h[p0] = 1.0, 0.0
                ones[i1], zeros[i0] = p1, p0
                stale += batch_size
            T *= cool_rate
        if stale > 8000 and restart_count < 20:
            h = best_h.copy()
            cur_c5 = best_c5
            ones = np.where(h > 0.5)[0].copy()
            zeros = np.where(h <= 0.5)[0].copy()
            T = best_c5 * 0.01 * (0.5 ** restart_count)
            stale = 0
            restart_count += 1
    return best_h, best_c5
