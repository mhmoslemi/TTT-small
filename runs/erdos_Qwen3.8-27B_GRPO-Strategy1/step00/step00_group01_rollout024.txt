import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def smooth_obj(x, n, T):
    h = x.reshape(n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    return (cm + np.log(np.sum(np.exp(c - cm)))) / T * dx


def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        d = h.sum() - target
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h > 1e-14
        else:
            m = h < 1.0 - 1e-14
        if not m.any():
            return h
        h[m] += -d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def do_lbfgs(h0, n, T, maxiter):
    target = n / 2.0
    h0 = project_sum(h0, target)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(n, T),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12,
                            'gtol': 1e-10, 'disp': False})
    return project_sum(res.x, target)


def quantize(h, thr):
    h = h.copy()
    h[h < thr] = 0.0
    h[h > 1.0 - thr] = 1.0
    return project_sum(h, len(h) / 2.0)


def coord_refine(h, n, max_rounds, t0, tlimit):
    target = n / 2.0
    best = compute_c5(h)
    h = h.copy()
    for rnd in range(max_rounds):
        if time.time() - t0 > tlimit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0, old,
                     max(0.0, old - 0.05), min(1.0, old + 0.05),
                     (old + 0.0) / 2.0, (old + 1.0) / 2.0]
            seen = set()
            for tv in cands:
                tv_r = round(tv, 12)
                if tv_r in seen:
                    continue
                seen.add(tv_r)
                h[i] = tv_r
                h = project_sum(h, target)
                c5 = compute_c5(h)
                if c5 < best - 1e-15:
                    best = c5
                    improved = True
                else:
                    h[i] = old
            h = project_sum(h, target)
        if not improved:
            break
    return h, best


def try_01(h, n, t0, tlimit):
    target = n / 2.0
    best_01 = None
    best_val = compute_c5(h)
    idx = np.argsort(-h)
    for k in range(max(0, int(target) - 5), min(n, int(target) + 6)):
        if time.time() - t0 > tlimit:

        h01 = np.zeros(n)
        h01[idx[:k]] = 1.0
        rem = target - k

        c5 = compute_c5(h01)
        if c5 < best_val:
            best_val = c5
            best_01 = h01.copy()
    return best_01, best_val
