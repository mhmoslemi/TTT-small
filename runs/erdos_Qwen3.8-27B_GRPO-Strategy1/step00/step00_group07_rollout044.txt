```python
import numpy as np
import time
import math


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def sa_optimize(h, time_budget, seed, T0=0.008, step=0.04, cool=0.9998):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    T = T0
    T_min = 1e-6
    start = time.time()
    no_improve = 0
    patience = 50000

    while T > T_min:
        if time.time() - start > time_budget:
            break
        if no_improve > patience:
            break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            T *= cool
            continue

        delta = rng.uniform(-step, step)
        nh_i = h[i] + delta
        nh_j = h[j] - delta
        if nh_i < 0 or nh_i > 1 or nh_j < 0 or nh_j > 1:
            T *= cool
            continue

        old_i, old_j = h[i], h[j]
        h[i] = nh_i
        h[j] = nh_j
        new_c5 = compute_c5(h)

        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if rng.random() < math.exp(-(new_c5 - current_c5) / T):
                current_c5 = new_c5
            else:
                h[i] = old_i
                h[j] = old_j

        T *= cool

    return best_h, best_c5


def local_refine(h, time_budget, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    start = time.time()
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    while time.time() - start < time_budget:
        improved = False
        for ss in step_sizes:
            for _ in range(2000):
                if time.time() - start > time_budget:
                    return h, current_c5
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for sign in (1, -1):
                    d = sign * ss
                    nh_i = h[i] + d
                    nh_j = h[j] - d
                    if nh_i < 0 or nh_i > 1 or nh_j < 0 or nh_j > 1:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i] = nh_i
                    h[j] = nh_j
                    new_c5 = compute_c5(h)
                    if new_c5 < current_c5 - 1e-12:
                        current_c5 = new_c5
                        improved = True
                        break
                    else:
                        h[i] = old_i
                        h[j] = old_j
        if not improved:
            break

    return h, current_c5


def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(80) * 0.5

    n = len(h)
    target_sum = n / 2.0
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # Re-fix sum exactly by adjusting one element
        residual = target_sum - h.sum()
        h[0] = h[0] + residual
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    elapsed = time.time() - t0
    remaining = budget_s - elapsed

    # Multiple SA runs with varying parameters
    sa_configs = [

        (0.25, T0=0.005, step=0.03, cool=0.9998, label=1),
        (0.20, T0=0.003, step=0.02, cool=0.9999, label=2),
        (0.10, T0=0.001, step=0.01, cool=0.99995, label=3),

        if new_c5 < best_c5:

            best_h = new_h

        if new_c5 < best_c5:
            best_c5 = new_c5

    # Final polish: very fine local search
    elapsed = time.time() - t0

    # Validate and clean up
    best_h = np.clip(best_h, 0.0, 1.0)

    s = best_h.sum()
    if abs(s - target_sum) > 1e-10:

        need = target_sum - s
        for idx in range(n):

    final_c5 = compute_c5(best_h)
    n_pts = len(best_h)
    return best_h, final_c5, n_pts
```
