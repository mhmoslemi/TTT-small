    n = len(h)
    c5_cur = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]

        h[best_j] = 1
        c5_cur = c5_cur - best_delta

        zeros = np.where(h <= 0.5)[0]

    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: base resolution (~12% of budget)
    t1 = t0 + budget_s * 0.12
    h, c5 = greedy_improve(h, rng, min(t1 - 10, t0 + 15))
    if c5 < c5_best:
        c5_best, h_best = c5, h.copy()

    for r in range(15):
        if time.time() > t1 - 5:
            break
        h_s = h_best.copy() if r == 0 else perturb(h_best, rng, rng.randint(3, 20))
        T0 = 1.0 / n
        dl = min(t1 - 5, time.time() + 4)
        h_s, c5s = sa_run(h_s, rng, 80000, T0, dl)
        h_s, c5s = greedy_improve(h_s, rng, min(t1 - 2, time.time() + 3))
        if c5s < c5_best:
            c5_best, h_best = c5s, h_s.copy()

    # Phase 2: upscaling (~85% of budget)
    t2 = t0 + budget_s - 10
    h_cur = h_best.copy()
    fracs = [0.15, 0.20, 0.25, 0.40]
    rem0 = max(1.0, t2 - time.time())

    for lvl, frac in enumerate(fracs):
        if time.time() > t2 - 5:
            break
        h_up = np.repeat(h_cur, 2)
        n_up = len(h_up)
        t_end = min(t2 - 5, time.time() + rem0 * frac)
        
        tg = min(t_end - 10, time.time() + rem0 * frac * 0.30)
        h_up, c5g = greedy_improve(h_up, rng, tg)
        
        T0 = 1.0 / n_up
        n_it = max(200000, int(rem0 * frac * 80000))
        t_sa = min(t_end - 5, time.time() + rem0 * frac * 0.45)
        h_s, c5s = sa_run(h_up, rng, n_it, T0, t_sa)
        h_s, c5s = greedy_improve(h_s, rng, min(t_end - 1, time.time() + rem0 * frac * 0.25))
        
        # Additional SA restart if time permits
        if time.time() < t_end - 10 and c5s >= c5_best:
            h_r = perturb(h_cur, rng, 10)  # perturb the parent, not the upsampled
            # Actually, let me perturb the current best at this resolution
            h_r = perturb(np.repeat(h_best, 2), rng, 15)
            h_r, c5r = sa_run(h_r, rng, 100000, T0, t_end - 5)
            h_r, c5r = greedy_improve(h_r, rng, t_end - 1)
            c5s = min(c5s, c5r)
            h_s = h_s if c5s != c5r else h_r
        
        c5c = compute_c5(h_s)
        if c5c < c5_best:
            c5_best, h_best = c5c, h_s.copy()
            h_cur = h_s.copy()
        else:
            break

    h_best = np.clip(h_best, 0.0, 1.0)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
