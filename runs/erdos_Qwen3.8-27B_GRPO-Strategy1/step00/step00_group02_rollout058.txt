import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Add/subtract uniformly from elements not at boundary
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            adj = diff / np.sum(mask)
            h[mask] += adj
            h = np.clip(h, 0, 1)
            # Re-adjust for clipping
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-10:
                h += diff2 / n
                h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (h < 1.0 - 1e-12)
        else:
            mask = (h > 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        # Final adjustment
        if abs(h.sum() - target) > 1e-10:
            h += (target - h.sum()) / n
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, n, t=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    return m + np.log(np.exp(t * (vals - m)).sum()) / t

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                if diff <= 1e-12:
                    break
                adj = min(1.0 - h[i], diff)
                h[i] += adj
                diff -= adj
        else:
            order = np.argsort(-h)
            for i in order:
                if diff >= -1e-12:
                    break
                adj = min(h[i], -diff)
                h[i] -= adj
                diff += adj
    return h


def smooth_obj(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    return m + np.log(np.exp(t * (vals - m)).sum()) / t


def objective_pen(h, n, t, pen):
    s = h.sum() - n / 2.0
    return smooth_obj(h, n, t) + pen * s * s


def interp_step(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)
