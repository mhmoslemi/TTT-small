                break
            h_new = h.copy()
            mt = int(rng.integers(0, 5))
            if mt == 0:
                i = int(rng.integers(0, n)); j = int(rng.integers(0, n))
                if i == j:
                    continue
                a = min(rng.uniform(0.005, 0.15), h[i], 1.0 - h[j])
                if a < 1e-6:
                    continue
                h_new[i] -= a; h_new[j] += a
            elif mt == 1:
                c = int(rng.integers(2, n - 2)); w = int(rng.integers(2, 5))
                lo, hi = max(0, c - w), min(n, c + w)
                s0 = h[lo:hi].sum()
                h_new[lo:hi] = np.clip(h[lo:hi] + rng.normal(0, 0.02, hi - lo), 0, 1)
                s1 = h_new[lo:hi].sum()
                if s1 > 1e-10 and abs(s1 - s0) > 1e-10:
                    h_new[lo:hi] *= s0 / s1
                    h_new[lo:hi] = np.clip(h_new[lo:hi], 0, 1)
                    r = s0 - h_new[lo:hi].sum()
                    m = h_new[lo:hi] < 1.0 if r > 0 else h_new[lo:hi] > 0.0
                    if m.any() and abs(r) > 1e-12:
                        h_new[lo:hi][m] = np.clip(h_new[lo:hi][m] + r / m.sum(), 0, 1)
            elif mt == 2:
                i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
                h_new[i], h_new[j] = h_new[j], h_new[i]
            elif mt == 3:
                i = int(rng.integers(0, n))
                d = rng.uniform(0.02, 0.08) * (1 if h_new[i] > 0.5 else -1)
                h_new[i] = np.clip(h_new[i] + d, 0, 1)
                j = int(rng.integers(0, n))
                if j != i:
                    h_new[j] = np.clip(h_new[j] + (h[i] - h_new[i]), 0, 1)
            else:
                i = int(rng.integers(0, max(1, n - 7)))
                w = int(rng.integers(3, min(8, n - i)))
                sh = int(rng.integers(1, w))
                h_new[i:i + w] = np.roll(h_new[i:i + w].copy(), sh)
            if abs(h_new.sum() - target) > 1e-8:
                h_new = _repair(h_new, target)
                if h_new is None:
                    continue
            c5n = _compute_c5(h_new, dx)
            if c5n < curr - 1e-14:
                h = h_new; curr = c5n; no_improve = 0
                if curr < best_local_c5 - 1e-14:
                    best_local_c5 = curr; best_local = h.copy()
            elif rng.random() < np.exp(-(c5n - curr) / temp):
                h = h_new; curr = c5n
            else:
                no_improve += 1
        it += 1
        if no_improve > 2000:
            kick = rng.normal(0, 0.04 + 0.02 * min(it / 50.0, 1.0), n)
            h = np.clip(best_local + kick, 0, 1)
            h = _repair(h, target)
            if h is not None:
                curr = _compute_c5(h, dx)
            no_improve = 0
        temp = max(0.0004, temp * 0.995)
    return best_local, best_local_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    except (NameError, TypeError):
        n = 200
        h_init = np.full(n, 0.5)
    dx = 2.0 / n
    target = n / 2.0
    h_init = _repair(h_init, target)
    if h_init is None:
        h_init = np.full(n, 0.5)
    best_h = h_init.copy()
    best_c5 = _compute_c5(h_init, dx)
    # Build starting points
    starts = [best_h.copy()]
    hb = (rng.random(n) > 0.5).astype(float)
    n1 = int(hb.sum()); tgt_int = int(target)
    if n1 > tgt_int:
        idx = np.where(hb == 1)[0]
        hb[rng.choice(idx, n1 - tgt_int, replace=False)] = 0
    elif n1 < tgt_int:
        idx = np.where(hb == 0)[0]
        hb[rng.choice(idx, tgt_int - n1, replace=False)] = 1
    starts.append(hb.copy())
    hblk = np.zeros(n); hblk[:tgt_int] = 1.0
    starts.append(hblk.copy())
    hpb = h_init.copy()
    for _ in range(300):
        hpb = np.where(hpb > 0.5, np.clip(hpb + 0.01, 0, 1), np.clip(hpb - 0.01, 0, 1))
        hpb = _repair(hpb, target)
        if hpb is None:
            hpb = h_init.copy(); break
    starts.append(hpb.copy())
    # Alternate periodic patterns
    for per in [4, 5, 8]:
        if n % per == 0 and per <= n:
            fill = int(target * per / n)
            if 0 < fill < per:
                pat = np.zeros(per); pat[:fill] = 1.0
                hp = np.tile(pat, n // per)
                hp = _repair(hp, target)
                if hp is not None:
                    starts.append(hp.copy())
    # Optimize each start with time allocation
    n_starts = len(starts)
    for si in range(n_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        per_start = remaining / (n_starts - si)
        per_start = min(per_start, 200.0)
        sub_dead = min(t0 + per_start, deadline)
        h_res, c5_res = _local_search(starts[si], target, dx, n, sub_dead, rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    # If time remains, do additional polishes from best
    if time.time() < deadline - 10:
        h_res, c5_res = _local_search(best_h.copy(), target, dx, n, deadline, rng, temp_init=0.001)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    return best_h, best_c5, n
