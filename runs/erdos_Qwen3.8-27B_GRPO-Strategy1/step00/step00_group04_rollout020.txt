        ones = np.where(h == 1)[0]
        idx = np.random.choice(ones, s - target, replace=False)
        h[idx] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        idx = np.random.choice(zeros, target - s, replace=False)
        h[idx] = 1
    return h

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h

def anneal(h, t_start, deadline):
    h = h.copy()
    T = t_start
    T_min = 5e-5
    rate = 0.99997
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    n = len(h)
    stuck = 0
    while time.time() < deadline:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        a = int(oi[np.random.randint(len(oi))])
        b = int(zi[np.random.randint(len(zi))])
        h[a] = 0.0
        h[b] = 1.0
        nc = compute_c5(h)
        d = nc - c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-12)):
            c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stuck += 1
        if stuck > 5000:
            # Kick: flip 5 random pairs
            for _ in range(5):
                oi2 = np.where(h == 1)[0]
                zi2 = np.where(h == 0)[0]
                if len(oi2) > 0 and len(zi2) > 0:
                    h[oi2[np.random.randint(len(oi2))]] = 0.0
                    h[zi2[np.random.randint(len(zi2))]] = 1.0
            c5 = compute_c5(h)
            stuck = 0
        T = max(T * rate, T_min)
    return best_h, best_c5

def hill_climb(h, deadline):
    h = h.copy()
    c5 = compute_c5(h)
    n = len(h)
    n_iter = 0
    while time.time() < deadline and n_iter < 500000:
        improved = False
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        for a in oi[np.random.permutation(len(oi))]:
            for b in zi[np.random.permutation(len(zi))][:30]:
                h[a] = 0.0
                h[b] = 1.0
                nc = compute_c5(h)
                if nc < c5:
                    c5 = nc
                    improved = True
                else:
                    h[a] = 1.0
                    h[b] = 0.0
            if improved:
                break
        if not improved:
            break
        n_iter += 1
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Get parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=float)
    else:
        hp = np.zeros(42)
        hp[::2] = 1.0
    n_par = len(hp)

    bh = None
    bc = float('inf')

    # Build list of n to try
    n_set = set()
    for x in [n_par, n_par*2, n_par*3, 20, 24, 28, 32, 36, 40, 44, 48, 52, 56, 64, 80, 96, 128]:
        if x >= 12 and x % 2 == 0:
            n_set.add(x)
    n_list = sorted(n_set)

    # Phase 1: Quick exploration across n values (35% budget)
    dl1 = t0 + budget_s * 0.35
    for ni, n in enumerate(n_list):
        if time.time() > dl1:
            break
        remaining_n = len(n_list) - ni
        per_n = (dl1 - time.time()) / max(remaining_n, 1)
        per_start = max(per_n / 3.0, 3.0)

        # Build candidates
        cands = []
        # Scaled parent
        h_s = np.zeros(n)
        for i in range(n):
            h_s[i] = hp[min(int(i * n_par / n), n_par - 1)]
        cands.append(fix_binary(h_s, n))
        # Random
        idx = np.random.choice(n, n // 2, replace=False)
        h_r = np.zeros(n)
        h_r[idx] = 1.0
        cands.append(h_r)
        # Thue-Morse
        cands.append(fix_binary(make_tm(n), n))

        for h0 in cands:
            if time.time() > dl1:
                break
            dd = min(time.time() + per_start, dl1)
            hh, cc = anneal(h0, 0.004, dd)
            if cc < bc:
                bc = cc
                bh = hh.copy()

    # Phase 2: Intensive annealing on best (up to 60% of total)
    if bh is not None:
        dl2 = t0 + budget_s * 0.75
        bh, bc = anneal(bh, 0.003, dl2)

    # Phase 3: Hill climb polish
    if bh is not None and time.time() < t0 + budget_s * 0.80:
        dl3 = t0 + budget_s * 0.90
        bh, bc = hill_climb(bh, dl3)

    # Phase 4: Final short anneal
    if bh is not None and time.time() < t0 + budget_s * 0.91:
        dl4 = t0 + budget_s * 0.96
        bh, bc = anneal(bh, 0.0003, dl4)

    # Verify
    fc = compute_c5(bh)
    assert abs(bh.sum() * (2.0 / len(bh)) - 1.0) < 1e-9, "Integral constraint violated"
    assert np.all(bh >= 0) and np.all(bh <= 1), "Bounds violated"

    return (bh, fc, len(bh))
