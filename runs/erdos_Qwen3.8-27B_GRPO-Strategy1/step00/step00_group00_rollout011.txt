def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        nfree = int(np.sum(mask))
        if nfree > 0:
            h[mask] = np.clip(h[mask] + diff / nfree, 0.0, 1.0)
        residual = target - np.sum(h)
        if abs(residual) > 1e-10:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h


def penalty_fn(x, n, w):
