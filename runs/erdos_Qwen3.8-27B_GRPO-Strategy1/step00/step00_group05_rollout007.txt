    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    # Fine-grained fix
    s = np.sum(h)
    if s > target + 1e-12:
        excess = s - target
        for i in np.argsort(h)[::-1]:
            if excess <= 1e-12:
                break
            d = min(excess, h[i])
            h[i] -= d
            excess -= d
    elif target - s > 1e-12:
        deficit = target - s
        for i in np.argsort(h):
            if deficit <= 1e-12:
                break
            d = min(deficit, 1.0 - h[i])
            h[i] += d
            deficit -= d
    return h

    c = np.correlate(h, 1.0 - h, mode='full')

    if n_best != n_work:
        h_work = project_feasible(upsample_interp(h_best, n_work), n_work)
    else:
        h_work = h_best.copy()

    if tl() > 60:
        for _ in range(40):
            if tl() < 30:
                break
            scale = 0.02 + 0.03 * np.random.rand()
            h_pert = project_feasible(h_work + scale * np.random.randn(n_work), n_work)
            h_opt, c5_opt = optimize_candidate(h_pert, n_work, min(tl() - 10, 50), t0)
            if c5_opt < c5_best:
                h_best, c5_best, n_best = h_opt, c5_opt, n_work
                h_work = h_best

    # Phase 4: Final coordinate polish
    if tl() > 15:
        h_best, c5_best = coordinate_descent(h_best, n_best, t_start=time.time(),
                                             time_limit=tl() - 5)

    h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
