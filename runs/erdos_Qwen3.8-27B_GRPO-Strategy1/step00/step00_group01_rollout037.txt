def project_onto_constraint(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    p = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-12:
        mask = (p < 1.0 - 1e-12) if diff > 0 else (p > 1e-12)
        if mask.sum() > 0:
            p[mask] += diff / mask.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def c5_value(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def smooth_max_val(corr, T):
    cmax = np.max(corr)
    return cmax + float(np.log(np.mean(np.exp(T * (corr - cmax))))) / T

def smooth_max_grad(h, dx, T):

def smooth_max_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    val = smooth_max_val(corr, T)
    
    cmax = np.max(corr)
    lw = T * (corr - cmax)
    lw -= np.max(lw)
    w = np.exp(lw)
    w /= w.sum()  # length 2n-1
    
    # For each j, we need:
    # A[j] = sum_k w[k] * h[j+k-(n-1)]  (0 if index out of range)
    # B[j] = sum_k w[k] * h[j-k+(n-1)]  (0 if index out of range)
    
    # Vectorized using Toeplitz-like matrices or broadcasting
    # j ranges 0..n-1, k ranges 0..2n-2
    j_idx = np.arange(n)[:, None]  # (n, 1)
    k_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    
    idx_A = j_idx + k_idx - (n-1)  # (n, 2n-1)
    idx_B = j_idx - k_idx + (n-1)  # (n, 2n-1)
    
    # A[j,k] = w[k] * h[idx_A[j,k]] if valid
    mask_A = (idx_A >= 0) & (idx_A < n)
    mask_B = (idx_B >= 0) & (idx_B < n)
    
    A = (w[None, :] * np.where(mask_A, h[np.where(mask_A, idx_A, 0)], 0.0)).sum(axis=1)
    B = (w[None, :] * np.where(mask_B, h[np.where(mask_B, idx_B, 0)], 0.0)).sum(axis=1)
    
    grad = dx * (1.0 - A - B)
    return val, grad
