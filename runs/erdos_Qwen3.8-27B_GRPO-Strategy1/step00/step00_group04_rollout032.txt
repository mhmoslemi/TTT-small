import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(n, rng, n_ones=None):
    if n_ones is None:
        n_ones = n // 2
    h = np.zeros(n)
    indices = rng.choice(n, n_ones, replace=False)
    h[indices] = 1.0
    return h

def hill_climb(h, n, rng, max_iter=10000, time_limit=5.0):
    start = time.time()
    current = compute_c5(h, n)
    best = current
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if n <= 80:
            pairs = [(i, j) for i in ones for j in zeros]
            rng.shuffle(pairs)

            pairs = [(rng.choice(ones), rng.choice(zeros)) for _ in range(200)]
        
        improved = False
        for i, j in pairs:
            if time.time() - start > time_limit:
                break
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            if new_c5 < current:
                current = new_c5
                if new_c5 < best:
                    best = new_c5
                    best_h = h.copy()
                improved = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(n, rng, n_ones=None):
    if n_ones is None:
        n_ones = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    return h

def hill_climb(h, n, rng, time_limit=5.0):
    t_s = time.time()
    cur = compute_c5(h, n)
    best = cur
    best_h = h.copy()
    stall = 0
    max_stall = 50
    
    while time.time() - t_s < time_limit:
        ones = np.flatnonzero(h)
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        found = False
        # Adaptive: try more for small n, fewer for large
        n_try = min(len(ones)*len(zeros), max(100, 2*n))
        for _ in range(n_try):
            if time.time() - t_s > time_limit:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i], h[j] = 0.0, 1.0
            new = compute_c5(h, n)
            if new < cur - 1e-15:
                cur = new
                found = True
                if new < best:
                    best = new
                    best_h = h.copy()
                ones = np.flatnonzero(h)
                zeros = np.where(h == 0)[0]
                break
            else:
                h[i], h[j] = 1.0, 0.0
        
        if not found:
            stall += 1
            if stall >= max_stall:
                # Perturb
                npert = 1 + int(rng.randint(0, 3))
                for _ in range(npert):
                    ones = np.flatnonzero(h)
                    zeros = np.where(h == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    a, b = int(rng.choice(ones)), int(rng.choice(zeros))
                    h[a], h[b] = 0.0, 1.0
                cur = compute_c5(h, n)
                stall = 0
    
    return best_h, best

def resize(h_src, n, rng):
    n_src = len(h_src)
    if n_src == n:
        return h_src.astype(float).copy()
    n1 = n // 2
    x0 = np.arange(n_src) / n_src
    x1 = np.arange(n) / n
    hi = np.interp(x1, x0, h_src)
    srt = np.sort(hi)[::-1]
    thr = srt[n1-1] if n1 < len(srt) else 0.5
    h = (hi > thr + 1e-12).astype(float)
    ac = int(h.sum())

        h[rng.choice(ex, ac-n1, replace=False)] = 0
    elif ac < n1:
        df = np.where(h==0)[0]
        h[rng.choice(df, n1-ac, replace=False)] = 1

def spread_pattern(n, rng):
    """Evenly spread n/2 ones."""
    h = np.zeros(n)
    n1 = n // 2
    positions = (np.arange(n1) * n + n1 // 2) // n1

        positions[dup[-1]] = (positions[dup[-1]] + 1) % n
        positions = np.sort(positions)
