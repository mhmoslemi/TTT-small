import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def generate_alternative(n, rng):
    """Generate a random valid starting point."""
    h = rng.random(n)
    h = fix_sum(h, 0.5 * n)
    return h


def structured_start(n, rng):
    """Generate a structured starting point with some pattern."""
    n_hill = rng.integers(3, 8)
    h = np.zeros(n)
    width = n // (2 * n_hill)
    for k in range(n_hill):
        center = int((2 * k + 1) * n / (2 * n_hill))
        spread = max(width, 1)
        hi = min(n, center + spread)
        lo = max(0, center - spread)
        h[lo:hi] += 1.0
    h = np.clip(h, 0, 1)
    h = fix_sum(h + 0.1 * rng.random(n), 0.5 * n)
    return h
