def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_constrained(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def alternating_lp(h, max_iter=200, start_time=None, time_limit=300.0):
    n = len(h)
    dx = 2.0 / n
    h = np.array(h, dtype=float)
    best_c5 = compute_c5(h)
    h_best = h.copy()

    for it in range(max_iter):
        if start_time is not None and time.time() - start_time > time_limit:
            break
        g = 1.0 - h
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_rows = []
        for s in range(-(n - 1), n):
            row = np.zeros(n + 1)
            row[-1] = -1.0
            i_start = max(0, -s)
            i_end = min(n, n - s)
            for i in range(i_start, i_end):
                row[i] = dx * g[i + s]
            A_rows.append(row)

        A_ub = np.array(A_rows)
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(None, None)]

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        h_new = res.x[:n]
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new.copy()
        # Damped update for stability
        h = 0.6 * h + 0.4 * h_new

    return h_best, best_c5


def smooth_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    lse = mc + tau * np.log(np.sum(np.exp((corr - mc) / tau)))
    penalty = 10.0 * (np.sum(h) - n / 2.0) ** 2
    return dx * lse + penalty


def local_refine(h, n, target_sum, start_time, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    res = minimize(smooth_obj, h, args=(0.02,), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 300, 'ftol': 1e-14})
    if time.time() < deadline:
        h2 = project_constrained(res.x, target_sum)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    time_left = deadline - time.time()
    if time_left > 5:
        h3, c5_3 = alternating_lp(best_h, max_iter=100, start_time=start_time,
                                  time_limit=time_left)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
    else:
        init = np.ones(47) * 0.5

    n = len(init)
    target_sum = n / 2.0
    init = project_constrained(init, target_sum)
    c5_init = compute_c5(init)
    best_c5 = c5_init
    best_h = init.copy()

    # Phase 1: Main alternating LP
    time_left = deadline - time.time()
    h_lp, c5_lp = alternating_lp(init, max_iter=500, start_time=t0,
                                 time_limit=time_left)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: Smooth refinement at multiple tau
    for tau in [0.05, 0.02, 0.01]:
        if time.time() > deadline:
            break
        res = minimize(smooth_obj, best_h, args=(tau,), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 200, 'ftol': 1e-14})
        h_opt = project_constrained(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # Alternate from refined point
        time_left2 = deadline - time.time()
        if time_left2 > 10:
            h_a, c5_a = alternating_lp(best_h, max_iter=200, start_time=t0,
                                       time_limit=time_left2)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Phase 3: Local refinement neighborhood
    if time.time() < deadline:
        for _ in range(5):
            if time.time() > deadline:
                break
            h_pert = best_h + rng.randn(n) * 0.02
            h_pert = project_constrained(h_pert, target_sum)
            h_r, c5_r = local_refine(h_pert, n, target_sum, t0, deadline)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 4: Random restarts
    while time.time() < deadline:
        if rng.rand() < 0.6 and best_h is not None:
            h_rand = best_h + rng.randn(n) * 0.1
        else:
            h_rand = rng.rand(n)
        h_rand = project_constrained(h_rand, target_sum)
        time_left4 = deadline - time.time()
        if time_left4 < 3:
            break
        h_rs, c5_rs = alternating_lp(h_rand, max_iter=20, start_time=t0,
                                     time_limit=time_left4)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()

    # Final: ensure constraints
    if best_h is not None:
        best_h = project_constrained(best_h, len(best_h) / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
