def coord_opt(h, m, n):
    """Find optimal h[m] ignoring sum constraint, given all other coords fixed."""
    off = n - 1
    # For each k, corr[k] = a_k * h[m] + b_k
    a = np.zeros(2*n - 1)
    b = np.zeros(2*n - 1)
    hm = h[m]
    for k in range(2*n - 1):
        shift = k - off
        a_k = 0.0
        b_k = 0.0
        # h[m] as first factor: need 0 <= m - shift < n
        if 0 <= m - shift < n:
            a_k += (1.0 - h[m - shift])
        # h[m] as second factor (1-h[m]): need 0 <= m + shift < n
        if 0 <= m + shift < n:
            a_k -= h[m + shift]
        # b_k = corr[k] - a_k * h[m]
        # But we can compute b_k from the current correlation
        a[k] = a_k
    
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    b = corr_full - a * hm
    
    # 1D convex: min t in [0,1] of max_k(a_k*t + b_k)
    lo_t, hi_t = 0.0, 1.0
    for _ in range(60):
        t_mid = 0.5 * (lo_t + hi_t)
        vals = a * t_mid + b
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-12)[0]
        a_max = np.max(a[active])
        a_min = np.min(a[active])
        if a_min > 0:  # increasing, move left
            lo_t = t_mid
        elif a_max < 0:  # decreasing, move right
            hi_t = t_mid
        else:
            break
    return 0.5 * (lo_t + hi_t)

def coordinate_descent(h_init, n, max_sweeps=200, tol=1e-10):
    S = n / 2.0
    h = project_sum_box(h_init.copy(), S)
    best = h.copy()
    best_score = compute_c5(best)
    
    for sweep in range(max_sweeps):
        improved = False
        for m in range(n):
            t_opt = coord_opt(h, m, n)
            h[m] = t_opt
            # Project to maintain sum
            h = project_sum_box(h, S)
        
        score = compute_c5(h)
        if score < best_score - tol:
            best_score = score
            best = h.copy()
            improved = True
        elif not improved:
            # Check if we should stop
            pass
        if not improved and sweep > 10:
            break
    
    return best, best_score

def subgradient_descent(h_init, n, max_iters=5000, lr_init=0.05):
    S = n / 2.0
    h = project_sum_box(h_init.copy(), S)
    best = h.copy()
    best_score = compute_c5(best)
    dx = 2.0 / n
    off = n - 1
    
    velocity = np.zeros(n)
    lr = lr_init
    beta = 0.9
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        active = np.where(np.abs(corr - max_val) < 1e-8)[0]
        
        grad = np.zeros(n)
        for k in active:
            shift = k - off
            idx1 = np.arange(n) - shift
            mask1 = (idx1 >= 0) & (idx1 < n)
            grad[mask1] += (1.0 - h[idx1[mask1]])
            idx2 = np.arange(n) + shift
            mask2 = (idx2 >= 0) & (idx2 < n)
            grad[mask2] -= h[idx2[mask2]]
        grad /= len(active)
        grad *= dx  # scale
        
        velocity = beta * velocity - lr * grad
        h_new = project_sum_box(h + velocity, S)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new.copy()
            h = h_new
        else:
            lr *= 0.95
            h = project_sum_box(h - 0.5 * velocity, S)
        
        if lr < 1e-10:
            break
    
    return best, best_score

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum_box(h, S):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def coord_optimize(h, m, n, grid=151):
    corr_orig = np.correlate(h, 1.0 - h, mode='full')
    nk = 2 * n - 1
    quad = np.zeros(nk)
    lin = np.zeros(nk)
    base = np.zeros(nk)
    hm = h[m]
    for k in range(nk):
        km = k - m
        if 0 <= km < n:
            if k == 2 * m:
                quad[k] = -1.0
                lin[k] = 1.0
                base[k] = corr_orig[k] - (hm - hm * hm)
            else:
                lin[k] = 1.0 - 2.0 * h[km]
                base[k] = corr_orig[k] - lin[k] * hm
        else:
            base[k] = corr_orig[k]
    ts = np.linspace(0.0, 1.0, grid)
    vals = np.max(quad[None, :] * ts[:, None] ** 2 + lin[None, :] * ts[:, None] + base[None, :], axis=1)
    return ts[np.argmin(vals)]

def coordinate_descent(h_init, n, max_sweeps=100, grid=151):
    S = n / 2.0
    h = project_sum_box(h_init.copy(), S)
    best = h.copy()
    best_score = compute_c5(best)
    for sweep in range(max_sweeps):
        for m in range(n):
            t_opt = coord_optimize(h, m, n, grid)
            h[m] = t_opt
        h = project_sum_box(h, S)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best = h.copy()
    return best, best_score

def subgradient_descent(h_init, n, max_iters=3000, lr=0.04):
    S = n / 2.0
    h = project_sum_box(h_init.copy(), S)
    best = h.copy()
    best_score = compute_c5(best)
    dx = 2.0 / n
    vel = np.zeros(n)
    beta = 0.85
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mv = np.max(corr)
        active = np.where(np.abs(corr - mv) < 1e-9)[0]
        grad = np.zeros(n)
        for k in active:
            m_arr = np.arange(n)
            km = m_arr - k  # k - j = k - m means j = m, need 0<=k-m<n
            valid1 = (k - m_arr >= 0) & (k - m_arr < n)
            # d/dh[m]: from j=m: (1-h[k-m]) if valid; from j=k-m: -h[m-(k-m)]=-h[2m-k] if valid
            grad[valid1] += (1.0 - h[k - m_arr[valid1]])
            valid2 = (m_arr + k >= 0) & (m_arr + k < n) & (m_arr != k - m_arr)
            # j=k-m: need 0<=k-m<n, derivative contribution is -h[k-m] for the (1-h[m]) term
            # Actually: term h[k-m]*(1-h[m]), d/dh[m] = -h[k-m], valid if 0<=k-m<n
            valid3 = (k - m_arr >= 0) & (k - m_arr < n) & (k - m_arr != m_arr)
            grad[valid3] -= h[k - m_arr[valid3]]
        grad /= max(len(active), 1)
        grad *= dx
        vel = beta * vel - lr * grad
        h_new = project_sum_box(h + vel, S)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new.copy()
            h = h_new
        else:
            lr *= 0.97
        if lr < 1e-8:
            break
    return best, best_score

def binary_local_search(h_init, n, max_iters=5000):
    S = n / 2.0
    h = project_sum_box(h_init.copy(), S)
    best = h.copy()
    best_score = compute_c5(best)
    ones = np.where(h > 0.99)[0]
    zeros = np.where(h < 0.01)[0]
    for it in range(max_iters):
        improved = False
        for i in ones[:30]:
            for j in zeros[:30]:
                h[i], h[j] = h[j], h[i]
                sc = compute_c5(h)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best = h.copy()
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best, best_score

def upsample(h_old, n_new):

    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

        if sc < best_score:
            best_score, best_h = sc, h_cd

    # Phase 5: Final refinement
    if time.time() < deadline:
        n_f = len(best_h)
        h_cd, sc = coordinate_descent(best_h, n_f, max_sweeps=100)
        if sc < best_score:
            best_score, best_h = sc, h_cd
        h_sg, sc = subgradient_descent(best_h, n_f, max_iters=1500)
        if sc < best_score:
            best_score, best_h = sc, h_sg

    # Final projection
    n_final = len(best_h)
    best_h = project_sum_box(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final
