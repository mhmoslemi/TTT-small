For arrays a (length n) and v (length n):
np.correlate(a, v, mode="full") has length 2n-1.
The element at index n-1 is the "zero lag" correlation: sum_i a[i]*v[i].
The element at index n-1+k (for k>0) is: sum_i a[i+k]*v[i] (where i ranges so both indices are valid).
The element at index n-1-k (for k>0) is: sum_i a[i]*v[i+k] (equivalently, sum_i a[i-k]*v[i]).

So np.correlate(h, 1-h, mode="full")[n-1] = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]² (for general h) or 0 (for binary h).

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        m = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def local_search_sa(h, n, rng, max_iters, t0, deadline, T_init=2e-4):
