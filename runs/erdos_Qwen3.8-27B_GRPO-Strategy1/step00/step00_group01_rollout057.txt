import numpy as np
import time

def project_onto_feasible(g, target_sum):
    lo = -np.max(g) - 1.0
    hi = 2.0 - np.min(g)
    for _ in range(200):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(g + mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(g + (lo + hi) * 0.5, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def weighted_gradient(h, dx, w):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    nh = -h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        kj = k - (n - 1)
        if kj >= 0:
            grad[:n - kj] += wk * omh[kj:]
            grad[kj:] += wk * nh[:n - kj]
        else:
            grad[-kj:] += wk * omh[:n + kj]
            grad[:n + kj] += wk * nh[-kj:]
    return grad * dx

def soft_max_weights(scaled, T):
    m = np.max(scaled)
    exps = np.exp(T * (scaled - m))
    s = np.sum(exps)
    if s < 1e-300:
        return np.full_like(scaled, 1.0 / len(scaled))
    return exps / s

def optimize_h(h_init, n, dx, target_sum, deadline, T0=20.0, lr0=0.05, max_iter=3000):
    h = h_init.copy()
    c5_cur = compute_c5(h, dx)
    best_local = c5_cur
    best_local_h = h.copy()
    T = T0
    lr = lr0
    stuck = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        scaled = corr * dx
        c5_cur = float(np.max(scaled))
        w = soft_max_weights(scaled, T)
        grad = weighted_gradient(h, dx, w) * T
        alpha = lr
        improved = False
        for _ in range(15):
            h_try = project_onto_feasible(h - alpha * grad, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < c5_cur - 1e-12:
                h = h_try
                c5_cur = c5_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stuck += 1
        else:
            stuck = 0
            if c5_cur < best_local - 1e-10:
                best_local = c5_cur
                best_local_h = h.copy()
        if stuck > 200:
            if time.time() > deadline - 1.0:
                break
            h = best_local_h + 0.05 * np.random.randn(n)
            h = project_onto_feasible(h, target_sum)
            c5_cur = compute_c5(h, dx)
            stuck = 0
            T = max(T * 0.5, 5.0)
        T = min(T * 1.003, 200.0)
        if it % 500 == 499:
            lr *= 0.92
    return best_local_h, best_local

def polish_with_scipy(h_init, n, dx, target_sum, deadline):
    from scipy.optimize import minimize
    end = min(deadline, time.time() + 60)
    T = 80.0

    def obj(x):
        corr = compute_corr(x)
        scaled = corr * dx
        m = np.max(scaled)
        return float(np.mean(np.exp(T * (scaled - m))) ** (1.0 / T)) * 1.0 + m

    def obj_grad(x):
        corr = compute_corr(x)
        scaled = corr * dx
        m = np.max(scaled)
        w = soft_max_weights(scaled, T)
        g = weighted_gradient(x, dx, w) * T
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-10})
    if res.success and np.all(res.x >= -1e-8) and np.all(res.x <= 1 + 1e-8):
        h_new = project_onto_feasible(res.x, target_sum)
        c5_new = compute_c5(h_new, dx)
        return h_new, c5_new
    return h_init, compute_c5(h_init, dx)
