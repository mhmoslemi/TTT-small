def compute_c5(h, n):
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)

            if np.sum(mask) > 0:
                h[mask] += diff2 / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
