import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def repair_h(h, n):
    """Ensure 0<=h<=1 and sum(h)=n/2 exactly."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Try to distribute among non-saturated elements
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        counts = np.sum(mask)
        if counts > 0:
            h[mask] += diff / counts
            h = np.clip(h, 0.0, 1.0)
        # If still off, adjust one element
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            if diff2 > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h

    """Compute subgradient of f_k at h for a given shift, and take a step."""
    n = len(h)
    grad = np.zeros(n)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h.copy()
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
            h[idx] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    g = np.zeros(n)
    for i in range(n):

            g[i] += 1.0 - h[i + k_shift]
        if 0 <= i - k_shift < n:
            g[i] -= h[i - k_shift]
