    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fine-tune to hit exact sum
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:

        h[free] += diff / free.sum()
        h = np.clip(h, 0, 1)

import numpy as np
import time

def project_feasible(v, target_sum):
    """Project v onto {h: sum(h)=target_sum, 0<=h[i]<=1} via bisection on dual."""
    lo = -target_sum - 1.0
    hi = 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h, dx):
    g = np.correlate(h, 1.0 - h, mode='full')
    return float(g.max() * dx)

def compute_grad(h):
    """Subgradient of max_k corr_k w.r.t. h."""
    n = len(h)
    g = np.correlate(h, 1.0 - h, mode='full')
    c5_val = g.max()
    active = np.where(g >= c5_val - 1e-10)[0]
    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx) - (n - 1)
        js1 = max(0, k)
        je1 = min(n, n + k)
        if js1 < je1:
            grad[js1:je1] += (1.0 - h[js1 - k:je1 - k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            grad[js2:je2] -= h[js2 + k:je2 + k]
    return grad / max(len(active), 1)
