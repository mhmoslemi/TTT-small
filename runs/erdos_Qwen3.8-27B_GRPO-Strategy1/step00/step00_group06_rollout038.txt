def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = 0.5 * n
    d = tgt - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        r = 1.0 - h
        s = r.sum()
        if s > 1e-15:
            h += r * (d / s)

        d = tgt - h.sum()
        if d > 1e-12:
