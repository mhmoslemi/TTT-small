import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor)) * dx

def project(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0, 1)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0, 1)

def scalar_subgrad(h, n):
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    maxv = cor.max()
    tol = 0.02 * (cor.max() - cor.min() + 1e-12)
    mask = cor >= maxv - tol
    grad = np.zeros(n)
    idx = np.where(mask)[0]
    for k in idx:
        s = k - (n - 1)
        # positive part: (1-h[j+s]) for valid j
        jp = np.arange(n) + s
        v1 = (jp >= 0) & (jp < n)
        grad[v1] += (1.0 - h[jp[v1]])
        # negative part: -h[j-s] for valid j
        jm = np.arange(n) - s
        v2 = (jm >= 0) & (jm < n)
        grad[v2] -= h[jm[v2]]
    grad *= dx
    return grad, maxv * dx

def optimize_subgrad(h0, n, max_iter=5000, seed=42):
    rng = np.random.default_rng(seed)
    target = n / 2.0
    h = h0.copy().astype(float)

    best_c = compute_c5(h, n)
    best_h = h.copy()
    nesterov_mu = 0.9

        grad, c_val = scalar_subgrad(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, n, mu):
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    cor_s = cor - cor.max()
    w = np.exp(mu * cor_s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        jp = np.arange(n) + s
        m1 = (jp >= 0) & (jp < n)
        if m1.any():
            grad[m1] += w[k] * (1.0 - h[jp[m1]])
        jm = np.arange(n) - s
        m2 = (jm >= 0) & (jm < n)
        if m2.any():
            grad[m2] -= w[k] * h[jm[m2]]
    return grad * dx
