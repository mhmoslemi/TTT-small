import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)
        # Final nudge
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                i = idx[0]
                h[i] += np.clip(diff2, -h[i], 1.0 - h[i])
    return h

def sa_continuous(h, rng, budget_end, T_init=0.003, T_min=1e-5, step_init=0.12):
    n = len(h)
    best_c = compute_c5(h)
    T = T_init
    step = step_init
    stall = 0
    while time.time() < budget_end:
        new_h = h.copy()
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        d = rng.uniform(-step, step)
        if d > 0:
            d = min(d, 1.0 - new_h[i], new_h[j])
        else:
            d = max(d, -new_h[i], -(1.0 - new_h[j]))
        if abs(d) < 1e-13:
            continue
        new_h[i] += d
        new_h[j] -= d
        new_c = compute_c5(new_h)
        if new_c < best_c:
            h = new_h
            best_c = new_c
            stall = 0
            step = max(step * 0.998, 0.01)
            T = max(T * 0.9995, T_min)
        elif rng.random() < np.exp(-(new_c - best_c) / T):
            h = new_h
            stall += 1
            T = max(T * 0.999, T_min)
        else:
            stall += 1
            T = max(T * 0.999, T_min)
        if stall > 5000:
            T = max(T * 3.0, T_init * 0.5)
            step = step_init
            stall = 0
    return h, best_c

def sa_binary(h, rng, budget_end, T_init=0.002, T_min=1e-5):
    n = len(h)
    best_c = compute_c5(h)
    T = T_init
    stall = 0
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    while time.time() < budget_end:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c < best_c:
            best_c = new_c
            stall = 0
            T = max(T * 0.999, T_min)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            if rng.random() < np.exp(-(new_c - best_c) / T):
                stall += 1
                T = max(T * 0.999, T_min)
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
                stall += 1
                T = max(T * 0.999, T_min)
        if stall > 3000:
            T = max(T * 3.0, T_init * 0.5)
            stall = 0
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Start from initial_h_values
    n_init = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    h0 = fix_sum(h0, n_init / 2.0)

    # Phase 1: Continuous SA on initial (n=80)
    budget_p1 = t0 + budget_s * 0.35

    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, len(h1)

    # Phase 2: Binary optimization for several n
    n_vals = [80, 96, 100, 120, 160]
    remaining = max(30, budget_s * 0.45)
    time_per_bin = remaining / len(n_vals)

    for nv in n_vals:
        phase_end = min(time.time() + time_per_bin, t0 + budget_s - 30)
        if time.time() > phase_end:
            break
        # Multiple random restarts for binary
        n_restarts = max(2, int(time_per_bin / 60))
        best_bin_c = float('inf')
        best_bin_h = None
        for r in range(n_restarts):
            rb_end = phase_end - (n_restarts - 1 - r) * 0.1
            hb = make_binary(nv, rng)
            hb, cb = sa_binary(hb, rng, rb_end, T_init=0.003, T_min=1e-5)
            if cb < best_bin_c:
                best_bin_c = cb
                best_bin_h = hb.copy()
        if best_bin_h is not None and best_bin_c < best_c5:
            best_h, best_c5, best_n = best_bin_h, best_bin_c, nv

    # Phase 3: Fine-tune the best solution
    if time.time() < t0 + budget_s - 15:
        h_fine = best_h.copy()
        fine_end = t0 + budget_s - 5
        T = 0.0008
        step = 0.03
        stall = 0
        n_f = len(h_fine)
        while time.time() < fine_end:
            new_h = h_fine.copy()
            i = rng.integers(n_f)
            j = rng.integers(n_f)
            while j == i:
                j = rng.integers(n_f)
            d = rng.uniform(-step, step)
            if d > 0:
                d = min(d, 1.0 - new_h[i], new_h[j])
            else:
                d = max(d, -new_h[i], -(1.0 - new_h[j]))
            if abs(d) < 1e-13:
                continue
            new_h[i] += d
            new_h[j] -= d
            new_c = compute_c5(new_h)
            if new_c < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c
                h_fine = new_h
                stall = 0
                step = max(step * 0.999, 0.005)
            elif rng.random() < np.exp(-(new_c - best_c5) / T):
                h_fine = new_h
                stall += 1
            else:
                stall += 1
            T = max(T * 0.9995, 1e-6)
            if stall > 4000:
                T = 0.0005
                step = 0.02
                stall = 0

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
