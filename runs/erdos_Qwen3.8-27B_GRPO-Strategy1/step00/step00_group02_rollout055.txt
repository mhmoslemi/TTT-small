    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            break
    return h

        return cmax

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def remaining():
        return max(0.0, budget_s - (time.time() - t_start))

    # Initialize
    if 'initial_h_values' in dir() and initial_h_values is not None and len(initial_h_values) > 0:
        n0 = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float).copy()
        h0 = project_feasible(h0, n0)
    else:
        n0 = 60
        h0 = np.ones(n0) * 0.5
        h0 = project_feasible(h0, n0)

    best_h = project_feasible(h0.copy(), n0)
    best_c5 = compute_c5(best_h)
    n_best = n0

    # Phase 1: Optimize at initial resolution
    if remaining() > 60:
        h_opt, c5_val = sg_optimize(best_h, n0, max_iter=6000, step0=0.4)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
        # Try SLSQP refinement
        if remaining() > 40:
            h_s, c5_s = slsqp_optimize(best_h, n0, T=30.0, max_iter=120)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = project_feasible(h_s, n0)
            # More SLSQP with higher T
            if remaining() > 30:
                h_s2, c5_s2 = slsqp_optimize(best_h, n0, T=80.0, max_iter=100)
                if c5_s2 < best_c5:
                    best_c5 = c5_s2
                    best_h = project_feasible(h_s2, n0)
        # Also try 1-h symmetry
        h_flip = project_feasible(1.0 - best_h, n0)
        c5_flip = compute_c5(h_flip)
        if c5_flip < best_c5:
            best_c5 = c5_flip
            best_h = h_flip
        # Final SG pass on best
        if remaining() > 30:
            h_f, c5_f = sg_optimize(best_h, n0, max_iter=3000, step0=0.2)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    # Phase 2: Try different resolutions
    resolutions = []
    for mult in [2, 3, 4]:
        r = n0 * mult
        if r <= 600:
            resolutions.append(r)
    for r in [60, 90, 120, 150, 180, 240, 300, 360]:
        if r > n0 and r <= 600 and r not in resolutions:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))

    for n_try in resolutions:
        if remaining() < 80:
            break
        h_up = upsample(best_h, n_best, n_try)
        h_up = project_feasible(h_up, n_try)
        # Add small noise to escape local optima
        h_noisy = h_up + rng.normal(0, 0.02, n_try)
        h_noisy = project_feasible(h_noisy, n_try)

        # Optimize
        h_opt1, c5_1 = sg_optimize(h_up, n_try, max_iter=4000, step0=0.3)
        h_opt2, c5_2 = sg_optimize(h_noisy, n_try, max_iter=3000, step0=0.3)
        cand_h = h_opt1 if c5_1 <= c5_2 else h_opt2
        cand_c5 = min(c5_1, c5_2)

        # SLSQP refinement
        if remaining() > 40:
            h_s, c5_s = slsqp_optimize(cand_h, n_try, T=30.0, max_iter=100)
            c5_s2 = c5_s
            h_s2 = h_s
            if remaining() > 30:
                h_s2t, c5_s2t = slsqp_optimize(cand_h, n_try, T=60.0, max_iter=80)
                if c5_s2t < c5_s2:
                    h_s2, c5_s2 = h_s2t, c5_s2t
            if c5_s2 < cand_c5:
                cand_h = project_feasible(h_s2, n_try)
                cand_c5 = c5_s2

        # Complement symmetry
        h_comp = project_feasible(1.0 - cand_h, n_try)
        if compute_c5(h_comp) < cand_c5:
            cand_h = h_comp
            cand_c5 = compute_c5(cand_h)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h
            n_best = n_try

    # Phase 3: Final local polish at best resolution
    if remaining() > 20:
        h_pol, c5_pol = sg_optimize(best_h, n_best, max_iter=2000, step0=0.15)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Validate
    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_best
