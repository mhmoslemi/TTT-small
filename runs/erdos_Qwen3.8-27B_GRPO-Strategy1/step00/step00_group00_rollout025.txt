        x0 = repair_sum(best_h + rng.normal(0, 0.01, n), target)
        try:
            r = minimize(pen_obj, x0, args=(n, beta, lam),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 400, 'ftol': 1e-13})
            hx = repair_sum(r.x, target)
            cx = compute_c5(hx, n)
            if cx < best_c5:
                best_h = hx
                best_c5 = cx
        except Exception:
            pass

    # Phase 3: Directed coordinate descent (steepest-greedy)
    for r in range(5):
        if not time_ok(20):
            break
        iters = int(min(25000, max(3000, (budget_s - 60 - (time.time() - t0)) * 500)))
        hc, cc = directed_cd(best_h, n, target, iters,
                             np.random.RandomState(seed + 500 + r * 7))
        if cc < best_c5:
            best_h = hc
            best_c5 = cc

    # Phase 4: SA polish
    for r in range(4):
        if not time_ok(15):
            break
        iters = int(min(40000, max(3000, (budget_s - 50 - (time.time() - t0)) * 400)))
        hc, cc = coord_sa(best_h, n, target, iters,
                          np.random.RandomState(seed + 900 + r * 13))
        if cc < best_c5:
            best_h = hc
            best_c5 = cc

    # Final: SLSQP polish on best
    if time_ok(10):
        try:
            r = minimize(obj_c5, best_h, args=(n,), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-13})
            hx = repair_sum(r.x, target)
            cx = compute_c5(hx, n)
            if cx < best_c5:
                best_h = hx
                best_c5 = cx
        except Exception:
            pass

    best_h = repair_sum(best_h, target)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
