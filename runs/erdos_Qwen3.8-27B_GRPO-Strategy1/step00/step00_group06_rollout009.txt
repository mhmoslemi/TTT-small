import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def best_improvement_ls(h, max_rounds, time_limit):
    n = len(h)
    h = h.copy()
    t0 = time.time()
    curr = eval_c5(h)
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        one_idx = np.where(h > 0.5)[0]
        zero_idx = np.where(h < 0.5)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        best_delta = 1e-14
        best_pair = None
        for i in one_idx:
            for j in zero_idx:
                h[i], h[j] = h[j], h[i]
                new_c = eval_c5(h)
                delta = curr - new_c
                h[i], h[j] = h[j], h[i]
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]], h[best_pair[1]] = h[best_pair[1]], h[best_pair[0]]
        curr -= best_delta
    return h, curr

def random_local_search(h, max_iter, time_limit, rng):
    n = len(h)
    h = h.copy()
    t0 = time.time()
    curr = eval_c5(h)

            if new_c < curr - 1e-14:
                curr = new_c
            else:
                h[i], h[j] = h[j], h[i]

    bounds = [(0.0, 1.0)] * n
    res = minimize(penalty_obj, h0, args=(target_sum,), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    h_opt = np.clip(res.x, 0, 1)

    h_opt = h_opt + diff / n
    h_opt = np.clip(h_opt, 0, 1)
    return h_opt, eval_c5(h_opt)

    target = n / 2.0
    h = h.copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        zeros = np.where(h < 0.5)[0]
        cnt = 0
        for k in zeros:
            if cnt >= int(diff):
                break
            h[k] = 1.0
            cnt += 1
        frac = diff - cnt
        if frac > 1e-10 and cnt < len(zeros):
            h[zeros[cnt]] = min(1.0, frac)
    else:
        ones = np.where(h > 0.5)[0]
        cnt = 0
        for k in ones:
            if cnt >= int(-diff):
                break
            h[k] = 0.0
            cnt += 1
        frac = -diff - cnt
        if frac > 1e-10 and cnt < len(ones):
            h[ones[cnt]] = max(0.0, 1.0 - frac)
    return h
