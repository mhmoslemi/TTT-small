        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(80) * 0.5
    
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    h0 = project_to_manifold(h0, target0)
    
    best_h = h0.copy()
    best_n = n0
    best_c5 = compute_c5(h0, dx0)
    
    # Phase 1: Optimize at initial resolution
    h, c5 = optimize_h(h0, n0, dx0, target0, 4000, 0.05, 0.08, rng, t0, budget_s)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n0
    
    # Phase 2: Upsample and refine
    for level in range(4):
        if time.time() - t0 > budget_s * 0.60:
            break
        n_new = best_n * 2
        if n_new > 1024:
            break
        dx_new = 2.0 / n_new
        target_new = n_new / 2.0
        x_c = np.linspace(0, 2, best_n, endpoint=False)
        x_f = np.linspace(0, 2, n_new, endpoint=False)
        h_up = np.interp(x_f, x_c, best_h)
        h_up = project_to_manifold(h_up, target_new)
        h_up, c5_up = optimize_h(h_up, n_new, dx_new, target_new, 2000, 0.025, 0.04, rng, t0, budget_s)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_new
    
    # Phase 3: Multiple restarts at original resolution
    for trial in range(12):
        if time.time() - t0 > budget_s * 0.90:
            break
        if trial == 0:
            h_r = best_h.copy()
        else:
            h_r = best_h.copy()
            h_r += rng.standard_normal(best_n) * (0.2 + 0.1 * trial / max(1, len(range(12))))
            h_r = project_to_manifold(h_r, best_n / 2.0)
        n_t = best_n
        dx_t = 2.0 / n_t
        tgt_t = n_t / 2.0
        h_r, c5_r = optimize_h(h_r, n_t, dx_t, tgt_t, 800, 0.03, 0.05, rng, t0, budget_s)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_t
    
    # Phase 4: Final polish at best resolution
    n_f = best_n
    dx_f = 2.0 / n_f
    tgt_f = n_f / 2.0
    h_final, c5_final = optimize_h(best_h, n_f, dx_f, tgt_f, 2000, 0.01, 0.01, rng, t0, budget_s)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    
    return (best_h, best_c5, best_n)
