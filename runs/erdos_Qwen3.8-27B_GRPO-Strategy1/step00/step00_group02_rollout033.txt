import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-10
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.clip(h[free] + diff / cnt, 0.0, 1.0)
        else:
            free = h > 1e-10
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.clip(h[free] - (-diff) / cnt, 0.0, 1.0)
    return h

def softmax_obj(h, temp=0.005):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return dx * (m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def softmax_grad(h, temp=0.005):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-8:
            continue
        lag = idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - lag < n:
                g += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                g -= h[j + lag]
            grad[j] += w[idx] * g
    grad *= dx
    return grad, dx * (m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sum_constraint(x):
    return np.sum(x) - 0.5 * len(x)

def sum_constraint_jac(x):
    return np.ones(len(x))

def project_h(h, n):
    target = 0.5 * n
    h = h.astype(np.float64).copy()
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-15:
            break
    return np.clip(h - mid, 0.0, 1.0)

def compute_grad_softmax(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    grad = np.zeros(n)
    j = np.arange(n)
    for idx in range(len(w)):
        if w[idx] < 1e-7:
            continue
        lag = idx - (n - 1)
        g = np.zeros(n)
        jl = j - lag
        vl = (jl >= 0) & (jl < n)
        g[vl] += (1.0 - h[jl[vl]])
        jp = j + lag
        vp = (jp >= 0) & (jp < n)
        g[vp] -= h[jp[vp]]
        grad += w[idx] * g
    grad *= dx
    return grad

def pgd_optimize(h, max_iter, lr, temp, t_deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_temp = temp
    step = lr
    stall = 0
    for it in range(max_iter):
        if time.time() >= t_deadline:
            break
        grad = compute_grad_softmax(h, cur_temp)
        s = step
        improved = False
        for ls in range(30):
            h_trial = project_h(h - s * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial <= best_c5:
                h = h_trial
                if c5_trial < best_c5 - 1e-14:
                    best_h = h.copy()
                    best_c5 = c5_trial
                    stall = 0
                    improved = True
                break
            s *= 0.5
            h = h_trial
        if not improved:
            stall += 1
        if stall > 80:
            cur_temp = max(cur_temp * 0.4, 1e-5)
            step = lr
            stall = 0
            h = project_h(h + 0.005 * rng.standard_normal(n), n)
    return best_h, best_c5
