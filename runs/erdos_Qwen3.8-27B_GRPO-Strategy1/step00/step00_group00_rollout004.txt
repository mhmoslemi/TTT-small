    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return temp * logsumexp(corr / temp) * dx


def sum_eq(x):
    return np.sum(x) - len(x) / 2.0


def sum_eq_jac(x):
    return np.ones(len(x))


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) > 1e-8 and s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-5) & (h < 1.0 - 1e-5)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def rand_start(n, rng):
    h = rng.rand(n) * 0.4 + 0.3
    return repair_h(h, n)


def structured_start(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 11)
    if n_blocks == 1:
        h[:] = 0.5
        return repair_h(h, n)
    bnds = sorted(rng.choice(range(1, n), size=n_blocks - 1, replace=False))
    start = 0
    for i in range(n_blocks):
        end = bnds[i] if i < len(bnds) else n
        amp = 0.2 + rng.rand() * 0.25
        val = 0.5 + amp if i % 2 == 0 else 0.5 - amp
        h[start:end] = val
        start = end
    return repair_h(h, n)


def cos_start(n, rng):
    x = np.linspace(0, 2, n)
    freq = rng.randint(3, 20)
    phase = rng.rand() * 6.28
    h = 0.5 + 0.3 * np.cos(freq * np.pi * x + phase)
    return repair_h(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1e9

    # Try parent
    if 'initial_h_values' in globals():
        try:
            init_h = np.array(initial_h_values, dtype=float)
            if init_h.size > 0 and np.all(np.isfinite(init_h)):
                pn = len(init_h)
                init_h = repair_h(init_h, pn)
                best_h = init_h.copy()
                best_c5 = compute_c5(init_h)
        except (NameError, TypeError, ValueError):
            pass

    if best_h is None:
        best_h = rand_start(45, rng)
        best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: multi-resolution multi-start
    resolutions = [45, 100, 200, 450]
    for n in resolutions:
        if elapsed() > budget_s - 80:
            break

        starts = []
        if len(best_h) == n:
            starts.append(best_h.copy())
        else:
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n)
            starts.append(repair_h(np.interp(xn, xo, best_h), n))
        for _ in range(3):
            starts.append(rand_start(n, rng))
        for _ in range(2):
            starts.append(structured_start(n, rng))
        for _ in range(2):
            starts.append(cos_start(n, rng))

        for h0 in starts:
            if elapsed() > budget_s - 30:
                break

            # SLSQP on max objective
            try:
                res = minimize(compute_c5, h0, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq', 'fun': sum_eq,
                                             'jac': sum_eq_jac}],
                               options={'maxiter': 400, 'ftol': 1e-10})
                h_r = repair_h(res.x, n)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5, best_h = c5_r, h_r.copy()
            except Exception:
                pass

            # COBYLA (no eq constraint, repair after)
            if elapsed() > budget_s - 30:
                break
            try:
                res = minimize(compute_c5, h0, method='COBYLA',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 800, 'rhobeg': 1.0})
                h_r = repair_h(res.x, n)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5, best_h = c5_r, h_r.copy()
            except Exception:
                pass

            # Smooth L-BFGS-B + SLSQP polish
            if elapsed() > budget_s - 30:
                break
            try:
                res = minimize(smooth_c5, h0, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 500, 'ftol': 1e-12})
                h_r = repair_h(res.x, n)
                res2 = minimize(compute_c5, h_r, method='SLSQP',
                                bounds=[(0.0, 1.0)] * n,
                                constraints=[{'type': 'eq', 'fun': sum_eq,
                                              'jac': sum_eq_jac}],
                                options={'maxiter': 300, 'ftol': 1e-12})
                h_r = repair_h(res2.x, n)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5, best_h = c5_r, h_r.copy()
            except Exception:
                pass

    # Phase 2: focused local search around best
    nb = len(best_h)
    for i in range(30):
        if elapsed() > budget_s - 10:
            break
        noise = 0.08 * (1.0 - 0.85 * i / 30.0)
        h0 = repair_h(best_h + rng.uniform(-noise, noise, nb), nb)
        for method in ['SLSQP', 'COBYLA']:
            if elapsed() > budget_s - 5:
                break
            try:
                if method == 'SLSQP':
                    res = minimize(compute_c5, h0, method='SLSQP',
                                   bounds=[(0.0, 1.0)] * nb,
                                   constraints=[{'type': 'eq', 'fun': sum_eq,
                                                 'jac': sum_eq_jac}],
                                   options={'maxiter': 400, 'ftol': 1e-11})
                else:
                    res = minimize(compute_c5, h0, method='COBYLA',
                                   bounds=[(0.0, 1.0)] * nb,
                                   options={'maxiter': 600, 'rhobeg': 0.4})
                h_r = repair_h(res.x, nb)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5, best_h = c5_r, h_r.copy()
            except Exception:
                pass

    # Final repair
    nf = len(best_h)
    best_h = repair_h(best_h, nf)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, nf
