import numpy as np
from scipy.optimize import minimize
import time

_PARAMS = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def smooth_max(corr, t):
    m = float(np.max(corr))
    if t < 1.0:
        return m
    shifted = t * (corr - m)
    shifted[shifted > 500] = 500
    return m + float(np.log(np.sum(np.exp(shifted))) / t)

def fix_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.argmin(1.0 - h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub
    return h

def interp_h_periodic(h_coarse, n_fine):
    n_c = len(h_coarse)
    x_c = np.concatenate([np.arange(-n_c) / n_c, np.arange(n_c) / n_c, np.arange(n_c, 2 * n_c) / n_c])
    h_ext = np.concatenate([h_coarse, h_coarse, h_coarse])
    x_f = np.arange(n_fine) / n_fine
    return np.interp(x_f, x_c, h_ext)

def _objective(h):
    dx = _PARAMS['dx']
    t = _PARAMS['t']
    corr = compute_corr(h)
    return smooth_max(corr, t) * dx

def _obj_grad(h):
    dx = _PARAMS['dx']
    t = _PARAMS['t']
    corr = compute_corr(h)
    m = float(np.max(corr))
    shifted = t * (corr - m)
    shifted[shifted > 500] = 500
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        for i in range(n):
            i1 = i + lag
            if 0 <= i1 < n:
                grad[i] += w[k] * (1.0 - h[i1])
                grad[i1] += w[k] * (-h[i])
    return grad * dx

def _constraint_fun(h):
    return np.sum(h) - _PARAMS['target']

def _constraint_jac(h):
    return np.ones(len(h))

def _run_slsqp(h_start, n, t_val, max_iter, ftol):
    _PARAMS['dx'] = 2.0 / n
    _PARAMS['t'] = t_val
    _PARAMS['target'] = n / 2.0
    constraints = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    h_start = fix_sum(h_start, n / 2.0)
    try:
        res = minimize(_objective, h_start, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = fix_sum(res.x, n / 2.0)
            c5 = compute_c5(h_out)
            return h_out, c5
    except Exception:
        pass
    return h_start, compute_c5(h_start)

def _try_cobyla(h_start, n, max_iter):
    _PARAMS['dx'] = 2.0 / n
    _PARAMS['t'] = 1e12
    _PARAMS['target'] = n / 2.0

    def obj_cobyla(h):
        return compute_c5(h)

    def con_cobyla(h):
        return np.sum(h) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_cobyla, h_start, method='COBYLA',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': con_cobyla}],
                       options={'maxiter': max_iter, 'rhobeg': 0.05, 'catol': 1e-6})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = fix_sum(res.x, n / 2.0)
            c5 = compute_c5(h_out)
            return h_out, c5
    except Exception:
        pass
    return h_start, compute_c5(h_start)

    t0 = time.time()

    best_h = None
