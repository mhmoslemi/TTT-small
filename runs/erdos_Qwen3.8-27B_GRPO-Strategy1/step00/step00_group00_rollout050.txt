def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)

            add = min(diff, 1.0 - h[i])

                continue
            h[i] += add
            diff -= add
