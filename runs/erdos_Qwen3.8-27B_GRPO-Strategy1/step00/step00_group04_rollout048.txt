import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_integral(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0, 1)
    return h

def interpolate_h(h, n_old, n_new):
    x_old = np.arange(n_old) * 2.0 / n_old + 1.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new + 1.0 / n_new
    if n_new <= n_old:
        x_q = (x_old + np.roll(x_old, -1)) / 2.0
        idx = np.searchsorted(x_old, x_q, side='right')
        idx = np.clip(idx, 1, n_old - 1)
        frac = (x_q - x_old[idx - 1]) / (x_old[idx] - x_old[idx - 1])
        h_new = (1 - frac) * h[idx - 1] + frac * h[idx]
    else:
        f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
        h_new = np.array(f(x_new))
    return np.clip(h_new, 0, 1)

def smooth_obj(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr / t
    c_max = np.max(c)
    log_sum = c_max + np.log(np.sum(np.exp(c - c_max)))
    return t * log_sum * dx

def optimize_at_n(h_start, n, max_iter, ftol):
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    t_sm = 0.003
    obj = lambda x: smooth_obj(x, n, t_sm)
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': max_iter, 'ftol': ftol})
    h_opt = np.clip(res.x, 0, 1)
    h_opt = fix_integral(h_opt, n)
    return h_opt

def try_analytic_start(n, kind):

    if kind == 0:
        h = np.where(x < 1.0, 1.0, 0.0)
    elif kind == 1:

    elif kind == 2:
        h = 1.0 - np.abs(x - 1.0)
    elif kind == 3:

    else:
        h = np.clip(0.5 + 0.4 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x), 0, 1)
    h = fix_integral(h, n)
    return h
