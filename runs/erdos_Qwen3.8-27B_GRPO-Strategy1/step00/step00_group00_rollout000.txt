                h_start = best_h + rng.normal(0, 0.12, plan_n)
            elif strategy == 1 and best_h is not None:
                h_start = best_h.copy()
                mask = rng.random(plan_n) < 0.25
                h_start[mask] = 1.0 - h_start[mask]
                h_start += rng.normal(0, 0.05, plan_n)
            elif strategy == 2:
                # Block pattern with noise
                h_start = np.zeros(plan_n)
                block = rng.integers(3, 8)
                phase = rng.integers(0, block)
                for i in range(plan_n):
                    h_start[i] = 1.0 if (i + phase) % block < block // 2 else 0.0
                h_start += rng.normal(0, 0.3, plan_n)
            else:
                h_start = rng.random(plan_n)
                h_start = (h_start > rng.random()).astype(float)

            h_start = repair(h_start, plan_n)

            method = 'SLSQP' if rng.random() < 0.8 else 'COBYLA'
            if method == 'SLSQP':
                res = minimize(compute_c5, h_start, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 400, 'ftol': 1e-12})
            else:
                res = minimize(compute_c5, h_start, method='COBYLA', bounds=bounds,
                               constraints=cons, options={'maxiter': 300, 'rhobeg': 0.1})

            if res.x is not None:
                h_opt = repair(res.x, plan_n)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_opt.copy(), plan_n

        # Polish phase: small perturbations around best
        remaining = budget_s * 0.92 - (time.time() - start_time)
        if remaining > 10 and best_h is not None:
            bounds = [(0.0, 1.0)] * best_n
            cons = [{'type': 'eq', 'fun': SumEq(best_n)}]
            n_polish = int(min(remaining / 3, 30))
            for p in range(n_polish):
                if time.time() - start_time > budget_s * 0.95:
                    break
                h_start = best_h + rng.normal(0, 0.05, best_n)
                h_start = repair(h_start, best_n)
                res = minimize(compute_c5, h_start, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 200, 'ftol': 1e-13})
                if res.x is not None:
                    h_opt = repair(res.x, best_n)
                    c5 = compute_c5(h_opt)
                    if c5 < best_c5:
                        best_c5, best_h, best_n = c5, h_opt.copy(), best_n

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
