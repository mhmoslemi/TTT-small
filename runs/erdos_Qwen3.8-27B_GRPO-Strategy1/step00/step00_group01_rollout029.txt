    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return h

def compute_gradient(h, active_lags):
    n = len(h)
    grad = np.zeros(n)
    for m in active_lags:
        k = m - (n - 1)
        # grad[j] += [0<=j-k<n]*(1-h[j-k]) - [0<=j+k<n]*h[j+k]

import numpy as np
import time

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / len(h)))

def project(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            return h
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            return np.clip(h, 0.0, 1.0)
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def corr_grad_single(h, k, n):
    grad = np.zeros(n)
    lo1 = max(0, k)
    hi1 = min(n, n + k)
    if lo1 < hi1:
        grad[lo1:hi1] += 1.0 - h[lo1 - k:hi1 - k]
    lo2 = max(0, -k)
    hi2 = min(n, n - k)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + k:hi2 + k]
    return grad

def optimize_from(h_init, target_sum, max_time, t_start):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.06
    patience = 0
    prev_c5 = best_c5

    while time.time() - t_start < max_time:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        # Adaptive tolerance: tighter as we converge
        frac = max(0.01, 0.08 - 0.06 * patience / 200.0)
        tol = frac * max_corr + 1e-10
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            grad += corr_grad_single(h, k, n)
        grad /= len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.06]:
            h_new = project(h - lr * factor * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new
                improved = True
                break

        if improved:
            patience = 0
            prev_c5 = best_c5
        else:
            patience += 1
            lr *= 0.82

        lr = max(lr * 0.99995, 1e-7)

        if patience > 400:
            break
        if patience > 150 and best_c5 > prev_c5 - 1e-13:
            h = best_h.copy()
            patience = 100
            lr = max(lr, 0.03)

    return best_h, best_c5
