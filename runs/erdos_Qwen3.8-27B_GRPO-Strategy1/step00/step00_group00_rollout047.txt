    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if beta < 1:
        return m * dx
    vals = np.exp((corr - m) * beta)
    return (m + np.log(np.sum(vals) / len(corr)) / beta) * dx

def _max_obj(x, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

def _sum_constraint(x, target):
    return np.sum(x) - target

def _repair_sum(x, target, n):
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    if diff > 0:

    else:

        if remaining < 1e-12:

        if diff > 0:
            add = min(remaining, 1.0 - x[i])
            x[i] += add
        else:
            sub = min(remaining, x[i])
            x[i] -= sub
        remaining = abs(target - np.sum(x))
    return x

    """Round x to near-binary: 22 ones, 22 zeros, one 0.5 (for target=22.5)"""
    sorted_idx = np.argsort(x)

        result[sorted_idx[(n+1)//2:]] = 1.0
        result[sorted_idx[(n-1)//2]] = 0.5

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def _max_obj(x, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx


def _smooth_obj(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if beta < 1:
        return m * dx
    shifted = np.exp((corr - m) * beta)
    return (m + np.log(np.sum(shifted) / len(corr)) / beta) * dx


def _sum_eq(x, target):
    return np.sum(x) - target


def _repair_sum(x, target):
    x = np.clip(x, 0.0, 1.0).copy()
    diff = target - np.sum(x)
    if abs(diff) < 1e-13:
        return x
    idx = np.argsort(x) if diff > 0 else np.argsort(-x)
    remaining = abs(diff)
    for i in idx:
        if remaining < 1e-13:
            break
        if diff > 0:
            a = min(remaining, 1.0 - x[i])
            x[i] += a
        else:
            a = min(remaining, x[i])
            x[i] -= a
        remaining = abs(target - np.sum(x))
    return x


def _round_binary(x, n):
    target = n / 2.0
    idx = np.argsort(x)
    result = np.zeros(n)
    if n % 2 == 0:
        result[idx[n // 2:]] = 1.0
    else:
        result[idx[(n + 1) // 2:]] = 1.0
        result[idx[(n - 1) // 2]] = 0.5
    return result


def _local_binary_search(x, dx, target, max_sweeps=50):
    x = x.copy()
    best_val = _max_obj(x, dx)
    for _ in range(max_sweeps):
        improved = False
        ones = np.where(x > 0.5)[0]
        zeros = np.where(x < 0.5)[0]
        for i in ones:
            for j in zeros:
                if abs(x[i] - 1.0) < 1e-6 and abs(x[j]) < 1e-6:
                    x[i] = 0.0
                    x[j] = 1.0
                    if abs(np.sum(x) - target) < 1e-8:
                        v = _max_obj(x, dx)
                        if v < best_val - 1e-12:
                            best_val = v
                            improved = True
                        else:
                            x[i] = 1.0
                            x[j] = 0.0
                    else:
                        x[i] = 1.0
                        x[j] = 0.0
        if not improved:
            break
    return x, best_val

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = _repair_sum(h0, target)
    best_h = h0.copy()
    best_val = _max_obj(h0, dx)

    def _time_left():
        return budget_s - 10 - (time.time() - t0)

    # Phase 1: SLSQP from initial point with increasing beta
    bounds = [(0.0, 1.0)] * n
    for beta in [30, 80, 200, 500, 1000]:
        if _time_left() < 15:

            hc = _repair_sum(res.x, target)
