    """Project v onto {x: 0<=x<=1, sum(x)=S}"""
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(v - tau, 0.0, 1.0).sum()
        if s > S:
            lo = tau
        else:
            hi = tau
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exps = np.exp(alpha * (corr - max_c))
    wsum = exps.sum()
    weights = exps / wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-10:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                grad[:n - d] += wk * (1.0 - h[d:n])
                grad[d:] += wk * (-h[:n - d])
        else:
            dd = -d
            if dd < n:
                grad[dd:] += wk * (1.0 - h[:n - dd])
                grad[:n - dd] += wk * (-h[dd:])
    return grad * dx


def c5_objective(x, *args):
    return compute_c5(np.clip(x, 0.0, 1.0))


def sum_eq(x, *args):
    return float(np.sum(x) - args[0])


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        h_init = np.zeros(47)
        h_init[:24] = 1.0
        h_init[23] = 0.5

    n = len(h_init)
    S = n / 2.0
    h_init = project_feasible(h_init, S)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n

    # Phase 1: PGD with smooth gradient, multiple settings
    for trial, (alpha, lr0) in enumerate(zip(
            [5.0, 10.0, 20.0, 50.0, 100.0, 200.0],
            [0.02, 0.03, 0.05, 0.08, 0.1, 0.15])):
        if time.time() - t0 > budget_s - 200:
            break
        h_curr = best_h.copy()
        if trial > 0:
            h_curr += 0.01 * np.random.randn(n)
            h_curr = project_feasible(h_curr, S)
        for it in range(4000):
            if time.time() - t0 > budget_s - 200:
                break
            grad = get_smooth_gradient(h_curr, alpha)
            lr = lr0 / (1.0 + 0.002 * it)
            h_new = project_feasible(h_curr - lr * grad, S)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_curr = h_new

    # Phase 2: SLSQP refinement
    if time.time() - t0 < budget_s - 100:
        bounds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': sum_eq, 'args': (S,)}
        res = minimize(c5_objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=[cons],
                       maxiter=500, options={'ftol': 1e-13})
        h_res = project_feasible(res.x, S)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5:
            best_h = h_res
            best_c5 = c5_res

    # Phase 3: PGD from SLSQP result
    if time.time() - t0 < budget_s - 150:
        h_curr = best_h.copy()
        for it in range(3000):
            if time.time() - t0 > budget_s - 150:
                break
            grad = get_smooth_gradient(h_curr, 80.0)
            lr = 0.04 / (1.0 + 0.002 * it)
            h_new = project_feasible(h_curr - lr * grad, S)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_curr = h_new

    # Phase 4: Try other n values (interpolate from best)
    for n_try in [46, 47, 48, 63, 81, 99, 127, 159]:
        if time.time() - t0 > budget_s - 80:
            break
        if n_try == best_n:
            continue
        S_try = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, S_try)
        for it in range(2000):
            if time.time() - t0 > budget_s - 80:
                break
            grad = get_smooth_gradient(h_try, 50.0)
            h_try = project_feasible(h_try - 0.04 / (1.0 + 0.002 * it) * grad, S_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            best_n = n_try

    # Phase 5: Final SLSQP on best
    if time.time() - t0 < budget_s - 30:
        n_f = best_n
        S_f = n_f / 2.0
        bounds = [(0.0, 1.0)] * n_f
        cons = {'type': 'eq', 'fun': sum_eq, 'args': (S_f,)}
        res = minimize(c5_objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=[cons],
                       maxiter=500, options={'ftol': 1e-14})
        h_res = project_feasible(res.x, S_f)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5:
            best_h = h_res
            best_c5 = c5_res
            best_n = n_f

    return (best_h, best_c5, best_n)
