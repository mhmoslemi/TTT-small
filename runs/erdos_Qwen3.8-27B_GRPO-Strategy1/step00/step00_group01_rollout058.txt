def softmax_obj_grad(args, tau):
    h = args
    n = len(h)
    dx = 2.0 / n
    raw_corr = np.correlate(h, 1 - h, mode='full')

```python
def project_to_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}"""
    y = x.copy()
    # Binary search for lambda
    lo, hi = -max(y) - 1, max(1 - y) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        z = np.clip(y + mid, 0, 1)
        s = z.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + (lo + hi) / 2, 0, 1)
    # Final fixup (should be very small)
    diff = target_sum - z.sum()
    if abs(diff) > 1e-12:
        z += diff / len(z)
        z = np.clip(z, 0, 1)
    return z
```

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def softmax_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw_corr = np.correlate(h, 1 - h, mode='full')
    corr = raw_corr * dx
    m = np.max(corr)
    shifted = tau * (corr - m)
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    exps_sum = exps.sum()
    weights = exps / exps_sum
    obj = m + (max_shifted + np.log(exps_sum)) / tau
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        t = k - (n - 1)
        if t == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif t > 0:
            grad[t:] += w * (1.0 - h[:n-t])
            grad[:n-t] -= w * h[t:]
        else:
            s = -t
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
    grad *= dx
    return obj, grad

def project_feasible(x, target_sum):
    y = x.copy()
    lo = -y.max() - 0.1
    hi = 1.0 - y.min() + 0.1
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        z = np.clip(y + mid, 0.0, 1.0)
        if z.sum() < target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + 0.5*(lo+hi), 0.0, 1.0)
    diff = target_sum - z.sum()
    if abs(diff) > 1e-14:
        # Distribute remaining among non-saturated entries
        mask = (z > 0.001) & (z < 0.999)
        if mask.sum() > 0:
            z[mask] += diff / mask.sum()
            z = np.clip(z, 0.0, 1.0)
    return z

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def softmax_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw_corr = np.correlate(h, 1 - h, mode='full')
    corr = raw_corr * dx
    m = np.max(corr)
    shifted = tau * (corr - m)
    ms = np.max(shifted)
    exps = np.exp(shifted - ms)
    exps_sum = exps.sum()
    weights = exps / exps_sum
    obj = m + (ms + np.log(exps_sum)) / tau
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        t = k - (n - 1)
        if t == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif t > 0:
            grad[t:] += w * (1.0 - h[:n - t])
            grad[:n - t] -= w * h[t:]
        else:
            s = -t
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    grad *= dx
    return obj, grad
