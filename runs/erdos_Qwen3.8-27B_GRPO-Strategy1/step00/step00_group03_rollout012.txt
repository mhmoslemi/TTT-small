from scipy.optimize import minimize, NonlinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(x, temperature):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    cmax = np.max(corr)
    return cmax + (1.0 / temperature) * np.log(np.sum(np.exp(temperature * (corr - cmax))))

def sum_pen_obj(x, target, penalty):
    return compute_c5(x) + penalty * (np.sum(x) - target) ** 2

def grad_smooth_max(x, temperature):
    """Numerical gradient of smooth max objective"""
    n = len(x)
    eps = 1e-7
    f0 = smooth_max_obj(x, temperature)
    grad = np.zeros(n)
    for i in range(n):
        x[i] += eps
        f1 = smooth_max_obj(x, temperature)
        x[i] -= eps
        grad[i] = (f1 - f0) / eps
    return grad

def project_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n, target / n)
    h = h * (target / s)

import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(x, temperature):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    cmax = np.max(corr)
    scaled = temperature * (corr - cmax)
    scaled = np.clip(scaled, -500.0, 0.0)
    return cmax + (1.0 / temperature) * np.log(np.sum(np.exp(scaled)))

def sum_pen_obj(x, target, penalty):
    return compute_c5(x) + penalty * (np.sum(x) - target) ** 2

def project_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n, target / n)
    h = h * (target / s)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        count = min(int(np.ceil(abs(diff))) + 2, n)
        if diff > 0:
            order = np.argsort(h)
            h[order[:count]] += diff / count
        else:
            order = np.argsort(h)[::-1]
            h[order[:count]] += diff / count
    return np.clip(h, 0.0, 1.0)

def nlc_sum(x):
    return np.array([np.sum(x)])

def nlc_sum_jac(x):
    return np.ones((1, len(x)))

def interpolate_to(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

            h2 = best.copy()
            h2[i] += delta
            h2 = project_feasible(h2, target)
            c2 = compute_c5(h2)
            if c2 < best_c - 1e-12:
                best = h2
                best_c = c2
                break
        step *= 0.98
        if step < 1e-5:
            break
    return best, best_c
