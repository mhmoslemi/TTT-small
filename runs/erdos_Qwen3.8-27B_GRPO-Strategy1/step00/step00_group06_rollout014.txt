import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.float64)
    perm = rng.permutation(n)
    h[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        h[perm[n // 2]] = 0.5
    return h

def steepest_descent(h, time_limit):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    while time.time() - start < time_limit:
        # Identify distinct values
        uv = np.unique(h)
        if len(uv) < 2:
            break
        groups = []
        for v in uv:
            pos = np.where(h == v)[0]
            if len(pos) > 0:
                groups.append((v, pos))
        
        best_delta = 1e-12
        best_move = None
        best_new = best_c5
        
        for ia in range(len(groups)):
            va, pa = groups[ia]
            for ib in range(ia + 1, len(groups)):
                vb, pb = groups[ib]
                for i in pa:
                    h[i] = vb  # temporarily set i to vb
                    for j in pb:
                        h[j] = va  # temporarily set j to va
                        c5 = compute_c5(h)
                        h[j] = vb  # restore j
                        if c5 < best_new - 1e-15:
                            best_new = c5
                            best_move = (i, j)
                    h[i] = va  # restore i
        
        if best_move is None:
            break
        
        i, j = best_move
        h[i], h[j] = h[j], h[i]
        best_c5 = best_new
        best_h = h.copy()
    
    return best_h, best_c5

def iterated_ls(h_init, seed, time_limit):
    n = len(h_init)
    rng = np.random.RandomState(seed)
    h = np.array(h_init, dtype=np.float64).copy()
    best = compute_c5(h)
    best_h = h.copy()

        round_t = min(45.0, remaining)

        if c5 < best:

        # Perturb

        for _ in range(K):
            i, j = rng.choice(n, 2, replace=False)
            if h[i] != h[j]:
                h[i], h[j] = h[j], h[i]
        # If perturbed solution is worse, that's OK - steepest descent will fix it

    return best_h, best
