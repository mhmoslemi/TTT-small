def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h.astype(float), 1 - h, mode="full")
    return float(np.max(c)) * dx

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = h.copy()
    target = n // 2
    # Greedy: sort by value, set top n/2 to 1
    idx = np.argsort(h)[::-1]
    h[:] = 0
    h[idx[:target]] = 1
    return h.astype(float)

def perturb(h, rng, n, num_flips):
    """Randomly flip num_flips pairs to get a new starting point."""
    h = h.copy()

import time

def evaluate_c5(h):

    dx = 2.0 / n

    h = h.astype(float).copy()
    dx = 2.0 / n
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    deadline = time.time() + time_limit
    stuck = 0
    kick_count = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)

            ii = np.repeat(np.arange(n1), n0)
            jj = np.tile(np.arange(n0), n1)
        else:
            ii = rng.randint(n1, size=n_try)
            jj = rng.randint(n0, size=n_try)
        
        best_delta = 0.0
        best_idx = -1
        c_arr = np.correlate(h, 1.0 - h, mode="full")
        current_max = float(np.max(c_arr))
        
        for k in range(len(ii)):
            i, j = ones[ii[k]], zeros[jj[k]]
            h[i] = 0.0
            h[j] = 1.0
            c_new = np.correlate(h, 1.0 - h, mode="full")
            new_max = float(np.max(c_new))
            h[i] = 1.0
            h[j] = 0.0
            delta = new_max - current_max
            if delta < best_delta - 1e-14:

                best_idx = k

            h[j] = 1.0

            kick_count = 0

import numpy as np
import time

def evaluate_c5(h):
    """Compute C5 value for a step function h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    out = np.zeros(n, dtype=float)
    out[idx[:target]] = 1.0
    return out

def upsample_binary(h_small, n_large):
    """Upsample binary h to larger n, maintaining sum = n/2."""
    n_small = len(h_small)
    x_small = np.linspace(0.5/n_small, 1.0 - 0.5/n_small, n_small)
    x_large = np.linspace(0.5/n_large, 1.0 - 0.5/n_large, n_large)
    h_cont = np.interp(x_large, x_small, h_small.astype(float))
    return project_binary(h_cont, n_large)

def swap_search(h, n, rng, time_limit=30.0, sample_size=None, kick_every=5, kick_size=5, max_kicks=30):
    """Steepest-descent swap local search with random kicks."""
    h = h.astype(float).copy()
    dx = 2.0 / n
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    deadline = time.time() + time_limit
    stuck = 0
    total_kicks = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        total_pairs = n1 * n0
        
        if sample_size is None or total_pairs <= sample_size:
            ii = np.repeat(np.arange(n1), n0)
            jj = np.tile(np.arange(n0), n1)
        else:
            ii = rng.randint(n1, size=sample_size)
            jj = rng.randint(n0, size=sample_size)
        
        best_delta = 0.0
        best_k = -1
        n_try = len(ii)
        
        for k in range(n_try):
            i, j = ones[ii[k]], zeros[jj[k]]
            h[i] = 0.0
            h[j] = 1.0
            c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
            h[i] = 1.0
            h[j] = 0.0
            if c5 < best_c5 + best_delta - 1e-14:
                best_delta = c5 - best_c5
                best_k = k
        
        if best_k >= 0:
            i, j = ones[ii[best_k]], zeros[jj[best_k]]
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
            stuck = 0
            total_kicks = 0
        else:
            stuck += 1
            if stuck >= kick_every:
                # Random kick
                for _ in range(kick_size):
                    ones_k = np.where(h == 1)[0]
                    zeros_k = np.where(h == 0)[0]
                    if len(ones_k) > 0 and len(zeros_k) > 0:
                        i = ones_k[rng.randint(len(ones_k))]
                        j = zeros_k[rng.randint(len(zeros_k))]
                        h[i], h[j] = h[j], h[i]
                best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
                stuck = 0
                total_kicks += 1
                if total_kicks > max_kicks:
                    break
    
    return h, best_c5
