    best_v = compute_c5(best)
    beta = beta_start
    h = h.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it > 0 and it % 200 == 0:
            beta = min(beta * 1.4, 2000.0)
        obj, grad = smooth_obj_grad(h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        lr = 0.01 / (gn + 1e-10)
        improved = False
        for _ in range(6):
            h_try = project_feasible(h - lr * grad, n)
            v_try = compute_c5(h_try)
            if v_try < compute_c5(h) - 1e-12:
                h = h_try
                if v_try < best_v:
                    best_v = v_try
                    best = h_try.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            # stagnation: try larger beta or skip
            if it % 100 == 0:
                beta = min(beta * 2.0, 2000.0)
    return best, best_v


def coord_descent(h, n, t_end, max_rounds=3000):
    best = h.copy()
    best_v = compute_c5(best)
    h = h.copy()
    deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.03, -0.03]
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        cur_v = compute_c5(h)
        any_improve = False
        for j in range(n):
            for d in deltas:
                h_try = h.copy()
                h_try[j] += d
                h_try = project_feasible(h_try, n)
                v_try = compute_c5(h_try)
                if v_try < cur_v - 1e-10:
                    h = h_try
                    cur_v = v_try
                    any_improve = True
                    if v_try < best_v:
                        best_v = v_try
                        best = h_try.copy()
                    break
        if not any_improve:
            break
    return best, best_v
