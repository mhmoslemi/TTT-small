import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12 and abs(s - target) > 1e-12:
        if s < target:
            deficit = target - s
            room = 1.0 - h
            tr = np.sum(room)
            if tr > 1e-12:
                h = h + (deficit / tr) * room
        else:
            excess = s - target
            tm = np.sum(h)
            if tm > 1e-12:
                frac = 1.0 - excess / tm
                h = h * frac
    return np.clip(h, 0.0, 1.0)


def penalty_obj(x, n):
    target = n / 2.0
    pen = 10.0 * abs(np.sum(x) - target)
    return compute_c5(x) + pen


def smooth_obj(x, n, alpha):
    target = n / 2.0
    pen = 20.0 * abs(np.sum(x) - target)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    shifted = corr - m
    sm = np.log(np.sum(np.exp(alpha * shifted)) / len(corr)) / alpha + m
    return sm + pen


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def upsample(h_from, n_to):
    n_from = len(h_from)
    xs_o = np.linspace(0.0, 2.0, n_from, endpoint=False)
    xs_n = np.linspace(0.0, 2.0, n_to, endpoint=False)
    h_new = np.interp(xs_n, xs_o, h_from)
    return repair_h(h_new, n_to)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h = None
    best_c5 = 2.0

    n45 = 45
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
        if len(h_init) == n45:
            h_init = repair_h(h_init, n45)
            c = compute_c5(h_init)
            if c < best_c5:
                best_c5 = c
                best_h = h_init.copy()

    if best_h is None:
        best_h = np.zeros(n45)
        best_h[:n45 // 2] = 1.0
        best_h = repair_h(best_h, n45)
        best_c5 = compute_c5(best_h)

    # Phase 1: COBYLA on n=45, multiple restarts
    bounds45 = [(0.0, 1.0)] * n45
    starts = [best_h.copy()]
    for i in range(6):
        s = np.random.uniform(0.05, 0.25)
        pert = best_h + s * np.random.randn(n45)
        starts.append(repair_h(pert, n45))
    for i in range(4):
        r = np.random.rand(n45)
        starts.append(repair_h(r, n45))

    for x0 in starts:
        if time.time() - t_start > budget_s - 200:
            break
        maxfev = 3000
        res = minimize(penalty_obj, x0, args=(n45,), method='COBYLA',
                      bounds=bounds45,
