import numpy as np
import time


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def upsample_to(h_src, n_target, rng):
    n_src = len(h_src)
    if n_target == n_src:
        return h_src.astype(np.float64).copy()
    idx = np.round(np.linspace(0, n_src - 1, n_target)).astype(int)
    h_new = h_src[idx].astype(np.float64)
    target_sum = n_target // 2
    current_sum = int(np.round(h_new.sum()))
    diff = target_sum - current_sum
    if diff > 0:
        zeros = np.where(h_new == 0.0)[0]
        perm = rng.permutation(len(zeros))
        h_new[zeros[perm[:diff]]] = 1.0
    elif diff < 0:
        ones = np.where(h_new == 1.0)[0]
        perm = rng.permutation(len(ones))
        h_new[ones[perm[:abs(diff)]]] = 0.0
    return h_new


def greedy_search(h, deadline, max_stall=15):
    n = len(h)
    h = h.copy().astype(np.float64)
    c5 = compute_c5(h)
    stall = 0
    while stall < max_stall:
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n1, n0 = len(ones), len(zeros)
        for ii in range(n1):
            if time.time() > deadline:
                return h, c5
            i = ones[ii]
            if h[i] != 1.0:
                continue
            for j in zeros:

                h[i] = 0.0
                h[j] = 1.0

                    c5 = nc
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if improved:
            stall = 0
        else:
            stall += 1
    return h, c5
