import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _soft_max_val(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    scaled = alpha * corr
    mx = float(np.max(scaled))
    return (mx + float(np.log(np.sum(np.exp(scaled - mx))))) / alpha

def _soft_max_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    scaled = alpha * corr
    mx = float(np.max(scaled))
    p = np.exp(scaled - mx)
    p /= float(np.sum(p))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if p[r] < 1e-12:
            continue
        lag = r - (n - 1)
        w = alpha * dx * p[r]
        lo = max(0, lag)
        hi = min(n, n + lag)
        if hi > lo:
            idx = np.arange(lo, hi)
            grad[idx] += w * (1.0 - h[idx - lag])
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if hi > lo:
            idx = np.arange(lo, hi)
            grad[idx] -= w * h[idx + lag]
    return grad

def _sfun(x, dx, alpha):
    return _soft_max_val(x, dx, alpha)

    return _soft_max_grad(x, dx, alpha)

def _con_fun(x, ts):
    return x.sum() - ts

        return h
    if diff > 0:
        order = np.argsort(h)
    else:
        order = np.argsort(-h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h
