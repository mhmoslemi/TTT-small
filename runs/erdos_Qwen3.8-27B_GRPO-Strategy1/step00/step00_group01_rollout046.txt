```python
import numpy as np
import time
from scipy.interpolate import interp1d

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(corr_full(h)) * dx

def project(y, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_grad(h, temp):
    n = len(h)
    corr = corr_full(h)
    cs = corr / temp
    cs -= cs.max()
    w = np.exp(cs)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, -lag)
        je = min(n, n - lag)
        if je > js:
            grad[js:je] += wk * (1.0 - h[js + lag:je + lag])
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if je2 > js2:
            grad[js2:je2] -= wk * h[js2 - lag:je2 - lag]
    return grad, np.max(corr)

def gradient_descent(h0, alloc_s, start_time):
    n = len(h0)
    target = n / 2.0
    h = project(h0.copy(), target)
    best_h = h.copy()
    best_score = compute_score(h)
    temp = 0.08
    temp_min = 0.002
    lr = 0.08
    patience = 0
    max_patience = 300
    it = 0
    max_it = 80000
    while it < max_it:
        if time.time() - start_time > start_time + alloc_s:
            break
        grad, _ = soft_max_grad(h, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-8:
            break
        h_new = project(h - lr * grad / max(gnorm, 1e-6) * np.linalg.norm(grad), target)
        s = compute_score(h_new)
        if s < best_score - 1e-9:
            best_score = s
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            lr *= 0.98
        h = h_new
        temp = max(temp_min, 0.08 * np.exp(-it / 8000.0))
        it += 1
    return best_h, best_score

def coordinate_refine(h, start_time, budget_s):
    n = len(h)
    target = n / 2.0
    h = project(h.copy(), target)
    best_score = compute_score(h)
    for epoch in range(30):
        if time.time() - start_time > budget_s - 5:
            break
        improved = False
        for j in range(n):
            old = h[j]
            for d in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.1, -0.1]:
                ht = h.copy()
                ht[j] = old + d
                ht = project(ht, target)
                s = compute_score(ht)
                if s < best_score - 1e-10:
                    h = ht
                    best_score = s
                    improved = True
                    break
        if not improved:
            break
    return h, best_score

def perturbation_search(h, start_time, budget_s, n_pert=8):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_score(h)
    for _ in range(n_pert):
        if time.time() - start_time > budget_s - 5:
            break
        noise = 0.15 * np.random.randn(n)
        hp = project(h + noise, target)
        h2, s2 = gradient_descent(hp, min(5.0, budget_s - (time.time() - start_time)), start_time)
        if s2 < best_score:
            best_h = h2
            best_score = s2
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    best_h = None
    best_score = float('inf')
    best_n = 0

    def try_update(h, s, n):
        nonlocal best_h, best_score, best_n
        if s < best_score:
            best_score = s
            best_h = h.copy()
            best_n = n

    # Phase 1: Optimize initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
        h_opt, s_opt = gradient_descent(h_init, budget_s * 0.3, start_time)
        h_opt, s_opt = coordinate_refine(h_opt, start_time, start_time + budget_s * 0.5)
        try_update(h_opt, s_opt, n_init)

    # Phase 2: Multiple random restarts at n=47
    if time.time() - start_time < budget_s * 0.55:
        structures = []
        # Symmetric tent
        x = np.linspace(0, 1, 47)
        structures.append(project(0.5 + 0.5 * np.cos(np.pi * x * 4), 23.5))
        # Two bumps
        structures.append(project(0.5 + 0.3 * np.sin(2 * np.pi * x * 2) + 0.1 * np.sin(4 * np.pi * x), 23.5))
        # Random
        structures.append(np.random.uniform(0, 1, 47))
        structures.append(project(np.random.uniform(0, 1, 47), 23.5))
        for h_start in structures:
            if time.time() - start_time > budget_s * 0.6:
                break
            h_opt, s_opt = gradient_descent(h_start, min(20.0, budget_s - (time.time() - start_time)), start_time)
            h_opt, s_opt = coordinate_refine(h_opt, start_time, start_time + budget_s * 0.65)
            try_update(h_opt, s_opt, 47)

    # Phase 3: Perturbation + re-optimize around best
    if best_h is not None and time.time() - start_time < budget_s * 0.75:
        h_p, s_p = perturbation_search(best_h, start_time, start_time + budget_s * 0.8, n_pert=6)
        try_update(h_p, s_p, len(h_p))

    # Phase 4: Upsample to n=94 and optimize
    if best_h is not None and time.time() - start_time < budget_s * 0.85:
        n_old = len(best_h)
        n_new = 2 * n_old
        x_old = np.arange(n_old, dtype=float)
        x_new = np.linspace(0, n_old - 1, n_new)
        f_interp = interp1d(x_old, best_h, kind='cubic', fill_value=(best_h[0], best_h[-1]))
        h_up = f_interp(x_new)
        h_up = project(h_up, n_new / 2.0)
        h_opt, s_opt = gradient_descent(h_up, min(60.0, budget_s - (time.time() - start_time) - 10), start_time)
        h_opt, s_opt = coordinate_refine(h_opt, start_time, start_time + budget_s - 5)

            # Another gradient pass

            if s2 < best_score:

            best_h, best_score = coordinate_refine(best_h, start_time, start_time + remaining)

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_score = compute_score(best_h)
    return best_h, final_score, len(best_h)
```
