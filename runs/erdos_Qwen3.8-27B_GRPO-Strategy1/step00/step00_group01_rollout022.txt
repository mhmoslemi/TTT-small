import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float((2.0 / n) * np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_feasible(h):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 0 and abs(diff) < 1.0:
        idx = int(np.argmax(h < 1) if diff > 0 else np.argmin(h > 0)) if cnt > 0 else 0
        adjust = min(max(diff, -h[idx]), 1.0 - h[idx])
        h[idx] += adjust
    return h

def random_hc(h, rng, n_iters, deadline):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    eps_list = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        for eps in eps_list:
            if h[i] + eps <= 1 + 1e-10 and h[j] - eps >= -1e-10:
                h[i] += eps; h[j] -= eps
                c = compute_c5(h)
                if c < best:
                    best = c
                else:
                    h[i] -= eps; h[j] += eps
            if h[j] + eps <= 1 + 1e-10 and h[i] - eps >= -1e-10:
                h[j] += eps; h[i] -= eps
                c = compute_c5(h)
                if c < best:
                    best = c
                else:
                    h[j] -= eps; h[i] += eps
    return h, best

def sys_hc(h, deadline):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    eps_list = [0.005, 0.015, 0.05]
    while time.time() < deadline:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for eps in eps_list:
                    if h[i] + eps <= 1 + 1e-10 and h[j] - eps >= -1e-10:
                        h[i] += eps; h[j] -= eps
                        c = compute_c5(h)
                        if c < best - 1e-15:
                            best = c; improved = True
                        else:
                            h[i] -= eps; h[j] += eps
                    if h[j] + eps <= 1 + 1e-10 and h[i] - eps >= -1e-10:
                        h[j] += eps; h[i] -= eps
                        c = compute_c5(h)
                        if c < best - 1e-15:
                            best = c; improved = True
                        else:
                            h[j] -= eps; h[i] += eps
        if not improved:
            break
    return h, best

def anneal(h, rng, n_iters, deadline):
    n = len(h)
    h = h.copy()
    bh = h.copy()
    best = compute_c5(h)
    cur = best
    T = 0.025
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        T = max(T * 0.99993, 2e-5)
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        eps = rng.uniform(-0.5, 0.5)
        ni, nj = h[i] + eps, h[j] - eps
        if not (-1e-10 <= ni <= 1 + 1e-10 and -1e-10 <= nj <= 1 + 1e-10):
            continue
        h[i], h[j] = ni, nj
        c = compute_c5(h)
        if c <= cur or (c - cur) / T < -np.log(max(rng.random(), 1e-15)):
            cur = c
            if c < best:
                best, bh = c, h.copy()
        else:
            h[i], h[j] = h[i] - eps, h[j] + eps
    return bh, best

    if len(h) == nn:
        return h.copy()
    xo = np.arange(len(h)) / len(h)
    xn = np.arange(nn) / nn
    return project_feasible(np.interp(xn, xo, h))

    return bh, final_c5, len(bh)
