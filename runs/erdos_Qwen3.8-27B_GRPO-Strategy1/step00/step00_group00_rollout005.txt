import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(g)

def repair_sum(h, S):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = S - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, dx, S, max_sweeps=50):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            # Try 0, 1, and small perturbations
            trials = [0.0, 1.0, old - 0.1, old - 0.05, old + 0.05, old + 0.1,
                      old - 0.2, old + 0.2]
            best_v = cur
            best_x = old
            for t in trials:
                t = min(1.0, max(0.0, t))
                h[i] = t
                hr = repair_sum(h, S)
                v = compute_c5(hr, dx)
                if v < best_v - 1e-12:
                    best_v = v
                    best_x = hr.copy()
            h = best_x
            if best_v < cur - 1e-12:
                cur = best_v
                improved = True
        if not improved:
            break
    return h, cur

def try_binary_patterns(h, n, dx, S):
    """Try to make the solution more binary."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Round toward binary: for each element, try 0 or 1
    h_bin = h.copy()
    for i in range(n):
        for val in [0.0, 1.0]:
            h_bin[i] = val
            h_try = repair_sum(h_bin, S)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            else:
                h_bin[i] = h[i]  # revert
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(g)

def repair_sum(h, S):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = S - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, dx, S, max_sweeps=200, time_limit=None, t0=0):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h, dx)

            break
        improved = False
        for i in range(n):
            old = h[i]
            trials = [0.0, 1.0, old - 0.25, old - 0.1, old - 0.03,
                      old + 0.03, old + 0.1, old + 0.25]
            best_v = cur
            best_arr = h.copy()
            for t in trials:
                t = min(1.0, max(0.0, t))
                if abs(t - old) < 1e-12:
                    continue
                h[i] = t
                hr = repair_sum(h, S)
                v = compute_c5(hr, dx)
                if v < best_v - 1e-13:
                    best_v = v
                    best_arr = hr.copy()
            if best_v < cur - 1e-13:
                h = best_arr
                cur = best_v
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, cur

def slsqp_step(x0, n, dx, S, maxiter=400):
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        c5 = compute_c5(x, dx)
        pen = 200.0 * (np.sum(x) - S) ** 2
        return c5 + pen

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S,
             'jac': lambda x: np.ones_like(x)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair_sum(np.clip(res.x, 0, 1), S)
    except Exception:
        return repair_sum(np.clip(x0, 0, 1), S)
