import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum}."""
    n = len(h)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient(h, n):
    """Compute subgradient of C5 at h (gradient of max correlation component)."""
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    k_star = int(np.argmax(corr))
    d = k_star - (n - 1)
    
    grad = np.zeros(n)
    # corr[k] = sum_i h[i]*(1-h[i+d]) for valid (i, i+d)
    # d/∂h[j] corr:
    #   if j+d valid: (1-h[j+d]) (j as first factor)
    #   if j-d valid: -h[j-d] (j as second factor: derivative of h[j-d]*(1-h[j]) w.r.t. h[j] = -h[j-d])
    for j in range(n):
        if 0 <= j + d < n:
            grad[j] += (1.0 - h[j + d])
        if 0 <= j - d < n:
            grad[j] += (-h[j - d])
    return dx * grad

    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    t0 = time.time()
    lr0 = 0.5
    gamma = 0.999
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if it % 500 == 0 and best_c5 < 0.38085:
            break
        
        grad = subgradient(h, n)
        lr = lr0 / (1.0 + gamma * it)

                best_h = h_new.copy()
                h = h_new

    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    momentum = np.zeros(n)
    beta = 0.9
    lr0 = 0.8
    no_improve = 0
    patience = 500

    for it in range(200000):
        if time.time() - t0 > time_limit:
            break
        grad = get_subgradient(h, n)
        momentum = beta * momentum + (1 - beta) * grad
        lr = lr0 / (1.0 + 0.001 * it) ** 0.7
        h_new = h - lr * momentum
        h_new = project_box_sum(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            if c5 < best_c5 - 1e-8:
                no_improve = 0
            else:
                no_improve += 1
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            if no_improve > patience:
                no_improve = 0
                beta *= 0.95
        else:
            no_improve += 1
            if no_improve > patience:
                no_improve = 0
                beta *= 0.9
                lr0 *= 0.7

    return best_h, best_c5


def upscale(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    return np.interp(x_new, x_old, h)


def make_block_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    # Alternating blocks
    block_size = max(1, n_ones // (n_ones // 2 + 1))
    pos = 0
    for i in range(n_ones):
        bsz = max(1, int(block_size * (1 + 0.3 * rng.random())))
        end = min(pos + bsz, n)
        h[pos:end] = 1.0
        pos = end
        if pos >= n:
            break
        gap = max(0, n - pos - (n_ones - i - 1) * bsz)
        pos += max(1, gap // max(1, n_ones - i))
    h = project_box_sum(h, n / 2.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_block_init(45, rng)

    n0 = len(best_h)
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Resolution path
    resolutions = [n0]
    cur = n0
    while cur < 400:
        nxt = min(cur * 2, 400)
        if nxt > cur:
            resolutions.append(nxt)
            cur = nxt
        else:
            break

    for n_pts in resolutions:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 40:
            break

        # Upscale current best to this resolution
        if n_pts > best_n:
            h_cur = upscale(best_h, n_pts)
        else:
            h_cur = best_h.copy()
        target_sum = n_pts / 2.0
        h_cur = project_box_sum(h_cur, target_sum)

        # Allocate time: 70% to restarts at optimal resolution
        if n_pts == resolutions[-1]:
            time_alloc_total = remaining * 0.7
        else:
            time_alloc_total = remaining * 0.15

        n_restarts = min(8, max(2, int(time_alloc_total / 25)))
        time_per = time_alloc_total / n_restarts

        for r in range(n_restarts):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 15:
                break

            if r == 0:
                h_start = h_cur.copy()
            elif r == 1:
                h_start = make_block_init(n_pts, rng)
            elif r == 2:
                # Lattice pattern
                h_start = np.zeros(n_pts)
                n_ones = n_pts // 2
                spacing = n_pts / n_ones
                for i in range(n_ones):
                    h_start[int(i * spacing) % n_pts] = 1.0
                h_start = project_box_sum(h_start, target_sum)
            else:
                noise = 0.03 * r * rng.standard_normal(n_pts)
                h_start = h_cur + noise
                h_start = project_box_sum(h_start, target_sum)

            alloc = min(time_per, remaining - 10)
            if alloc < 5:
                break

            h_opt, c5_opt = optimize_subgrad(h_start, target_sum, rng, time_limit=alloc)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_pts

        # Update for next resolution
        best_h = project_box_sum(best_h, n_pts / 2.0)
        best_n = n_pts

    # Final polish at best resolution with whatever time remains
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        h_opt, c5_opt = optimize_subgrad(best_h, best_n / 2.0, rng,
                                         time_limit=remaining - 10)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    # Fix rounding to get exact sum
    diff = n_final / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-10:
        # Distribute the difference to non-boundary elements
        interior = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        n_int = np.sum(interior)
        if n_int > 0:
            best_h[interior] += diff / n_int
            best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
