```python
def smoothed_max_gradient(h, n_pts, T):
    """Compute gradient of (1/T)*log(sum(exp(T*scores))) w.r.t. h"""
    scores = get_scores(h, n_pts)
    dx = 2.0 / n_pts
    scaled = T * scores
    m = np.max(scaled)
    logits = scaled - m
    exp_logits = np.exp(logits)
    weights = exp_logits / exp_logits.sum()
    
    grad = np.zeros(n_pts)
    for k in range(2 * n_pts - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n_pts - 1)
        if s == 0:
            continue  # self-correlation, always 0
        if s > 0:
            # First term: j=0 to n-1-s: (1-h[j+s])
            idx1 = np.arange(n_pts - s)
            grad[idx1] += w * (1.0 - h[idx1 + s])
            # Second term: j=s to n-1: -h[j-s]
            idx2 = np.arange(s, n_pts)
            grad[idx2] -= w * h[idx2 - s]
        else:
            t = -s
            # First term (valid_i): j=t to n-1: (1-h[j-t])
            idx1 = np.arange(t, n_pts)
            grad[idx1] += w * (1.0 - h[idx1 - t])
            # Second term: j=0 to n-1-t: -h[j+t]
            idx2 = np.arange(n_pts - t)
            grad[idx2] -= w * h[idx2 + t]
    
    return dx * grad
```

import numpy as np
import time

def project_h(v, target_sum):
    n = len(v)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Correct sum
    h += (target_sum - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    return h

def scores_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def c5_of(h):
    return scores_of(h).max()

def grad_smoothed_max(h, T):
    n = len(h)
    dx = 2.0 / n
    sc = scores_of(h)
    Tsc = T * sc
    m = Tsc.max()
    ew = np.exp(Tsc - m)
    pw = ew / ew.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = pw[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s == 0:
            continue
        if s > 0:
            # j = 0..n-1-s: +(1-h[j+s])
            grad[:n-s] += w * (1.0 - h[s:])
            # j = s..n-1: -h[j-s]
            grad[s:] -= w * h[:n-s]
        else:
            t = -s
            # j = t..n-1: +(1-h[j-t])
            grad[t:] += w * (1.0 - h[:n-t])
            # j = 0..n-1-t: -h[j+t]
            grad[:n-t] -= w * h[t:]
    return dx * grad

    c5_best = c5_of(h_best)
    
    # Main optimization loop
    max_iterations = 2000

    T_max = 100.0

    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iterations):

        if elapsed > budget_s - 60:
            break

        # Nesterov momentum
        momentum = beta * momentum + (1 - beta) * g
        h_new = h_best - alpha * momentum

        if c5_new < c5_best - 1e-8:

  for m in valid_m:
    # h[m] * (1-h[m+s])
    # ∂/∂h[m] = (1-h[m+s])
    # ∂/∂h[m+s] = -h[m]
    grad[m] += w * (1 - h[m+s])
    grad[m+s] += w * (-h[m])

This is O(n) per shift and O(n) shifts = O(n²) total. For n=47, that's ~2200 operations. Very fast.

Let me implement this:

```python
def grad_smoothed_max(h, T):
    n = len(h)
    dx = 2.0 / n
    sc = scores_of(h)
    Tsc = T * sc
    m = Tsc.max()
    ew = np.exp(Tsc - m)
    pw = ew / ew.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = pw[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            continue
        m_lo = max(0, -s)
        m_hi = min(n, n - s)
        for m in range(m_lo, m_hi):
            grad[m] += w * (1.0 - h[m + s])
            grad[m + s] += w * (-h[m])
    return dx * grad
```

This is correct but has a Python loop. For n=47, 2n-1=93 shifts, each with up to 47 terms. Total iterations: ~93*23 ≈ 2139. Each is a few ops. Should be fast enough.

```python
def grad_smoothed_max(h, T):
    n = len(h)
    dx = 2.0 / n
    sc = scores_of(h)
    Tsc = T * sc
    m = Tsc.max()
    ew = np.exp(Tsc - m)
    pw = ew / ew.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = pw[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            continue
        m_lo = max(0, -s)
        m_hi = min(n, n - s)
        ms = np.arange(m_lo, m_hi)
        grad[ms] += w * (1.0 - h[ms + s])
        grad[ms + s] += -w * h[ms]
    return dx * grad
```

This should be efficient. For n=47, each of the ~93 iterations does a vectorized op on arrays of length ≤47. Very fast.
