    diff = target - np.sum(h)
    # Adjust iteratively
    for _ in range(10):
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)

                    c = c_new
                    improved = True
                    break

import numpy as np
import time
from scipy.optimize import minimize

def compute_cost(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_cost(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(corr - m))) / T) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argsort(h)[::-1] if diff > 0 else np.argsort(h)
        remaining = abs(diff)
        for i in idx:
            step = min(remaining, h[i] if diff > 0 else 1.0 - h[i])
            h[i] -= step if diff > 0 else -step
            remaining -= step
            if remaining < 1e-14:
                break
    return h

def coordinate_polish(h, n, max_rounds=200, time_limit=30.0):

                if c_try < c - 1e-12:
