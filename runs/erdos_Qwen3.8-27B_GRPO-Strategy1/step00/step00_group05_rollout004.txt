import numpy as np
from scipy.optimize import minimize
import time

def c5_from_h(h, dx):
    """Compute C5 value from h array."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    """Adjust h to have the correct sum while staying in [0,1]."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to add: increase zeros
            mask = (h < 1.0)
        else:
            # Need to subtract: decrease ones
            mask = (h > 0.0)
        count = np.sum(mask)
        if count > 0:
            step = diff / count
            h[mask] = np.clip(h[mask] + step, 0, 1)
    return h

def upsample(h, new_n):
    """Upsample h to new_n points."""
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def smooth_max(corr, dx, alpha=100):
    """Smooth approximation of max(corr) * dx."""
    max_val = np.max(corr)
    return dx * max_val + dx / alpha * np.log(np.sum(np.exp(alpha * (corr - max_val))))

    """Subgradient optimization for the max correlation."""
    h = h0.copy()
    dx = 2.0 / n
    target_sum = n / 2.0

        # represents the correlation at lag (k - n + 1)
        # corr_full[k] = sum_{valid i} h[i] * (1-h[i + (k-n+1)])

        lag = k_max - n + 1
        grad = np.zeros(n)
        
        if lag >= 0:

            # d/dh[j]: 
            #   if j = i+lag (i.e., i = j-lag): contributes (1-h[j-lag]) for i=j-lag
            #   if j = i (i.e., i = j): contributes -h[j+lag] for i=j
            for i in range(n - lag):

                j1 = i + lag  # position in h that multiplies (1-h[i])
                grad[j1] += (1.0 - h[i])
                j2 = i  # position in h that's subtracted
                grad[j2] -= h[i + lag]
        else:

            L = -lag

            for i in range(L, n):

            hb[i], hb[j] = hb[j], hb[i]
            vc = c5_value(hb)
            if vc >= vb:
                hb[i], hb[j] = hb[j], hb[i]
            else:
                vb = vc
        try_update(hb, np_)

    # Higher resolutions
    for n in [140, 210, 280, 350, 490, 630]:
        if elapsed() > budget_s * 0.85:
            break
        remaining = (budget_s * 0.85 - elapsed()) / max(1, 630 // n)

        x_old = np.linspace(0, 1, np_, endpoint=False)
        x_new = np.linspace(0, 1, n, endpoint=False)
        h_up = np.interp(x_new, x_old, parent)
        h_bin = binary_upsample(parent, n)
        h_bin = fix_sum(h_bin, n / 2.0)
        h_rand = np.zeros(n)
        h_rand[rng.permutation(n)[:n // 2]] = 1.0

        for h_init in [h_up, h_bin, h_rand]:
            if elapsed() > budget_s * 0.88:
                break
            # SLSQP for smaller n
            if n <= 280 and elapsed() < budget_s * 0.85:
                dx = 2.0 / n
                target = n / 2.0
                bounds2 = [(0.0, 1.0)] * n
                cons2 = [{'type': 'eq', 'fun': lambda x, t=target: x.sum() - t,
                          'jac': lambda x: np.ones(n)}]
                try:
                    r = minimize(lambda x: float(np.max(np.correlate(x, 1.0-x, mode="full")) * dx),
                                 h_init.copy(), method='SLSQP', bounds=bounds2,
                                 constraints=cons2, options={'maxiter': 800, 'ftol': 1e-12})
                    try_update(r.x, n)
                except Exception:
                    pass

            # Subgradient
            if elapsed() < budget_s * 0.88:
                max_it = max(1000, int(300000 / n))
                hr, vr = subgrad_optimize(h_init.copy(), max_it, lr_init=0.06)
                try_update(hr, n)

    # Final polish: subgradient on best solution found
    if best_h is not None and elapsed() < budget_s * 0.92:
        max_it = max(2000, int(500000 / best_n))
        hr, vr = subgrad_optimize(best_h.copy(), max_it, lr_init=0.03)
        try_update(hr, best_n)

    if best_h is None:
        best_h = parent
        best_n = np_
        best_c5 = c5_value(parent)

    return best_h, best_c5, best_n
