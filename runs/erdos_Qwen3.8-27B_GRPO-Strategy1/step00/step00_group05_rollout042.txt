from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, dx):
    """Compute C5 = max_k sum_i h[i] * (1-h[i+k]) * dx"""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr) * dx

def make_binary(h, n):
    """Convert continuous h to binary with exactly n//2 ones."""
    h_sorted = np.argsort(h)[::-1]  # indices of largest values first
    binary = np.zeros(n, dtype=np.float64)
    binary[h_sorted[:n//2]] = 1.0
    return binary

def swap_delta(h, i, j, n):
    """Compute delta c[k] for swapping h[i]=1->0, h[j]=0->1.
    Returns array of length 2n-1 with the changes."""
    k = np.arange(-(n-1), n)
    # delta c[k] = h[i+k] + h[i-k] - h[j+k] - h[j-k] (with 0 for out of range)
    def safe_h(idx):
        mask = (idx >= 0) & (idx < n)
        result = np.zeros(len(idx))
        result[mask] = h[idx[mask]]
        return result
    ik = i + k
    ik2 = i - k
    jk = j + k
    jk2 = j - k
    return safe_h(ik) + safe_h(ik2) - safe_h(jk) - safe_h(jk2)

def local_search_binary(h, max_time=50, seed=42):
    """Greedy + SA local search for binary h."""
    rng = np.random.default_rng(seed)
    n = len(h)
    dx = 2.0 / n

    zeros = np.where(h == 0)[0]

    best_c5 = np.max(corr) * dx
    current_c5 = best_c5
    
    start_time = time.time()
    no_improve = 0
    temp = 0.1

        if i == j:

        else:

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n):
    n_half = n // 2
    # Handle ties by using index as tiebreaker
    h_with_idx = h + np.arange(n) * 1e-10
    h_sorted = np.argsort(h_with_idx)[::-1]
    binary = np.zeros(n, dtype=np.float64)
    binary[h_sorted[:n_half]] = 1.0
    return binary

def sturmian_init(n, alpha):
    h = np.zeros(n, dtype=np.float64)
    for i in range(n):
        h[i] = np.floor((i + 1) * alpha) - np.floor(i * alpha)
    return make_binary(h, n)

def simulated_anneal(h_init, dx, max_time, seed):
    rng = np.random.default_rng(seed)
    n = len(h_init)
    h = h_init.astype(np.float64).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    temp = 0.02
    min_temp = 0.002
    start = time.time()
    stall = 0
    max_stall = 30000
    
    while time.time() - start < max_time and stall < max_stall:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        
        if temp > min_temp:
            temp *= 0.99999
    
    return best_h, best_c5

def greedy_step(h, dx):
    """One greedy step: find the single best swap."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    cur_max = np.max(corr)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_improve = 0
    best_i, best_j = -1, -1

        for j in zeros:
            h[i], h[j] = 0.0, 1.0
            new_max = np.max(np.correlate(h, 1-h, mode="full"))
            h[i], h[j] = 1.0, 0.0

            if improve > best_improve + 1e-15:

                best_i, best_j = i, j
    
    if best_i >= 0:
        h[best_i], h[best_j] = 0.0, 1.0
        return True
    return False

    start = time.time()

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n):
    n_half = n // 2
    h_tie = h + np.arange(n, dtype=np.float64) * 1e-10
    order = np.argsort(h_tie)[::-1]
    binary = np.zeros(n, dtype=np.float64)
    binary[order[:n_half]] = 1.0
    return binary

def sturmian_init(n, alpha):
    h = np.zeros(n, dtype=np.float64)
    for i in range(n):
        h[i] = np.floor((i + 1) * alpha) - np.floor(i * alpha)
    return make_binary(h, n)

def random_binary(n, seed):
    rng = np.random.default_rng(seed)
    idx = rng.choice(n, n // 2, replace=False)
    r = np.zeros(n, dtype=np.float64)
    r[idx] = 1.0
    return r

def simulated_anneal(h_init, dx, max_time, seed):
    rng = np.random.default_rng(seed)
    n = len(h_init)
    h = h_init.astype(np.float64).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    temp = 0.025
    min_temp = 0.003
    cool_rate = 0.999995
    start = time.time()
    stall = 0

    while time.time() - start < max_time and stall < 40000:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        if temp > min_temp:
            temp *= cool_rate
    return best_h, best_c5

def greedy_refine(h, dx, max_time):
    h = h.copy()
    n = len(h)
    start = time.time()
    while time.time() - start < max_time:
        corr = np.correlate(h, 1 - h, mode="full")
        cur_max = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_improve = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                nm = float(np.max(np.correlate(h, 1 - h, mode="full")))
                h[i], h[j] = 1.0, 0.0
                imp = cur_max - nm
                if imp > best_improve + 1e-15:
                    best_improve = imp
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h[best_i], h[best_j] = 0.0, 1.0
    return h

def perturb_h(h, rng, n_swaps):
    h = h.copy()
    for _ in range(n_swaps):
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        ii = int(oi[rng.integers(len(oi))])
        jj = int(zi[rng.integers(len(zi))])
        h[ii], h[jj] = 0.0, 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    iv = globals().get('initial_h_values', None)
    if iv is not None:
        h_init = np.array(iv, dtype=np.float64)
    else:
        h_init = np.zeros(70, dtype=np.float64)
        h_init[::2] = 1.0

    n = len(h_init)
    dx = 2.0 / n
    h_init = make_binary(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, dx)

    # Diverse initializations
    inits = [h_init.copy()]
    a1 = (np.sqrt(5.0) + 1.0) / 4.0
    a2 = (np.sqrt(5.0) - 1.0) / 4.0
    inits.append(sturmian_init(n, a1))
    inits.append(sturmian_init(n, a2))
    x_arr = np.linspace(0.0, 2.0, n, endpoint=False)
    for freq in [3.7, 6.3, 9.1]:
        inits.append(make_binary(np.sin(freq * x_arr + 0.5) * 0.5 + 0.5, n))
    for r in range(3):
        inits.append(random_binary(n, seed + r * 997))

    # Phase 1: SA on each init (use 40% of budget)
    phase1_end = budget_s * 0.35
    for trial, h_try in enumerate(inits):
        elapsed = time.time() - t0
        if elapsed >= phase1_end:
            break
        t_alloc = min(25.0, phase1_end - elapsed)
        if t_alloc < 3:
            break
        h_res, c5_res = simulated_anneal(h_try, dx, t_alloc, seed + trial * 131)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 2: Greedy refinement (up to 15% more)
    phase2_end = budget_s * 0.5
    elapsed = time.time() - t0
    if elapsed < phase2_end:
        h_res = greedy_refine(best_h, dx, max_time=phase2_end - elapsed)
        c5_res = compute_c5(h_res, dx)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 3: Repeated perturbation + SA + greedy (up to 85%)
    phase3_end = budget_s * 0.85
    rnum = 0
    while time.time() - t0 < phase3_end:
        remaining = phase3_end - (time.time() - t0)
        if remaining < 15:
            break
        rng_p = np.random.default_rng(seed + 50000 + rnum * 17)
        n_pert = 2 + (rnum % 5)
        h_pert = perturb_h(best_h, rng_p, n_pert)
        t_alloc = min(70.0, remaining - 10)
        if t_alloc < 5:
            break
        h_res, c5_res = simulated_anneal(h_pert, dx, t_alloc, seed + 80000 + rnum * 23)
        if c5_res < best_c5:
            best_c5 = c5_res
        h_res = greedy_refine(h_res, dx, max_time=min(20, phase3_end - (time.time() - t0) - 3))
        c5_res = compute_c5(h_res, dx)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
        rnum += 1

    # Phase 4: Final greedy polish
    remaining = budget_s * 0.97 - (time.time() - t0)
    if remaining > 3:
        h_res = greedy_refine(best_h, dx, max_time=remaining)
        c5_res = compute_c5(h_res, dx)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Validate: ensure exactly n//2 ones
    n_out = len(best_h)
    dx_out = 2.0 / n_out
    n_ones = int(np.round(np.sum(best_h)))
    target = n_out // 2
    if n_ones != target:
        if n_ones > target:
            idxs = np.where(best_h == 1)[0]
            for idx in idxs:
                if int(np.sum(best_h)) <= target:
                    break
                best_h[idx] = 0.0
        else:
            idxs = np.where(best_h == 0)[0]
            for idx in idxs:
                if int(np.sum(best_h)) >= target:
                    break
                best_h[idx] = 1.0

    final_c5 = compute_c5(best_h, dx_out)
    return list(best_h), final_c5, n_out
