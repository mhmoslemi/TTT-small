    a, b = max(0, k), min(n, n+k)
    if a < b:
        g[a:b] += 1.0 - h[a-k:b-k]
    a, b = max(0, -k), min(n, n-k)
    if a < b:
        g[a:b] -= h[a+k:b+k]
    return g

def _pgd_step(h, n, target, deadline, lr, mom, mu=0.3, top_k=5):
    """One projected gradient step. Returns (h, c, lr, mom, improved)."""
    ov = _overlaps(h)
    c = float(ov.max())
    ti = np.argpartition(ov, -top_k)[-top_k:]
    g = np.zeros(n)
    for t in ti:
        g += _glag(h, int(t) - (n-1), n)
    dx = 2.0 / n
    g *= dx / top_k
    mom = mu * mom + (1.0 - mu) * g
    h2 = _proj(h - lr * mom, target)
    if time.time() > deadline:
        return h, c, lr, mom, False
    c2 = float(_overlaps(h2).max())
    if c2 < c:
        return h2, c2, min(lr*1.02, 0.5), mom, True
    else:
        return h, c, lr*0.5, mom*0.5, False

import numpy as np
import time

def _ov(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _prj(y, tgt):
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        if np.clip(y - m, 0, 1).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def _gl(h, k, n):
    g = np.zeros(n)
    a1, b1 = max(0, -k), min(n, n - k)
    if a1 < b1:
        g[a1:b1] += 1.0 - h[a1+k:b1+k]
    a2, b2 = max(0, k), min(n, n + k)
    if a2 < b2:
        g[a2:b2] -= h[a2-k:b2-k]
    return g

def _pgd(h, n, tgt, dl, niter, topk):
    dx = 2.0 / n
    lr = 0.08
    mom = np.zeros(n)
    mu = 0.35
    for _ in range(niter):

            break
        ov = _ov(h)

        ti = np.argpartition(ov, -topk)[-topk:]
        g = np.zeros(n)
        for t in ti:
            g += _gl(h, int(t) - (n - 1), n)
        g *= dx / topk
