    k_star = np.argmax(corr)
    shift = k_star - (n - 1)

        if 0 <= idx2 < n:
            grad[j] -= h[idx2]
    return grad * dx

def project_feasible(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

        return h
    # Binary search on lambda
    lo, hi = -2.0, 2.0

def smooth_grad(h, tau):
