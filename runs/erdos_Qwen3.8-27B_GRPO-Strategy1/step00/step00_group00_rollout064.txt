    h = np.clip(h, 0.0, 1.0)

    if abs(diff) < 1e-12:

        mask = h < 1.0
        room = 1.0 - h[mask]

        mask = h > 0.0
        room = h[mask]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def fix_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            tot = np.sum(room)
            if tot < 1e-15:
                break
            h[mask] += diff * room / tot
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            tot = np.sum(room)
            if tot < 1e-15:
                break
            h[mask] += diff * room / tot
        h = np.clip(h, 0.0, 1.0)
    return h

_gs = 0.0

def _cf(x):
    return np.sum(x) - _gs

def _cj(x):
    return np.ones(len(x))

def _slsqp(h0, target, maxiter, ftol):
    global _gs
    _gs = target
    n = len(h0)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return None
    if res.x is not None and np.all(np.isfinite(res.x)):
        return fix_sum(res.x, target)
    return None

def _local_search(h, target, n, max_rounds=200):
    best = fix_sum(h, target)
    best_c5 = compute_c5(best)
    for _ in range(max_rounds):
        improved = False
        for attempt in range(80):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            new_h = best.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_h = fix_sum(new_h, target)
            c5v = compute_c5(new_h)
            if c5v < best_c5 - 1e-14:
                best = new_h
                best_c5 = c5v
                improved = True
                break
        if not improved:
            break
    return best, best_c5
