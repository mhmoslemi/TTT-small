Wait, I have a problem. The `softmax_obj` and `softmax_grad` functions use `project_feasible` internally, but `minimize` will call them with arbitrary x. The projection inside is fine for evaluation, but the gradient should be consistent with the objective. Since we're projecting inside both, it should be approximately consistent (the projection is a "clamp" that's differentiable almost everywhere).

Actually, there's a subtlety: L-BFGS-B will use the gradient to estimate the Hessian and take steps. If the gradient is computed at the projected point but the step is applied to the unprojected point, there could be inconsistencies. However, since we have box constraints [0,1] in L-BFGS-B, the primary "projections" are handled by the solver. The sum constraint is the issue.

Let me add a penalty for the sum constraint in the objective and gradient:

```python
def softmax_obj(x, n, tau):
    ts = n / 2.0
    dx = 2.0 / n
    corrs = np.correlate(x, 1.0 - x, mode="full")
    mx = corrs.max()
    mg = corrs - mx
    # Add penalty for sum constraint
    penalty = 10.0 * (x.sum() - ts) ** 2
    return (mx + tau * np.log(np.sum(np.exp(mg / tau)))) * dx + penalty

def softmax_grad(x, n, tau):
    ts = n / 2.0
    dx = 2.0 / n
    corrs = np.correlate(x, 1.0 - x, mode="full")
    mx = corrs.max()
    mg = corrs - mx
    w = np.exp(mg / tau)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-12:
            g += w[k] * subgrad_c(x, k)
    g *= dx
    g += 20.0 * (x.sum() - ts) / n  # gradient of penalty
    return g
```

This way, L-BFGS-B works with the box constraints [0,1] and the penalty pushes toward the sum constraint. At the end, I project exactly.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def subgrad_c(h, k):
    n = len(h)
    m = k - (n - 1)
    g = np.ones(n)
    for j in range(n):
        jm = j + m
        if 0 <= jm < n:
            g[j] -= h[jm]
        jn = j - m
        if 0 <= jn < n:
            g[j] -= h[jn]
    return g


def project_feasible(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def sgd_opt(h0, n, iters=3000, lr0=0.08, t_end=None):
    ts = n / 2.0
    h = project_feasible(h0.copy(), ts)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    stuck = 0
    for it in range(iters):
        if t_end is not None and time.time() > t_end:
            break
        corrs = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corrs) * (2.0 / n))
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 500:
                if lr < 1e-8:
                    break
                lr *= 0.85
                stuck = 0
        k = int(np.argmax(corrs))
        g = subgrad_c(h, k)
        accepted = False
        for _ in range(12):
            hnew = project_feasible(h - lr * g, ts)
            c5n = compute_c5(hnew)
            if c5n <= c5 * (1.0 + 1e-8) + 1e-12:
                h = hnew
                accepted = True
                break
            lr *= 0.5
        if not accepted and lr < 1e-10:
            break
    return best_h, best_c5


def _softmax_obj(x, n, tau, ts):
    dx = 2.0 / n
    corrs = np.correlate(x, 1.0 - x, mode="full")
    mx = corrs.max()
    mg = corrs - mx
    base = (mx + tau * np.log(np.sum(np.exp(mg / tau)))) * dx
    penalty = 50.0 * (x.sum() - ts) ** 2
    return base + penalty


def _softmax_grad(x, n, tau, ts):
    dx = 2.0 / n
    corrs = np.correlate(x, 1.0 - x, mode="full")
    mx = corrs.max()
    mg = corrs - mx
    w = np.exp(mg / tau)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-12:
            g += w[k] * subgrad_c(x, k)
    g *= dx
    g += 100.0 * (x.sum() - ts) / n
    return g


def softmax_opt(h0, n, t_end=None):
    ts = n / 2.0
    h = project_feasible(h0.copy(), ts)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for tau in [0.2, 0.1, 0.05, 0.03, 0.015]:
        if t_end is not None and time.time() > t_end:
            break
        for _ in range(2):
            if t_end is not None and time.time() > t_end:
                break
            try:
                res = minimize(
                    lambda x: _softmax_obj(x, n, tau, ts),
                    h.copy(),
                    jac=lambda x: _softmax_grad(x, n, tau, ts),
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n,
                    options={'maxiter': 400, 'ftol': 1e-15}
                )
            except Exception:
                break
            h = project_feasible(res.x, ts)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def extend_res(h, no, nn):
    xo = np.arange(no, dtype=float) * (2.0 / no) + 1.0 / no
    xn = np.arange(nn, dtype=float) * (2.0 / nn) + 1.0 / nn
    hn = np.interp(xn, xo, h)
    return project_feasible(hn, nn / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    global initial_h_values
    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    else:
        n0 = 45
        h0 = np.full(n0, 0.5)

    ts = n0 / 2.0
    best_h = project_feasible(h0, ts)
    best_c5 = compute_c5(best_h)

    t1 = t0 + budget_s * 0.22
    t2 = t0 + budget_s * 0.50
    t3 = t0 + budget_s * 0.78
    t4 = t0 + budget_s * 0.95

    # Phase 1: Subgradient descent at base resolution
    h, c5 = sgd_opt(best_h, n0, iters=4000, lr0=0.10, t_end=t1)
    if c5 < best_c5:
        best_h, best_c5 = h, c5

    # Phase 2: Softmax smoothed optimization
    h, c5 = softmax_opt(best_h, n0, t_end=t2)
    if c5 < best_c5:
        best_h, best_c5 = h, c5

    # Phase 3: Random restarts at same resolution
    n_restarts = 0
    while time.time() < t3 and n_restarts < 30:
        n_restarts += 1
        scale = 0.25 / (1.0 + 0.1 * n_restarts)
        noise = np.random.randn(n0) * scale
        hr = project_feasible(best_h + noise, ts)
        h, c5 = sgd_opt(hr, n0, iters=1200, lr0=0.06, t_end=min(time.time() + 15, t3))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Try higher resolution
    for mult in [2, 3]:
        if time.time() > t4:
            break
        nn = int(n0 * mult)
        he = extend_res(best_h, len(best_h), nn)
        h, c5 = sgd_opt(he, nn, iters=2500, lr0=0.07, t_end=t4)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Final polish at best resolution
    if time.time() < t0 + budget_s - 5:
        n_best = len(best_h)
        ts_best = n_best / 2.0
        h, c5 = sgd_opt(best_h, n_best, iters=2000, lr0=0.03, t_end=t0 + budget_s - 3)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = project_feasible(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
