            h[a] = 0.0
            h[b] = 1.0
            c5n = evaluate_c5(h)
            if c5n < c5:
                c5 = c5n
                improved = True
                if c5n < bc5:
                    bc5 = c5n
                    bh = h.copy()
                break
            else:
                h[a] = 1.0
                h[b] = 0.0
        if not improved:
            # Random swap to escape
            a = nz[rng.integers(nz.size)]
            b = zz[rng.integers(zz.size)]
            h[a] = 0.0
            h[b] = 1.0
            c5n = evaluate_c5(h)
            if c5n <= c5 + 0.002:
                c5 = c5n
            else:
                h[a] = 1.0
                h[b] = 0.0
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    N = 500

    # Get initial solution from parent
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        h0 = scale_initial(parent, N, rng)
    except (NameError, Exception):
        h0 = random_binary(N, rng)

    # Phase 1: Main SA from scaled parent (35% budget)
    t1 = t0 + 0.35 * budget_s
    bh, bc5 = sa_optimize(h0, rng, t1, T0=0.012, Tlo=0.001, cool=0.99998)

    # Phase 2: Multi-restart SA (35% budget)
    t2 = t0 + 0.70 * budget_s
    rem = max(t2 - time.time(), 0)
    if rem > 10:
        n_rest = int(min(8, max(3, rem / 55)))
        seg = rem / n_rest
        for r in range(n_rest):
            if time.time() >= t2:
                break
            if r == 0:
                hp = bh.copy()
            else:
                # Perturb best or use fresh random start
                if rng.random() < 0.6:
                    hp = bh.copy()
                    nsw = int(rng.integers(4, 20))
                    for _ in range(nsw):
                        nz = np.nonzero(hp)[0]
                        zz = np.nonzero(1.0 - hp)[0]
                        a = nz[rng.integers(nz.size)]
                        b = zz[rng.integers(zz.size)]
                        hp[a] = 0.0
                        hp[b] = 1.0
                else:
                    hp = random_binary(N, rng)
            te = min(time.time() + seg, t2)
            if te - time.time() > 3:
                hn, cn = sa_optimize(hp, rng, te, T0=0.007, Tlo=0.0008, cool=0.99999)
                if cn < bc5:
                    bc5, bh = cn, hn

    # Phase 3: Targeted refinement (15% budget)
    t3 = t0 + 0.85 * budget_s
    if time.time() < t3:
        bh, bc5 = targeted_refinement(bh, rng, t3)

    # Phase 4: Final SA polish (13% budget)
    t4 = t0 + 0.98 * budget_s
    if time.time() < t4:
        bh, bc5 = sa_optimize(bh, rng, t4, T0=0.003, Tlo=0.0002, cool=0.999995)

    c5_final = evaluate_c5(bh)
    assert abs(np.sum(bh) * 2.0 / N - 1.0) < 1e-6
    assert np.all(bh >= 0) and np.all(bh <= 1)
    return bh, c5_final, N
