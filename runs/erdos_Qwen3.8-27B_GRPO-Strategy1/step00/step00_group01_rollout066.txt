from scipy.optimize import minimize
import time

_current_alpha = 80.0
_current_target = 23.5


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)


def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    e = np.exp(alpha * (c - m))
    s = np.sum(e)
    obj = m + np.log(s) / alpha
    w = e / s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            j1 = np.arange(0, n - k)
            grad[j1] += wk * (1.0 - h[j1 + k])
            j2 = np.arange(k, n)
            grad[j2] -= wk * h[j2 - k]
        else:
            kk = -k
            j1 = np.arange(kk, n)
            grad[j1] += wk * (1.0 - h[j1 - kk])
            j2 = np.arange(0, n - kk)
            grad[j2] -= wk * h[j2 + kk]
    grad *= dx
    return obj, grad
