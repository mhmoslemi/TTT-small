                    best_c5 = c5t
                    best_h = h.copy()
                    break
                h = best_h.copy()
            if (it + 1) % 5000 == 0:
                lr *= 0.9

    # 6. Final polish with simulated-annealing-like perturbations
    if best_h is not None and time_left() > 8:
        n = len(best_h)
        h = best_h.copy()
        T = 0.02
        for it in range(50000):
            if time_left() < 2:
                break
            T *= 0.9998
            i = rng.randint(n)
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + rng.uniform(-0.2, 0.2), 0, 1)
            h2 = repair(h2, n)
            c5t = compute_c5(h2)
            if c5t < best_c5 - 1e-12:
                best_c5 = c5t
                best_h = h2
                h = h2
            elif rng.rand() < np.exp((best_c5 - c5t) / max(T, 1e-6)):
                h = h2

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
