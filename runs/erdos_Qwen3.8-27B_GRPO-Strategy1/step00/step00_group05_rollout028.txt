1. Evaluate f at h_j = 0 and h_j = 1

import numpy as np
from scipy.optimize import minimize
import time

_g_dx = 0.0
_g_beta = 1.0
_g_target = 0.0

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _obj_soft(h):
    corr = np.correlate(h, 1.0 - h, mode="full") * _g_dx
    bc = _g_beta * corr
    bc = bc - bc.max()
    w = np.exp(bc)
    w /= w.sum()
    return float(corr @ w)

def _csum_eq(h):
    return h.sum() - _g_target

def _csum_jac(h):
    return np.ones(len(h))

def _fix_sum(h, target):
    s = h.sum()
    if s < 1e-15:
        return np.full(len(h), target / len(h))
    h = h * (target / s)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()

        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():

    return h

def _opt_slsqp(h0, n, dx, target, betas, max_iter, deadline):

    _g_dx = dx
    _g_target = target

    bounds = [(0.0, 1.0)] * n

        if time.time() > deadline:
            break

            hn = np.clip(r.x, 0.0, 1.0)
            hn = _fix_sum(hn, target)
            c5 = _eval_c5(hn)
            if c5 < bc5:

                bh = hn
        except Exception:

def _coord_descent(h, target, deadline, max_sweeps=80):
    n = len(h)
    c5 = _eval_c5(h)
    grid = np.linspace(0.0, 1.0, 21)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_val = c5
            best_hj = old
            for hj in grid:
                h[j] = hj
                v = _eval_c5(h)
                if v < best_val:
                    best_val = v
                    best_hj = hj
            h[j] = best_hj
            if best_hj != old:
                improved = True
            c5 = best_val
        h = _fix_sum(h, target)
        if not improved:
            break
    return h, _eval_c5(h)

def _make_h(n, target, rng, style):

        h[::2] = 2.0 * target / (n // 2 + 1)
