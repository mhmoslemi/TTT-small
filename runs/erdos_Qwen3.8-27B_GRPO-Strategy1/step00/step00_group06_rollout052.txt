def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    # Rank by value, take top n/2
    k = n // 2
    if n % 2 == 0:
        threshold = np.sort(h)[::-1][k-1] if k > 0 else 0
        b = (h >= threshold).astype(float)
        # Fix count
        count = int(b.sum())
        if count > k:
            # Turn off some 1s (those closest to threshold)
            ones = np.where(b == 1)[0]
            vals = h[ones]
            order = np.argsort(vals)
            for i in range(count - k):
                b[ones[order[i]]] = 0
        elif count < k:
            zeros = np.where(b == 0)[0]
            vals = h[zeros]
            order = np.argsort(vals)[::-1]
            for i in range(k - count):
                b[zeros[order[i]]] = 1
    else:
        # n is odd, n/2 is not integer - use continuous
        # For now, just round and adjust
        b = (h >= 0.5).astype(float)
        count = int(b.sum())
        target = n / 2.0  # e.g., 47.5
        if count > int(np.ceil(target)):
            ones = np.where(b == 1)[0]
            order = np.argsort(h[ones])
            b[ones[order[0]]] = 0.5  # fractional
            # This gets complicated, just use continuous
    return b
