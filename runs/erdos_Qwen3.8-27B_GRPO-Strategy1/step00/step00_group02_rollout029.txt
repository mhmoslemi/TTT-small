import time

def compute_c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    v = 1 - h
    corr = np.correlate(h, v, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, n, beta, lam):
    dx = 2.0 / n
    v = 1 - h
    corr = np.correlate(h, v, mode='full') * dx
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    f = cmax + np.log(np.sum(exps)) / beta
    # penalty for constraint

    f += lam * s * s
    return f

def smooth_grad(h, n, beta, lam):

    v = 1 - h
    corr = np.correlate(h, v, mode='full') * dx
    cmax = np.max(corr)

    weights = exps / np.sum(exps)  # softmax weights over correlation terms

            if attempt == 0:
                h_b = binarize(h_track if len(h_track) == n_bs else interp_to(h_track, n_track, n_bs), n_bs)
            else:
                h_b = np.zeros(n_bs)
                k = n_bs // 2
                h_b[np.random.choice(n_bs, k, replace=False)] = 1.0
            h_bs, c5_bs = local_binary_search(h_b, n_bs, max_iters=200)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()

    # Phase 4: random restarts at moderate resolution
    time_left = budget_s - elapsed()
    if time_left > 20:
        n_rr = min(180, max(n_track, 90))
        if n_rr % 2 == 1:
            n_rr -= 1
        n_restarts = int(min(30, time_left / 10))
        for r in range(n_restarts):
            if elapsed() > budget_s * 0.95:
                break
            h_r = np.zeros(n_rr)
            h_r[np.random.choice(n_rr, n_rr // 2, replace=False)] = 1.0
            # Perturb: swap a few
            for _ in range(3):
                zi = np.where(h_r == 0)[0]
                oi = np.where(h_r == 1)[0]
                if len(zi) > 0 and len(oi) > 0:
                    h_r[np.random.choice(oi)] = 0
                    h_r[np.random.choice(zi)] = 1
            try:
                res = minimize(obj_smooth, h_r, args=(n_rr, 100, 10.0),
                               jac=grad_smooth, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n_rr,
                               options={'maxiter': 100, 'ftol': 1e-12})
                h_r = project_feasible(res.x, n_rr)
            except Exception:
                pass
            c5r = compute_c5(h_r)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h_r.copy()
            h_rb = binarize(h_r, n_rr)
            if compute_c5(h_rb) < best_c5:
                best_c5 = compute_c5(h_rb)
                best_h = h_rb.copy()
            h_rb2, c5_rb2 = local_binary_search(binarize(h_r, n_rr), n_rr, max_iters=100)
            if c5_rb2 < best_c5:
                best_c5 = c5_rb2
                best_h = h_rb2.copy()

    # Final polish: local binary search on best
    n_best = len(best_h)
    if n_best % 2 == 0 and elapsed() < budget_s * 0.98:
        h_final, c5_final = local_binary_search(best_h, n_best, max_iters=300)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
