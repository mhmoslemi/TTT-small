                intensity = 0.03 + 0.15 * (it + 1) / max_bh
                h_p = make_perturbation(bh_h, n, rng, intensity)

            h_p = slsqp_opt(h_p, n, 200, maxiter=200)
            c_p, _ = compute_c5(h_p)

            if c_p < bh_c5 - 1e-10:
                bh_h = h_p.copy()
                bh_c5 = c_p
                if c_p < c5_best - 1e-12:
                    c5_best = c_p
                    h_best = h_p.copy()
                    n_best = n

        h = bh_h

    elapsed = time.time() - t0
    if budget_s - elapsed > 5:
        h_best = slsqp_opt(h_best, n_best, 1000, maxiter=500)
        h_best = slsqp_opt(h_best, n_best, 3000, maxiter=500)
        c5_best, _ = compute_c5(h_best)

    return h_best, c5_best, n_best
