import numpy as np
from scipy.optimize import minimize
import time

_N = 45

def c5_obj(x):
    dx = 2.0 / _N
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

def sum_eq(x):
    return np.sum(x) - _N / 2.0

def project(x, n):
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(x)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = x < 1.0
        else:
            m = x > 0.0
        c = m.sum()
        if c == 0:
            break
        x[m] = np.clip(x[m] + d / c, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.asarray(initial_h_values, dtype=float).copy()
    else:
        init = np.full(45, 0.5)

    n = len(init)
    _N = n
    init = project(init, n)
    best_h = init.copy()
    best_c5 = c5_obj(init)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]

    # Phase 1: SLSQP from initial point
    if (time.time() - t0) < budget_s * 0.05:
        try:
            r = minimize(c5_obj, best_h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 5000, 'ftol': 1e-16})
            if r.x is not None and np.all(np.isfinite(r.x)):
                r.x = project(r.x, n)
                v = c5_obj(r.x)
                if v < best_c5:
                    best_c5 = v
                    best_h = r.x.copy()
        except Exception:
            pass

    # Phase 2: Multi-start SLSQP with varying perturbation scales
    scales = [0.02, 0.05, 0.1, 0.15, 0.25, 0.35, 0.5]
    trials_per_scale = 4
    total_trials = len(scales) * trials_per_scale
    for si, sc in enumerate(scales):
        for tr in range(trials_per_scale):
            if (time.time() - t0) > budget_s * 0.55:
                break
            if si == 0 and tr == 0:
                h0 = best_h.copy()
            else:
                h0 = best_h + np.random.uniform(-sc, sc, n)
                h0 = project(h0, n)
            try:
                r = minimize(c5_obj, h0, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
                if r.x is not None and np.all(np.isfinite(r.x)):
                    r.x = project(r.x, n)
                    v = c5_obj(r.x)
                    if v < best_c5:
                        best_c5 = v
                        best_h = r.x.copy()
            except Exception:
                pass

    # Phase 3: COBYLA (derivative-free, handles non-smoothness better)
    if (time.time() - t0) < budget_s * 0.6:
        try:
            r = minimize(c5_obj, best_h, method='COBYLA', bounds=bounds,
                         constraints=cons, options={'maxiter': 10000, 'rhobeg': 0.05})
            if r.x is not None and np.all(np.isfinite(r.x)):
                r.x = project(r.x, n)
                v = c5_obj(r.x)
                if v < best_c5:
                    best_c5 = v
                    best_h = r.x.copy()
        except Exception:
            pass

    # Phase 4: Simulated annealing with paired coordinate moves
    if (time.time() - t0) < budget_s * 0.65:
        ch = best_h.copy()
        cc = best_c5
        T = 0.003
        for it in range(50000):
            if (time.time() - t0) > budget_s * 0.85:
                break
            i = np.random.randint(n)
            j = (i + 1 + np.random.randint(n - 1)) % n
            if j == i:
                j = (i + 1) % n
            nh = ch.copy()
            di = np.random.normal(0, T)
            dh = T * (1.0 - 2.0 * np.random.random())
            nh[i] += di
            nh[j] += dh
            nh = project(nh, n)
            nv = c5_obj(nh)
            if nv < cc or np.random.random() < np.exp(-(nv - cc) / max(T, 1e-15)):
                ch = nh
                cc = nv
                if nv < best_c5:
                    best_c5 = nv
                    best_h = nh.copy()
            T *= 0.9999
            if T < 1e-6:
                T = 1e-6

    # Phase 5: Short SLSQP from best SA point
    if (time.time() - t0) < budget_s * 0.9:
        try:
            r = minimize(c5_obj, best_h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
            if r.x is not None and np.all(np.isfinite(r.x)):
                r.x = project(r.x, n)
                v = c5_obj(r.x)
                if v < best_c5:
                    best_c5 = v
                    best_h = r.x.copy()
        except Exception:
            pass

    # Phase 6: Final round of small-perturbation SLSQP
    for k in range(10):
        if (time.time() - t0) > budget_s * 0.97:
            break
        h0 = best_h + np.random.uniform(-0.01, 0.01, n)
        h0 = project(h0, n)
        try:
            r = minimize(c5_obj, h0, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 1000, 'ftol': 1e-16})
            if r.x is not None and np.all(np.isfinite(r.x)):
                r.x = project(r.x, n)
                v = c5_obj(r.x)
                if v < best_c5:
                    best_c5 = v
                    best_h = r.x.copy()
        except Exception:
            pass

    best_h = project(best_h, n)
    best_c5 = c5_obj(best_h)
    return best_h, best_c5, n
