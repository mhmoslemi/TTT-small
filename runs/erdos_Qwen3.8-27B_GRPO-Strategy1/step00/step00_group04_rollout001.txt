                a = o[rng.randint(len(o))]
                b = z[rng.randint(len(z))]
                h[a] = 0.0
                h[b] = 1.0
                o = np.where(h == 1)[0]
                z = np.where(h == 0)[0]
            c5_cur = evaluate_c5(h)

        it += 1

    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    has_parent = ('initial_h_values' in globals() and initial_h_values is not None)
    if has_parent:
        h_parent = np.asarray(initial_h_values, dtype=np.float64)
    else:
        h_parent = np.zeros(42, dtype=np.float64)
        h_parent[1::2] = 1
        if int(h_parent.sum()) != 21:
            idx = np.where(h_parent == 0)[0]
            h_parent[idx[:21 - int(h_parent.sum())]] = 1

    best_h = None
    best_c5 = float('inf')

    # First: quick SA at parent resolution
    h_qs = h_parent.copy()
    c5_qs = evaluate_c5(h_qs)
    h_qb, c5_qb = sa_optimize(h_qs, t0 + 30, rng, temp_init=0.003)
    if c5_qb < best_c5:
        best_c5 = c5_qb
        best_h = h_qb

    # Main search at multiple resolutions
    configs = [(168, 2), (200, 2), (280, 2), (350, 1), (500, 1)]

    for n_pts, n_restarts in configs:
        if time.time() - t0 > budget_s - 15:
            break
        remaining = budget_s - 15 - (time.time() - t0)
        per_run = remaining / n_restarts
        per_run = max(per_run, 15)

        for r in range(n_restarts):
            deadline = time.time() + per_run
            if time.time() > deadline:
                break

            if has_parent and rng.random() < 0.6:
                h = scale_parent(h_parent, n_pts, rng)
            elif has_parent:
                h = scale_parent(h_qs, n_pts, rng)
            else:
                h = make_random_binary(n_pts, rng)

            h_local, c5_local = sa_optimize(h, deadline, rng, temp_init=0.005)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local

    if best_h is None:
        n = 200
        best_h = make_random_binary(n, rng)
        best_c5 = evaluate_c5(best_h)

    # Final: refine best solution for remaining time
    if time.time() - t0 < budget_s - 5:
        h_final, c5_final = sa_optimize(best_h.copy(), time.time() + (budget_s - 5 - (time.time() - t0)), rng, temp_init=0.002)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    return (best_h, best_c5, len(best_h))
