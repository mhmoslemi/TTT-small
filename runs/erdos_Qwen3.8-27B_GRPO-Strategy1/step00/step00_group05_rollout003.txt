            best_h = (best_h > 0.5).astype(float)
            cmp_ = int(best_h.sum())
            tgt = n // 2
            if cmp_ > tgt:
                idx = np.where(best_h == 1)[0]
                best_h[idx[rng.permutation(len(idx))[:cmp_ - tgt]]] = 0
            elif cmp_ < tgt:
                idx = np.where(best_h == 0)[0]
                best_h[idx[rng.permutation(len(idx))[:tgt - cmp_]]] = 1

    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
