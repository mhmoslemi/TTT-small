                if nm < current_max - best_improve - 1e-12:
                    best_improve = current_max - nm
                    best_i, best_j = i, j
                    current_max = nm
            h[i] = 1.0
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
    return h


def stochastic_hill_climb(h, n_trials=3000, dl=None):
    n = len(h)
    rng_s = np.random.default_rng(777)
    h = h.copy()
    cur = max_corr(h)
    t_deadline = time.time() + dl if dl else float('inf')
    for _ in range(n_trials):
        if time.time() > t_deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng_s.integers(len(ones))]
        j = zeros[rng_s.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nm = max_corr(h)
        if nm > cur + 1e-12:
            h[i], h[j] = 1.0, 0.0
        else:
            cur = nm
            # Try additional swap from here
            for _ in range(3):
                ones2 = np.where(h == 1)[0]
                zeros2 = np.where(h == 0)[0]
                i2 = ones2[rng_s.integers(len(ones2))]
                j2 = zeros2[rng_s.integers(len(zeros2))]
                h[i2], h[j2] = 0.0, 1.0
                nm2 = max_corr(h)
                if nm2 < cur:
                    cur = nm2
                else:
                    h[i2], h[j2] = 1.0, 0.0
                    break
    return h


def smooth_obj(h, n, beta=10.0, pw=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    smax = m + np.log(np.sum(np.exp(corr - m + 1e-12))) / beta
    pen = pw * (np.sum(h) - 0.5 * n) ** 2
    return smax + pen


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial
    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 70
        h0 = make_binary(n, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Round to binary
    h_bin = (best_h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        for idx in ones:
            h_bin[idx] = 0.0
            if int(h_bin.sum()) == target:
                break
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        for idx in zeros:
            h_bin[idx] = 1.0
            if int(h_bin.sum()) == target:
                break

    # Phase 1: Local search from initial
    h_ls = local_search_binary(h_bin, max_rounds=40)
    c5 = compute_c5(h_ls)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_ls.copy()

    # Stochastic hill climb on best
    h_shc = stochastic_hill_climb(best_h, n_trials=5000, dl=30)
    c5 = compute_c5(h_shc)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_shc.copy()
    h_ls = local_search_binary(best_h, max_rounds=20)
    c5 = compute_c5(h_ls)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_ls.copy()

    # Phase 2: Random restarts
    elapsed = time.time() - t0
    remaining = budget_s * 0.75 - elapsed
    per_start = 0.08
    n_starts = max(50, int(remaining / per_start))

    for r in range(n_starts):
        if time.time() - t0 > budget_s * 0.8:
            break
        h_rand = make_binary(n, rng)
        h_ls = local_search_binary(h_rand, max_rounds=8)
        c5 = compute_c5(h_ls)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_ls.copy()
            # Extended refinement
            t_ref = time.time()
            h_ext = local_search_binary(h_ls, max_rounds=30, dl=5)
            c5e = compute_c5(h_ext)
            if c5e < best_c5:
                best_c5 = c5e
                best_h = h_ext.copy()
            h_shc2 = stochastic_hill_climb(best_h, n_trials=2000, dl=5)
            c5s = compute_c5(h_shc2)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = h_shc2.copy()
            local_search_binary(best_h, max_rounds=10, dl=5)
            c5f = compute_c5(best_h)
            if c5f < best_c5:
                best_c5 = c5f

    # Phase 3: Continuous refinement on best binary
    if time.time() - t0 < budget_s * 0.85:
        bounds_cont = [(0.0, 1.0)] * n
        h_start = best_h + rng.normal(0, 0.01, n)
        h_start = np.clip(h_start, 0, 1)
        res = minimize(
            smooth_obj, h_start, args=(n, 8.0, 20.0),
            method='L-BFGS-B', bounds=bounds_cont,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        h_cont = np.clip(res.x, 0, 1)
        s = np.sum(h_cont)
        if s > 1e-6:
            h_cont = np.clip(h_cont * (n / 2.0) / s, 0, 1)
        c5c = compute_c5(h_cont)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h_cont.copy()

    # Final: verify and return
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    if abs(s - n / 2.0) > 1e-6 and s > 1e-6:
        best_h = best_h * (n / 2.0) / s
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
