def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    """Ensure h sums to n/2 and is in [0,1]"""
    h = np.clip(h, 0.0, 1.0)
    deficit = n / 2.0 - np.sum(h)
    if abs(deficit) > 1e-10:
        # Distribute deficit
        if deficit > 0:
            for i in range(n):
                room = 1.0 - h[i]
                add = min(room, deficit)
                h[i] += add
                deficit -= add
                if deficit < 1e-10:

        else:

                sub = min(room, deficit)
                h[i] -= sub
                deficit -= sub

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    deficit = n / 2.0 - np.sum(h)
    if abs(deficit) > 1e-10:
        if deficit > 0:
            for i in range(n):
                room = 1.0 - h[i]
                add = min(room, deficit)
                if add > 0:
                    h[i] += add
                    deficit -= add
                    if deficit <= 1e-10:
                        break
        else:
            deficit = -deficit
            for i in range(n):
                room = h[i]
                sub = min(room, deficit)
                if sub > 0:
                    h[i] -= sub
                    deficit -= sub
                    if deficit <= 1e-10:
                        break
    return h
