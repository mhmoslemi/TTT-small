from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_c5(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    v = corr * alpha
    v_max = np.max(v)
    return float((v_max + np.log(np.sum(np.exp(v - v_max)))) / alpha)

def repair_sum(h, target):
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final precise fix: adjust one element
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    def objective(x):
        return smooth_max_c5(x, dx, alpha)
    def sum_eq(x):
        return np.sum(x) - S
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq}]
    res = minimize(objective, h, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x, res.fun

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_c5(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    v = corr * alpha
    v_max = np.max(v)
    return float((v_max + np.log(np.sum(np.exp(v - v_max)))) / alpha)

def repair_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, n, dx, S, max_iter=1000, alpha=80):
    def objective(x):
        return smooth_max_c5(x, dx, alpha)
    def sum_eq(x):
        return np.sum(x) - S
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq}]
    res = minimize(objective, h, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x, res.fun

def pair_descent(h, n, dx, S, max_rounds=10):

    best_c5 = compute_c5(h, dx)

                        best_h[j] += d
                        c5 = compute_c5(best_h, dx)
                        if c5 < best_c5 - 1e-12:
                            best_c5 = c5
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[j] >= d and best_h[i] + d <= 1.0:
                        best_h[j] -= d

                        c5 = compute_c5(best_h, dx)

                            best_c5 = c5
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
        if not improved:

    return best_h, best_c5
