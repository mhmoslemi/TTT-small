import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                h[:] = target_sum / len(h)
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                h[:] = target_sum / len(h)
                break
            h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if abs(diff) < 1e-14:
                break
            step = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += step * (1.0 if diff > 0 else -1.0)
            diff = target_sum - h.sum()
    return h

def objective_smooth(h, target_sum, lam):
    h2 = project_h(h, target_sum)
    return compute_score(h2) + lam * (h2.sum() - target_sum) ** 2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(47, 0.5)

    n = len(h0)
    target = n / 2.0
    h = project_h(h0, target)

    best_h = h.copy()
    best_score = compute_score(h)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Coordinate descent with grid search
    grid_size = 40
    grid = np.linspace(0.0, 1.0, grid_size)
    max_sweeps = 300
    for sweep in range(max_sweeps):
        if remaining() < 180:
            break
        improved = False
        for j in range(n):
            if remaining() < 180:
                break
            old = h[j]
            best_x = old
            best_s = np.inf
            for x in grid:
                h[j] = x
                s = compute_score(h)
                if s < best_s:
                    best_s = s
                    best_x = x
            h[j] = old
            if best_x != old:
                h[j] = best_x
                improved = True
        h = project_h(h, target)
        sc = compute_score(h)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h.copy()
        if not improved and sweep > 20:
            # Perturb to escape local opt
            if remaining() > 200:
                h = h + rng.randn(n) * 0.05
                h = project_h(h, target)

    # Phase 2: Local refinement with scipy Powell
    if remaining() > 120:
        def obj_powell(x):
            return compute_score(project_h(x, target))
        res = minimize(obj_powell, best_h.copy(), method='Powell',
                       options={'maxiter': 1000, 'xtol': 1e-12, 'ftol': 1e-14})
        if res.fun < best_score:
            best_score = res.fun
            best_h = project_h(res.x, target)

    # Phase 3: Nelder-Mead refinement
    if remaining() > 60:
        def obj_nm(x):
            return compute_score(project_h(x, target))
        res = minimize(obj_nm, best_h.copy(), method='Nelder-Mead',
                       options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14,
                                'initial_simplex_size': 0.1})
        if res.fun < best_score:
            best_score = res.fun
            best_h = project_h(res.x, target)

    # Phase 4: Try binary-like solutions
    if remaining() > 30:
        for trial in range(20):
            if remaining() < 30:
                break
            # Start from best, push towards binary
            h_bin = (best_h > 0.5).astype(float)
            # Adjust to get sum right
            diff = target - h_bin.sum()
            if diff > 0:
                order = np.argsort(best_h)
                for idx in reversed(order[:30]):
                    if h_bin[idx] < 1.0:
                        h_bin[idx] += min(diff, 1.0 - h_bin[idx])
                        diff -= min(diff, 1.0 - h_bin[idx])
                        if abs(diff) < 1e-10:
                            break
            elif diff < 0:
                order = np.argsort(best_h)
                for idx in reversed(order[-30:]):
                    if h_bin[idx] > 0.0:
                        h_bin[idx] += max(diff, -h_bin[idx])
                        diff -= max(diff, -h_bin[idx])
                        if abs(diff) < 1e-10:
                            break
            h_bin = project_h(h_bin, target)
            sc = compute_score(h_bin)
            if sc < best_score:
                best_score = sc
                best_h = h_bin.copy()
            # Perturb the binary and refine
            h2 = h_bin + rng.randn(n) * 0.02
            h2 = project_h(h2, target)
            if remaining() > 40:
                res = minimize(obj_nm, h2, method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-13})
                if res.fun < best_score:
                    best_score = res.fun
                    best_h = project_h(res.x, target)

    # Phase 5: Random restarts with short optimization
    if remaining() > 60:
        for restart in range(10):
            if remaining() < 60:
                break
            h_r = best_h + rng.randn(n) * 0.3
            h_r = project_h(h_r, target)
            res = minimize(obj_nm, h_r, method='Nelder-Mead',
                           options={'maxiter': 400, 'xatol': 1e-10, 'fatol': 1e-13})
            if res.fun < best_score:
                best_score = res.fun
                best_h = project_h(res.x, target)

    # Ensure feasibility
    best_h = project_h(best_h, target)
    final_score = compute_score(best_h)
    best_score = min(best_score, final_score)

    return best_h, best_score, n
