def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    if T < 1e-10:
        return mc
    s = (corr - mc) / T
    return mc + T * np.log(np.sum(np.exp(s)))

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            if m.sum() == 0:
                break
            h[m] += d / m.sum()
        else:
            m = h > 0.0
            if m.sum() == 0:
                break
            h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)
