import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    grad = np.zeros(n)
    offset = k_star - (n - 1)
    for p in range(n):
        i1 = p + offset
        if 0 <= i1 < n:
            grad[p] += (1.0 - h[i1])
        i2 = p - offset
        if 0 <= i2 < n:
            grad[p] -= h[i2]
    return dx * grad

def project_to_feasible(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    # Project onto {0 <= h <= 1, sum(h) = target_sum}
    # Use binary search on lambda
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target_sum - np.sum(projected)
    # Distribute among non-bound elements
    mask = (projected > 1e-10) & (projected < 1.0 - 1e-10)
    if np.sum(mask) > 0:
        projected[mask] += diff / np.sum(mask)
        projected = np.clip(projected, 0.0, 1.0)
    return projected

def projected_gradient_step(h, lr, target_sum=None):
    grad = compute_subgradient(h)
    h_new = h - lr * grad
    h_new = project_to_feasible(h_new, target_sum)
    return h_new, compute_c5(h_new)

def optimize_h(h0, max_iters=5000, lr_init=0.05, lr_min=1e-6, time_limit=None, start_time=None):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    lr = lr_init
    target_sum = n / 2.0
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        grad = compute_subgradient(h)
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            # Slightly increase lr if improving
            lr = min(lr * 1.01, lr_init * 2)
        else:
            lr *= 0.85
        
        # Check convergence
        if lr < lr_min:
            break
        if it > 100 and abs(c5 - best_c5) < 1e-10:
            # Stuck, try largest update
            break
    
    return best_h, best_c5

def random_feasible_h(n, seed=None):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, n)
    h = project_to_feasible(h, n / 2.0)
    return h

def perturb_h(h, sigma=0.05, seed=None):
    rng = np.random.RandomState(seed)
    h_pert = h + rng.normal(0, sigma, len(h))
    return project_to_feasible(h_pert, len(h) / 2.0)

def upscale(h):
    """Upscale by factor 2 (duplicate each sample)"""
    return np.repeat(h, 2)

    for k in active:
        offset = k - (n - 1)
        g = np.zeros(n)
        if offset >= 0:
            g[:n-offset] += (1.0 - h[offset:n])
            g[offset:] -= h[:n-offset]
        else:
            g[-offset:] += (1.0 - h[:n+offset])
            g[:n+offset] -= h[-offset:]
        grad += g
    grad /= len(active)
    return dx * grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _single_subgrad(h, n, k):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        g[:n - offset] += (1.0 - h[offset:])
        g[offset:] -= h[:n - offset]
    else:
        g[-offset:] += (1.0 - h[:n + offset])
        g[:n + offset] -= h[-offset:]
    return g

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    return dx * _single_subgrad(h, n, k_star)

def compute_avg_subgradient(h, rel_eps=0.02):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - rel_eps * max_val)[0]
    grad = np.zeros(n)
    for k in active:
        grad += _single_subgrad(h, n, int(k))
    grad /= len(active)
    return dx * grad

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(30):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 5e-5) & (h < 1.0 - 5e-5)
        cnt = np.sum(free)
        if cnt == 0:
            break
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_pd(h0, max_time=200.0, start_time=None, use_avg=True):
    if start_time is None:
        start_time = time.time()
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.8
    no_improve = 0
    patience = 800

    while time.time() - start_time < max_time:
        if use_avg:
            grad = compute_avg_subgradient(h, 0.015)
        else:
            grad = compute_subgradient(h)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = lr
        c5_curr = compute_c5(h)
        improved = False
        for _ in range(35):
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr - 1e-16:
                h = h_new
                improved = True

                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                lr = min(lr * 1.03, 8.0)
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.92
            if no_improve > patience:
                break
    return best_h, best_c5

def optimize_smooth(h0, max_time=150.0, start_time=None, tau=60.0):
    """Fine-tune with log-sum-exp smoothing."""
    if start_time is None:
        start_time = time.time()
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0, target)
    best_h = h.copy()

def upscale(h, factor=2):
    return np.repeat(h, factor)

def random_feasible(n, rng):
