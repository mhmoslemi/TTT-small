import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute C5 value."""
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    return float(np.max(c) * dx)

def smooth_max(c, beta):
    """Smooth approximation of max using log-sum-exp."""
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def smooth_max_grad(c, beta):
    """Gradient (weights) of smooth max."""
    m = np.max(c)
    shifted = beta * (c - m)
    e = np.exp(shifted)
    return e / np.sum(e)

def grad_of_c(h, w, dx, n):
    """Compute gradient of c * dx w.r.t. h, given weights w on each lag."""
    # w[k] for k = 0, ..., 2n-2, lag = k - (n-1)
    grad = np.zeros(n)
    i_arr = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        i1 = i_arr + lag
        i2 = i_arr - lag
        v1 = (i1 >= 0) & (i1 < n)
        v2 = (i2 >= 0) & (i2 < n)
        if np.any(v1):
            grad[v1] += wk * (1.0 - h[i1[v1]])
        if np.any(v2):
            grad[v2] -= wk * h[i2[v2]]
    return grad

def project_and_scale(h, target_sum):
    """Project h to [0,1]^n and adjust sum to target_sum."""
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def optimize_h(h0, n_points, dx, max_iters=200, budget_fn=None):
    """Optimize h using projected gradient descent on smooth max."""
    target_sum = n_points / 2.0
    h = project_and_scale(h0.copy(), target_sum)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    
    beta = 5.0

        c = np.correlate(h, g, mode='full') * dx
        obj_val = smooth_max(c, beta)
        w = smooth_max_grad(c, beta)
        grad_h = grad_of_c(h, w, dx, n_points)

        h = h - lr * grad_h
        h = project_and_scale(h, target_sum)

        # Check convergence
        if it > 0 and abs(obj_val - best_val) < 1e-8:

def compute_c5(h, dx):

    x_new = np.arange(n) * 2.0 / n

    x_old_ext = np.concatenate([[x_old[0]-2.0/len(parent)], x_old, [x_old[-1]+2.0/n]])

        h = np.clip(h, 0.0, 1.0)

    if len(parent) == n:
