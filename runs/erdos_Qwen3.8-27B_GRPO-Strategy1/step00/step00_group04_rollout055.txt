    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_and_weights(vals, temp):
    m = np.max(vals)
    if temp < 1e-12:
        obj = m
        w = np.zeros(len(vals))
        w[np.argmax(vals)] = 1.0
        return obj, w
    z = (vals - m) / temp
    e = np.exp(z)
    s = np.sum(e)
    obj = m + temp * np.log(s)
    w = e / s
    return obj, w

    for m in range(2 * n - 1):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def smooth_max_weights(vals, temp):
    m = np.max(vals)
    if temp < 1e-12:
        w = np.zeros(len(vals))
        w[np.argmax(vals)] = 1.0
        return m, w
    z = (vals - m) / temp
    e = np.exp(z)
    s = e.sum()
    return m + temp * np.log(s), e / s

def grad_obj(h, temp):
    n = len(h)
    corr_s = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    _, w = smooth_max_weights(corr_s, temp)
    grad = np.zeros(n)
    dx = 2.0 / n
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        s = m - (n - 1)
        lo1, hi1 = max(0, s), min(n - 1, n - 1 + s)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wm * (1.0 - h[lo1 - s:hi1 - s + 1])
        lo2, hi2 = max(0, -s), min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= wm * h[lo2 + s:hi2 + s + 1]
    return dx * grad

def project_feasible(h, target):
    def f_alpha(alpha):
        return np.sum(np.clip(h - alpha, 0, 1)) - target
    lo, hi = -1.5, 1.5
    if f_alpha(lo) < 0 or f_alpha(hi) > 0:
        return np.clip(h, 0, 1)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if f_alpha(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def binary_local_search(h, target, max_time_fn):
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0)
    while h.sum() < target:
        idx = np.where(h < 1.0)[0]
        h[idx[np.argmin(h[idx])]] = 1.0
    while h.sum() > target:
        idx = np.where(h > 0.0)[0]
        h[idx[np.argmax(h[idx])]] = 0.0
    best = h.copy()
    best_c5 = compute_c5(best)
    improved = True
    while improved and max_time_fn() > 2:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        order = np.random.permutation(len(ones) * len(zeros))
        for o in order:
            i, j = ones[o % len(ones)], zeros[o // len(ones)]
            h[i], h[j] = 0.0, 1.0
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best = h.copy()
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    return best, best_c5

def proj_grad_descend(h, target, temps, max_iters, lr_init, max_time_fn):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for temp in temps:
        if max_time_fn() < 5:
            break
        h_cur = best_h.copy()
        for it in range(max_iters):
            if max_time_fn() < 3:
                break
            g = grad_obj(h_cur, temp)
            h_new = project_feasible(h_cur - lr * g, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
            h_cur = h_new
            if np.max(np.abs(g)) < 1e-9:
                break
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    from scipy.interpolate import interp1d
    f = interp1d(x_old, h, kind='linear', bounds_error=False, fill_value=(h[0], h[-1]))
    h_new = f(x_new)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0

    if time_left() > 30:

        if c5b < best_c5:

    # Phase 2: Projected gradient at original resolution
    if time_left() > 20:

        hg, c5g = proj_grad_descend(best_h.copy(), target0, temps, 3000, 0.08, time_left)
        if c5g < best_c5:

    for n_new in [84, 168, 200, 300]:

        h_up = upsample(best_h, n_new)
        target_new = n_new / 2.0

    for trial in range(20):
