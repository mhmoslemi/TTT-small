def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def local_search_binary(h, rng, max_iters=100000, patience=2000):
    """Local search by swapping a 1 and a 0."""
    n = len(h)
    h = h.copy().astype(float)
    # Make it binary
    # Sort by value, top n/2 = 1, rest = 0
    idx = np.argsort(-h)
    h[:] = 0.0
    h[idx[:n//2]] = 1.0
    
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    stale = 0
    for it in range(max_iters):
        # Pick random 1 and random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
        
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            stale = 0
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            stale += 1
            if stale > patience:
                break
    
    return h, best_c5

        h_new[zeros[order[:target - s]]] = 1
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(44)
        h0[::2] = 1.0

    best_h = _make_binary_from_continuous(h0, 44, rng) if len(h0) >= 22 else h0.copy()
    best_c5 = compute_c5(best_h)

    sizes = [44, 46, 50, 54, 56, 60, 64, 70, 72, 80, 88, 100, 108, 120, 140, 160, 200]

    for n in sizes:
        if rem() < 150:
            break
        # Estimate time per SA run
        est = max(2.0, n * 40000 / 8e6)
        n_trials = max(2, min(10, int(rem() * 0.7 / est / 3)))

        for trial in range(n_trials):
            if rem() < 40:
                break
            # Initial configuration
            if trial == 0:
                h = _make_binary_from_continuous(h0, n, rng)
            elif trial == 1:
                h = np.zeros(n)
                h[::2] = 1.0
                for _ in range(n // 4):
                    a = np.flatnonzero(h)[rng.randint(n // 2)]
                    b = np.flatnonzero(1 - h)[rng.randint(n // 2)]
                    h[a], h[b] = h[b], h[a]
            else:
                h = np.zeros(n)
                perm = rng.permutation(n)
                h[perm[:n // 2]] = 1.0

            n_iters = max(15000, int(rem() / (n_trials * 2) * 8000))
            n_iters = min(n_iters, 300000)
            T0 = 0.02 + 0.01 * (1.0 - n / 200.0)
            cool = 1.0 - 1.0 / (n_iters * 1.5)

            h, c5 = _sa(h, n_iters, T0, cool, rng)
            h, c5 = _hill_climb(h, 200, rng)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Try embedding best solution into this size and re-optimize
        if n > 44 and rem() > 60:
            h_emb = _embed(best_h, n, rng)
            n_iters = max(10000, int(rem() * 0.3 * 5000))
            n_iters = min(n_iters, 150000)
            T0 = 0.01
            cool = 1.0 - 1.0 / (n_iters * 1.2)
            h_emb, c5 = _sa(h_emb, n_iters, T0, cool, rng)
            h_emb, c5 = _hill_climb(h_emb, 200, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_emb.copy()

    # Final polish on best
    if rem() > 20:
        n_best = len(best_h)
        n_iters = max(5000, int(rem() * 1500))
        cool = 1.0 - 1.0 / (n_iters * 1.0)
        best_h, best_c5 = _sa(best_h, n_iters, 0.005, cool, rng)
        best_h, best_c5 = _hill_climb(best_h, 500, rng)

    return best_h, best_c5, len(best_h)
