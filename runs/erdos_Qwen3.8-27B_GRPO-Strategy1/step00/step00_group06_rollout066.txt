But I can do better: incremental update. When I swap h[i] and h[j] (say h[i]: 1→0, h[j]: 0→1):

c(k) changes for lags where i or j is involved.
Specifically, c(k) = sum_n h[n+k]*(1-h[n]).

The change in c(k) when h[i] goes 1→0 and h[j] goes 0→1:
- Effect of h[i]: 1→0
  - As h[n+k] where n+k=i, n=i-k: term changes from 1*(1-h[i-k]) to 0. Δ = -(1-h[i-k]) = h[i-k]-1 (if i-k valid)
  - As (1-h[n]) where n=i: term changes from h[i+k]*(1-1)=0 to h[i+k]*(1-0)=h[i+k]. Δ = h[i+k] (if i+k valid)
  
- Effect of h[j]: 0→1
  - As h[n+k] where n+k=j, n=j-k: term changes from 0 to 1*(1-h[j-k]) = 1-h[j-k]. Δ = 1-h[j-k] (if j-k valid)
  - As (1-h[n]) where n=j: term changes from h[j+k]*1 to h[j+k]*0. Δ = -h[j+k] (if j+k valid)

So Δc(k) = [h[i-k]-1 if 0≤i-k<n else 0] + [h[i+k] if 0≤i+k<n else 0]

But this is per lag k, and k ranges from -(n-1) to n-1. So we'd need to update O(n) lags. For n=100, that's 100 updates per move, each O(1). Total O(n) per move.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def random_binary(n, rng):
    h = np.zeros(n, dtype=np.float64)
    if n % 2 == 0:
        pos = rng.permutation(n)[:n//2]
        h[pos] = 1.0
    else:
        pos = rng.permutation(n)[:n//2]
        h[pos] = 1.0
        extra = rng.choice(n)
        h[extra] += 0.5
        h[extra] = 1.0  # make it binary, adjust elsewhere
        # Actually for odd n, sum = n/2 = x.5
        # With n//2 ones and one 0.5: sum = n//2 + 0.5 = (n-1)/2 + 0.5 = n/2 ✓
        # But we set n//2 positions to 1 and one to 0.5, total = n//2 + 0.5 ✓
        # Need to unset one of the 1s
        h[pos[0]] = 0.0
        h[extra] = 0.5
    return h

def anneal(h, n, rng, t_limit):
    """Simulated annealing until time limit. Returns (best_h, best_c5)."""
    curr_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = curr_c5
    
    T = 0.05
    T_min = 0.0005
    # Estimate: ~10000 iterations expected
    cooling = (T_min / T) ** (1.0 / 10000)
    
    no_improve = 0
    
    while time.time() < t_limit:
        # Get indices
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, n)
        delta = new_c5 - curr_c5
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        
        T *= cooling
        if T < T_min:
            T = T_min
            # Restart with slight perturbation
            if no_improve > 5000:
                # Kick
                for _ in range(5):
                    a, b = rng.randint(n), rng.randint(n)
                    if h[a] != h[b]:
                        h[a], h[b] = h[b], h[a]
                curr_c5 = compute_c5(h, n)
                T = 0.02
                no_improve = 0
    
    # Final local improvement
    improved = True
    while improved:
        improved = False
        best_delta = 0
        best_pair = None
        for i in range(n):
            for j in range(i+1, n):
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
                    c = compute_c5(h, n)
                    delta = c - curr_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i, j)
                    h[i], h[j] = h[j], h[i]
        if best_pair is not None and best_delta < -1e-10:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            curr_c5 += best_delta
            best_h = h.copy()
            best_c5 = curr_c5
            improved = True
```

```python
    bc = curr
    T = 0.04
    Tmin = 0.001

                stall = 0

            T = 0.03
            stall = 0

    c = compute_c5(h, n)

            h[i], h[j] = h[j], h[i]
```
