import numpy as np
from scipy.optimize import minimize
import time

_n_active = 45
_temp_active = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _smooth_obj(x):
    temp = _temp_active
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    log_s = np.log(np.sum(np.exp(temp * (corr - m))))
    return (m + log_s / temp) * dx


def _obj_direct(x):
    return compute_c5(x)


def _constr_eq(x):
    return np.sum(x) - _n_active / 2.0


def project_feasible(h, n):
    target = n / 2.0
    for _ in range(80):
        lo, hi = -2.0, 2.0
        for _ in range(70):
            mid = (lo + hi) * 0.5
            hp = np.clip(h + mid, 0.0, 1.0)
            if np.sum(hp) < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
        if abs(np.sum(h) - target) < 1e-15:
            break
    return h


def binary_local_search(h_init, n, max_iters=500):
    half = n // 2
    idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[idx[:half]] = 1.0
    current = compute_c5(h)
    for _ in range(max_iters):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                s = compute_c5(h)
                if s < current - 1e-16:
                    current = s
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, current

    global _n_active, _temp_active

    score_best = compute_c5(h_best)

    n_list = sorted(set([n_best, 44, 46, 48, 60, 72, 90]))

    for n in n_list:

            break
        _n_active = n
        bounds = [(0.0, 1.0)] * n

        if n == n_best:

            xn = np.linspace(0.0, 2.0, n)
            h0 = np.interp(xn, xo, h_best)
        h0 = project_feasible(h0, n)

        # SLSQP with different temperatures
        for temp in [150.0, 300.0]:
            _temp_active = temp
            try:

                              bounds=bounds,
