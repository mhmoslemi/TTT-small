import numpy as np
import time
from scipy.optimize import minimize

_NS = [47]
_BETA = [50.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    smax = m + np.log(np.sum(np.exp(corr - m))) / _BETA[0]
    return smax * (2.0 / _NS[0])

def sum_con_fun(h):
    return h.sum() - _NS[0] * 0.5

def sum_con_jac(h):
    return np.ones(_NS[0])

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = n * 0.5
    d = target - h.sum()
    if abs(d) > 1e-14:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = m.sum()
        if cnt > 0:
            h[m] += d / cnt
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h += d / n
            np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_solve(h0, n, beta, maxiter):
    _NS[0] = n
    _BETA[0] = beta
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    bnds = [(0.0, 1.0)] * n
    r = minimize(smooth_obj, h0, method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    if r.x is not None:
        return proj(r.x, n)
    return h0

    t0 = time.time()
    rng = np.random.default_rng(seed)

    iv = globals().get('initial_h_values', None)
    if iv is not None:
        iv = np.array(iv, dtype=float)

            n = len(iv)
            h0 = proj(iv, n)

        if c5n < bc5 - 1e-15:
