import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def grad_corr_k(h, k):
    n = len(h)
    ak = abs(k)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
    else:
        if ak < n:
            grad[:n - ak] += (1.0 - h[ak:])
            grad[ak:] -= h[:n - ak]
    return grad


def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
