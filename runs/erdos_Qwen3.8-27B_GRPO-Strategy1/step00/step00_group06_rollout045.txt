    d = target - np.sum(h)
    if abs(d) < 1e-12:
        return h
    mask = (h > 0.005) & (h < 0.995)
    if np.sum(mask) > 0:
        h = h.copy()
        h[mask] += d / np.sum(mask)
    else:
        h = h.copy()
        h += d / n
    return np.clip(h, 0.0, 1.0)

def to_logit(h):
    h = np.clip(h, 1e-5, 1.0 - 1e-5)
    return np.log(h / (1.0 - h))

def from_logit(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -40, 40)))

def binary_push(h, n):
    target = n / 2.0
    nb = (h > 0.5).astype(float)
    s = np.sum(nb)
    if s > target:
        idx = np.where(nb == 1)[0]
        order = np.argsort(h[idx])[::-1]
        for i in order:
            if s > target:
                nb[idx[i]] = 0.0
                s -= 1
    elif s < target:
        idx = np.where(nb == 0)[0]
        order = np.argsort(h[idx])[::-1]
        for i in order:
            if s < target:
                nb[idx[i]] = 1.0
                s += 1
    diff = target - np.sum(nb)
    if abs(diff) > 1e-10:
        candidates = np.where((nb > 0) & (nb < 1))[0]
        if len(candidates) == 0:
            candidates = np.where(nb < 1)[0]
        if len(candidates) > 0:
            best_i = candidates[0]
            if diff > 0:
                nb[best_i] = min(1.0, nb[best_i] + diff)
            else:
                nb[best_i] = max(0.0, nb[best_i] + diff)
    return nb

def coord_descent(h, n, max_sweeps=5):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for trial_val in [0.0, 1.0]:
                h_t = best.copy()
                h_t[i] = trial_val
                h_t = fix_sum(h_t, n)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5 - 1e-12:
                    best = h_t
                    best_c5 = c5_t
                    improved = True
        if not improved:
            break
    return best, best_c5

def gradient_step(h, n, alpha, lr, steps):
    x = to_logit(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for s in range(steps):
        grad = np.zeros(n)
        eps = 1e-5
        for i in range(n):
            xp = x.copy(); xp[i] += eps
            xm = x.copy(); xm[i] -= eps
            fp = obj_func(xp, n, alpha, n / 2.0)
            fm = obj_func(xm, n, alpha, n / 2.0)
            grad[i] = (fp - fm) / (2 * eps)
        x -= lr * grad
        h_new = from_logit(x)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new
            best_c5 = c5_new
        else:
            lr *= 0.8
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is None or len(h0) == 0:
        np.random.seed(seed)
        h0 = (np.random.rand(95) < 0.5).astype(float)
        h0 = fix_sum(h0, 95)

    n = len(h0)
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    print(f"Start: n={n}, C5={best_c5:.6f}")

    def try_update(h_new):
        nonlocal best_h, best_c5
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            return True
        return False

    # Phase 1: L-BFGS-B with increasing alpha
    x0 = to_logit(h0)
    alphas = [10, 30, 100, 300, 1000, 3000]
    for alpha in alphas:
        if time.time() - t0 > budget_s * 0.45:
            break
        for restart in range(2):
            if time.time() - t0 > budget_s * 0.45:
                break
            try:
                res = minimize(obj_func, x0, args=(n, alpha, target_sum),
                               method='L-BFGS-B',
                               options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
                h_new = from_logit(res.x)
                if try_update(h_new):
                    x0 = to_logit(best_h)
            except Exception:
                pass
        if time.time() - t0 > budget_s * 0.45:
            break

    # Phase 2: Binary push + re-optimize
    if time.time() - t0 < budget_s * 0.75:
        hb = binary_push(best_h, n)
        if try_update(hb):
            x0 = to_logit(best_h)
        for alpha in [100, 500, 2000]:
            if time.time() - t0 > budget_s * 0.75:
                break
            try:
                res = minimize(obj_func, x0, args=(n, alpha, target_sum),
                               method='L-BFGS-B',
                               options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10})
                h_new = from_logit(res.x)
                if try_update(h_new):
                    x0 = to_logit(best_h)
            except Exception:
                pass

    # Phase 3: Coordinate descent
    if time.time() - t0 < budget_s * 0.85:
        best_h, best_c5 = coord_descent(best_h, n, max_sweeps=8)
        print(f"After coord descent: C5={best_c5:.6f}")

    # Phase 4: Random perturbations + gradient
    if time.time() - t0 < budget_s * 0.95:
        for trial in range(300):
            if time.time() - t0 > budget_s * 0.95:
                break
            sigma = 0.005 + 0.02 * np.random.random()
            h_p = best_h + np.random.randn(n) * sigma
            try_update(h_p)
        # Final polish
        if time.time() - t0 < budget_s * 0.97:
            x0 = to_logit(best_h)
            try:
                res = minimize(obj_func, x0, args=(n, 5000, target_sum),
                               method='L-BFGS-B',
                               options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12})
                h_new = from_logit(res.x)
                try_update(h_new)
            except Exception:
                pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, n)
    final_c5 = compute_c5(best_h)
    s = np.sum(best_h)
    if abs(s - target_sum) > 1e-6:
        best_h += (target_sum - s) / n
        best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = compute_c5(best_h)

    print(f"Final: n={n}, C5={final_c5:.6f}, sum={np.sum(best_h):.8f}")
    return best_h, final_c5, n
