    best_val = min(np.max(base), np.max(base + delta))

        return h
    if diff > 0:

            ts = np.zeros((m, m))
            with np.errstate(divide='ignore', invalid='ignore'):
                np.divide(B_mat, D_mat, out=ts, where=mask)
            valid_t = ts[mask]
            valid_t = valid_t[np.isfinite(valid_t)]
            valid_t = valid_t[(valid_t >= 0.0) & (valid_t <= 1.0)]
            if len(valid_t) > 0:
                vals = np.max(base[None, :] + delta[None, :] * valid_t[:, None], axis=1)
                idx = int(np.argmin(vals))
                if vals[idx] < best_val:
                    best_val = float(vals[idx])
                    best_t = float(valid_t[idx])
    return best_t, best_val

def coord_descent_pass(h, dx, S):
    n = len(h)
    h = h.copy()
    total_improve = 0.0
    for i in range(n):
        old_val = h[i]
        h[i] = 0.0
        base = np.correlate(h, 1.0 - h, mode='full') * dx
        h[i] = 1.0
        full = np.correlate(h, 1.0 - h, mode='full') * dx
        delta = full - base
        best_t, _ = solve_1d_minimax(base, delta)
        h[i] = best_t
        total_improve += abs(best_t - old_val)
    h = project_to_sum(h, S)
    return h, total_improve

def run_coord_descent(h, dx, S, max_passes=200):
    h = h.copy()
    for p in range(max_passes):
        h, improvement = coord_descent_pass(h, dx, S)
        if improvement < 1e-11:
            break
    return h

def local_search_swap(h, dx, S, rng, max_iters=10000):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    deltas = np.array([0.1, 0.05, 0.02, 0.01, 0.005])
    for it in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        found = False
        for d in deltas:
            ni = h[i] + d
            nj = h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    found = True
                else:
                    h[i], h[j] = ni - d, nj + d
                if found:
                    break
            nj = h[j] + d
            ni = h[i] - d
            if 0.0 <= nj <= 1.0 and 0.0 <= ni <= 1.0:
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    found = True
                else:
                    h[i], h[j] = ni + d, nj - d
                if found:
                    break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    S = n / 2.0
    h0 = project_to_sum(h0, S)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent from initial
    if time_left() > 20:
        h = run_coord_descent(best_h, dx, S, max_passes=300)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Local search
    if time_left() > 30:
        h, c5 = local_search_swap(best_h, dx, S, rng, max_iters=30000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h = run_coord_descent(h, dx, S, max_passes=100)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Multi-start perturbations
    for restart in range(40):
        if time_left() < 20:
            break
        sigma = 0.03 + 0.04 * (1.0 - restart / 40.0)
        h_pert = best_h + rng.normal(0, sigma, n)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = project_to_sum(h_pert, S)
        h = run_coord_descent(h_pert, dx, S, max_passes=80)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h2, c5_2 = local_search_swap(h, dx, S, rng, max_iters=5000)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        h3 = run_coord_descent(h2, dx, S, max_passes=50)
        c5_3 = compute_c5(h3, dx)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Try other n values
    n_current = len(best_h)
    dx_current = 2.0 / n_current
    S_current = n_current / 2.0
    for trial_n in [44, 42, 40, 46, 48, 38, 50, 36, 52, 60, 64, 80]:
        if time_left() < 40:
            break
        n_t = trial_n
        dx_t = 2.0 / n_t
        S_t = n_t / 2.0
        x_old = np.linspace(0, 2, n_current, endpoint=False) + dx_current / 2
        x_new = np.linspace(0, 2, n_t, endpoint=False) + dx_t / 2
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_sum(h_interp, S_t)
        h_t = run_coord_descent(h_interp, dx_t, S_t, max_passes=150)
        c5_t = compute_c5(h_t, dx_t)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
            n_current, dx_current, S_current = n_t, dx_t, S_t
            if time_left() > 30:
                h2, c5_2 = local_search_swap(best_h, dx_current, S_current, rng, max_iters=15000)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2
                h3 = run_coord_descent(best_h, dx_current, S_current, max_passes=80)
                c5_3 = compute_c5(h3, dx_current)
                if c5_3 < best_c5:
                    best_h, best_c5 = h3, c5_3

    # Phase 5: Binary-like initialization for current best n
    if time_left() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        S_b = n_b / 2.0
        # Create binary-like: sort indices, put 1s first
        n_ones = int(np.floor(S_b))
        h_bin = np.zeros(n_b)
        h_bin[:n_ones] = 1.0
        if abs(S_b - n_ones) > 1e-12:
            h_bin[n_ones] = S_b - n_ones
        # Randomize the positions
        h_bin = h_bin[rng.permutation(n_b)]
        h_bin = project_to_sum(h_bin, S_b)
        h_b = run_coord_descent(h_bin, dx_b, S_b, max_passes=200)
        c5_b = compute_c5(h_b, dx_b)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b
            h2, c5_2 = local_search_swap(best_h, dx_b, S_b, rng, max_iters=10000)
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2

    # Final polish
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    S_f = n_f / 2.0
    for _ in range(10):
        if time_left() < 5:
            break
        h = run_coord_descent(best_h, dx_f, S_f, max_passes=50)
        c5 = compute_c5(h, dx_f)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Final projection and verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, S_f)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, len(best_h)
