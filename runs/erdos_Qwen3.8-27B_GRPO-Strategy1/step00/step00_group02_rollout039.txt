import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def obj_func(h):
    return compute_c5(h)


def sum_con_fun(x):
    return np.sum(x) - len(x) / 2.0


def sum_con_jac(x):
    return np.ones(len(x))


def make_constraint():
    return {'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}


def interp_solution(h_parent, n_parent, n_child):
    if n_parent == n_child:
        return h_parent.copy()
    x_parent = np.linspace(0, 2, n_parent, endpoint=False)
    x_child = np.linspace(0, 2, n_child, endpoint=False)
    f = interp1d(x_parent, h_parent, kind='linear', bounds_error=False, fill_value='extrapolate')
    h_child = np.array(f(x_child))
    return np.clip(h_child, 0, 1)


def fix_sum(h):
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute evenly among elements not at bounds
        h = h + diff / n
    h = np.clip(h, 0, 1)
    # After clipping, might need another pass
    remaining = target - np.sum(h)
    if remaining > 0:
        mask = h < 1.0
        if np.sum(mask) > 0:
            h[mask] += remaining / np.sum(mask)
            h = np.clip(h, 0, 1)
    elif remaining < 0:
        mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += remaining / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h


def slsqp_optimize(h0, n, maxiter=5000, ftol=1e-10):
    bounds = [(0.0, 1.0)] * n
    cons = make_constraint()
    res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    if res.fun < compute_c5(h0) + 1e-12:
        return res.x, res.fun
    return h0, compute_c5(h0)

        h_opt, f_opt = slsqp_optimize(h_pert, n, maxiter=3000, ftol=1e-9)
        if f_opt < best_f:
            best_h, best_f = h_opt, f_opt
    return best_h, best_f

def compute_c5(h):

def objective(h):

def sum_con_fun(x):
    return np.sum(x) - len(x) / 2.0


def sum_con_jac(x):

    n = len(h)

            h = np.clip(h, 0.0, 1.0)

    return best, best_f

    h_ext = np.concatenate([h_p, h_p[:1]])
    h_c = np.interp(xc, xp, h_ext)
    return np.clip(h_c, 0, 1)
