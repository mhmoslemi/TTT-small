        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.999999
        else:
            mask = h > 1e-7
        cnt = np.sum(mask)
        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j - lag < n:
            g += 1.0 - h[j - lag]
        if 0 <= j + lag < n:
            g -= h[j + lag]
        grad[j] = g
    return grad * dx, np.max(corr) * dx

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ec = np.exp(tau * (corr - m))
    p = ec / np.sum(ec)
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        v1 = (j - lags >= 0) & (j - lags < n)
        v2 = (j + lags >= 0) & (j + lags < n)

        if np.any(v2):
            s -= np.dot(p[lags[v2] + n - 1], h[j + lags[v2]])
        grad[j] = s

        best_c5 = final_c5
    
    return (best_h, best_c5, best_n)
