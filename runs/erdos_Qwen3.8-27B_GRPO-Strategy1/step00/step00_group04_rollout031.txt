            grad[i] = (smoothed_max(hp, T) - smoothed_max(hm, T)) / (2.0 * eps)

    # Project
    h_new = project_feasible(h_new, n)

        return smooth_obj(x, T)
    def obj_grad(x):
        _, g = num_grad(x, T, eps=1e-5)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       constraints=cons, bounds=bounds,
                       options={'maxiter': 150, 'ftol': 1e-10})
        if res.x is not None:
            h_opt = project_feasible(np.clip(res.x, 0, 1), n)
            c5_opt = compute_c5(h_opt)
            if c5_opt < compute_c5(h0):
                return h_opt, c5_opt
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.93
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def try_optimize(h_candidate, n, max_iters):
        nonlocal best_h, best_c5
        if time.time() > deadline:
            return
        h_opt, c5_opt = pgd_optimize(h_candidate, n, deadline, max_iters)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
        if time.time() < deadline and n <= 150:
            h_sls, c5_sls = slsqp_polish(h_opt, n, deadline)
            if c5_sls < best_c5:
                best_h = h_sls.copy()
                best_c5 = c5_sls

    # Phase 1: Optimize at initial resolution
    h_init_feasible = project_feasible(h_init.copy(), n_init)
    try_optimize(h_init_feasible, n_init, 500)

    # Phase 2: Try upsampling to finer grids
    ns_fine = []
    for mult in [2, 3, 4, 5]:
        n_try = n_init * mult
        if n_try % 2 == 1:
            n_try += 1
        if n_try <= 400:
            ns_fine.append(n_try)
    ns_fine = sorted(set(ns_fine))

    for n_fine in ns_fine:
        if time.time() > deadline:
            break
        h_up = np.interp(
            np.linspace(0, n_init - 1, n_fine),
            np.arange(n_init, dtype=np.float64),
            h_init
        )
        h_up = project_feasible(h_up, n_fine)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        iters = 350 if n_fine <= 120 else 200
        try_optimize(h_up, n_fine, iters)

        # Restart with small perturbation
        for r in range(3):
            if time.time() > deadline:
                break
            h_r = best_h.copy() if len(best_h) == n_fine else h_up.copy()
            h_r = h_r + rng.normal(0, 0.015, n_fine)
            h_r = project_feasible(h_r, n_fine)
            try_optimize(h_r, n_fine, iters)

    # Phase 3: Block-structure starting point
    if time.time() < deadline:
        n_blk = n_init
        h_blk = np.zeros(n_blk)
        h_blk[:n_blk // 2] = 1.0
        # Perturb to break symmetry
        h_blk = h_blk + rng.normal(0, 0.1, n_blk)
        h_blk = project_feasible(h_blk, n_blk)
        try_optimize(h_blk, n_blk, 400)

    # Phase 4: Downsample best to n_init if needed, or keep as is
    n_best = len(best_h)
    if n_best > n_init and time.time() < deadline:
        h_ds = np.interp(
            np.linspace(0, n_best - 1, n_init),
            np.arange(n_best, dtype=np.float64),
            best_h
        )
        h_ds = project_feasible(h_ds, n_init)
        c5_ds = compute_c5(h_ds)
        if c5_ds < best_c5:
            best_h = h_ds
            best_c5 = c5_ds
        # Also try optimizing the downsampled version
        try_optimize(h_ds, n_init, 300)

    # Final projection and evaluation
    best_h = project_feasible(np.clip(best_h, 0.0, 1.0), len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
