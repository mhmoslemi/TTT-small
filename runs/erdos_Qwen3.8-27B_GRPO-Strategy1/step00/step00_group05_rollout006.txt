import numpy as np
import time
import math

def compute_c5(h, dx):
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full") * dx
    return float(np.max(corr))

def resample(h, new_n):
    """Resample h from len(h) to new_n, maintaining sum = new_n/2"""
    old_n = len(h)
    h_float = h.astype(float)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new_float = np.interp(x_new, x_old, h_float, period=2)
    h_new = (h_new_float >= 0.5).astype(float)
    # Adjust to get exactly new_n/2 ones
    current_sum = int(h_new.sum())
    target_sum = new_n // 2
    if current_sum > target_sum:
        # Remove excess ones
        indices = np.where(h_new == 1)[0]
        np.random.shuffle(indices)
        for i in indices[:current_sum - target_sum]:
            h_new[i] = 0
    elif current_sum < target_sum:
        indices = np.where(h_new == 0)[0]
        np.random.shuffle(indices)
        for i in indices[:target_sum - current_sum]:
            h_new[i] = 1
    return h_new

def local_search(h, dx, max_iters=50000, seed=42, verbose=False):
    """Greedy local search by swapping 0 and 1"""
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    n = len(h)
    best_c5 = compute_c5(h, dx)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    stagnant = 0
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        c5 = compute_c5(h, dx)

            best_c5 = c5
            stagnant = 0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1
            h[j] = 0
            stagnant += 1
            if stagnant > 5000:
                break
    return h, best_c5

def simulated_annealing(h, dx, max_iters=100000, seed=42, T_start=0.1, T_end=0.001):
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        T = T_start * (T_end / T_start) ** (it / max_iters)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        c5 = compute_c5(h, dx)
        delta = c5 - curr_c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            curr_c5 = c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def resample_binary(h, new_n, seed=42):

    old_n = len(h)

    h_ext = np.concatenate([h_float, h_float, h_float])
