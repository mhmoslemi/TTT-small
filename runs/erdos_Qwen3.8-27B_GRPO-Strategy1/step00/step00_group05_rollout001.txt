import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]))
    h_new = np.clip(h_new, 0, 1)
    return adjust_sum(h_new, n_new / 2.0)

def adjust_sum(h, target_sum):
    h = np.clip(h, 0, 1).astype(float)
    s = h.sum()
    if s > 1e-12:
        h *= target_sum / s
    h = np.clip(h, 0, 1)
    remaining = target_sum - h.sum()
    if remaining > 1e-10:
        mask = h < 0.9999
        if mask.any():
            h[mask] = np.clip(h[mask] + remaining / mask.sum(), 0, 1)
    elif remaining < -1e-10:
        mask = h > 0.0001
        if mask.any():
            h[mask] = np.clip(h[mask] + remaining / mask.sum(), 0, 1)
    return h

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def make_smooth(n, rng):
    freqs = rng.randint(1, 8, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.1, 0.4, size=3)
    x = np.arange(n) / n
    h = 0.5
    for f, p, a in zip(freqs, phases, amps):
        h += a * np.sin(2 * np.pi * f * x + p)
    return adjust_sum(h, n / 2.0)

def make_periodic(n, period, duty, rng):
    h = np.zeros(n)
    w = max(1, int(period * duty))
    shift = rng.randint(0, period)
    for i in range(n):
        if (i - shift) % period < w:
            h[i] = 1.0
    return adjust_sum(h, n / 2.0)

def slsqp_optimize(h0, n, maxiter=100):
    target_sum = n / 2.0
    h0 = adjust_sum(h0, target_sum)

    def obj(x):
        return compute_c5(np.clip(x, 0, 1), n)

    def cons_fn(x):
        return x.sum() - target_sum

    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cons_fn}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return adjust_sum(res.x, target_sum)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    n_parent = 70
    h_parent = None
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        best_c5 = compute_c5(h_parent, n_parent)
        best_h = h_parent.copy()
        best_n = n_parent

    def remaining():
        return budget_s - (time.time() - t0)

    resolutions = [70, 140, 210, 280, 420, 560]

    for n in resolutions:
        if remaining() < 60:
            break
        target_sum = n / 2.0

        candidates = []
        if h_parent is not None and n != n_parent:
            candidates.append(upsample(h_parent, n))
        if best_h is not None and best_n != n:
            candidates.append(upsample(best_h, n))
        elif best_h is not None and best_n == n:
            pass  # already at this resolution

        for _ in range(6):
            candidates.append(make_smooth(n, rng))
        for _ in range(6):
            candidates.append(make_binary(n, rng))
        for p in [4, 6, 8, 10, 12]:
            candidates.append(make_periodic(n, p, 0.5, rng))
        for _ in range(4):

                                            rng.uniform(0.3, 0.6), rng))

        # Perturb best if at this n
