    dx = 2.0 / n
    g = np.zeros(n)

            g[k:] += (1.0 - h[:n-k])

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def grad_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if 0 <= k < n:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += -h[k:]
    elif -n < k < 0:
        kp = -k
        g[:n - kp] += (1.0 - h[kp:])
        g[kp:] += -h[:n - kp]
    return g * dx


def project_box_sum(v, target):
    lo, hi = -3.0, 3.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def optimize_h(h_init, max_iters, t0, budget, lr=1.0, beta_init=30.0, beta_end=300.0):
    n = len(h_init)
    target = n / 2.0
    h = project_box_sum(h_init.copy(), target)
    best_h = h.copy()
    best_sc = compute_c5(h)
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > budget - 5.0:
            break

        frac = min(1.0, it / max(1, max(1, max_iters // 3)))
        beta = beta_init + (beta_end - beta_init) * frac

        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        cur = float(np.max(scores))
        if cur < best_sc - 1e-12:
            best_sc = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break

        sh = scores - cur
        w = np.exp(beta * sh)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros_like(scores)
            w[np.argmax(scores)] = 1.0
        else:
            w /= wsum

        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] > 1e-5:
                k = m - (n - 1)
                grad += w[m] * grad_shift(h, k)

        h_new = project_box_sum(h - lr * grad, target)
        new_sc = compute_c5(h_new)
        if new_sc < cur:
            h = h_new
            lr = min(lr * 1.08, 5.0)
        else:
            lr *= 0.6
            if lr < 1e-5:
                break

    return best_h, best_sc

        h[start:end] = rng.choice([0.0, 0.5, 1.0])
    h += 0.05 * np.random.RandomState(seed_val + 100).randn(n)
    return h
