    for n in n_list:
        if time.time() - t0 > budget_s - 15:
            break
        dx = 2.0 / n
        _sum_target[0] = n / 2.0
        bounds = [(0.0, 1.0)] * n

        starts = []
        if n == init_n and init_h is not None:
            starts.append(repair_h(init_h, n))
        for m in range(4):
            starts.append(make_start(n, seed * 11 + m * 7 + n, m))

        for st in starts:
            if time.time() - t0 > budget_s - 15:
                break
            h_work = st.copy()

            for beta in [50.0, 200.0, 800.0]:
                if time.time() - t0 > budget_s - 15:
                    break
                try:
                    res = minimize(_obj, h_work, args=(dx, beta),
                                  method='SLSQP', bounds=bounds,
                                  constraints=constraints,
                                  options={'maxiter': 200, 'ftol': 1e-11})
                    h_work = repair_h(res.x, n)
                except Exception:
                    pass

            try:
                res = minimize(_obj2, h_work, args=(dx, 100.0),
                              method='SLSQP', bounds=bounds,
                              constraints=constraints,
                              options={'maxiter': 100, 'ftol': 1e-12})
                h_work = repair_h(res.x, n)
            except Exception:
                pass

            c5 = compute_c5(h_work, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_work.copy()
                best_n = n

            if time.time() - t0 < budget_s - 20 and best_h is not None:
                remaining = max(0.5, (budget_s - 20) - (time.time() - t0)) / 4.0
                h_p, c5_p = perturb_h(best_h, best_n, 2.0 / best_n, remaining, rng)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p
                    best_n = best_n

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_c5 = 0.5
        best_n = n

    return best_h, best_c5, best_n
