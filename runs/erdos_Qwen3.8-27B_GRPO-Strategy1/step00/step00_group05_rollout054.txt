            h_opt, c5_opt = _slsqp(best_h.copy(), best_n, maxiter=250, smooth=True, beta=80.0)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Try upscaling and optimizing at higher resolutions
    for target_n in [140, 210, 280, 350, 420, 560, 700]:
        if _remaining() < 200:
            break
        if target_n % best_n != 0:
            continue
        factor = target_n // best_n
        h_up = upsample_signal(best_h, factor)
        maxiter = max(40, 350 - target_n // 2)

        # Direct SLSQP
        h_opt, c5_opt = _slsqp(h_up, target_n, maxiter=maxiter)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, target_n

        # Smooth SLSQP
        if _remaining() > 150:
            h_opt, c5_opt = _slsqp(h_up, target_n, maxiter=maxiter, smooth=True, beta=120.0)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, target_n

        # Binarize then re-optimize
        if _remaining() > 150:
            h_bin = _try_binarize(best_h * factor + h_up * 0 + best_h, target_n) if False else _try_binarize(h_up, target_n)
            h_opt, c5_opt = _slsqp(h_bin, target_n, maxiter=max(40, maxiter // 2))
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, target_n

        # Perturbed initialization
        if _remaining() > 150:
            h_pert = h_up + 0.08 * rng.randn(target_n)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = _fix_sum(h_pert, target_n / 2.0)
            h_opt, c5_opt = _slsqp(h_pert, target_n, maxiter=max(40, maxiter // 2), smooth=True, beta=60.0)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, target_n

    # Final: try one more round at the best resolution with higher iterations
    if _remaining() > 120:
        n_try = best_n
        if n_try < 140 and _remaining() > 200:
            n_try = 140
            h_up = upsample_signal(best_h, 140 // best_n)
        else:
            h_up = best_h.copy()
        h_opt, c5_opt = _slsqp(h_up, n_try, maxiter=min(500, max(100, int(_remaining() * 0.5))))
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
