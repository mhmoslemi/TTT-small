    """Project h onto {0<=x<=1, sum(x)=n_points/2}"""
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on shift
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Need to find alpha: sum(clamp(h - alpha, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):

        h_mid = np.clip(h - mid, 0.0, 1.0)
        s_mid = np.sum(h_mid)
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    diff = target - np.sum(h_proj)

def compute_c5(h, dx):
    """Compute the C5 value for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n_points):
    """Project h onto {0<=x<=1, sum(x)=n_points/2}."""
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Bisection on alpha: find alpha such that sum(clamp(h-alpha,0,1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)

            else:
                idx = np.argmax(h_proj)
            h_proj[idx] += diff
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_grad(h, n_points, dx, T):
    """Compute gradient of smoothed max of correlation."""
    N = n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(corr)
    exp_val = np.exp((corr - c_max) * T)
    weights = exp_val / np.sum(exp_val)
    lags = np.arange(2 * N - 1) - (N - 1)
    
    grad = np.zeros(N)
    for m in range(2 * N - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = lags[m]
        if lag >= 0:
            # Term 1: (1-h[j-lag]) for j in [lag, N-1]

            grad[0:N+lag] += w * (1.0 - h[-lag:N])
            # Term 2: -h[j+lag] = -h[j-|lag|] for j in [|lag|, N-1]
            grad[-lag:N] -= w * h[0:N+lag]
    
    grad *= dx
    obj = (np.log(np.sum(exp_val)) / T + c_max) * dx
    return obj, grad

    h = h0.copy()
    h = project_onto_constraint(h, n_points)
    T = T_init
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    
    # Initial learning rate
    grad_norm = np.linalg.norm(compute_grad(h, n_points, dx, T)[1])
    lr = 1.0 / (grad_norm + 1e-6) if grad_norm > 0 else 0.01
    lr = min(lr, 1.0 / (T * dx))
    
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        obj, grad = compute_grad(h, n_points, dx, T)

        h_new = project_onto_constraint(h - lr * grad, n_points)
        c5_new = compute_c5(h_new, dx)

            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        elif c5_new < compute_c5(h, dx):
            h = h_new
            stall = 0
        else:

            lr *= 0.7

These are NOT the same in general. corr[+k] = sum_i h[i+k] * (1-h[i]) (by substitution j=i+k).

Actually:
corr at lag k (m = k + n - 1): sum_i h[i+k] * (1-h[i]) for valid i
corr at lag -k (m = -k + n - 1): sum_i h[i-k] * (1-h[i]) for valid i = sum_j h[j] * (1-h[j+k]) for valid j

So corr[-k] = sum_j h[j] * (1-h[j+k]) which is our integral for shift k.
And corr[+k] = sum_i h[i+k] * (1-h[i]) = sum_j h[j] * (1-h[j-k]) which is the integral for shift -k.

Since we take max over all lags (both signs), we're covering all shifts. Good.

Now, one important observation: the integral for shift k=0 is:
∫ h(x)(1-h(x)) dx = 1 - ∫ h²(x) dx

This is minimized when ∫h² is maximized (i.e., h is as "peaked" as possible).
The integrals for large |k| are small (small overlap).
The "dangerous" shifts are moderate |k| where there's significant overlap but h and h(x+k) can be very different.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
        else:
            idx = np.argmin(h_proj) if diff > 0 else np.argmax(h_proj)
            h_proj[idx] = np.clip(h_proj[idx] + diff, 0, 1)
    return h_proj

def compute_grad(h, N, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(corr)
    exp_val = np.exp((corr - c_max) * T)
    total = np.sum(exp_val)
    weights = exp_val / total
    lags = np.arange(2 * N - 1) - (N - 1)
    grad = np.zeros(N)
    for m in range(2 * N - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = lags[m]
        if lag >= 0:
            grad[lag:N] += w * (1.0 - h[0:N - lag])
            grad[0:N - lag] -= w * h[lag:N]
        else:
            grad[0:N + lag] += w * (1.0 - h[-lag:N])
            grad[-lag:N] -= w * h[0:N + lag]
    grad *= dx
    obj = (np.log(total) / T + c_max) * dx
    return obj, grad

def optimize_pgd(h0, N, dx, T_init, max_iters, time_limit):
    h = project_onto_constraint(h0.copy(), N)
    T = T_init
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    obj, grad = compute_grad(h, N, dx, T)
    gn = np.linalg.norm(grad)
    lr = 1.0 / (gn + 1e-4) if gn > 1e-4 else 0.01
    lr = min(lr, 2.0 / (T * dx * N))
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        _, grad = compute_grad(h, N, dx, T)
        accepted = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            h_new = project_onto_constraint(h - lr * shrink * grad, N)
            c5_new = compute_c5(h_new, dx)
            c5_cur = compute_c5(h, dx)
            if c5_new < c5_cur:
                h = h_new
                accepted = True
                break
        if accepted:
            stall = 0
            c5_cur = compute_c5(h, dx)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            stall += 1
            lr *= 0.6
            if stall > 20:
                T = max(T * 0.85, 5.0)
                _, g2 = compute_grad(h, N, dx, T)
                gn2 = np.linalg.norm(g2)
                lr = min(1.0 / (gn2 + 1e-4), 2.0 / (T * dx * N))
                stall = 0
                if T <= 5.5 and stall > 0:
                    break
        if it % 20 == 0 and it > 0:
            T = max(T * 0.998, 8.0)
        if it % 100 == 0:
            _, g3 = compute_grad(h, N, dx, T)
            gn3 = np.linalg.norm(g3)
            if gn3 > 1e-4:
                lr = min(1.0 / (gn3 + 1e-4) * 0.8, 2.0 / (T * dx * N))
    return best_h, best_c5
