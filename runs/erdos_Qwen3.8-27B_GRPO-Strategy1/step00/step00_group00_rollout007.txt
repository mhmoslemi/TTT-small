from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(np.sum(free))
        if cnt == 0:
            break
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def soft_max(c, tau):
    cm = float(np.max(c))
    return cm + tau * float(np.log(np.sum(np.exp((c - cm) / tau))))

def penalized_objective(h, dx, tau, target_sum, penalty_w):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = soft_max(c, tau)
    val += penalty_w * (np.sum(h) - target_sum) ** 2
    return val
