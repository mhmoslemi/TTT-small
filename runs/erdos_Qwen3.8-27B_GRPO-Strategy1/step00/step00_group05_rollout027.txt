            res = minimize(c5_objective, best_h, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 600, 'ftol': 1e-12})
            if res.fun < best_c5:
                h_new = np.clip(res.x, 0.0, 1.0)
                adj = (n / 2.0 - np.sum(h_new)) / n
                h_new = np.clip(h_new + adj, 0.0, 1.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 3: Higher resolutions with local search
    for factor in [2, 4]:
        elapsed = time.time() - t0
        if budget_s - elapsed < 50:
            break
        n = n_parent * factor
        if n > 2500:
            break
        h_up = np.repeat(best_h, factor)
        h_bin_up = np.where(h_up >= 0.5, 1.0, 0.0)
        s = int(np.sum(h_bin_up))
        target = n // 2
        if s != target:
            if s > target:
                idx = np.random.choice(np.where(h_bin_up == 1)[0], s - target, replace=False)
                h_bin_up[idx] = 0.0
            else:
                idx = np.random.choice(np.where(h_bin_up == 0)[0], target - s, replace=False)
                h_bin_up[idx] = 1.0
        tb = min(30, max(8, int((budget_s - (time.time() - t0)) // 4)))
        h_new, c5_new = local_search_binary(h_bin_up, 40, tb)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new

    # Phase 4: Kick + local search (escape local optima)
    elapsed = time.time() - t0
    if budget_s - elapsed > 50 and n_parent <= 300:
        n = n_parent
        for kick in range(12):
            elapsed = time.time() - t0
            if budget_s - elapsed < 30:
                break
            h_pert = np.where(best_h >= 0.5, 1.0, 0.0)
            n_kicks = int(np.random.randint(3, 8))
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_kicks):
                if len(ones) < 1 or len(zeros) < 1:
                    break
                i = int(np.random.choice(ones))
                j = int(np.random.choice(zeros))
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            h_k, c5_k = local_search_binary(h_pert, 30, 10)
            if c5_k < best_c5:
                best_h, best_c5 = h_k.copy(), c5_k

    # Phase 5: Random restarts
    elapsed = time.time() - t0
    if budget_s - elapsed > 40:
        n = n_parent
        n_trials = min(6, int((budget_s - elapsed) // 20))
        for trial in range(n_trials):
            elapsed = time.time() - t0
            if budget_s - elapsed < 25:
                break
            h_rand = np.zeros(n)
            h_rand[np.random.choice(n, n // 2, replace=False)] = 1.0
            h_rs, c5_rs = local_search_binary(h_rand, 30, 10)
            if c5_rs < best_c5:
                best_h, best_c5 = h_rs.copy(), c5_rs

    # Phase 6: Final polish with local search
    elapsed = time.time() - t0
    if budget_s - elapsed > 15 and len(best_h) <= 200:
        h_fin = np.where(best_h >= 0.5, 1.0, 0.0)
        h_fs, c5_fs = local_search_binary(h_fin, 40, 12)
        if c5_fs < best_c5:
            best_h, best_c5 = h_fs.copy(), c5_fs

    # Ensure valid return
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, int(len(best_h)))
