from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    """Project h to satisfy 0<=h[i]<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    # Distribute the adjustment
    if diff > 0:
        # Need to increase sum: add to elements with room
        room = 1.0 - h
        mask = room > 1e-10
        if np.sum(room[mask]) < diff:

import numpy as np
from scipy.optimize import minimize
import time

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraints(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += diff * room / total
    else:
        total = h.sum()
        if total > 1e-12:
            h += diff * h / total
    return np.clip(h, 0.0, 1.0)

def sum_eq(h):
    return h.sum() - _global_target

def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h

def simulated_anneal(h, n, rng, max_iters, t0, budget):
    global _global_target
    _global_target = n / 2.0
    val = compute_c5(h)
    T = 0.02
    for it in range(max_iters):
        if time.time() - t0 > budget - 5:
            break
        move_type = rng.random()
        if move_type < 0.5:
            # Swap two elements
            i, j = rng.choice(n, 2, replace=False)
            old_i, old_j = h[i], h[j]
            h[i], h[j] = old_j, old_i
            new_val = compute_c5(h)
            delta = new_val - val
            if delta < 0 or rng.random() < np.exp(-max(delta, 0) / T):
                val = new_val
            else:
                h[i], h[j] = old_i, old_j
        elif move_type < 0.8:
            # Perturb one element
            i = rng.randint(n)
            old = h[i]
            h[i] = np.clip(old + rng.normal(0, 0.3), 0, 1)
            h = project_onto_constraints(h, n)
            new_val = compute_c5(h)
            delta = new_val - val
            if delta < 0 or rng.random() < np.exp(-max(delta, 0) / T):
                val = new_val
            else:
                h[i] = old
                h = project_onto_constraints(h, n)
        else:
            # Perturb two elements
            i, j = rng.choice(n, 2, replace=False)
            old_i, old_j = h[i], h[j]
            h[i] = np.clip(old_i + rng.normal(0, 0.3), 0, 1)
            h[j] = np.clip(old_j + rng.normal(0, 0.3), 0, 1)
            h = project_onto_constraints(h, n)
            new_val = compute_c5(h)
            delta = new_val - val
            if delta < 0 or rng.random() < np.exp(-max(delta, 0) / T):
                val = new_val
            else:
                h[i], h[j] = old_i, old_j
                h = project_onto_constraints(h, n)
        T *= 0.999
        T = max(T, 0.001)
    return h, val

def coordinate_descent(h, n, max_rounds, t0, budget):
    global _global_target
    _global_target = n / 2.0
    val = compute_c5(h)
    for r in range(max_rounds):
        if time.time() - t0 > budget - 5:
            break
        improved = False
        for m in range(n):
            # Try changing h[m] while adjusting h[m+1] (mod n) to maintain sum
            partner = (m + 1) % n
            old_m, old_p = h[m], h[partner]
            # Vary h[m] and adjust partner
            for frac in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5]:
                new_m = np.clip(h[m] + frac, 0, 1)
                new_p = np.clip(h[partner] + (h[m] - new_m), 0, 1)
                # Check if sum is preserved
                if abs((h[m] + h[partner]) - (new_m + new_p)) < 1e-10:
                    h[m], h[partner] = new_m, new_p
                    new_val = compute_c5(h)
                    if new_val < val:
                        val = new_val
                        improved = True
                        break
                    else:
                        h[m], h[partner] = old_m, old_p
        if not improved:
            break
    return h, val

        _global_target = n0 / 2.0

        best_h, best_val, best_n = h0.copy(), val0, n0
        print(f"Initial: n={n0}, C5={val0:.6f}")
    
    # Optimize the initial with SLSQP
    if still_ok():
        n = best_n
        _global_target = n / 2.0
        bounds = [(0, 1)] * n
        cons = [{'type': 'eq', 'fun': sum_eq}]
        for method in ['SLSQP', 'Nelder-Mead']:
            if not still_ok(20):
                break
            try:
                kwargs_opt = {'maxiter': 2000, 'disp': False}
                if method == 'SLSQP':
                    kwargs_opt['ftol'] = 1e-12
                res = minimize(compute_c5, best_h.copy(), method=method,
                             bounds=bounds if method == 'SLSQP' else None,
                             constraints=cons if method == 'SLSQP' else [],
                             options=kwargs_opt)
                if res.x is not None:
                    h_opt = project_onto_constraints(res.x, n)
                    v = compute_c5(h_opt)
                    if v < best_val:

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-12:
            h = h + diff * room / tot
    else:
        tot = h.sum()
        if tot > 1e-12:
            h = h + diff * h / tot
    return np.clip(h, 0.0, 1.0)

    return h.sum() - _global_target

    h = np.zeros(p)

            h[i] = 1.0
