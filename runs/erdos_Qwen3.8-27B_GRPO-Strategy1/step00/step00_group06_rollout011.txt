        target_sum = n / 2.0
        starts = make_starts(n, rng)
        if h_init is not None and n == n_init:
            starts.insert(0, h_init.copy())

        for h0 in starts:
            if time.time() - t_start_global > budget_s - 20:
                break
            h0r = repair(h0, target_sum)
            c5 = compute_C5(h0r, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0r.copy()

            # Nelder-Mead
            if time.time() - t_start_global > budget_s - 50:
                break
            res = minimize(nm_obj, h0r, args=(target_sum, dx), method='Nelder-Mead',
                           options={'maxiter': 800, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
            h_nm = repair(res.x, target_sum)
            c5_nm = compute_C5(h_nm, dx)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm.copy()

            # Peak reduction (simulated annealing focused on peak)
            if time.time() - t_start_global > budget_s - 30:
                break
            hs, cs = peak_reduce(h_nm, dx, target_sum, rng, t_start_global, budget_s, n_iters=2500)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs

            # Coordinate descent from best so far
            if time.time() - t_start_global > budget_s - 20:
                break
            start_cd = h_nm if c5_nm <= cs else hs
            hs2, cs2 = coord_descent(start_cd, dx, target_sum, n_iters=8000)
            if cs2 < best_c5:
                best_c5 = cs2
                best_h = hs2

    if best_h is None:
        n = 94
        h = np.zeros(n)
        h[0::2] = 1.0
        best_h = repair(h, n / 2.0)
        best_c5 = compute_C5(best_h, 2.0 / n)

    return best_h, best_c5, len(best_h)
