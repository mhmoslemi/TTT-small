    remaining = budget_s - elapsed - 20
    for n_try in [30, 40, 50, 60, 90, 120]:
        if remaining < 60:
            break
        if time.time() - start_time > budget_s - 30:
            break
        t_left = min(120, remaining * 0.15)
        h_opt, c5_opt = try_resolution(n_try, best_h, best_c5, seed=seed + n_try, time_limit=t_left)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            n = n_try
        remaining = budget_s - (time.time() - start_time) - 20

    # Phase 4: Coordinate descent polish
    if time.time() - start_time < budget_s - 5:
        cur_n = len(best_h)
        cur_sum = cur_n / 2.0
        improved = True
        while improved and time.time() - start_time < budget_s - 5:
            improved = False
            for i in range(cur_n):
                for val in [0.0, 1.0, 0.5]:
                    h_try = best_h.copy()
                    h_try[i] = val
                    h_try = project_h(h_try, cur_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        improved = True

    # Phase 5: Final Nelder-Mead polish
    if time.time() - start_time < budget_s - 10:
        try:
            cur_n = len(best_h)
            cur_sum = cur_n / 2.0
            def obj_nm(x):
                return compute_c5(project_h(x, cur_sum))
            res = minimize(obj_nm, best_h, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
            h_nm = project_h(res.x, cur_sum)
            c5_nm = compute_c5(h_nm)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm
        except Exception:
            pass

    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = project_h(final_h, len(final_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
