import numpy as np
from scipy.optimize import minimize
import time

def obj_max(h, dx):
    """Compute exact C5 value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def obj_smooth(h, dx, temp=5.0):
    """Smooth max using log-sum-exp."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < 1e-10:
        return 0.0
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp))) / (2*len(h)-1)) * dx

def fix_constraint(h, target):
    """Fix sum(h) = target while keeping 0 <= h <= 1."""
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

import time

def c5_exact(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_smooth(h, dx, beta=2.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < 1e-12:
        return 0.0
    # log-sum-exp smoothing
    normalized = (corr - m) / beta
    lse = m + beta * np.log(np.sum(np.exp(normalized)) / len(corr))
    return lse * dx

def fix_sum(h, target):
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def c5_val(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < 1e-14:
        return 0.0
    return (m + beta * np.log(np.sum(np.exp((corr - m) / beta))) / (2 * len(h) - 1)) * dx

def fix_sum(h, target):
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h, target, rng, max_iter, cool_start=1.0, cool_rate=0.999):
    """Local search for binary-like sequences with swaps."""
    n = len(h)
    best_c5 = c5_val(h, 2.0 / n)
    cur = h.copy()
    T = cool_start
    for it in range(max_iter):
        T *= cool_rate

        j = rng.randint(n)
        if i == j:
            continue

        delta = rng.uniform(-0.3, 0.3)
        cur[i] += delta
        cur[j] -= delta
        if cur[i] < 0 or cur[i] > 1 or cur[j] < 0 or cur[j] > 1:
            cur[i] -= delta
            cur[j] += delta
            continue
        cur = fix_sum(cur, target)
        new_c5 = c5_val(cur, 2.0 / n)
        if new_c5 < best_c5 or rng.random() < np.exp(-(new_c5 - best_c5) / max(T, 1e-10)):

                best_c5 = new_c5
            # Accept

import numpy as np
from scipy.optimize import minimize
import time

def c5_val(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < 1e-14:
        return 0.0
    return (m + beta * np.log(np.sum(np.exp((corr - m) / beta))) / (2 * len(h) - 1)) * dx

def fix_sum(h, target):
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def swap_search(h, target, rng, max_iter, T0=0.01):
    n = len(h)
    dx = 2.0 / n
    best = c5_val(h, dx)
    cur = h.copy()
    T = T0
    for it in range(max_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        c5_new = c5_val(cur, dx)
        if c5_new < best or (T > 1e-8 and rng.random() < np.exp(-max(c5_new - best, 0) / T)):
            if c5_new < best:
                best = c5_new
        else:
            cur[i], cur[j] = cur[j], cur[i]
        T *= 0.9995
    return cur, best

def perturb_search(h, target, rng, max_iter):

            cur[i] -= d

            cur = fix_sum(cur, target)
