import numpy as np
from scipy.optimize import minimize, linprog
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def corr_grad_at(h, m, n):
    """Gradient of correlate(h, 1-h, 'full')[m] w.r.t. h"""
    lag = m - (n - 1)
    grad = np.zeros(n)
    # First term: d/dh[p] of sum h[n+lag]*(1-h[n]) = (1-h[p-lag]) for valid p
    lo1 = max(0, lag)
    hi1 = min(n, lag + n)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[np.arange(lo1 - lag, hi1 - lag)])
    # Second term: -h[p+lag] for valid p
    lo2 = max(0, -lag)
    hi2 = min(n, n - lag)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[np.arange(lo2 + lag, hi2 + lag)]
    return grad

def project_onto_box_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    # Binary search for alpha such that sum(clip(h+alpha, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    alpha = (lo + hi) / 2
    return np.clip(h + alpha, 0.0, 1.0)

def subgradient_optimize(h0, dx, max_iter=5000, tol=1e-8):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    h = project_onto_box_sum(h, target_sum)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    alpha = 2.0 / (n + 1.0)  # initial step size
    beta = 0.9  # momentum
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Subgradient: use top-3 active constraints
        top_idx = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for m in top_idx:
            g += corr_grad_at(h, m, n)
        g /= len(top_idx)
        
        # Heavy ball
        v = beta * v - alpha * g
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target_sum)
        h = h_new
        
        # Adapt step size
        alpha *= 0.999
        
        if abs(c5 - best_c5) < tol and it > 100:
            break
    
    return best_h, best_c5

def coordinate_descent(h0, dx, max_iter=100):
    """Try to improve by changing individual coordinates."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for it in range(max_iter):
        improved = False
        for i in range(n):
            # Try setting h[i] to various values
            # If we increase h[i] by d, we must decrease another coordinate by d
            # For simplicity, try small perturbations
            old_val = h[i]
            # Try h[i] = 0
            h[i] = 0.0
            h = project_onto_box_sum(h, target_sum)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                improved = True
            else:
                h[i] = old_val
                h = project_onto_box_sum(h, target_sum)
            # Try h[i] = 1
            h[i] = 1.0
            h = project_onto_box_sum(h, target_sum)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                improved = True
            else:
                h[i] = old_val
                h = project_onto_box_sum(h, target_sum)
        
        if not improved:
            break
    
    return h, best_c5

def interpolate_h(h_parent, n_new):
    """Interpolate parent solution to new resolution."""
    n_old = len(h_parent)
    dx_old = 2.0 / n_old

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def corr_grad_at(h, m, n):
    lag = m - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, lag)
    hi1 = min(n, lag + n)
    if lo1 < hi1:
        idx = np.arange(lo1 - lag, hi1 - lag)
        grad[lo1:hi1] += (1.0 - h[idx])
    lo2 = max(0, -lag)
    hi2 = min(n, n - lag)
    if lo2 < hi2:
        idx = np.arange(lo2 + lag, hi2 + lag)
        grad[lo2:hi2] -= h[idx]
    return grad

def project(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def interp_h(h_par, n_new):
    n_old = len(h_par)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_par)

    target = n / 2.0
    h = project(h0, target).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    t0 = time.time()
    alpha = 0.4 / np.sqrt(n)
    beta = 0.85
    patience = 0
    for it in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        k_top = min(7, len(corr))
        top = np.argsort(corr)[-k_top:]
        g = np.zeros(n)
        for m in top:
            g += corr_grad_at(h, m, n)
        g /= k_top
        v = beta * v - alpha * g
        h = project(h + v, target)
        alpha *= 0.9992
        if patience > 800:
            break
    return best_h, best_c5
