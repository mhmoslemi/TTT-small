def evaluate(h):
    """Compute C5 for given h array"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Bisection on constant offset
    for _ in range(40):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        if s > target:
            h -= (s - target) / n
        else:
            h += (target - s) / n
        h = np.clip(h, 0, 1)
    return h

def grad_single(h, s, n):
    """Gradient of correlation at shift s"""
    dx = 2.0 / n
    g = np.ones(n)
    # h[j+s] contribution
    if s == 0:
        g -= h
    elif s > 0:
        g[:n-s] -= h[s:]
        # g[n-s:] unchanged (h[j+s] = 0)
    else:
        g[-s:] -= h[:n+s]
    # h[j-s] contribution (negative)
    if s == 0:
        g -= h
    elif s > 0:
        g[s:] -= h[:n-s]
    else:
        g[:n+s] -= h[-s:]
    return g * dx

    return best_h, best_c5

def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h)

def make_starts(n, parent, rng):
    starts = []
    if parent is not None:
        starts.append(project_feasible(upscale(parent, n), n))
    x = np.arange(n) / n
    for f in [2, 3, 4, 5]:
        starts.append(project_feasible(0.5 + 0.12 * np.cos(2.0 * np.pi * f * x), n))
    h_v = 1.0 - np.abs(2.0 * x - 1.0)
    starts.append(project_feasible(h_v, n))
    starts.append(project_feasible(0.5 + 0.08 * rng.randn(n), n))
    h_b = np.zeros(n)
    w = max(2, n // 8)
    h_b[n//4 - w//2 : n//4 + w//2 + 1] = 1.0
    h_b[3*n//4 - w//2 : 3*n//4 + w//2 + 1] = 1.0
    starts.append(project_feasible(h_b, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    best_c5 = 1.0
    best_h = None
    n_list = [42, 50, 60, 80, 100, 120, 150, 200]
    rng = np.random.RandomState(seed)

    for ni, n in enumerate(n_list):
        elapsed = time.time() - t0
        remaining = budget_s - 120 - elapsed
        if remaining <= 0:
            break
        n_remain = len(n_list) - ni
        time_per_n = max(20, remaining // n_remain)
        starts = make_starts(n, parent, rng)
        time_per_start = max(5, time_per_n // max(1, len(starts)))

        for si, h0 in enumerate(starts):
            if time.time() - t0 > budget_s - 60:
                break
            gd_iters = max(500, int(time_per_start * 400))
            h_gd, c5_gd = grad_descent(h0, n, gd_iters, seed=seed + si * 100)
            if c5_gd < best_c5:
                best_c5 = c5_gd
                best_h = h_gd.copy()
            ls_trials = max(500, int(time_per_start * 200))
            h_ls, c5_ls = local_search(h_gd, n, ls_trials, seed=seed + si * 200 + 1)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    if best_h is not None:
        n_best = len(best_h)
        h_f, c5_f = local_search(best_h, n_best, n_trials=4000, seed=77777)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        if time.time() - t0 < budget_s - 60:
            h_gd2, c5_gd2 = grad_descent(best_h, n_best, max_iter=2000, seed=88888)
            if c5_gd2 < best_c5:
                best_c5 = c5_gd2
                best_h = h_gd2
            h_ls2, c5_ls2 = local_search(best_h, n_best, n_trials=3000, seed=99999)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2

    if best_h is None:
        n = 100
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
