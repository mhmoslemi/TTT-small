def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective(h):
    c5 = compute_c5(h)
    # Penalty for constraint violation (sum should be n/2)
    n = len(h)
    viol = np.sum(h) - n / 2.0
    return c5 + 10.0 * viol ** 2

def objective_grad(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    
    grad = np.zeros(n)
    lo1 = max(0, -k)
    hi1 = min(n, n - k)
    if lo1 < hi1:
        grad[lo1:hi1] += 1.0 - h[lo1+k:hi1+k]
    lo2 = max(0, k)
    hi2 = min(n, n + k)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2-k:hi2-k]
    grad *= dx

    viol = np.sum(h) - n / 2.0
    grad += 20.0 * viol * np.ones(n)

    """Project h to satisfy 0<=h[i]<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    diff = n / 2.0 - np.sum(h)
    # Adjust by adding/subtracting from elements not at boundary
    # Simple: add diff/n to all
    h += diff / n
    h = np.clip(h, 0, 1)
    # Second pass
    diff = n / 2.0 - np.sum(h)
    mask_free = (h > 0.01) & (h < 0.99)
    if np.any(mask_free):
        h[mask_free] += diff / np.sum(mask_free)
    h = np.clip(h, 0, 1)
    # Final adjustment
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:

        sorted_idx = np.argsort(np.minimum(h, 1-h))[::-1]

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

    for _ in range(30):
        diff = 0.5 * n - np.sum(h)
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
        else:
            room = h
        free = room > 1e-10
        if not np.any(free):
            break
        per = diff / np.sum(free.astype(float))
        h[free] += per
        h = np.clip(h, 0.0, 1.0)
    diff = 0.5 * n - np.sum(h)
    if abs(diff) > 1e-10:
        idxs = np.argsort(np.minimum(h, 1.0 - h))[::-1]
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            step = np.sign(diff) * min(abs(diff), min(h[i], 1.0 - h[i]) + 1e-10)
            h[i] += step
            diff = 0.5 * n - np.sum(h)

    return proj_feasible(np.clip(h_new, 0, 1), new_n)

            i = pi // (n - 1)

                h2[i] -= sign * delta
                h2[j] += sign * delta

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def proj_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = 0.5 * n - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        free = room > 1e-10
        if not np.any(free):
            break
        per = diff / np.sum(free.astype(float))
        h[free] += per
        h = np.clip(h, 0.0, 1.0)
    diff = 0.5 * n - np.sum(h)
    if abs(diff) > 1e-10:
        idxs = np.argsort(np.minimum(h, 1.0 - h))[::-1]
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            mx = min(abs(h[i]), abs(1.0 - h[i]))
            step = np.sign(diff) * min(abs(diff), mx + 1e-10)
            h[i] += step
            diff = 0.5 * n - np.sum(h)
    return h


def upsample(h, factor=2):
    n = len(h)
    new_n = n * factor
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0.0, n - 1.0, new_n)
    h_new = np.interp(x_new, x_old, h)
    return proj_feasible(np.clip(h_new, 0.0, 1.0), new_n)


def pair_descent(h, n, max_sweeps=80, delta=0.015, seed=0):
    best_c = compute_c5(h)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        improved = False
        for _ in range(300):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for sign in (1, -1):
                h2 = h.copy()
                h2[i] -= sign * delta
                h2[j] += sign * delta
                if h2[i] < 0.0 or h2[i] > 1.0 or h2[j] < 0.0 or h2[j] > 1.0:
                    continue
                c2 = compute_c5(h2)
                if c2 < best_c - 1e-12:
                    h = h2
                    best_c = c2
                    improved = True
                    break
        if not improved:
            break
    return h, best_c


def sum_constraint(x):
    return np.sum(x) - 0.5 * len(x)


def sum_jacobian(x):
    return np.ones(len(x))


def obj_penalty(x):
    n = len(x)
    c5 = compute_c5(x)
    viol = np.sum(x) - 0.5 * n
    return c5 + 150.0 * viol * viol
