        js = np.arange(kp, n)
        grad[js] -= h[js - kp]

def upsample(h, target_n):
    n = len(h)
    if n == target_n:

def project(h, target_sum):

def compute_subgrad(h):

    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    mi = int(np.argmax(corr))
    k = mi - (n - 1)
    dx = 2.0 / float(n)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n - k])  # 1-h[j-k] for j in [k, n-1], i.e., h[0:n-k]
        if n - k > 0:

        if kp < n:
            g[kp:] -= h[:n - kp]  # -h[j-kp] for j in [kp, n-1]
    return g * dx, float(corr[mi] * dx)

import numpy as np
import time


def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def upsample(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) / float(n)
    x_new = np.arange(target_n) / float(target_n)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])


def project(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(200):
        np.clip(h, 0.0, 1.0, out=h)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_subgrad(h):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    mi = int(np.argmax(corr))
    k = mi - (n - 1)
    dx = 2.0 / float(n)
    g = np.zeros(n)
    if k >= 0:
        kk = min(k, n - 1)
        g[kk:] += (1.0 - h[:n - kk])
        g[:n - kk] -= h[kk:]
    else:
        kp = -k
        g[:n - kp] += (1.0 - h[kp:])
        g[kp:] -= h[:n - kp]
    return g * dx

            best_h = h_new.copy()
        h = h_new

            lr *= 0.6
    return best_h, best_c5

                best_h = h_pol.copy()

    # Fallback
    if best_h is None:
        best_h = np.full(100, 0.5)
        best_c5 = evaluate_c5(best_h)

    # Validate constraint
    n = len(best_h)
    target = n // 2
    if abs(best_h.sum() - target) > 1e-6:
        h_fix = np.round(best_h).astype(float)
        while h_fix.sum() > target:
            idx = rng.choice(np.where(h_fix == 1)[0])
            h_fix[idx] = 0
        while h_fix.sum() < target:
            idx = rng.choice(np.where(h_fix == 0)[0])
            h_fix[idx] = 1
        best_h = h_fix
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
