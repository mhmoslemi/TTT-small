import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def cs5_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(alpha*(corr - m), -700, 700)))/len(corr))/alpha

def grad_corr_k(h, shift):
    """Gradient of corr[k] w.r.t. h, where shift = k - (n-1)"""
    n = len(h)
    grad = np.zeros(n)
    # shift = k - (n-1)
    # corr[k] = sum_j h[j]*(1-h[j-shift])
    # d/dh[i] = (1-h[i-shift]) if valid + (-h[i+shift]) if valid
    for i in range(n):
        g = 0.0
        j1 = i - shift
        if 0 <= j1 < n:
            g += 1.0 - h[j1]
        j2 = i + shift
        if 0 <= j2 < n:
            g -= h[j2]
        grad[i] = g
    return grad

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    logits = np.clip(alpha * (corr - m), -700, 700)
    weights = np.exp(logits)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)

    if abs(diff) > 1e-14:
        free = (p > 1e-12) & (p < 1.0 - 1e-12)
        if free.any():
            p[free] += diff / float(free.sum())
            np.clip(p, 0.0, 1.0, out=p)
    return p

def subgrad_descent(h, n, steps, lr_init, t_end):
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr_init
    for it in range(steps):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = grad_corr_shift(h, s, n)
        h_new = h - lr * grad
        h_new = proj_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        lr *= 0.9995
    return best_h, best_c

def slsqp_run(h0, n, obj_fn, jac_fn, bounds, cons, maxiter, t_end):
    if time.time() > t_end:
        return h0, compute_c5(h0)
    kwargs = {'method': 'SLSQP', 'bounds': bounds, 'constraints': cons,
              'options': {'maxiter': maxiter, 'ftol': 1e-14}}
    if jac_fn is not None:
        kwargs['jac'] = jac_fn
    try:
        res = minimize(obj_fn, h0, **kwargs)
        if res.success or res.fun < compute_c5(h0):
            cand = proj_feasible(res.x, n)
            return cand, compute_c5(cand)
    except Exception:
        pass
    return h0, compute_c5(h0)

def pair_swap_cd(h, n, n_passes, t_end):
    target = n / 2.0
    best_c = compute_c5(h)
    for p in range(n_passes):
        if time.time() > t_end:
            break
        for i in range(n - 1):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                t_sum = h[i] + h[j]
                t_lo = max(0.0, t_sum - 1.0)
                t_hi = min(1.0, t_sum)
                if t_hi - t_lo < 1e-14:
                    continue
                best_t = h[i]
                best_f = compute_c5(h)
                for tval in [t_lo, t_hi, (t_lo + t_hi) * 0.5, h[i]]:
                    tval = float(np.clip(tval, t_lo, t_hi))
                    h[i] = tval
                    h[j] = t_sum - tval
                    f = compute_c5(h)
                    if f < best_f:
                        best_f = f
                        best_t = tval
                h[i] = best_t
                h[j] = t_sum - best_t
                if best_f < best_c:
                    best_c = best_f
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = proj_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(x.sum() - n / 2)}]

    # Phase 1: SLSQP with annealed smoothing
    for alpha in [30.0, 80.0, 200.0, 500.0, 1500.0]:
        if time.time() > t_end * 0.4:
            break
        obj = lambda x, a=alpha: smooth_obj(x, a)
        jac = lambda x, a=alpha: smooth_grad(x, a)
        best_h, best_c = slsqp_run(best_h, n, obj, jac, bounds, cons, 400, t_end * 0.95)

    # Phase 2: SLSQP on raw max (no smoothing)
    if time.time() < t_end * 0.5:
        best_h, best_c = slsqp_run(best_h, n, lambda x: compute_c5(x), None, bounds, cons, 600, t_end * 0.9)

    # Phase 3: Subgradient descent from best
    if time.time() < t_end * 0.65:
        h2, c2 = subgrad_descent(best_h, n, 5000, 0.003, t_end * 0.75)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 4: Pair-swap coordinate descent
    if time.time() < t_end * 0.75:
        best_h, best_c = pair_swap_cd(best_h, n, 3, t_end * 0.82)

    # Phase 5: More SLSQP refinement
    if time.time() < t_end * 0.85:
        obj = lambda x: smooth_obj(x, 1000.0)
        jac = lambda x: smooth_grad(x, 1000.0)
        best_h, best_c = slsqp_run(best_h, n, obj, jac, bounds, cons, 500, t_end * 0.92)

    # Phase 6: Random restarts
    while time.time() < t_end * 0.93:
        for scale in [0.05, 0.15, 0.3]:
            if time.time() > t_end * 0.93:
                break
            pert = np.random.randn(n) * scale
            h_pert = proj_feasible(best_h + pert, n)
            obj = lambda x: smooth_obj(x, 200.0)
            jac = lambda x: smooth_grad(x, 200.0)
            h_cand, c_cand = slsqp_run(h_pert, n, obj, jac, bounds, cons, 200, t_end * 0.94)
            if c_cand < best_c:
                best_h, best_c = h_cand, c_cand
            # Quick subgradient polish
            if c_cand <= best_c + 1e-6 and time.time() < t_end * 0.94:
                h_cand, c_cand = subgrad_descent(h_cand, n, 1000, 0.002, t_end * 0.94)
                if c_cand < best_c:
                    best_h, best_c = h_cand, c_cand

    # Final polish: project and verify
    best_h = proj_feasible(best_h, n)
    best_c = compute_c5(best_h)

    return best_h, best_c, n
