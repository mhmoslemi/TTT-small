def compute_c5(h):

def compute_gradient(h):
    """Subgradient of max correlation term"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        kp = -k
        grad[:n-kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n-kp]
    return grad * dx

def project_feasible(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2} using bisection"""
    target = n / 2.0
    # Bisection: find shift s such that sum(clip(h+s, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5*(lo+hi), 0.0, 1.0)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        kp = -k
        grad[:n-kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n-kp]

def project_feasible(h, n):
    target = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(50):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))

        else:

def upscale(h, n_old, n_new, method='rep'):
    if method == 'rep':
        idx = (np.arange(n_new) * n_old / n_new).astype(int)
        idx = np.minimum(idx, n_old - 1)
        h_new = h[idx].copy()
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_binary_pattern(n, pattern_len, ones_per_period, seed=0):

    h = np.zeros(n)

                    h[i] = 1.0
                    diff -= 1
        else:
            for i in range(n):
                if h[i] == 1 and diff < 0:
                    h[i] = 0.0
                    diff += 1
    return h

                break
            h_up = upscale(best_h, best_n, n_t, method='interp')
            h_opt, c5 = optimize_anneal(h_up, n_t, 12000, lr0=0.1, seed=seed + n_t)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt.copy(), n_t

    # Phase 3: Try structured binary initializations
    if remaining_frac() > 0.35:
        structures = [
            (10, 5), (14, 7), (20, 10), (22, 11), (24, 12),
            (28, 14), (30, 15), (34, 17), (40, 20)
        ]
        for pi, (plen, ones) in enumerate(structures):
            if remaining_frac() < 0.2:
                break
            n_r = max(45, (plen * 5) // 1 * 9)
            n_r = min(n_r, 360)
            h_r = make_binary_pattern(n_r, plen, ones, seed=seed + 100 + pi)
            h_r = project_feasible(h_r, n_r)
            h_opt, c5 = optimize_anneal(h_r, n_r, 8000, lr0=0.2, seed=seed + 100 + pi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt.copy(), n_r

    # Phase 4: Random continuous restarts
    if remaining_frac() > 0.2:
        for trial in range(20):
            if remaining_frac() < 0.1:
                break
            n_r = [45, 60, 90, 120, 180][trial % 5]
            t_norm = np.linspace(0, 2, n_r)
            freq = (trial + 1) * 0.7 + rng.rand() * 2.0
            h_r = 0.5 + 0.45 * np.sin(freq * t_norm + rng.rand() * 6.28)
            h_r = h_r + 0.3 * rng.randn(n_r)
            h_r = project_feasible(h_r, n_r)
            h_opt, c5 = optimize_anneal(h_r, n_r, 6000, lr0=0.15, seed=seed + 200 + trial)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt.copy(), n_r

    # Phase 5: Fine-tune best solution with smaller steps
    if best_h is not None and remaining_frac() > 0.05:
        iters = min(30000, int(remaining_frac() * budget_s * 50))
        h_opt, c5 = optimize(best_h, best_n, iters, lr0=0.03, seed=seed + 999)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Final: ensure constraint and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
