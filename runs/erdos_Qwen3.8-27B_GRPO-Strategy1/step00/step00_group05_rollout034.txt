import time

def _eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(h, n):
    """Project to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    d = n * 0.5 - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        order = np.argsort(1.0 - h, kind='stable')
    else:
        order = np.argsort(h, kind='stable')
        d = -d
    for idx in order:
        if d > 0:
            a = min(1.0 - h[idx], d)
            h[idx] += a
        else:
            a = min(h[idx], -d)
            h[idx] -= a
        d -= a
        if abs(d) < 1e-14:
            break
    return h

def _binary_local(h, n, dx, max_sweeps=10):

    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    # Ensure exactly n/2 ones

import numpy as np

def _eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = n * 0.5 - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        order = np.argsort(1.0 - h, kind='stable')
    else:
        order = np.argsort(h, kind='stable')
        d = -d
    for idx in order:
        if d > 0:
            a = min(1.0 - h[idx], d)
            h[idx] += a
        else:
            a = min(h[idx], -d)
            h[idx] -= a
        d -= a
        if abs(d) < 1e-14:
            break
    return h

def _binary_local(h, n, dx, rng, max_sweeps=5):

        improved = False
