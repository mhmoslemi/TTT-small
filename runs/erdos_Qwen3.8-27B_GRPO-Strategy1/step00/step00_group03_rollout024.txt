def make_feasible(h, n):

    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = n/2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the correction
        if diff > 0:
            # Need to increase sum

        else:

                h[idx] += add
                diff -= add

                h[idx] -= rem

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def ensure_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
