import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.minimum(h[m] + diff / c, 1.0)
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.maximum(h[m] + diff / c, 0.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(h - 0.5))
        for i in order:
            if abs(diff) < 1e-14:
                break
            a = np.clip(diff, -h[i], 1.0 - h[i])
            h[i] += a
            diff -= a
    return h

def smax(h, dx, tau):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    return m + np.log(np.exp(tau * (c - m)).sum()) / tau

def smax_grad(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    w = np.exp(tau * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        for j in range(n):
            if 0 <= j - s < n:
                g[j] += wk * (1.0 - h[j - s])
            if 0 <= j + s < n:
                g[j] += wk * (-h[j + s])
    return g

def lfbgsopt(h, dx, n, tau, mu, maxiter, t0, budget):
    target = 0.5 * n
    def obj(x):
        return smax(x, dx, tau) + mu * (x.sum() - target) ** 2
    def jac(x):
        return smax_grad(x, dx, tau) + 2 * mu * (x.sum() - target)
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def proj_grad_descent(h, dx, n, tau, t0, budget):
    lr = 0.005
    target = 0.5 * n
    c_cur = compute_c5(h, dx)
    for it in range(20000):
        if time.time() - t0 > budget - 5:
            break
        g = smax_grad(h, dx, tau) + 2e-3 * (h.sum() - target)
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, dx)
        if c_new < c_cur - 1e-15:
            c_cur = c_new
            h = h_new
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.9
            if lr < 1e-5:
                break
    return h

def coord_descent(h, dx, n, t0, budget):
    best = compute_c5(h, dx)
    deltas = [0.0005, -0.0005, 0.002, -0.002, 0.01, -0.01, 0.03, -0.03]
    for sweep in range(50):
        if time.time() - t0 > budget - 5:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 5:
                break
            for d in deltas:
                h2 = h.copy()
                h2[j] += d
                h2 = project_feasible(h2, n)
                c = compute_c5(h2, dx)
                if c < best - 1e-15:
                    best = c
                    h[:] = h2
                    improved = True
                    break
        if not improved:
            break
    return h, best

def binary_push(h, dx, n, t0, budget):
    best = compute_c5(h, dx)
    best_h = h.copy()
    target = 0.5 * n
    for _ in range(2000):
        if time.time() - t0 > budget - 5:
            break
        j = np.random.randint(n)
        if h[j] > 0.5:

        else:
            h[j] = 0.0
        h = project_feasible(h, n)
        c = compute_c5(h, dx)
        if c < best - 1e-14:
            best = c
            best_h = h.copy()
        else:
            h = best_h.copy()
    return best_h, best

def random_binary_start(n, rng):
    h = np.zeros(n)
    num_ones = int(n // 2)
    idx = rng.permutation(n)
    h[idx[:num_ones]] = 1.0
    if n % 2 == 1:
        h[idx[num_ones]] = 0.5
    return h
