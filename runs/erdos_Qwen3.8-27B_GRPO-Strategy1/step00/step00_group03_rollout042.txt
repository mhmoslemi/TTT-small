```python
def project_box_sum(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}."""
    n = len(h)
    # Binary search on lambda
    lo, hi = h.min() - target_sum, h.max()  # rough bounds
    # Actually: we want sum(clip(h - lambda, 0, 1)) = target_sum
    # sum(clip(h - lambda, 0, 1)) is monotonically decreasing in lambda
    # When lambda -> -inf, sum -> n
    # When lambda -> +inf, sum -> 0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-12:
            break
    return np.clip(h - (lo + hi) / 2, 0, 1)
```

```python
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    # Ensure feasibility
    h = project_box_sum(h, target_sum)
    
    best_h = h.copy()
    best_c = compute_c5(h)
    
    alpha = 0.1
    for it in range(n_max_iter):
        if budget_check():
            break
        
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_val = np.max(corr)
        
        # Find active constraints (top few)
        thresh = c_val - c_val * 0.05
        active = np.where(corr >= thresh)[0]
        
        # Compute gradient (average over active constraints)
        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            for j in range(n):
                g = 1.0
                if 0 <= j - k < n:
                    g -= h[j - k]
                if 0 <= j + k < n:
                    g -= h[j + k]
                grad[j] += g
        grad /= len(active)
        
        # Line search
        for _ in range(20):
            h_new = project_box_sum(h - alpha * grad, target_sum)

            alpha *= 0.5
            if alpha < 1e-8:
                break
        else:
            alpha *= 0.9
            
        if alpha < 1e-10:
```

```python
    else:
```
