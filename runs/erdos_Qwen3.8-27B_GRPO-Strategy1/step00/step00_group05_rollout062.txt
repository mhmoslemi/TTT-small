import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, dx, beta, target_sum, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    smooth_max = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    viol = np.sum(h) - target_sum
    return smooth_max * dx + penalty * viol * viol

def upsample_to(parent, new_n):
    old_n = len(parent)
    if new_n == old_n:
        return parent.astype(float).copy()
    idx = np.floor(np.linspace(0, old_n, new_n + 1)[:-1]).astype(int)
    h = parent[idx].astype(float)
    target = new_n / 2.0
    h += (target - np.sum(h)) / new_n
    return np.clip(h, 0.0, 1.0)

def fix_constraints(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h += (target - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

    order = np.argsort(-h)

    return h_b
