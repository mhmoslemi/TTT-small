def project_to_feasible(h):
    """Project h to satisfy sum(h) = n/2, 0 <= h <= 1"""
    n = len(h)
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        # Add to those < 1, subtract from those > 0
        if diff > 0:
            candidates = np.where(h < 1)[0]
            room = 1 - h[candidates]
            total_room = room.sum()
            if total_room > 0:
                frac = min(diff / total_room, 1)
                h[candidates] += room * frac
            else:
                h += diff / n
        else:
            candidates = np.where(h > 0)[0]
            room = h[candidates]
            total_room = room.sum()
            if total_room > 0:
                frac = min(-diff / total_room, 1)
                h[candidates] -= room * frac
            else:
                h += diff / n
    h = np.clip(h, 0, 1)
    return h

    best_c = compute_c5(h)
    for step in range(n_steps):
