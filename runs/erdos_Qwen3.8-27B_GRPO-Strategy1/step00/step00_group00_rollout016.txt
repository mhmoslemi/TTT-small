def eval_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_worst(h, worst_idx):
    n = len(h)
    k = worst_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
    else:
        kk = -k
        if n - kk > 0:
            grad[:n - kk] += (1.0 - h[kk:])
            grad[kk:] -= h[:n - kk]
    return grad
