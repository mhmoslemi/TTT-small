import numpy as np
from scipy.optimize import minimize
import time

def clip_sum_project(h, S):
    """Project h onto {x: 0<=x_i<=1, sum(x)=S} via bisection."""
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - S
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_fg(h, beta):
    """Smoothed max-correlate objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    mx = np.max(vals)
    e = np.exp(beta * (vals - mx))
    se = np.sum(e)
    w = e / se
    f = mx + np.log(se) / beta
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        lo1, hi1 = max(0, -k), min(n, n - k)
        for j in range(lo1, hi1):
            grad[j] += wm * (1.0 - h[j + k])
        lo2, hi2 = max(0, k), min(n, n + k)
        for j in range(lo2, hi2):
            grad[j] -= wm * h[j - k]
    grad *= dx
    return f, grad

def solve_slsqp(h0, S, beta, maxiter=300):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - S, 'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_fg, h0, args=(beta,), jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = np.clip(res.x, 0, 1)
    except:
        h_out = h0.copy()
    h_out = clip_sum_project(h_out, S)
    return h_out, c5_of(h_out)

def pgd_refine(h0, S, n_iter, lr, time_limit=40):
    t_start = time.time()
    h = h0.astype(float).copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = c5_of(h)
    for it in range(n_iter):
        if it % 5000 == 4999 and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = int(np.argmax(corr))
        k = m - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j + k < n:
                grad[j] += (1.0 - h[j + k])
            if 0 <= j - k < n:
                grad[j] -= h[j - k]
        grad *= dx
        h = clip_sum_project(h - lr * grad, S)
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        if it % 3000 == 2999:
            lr *= 0.92
    return best_h, best_c5
