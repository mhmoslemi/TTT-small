def project_simplex(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_score(h, n):
    return float(np.max(corr_vec(h)) * (2.0 / n))

def slopes_for_j(h, j):
    n = len(h)
    m = 2 * n - 1
    slopes = np.zeros(m)
    for k in range(m):
        i1 = j - k + n - 1
        i2 = j + k - n + 1
        s = 0.0
        if 0 <= i1 < n:
            s += 1.0 - h[i1]
        if 0 <= i2 < n:
            s -= h[i2]
        slopes[k] = s
    return slopes

def coord_desc(h, n, passes=20, n_grid=50):
    target = float(n) / 2.0
    for _ in range(passes):
        max_gain = 0.0
        for j in range(n):
            c = corr_vec(h)
            sl = slopes_for_j(h, j)
            old = h[j]
            best_t = old
            best_obj = np.max(c)
            for t in np.linspace(0.0, 1.0, n_grid):
                obj = np.max(c + (t - old) * sl)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_t = t
            if best_t != old:
                max_gain = max(max_gain, abs(best_t - old))
                h[j] = best_t
        h = project_simplex(h, target)
        if max_gain < 1e-8:
            break
    return h

def softmax_grad_descent(h, n, lr, alpha, iters=100):
    target = float(n) / 2.0
    for _ in range(iters):
        c = corr_vec(h)
        # Softmax weights
        ac = alpha * (c - np.max(c))
        p = np.exp(ac) / np.exp(ac).sum()
        # Weighted gradient
        m = len(c)
        grad = np.zeros(n)
        for k in range(m):
            if p[k] > 1e-10:
                for j in range(n):

                    i2 = j + k - n + 1
                    g = 0.0
                    if 0 <= i1 < n:
                        g += 1.0 - h[i1]
                    if 0 <= i2 < n:
                        g -= h[i2]
                    grad[j] += p[k] * g
        h = h - lr * grad
        h = project_simplex(h, target)
    return h
