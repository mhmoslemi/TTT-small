from scipy.optimize import minimize
import time

_g_n = 45

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h):
    n = _g_n
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = float(np.sum(h))
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
            c = int(np.sum(m))
            if c == 0:
                break
            h[m] = np.minimum(h[m] + d / c, 1.0)
        else:
            m = h > 1e-15
            c = int(np.sum(m))
            if c == 0:
                break
            h[m] = np.maximum(h[m] - d / c, 0.0)
    return h

def _mm_obj(v):
    h = v[:_g_n]
    t = v[_g_n]
    dx = 2.0 / _g_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr) * dx)
    if mc > t:
        return t + 500.0 * (mc - t) ** 2
    return t

def _eq_con(v):
    return float(np.sum(v[:_g_n]) - _g_n / 2.0)

def _penalty_c5(h):
    c5 = compute_c5(h)
    s = float(np.sum(h))
    target = _g_n / 2.0
    pen = (s - target) ** 2 * 1e4
    lo = np.sum(np.minimum(h, 0.0) ** 2)
    hi = np.sum(np.maximum(h - 1.0, 0.0) ** 2)
    pen += 1e4 * (lo + hi)
    return c5 + pen

def _try_slsqp(x0_h, bounds, cons):

    t_init = compute_c5(x0_h) + 0.05
    xv = np.concatenate([x0_h, [t_init]])
    try:
        res = minimize(_mm_obj, xv, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 400, 'ftol': 1e-12})
        return repair_sum(res.x[:n])
    except Exception:
        return None

            break

            c5_try2 = compute_c5(h_try2)
            if c5_try2 < c5_best:

                h_best = h_try2
    return h_best, c5_best
