def upsample(h_init, n_target):
    n_init = len(h_init)
    if n_init == n_target:
        return h_init.copy()
    # Linear interpolation and threshold
    x_old = np.linspace(0, 2, n_init, endpoint=False)
    x_new = np.linspace(0, 2, n_target, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_init)
    # Threshold to binary, then fix sum
    h_bin = (h_interp >= 0.5).astype(float)
    # Fix sum to n_target/2
    target_sum = n_target // 2
    current_sum = int(h_bin.sum())
    if current_sum > target_sum:
        # Flip some 1s to 0s (the ones closest to 0.5)
        ones = np.where(h_bin == 1)[0]
        # Sort by h_interp value (lowest first)
        order = np.argsort(h_interp[ones])
        for idx in order[:current_sum - target_sum]:
            h_bin[ones[idx]] = 0
    elif current_sum < target_sum:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_interp[zeros])
        for idx in order[:target_sum - current_sum]:
            h_bin[zeros[idx]] = 1
    return h_bin

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def upsample_binary(h_init, n_target):
    n_init = len(h_init)
    x_old = np.arange(n_init) / n_init * 2.0
    x_new = np.arange(n_target) / n_target * 2.0
    h_interp = np.interp(x_new, x_old, h_init)
    h_bin = (h_interp >= 0.5).astype(float)
    target = n_target // 2
    cur = int(h_bin.sum())
    if cur > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h_interp[idx])
        h_bin[idx[order[:cur - target]]] = 0
    elif cur < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h_interp[idx])
        h_bin[idx[order[:target - cur]]] = 1
    return h_bin

def steepest_descent(h, max_rounds, deadline):
    n = len(h)
    h = h.copy()
    c5 = evaluate_c5(h)
    
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_impr = 1e-15
        best_pair = None
        best_val = c5
        
        n_swaps = min(len(ones) * len(zeros), 3000)
        if n_swaps <= 3000:
            pairs = [(i, j) for i in ones for j in zeros]
        else:
            rng = np.random.RandomState(int(time.time() * 1000000) % 2**31)
            pairs = [(ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]) for _ in range(3000)]
        
        for i, j in pairs:
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            v = evaluate_c5(h)
            h[i], h[j] = 1.0, 0.0
            if v < best_val:
                best_val = v
                best_pair = (i, j)
                best_impr = c5 - v
        
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = 0.0, 1.0
        c5 = best_val
    
    return h, c5

    # Final verification and fix
    if best_h is None:
        best_h = thue_morse_bin(64)
        best_c5 = evaluate_c5(best_h)
    n_final = len(best_h)
    tgt = n_final // 2
    c = int(best_h.sum())
    if c != tgt:
        if c > tgt:
            idx = np.where(best_h == 1)[0]
            np.random.shuffle(idx)
            best_h[idx[:c - tgt]] = 0
        else:
            idx = np.where(best_h == 0)[0]
            np.random.shuffle(idx)
            best_h[idx[:tgt - c]] = 1
        best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, len(best_h))
