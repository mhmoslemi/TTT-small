        term2 = np.zeros(n)

        hi2 = min(n, n - s)
        term2[lo2:hi2] = h[lo2+s:hi2+s]
        grad += term1 - term2

def compute_c5_and_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def project_box_sum(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    h = np.array(h, dtype=float)
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    proj = np.clip(h - mid, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # Adjust non-boundary elements
        mask = (proj > 1e-10) & (proj < 1 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def subgradient(h, active_shifts):
    """Compute subgradient of max active shift sum."""
    n = len(h)

    for s in active_shifts:

            t2[s_abs:] = h[:n-s_abs]
        grad += t1 - t2
    grad /= len(active_shifts)
    return grad
