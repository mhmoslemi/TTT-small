import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def subgradient(h, corr, n, top_k=5):
    """Compute subgradient of max(corr) w.r.t. h"""
    grad = np.zeros(n)
    max_val = np.max(corr)
    # Lags near maximum
    candidates = np.argpartition(-corr, min(top_k, len(corr)))[:top_k]
    for m in candidates:
        p = m - (n - 1)
        if p >= 0:
            j_lo, j_hi = 0, n - p
        else:
            j_lo, j_hi = -p, n
        j = np.arange(j_lo, j_hi)
        jp = j + p
        grad[j] += (1.0 - h[jp])
        if p != 0:
            grad[jp] -= h[j]
    grad /= len(candidates)
    return grad

So h(x+0.5)=1 on [0.5,1].
h(x)=1 on [0,0.5] ∪ [1,1.5].
Overlap: [0,0.5]∩[0.5,1] = ∅, [1,1.5]∩[0.5,1] = ∅.
So overlap is 0, and ∫h(1-h(x+0.5)) = 1*1 = 1 (since all of h(x)=1 has h(x+0.5)=0).

So for any single interval, C₅ = 1 (at large shifts). The issue is that for large shifts, the overlap goes to 0.

The fundamental problem: for the "full" correlate, at step size k (in index), the number of overlapping terms decreases. At the maximum shift (k = n-1), only 1 term overlaps, and the value is h[0]*(1-h[n-1]) or h[n-1]*(1-h[0]).

So for any h, c[n-1] = h[0](1-h[n-1]) and c[0] = h[n-1](1-h[0]).

For h[0]=h[n-1]=0.5: this equals 1 - 0.5 = 0.5.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr

def project_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1.0 - 1e-14) if d > 0 else (h > 1e-14)
        if not m.any():
            break
        h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def subgrad_for_top(h, corr, n, top_k=8):
    grad = np.zeros(n)
    idx = np.argsort(-corr)[:top_k]
    for m in idx:
        p = m - (n - 1)
        if p >= 0:
            j0, j1 = 0, n - p
        else:
            j0, j1 = -p, n
        if j1 - j0 <= 0:
            continue
        j = np.arange(j0, j1)
        jp = j + p
        grad[j] += (1.0 - h[jp])
        if p != 0:
            grad[jp] -= h[j]
    grad /= len(idx)
    return grad

def coordinate_descend(h, tgt, n, max_iter=200):
    c5, _ = compute_c5(h)
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            old = h[i]
            for dv in [0.5, -0.5, 0.25, -0.25, 0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
                h[i] = np.clip(old + dv, 0.0, 1.0)
                h = project_sum(h, tgt)
                c5_new, _ = compute_c5(h)
                if c5_new < c5 - 1e-13:
                    c5 = c5_new
                    improved = True
                else:
                    h[i] = np.clip(old, 0.0, 1.0)
                    h = project_sum(h, tgt)
        if not improved:
            break
    return h, c5

def optimize_single(h, tgt, n, t0, budget_s):
    c5, _ = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    lr = 0.04
    mom = 0.85
    vel = np.zeros(n)
    stall = 0
    
    while time.time() - t0 < budget_s - 8:
        _, corr = compute_c5(best_h)
        grad = subgrad_for_top(best_h, corr, n, top_k=10)
        vel = mom * vel - lr * grad
        h_new = project_sum(best_h + vel, tgt)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            c5 = c5_new
            best_h = h_new
            if c5 < best_c5:
                best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall > 50:
                h_kick = project_sum(best_h + np.random.normal(0, lr * 0.6, n), tgt)
                c5_k, _ = compute_c5(h_kick)
                if c5_k < best_c5:
                    best_c5 = c5_k
                    best_h = h_kick
                stall = 0
                lr = max(0.005, lr * 0.7)
                vel *= 0.1
        lr *= 0.9998
    
    # Coordinate descent polish
    h_cd, c5_cd = coordinate_descend(best_h, tgt, n, max_iter=100)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
