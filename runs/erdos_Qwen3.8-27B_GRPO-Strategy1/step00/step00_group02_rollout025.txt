        if time.time() - t0 > budget_s - 30:
            break
        obj = C5TopSum(tk)
        try:
            res = minimize(
                fun=obj.obj, x0=best_h.copy(), jac=obj.grad,
                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-14}
            )
            h_c = fix_integral(res.x, n0)
            c5 = compute_c5(h_c)
            if c5 < best_c5:
                best_h = h_c
                best_c5 = c5
        except Exception:
            pass

    # Phase 3: Perturbed restarts
    for restart in range(5):
        if time.time() - t0 > budget_s - 60:
            break
        h_pert = best_h + rng.normal(0, 0.02, n0)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = fix_integral(h_pert, n0)
        obj = C5SmoothMax(300, 3)
        try:
            res = minimize(
                fun=obj.obj, x0=h_pert, jac=obj.grad,
                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-14}
            )
            h_c = fix_integral(res.x, n0)
            c5 = compute_c5(h_c)
            if c5 < best_c5:
                best_h = h_c
                best_c5 = c5
        except Exception:
            pass

    # Phase 4: Coordinate descent polish
    for it in range(3000):
        if time.time() - t0 > budget_s - 15:
            break
        idx1 = rng.randint(n0)
        idx2 = rng.randint(n0)
        if idx1 == idx2:
            continue
        step = rng.choice([-1, 1]) * rng.uniform(0.003, 0.02)
        h_trial = best_h.copy()
        h_trial[idx1] += step
        h_trial[idx2] -= step
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = fix_integral(h_trial, n0)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-12:
            best_h = h_trial
            best_c5 = c5_trial

    # Phase 5: Higher resolution refinement
    if time.time() - t0 < budget_s - 90:
        n1 = min(n0 * 2, 180)
        x_old = np.linspace(0, 2, n0, endpoint=False) + 1.0 / (2.0 * n0)
        x_new = np.linspace(0, 2, n1, endpoint=False) + 1.0 / (2.0 * n1)
        f_int = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_high = np.clip(f_int(x_new), 0.0, 1.0)
        h_high = fix_integral(h_high, n1)

        bounds1 = [(0.0, 1.0)] * n1
        ic1 = IntegralConstraint(n1)
        constraints1 = [{'type': 'eq', 'fun': ic1.fun, 'jac': ic1.jac}]

        for beta in [100, 300]:
            if time.time() - t0 > budget_s - 30:
                break
            obj = C5SmoothMax(beta, 3)
            try:
                res = minimize(
                    fun=obj.obj, x0=h_high.copy(), jac=obj.grad,
                    method='SLSQP', bounds=bounds1,
                    constraints=constraints1,
                    options={'maxiter': 300, 'ftol': 1e-14}
                )
                h_c = fix_integral(res.x, n1)
                c5 = compute_c5(h_c)
                if c5 < best_c5:
                    best_h = h_c
                    best_c5 = c5
            except Exception:
                pass

    # Final SLSQP polish on best
    if time.time() - t0 < budget_s - 5:
        n_final = len(best_h)
        bounds_f = [(0.0, 1.0)] * n_final
        ic_f = IntegralConstraint(n_final)
        cons_f = [{'type': 'eq', 'fun': ic_f.fun, 'jac': ic_f.jac}]
        obj = C5SmoothMax(500, 3)
        try:
            res = minimize(
                fun=obj.obj, x0=best_h, jac=obj.grad,
                method='SLSQP', bounds=bounds_f, constraints=cons_f,
                options={'maxiter': 200, 'ftol': 1e-14}
            )
            h_c = fix_integral(res.x, n_final)
            c5 = compute_c5(h_c)
            if c5 < best_c5:
                best_h = h_c
                best_c5 = c5
        except Exception:
            pass

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
