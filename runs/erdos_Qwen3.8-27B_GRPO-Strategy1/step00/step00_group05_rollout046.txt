import numpy as np
import time


def compute_c5(h, n_points=None):
    if n_points is None:
        n_points = len(h)
    dx = 2.0 / n_points
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * dx)


def make_binary(h):
    n = len(h)
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    target = n // 2
    if s < target:
        zeros = np.where(h == 0)[0]
        perm = zeros[np.random.default_rng(1).permutation(len(zeros))]
        for k in range(target - s):
            h[perm[k]] = 1.0
    elif s > target:
        ones = np.where(h == 1)[0]
        perm = ones[np.random.default_rng(2).permutation(len(ones))]
        for k in range(s - target):
            h[perm[k]] = 0.0
    return h


def upsample(h, factor=2):
    return np.repeat(h, factor)


def local_search(h, time_limit=30.0):
    n = len(h)
    h = make_binary(h.copy())
    best_c5 = compute_c5(h, n)
    start = time.time()
    while time.time() - start < time_limit:
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        best_swap = None
        best_new = best_c5
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h, n)
                h[i] = 1.0
                h[j] = 0.0
                if c5 < best_new - 1e-12:
                    best_new = c5
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_new
        else:
            break
    return h, best_c5


def anneal(h, time_limit=20.0, t0=0.05, cooling=0.97, seed=42):
    n = len(h)
    h = make_binary(h.copy())
    best_h = h.copy()

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
    return best_h, best_c5


def random_binary(n, seed=0):
    rng = np.random.default_rng(seed)
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def perturb(h, n_flips, seed=0):
    h = h.copy()

        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
    return h
