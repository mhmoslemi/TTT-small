                hp = fix_sum(hp, target_sum)
                x0 = np.append(hp, evaluate_c5(hp, dx) + 0.01)
            try:
                res = minimize(slsqp_obj, x0, method='SLSQP', bounds=bnds,
                              constraints=[nc_ineq, nc_eq],
                              options={'maxiter': 2000, 'ftol': 1e-14})
                hx = np.clip(res.x[:n], 0.0, 1.0)
                hx = fix_sum(hx, target_sum)
                c5x = evaluate_c5(hx, dx)
                if c5x < best_c5:
                    best_c5 = c5x
                    best_h = hx
            except Exception:
                pass

    # Phase 3: Final coordinate descent polish
    if time.time() - t0 < budget_s - 10:
        h = best_h.copy()
        def time_left3():
            return time.time() - t0 < budget_s - 5
        h, best_h, best_c5 = coord_descent(h, best_h, best_c5, dx, rng, 15, time_left3)

    # Ensure final feasibility
    best_h = fix_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h, dx)

    return (best_h, best_c5, n)
