import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def swap_perturb(h, rng, max_step=0.5):
    n = len(h)
    h_new = h.copy()
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h_new
    if rng.random() < 0.5:
        mt = min(h_new[i], 1.0 - h_new[j])
        if mt <= 0:
            return h_new
        t = rng.random() * max_step * mt
        h_new[i] -= t
        h_new[j] += t
    else:
        mt = min(h_new[j], 1.0 - h_new[i])
        if mt <= 0:
            return h_new
        t = rng.random() * max_step * mt
        h_new[j] -= t
        h_new[i] += t
    return h_new


def ensure_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize_heuristic(h, target):
    n = len(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0).copy()
    diff = target - np.sum(h_bin)
    if diff > 1e-9:
        zi = np.where(h_bin == 0)[0]
        zi = zi[np.argsort(-h[zi])]
        nf = int(diff)
        if nf <= len(zi):
            h_bin[zi[:nf]] = 1.0
            frac = diff - nf
            if frac > 1e-9 and nf < len(zi):
                h_bin[zi[nf]] = frac
    elif diff < -1e-9:
        oi = np.where(h_bin == 1)[0]
        oi = oi[np.argsort(h[oi])]
        nf = int(-diff)
        if nf <= len(oi):
            h_bin[oi[:nf]] = 0.0
            frac = -diff - nf
            if frac > 1e-9 and nf < len(oi):
                h_bin[oi[nf]] = 1.0 - frac
    return h_bin
