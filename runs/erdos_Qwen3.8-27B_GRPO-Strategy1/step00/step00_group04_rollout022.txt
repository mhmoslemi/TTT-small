    m = np.max(vals)
    if m < 1e-12:
        w = np.zeros(len(vals))
        w[np.argmax(vals)] = 1.0
        return m, w
    shifted = beta * (vals - m)
    shifted = np.minimum(shifted, 500)
    shifted = np.maximum(shifted, -500)
    exp_v = np.exp(shifted)
    s = np.sum(exp_v)
    result = m + np.log(s) / beta
    w = exp_v / s
    return result, w

    dx = 2.0 / n
    lags = compute_all_lags(h)
    obj, w = smoothed_max_and_weights(lags, beta)

    exp_v = np.exp(shifted)
    s = np.sum(exp_v)
    obj = m + np.log(s) / beta
    w = exp_v / s
    grad = np.zeros(n)
    nm1 = n - 1
    for ii in range(2 * n - 1):
        if w[ii] < 1e-10:
            continue
        lag = ii - nm1
        wdx = w[ii] * dx
        if lag >= 0:
            L = n - lag
            if L > 0:
                grad[lag:] += wdx * (1.0 - h[:L])
                grad[:L] -= wdx * h[lag:]
        else:
            kk = -lag
            L = n - kk
            if L > 0:
                grad[:L] += wdx * (1.0 - h[kk:])
                grad[kk:] -= wdx * h[:L]
    return obj, grad

def project_sum_box(x, target):
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def upsample_nn(h, new_n):
    n = len(h)
    old_dx = 2.0 / n
    old_x = (np.arange(n) + 0.5) * old_dx
    new_dx = 2.0 / new_n
    new_x = (np.arange(new_n) + 0.5) * new_dx
    idx = np.searchsorted(old_x, new_x, side='right') - 1
    idx = np.clip(idx, 0, n - 1)
    return project_sum_box(h[idx].astype(float), new_n * 0.5)

def pg_opt(h0, t_budget):
    n = len(h0)
    target = n * 0.5
    h = project_sum_box(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr0 = 0.08
    while time.time() - t0 < t_budget:
        elapsed = time.time() - t0
        frac = elapsed / t_budget
        if frac >= 1.0:
            break
        beta = 20.0 + 280.0 * frac * frac
        obj, grad = obj_and_grad(h, beta)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        step = lr0 * np.exp(-2.0 * frac)
        for _ in range(30):
            h_new = project_sum_box(h - step * grad, target)
            if np.max(np.abs(h_new - h)) < 1e-15:
                break
            obj_new, _ = obj_and_grad(h_new, beta)
            if obj_new <= obj + 1e-13:
                h = h_new
                break
            step *= 0.5
        else:
            h = project_sum_box(h - step * grad, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 42
        h0 = np.zeros(n0)
        h0[::2] = 1.0
        h0 = project_sum_box(h0, n0 * 0.5)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = len(h0)

    def try_update(h_op, c5_op):
        nonlocal best_h, best_c5, best_n
        if c5_op < best_c5 - 1e-12:
            best_c5 = c5_op
            best_h = h_op.copy()
            best_n = len(h_op)

    # Phase 1: optimize at original resolution
    rem = budget_s - (time.time() - t_start)
    if rem > 20:
        tb = min(40.0, rem * 0.06)
        h_op, c5_op = pg_opt(h0, tb)
        try_update(h_op, c5_op)

    # Phase 2: coarse-to-fine
    for res in [84, 168, 336]:
        rem = budget_s - (time.time() - t_start)
        if rem < 30:
            break
        tb = min(rem * 0.22, 140.0)
        h_up = upsample_nn(best_h, res)
        h_op, c5_op = pg_opt(h_up, tb)
        try_update(h_op, c5_op)

    # Phase 3: random restarts
    rem = budget_s - (time.time() - t_start)
    for r in range(5):
        rem = budget_s - (time.time() - t_start)
        if rem < 35:
            break
        n_r = 168
        h_base = upsample_nn(best_h, n_r)
        noise = 0.12 * np.random.randn(n_r)
        h_p = project_sum_box(h_base + noise, n_r * 0.5)
        tb = min(rem * 0.18, 70.0)
        h_op, c5_op = pg_opt(h_p, tb)
        try_update(h_op, c5_op)

    # Phase 4: final polish
    rem = budget_s - (time.time() - t_start)
    if rem > 15:
        tb = min(rem * 0.5, 80.0)
        h_op, c5_op = pg_opt(best_h, tb)
        try_update(h_op, c5_op)

    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final
    return (best_h, best_c5, best_n)
