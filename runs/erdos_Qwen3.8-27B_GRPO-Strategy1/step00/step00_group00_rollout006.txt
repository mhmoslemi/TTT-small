import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

    h = np.clip(h.astype(float), 0.0, 1.0)
    d = S - h.sum()
    if abs(d) < 1e-13:
        return h
    if d > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(1.0 - h[i], d)
            h[i] += add
            d -= add
            if d < 1e-13:
                break
    else:
        idx = np.argsort(-h)
        for i in idx:
            sub = min(h[i], -d)
            h[i] -= sub
            d += sub
            if d > -1e-13:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def c5_obj(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project(h, S):
    h = np.clip(np.array(h, dtype=np.float64), 0.0, 1.0)
    d = S - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(1.0 - h[i], d)
            h[i] += add
            d -= add
            if d < 1e-14:
                break
    else:
        idx = np.argsort(-h)
        for i in idx:
            sub = min(h[i], -d)
            h[i] -= sub
            d += sub
            if d > -1e-14:
                break
    return h

def make_binary_start(n, S, rng):
    h = np.zeros(n)
    h[:int(S)] = 1.0
    frac = S - int(S)
    if frac > 1e-10:
        h[int(S)] = frac
    perm = rng.permutation(n)
    h = h[perm]
    return h
