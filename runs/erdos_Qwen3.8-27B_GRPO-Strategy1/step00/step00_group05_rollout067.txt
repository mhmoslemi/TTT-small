import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = np.sum(interior)
    if cnt > 0:
        h[interior] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = max(diff, -h[i])
                h[i] += a
                diff -= a
            if abs(diff) < 1e-14:
                break
    return h

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def smooth_max_cc(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) * beta
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta

def make_bounds(n):
    return [(0.0, 1.0)] * n

        def obj(x):
            return smooth_max_cc(x, beta)

        obj = compute_c5

    res = minimize(obj, h_init, method='SLSQP', bounds=make_bounds(n),
                   constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13})
