    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""

            hc = project_feasible(res.x, _target_sum)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc

    # Phase 4: Very fine local search
    for i in range(600):
        if time.time() - t0 > budget_s - 5:
            break
        x0 = project_feasible(best_h + rng.normal(0, 0.002, n), _target_sum)
        cons = make_constraints(n)
        res = minimize(compute_c5, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 300, 'ftol': 1e-17})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hc = project_feasible(res.x, _target_sum)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc

    # Final: nudge individual coordinates for last improvements
    for i in range(200):
        if time.time() - t0 > budget_s - 1:
            break
        idx = rng.randint(n)
        for delta in [0.001, -0.001, 0.0005, -0.0005, 0.01, -0.01]:
            trial = best_h.copy()
            trial[idx] += delta
            trial = project_feasible(trial, _target_sum)
            c = compute_c5(trial)
            if c < best_c5:
                best_c5, best_h = c, trial

    best_h = project_feasible(best_h, _target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
