from scipy.optimize import minimize

def c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        room = (1.0 - h) if diff > 0 else h.copy()
        tr = np.sum(room)
        if tr < 1e-14:
            break
        h = np.clip(h + (diff / tr) * room, 0.0, 1.0)
    return h

def con_fun(x):
    return np.sum(x) - len(x) / 2.0

def try_binary(h, target):
    n = len(h)
    h_r = np.where(h > 0.5, 1.0, 0.0)
    diff = target - np.sum(h_r)
    if abs(diff) < 1e-12:
        return h_r
    if diff > 0:
        idxs = np.where(h_r == 0)[0]
        order = idxs[np.argsort(-h[idxs])]
        d = diff
        for i in order:
            if d < 1e-12:

            a = min(d, 1.0)
            h_r[i] = a
            d -= a
    else:

        order = idxs[np.argsort(h[idxs])]

        for i in order:
            if d < 1e-12:
                break
