def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(mask.sum())

            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            idx = int(h.argmin()) if diff > 0 else int(h.argmax())
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
