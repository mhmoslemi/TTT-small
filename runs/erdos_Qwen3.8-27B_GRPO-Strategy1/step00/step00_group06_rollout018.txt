import numpy as np
import time

def compute_C5_full(h, n):
    """Compute C5 = max(correlation * dx)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_to_sum(h, target_sum):
    """Project h to have exact sum, maintaining [0,1] bounds"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add diff
        can_add = 1.0 - h
        mask = can_add > 1e-12
        if mask.sum() == 0:
            return h
        frac = diff / can_add[mask].sum()
        if frac > 1:
            h[mask] = 1.0
        else:
            h[mask] += frac * can_add[mask]
    else:
        # Need to subtract (-diff)
        can_sub = h
        mask = can_sub > 1e-12
        if mask.sum() == 0:
            return h
        frac = (-diff) / can_sub[mask].sum()
        if frac > 1:
            h[mask] = 0.0
        else:
            h[mask] -= frac * can_sub[mask]
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        # t(i) = t(i//2) XOR (i%2)
        t[i] = t[i // 2] ^ (i % 2)
    return t.astype(float)

def make_binary_with_fraction(ones_count, zeros_count, frac_val, frac_count, n):
    """Create a binary sequence with one fractional value"""
    h = np.zeros(n)
    h[:ones_count] = 1.0
    # Place fractional in the middle
    idx = (ones_count + frac_count) // 2

            h[idx] = np.clip(h[idx] + resid, 0.0, 1.0)
    return h

def thue_morse(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def generate_initial(n, seed, target):
    rng = np.random.RandomState(seed)
    candidates = []
    tm = thue_morse(n)
    candidates.append(project_sum(tm, target))
    candidates.append(project_sum(tm[::-1], target))
    for _ in range(4):
        rb = (rng.rand(n) < 0.5).astype(float)
        candidates.append(project_sum(rb, target))
    best_c = 1e9
    best_h = candidates[0]
    for c in candidates:
        c5 = compute_C5(c, n)
        if c5 < best_c:
            best_c = c5
            best_h = c
    return best_h
