import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def soft_max_val(corr, tau):
    m = np.max(corr)
    s = np.exp(tau * (corr - m))
    return m + np.log(np.sum(s)) / tau, s / np.sum(s)

def analytic_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    _, w = soft_max_val(corr, tau)
    g = np.zeros(n)
    for lag in range(-(n - 1), n):
        wi = w[n - 1 + lag] * dx
        if wi < 1e-10:
            continue
        L = n - abs(lag)
        if lag >= 0:
            g[lag:] += wi * omh[:L]
            g[:L] -= wi * h[lag:]
        else:
            g[:L] += wi * omh[-lag:]
            g[-lag:] -= wi * h[:L]
    return g

def project(h, ts):
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > ts:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    d = ts - r.sum()
    if abs(d) > 1e-12:
        i = int(np.argmax(1.0 - r)) if d > 0 else int(np.argmax(r))
        r[i] = float(np.clip(r[i] + d, 0.0, 1.0))
    return r

def project_binary(h, ts):
    hb = (h > 0.5).astype(float)
    ones = int(np.sum(hb))
    if ones > ts:
        idx = np.where(hb == 1)[0]
        idx = idx[np.argsort(h[idx])]
        for i in range(ones - ts):
            hb[idx[i]] = 0.0
    elif ones < ts:
        idx = np.where(hb == 0)[0]
        idx = idx[np.argsort(-h[idx])]
        for i in range(ts - ones):
            hb[idx[i]] = 1.0
    return hb

def optimize_local(h, ts, tau_start, max_iter, deadline, noise_period=0):
    n = len(h)
    dx = 2.0 / n
    tau = tau_start
    alpha = 0.02
    bh = h.copy()
    bc, _ = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = analytic_grad(h, tau)
        gn = np.linalg.norm(g)
        if gn < 1e-9:
            tau = min(tau * 1.5, 500.0)
            continue
        cc, _ = soft_max_val(np.correlate(h, 1 - h, mode='full') * dx, tau)
        accepted = False
        for _ in range(15):
            hn = project(h - alpha * g, ts)
            cn, _ = soft_max_val(np.correlate(hn, 1 - hn, mode='full') * dx, tau)
            if cn <= cc + 1e-12:
                h = hn
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            alpha = max(alpha * 0.5, 1e-6)
            if tau < 300:
                tau *= 1.2
            continue
        alpha = min(alpha * 1.03, 0.1)
        if it % 3000 == 0 and tau < 300:
            tau *= 1.3
        if noise_period > 0 and it % noise_period == noise_period - 1:
            h = project(h + 0.03 * np.random.randn(n), ts)
            alpha = 0.02
        ce, _ = compute_c5(h)
        if ce < bc:
            bc = ce
            bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0

    try:

        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
