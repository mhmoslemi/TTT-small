def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target_sum):
    """Project h to {0<=h<=1, sum(h)=target_sum} via bisection on lambda"""
    def do_project(shift):
        return np.sum(np.clip(h - shift, 0, 1))
    
    lo, hi = -max(0, np.sum(h) - target_sum) * 1.1, max(0, target_sum - np.sum(h)) * 1.1
    # Actually simpler: just bisection

        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    shift = (lo + hi) / 2
    return np.clip(h + shift, 0, 1)

import numpy as np

def project_constraint(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h):
    N = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / N))


def subgrad(h, ci, N):
    g = np.zeros(N)
    for idx in ci:
        k = idx - (N - 1)
        if k >= 0:
            if k < N:
                g[k:] += 1.0 - h[:N - k]
                g[:N - k] -= h[k:]
        else:
            d = -k
            if d < N:
                g[:N - d] += 1.0 - h[d:]
                g[d:] -= h[:N - d]
    return g
