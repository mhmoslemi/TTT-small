from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h_new, target_sum):
    """Project h_new onto {h : sum(h)=target_sum, 0<=h<=1}"""
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h_new - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h_new - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_obj_grad(h, T=80.0):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full') * dx
    m = np.max(corr)
    shifted = T * (corr - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = m + np.log(sum_exp) / T

        if wk < 1e-14:

        shift = k - (n - 1)

        j_end = min(n, n + shift)

            idx2 = np.arange(j_start2, j_end2) + shift
            grad[j_start2:j_end2] -= wk * h[idx2]
    grad *= dx
    return obj, grad
