            max_stuck = min(max_stuck * 2, 50000)
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        a = oi[rng.randint(len(oi))]
        b = zi[rng.randint(len(zi))]
        old = _mc(h)
        h[a] = 0.0
        h[b] = 1.0
        new = _mc(h)
        dE = (new - old) * dx
        if new < old or rng.random() < np.exp(-dE / max(T, 1e-10)):
            if new < best_mc:
                best_mc = new
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stuck += 1
        T = max(0.05, T * cool)
    return best_h, best_mc

def _greedy(h, best_h, best_mc, rng, t_end):
    h = h.copy()
    while time.time() < t_end:
        oi = np.where(h == 1).tolist()
        zi = np.where(h == 0).tolist()
        base = _mc(h)
        improved = False
        for a in oi:
            for b in zi:
                h[a] = 0.0
                h[b] = 1.0
                m = _mc(h)
                h[a] = 1.0
                h[b] = 0.0
                if m < base:
                    h[a] = 0.0
                    h[b] = 1.0
                    best_mc = m
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_mc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 70
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
        h0 = np.where(h0 >= 0.5, 1.0, 0.0)
    except Exception:
        h0 = np.zeros(n)
        h0[rng.permutation(n)[:n // 2]] = 1.0
    tgt = n // 2
    cnt = int(h0.sum())
    if cnt > tgt:
        idx = np.where(h0 == 1)[0]
        h0[rng.choice(idx, cnt - tgt, replace=False)] = 0.0
    elif cnt < tgt:
        idx = np.where(h0 == 0)[0]
        h0[rng.choice(idx, tgt - cnt, replace=False)] = 1.0
    dx = 2.0 / n
    best_h = h0.copy()
    best_mc = int(_mc(best_h))
    # Phase 1: SA on original n (60% budget)
    t1 = t0 + 0.60 * budget_s
    best_h, best_mc = _sa_loop(h0, best_h, best_mc, dx, rng, t0, t1)
    # Phase 2: Greedy polish (10% budget)
    t2 = t0 + 0.70 * budget_s
    best_h, best_mc = _greedy(best_h, best_h, best_mc, rng, t2)
    # Phase 3: Try doubled resolution (remaining time)
    n2 = n * 2
    dx2 = 2.0 / n2
    c5_cur = best_mc * dx
    h2 = np.repeat(best_h, 2).copy()
    mc2 = int(_mc(h2))
    if mc2 * dx2 < c5_cur or rng.randint(0, 2) == 0:
        h2_best_mc = mc2
        h2_best = h2.copy()
        t3 = t0 + budget_s - 5
        h2_best, h2_best_mc = _sa_loop(h2, h2_best, h2_best_mc, dx2, rng, time.time(), t3, T0=3.0, cool=0.99998, max_stuck0=3000)
        if h2_best_mc * dx2 < c5_cur:
            best_h = h2_best
            best_mc = h2_best_mc
            n = n2
            c5_cur = best_mc * dx2
    # Phase 4: Try a few more random restarts at original n if times allows
    t4 = t0 + budget_s - 5
    c5_best = best_mc * (2.0 / n)
    for trial in range(3):
        if time.time() > t4:
            break
        T = 5.0
        h_try = best_h.copy()
        rng2 = np.random.RandomState(seed + 1000 * (trial + 1))
        # Perturb
        for _ in range(rng2.randint(5, 20)):
            oi = np.where(h_try == 1)[0]
            zi = np.where(h_try == 0)[0]
            a = oi[rng2.randint(len(oi))]
            b = zi[rng2.randint(len(zi))]
            h_try[a] = 0.0
            h_try[b] = 1.0
        mc_try = int(_mc(h_try))
        if mc_try * (2.0 / n) < c5_best:
            t_sub = min(t4 - (trial + 1) * 20, t4)
            h_try, mc_try = _sa_loop(h_try, h_try, mc_try, 2.0 / n, rng2, time.time(), t_sub, T0=T, cool=0.99998)
            if mc_try * (2.0 / n) < c5_best:
                best_h = h_try
                best_mc = mc_try
                c5_best = mc_try * (2.0 / n)
    final_c5 = float(best_mc * 2.0 / n)
    return (best_h, final_c5, n)
