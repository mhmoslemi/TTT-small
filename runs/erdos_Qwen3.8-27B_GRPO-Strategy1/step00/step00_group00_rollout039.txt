import numpy as np
from scipy.optimize import minimize
import time

_G = {'n': 45, 'start': 0.0, 'max_time': 999.0, 'pw': 500.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = -float(np.max(np.abs(h))) - 1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    h_out = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - float(h_out.sum())
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(h_out - 0.5)))
        h_out[idx] = min(max(float(h_out[idx]) + diff, 0.0), 1.0)
    return h_out

def _obj_nlm(x):
    n = _G['n']
    if time.time() - _G['start'] > _G['max_time']:
        return 1e10
    c5 = compute_c5(x)
    pen = _G['pw'] * (float(x.sum()) - n * 0.5) ** 2
    return c5 + pen

def _obj_slsqp(x):
    return float(x[-1])

def _cons_eq(x):
    n = _G['n']
    return float(x[:-1].sum()) - n * 0.5

def _cons_ineq(x):
    n = _G['n']
    h = x[:-1]
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(x[-1]) - cor

def do_nelder_mead(h_init, max_time, rng):
    n = len(h_init)
    _G['n'] = n
    _G['start'] = time.time()
    _G['max_time'] = max_time
    _G['pw'] = 500.0
    h0 = project_feasible(h_init, n)

    try:
        res = minimize(_obj_nlm, h0, method='Nelder-Mead', options=opts)
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h_init, compute_c5(h_init)

def do_slsqp(h_init, max_time):
    n = len(h_init)
    _G['n'] = n
    _G['start'] = time.time()
    _G['max_time'] = max_time
    h = project_feasible(h_init, n)
    t0 = compute_c5(h) + 0.05
    x0 = np.concatenate([h, np.array([t0])])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': _cons_eq}, {'type': 'ineq', 'fun': _cons_ineq}]
    try:
        res = minimize(_obj_slsqp, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
        if res.fun < 1e8:

            return h_out, compute_c5(h_out)
    except Exception:

def do_coord_descent(h_init, max_time, rng):
    n = len(h_init)
    t0 = time.time()
    h = project_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
