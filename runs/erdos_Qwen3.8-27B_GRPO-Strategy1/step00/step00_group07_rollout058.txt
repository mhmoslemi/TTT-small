def active_gradient_step(h, k_star, lr):
    """Reduce correlation at lag k_star by adjusting h values."""
    n = len(h)
    # For np.correlate(h, 1-h, mode='full'), the lag mapping:
    # index m in full output corresponds to shift (m - (n-1))
    shift = k_star - (n - 1)
    # c[m] = Σ_i h[i + shift] * (1-h[i]) for valid i
    # So to reduce c, decrease h[i+shift] or increase h[i]
    
    # Which indices contribute most?
    if shift >= 0:
        valid_i = range(n - shift)
        a = np.array([i + shift for i in valid_i])
        b = np.array(list(valid_i))
    else:
        valid_i = range(-shift)
        a = np.array(list(valid_i))
        b = np.array([i - shift for i in valid_i])
    
    products = h[a] * (1 - h[b])
    # Gradient: ∂c/∂h[a_i] = (1-h[b_i]), ∂c/∂h[b_i] = -h[a_i]
    # To minimize c, move h[a_i] down (proportional to 1-h[b_i]) and h[b_i] up (proportional to h[a_i])
    
    # Simple: push top contributors
    top_idx = np.argsort(products)[-10:]
    for idx in top_idx:
        h[a[idx]] = max(0, h[a[idx]] - lr)
        h[b[idx]] = min(1, h[b[idx]] + lr)
    return h
