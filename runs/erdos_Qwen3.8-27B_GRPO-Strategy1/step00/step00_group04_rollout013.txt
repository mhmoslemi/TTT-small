from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def thue_morse(n):
    # Generate Thue-Morse sequence of length n (padded)
    seq = [0]
    for _ in range(n.bit_length()):
        seq += [1 - s for s in seq]
    return np.array(seq[:n], dtype=float)

def local_search_binary(h, n, max_iter=10000, time_limit=30):
    # h is binary, sum(h) = n/2
    # Swap-based local search
    best_c5 = evaluate_c5(h)
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    start = time.time()
    for iteration in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Pick random 1 and random 0
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def thue_morse_gen(n):
    seq = np.array([0])
    while len(seq) < n:
        seq = np.concatenate([seq, 1 - seq])
    return seq[:n].astype(float)

def adjust_ones(h, target_ones):
    h = h.copy()
    cs = h.sum()
    if cs > target_ones:
        one_idx = np.where(h > 0.5)[0]
        np.random.shuffle(one_idx)
        for idx in one_idx[:int(cs - target_ones)]:
            h[idx] = 0
    elif cs < target_ones:
        zero_idx = np.where(h < 0.5)[0]
        np.random.shuffle(zero_idx)
        for idx in zero_idx[:int(target_ones - cs)]:
            h[idx] = 1
    return h

def local_search_binary(h, n, iterations, time_limit, anneal=False):
    """Local search with optional simulated annealing for binary h."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = np.max(corr)
    best_max = current_max
    best_h = h.copy()
    
    start = time.time()
    one_pos = np.where(h == 1)[0]
    zero_pos = np.where(h == 0)[0]
    
    T0 = 0.5 / n  # initial temperature
    T_end = 0.01 / n
    total_iters = iterations
    
    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        # Progress-based cooling
        T = T0 * (T_end / T0) ** (it / max(total_iters, 1))
        
        # Pick random positions
        a = one_pos[np.random.randint(len(one_pos))]
        b = zero_pos[np.random.randint(len(zero_pos))]
        if a == b:
            continue
        
        # Compute delta for all shifts
        ks = np.arange(2 * n - 1) - (n - 1)
        # h[a+k] + h[a-k] - h[b+k] - h[b-k], with h[x]=0 if x out of [0,n-1]
        ak = a + ks
        ak2 = a - ks
        bk = b + ks
        bk2 = b - ks
        
        delta = np.zeros(2 * n - 1)
        mask = (ak >= 0) & (ak < n)
        delta[mask] += h[ak[mask]]
        mask = (ak2 >= 0) & (ak2 < n)
        delta[mask] += h[ak2[mask]]
        mask = (bk >= 0) & (bk < n)
        delta[mask] -= h[bk[mask]]
        mask = (bk2 >= 0) & (bk2 < n)
        delta[mask] -= h[bk2[mask]]
        
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        # Acceptance
        accept = False
        if new_max <= current_max:
            accept = True
        elif anneal and T > T_end:
            delta_c5 = (new_max - current_max) * dx
            if np.random.random() < np.exp(-delta_c5 / max(T, 1e-10)):
                accept = True
        
        if accept:
            corr = new_corr
            h[a], h[b] = h[b], h[a]
            current_max = new_max
            one_pos = np.where(h == 1)[0]
            zero_pos = np.where(h == 0)[0]
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
    
    return best_h, best_max * dx

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def thue_morse_gen(n):
    seq = np.array([0])
    while len(seq) < max(n + 1, 4):
        seq = np.concatenate([seq, 1 - seq])
    return seq[:n].astype(float)

def adjust_to_target(h, target):
    h = h.copy()
    diff = int(round(h.sum() - target))
    if diff > 0:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:diff]] = 0
    elif diff < 0:
        idx = np.where(h < 0.5)[0]
        np.random.shuffle(idx)
        h[idx[: -diff]] = 1
    return h

def local_search(h, n, max_iter, time_limit, anneal=False):
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    start = time.time()
    T0 = 0.4 / n
    T_f = 0.005 / n
    ks = np.arange(2 * n - 1, dtype=np.float64) - (n - 1)
    
    one_pos = np.where(h == 1)[0]
    zero_pos = np.where(h == 0)[0]
    no_improve = 0
    kick_interval = 3000
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if one_pos.size == 0 or zero_pos.size == 0:
            break
        T = T0 * (T_f / T0) ** (float(it) / float(max(max_iter, 1)))
        a = int(one_pos[np.random.randint(len(one_pos))])
        b = int(zero_pos[np.random.randint(len(zero_pos))])
        ak_p = a + ks
        ak_m = a - ks
        bk_p = b + ks
        bk_m = b - ks
        delta = np.zeros(2 * n - 1, dtype=np.float64)
        m = (ak_p >= 0) & (ak_p < n)
        delta[m] += h[ak_p[m].astype(int)]
        m = (ak_m >= 0) & (ak_m < n)
        delta[m] += h[ak_m[m].astype(int)]
        m = (bk_p >= 0) & (bk_p < n)
        delta[m] -= h[bk_p[m].astype(int)]
        m = (bk_m >= 0) & (bk_m < n)
        delta[m] -= h[bk_m[m].astype(int)]
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        accept = (new_max <= current_max + 1e-12)
        if not accept and anneal and T > T_f * 0.5:
            dc = (new_max - current_max) * dx
            if dc < 0.1 and np.random.random() < np.exp(-dc / max(T, 1e-12)):
                accept = True
        if accept:
            corr = new_corr
            h[a], h[b] = h[b], h[a]
            current_max = new_max
            one_pos = np.where(h == 1)[0]
            zero_pos = np.where(h == 0)[0]
            no_improve = 0
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > kick_interval and np.random.random() < 0.01:
                for _ in range(3):
                    if one_pos.size > 0 and zero_pos.size > 0:
                        aa = int(one_pos[np.random.randint(len(one_pos))])
                        bb = int(zero_pos[np.random.randint(len(zero_pos))])
                        if aa != bb:
                            h[aa], h[bb] = h[bb], h[aa]
                corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                current_max = float(np.max(corr))
                one_pos = np.where(h == 1)[0]
                zero_pos = np.where(h == 0)[0]
                no_improve = 0
    return best_h, best_max * dx
