def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        lam = (lo + hi) / 2.0
        s = np.sum(np.clip(h - lam, 0.0, 1.0))
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    offset = n - 1
    corrs = np.correlate(h, 1.0 - h, mode='full')
    vals = corrs * dx
    m = np.max(vals)
    exps = np.exp(alpha * (vals - m))
    Z = np.sum(exps)
    weights = exps / Z
    obj = m + np.log(Z) / alpha
    grad = np.zeros(n)
    ncorr = len(corrs)

        j2 = idx + k - offset
        grad[idx] += w * (1.0 - h[j2])
        grad[j2] += w * (-h[idx])
    grad *= dx
