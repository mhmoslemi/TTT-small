```python
import numpy as np
from scipy.optimize import minimize
import time
import sys


def _proj(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum} via bisection."""
    lo = -v.max() - 0.1
    hi = 1.0 - v.min() + 0.1
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        p = np.clip(v - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _smax(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    return float(m + np.log(np.sum(np.exp(beta * (c - m)))) / beta)


def _smax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    e = np.exp(beta * (c - m))
    s = e.sum()
    w = e / s
    shifts = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = shifts[k]
        a1, b1 = max(0, sh), min(n, n + sh)
        if b1 > a1:
            g[a1:b1] += wk * (1.0 - h[a1-sh:b1-sh])
        a2, b2 = max(0, -sh), min(n, n - sh)
        if b2 > a2:
            g[a2:b2] += wk * (-h[a2+sh:b2+sh])
    return g * dx


def _pgd(h0, beta, n_iter, lr0, tlim, momentum=0.9):
    n = len(h0)
    ts = n / 2.0
    h = _proj(h0.copy(), ts)
    best = h.copy()
    bs = _score(h)
    v = np.zeros(n)
    t_st = time.time()
    for it in range(n_iter):
        if time.time() - t_st > tlim:
            break
        lr = lr0 / (1.0 + 0.02 * it)
        g = _smax_grad(h, beta)
        v = momentum * v - lr * g
        h_new = _proj(h + v, ts)
        s = _score(h_new)
        if s < bs:
            bs = s
            best = h_new.copy()
        h = h_new
    return best, bs


def _slsqp_fun(x, b):
    return _smax(x, b)


def _slsqp_jac(x, b):
    return _smax_grad(x, b)


def _slsqp_eq(x, ts):
    return x.sum() - ts


def _slsqp_eqj(x, ts):
    return np.ones(len(x))


def _cd_pass(h, n_passes=3):
    """Coordinate descent: for each coordinate, do 1D minimization of max corr."""
    n = len(h)
    ts = n / 2.0
    h = h.copy()
    best_s = _score(h)
    for p in range(n_passes):
        for i in range(n):
            old_val = h[i]
            # 1D search: try different values for h[i]
            # The score as function of h[i] is piecewise linear (max of linear fns)
            # Use ternary search since it's convex
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[i] = m1
                h_t = _proj(h, ts)
                s1 = _score(h_t)
                h[i] = m2
                h_t = _proj(h, ts)
                s2 = _score(h_t)
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
            h[i] = 0.5 * (lo + hi)
            h = _proj(h, ts)
            if _score(h) < best_s:
                best_s = _score(h)
    return h, best_s
```

```python
        f1 = _smax(h, beta)
        h[j] -= 2*eps
        f2 = _smax(h, beta)
        h[j] += eps  # restore
        g[j] = (f1 - f2) / (2*eps)
    return g
```

This is clean and correct by construction. The only downside is 2n correlation computations per gradient step, but for n=47 that's trivial.

Actually, even better: I can compute the gradient of the MAX (not smoothed max) as a subgradient. At the argmax k*, the gradient is just the gradient of corr[k*] * dx with respect to h.

And for corr[k*] (which is a bilinear form in h), the gradient is easy to compute without knowing numpy's exact indexing, as long as I know which pairs of indices are multiplied.

But to avoid confusion, let me just use the numerical gradient of the smoothed max. It's simpler and I'm confident it's correct.
