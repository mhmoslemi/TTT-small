def _grad_smooth(h, dx, beta):
    n = len(h)
    J = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    shifted = np.minimum(beta * (corr - M), 50.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)  # shape (J,)
    
    grad = np.zeros(n)
    for j in range(J):
        lag = j - (n - 1)
        if weights[j] == 0:
            continue
        # dC_j/dh[m] = (1-h[m+lag]) * I(0≤m+lag<n) - h[m-lag] * I(0≤m-lag<n)
        m_plus = np.arange(n) + lag
        m_minus = np.arange(n) - lag
        valid_plus = (m_plus >= 0) & (m_plus < n)
        valid_minus = (m_minus >= 0) & (m_minus < n)
        
        contrib = np.zeros(n)
        contrib[valid_plus] = (1.0 - h[m_plus[valid_plus]])
        contrib[valid_minus] -= h[m_minus[valid_minus]]
        
        grad += weights[j] * contrib
    
    return dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def _corr_max(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _c5_val(h, dx):
    return _corr_max(h) * dx

def _smooth_f(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    v = np.minimum(beta * (corr - M), 50.0)
    return (M + np.log(np.mean(np.exp(v))) / beta) * dx

def _smooth_g(h, dx, beta):
    n = len(h)
    J = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    v = np.minimum(beta * (corr - M), 50.0)
    exp_v = np.exp(v)
    w = exp_v / exp_v.sum()
    grad = np.zeros(n)
    for j in range(J):
        if w[j] < 1e-15:
            continue
        lag = j - (n - 1)
        mp = np.arange(n) + lag
        mm = np.arange(n) - lag
        vp = (mp >= 0) & (mp < n)
        vm = (mm >= 0) & (mm < n)
        contrib = np.zeros(n)
        if np.any(vp):
            contrib[vp] = 1.0 - h[mp[vp]]
        if np.any(vm):
            contrib[vm] -= h[mm[vm]]
        grad += w[j] * contrib
    return dx * grad

def _repair_sum(h, n):
    """Repair h to satisfy sum(h) = n/2, 0 <= h <= 1."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(500):
        deficit = target - h.sum()
        if abs(deficit) < 1e-10:
            break
        if deficit > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(deficit / cnt, (1.0 - h[mask]).min())
            h[mask] += step
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(-deficit / cnt, h[mask].min())
            h[mask] -= step
    # Final precision fix
    h = np.clip(h, 0, 1)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        h += deficit / n
        h = np.clip(h, 0, 1)
    return h

def _coord_descent(h, dx, max_sweeps=50):
    """Coordinate descent with sum repair."""
    n = len(h)
    best = _c5_val(h, dx)
    for sweep in range(max_sweeps):
        improved = False
        for m in range(n):
            # Ternary search for optimal h[m]
            a, b = 0.0, 1.0
            for _ in range(40):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[m] = m1
                f1 = _corr_max(h)
                h[m] = m2
                f2 = _corr_max(h)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            h[m] = (a + b) / 2.0
            # Repair sum
            h = _repair_sum(h, n)
            c = _c5_val(h, dx)
            if c < best - 1e-12:
                best = c
                improved = True
        if not improved:
            break
    return h

    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = h0.copy()
    best_c5 = _c5_val(h0, dx)
    
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
            'jac': lambda x: np.ones(len(x))}]
    bounds = [(0.0, 1.0)] * n
    
    beta_seq = [20, 50, 100, 200, 500]

    for beta in beta_seq:
        if time.time() > deadline:

        try:

                          bounds=bounds, constraints=con,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:

                c5 = _c5_val(h_c, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_c
                    h0 = h_c.copy()
        except Exception:

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd

    n_restarts = 0
    while time.time() < deadline:

        # Generate perturbed starting point
        scale = 0.05 * max(0.1, 1.0 - n_restarts / 30.0)
        h_start = best_h + np.random.normal(0, scale, n)
        h_start = _repair_sum(h_start, n)

                    best_c5 = c5
                    best_h = h_c
        except Exception:
            pass
    
    return (best_h, best_c5, n)

    return (M + np.log(np.mean(np.exp(v))) / _BETA) * _DX

def _sm_g(x):
    n = len(x)
    J = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = np.max(corr)
    v = np.minimum(_BETA * (corr - M), 50.0)
    ev = np.exp(v)
    w = ev / ev.sum()
    grad = np.zeros(n)
    for j in range(J):
        if w[j] < 1e-15:
            continue
        lag = j - (n - 1)
        mp = np.arange(n) + lag
        mm = np.arange(n) - lag
        vp = (mp >= 0) & (mp < n)
        vm = (mm >= 0) & (mm < n)
        contrib = np.zeros(n)
        if np.any(vp):
            contrib[vp] = 1.0 - x[mp[vp]]
        if np.any(vm):
            contrib[vm] -= x[mm[vm]]
        grad += w[j] * contrib
    return _DX * grad

def _constraint(x):
    return x.sum() - _N / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _repair(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            step = d / cnt
            for i in np.where(mask)[0]:
                h[i] = min(1.0, h[i] + step)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            step = -d / cnt
            for i in np.where(mask)[0]:
                h[i] = max(0.0, h[i] - step)
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        h[0] = np.clip(h[0] + d, 0, 1)
        h[-1] = np.clip(h[-1] + (target - h.sum()), 0, 1)
    return h

def _c5(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * _DX

def _coord_descent(h, max_sweeps=50):
    n = len(h)
    best = _c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for m in range(n):
            a, b = 0.0, 1.0
            for _ in range(45):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[m] = m1
                f1 = np.max(np.correlate(h, 1.0 - h, mode='full'))
                h[m] = m2
                f2 = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            h[m] = (a + b) / 2.0
            h = _repair(h)
            c = _c5(h)
            if c < best - 1e-10:
                best = c
                improved = True
        if not improved:
            break
    return h
