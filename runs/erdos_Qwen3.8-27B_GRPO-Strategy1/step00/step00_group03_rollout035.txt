from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h to feasible set: sum=h_n_half, 0<=h<=1"""
    target = n / 2
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to add (1-h) capacity
            cap = 1 - h
            if cap.sum() > 0:
                h += diff * cap / cap.sum()
        else:
            # Need to remove (-diff) 
            if h.sum() > 0:
                h += diff * h / h.sum()
        h = np.clip(h, 0, 1)
        # Final fix
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-10:
            eps = 1e-10
            if diff2 > 0:
                h = h + diff2 / n
            else:
                h = h + diff2 / n

                else:
                    h_cur = h_start + 0.12 * rng.randn(n)
                    h_cur = project_feasible(h_cur, n)

            # Adam gradient descent
            max_iter = int(min(remaining - 30, 600) * 30)
            max_iter = max(max_iter, 1000)
            h_cur, c_cur = adam_descent(h_cur, n, lr=0.015, max_iter=max_iter, seed_val=seed + rnd * 7 + n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()

            # SLSQP refinement if n <= 140 and time allows
            elapsed = time.time() - t0
            remaining = budget_s - elapsed
            if n <= 140 and remaining > 80:
                h_sl, c_sl = slsqp_refine(h_cur, n, maxiter=60)
                if c_sl < best_c:
                    best_c = c_sl
                    best_h = h_sl.copy()

    # Final micro-refinement with small lr
    elapsed = time.time() - t0
    if best_c > 0.38085 and (budget_s - elapsed) > 40:
        n_best = len(best_h)
        h_final = adam_descent(best_h, n_best, lr=0.005, max_iter=2000, seed_val=seed + 99999)
        if f '< best_c:
            best_c = c_final
            best_h = h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
