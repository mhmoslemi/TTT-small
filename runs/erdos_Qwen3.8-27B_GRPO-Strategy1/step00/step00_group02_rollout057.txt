def subgrad_softmax(h, corr, n, k=5):

    # Get top k correlation indices
    top_k = np.argsort(corr)[-k:][::-1]

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr, dx

def project_manifold(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = int(free.sum())
        if nf == 0:
            idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff, 0, 1))
        else:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_at_max(h, corr, n):
    k_star = int(np.argmax(corr))
    tau = k_star - (n - 1)
    lo = max(0, -tau)
    hi = min(n - 1, n - 1 - tau)
    grad = np.zeros(n)
    for j in range(lo, hi + 1):
        i1 = j + tau
        if 0 <= i1 < n:
            grad[j] += 1.0 - h[i1]
        i2 = j - tau
        if lo <= i2 <= hi:
            grad[j] -= h[i2]
    return grad

def subgrad_topk(h, corr, n, k=3):
