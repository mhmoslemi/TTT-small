        h = h_flat.copy()
        h = np.clip(h, 0, 1)

        ]

            method='SLSQP',

            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9, 'disp': False}
        )
        return res.x

        perturb = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)

        return perturbed_norm
