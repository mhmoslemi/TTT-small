from scipy.optimize import minimize
from scipy.signal import correlate
from functools import partial
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):

    dx = 2.0 / n_points
    h_initial = initial_h.copy()

    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return res.x

    def perturb(h, perturbation_scale=0.05):
        perturb = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        perturbed_norm = perturbed * (n_points / 2) / np.sum(perturbed)
        return perturbed_norm

    def refine_solution(h_initial, iterations=10):
        optimized_h = optimize_h(h_initial)
        for _ in range(iterations):
            perturbed_h = perturb(optimized_h, perturbation_scale=0.05)
            optimized_h = optimize_h(perturbed_h)
        return optimized_h

    def run_multiple_starts(num_starts=10, perturbation_scale=0.1):
        best_h = h_initial.copy()
        best_c5 = compute_overlap(best_h)
        for _ in range(num_starts):
            perturbed_h = perturb(h_initial, perturbation_scale=perturbation_scale)
            optimized_h = refine_solution(perturbed_h, iterations=10)
            current_c5 = compute_overlap(optimized_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5

    best_h, best_c5 = run_multiple_starts(num_starts=20, perturbation_scale=0.05)
    final_h = refine_solution(best_h, iterations=20)
    final_c5 = compute_overlap(final_h)

    return (final_h, final_c5, n_points)
