```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 samples, can increase if needed
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1  # Initial guess
    
    # Normalize to satisfy the integral constraint
    h = h / np.sum(h) * (n_points / 2)
    
    # Evaluate initial overlap
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        overlap = correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    initial_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = initial_c5
    time_used = 0
    
    # Optimization
    for _ in range(200):  # Limited iterations to stay within time
        start_time = time.time()
        
        def objective(x):
            nonlocal best_c5
            x = np.clip(x, 0, 1)
            overlap = correlate(x, 1 - x, mode="full") * dx
            current_c5 = np.max(overlap)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x.copy()
            return current_c5
        
        # Use a simple gradient descent with random perturbation
        perturbation = np.random.randn(n_points) * 0.05
        h = np.clip(h + perturbation, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
```
