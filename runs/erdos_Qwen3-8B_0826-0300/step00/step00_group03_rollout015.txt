import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values as a starting point
    n_points_initial = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=float)
    
    # Use a smaller number of points, starting with n_points = 100
    n_points = 100
    while time.time() - start_time < budget_s:
        # Discretize the interval [0, 2] with n_points samples
        dx = 2.0 / n_points
        h = h_values.copy()
        h = np.resize(h, n_points)  # Resize to n_points

        # Ensure normalization constraint
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = h * (n_points / 2) / sum_h

        # Calculate the overlap integral
        def compute_c5(h):
            h_reshaped = np.tile(h, (n_points, 1))
            shifted_h = np.roll(h_reshaped, np.arange(n_points), axis=1)
            overlap = np.sum(h_reshaped * (1 - shifted_h), axis=1)
            return np.max(overlap) * dx

        c5 = compute_c5(h)
        if c5 < 0.380876:
            print(f"Found better solution with n_points={n_points}, C5={c5}")
            best_h = h.copy()
            best_c5 = c5
            break

        # Use a gradient-based optimizer to refine the solution
        def objective(x):
            h = x.reshape(n_points)
            # Ensure h is in [0, 1]
            h = np.clip(h, 0, 1)
            # Normalize
            sum_h = np.sum(h)
            if sum_h != n_points / 2:
                h = h * (n_points / 2) / sum_h
            # Compute C5
            h_reshaped = np.tile(h, (n_points, 1))
            shifted_h = np.roll(h_reshaped, np.arange(n_points), axis=1)
            overlap = np.sum(h_reshaped * (1 - shifted_h), axis=1)
            c5 = np.max(overlap) * dx
            return c5

        x0 = h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if result.success:
            h_values = result.x
            c5 = result.fun
            if c5 < best_c5 or best_c5 is None:
                best_h = h_values.copy()
                best_c5 = c5
                print(f"Updated best solution with n_points={n_points}, C5={c5}")
        else:
            print(f"Optimization failed for n_points={n_points}")
        
        # Try increasing the number of points gradually
        n_points += 50
        
    # If no better solution found, return the best found
    if best_h is None:
        best_h = h_values
        best_c5 = c5
    return best_h, best_c5, n_points
