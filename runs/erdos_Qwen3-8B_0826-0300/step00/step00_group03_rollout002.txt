import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100  # Start with 100 points, can increase if needed
    
    # Initial construction (from global variable if provided)
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial = np.array(h_initial)
        if len(h_initial) != n_points:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        h = h_initial
    else:
        h = np.zeros(n_points)
    
    # Ensure the initial h is in [0,1] and normalized
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Define the function to compute C5
    def compute_c5(h):
        dx = 2.0 / n_points
        h_extended = np.concatenate([h, h[1:]])
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to optimize
    def objective(x):
        h = x.reshape(n_points)
        c5 = compute_c5(h)
        return c5
    
    # Function to compute the integral constraint
    def constraint(x):
        h = x.reshape(n_points)
        integral = np.sum(h) * (2.0 / n_points)
        return integral - 1.0
    
    # Bounds and constraints
    x_bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization settings
    maxiter = 100
    tol = 1e-6
    
    # Initial optimization with SLSQP
    result = optimize.minimize(
        objective,
        h.flatten(),
        bounds=x_bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': tol, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Try a few more iterations with different starting points if time allows
    elapsed = time.time() - time.time()
    if elapsed < budget_s:
        for _ in range(3):
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            result = optimize.minimize(
                objective,
                perturbed_h.flatten(),
                bounds=x_bounds,
                constraints=cons,
                method='SLSQP',
                options={'maxiter': 20, 'ftol': tol, 'eps': 1e-8}
            )
            if result.fun < best_c5:
                best_h = result.x
                best_c5 = result.fun
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, n_points
