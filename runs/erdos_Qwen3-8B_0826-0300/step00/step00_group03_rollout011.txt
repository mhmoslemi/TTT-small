def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    c5 = 0.0
    for k in range(1, n_points // 2 + 1):
        overlap = np.correlate(h, 1 - h, mode="full")[k]
        c5 = max(c5, overlap * dx)
    return c5

def run(seed=42, budget_s=60, **kwargs):
    # Start time
    start_time = time.time()
    
    # Initial parameters
    n_points = 200  # Less than 1k samples
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points].copy()  # Use initial construction
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    # Optimization bounds
    bounds = Bounds(0.0, 1.0)
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        # Reshape for numpy
        h = h.reshape(n_points)
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute maximum overlap
        c5 = np.max(corr[1:n_points // 2 + 1]) * dx
        return c5

    # Constraints: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
    
    # Initial guess
    h_initial = initial_h
    
    # Optimization
    result = minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints=constraint,
        method='SLSQP',
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 1000}
    )
    
    # Check elapsed time and return best solution
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (initial_h, evaluate_erdos_solution(initial_h), len(initial_h))
    
    # Return the best solution found
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values)
    n_points = len(h_values)
    return (h_values, c5_bound, n_points)
