import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assume this is loaded with n_points=47

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be adjusted if needed
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5  # Random initial guess
    h = h / np.sum(h) * (n_points / 2)  # Ensure integral condition
    
    # Optimize using gradient descent with line search
    max_iterations = 1000
    learning_rate = 0.01
    tolerance = 1e-4
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for _ in range(max_iterations):
        # Compute current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Compute gradients for each point
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Perturb h[i] slightly to estimate gradient
            h_perturb = h.copy()
            h_perturb[i] += 1e-5
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
            c5_perturb = evaluate_erdos_solution(h_perturb)
            grad[i] = (c5_perturb - current_c5) / 1e-5
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Check for improvement
        if evaluate_erdos_solution(h) < current_c5 - tolerance:
            best_h = h.copy()
            best_c5 = evaluate_erdos_solution(h)
        elif time.time() - start_time > budget_s - 1:
            break
    
    # Additional refinement with local search
    for _ in range(100):
        # Random perturbation
        idx = np.random.randint(n_points)
        h[idx] += np.random.uniform(-0.05, 0.05)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5 - tolerance:
            best_h = h.copy()
            best_c5 = c5
        elif time.time() - start_time > budget_s - 1:
            break
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
