import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 70  # Starting with the initial sample count
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h_values):
        h = h_values.reshape(n_points)
        # Ensure h stays within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute integral of h
        if np.sum(h) * dx < 1 - 1e-8 or np.sum(h) * dx > 1 + 1e-8:
            return float('inf')
        # Compute the overlap integral
        h_pad = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        shifted = np.zeros_like(h_pad)
        for k in range(1, n_points):
            shifted[k:] = h_pad[:n_points - k]
        integral = np.sum(h * (1 - shifted)) * dx
        return integral
    
    def evaluate_bound(h_values):
        h = h_values.reshape(n_points)
        h = np.clip(h, 0, 1)
        if np.sum(h) * dx < 1 - 1e-8 or np.sum(h) * dx > 1 + 1e-8:
            return float('inf')
        h_pad = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        shifted = np.zeros_like(h_pad)
        for k in range(1, n_points):
            shifted[k:] = h_pad[:n_points - k]
        integrals = np.sum(h * (1 - shifted)) * dx
        return np.max(integrals)
    
    def compute_c5(h_values):
        return evaluate_bound(h_values)
    
    best_h = h
    best_c5 = compute_c5(h)
    
    # Optimize with gradient descent
    max_iter = 100
    learning_rate = 0.01
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute gradient approximation
        perturb = np.random.normal(0, 0.01, size=n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb_sum = np.sum(h_perturb) * dx
        if h_perturb_sum < 1 - 1e-8 or h_perturb_sum > 1 + 1e-8:
            continue
        c5_perturb = compute_c5(h_perturb)
        grad = (c5_perturb - best_c5) * perturb
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        if h_sum < 1 - 1e-8 or h_sum > 1 + 1e-8:
            h = h + (1 - h_sum / dx) / n_points
            h = np.clip(h, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Try reducing n_points to a smaller value
    for n_points in range(50, 30, -1):
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        if h_sum < 1 - 1e-8 or h_sum > 1 + 1e-8:
            h = h + (1 - h_sum / dx) / n_points
            h = np.clip(h, 0, 1)
        # Optimize with gradient descent
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.01, size=n_points)
            h_perturb = h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb_sum = np.sum(h_perturb) * dx
            if h_perturb_sum < 1 - 1e-8 or h_perturb_sum > 1 + 1e-8:
                continue
            c5_perturb = compute_c5(h_perturb)
            grad = (c5_perturb - best_c5) * perturb
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h) * dx
            if h_sum < 1 - 1e-8 or h_sum > 1 + 1e-8:
                h = h + (1 - h_sum / dx) / n_points
                h = np.clip(h, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    # Final check with best_h
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h) * dx
    if best_h_sum < 1 - 1e-8 or best_h_sum > 1 + 1e-8:
        best_h = best_h + (1 - best_h_sum / dx) / n_points
        best_h = np.clip(best_h, 0, 1)
    
    return (best_h, best_c5, n_points)
