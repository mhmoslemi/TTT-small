import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    # Initial construction (pre-loaded)
    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7088, 0.6926, 0.6750,
                                 0.6562, 0.6364, 0.6158, 0.5944, 0.5723, 0.5500, 0.5276, 0.5052, 0.4830, 0.4613,
                                 0.4398, 0.4184, 0.3970, 0.3761, 0.3555, 0.3354, 0.3157, 0.2966, 0.2781, 0.2600,
                                 0.2425, 0.2253, 0.2085, 0.1921, 0.1761, 0.1605, 0.1453, 0.1305, 0.1161, 0.1021,
                                 0.0885, 0.0753, 0.0625, 0.0501, 0.0381, 0.0265, 0.0153, 0.0045, 0.0000])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    best_c5 = float('inf')
    best_h = h.copy()
    time_start = time.time()
    
    def evaluate_c5(h_values):
        h = h_values
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        h = h.reshape(-1, 1)
        ones = np.ones_like(h)
        one_minus_h = 1 - h
        cross_correlation = np.convolve(h, one_minus_h, mode='full')
        c5 = np.max(cross_correlation) * dx
        return c5

    def optimize_h():
        nonlocal best_c5, best_h
        h_init = h.copy()
        result = optimize.minimize(evaluate_c5, h_init, method='L-BFGS-B', bounds=[(0, 1)]*n_points)
        h = result.x
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    while time.time() - time_start < budget_s:
        c5 = optimize_h()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Early stopping if we're close to the target
        if best_c5 <= 0.380876:
            break

    return best_h, best_c5, n_points
