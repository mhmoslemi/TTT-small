import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
initial_h_values = np.random.rand(47)  # Placeholder, replace with actual initial solution


def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for balance between speed and quality
    
    # Try with different numbers of points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initialize h with a random starting point
        h = np.random.rand(n_points)
        h = h * 2 - 1  # Adjust to [0, 1] range
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
        
        # Optimization setup
        x = cp.Variable(n_points)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        
        # Define the objective function to minimize the maximum overlap
        # We use a convex relaxation by using a minimax formulation
        # This is an approximation due to computational limitations
        overlap = correlate(x.value, 1 - x.value, mode="full") * (2 / n_points)
        c5 = np.max(overlap)
        
        # Perform gradient descent with a learning rate
        learning_rate = 0.01
        for _ in range(100):
            # Compute current overlap
            overlap = correlate(x.value, 1 - x.value, mode="full") * (2 / n_points)
            current_c5 = np.max(overlap)
            
            # Gradient descent step
            gradient = np.zeros_like(x.value)
            for i in range(n_points):
                # For each position, compute the gradient of max overlap
                # This is a simplified approximation
                for k in range(-n_points + 1, n_points):
                    if k >= 0:
                        overlap_k = correlate(x.value, 1 - x.value, mode="full")[k]
                    else:
                        overlap_k = correlate(1 - x.value, x.value, mode="full")[-k]
                    if overlap_k == current_c5:
                        # Approximate gradient
                        grad = np.zeros_like(x.value)
                        grad[i] = (correlate(x.value + 1e-5, 1 - x.value, mode="full") * (2 / n_points)
                                   - correlate(x.value, 1 - x.value, mode="full") * (2 / n_points))[i]
                        gradient[i] = grad[i]
            x.value -= learning_rate * gradient
            
            # Check for convergence
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x.value
                print(f"n_points={n_points}, best_c5={best_c5}")
        
        # Check if we have a better solution
        if best_c5 < c5:
            best_c5 = c5
            best_h = h
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
