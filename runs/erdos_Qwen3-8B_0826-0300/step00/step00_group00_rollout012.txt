import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Preloaded functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # placeholder for actual data

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # start with 100 points, can be increased if time permits
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points // 2] = 1.0  # initial guess: half the interval is 1
    h = np.random.permutation(h)  # randomize to avoid local minima

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points

    # Try increasing n_points if time allows
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0  # initial guess: half the interval is 1
        h = np.random.permutation(h)  # randomize to avoid local minima

        # Optimization: try to minimize max overlap
        def objective_func(h_vals):
            h_vals = np.array(h_vals)
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap

        # Use scipy's minimize to find a better solution
        result = opt.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-4,
            options={'maxiter': 100}
        )

        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                best_n_points = n_points

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
