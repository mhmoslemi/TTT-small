import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])
    total_area = np.sum(h) * dx
    if not np.isclose(total_area, 1.0):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * 1.0
        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Try increasing n_points if time allows
    time_remaining = budget_s
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time_remaining <= 0:
            break
        start_time = time.time()
        h = np.random.rand(n_points)
        h = h / np.sum(h) * 1.0
        h = np.clip(h, 0.0, 1.0)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        time_remaining -= time.time() - start_time
    
    # Use optimization to improve the best_h
    if time_remaining > 0:
        def objective(x):
            h = x.reshape((-1, 1))
            overlap = np.max(np.correlate(h, 1 - h, mode="full")) * dx
            return overlap
        
        bounds = [(0.0, 1.0) for _ in range(len(best_h))]
        result = optimize.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6},
            jac=None
        )
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Final check to ensure area constraint
    total_area = np.sum(best_h) * dx
    if not np.isclose(total_area, 1.0):
        best_h = np.random.rand(len(best_h))
        best_h = best_h / np.sum(best_h) * 1.0
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, len(best_h))
