```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros(2 * n_points)
    shifted_h[n_points:] = h
    shifted_1_minus_h = 1 - h
    correlate = np.correlate(shifted_h, shifted_1_minus_h, mode="full")
    c5 = np.max(correlate) * dx
    return c5

def initial_h_values():
    return np.loadtxt("initial_h_values.txt")

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start from the initial construction
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = initial_h * (n_points / 2) / sum_h
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try reducing the number of points
    for n_points in range(50, 100, -10):
        if time.time() - start_time > budget_s - 2:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization with CVXPY
        h_var = cp.Variable(n_points)
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        obj = cp.Maximize(cp.Max([cp.sum(h_var[i] * (1 - h_var[(i + k) % n_points])) for k in range(1, n_points // 2 + 1)]))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iter=100)
            if prob.status == 'optimal':
                h_opt = h_var.value
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
```
