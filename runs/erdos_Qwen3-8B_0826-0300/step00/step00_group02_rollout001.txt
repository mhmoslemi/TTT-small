import numpy as np

import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute (1 - h) for convolution
    one_minus_h = 1.0 - h
    # Perform full convolution
    correlation = signal.correlate(h, one_minus_h, mode="full")
    # Multiply by dx to get the integral
    c5 = np.max(correlation * dx)
    return c5

    h = cp.Variable(n_points, nonneg=True, lb=0.0, ub=1.0)
    
    # Objective: Minimize the maximum overlap integral
    def objective_function(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        one_minus_h = 1.0 - h_vec
        correlation = signal.correlate(h_vec, one_minus_h, mode="full")
        c5 = np.max(correlation * dx)
        return c5
    
    # Constraint: integral of h is 1
    constraint = cp.sum(h) == n_points / 2.0
    
    # Use a solver with a time budget
    start_time = time.time()
    best_c5 = float('inf')
    best_h_values = h_initial.copy()
    
    # Try a few gradient-based optimization steps
    for _ in range(3):
        # Use scipy optimization for initial refinement
        def objective(x):
            return objective_function(x)
        
        def constraint_func(x):
            return cp.sum(x) - n_points / 2.0
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=1e-6)
