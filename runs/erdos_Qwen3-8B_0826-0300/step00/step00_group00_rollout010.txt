import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions (simulated)
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
        overlap = np.dot(h, 1 - shifted) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def initial_h_values():
    return [0.5 if i % 3 == 0 else 0.0 for i in range(45)]

# Strategy:
# 1. Start with the initial_h_values as a baseline and improve from there.
# 2. Use a gradient-based optimization to adjust the step function h.
# 3. Use a sliding window approach to evaluate the overlap integral efficiently.
# 4. Minimize the maximum overlap integral using a constrained optimization.
# 5. Use a moderate number of samples (n_points = 100) for a good balance between accuracy and speed.
# 6. Employ a numerical solver with bounds on the h values.

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Start with a simple step function based on the initial_h_values
    h = np.array([0.5 if i % 3 == 0 else 0.0 for i in range(n_points)])
    h = h * (n_points / 2) / np.sum(h)  # Ensure the integral condition is satisfied
    h = np.clip(h, 0, 1)  # Ensure h is in [0, 1]

    max_overlap = evaluate_erdos_solution(h.tolist())

    # Optimization parameters
    max_iterations = 20
    epsilon = 1e-4
    improvement_threshold = 1e-5
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Function to evaluate the maximum overlap
        def objective(x):
            h_vals = x.reshape(n_points)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)  # Ensure the integral condition is satisfied
            overlap = evaluate_erdos_solution(h_vals.tolist())
            return overlap

        # Optimization using gradient descent with bounds
        h_opt = cp.Variable(n_points)
        objective_func = cp.Minimize(objective(h_opt))
        constraints = [cp.sum(h_opt) == n_points / 2, h_opt >= 0, h_opt <= 1]
        prob = cp.Problem(objective_func, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100, eps=epsilon)
            if prob.status == 'optimal':
                h = h_opt.value
                max_overlap = objective(h)
                if max_overlap < max_overlap - improvement_threshold:
                    break
        except:
            pass

    h_values = h.tolist()
    c5_bound = max_overlap
    return h_values, c5_bound, n_points
