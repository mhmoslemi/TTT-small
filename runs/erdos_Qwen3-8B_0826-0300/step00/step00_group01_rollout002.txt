    def evaluate_c5(h_vals):

        h_extended = np.concatenate([h, np.zeros_like(h)])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / len(h))
        return c5
    
    # Optimization function: minimize C5
    def objective_function(h_vals):
        c5 = evaluate_c5(h_vals)
        return c5

    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2

    h_initial = np.array(initial_h)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    max_iter = 100
    tol = 1e-5
    
    # Optimization using scipy.optimize
    result = optimize.minimize(
        objective_function,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        tol=tol,
        options={'maxiter': max_iter}
    )
    
    # If optimization fails, return initial
    if result.success:
        best_h = result.x
    else:
        best_h = h_initial

    best_c5 = evaluate_c5(best_h)
