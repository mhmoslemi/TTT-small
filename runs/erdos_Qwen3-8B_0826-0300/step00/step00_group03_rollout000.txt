```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        h = np.array(h)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:-k]
        overlap = correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to compute the objective for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraints
    def constraints(h):
        return np.array([np.sum(h) - n_points / 2])
    
    # Optimization
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Try optimizing with CVXPY
    try:
        # Convert to CVXPY variables
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(objective(h_var))
        cons = [cp.sum(h_var) == n_points / 2]
        prob = cp.Problem(obj, cons)
        result = prob.solve(solver=cp.MOSEK, eps=1e-6)
        if result is not None and result >= 0:
            h = h_var.value
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    except:
        pass
    
    # Try optimizing with scipy.optimize
    try:
        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        # Optimization using L-BFGS-B
        result = optimize.minimize(objective, h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-6, options={'ftol': 1e-9, 'maxiter': 100})
        if result.success:
            h = result.x
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    except:
        pass
    
    # Try reducing the number of points if time permits
    if time.time() - start_time < budget_s - 5:
        # Try lower n_points if possible
        for n_new in range(n_points - 10, 50, -1):
            dx_new = 2.0 / n_new
            h_new = np.zeros(n_new)
            # Distribute the integral evenly
            h_new = np.linspace(0, 1, n_new)
            h_new = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(best_h)), best_h)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_new / 2) / np.sum(h_new)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
                n_points = n_new
    
    return best_h, best_c5, n_points
```
