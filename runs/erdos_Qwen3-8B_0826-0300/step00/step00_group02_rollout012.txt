import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.load('initial_h_values.npy')  # Load initial construction (n=45 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 45  # Start with the initial construction
    
    # Function to compute overlap integral
    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
    
    # Function to optimize h using convex optimization
    def optimize_h(n_points, initial_h=None):
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        
        # Objective: minimize the maximum overlap integral
        overlap = cp.Maximize(cp.Max([cp.sum(cp.multiply(h[i], 1 - h[i + k])) for k in range(1, n_points)]))
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # Integral constraint
            0 <= h,
            h <= 1
        ]
        
        # Solve the optimization problem
        prob = cp.Problem(overlap, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-5, verbose=True)
            if prob.status != 'OPTIMAL':
                return None
            return h.value
        except:
            return None
    
    # Function to perturb a solution slightly
    def perturb_h(h, perturb_strength=0.05):
        h_perturbed = h.copy()
        for i in range(len(h_perturbed)):
            h_perturbed[i] += np.random.uniform(-perturb_strength, perturb_strength)
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        return h_perturbed
    
    # Function to improve solution using local search
    def improve_solution(h, n_points, max_iterations=100):
        for _ in range(max_iterations):
            # Compute current C₅
            current_c5 = compute_c5(h)
            
            # Perturb the solution slightly
            h_perturbed = perturb_h(h)
            
            # Compute C₅ for perturbed solution
            perturbed_c5 = compute_c5(h_perturbed)
            
            if perturbed_c5 < current_c5:
                h = h_perturbed
            else:
                # If perturbed solution is worse, try a different perturbation
                h = perturb_h(h, perturb_strength=0.1)
        
        return h
    
    # Start with the initial construction
    h = initial_h_values[:n_points].copy()
    
    # Optimize using convex optimization
    h_opt = optimize_h(n_points, initial_h=h)
    if h_opt is not None:
        h = h_opt
    else:
        # If optimization fails, use local search
        h = improve_solution(h, n_points)
    
    # Evaluate the current solution
    current_c5 = compute_c5(h)
    
    # Try increasing n_points if time allows
    while time.time() - start_time < budget_s - 5:
        n_points += 5
        dx = 2.0 / n_points
        # Resample h to new number of points
        h_resampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h_resampled = np.clip(h_resampled, 0, 1)
        
        # Optimize using convex optimization
        h_opt = optimize_h(n_points, initial_h=h_resampled)
        if h_opt is not None:
            h = h_opt
        else:
            # If optimization fails, use local search
            h = improve_solution(h_resampled, n_points)
        
        # Evaluate the current solution
        current_c5 = compute_c5(h)
        print(f"n_points = {n_points}, C₅ = {current_c5}")
    
    # Final evaluation
    final_c5 = compute_c5(h)
    print(f"Final n_points = {n_points}, C₅ = {final_c5}")
    
    return h, final_c5, n_points
