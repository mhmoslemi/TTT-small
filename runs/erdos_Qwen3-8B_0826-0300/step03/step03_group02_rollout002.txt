import random
import time
from deap import base, creator, tools, algorithms

    def evaluate(individual):

        return max_overlap,
    
    # Create fitness and individual classes
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    # Setup tools
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: random.uniform(0, 1))
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutUniform, low=0, up=1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint_pop(population):
        for ind in population:
            if abs(np.sum(ind) - n_points / 2) > 1e-6:
                raise ValueError("Constraint violated")
    
    # Register constraint-checking function
    toolbox.register("check_constraints", constraint_pop)
    
    # Define the GA
    def run_ga():
        # Initialize population
        pop = toolbox.population(n=50)
        # Evaluate the entire population
        fitnesses = map(evaluate, pop)
        for ind, fit in zip(pop, fitnesses):
            ind.fitness.values = fit
        # Check initial constraints
        toolbox.check_constraints(pop)
        
        # Set up the GA
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)
        
        # Run the GA
        pop, logbook = algorithms.eaSimple(pop, toolbox, cxpb=0.5, mutpb=0.1, ngen=100,

    start_time = time.time()

    # Run the GA with early stopping if budget runs out

                if current_c5 < best_c5:

                    best_c5 = current_c5
            except Exception as e:
