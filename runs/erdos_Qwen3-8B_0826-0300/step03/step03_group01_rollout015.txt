import time

from scipy.optimize import minimize
from scipy.optimize import Bounds

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def create_initial_guess_from_initial(n_points, initial_h, seed=42):
    """Create a more informed initial guess based on the given initial_h."""
    np.random.seed(seed)
    h = initial_h.copy()
    # Introduce small perturbations to avoid local optima
    noise = np.random.rand(n_points) * 0.05
    h += noise
    h = np.clip(h, 0, 1)
    # Normalize to ensure sum(h) = n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    return h

        start_time = time.time()
