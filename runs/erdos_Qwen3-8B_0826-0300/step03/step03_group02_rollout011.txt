import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def generate_uniform_initial(n_points):
    """Generate a uniform initial guess for the step function."""
    return np.full(n_points, 0.5)

def generate_random_initial(n_points):
    """Generate a random initial guess for the step function."""
    return np.random.uniform(0, 1, n_points)

def generate_perturbed_initial(n_points, base_h, perturbation=0.05):
    """Generate a perturbed initial guess based on a base function."""
    return np.clip(base_h + np.random.uniform(-perturbation, perturbation, size=n_points), 0, 1)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Test several values of n_points (discretization)
    n_points_list = [50, 100, 150, 200]
    best_h = np.array([])
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Initialize h with multiple strategies
        initial_h = generate_random_initial(n_points)
        initial_h = generate_perturbed_initial(n_points, initial_h, perturbation=0.01)
        initial_h = generate_uniform_initial(n_points)
        
        dx = 2.0 / n_points
        
        # Constraint: sum(h) == n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Bounds for h[i] in [0, 1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function to minimize the maximum overlap
        def objective(h):
            overlap = np.correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap) * dx
            return max_overlap
        
        # Optimization
        result = optimize.minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 100}
        )
        
        # Check time budget

            break
        
        current_c5 = evaluate_erdos_solution(result.x)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
    
    return (best_h, best_c5, len(best_h))
