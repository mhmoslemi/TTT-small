import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def create_initial_guess(n_points, seed=42):
    """Create a more informed initial guess with diverse patterns."""
    np.random.seed(seed)
    h = np.zeros(n_points)
    for i in range(n_points):
        # Randomly assign 0 or 1 with some variation
        if np.random.rand() < 0.5:
            h[i] = 0.0
        else:
            h[i] = 1.0
    # Add some sine wave pattern for diversity
    wave = 0.5 * np.sin(np.linspace(0, np.pi, n_points))
    h += wave
    h = np.clip(h, 0, 1)
    # Normalize to ensure sum(h) = n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    best_h = None
    best_c5_bound = float('inf')
    best_n_points = 0
    time_used = 0
    
    # Try multiple initial configurations and different n_points
    for n_points in [100, 150, 200, 250, 300]:  # Explore different sample sizes
        # Generate a diverse initial guess
        h_initial = create_initial_guess(n_points)
        
        dx = 2.0 / n_points
        
        # Objective function to minimize the maximum overlap
        def objective(h):
            overlap = np.correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap) * dx
            return max_overlap
        
        # Constraint: sum(h) == n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Bounds for h[i] in [0, 1]
        bounds = Bounds(0, 1)
        
        # Initial guess
        h0 = h_initial.copy()
        
        # Optimization
        start_time = time.time()
        result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        elapsed_time = time.time() - start_time
        time_used += elapsed_time
        
        # Check if we have enough time to try a few more iterations
        if elapsed_time < budget_s - 1:
            # Try a few more iterations with tighter tolerances
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        
        # Evaluate the solution
        c5_bound = evaluate_erdos_solution(result.x)
        if c5_bound < best_c5_bound:
            best_c5_bound = c5_bound
            best_h = result.x
            best_n_points = n_points
        
        # Early stopping if we have no time left
        if time_used > budget_s - 1:
            break
    
    # Return the best solution found
    return (best_h, best_c5_bound, best_n_points)
