from scipy.optimize import Bounds

evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def create_initial_guess(n_points, seed=42):
    """Create a random initial guess with uniform distribution and proper normalization."""
    np.random.seed(seed)
    h = np.random.rand(n_points)
    # Normalize to ensure sum(h) = n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    return h

def run(seed=42, budget_s=60, **kwargs):

        def constraint(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
