    def perturb_solution(solution, perturb_strength=0.05):
        perturbed = solution.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                perturbed[i] += np.random.uniform(-perturb_strength, perturb_strength)
        perturbed = np.clip(perturbed, 0, 1)

        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        return perturbed

    def simulated_annealing(initial_solution, temp=1.0, cooling_rate=0.995, steps=500):

        best_solution = current_solution.copy()

                best_cost = current_cost

            # Cool down
