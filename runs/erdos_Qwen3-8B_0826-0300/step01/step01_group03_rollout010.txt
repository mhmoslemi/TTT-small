def calculate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h_array = np.array(h)
    one_minus_h = 1.0 - h_array
    corr = correlate(h_array, one_minus_h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap
