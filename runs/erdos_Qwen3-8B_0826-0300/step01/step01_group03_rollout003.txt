import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    
    # Initial guess from pre-imported initial_h_values
    try:
        initial_h_values = initial_h_values
    except NameError:
        # If initial_h_values is not available, create a uniform distribution
        initial_h_values = np.full(n_points, 1.0 / n_points * 2.0)
    
    # Ensure initial_h_values is of correct length
    if len(initial_h_values) != n_points:
        initial_h_values = np.random.rand(n_points)
    
    # Normalize to satisfy the constraint sum(h) == n_points / 2
    initial_h_values = initial_h_values * (n_points / 2.0) / np.sum(initial_h_values)
    
    # Define function to compute the overlap integral
    def compute_c5(h):
        h = np.concatenate([h, h[1:]]).astype(float)  # Extend to handle full convolution
        one_minus_h = 1.0 - h
        # Compute the cross-correlation
        corr = np.correlate(h, one_minus_h, mode="full")
        # Take the maximum value and multiply by dx
        return np.max(corr) * dx
    
    # Define function to compute the objective for optimization
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    # Define constraints for optimization
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    maxiter = 100
    tol = 1e-5
    
    # Use SLSQP for constrained optimization
    result = optimize.minimize(
        objective,
        initial_h_values,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': maxiter, 'ftol': tol, 'disp': False}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we need to refine further
    if best_c5 > 0.380876:
        # Try increasing n_points if still within budget
        n_points = 200
        dx = 2.0 / n_points
        # Re-initialize with a random distribution
        initial_h_values = np.random.rand(n_points)
        initial_h_values = initial_h_values * (n_points / 2.0) / np.sum(initial_h_values)
        
        # Re-define compute_c5 and objective with new n_points
        def compute_c5_new(h):
            h = np.concatenate([h, h[1:]]).astype(float)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode="full")
            return np.max(corr) * dx
        
        def objective_new(x):
            h = x.reshape(n_points)
            return compute_c5_new(h)
        
        bounds_new = [(0.0, 1.0) for _ in range(n_points)]
        
        result_new = optimize.minimize(
            objective_new,
            initial_h_values,
            method='SLSQP',
            bounds=bounds_new,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0},
            options={'maxiter': maxiter, 'ftol': tol, 'disp': False}
        )
        
        best_h = result_new.x
        best_c5 = compute_c5_new(best_h)
    
    # Ensure we do not exceed the time budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Add some heuristic to perturb the solution slightly for exploration
        perturbation = np.random.rand(n_points) * 0.05
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2.0) / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution
        perturbed_c5 = compute_c5(perturbed_h)
        
        # Accept better solution
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    return best_h, best_c5, n_points
