import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder for the initial construction
evaluate_erdos_solution = lambda h: None  # Placeholder for the evaluation function

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 42  # Start with the initial construction's n_points
    
    # Try multiple starting points to avoid local optima
    for _ in range(3):  # Try 3 different initial constructions
        h = np.random.rand(n_points) * 2 - 1  # Random initial guess
        h = np.clip(h, 0, 1)  # Ensure h is within [0, 1]
        h = h * (n_points / 2) / np.sum(h)  # Normalize the integral to 1
        
        # Define the problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Objective: minimize the max of overlaps
        max_overlap = cp.Maximize(cp.Minimize(cp.maximum(cp.sum(h_var * (1 - h_var[k:])) for k in range(1, n_points))))
        prob = cp.Problem(max_overlap, constraints)
        
        # Solve the problem
        try:
            result = prob.solve(solver=cp.SCS, max_iter=100, time_limit=30)
            if result is not None:
                h_opt = h_var.value
                # Compute C5
                c5 = 0
                for k in range(1, n_points):
                    overlap = np.sum(h_opt * (1 - h_opt[k:])) * (2 / n_points)
                    c5 = max(c5, overlap)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        
        # Try increasing the number of points if within budget
        if time.time() - start_time < budget_s - 10:
            n_points = 60
            h = np.random.rand(n_points) * 2 - 1
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check for early termination
        if time.time() - start_time >= budget_s:
            break
    
    # Final check with best_h
    if best_h is not None:
        c5 = 0
        for k in range(1, len(best_h)):
            overlap = np.sum(best_h * (1 - best_h[k:])) * (2 / len(best_h))
            c5 = max(c5, overlap)
        return (best_h, c5, len(best_h))
    else:
        return (initial_h_values, 0.38092, 42)
