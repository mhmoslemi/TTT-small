import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from functools import partial
from time import time
from collections import deque

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points

    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-9}
        )
        return res.x

    def perturb_solution(solution, perturb_strength=0.05):
        perturbed = solution.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                perturbed[i] += np.random.uniform(-perturb_strength, perturb_strength)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        return perturbed

    def perturb_and_refine(solution, steps=10, perturb_strength=0.05):
        best_solution = solution.copy()
        best_fitness = compute_overlap(best_solution)
        for _ in range(steps):
            perturbed = perturb_solution(best_solution, perturb_strength)
            optimized = optimize_h(perturbed)
            current_fitness = compute_overlap(optimized)
            if current_fitness < best_fitness:
                best_solution = optimized
                best_fitness = current_fitness
        return best_solution, best_fitness

    def adaptive_search(solution, max_steps=20, perturb_strength=0.05):
        best_solution = solution.copy()
        best_fitness = compute_overlap(best_solution)
        for step in range(max_steps):
            perturb_strength = max(0.01, perturb_strength * 0.95)
            perturbed = perturb_solution(best_solution, perturb_strength)
            optimized = optimize_h(perturbed)
            current_fitness = compute_overlap(optimized)
            if current_fitness < best_fitness:
                best_solution = optimized
                best_fitness = current_fitness
        return best_solution, best_fitness

    def generate_initial_h():
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                initial_h[i] = 1.0
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        return initial_h

    start_time = time()
    initial_h = generate_initial_h()
    best_solution, best_fitness = perturb_and_refine(initial_h, steps=15)
    best_solution, best_fitness = adaptive_search(best_solution, max_steps=20)

    while time() - start_time < budget_s - 5:
        best_solution, best_fitness = perturb_and_refine(best_solution, steps=10, perturb_strength=0.05)

    return (best_solution, best_fitness, n_points)
