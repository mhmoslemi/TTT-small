from scipy.optimize import minimize
from scipy.signal import correlate
from functools import partial
from time import time
from collections import deque

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points

    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

        bounds = [(0, 1) for _ in range(n_points)]

        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return res.x

        perturbed = solution.copy()

        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
