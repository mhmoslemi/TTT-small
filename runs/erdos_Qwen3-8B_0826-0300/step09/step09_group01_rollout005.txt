import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from functools import partial
from time import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points / 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    initial_h = np.concatenate([initial_h, initial_h[:int(n_points/2)]])
    initial_h = np.random.permutation(initial_h)
    initial_h = initial_h[:n_points]
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return res.x

    def perturb_solution(solution, perturb_strength=0.05):
        perturbed = solution.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                perturbed[i] += np.random.uniform(-perturb_strength, perturb_strength)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        return perturbed

    def simulated_annealing(initial_solution, temp=1.0, cooling_rate=0.995, steps=1000):
        current_solution = initial_solution.copy()
        current_cost = compute_overlap(current_solution)
        best_solution = current_solution.copy()
        best_cost = current_cost

        for step in range(steps):
            perturbed = perturb_solution(current_solution)
            perturbed_cost = compute_overlap(perturbed)

            delta = perturbed_cost - current_cost
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_solution = perturbed
                current_cost = perturbed_cost

            if current_cost < best_cost:
                best_solution = current_solution.copy()
                best_cost = current_cost

            temp *= cooling_rate

        return best_solution, best_cost

    def multi_start_optimization(initial_solution, num_starts=15):
        best_solution = initial_solution.copy()
        best_fitness = compute_overlap(best_solution)
        for _ in range(num_starts):
            perturbed = perturb_solution(best_solution)
            optimized = simulated_annealing(perturbed)
            current_fitness = optimized[1]
            if current_fitness < best_fitness:
                best_solution = optimized[0]
                best_fitness = current_fitness
        return best_solution, best_fitness

    start_time = time()
    best_solution, best_fitness = multi_start_optimization(initial_h, num_starts=15)

    while time() - start_time < budget_s - 5:
        perturbed = perturb_solution(best_solution)
        optimized = simulated_annealing(perturbed)
        current_fitness = optimized[1]
        if current_fitness < best_fitness:
            best_solution = optimized[0]
            best_fitness = current_fitness

    return (best_solution, best_fitness, n_points)
