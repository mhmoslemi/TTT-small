from scipy.optimize import minimize
from scipy.signal import correlate

from collections import deque

        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return res.x

    def perturb_solution(solution, perturb_strength=0.05):
        perturbed = solution.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                perturbed[i] += np.random.uniform(-perturb_strength, perturb_strength)
        perturbed = np.clip(perturbed, 0, 1)
        # Normalize to maintain the integral constraint
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        return perturbed

    def local_search(solution, steps=10):
        best_solution = solution.copy()
        best_fitness = compute_overlap(best_solution)
        for _ in range(steps):
            perturbed = perturb_solution(best_solution)
            optimized = optimize_h(perturbed)
            current_fitness = compute_overlap(optimized)
            if current_fitness < best_fitness:
                best_solution = optimized
                best_fitness = current_fitness
        return best_solution, best_fitness

    def multi_start_optimization(initial_solution, num_starts=25, perturb_strength=0.05):
        best_solution = initial_solution.copy()
        best_fitness = compute_overlap(best_solution)
        for _ in range(num_starts):
            perturbed = perturb_solution(best_solution, perturb_strength)
            optimized = optimize_h(perturbed)
            current_fitness = compute_overlap(optimized)
            if current_fitness < best_fitness:
                best_solution = optimized
                best_fitness = current_fitness
        return best_solution, best_fitness

    def greedy_search(solution, max_steps=15):
        best_solution = solution.copy()
        best_fitness = compute_overlap(best_solution)
        for _ in range(max_steps):
            perturbed = perturb_solution(best_solution)
            optimized = optimize_h(perturbed)
            current_fitness = compute_overlap(optimized)
            if current_fitness < best_fitness:
                best_solution = optimized
                best_fitness = current_fitness
        return best_solution, best_fitness

    def refine_solution(solution, steps=15):
        best_solution = solution.copy()
        best_fitness = compute_overlap(best_solution)
        for _ in range(steps):
            perturbed = perturb_solution(best_solution)
            optimized = optimize_h(perturbed)
            current_fitness = compute_overlap(optimized)
            if current_fitness < best_fitness:
                best_solution = optimized
                best_fitness = current_fitness
        return best_solution, best_fitness

    best_solution, best_fitness = multi_start_optimization(initial_h, num_starts=25)

        current_fitness = compute_overlap(optimized)
        if current_fitness < best_fitness:
            best_solution = optimized
            best_fitness = current_fitness
