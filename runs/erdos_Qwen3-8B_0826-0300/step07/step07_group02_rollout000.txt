import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from functools import partial

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 256
    dx = 2.0 / n_points
    integral_constraint = n_points / 2

    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_constraint}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return res.x

    def perturb_h(h_initial, perturb_scale=0.05):
        perturb = np.random.uniform(-perturb_scale, perturb_scale, size=n_points)
        perturbed_h = np.clip(h_initial + perturb, 0, 1)
        if np.sum(perturbed_h) == 0:
            return np.zeros(n_points)
        perturbed_h = perturbed_h * (integral_constraint / np.sum(perturbed_h))
        return perturbed_h

    def refine_solution(h_initial):
        optimized_h = optimize_h(h_initial)
        for _ in range(10):
            perturbed_h = perturb_h(optimized_h, perturb_scale=0.01)
            optimized_h = optimize_h(perturbed_h)
        return optimized_h

    def generate_initial_h():
        h = np.zeros(n_points)
        for i in range(n_points // 3):
            h[i] = 1
        for i in range(n_points // 2, 2 * n_points // 3):
            h[i] = 1
        h = h * (integral_constraint / np.sum(h))
        return h

    best_c5 = compute_overlap(best_h)

    for _ in range(10):
        perturbed_h = perturb_h(best_h, perturb_scale=0.1)
        optimized_h = refine_solution(perturbed_h)
        current_c5 = compute_overlap(optimized_h)
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break

    optimized_h = optimize_h(best_h)
    final_c5 = compute_overlap(optimized_h)
    if final_c5 < best_c5:
        best_h = optimized_h
        best_c5 = final_c5

    return (best_h, best_c5, n_points)
