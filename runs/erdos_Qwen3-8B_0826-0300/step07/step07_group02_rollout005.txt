import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from functools import partial

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    integral_constraint = n_points / 2
    
    # Initialize with a random but normalized distribution
    h_initial = np.random.uniform(0, 1, n_points)
    h_initial = h_initial * (integral_constraint / np.sum(h_initial))
    
    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        return np.max(corr) * dx
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)
    
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_constraint}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return res.x
    
    def refine_solution(h_initial, num_refines=8, perturb_scale=0.01):
        optimized_h = optimize_h(h_initial)
        for _ in range(num_refines):
            perturb = np.random.uniform(-perturb_scale, perturb_scale, size=n_points)
            perturbed_h = np.clip(optimized_h + perturb, 0, 1)
            if np.sum(perturbed_h) == 0:
                perturbed_h = np.zeros(n_points)
            else:
                perturbed_h = perturbed_h * (integral_constraint / np.sum(perturbed_h))
            optimized_h = optimize_h(perturbed_h)
        return optimized_h
    
    def perturb_and_optimize(h_initial, perturb_scale=0.1):
        perturb = np.random.uniform(-perturb_scale, perturb_scale, size=n_points)
        perturbed_h = np.clip(h_initial + perturb, 0, 1)
        if np.sum(perturbed_h) == 0:
            perturbed_h = np.zeros(n_points)
        else:
            perturbed_h = perturbed_h * (integral_constraint / np.sum(perturbed_h))
        optimized_h = optimize_h(perturbed_h)
        return optimized_h
    
    best_h = h_initial.copy()
    best_c5 = compute_overlap(best_h)
    
    num_starts = 15
    for _ in range(num_starts):
        perturbed_h = perturb_and_optimize(best_h)
        refined_h = refine_solution(perturbed_h)
        current_c5 = compute_overlap(refined_h)
        if current_c5 < best_c5:
            best_h = refined_h
            best_c5 = current_c5
    
    refined_optimized_h = refine_solution(best_h)
    final_c5 = compute_overlap(refined_optimized_h)
    if final_c5 < best_c5:
        best_h = refined_optimized_h
        best_c5 = final_c5
    
    final_optimized_h = optimize_h(best_h)
    final_c5 = compute_overlap(final_optimized_h)
    if final_c5 < best_c5:
        best_h = final_optimized_h
        best_c5 = final_c5
    
    return (best_h, best_c5, n_points)
