import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from global variable if provided
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        n_initial = len(initial_h_values)
    else:
        initial_h_values = np.zeros(200)
        n_initial = 200
    
    # Start with a moderate number of points
    n_points = 100
    max_points = 250  # Maximum number of points to consider
    best_c5 = 1.0
    best_h = np.zeros(n_points)
    best_n_points = n_points
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        dx = 2.0 / n_points
        h_extended = np.concatenate([h_values, h_values[1:]])
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to compute the integral constraint
    def constraint(x):
        h = x.reshape(n_points)
        integral = np.sum(h) * (2.0 / n_points)
        return integral - 1.0
    
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    # Optimization with bounds and constraints
    x_bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def optimize_with_initial(initial_h_values, n_points):
        if len(initial_h_values) != n_points:
            initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h_initial = np.clip(initial_h_values, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        result = optimize.minimize(
            objective,
            h_initial.flatten(),
            bounds=x_bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-6, 'eps': 1e-8}
        )
        return result.fun, result.x
    
    # Initial optimization
    if 'initial_h_values' in globals():
        initial_c5, initial_h = optimize_with_initial(initial_h_values, n_points)
        best_c5 = initial_c5
        best_h = initial_h
        best_n_points = n_points
    
    # Try increasing the number of points if within budget
    start_time = time.time()
    for n_points in range(150, max_points + 1, 50):
        if time.time() - start_time > budget_s:
            break
        
        # Interpolate initial h to new number of points
        new_x = np.linspace(0, 2, n_points)
        old_x = np.linspace(0, 2, len(initial_h_values))
        h_initial = np.interp(new_x, old_x, initial_h_values)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Re-define compute_c5 and objective for new n_points
        def compute_c5_new(h_values):
            dx = 2.0 / n_points
            h_extended = np.concatenate([h_values, h_values[1:]])
            correlation = np.correlate(h_values, 1 - h_values, mode='full')
            c5 = np.max(correlation) * dx
            return c5
        
        def objective_new(x):
            h = x.reshape(n_points)
            return compute_c5_new(h)
        
        def constraint_new(x):
            h = x.reshape(n_points)
            integral = np.sum(h) * (2.0 / n_points)
            return integral - 1.0
        
        x_bounds_new = [(0.0, 1.0) for _ in range(n_points)]
        cons_new = [{'type': 'eq', 'fun': constraint_new}]
        
        result_new = optimize.minimize(
            objective_new,
            h_initial.flatten(),
            bounds=x_bounds_new,
            constraints=cons_new,
            method='SLSQP',
            options={'maxiter': 30, 'ftol': 1e-6, 'eps': 1e-8}
        )
        new_c5 = result_new.fun
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = result_new.x
            best_n_points = n_points
    
    # Final check and normalization
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (best_n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, best_n_points
