import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from functools import partial
from time import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute overlap integral
    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Objective function for optimization
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)
    
    # Constraint for integral
    def integral_constraint(x):
        return np.sum(x) - n_points / 2
    
    # Optimization function with bounds and constraints
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': integral_constraint}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-10}
        )
        return res.x
    
    # Strategy: Use a genetic algorithm for global search and refine with local optimization
    def genetic_algorithm(population_size, generations, mutation_rate, crossover_rate):
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([compute_overlap(ind) for ind in population])
            
            # Select parents
            parents_indices = np.argsort(fitness)[:int(population_size / 2)]
            parents = population[parents_indices]
            
            # Crossover
            children = []
            for _ in range(population_size - len(parents)):
                parent1 = parents[np.random.choice(parents_indices)]
                parent2 = parents[np.random.choice(parents_indices)]
                child = np.where(np.random.rand(n_points) < crossover_rate, parent1, parent2)
                children.append(child)
            
            # Mutation
            for child in children:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.uniform(-0.05, 0.05)
                        child[i] = np.clip(child[i], 0, 1)
                        child[i] = child[i] * (n_points / 2) / np.sum(child)
            
            # Combine population
            population = np.vstack([parents, np.array(children)])
            
            # Keep only the best individuals
            population = population[np.argsort(fitness)[:population_size]]
        
        best_idx = np.argmin(fitness)
        return population[best_idx], fitness[best_idx]
    
    # Start with genetic algorithm
    start_time = time()
    best_solution, best_fitness = genetic_algorithm(population_size=50, generations=20, mutation_rate=0.1, crossover_rate=0.8)
    
    # Refine with local optimization
    for _ in range(10):
        if time() - start_time > budget_s - 5:
            break
        perturbed = best_solution.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                perturbed[i] += np.random.uniform(-0.01, 0.01)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        optimized = optimize_h(perturbed)
        current_fitness = compute_overlap(optimized)
        if current_fitness < best_fitness:
            best_solution = optimized
            best_fitness = current_fitness
    
    return (best_solution, best_fitness, n_points)
