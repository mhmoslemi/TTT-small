            method='SLSQP',

            constraints=constraints,

        )
        return res.x

    def kmeans_clustering(h, n_clusters):
        h_array = np.array(h)
        h_rescaled = h_array / np.max(h_array)
        kmeans = KMeans(n_clusters=n_clusters, n_init=10).fit(h_rescaled.reshape(-1, 1))
        labels = kmeans.labels_
        centers = kmeans.cluster_centers_.flatten()
        h_clustered = np.zeros_like(h_array)
        for i in range(n_clusters):
            h_clustered[labels == i] = centers[i]
        return h_clustered
