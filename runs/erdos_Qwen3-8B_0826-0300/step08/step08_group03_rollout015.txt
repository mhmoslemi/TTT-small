from scipy.optimize import minimize, Bounds

        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [

        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return res.x

    def perturb_and_normalize(h_initial, perturb_scale=0.05):
        perturb = np.random.uniform(-perturb_scale, perturb_scale, size=n_points)

        if np.sum(perturbed_h) == 0:
            return np.zeros(n_points)
        perturbed_h = perturbed_h * (integral_constraint / np.sum(perturbed_h))
        return perturbed_h

        optimized_h = optimize_h(h_initial)

            perturbed_h = perturb_and_normalize(optimized_h, perturb_scale=perturb_scale)
