from scipy.optimize import minimize
from scipy.signal import correlate

    def compute_overlap(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    # Function to optimize with bounds and constraints
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_constraint}
        ]
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return res.x

    # Function to perturb and normalize the solution
    def perturb_and_normalize(h_current, perturbation_scale=0.05):
        h = h_current.copy()
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros(n_points)
        h = h * (integral_constraint / np.sum(h))
