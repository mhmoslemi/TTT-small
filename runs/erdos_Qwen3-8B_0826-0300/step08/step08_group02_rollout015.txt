    def compute_overlap(h):

        corr = correlate(h_array, 1 - h_array, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    # Objective function for optimization with bounds and constraints
    def objective_with_bounds(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_overlap(h)

    # Constraint to maintain integral constraint
    def integral_constraint(x):
        return np.sum(x) - integral_constraint

    # Bounds for the variables
    bounds = [(0, 1) for _ in range(n_points)]

    # Function to perform local optimization with SLSQP
    def optimize_local(h_initial):
        result = minimize(
            objective_with_bounds,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': integral_constraint}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x

    # Function to perform local optimization with L-BFGS-B
    def optimize_lbfgs(h_initial):
        result = minimize(
            objective_with_bounds,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': integral_constraint}],
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x
