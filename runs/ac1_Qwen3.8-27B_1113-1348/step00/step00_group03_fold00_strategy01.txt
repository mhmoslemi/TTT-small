# Numerical gradient function for the objective
def numerical_gradient(f, x, epsilon=1e-8):
    n = len(x)
    grad = np.zeros(n)
    for i in range(n):
        # We'll backow the gradient using central difference
        perturbed = x.copy()
        perturbed[i] = x[i] + epsilon if x[i] > 0 else x[i]
        f_forward = f(perturbed)
        
        perturbed[i] = x[i] - epsilon if x[i] > 0 else x[i]
        if x[i] - epsilon < 0:  # Underflow safety in function evaluation
            f_backward = np.inf
        else:
            f_backward = f(perturbed)
            
        grad[i] = (f_forward - f_backward) / (2 * epsilon)
    return grad
