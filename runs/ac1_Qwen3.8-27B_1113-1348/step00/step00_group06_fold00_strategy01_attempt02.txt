            return np.inf
            
    # Convert to floats for consistency and protect against negatives
    seq_array = np.array([np.float64(x) for x in seq_array])

    # Ensure non-negative
    if np.any(seq_array < 0):
        return np.inf
        
    # Square and bound; clipping is costly but might be okay since n厳しいis limited
    if np.max(seq_array) > 1000:
        return np.inf
        
    n = seq_length
    b = np.convolve(seq_array, seq_array, mode="full")
    max_b_val = np.max(b)
    sum_a_val = np.sum(seq_array)
    
    # If the sum is too close to zero, it would be considered invalid by the evaluator
    if sum_a_val < 0.01:
        return np.inf

def normalize_sequence(seq):
    """Normalize a sequence to have a fixed L1 norm."""
    current_sum = np.sum(seq)
    desired_sum = 0.01  # Chosen for numerical stability and meaningful comparison
    scaling_factor = desired_sum * (1 + 1e-7) / current_sum  # Prevent exactly zero
    return seq * scaling_factor / (np.max(seq) if np.any(seq) else 1)  # Clip to prevent overflow
