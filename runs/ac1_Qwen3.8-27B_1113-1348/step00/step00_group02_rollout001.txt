```
def gaussian_block_perturb(base, rng, sigma):
    n=base.size
    options = (16, 48, 128, 256, 512, 1024)
    # choose constrained by n
    valid=[]
    for w in options:
       if w < n: valid.append(w)
    if not valid: valid=[max(1,n//2)]
    w = int(valid[int(rng.integers(0, len(valid)))])
    s = int(rng.integers(0, max(1, n - w + 1)))
    cand = np.array(base, dtype=np.float64, copy=True)
    # generate center offset
    center = s + 0.5*w
    idx = np.arange(s, s+w, dtype=np.float64)
    t = (idx - center) / (0.25*w)
    mask = np.exp(-t*t)
    noise = rng.normal(0.0, sigma, w) * mask
    cand[s:s+w] *= np.exp(noise)
    cand = np.clip(cand, 0.0, 1000.0)
    return normalize_mean(cand, 1.0)
```

```
def lowfreq_mult_perturb(base, rng, sigma, time_vec, max_freq):
```

```python
"""
Search for a nonnegative low-autoconvolution step sequence.

The evaluator computes
    2*n*max(conv(a,a)) / (sum(a))^2.
A strong baseline is a discrete quadratic ramp:
    a_i ~= 3*(1 - i/n)^2,
whose continuous limit gives score about 1.4748, below 1.5027.
The program then searches smooth perturbations of this baseline and simple
smooth power-law shapes using fast FFT convolution estimates.  It keeps a
verified exact-evaluation safe fallback and returns the best exact-verified
sequence when possible.
"""

import math
import time
import numpy as np


def next_power_of_two(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fast_convmax(a):
    """Return the maximum coefficient of the real full convolution a*a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0

    m = next_power_of_two(2 * n - 1)
    fa = np.fft.rfft(a, m)
    fa = fa * fa
    c = np.fft.irfft(fa, m)

    out_len = 2 * n - 1
    if c.size > out_len:
        c = c[:out_len]
    else:
        c = np.pad(c, (0, out_len - c.size), mode="constant")

    mx = float(np.max(c))
    if mx < 0.0 and mx > -1e-9:
        mx = 0.0
    return mx


def fast_score(a):
    """Fast FFT estimate of evaluate_sequence, with clipping safeguards."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf

    cmax = fast_convmax(a)
    return float(2.0 * a.size * cmax / (s * s))


def exact_score(a):
    """Call the authoritative evaluator if it is available."""
    a = np.asarray(a, dtype=np.float64)
    fn = globals().get("evaluate_sequence")
    if not callable(fn):
        return np.inf
    try:
        return float(fn(a.tolist()))
    except Exception:
        return np.inf


def normalize_mean(a, target=1.0):
    """Clip to valid range and rescale to the desired mean."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(np.sum(a))
    if s <= 0.0:
        return a

    a *= target / s
    a = np.clip(a, 0.0, 1000.0)
    return a


def make_power_sequence(n, p=1.0, alpha=2.0):
    """
    Make a smooth one-sided power-law sequence:
        a_i ~ (1 - (i/n)^p)^alpha,
    normalized to mean 1.  For p=1 and alpha=2 this is the quadratic baseline.
    """
    if n <= 0:
        return np.array([], dtype=np.float64)

    x = np.arange(n, dtype=np.float64) / float(n)
    base = 1.0 - np.power(x, float(p))
    base = np.clip(base, 0.0, 1.0)
    v = np.power(base, float(alpha))
    return normalize_mean(v, 1.0)


def lowfreq_mult_perturb(base, rng, sigma, time_vec, max_freq=24):
    """Multiplicative smooth perturbation using a few low-frequency sines."""
    n = base.size
    comps = int(rng.integers(1, 4))
    noise = np.zeros(n, dtype=np.float64)
    scale = float(sigma) / math.sqrt(float(max(1, comps)))

    for _ in range(comps):
        freq = float(rng.integers(1, max_freq + 1))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        amp = float(rng.normal(0.0, scale))
        noise += amp * np.cos(2.0 * math.pi * freq * time_vec + phase)

    cand = base * np.exp(noise)
    cand = np.where(np.isfinite(cand), cand, 0.0)
    cand = np.clip(cand, 0.0, 1000.0)
    return normalize_mean(cand, 1.0)


def gaussian_block_perturb(base, rng, sigma):
    """Multiplicative Gaussian-block perturbation."""
    n = base.size
    cand = np.array(base, dtype=np.float64, copy=True)

    sizes = (16, 32, 64, 128, 256, 512, 1024)
    valid = [w for w in sizes if w < n]
    if not valid:
        valid = [max(1, n // 2)]

    w = int(valid[int(rng.integers(0, len(valid)))]))
    s = int(rng.integers(0, max(1, n - w + 1)))

    idx = np.arange(s, s + w, dtype=np.float64)
    center = float(s + 0.5 * w)
    t = (idx - center) / (0.25 * float(w))
    mask = np.exp(-t * t)
    noise = rng.normal(0.0, float(sigma), w) * mask

    cand[s:s + w] *= np.exp(noise)
    cand = np.where(np.isfinite(cand), cand, 0.0)
    cand = np.clip(cand, 0.0, 1000.0)
    return normalize_mean(cand, 1.0)


def local_search(base, score, rng, deadline, max_tries=40000):
    """Smooth multiplicative hill climbing."""
    best = np.array(base, dtype=np.float64, copy=True)
    best_score = float(score)
    n = best.size
    if n == 0:
        return best, best_score

    time_vec = np.arange(n, dtype=np.float64) / float(n)
    sigma = 0.08
    no_improve = 0
    improvements = 0
    tries = 0

    while time.monotonic() < deadline and tries < max_tries:
        mode = float(rng.random())
        if mode < 0.70:
            cand = lowfreq_mult_perturb(best, rng, sigma, time_vec, 24)
        elif mode < 0.90:
            cand = gaussian_block_perturb(best, rng, sigma)
        else:
            cand = gaussian_block_perturb(best, rng, sigma * 0.45)

        sc = fast_score(cand)
        if math.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            no_improve = 0
            improvements += 1
            sigma = min(0.25, sigma * 1.03)
            if improvements % 10 == 0:
                print("local improvement {}".format(improvements), best_score)
        else:
            no_improve += 1
            if no_improve > 250:
                sigma = max(0.005, sigma * 0.90)
                no_improve = 0

        tries += 1

    print("local search finished", best_score, "tries", tries)
    return best, best_score


def scan_power_shapes(n_scan, deadline, top_k=6):
    """Scan a small family of smooth power-law shapes by fast score."""
    p_vals = (0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0, 3.0)
    a_vals = (1.3, 1.5, 1.7, 1.8, 1.9, 2.0, 2.1, 2.3, 2.6)

    top = []
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        for alpha in a_vals:
            if time.monotonic() >= deadline:
                break

            seq = make_power_sequence(n_scan, p, alpha)
            sc = fast_score(seq)
            if math.isfinite(sc):
                top.append([sc, float(p), float(alpha)])
                top.sort()
                if len(top) > top_k:
                    top = top[:top_k]

    return top

    if time.monotonic() < hard_deadline - 120.0:
        scan_deadline = min(hard_deadline - 120.0, t_start + 60.0)
        top_shapes = scan_power_shapes(512, scan_deadline, top_k=6)
        for item in top_shapes:
            scan_sc, p, alpha = item
            if time.monotonic() >= hard_deadline - 180.0:
                break
            seq = make_power_sequence(N, p, alpha)
            sc = fast_score(seq)
            if sc < search_fast - 1e-12:
                search_seq = seq
                search_fast = sc
                print("parameter shape improved", p, alpha, sc)

    # Polish the best FFT-scored sequence with smooth multiplicative local search.
    local_deadline = hard_deadline - 30.0
    if time.monotonic() < local_deadline and search_seq.size > 0:
        search_seq, search_fast = local_search(
            search_seq,
            search_fast,
            rng,
            local_deadline,
            max_tries=40000
        )

    # Exact verification of the searched sequence if time remains.
    search_exact = np.inf
    if time.monotonic() < hard_deadline - 10.0:
        search_exact = exact_score(search_seq)
        print("searched candidate exact score", search_exact)

    # Prefer exact-verified improvements; otherwise conservatively return the safe baseline.
    if math.isfinite(safe_exact):
        if math.isfinite(search_exact) and search_exact + 1e-12 < safe_exact:
            return search_seq.tolist()
        return safe_seq.tolist()

    # If exact evaluation was unavailable, use fast scores.
    if search_fast < safe_fast:
        return search_seq.tolist()
    return safe_seq.tolist()
```
