def _f(x):
    n = _state["n"]; p = _state["p"]; N = _state["N"]
    a = _softmax(x.astype(np.float64, copy=False))
    S = float(a.sum())
    b = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(a, N), N)[:2 * n - 1]
    b = np.clip(b, 0.0, None)
    bmax = float(b.max())
    if bmax <= 0:
        return 2.0 * n  # should not happen
    xb = b / bmax
    m = bmax * float(np.sum(xb ** p)) ** (1.0 / p)
    W = xb ** (p - 1)
    _state["b"] = b; _state["W"] = W; _state["bmax"] = bmax
    _state["m"] = m; _state["ok"] = True
    return float((2.0 * n * m) / (S * S))

"""
Search for a non-negative height sequence minimizing

    O(a) = 2*n*max(a*a) / (sum(a))^2

where a*a is the full linear convolution (length 2n-1) and n = len(a).

Method
------
1. O is scale-invariant, so we parameterize a = softmax_e(u) (sum a = 1)
   and replace the max of b = a*a by a p-norm soft-max.  p is annealed
   (8 -> 16 -> 32 -> 80 -> 160 -> 256) so the final points approximate the
   true max very closely while gradients stay smooth.
2. A full analytic gradient of the soft-max objective w.r.t. log-heights is
   available, computed with three FFT convolutions per evaluation.
3. SLSQP (L-BFGS-B fallback, then 2-point finite differences as a last
   resort) is run from many seeds: gaussian bumps, cosine arches of several
   exponents, dense geometric ramps (several ratios), trapezoids and
   plateau bumps, plus the imported previous winners.
4. Best candidates are re-polished at resampled nearby lengths.
5. A cheap final exact-evaluation "hot mutation" phase (single-site, block
   and mass-move perturbations) squeezes out residual gain.
6. Every accepted improvement is validated by the official
   evaluate_sequence; a constant baseline is guaranteed valid.
"""

import time
import math

import numpy as np
from scipy import optimize

_TIME_BUDGET = 2400.0
_HARD_DEADLINE = _TIME_BUDGET - 10.0

_GRAD_OK = [True]

_best_list = [None]


def _record(seq):
    """Evaluate exactly; update global best if better (finite)."""
    try:
        score = float(evaluate_sequence(seq))
    except Exception:
        return
    if not np.isfinite(score):
        return
    cur = _best_score[0]

        _best_score[0] = score
        _best_list[0] = [float(v) for v in seq]
        print("record: score=%.9f  n=%d" % (score, len(seq)), flush=True)


def _next_pow2(m):
    p = 1
    while p < m:
        p *= 2
    return p


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    t = u - u.max()
    e = np.exp(t)
    return e / e.sum()


# --- objective / gradient state -------------------------------------------
_state = {"n": None, "p": None, "N": None, "ok": False}


def _f(x):
    n = _state["n"]
    N = _state["N"]
    p = _state["p"]
    a = _softmax(x)
    b = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(a, N), N)[:2 * n - 1]
    b = np.clip(b, 0.0, None)
    bmax = float(b.max())
    if bmax <= 1e-300:
        return 2.0 * n
    xb = b / bmax
    m = bmax * float(np.sum(xb ** p)) ** (1.0 / p)
    W = xb ** (p - 1)
    S = float(a.sum())
    _state["W"] = W
    _state["m"] = m
    _state["B"] = b
    _state["ok"] = True
    return float((2.0 * n * m) / (S * S))


def _g(x):
    n = _state["n"]
    N = _state["N"]
    m = _state["m"]
    W = _state["W"]
    a = _softmax(x)
    A = np.fft.rfft(a, N)
    Wf = np.fft.rfft(W, N)
    Wh = np.fft.rfft(W[::-1], N)
    AW = np.fft.irfft(A * Wf, N)          # (a * W)
    Cafe = np.fft.irfft(A * Wh, N)        # (a * rev(W))
    G = Cafe[n - 1:2 * n - 1][::-1].copy()
    G += AW[0:n]
    dm = float(np.dot(a, a))
    da = (2.0 * n / (m * dm)) * G
    dm1 = float(np.dot(da, a))
    return a * (da - dm1)


def _set_lp(n, p):
    n = int(n)
    _state["n"] = n
    _state["p"] = int(p)
    _state["N"] = _next_pow2(3 * n - 1)
    _state["ok"] = False
    _state["W"] = None
    _state["m"] = None
    _state["B"] = None


def _grad_check(x):
    n = x.size
    if n > 64:
        return True
    tol = 5e-4
    f0 = _f(x)
    g0 = _g(x)
    idx = np.arange(0, n, max(1, n // 16))
    eps = 1e-6
    mx = 0.0
    for i in idx:
        xp = x.copy()
        xp[i] += eps
        fm = x.copy()
        fm[i] -= eps
        fd = (_f(xp) - _f(fm)) / (2.0 * eps)
        d = abs(fd - g0[i])
        mx = max(mx, d / max(1.0, abs(g0[i])))
    return mx < tol


def _one_phase(u, n, p, maxiter, soft_tol):
    _set_lp(n, p)
    v0 = _f(u)
    if not np.isfinite(v0):
        return u, v0
    last_v = v0
    for tries in range(2 if p >= 120 else 1):
        try:
            res = optimize.minimize(
                _f, u, jac=(_g if _GRAD_OK[0] else "2-point"),
                method="SLSQP", options={"maxiter": maxiter, "ftol": 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                v1 = float(_f(res.x))
                if np.isfinite(v1) and v1 < last_v:
                    last_v = v1
                    u = res.x
                if last_v > v0 * (1.0 - soft_tol):
                    break
        except Exception as e:
            print("slsqp n=%d p=%d: %s" % (n, p, e), flush=True)
            break
    return u, last_v


def _polish(u, n, pchain, deadline, maxiter):
    """Run the p-annealing; return (u_with_best_exact, exact_score)."""
    best_u = u.copy()
    best_exact = float("inf")
    a = _softmax(u)
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc):
        best_exact = sc
    for p in pchain:
        if time.time() > deadline - 60.0:
            break
        _set_lp(n, p)
        ok = _grad_check(u) if _GRAD_OK[0] else False
        if not ok:
            print("gradient check failed (n=%d, p=%d); using fd" % (n, p), flush=True)
            _GRAD_OK[0] = False
            u, _ = _one_phase(u, n, p, min(60, maxiter), 0.0)
        else:
            u, v = _one_phase(u, n, p, maxiter, 0.0)
        a = _softmax(u)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_u = u.copy()
    return best_u, best_exact


def _mk_seeds(n, rng):
    out = []
    x = (np.arange(n) - (n - 1) / 2.0)
    for alpha in (0.30, 0.42, 0.55):
        out.append(-0.5 * (x / (alpha * n)) ** 2)
    z = (np.arange(n) + 0.5) / n
    for q in (0.5, 1.0, 2.5, 6.0):
        out.append(np.sin(math.pi * z) ** q)
    for r in (0.72, 0.84, 1.20, 1.39):
        out.append(np.log(r) * np.arange(n))
    c = np.clip(1.0 - np.abs(x) / (0.42 * n), 0.10, 1.0)
    out.append(np.log(c + 1.0e-9))
    out.append(np.log(z + 0.1) + np.log((1.0 - z) + 0.1) * 0.5)   # doorway
    return [np.clip(np.asarray(s, dtype=np.float64), -40.0, 40.0) for s in out]


def _resample(a, m):
    n = len(a)
    if m == n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, m)
    return np.interp(x1, x0, a) + 1.0e-12


def _hot_loop(a_best, deadline, rng):
    a = np.asarray(a_best, dtype=np.float64).copy()
    n = len(a)
    best = float(evaluate_sequence(a.tolist()))
    it = 0
    while time.time() < deadline and (it <= 2000 or n <= 900):
        it += 1
        r = float(rng.random())
        if r < 0.5:
            cand = a.copy()
            size = max(1, min(n, int(rng.integers(1, max(2, n // 30)))))
            start = int(rng.integers(0, max(1, n - size + 1)))
            stop = min(n, start + size)
            cand[start:stop] *= math.exp(float(rng.normal(0.0, 0.22)))
            sc = float(evaluate_sequence(cand.tolist()))
        elif r < 0.8:
            cand = a.copy()
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2:
                d = float(rng.uniform(0.15, 0.5)) * a[i1]
                if a[i1] - d > 1e-6:
                    cand[i1] = a[i1] - d
                    cand[i2] = a[i2] + d
                    sc = float(evaluate_sequence(cand.tolist()))
                else:
                    continue
            else:
                continue
        else:
            cand = a.copy()
            j = int(rng.integers(n))
            cand[j] *= math.exp(float(rng.normal(0.0, 0.55)))
            sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best - 1e-12:
            best = sc
            a = cand
            _record(a.tolist())
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search entry point; always returns a valid best sequence."""
    t0 = time.time()
    budget = min(float(budget_s), _TIME_BUDGET)
    d_pool = t0 + budget - 380.0
    d_res = t0 + budget - 150.0
    d_hot = t0 + budget - 8.0
    rng = np.random.default_rng(987654321 + 1000003 * int(seed))
    print("starting, budget=%.0f s" % budget, flush=True)

    # Guaranteed valid baseline.
    base = [0.25] * 40
    _record(base)

    # Imported previous winners as seeds / candidates.
    imports = []
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        pv = np.clip(pv[~np.isnan(pv) & ~np.isinf(pv)], 0.0, 1000.0)
        if pv.size >= 2 and np.isfinite(pv).all() and pv.sum() >= 0.01:
            imports.append(pv)
            _record(pv.tolist())
    except Exception:
        pass

    pchain_short = (8, 16, 48, 120)
    pchain_full = (8, 16, 32, 80, 160, 256)

    # ---- Phase 1: main pool over lengths ---------------------------------
    npool = [22, 28, 36, 44, 54, 66, 80, 98, 120, 146, 178, 216, 262,
             318, 384, 460, 548, 656, 780, 920, 1080]
    for k, n in enumerate(npool):
        if time.time() > d_pool - 25.0:
            break
        remaining = d_pool - time.time()
        per_n = max(12.0, remaining / max(1.0, (len(npool) - k)))
        seeds = _mk_seeds(n, rng)
        cands = []
        for idx, s in enumerate(seeds):
            if time.time() > d_pool - 12.0 or time.time() > t0 + per_n * 2.0:
                break
            a = _softmax(s)
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc):
                cands.append((sc, idx, s))
        cands.sort()
        if n >= 260:
            K = 2
        elif n >= 120:
            K = 3
        else:
            K = 4
        chain = pchain_full if n <= 320 else pchain_short
        for i, (sc0, _, s) in enumerate(cands[:K]):
            if time.time() > d_pool - 20.0:
                break
            if sc0 > 2.12 and i > 0:
                continue
            maxit = int(min(420, max(50, 240.0 * 340.0 / max(120, n))))
            u, ex = _polish(s.copy(), n, chain, d_pool, maxit)
            print("n=%d seed=%d exact=%.9f (t=%.0f)" % (n, i, ex, time.time() - t0),
                  flush=True)

    # ---- Phase 2: resampled lengths around best candidate ----------------
    b = _best_list[0]
    if b is not None:
        ab = np.asarray(b, dtype=np.float64)
        m0 = len(ab)
        mset = [17, 27, 43, 71, 111, 179, 288, 463, 748, 1200]
        fact = [0.70, 0.80, 0.90, 1.12, 1.30]
        for f in fact:
            m = max(15, int(round(m0 * f)))
            if m > 1300 or m in mset:
                continue
            mset.append(m)
        mset = sorted(mset)
        for idx, m in enumerate(mset):
            if time.time() > d_res - 15.0:
                break
            ar = _resample(ab, m)
            u0 = np.log(ar + 1.0e-9)
            maxit = int(min(300, max(40, 180.0 * 300.0 / max(120, m))))
            chain2 = (16, 48, 120) if m > 400 else (8, 24, 96, 192)
            u, ex = _polish(u0, m, chain2, d_res, maxit)
            print("resample m=%d exact=%.9f (t=%.0f)" % (m, ex, time.time() - t0),
                  flush=True)
        # Re-polish imported long winners briefly (cheap FFT at any n).
        for pv in imports:
            if time.time() > d_res - 10.0:
                break
            n0 = len(pv)
            am = np.clip(pv, 1e-300, 1000.0)
            u = np.log(am / am.sum())
            maxit = max(15, int(1500000.0 / max(500, n0)))
            u, ex = _polish(u, n0, (32, 120), d_res, maxit)

    # ---- Phase 3: exact hot mutation --------------------------------------
    b = _best_list[0]
    if b is None:
        b = base
    a = _hot_loop(np.asarray(b, dtype=np.float64), d_hot, rng)
    print("final: score=%.9f n=%d (t=%.0f)" % (_best_score[0],
                                               len(_best_list[0] or b),
                                               time.time() - t0), flush=True)
    return _best_list[0] if _best_list[0] is not None else base
