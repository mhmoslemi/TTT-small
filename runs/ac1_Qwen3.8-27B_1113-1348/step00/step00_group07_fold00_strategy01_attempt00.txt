            exponents.append(f)
        else:
            exponents.append(0.0 if value < 0.1 else 1.0)
    weights = np.asarray(exponents) ** axis_weight
    weights = np.clip(weights, 1e-2, None)
    return weights

def find_optimal_scale(sequence, bounds, multiplier_index, max_time):
    """Coordinates aggressive inward/outward scaling."""
    sequence_np = np.array(sequence)
    target_sum = np.sqrt(2 * len(sequence_np))

    # Prepare bounds and explore all coordinates
    low_bounds = np.log(np.float(1e-4)) + np.inf
    high_bounds = (sequence_np + 0.1) * bounds[multiplier_index] 

    # Setup optimization metric
    strategy = 'trust-constr' if len(sequence_np) > 512 else 'Nelder-Mead'

    # Time constraints
    time_step = max_time / (len(sequence_np) + 10)
    start_time = time.time()
    step = 1e-2

    # Initial guess with aggressive amplitude adjustment
    if np.sum(sequence_np) > 0.1:

    else:
        scaled_guess = sequence_np.copy()
        for i, (low, high) in enumerate(zip(low_bounds, high_bounds)):
            scaled_guess[i] = (high - low) / 2 + np.exp(np.random.uniform(low, high, 1))

    # Optimization function being an enemy of exponential terms
    def laguerre_objective(x):
        seq = np.exp(x) if strategy == 'trust-constr' else x
        laguerre = convolve_without_bounds(seq)
        objective_base = max(laguerre)
        return objective_base if objective_base > 0.0 else np.inf

    # Wrapper with time limits
    time_budget = 5.0  # seconds
    best_score = np.inf
    best_params = {'x': scaled_guess}

    # The dimensionality of sequence is n-dimensional, we only care about each individual component
    while (time.time() - start_time) < time_budget and best_score > 1.50:
        # Adaptive step size
        new_step = max(step * 0.9, 0.001)

        # Use bounds to inform the optimization
        # Note: The bounds are not currently directly passed to the optimizer.
        result = minimize(laguerre_objective, scaled_guess, method=strategy, options={'x0': 0.5, 'step': new_step})

                best_params['x'] = candidate
                print("New optimal scaling found with score", best_score)
