def evaluate_sequence(sequence):
    """Evaluates the sequence against the defined metric."""
    if not isinstance(sequence, list) or not all(isinstance(x, float) for x in sequence):
        return np.inf
    
    # Safety filtering
    error_flag = False
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (float, int)) or np.isnan(x) or np.isinf(x) or not 0<=x<=1000:
            error_flag = True
            break
    
    if error_flag:
        return np.inf
    
    conv = np.convolve(sequence, sequence, mode="full")
    total_sum = np.sum(sequence)
    
    if total_sum < 0.01:
        return np.inf
    
    if len(sequence) < 1 or total_sum == 0:
        return np.inf
    
    max_conv_val = np.max(conv)
    wrapper = float(2 * len(sequence) * max_conv_val) / (total_sum ** 2)
    if np.isinf(wrapper) or wrapper <= 0 or not np.isfinite(wrapper):
        # fallback to finite evaluation
        total_sum_clip = min(total_sum, 1.0)
        normalized_seq = [x / total_sum_clip for x in sequence]
        conv_normalized = np.convolve(normalized_seq, normalized_seq, mode="full")
        max_conv_normalized = np.max(conv_normalized)
        wrapper = (2 * len(sequence) * max_conv_normalized) / (total_sum_clip ** 2)
    
    return wrapper
