    max(conv(x, x)) >= (conv(x, x))[0] = sum_i x_i^2 >= (sum_i x_i)^2 / n

    evaluate_sequence(x) = 2 * n * max(conv(x, x)) / (sum_i x_i)^2 >= 2.
