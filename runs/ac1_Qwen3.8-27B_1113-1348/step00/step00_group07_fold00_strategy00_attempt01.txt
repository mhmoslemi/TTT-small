    n_steps_total = 10  # Steps until cooldown restart (setting number of restarts: 2400/ (10 * steps_per_restart))
    
    # Setup starting sequence if budget was met by parent.
    best_score = np.inf
    best_sequence = None
    best_len = None
    
    # First, convert and check the current sequence.
    current_sequence = np.asarray(current_sequence, dtype=np.float64)
    current_sequence = np.clip(current_sequence, 0., 1000.)[:n]  # Adjust length and clip
    
    # Check for NaN or inf.
    if ~np.isfinite(current_sequence).all():
        # If invalid set to ones and re-eval.
        current_sequence = np.ones(n, dtype=np.float64)
    
    # Evaluate current_sequence to set the bar.
    try:
        current_score = evaluate_sequence(current_sequence.tolist())
    except:
        current_sequence = np.ones(n)
        current_score = evaluate_sequence(current_sequence.tolist())
        pass
    
    # Set the initial best.
    if np.isfinite(current_score):
        best_sequence = current_sequence.copy()
        best_score = current_score
        best_len = n
        
    def create_candidate_from_perturbation(seq, rng, width_range, reflection_prob):
        """Create a new candidate from perturbing a contiguous segment."""
        n_here = len(seq)
        # Choose width from the range provided.
        width = rng.integers(max(1, n_here//200), max(2, n_here//12))
        start_pos = rng.integers(1, n_here - width + 1)
        end_pos = start_pos + width
        
        # Perturb the block by a random factor.
        factor = np.exp(rng.normal(0.0, 0.12))
        # Ensure the perturb doesn't introduce negative numbers by resetting to positive.
        new_block = factor * np.abs(seq[start_pos:end_pos])
        new_seq = seq.copy()
        new_seq[start_pos:end_pos] = new_block
        
        # Apply reflection with some probability.
        if rng.random() < reflection_prob:
            new_seq = 0.5 * (new_seq + new_seq[::-1])
            # Convert any off-bound or negative values back to bounds.
            new_seq = np.clip(new_seq, 0., 1000.)
        else:
            new_seq = np.clip(new_block, 0., 1000.)  # Actually, we don't need this because factor is positive and the block had absolute value.
        return new_seq
    
    # Create a very specific random sequence for diversification, not necessarily retaining the length.
    def create_random_sequence_at_length(length, rng):
        ret = np.ones(length) * 1.0
        # Randomly perturb coefficients for diversity.
        indices = rng.choice(len(ret), size=len(ret)//3, replace=False)
        factors = np.exp(rng.normal(0.0, 0.15, len(indices)))
        for i, idx in enumerate(indices):
            # Add enough perturbations so there's a peak but uniform.
            change = factors[i] * 10.0 - 1.0  # More than double or less than triple based on factor range.
            if i % 2 == 0:
                ret[idx] = 1000.0 * change  # This biases toward peaks but careful.
            else:
                ret[idx] = 100.0 * change    # But actually, the goal is to achieve a better ratio.
        return np.clip(ret, 0., 1000.) / 10.0  # Normalize to a non-new number.
    
    # Set up length options as powers of two up to 256.
    lengths = [8, 16, 32, 64, 128, 256]
    reflection_p = 0.35  # Keep the reflection probability as in the previous program.
    
    # Track our candidates and the necessary replacements.
    best_score = current_score
    best_curr_seq = current_sequence.copy()
    
    # Design an SA-like search loop.
    t0 = budget_s
    start_time = time.time()
    counter = 0
    
    while time.time() < start_time + max(budget_s, 0.0):
        # Determine if it's time to try a new length.
        # Every n_steps_total//2 steps we retry a new random sequence at other lengths.
        if counter % (n_steps_total // 2) == 0:
            # Choose a new random length.
            new_len = rng.choice(lengths, replace=True)
            # If we already tried this length keep trying.
            if new_len != best_len:
                best_len = new_len
                # Create a new sequence at the new length.
                new_seq = create_random_sequence_at_length(new_len, rng)
                try:
                    new_score = evaluate_sequence(new_seq.tolist())
                except:
                    continue
                if np.isfinite(new_score):
                    print(f"Trying random sequence at length {new_len}, score {new_score}")
                    # Update if better than current.
                    if new_score < current_score:
                        current_score = new_score
                        current_sequence = new_seq
                    
                    # Update best if better.
                    if new_score < best_score:
                        best_score = new_score
                        best_curr_seq = new_seq.copy()
            else:
                # Randomly pick another length.
                new_len = [l for l in lengths if l != best_len][0]
                new_seq = create_random_sequence_at_length(new_len, rng)
                # Evaluate if needed, but skip for efficiency if already computed.
                if np.isfinite(best_score):
                    new_score = evaluate_sequence(new_seq.tolist())
                else:
                    try:
                        new_score = evaluate_sequence(new_seq.tolist())
                    except:
                        continue
                    pass
        
        # Now, within one temperature setting.
        # In the previous program, they set T = T_initial and then decayed by a factor every k steps.
        # We'll follow a similar method but with the following constraints.
        # Temperature control.
        T = T_initial
        T_decay_needed = False
        
        # We'll do a standard SA cycle until time is out.
        while T > T_min and time.time() < start_time + max(budget_s, 0):
            T = 0.9995 * T
            
            # If T decays enough, check if we need to restart.
            T_decay_needed = T_decay_needed or (counter % n_steps_total == 0 and sum(np.log(T)) < -10)
            
            counter += 1
            
            # Create candidate via block perturbation (local candidate).
            try:
                new_seq_here = create_candidate_from_perturbation(best_curr_seq, rng, ... reflection_p)
                new_score_here = evaluate_sequence(new_seq_here.tolist())
            except:
                try:
                    new_seq_here_crit = create_candidate_from_perturbation(best_curr_seq, rng, width_range=(1,100), reflection_p=reflection_p)
                    new_score_here = evaluate_sequence(new_seq_here_crit.tolist())
                except:
                    continue
            
            # If evaluation fails, skip.
            if ~np.isfinite(new_score_here):
                # Skip this candidate.
                pass
            else:
                # When T is large (beginning of SA), we only require improvement; as T drops, accept worse solutions.
                # Check if better than best_curr_seq or if we are in a temperature context for candidate.
                if new_score_here < best_curr_seq_score:
                    best_curr_seq_score = new_score_here
                    best_curr_seq = new_seq_here.copy()
                    # Print improvement if found.
                    print(f"Improvement: {best_curr_seq_score}")
                
                # If reached best_score, continue with cooling without reflection until T_stuck.
                # Standard SA transition: Exponential function.
                if new_score_here < best_curr_seq_score:  # This candidate is better
                    # No need for reflection here, accept it directly.
                    best_curr_seq = new_seq_here.copy()
                    best_curr_seq_score = new_score_here
                else:
                    # Decide to accept on the basis of probability.
                    delta_score = best_curr_seq_score - new_score_here
                    accept_prob = np.exp((delta_score / T))
                    
                    # Random selection to decide acceptance.
                    if rng.random() < accept_prob:
                        best_curr_seq = new_seq_here.copy()
                        best_curr_seq_score = new_score_here
                        # Allow updating if temporarily improving.
                    else:
                        # Reject candidate.
                        pass
                # Every few acceptance steps, clear cache.
        
        # After full SA cycle, we have the best sequence so far.
        best_sequence = best_curr_seq.tolist()
        return best_sequence

- Focus on power-of-two lengths (8, 16, 32,...) to ensure diversity without excessive convolutions.
- SA temperature cools exponentially, restarting the cooling cycle every `n_steps_total`.
- SA step time validation—each temperature step is handled punctually without missing deadlines.
- Number of steps depends on the cooling parameters and the allowed time— we expect around 10^7 steps or fewer.
- Reflection probability and sequence length are adjusted as needed.

**Budget Estimate and Validation:**
- The SA step computes in constant time (O(n) convolution) due to minimal computations.
- The number of steps can be estimated by cooling decay— say 10 steps per cooling cycle, with a total time allowing at least 10 cycles.
- We'll evaluate the best sequence found and return it once the time is nearly-up.

**Fallback and Validation Steps:**
- If the parent is of poor quality, the function will generate random sequences from the prescribed lengths.
- After time is up, if the quality does not improve or stabilizes, the final sequence is evaluated independently.

**Return Contract:**
- The function returns a non-empty list of floats, representing a valid sequence with the minimal objective found.
- The sequence must be non-negative and within the 1000 cap, and must independently be validated by the `evaluate_sequence` function.
