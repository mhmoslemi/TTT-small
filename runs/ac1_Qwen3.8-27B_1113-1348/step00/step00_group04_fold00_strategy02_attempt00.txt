def make_block_perturbation(sequence, rng):
    """Generate a next candidate sequence via stochastic block perturbation."""
    n = len(sequence)
    width_param = rng.integers(max(1, int(n/200)), max(100, int(n/10)))
    start = rng.integers(0, n - width_param)
    end = start + width_param
    
    # Convert to array for mutation operations
    seq_arr = np.asarray(sequence, dtype=np.float64).copy()
    seq_arr[start:end] = np.clip(seq_arr[start:end] * (1 + rng.normal(0.0, 0.05)), 0.0, 1000.0)
    
    # Random chance for reflection mutation
    if rng.random() < 0.35:
        # Reflect and average with a 0.5 safety factor
        seq_arr = np.clip(0.5 * (seq_arr + np.flip(seq_arr)), 0.0, 1000.0)
        
    return seq_arr
