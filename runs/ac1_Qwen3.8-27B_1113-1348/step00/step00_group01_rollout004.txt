```python
"""
Search for non-negative sequences minimizing
    2*n*max(convolve(a,a))/(sum(a))**2
scale-invariantly, by keeping sum(a)=1 and minimizing the peak of a*a.

The main optimizer is a natural-gradient descent on the smooth maximum
(log-sum-exp) of the convolution coefficients, interleaved with:
  - several smooth/multipartite starting profiles,
  - a coarse-to-fine warm start,
  - LP-inspired inverse-coverage mixture moves,
  - simulated-annealing-like restarts on stagnation.
"""

import time
import math
import numpy as np


def _fft_len(m: int) -> int:
    """Next power-of-two FFT length >= m."""
    m = max(2, int(m))
    L = 1
    while L < m:
        L <<= 1
    return L


def _normalize(a: np.ndarray) -> np.ndarray:
    """Make a non-negative finite vector of sum one."""
    x = np.asarray(a, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if not math.isfinite(s) or s <= 0.0:
        x = np.ones_like(x, dtype=np.float64)
        s = float(x.size)
    x = x / s
    x = np.maximum(x, 1e-300)
    x = x / float(x.sum())
    return x


def _conv_square(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a for seed scoring."""
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    L = _fft_len(2 * n - 1)
    fa = np.fft.rfft(a, L)
    p = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    p = np.where(np.isfinite(p), p, 0.0)
    return np.maximum(p, 0.0)


def _score_from_a(a: np.ndarray) -> float:
    """Fast proxy for evaluate_sequence, assuming sum(a)=1 after normalization."""
    a = _normalize(a)
    n = int(a.size)
    if n == 0:
        return float("inf")
    p = _conv_square(a)
    if p.size == 0:
        return float("inf")
    pm = float(np.max(p))
    if not math.isfinite(pm):
        return float("inf")
    return float(2.0 * n * pm)


def _smooth_max(p: np.ndarray, beta: float) -> float:
    """Smooth max (beta inverse-temperature)."""
    if p.size == 0:
        return 0.0
    if beta <= 1.0:
        return float(np.max(p))
    m = float(np.max(p))
    z = beta * (p - m)
    z = np.clip(z, -700.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        return m
    return m + math.log(s) / beta


def _weights_from_p(p: np.ndarray, beta: float) -> np.ndarray:
    """Softmax weights over convolution coefficients."""
    if p.size == 0:
        return np.array([], dtype=np.float64)
    if beta <= 1.0:
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    m = float(np.max(p))
    z = beta * (p - m)
    z = np.clip(z, -700.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        w = np.zeros(p.size, dtype=np.float64)
        w[int(np.argmax(p))] = 1.0
        return w
    w = ez / s
    w = np.where(np.isfinite(w), w, 0.0)
    s2 = float(w.sum())
    if s2 > 0.0:
        w /= s2
    return w


def _cross_with_fa(a: np.ndarray, w: np.ndarray, fa: np.ndarray, L: int) -> np.ndarray:
    """
    d_j = sum_k w_k a_{k-j}  for j=0..n-1,
    where a has length n and w has length 2n-1.
    """
    n = int(a.size)
    m = 2 * n - 1
    d = np.zeros(n, dtype=np.float64)
    if n == 0 or w.size == 0:
        return d

    maxw = float(np.max(w)) if w.size else 0.0
    if maxw <= 0.0 or not math.isfinite(maxw):
        return d

    thresh = max(1e-14, 1e-5 * maxw)
    idx = np.nonzero(w > thresh)[0]

    # Use a sparse path only when a few diagonals carry almost all mass.
    if idx.size <= 32 and float(w[idx].sum()) > 0.90:
        for k in idx.tolist():
            weight = float(w[k])
            if weight <= 0.0:
                continue
            start = max(0, k - n + 1)
            stop = min(n - 1, k)
            if start > stop:
                continue
            seg = a[k - stop: k - start + 1][::-1]
            d[start:stop + 1] += weight * seg
        return d

    rev_w = w[::-1].copy()
    fwr = np.fft.rfft(rev_w, L)
    conv = np.fft.irfft(fa * fwr, L)
    if n == 1:
        return np.array([float(conv[0]) if conv.size > 0 else 0.0], dtype=np.float64)
    out = conv[n - 1: 2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def _recompute_p(a: np.ndarray, L: int, m: int):
    """Return fa and p=a*a using the existing FFT length L."""
    fa = np.fft.rfft(a, L)
    p = np.fft.irfft(fa * fa, L)[:m]
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    return fa, p


def _candidate_from_d(d: np.ndarray):
    """Inverse-coverage LP-style candidate g from d_j >= 0."""
    if d is None or d.size == 0:
        return None
    den = np.maximum(np.asarray(d, dtype=np.float64), 1e-300)
    g = 1.0 / den
    g = np.where(np.isfinite(g), g, 0.0)
    s = float(g.sum())
    if s <= 0.0 or not math.isfinite(s):
        return None

    med = float(np.median(g))
    if med > 0.0 and math.isfinite(med):
        g = np.minimum(g, 100.0 * med)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 0.0 or not math.isfinite(s):
        return None
    return g / s


def _try_mix(a: np.ndarray, g: np.ndarray, fa: np.ndarray,
             p_f: np.ndarray, L: int, m: int, maxp: float):
    """
    Minimize max((1-t)a + t g)^2 convolution over t by precomputing a*a, g*g, a*g.
    Returns (t, None) if no useful improvement.
    """
    if g is None or g.size != a.size:
        return 0.0, None

    fg = np.fft.rfft(g, L)
    p_fg = np.fft.irfft(fa * fg, L)[:m]
    p_g = np.fft.irfft(fg * fg, L)[:m]
    p_fg = np.maximum(np.where(np.isfinite(p_fg), p_fg, 0.0), 0.0)
    p_g = np.maximum(np.where(np.isfinite(p_g), p_g, 0.0), 0.0)

    best_t = 0.0
    best_val = float(p_f.max()) if p_f.size else float("inf")

    trial_ts = [0.01, 0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.70]
    for t in trial_ts:
        one = 1.0 - t
        pm = one * one * p_f + 2.0 * one * t * p_fg + t * t * p_g
        v = float(np.max(pm))
        if v < best_val - 1e-15:
            best_val = v
            best_t = t

    if best_t > 0.0:
        lo = max(0.0, best_t - 0.10)
        hi = min(1.0, best_t + 0.10)
        for r in range(25):
            t = lo + (hi - lo) * (r / 24.0)
            one = 1.0 - t
            pm = one * one * p_f + 2.0 * one * t * p_fg + t * t * p_g
            v = float(np.max(pm))
            if v < best_val - 1e-15:
                best_val = v
                best_t = t

    if best_t > 0.0 and best_val < maxp - 1e-12:
        return float(best_t), None
    return 0.0, None


def _seed_score_key(item):
    return float(item[0])


def _make_seed_arrays(n: int, rng: np.random.Generator) -> list:
    """A diverse pool of starting sequences."""
    arrs = []
    if n <= 0:
        return arrs

    # Uniform.
    arrs.append(np.ones(n, dtype=np.float64))

    i = np.arange(n, dtype=np.float64)
    r = n - 1.0 - i

    # Power-law profiles, especially left endpoint singularity-like shapes.
    shifts = [0.5, 0.75, 1.0, 1.5]
    pw = [0.25, 0.45, 0.55, 0.70]
    qw = [0.0, 0.30, 0.60, 1.00]

    for sh in shifts:
        base_l = i + sh
        base_r = r + sh
        for alpha in pw:
            if alpha == 0.0:
                l = np.ones(n, dtype=np.float64)
            else:
                l = np.power(base_l, -alpha)
            for beta in qw:
                if beta == 0.0:
                    rr = np.ones(n, dtype=np.float64)
                else:
                    rr = np.power(base_r, -beta)
                arrs.append(l * rr)

    # Smooth sinusoidal profiles.
    x = (i + 0.5) / float(n)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        arrs.append(np.sin(math.pi * x) ** p)

    for f in [1, 2, 3]:
        for ph in [0.0, 0.7, 1.4]:
            arrs.append(1.0 + 0.8 * np.sin(2.0 * math.pi * f * x + ph))

    # Geometric decays both ways.
    for val in [0.75, 0.90, 0.98]:
        arrs.append(val ** i)
        arrs.append(val ** r)

    # Random low-pass smooth profiles.
    for amp in [0.25, 0.60, 1.10]:
        for _ in range(2):
            noise = rng.standard_normal(n)
            uf = np.fft.rfft(noise, n)
            cut = max(2, n // 40)
            if cut < uf.size:
                uf[cut:] = 0.0
            smooth = np.fft.irfft(uf, n)
            smooth = smooth - float(np.mean(smooth))
            arrs.append(np.exp(amp * smooth))

    # A few blunt random blocks (for rare escapes).
    for _ in range(3):
        arr = np.ones(n, dtype=np.float64)
        B = int(max(4, n // 100))
        for _b in range(B):
            start = int(rng.integers(0, n))
            length = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            bloc = slice(start, min(n, start + length))
            arr[bloc] *= float(np.exp(rng.normal(0.0, 0.35)))
        arrs.append(arr)

    return arrs


def _upsample(coarse: np.ndarray, n_full: int) -> np.ndarray:
    """Lift a coarse profile to full length by log-linear interpolation."""
    if coarse is None or coarse.size == 0:
        return np.ones(n_full, dtype=np.float64)
    if coarse.size == n_full:
        return _normalize(coarse)
    c = np.maximum(np.asarray(coarse, dtype=np.float64), 1e-300)
    xc = np.linspace(0.0, 1.0, c.size, endpoint=False)
    xf = np.linspace(0.0, 1.0, n_full, endpoint=False)
    logc = np.log(c)
    interp = np.exp(np.interp(xf, xc, logc))
    return _normalize(interp)


def _coarse_seed(n_full: int, deadline: float, rng: np.random.Generator):
    """Optional coarse warm start."""
    now = time.monotonic()
    if n_full <= 0 or deadline - now < 25.0:
        return None

    n_c = int(min(512, max(128, n_full // 10)))
    if n_c >= n_full:
        return None

    i = np.arange(n_c, dtype=np.float64)
    rr = n_c - 1.0 - i
    arrs = [np.ones(n_c, dtype=np.float64)]

    for sh in [0.5, 1.0]:
        for alpha in [0.30, 0.45, 0.55, 0.70]:
            for beta in [0.0, 0.40, 0.80]:
                l = np.power(i + sh, -alpha)
                b = np.power(rr + sh, -beta)
                arrs.append(l * b)

    scored = []
    for arr in arrs:
        try:
            arrn = _normalize(arr)
            sc = _score_from_a(arrn)
            if math.isfinite(sc):
                scored.append((sc, arrn))
        except Exception:
            pass

    if not scored:
        return None

    scored.sort(key=_seed_score_key)
    best_arr = scored[0][1]
    best_score = scored[0][0]

    top = min(2, len(scored))
    for rank in range(top):
        rem = deadline - time.monotonic()
        if rem < 15.0:
            break
        end = time.monotonic() + rem / float(top - rank)
        out = _optimize_sequence(scored[rank][1], end, rng)
        sc = _score_from_a(out)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = out

    return _upsample(best_arr, n_full)


def _optimize_sequence(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Natural-gradient / annealed local search around one starting sequence."""
    a = _normalize(a0)
    n = int(a.size)
    if n <= 0:
        return a

    m = 2 * n - 1
    L = _fft_len(3 * n - 2 if n > 1 else 2)
    fa, p = _recompute_p(a, L, m)
    maxp = float(np.max(p))
    score = float(2.0 * n * maxp)
    if not math.isfinite(score):
        a = np.ones(n, dtype=np.float64) / float(n)
        fa, p = _recompute_p(a, L, m)
        maxp = float(np.max(p))
        score = float(2.0 * n * maxp)

    best_a = a.copy()
    best_score = score

    start = time.monotonic()
    total = max(1.0, max(0.0, deadline - start))
    step = 0.15
    min_step = 0.003
    max_step = 1.00
    it = 0
    stall = 0
    last_print = start
    last_good_time = start

    while time.monotonic() < deadline - 1.0:
        try:
            it += 1
            progress = min(1.0, (time.monotonic() - start) / total)
            target_B = 0.05 + (45.0 - 0.05) * (progress ** 0.65)
            B = target_B * (0.75 + 0.50 * float(rng.random()))
            beta = max(1.0, B * float(n))

            J = _smooth_max(p, beta)
            w = _weights_from_p(p, beta)
            d = _cross_with_fa(a, w, fa, L)
            if not np.all(np.isfinite(d)):
                d = np.zeros(n, dtype=np.float64)

            grad_a = 2.0 * d
            mu = float(np.dot(a, grad_a))
            gc = grad_a - mu
            norm2 = float(np.dot(a, gc * gc))

            if not math.isfinite(norm2) or norm2 <= 1e-30:
                stall += 2
            else:
                direction = -gc / math.sqrt(norm2)
                direction = np.where(np.isfinite(direction), direction, 0.0)
                direction = np.clip(direction, -4.0, 4.0)

                # Small randomized perturbation of the natural-gradient direction.
                if rng.random() < 0.04:
                    nv = rng.standard_normal(n)
                    nv = nv - float(np.dot(a, nv))
                    nv_norm = math.sqrt(max(1e-30, float(np.dot(a, nv * nv))))
                    if nv_norm > 1e-16:
                        direction = direction + 0.07 * (nv / nv_norm)
                        direction = np.clip(direction, -4.0, 4.0)

                accepted = False
                trial_steps = [step]
                if step > 0.04:
                    trial_steps.append(step * 0.55)
                    trial_steps.append(step * 0.25)
                else:
                    trial_steps.append(step * 1.6)
                    trial_steps.append(step * 2.5)

                for st in trial_steps:
                    if st < min_step or st > max_step:
                        continue
                    cand = a * np.exp(st * direction)
                    cand = np.where(np.isfinite(cand), cand, 0.0)
                    cand = np.maximum(cand, 1e-300)
                    cs = float(cand.sum())
                    if cs <= 0.0 or not math.isfinite(cs):
                        continue
                    cand /= cs

                    fac, pc = _recompute_p(cand, L, m)
                    cmax = float(np.max(pc))
                    cJ = _smooth_max(pc, beta)

                    hard_better = cmax < maxp - 1e-14
                    soft_better = (cJ < J - 1e-12) and (cmax <= maxp * 1.002 + 1e-18)
                    if hard_better or (soft_better and (stall > 0 or rng.random() < 0.35)):
                        a = cand
                        fa = fac
                        p = pc
                        maxp = cmax
                        score = float(2.0 * n * cmax)
                        J = cJ
                        stall = 0
                        accepted = True

                        if score < best_score - 1e-12:
                            best_score = score
                            best_a = a.copy()
                            last_good_time = time.monotonic()

                        step = min(max_step, st * 1.04)
                        break

                if not accepted:
                    step = max(min_step, step * 0.72)
                    stall += 1

            # LP-inspired inverse-coverage mixture move.
            if (stall >= 18 or (it % 180 == 0)) and time.monotonic() < deadline - 2.0:
                g = _candidate_from_d(d)
                if g is not None:
                    t, _ = _try_mix(a, g, fa, p, L, m, maxp)
                    if t > 0.0:
                        mixed = (1.0 - t) * a + t * g
                        mixed = np.where(np.isfinite(mixed), mixed, 0.0)
                        mixed = np.maximum(mixed, 1e-300)
                        ms = float(mixed.sum())
                        if ms > 0.0 and math.isfinite(ms):
                            mixed /= ms
                            mfa, mp = _recompute_p(mixed, L, m)
                            mmax = float(np.max(mp))
                            mscore = float(2.0 * n * mmax)
                            if math.isfinite(mscore) and mmax < maxp - 1e-12:
                                a = mixed
                                fa = mfa
                                p = mp
                                maxp = mmax
                                score = mscore
                                stall = 0
                                step = max(0.05, min(step, 0.15))
                                if score < best_score - 1e-12:
                                    best_score = score
                                    best_a = a.copy()
                                    last_good_time = time.monotonic()

            # Stagnation recovery.
            if stall >= 70 or (time.monotonic() - last_good_time > 300.0 and stall >= 20):
                # Either mutate best solution or return to it.
                if rng.random() < 0.65:
                    base = best_a.copy()
                else:
                    base = a.copy()

                sig = 0.04 + 0.18 * float(rng.random())
                pert = base * np.exp(sig * rng.standard_normal(n))
                pert = _normalize(pert)
                pfa, pp = _recompute_p(pert, L, m)
                pmax = float(np.max(pp))
                psc = float(2.0 * n * pmax)

                accept_pert = (
                    math.isfinite(psc) and
                    (pmax < maxp * 1.03 + 1e-12) and
                    (psc < best_score * 1.05 + 1e-12)
                )
                if accept_pert or rng.random() < 0.05:
                    a = pert
                    fa = pfa
                    p = pp
                    maxp = pmax
                    score = psc
                else:
                    a = best_a.copy()
                    fa, p = _recompute_p(a, L, m)
                    maxp = float(np.max(p))
                    score = float(2.0 * n * maxp)

                step = 0.12
                stall = 0

            if time.monotonic() - last_print >= 45.0:
                try:
                    print(
                        f"[opt] n={n} best={best_score:.7f} current={score:.7f} "
                        f"step={step:.4f} beta={beta:.1f} it={it} stall={stall}"
                    )
                except Exception:
                    pass
                last_print = time.monotonic()

        except Exception:
            break

    if best_a.size != n or not np.all(np.isfinite(best_a)):
        best_a = np.ones(n, dtype=np.float64) / float(n)
    return _normalize(best_a)


def _choose_n(parent) -> int:
    if parent is not None:
        try:
            sz = int(np.size(parent))
            if 512 <= sz <= 10000:
                return sz
        except Exception:
            pass
    return 6121


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main entry point. Returns the best non-negative sequence found."""
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    budget = max(120.0, budget - 8.0)

    t0 = time.monotonic()
    deadline = t0 + budget
    rng = np.random.default_rng(seed)

    # Recover the optional public construction, if present.
    parent = None
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)
        if par.ndim == 1 and par.size > 0:
            par = np.where(np.isfinite(par), par, 0.0)
            par = np.maximum(par, 0.0)
            if float(par.sum()) > 0.0:
                parent = par
    except Exception:
        parent = None

    n = _choose_n(parent)

    # Coarse warm start, when there is enough time.
    coarse = None
    if budget > 450.0:
        coarse_end = t0 + min(60.0, 0.10 * budget)
        try:
            coarse = _coarse_seed(n, coarse_end, rng)
        except Exception:
            coarse = None

    # Full-length seed pool.
    arrays = []
    if parent is not None:
        if parent.size == n:
            arrays.append(parent)
        else:
            arrays.append(_upsample(parent, n))
    if coarse is not None:
        arrays.append(coarse)
    arrays.extend(_make_seed_arrays(n, rng))

    # Score and keep only the best seeds.
    scored = []
    seen_scores = []
    for arr in arrays:
        try:
            arrn = _normalize(arr)
            sc = _score_from_a(arrn)
            if math.isfinite(sc):
                scored.append((sc, arrn))
        except Exception:
            pass

    if not scored:
        scored.append((2.0, np.ones(n, dtype=np.float64) / float(n)))

    scored.sort(key=_seed_score_key)

    best_score = scored[0][0]
    best_arr = scored[0][1]

    # Run several top seeds for controlled time slices.
    K = min(5, len(scored))
    if budget > 700.0:
        K = min(6, len(scored))

    weights = []
    for rank in range(K):
        weights.append(1.0 / float((rank + 1) ** 0.85))
    sum_w = float(sum(weights))

    final_reserve = max(30.0, 0.22 * budget)
    exploration_deadline = deadline - final_reserve

    rank = 0
    while rank < K:
        now = time.monotonic()
        if now >= exploration_deadline - 2.0:
            break
        remaining = exploration_deadline - now
        rem_w = float(sum(weights[rank:]))
        if rem_w <= 0.0:
            break
        slice_time = remaining * weights[rank] / rem_w
        if slice_time < 8.0 and rank + 1 < K:
            rank += 1
            continue
        end = min(exploration_deadline, now + slice_time)
        try:
            out = _optimize_sequence(scored[rank][1], end, rng)
            sc = _score_from_a(out)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = out
                try:
                    print(f"[main] new best after seed rank {rank}: {best_score:.7f}")
                except Exception:
                    pass
        except Exception:
            pass
        rank += 1

    # Final polish of the current champion.
    if time.monotonic() < deadline - 10.0:
        try:
            out = _optimize_sequence(best_arr, deadline - 8.0, rng)
            sc = _score_from_a(out)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = out
                try:
                    print(f"[main] final polish improved to {best_score:.7f}")
                except Exception:
                    pass
        except Exception:
            pass

    # Clean final output: finite, non-negative, keep tiny noise under control.
    out = np.asarray(best_arr, dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    mx = float(np.max(out)) if out.size else 0.0
    if mx > 0.0 and math.isfinite(mx):
        out[out < 1e-18 * mx] = 0.0
    out = np.asarray(out, dtype=np.float64)
    s = float(out.sum())
    if not math.isfinite(s) or s < 1e-6:
        out = np.ones(n, dtype=np.float64)
        s = float(out.size)
    out /= s

    # Optional authoritative validation. If invalid, fall back to uniform.
    try:
        ev = float(evaluate_sequence([float(x) for x in out]))
        if not math.isfinite(ev):
            out = np.ones(n, dtype=np.float64) / float(n)
            try:
                ev2 = float(evaluate_sequence([float(x) for x in out]))
                if math.isfinite(ev2):
                    best_score = ev2
            except Exception:
                pass
    except Exception:
        pass

    try:
        print(f"[main] returning n={n} best_internal={best_score:.7f}")
    except Exception:
        pass

    return [float(x) for x in out]
```
