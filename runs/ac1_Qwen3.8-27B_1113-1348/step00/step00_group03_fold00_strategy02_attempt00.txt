            current_s = time.monotonic()
            if current_s > timeout - check_period:
                if best_score >= 3.0:
                    # Strategy stabilization
                    break
                # Recompute evaluation budget early to save time
                checkpoint = current_s + check_period
                evaluation_budget = int((timeout - checkpoint) * 1.5)
                config['cooldown_length'] = int(evaluation_budget * 0.3)
            
            # Limit evaluation count
            t_next = timeout - evaluation_budget
            if time.monotonic() > t_next and best_score < target_lower_bound:
                # Schedule termination
                time.sleep(t_next - time.monotonic())
                evaluation_budget = 0
                
            # Evaluate candidate and update state if better
            cand_score = evaluate_sequence(new_candidate.tolist())
            if cand_score < best_score and not np.isinf(cand_score):
                best_score = cand_score
                best_sequence = new_candidate.tolist()
                best_candidate = np.asarray(best_sequence, dtype=np.float64)
            elif cand_score is np.inf:
                # Bad candidate, continue search
                continue
                
        # Step 3: When bound at 0, restart with fallback
        if best_score >= config['cooldown_length']:
            print("Early termination: hitting bound")
            best_score = fallback_score  # Reset to make progress
            best_sequence = fallback.tolist()
            best_candidate = fallback.copy()
            
    return list(best_sequence)
