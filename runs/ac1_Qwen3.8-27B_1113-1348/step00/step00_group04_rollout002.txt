def _fft_conv(a):
    n = int(a.size)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    target = 2*n - 1
    L = 1
    while L < target:
        L <<= 1
    A = np.fft.rfft(a, L)
    out = np.fft.irfft(A*A, L)
    return out[:target]

"""
Minimize  score(a) = 2*n*max_k(c_k) / (sum(a))^2  where c = conv(a,a) [full].

Because the objective is scale-invariant, we optimise with sum(a)=1 (so the
score becomes 2n*max_k(c_k)) and the true minimisation of max(c) over the
simplex {a >= 0, sum a = 1}.  The main structure of a low-score sequence is
a "flatten the self-convolution" problem with a non-symmetric profile (the
all-ones symmetric sequence is only a local baseline with score 2).

Search (heuristic, memory-bounded):
  * The internal oracle uses an FFT-based convolution (O(n log n)) so thousands
    of inner updates are possible inside the 2400 s budget; evaluate_sequence()
    is called only sporadically and only as the authoritative scorer for the
    hall-of-fame finalists.
  * Each "trial" runs a projected subgradient step:
        - find the active lag k* = argmax c
        - multiply the whole contributing slice i in [k*-n+1..k*] by (1-alpha)
          (this is safe, always keeps a>=0, and scales the active lag by
           (1-alpha)^2),
        - add a light multiplicative log-normal shock for escape and rescale
          back to sum 1,
        - track the best-seen point, with occasional "shock around best" when
          stuck.
  * We run many diversified trials at different sequence lengths (from 2048 up
    to 32768) and random initial shapes (ramps, exponentials, smooth/comb
    modulated, random, etc.), keep a small hall-of-fame of the best internal
    offenders, then exploit the top entries and polish the single best, and
    finally confirm the top 10 hall entries with the authoritative
    evaluate_sequence() and return the best exact one.
"""

def _as_valid_array(x, low=0.0, high=1000.0):
    """Return a 1-D float64 array clipped to [low, high] or None if unusable."""
    if x is None:
        return None
    try:
        if isinstance(x, bool):
            return None
        arr = np.asarray(x, dtype=np.float64)
    except Exception:
        return None
    if arr.ndim != 1 or arr.size == 0:
        return None
    if not np.all(np.isfinite(arr)):
        return None
    return np.clip(arr, float(low), float(high))


def _normalize_sum1(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = float(a.sum()) if a.size else 0.0
    if not np.isfinite(S) or S <= 0:
        return a.copy()
    return a / S


def _fft_conv(a):
    """Full linear convolution of 1-D real a with itself via FFT, length 2n-1."""
    n = int(a.size)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    target = 2 * n - 1
    L = 1
    while L < target:
        L <<= 1
    A = np.fft.rfft(a, L)
    out = np.fft.irfft(A * A, L)
    return out[:target]


def _score_internal(a):
    """Fast proxy for the authoritative evaluate_sequence using FFT."""
    arr = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = float(arr.sum())
    if not np.isfinite(S) or S < 0.01:
        return np.inf
    n = arr.size
    if n <= 0:
        return np.inf
    c = _fft_conv(arr)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return 2.0 * float(n) * m / (S * S)


def _make_start(n, kind, rng):
    t = np.linspace(0.0, 1.0, int(n))
    k = kind if kind in (
        'ones', 'ramp', 'ramprev', 'expdec', 'expinc',
        'rand', 'smooth1', 'smooth2', 'comb', 'bi', 'hann'
    ) else 'smooth1'
    if k == 'ones':
        a = np.ones(n)
    elif k == 'ramp':
        a = 0.05 + t
    elif k == 'ramprev':

    elif k == 'expdec':
        a = np.exp(-8.0 * t) + 1e-4
    elif k == 'expinc':
        a = np.exp(8.0 * (t - 1.0)) + 1e-4
    elif k == 'rand':
        a = 0.10 + 0.9 * rng.random(n)
    elif k == 'smooth1':

def run_trial(a0, duration, rng):
    a = _normalize_sum1(a0)
    n = a.size
    if n <= 0:
        return a0, _score_internal(a0)
    best_a = a.copy()
    best_val = _score_internal(a)
    t0 = time.monotonic()
    alpha = 0.18          # slice attenuation peek factor
    sigma = 0.10          # log-normal shock
    step = 0
    no_imp = 0
    while (time.monotonic() - t0) < duration:
        c = _fft_conv(a)
        if c.size == 0:
            break
        m = float(np.max(c))
        kstar = int(np.argmax(c))
        i0 = max(0, kstar - n + 1)
        i1 = min(n - 1, kstar)
        if i1 >= i0:
            idx = np.arange(i0, i1 + 1)
            a = a.copy()
            a[idx] *= (1.0 - alpha)
        a = a * np.exp(sigma * rng.standard_normal(n))
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if not np.isfinite(S) or S <= 0:
            a = best_a.copy()
            S = float(a.sum())
        a = a / S
        val = _score_internal(a)
        step += 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        best_exact_val, best_exact_arr = _eval_hall_top(hall, 6, best_exact_val, best_exact_arr)

    # ---------------- Final authoritative confirmation ----------------
    best_exact_val, best_exact_arr = _eval_hall_top(hall, 10, best_exact_val, best_exact_arr)
    # Also ping the in-memory best once more.
    if best_internal_arr is not None:
        try:
            sc = float(evaluate_sequence(best_internal_arr.tolist()))
            if np.isfinite(sc):
                if best_exact_arr is None or sc < best_exact_val:
                    best_exact_val = sc
                    best_exact_arr = best_internal_arr.copy()
        except Exception:
            pass
    # Fallbacks.
    if best_exact_arr is None:
        if parent is not None:
            best_exact_arr = parent
        else:
            best_exact_arr = np.ones(2048, dtype=np.float64)
    out = np.clip(np.asarray(best_exact_arr, dtype=np.float64), 0.0, 1000.0)
    S = float(out.sum())
    if not np.isfinite(S) or S < 0.01:
        out = _normalize_sum1(np.clip(parent if parent is not None else np.ones(2048, dtype=np.float64), 0.0, 1000.0))
    dt = time.monotonic() - t_start
    try:
        final_exact = evaluate_sequence(out.tolist())
    except Exception:
        final_exact = np.nan
    print(f"[final] exact={final_exact} internal_best={best_internal_val} t={dt:.1f}s len={out.size}")
    return [float(x) for x in out.tolist()]
