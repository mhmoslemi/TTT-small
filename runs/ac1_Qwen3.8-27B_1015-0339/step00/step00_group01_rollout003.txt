            if time.monotonic() > deadline:
                break
            candidate = best.copy()
            # Random block multiplicative perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 8)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.03)))
            # Occasionally apply a global rescale to explore magnitudes
            if rng.random() < 0.15:
                candidate *= float(np.exp(rng.normal(0.0, 0.02)))
            candidate = np.clip(candidate, 0.0, 1000.0)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score - 1e-15:
                best = candidate
                best_score = score
                improvement = True

    # Final safety: ensure we return a valid non-empty list
    result = [float(x) for x in best]
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0] * n
    return result
