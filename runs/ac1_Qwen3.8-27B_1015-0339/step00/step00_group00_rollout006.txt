        if np.isfinite(p_score) and p_score < best_score:
            best = parent
            best_score = p_score

    # If somehow above 2.0 (numerical edge), try a few quick fixes
    if best_score > 2.0 + 1e-9:
        deadline = time.monotonic() + min(30.0, float(budget_s) * 0.3)
        rng = np.random.default_rng(seed)
        # Smooth the parent toward constant via exponential moving average
        s = float(best.sum())
        if s > 0:
            target = np.full(n, s / n, dtype=np.float64)
            for alpha in [0.5, 0.7, 0.9, 0.99]:
                candidate = (1 - alpha) * best + alpha * target
                candidate = np.clip(candidate, 0.0, 1000.0)
                if candidate.sum() < 0.01:
                    continue
                sc = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = candidate
                    best_score = sc
                if time.monotonic() > deadline:
                    break
            if best_score > 2.0 + 1e-9:
                # Fall back to pure constant
                best = np.ones(n, dtype=np.float64)
                best_score = float(evaluate_sequence(best.tolist()))

    return [float(x) for x in best]
