import numpy as np
import time


def _make_geo(n, r, rng):
    """Geometric sequence, normalized to sum n."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    a = a / a.sum() * n
    return a


def _make_hann(n):
    """Hann window, normalized to sum n."""
    idx = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    return a / a.sum() * n


def _make_uniform(n):
    return np.ones(n, dtype=np.float64)


def _block_perturb(a, rng, n):
    """Perturb a random block of a."""
    b = a.copy()
    w = rng.integers(max(2, n // 50), max(3, n // 5))
    s = rng.integers(0, max(1, n - w))
    mult = np.exp(rng.normal(0.0, 0.05, size=w))
    b[s:s+w] *= mult
    return np.clip(b, 0.0, 1000.0)


def _coord_move(a, idx, delta):
    """Move one coordinate by delta."""
    b = a.copy()
    b[idx] += delta
    b[idx] = max(0.0, min(1000.0, b[idx]))
    return b


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    # Get and normalize parent

    n = a.size

        a = np.ones(n, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a * (n / s)
    
    # Initial candidates
    candidates = [a.copy()]
    for r in [1 - 2.0/n, 1 - 5.0/n, 1 - 10.0/n, 1 - 20.0/n]:
        candidates.append(_make_geo(n, r, rng))
        candidates.append(_make_geo(n, 1/r, rng))  # increasing
    candidates.append(_make_hann(n))
    
    best = None
    best_score = np.inf
    for cand in candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    
    # Local search

    iters = 0
    while time.monotonic() < deadline - 30:
        cand = _block_perturb(best, rng, n)
        sc = float(evaluate_sequence(cand.tolist()))
        iters += 1
        if sc < best_score:

            best = cand

    while time.monotonic() < deadline:

        sc = float(evaluate_sequence(cand.tolist()))
        iters += 1
        if sc < best_score:
            best_score = sc
            best = cand
    
    return [float(x) for x in best]
