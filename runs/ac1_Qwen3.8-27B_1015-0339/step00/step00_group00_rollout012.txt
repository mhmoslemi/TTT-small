def build_toeplitz_sparse(f, n):
    """Build the (2n-1) x n Toeplitz constraint matrix for conv(f, g) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    n_rows = 2 * n - 1
    n_cols = n
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(n_rows, n_cols))
    return A
