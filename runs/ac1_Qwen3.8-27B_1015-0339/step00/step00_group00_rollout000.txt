import time


def _clip_nonneg(arr, cap=1000.0):
    return np.clip(np.asarray(arr, dtype=np.float64), 0.0, cap)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
