import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(x, n, temp):
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    if m > 700:
        return m / temp  # overflow protection
    sm = np.log(np.sum(np.exp(scaled - m)) + 1e-30) + m
    return sm / temp * (2.0 / n)

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def repair(h, n):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 0.005) & (h < 0.995)
        cnt = np.sum(mask)
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0, 1)
            continue
        h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
    if abs(np.sum(h) - target) > 1e-10:
        # Force exact
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += target - np.sum(h)
        h = np.clip(h, 0, 1)
    return h

def slsqp(x0, n, temp, maxiter=200, ftol=1e-14):
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    res = minimize(smooth_max_obj, x0, method='SLSQP',
                   bounds=bounds, constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def fourier_basis(n, K):
    """Return basis matrix for K Fourier modes (excluding DC)."""
    i = np.arange(n)
    B = []
    for k in range(1, K+1):
        B.append(np.cos(2*np.pi*k*i/n))
        B.append(np.sin(2*np.pi*k*i/n))
    return np.column_stack(B) if B else None

def de_fourier(best_h, n, K, popsize, maxiter, t_start, deadline):
    """Run DE in Fourier space."""
    B = fourier_basis(n, K)
    if B is None:
        return best_h
    nparams = 2 * K
    h_centered = best_h - 0.5
    x0 = B.T @ h_centered  # project onto Fourier subspace
    
    def de_obj(params):
        h = B @ params + 0.5
        h = np.clip(h, 0, 1)
        h = repair(h, n)
        return compute_c5(h, n)
    
    bounds_f = [(-0.5, 0.5)] * nparams
    res = differential_evolution(de_obj, bounds_f, 
                                  popsize=popsize, maxiter=maxiter,
                                  seed=42, tol=1e-8, mutation=(0.3, 1.0),
                                  recombination=0.9, polish=True,
                                  init='sobol')
    h = B @ res.x + 0.5
    h = np.clip(h, 0, 1)
    h = repair(h, n)
    return h

def subgradient_step(h, n, step_size):
    """One subgradient step: move against the worst correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    # Gradient of c_k with respect to h
    # c_k = sum_i h[i] * (1-h[i+k]) (roughly)
    # dc_k/dh[i] = (1-h[i+k]) - h[i-(* if valid)]
    # This is approximate for linear correlation
    g = np.zeros(n)
    offset = n - 1  # lag corresponding to argmax
    k = k_star - offset  # actual lag (can be neg)
    for i in range(n):
        # contribution to corr[k_star] from h[i]
        j = i + k
        if 0 <= j < n:
            g[i] += (1.0 - h[j])  # d(h[i]*(1-h[j]))/dh[i]
        j2 = i - k
        if 0 <= j2 < n:
            g[i] -= h[j2]  # d(h[j2]*(1-h[i]))/dh[i]
    h_new = h - step_size * g
    h_new = np.clip(h_new, 0, 1)
    h_new = repair(h_new, n)
    return h_new, k_star

        if c5 < best_c5:

            best_h = x

        c5 = compute_c5(h_new, n)
        if c5 < best_c5:

            best_h = h_new
        # SLSQP after DE
        x = slsqp(best_h, n, 200, maxiter=200)
        x = repair(x, n)
        c5 = compute_c5(x, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, temp):
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = temp * corr
    m = float(s.max())
    return (np.log(np.sum(np.exp(s - m)) + 1e-30) + m) / temp * (2.0 / n)

def sum_eq(x, n):
    return float(x.sum() - n / 2.0)

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - float(h.sum())
        if abs(d) < 1e-14:
            break
        m = (h > 0.01) & (h < 0.99)
        c = int(m.sum())
        if c == 0:
            h += d / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[m] += d / c
            h = np.clip(h, 0.0, 1.0)
    d = tgt - float(h.sum())
    if abs(d) > 1e-12:
        j = int(np.argmin(np.abs(h - 0.5)))
        h[j] = float(np.clip(h[j] + d, 0.0, 1.0))
    return h

def slsqp_solve(x0, n, temp, maxiter):
    res = minimize(smooth_obj, x0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
                   args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def fourier_eval(params, n, K):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += params[2*(k-1)] * np.cos(2.0 * np.pi * k * i / n)
        h += params[2*(k-1)+1] * np.sin(2.0 * np.pi * k * i / n)
    return h

def de_fourier_obj(p, n, K):
    h = fourier_eval(p, n, K)
    h = np.clip(h, 0.0, 1.0)
    h = repair(h, n)
    return compute_c5(h, n)
