```python
import time

def compute_c5(h, dx):

    return float(np.max(corr) * dx)

    for _ in range(200):

        d = target_sum - h.sum()
        if abs(d) < 1e-14:

def slsqp_optimize(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    # Objective: minimize t
    # Constraints: corr_j * dx <= t for all j, sum(h) = target_sum, 0<=h<=1
    # Variable: x = [h_0, ..., h_{n-1}, t]
    
    def objective(x):
        return x[n]
    
    def corr_constraints(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return x[n] * np.ones(len(corr)) - corr  # Need >= 0

    x0[:n] = h0

        {'type': 'eq', 'fun': eq_constraint}

        if res.success or res.x is not None:
            h_opt = project_sr(res.x[:n], target_sum)
            return h_opt
    except Exception:
        pass
    return h0.copy()
```

```python
def corr_jacobian(h):
    n = len(h)
    v = 1.0 - h
    J = np.zeros((2*n-1, n))
    # For each lag l from -(n-1) to (n-1):
    # J[n-1+l, k] = v[k+l] (if valid) - h[k-l] (if valid)
    # 
    # Term 1: v[k+l] for k+l in [0, n-1] → k in [-l, n-1-l]
    # Term 2: h[k-l] for k-l in [0, n-1] → k in [l, n-1+l]
    
    for l in range(-(n-1), n):
        j = n - 1 + l
        # Term 1
        k_lo1 = max(0, -l)
        k_hi1 = min(n-1, n-1-l)
        if k_lo1 <= k_hi1:
            J[j, k_lo1:k_hi1+1] += v[k_lo1+l:k_hi1+l+1]
        # Term 2
        k_lo2 = max(0, l)
        k_hi2 = min(n-1, n-1+l)
        if k_lo2 <= k_hi2:
            J[j, k_lo2:k_hi2+1] -= h[k_lo2-l:k_hi2-l+1]
    return J
```

```python
        grad = np.zeros(n)

                if 0 <= k - lag <= n-1:
                    g -= h[k - lag]
                grad[k] += g / n_active

        # (Simplified: use diminishing step size)
        lr = lr0 / (1.0 + 0.01 * it)
        
        h_new = project_sr(h - lr * grad, target_sum)
        h = h_new
    
    return best_h, best_c5
```

```python
def compute_subgradient(h, active_mask, dx):
    n = len(h)
    grad = np.zeros(n)
    v = 1.0 - h
    count = active_mask.sum()
    if count == 0:
        return grad

        lag = j - (n-1)
        # Term 1: (1-h[k+lag]) for k in [-lag, n-1-lag] ∩ [0, n-1]
        k_lo1 = max(0, -lag)
        k_hi1 = min(n-1, n-1-lag)
        if k_lo1 <= k_hi1:
            grad[k_lo1:k_hi1+1] += v[k_lo1+lag:k_hi1+lag+1]
        # Term 2: -h[k-lag] for k in [lag, n-1+lag] ∩ [0, n-1]
        k_lo2 = max(0, lag)
        k_hi2 = min(n-1, n-1+lag)
        if k_lo2 <= k_hi2:
            grad[k_lo2:k_hi2+1] -= h[k_lo2-lag:k_hi2-lag+1]
    grad /= count
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_sr(h, ts):
    h = np.array(h, dtype=float)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def corr_jac(h):
    n = len(h)
    v = 1.0 - h
    J = np.zeros((2 * n - 1, n))
    for l in range(-(n - 1), n):
        j = n - 1 + l
        a, b = max(0, -l), min(n - 1, n - 1 - l)
        if a <= b:
            J[j, a:b + 1] += v[a + l:b + l + 1]
        c, d = max(0, l), min(n - 1, n - 1 + l)
        if c <= d:
            J[j, c:d + 1] -= h[c - l:d - l + 1]
    return J

def slsqp_opt(h0, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    nc = 2 * n - 1

    def _obj(x):
        return x[n]

    def _og(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def _ine(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

    def _inej(x):
        return np.hstack([-corr_jac(x[:n]), np.ones((nc, 1))])

    def _eqc(x):
        return x[:n].sum() - ts

    def _eg(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    t_init = compute_c5(h0, dx) / dx + 0.005
    x0 = np.concatenate([h0, [t_init]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': _ine, 'jac': _inej},
            {'type': 'eq', 'fun': _eqc, 'jac': _eg}]
    try:
        r = minimize(_obj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, jac=_og,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        if r.x is not None:
            return project_sr(r.x[:n], ts)
    except Exception:
        pass
    return h0.copy()

def proj_subgrad(h0, n_iter, t_end, af=0.01, rseed=0):
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    rng = np.random.RandomState(rseed)
    h = h0.copy()
    bh = h.copy()
    bc5 = compute_c5(h, dx)
    stall = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        c5 = m * dx
        if c5 < bc5 - 1e-15:
            bc5 = c5
            bh = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            h = project_sr(h + 0.04 * rng.randn(n), ts)
            stall = 0
            continue
        thr = m * (1.0 - af)
        act = np.where(corr >= thr)[0]
        cnt = max(act.size, 1)
        v = 1.0 - h
        g = np.zeros(n)
        for j in act:
            lag = j - (n - 1)
            a, b = max(0, -lag), min(n - 1, n - 1 - lag)
            if a <= b:
                g[a:b + 1] += v[a + lag:b + lag + 1]
            c, d = max(0, lag), min(n - 1, n - 1 + lag)
            if c <= d:
                g[c:d + 1] -= h[c - lag:d - lag + 1]
        g /= cnt
        lr = 0.15 / (1.0 + 0.001 * it)
        h = project_sr(h - lr * g, ts)
    return bh, bc5

def upsample_fn(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.arange(n_old) * 2.0 / n_old
    xn = np.arange(n_new) * 2.0 / n_new
    return np.interp(xn, xo, h)

    for idx, n_try in enumerate([94, 141, 188, 235, 282, 330, 377]):

        ts_t = n_try / 2.0
        h_up = project_sr(upsample_fn(best_h, n_try), ts_t)
```
