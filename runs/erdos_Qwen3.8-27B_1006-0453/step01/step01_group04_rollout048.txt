import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient(h, top_k=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    # Average gradient over top-k for smoother direction
    grad = np.zeros(n)
    for j in idx:
        lag = j - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            ii = np.arange(lo1, hi1) - lag
            grad[lo1:hi1] += (1.0 - h[ii])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            ii = np.arange(lo2, hi2) + lag
            grad[lo2:hi2] -= h[ii]
    return grad * dx / top_k

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        if s > target:
            excess = s - target
            pos = h > 0
            if pos.sum() == 0:
                break
            dec = min(1.0, excess / pos.sum())
            h[pos] = np.maximum(0.0, h[pos] - dec)
        else:
            deficit = target - s
            pos = h < 1.0
            if pos.sum() == 0:
                break
            inc = min(1.0 - h[pos].max() if pos.sum() > 0 else 1.0, deficit / pos.sum())
            h[pos] = np.minimum(1.0, h[pos] + inc)

    return sm * dx + penalty * (integral - 1.0) ** 2

        h_pert, cp = lbgf_polish(h_pert, t0, budget_s, budget_s * 0.06)
        if cp < best_c5:
            best_c5, best_h = cp, h_pert.copy()
        h_pert, cp = coord_descent(h_pert, t0, budget_s, budget_s * 0.03)
        if cp < best_c5:
            best_c5, best_h = cp, h_pert.copy()

    # Final check
    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
