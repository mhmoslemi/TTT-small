The problem is: minimize max_k ∫ h(x)(1-h(x+k)) dx subject to ∫ h = 1, 0 ≤ h ≤ 1.

If h is the indicator of a set A of measure 1 (within [0,2]), then:
∫ h(x)(1-h(x+k)) dx = measure of A ∩ {x : x+k ∉ A}

This is the measure of points in A whose k-neighbor (at distance k) is NOT in A.

For the Erdős problem, we want a set A of measure 1 in [0,2] such that for every shift k, the "boundary" of A with respect to shift k is minimized.

A "thick" set (like an interval) has large boundary for small k. A "spread out" set (like a fat Cantor set) has small boundary for all k.

The optimal solution is likely not binary but has fractional values at the "transition" regions.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def slsqp_full(h0, dx, target_sum, maxiter=400, ftol=1e-14):
    n = len(h0)
    def obj(x):
        return x[-1]
    def ineq(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1.0 - h, mode='full')
    def eq_c(x):
        return np.sum(x[:-1]) - target_sum
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = float(np.max(corr0)) / dx + 0.003
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq_c}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_h(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def slsqp_topm(h0, dx, target_sum, topm=8, maxiter=300, ftol=1e-14):
    n = len(h0)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    top_idx = np.argsort(corr0)[-topm:]
    def obj(x):
        return x[-1]
    def ineq(x):
        h = x[:-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return x[-1] - corr[top_idx]
    def eq_c(x):
        return np.sum(x[:-1]) - target_sum
    t0 = float(np.max(corr0)) / dx + 0.003
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq_c}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_h(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def pair_local_search(h, dx, target_sum, rounds=15, deltas=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    if deltas is None:
        deltas = [0.001, 0.003, 0.01, 0.03, 0.1]
    for _ in range(rounds):
        improved = False
        for i in range(n):
            hi_curr = best_h[i]
            for j in range(i + 1, n):
                hj_curr = best_h[j]
                for d in deltas:
                    for s in (1, -1):
                        ni = hi_curr + s * d
                        nj = hj_curr - s * d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        best_h[i] = ni
                        best_h[j] = nj
                        c = compute_c5(best_h, dx)
                        if c < best_c5 - 1e-14:
                            best_c5 = c
                            hi_curr = ni
                            hj_curr = nj
                            improved = True
                        else:
                            best_h[i] = hi_curr
                            best_h[j] = hj_curr
        if not improved:
            break
    return best_h, best_c5

def triple_local_search(h, dx, target_sum, rounds=5):
