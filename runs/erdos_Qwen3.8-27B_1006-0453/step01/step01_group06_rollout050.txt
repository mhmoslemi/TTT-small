```python
def subgrad_c5(h, k_star_idx):
    """Compute subgradient of C5 at h, where k_star_idx is the argmax in the full correlate output."""
    n = len(h)
    dx = 2.0 / n
    k = k_star_idx - (n - 1)  # actual lag
    sg = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            sg[j] += (1.0 - h[j - k])
        for j in range(0, n - k):
            sg[j] -= h[j + k]
    else:
        p = -k
        for j in range(0, n - p):
            sg[j] += (1.0 - h[j + p])
        for j in range(p, n):
            sg[j] -= h[j - p]
    return dx * sg
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    val = m + float(np.log(np.sum(np.exp(T * (corr - m))))) / T
    return val * dx

def project_sum(h, target):
    h = h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_c5(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    k = k_idx - (n - 1)
    sg = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            sg[j] += (1.0 - h[j - k])
        for j in range(0, n - k):
            sg[j] -= h[j + k]
    else:
        p = -k
        for j in range(0, n - p):
            sg[j] += (1.0 - h[j + p])
        for j in range(p, n):
            sg[j] -= h[j - p]
    return dx * sg

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0

    n = len(best_h)
    best_h = project_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP smooth max at increasing T (0-30%)
    bnds = [(0.0, 1.0)] * n
    for T in [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]:
        if elapsed() > budget_s * 0.30:
            break
        cons = [{'type': 'eq', 'fun': lambda x, nn=n: x.sum() - nn/2.0,
                 'jac': lambda x: np.ones(len(x))}]
        try:
            res = minimize(smooth_c5, best_h, args=(T,),
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-16})
            hc = project_sum(np.clip(res.x, 0, 1), n / 2.0)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Subgradient method with backtracking (30-55%)
    lr = 0.05
    no_improve = 0
    for step in range(5000):
        if elapsed() > budget_s * 0.55:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_star = int(np.argmax(corr))
        sg = subgrad_c5(best_h, k_star)
        # Try current lr, back off if no improvement
        for trial in range(5):
            new_h = project_sum(best_h - lr * sg, n / 2.0)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = new_h
                no_improve = 0
                break
            lr *= 0.5
        else:
            no_improve += 1
            lr *= 1.1
            if no_improve > 200:
                break
    # Reset lr for next phases
    lr = 0.03

    # Phase 3: Nelder-Mead multi-start (55-75%)
    n_starts = 50
    for i in range(n_starts):
        if elapsed() > budget_s * 0.75:
            break
        scale = 0.08 * np.exp(-0.025 * i) + 0.003
        h0 = best_h + np.random.uniform(-scale, scale, size=n)
        h0 = project_sum(h0, n / 2.0)
        try:
            def obj_nm(x, nn=n):
                return compute_c5(x) + 200.0 * abs(x.sum() - nn / 2.0)
            res = minimize(obj_nm, h0, method='Nelder-Mead',
                          options={'maxiter': 400, 'xatol': 1e-11, 'fatol': 1e-13})
            hc = project_sum(np.clip(res.x, 0, 1), n / 2.0)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 4: L-BFGS-B with high-T smooth and penalty (75-88%)
    n_cur = len(best_h)
    bnds_c = [(0.0, 1.0)] * n_cur
    for T in [200, 1000, 5000, 20000, 100000]:
        if elapsed() > budget_s * 0.88:
            break
        def obj_lbfgs(x, TT=T, nn=n_cur):
            return smooth_c5(x, TT) + 1000.0 * (x.sum() - nn / 2.0) ** 2
        try:
            res = minimize(obj_lbfgs, best_h, method='L-BFGS-B',
                          bounds=bnds_c, options={'maxiter': 2000, 'gtol': 1e-14,
                                                   'ftol': 1e-15})
            hc = project_sum(np.clip(res.x, 0, 1), n_cur / 2.0)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Subgradient refinement (88-94%)
    lr = 0.02
    no_improve2 = 0
    for step in range(3000):
        if elapsed() > budget_s * 0.94:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_star = int(np.argmax(corr))
        sg = subgrad_c5(best_h, k_star)
        improved = False
        for trial in range(6):
            new_h = project_sum(best_h - lr * sg, n_cur / 2.0)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = new_h
                improved = True
                break
            lr *= 0.5
        if not improved:
            no_improve2 += 1
            lr *= 1.05
            if no_improve2 > 300:
                break
        else:
            no_improve2 = 0

    # Phase 6: Final SLSQP polish (94-99%)
    if elapsed() < budget_s * 0.99:
        bnds_f = [(0.0, 1.0)] * n_cur
        cons_f = [{'type': 'eq', 'fun': lambda x, nn=n_cur: x.sum() - nn/2.0,
                   'jac': lambda x: np.ones(len(x))}]
        for T in [50000, 200000, 1000000]:
            if elapsed() > budget_s * 0.99:
                break
            try:
                res = minimize(smooth_c5, best_h, args=(T,),
                              method='SLSQP', bounds=bnds_f, constraints=cons_f,
                              options={'maxiter': 1000, 'ftol': 1e-17})
                hc = project_sum(np.clip(res.x, 0, 1), n_cur / 2.0)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Final feasibility and evaluation
    best_h = project_sum(best_h, len(best_h) / 2.0)
    # Ensure exact sum
    diff = len(best_h) / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-8) & (best_h < 1.0 - 1e-8))[0]
        if len(free) > 0:
            best_h[free] += diff / len(free)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
