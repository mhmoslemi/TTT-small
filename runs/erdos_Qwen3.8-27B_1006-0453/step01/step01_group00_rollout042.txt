def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sum_constraint_grad(n):
    return np.ones(n)

def project_onto_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = h.copy()
    target = n * 0.5
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        h_flat = h - (s - target) / n
        h_new = np.clip(h_flat, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-14):
            h = h_new
            break
        h = h_new
    # Fine-tune
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_objective(h, T, n):

    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on offset"""
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_try = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_final = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h_final)
    if abs(diff) > 1e-13:
        interior = (h_final > 1e-10) & (h_final < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h_final[interior] += diff / np.sum(interior)
            h_final = np.clip(h_final, 0.0, 1.0)
    return h_final

def softmax_gradient(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    W = np.exp((corr - c_max) / T)
    W /= np.sum(W)
    grad = np.zeros(n)

        if w < 1e-14:

        seg = np.arange(i_lo, i_hi + 1)
        grad[i_lo:i_hi+1] += w * (1.0 - h[seg+d] - h[seg-d])

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h_out)
    if abs(diff) > 1e-13:
        mask = (h_out > 1e-10) & (h_out < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h_out[mask] += diff / cnt
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def softmax_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    sh = corr - cm
    return (cm + T * np.log(np.sum(np.exp(sh / T)))) * dx

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    W = np.exp((corr - cm) / T)
    W /= np.sum(W)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = W[m]
        if w < 1e-13:
            continue
        d = m - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        seg = np.arange(i_lo, i_hi + 1)
        grad[i_lo:i_hi+1] += w * (1.0 - h[seg + d] - h[seg - d])
    return grad * dx

def proj_grad_step(h, T, n, lr):
    g = softmax_gradient(h, T, n)
    step = lr
    c5_cur = compute_c5(h)
    for _ in range(30):
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-16:
            return h_new, c5_new
        step *= 0.5
    return h, c5_cur

def nesterov_loop(h, T, n, max_iter, lr_init):
    h_curr = h.copy()
    v = np.zeros(n)
    c5_best_local = compute_c5(h_curr)
    lr = lr_init
    for it in range(max_iter):
        h_y = h_curr + v
        g_y = softmax_gradient(h_y, T, n)
        v = 0.9 * v - lr * g_y
        h_new_raw = h_curr + v
        h_new = project_feasible(h_new_raw, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best_local:
            c5_best_local = c5_new
        h_curr = h_new
        lr *= 0.999
    return h_curr, c5_best_local

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n)

    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Nesterov gradient with T annealing at current resolution
    T_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for T in T_list:
        if remaining() < 150:
            break
        h = h_best.copy()
        lr = 0.5 / T
        iters = min(600, int(remaining() * 8))
        for it in range(iters):
            if it % 100 == 0 and remaining() < 130:
                break
            h, c5_curr = nesterov_loop(h, T, n_best, max_iter=100, lr_init=lr)
            if c5_curr < c5_best:
                h_best = h.copy()
                c5_best = c5_curr
            # Occasionally reduce lr
            lr *= 0.7

    # Phase 2: SLSQP polishing with soft-max
    for T in [0.01, 0.003, 0.001]:
        if remaining() < 120:
            break
        t_save = T
        res = opt.minimize(
            lambda x: softmax_objective(x, t_save),
            h_best.copy(),
            jac=lambda x: softmax_gradient(x, t_save),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_best,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_best * 0.5}],
            options={'maxiter': 300, 'ftol': 1e-15}
        )
        h_c = project_feasible(res.x, n_best)
        c5_c = compute_c5(h_c)
        if c5_c < c5_best:
            h_best, c5_best = h_c, c5_c

                lr *= 0.75

    # Phase 4: Fine-tuning at best resolution with very small T
    if remaining() > 100:
        h = h_best.copy()
        n_cur = len(h)
        for T in [0.001, 0.0005, 0.0003]:
            if remaining() < 50:
                break
            lr = 0.3 / T
            for it in range(min(800, int(remaining() * 4))):

                    break

                if c5_curr < c5_best:
                    h_best, c5_best = h.copy(), c5_curr
                lr *= 0.995

    # Phase 5: Coordinate descent on worst shifts for final polish
    if remaining() > 30:
        h = h_best.copy()
        n_cur = len(h)
        dx = 2.0 / n_cur
        for round_pass in range(200):
            if remaining() < 20:
                break

            c5_val = np.max(corr) * dx
            if c5_val < c5_best:
                h_best, c5_best = h.copy(), c5_val
            # Find worst lag
            worst_m = np.argmax(corr)
            d = worst_m - (n_cur - 1)
            i_lo = max(0, -d)
            i_hi = min(n_cur - 1, n_cur - 1 - d)
            # Try adjusting each h[i] in the valid range
            for i in range(i_lo, i_hi + 1):
                i_hi2 = min(n_cur - 1, n_cur - 1 - d)
                if i > i_hi2:
                    break
                # Try a small adjustment
                for delta in [0.01, -0.01, 0.005, -0.005]:
                    h_trial = h.copy()
                    h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
                    h_trial = project_feasible(h_trial, n_cur)
                    c5_t = compute_c5(h_trial)
                    if c5_t < c5_best:
                        h = h_trial
                        h_best, c5_best = h.copy(), c5_t
                        break

    h_best = project_feasible(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
