        grad, c5, _ = subgradient(best_h, n, top_k=15, temp=8.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        best_h = best_h - step * grad
        best_h = project_feasible(best_h, n)
        c5_new = compute_c5(best_h)[0]
        if c5_new < best_c5:
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            # Perturb and restart step
            best_h = best_h + 0.02 * np.random.randn(n)
            best_h = project_feasible(best_h, n)
            step = 0.06
            no_improve = 0
        step *= step_decay
        if it % 5000 == 4999:
            step = max(0.02, step)

    # Phase 1b: SLSQP polish at current n
    for T in [20, 60, 200]:
        if elapsed() > budget_s * 0.65:
            break
        hc, c5v = slsqp_polish(best_h, n, T, maxiter=250)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Nelder-Mead local search (10% time)
    nm_t = budget_s * 0.75
    for i in range(30):
        if elapsed() > nm_t:
            break
        scale = 0.08 * np.exp(-0.06 * i) + 0.005
        h0 = best_h + np.random.uniform(-scale, scale, size=n)
        h0 = project_feasible(h0, n)

        def nm_obj(h):
            c5v, _ = compute_c5(h)
            penalty = abs(h.sum() - n / 2.0) * 100.0
            penalty += np.sum(np.maximum(0, h - 1)) * 100.0 + np.sum(np.maximum(0, -h)) * 100.0
            return c5v + penalty

        try:
            res = minimize(nm_obj, h0, method='Nelder-Mead',
                           options={'maxiter': 600, 'xatol': 1e-10, 'fatol': 1e-12, 'adapt': True})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)[0]
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 3: Try higher dimensions (15% time)
    dim_t = budget_s * 0.90
    n_candidates = sorted(set([n * 2, 100, 120, 150, 200]))
    for n_try in n_candidates:
        if elapsed() > dim_t:
            break
        if n_try == n or n_try > 250:
            continue
        h_try = resample(best_h, n_try)
        c5_try = compute_c5(h_try)[0]
        # Subgradient on new dimension
        step_s = 0.05
        for it in range(8000):
            if elapsed() > dim_t:
                break
            grad, c5, _ = subgradient(h_try, n_try, top_k=min(20, n_try), temp=10.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            h_try = h_try - step_s * grad
            h_try = project_feasible(h_try, n_try)
            c5_try = compute_c5(h_try)[0]
            step_s *= 0.9995
        # SLSQP polish
        hc, c5v = slsqp_polish(h_try, n_try, T=150, maxiter=200)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            n = n_try

    # Phase 4: Final high-T SLSQP + NM polish
    if elapsed() < budget_s * 0.96:
        for T in [300, 800]:
            if elapsed() > budget_s * 0.96:
                break
            hc, c5v = slsqp_polish(best_h, n, T, maxiter=400)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        # One more NM at very small scale
        h0 = best_h + np.random.uniform(-0.01, 0.01, size=n)
        h0 = project_feasible(h0, n)
        try:
            def fin_obj(h):
                c5v, _ = compute_c5(h)
                return c5v + abs(h.sum() - n / 2.0) * 200.0
            res = minimize(fin_obj, h0, method='Nelder-Mead',
                           options={'maxiter': 400, 'xatol': 1e-11, 'fatol': 1e-13})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)[0]
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final feasibility enforcement
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)[0]
    return best_h, best_c5, len(best_h)
