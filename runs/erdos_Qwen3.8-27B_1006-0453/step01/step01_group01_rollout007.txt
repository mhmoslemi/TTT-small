def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)
    return h
