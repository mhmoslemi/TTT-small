import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_top_lags(h, lags, N):
    """Sum of gradients for given lags."""
    g = np.zeros(N)
    for lag in lags:
        if lag >= 0:
            nv = N - lag
            if nv > 0:
                g[:nv] += (1.0 - h[lag:N])
                if lag > 0:
                    g[lag:] -= h[:nv]
        else:
            l = -lag
            nv = N - l
            if nv > 0:
                g[l:] += (1.0 - h[:nv])
                g[:nv] -= h[l:]
    return g

def optimize_focused(h, N, target, dx, t0, budget_s, rng, n_restarts=3):
    """Optimize using top-T lag focused gradient descent."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for restart in range(n_restarts):
        h = best_h.copy()
        if restart > 0:
            h += rng.randn(N) * (0.02 + 0.01 * restart)
            h = project_sum(h, target)
        
        T = 3  # top lags
        lr = 0.015
        best_local = compute_c5(h, dx)
        best_h_local = h.copy()
        stale = 0
        
        for it in range(3000):
            if time.time() - t0 > budget_s - 15:
                return best_h, best_c5
            if lr < 1e-10:
                break
            
            corr = get_corr(h)
            c5_val = float(corr.max() * dx)
            
            if c5_val < best_local:
                best_local = c5_val
                best_h_local = h.copy()
                stale = 0
            else:
                stale += 1

            top_idx = np.argsort(corr)[-T:]
            top_lags = top_idx - (N - 1)
            
            # Gradient
            g = grad_top_lags(h, top_lags, N)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:

            step = lr * g / g_norm  # normalized direction
            for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:

                h_new = project_sum(h_new, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new <= c5_val + 1e-12:
                    h = h_new
                    c5_val = c5_new
                    break
            else:

                h = project_sum(h, target)

                lr *= 0.8
                if stale > 200:

                    stale = 0

            best_h = best_h_local

def compute_c5(h, dx):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-12:
        return h

def grad_lag(h, k, N):
