def project_h(h, target_sum):
    """Project to [0,1]^n with sum = target_sum"""
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return h

def softmax_gradient(h, dx, beta=50.0):
    """Compute gradient of softmax-smoothed max of correlation values"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_c = corr.max()
    w = np.exp(beta * (corr - max_c))
    w /= w.sum()
    # Gradient computation
    grad = np.zeros(n)
