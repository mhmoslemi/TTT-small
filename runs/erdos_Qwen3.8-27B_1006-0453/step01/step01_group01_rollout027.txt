import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smax_obj(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    mc = np.max(corr)
    val = (mc + np.log(np.sum(np.exp(t * (corr - mc)))) / t) * dx
    s = np.sum(h) - n / 2.0
    return val + lam * s * s

def _smax_grad(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    mc = np.max(corr)
    et = np.exp(t * (corr - mc))
    p = et / np.sum(et)
    S1 = np.zeros(n)
    S2 = np.zeros(n)
    for j in range(n):
        S2[j] = np.dot(h, p[n - 1 - j: 2 * n - 1 - j])
        hist = min(2 * j + 1, n)
        S1[j] = np.dot(h[:hist], p[n - 1 - j: n - 1 - j + hist])
    g = dx * (1.0 - S1 - S2)
    s = np.sum(h) - n / 2.0
    g = g + 2.0 * lam * s
    return g

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return _proj(np.interp(xd, xs, h), nd)

def _make_obj(t, lam):
    def f(x):
        return _smax_obj(x, t, lam)
    return f

def _make_grad(t, lam):
    def g(x):
        return _smax_grad(x, t, lam)
    return g

def _run_lbfgsb(h0, n, t, lam, maxiter, deadline):
    if time.time() > deadline:
        return h0
    f = _make_obj(t, lam)
    g = _make_grad(t, lam)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
        return _proj(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bh = None
    bc = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        bh = _proj(h_init, len(h_init))
        bc = compute_c5(bh)
    except Exception:
        bh = _proj(np.full(280, 0.5) + rng.randn(280) * 0.1, 280)
        bc = compute_c5(bh)

    # Phase 1: Multi-resolution coarse-to-fine
    resolutions = [40, 80, 160, 280]
    time_per_res = budget_s * 0.35 / len(resolutions)
    for n in resolutions:
        if time.time() > t0 + budget_s * 0.5:
            break
        if len(bh) == n:
            h0 = bh.copy()
        else:
            h0 = _interp(bh, n)
        for t in [50.0, 200.0, 600.0]:
            deadline = t0 + budget_s * 0.5
            if time.time() > deadline - 5:
                break
            h0 = _run_lbfgsb(h0, n, t, 200.0, 400, deadline)
            c5 = compute_c5(h0)
            if c5 < bc:
                bc = c5
                bh = h0.copy()

    # Phase 2: Multiple restarts at n=280
