Actually, let me think about this differently. The problem has a lot of symmetry and structure. Let me think about what the optimal solution looks like.

For the integral to be 1 on [0,2], the function h has "mass" 1. The overlap at lag k measures how much h(x) and h(x+k) "disagree" (one is 1, other is 0). We want to minimize the maximum disagreement over all lags.

If h is 1 on a set A of measure 1, the overlap at lag k is measure({x: x∈A, x+k∉A}) = measure(A) - measure(A ∩ (A-k)) = 1 - measure(A ∩ (A-k)).

So minimizing max_k [1 - measure(A ∩ (A-k))] is equivalent to maximizing min_k measure(A ∩ (A-k)).

This is a classic problem! We want a set A of measure 1 in [0,2] such that A overlaps with all its translates as much as possible.

The trivial upper bound: if A = [0,1], then for k near 0, A∩(A-k) ≈ [0,1-k] has measure 1-k, and for k near 1, A∩(A-k) = {0} has measure 0. So max overlap = 1.

A better strategy: make A a "fat" set that's spread out, like A = union of many small intervals distributed evenly. Then A∩(A-k) will be more uniform.

For a 2-coloring (h is 0 or 1), the problem reduces to: partition [0,2] into A (measure 1) and B (measure 1) to minimize max_k measure(A ∩ B shifted by k).

This is related to the "no-three-in-line" type problems and has connections to additive combinatorics.

                A[:n - nl] += w * (1.0 - h[nl:])
                B[nl:] += w * h[:n - nl]
    grad = dx * (A - B) + 2.0 * penalty * (integral - 1.0) * dx
    return val, grad

def _obj(x, temp, penalty):
    v, _ = smooth_obj_and_grad(x, temp, penalty)
    return v

def _grad(x, temp, penalty):
    _, g = smooth_obj_and_grad(x, temp, penalty)
    return g

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def lbfgsb_opt(h_in, temp, penalty, max_iter):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h_in.copy(), jac=_grad, args=(temp, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-7})
    return np.clip(res.x, 0.0, 1.0)

def local_subgradient(h, max_steps, alpha_init):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    alpha = alpha_init
    for step in range(max_steps):
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] -= (1.0 - cur[lag:]) * dx
                if lag > 0:
                    grad[lag:] -= cur[:n - lag] * dx
        else:
            nl = -lag
            if nl < n:
                grad[nl:] -= (1.0 - cur[:n - nl]) * dx
                grad[:n - nl] -= cur[nl:] * dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        for _ in range(25):
            new_h = cur - alpha * grad / gnorm
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = repair_h(new_h)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                cur = new_h
                best_c5 = new_c5
                best_h = new_h.copy()
                alpha *= 1.05
                break
            alpha *= 0.6
        if step % 15 == 14:
            idx = np.random.choice(n, size=max(1, n // 15), replace=False)
            cur[idx] = np.clip(cur[idx] + np.random.randn(len(idx)) * 0.04, 0.0, 1.0)
            cur = repair_h(cur)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_init.copy()
    cur_n = len(cur_h)

    # Phase 1: Multi-temperature L-BFGS-B at initial resolution
    temps1 = [1.0, 0.3, 0.08, 0.02, 0.005]
    for temp in temps1:
        if time.time() > deadline * 0.2:
            break
        cur_h = lbfgsb_opt(cur_h, temp, 50.0, 150)
        c5v = compute_c5(cur_h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = cur_h.copy()

    # Phase 2: Upscale to 2x and optimize
    if time.time() < deadline * 0.35:
        cur_h = repair_h(interp_upscale(cur_h, 2))
        cur_n = len(cur_h)
        temps2 = [0.15, 0.04, 0.01, 0.003]
        for temp in temps2:
            if time.time() > deadline * 0.5:
                break
            cur_h = lbfgsb_opt(cur_h, temp, 80.0, 120)
            c5v = compute_c5(cur_h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cur_h.copy()

    # Phase 3: Upscale to 2x again and optimize
    if time.time() < deadline * 0.55:
        cur_h = repair_h(interp_upscale(cur_h, 2))
        cur_n = len(cur_h)
        temps3 = [0.05, 0.01, 0.002]
        for temp in temps3:
            if time.time() > deadline * 0.72:
                break
            cur_h = lbfgsb_opt(cur_h, temp, 50.0, 100)
            c5v = compute_c5(cur_h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cur_h.copy()

    # Phase 4: Subgradient local search
    if time.time() < deadline * 0.82:
        for _ in range(3):
            if time.time() > deadline * 0.82:
                break
            cur_h, lc5 = local_subgradient(cur_h, 150, 0.3)
            if lc5 < best_c5:
                best_c5 = lc5
                best_h = cur_h.copy()

    # Phase 5: Random restarts with diverse perturbations
    if time.time() < deadline * 0.92:
        n_restarts = 4
        for ri in range(n_restarts):
            if time.time() > deadline * 0.93:
                break
            sigma = 0.12 / (ri + 1)
            h_pert = np.clip(cur_h + np.random.randn(cur_n) * sigma, 0.0, 1.0)
            h_pert = repair_h(h_pert)
            # Quick multi-temp optimization
            for temp in [0.05, 0.01]:
                if time.time() > deadline * 0.93:
                    break
                h_pert = lbfgsb_opt(h_pert, temp, 40.0, 80)
            c5v = compute_c5(h_pert)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_pert.copy()
                cur_h = h_pert.copy()
            else:
                cur_h = h_pert.copy()

    # Final repair
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
