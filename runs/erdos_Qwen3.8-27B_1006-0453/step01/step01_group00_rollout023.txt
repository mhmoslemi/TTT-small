import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def obj_lse(x, T):
    c = c5_vec(x)
    m = np.max(c)
    if T < 1e-12:
        return m
    return m + T * np.log(np.sum(np.exp((c - m) / T)))

def con_sum(x, n):
    return np.sum(x) - n * 0.5

def obj_pnorm(x, p, lam, n):
    c = c5_vec(x)
    c = np.maximum(c, 0.0)
    if p < 50:
        val = np.mean(c ** p) ** (1.0 / p)
    else:
        val = np.max(c)
    return val + lam * (np.sum(x) - n * 0.5) ** 2

def obj_sym(x, T):
    nh = len(x)
    h = np.empty(2 * nh)
    h[:nh] = x
    h[nh:] = 1.0 - x[::-1]
    return obj_lse(h, T)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    ni = np.sum(interior)
    if ni > 0:
        h[interior] += diff / ni
    return np.clip(h, 0.0, 1.0)

    x_new = np.arange(nn) * (2.0 / nn)
    return fix_sum(np.interp(x_new, x_old, h), nn)

                     bounds=bs, constraints=[{'type': 'eq', 'fun': con_sum, 'args': (n,)}],
                     options={'maxiter': mi, 'ftol': 1e-16})
    return np.clip(r.x, 0.0, 1.0)

                     bounds=bs, options={'maxiter': mi, 'ftol': 1e-16})
    return np.clip(r.x, 0.0, 1.0)

def lbfgsb_full(h0, n, p, mi, lam):

                     bounds=bs, options={'maxiter': mi, 'gtol': 1e-10})
    return np.clip(r.x, 0.0, 1.0)

    h2 = h.copy()
    h2[h2 < th] = 0.0
    h2[h2 > 1.0 - th] = 1.0
    h2 = fix_sum(h2, n)
    free = (h2 > 1e-6) & (h2 < 1.0 - th * 0.5)
    fi = np.where(free)[0]
    if len(fi) < 3:
        return h2

        ht = h2.copy()
        ht[fi] = x

    h2[fi] = np.clip(r.x, 0.0, 1.0)
