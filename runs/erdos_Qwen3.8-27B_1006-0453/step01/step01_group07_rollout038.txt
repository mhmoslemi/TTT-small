def upsample(parent, new_n):
    old_n = len(parent)
    old_x = np.arange(old_n) / old_n
    new_x = np.arange(new_n) / new_n
    v = np.interp(new_x, old_x, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0, 1)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0, 1)

def fix_sum(h, n):
    h = np.array(h, dtype=np.float64, copy=True)
    h += (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0, 1)

def _obj(x, dx, beta, kf, half_n):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

        k = min(kf, len(corr) - 1)
        idx = np.argpartition(-corr, k)[:kf]

        tm = tv.max()
        s = np.sum(np.exp((tv - tm) / beta))
        smax = tm + beta * np.log(s)
    else:
        s = np.sum(np.exp((corr - m) / beta))
        smax = m + beta * np.log(s)

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr

def upsample(parent, new_n):
    old_n = len(parent)
    v = np.interp(np.arange(new_n) / new_n, np.arange(old_n) / old_n, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0, 1)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0, 1)

def fix_sum(h, n):
    h = np.array(h, dtype=np.float64, copy=True)
    h += (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0, 1)

def obj_func(x, dx, beta, kf, half):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if kf > 0 and kf < len(corr):
        idx = np.argpartition(-corr, min(kf, len(corr) - 1))[:kf]
        tv = corr[idx]
        tm = tv.max()

        smax = tm + beta * np.log(s)
