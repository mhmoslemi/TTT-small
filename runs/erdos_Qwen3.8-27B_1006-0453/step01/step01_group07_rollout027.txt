import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / beta, -50.0, 0.0)
    s = np.sum(np.exp(shifted))
    return dx * (m + beta * np.log(s))

def smooth_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / beta, -50.0, 0.0)
    w = np.exp(shifted)
    w /= np.sum(w)

    nfull = len(corr)

        wk = w[k]

        # Term 2: -h[j-L] for j where 0<=j<n and 0<=j-L<n
        js2 = max(0, L)
        je2 = min(n, n + L)
        if je2 > js2:
            j = np.arange(js2, je2)
            grad[j] += wk * (-x[j - L])
    return dx * grad

    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v = project_feasible(v, new_n / 2.0)
    return v

    m = np.max(corr)

        L = k - (n - 1)
        js1 = max(0, -L)
        je1 = min(n, n - L)
        if je1 > js1:
            j = np.arange(js1, je1)
            grad[j] += wk * (1.0 - h[j + L])
        js2 = max(0, L)
        je2 = min(n, n + L)
        if je2 > js2:
            j = np.arange(js2, je2)
            grad[j] += wk * (-h[j - L])
    return grad, m

    corr = np.correlate(x, 1.0 - x, mode="full")

    s = np.sum(np.exp(shifted))

    for freq in [4, 6]:
        s = 0.5 + 0.42 * np.sin(np.pi * freq * xgrid + rng.randn() * 0.5)
        inits.append(project_feasible(s, n / 2.0))

    phase1_end = budget_s * 0.40
    for h_init in inits:
        if time.time() - t0 > phase1_end:
            break
        for beta in [0.015, 0.006, 0.0025]:
            if time.time() - t0 > phase1_end:
                break
            try:
                res = _slsqp_run(h_init.copy(), n, beta, bounds, cons, 700, 1e-12)
                h_new = project_feasible(res.x, n / 2.0)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = float(c5)
                    best_h = h_new.copy()
                    best_n = n
            except Exception:
                pass

    phase2_end = budget_s * 0.82
    h = best_h.copy()
    nn = best_n
    cur = compute_c5(h)
    max_cd = int((phase2_end - (time.time() - t0)) * 55)
    for it in range(max(1, max_cd)):
        if time.time() - t0 > phase2_end:
            break
        scale = 0.04 * max(0.15, 1.0 - 0.00003 * it)
        i = rng.randint(nn)
        j = rng.randint(nn)
        if i == j:
            continue
        d = rng.uniform(0.4, 1.6) * scale * rng.choice([-1.0, 1.0])
        h_tr = h.copy()
        h_tr[i] = h[i] + d
        h_tr[j] = h[j] - d
        if h_tr[i] < 0.0 or h_tr[i] > 1.0 or h_tr[j] < 0.0 or h_tr[j] > 1.0:
            h_tr[i] = np.clip(h_tr[i], 0, 1)
            h_tr[j] = np.clip(h_tr[j], 0, 1)
            diff = nn / 2.0 - np.sum(h_tr)
            if abs(diff) > 1e-8:
                h_tr[i] = np.clip(h_tr[i] + diff, 0, 1)
                h_tr[j] = np.clip(h_tr[j] - diff, 0, 1)
            if abs(np.sum(h_tr) - nn / 2.0) > 1e-4:
                continue
        c_new = compute_c5(h_tr)
        if c_new < cur - 1e-14:
            h = h_tr
            cur = float(c_new)
    if cur < best_c5:
        best_c5 = cur
        best_h = h.copy()
        best_n = nn

    phase3_end = budget_s * 0.93
    if time.time() - t0 < phase3_end:
        n2 = best_n
        bounds2 = list(zip(np.zeros(n2), np.ones(n2)))
        cons2 = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n2, n2)}]
        for beta in [0.002, 0.0008]:
            if time.time() - t0 > phase3_end:
                break
            try:
                res = _slsqp_run(best_h.copy(), n2, beta, bounds2, cons2, 900, 1e-14)
                h_new = project_feasible(res.x, n2 / 2.0)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = float(c5)
                    best_h = h_new.copy()
                    best_n = n2
            except Exception:
                pass

    if time.time() - t0 < budget_s * 0.97:
        h = best_h.copy()
        nn = best_n
        cur = compute_c5(h)
        steps2 = int((budget_s * 0.97 - (time.time() - t0)) * 60)
        for it in range(max(1, min(steps2, 15000))):
            if time.time() - t0 > budget_s * 0.975:
                break
            i = rng.randint(nn)
            j = rng.randint(nn)
            if i == j:
                continue
            d = rng.uniform(0.3, 1.0) * 0.004 * rng.choice([-1.0, 1.0])
            h[i] = np.clip(h[i] + d, 0, 1)
            h[j] = np.clip(h[j] - d, 0, 1)
            diff = nn / 2.0 - np.sum(h)
            if abs(diff) > 1e-8:
                h[i] = np.clip(h[i] + diff, 0, 1)
                h[j] = np.clip(h[j] - diff, 0, 1)
            c_new = compute_c5(h)
            if c_new < cur - 1e-14:
                cur = float(c_new)
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            best_n = nn

    return best_h, best_c5, best_n
