    w_lag = w  # w_lag[m + N - 1] = weight for lag m
    for j in range(N):
        # Plus term: m in [-min(j,N-1), min(j,N-1)], contribution w[m]*(1-h[j-m])
        m_lo = max(-j, -(N-1))
        m_hi = min(j, N-1)
        if m_lo <= m_hi:
            m = np.arange(m_lo, m_hi + 1)
            idx = m + N - 1
            h_idx = j - m
            G[j] += np.dot(w_lag[idx], 1.0 - h[h_idx])
        # Minus term: m in [-min(N-1-j,N-1), min(N-1-j,N-1)], contribution -w[m]*h[j+|m|]
        m_lo2 = max(-(N-1-j), -(N-1))
        m_hi2 = min(N-1-j, N-1)
        if m_lo2 <= m_hi2:
            m2 = np.arange(m_lo2, m_hi2 + 1)
            idx2 = m2 + N - 1
            h_idx2 = j + np.abs(m2)
            G[j] -= np.dot(w_lag[idx2], h[h_idx2])
    return G

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_gradient(h, w, N):
    G = np.zeros(N)
    for j in range(N):
        m_lo = max(-j, -(N-1))
        m_hi = min(j, N-1)
        if m_lo <= m_hi:
            m = np.arange(m_lo, m_hi + 1)
            G[j] += np.dot(w[m + N - 1], 1.0 - h[j - m])
        r = N - 1 - j
        m_lo2 = max(-r, -(N-1))
        m_hi2 = min(r, N-1)
        if m_lo2 <= m_hi2:
            m2 = np.arange(m_lo2, m_hi2 + 1)
            G[j] -= np.dot(w[m2 + N - 1], h[j + np.abs(m2)])
    return G

def adam_smooth(h, N, target, dx, t0, time_limit, rng, lr_init=0.04, tau_init=6.0, max_iter=4000):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    tau = tau_init
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit - 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v < best_c5 - 1e-10:
            best_c5 = c5v
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        sc = tau * corr - tau * corr.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(2*N-1) / (2*N-1)
        else:
            w /= ws
        grad = weighted_gradient(h, w, N)
        m_buf = b1 * m_buf + (1-b1) * grad
        v_buf = b2 * v_buf + (1-b2) * grad*grad
        mh = m_buf / (1 - b1**(it+1))
        vh = v_buf / (1 - b2**(it+1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if (it+1) % 300 == 0:
            lr *= 0.82
            tau *= 2.0
            if stall > 200:
                lr *= 0.5
                h = h + rng.randn(N) * 0.02 * (stall / 200.0)
                h = project_sum(h, target)
                stall = 0
        if lr < 1e-9:
            break
    return best_h, best_c5

def coord_descent_polish(h, N, target, dx, t0, time_limit, n_passes=3):
    best_c5 = compute_c5(h, dx)
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    for p in range(n_passes):
        if time.time() - t0 > time_limit - 20:
            break
        for j in range(N):
            a_k = np.zeros(2*N-1)
            b_k = corr_full.copy()
            for k_idx in range(2*N-1):
                m = k_idx - (N-1)
                # d(corr_m)/dh[j]: plus term if j-m in [0,N-1] and m >= 0 (or |m| pattern)
                # For m >= 0: (1-h[j-m])*I(j>=m) - h[j+m]*I(j<=N-1-m)
                # For m < 0, l=-m: (1-h[j-l])*I(j>=l) - h[j+l]*I(j<=N-1-l)
                if m >= 0:
                    if j >= m and j < N:
                        b_k[k_idx] += h[j-m]
                        a_k[k_idx] -= h[j-m]
                    if j <= N-1-m:
                        b_k[k_idx] += h[j+m]
                        a_k[k_idx] += h[j+m]
                else:
                    l = -m
                    if j >= l:
                        b_k[k_idx] += h[j-l]
                        a_k[k_idx] -= h[j-l]
                    if j <= N-1-l:
                        b_k[k_idx] += h[j+l]
                        a_k[k_idx] += h[j+l]
            current = h[j]
            b_k_adj = b_k - a_k * current
            lo_x, hi_x = 0.0, 1.0
            for _ in range(40):
                mid = (lo_x + hi_x) / 2.0
                f_lo = np.max(a_k * 0.0 + b_k_adj)
                f_mid = np.max(a_k * mid + b_k_adj)
                f_hi = np.max(a_k * 1.0 + b_k_adj)
                if f_mid <= f_lo and f_mid <= f_hi:
                    break
                elif f_lo < f_mid:
                    hi_x = mid
                else:
                    lo_x = mid
            new_val = (lo_x + hi_x) / 2.0
            # Adjust for sum constraint: shift all others slightly
            delta = new_val - current
            if abs(delta) < 1e-10:
                continue
            h[j] = new_val
            s = h.sum()
            if abs(s - target) > 1e-10:
                h = project_sum(h, target)
            # Recompute corr for this j
            corr_full = np.correlate(h, 1.0 - h, mode='full')
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
    return h, best_c5

        pos = (i + phase) % period
        if pos < 0.5:
            h[i] = 1.0
    h = project_sum(h, target)
    return h
