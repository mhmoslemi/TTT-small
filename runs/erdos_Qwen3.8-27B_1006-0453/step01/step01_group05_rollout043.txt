```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    t = temp * corr
    m = np.max(t)
    return (np.log(np.sum(np.exp(t - m))) / temp + m / temp) * dx

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def cd_pass(h, n, sweeps=30):
    """Coordinate descent on max of correlations."""
    m = n
    two_m_1 = 2 * m - 1
    idx = np.arange(two_m_1)
    for sw in range(sweeps):
        max_ch = 0.0
        for i in range(m):
            old = h[i]
            comp = 1.0 - h
            corr = np.correlate(h, comp, mode='full')
            
            # a[k] and b[k] for each correlation index k
            d_vals = idx - (m - 1)  # d for each k
            i_d = i - d_vals  # i - d for each k
            i_d2 = i + d_vals  # i + d for each k
            A = ((i_d >= 0) & (i_d < m)).astype(float)
            B = ((i_d2 >= 0) & (i_d2 < m)).astype(float)
            hd = np.where(A > 0, h[i_d.astype(int)], 0.0)
            hd2 = np.where(B > 0, h[i_d2.astype(int)], 0.0)
            a = (1.0 - hd) * A - hd2 * B
            b = corr - old * a
            
            # Find t* minimizing max_k (a_k * t + b_k) for t in [0,1]
            # Collect all candidate points
            candidates = np.array([0.0, 1.0])
            # Pairwise intersections (vectorized)
            a_diff = a[:, None] - a[None, :]
            b_diff = b[None, :] - b[:, None]
            valid = np.abs(a_diff) > 1e-12
            t_mat = np.zeros_like(a_diff)
            np.divide(b_diff, a_diff, out=t_mat, where=valid)
            t_mat[~valid] = -1.0
            cand = t_mat[(t_mat >= 0.0) & (t_mat <= 1.0)]
            if len(cand) > 0:
                candidates = np.concatenate([candidates, cand])
            
            # Evaluate at all candidates
            vals = np.max(a[None, :] * candidates[:, None] + b[None, :], axis=1)
            best_t = candidates[np.argmin(vals)]
            
            h[i] = best_t
            max_ch = max(max_ch, abs(best_t - old))
        
        h = repair_h(h, n)
        if max_ch < 1e-12:
            break
    return h

def slsqp_solve(x0, n, bounds, temp, maxiter):
    res = minimize(
        smooth_obj, x0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
        args=(n, temp),
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return res.x
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

def cd_optimize(h, n, sweeps=30):
    """Coordinate descent minimizing max correlation."""
    m = n
    two_m = 2 * m - 1
    for sw in range(sweeps):
        max_ch = 0.0
        for i in range(m):
            old = h[i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # For each correlation index k, compute slope a and intercept b wrt h[i]
            d_vals = np.arange(two_m) - (m - 1)
            i_md = i - d_vals
            i_pd = i + d_vals
            A = ((i_md >= 0) & (i_md < m)).astype(float)
            B = ((i_pd >= 0) & (i_pd < m)).astype(float)
            i_md_c = np.clip(i_md, 0, m - 1).astype(int)
            i_pd_c = np.clip(i_pd, 0, m - 1).astype(int)
            hd = np.where(A > 0, h[i_md_c], 0.0)
            hd2 = np.where(B > 0, h[i_pd_c], 0.0)
            
            a = (1.0 - hd) * A - hd2 * B
            b = corr - old * a
            
            # Ternary search for min max_k (a_k * t + b_k) on [0,1]
            lo, hi_t = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi_t - lo) / 3.0
                m2 = hi_t - (hi_t - lo) / 3.0
                if np.max(a * m1 + b) < np.max(a * m2 + b):
                    hi_t = m2
                else:
                    lo = m1
            best_t = (lo + hi_t) / 2.0
            
            h[i] = best_t
            max_ch = max(max_ch, abs(best_t - old))
        
        h = repair_h(h, n)
        if max_ch < 1e-11:
            break
    return h

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    t = temp * corr
    mx = np.max(t)
    return (np.log(np.sum(np.exp(t - mx))) + mx) / temp * dx

    return np.sum(x) - n / 2.0

def slsqp(x0, n, bounds, temp, maxiter):

                   constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                   args=(n, temp), options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def cd_optimize(h, n, sweeps=50):
    m = n
    two_m = 2 * m - 1
    d_vals = np.arange(two_m) - (m - 1)
    mid = m - 1  # index for d=0
    for sw in range(sweeps):
        max_ch = 0.0
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
        for i in range(m):
            old = h[i]
            i_md = i - d_vals
            i_pd = i + d_vals
            A = ((i_md >= 0) & (i_md < m)).astype(float)
            B = ((i_pd >= 0) & (i_pd < m)).astype(float)
            i_md_c = np.clip(i_md, 0, m - 1).astype(int)
            i_pd_c = np.clip(i_pd, 0, m - 1).astype(int)
            hd = np.where(A > 0, h[i_md_c], 0.0)
            hd2 = np.where(B > 0, h[i_pd_c], 0.0)
            a = (1.0 - hd) * A - hd2 * B
            b = corr - old * a
            # R0 for d=0: sum_{j!=i} h[j]*(1-h[j])
            R0 = float(np.sum(h * (1.0 - h)) - old * (1.0 - old))
            # Generate candidates
            cands = np.array([0.0, 1.0, 0.5])
            # Line-line intersections (exclude mid)
            mask = np.ones(two_m, dtype=bool)
            mask[mid] = False
            a_m = a[mask]
            b_m = b[mask]
            a_diff = a_m[:, None] - a_m[None, :]
            b_diff = b_m[None, :] - b_m[:, None]
            valid = np.abs(a_diff) > 1e-12
            safe = np.where(valid, a_diff, 1.0)
            t_mat = b_diff / safe
            t_mat[~valid] = -999.0
            in_range = (t_mat >= 0.0) & (t_mat <= 1.0)
            if in_range.any():
                cands = np.concatenate([cands, t_mat[in_range]])
            # Parabola-line: t^2+(a_k-1)t+(b_k-R0)=0
            Bq = a_m - 1.0
            Cq = b_m - R0
            disc = Bq * Bq - 4.0 * Cq
            pos = disc >= 0
            sq = np.sqrt(np.maximum(disc, 0.0))
            t1 = (-Bq + sq) / 2.0
            t2 = (-Bq - sq) / 2.0
            extra = np.concatenate([t1[pos], t2[pos]])
            er2 = (extra >= 0.0) & (extra <= 1.0)
            if er2.any():
                cands = np.concatenate([cands, extra[er2]])
            # Evaluate true objective at candidates
            vm = a[None, :] * cands[:, None] + b[None, :]
            vm[:, mid] = cands - cands * cands + R0
            tv = np.max(vm, axis=1)
            best_t = float(cands[np.argmin(tv)])
            # Update correlation incrementally (exact for linear terms)
            corr += (best_t - old) * a
            corr[mid] = np.sum(h * (1.0 - h))  # fix d=0
            h[i] = best_t
            max_ch = max(max_ch, abs(best_t - old))
        h = repair_h(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
        if max_ch < 1e-12:
            break
    return h


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    t = temp * corr
    mx = np.max(t)
    return (np.log(np.sum(np.exp(t - mx))) + mx) / temp * dx


def sum_con(x, n):
    return np.sum(x) - n / 2.0


def slsqp_solve(x0, n, bounds, temp, maxiter):
    res = minimize(
        smooth_obj, x0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
        args=(n, temp),
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return res.x


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    h0 = repair_h(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    def left():
        return budget_s - 20 - (time.time() - t0)

    def try_update(ht):
        nonlocal best_h, best_c5
        ht = repair_h(ht, n)
        c5t = compute_c5(ht, n)
        if c5t < best_c5:
            best_c5 = c5t
            best_h = ht

    # Phase 1: Coordinate descent from parent
    for ns in [100, 50, 25, 10]:
        if left() < 10:
            break
        try_update(cd_optimize(best_h.copy(), n, sweeps=ns))

    # Phase 2: SLSQP with temperature annealing
    for temp in [15, 40, 100, 300, 800, 2000]:
        if left() < 15:
            break
        mi = min(500, max(80, int(left() * 1.5)))
        ht = slsqp_solve(best_h.copy(), n, bounds, temp, mi)
        try_update(ht)
        if c5t < best_c5:
            try_update(cd_optimize(best_h.copy(), n, sweeps=25))

    # Phase 3: Random perturbations + CD
    for trial in range(60):
        if left() < 15:
            break
        scale = 0.06 * np.exp(-trial / 25.0) + 0.005
        cand = best_h + np.random.uniform(-scale, scale, n)
        try_update(cd_optimize(cand, n, sweeps=20))

    # Phase 4: Structured starts
    if left() > 50:
        for th in [0.25, 0.35, 0.5, 0.65, 0.75]:
            if left() < 15:
                break
            bb = (best_h > th).astype(float)
            bb = repair_h(bb, n)
            try_update(cd_optimize(bb, n, sweeps=50))
        # Periodic patterns
        for per in [2, 3, 4, 5, 6, 9]:
            if left() < 15:
                break
            if n % per == 0:
                base = np.random.uniform(0.1, 0.9, per)
                pat = np.tile(base, n // per)
                try_update(cd_optimize(pat, n, sweeps=40))

    # Phase 5: Alternating CD-SLSQP refinement
    if left() > 60:
        for _ in range(3):
            if left() < 20:
                break
            mi = min(300, max(80, int(left() * 0.8)))
            ht = slsqp_solve(best_h.copy(), n, bounds, 60, mi)
            try_update(ht)
            try_update(cd_optimize(best_h.copy(), n, sweeps=40))

    # Phase 6: Final deep optimization
    if left() > 40:
        try_update(cd_optimize(best_h.copy(), n, sweeps=150))
    if left() > 15:
        mi = min(400, max(100, int(left() * 4)))
        ht = slsqp_solve(best_h.copy(), n, bounds, 25, mi)
        try_update(ht)
        try_update(cd_optimize(best_h.copy(), n, sweeps=30))

    best_h = repair_h(best_h, n)
    return best_h, compute_c5(best_h, n), n
```
