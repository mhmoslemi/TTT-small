def binarize_local_search(h, n, max_iters=10000):
    # Binarize: threshold at 0.5
    b = (h > 0.5).astype(float)
    # Ensure sum is n/2
    # Count ones
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        # Pick the smallest h values among the 1s
        ones_idx = np.where(b == 1)[0]
        sort_order = np.argsort(h[ones_idx])
        for idx in ones_idx[sort_order[:n_ones - target]]:
            b[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(b == 0)[0]
        sort_order = np.argsort(h[zeros_idx])[::-1]
        for idx in zeros_idx[sort_order[:target - n_ones]]:
            b[idx] = 1
    
    c5_cur = compute_c5(b)
    ones_set = set(np.where(b == 1)[0].tolist())
    zeros_set = set(np.where(b == 0)[0].tolist())
    
    for _ in range(max_iters):
        # Try a random flip pair
        if not ones_set or not zeros_set:
            break
        i = np.random.choice(list(ones_set))
        j = np.random.choice(list(zeros_set))
        b[i] = 0; b[j] = 1
        ones_set.remove(i); zeros_set.remove(j)
        ones_set.add(j); zeros_set.add(i)
        c5_new = compute_c5(b)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            # Undo
            b[i] = 1; b[j] = 0
            ones_set.remove(j); zeros_set.remove(i)
            ones_set.add(i); zeros_set.add(j)
    
    return b, c5_cur
