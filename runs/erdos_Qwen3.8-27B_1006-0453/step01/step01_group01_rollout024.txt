    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_feasible(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h2 = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h2)
    if abs(diff) > 1e-12:
        mask = (h2 > 1e-10) & (h2 < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h2[mask] += diff / cnt
            h2 = np.clip(h2, 0.0, 1.0)
    return h2

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    grad += w[n-1] * (1.0 - 2.0 * h)
    for j in range(2*n-1):
        sv = j - (n-1)
        if sv == 0:
            continue
        wj = w[j]
        if wj < 1e-14:
            continue
        if sv > 0:
            m1 = np.arange(sv, n)
            grad[m1] += wj * (1.0 - h[m1 - sv])
            m2 = np.arange(n - sv)
            grad[m2] += wj * (-h[m2 + sv])
        else:
            p = -sv
            m1 = np.arange(n - p)
            grad[m1] += wj * (1.0 - h[m1 + p])
            m2 = np.arange(p, n)
            grad[m2] += wj * (-h[m2 - p])
    return dx * grad

def projected_gradient(h, target_sum, t, n_iter, lr_init, time_limit):
    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_val = compute_c5(h_curr)
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        mu = 0.85 + 0.1 * min(1.0, it / 200.0)
        y = h_curr + mu * (h_curr - h_prev)
        g = smooth_grad(y, t)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        lr = lr_init / (1.0 + 0.005 * it)
        h_new = proj_feasible(y - lr * g, target_sum)
        h_prev = h_curr.copy()
        h_curr = h_new
        if it % 500 == 0:
            val = compute_c5(h_curr)
            if val < best_val:
                best_val = val
                best_h = h_curr.copy()
    val = compute_c5(h_curr)
    if val < best_val:
        best_h = h_curr.copy()
    return best_h

def slsqp_polish(h, t, maxiter=3000):
    n = len(h)
    target_sum = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(hh):
        return smooth_obj(hh, t)
    def objg(hh):
        return smooth_grad(hh, t)
    def ceq(hh):
        return np.array([np.sum(hh) - target_sum])
    def ceqj(hh):
        return np.ones((1, n))
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    try:
        r = minimize(obj, h, jac=objg, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return proj_feasible(np.clip(r.x, 0.0, 1.0), target_sum)
    except Exception:
        return h

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return proj_feasible(np.interp(xd, xs, h, left=h[0], right=h[-1]), nd)
