import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_obj(t, lam):
    def obj(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        s = t * (corr - mx)
        logsum = np.log(np.sum(np.exp(s - s.max())) + 1e-300)
        smooth = mx + (s.max() + logsum) / t
        penalty = lam * (np.sum(h) - n / 2.0) ** 2
        return smooth * dx + penalty
    return obj

    return np.sum(x) - len(x) / 2.0

    return np.ones((1, len(x)))

    xs = np.arange(n_from)
    xd = np.arange(n_to)
    h_new = np.interp(xd, xs, h_from)
    return project(h_new, n_to)

            for _ in range(3):
                if time.time() > t0 + budget_s - 3:
                    break
                try:
                    _params['t'] = temp
                    bnds = [(0.0, 1.0)] * n_fin
                    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
                    r = minimize(obj_func, best_h, method='SLSQP',
                                 bounds=bnds, constraints=cons,
                                 options={'maxiter': 8000, 'ftol': 1e-16})
                    hc = project(np.clip(r.x, 0.0, 1.0), n_fin)
                    c5 = compute_c5(hc)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hc
                except Exception:
                    pass

    final_h = project(best_h, len(best_h))
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
