import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, rng):
    target = n // 2
    idx = np.argpartition(h, n - target)[n - target:]
    h_b = np.zeros(n)
    h_b[idx] = 1.0
    return h_b

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def sa_binary(h_init, n, rng, n_iter, T0, T1):
    h = h_init.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < c5_cur or rng.rand() < np.exp((c5_cur - c5_new) / max(T, 1e-12)):
            c5_cur = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def greedy_local_search(h, n, rng, max_passes=30):
    h = h.copy()
    c5_cur = compute_c5(h)
    for p in range(max_passes):
        improved = False
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(ones)
        for i in ones:
            if h[i] != 1:
                continue
            best_j = -1
            best_c5 = c5_cur
            for j in zeros:
                if h[j] != 0:
                    continue
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_j = j
                h[i], h[j] = h[j], h[i]
            if best_j >= 0:
                h[i], h[best_j] = h[best_j], h[i]
                c5_cur = best_c5
                improved = True
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0].copy()
        if not improved:
            break
    return h, c5_cur

def slsqp_refine(h, n, T, max_iter):
    def obj(v):
        vv = np.clip(v, 0, 1)
        corr = np.correlate(vv, 1.0 - vv, mode='full')
        cm = np.max(corr)
        s = np.sum(np.exp((corr - cm) / T))
        return (cm + T * np.log(s)) * (2.0 / n)
    def cons(v):
        return np.sum(v) - n * 0.5
    res = opt.minimize(obj, h, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': max_iter, 'ftol': 1e-13})
    return res.x

def upsample_binary(h, n_old, n_new, rng):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_interp = np.interp(x_new, x_old, h, period=2.0)
    h_interp = h_interp * (n_new * 0.5 / max(h_interp.sum(), 1e-15))
    h_interp = np.clip(h_interp, 0, 1)
    return to_binary(h_interp, n_new, rng)

def intervals_to_binary(n, n_intervals, rng):
    h = np.zeros(n)
    target = n // 2
    boundary = np.sort(rng.uniform(0, n, size=2 * n_intervals))
    lengths = target / n_intervals
    for k in range(n_intervals):
        start = boundary[2 * k]
        end = min(n, start + lengths)
        h[int(start):int(end)] = 1.0
    return h
