def compute_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))

            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 5: Multi-swap escapes + final polish (10% time)
    d5 = t0 + budget_s * 0.97
    if time.time() < d5 - 20:
        h_m, c5_m = multi_swap_escape(best_h, t0, d5 - 15)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()
    if time.time() < d5 - 5:
        h_f, c5_f = lbfgs_refine(best_h, t0, d5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    integ = np.sum(best_h) * dx
    if integ > 1e-10:
        best_h = best_h / integ
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
