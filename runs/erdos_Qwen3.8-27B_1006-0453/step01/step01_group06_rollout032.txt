    h = h_start.copy()
    cur_max = np.max(get_correlations(h))
    best_local_max = cur_max
    best_local = h.copy()

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlations(h)
    m = np.max(corr)
    if T < 1e-9:
        return m * dx
    shifted = np.minimum(T * (corr - m), 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sa_binary(h_start, deadline, max_iters=1000000, T_hi=0.5, T_lo=0.005):
    n = len(h_start)
    h = h_start.copy()
    corr = get_correlations(h)
    cur_max = np.max(corr)
    best_max = cur_max
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(max_iters - 1, 1)
        T = T_hi * (T_lo / T_hi) ** frac
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if np.random.rand() < 0.6:
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
        else:
            worst_k = np.argmax(corr)
            d = worst_k - (n - 1)
            lo = max(0, -d)
            hi = min(n, n - d)
            valid_i = np.arange(lo, hi)
            mask = (h[valid_i] == 1) & (h[valid_i + d] == 0)
            cands = valid_i[mask]
            if len(cands) > 0:
                i = cands[np.random.randint(len(cands))]
            else:
                i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = get_correlations(h)
        new_max = np.max(new_corr)
        delta = new_max - cur_max
        
        if delta < 0 or (T > 0.001 and np.random.rand() < np.exp(-min(delta / T, 50.0))):
            cur_max = new_max
            corr = new_corr
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_max

def steepest_descent_binary(h_start, deadline, max_rounds=500):
    n = len(h_start)
    h = h_start.copy()
    cur_max = np.max(get_correlations(h))
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_max = np.max(get_correlations(h))
                delta = new_max - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        cur_max += best_delta
    
    return h, cur_max
