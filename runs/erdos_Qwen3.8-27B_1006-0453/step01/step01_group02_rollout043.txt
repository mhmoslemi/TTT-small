dh_i/dp_j = sigmoid'(p_i - offset) * (δ_{ij} - d(offset)/dp_j)

And d(offset)/dp_j is determined by the constraint:
Σ_i dh_i/dp_j = 0 → Σ_i sigmoid'(p_i - offset) * (δ_{ij} - d(offset)/dp_j) = 0
→ sigmoid'(p_j - offset) = d(offset)/dp_j * Σ_i sigmoid'(p_i - offset)
→ d(offset)/dp_j = sigmoid'(p_j - offset) / Σ_i sigmoid'(p_i - offset)

= s_j * grad_h_j - s_j/S * Σ_i grad_h_i * s_i

This is the projection of (grad_h * s) onto the tangent space of the constraint.

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = float(h.sum()) - target
    if abs(diff) < 1e-12:
        return h
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_grad(h, corr, w, N):
    """Weighted gradient of max-corr surrogate."""
    grad = np.zeros(N)
    thresh = 1.0e-5
    for k in range(len(corr)):
        wk = w[k]
        if wk < thresh:
            continue
        lag = k - (N - 1)
        if lag >= 0:
            n = N - lag
            grad[lag:] += wk * (1.0 - h[:n])
            grad[:n] -= wk * h[lag:]
        else:
            l = -lag
            n = N - l
            grad[l:] += wk * (1.0 - h[:n])
            grad[:n] -= wk * h[l:]
    return grad

def optimize_smooth(h, N, target, dx, t0, budget_s, tau_init, lr_init, max_iters):

    best_h = h.copy()

def sa_binary(h_init, N, target, dx, t0, budget_s, rng, max_iters=50000):
    """Simulated annealing on binary solution."""
    h = (h_init > 0.5).astype(float)
    h = project_sum(h, target)
    # Ensure exactly N/2 ones
    ones = np.where(h > 0.5)[0]
    if len(ones) != target:
        # Fix
        h = np.zeros(N)
        if len(ones) > target:
            keep = rng.choice(ones, size=target, replace=False)
            h[keep] = 1.0
        else:
            zeros = np.where(h < 0.5)[0]
            need = target - len(ones)
            new_ones = rng.choice(zeros, size=need, replace=False)
            h[ones] = 1.0
            h[new_ones] = 1.0

        corr = np.correlate(h, 1-h, mode='full')
        return float(np.max(corr))

    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-6:

        lag = k - (N - 1)

            grad[lag:] += wk * (1.0 - h[:n])
