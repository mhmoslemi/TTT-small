def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def proj_constraint(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

def smooth_subgrad(h, beta):

    corr = np.correlate(h, 1.0 - h, mode="full")

            grad[LL:] -= wk * h[:-LL]           # h[m-LL] for m=LL..n-1
