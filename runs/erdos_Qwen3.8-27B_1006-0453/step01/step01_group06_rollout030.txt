                            best[j] -= d
                            best[i] += d
                    if best[j] >= d and best[i] <= 1.0 - d:
                        best[i] += d
                        best[j] -= d
                        c = np.max(np.correlate(best, 1.0 - best, mode='full'))
                        if c < best_corr - 1e-13:
                            best_corr = c
                            improved = True
                        else:
                            best[i] -= d
                            best[j] += d
        if not improved:
            break
    return best, best_corr * 2.0 / n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up with rising T
    for T in [300, 1500, 6000]:
        if time.time() - t0 > budget_s * 0.18:
            break
        cons = [{'type': 'eq', 'fun': sum_constraint}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_c5, best_h, args=(T,), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-14})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Snap to binary + local search
    if time.time() - t0 < budget_s * 0.52:
        bh, bc = binary_local_search(best_h, n, t0, budget_s, 0.52)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh

    # Phase 3: SLSQP from near-binary with very high T
    if time.time() - t0 < budget_s * 0.74:
        for T in [8000, 30000, 100000]:
            if time.time() - t0 > budget_s * 0.76:
                break
            cons = [{'type': 'eq', 'fun': sum_constraint}]
            bnds = [(0.0, 1.0)] * n
            try:
                res = minimize(smooth_c5, best_h, args=(T,), method='SLSQP',
                              bounds=bnds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-14})
                hc = project_feasible(np.clip(res.x, 0, 1), n)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 4: Fine coordinate descent (coarse deltas)
    if time.time() - t0 < budget_s * 0.86:
        fh, fc = fine_coord(best_h, n, t0, budget_s, 0.86,
                           [0.003, 0.008, 0.02, 0.05])
        if fc < best_c5:
            best_c5 = fc
            best_h = fh

    # Phase 5: Ultra-fine coordinate descent
    if time.time() - t0 < budget_s * 0.94:
        uh, uc = fine_coord(best_h, n, t0, budget_s, 0.94,
                           [0.0005, 0.001, 0.003, 0.008])
        if uc < best_c5:
            best_c5 = uc
            best_h = uh

    # Phase 6: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.995:
        for T in [50000, 200000]:
            if time.time() - t0 > budget_s * 0.998:
                break
            cons = [{'type': 'eq', 'fun': sum_constraint}]
            bnds = [(0.0, 1.0)] * n
            try:
                res = minimize(smooth_c5, best_h, args=(T,), method='SLSQP',
                              bounds=bnds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-15})
                hc = project_feasible(np.clip(res.x, 0, 1), n)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
