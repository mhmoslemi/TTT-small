import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(p - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgradient(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:n] += (1.0 - h[0:n-k])
            g[0:n-k] -= h[k:n]
    else:
        k_abs = -k
        if k_abs < n:
            g[0:n-k_abs] += (1.0 - h[k_abs:n])
            g[k_abs:n] -= h[0:n-k_abs]
    return g

def subgradient_loop(h, n, t_start, budget_s, margin=60):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.05
    for it in range(5000):
        if time.time() - t_start > budget_s - margin:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        k_shift = k_star - (n - 1)
        g = compute_subgradient(h, n, k_shift)
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h + 0.1 * step * g  # accept-reject: step back a bit
            h = project_feasible(h, n)
            step *= 0.97
        if it % 500 == 0:
            step = max(0.001, step * 0.9)
    return best_h, best_c5

def soft_max_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    scaled = scaled - m
    return (np.log(np.sum(np.exp(scaled))) / temp + m) * dx

def slsqp_step(x0, n, temp, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}
    res = minimize(soft_max_obj, x0, method='SLSQP',
                   bounds=bounds, constraints=[cons],
                   args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def nelder_mead_step(x0, n, temp, maxiter):
    # Use soft-max as proxy for smoothness
    res = minimize(soft_max_obj, x0, method='Nelder-Mead',
                   args=(n, temp),
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
    return res.x

def repair_h(h, n):

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = project_feasible(h + diff/n, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float)
    lo, hi = -1.5, 2.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(p - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)


def compute_grad(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        ka = -k
        if ka < n:
            g[:n-ka] += (1.0 - h[ka:])
            g[ka:] -= h[:n-ka]
    return g


def subgradient_loop(h, n, t_start, budget_s, margin=60, max_iter=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.04
    lr = 0.04
    v = np.zeros(n)
    momentum = 0.7
    for it in range(max_iter):
        if time.time() - t_start > budget_s - margin:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * 2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        top_k = np.argsort(corr)[-4:]
        weights = np.exp((corr[top_k] - corr[top_k[-1]]) * 20.0)
        weights /= weights.sum()
        g = np.zeros(n)
        for wi, ki in zip(weights, top_k):
            k_shift = int(ki) - (n - 1)
            g += wi * compute_grad(h, n, k_shift)
        v = momentum * v + (1.0 - momentum) * g
        h_new = project_feasible(h - lr * v, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            h = h_new
            lr *= 1.01
        else:
            lr *= 0.85
            v *= 0.5
        if it % 800 == 799:
            momentum = max(0.3, momentum * 0.95)
    return best_h, best_c5


def soft_max_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    scaled = scaled - m
    return (np.log(np.sum(np.exp(scaled))) / temp + m) * dx


def slsqp_step(x0, n, temp, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}
    res = minimize(soft_max_obj, x0, method='SLSQP',
                   bounds=bounds, constraints=[cons],
                   args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, n)


def nm_step(x0, n, temp, maxiter):
    res = minimize(soft_max_obj, x0, method='Nelder-Mead',
                   args=(n, temp),
                   options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-14})
    return project_feasible(res.x, n)


def repair_h(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    return project_feasible(h, n)
