    best_c5 = compute_c5(h, n)
    for _ in range(n_rounds):
        improved = False
        h1_idx = np.where(h > 0.5)[0]
        h0_idx = np.where(h < 0.5)[0]
        for a in h1_idx:
            for b in h0_idx:
                if a == b:
                    continue
                h[a], h[b] = h[b], h[a]
                # Repair to maintain sum
                h = repair_h(h, n)
                v = compute_c5(h, n)
                if v < best_c5 - 1e-12:
                    best_c5 = v
                    improved = True
                else:
                    h[a], h[b] = h[b], h[a]
        if not improved:
            break
    return h.copy(), best_c5

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return dx * (m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(np.sum(free))
        if nf == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            per = diff / nf
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def coord_descend(h, n, n_passes=100):
    best_c5 = compute_c5(h, n)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            bv = best_c5
            bt = h[i]
            for t in np.linspace(0.0, 1.0, 17):
                h[i] = t
                v = compute_c5(h, n)
                if v < bv:
                    bv, bt = v, t
            lo = max(0.0, bt - 0.0625)
            hi = min(1.0, bt + 0.0625)
            for t in np.linspace(lo, hi, 9):
                h[i] = t
                v = compute_c5(h, n)
                if v < bv:
                    bv, bt = v, t
            h[i] = bt
            if bv < best_c5 - 1e-12:
                best_c5 = bv
                improved = True
        if not improved:
            break
    return h.copy(), best_c5

def nm_smooth(h0, n, temp, maxiter):

                   args=(n, temp),
                   options={'maxiter': maxiter, 'xatol': 1e-10,
                            'fatol': 1e-12, 'adaptive': True})
    return res.x

def binary_swap_h(h, n, rounds=80):

    for _ in range(rounds):
        improved = False
        h1 = list(np.where(h > 0.7)[0])
        h0 = list(np.where(h < 0.3)[0])
        for a in h1:
            for b in h0:
                if a == b:
                    continue
                h[a], h[b] = h[b], h[a]
                v = compute_c5(h, n)

                else:

        if not improved:
            break
    return h.copy(), best_c5
