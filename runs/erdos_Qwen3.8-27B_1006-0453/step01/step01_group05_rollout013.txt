import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(x, n, temp, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    val = temp * np.log(np.sum(np.exp((c - m) / temp))) + m
    pen = lam * (np.sum(x) - n / 2.0) ** 2
    return float(val * dx + pen)


def repair_h(h, n, iters=2000):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(iters):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            if diff > 0:
                i = int(np.argmin(h))
                j = int(np.argmax(h))
                step = min(diff / 2.0, 1.0 - h[i], h[j])
            else:
                i = int(np.argmax(h))
                j = int(np.argmin(h))
                step = min(-diff / 2.0, h[i], 1.0 - h[j])
            if step < 1e-15:
                break
            if diff > 0:
                h[i] += step
                h[j] -= step
            else:
                h[i] -= step
                h[j] += step
        else:
            h[free] += diff / nf
            h[free] = np.clip(h[free], 0.0, 1.0)
    return h


def nm_optimize(h0, n, temp, lam, maxiter):
    x0 = np.clip(h0, 1e-9, 1.0 - 1e-9)
    res = minimize(smooth_obj, x0, method='Nelder-Mead',
                   args=(n, temp, lam),
                   options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14})
    return res.x

    if d >= 0:
        for i in range(d, n):
            grad[i] += (1.0 - h[i - d])
        for i in range(0, n - d):
            grad[i] -= h[i + d]
    elif d < 0:

        for i in range(0, n - dd):
            grad[i] += (1.0 - h[i + dd])
        for i in range(dd, n):
            grad[i] -= h[i - dd]
    else:
        for i in range(n):
            grad[i] = 1.0 - 2.0 * h[i]
    return k_star, grad

    best_c5 = compute_c5(h, n)

            if time.time() - t_start > budget_s - 40:

            x = nm_optimize(best_h, n, temp, 200.0, 5000)
            x = repair_h(x, n)
            c5 = compute_c5(x, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()

    # Phase 5: Another pair-swap pass (finer)
    if time.time() - t_start < budget_s - 40:
        h_ps, c5_ps = pair_swap_local(best_h, n, n_sweeps=40)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 6: Multi-start with diverse structures
    n_starts = 12
    for s in range(n_starts):
        if time.time() - t_start > budget_s - 50:
            break
        scale = 0.03 + 0.04 * rng.rand()
        cand = best_h + rng.uniform(-scale, scale, n)
        cand = repair_h(cand, n)
        x = nm_optimize(cand, n, 30, 100.0, 2500)
        x = repair_h(x, n)
        c5 = compute_c5(x, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        h_ps, c5_ps = pair_swap_local(x, n, n_sweeps=15)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 7: Structured starts from scratch
    for mode in range(4):
        if time.time() - t_start > budget_s - 40:
            break
        x_start = gen_structured_start(n, mode, rng)
        x = nm_optimize(x_start, n, 50, 80.0, 3000)
        x = repair_h(x, n)
        c5 = compute_c5(x, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        h_ps, c5_ps = pair_swap_local(x, n, n_sweeps=20)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 8: Try alternative n values (40, 50, 36)
    for n_try in [40, 50, 36]:
        if time.time() - t_start > budget_s - 30:
            break
        x_try = np.full(n_try, 0.5)
        x_try = repair_h(x_try, n_try)
        # Add some structure
        for mode in range(3):
            if time.time() - t_start > budget_s - 20:
                break
            x_s = gen_structured_start(n_try, mode, rng)
            x_opt = nm_optimize(x_s, n_try, 40, 100.0, 3000)
            x_opt = repair_h(x_opt, n_try)
            c5_try = compute_c5(x_opt, n_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = x_opt.copy()
                best_n = n_try
            h_ps, c5_ps = pair_swap_local(x_opt, n_try, n_sweeps=15)
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()
                best_n = n_try

    # Phase 9: Final subgradient + pair-swap polish
    if time.time() - t_start < budget_s - 10:
        h_sg, c5_sg = subgradient_method(best_h, best_n, 200, 0.3)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
        h_ps, c5_ps = pair_swap_local(best_h, best_n, n_sweeps=30)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Final repair and evaluation
    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
