from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    
    # Compute gradient analytically
    # corr[k] = sum_i h[i]*(1-h[i+k-n+1]) for valid i
    # dcorr[k]/dh[m] = (1-h[m+k-n+1])*I - h[m-k+n-1]*I

        for i in range(lo, hi):
            j = i + k - n + 1
            grad[i] += wk * (1.0 - h[j])
            grad[j] -= wk * h[i]
    return dx * T * grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 1e-14:
            h += room * (diff / s)
    else:
        s = h.sum()
        if s > 1e-14:
            h *= (target / s)
    return np.clip(h, 0.0, 1.0)

    return h.sum() - len(h) / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n, k + 1)
        if lo >= hi:
            continue
        i_idx = np.arange(lo, hi)
        j_idx = i_idx + k - n + 1
        grad[i_idx] += wk * (1.0 - h[j_idx])
        grad[j_idx] -= wk * h[i_idx]
    return dx * T * grad

def project(h, n):

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:

        s = room.sum()
        if s > 1e-14:
            h = h + room * (diff / s)
    else:
        s = h.sum()
        if s > 1e-14:
            h = h + h * (diff / s)
    return np.clip(h, 0.0, 1.0)
