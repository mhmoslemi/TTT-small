def sum_eq(x, n):
    return x.sum() - 0.5 * n


def slsqp_solve(h0, n, T, maxiter=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    res = minimize(obj_lse, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, T),
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)


def coord_polish(h, n, max_sweeps=50):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for j in [(i + 1) % n, (i + n // 2) % n, (i + 7) % n]:
                if j == i:
                    continue
                dmax = min(best[i], 1.0 - best[j])
                dmin = max(-best[i], best[j] - 1.0)
                if dmax - dmin < 1e-9:
                    continue
                lo, hi = dmin, dmax
                for _ in range(25):
                    m1 = lo + 0.382 * (hi - lo)
                    m2 = lo + 0.618 * (hi - lo)
                    h1 = best.copy()
                    h1[i] += m1
                    h1[j] -= m1
                    h2 = best.copy()
                    h2[i] += m2
                    h2[j] -= m2
                    v1 = compute_c5(h1, n)
                    v2 = compute_c5(h2, n)
                    if v1 < v2:
                        hi = m2
                    else:
                        lo = m1
                d = (lo + hi) * 0.5
                if abs(d) > 1e-9:
                    ht = best.copy()
                    ht[i] += d
                    ht[j] -= d
                    ht = project_feasible(ht, n)
                    vt = compute_c5(ht, n)
                    if vt < best_c5 - 1e-11:
                        best = ht
                        best_c5 = vt
                        improved = True
        if not improved:
            break
    return best, best_c5


def nelder_refine(h0, n, maxiter=200):
    def obj_fn(x):
        xp = np.clip(x, 0.0, 1.0)
        cv = corr_scaled(xp, n)
        m = cv.max()
        penalty = 100.0 * (xp.sum() - 0.5 * n) ** 2
        return m + np.log(1.0 + np.sum(np.exp(cv - m)) * 0.01) + penalty
    res = minimize(obj_fn, h0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
    return project_feasible(res.x, n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def time_left(margin=30):
        return budget_s - margin - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(45, 0.5)
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP with temperature cooling
    for T in [30, 20, 15, 10, 8, 5, 4, 3, 2]:
        if time_left(300) < 5:
            break
        hi = slsqp_solve(best_h, n, T, maxiter=600)
        ci = compute_c5(hi, n)
        if ci < best_c5:
            best_c5 = ci
            best_h = hi

    # Phase 2: Coordinate descent polish
    if time_left(250) > 10:
        best_h, best_c5 = coord_polish(best_h, n, max_sweeps=40)

    # Phase 3: Structured multi-start
    n_starts = 0
    max_starts = 30
    while n_starts < max_starts and time_left(120) > 15:
        n_starts += 1
        scale = 0.15 * np.exp(-n_starts / 12.0) + 0.02
        cand = best_h + np.random.uniform(-scale, scale, n)
        cand = project_feasible(cand, n)
        # Quick SLSQP
        T_use = 15 if n_starts < 8 else 8
        hi = slsqp_solve(cand, n, T_use, maxiter=400)
        ci = compute_c5(hi, n)
        if ci < best_c5:
            best_c5 = ci
            best_h = hi
            # Quick coord polish on improvement
            if time_left(120) > 20:
                best_h, best_c5 = coord_polish(best_h, n, max_sweeps=15)

    # Phase 4: Nelder-Mead polish
    if time_left(60) > 10:
        hn = nelder_refine(best_h, n, maxiter=300)
        cn = compute_c5(hn, n)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    # Phase 5: Final SLSQP + coord polish
    if time_left(20) > 10:
        hi = slsqp_solve(best_h, n, 5, maxiter=800, ftol=1e-15)
        ci = compute_c5(hi, n)
        if ci < best_c5:
            best_c5 = ci
            best_h = hi
        if time_left(10) > 5:
            best_h, best_c5 = coord_polish(best_h, n, max_sweeps=20)

    # Final SLSQP with very tight tolerance
    if time_left(5) > 3:
        hi = slsqp_solve(best_h, n, 3, maxiter=400, ftol=1e-15)
        ci = compute_c5(hi, n)
        if ci < best_c5:
            best_c5 = ci
            best_h = hi

    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
