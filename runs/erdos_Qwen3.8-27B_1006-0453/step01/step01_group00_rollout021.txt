def softmax_obj(h, T):
    n = len(h)

    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))

And ≥ 1 - 2*n/n = -1 (loose lower bound).

Actually for h ∈ [0,1]: sum(h²) ≥ sum(h)²/n = (n/2)²/n = n/4 (by Cauchy-Schwarz).
So center value ≤ 1 - 2*(n/4)/n = 1 - 1/2 = 1/2.
And center value ≥ 1 - 2*(n/2)/n = 0.

So the center value is in [0, 1/2]. The max C₅ is at least the center value.

For the current solution with C₅ ≈ 0.38, the center value must be ≤ 0.38, and some off-center lag must be the actual max.

Let me also consider: what happens at lag n-1+s for small s?
correlate(h, 1-h, 'full')[n-1+s] = sum_i h[i+s]*(1-h[i]) for valid i (0 ≤ i < n-s)
= sum_i h[i+s] - sum_i h[i+s]*h[i]

The first term is sum of h over indices s to n-1 = sum(h) - sum(h[0:s]) = n/2 - sum(h[0:s]).
The second term is the (s-lag) autocorrelation.

For small s, the first term is ≈ n/2 (minus a few terms), and the second term is the autocorrelation at lag s.

So the value at lag n-1+s is approximately (n/2 - autocorr_s) * dx.

For this to be small, we need autocorr_s ≈ n/2, i.e., the autocorrelation at small lags should be close to the zero-lag value. This means h should be very "flat" or have very smooth transitions.

But h can't be too flat because it must go from 0 to 1 (or have high variance) to keep the center value low.

This is the fundamental tension: we want h to be smooth (to keep off-center correlations small) but also to have low "energy" (sum of h²) to keep the center correlation small.

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    h = h.copy()
    interior = (h > 1e-7) & (h < 1.0 - 1e-7)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        interior = (h > 1e-9) & (h < 1.0 - 1e-9)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def sum_constraint(h, n):
    return np.sum(h) - n * 0.5

def slsqp_optimize(h_start, n, T, max_iter):
    try:
        res = opt.minimize(
            softmax_obj, h_start, args=(T,), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-13}
        )
        return res.x
    except Exception:
        return h_start

def coordinate_descent(h, n, max_sweeps, time_limit):
    t0 = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        total_change = 0.0
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            v0 = compute_c5(h)
            h[i] = 1.0
            v1 = compute_c5(h)
            h[i] = old
            v_old = compute_c5(h)
            best_val = min(v0, v1, v_old)
            best_t = old
            if v0 <= best_val:
                best_t = 0.0
            if v1 <= best_val:
                best_t = 1.0
            lo = max(0.0, best_t - 0.12)
            hi = min(1.0, best_t + 0.12)
            if hi - lo > 1e-9:
                a = lo + (1.0 - gr) * (hi - lo)
                b = lo + gr * (hi - lo)
                h[i] = a
                fa = compute_c5(h)
                h[i] = b
                fb = compute_c5(h)
                for _ in range(12):
                    if fa <= fb:
                        hi = b; b = a; fb = fa
                        a = lo + (1.0 - gr) * (hi - lo)
                        h[i] = a; fa = compute_c5(h)
                    else:
                        lo = a; a = b; fa = fb
                        b = lo + gr * (hi - lo)
                        h[i] = b; fb = compute_c5(h)
                best_t = (lo + hi) * 0.5
                h[i] = best_t
                best_val = compute_c5(h)
            h[i] = best_t
            total_change += abs(best_t - old)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if total_change < 1e-11 * n:
            break
    return h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_new = np.interp(x_new, x_ext, h_ext)
    s = np.sum(h_new)
    if s > 1e-15:
        h_new *= (new_n * 0.5) / s
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP with decreasing temperature
    for T in [0.05, 0.015, 0.005, 0.002]:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 150:
            break
        h_opt = slsqp_optimize(h_best.copy(), n_best, T, max_iter=600)
        h_opt = fix_sum(h_opt, n_best)
        c5_cand = compute_c5(h_opt)
        if c5_cand < c5_best:
            h_best = h_opt
            c5_best = c5_cand

    # Phase 2: Coordinate descent (main workhorse)
    elapsed = time.time() - t_start
    remaining = min(budget_s - elapsed - 300, 350)
    if remaining > 30:
        h_best, c5_best = coordinate_descent(h_best, n_best, max_sweeps=500, time_limit=remaining)

    # Phase 3: Random perturbations + re-optimization
    for trial in range(6):
        elapsed = time.time() - t_start
        if budget_s - elapsed < 200:
            break
        scale = 0.04 * (1.0 + 0.3 * trial)
        noise = np.random.randn(n_best) * scale
        h_pert = np.clip(h_best + noise, 0.0, 1.0)
        h_pert = fix_sum(h_pert, n_best)
        h_pert, c5_cand = coordinate_descent(h_pert, n_best, max_sweeps=150, time_limit=35)
        if c5_cand < c5_best:
            h_best = h_pert
            c5_best = c5_cand

    # Phase 4: Multi-resolution exploration
    for n_try in [140, 100, 200, 70]:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 160:
            break
        h_try = upsample(h_best, n_try)
        h_opt = slsqp_optimize(h_try, n_try, 0.02, max_iter=400)
        h_opt = fix_sum(h_opt, n_try)
        h_opt, c5_cand = coordinate_descent(h_opt, n_try, max_sweeps=200, time_limit=50)
        if c5_cand < c5_best:
            h_best = h_opt
            c5_best = c5_cand
            n_best = n_try

    # Phase 5: Final coordinate descent polish
    elapsed = time.time() - t_start
    if budget_s - elapsed > 40:
        h_best, c5_best = coordinate_descent(h_best, n_best, max_sweeps=400, time_limit=budget_s - elapsed - 15)

    # Phase 6: Very fine SLSQP at end
    elapsed = time.time() - t_start
    if budget_s - elapsed > 30:
        h_opt = slsqp_optimize(h_best.copy(), n_best, 0.001, max_iter=800)
        h_opt = fix_sum(h_opt, n_best)
        c5_cand = compute_c5(h_opt)
        if c5_cand < c5_best:
            h_best = h_opt
            c5_best = c5_cand
        h_best, c5_best = coordinate_descent(h_best, n_best, max_sweeps=100, time_limit=20)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
