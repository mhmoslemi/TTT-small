        swap[j] = h[i]
        c5_new = compute_c5(swap)
        if c5_new <= best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / T):
            h = swap
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.zeros(60)
        parent[:30] = 1.0

    target_ns = [48, 54, 60, 66, 72, 78, 84, 90, 96, 108, 120]
    best_h = parent.copy()
    best_c5 = compute_c5(best_h)

    for n in target_ns:
        if remaining() < 40:
            break
        h0 = interp_to_n(parent, n)
        h0 = project_feasible(h0, n)

        # Coordinate descent phase
        if remaining() > 50:
            h_cd = coord_descent(h0, n, n_sweeps=50, rng=rng, t_limit=min(40, remaining() * 0.25))
            if compute_c5(h_cd) < compute_c5(h0):
                h0 = h_cd

        # SLSQP polish
        for T in [80, 300, 1000]:
            if remaining() < 20:
                break
            h_sl = slsqp_opt(h0, n, T, maxiter=500)
            if compute_c5(h_sl) < compute_c5(h0):
                h0 = h_sl

        # Second coordinate descent
        if remaining() > 30:
            h_cd2 = coord_descent(h0, n, n_sweeps=30, rng=rng, t_limit=min(25, remaining() * 0.2))
            if compute_c5(h_cd2) < compute_c5(h0):
                h0 = h_cd2

        c5_this = compute_c5(h0)
        if c5_this < best_c5:
            best_c5 = c5_this
            best_h = h0.copy()

    # Binary annealing at a few resolutions, then continuous optimization
    if remaining() > 120:
        for n_bin in [60, 72, 84]:
            if remaining() < 50:
                break
            h_bin, c5_bin = binary_anneal(n_bin, rng, n_iter=8000, T_start=0.008, T_end=0.0005)
            # Refine: use binary as starting point for continuous
            h_start = h_bin.astype(float).copy()
            h_r = coord_descent(h_start, n_bin, n_sweeps=40, rng=rng, t_limit=20)
            if compute_c5(h_r) < compute_c5(h_start):
                h_start = h_r
            for T in [100, 500]:
                if remaining() < 20:
                    break
                h_s = slsqp_opt(h_start, n_bin, T, maxiter=400)
                if compute_c5(h_s) < compute_c5(h_start):
                    h_start = h_s
            h_r2 = coord_descent(h_start, n_bin, n_sweeps=20, rng=rng, t_limit=10)
            if compute_c5(h_r2) < compute_c5(h_start):
                h_start = h_r2
            c5_b = compute_c5(h_start)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_start.copy()

    # Final polish on best solution
    n_best = len(best_h)
    if remaining() > 20:
        for T in [600, 2000, 5000]:
            if remaining() < 10:
                break
            h_f = slsqp_opt(best_h, n_best, T, maxiter=600)
            if compute_c5(h_f) < best_c5:
                best_c5 = compute_c5(h_f)
                best_h = h_f

    if remaining() > 10:
        h_f2 = coord_descent(best_h, len(best_h), n_sweeps=25, rng=rng, t_limit=10)
        if compute_c5(h_f2) < best_c5:
            best_c5 = compute_c5(h_f2)
            best_h = h_f2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
