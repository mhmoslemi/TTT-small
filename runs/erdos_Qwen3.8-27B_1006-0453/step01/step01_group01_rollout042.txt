import time

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.fft.ifft(np.conj(H) * G).real * dx
    return float(np.max(corr)), corr

def gradient(h, t):
    """Gradient of (1/t)*log(sum exp(t*c_k)) where c_k = circ_corr(h, 1-h) * dx"""
    n = len(h)
    dx = 2.0 / n
    _, c = compute_c5_fft(h)  # c already includes dx
    
    # Softmax weights
    mc = np.max(c)
    s = t * (c - mc)
    w = np.exp(s)
    sw = np.sum(w)
    if sw == 0:
        w = np.ones(n) / n
    else:
        w /= sw

    W = np.fft.fft(w)
    H = np.fft.fft(h)
    p1 = np.fft.ifft(np.conj(W) * H).real
    p2 = np.fft.ifft(W * H).real

def project(h, n):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def obj_val(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(c)
    s = t * (c - mc)
    return mc + np.log(np.sum(np.exp(s))) / t

def grad_fn(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(c)
    s = t * (c - mc)
    w = np.exp(s)
    sw = w.sum()
    if sw < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= sw
    g = np.zeros(n)
    for k in range(2 * n - 1):
        sh = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g * dx

def proj(h, n):
    tgt = n * 0.5
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        sv = float(np.clip(h - mid, 0.0, 1.0).sum())
        if sv > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def pg_solve(h0, t, max_it, mom, deadline):
    n = len(h0)
    h = h0.astype(float).copy()
    v = np.zeros(n)
    alpha = 1.0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        g = grad_fn(h, t)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = mom * v + g
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        f0 = obj_val(h, t)
        a = alpha
        ok = False
        h1 = h
        for _ in range(25):
            if a < 1e-12:
                break
            h1 = proj(h - a * v, n)
            f1 = obj_val(h1, t)
            if f1 < f0 - 1e-4 * a * gn * vn:
                ok = True
                break
            a *= 0.5
        if ok:
            alpha = min(a * 1.1, 2.0 / max(gn, 1e-8))
            h = h1
        else:
            h1 = proj(h - a * v, n)
            if obj_val(h1, t) < f0:
                h = h1
            else:
                break
    return h
