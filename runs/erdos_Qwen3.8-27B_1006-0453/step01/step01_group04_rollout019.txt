import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import CubicSpline
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_obj(h, temp, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / temp
    sm = cmax + temp * np.log(np.sum(np.exp(shifted)))
    objective = sm * dx
    integral = np.sum(h) * dx
    objective += mu * (integral - 1.0) ** 2
    return objective

def smooth_c5_grad(h, temp, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    weights = np.exp((corr - cmax) / temp)
    wsum = np.sum(weights)
    grad_corr = np.zeros(2 * n - 1)
    one_minus_h = 1.0 - h
    # For each lag k (stored at index k+(n-1) in full output):
    # corr[k+(n-1)] = sum_j h[j+k]*(1-h[j])
    # d/dh[m] = (1-h[m-k])*I(valid) - h[m+k]*I(valid)
    # We compute the gradient of the smooth max:
    # grad[h[m]] = dx * sum_k (w_k/wsum) * d(corr_k)/dh[m]
    # + 2*mu*(integral-1)*dx
    # Efficient: for each m, grad[m] = dx * sum_k (w_k/wsum)*[(1-h[m-offset_k])*I - h[m+offset_k]*I]
    # where offset_k = k - (n-1)
    # This is O(n^2) but n is at most ~3000, so ~9M ops (fine)
    # Better: use the correlation structure
    # d(corr_{k})/dh[m] = (1-h[m-k'])*I(0<=m-k'<n) - h[m+k']*I(0<=m+k'<n) where k'=k-(n-1)
    # Sum over k of (w_k/wsum) * above
    # = sum_k' (w_{k'+n-1}/wsum) * [(1-h[m-k'])*I(0<=m-k'<n) + (-h[m+k'])*I(0<=m+k'<n)]
    # First part: sum_k' w'[k']*(1-h[m-k'])*I = correlate-style sum
    # Let's just do it directly with the weights
    offsets = np.arange(-(n-1), n)  # k' values
    w = weights / wsum
    # Part 1: sum_k' w[k'] * (1-h[m-k']) for valid m-k'
    # Part 2: -sum_k' w[k'] * h[m+k'] for valid m+k'
    grad = np.zeros(n)
    for k_prime in range(-(n-1), n):
        wp = w[k_prime + (n-1)]
        if wp < 1e-12:
            continue
        if k_prime >= 0:
            # 1-h[m-k'] valid when m >= k'
            grad[k_prime:] += wp * one_minus_h[:n - k_prime]
            # -h[m+k'] valid when m < n-k'
            grad[:n - k_prime] -= wp * h[k_prime:]
        elif k_prime < 0:
            o = -k_prime
            # 1-h[m+k'] valid when m < n-o
            grad[:n - o] += wp * one_minus_h[o:]
            # -h[m-o] valid when m >= o
            grad[o:] -= wp * h[:n - o]
    grad *= dx
    integral = np.sum(h) * dx
    grad += 2.0 * mu * (integral - 1.0) * dx
    return grad

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h, 0.0, 1.0)
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h = h / integ
    return np.clip(h, 0.0, 1.0)

def cubic_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    cs = CubicSpline(x_old, h, bc_type='periodic')
    h_new = cs(x_new)
    return np.clip(h_new, 0.0, 1.0)

def subgradient_step(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    offset = k_star - (n - 1)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    if offset >= 0:
        grad[offset:] += one_minus_h[:n - offset]
        grad[:n - offset] -= h[offset:]
    else:
        o = -offset
        grad[:n - o] += one_minus_h[o:]
        grad[o:] -= h[:n - o]
    h_new = h - alpha * grad * dx
    return h_new

def structured_perturb(h, rng):
    n = len(h)
    h2 = h.copy()
    # Shift a random window
    wlen = max(3, n // 12)
    start = rng.randint(0, n - wlen)
    shift = rng.randint(1, max(2, n // 60))
    window = h2[start:start + wlen]
    new_window = np.zeros(wlen)
    new_window[:wlen - shift] = window[shift:]
    new_window[wlen - shift:] = window[:shift]
    h2[start:start + wlen] = new_window
    # Small Gaussian noise
    h2 += rng.randn(n) * 0.05
    return repair_h(np.clip(h2, 0.0, 1.0))

def optimize_block(h_in, n, t0, deadline, max_iter):
    bounds = [(0.0, 1.0)] * n
    cur = h_in.copy()
    best_local = compute_c5(cur)
    stages_t = [0.005, 0.001, 0.0002, 0.00005]
    stages_mu = [2000.0, 10000.0, 50000.0, 200000.0]
    for si in range(len(stages_t)):
        if time.time() >= deadline:
            break
        temp = stages_t[si]
        mu = stages_mu[si]
        frac_iter = 1.0 / (1.0 + 0.4 * si)
        ni = max(30, int(max_iter * frac_iter))
        try:
            res = minimize(
                smooth_c5_obj, cur, args=(temp, mu),
                jac=smooth_c5_grad,
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': ni, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': ni * 3}
            )
        except Exception:
            res = minimize(
                smooth_c5_obj, cur, args=(temp, mu),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': ni, 'ftol': 1e-14, 'gtol': 1e-10}
            )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best_local:
            best_local = c5v
        cur = nh
    return cur, best_local
