def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def lse_gradient(h, dx, beta):
    n = len(h)
    corr = compute_corrs(h)
    m = corr.max()
    w = np.exp(beta * (corr - m))
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(-(n-1), n):
            wc = w[k + (n-1)]
            if wc < 1e-15:
                continue
            mk = j - k
            if 0 <= mk < n:
                g += wc * (1.0 - h[mk])
            pk = j + k
            if 0 <= pk < n:
                g -= wc * h[pk]
        grad[j] = g
    return grad * dx

def lse_optimize(h0, dx, target_sum, n_iter=400, t_deadline=None):

                stale = 0
                break

            lr *= 0.92
            if stale > 80:
                break
        else:
            lr = min(lr * 1.03, 0.3)
    return best_h, best_c5

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        h = x[:n]
        return x[n] - np.correlate(h, 1.0 - h, mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.004]])
    bnds = [(0.0, 1.0)] * n + [(0.0, n * 0.6)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
