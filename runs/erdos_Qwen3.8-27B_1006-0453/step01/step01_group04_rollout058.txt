from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:

    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

    n = len(h)
    m = np.arange(-(n-1), n)  # offsets

    term2 = np.zeros(2*n-1)
    term2[valid2] = -h[idx2[valid2]]

        t_best = ts_fine[np.argmin(vals_fine)]

            ib = np.argmin(vals)
            t_best = ts_grid[ib]
            
            if t_best > 0.02 and t_best < 0.98:
                lo = max(0.0, t_best - 0.05)
                hi = min(1.0, t_best + 0.05)
                ts_fine = np.linspace(lo, hi, 32)
                vals_f = np.max(S[None, :] * ts_fine[:, None] + B[None, :], axis=1)
                t_best = ts_fine[np.argmin(vals_f)]
            
            if abs(t_best - h[i]) > 1e-12:
                h[i] = t_best
            
            if (cnt + 1) % 200 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full')
                top_idx = np.argsort(corr)[::-1][:tk]
                top_off = all_off[top_idx]
        
        h = repair_h(h)
    
    return h

def lbgfs_polish(h, t0, deadline, n_sweeps_dbg):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    schedules = [
        (0.06, 200.0, 300),
        (0.02, 800.0, 250),
        (0.005, 3000.0, 200),
        (0.001, 10000.0, 150),
    ]
    cur = h.copy()
    for temp, pen, maxit in schedules:
        if time.time() > deadline:
            break
        res = minimize(smooth_c5, cur, args=(temp, pen),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10})
        nh = repair_h(np.clip(res.x, 0.0, 1.0))
        cur = nh
    return cur

def perturb_structured(h, rng):
    n = len(h)
    mode = rng.integers(3)
    if mode == 0:
        k = rng.integers(-n // 4, n // 4 + 1)
        h2 = np.roll(h, k)
    elif mode == 1:
        h2 = h[::-1].copy()
    else:
        a = rng.uniform(0.2, 0.5)
        h2 = a * h + (1 - a) * h[::-1]
    return repair_h(h2)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float).copy()
    h_init = repair_h(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent at initial resolution n=360
    h1 = h_init.copy()
    h1 = coord_descent(h1, t0, t0 + budget_s * 0.25, n_sweeps=40, top_k=120)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
    
    # Phase 2: Upscale to n=720, coordinate descent
    if time.time() < t0 + budget_s * 0.35:
        h2 = interp_upscale(h1, 2)
        h2 = repair_h(h2)
        h2 = coord_descent(h2, t0, t0 + budget_s * 0.6, n_sweeps=30, top_k=200)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
    else:
        h2 = h1.copy()
    
    # Phase 3: L-BFGS-B polishing at n=720
    if time.time() < t0 + budget_s * 0.8:
        h3 = lbgfs_polish(h2, t0, t0 + budget_s * 0.8, 4)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
    
    # Phase 4: Try structured perturbations + re-optimization
    if time.time() < t0 + budget_s * 0.88:
        for _ in range(2):
            if time.time() > t0 + budget_s * 0.88:
                break
            h_p = perturb_structured(best_h, rng)
            h_p = coord_descent(h_p, t0, t0 + budget_s * 0.88, n_sweeps=15, top_k=150)
            cp = compute_c5(h_p)
            if cp < best_c5:
                best_c5 = cp
                best_h = h_p.copy()
    
    # Phase 5: Final L-BFGS polish on best
    if time.time() < t0 + budget_s * 0.93:
        h_f = lbgfs_polish(best_h, t0, t0 + budget_s * 0.95, 3)
        cf = compute_c5(h_f)
        if cf < best_c5:
            best_c5 = cf
            best_h = h_f.copy()
    
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
