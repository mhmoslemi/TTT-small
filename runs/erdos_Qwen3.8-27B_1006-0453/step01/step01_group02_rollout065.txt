def project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Binary search for the right shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        s = np.clip(h+mid, 0, 1).sum()
        if s > target: lo = mid
        else: hi = mid
    return np.clip(h+0.5*(lo+hi), 0, 1)

def corr_and_topk_grad(h, N, topk=5):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    k = min(topk, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    idx = idx[np.argsort(corr[idx])[::-1]]
    
    grad = np.zeros(N)
    for p in idx:
        lag = int(p) - (N-1)
        if lag >= 0:
            nv = N - lag
            if nv > 0 and lag < N:
                grad[:nv] += (1.0 - h[lag:lag+nv])
                if lag > 0:
                    grad[lag:] -= h[:nv]
        else:
            l = -lag
            nv = N - l
            if nv > 0 and l < N:
                grad[l:] += (1.0 - h[:nv])
                if l > 0:
                    grad[:nv] -= h[l:l+nv]
    return corr, grad, float(np.max(corr))

def gradient_optimize(h, N, target, dx, t0, t_end, topk=5, lr=0.02, max_iter=5000):
    best_h, best_c5 = h.copy(), float(np.max(np.correlate(h,1-h,mode='full'))*dx)
    for it in range(max_iter):
        if time.time() > t_end: break
        corr, grad, cmax = corr_and_topk_grad(h, N, topk)
        c5 = cmax * dx
        if c5 < best_c5: best_h, best_c5 = h.copy(), c5
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10: break
        h_new = project_sum(h - lr * grad / gnorm, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new
            h = h_new
            lr = min(lr * 1.1, 0.1)
        else:
            h = h_new  # still update to maintain trajectory
            lr *= 0.85
        if lr < 1e-8: break
    return best_h, best_c5

def sa_continuous(h, N, target, dx, t0, t_end, rng, T0=0.01, T1=0.001, n_iter=20000):
    """Simulated annealing in continuous space."""
    best_h, best_c5 = h.copy(), float(np.max(np.correlate(h,1-h,mode='full'))*dx)
    curr_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end: break
        T = T0 * (T1/T0) ** (it / n_iter)
        # Random move: pair exchange
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j: continue
        delta = h[j] - h[i]
        t = rng.uniform(0.1, 1.0)
        h[i] += t * delta
        h[j] -= t * delta
        h = project_sum(h, target)
        c5_new = compute_c5(h, dx)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def corr_topk_grad(h, N, topk):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    k = min(topk, 2 * N - 1)
    part = np.argpartition(corr, -k)[-k:]
    part = part[np.argsort(corr[part])[::-1]]
    grad = np.zeros(N)
    for p in part:
        lag = int(p) - (N - 1)
        if lag >= 0:
            nv = N - lag
            if nv > 0 and lag < N:
                grad[:nv] += (1.0 - h[lag:lag + nv])
                if lag > 0:
                    grad[lag:] -= h[:nv]
        else:
            l = -lag
            nv = N - l
            if nv > 0 and l < N:
                grad[l:] += (1.0 - h[:nv])
                if l > 0:
                    grad[:nv] -= h[l:l + nv]
    return corr, grad, float(np.max(corr))

def grad_opt(h, N, target, dx, t_deadline, topk, lr_init, max_iter):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    lr = lr_init
    omh = 1.0 - h
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        corr, grad, cmax = corr_topk_grad(h, N, topk)
        c5v = cmax * dx
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        step = lr / gn
        h_new = project_sum(h - step * grad, target)
        omh_new = 1.0 - h_new
        c5_new = float(np.max(np.correlate(h_new, omh_new, mode='full')) * dx)
        if c5_new <= c5v + 1e-14:
            h = h_new
            lr = min(lr * 1.05, lr_init * 2)
        else:
            lr *= 0.8
        if lr < 1e-9:
            break
    return best_h, best_c5

def sa_binary(h_bin, N, dx, t_deadline, rng, T0=0.003, T1=0.0002, n_iter=80000):
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    T = T0
    cool_rate = (T1 / T0) ** (1.0 / n_iter)
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        ones_mask = h > 0.5
        zeros_mask = ~ones_mask
        ni = np.count_nonzero(ones_mask)
        nj = np.count_nonzero(zeros_mask)
        if ni == 0 or nj == 0:
            break
        idx_ones = np.flatnonzero(ones_mask)
        idx_zeros = np.flatnonzero(zeros_mask)
        i = idx_ones[rng.randint(ni)]
        j = idx_zeros[rng.randint(nj)]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h, dx)
        d = c5_new - curr_c5
        if d < 0 or rng.random() < np.exp(-d / T):
            curr_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool_rate
    return best_h, best_c5

def sa_continuous(h, N, target, dx, t_deadline, rng, T0=0.004, T1=0.0003, n_iter=30000):

            h[j] += t * delta
