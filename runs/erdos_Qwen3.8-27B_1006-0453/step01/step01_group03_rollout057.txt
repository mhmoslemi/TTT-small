import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:

        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0.copy(), [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_feasible(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def local_search_pairs(h, dx, target_sum, n_sweeps=5, deadline=None):
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur, dx)
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if h_cur[i] <= 0 or h_cur[j] >= 1:
                    continue
                if h_cur[i] >= 1 or h_cur[j] <= 0:
                    continue
                delta = h_cur[i] - h_cur[j]
                if abs(delta) < 1e-6:
                    continue
                t_max = min(0.5, abs(delta) * 0.8)
                for t in [t_max, t_max * 0.5, t_max * 0.25, -t_max, -t_max * 0.5, -t_max * 0.25]:
                    if h_cur[i] + t < 0 or h_cur[i] + t > 1:
                        continue
                    if h_cur[j] - t < 0 or h_cur[j] - t > 1:
                        continue
                    h_try = h_cur.copy()
                    h_try[i] += t
                    h_try[j] -= t
                    c5_try = compute_c5(h_try, dx)
                    if c5_try < best_c5 - 1e-12:
                        h_cur = h_try
                        best_c5 = c5_try
                        improved = True
                        break
        if not improved:
            break
    return h_cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.zeros(47)
        h0[:23] = 1.0
        h0[23] = 0.5
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Annealed smooth gradient (main workhorse)
    dl1 = t_start + budget_s * 0.35
    h1, c1 = anneal_optimize(h0, dx, target_sum, 8000, dl1)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    if time_left() > 30:
        h2 = slsqp_polish(best_h, dx, target_sum, 250)
        c2 = compute_c5(h2, dx)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Pair local search
    if time_left() > 30:
        dl3 = min(t_start + budget_s - 30, time.time() + 30)
        h3, c3 = local_search_pairs(best_h, dx, target_sum, 3, dl3)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Multi-start with diverse perturbations
    strategies = [
        lambda bh: bh + 0.03 * np.random.randn(n),
        lambda bh: bh + 0.06 * np.random.randn(n),
        lambda bh: bh + 0.1 * np.random.randn(n),
        lambda bh: np.clip(bh + 0.15 * np.random.randn(n), 0, 1),
    ]
    for r in range(10):
        if time_left() < 80:
            break
        if r < len(strategies):
            h_r = strategies[r % len(strategies)](best_h)
        elif r < 8:
            h_r = best_h.copy()
            s = np.random.randint(0, max(n - 5, 1))
            l = np.random.randint(2, min(8, n - s))
            h_r[s:s+l] = np.clip(h_r[s:s+l] + 0.4 * np.random.randn(l), 0, 1)
        else:
            h_r = np.zeros(n)
            perm = np.random.permutation(n)
            h_r[perm[:int(target_sum)]] = 1.0
            frac = target_sum - int(target_sum)
            if frac > 0.01:
                h_r[perm[int(target_sum)]] = frac
        h_r = project_feasible(h_r, target_sum)
        c5_r = compute_c5(h_r, dx)
        # Short optimization from this start
        dl_r = min(t_start + budget_s - 40, time.time() + 20)
        if r % 2 == 0:
            h_r2, c5_r2 = anneal_optimize(h_r, dx, target_sum, 2000, dl_r)
        else:
            h_r2, c5_r2 = momentum_optimize(h_r, dx, target_sum, 1500, dl_r)
        if c5_r2 < best_c5:
            best_h, best_c5 = h_r2, c5_r2
        # Quick SLSQP on best of (h_r, h_r2)
        if time_left() > 25 and c5_r2 < best_c5 + 0.002:
            h_r3 = slsqp_polish(h_r2, dx, target_sum, 100)
            c5_r3 = compute_c5(h_r3, dx)
            if c5_r3 < best_c5:
                best_h, best_c5 = h_r3, c5_r3

    # Phase 5: Final SLSQP + local search
    if time_left() > 20:
        h_f = slsqp_polish(best_h, dx, target_sum, 300)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    if time_left() > 15:
        dl_f = min(t_start + budget_s - 5, time.time() + 10)
        h_fs, c_fs = local_search_pairs(best_h, dx, target_sum, 3, dl_f)
        if c_fs < best_c5:
            best_h, best_c5 = h_fs, c_fs

    # Final verification
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
