Wait, but the smooth_grad function as I wrote it computes the gradient of SUM of w[k]*corr[k], not of (1/tau)*log(sum(exp(tau*corr[k]))).

The gradient of (1/tau)*log(sum(exp(tau*corr[k]))) is:
= (1/tau) * (1/sum(exp(tau*corr))) * sum(exp(tau*corr[k]) * tau * d(corr[k])/dh)
= SUM_k [exp(tau*corr[k]) / sum_j(exp(tau*corr[j]))] * d(corr[k])/dh
= sum_k w[k] * d(corr[k])/dh

So it IS the same as what I computed! The gradient of the smooth max is the weighted sum of individual gradients with softmax weights. Good.

import time
from scipy.optimize import minimize

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):

    if abs(float(h.sum()) - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(corr, h, N, tau):
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        L = k - (N - 1)
        if L >= 0:
            if L < N:
                grad[:N-L] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:N-L]
        else:
            l = -L
            if l < N:
                grad[l:] += wk * (1.0 - h[:N-l])
                grad[:N-l] -= wk * h[l:]
    return grad

                      options={'maxiter': 200, 'ftol': 1e-10})
