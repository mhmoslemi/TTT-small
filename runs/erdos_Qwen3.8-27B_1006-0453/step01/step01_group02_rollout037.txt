        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[:N-m] += 1.0 - h[m:]
            if m > 0:
                g[m:] -= h[:N-m]
    else:
        l = -m
        if l < N:
            g[l:] += 1.0 - h[:N-l]
            if l > 0:
                g[:N-l] -= h[l:]
    return g

def optimize_adaptive(h, N, target, dx, t0, budget_s, lr_init, tau_init, n_iters):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    tau = tau_init
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(corr.max() * dx)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        
        # Smooth-max weights
        shifted = corr - corr.max()
        w = np.exp(tau * shifted)
        sw = w.sum()
        if sw < 1e-30:
            break
        w /= sw
        
        # Gradient from significant lags
        thresh = 0.5 / N
        active_idx = np.where(w > thresh)[0]
        if len(active_idx) == 0:
            break
        grad = np.zeros(N)
        for idx in active_idx:
            lag = int(idx) - (N - 1)
            grad += w[idx] * grad_lag(h, lag, N)
        
        # Adam step
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * grad * grad
        m_hat = m_buf / (1 - b1**(it+1))
        v_hat = v_buf / (1 - b2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_sum(h, target)
        
        # Annealing
        if (it+1) % 200 == 0:
            lr *= 0.88
            tau *= 1.6
        if (it+1) % 500 == 0:
            # Small perturbation to escape plateaus
            h += 0.02 * np.random.randn(N)
            h = project_sum(h, target)
        if lr < 1e-10:
            break
    
    return best_h, best_c5

def optimize_top1(h, N, target, dx, t0, budget_s, n_iters=3000):
    """Focus entirely on the single worst lag."""
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.015
    
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(corr.max() * dx)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        
        # Focus on top 2 lags
        top2 = np.argsort(corr)[-2:]
        grad = np.zeros(N)
        for idx in top2:
            lag = int(idx) - (N - 1)
            grad += grad_lag(h, lag, N)
        grad /= len(top2)
        
        h = h - lr * grad
        h = project_sum(h, target)
        
        lr *= 0.9995
        if it % 400 == 0 and it > 0:
            h += 0.015 * np.random.randn(N)
            h = project_sum(h, target)
        if lr < 1e-9:
            break
    
    return best_h, best_c5

def binary_search(h_init, N, target, dx, t0, budget_s, rng, n_iters=8000):
    """Greedy/annealed binary local search."""
    h = (h_init > 0.5).astype(float)
    # Ensure exact count
    n_ones = int(h.sum())
    tgt = N // 2
    if n_ones > tgt:
        idx = np.flatnonzero(h == 1)
        rm = rng.choice(idx, n_ones - tgt, replace=False)
        h[rm] = 0
    elif n_ones < tgt:
        idx = np.flatnonzero(h == 0)
        add = rng.choice(idx, tgt - n_ones, replace=False)
        h[add] = 1
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 3:
            break
        if it % 30 == 0:
            c5v = compute_c5(h, dx)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        
        # Accept if not too much worse (simulated annealing)
        if it < 1000:
            c5_new = compute_c5(h, dx)
            if c5_new > best_c5 + 0.002:
                h[i], h[j] = h[j], h[i]
        else:
            c5_new = compute_c5(h, dx)
            if c5_new > compute_c5(h, dx):  # always accept or revert
                pass
    
    return best_h, best_c5

min max_k corr_k(h)
s.t. 0 ≤ h ≤ 1, sum(h) = N/2

This is a convex problem! (min of max of linear functions, with linear constraints). Wait, is corr_k(h) linear in h?

corr_k = ∑ h[i]·(1-h[i+k]) = ∑ h[i] - ∑ h[i]·h[i+k]

The first term ∑ h[i] is linear. The second term ∑ h[i]·h[i+k] is quadratic (bilinear). So the objective is NOT convex!

If h is binary (0 or 1), then h·(1-h) = 0, and corr_k = ∑ h[i]·(1-h[i+k]) is linear in h. But we're not restricted to binary.

So the problem is non-convex in general. However, if we restrict to binary h, it becomes a combinatorial optimization.
