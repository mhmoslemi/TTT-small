import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_from_weights(h, w, N):
    grad = np.zeros(N)
    c = N - 1
    active = np.where(w > 1e-12)[0]
    for idx in active:
        wi = w[idx]
        lag = int(idx) - c
        if 0 <= lag < N:
            grad[:N - lag] += wi * (1.0 - h[lag:])
            grad[lag:] -= wi * h[:N - lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += wi * (1.0 - h[:N - l])
            grad[:N - l] -= wi * h[l:]
    return grad

def optimize(h, N, dx, t0, t_end, rng, lr0=0.02, tau0=6.0, mu=0.0):
    target = N / 2.0
    best_c = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    mb = np.zeros(N)
    vb = np.zeros(N)
    tau = tau0
    for it in range(12000):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        s = tau * (corr - corr.max())
        w = np.exp(s)
        w /= w.sum()
        thr = max(1.0 / N, 0.001)
        w = np.where(w > thr, w, 0.0)
        w /= w.sum()
        g = grad_from_weights(h, w, N)
        if mu > 0:
            g += mu * (1.0 - 2.0 * h)
        mb = b1 * mb + (1.0 - b1) * g
        vb = b2 * vb + (1.0 - b2) * g * g
        mh = mb / (1.0 - b1 ** (it + 1))
        vh = vb / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if (it + 1) % 400 == 0:
            lr *= 0.78
            tau *= 3.0
            if lr > 0.003:
                h += rng.randn(N) * 0.012
                h = project_sum(h, target)
        if lr < 1e-7:

    return best_h, best_c

def binary_sa(h, N, dx, t0, t_end, rng):
    target = N / 2.0
    h = (h > 0.5).astype(float)
    c_cur = compute_c5(h, dx)
    best_c = c_cur
    best_h = h.copy()
    T = 0.003
    for step in range(50000):
        if time.time() > t_end:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h, dx)
        if c_new < best_c:

                h[i] = 1.0
            else:
                c_cur = c_new
        else:
            c_cur = c_new
        if step % 3000 == 0:
            T *= 0.9
    return best_h, best_c
