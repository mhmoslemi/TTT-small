import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fix_sum(h, n):
    return project_feasible(h, n * 0.5)

def power_mean_obj(h, n, p=20):
    """Power-mean approximation to max of correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    # Only consider positive correlations (negative can't be max)
    vals = np.maximum(corr, 0)
    return (np.sum(vals**p) / len(vals))**(1.0/p) * dx

def soft_max_obj(h, n, T=0.01):
    """Soft-max using exp."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_max = np.max(corr)
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def fourier_to_h(params, n, K):
    x = np.arange(n) * (2.0 / n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += params[2*(k-1)] * np.cos(k * np.pi * x)
        h += params[2*(k-1)+1] * np.sin(k * np.pi * x)
    return h

def fourier_obj(params, n, K, p):
    h = fourier_to_h(params, n, K)
    # Penalty for out of [0,1]
    viol = np.sum(np.maximum(0, h - 1)**2) + np.sum(np.maximum(0, -h)**2)
    # Clip and project
    h_proj = project_feasible(h, n * 0.5)
    corr = np.correlate(h_proj, 1.0 - h_proj, mode='full')
    dx = 2.0 / n
    vals = np.maximum(corr, 0)
    pm = (np.sum(vals**p) / len(vals))**(1.0/p) * dx
    return pm + 50.0 * viol

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n, dtype=float) 
    x_new = np.arange(new_n, dtype=float)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, new_n * 0.5)
    return h_new

def slsqp_step(h, n, T, max_iter=500, ftol=1e-14):
    target = n * 0.5
    def obj(x):
        return soft_max_obj(x, n, T)
    def con(x):
        return np.sum(x) - target
    res = opt.minimize(
        obj, h.copy(),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
        options={'maxiter': max_iter, 'ftol': ftol}
    )
    if res.success or res.fun < obj(h):
        return res.x
    return h

def subgradient_step(h, n, step_size=0.1, steps=50):
    """Subgradient method targeting the max lag."""
    target = n * 0.5
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)  # lag index in full correlation
        # Convert to actual lag offset
        # full correlation: index 0 corresponds to lag -(n-1), index n-1 to lag 0
        lag = k_star - (n - 1)
        # Gradient of corr[lag] w.r.t. h

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        m = 0.5*(lo+hi)
        if np.sum(np.clip(h-m, 0, 1)) > target: lo = m
        else: hi = m
    return np.clip(h-0.5*(lo+hi), 0, 1)

def upsample(h, N):
    return project(np.interp(np.arange(N, dtype=float), np.arange(len(h), dtype=float), h), N*0.5)

def sm_obj(x, n, T):
    c = np.correlate(x, 1.0-x, mode='full')
    mx = np.max(c)
    return (mx + T*np.log(np.sum(np.exp((c-mx)/T)))) * 2.0/n

def do_slsq(h, n, T, it):
    t = n*0.5; j = np.ones(n)
    r = opt.minimize(sm_obj, h.copy(), (n,T), 'SLSQP',
        bounds=[(0,1)]*n, constraints=[{'type':'eq','fun':lambda x:np.sum(x)-t,'jac':lambda x:j}],
        options={'maxiter':it,'ftol':1e-15})
    return project(r.x if (np.isfinite(r.x).all() and r.fun<sm_obj(h,n,T)) else h, t)

def subgrad(h, n, stps, lr0):
    t = n*0.5; h = h.copy(); lr = lr0
    for _ in range(stps):
        c = np.correlate(h, 1.0-h, mode='full')

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(h - m, 0.0, 1.0)) > target:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, N):
    x_old = np.arange(len(h), dtype=float)
    x_new = np.arange(N, dtype=float)
    return project(np.interp(x_new, x_old, h), N * 0.5)

def sm_obj(x, n, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(c)
    return (mx + T * np.log(np.sum(np.exp((c - mx) / T)))) * 2.0 / n

def do_slsq(h, n, T, it):
    t = n * 0.5
    jn = np.ones(n)
    r = opt.minimize(sm_obj, h.copy(), (n, T), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - t, 'jac': lambda x: jn}],
                     options={'maxiter': it, 'ftol': 1e-15})
    if np.isfinite(r.x).all() and r.fun < sm_obj(h, n, T):
        return project(r.x, t)
    return h

def _grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    for jj in range(n):
        if 0 <= jj - k < n:
            g[jj] += 1.0 - h[jj - k]
        if 0 <= jj + k < n:
            g[jj] -= h[jj + k]
    return g

def subgrad(h, n, stps, lr0, top_m=1):
    t = n * 0.5
    h = h.copy()
    lr = lr0
    for _ in range(stps):
        c = np.correlate(h, 1.0 - h, mode='full')
        if top_m <= 1:
            ks = int(np.argmax(c))
            active = [ks]
        else:
            thr = np.max(c) - 0.02 * np.max(c)
            active = [i for i in np.argsort(c)[-top_m:] if c[i] >= thr]
        g = np.zeros(n)
        for idx in active:
            g += _grad_for_lag(h, idx - (n - 1))
        g /= len(active)
        hn = project(h - lr * g, t)
        if compute_c5(hn) < compute_c5(h):
            h = hn
        else:
            lr *= 0.8
        if lr < 1e-7:
            break
    return h
