import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h -= room * ((-diff) / total)
    return np.clip(h, 0.0, 1.0)

def compute_contrib(h, j):
    """Compute linear contribution of h[j] to each corr element (excluding d=0)."""
    n = len(h)
    contrib = np.zeros(2 * n - 1)
    for i in range(2 * n - 1):
        d = i - (n - 1)
        if d == 0:
            continue
        a = max(0, -d)
        b = min(n - 1, n - 1 - d)
        # First factor: j in [a,b]
        if a <= j <= b:
            jd = j + d
            contrib[i] += (1.0 - h[jd])
        # Second factor: j-d in [a,b]
        jd = j - d
        if a <= jd <= b:
            contrib[i] += (-h[jd])
    return contrib

def coord_update(h, j, n_grid=400):
    """Update h[j] to minimize max corr, given all other coords fixed."""
    n = len(h)
    old_t = h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    contrib = compute_contrib(h, j)
    d0_base = corr[n-1] - old_t * (1.0 - old_t)
    
    t_grid = np.linspace(0.0, 1.0, n_grid)
    best_t = old_t
    best_obj = np.inf
    
    for t in t_grid:
        delta = t - old_t
        new_corr = corr + delta * contrib
        new_corr[n-1] = d0_base + t * (1.0 - t)
        obj = np.max(new_corr)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    h[j] = best_t
    return best_t

        d = i - (n - 1)
        if d == 0:
            continue
        a = max(0, -d)
        b = min(n - 1, n - 1 - d)
        if a <= j <= b:
            jd = j + d
            contrib[i] += (1.0 - h[jd])
        jd = j - d
        if a <= jd <= b:
            contrib[i] += (-h[jd])
    return contrib

def coord_update(h, j, n_grid=300):
    n = len(h)
    old_t = h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    contrib = compute_contrib(h, j)
    d0_base = corr[n-1] - old_t * (1.0 - old_t)
    
    t_grid = np.linspace(0.0, 1.0, n_grid)
    best_t = old_t
    best_obj = np.inf
    
    for t in t_grid:
        delta = t - old_t
        new_corr = corr + delta * contrib
        new_corr[n-1] = d0_base + t * (1.0 - t)
        obj = np.max(new_corr)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    h[j] = best_t
    return best_t

def swap_update(h, j, k, n_grid=200):
    n = len(h)
    hj, hk = h[j], h[k]
    corr = np.correlate(h, 1.0 - h, mode='full')
    contrib_j = compute_contrib(h, j)
    contrib_k = compute_contrib(h, k)
    d_comb = contrib_j - contrib_k  # change per unit delta
    
    d0_base = corr[n-1] - hj*(1-hj) - hk*(1-hk)
    
    delta_lo = max(-hj, hk-1.0)
    delta_hi = min(1.0-hj, hk)
    if delta_lo >= delta_hi:
        return
    
    delta_grid = np.linspace(delta_lo, delta_hi, n_grid)
    best_delta = 0.0
    best_obj = np.inf
    
    for delta in delta_grid:
        new_corr = corr + delta * d_comb
        new_tj = hj + delta
        new_tk = hk - delta
        new_corr[n-1] = d0_base + new_tj*(1-new_tj) + new_tk*(1-new_tk)
        obj = np.max(new_corr)
        if obj < best_obj:
            best_obj = obj
            best_delta = delta
    
    h[j] = hj + best_delta
    h[k] = hk - best_delta

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

    for t in t_grid:
        delta = t - old_t
        new_m = np.max(corr + delta * contrib)  # for i≠n-1
        new_d0 = d0_base + t * (1.0 - t)

def coord_update_cached(h, j, corr, n_grid=300):
    n = len(h)
    old_t = h[j]
    contrib = compute_contrib(h, j)
    d0_base = corr[n-1] - old_t * (1.0 - old_t)
    
    t_grid = np.linspace(0.0, 1.0, n_grid)
    deltas = t_grid - old_t  # shape (n_grid,)

    # But this is n_grid * (2n-1) memory. For n=120, n_grid=300: 300*239 = 71K. Fine.
    new_corrs = corr[None, :] + deltas[:, None] * contrib[None, :]
    # Override d=0 column
    d0_vals = d0_base + t_grid * (1.0 - t_grid)
    new_corrs[:, n-1] = d0_vals
    
    obj = new_corrs.max(axis=1)
    best_idx = np.argmin(obj)
    best_t = t_grid[best_idx]
    
    delta = best_t - old_t
    h[j] = best_t
    corr = corr + delta * contrib
    corr[n-1] = d0_base + best_t * (1.0 - best_t)
    return corr
