import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def active_gradient(h, shifts, weights):
    """Compute weighted sum of subgradients for given shifts."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    idx = np.arange(n)
    for s, w in zip(shifts, weights):
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            # (1-h[j+s]) for j < n-s; -h[j-s] for j >= s
            pos_mask = idx < (n - s)
            neg_mask = idx >= s
            shifted_pos = (1.0 - h[s:])*pos_mask  # h[j+s] for j in [0, n-1-s]
            shifted_neg = h[:n-s]*neg_mask  # h[j-s] for j in [s, n-1]
            grad += w * shifted_pos
            grad -= w * shifted_neg
        else:
            t = -s
            pos_mask = idx >= t
            neg_mask = idx < (n - t)
            shifted_pos = (1.0 - h[:n-t])*pos_mask  # h[j-t] for j in [t, n-1]
            shifted_neg = h[t:]*neg_mask  # h[j+t] for j in [0, n-1-t]
            grad += w * shifted_pos
            grad -= w * shifted_neg
    return grad * dx

def subgradient_optimize(h, t0, budget_frac, n_iters_max, top_k=5):
    """Subgradient descent with momentum."""
    n = len(h)
    cur = h.copy()
    best = cur.copy()
    best_c5 = compute_c5(cur)
    vel = np.zeros(n)
    momentum = 0.9
    step = 0.05
    no_improve = 0
    
    for it in range(n_iters_max):
        if time.time() - t0 > budget_frac:
            break
        
        corrs = compute_corrs(cur)
        # Find top-k shifts
        top_indices = np.argsort(corrs)[-top_k:]
        top_values = corrs[top_indices]
        max_val = corrs[np.argmax(corrs)]
        
        # Weights: exponential of (value - max) / temp
        temp = max(0.01, 0.05 * (1.0 - it / float(n_iters_max)))
        weights = np.exp((top_values - max_val) / temp)
        weights /= weights.sum()
        
        # Convert correlation indices to shifts
        shifts = top_indices - (n - 1)
        
        # Compute subgradient
        grad = active_gradient(cur, shifts, weights)
        
        # Nesterov momentum step
        vel = momentum * vel - step * grad
        new_h = cur + vel
        
        # Project: clip to [0,1] and normalize
        new_h = repair_h(new_h)
        
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = new_h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur = new_h
        
        # Adaptive step
        if no_improve > 30:
            step *= 0.95
            no_improve = 0
        if step < 0.001:
            break
    
    return best, best_c5

def smooth_c5(h, temp, penalty):

    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def repair_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        if s > target:
            pos = h > 1e-12
            cnt = pos.sum()
            if cnt == 0:
                break
            h[pos] -= (s - target) / cnt
        else:
            room = h < 1.0 - 1e-12
            cnt = room.sum()
            if cnt == 0:
                break
            h[room] += (target - s) / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def active_gradient(h, shifts, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    idx = np.arange(n)
    for s, w in zip(shifts, weights):
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            pos = np.zeros(n)
            pos[:n - s] = 1.0 - h[s:]
            neg = np.zeros(n)
            neg[s:] = h[:n - s]
            grad += w * pos - w * neg

            t = -s
            pos = np.zeros(n)
            pos[t:] = 1.0 - h[:n - t]
            neg = np.zeros(n)
            neg[:n - t] = h[t:]
            grad += w * pos - w * neg

    mom0 = 0.85

    no_improve = 0
    step_mult = 1.0

        frac = it / float(max_iters)
        top_k = max(top_k_start, int(top_k_start + (top_k_end - top_k_start) * frac))
        corrs = compute_corrs(cur)
        max_val = np.max(corrs)
        top_indices = np.argpartition(corrs, -top_k)[-top_k:]
        top_values = corrs[top_indices]
        temp = 0.03 * (1.0 - 0.5 * frac) + 0.005
        weights = np.exp((top_values - max_val) / temp)
        wsum = weights.sum()

def smooth_c5(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def smooth_c5_grad(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_vals = np.exp(shifted / temp)
    weights = exp_vals / exp_vals.sum()
    threshold = 1e-5
    active = weights > threshold
    if active.sum() == 0:
        active[np.argmax(weights)] = True
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = active_gradient(h, shifts[active], weights[active])
    dx = 2.0 / n
    grad += 2.0 * penalty * (np.sum(h) * dx - 1.0) * dx
    return grad
