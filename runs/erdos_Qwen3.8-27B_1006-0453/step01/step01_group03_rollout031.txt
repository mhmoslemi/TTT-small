def ineq_jac(x):
    J = np.zeros((2*n-1, n+1))
    J[:, n] = 1.0
    h = x[:n]
    # For lag k (0 to 2n-2), coordinate m (0 to n-1):
    # j1 = m + k - (n-1), valid if 0 <= j1 < n
    # j2 = m - k + (n-1), valid if 0 <= j2 < n
    # J[k,m] = -((1-h[j1])*[valid j1] - h[j2]*[valid j2])
    
    k_arr = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    m_arr = np.arange(n)[None, :]  # (1, n)
    j1 = m_arr + k_arr - (n-1)  # (2n-1, n)
    j2 = m_arr - k_arr + (n-1)
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    # (1-h[j1]) for valid entries
    h_safe = np.clip(j1, 0, n-1)
    h_safe2 = np.clip(j2, 0, n-1)
    term1 = (1.0 - h[h_safe]) * valid1
    term2 = h[h_safe2] * valid2
    J[:, :n] = -(term1 - term2)
    return J

def swap_search(h, dx, target_sum, n_iters=5000, time_limit=None):
    """Simulated annealing on binary-ish structure using swaps."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    temp = 0.005
    
    for it in range(n_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        # Pick a random pair to swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        h = project_sr(h, target_sum)
        c5_new = compute_c5(h, dx)
        delta = c5_new - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-8)):
            best_c5 = min(best_c5, c5_new)
            # Keep the swap
        else:
            # Undo
            h[i], h[j] = h[j], h[i]
            h = project_sr(h, target_sum)
        
        if it % 500 == 0:
            temp = max(temp * 0.95, 1e-5)
        
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def coord_descent(h, dx, target_sum, n_passes=35, time_limit=None):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for p in range(n_passes):
        if time_limit is not None and time.time() > time_limit:
            break
        if p < 10:
            half_w = 1.0
        elif p < 25:
            half_w = 0.4
        else:
            half_w = 0.15
        improved = False
        for m in range(n):
            lo = max(0.0, h[m] - half_w)
            hi = min(1.0, h[m] + half_w)
            grid = np.linspace(lo, hi, 40)
            old = h[m]
            best_t = old
            best_obj = compute_c5(h, dx)
            for t in grid:
                if abs(t - old) < 1e-8:
                    continue
                h[m] = t
                obj = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            if best_t != old:
                h[m] = best_t
                h = project_sr(h, target_sum)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_solve(h0, dx, target_sum, maxiter=120):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.003]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()
