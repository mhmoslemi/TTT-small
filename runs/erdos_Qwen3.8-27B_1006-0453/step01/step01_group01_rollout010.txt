def compute_grad_at_max(h):
    n = len(h)
    corr = np.correlate(h, 1-h, 'full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    # Part 1: (1-h[j-lag]) for 0 <= j-lag < n
    lo1 = max(0, lag)
    hi1 = min(n, n + lag)
    # j-lag in [0, n-1] means j in [lag, lag+n-1], intersect with [0, n-1]
    # If lag >= 0: j in [lag, n-1], j-lag in [0, n-1-lag]
    # If lag < 0: j in [0, n-1+lag], j-lag in [0-lag, n-1]
    if lag >= 0:
        j_range = range(lag, n)
        i_range = range(0, n - lag)
        for j, i in zip(j_range, i_range):
            grad[j] += (1 - h[i])
    else:
        j_range = range(0, n + lag)
        i_range = range(-lag, n)
        for j, i in zip(j_range, i_range):
            grad[j] += (1 - h[i])
    
    # Part 2: -h[j+lag] for 0 <= j+lag < n
    if lag >= 0:
        j_range = range(0, n - lag)
        i_range = range(lag, n)
        for j, i in zip(j_range, i_range):
            grad[j] -= h[i]
    else:
        j_range = range(-lag, n)
        i_range = range(0, n + lag)
        for j, i in zip(j_range, i_range):
            grad[j] -= h[i]
    
    return grad

def compute_grad_at_max(h):
    n = len(h)
    corr = np.correlate(h, 1-h, 'full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        grad[:n+lag] += (1 - h[-lag:])
        grad[-lag:] -= h[:n+lag]
    return grad

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        g[:n + lag] += (1.0 - h[-lag:])
        g[-lag:] -= h[:n + lag]
    return g

def grad_topk(h, k):
    n = len(h)
    corr = compute_corr(h)
    if k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -k)[-k:]
    g = np.zeros(n)
    for m in idx:
        g += grad_lag(h, m - (n - 1))
    return g / float(k)

def project_feasible(h, n):
