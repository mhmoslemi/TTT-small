def project_constraints(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet sum constraint
    diff = n / 2.0 - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)

    diff = n / 2.0 - np.sum(h)
    h += diff / n
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_set(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < 2*n - 1:
        idx = np.argsort(corr)[-top_k:]
        w = np.zeros(2*n - 1)
        sv = corr[idx]
        w[idx] = np.exp((sv - sv.max()) / beta)
        w /= w.sum()
    else:
        w = np.exp((corr - corr.max()) / beta)
        w /= w.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            grad[:n-s] += wm * (1.0 - h[s:])
            grad[s:] -= wm * h[:n-s]
        else:
            ns = -s
            grad[ns:] += wm * (1.0 - h[:n-ns])
            grad[:n-ns] -= wm * h[ns:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_set(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf == 0:
            h += diff / n
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = 2 * n - 1
    if top_k is not None and top_k < nc:
        idx = np.argsort(corr)[-top_k:]
        w = np.zeros(nc)
        sv = corr[idx]
        w[idx] = np.exp((sv - sv.max()) / beta)
        w /= w.sum()
    else:
        w = np.exp((corr - corr.max()) / beta)
        w /= w.sum()
    grad = np.zeros(n)
    for m in range(nc):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            grad[:n-s] += wm * (1.0 - h[s:])
            grad[s:] -= wm * h[:n-s]
        else:
            ns = -s
            grad[ns:] += wm * (1.0 - h[:n-ns])
            grad[:n-ns] -= wm * h[ns:]
    return grad * dx

def slsqp_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    return dx * (m + beta * np.log(np.sum(np.exp((corr - m) / beta))))

    return np.sum(x) - len(x) / 2.0
