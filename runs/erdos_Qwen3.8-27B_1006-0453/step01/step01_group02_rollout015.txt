import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_grad(h, pwr):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.maximum(corr, 0.0)
    w = np.power(c, pwr)
    ws = w.sum()
    if ws < 1e-30:
        idx = int(np.argmax(corr))
        w = np.zeros_like(corr)
        w[idx] = 1.0
    else:
        w /= ws
    grad = np.zeros(N)
    Nm1 = N - 1
    active = np.where(w > 1e-10)[0]
    for idx in active:
        lag = int(idx) - Nm1
        wi = w[idx]
        if lag >= 0:
            if lag < N:
                grad[lag:] += wi * (1.0 - h[:N - lag])
                grad[:N - lag] -= wi * h[lag:]
        else:
            l = -lag
            if l < N:
                grad[:N - l] += wi * (1.0 - h[l:])
                grad[l:] -= wi * h[:N - l]
    return float(np.max(corr)), grad
