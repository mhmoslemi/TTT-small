```python
def coord_step(h, j, n, corr):
    """Compute optimal h[j] given corr for all lags."""
    # For each lag s, R(s) = a_s * h[j] + b_s where a_s and b_s depend on neighbors
    # a_s = (1-h[j-s])*[0≤j-s<n] - h[j+s]*[0≤j+s<n]
    # b_s = R(s) - h[j]*a_s
    # We want to minimize max_s (a_s * v + b_s) over v ∈ [0,1]
    
    n_corr = 2*n - 1
    a = np.zeros(n_corr)
    b = np.zeros(n_corr)
    center = n - 1
    for idx in range(n_corr):
        s = idx - center
        a_s = 0.0
        if 0 <= j - s < n:
            a_s += (1.0 - h[j - s])
        if 0 <= j + s < n:
            a_s -= h[j + s]
        a[idx] = a_s
        b[idx] = corr[idx] - h[j] * a_s
    
    # Minimize max(a_s * v + b_s) for v in [0,1]
    # The maximum of linear functions: find the v where the max is minimized
    # This is a 1D convex minimization problem
    # The minimum is at a breakpoint or at v=0 or v=1
    # Breakpoints: where a_i * v + b_i = a_j * v + b_j, i.e., v = (b_j - b_i)/(a_i - a_j)
    
    # Simple approach: ternary search or check critical points
    # For efficiency, just do a golden section search on [0,1]
    
    def f(v):
        return np.max(a * v + b)
    
    # Golden section search
    golden = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - golden * (hi - lo)
    d = lo + golden * (hi - lo)
    for _ in range(50):
        if f(c) < f(d):
            hi = d
        else:
            lo = c
        c = hi - golden * (hi - lo)
        d = lo + golden * (hi - lo)
    
    v_opt = (lo + hi) / 2
    return v_opt
```

Actually, for a given j, a[idx] depends on:

For a fixed j, as s varies:
- The first term (1-h[j-s]) is nonzero when 0≤j-s<n, i.e., -j < s < n-j, i.e., s ∈ (-j, n-j)
- The second term -h[j+s] is nonzero when 0≤j+s<n, i.e., -j < s < n-j, i.e., s ∈ (-j, n-j)

Both are nonzero for the same range of s! So a_s = (1-h[j-s]) - h[j+s] for s ∈ (-j, n-j), and 0 otherwise.

In terms of the correlation index idx = s + (n-1):
idx ranges from n-1-j to n-1+j (for the first term) and from n-1-j to n-1+j (for the second term).

So a is nonzero only in a window of width min(2j+1, 2(n-1-j)+1, 2n-1) around the center.

This means for each coordinate j, I only need to evaluate a over a window. The golden section still needs to evaluate max over all 2n-1 values, but most of b = corr - h[j]*a is just corr (since a=0 outside the window).

So max(a*v + b) = max(max(a*v + b in window), max(corr outside window)).

```python
def project(h, n):
    """Project h onto {0≤x≤1, sum(x)=n/2}."""
    h = h.copy()
    target = n / 2.0
    # First clip
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        # Need to reduce sum by s - target
        # Reduce from those not at 0, starting with those closest to 1
        excess = s - target
        # Water-filling: reduce uniformly from those above a threshold
        # Simple: reduce proportionally from those in (0,1)
        mask = (h > 0) & (h < 1)
        if np.sum(mask) > 0:
            h[mask] -= excess / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        # If still over (some hit 0), do again
        # Repeat a few times
        for _ in range(5):
            s = np.sum(h)
            if s <= target + 1e-14:
                break
            excess = s - target
            mask = h > 0
            if not np.any(mask):
                break
            h[mask] -= excess / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    else:
        # Need to increase sum by target - s
        deficit = target - s
        for _ in range(5):
            s = np.sum(h)
            if s >= target - 1e-14:
                break
            deficit = target - s
            mask = h < 1
            if not np.any(mask):
                break
            h[mask] += deficit / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
    g = np.zeros(n)

        if s > 0:
            g[s:] += (1.0 - h[:n-s])
        g[:n-s] -= h[s:]
    else:
        t = -s
        g[:n-t] += (1.0 - h[t:])
        g[t:] -= h[:n-t]
    return g

def fix_sum(h, n):

    nx = np.linspace(0.0, 1.0, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
```
