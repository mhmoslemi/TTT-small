```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, pen):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + pen * (integral - 1.0) ** 2

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -np.max(h) - 1.0, -np.min(h) + float(n)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_step(h, n_active=5, alpha=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top n_active indices
    top_idx = np.argsort(corr)[-n_active:]
    w = np.zeros(2 * n - 1)
    active_corr = corr[top_idx]
    w[top_idx] = active_corr / np.sum(active_corr)  # weight by correlation value
    # Compute subgradient
    sg = np.zeros(n)
    for m in top_idx:
        lag = n - 1 - m
        wm = w[m]
        if lag == 0:
            sg += wm * (1.0 - 2.0 * h)
        elif lag > 0:
            lhs = np.zeros(n)
            lhs[:n - lag] = 1.0 - h[lag:]
            rhs = np.zeros(n)
            rhs[lag:] = h[:n - lag]
            sg += wm * (lhs - rhs)
        else:
            a = -lag
            lhs = np.zeros(n)
            lhs[a:] = 1.0 - h[:n - a]
            rhs = np.zeros(n)
            rhs[:n - a] = h[a:]
            sg += wm * (lhs - rhs)
    h_new = h - alpha * sg
    return project_feasible(h_new)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def optimize_smooth(h_in, t0, budget_s, time_frac):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    end_time = t0 + budget_s * time_frac
    cur = h_in.copy()
    best_local = compute_c5(cur)
    
    schedules = [
        (0.05, 100.0, 300),
        (0.02, 500.0, 200),
        (0.005, 2000.0, 200),
        (0.001, 10000.0, 150),
    ]
    for temp, pen, maxit in schedules:
        if time.time() > end_time:
            break
        res = minimize(
            smooth_obj, cur, args=(temp, pen),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = project_feasible(nh)
        c5v = compute_c5(nh)
        if c5v < best_local:
            best_local = c5v
        cur = nh
    
    # Subgradient polishing
    sg_end = min(time.time() + 20, end_time)
    alpha = 0.02
    while time.time() < sg_end:
        cur = subgradient_step(cur, n_active=3, alpha=alpha)
        if compute_c5(cur) < best_local:
            best_local = compute_c5(cur)
        alpha *= 0.999
        if alpha < 0.001:
            break
    return cur, best_local

def coord_descent_polish(h, t0, budget_s, time_frac):
    n = len(h)
    end_time = t0 + budget_s * time_frac
    cur = h.copy()
    best = compute_c5(cur)
    while time.time() < end_time:
        improved = False
        for i in range(n):
            if time.time() > end_time:
                break
            old = cur[i]
            # Try a few values around old
            for trial in [0.0, 1.0, old * 0.5, old * 1.5, (old + 0.5) / 2]:
                trial = np.clip(trial, 0.0, 1.0)
                cur[i] = trial
                c5v = compute_c5(cur)
                if c5v < best:
                    best = c5v
                    improved = True
                    old = trial
            cur[i] = old
        if not improved:
            break
    return cur, best

    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    h_init = project_feasible(h_init)

    best_c5 = compute_c5(best_h)

    n0 = len(h_init)
    cur = h_init.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, pen):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + pen * (integral - 1.0) ** 2

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -np.max(h) - 1.0, -np.min(h) + float(n)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_step(h, n_active, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_active:]
    w = np.zeros(2 * n - 1)
    ac = corr[top_idx]
    w[top_idx] = ac / np.sum(ac)
    sg = np.zeros(n)
    for m in top_idx:
        lag = n - 1 - m
        wm = w[m]
        if lag == 0:
            sg += wm * (1.0 - 2.0 * h)
        elif lag > 0:
            lhs = np.zeros(n)
            lhs[:n - lag] = 1.0 - h[lag:]
            rhs = np.zeros(n)
            rhs[lag:] = h[:n - lag]
            sg += wm * (lhs - rhs)
        else:
            a = -lag
            lhs = np.zeros(n)
            lhs[a:] = 1.0 - h[:n - a]
            rhs = np.zeros(n)
            rhs[:n - a] = h[a:]
            sg += wm * (lhs - rhs)
    return project_feasible(h - alpha * sg)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def optimize_smooth(h_in, t0, budget_s, time_frac):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    end_time = t0 + budget_s * time_frac
    cur = h_in.copy()
    best_local = compute_c5(cur)
    schedules = [
        (0.10, 50.0, 200),
        (0.03, 200.0, 200),
        (0.008, 1000.0, 200),
        (0.002, 5000.0, 200),
        (0.0005, 20000.0, 150),
    ]
    for temp, pen, maxit in schedules:
        if time.time() > end_time:
            break
        res = minimize(
            smooth_obj, cur, args=(temp, pen),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = project_feasible(nh)
        c5v = compute_c5(nh)
        if c5v < best_local:
            best_local = c5v
        cur = nh
    # Subgradient polishing
    sg_end = min(time.time() + 15, end_time)
    alpha = 0.03
    for _ in range(500):
        if time.time() > sg_end:
            break
        cur = subgradient_step(cur, 3, alpha)
        alpha *= 0.995
        if alpha < 0.0005:
            break
    c5v = compute_c5(cur)
    if c5v < best_local:
        best_local = c5v
    return cur, best_local

def coord_descent_polish(h, t0, budget_s, time_frac):
    n = len(h)
    end_time = t0 + budget_s * time_frac
    cur = h.copy()
    best = compute_c5(cur)
    sweeps = 0
    while time.time() < end_time and sweeps < 10:
        sweeps += 1
        improved = False
        for i in range(0, n, max(1, n // 300)):
            if time.time() > end_time:
                break
            old = cur[i]
            candidates = [0.0, 1.0, old * 0.5, min(1.0, old + 0.2), max(0.0, old - 0.2)]
            for trial in candidates:
                trial = np.clip(trial, 0.0, 1.0)
                cur[i] = trial
                c5v = compute_c5(cur)
                if c5v < best - 1e-12:
                    best = c5v
                    improved = True
            if not improved:
                cur[i] = old
    return cur, best

def binary_snap(h):
    h2 = (h > 0.5).astype(float)
    h2 = project_feasible(h2)
    return h2

def soft_binary_snap(h):
    h2 = np.where(h > 0.65, 1.0, np.where(h < 0.35, 0.0, h))
    return project_feasible(h2)
```
