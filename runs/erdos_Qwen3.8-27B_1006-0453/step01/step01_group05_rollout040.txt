```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    # Final exact fix
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Find the value closest to 0.5 to adjust
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += (target - s)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)) / len(corr))

    """Project x onto {h: sum(h)=target, 0<=h<=1} using water-filling."""
    x = np.clip(x, 0.0, 1.0)
    # Bisect on lambda
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

    def obj(x):
        return soft_max_obj(x, n, temp)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
    return res.x

def active_set_step(h, n, step_size):
    """One step of projected gradient on the active (max) lag."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    
    # Compute gradient of corr[max_idx] w.r.t. h
    # np.correlate(h, 1-h, 'full')[k] = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i
    # Let's derive: for output index k (0 to 2n-2), shift = k - (n-1)
    # corr[k] = sum_i h[i] * (1 - h[i + shift]) where 0<=i<n and 0<=i+shift<n
    #         = sum_i h[i] - sum_i h[i]*h[i+shift]
    
    shift = max_idx - (n - 1)
    grad = np.zeros(n)
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nfree = np.sum(free)
        if nfree == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    # Exact final adjustment
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def soft_minus_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)) / len(corr))

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def slsqp_opt(h0, n, bounds, temp, maxiter):

                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
                   args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
    return res.x

def proj_feasible(x, target):
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        nfree = np.sum(free)
        if nfree == 0:
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def soft_max_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)) / len(corr))


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def slsqp_opt(h0, n, bounds, temp, maxiter):
    try:
        res = minimize(soft_max_obj, h0, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
                       args=(n, temp),
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return h0.copy()


def proj_feasible(x, target, n):
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = np.sum(free)
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h


def active_gradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        s2 = -s
        grad[:n - s2] += (1.0 - h[s2:])
        grad[s2:] += (-h[:n - s2])
    return grad * dx


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    h0 = repair_h(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    # Phase 1: SLSQP temperature annealing
    temps = [100, 50, 25, 12, 6, 3, 1.5, 0.8, 0.4, 0.2]
    for temp in temps:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 250:
            break
        avail = budget_s - elapsed - 250
        if avail < 8:
            break
        maxiter = min(400, max(60, int(avail * 0.35)))
        h_t = slsqp_opt(best_h, n, bounds, temp, maxiter)
        h_t = repair_h(h_t, n)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 2: Active-set projected gradient descent
    if budget_s - (time.time() - t_start) > 130:
        h_curr = best_h.copy()
        for lr in [0.1, 0.05, 0.025, 0.01, 0.005]:
            for it in range(600):
                if budget_s - (time.time() - t_start) < 130:
                    break
                g = active_gradient(h_curr, n)
                h_new = h_curr - lr * g
                h_new = proj_feasible(h_new, target, n)
                c5_new = compute_c5(h_new, n)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_curr = h_new.copy()
                else:
                    h_curr = 0.5 * h_curr + 0.5 * h_new

    # Phase 3: Random restarts with SLSQP
    n_restarts = 50
    for r in range(n_restarts):
        if budget_s - (time.time() - t_start) < 100:
            break
        scale = 0.25 * np.exp(-r / 18.0) + 0.02
        cand = best_h + np.random.uniform(-scale, scale, n)
        cand = repair_h(cand, n)
        for temp in [8, 3, 1]:
            if budget_s - (time.time() - t_start) < 100:
                break
            maxiter = min(200, max(40, int((budget_s - 100 - (time.time() - t_start)) * 0.25)))
            h_t = slsqp_opt(cand, n, bounds, temp, maxiter)
            h_t = repair_h(h_t, n)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t
                cand = h_t.copy()

    # Phase 4: Structured sinusoidal perturbations
    if budget_s - (time.time() - t_start) > 80:
        x_grid = np.arange(n) * 2.0 / n
        for freq in [1, 2, 3, 4, 5]:
            for amp in [0.15, 0.3, 0.5]:
                if budget_s - (time.time() - t_start) < 70:
                    break
                pat = amp * np.sin(2.0 * np.pi * freq * x_grid / 2.0)
                cand = best_h + pat
                cand = repair_h(cand, n)
                h_t = slsqp_opt(cand, n, bounds, 5, 150)
                h_t = repair_h(h_t, n)
                c5_t = compute_c5(h_t, n)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t

    # Phase 5: Coordinate descent fine-tuning
    if budget_s - (time.time() - t_start) > 40:
        for _ in range(400):
            if budget_s - (time.time() - t_start) < 40:
                break
            improved = False
            for j in range(n):
                old = best_h[j]
                for delta in [0.01, -0.01, 0.03, -0.03, 0.06, -0.06]:
                    trial = best_h.copy()
                    trial[j] = np.clip(old + delta, 0.0, 1.0)
                    trial = repair_h(trial, n)
                    c5t = compute_c5(trial, n)
                    if c5t < best_c5 - 1e-12:
                        best_c5 = c5t
                        best_h = trial
                        improved = True
                        break
            if not improved:
                break

    # Phase 6: Final SLSQP polish
    elapsed = time.time() - t_start
    if budget_s - elapsed > 15:
        maxiter = max(150, int((budget_s - 15 - elapsed) * 4))
        h_t = slsqp_opt(best_h, n, bounds, 0.3, maxiter)
        h_t = repair_h(h_t, n)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
```
