```python
def compute_grad(h, T):
    n = len(h)
    dx = 2.0/n
    # Compute all c_k
    corr = np.correlate(h, 1.0-h, mode='full')  # length 2n-1
    c = corr * dx  # length 2n-1
    c_max = np.max(c)
    # Weights
    shifted = c - c_max
    exp_vals = np.exp(shifted / T)
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient: ∂f/∂h[j] = dx * sum_k w_k * [(1-h[j-k]) - h[j+k]]
    # where k = idx - (n-1) for idx in [0, 2n-2]
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for idx in range(2*n-1):
            k = idx - (n - 1)
            term = 0.0
            if 0 <= j - k <= n - 1:
                term += (1 - h[j - k])
            if 0 <= j + k <= n - 1:
                term -= h[j + k]
            s += w[idx] * term
        grad[j] = dx * s
    return grad
```

Actually: c_k = dx * sum_j h[j]*(1-h[j+k])
If I change h[i] from old to new:
- In the sum, h[i] appears as h[j] when j=i: change = (new-old)*(1-h[i+k])
- h[i] appears as h[j+k] when j=i-k: change = h[i-k]*(0-(new-old)) = -h[i-k]*(new-old)

So Δc_k = dx * (new - old} * [(1-h[i+k])*I(0≤i+k<n) - h[i-k]*I(0≤i-k<n)]

This is linear in (new-old) for each k! So c_k as a function of t is:
c_k(t) = c_k_old + dx * (t - old) * [(1-h[i+k])*I(valid) - h[i-k]*I(valid)]

So c_k is linear in t (for k≠0) or... wait, for k=0:
c_0 = dx * sum_j h[j]*(1-h[j]) = dx * (sum h[j] - sum h[j]²)
This is quadratic in t when i is the variable: c_0(t) = c_0_old + dx*(t-old)*(1-2*h[i]) + dx*(t²-old²)
= c_0_old + dx*(t-old)*(1-2*old) + dx*(t-old)*(t+old)

OK so it's at most quadratic in t. The soft-max of linear/quadratic functions can be minimized with Newton's method in 1D.

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_obj(h, T):
    corr = compute_corr_vals(h)
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max
    shifted = corr - c_max
    return c_max + T * np.log(np.sum(np.exp(shifted / T)))

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_vals(h)
    c_max = np.max(corr)
    shifted = corr - c_max
    exp_vals = np.exp(shifted / T)
    w = exp_vals / np.sum(exp_vals)
    
    J = np.arange(n)[:, None]
    K = np.arange(2 * n - 1) - (n - 1)
    jmk = J - K
    jpk = J + K
    
    valid1 = (jmk >= 0) & (jmk < n)
    valid2 = (jpk >= 0) & (jpk < n)
    
    h1 = np.zeros(2 * n)
    h2 = np.zeros(2 * n)
    h1[:n] = 1.0 - h
    h2[:n] = h
    
    t1 = np.where(valid1, h1[jmk], 0.0)
    t2 = np.where(valid2, h2[jpk], 0.0)
    
    return dx * ((t1 - t2) @ w)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    # If still not matching, distribute to remaining
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Add to elements with most room
        diff = target - s
        if diff > 0:
            idx = np.argsort(1.0 - h)  # most room first
        else:
            idx = np.argsort(h)  # most to remove first
        rem = abs(diff)
        for i in idx:
            step = min(rem, abs((1.0 if diff > 0 else 0.0) - h[i]))
            if diff > 0:
                h[i] += step
            else:
                h[i] -= step
            rem -= step
            if rem < 1e-14:
                break
    return h

def l_bfgs_step(h, n, T, max_iter=300, maxfun=5000):
    res = opt.minimize(
        soft_max_obj, h.copy(),
        jac=lambda x: soft_max_grad(x, T),
        args=(T,),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': maxfun}
    )
    return res.x

def coord_desc(h, n, T, n_sweeps=3):
    """1D coordinate descent using golden section on each coordinate"""
    h = h.copy()
    for sweep in range(n_sweeps):
        for i in range(n):
            t0 = h[i]
            def f1d(t):
                h[i] = t
                return soft_max_obj(h, T)
            # Golden section in [max(0,t0-0.2), min(1,t0+0.2)]
            lo = max(0.0, t0 - 0.2)
            hi = min(1.0, t0 + 0.2)
            if hi - lo < 1e-10:
                continue
            res = opt.minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-8})
            h[i] = res.x if res.fun < f1d(t0) else t0
        h = fix_sum(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    h_best = np.array(initial_h_values, dtype=float).copy()
    n = len(h_best)
    c5_best = compute_c5(h_best)
    
    # Phase 1: L-BFGS-B with decreasing T
    T_schedule = [0.1, 0.05, 0.02, 0.008, 0.003, 0.001, 0.0005]
    for T in T_schedule:
        elapsed = time.time() - t_start
        if elapsed > budget_s - 180:
            break
        h_opt = l_bfgs_step(h_best.copy(), n, T, max_iter=200, maxfun=3000)
        h_cand = fix_sum(h_opt, n)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand
    
    # Phase 2: Coordinate descent refinement
    elapsed = time.time() - t_start
    if elapsed < budget_s - 300:
        for T in [0.01, 0.003, 0.001]:
            elapsed = time.time() - t_start
            if elapsed > budget_s - 150:
                break
            h_opt = coord_desc(h_best.copy(), n, T, n_sweeps=2)
            h_cand = fix_sum(h_opt, n)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                h_best = h_cand
                c5_best = c5_cand
    
    # Phase 3: Random perturbations + re-optimize
    n_perturb = 8
    for i in range(n_perturb):
        elapsed = time.time() - t_start
        if elapsed > budget_s - 120:
            break
        noise = np.random.randn(n) * 0.03
        h_pert = np.clip(h_best + noise, 0, 1)
        h_pert = fix_sum(h_pert, n)
        h_opt = l_bfgs_step(h_pert, n, 0.01, max_iter=150, maxfun=2000)
        h_cand = fix_sum(h_opt, n)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand
    
    # Phase 4: Final polish
    elapsed = time.time() - t_start
    if elapsed < budget_s - 60:
        for T in [0.001, 0.0003]:
            elapsed = time.time() - t_start
            if elapsed > budget_s - 30:
                break
            h_opt = l_bfgs_step(h_best.copy(), n, T, max_iter=300, maxfun=5000)
            h_cand = fix_sum(h_opt, n)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                h_best = h_cand
                c5_best = c5_cand
    
    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_obj(h, T):
    corr = compute_corr_vals(h)
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max
    shifted = corr - c_max
    return c_max + T * np.log(np.sum(np.exp(shifted / T)))

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_vals(h)
    c_max = np.max(corr)
    shifted = corr - c_max
    exp_vals = np.exp(shifted / T)
    w = exp_vals / np.sum(exp_vals)
    J = np.arange(n)[:, None]
    K = np.arange(2 * n - 1) - (n - 1)
    jmk = J - K
    jpk = J + K
    valid1 = (jmk >= 0) & (jmk < n)
    valid2 = (jpk >= 0) & (jpk < n)
    h1 = np.zeros(2 * n)
    h2 = np.zeros(2 * n)
    h1[:n] = 1.0 - h
    h2[:n] = h
    t1 = np.where(valid1, h1[jmk], 0.0)
    t2 = np.where(valid2, h2[jpk], 0.0)
    return dx * ((t1 - t2) @ w)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        diff = target - s
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        rem = abs(diff)
        for i in idx:
            step = min(rem, abs((1.0 if diff > 0 else 0.0) - h[i]))
            h[i] += step if diff > 0 else -step
            rem -= step
            if rem < 1e-14:
                break
    return h

def l_bfgs_step(h, n, T, max_iter=200, maxfun=3000):
    res = opt.minimize(
        soft_max_obj, h.copy(),
        jac=lambda x: soft_max_grad(x, T),
        args=(T,),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                 'maxfun': maxfun, 'maxls': 30}
    )
    return res.x

def coord_desc(h, n, T, n_sweeps=2):
    h = h.copy()
    for _ in range(n_sweeps):
        for i in range(n):
            t0 = h[i]
            lo = max(0.0, t0 - 0.15)
            hi = min(1.0, t0 + 0.15)
            if hi - lo < 1e-9:
                continue
            def f1d(t):
                h[i] = t
                return soft_max_obj(h, T)
            res = opt.minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-7})
            h[i] = res.x if res.fun < f1d(t0) else t0
        h = fix_sum(h, n)
    return h

def targeted_reduce(h, n, T, n_iters=20):
    for _ in range(n_iters):
        corr = compute_corr_vals(h)
        k_worst = int(np.argmax(corr))
        idx_worst = k_worst + (n - 1)
        w = np.ones(2 * n - 1) * 1e-4
        w[idx_worst] = 1.0
        w_top = np.sort(corr)[-5:]
        top_idx = np.argsort(corr)[-5:]
        w[top_idx] = 1.0
        w = w / np.sum(w)
        def obj_targeted(x):
            c = compute_corr_vals(x)
            return float(np.dot(w, c))
        h = l_bfgs_step(h, n, T, max_iter=80, maxfun=800)
        h = fix_sum(h, n)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n * 0.5
    s = np.sum(h_new)
    if s > 1e-15:
        h_new = h_new * (target / s)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n = len(h_best)

        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand
            n = len(h_cand)
            return True
        return False

    # Phase 1: L-BFGS-B with soft-max, multi-T schedule
    T_schedule = [0.08, 0.04, 0.02, 0.008, 0.003, 0.001, 0.0004]
    for T in T_schedule:

        h_opt = l_bfgs_step(h_best.copy(), n, T, max_iter=250, maxfun=4000)
        try_update(h_opt)

    # Phase 2: Coordinate descent
    if elapsed() < budget_s - 300:
        for T in [0.01, 0.004, 0.001]:
            if elapsed() > budget_s - 150:
                break
            h_opt = coord_desc(h_best.copy(), n, T, n_sweeps=2)
            try_update(h_opt)

    # Phase 3: Targeted worst-constraint reduction
    if elapsed() < budget_s - 200:
        for T in [0.02, 0.005]:
            if elapsed() > budget_s - 100:
                break
            h_opt = targeted_reduce(h_best.copy(), n, T, n_iters=10)
            try_update(h_opt)

    # Phase 4: Random perturbations + re-optimize
    n_perturb = 10
    for i in range(n_perturb):
        if elapsed() > budget_s - 120:
            break
        scale = 0.04 * (1.0 - i / float(n_perturb + 1))
        noise = np.random.randn(n) * scale
        h_pert = np.clip(h_best + noise, 0, 1)
        h_pert = fix_sum(h_pert, n)
        T_use = 0.02 if i < 5 else 0.005
        h_opt = l_bfgs_step(h_pert, n, T_use, max_iter=150, maxfun=2000)
        try_update(h_opt)

    # Phase 5: Try slightly different resolution
    if elapsed() < budget_s - 250:
        for n_try in [200, 350]:
            if elapsed() > budget_s - 120:
                break
            h_up = upsample(h_best, n_try)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best - 1e-5:
                h_best = h_up
                c5_best = c5_up
                n = n_try
            for T in [0.02, 0.005]:
                if elapsed() > budget_s - 80:
                    break
                h_opt = l_bfgs_step(h_best.copy(), n, T, max_iter=200, maxfun=2500)
                try_update(h_opt)

    # Phase 6: Final fine polish
    if elapsed() < budget_s - 50:
        for T in [0.001, 0.0003, 0.0001]:
            if elapsed() > budget_s - 20:
                break
            h_opt = l_bfgs_step(h_best.copy(), n, T, max_iter=300, maxfun=4000)
            try_update(h_opt)

    # Phase 7: Coordinate descent final pass
    if elapsed() < budget_s - 15:
        h_opt = coord_desc(h_best.copy(), n, 0.002, n_sweeps=1)
        try_update(h_opt)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
```
