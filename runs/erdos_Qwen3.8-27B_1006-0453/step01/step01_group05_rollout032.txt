import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h)) * (2.0 / n))

def _obj_lse(x, n, t):
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    if t < 1e-8:
        return m * dx
    s = np.clip((c - m) / t, -50, 0)
    return (t * np.log(np.sum(np.exp(s))) + m) * dx

def _obj_psum(x, n, p):
    dx = 2.0 / n
    c = np.clip(_corr(x), -200, 200)
    return np.sum(c ** p) ** (1.0 / p) * dx

def _obj_topk(x, n, k):
    dx = 2.0 / n
    c = _corr(x)
    return np.sum(np.partition(c, -k)[-k:]) * dx

def _sc(x, n):
    return np.sum(x) - n / 2.0

def _repair(h, n):
    tg = n / 2.0
    h = np.clip(h.astype(float).copy(), 0, 1)
    for _ in range(600):
        d = tg - h.sum()
        if abs(d) < 1e-14:
            break
        fr = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(fr.sum())
        if nf == 0:
            h = np.clip(h + d / len(h), 0, 1)
        else:
            h[fr] = np.clip(h[fr] + d / nf, 0, 1)
    return h

def _slsqp(x0, n, obj, args, mi=150):
    try:
        b = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sc, 'args': (n,)}]
        r = minimize(obj, x0, method='SLSQP', bounds=b, constraints=cons,
                     args=args, options={'maxiter': mi, 'ftol': 1e-15})
        return r.x
    except Exception:
        return x0.copy()

def _sa_binary(n, rng, n_iter, ts, bud, mg=50):
    """Simulated annealing on binary sequences (n must be even)"""
    h = (np.arange(n) % 2 == 0).astype(float)
    dc = _c5(h, n)
    best_h = h.copy()
    best_c = dc
    T = 0.8
    for i in range(n_iter):
        if time.time() - ts > bud - mg:
            break
        T = max(0.02, 0.8 * np.exp(-i / 8000.0))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        nc = _c5(h, n)
        if nc < dc or rng.random() < np.exp(-(nc - dc) / max(T, 1e-9)):
            dc = nc
        else:
            h[a], h[b] = 1.0, 0.0
        if dc < best_c:
            best_c = dc
            best_h = h.copy()
    return best_h, best_c

def _gauss_search(h, n, rng, n_iter, ts, bud, mg=50):
