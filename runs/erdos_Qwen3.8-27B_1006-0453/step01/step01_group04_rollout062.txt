def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h.copy()
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def repair_h_exact(h):

    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    cur = h_in.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()

Minimizing C₅ is equivalent to maximizing min_k A_k(h).

This is a **maximin** problem: max_h min_k A_k(h).

- ∫h = 1
- A_k(h) ≥ t for all k

Maximize f(h) = min_k A_k(h) = min_k (1 - c_k(h)) where c_k(h) = ∫h(x)(1-h(x+k))dx.

Wait, f(h) = min_k A_k(h) = min_k (1 - c_k(h)) = 1 - max_k c_k(h) = 1 - C₅(h).

        return h + rng.randn(n) * sig
    else:
        i1 = int(rng.randint(0, n))
        i2 = int(rng.randint(0, n))
        h2 = h.copy()
        val = (h2[i1] + h2[i2]) / 2.0
        h2[i1] = val
        h2[i2] = val
        return h2

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Polish at 360 (short)
    t_lim = t0 + budget_s * 0.05
    h360, c5_360 = optimize_lbfgsb(h_init, t0, t_lim)
    if c5_360 < best_c5:
        best_c5 = c5_360
        best_h = h360.copy()

    # Phase 2: Upscale to 720, main optimization
    h720 = interp_upscale(best_h, 2)
    h720 = repair_h(h720)
    t_lim = t0 + budget_s * 0.35
    h720, c5_720 = optimize_lbfgsb(h720, t0, t_lim)
    if c5_720 < best_c5:
        best_c5 = c5_720
        best_h = h720.copy()

    # Phase 3: Subgradient refinement at 720
    if time.time() - t0 < budget_s * 0.55:
        t_lim = t0 + budget_s * 0.6
        h_sub, c5_sub = optimize_subgrad(best_h, t0, t_lim, n_steps=5000)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 4: Restart loop at 720
    n_cur = len(best_h)
    t_budget_restart_start = time.time() - t0
    max_restarts = 6
    for ri in range(max_restarts):
        if time.time() - t0 > budget_s * 0.90:
            break
        cur = perturb_structured(best_h, rng)
        cur = repair_h(cur)
        alloc = budget_s * 0.05
        t_lim = time.time() + alloc
        h_opt, c5_opt = optimize_lbfgsb(cur, time.time() - alloc, t_lim)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # Quick subgrad polish on current best
        if time.time() - t0 < budget_s * 0.88:
            t_lim2 = time.time() + budget_s * 0.02
            h_sg, c5_sg = optimize_subgrad(best_h, t0 - budget_s * 0.02, t_lim2, n_steps=1000)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Phase 5: Try upscaling to 1440 if time permits
    if time.time() - t0 < budget_s * 0.80:
        h1440 = interp_upscale(best_h, 2)
        h1440 = repair_h(h1440)
        t_lim = t0 + budget_s * 0.88
        h1440, c5_1440 = optimize_lbfgsb(h1440, t0, t_lim)
        if c5_1440 < best_c5:
            best_c5 = c5_1440
            best_h = h1440.copy()
        if time.time() - t0 < budget_s * 0.93:
            t_lim2 = t0 + budget_s * 0.95
            h1440, c5_1440 = optimize_subgrad(best_h, t0, t_lim2, n_steps=2000)
            if c5_1440 < best_c5:
                best_c5 = c5_1440
                best_h = h1440.copy()

    # Final repair
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
