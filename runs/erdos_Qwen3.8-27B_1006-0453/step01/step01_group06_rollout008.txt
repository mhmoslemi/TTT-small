def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    # d corr[m] / d h[i] = (1 - h[i + (m-(n-1))]) - h[i - (m-(n-1))])
    #   with h[out of bounds] = 0
    
    # This is complex to vectorize efficiently. Let me use a simpler approach.
    # Actually for SLSQP, finite difference gradient is fine.
    return None  # Let SLSQP use finite differences

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_desc_step(h, penalty=100.0):
    """One full coordinate descent sweep."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    
    for i in range(n):
        old_val = h[i]
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Compute c_k for each correlation index
        k = np.arange(-(n-1), n)  # lags
        ip = i + k
        im = i - k
        valid_p = (ip >= 0) & (ip < n)
        valid_m = (im >= 0) & (im < n)
        h_ip = np.where(valid_p, 0.0, 0.0)
        h_im = np.where(valid_m, 0.0, 0.0)
        h_ip[valid_p] = h[ip[valid_p]]
        h_im[valid_m] = h[im[valid_m]]
        
        c = 1.0 - h_ip - h_im
        d = corr - c * old_val
        
        # S_partial = sum of h[j] for j != i
        S_partial = h.sum() - old_val
        
        # 1D optimization: min_{x in [0,1]} max_k(d_k + c_k*x)*dx + P*max(0,|S_partial+x-target|)
        # Use grid
        xs = np.linspace(0, 1, 200)
        base = np.max(d[None, :] + c[None, :] * xs[:, None], axis=1) * dx
        penalty_term = penalty * np.maximum(0, np.abs(S_partial + xs - target) - 1e-10)
        vals = base + penalty_term
        argmin = np.argmin(vals)
        best_x = xs[argmin]
        
        # Refine
        if argmin > 0 and argmin < len(xs) - 1:
            lo, hi = xs[argmin-1], xs[argmin+1]
            xs_f = np.linspace(lo, hi, 100)
            base_f = np.max(d[None, :] + c[None, :] * xs_f[:, None], axis=1) * dx
            pen_f = penalty * np.maximum(0, np.abs(S_partial + xs_f - target) - 1e-10)
            vals_f = base_f + pen_f
            best_x = xs_f[np.argmin(vals_f)]
        
        h[i] = best_x
    
    # Project to feasible
    h = project_feasible(h, n)
    return h

            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 6: Final polish
    n_fin = len(best_h)
    if elapsed() < budget_s - 5:
        hc = coord_desc_sweep(best_h, penalty=500.0, n_inner=20, rng=rng)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        if elapsed() < budget_s - 3:
            hc = slsqp_opt(best_h, n_fin, 2000, 400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
            hc = coord_desc_sweep(best_h, penalty=200.0, n_inner=10, rng=rng)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
