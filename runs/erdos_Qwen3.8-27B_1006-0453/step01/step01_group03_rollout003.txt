def project_sum(h, target_sum):

        d = target_sum - h.sum()

        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def coord_descent(h, target_sum, n_sweeps=100, deadline=None):
    n = len(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        h = project_sum(h, target_sum)
        for i in range(n):
            t = h[i]
            corr_cur = get_corr(h)
            D = np.zeros(2 * n - 1)
            C = corr_cur.copy()
            for k in range(-(n - 1), n):
                j = k + n - 1
                if 0 <= i + k < n:
                    D[j] += (1.0 - h[i + k])
                    C[j] -= t * (1.0 - h[i + k])
                if 0 <= i - k < n:
                    D[j] -= h[i - k]
                    C[j] -= h[i - k] * (1.0 - t)
            lo, hi = 0.0, 1.0

                vals = C + mid * D

                    hi = mid
                elif D[imax] < -1e-13:
                    lo = mid
                else:
                    break
            h[i] = (lo + hi) * 0.5
        h = project_sum(h, target_sum)
    return h

def slsqp_solve(h0, dx, target_sum, maxiter=500, ftol=1e-14):

        return x[n]

            return project_sum(np.clip(res.x[:n], 0, 1), target_sum)
    except Exception:
        pass
    return h0.copy()
