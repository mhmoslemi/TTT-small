    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint_fn(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            total = room[mask].sum()
            if total < 1e-14:
                break
            h[mask] += room[mask] * (diff / total)
        else:
            mask = h > 1e-12
            total = h[mask].sum()
            if total < 1e-14:
                break
            h[mask] += h[mask] * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
