This dampens the bottom (n_c - k) lags. So the top k lags get full weight. That's correct.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    diff = s - target
    h2 = np.clip(h - diff / len(h), 0.0, 1.0)
    if abs(float(h2.sum()) - target) < 1e-6:
        return h2
    lo, hi = -abs(diff) - 1.0, abs(diff) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += (1.0 - h[:N - m])
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize_phase(h, N, target, dx, t0, budget_s, tau, topk, lr, n_iter, rng):
    best = compute_c5(h, dx)
    best_h = h.copy()
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-7
    n_c = 2 * N - 1
    no_improve = 0

    for it in range(n_iter):
        if time.time() - t0 > budget_s - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v < best:
            best = c5v
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve == 400:
            h = h + rng.randn(N) * 0.015
            h = project_sum(h, target)
            m_buf *= 0.5
            v_buf *= 0.5
            no_improve = 0

        if (it + 1) % 8 == 7:
            argmax_idx = int(np.argmax(corr))
            lag = argmax_idx - (N - 1)
            grad = grad_lag(h, lag, N)
        else:
            sc = tau * corr
            sc -= sc.max()
            w = np.exp(sc)
            w /= w.sum()
            if topk < n_c:
                idx_sorted = np.argsort(corr)
                w[idx_sorted[:n_c - topk]] *= 0.03
                w /= w.sum()
            thresh = 0.3 / n_c
            active = np.where(w > thresh)[0]
            grad = np.zeros(N)
            for idx in active:
                lag = int(idx) - (N - 1)
                grad += w[idx] * grad_lag(h, lag, N)

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * grad * grad
        m_hat = m_buf / (1.0 - b1 ** (it + 1))
        v_hat = v_buf / (1.0 - b2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)

        if (it + 1) % 800 == 0:
            lr *= 0.86
            tau *= 1.35
        if lr < 1e-10:
            break

    return best_h, best
