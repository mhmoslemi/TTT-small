```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    obj = dx * (m + beta * np.log(s))
    penalty = lambda_sum * (np.sum(h) - n/2.0)**2
    return obj + penalty
```

With large sharpness (e.g., 20-50), the parametrization pushes h toward binary, which is physically sensible.

But wait, with L-BFGS-B and this parametrization, the derivative is:
dh/dt = s * h * (1-h) (where s is the sharpness parameter)

This can be very small for large s, making the optimization slow. I'll need to use adaptive step sizes or a moderate s.

Actually, there's a simpler parametrization: h = clip(t, 0, 1) but that's not differentiable at the boundaries.

Let me try another idea: use the **water-filling** approach.
- Start with h = 1/2 everywhere (satisfies sum = n/2)
- Iteratively push h values toward 0 or 1 based on the gradient of the max correlation
- The "push" is: h_new = clip(h - alpha * grad, 0, 1)
- Then redistribute to maintain sum = n/2: use a water-filling algorithm

The water-filling (projection onto {h: 0≤h≤1, sum h = n/2}):
1. Start with h' = h - alpha * g
2. Need to find λ such that sum(clip(h' - λ, 0, 1)) = n/2
3. This is a 1D bisection problem!

This is clean and efficient. Let me implement this.

```python
def project_onto_simplex_box(v, target_sum):
    """Project v onto {h: 0≤h≤1, sum h = target_sum} using bisection on lambda."""
    # Need sum(clip(v - lam, 0, 1)) = target_sum
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0.0, 1.0)
```

This is O(n) per bisection step, 100 steps, so O(100n) total. Very fast.

```python
def projected_subgradient(h, n, lr, n_iter, rng):

        # c[k] = sum_{i} h[i] * (1-h[i-d]) for valid i

        if d > 0:

        elif d < 0:
            d_abs = -d
            g[d_abs:] = g[d_abs:] - h[:n-d_abs]
            g[:n-d_abs] = g[:n-d_abs] - h[d_abs:]
        else:

        if new_max < cur_max:

        else:
            # Reduce learning rate
            lr *= 0.9
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def project_box_sum(v, target_sum):
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 1.0, old_n, endpoint=False)
    nx = np.linspace(0.0, 1.0, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    return project_box_sum(v, new_n / 2.0)

def grad_lag(h, lag):
    n = len(h)
    g = np.ones(n)
    if lag > 0:
        g[lag:] -= h[:n - lag]
        g[:n - lag] -= h[lag:]
    elif lag < 0:
        d = -lag
        g[d:] -= h[:n - d]
        g[:n - d] -= h[d:]
    else:
        g -= 2.0 * h
    return g

def projected_subgrad(h, n, max_iter, lr0, rng):
    h = h.copy()
    lr = lr0
    cur_max = np.max(compute_corr_full(h))
    best = cur_max
    for it in range(max_iter):
        corr = compute_corr_full(h)
        k_star = np.argmax(corr)
        d = k_star - n + 1
        g = grad_lag(h, d)
        h_new = project_box_sum(h - lr * g, n / 2.0)
        new_max = np.max(compute_corr_full(h_new))
        if new_max < cur_max - 1e-12:
            h = h_new
            cur_max = new_max
            lr = min(lr * 1.02, 0.08)
            if cur_max < best:
                best = cur_max
        else:
            lr *= 0.92
            if lr < 1e-6:
                lr = 0.001
    return h

def coord_descent_target(h, n, max_iter, rng):
    h = h.copy()
    cur = np.max(compute_corr_full(h))
    for it in range(max_iter):
        corr = compute_corr_full(h)
        k_star = np.argmax(corr)
        d = k_star - n + 1
        d_abs = abs(d)
        if d_abs == 0:
            i = rng.randint(n)
            j = rng.randint(n)
        else:
            lo = max(0, d_abs)
            hi = min(n - 1, n - 1 - d_abs)
            if lo > hi:
                i = rng.randint(n)
                j = rng.randint(n)
            else:
                i = rng.randint(lo, hi + 1)
                j = i + d_abs
        for _ in range(6):
            for sign in [1.0, -1.0]:
                delta = sign * rng.uniform(0.005, 0.06)
                h_tr = h.copy()
                h_tr[i] = min(1.0, max(0.0, h_tr[i] + delta))
                h_tr[j] = min(1.0, max(0.0, h_tr[j] - delta))
                if abs(np.sum(h_tr) - n / 2.0) > 1e-9:
                    continue
                c = np.max(compute_corr_full(h_tr))
                if c < cur - 1e-13:
                    h = h_tr
                    cur = c
                    break
    return h

def slsqp_phase(h, n, betas):
    best_h = h.copy()
    best_val = smooth_max_obj(h, n, betas[0])
    dx = 2.0 / n
    for beta in betas:
        def obj(x):
            return smooth_max_obj(x, n, beta)
        def eq_fun(x):
            return np.sum(x) - n / 2.0
        def eq_jac(x):
            return np.ones(n)
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 600, 'ftol': 1e-12})
            h_new = project_box_sum(res.x, n / 2.0)
            v = smooth_max_obj(h_new, n, beta)
            if v < best_val:
                best_val = v
                best_h = h_new
        except Exception:
            pass
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    def time_left():
        return budget_s - (time.time() - t0)

    resolutions = [x for x in [pn, pn * 2, pn * 3] if x <= 600]
    if pn not in resolutions:
        resolutions.insert(0, pn)

    for n in resolutions:
        if time_left() < 45:
            break
        h0 = upsample(parent, n) if n != pn else parent.copy()

        h_s = slsqp_phase(h0, n, [0.04, 0.02, 0.01, 0.005, 0.002])
        c5 = compute_c5(h_s)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_s.copy(), n

        if time_left() < 40:
            continue
        n_iter_sg = min(3000, max(500, int(time_left() * 3)))
        h_sg = projected_subgrad(h_s, n, n_iter_sg, 0.03, rng)
        c5 = compute_c5(h_sg)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_sg.copy(), n

        if time_left() < 25:
            continue
        n_cd = min(6000, max(1000, int(time_left() * 2)))
        h_cd = coord_descent_target(h_sg if 'h_sg' in dir() else h_s, n, n_cd, rng)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_cd.copy(), n

    if time_left() < 15:
        return best_h, best_c5, best_n

    h = best_h.copy()
    n = best_n
    for restart in range(5):
        if time_left() < 20:
            break
        h_p = project_box_sum(h + rng.randn(n) * 0.012, n / 2.0)
        h_r = projected_subgrad(h_p, n, 800, 0.02, rng)
        c5r = compute_c5(h_r)
        if c5r < best_c5:
            best_c5, best_h, best_n = c5r, h_r.copy(), n
            h = h_r
        h_cd2 = coord_descent_target(h_r, n, 3000, rng)
        c5r = compute_c5(h_cd2)
        if c5r < best_c5:
            best_c5, best_h, best_n = c5r, h_cd2.copy(), n
            h = h_cd2

    if time_left() > 10:
        n_final = min(420, best_n)
        h_up = upsample(best_h, n_final) if n_final != best_n else best_h.copy()
        h_f = projected_subgrad(h_up, n_final, min(2000, int(time_left() * 2)), 0.02, rng)
        h_f = coord_descent_target(h_f, n_final, min(4000, int(time_left() * 1.5)), rng)
        c5f = compute_c5(h_f)
        if c5f < best_c5:
            best_c5, best_h, best_n = c5f, h_f.copy(), n_final

    return best_h, best_c5, best_n
```
