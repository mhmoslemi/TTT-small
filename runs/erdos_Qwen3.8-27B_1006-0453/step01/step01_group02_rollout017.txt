    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):

    target = N / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    t = tau * corr

    w = exp_t / s

    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)
    diff = float(h.sum()) - target
    if abs(diff) < 1e-12:
        return h

        if s > target:

        else:
            hi = mid

def smooth_obj_grad(h, N, tau, M_pen):
