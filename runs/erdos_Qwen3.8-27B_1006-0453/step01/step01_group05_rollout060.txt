    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp((corr - m) * temp))) / temp) * dx


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / np.sum(free), 0.0, 1.0)
    return h


def fourier_h(params, n, K):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    osc = np.zeros(n)
    for k in range(1, K + 1):

        osc += params[2*(k-1)+1] * np.sin(k * np.pi * x)
    lo, hi = -3.0, 3.0

        mid = (lo + hi) * 0.5
        if np.sum(np.clip(mid + osc, 0.0, 1.0)) < n / 2.0:
            lo = mid
        else:
            hi = mid
    return np.clip((lo + hi) * 0.5 + osc, 0.0, 1.0)

        if time.time() - t0 > budget_s - 18:

        kick = best_h + np.random.uniform(-0.07, 0.07, n)
        kick = repair_h(kick, n)
        for temp in [60, 250]:
            if time.time() - t0 > budget_s - 12:
                break
            h_t = slsqp_polish(kick, n, temp, bounds, 130)
            c = compute_c5(h_t, n)
            if c < best_c5:
                best_c5, best_h = c, h_t

    # Final polish
    if time.time() - t0 < budget_s - 3:
        rem = budget_s - 3 - (time.time() - t0)
        h_t = slsqp_polish(best_h, n, 500, bounds, max(80, int(rem * 6)))
        c = compute_c5(h_t, n)
        if c < best_c5:
            best_c5, best_h = c, h_t

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
