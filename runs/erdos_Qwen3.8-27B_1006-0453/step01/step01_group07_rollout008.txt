def smooth_max_val(corr, beta):
    m = np.max(corr)
    s = np.logaddexp.reduce((corr - m) / beta)
    return m + beta * s

def obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    z = (corr - m) / beta
    # Use log-sum-exp for numerical stability
    z_max = np.max(z)
    log_s = z_max + np.log(np.sum(np.exp(z - z_max)))
    val = dx * (m + beta * log_s)
    
    # Weights for gradient
    w = np.exp(z - z_max)
    w /= np.sum(w)
    
    # Compute gradient
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if wd < 1e-14:
            continue

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def lse_corr(corr, beta):
    m = np.max(corr)
    z = (corr - m) / beta
    zm = np.max(z)
    log_s = zm + np.log(np.sum(np.exp(z - zm)))
    return m + beta * log_s

def weights_from_corr(corr, beta):
    m = np.max(corr)
    z = (corr - m) / beta
    zm = np.max(z)
    w = np.exp(z - zm)
    w /= np.sum(w)
    return w

def obj_val(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return dx * lse_corr(corr, beta)

def obj_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    val = dx * lse_corr(corr, beta)
    w = weights_from_corr(corr, beta)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if wd < 1e-14:
            continue
        i_lo = max(0, d)
        i_hi = min(n - 1, n - 1 + d)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wd * (1.0 - x[idx - d])
        i_lo2 = max(0, -d)
        i_hi2 = min(n - 1, n - 1 - d)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            grad[idx2] -= wd * x[idx2 + d]
    grad *= dx
    return val, grad
