def grad_overlap(h, n, k):
    """Gradient of overlap at shift k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    # First factor: h[j] * (1-h[j+offset])
    # j valid: 0 <= j < n (always true in our loop)
    # j+offset valid: 0 <= j+offset < n
    lo1 = max(0, -offset)
    hi1 = min(n, n - offset)
    for j in range(lo1, hi1):
        grad[j] += (1.0 - h[j + offset])
    # Second factor: -h[j-offset] * h[j]
    # j-offset valid: 0 <= j-offset < n => offset <= j < n+offset
    # j valid: 0 <= j < n
    lo2 = max(0, offset)
    hi2 = min(n, n + offset)
    for j in range(lo2, hi2):
        grad[j] -= h[j - offset]
    return grad

import numpy as np
from scipy.optimize import minimize, differential_evolution

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n, nmax=300):
    t = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(nmax):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        if m.sum() == 0:
            i = int(np.argmin(h)) if d > 0 else int(np.argmax(h))
            h[i] = float(np.clip(h[i] + d, 0, 1))
        else:
            h[m] = np.clip(h[m] + d / m.sum(), 0, 1)
    return h


def slsqp_obj(x, n, temp):
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = temp * corr
    m = s.max()
    return (np.log(np.sum(np.exp(s - m))) + m) / temp * (2.0 / n)


def slsqp_con(x, n):
    return float(np.sum(x) - n / 2.0)


def run_slsqp(x0, n, temp, maxiter):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}]
    r = minimize(slsqp_obj, x0, method='SLSQP', bounds=b, constraints=c,
                 args=(n, temp), options={'maxiter': maxiter, 'ftol': 1e-15})
    return r.x


def grad_overlap(h, n, k):
    off = k - (n - 1)
    g = np.zeros(n)
    a = max(0, -off)
    b = min(n, n - off)
    if b > a:
        g[a:b] += 1.0 - h[a + off:b + off]
    c = max(0, off)
    d = min(n, n + off)
    if d > c:
        g[c:d] -= h[c - off:d - off]
    return g


def subgrad_opt(h, n, t0, budget, margin, iters, a0, beta):
    h = h.copy()
    v = np.zeros(n)
    bh = h.copy()
    bc = compute_c5(h, n)
    a = a0
    for _ in range(iters):
        if time.time() - t0 > budget - margin:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        km = int(corr.argmax())
        g = grad_overlap(h, n, km)
        v = beta * v - a * g
        hn = repair_h(h + v, n, 50)
        c5 = compute_c5(hn, n)
        if c5 < bc:
            bc = c5
            bh = hn.copy()
            h = hn
            a = min(a * 1.02, 0.05)
        else:
            a = max(a * 0.9, 1e-6)
    return bh, bc


def free_obj(x, h_full, n, fs, temp):
    hh = h_full.copy()
    hh[fs] = x
    corr = np.correlate(hh, 1.0 - hh, mode='full')
    s = temp * corr
    m = s.max()
    return (np.log(np.sum(np.exp(s - m))) + m) / temp * (2.0 / n)


def free_con(x, tgt):
    return float(np.sum(x) - tgt)

            fs = mid[:min(8, len(mid))]

            dist = np.minimum(h[fs], 1.0 - h[fs])
            order = np.argsort(dist)
            fs = fs[order[:12]]
        non_fs = np.setdiff1d(all_idx, fs)

        if tgt < -0.01 or tgt > len(fs) + 0.01:
