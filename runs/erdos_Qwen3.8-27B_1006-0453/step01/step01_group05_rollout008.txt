    # Compute all lag correlations
    # c[k] for k in range(-(n-1), n)
    values = np.zeros(2*n - 1)
    for k in range(-(n-1), n):
        idx1 = np.arange(max(0, -k), min(n, n-k))
        idx2 = idx1 + k
        values[k + n - 1] = np.sum(x[idx1] * (1 - x[idx2])) * dx
