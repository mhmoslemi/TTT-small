    dx = 2.0 / n

    ev = np.exp(t * (corr - m))

def _sum_jac(h):
    return np.ones(len(h))

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ev = np.exp(t * (corr - m))
    return (m + np.log(np.sum(ev)) / t) * dx

def _grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ev = np.exp(t * (corr - m))
    w = ev / np.sum(ev)
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j] * dx
        if wj < 1e-18:
            continue
        s = j - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n + s, n)
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo1 < hi1:
            g[lo1:hi1] += wj * (1.0 - h[lo1 - s:hi1 - s])
        if lo2 < hi2:
            g[lo2:hi2] -= wj * h[lo2 + s:hi2 + s]
    return g

def _sum_con(h):
    return np.sum(h) - len(h) / 2.0

def _sum_jac(h):
    return np.ones(len(h))

def _proj(h, n):

            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = np.count_nonzero(m)
        if c == 0:
            break
        step = d / c
        h[m] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h, nd):
    n = len(h)
    xs = np.arange(n) * 2.0 / n
    xd = np.arange(nd) * 2.0 / nd
    return _proj(np.interp(xd, xs, h), nd)

def _slsqp(h0, t, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
    try:
        r = minimize(_obj, h0, args=(t,), jac=_grad, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        hc = _proj(np.clip(r.x, 0, 1), n)
        return hc, compute_c5(hc)
    except Exception:
        return h0, compute_c5(h0)

def _lbfgsb(h0, t, maxiter, gtol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_obj, h0, args=(t,), jac=_grad, method='L-BFGS-B',
                     bounds=bnds,
                     options={'maxiter': maxiter, 'gtol': gtol})
        hc = _proj(np.clip(r.x, 0, 1), n)
        return hc, compute_c5(hc)
    except Exception:
        return h0, compute_c5(h0)
