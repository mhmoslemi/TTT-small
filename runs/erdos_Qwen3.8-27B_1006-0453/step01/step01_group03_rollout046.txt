    corr = np.correlate(h, 1.0 - h, mode='full')

    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.array(h, dtype=float).copy()

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def slsqq_solve(h0, dx, target_sum, maxiter=400, ftol=1e-13):
    n = len(h0)
    def obj(x):
        return x[n]
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def ineq(x):
        return x[n] - dx * np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eq(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx + 0.005
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       jac=obj_grad, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_res = project_feasible(res.x[:n], target_sum)
            if compute_c5(h_res, dx) <= compute_c5(h0, dx) + 1e-9:
                return h_res
    except Exception:
        pass
    return h0.copy()

def slsqq_solve_relaxed(h0, dx, target_sum, maxiter=300, ftol=1e-12):
    """SLSQP with only top-10 correlation constraints to help convergence."""
    n = len(h0)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    top_k = np.argsort(corr)[-12:]
    def obj(x):
        return x[n]
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def ineq(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - dx * c[top_k]
    def eq(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx + 0.005
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       jac=obj_grad, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_res = project_feasible(res.x[:n], target_sum)
            return h_res
    except Exception:
        pass
    return h0.copy()

def gen_perturbation(best_h, n, rng, ptype):
    if ptype == 0:
        return best_h + 0.02 * rng.randn(n)
    elif ptype == 1:
        return best_h + 0.05 * rng.randn(n)
    elif ptype == 2:
        return best_h + 0.10 * rng.randn(n)
    elif ptype == 3:
        freq = rng.randint(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        return best_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif ptype == 4:
        h_try = best_h.copy()
        b1 = rng.integers(0, n - 6)
        b2 = rng.integers(0, n - 6)
        blen = rng.integers(2, 6)
        b1, b2 = int(b1), int(b2)
        tmp = h_try[b1:b1+blen].copy()
        h_try[b1:b1+blen] = h_try[b2:b2+blen]
        h_try[b2:b2+blen] = tmp
        return h_try
    elif ptype == 5:
        return 1.0 - best_h
    elif ptype == 6:
        return best_h[::-1].copy()
    elif ptype == 7:
        return best_h + 0.15 * rng.randn(n)
    else:
        h_try = best_h.copy()
        idx = rng.choice(n, size=rng.integers(2, 5), replace=False)
        h_try[idx] += 0.1 * rng.randn(len(idx))
        return h_try

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]))
    return h_new

    rng = np.random.RandomState(seed)

    except NameError:

    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h_raw, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    best_n = n

    # Phase 1: Initial SLSQP polish (full constraints)
    if time_left() > 20:
        h1 = slsqq_solve(h0, dx, target_sum, 500, 1e-14)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Relaxed SLSQP (top constraints only) for better local move
    if time_left() > 20:
        h1r = slsqq_solve_relaxed(best_h, dx, target_sum, 300, 1e-13)
        c1r = compute_c5(h1r, dx)
        if c1r < best_c5:
            best_h = h1r
            best_c5 = c1r
            h1f = slsqq_solve(h1r, dx, target_sum, 300, 1e-14)
            c1f = compute_c5(h1f, dx)
            if c1f < best_c5:
                best_h, best_c5 = h1f, c1f

    # Phase 3: Diverse restarts
    num_tries = 0
    while time_left() > 25:
        num_tries += 1
        ptype = num_tries % 9
        h_try = gen_perturbation(best_h, n, rng, ptype)
        h_try = project_feasible(h_try, target_sum)

        h_new = slsqq_solve(h_try, dx, target_sum, 250, 1e-12)
        c_new = compute_c5(h_new, dx)

        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
            # Polish the new best with tighter SLSQP
            if time_left() > 25:
                h_pol = slsqq_solve(h_new, dx, target_sum, 400, 1e-14)
                c_pol = compute_c5(h_pol, dx)
                if c_pol < best_c5:
                    best_h, best_c5 = h_pol, c_pol

    # Phase 4: Resolution sweep
    for n_try in [n + 2, n + 4, n + 6, n - 2, n - 4]:
        if time_left() < 50 or n_try < 10:
            continue
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        h_interp = project_feasible(h_interp, target_try)

        h_new = slsqq_solve(h_interp, dx_try, target_try, 350, 1e-13)
        c_new = compute_c5(h_new, dx_try)

        if c_new < best_c5:
            best_h, best_c5, best_n = h_new, c_new, n_try
            # Extra polish at new resolution
            if time_left() > 30:
                h_pol = slsqq_solve(h_new, dx_try, target_try, 300, 1e-14)
                c_pol = compute_c5(h_pol, dx_try)
                if c_pol < best_c5:
                    best_h, best_c5 = h_pol, c_pol
        elif time_left() > 120:
            # Try a few perturbations at this resolution
            for _ in range(3):
                if time_left() < 30:
                    break
                h_p = h_interp + 0.05 * rng.randn(n_try)
                h_p = project_feasible(h_p, target_try)
                h_new = slsqq_solve(h_p, dx_try, target_try, 200, 1e-12)
                c_new = compute_c5(h_new, dx_try)
                if c_new < best_c5:
                    best_h, best_c5, best_n = h_new, c_new, n_try
                    break

    # Phase 5: Final intensive polish
    if time_left() > 10:
        dx_best = 2.0 / best_n
        target_best = best_n / 2.0
        h_final = slsqq_solve(best_h, dx_best, target_best, 600, 1e-15)
        c_final = compute_c5(h_final, dx_best)
        if c_final < best_c5:
            best_h, best_c5 = h_final, c_final

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
