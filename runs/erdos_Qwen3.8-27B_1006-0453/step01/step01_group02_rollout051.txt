        hr2o, c5r2 = refine(hr2, Nb, tb, dxb, t0, budget_s - 5, rng, 1500, 0.4)
        if c5r2 < best_c5:
            best_c5, best_h = c5r2, hr2o.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
