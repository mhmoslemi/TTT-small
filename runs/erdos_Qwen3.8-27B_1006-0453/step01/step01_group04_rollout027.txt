import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm + penalty * (integral - 1.0)**2

def subgrad_corr(h, k_star):
    n = len(h)
    dx = 2.0 / n
    s = k_star - (n - 1)
    j = np.arange(n)
    valid1 = (j - s >= 0) & (j - s < n)
    valid2 = (j + s >= 0) & (j + s < n)
    term1 = np.zeros(n)
    term1[valid1] = 1.0 - h[j[valid1] - s]
    term2 = np.zeros(n)
    term2[valid2] = -h[j[valid2] + s]
    return (term1 + term2) * dx

            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

    target = n / 2
    h = project_h(np.clip(h, 0.0, 1.0), target)

    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    alpha = 1.0
    
    while time.time() < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr)
        c5_scaled = c5_val * (2.0/n)
        if c5_scaled < best_c5:
            best_c5 = c5_scaled
            best_h = h.copy()

            top_idx = np.argpartition(corr, -top_k)[-top_k:]

            top_idx = [np.argmax(corr)]
        
        grad = np.zeros(n)
        for k in top_idx:

        grad /= len(top_idx)

            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k in top_idx:
            grad += subgrad_corr(h, int(k))
        grad /= len(top_idx)

        v = mom * v + alpha * grad
        h_new = project_h(h - v, target)
        h = h_new
        alpha *= 0.9985
        alpha = max(alpha, 0.02)

    return best_h, best_c5

def smooth_optimize(h, maxiter=150, temp=0.02, penalty=200.0):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h, args=(temp, penalty),
                  method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-9})
    h_out = np.clip(res.x, 0.0, 1.0)
    h_out = project_h(h_out, n / 2.0)
    return h_out

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def random_perturb(h, sigma, rng):
    n = len(h)
    mode = rng.random()
    if mode < 0.6:
        noise = rng.standard_normal(n) * sigma
        h2 = np.clip(h + noise, 0.0, 1.0)
    elif mode < 0.8:
        k = rng.randint(-n // 4, n // 4)
        h2 = np.roll(h, k)
    else:
        f = rng.randint(1, 4)
        amps = rng.uniform(0.05, 0.2)
        h2 = np.clip(h + amps * np.sin(2.0 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28)), 0.0, 1.0)
    return project_h(h2, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    h_init = project_h(h_init, len(h_init) / 2.0)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth optimization at n=360
    if time.time() - t0 < budget_s * 0.10:
        h1 = smooth_optimize(h_init, maxiter=120, temp=0.015, penalty=200.0)
        c5v = compute_c5(h1)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h1.copy()

    # Phase 2: Subgradient at n=360
    cur_h = best_h.copy()
    t_lim2 = t0 + budget_s * 0.32
    if time.time() < t_lim2:
        h2, c5v = subgrad_optimize(cur_h, t_lim2, top_k=5, mom=0.9, alpha_init=0.4)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()
        cur_h = h2

    # Phase 3: Upscale to n=720
    if time.time() - t0 < budget_s * 0.52:
        h3 = interp_upscale(cur_h, 2)
        h3 = project_h(h3, len(h3) / 2.0)
        if time.time() - t0 < budget_s * 0.62:
            h3 = smooth_optimize(h3, maxiter=100, temp=0.008, penalty=100.0)
        t_lim3 = t0 + budget_s * 0.76
        if time.time() < t_lim3:
            h3, c5v = subgrad_optimize(h3, t_lim3, top_k=5, mom=0.88, alpha_init=0.3)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h3.copy()
        cur_h = h3

    # Phase 4: Random restarts
    cur_n = len(best_h)
    cur_h = best_h.copy()
    n_restarts = 4
    for ri in range(n_restarts):
        t_start = t0 + budget_s * (0.78 + ri * 0.035)
        t_end = t0 + budget_s * (0.815 + ri * 0.035)
        if time.time() > t_end:
            break
        sigma = 0.12 / (ri + 1)
        h_pert = random_perturb(cur_h, sigma, rng)
        t_lim = min(t_end, time.time() + budget_s * 0.03)
        h_pert, c5v = subgrad_optimize(h_pert, t_lim, top_k=5, mom=0.85, alpha_init=0.25)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_pert.copy()
        cur_h = h_pert

    # Phase 5: Final smooth polish
    if time.time() < t0 + budget_s * 0.97:
        n_f = len(best_h)
        h_f = smooth_optimize(best_h, maxiter=80, temp=0.003, penalty=500.0)
        c5v = compute_c5(h_f)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_f

    # Final projection and evaluation
    best_h = project_h(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
