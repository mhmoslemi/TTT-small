```python
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
```

```python
def compute_d_i(h, n, i):
    """Compute d_i = correlate(e_i, g) - correlate(h, e_i) for a single i."""
    g = 1.0 - h
    d = np.zeros(2*n-1)
    # First term: g[n-1-m+i] for i ≤ m ≤ i+n-1
    # g[n-1-m+i] = g_rev[m-i] where g_rev = g[::-1]
    d[i:i+n] += g[::-1]
    # Second term: h[m-n+1+i] for n-1-i ≤ m ≤ 2n-2-i
    d[n-1-i : 2*n-1-i] -= h
    return d
```

```python
    """Local search on binary sequence: try swapping 0↔1."""
    t0 = time.time()
    # h_b is binary: 0 or 1, with exactly n/2 ones
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    
    current_max = np.max(np.correlate(h_b.astype(float), 1-h_b.astype(float), mode='full'))
    
    while (time.time() - t0) < max_time:
        improved = False
        # Random order
        np.random.shuffle(ones)

            if (time.time() - t0) > max_time:
                break
            for j in zeros:
                # Try swapping h[i]=1 and h[j]=0
                h_b[i] = 0
                h_b[j] = 1
                new_max = np.max(np.correlate(h_b.astype(float), 1-h_b.astype(float), phase='full'))

                    ones = np.append(ones, j)
                    improved = True
                    break
                else:
                    h_b[i] = 1
                    h_b[j] = 0

            break
    return h_b, current_max
```
