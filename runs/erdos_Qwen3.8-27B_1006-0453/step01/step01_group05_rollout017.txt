import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def soft_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp((corr - m) * beta))) / beta) * dx


def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def make_subgrad(h, n, k_lag):
    grad = np.zeros(n)
    shift = k_lag - (n - 1)
    if shift >= 0:
        for i in range(n - shift):
            grad[i + shift] += (1.0 - h[i])
            grad[i] -= h[i + shift]
    else:
        s = -shift
        for i in range(n - s):
            grad[i] += (1.0 - h[i + s])
            grad[i + s] -= h[i]
    return grad


def subgrad_step(h, n, beta, step, n_lags=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_lags:]
    wts = np.exp(beta * (corr[top_idx] - corr[top_idx[-1]]))
    wts /= np.sum(wts)
    grad = np.zeros(n)
    for idx, w in zip(top_idx, wts):
        grad += w * make_subgrad(h, n, int(idx))
    h_new = h - step * grad
    return repair_h(h_new, n)


def fourier_to_h(coeffs, n):
    m = len(coeffs) // 2
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(m):
        kk = k + 1
        h += coeffs[2 * k] * np.cos(2 * np.pi * kk * i / n)
        h += coeffs[2 * k + 1] * np.sin(2 * np.pi * kk * i / n)
    return np.clip(h, 0.0, 1.0)


def h_to_fourier(h, n, m):
    i = np.arange(n)
    dc = np.mean(h) - 0.5
    coeffs = np.zeros(2 * m)
    for k in range(m):
        kk = k + 1
        coeffs[2 * k] = (2.0 / n) * np.sum((h - 0.5 - dc) * np.cos(2 * np.pi * kk * i / n))
        coeffs[2 * k + 1] = (2.0 / n) * np.sum((h - 0.5 - dc) * np.sin(2 * np.pi * kk * i / n))
    return coeffs, dc


def fourier_obj(coeffs, n, m, beta):
    h_tmp = fourier_to_h(coeffs, n)
    return soft_obj(h_tmp, n, beta)


def slsqp_run(x0, n, bounds, temp, maxiter):
    res = minimize(
        soft_obj, x0, method='SLSQP', bounds=bounds,
        args=(n, temp),
        constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
        options={'maxiter': maxiter, 'ftol': 1e-11}
    )
    return res.x
