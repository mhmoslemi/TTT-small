    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h.copy()
    h = h.copy()
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    diff2 = target - s2
    if abs(diff2) > 1e-13:
        mask = (h > 0) & (h < 1)
        nm = np.sum(mask)
