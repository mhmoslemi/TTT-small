def repair_h(h, n):

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):

        diff = target - s

            break
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def softmax_obj(x, n, temp):

    corr = np.correlate(x, 1.0 - x, mode='full')

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def sum_eq(x):
    return np.sum(x) - len(x) / 2.0


def softmax_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    t = temp * corr
    m = np.max(t)
    if m < -100:
        return 0.0
    t = t - m
    return (np.log(np.sum(np.exp(t))) / temp + m / temp) * dx


def lp_obj(x, n, p):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    if mc < 1e-15:
        return 0.0
    shifted = corr / mc
    val = mc * (np.sum(shifted ** p) ** (1.0 / p))
    return val * dx


def slsqp_solve(x0, n, obj_func, obj_args, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    res = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=obj_args,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

    return res.x
