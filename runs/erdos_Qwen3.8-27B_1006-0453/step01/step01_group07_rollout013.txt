def linearize_corr(h0, n):
    """Linearize correlations around h0. Returns A matrix and b vector such that
    corr_k ≈ A[k,:] @ h + b[k] for k=0..2n-2"""
    A = np.zeros((2*n-1, n))
    b = np.zeros(2*n-1)
    q0 = 1.0 - h0

    # Linearize: h[i]*q[j] ≈ h0[i]*q0[j] + q0[j]*(h[i]-h0[i]) - h0[i]*(h[j]-h0[j])

So: A[k, p] = (1-h0[p+s]) * 1_{0≤p+s<n} + (-h0[p-s]) * 1_{0≤p-s<n}

where s = k - n + 1.

This is clean! Let me implement:

```python
def build_A_b(h0, n):
    q0 = 1.0 - h0
    A = np.zeros((2*n-1, n))
    b = np.zeros(2*n-1)
    for k in range(2*n-1):
        s = k - n + 1
        # b[k] = sum_{valid i} h0[i]*q0[i+s]
        i_start = max(0, -s)
        i_end = min(n-1, n-1-s)
        ii = np.arange(i_start, i_end+1)
        b[k] = np.sum(h0[ii] * q0[ii+s])
        
        # A[k, p] = q0[p+s]*1_{0≤p+s<n} + (-h0[p-s])*1_{0≤p-s<n}
        p = np.arange(n)
        # First term: p+s in [0,n-1] => p in [-s, n-1-s]
        ps = p + s
        mask1 = (ps >= 0) & (ps < n)
        A[k, mask1] += q0[ps[mask1]]
        # Second term: p-s in [0,n-1] => p in [s, n-1+s]
        pm = p - s
        mask2 = (pm >= 0) & (pm < n)
        A[k, mask2] += -h0[pm[mask2]]
    return A, b
```

This is O(n * n) = O(n²) for the loop, each iteration is O(n). For n=280, that's 559 * 280 ≈ 157k operations. Fast.

    corr = np.correlate(h, 1.0-h, mode="full")
    m = np.max(corr)
    # Active set: top k correlations
    thresh = np.percentile(corr, 100 - 100.0*k_active/len(corr))
    active = corr >= thresh
    # Gradient of corr_k w.r.t. h
    # = A[k,:] as derived above
    # Subgradient of max = weighted average of active gradients
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        s = k - n + 1
        p = np.arange(n)
        ps = p + s
        mask1 = (ps >= 0) & (ps < n)
        grad[mask1] += (1.0 - h[ps[mask1]])
        pm = p - s
        mask2 = (pm >= 0) & (pm < n)
        grad[mask2] += -h[pm[mask2]]
    grad /= np.sum(active)
    return grad, m

    for n in resolutions:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h = upsample(parent, n) if n != pn else parent.copy()
        cur = compute_c5(h)

        # Phase 1: subgradient descent with Nesterov momentum
        n_sgd = min(3000, int(rem * 0.25 * 8))
        h, cur = proj_subgrad(h, n, max(200, n_sgd), rng)
        if cur < best_c5:
            best_c5, best_h, best_n = cur, h.copy(), n

        # Phase 2: LP re-linearization
        rem = budget_s - (time.time() - t0)
        n_lp = min(4, int(rem / 50))
        for _ in range(max(1, n_lp)):
            if time.time() - t0 > budget_s - 100:
                break
            h_lp = solve_lp_step(h, n, time_limit=20)
            if h_lp is not None:
                c_lp = compute_c5(h_lp)
                if c_lp < cur:
                    h, cur = h_lp, c_lp
                    h, cur = proj_subgrad(h, n, 500, rng)
                    if cur < best_c5:
                        best_c5, best_h, best_n = cur, h.copy(), n
                else:
                    # Try blending
                    for frac in [0.5, 0.3]:
                        h_blend = project_feasible(frac * h + (1 - frac) * h_lp, n)
                        c_b = compute_c5(h_blend)
                        if c_b < cur:
                            h, cur = h_blend, c_b
                            h, cur = proj_subgrad(h, n, 300, rng)
                            if cur < best_c5:
                                best_c5, best_h, best_n = cur, h.copy(), n
            # Re-refine
            h, cur = target_cd(h, n, min(3000, int((budget_s - (time.time()-t0)) * 3)), rng)
            if cur < best_c5:
                best_c5, best_h, best_n = cur, h.copy(), n

        # Phase 3: coordinate descent
        rem = budget_s - (time.time() - t0) - 40
        if rem > 10:
            n_cd = int(rem * 10)
            h, cur = coord_descent(h, n, min(80000, n_cd), rng)
            if cur < best_c5:
                best_c5, best_h, best_n = cur, h.copy(), n
            h, cur = target_cd(h, n, min(40000, n_cd // 2), rng)
            if cur < best_c5:
                best_c5, best_h, best_n = cur, h.copy(), n

    # Final refinement on best
    rem = budget_s - (time.time() - t0) - 3
    if rem > 5:
        best_h, best_c5 = coord_descent(best_h, best_n, int(min(50000, rem * 12)), rng)
        best_h, best_c5 = target_cd(best_h, best_n, int(min(30000, rem * 8)), rng)

    return best_h, best_c5, best_n
