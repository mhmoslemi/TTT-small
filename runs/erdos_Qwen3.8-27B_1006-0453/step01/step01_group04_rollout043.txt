import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def norm_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-10:
        return h.copy()
    h = h * (n * 0.5) / s
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, temp, n, dx):
    hh = np.clip(h, 0.0, 1.0)
    s = hh.sum()
    if s < 1e-10:
        return 1e10
    hh = hh * (n * 0.5) / s
    hh = np.clip(hh, 0.0, 1.0)
    corr = np.correlate(hh, 1.0 - hh, mode='full')
    cmax = corr.max()
    sh = np.clip((corr - cmax) / temp, -500, 0)
    return (cmax + temp * np.log(np.sum(np.exp(sh)))) * dx

def interp_up(h, f):
    n = len(h)
    m = n * f
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(xn, xo, h)

def lse_phase(h, n, temps, maxit, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    best = compute_c5(cur)
    best_h = cur.copy()
    for temp in temps:
        if time.time() > deadline:
            break
        r = minimize(lse_obj, cur, args=(temp, n, dx),
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-9})
        nh = norm_h(r.x, n)
        c = compute_c5(nh)
        if c < best:
            best = c
            best_h = nh.copy()
        cur = nh
    return cur, best_h, best
