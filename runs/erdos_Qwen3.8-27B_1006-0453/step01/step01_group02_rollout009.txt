    lr = lr_init
    for it in range(max_iter):
        if time.time() - t0 > budget_s - 12:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local:
            best_local = c5_val
            best_h = h.copy()
        idx_k = np.argsort(corr)[::-1][:top_k]
        lags = (idx_k - (N - 1)).astype(int)
        g = grad_lags(h, lags, N)
        step = lr
        accepted = False
        for _bt in range(25):
            h_new = h - step * g
            h_new = project_sum(h_new, target)
            c_new = compute_c5(h_new, dx)
            if c_new <= c5_val + 1e-12:
                h = h_new
                accepted = True
                break
            step *= 0.65
        if not accepted:
            h = project_sum(h - lr * 0.01 * g, target)
        if (it + 1) % 300 == 0:
            lr *= 0.82
            top_k = min(top_k + 1, 10)
        if step < 1e-12:
            break
    return best_h, best_local

def coord_descend(h, N, target, dx, t0, budget_s, rng):
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    for sweep in range(80):
        if time.time() - t0 > budget_s - 15:
            break
        improved = False
        order = rng.permutation(N)
        for i in order:
            old = h[i]
            for nv in (0.0, 1.0):
                if abs(nv - old) < 1e-8:
                    continue
                h[i] = nv
                h = project_sum(h, target)
                c = compute_c5(h, dx)
                if c < best_local - 1e-11:
                    best_local = c
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()
            h = project_sum(h, target)
        if not improved:
            break
    return best_h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    sizes = sorted(set([s for s in [n0, 2*n0, 3*n0, 200, 300, 400, 500, 600, 800, 950] if 80 <= s <= 1000]))

    for N in sizes:
        if time.time() - t0 > budget_s - 50:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)

        h_interp = project_sum(np.interp(xN, x0, h0), target)
        h_threshold = project_sum((np.interp(xN, x0, h0) > 0.5).astype(float), target)

        h_rand = np.zeros(N)
        h_rand[rng.choice(N, size=int(target), replace=False)] = 1.0

        h_block = np.zeros(N)
        bs = max(1, N // 16)
        for st in range(0, N, bs):
            if rng.rand() > 0.45:
                h_block[st:min(st + bs, N)] = 1.0
        h_block = project_sum(h_block, target)

        h_pert = h_interp + rng.randn(N) * 0.05
        h_pert = project_sum(h_pert, target)

        inits = [h_interp, h_threshold, h_rand, h_block, h_pert]

        for h_init in inits:
            if time.time() - t0 > budget_s - 35:
                break
            h_opt, c5_opt = optimize_bt(h_init.copy(), N, target, dx, t0, budget_s, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                if time.time() - t0 < budget_s - 40:
                    h_cd, c5_cd = coord_descend(best_h.copy(), N, target, dx, t0, budget_s, rng)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd.copy()

    # Final polish
    N_b = len(best_h)
    dx_b = 2.0 / N_b
    tgt_b = N_b / 2.0
    if time.time() - t0 < budget_s - 25:
        h_p, c_p = optimize_bt(best_h.copy(), N_b, tgt_b, dx_b, t0, budget_s, rng, lr_init=0.008, top_k=6)
        if c_p < best_c5:
            best_c5 = c_p
            best_h = h_p
    if time.time() - t0 < budget_s - 20:
        h_cd, c_cd = coord_descend(best_h.copy(), N_b, tgt_b, dx_b, t0, budget_s, rng)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
