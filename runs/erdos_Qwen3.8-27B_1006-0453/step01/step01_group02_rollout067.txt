```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_box_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Binary search for the right shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        shifted = np.clip(h - mid, 0.0, 1.0)
        if shifted.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_c5(h, dx, mu=0.01):
    """Smooth approximation of C5 = max_k corr[k]*dx."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    scaled = corr / mu
    scaled -= scaled.max()
    return mu * (scaled.max() + np.log(np.exp(scaled).sum()))
```

```python
                    l = -lag
                    if l < N and l > 0:
                        grad[:N - l] += wi * (1.0 - x[l:])
                        grad[l:] -= wi * x[:N - l]
            
            s = x.sum()
            grad += 2.0 * penalty * (s - target)
            
            return smax + penalty * (s - target) ** 2, grad
        
        res = scipy_minimize(fun_grad, h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'maxfun': 1000, 'ftol': 1e-12, 'gtol': 1e-10})
        h = np.clip(res.x, 0.0, 1.0)
        h = project_sum(h, target)
        mu *= 0.5
        penalty *= 3
    
    return h
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(x, dx, mu, target, pen):
    N = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    e = np.exp((corr - m) / mu)
    S = e.sum()
    val = m + mu * np.log(S) + pen * (x.sum() - target) ** 2
    w = e / S
    return val, w, corr

def smooth_max_grad(x, dx, mu, target, pen):
    N = len(x)
    val, w, corr = smooth_max_obj(x, dx, mu, target, pen)
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        wi = w[idx] * dx
        if wi < 1e-14:
            continue
        lag = idx - (N - 1)
        if lag >= 0:
            if lag < N:
                if lag == 0:
                    grad[:] += wi * (1.0 - 2.0 * x)
                else:
                    grad[lag:] += wi * (1.0 - x[:N - lag])
                    grad[:N - lag] -= wi * x[lag:]
        else:
            l = -lag
            if 0 < l < N:
                grad[:N - l] += wi * (1.0 - x[l:])
                grad[l:] -= wi * x[:N - l]
    s = x.sum()
    grad[:] += 2.0 * pen * (s - target)
    return val, grad

def optimize_lbfgs(h0, N, target, dx, t0, budget_s, mu_start=0.08, n_stages=5, max_iter=300):
    bounds = [(0.0, 1.0)] * N
    pen = 200.0
    mu = mu_start
    h = h0.copy()
    
    for stage in range(n_stages):
        if time.time() - t0 > budget_s - 25:
            break
        
        def f(x):
            return smooth_max_obj(x, dx, mu, target, pen)[0]
        
        def fg(x):
            return smooth_max_grad(x, dx, mu, target, pen)
        
        res = sp_min(fg, h, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                              'ftol': 1e-14, 'gtol': 1e-10, 'iprint': 0})
        h = np.clip(res.x, 0.0, 1.0)
        h = project_sum(h, target)
        mu *= 0.55
        pen *= 2.5
    
    return h

def local_search(h, N, target, dx, t0, budget_s, rng, n_iter=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 12:
            break
        
        j = rng.randint(N)
        old = h[j]
        
        # Try several candidate values
        candidates = np.clip(np.array([
            old + rng.normal(0, 0.08),
            1.0 - old,
            min(1.0, old + 0.15),
            max(0.0, old - 0.15),
            rng.random()
        ]), 0.0, 1.0)
        
        best_val = old
        best_c5_trial = best_c5
        
        for cv in candidates:
            if abs(cv - old) < 0.005:
                continue
            h[j] = cv
            h = project_sum(h, target)
            c5_t = compute_c5(h, dx)
            if c5_t < best_c5_trial:
                best_c5_trial = c5_t
                best_val = cv
            # Restore
            h[j] = old
            h = project_sum(h, target)
        
        if best_val != old:
            h[j] = best_val
            h = project_sum(h, target)
            best_c5 = best_c5_trial
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _smax_and_weights(x, dx, mu):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    e = np.exp((corr - m) / mu)
    S = e.sum()
    return m + mu * np.log(S), e / S

def _smooth_max_grad(x, dx, mu, target, pen):
    N = len(x)
    smax, w = _smax_and_weights(x, dx, mu)
    s = x.sum()
    val = smax + pen * (s - target) ** 2
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        wi = w[idx] * dx
        if wi < 1e-14:
            continue
        lag = idx - (N - 1)
        if lag >= 0:
            if lag < N:
                if lag == 0:
                    grad[:] += wi * (1.0 - 2.0 * x)
                else:
                    grad[lag:] += wi * (1.0 - x[:N - lag])
                    grad[:N - lag] -= wi * x[lag:]
        else:
            l = -lag
            if 0 < l < N:
                grad[:N - l] += wi * (1.0 - x[l:])
                grad[l:] -= wi * x[:N - l]
    grad[:] += 2.0 * pen * (s - target)
    return val, grad

def optimize_lbfgs(h0, N, target, dx, t0, budget_s, mu_start=0.12, n_stages=5, max_iter=250):
    bounds = [(0.0, 1.0)] * N
    pen = 300.0
    mu = mu_start
    h = h0.copy()
    for stage in range(n_stages):
        if time.time() - t0 > budget_s - 25:
            break
        def fg(x):
            return _smooth_max_grad(x, dx, mu, target, pen)
        res = sp_min(fg, h, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                              'ftol': 1e-15, 'gtol': 1e-11, 'iprint': 0})
        h = np.clip(res.x, 0.0, 1.0)
        h = project_sum(h, target)
        mu *= 0.5
        pen *= 3.0
    return h

def local_search(h, N, target, dx, t0, budget_s, rng, n_iter=4000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 12:
            break
        j = rng.randint(N)
        old = h[j]
        cands = np.clip(np.array([
            old + rng.normal(0, 0.06), 1.0 - old,
            min(1.0, old + 0.12), max(0.0, old - 0.12), rng.random()
        ]), 0.0, 1.0)
        best_val, best_c5t = old, best_c5
        for cv in cands:
            if abs(cv - old) < 0.004:
                continue
            h[j] = cv
            h = project_sum(h, target)
            c5t = compute_c5(h, dx)
            if c5t < best_c5t:
                best_c5t, best_val = c5t, cv
            h[j] = old
            h = project_sum(h, target)
        if best_val != old:
            h[j] = best_val
            h = project_sum(h, target)
            best_c5, best_h = best_c5t, h.copy()
    return best_h, best_c5

def greedy_sweep(h, N, target, dx, t0, budget_s, n_sweeps=8):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    order = np.arange(N)
    for sw in range(n_sweeps):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        for j in order:
            old = best_h[j]
            for tv in (0.0, 1.0, old + (1.0 - old) * 0.3, old * 0.5):
                if abs(tv - old) < 0.008:
                    continue
                h = best_h.copy()
                h[j] = tv
                h = project_sum(h, target)
                c5t = compute_c5(h, dx)
                if c5t < best_c5 - 1e-12:
                    best_c5, best_h = c5t, h
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
```
