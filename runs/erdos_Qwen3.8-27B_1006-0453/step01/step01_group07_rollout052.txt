import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_lse(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / beta
    log_s = np.log(np.sum(np.exp(np.clip(shifted, -60, 0))))
    return dx * (m + beta * log_s)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = (np.arange(old_n) + 0.5) / old_n
    nx = (np.arange(new_n) + 0.5) / new_n
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

def fix_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def make_constraint(n):
    def fun(x):
        return np.sum(x) - n / 2.0
    def jac(x):
        return np.ones(n)
    return {'type': 'eq', 'fun': fun, 'jac': jac}

def grad_worst(h, idx_max, n):
    k = idx_max - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    else:
        m = -k
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    grad -= np.mean(grad)
    return grad
