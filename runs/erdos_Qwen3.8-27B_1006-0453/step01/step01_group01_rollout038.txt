def obj_smooth_grad(h):
    n = _N
    t = _T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    sm = np.max(s)
    exp_s = np.exp(s - sm)
    p = exp_s / np.sum(exp_s)  # softmax weights, length 2n-1
    fval = (mx + (sm + np.log(np.sum(exp_s))) / t) * dx
    
    # Gradient
    # dcorr[j]/dh[i] = (1 - h[i+j-(n-1)]) - h[i-j+(n-1)] (valid indices)
    # grad[i] = dx/t * sum_j p[j] * dcorr[j]/dh[i]
    #         = dx/t * (1 - sum_j p[j]*h[i+j-(n-1)]*1_{0<=i+j-(n-1)<n} - sum_j p[j]*h[i-j+(n-1)]*1_{0<=i-j+(n-1)<n})
    
    grad = np.full(n, 1.0)  # the "1" term: dx/t * sum_j p[j] * 1 = dx/t * 1
    
    # First correction: sum_j p[j] * h[i+j-(n-1)] for valid
    # For fixed i, j ranges over [max(0, n-1-i), min(2n-2, 2(n-1)-i)]
    for i in range(n):
        j_min = max(0, n - 1 - i)
        j_max = min(2*n - 2, 2*(n-1) - i)
        aj = i + np.arange(j_min, j_max + 1) - (n - 1)
        grad[i] -= np.dot(p[j_min:j_max+1], h[aj])
    
    # Second correction: sum_j p[j] * h[i-j+(n-1)] for valid
    for i in range(n):
        j_min = max(0, i)
        j_max = min(2*n - 2, i + n - 1)
        bj = i - np.arange(j_min, j_max + 1) + (n - 1)
        grad[i] -= np.dot(p[j_min:j_max+1], h[bj])
    
    grad *= dx / t
