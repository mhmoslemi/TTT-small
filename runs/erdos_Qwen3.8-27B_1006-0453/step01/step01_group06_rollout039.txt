import numpy as np
from scipy.optimize import minimize
import time

_constraint_n = 60

def _sum_con(x):
    return x.sum() - _constraint_n / 2.0

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-13:
            h += room * (diff / t)
    else:
        t = h.sum()
        if t > 1e-13:
            h += h * (diff / t)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T):
    n = len(x)
    corr = compute_corr(x)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * 2.0 / n

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        M = abs(idx - (n - 1))
        if M == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros(n)
            g[M:] += 1.0 - h[:n - M]
            g[:n - M] -= h[M:]
        grad += w[idx] * g
    return grad * dx

def pair_improve(h, n, t0, deadline):
    h = h.copy()

        k = rng.randint(n)
        if j == k:

            continue

        for d in deltas:
            if abs(d) < 1e-8:
                continue

            trial[j] += d
            trial[k] -= d

            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-12:
                best_c5 = c5t
                h = trial
                break
    return h, best_c5
