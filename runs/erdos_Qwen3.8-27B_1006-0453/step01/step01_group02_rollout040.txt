        w = w_new

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_smooth(h, corr, tau, dx):
    N = len(h)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(2 * N - 1) / (2 * N - 1)
    else:
        w /= ws
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        d = k - (N - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, d)
        je = min(N, N + d)
        if js < je:
            idx = np.arange(js, je) - d
            grad[js:je] += wk * (1.0 - h[idx])
        js2 = max(0, -d)
        je2 = min(N, N - d)
        if js2 < je2:
            idx2 = np.arange(js2, je2) + d
            grad[js2:je2] -= wk * h[idx2]
    return grad * dx
