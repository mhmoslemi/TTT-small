        if m >= 0:

                grad[l:] += wt * (1.0 - h[:N-l])
                # Negative terms: j from 0 to N-1-l
                grad[:N-l] -= wt * h[l:]
    return grad

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def project_binary(h, target):
    idx = np.argsort(h)[::-1][:int(target)]
    out = np.zeros(len(h))
    out[idx] = 1.0
    return out


def soft_obj(x, N, target, tau, mu):
    h = x.reshape(N)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = tau * corr
    sc_max = sc.max()
    e = np.exp(sc - sc_max)
    return (sc_max + np.log(e.sum())) + mu * (h.sum() - target) ** 2


def soft_grad(x, N, target, tau, mu):
    h = x.reshape(N)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = tau * corr
    sc_max = sc.max()
    e = np.exp(sc - sc_max)
    w = e / e.sum()
    grad = np.zeros(N)
    for t in range(2 * N - 1):
        if w[t] < 1e-11:
            continue
        m = t - (N - 1)
        wt = w[t]
        if m == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif m > 0:
            grad[:N - m] += wt * (1.0 - h[m:])
            grad[m:] -= wt * h[:N - m]
        else:
            l = -m
            grad[l:] += wt * (1.0 - h[:N - l])
            grad[:N - l] -= wt * h[l:]
    grad += 2.0 * mu * (h.sum() - target)
    return grad


def subgrad_refine(h, N, target, dx, n_iters, t0, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 0.008
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        m = max_idx - (N - 1)
        if m == 0:
            g = (1.0 - 2.0 * h) * dx
        elif m > 0:
            g = np.zeros(N)
            g[:N - m] += (1.0 - h[m:]) * dx
            g[m:] -= h[:N - m] * dx
        else:
            l = -m
            g = np.zeros(N)
            g[l:] += (1.0 - h[:N - l]) * dx
            g[:N - l] -= h[l:] * dx
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            a = lr * frac
            h_try = project_sum(h - a * g, target)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
        if improved:
            lr = min(lr * 1.05, 0.04)
        else:
            lr *= 0.6
    return best_h, best_c5


def local_search(h, N, target, dx, t0, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(777)
    n_tries = 30000
    for it in range(n_tries):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j:
            continue
        h_try = h.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = project_sum(h_try, target)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
    return best_h, best_c5


def lbfads_run(h_init, N, target, t0, budget_s):
    h_curr = h_init.copy()
    bounds = [(0.0, 1.0)] * N
    tau_seq = [8.0, 25.0, 80.0, 250.0]
    for tau in tau_seq:
        if time.time() - t0 > budget_s - 30:
            break
        mu = 80.0
        try:
            res = minimize(soft_obj, h_curr, args=(N, target, tau, mu),
                           jac=soft_grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-12, 'gtol': 1e-8})
            h_curr = project_sum(res.x, target)
        except Exception:
            pass
    return h_curr


def gen_inits(N, target, h_parent, x_parent, rng):
    inits = []
    xN = np.linspace(0.0, 2.0, N, endpoint=False)
    h_interp = project_sum(np.interp(xN, x_parent, h_parent), target)
    inits.append(h_interp)
    inits.append(project_sum(h_interp + rng.randn(N) * 0.12, target))
    inits.append(project_sum(h_interp + rng.randn(N) * 0.25, target))
    h_bin = np.zeros(N)
    h_bin[rng.choice(N, size=int(target), replace=False)] = 1.0
    inits.append(h_bin)
    for period in [7, 11, 13, 19, 23]:
        if period >= N:
            continue
        pat = (rng.rand(period) > 0.5).astype(float)
        h_pat = np.tile(pat, N // period + 1)[:N]
        inits.append(project_sum(h_pat, target))
    if N % 2 == 0:
        half = N // 2
        h_block = np.zeros(N)
        h_block[:half] = 1.0
        inits.append(h_block)
    return inits


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    sizes = [n0]
    if 2 * n0 <= 500:
        sizes.append(2 * n0)
    sizes = sorted(set(sizes))

    for N in sizes:
        if time.time() - t0 > budget_s - 80:
            break
        dx = 2.0 / N
        target = N / 2.0
        inits = gen_inits(N, target, h0, x0, rng)

        for h_init in inits:
            if time.time() - t0 > budget_s - 40:
                break
            h_curr = lbfads_run(h_init, N, target, t0, budget_s)
            h_curr, c5_sub = subgrad_refine(h_curr, N, target, dx, 400, t0, budget_s)

            h_bin = project_binary(h_curr, target)
            c5_bin = compute_c5(h_bin, dx)
            if c5_bin < c5_sub:
                h_curr = h_bin
                c5_sub = c5_bin

            h_curr, c5_ls = local_search(h_curr, N, target, dx, t0, budget_s)

            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_curr.copy()

    N_best = len(best_h)
    dx_best = 2.0 / N_best
    if time.time() - t0 < budget_s - 40:
        best_h, best_c5 = subgrad_refine(best_h, N_best, N_best / 2.0, dx_best,
                                         600, t0, budget_s)
    if time.time() - t0 < budget_s - 15:
        best_h, best_c5 = local_search(best_h, N_best, N_best / 2.0, dx_best,
                                       t0, budget_s)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
