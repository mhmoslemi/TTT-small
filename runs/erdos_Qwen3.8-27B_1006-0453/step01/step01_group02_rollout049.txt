import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, lag, N):
    g = np.zeros(N)
    if 0 <= lag < N:
        n = N - lag
        g[lag:] += 1.0 - h[:n]
        g[:n] -= h[lag:]
    elif -N < lag < 0:
        n = N + lag
        g[:n] += 1.0 - h[-lag:]
        g[-lag:] -= h[:n]
    return g

def optimize_adam(h, N, target, dx, t0, budget, rng, lr=0.012, tau0=6.0, n_valid=8000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    mu = np.zeros(N)
    sq = np.zeros(N)
    b1, b2, eps = 0.88, 0.999, 1e-8
    tau = tau0
    it = 0
    while it < n_valid:
        if time.time() - t0 > budget - 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        scaled = tau * corr - tau * corr.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w /= wsum
        thr = 2.0 / (N * 3.0)
        active = np.where(w > thr)[0]
        grad = np.zeros(N)
        for idx in active:
            lag = int(idx) - (N - 1)
            grad += w[idx] * grad_lag(h, lag, N)
        it += 1
        mu = b1 * mu + (1.0 - b1) * grad
        sq = b2 * sq + (1.0 - b2) * (grad * grad)
        mu_hat = mu / (1.0 - b1 ** it)
        sq_hat = sq / (1.0 - b2 ** it)
        h = h - lr * mu_hat / (np.sqrt(sq_hat) + eps)
        h = project_sum(h, target)
        if it % 250 == 0:
            lr *= 0.94
            tau *= 1.6
        if lr < 1e-10:
            break
    return best_h, best_c5

    target = N / 2.0

    h = np.zeros(N)

    T = T_init
    decay = (T_end / T_init) ** (1.0 / max(n_steps, 1))

            z = rng.randint(N)
