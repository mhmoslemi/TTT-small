import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, tau):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    grad = np.zeros(N)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = int(idx) - (N - 1)
        wi = w[idx]
        if k >= 0:
            if k == 0:
                grad += wi * (1.0 - 2.0 * h)
            else:
                grad[k:] += wi * (1.0 - h[:N - k])
                grad[:N - k] -= wi * h[k:]
        else:
            l = -k
            if l < N:
                grad[l:] += wi * (1.0 - h[:N - l])
                grad[:N - l] -= wi * h[l:]
    return grad

def grad_single_lag(h, m):
    N = len(h)
    grad = np.zeros(N)
    if m >= 0:
        if m < N:
            grad[m:] += (1.0 - h[:N - m])
            grad[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            grad[:N - l] += (1.0 - h[l:])
            grad[l:] -= h[:N - l]
    return grad

def optimize_adam(h, N, target, dx, t0, budget, rng, tau_start=5.0, tau_end=100.0,
                  n_iter=6000, lr_init=0.012, stuck_patience=400):
    best_local = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    last_improve = 0
    reheat_count = 0

    for it in range(n_iter):
        if time.time() - t0 > budget - 8:
            break
        progress = min(it / max(n_iter - 1, 1), 1.0)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local - 1e-12:
            best_local = c5_val
            best_h = h.copy()
            last_improve = it

        grad = smooth_max_grad(h, tau)
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)

        if (it + 1) % 300 == 0:
            lr *= 0.93
            if lr < 1e-10:
                break

        if it - last_improve > stuck_patience and reheat_count < 4:
            reheat_count += 1
            lr = lr_init * 0.6
            m_buf *= 0.3
            v_buf *= 0.5
            last_improve = it
            h = best_h.copy() + rng.randn(N) * 0.01
            h = project_sum(h, target)
            r = it - last_improve

    return best_h, best_local

def polish_worst_lags(h, N, target, dx, t0, budget):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    for round in range(3):
        if time.time() - t0 > budget - 4:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_lags = np.argsort(corr)[-4:][::-1]
        for lag_idx in top_lags:
            if time.time() - t0 > budget - 3:
                break
            m = int(lag_idx) - (N - 1)
            lr = 0.003
            for it in range(1500):
                if time.time() - t0 > budget - 2:
                    break
                g = grad_single_lag(best_h, m)
                h_try = project_sum(best_h - lr * g, target)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try
                else:
                    lr *= 0.97
                    if lr < 1e-12:
                        break
    return best_h, best_c5

def walk_search(h, N, target, dx, t0, budget, rng, n_steps=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    step_size = 0.02
    for step in range(n_steps):
        if time.time() - t0 > budget - 5:
            break
        mask = rng.rand(N) < 0.15
        pert = (rng.rand(N) - 0.5) * 2.0 * step_size
        pert *= mask
        h_try = project_sum(best_h + pert, target)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            step_size = min(step_size * 1.01, 0.04)
        elif step % 50 == 0:
            step_size *= 0.95
    return best_h, best_c5

def scipy_phase(h, N, target, dx, t0, budget, tau=30.0):
    def obj_pen(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode='full')
        sc = tau * corr
        sc -= sc.max()
        val = float(np.sum(np.exp(sc)) / tau + sc.max() / tau)
        val += 50.0 * (x.sum() - target) ** 2
        return val

    def grad_pen(x):
        x = np.clip(x, 0.0, 1.0)
        g = smooth_max_grad(x, tau)
        g += 100.0 * (x.sum() - target)
        return g

    try:
        res = scipy_minimize(obj_pen, h.copy(), jac=grad_pen, method='L-BFGS-B',
                             bounds=[(0.0, 1.0)] * N,
                             options={'maxiter': 300, 'gtol': 1e-8, 'maxfun': 3000})
        h_opt = project_sum(np.clip(res.x, 0.0, 1.0), target)
        return h_opt, compute_c5(h_opt, dx)
    except Exception:
        return h.copy(), compute_c5(h, dx)
