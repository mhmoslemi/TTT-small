def project_simplex(h, target_sum):

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    vals = corr * dx
    m = vals.max()
    es = np.exp(beta * (vals - m))

    f = m + np.log(s) / beta

        jlo = ilo + lag

            hslice = slice(jlo_c - lag, jhi_c - lag + 1)
            grad[jlo_c:jhi_c+1] -= wi * h[hslice.start:hslice.stop]

def lbfgs_penalty(h0, dx, target_sum, beta, mu, n_iter=2000):

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_simplex(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    vals = corr * dx
    m = vals.max()
    es = np.exp(beta * (vals - m))
    s = es.sum()
    f = m + np.log(s) / beta
    w = es / s
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-14:

        lag = mi - (n - 1)
        ilo = max(0, -lag)
        ihi = min(n - 1, n - 1 - lag)
        if ilo > ihi:
            continue
        a = ilo + lag
        b = ihi + 1 + lag
        grad[ilo:ihi + 1] += wi * omh[a:b]
        jlo = ilo + lag
        jhi = ihi + lag
        jlo_c = max(jlo, 0)
        jhi_c = min(jhi, n - 1)
        if jlo_c <= jhi_c:
            ga = jlo_c - lag
            gb = jhi_c - lag + 1
            grad[jlo_c:jhi_c + 1] -= wi * h[ga:gb]
    return f, grad
