def compute_c5(h, dx):

    return float(np.max(corr) * dx)


def project(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(400):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:

        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h


class SlqProblem:
    def __init__(self, n, target_sum):
        self.n = n
        self.target_sum = target_sum

    def obj(self, x):
        return x[self.n]

    def ineq(self, x):
        corr = np.correlate(x[:self.n], 1.0 - x[:self.n], mode='full')
        return x[self.n] - corr

    def eq(self, x):
        return np.sum(x[:self.n]) - self.target_sum


def slsqp_minimize(h0, dx, target_sum, maxiter=300):
    n = len(h0)
    t0 = compute_c5(h0, dx) / dx + 0.005
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    prob = SlqProblem(n, target_sum)
    cons = [{'type': 'ineq', 'fun': prob.ineq}, {'type': 'eq', 'fun': prob.eq}]
    try:
        res = minimize(prob.obj, x0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = project(res.x[:n], target_sum)
            if compute_c5(h_new, dx) <= compute_c5(h0, dx) + 1e-10:
                return h_new
    except Exception:
        pass
    return h0.copy()

    elapsed = time.time() - t0
    for r in range(12):
        elapsed = time.time() - t0
        if budget_s - elapsed < 60:
            break
        h_b = np.zeros(n_best)
        pos = np.random.permutation(n_best)
        k_ones = int(target_best)
        h_b[pos[:k_ones]] = 1.0
        frac = target_best - k_ones
        if frac > 0.01:
            h_b[pos[k_ones]] = frac
        h_b = project(h_b, target_best)
        h_b = slsqp_minimize(h_b, dx_best, target_best, 100)
        c5_b = compute_c5(h_b, dx_best)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b

    # Final polish
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    elapsed = time.time() - t0
    if budget_s - elapsed > 30:
        h_f = slsqp_minimize(best_h, dx_final, n_final / 2.0, 200)
        c5_f = compute_c5(h_f, dx_final)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    best_h = project(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n_final)
    return best_h, best_c5, n_final
