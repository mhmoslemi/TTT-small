def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0, 1, old_n, endpoint=False)
    nx = np.linspace(0, 1, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

def coord_sweep(h, n_sweeps=5):
    """Coordinate descent: optimize one coordinate at a time (exact 1D minimax)."""
    n = len(h)
    for _ in range(n_sweeps):
        for i in range(n):
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
            A = c1 - c0
            B = c0
            # Minimize max_k (A[k]*t + B[k]) for t in [0,1]
            best_val = float(np.max(B))
            best_t = 0.0
            v1 = float(np.max(A + B))
            if v1 < best_val:
                best_val = v1
                best_t = 1.0
            # Candidate lines: top at t=0 and t=1
            idx0 = np.argsort(-B)[:20]
            idx1 = np.argsort(-(A + B))[:20]
            cands = list(set(list(idx0) + list(idx1)))
            cl = np.array(cands, dtype=np.float64)
            for ii in range(len(cands)):
                for jj in range(ii + 1, len(cands)):
                    a, b = cands[ii], cands[jj]
                    da = A[a] - A[b]
                    if abs(da) < 1e-15:
                        continue
                    t = (B[b] - B[a]) / (A[a] - A[b])
                    if t < -0.01 or t > 1.01:
                        continue
                    t = float(np.clip(t, 0.0, 1.0))
                    # Check only candidate lines for max
                    vals = A[cl] * t + B[cl]
                    val = float(np.max(vals))
                    # Also check the true max occasionally
                    if val < best_val - 1e-10:
                        true_val = float(np.max(A * t + B))
                        if true_val < best_val - 1e-12:
                            best_val = true_val
                            best_t = t
            h[i] = best_t
    return h

def binary_sa(n, rng, max_iter=8000, t0_temp=0.3, t1_temp=0.01):
    """Simulated annealing on binary sequences with n/2 ones."""
    m = n // 2
    h = np.zeros(n)
    h[rng.choice(n, m, replace=False)] = 1.0
    cur = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    best = cur
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iter):
        frac = it / max_iter
        temp = t0_temp * (t1_temp / t0_temp) ** frac
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
        new = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        if new < cur or rng.rand() < np.exp(-(new - cur) / max(temp, 1e-10)):
            cur = new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best

def obj_smooth(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def eq_fun(x, n):
    return np.sum(x) - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_c5 = compute_c5(parent)
    best_h = parent.copy()
    best_n = pn

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Binary SA at multiple resolutions
    for n in [100, 140, 160, 200, 280]:
        if remaining() < 80:
            break
        # Initialize from parent or random
        if n <= pn:
            # Downsample parent by stride
            stride = pn // n
            h_init = parent[::stride][:n]
            if len(h_init) < n:
                h_init = np.pad(h_init, (0, n - len(h_init)), mode='wrap')
        else:
            h_init = upsample(parent, n)
        # Binarize
        thresh = 0.5
        h_bin = (h_init >= thresh).astype(np.float64)
        # Fix count
        n_ones = int(np.sum(h_bin))
        if n_ones > n // 2:
            excess = np.where(h_bin == 1)[0]
            remove = rng.choice(excess, n_ones - n // 2, replace=False)
            h_bin[remove] = 0
        elif n_ones < n // 2:
            deficit = np.where(h_bin == 0)[0]
            add = rng.choice(deficit, n // 2 - n_ones, replace=False)
            h_bin[add] = 1
        # Also try a random binary init
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0

        for h_start in [h_bin, h_rand]:
            if remaining() < 15:
                break
            h_sa, _ = binary_sa(n, rng, max_iter=6000)
            # Continuous refinement via coordinate descent
            h_cont = h_sa.copy()
            h_cont = coord_sweep(h_cont, n_sweeps=10)
            h_cont = fix_sum(h_cont, n)
            c5 = compute_c5(h_cont)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cont.copy()
                best_n = n

    # Phase 2: SLSQP refinement at best resolution and nearby
    n_best = best_n
    for n_try in [n_best, n_best - 20, n_best + 20]:
        if n_try < 50 or remaining() < 40:
            continue
        if n_try != n_best:
            h_init = upsample(best_h, n_try)
        else:
            h_init = best_h.copy()
        # A few SLSQP runs with different betas
        for beta in [0.002, 0.0005]:
            if remaining() < 15:
                break
            bounds = list(zip(np.zeros(n_try), np.ones(n_try)))
            cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n_try,)}]
            try:
                res = minimize(obj_smooth, h_init, args=(n_try, beta),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-12})
                h_new = fix_sum(np.clip(res.x, 0, 1), n_try)
                # Coordinate descent polish
                h_new = coord_sweep(h_new, n_sweeps=8)
                h_new = fix_sum(h_new, n_try)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    best_n = n_try
            except Exception:
                pass

    # Phase 3: Aggressive coordinate descent on the best solution
    if remaining() > 10:
        h = best_h.copy()
        n = best_n
        # Multiple coordinate sweeps
        n_sweeps = min(30, int(remaining() / 5))
        h = coord_sweep(h, n_sweeps=max(1, n_sweeps))
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 4: Targeted local search - perturb top correlation lags
    if remaining() > 8:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        dx = 2.0 / n
        total_iters = min(20000, int((remaining() - 3) / 0.002))
        for it in range(max(1, total_iters)):
            if remaining() < 3:
                break
            if it % 200 == 0 and cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.002, 0.03) * rng.choice([-1.0, 1.0])
            hi = h[i] + d
            hj = h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            h[i] = hi
            h[j] = hj
            c = compute_c5(h)
            if c >= cur - 1e-14:
                h[i] -= d
                h[j] += d
            else:
                cur = c
        if cur < best_c5:
            best_c5 = cur
            best_h = h

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
