import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    h = h.copy()
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += (target - s) / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smax_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if T < 1e-14:
        return m * dx
    return (m + T * np.log(np.exp((corr - m) / T).sum())) * dx

def sum_eq(x, n):
    return x.sum() - n * 0.5

def upsample(h, nn):
    no = len(h)
    xo = np.arange(no) * (2.0 / no)
    xn = np.arange(nn) * (2.0 / nn)
    return fix_sum(np.interp(xn, xo, h), nn)

def push_binary(h, n):
    h = h.copy()
    lo = (h > 0) & (h < 0.1)
    hi = (h > 0.9) & (h < 1.0)
    h[lo] *= 0.3
    h[hi] = 1.0 - (1.0 - h[hi]) * 0.3
    return fix_sum(h, n)
