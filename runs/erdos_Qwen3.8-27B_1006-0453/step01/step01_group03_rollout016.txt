        if not time_ok(45):
            break
        h_r = best_h + scales[r] * np.random.randn(n)
        h_r = project_feasible(h_r, target_sum)
        h_r, c_r = coord_descent(h_r, target_sum, dx, 40, t0 + budget_s - 45)
        if c_r < best_c5 and time_ok(45):
            h_r = slsqp_top_lags(h_r, top_k=5, maxiter=100, target_sum=target_sum, dx=dx)
            h_r, c_r = coord_descent(h_r, target_sum, dx, 30, t0 + budget_s - 45)
        if c_r < best_c5:
            best_h, best_c5 = h_r, c_r

    # Phase 6: Try different n values
    for n_try in [43, 51, 31]:
        if not time_ok(50):
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        # Interpolate
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, ts_try)
        h_try, c_try = coord_descent(h_try, ts_try, dx_try, 60, t0 + budget_s - 50)
        if c_try < best_c5 and time_ok(40):
            h_try = slsqp_top_lags(h_try, top_k=5, maxiter=150, target_sum=ts_try, dx=dx_try)
            h_try, c_try = coord_descent(h_try, ts_try, dx_try, 40, t0 + budget_s - 50)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try
            n = n_try
            dx = dx_try
            target_sum = ts_try

    # Phase 7: Binary-ish starts
    for r in range(5):
        if not time_ok(30):
            break
        h_b = np.zeros(n)
        pos = np.random.permutation(n)
        n_ones = int(target_sum)
        h_b[pos[:n_ones]] = 1.0
        frac = target_sum - n_ones
        if frac > 0.005:
            h_b[pos[n_ones]] = frac
        h_b = project_feasible(h_b, target_sum)
        h_b, c_b = coord_descent(h_b, target_sum, dx, 50, t0 + budget_s - 30)
        if c_b < best_c5 and time_ok(25):
            h_b = slsqp_top_lags(h_b, top_k=5, maxiter=100, target_sum=target_sum, dx=dx)
            h_b, c_b = coord_descent(h_b, target_sum, dx, 30, t0 + budget_s - 25)
        if c_b < best_c5:
            best_h, best_c5 = h_b, c_b

    # Final polish
    if time_ok(5):
        h_f, c_f = coord_descent(best_h, target_sum, dx, 50, t0 + budget_s - 3)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, len(best_h)
