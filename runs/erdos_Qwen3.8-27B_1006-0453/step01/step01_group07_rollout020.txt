        # j ranges from 0 to m-1
        j = np.arange(m)
        # Term 1: h[j+lag]*(1-h[j]), grad w.r.t. h[j+lag] is (1-h[j])
        grad[j + lag] += (1 - h[j])
        # Term 2: h[j+lag]*(1-h[j]), grad w.r.t. h[j] is -h[j+lag]
        grad[:m] += -h[j + lag]
    else:
        L = -lag
        m = n - L
        # j ranges from L to n-1
        j = np.arange(L, n)
        # Term 1: h[j]*(1-h[j-L]), grad w.r.t. h[j] is (1-h[j-L])
        grad[j] += (1 - h[j - L])
        # Term 2: h[j]*(1-h[j-L]), grad w.r.t. h[j-L] is -h[j]
        grad[:n-L] += -h[j]
    return grad
