```python
def project_onto_constraint(g, target):
    """Project g onto {0<=h<=1, sum(h)=target}"""
    n = len(g)
    lo, hi = g.min() - target, 1.0 - (g.max() - target)
    # Actually: h_i = clip(g_i + lam, 0, 1), need sum = target
    # sum(clip(g_i + lam, 0, 1)) is increasing in lam
    # At lam -> -inf: sum -> 0
    # At lam -> +inf: sum -> n
    lo = -max(g) - 1.0
    hi = 1.0 - min(g) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(g + mid, 0.0, 1.0)
        s = np.sum(h)
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g + (lo+hi)/2, 0.0, 1.0)
    # Fix any residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.minimum((corr - c_max) / T, 50.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = (c_max + T * np.log(sum_exp)) * dx
    w = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        elif -n < lag < 0:
            l = -lag
            grad[:n - l] += wk * (1.0 - h[l:])
            grad[l:] -= wk * h[:n - l]
    return val, grad * dx

def project_box_sum(g, target):
    n = len(g)
    lo = -np.max(g) - 1.0
    hi = 1.0 - np.min(g) + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(g + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = int(np.sum(mask))
    if cnt > 0:
        h[mask] += (target - s) / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    x = np.arange(new_n) * (2.0 / new_n)
    h_ext = np.concatenate([h, h])
    x_ext = np.arange(2 * old_n) * (2.0 / old_n)
    h_new = np.interp(x, x_ext, h_ext)
    target = new_n * 0.5
    return project_box_sum(h_new, target)

def constraint_sum(x, n):
    return np.sum(x) - n * 0.5

def slsqp_obj(x):
    return compute_c5(x)

def slsqp_grad(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * x
    elif 0 < lag < n:
        g[lag:] += (1.0 - x[:n - lag])
        g[:n - lag] -= x[lag:]
    elif -n < lag < 0:
        l = -lag
        g[:n - l] += (1.0 - x[l:])
        g[l:] -= x[:n - l]
    return g * (2.0 / n)

def slsqp_polish(h, n, max_iter):
    res = opt.minimize(slsqp_obj, h, jac=slsqp_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}],
                       options={'maxiter': max_iter, 'ftol': 1e-13})
    if res.fun < compute_c5(h):
        return project_box_sum(res.x, n * 0.5)
    return h

def pgd(h, n, T_sched, t_start, budge_s, min_rem):
    target = n * 0.5
    h_curr = h.copy()
    for T in T_sched:
        if time.time() - t_start > budge_s - min_rem:
            break
        alpha = 1.0
        mom = np.zeros(n)
        beta = 0.85
        stalled = 0
        best_val, _ = softmax_val_grad(h_curr, T)
        for it in range(8000):
            if time.time() - t_start > budge_s - min_rem:
                break
            val, grad = softmax_val_grad(h_curr, T)
            if val < best_val - 1e-14:
                best_val = val
                stalled = 0
            else:
                stalled += 1
            if stalled > 500:
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            mom = beta * mom + (1.0 - beta) * grad
            gnorm = np.linalg.norm(mom)
            if gnorm < 1e-13:
                break
            step = min(alpha / (gnorm + 1e-8), 0.04)
            h_try = project_box_sum(h_curr - step * mom, target)
            val_try, _ = softmax_val_grad(h_try, T)
            if val_try < val:
                h_curr = h_try
                alpha = min(alpha * 1.06, 80.0)
            else:
                alpha *= 0.65
                mom *= 0.5
                if alpha < 1e-8:
                    break
        h_curr = project_box_sum(h_curr, target)
    return h_curr
```
