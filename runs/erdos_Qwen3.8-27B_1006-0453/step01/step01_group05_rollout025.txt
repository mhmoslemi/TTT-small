def active_set_step(h, n, top_k=5, alpha_max=1.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full')
    C5 = np.max(corr) * dx
    
    # Find active lags
    target = C5 / dx
    active = np.where(corr >= target - 1e-6 * max(1, np.max(corr)))[0]
    if len(active) > top_k:
        # Keep only top_k
        top_indices = np.argsort(corr)[::-1][:top_k]
        active = top_indices
    
    # Compute gradients
    n_active = len(active)
    grad = np.zeros((n_active, n))
    for idx, lag_j in enumerate(active):
        k = lag_j - (n - 1)  # actual shift
        g = compute_grad_corr(h, n, k)
        grad[idx] = g
    
    # Solve LP:
    # minimize t (variable index n)
    # subject to: grad_m^T delta - t <= -f_m  for m in active
    #             sum(delta) = 0
    #             -h[i] <= delta[i] <= 1-h[i]
    
    # Variables: delta[0..n-1], t  (total n+1 variables)
    # c = [0,...,0, 1] (n+1 elements)
    
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag m: grad_m^T delta - t <= -corr[lag_j]
    # [grad_m, -1] @ [delta; t] <= -corr[lag_j]
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for idx in range(n_active):
        A_ub[idx, :n] = grad[idx]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -corr[active[idx]]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i] <= delta[i] <= 1-h[i], t unbounded (or lower bounded)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((None, None))  # t unbounded
    
    from scipy.optimize import linprog
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        delta = res.x[:n]
        # Line search
        step = delta
        best_alpha = alpha_max
        best_c5 = C5
        for alpha in [alpha_max, alpha_max*0.5, alpha_max*0.25, alpha_max*0.1, 0.05, 0.01]:

            h_try = np.clip(h_try, 0, 1)

            if c5_try < best_c5:
                best_c5 = c5_try
                best_alpha = alpha
        return h + best_alpha * step, best_c5
    return h, C5

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr(h, n, s):
    """Gradient of c(s) w.r.t. h, where s is the shift (0-indexed in correlate output = s + (n-1))"""
    # s is the actual shift: s = j - (n-1) where j is 0-index in correlate output
    g = np.zeros(n)
    if s >= 0:
        for m in range(s, n):
            g[m] += (1.0 - h[m - s])
        for m in range(0, n - s):
            g[m] -= h[m + s]
    elif s < 0:
        r = -s
        for m in range(0, n - r):
            g[m] += (1.0 - h[m + r])
        for m in range(r, n):
            g[m] -= h[m - r]
    else:
        g = 1.0 - 2.0 * h
    return g

def grad_corr(h, n, s):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        for m in range(s, n):
            g[m] += (1.0 - h[m - s])
        for m in range(0, n - s):
            g[m] -= h[m + s]
    else:
        r = -s
        for m in range(0, n - r):
            g[m] += (1.0 - h[m + r])
        for m in range(r, n):
            g[m] -= h[m - r]
    return g

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)

            h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr(h, n, s):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        for m in range(s, n):
            g[m] += (1.0 - h[m - s])
        for m in range(0, n - s):
            g[m] -= h[m + s]
    else:
        r = -s
        for m in range(0, n - r):
            g[m] += (1.0 - h[m + r])
        for m in range(r, n):
            g[m] -= h[m - r]
    return g


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        nf = int(np.sum(free))
        if nf == 0:
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def sum_con(x, n):
    return np.sum(x) - n / 2.0


def smooth_obj(x, n, temp):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    shifted = (c - m) * temp
    shifted = np.clip(shifted, -700, 0)
    return (m + np.log(np.sum(np.exp(shifted))) / temp) * (2.0 / n)


def do_linprog_step(h, n, active, corr):
    n_act = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    for i, j in enumerate(active):
        s = j - (n - 1)
        A_ub[i, :n] = grad_corr(h, n, s)
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[j]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((-50.0, 50.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def active_set_opt(h, n, top_k, max_iters, t_start, budget):
    best_c5 = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() - t_start > budget - 40:
            break
        corr = compute_corr(h, n)
        c5c = float(np.max(corr))
        thr = c5c - max(1e-9, 5e-7 * c5c)
        act = np.where(corr >= thr)[0]
        if len(act) > top_k:
            act = np.argsort(corr)[::-1][:top_k]
        act = np.sort(act)
        delta = do_linprog_step(h, n, act, corr)
        if delta is None or np.linalg.norm(delta) < 1e-14:
            j_max = int(np.argmax(corr))
            s = j_max - (n - 1)
            g = grad_corr(h, n, s)
            improved = False
            for sc in [0.02, 0.01, 0.005]:
                h_try = repair_h(h - sc * g, n)
                c5_try = compute_c5(h_try, n)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    h = h_try
                    improved = True
                    break
            if not improved:
                break
            continue
        best_alpha = 0.0
        best_new = best_c5
        for alpha in [0.5, 0.3, 0.15, 0.07, 0.03, 0.01]:
            h_t = repair_h(h + alpha * delta, n)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_new:
                best_new = c5_t
                best_alpha = alpha
        if best_alpha == 0.0:
            break
        h = repair_h(h + best_alpha * delta, n)
        best_c5 = compute_c5(h, n)
        if best_c5 < 0.38080:
            break
    return h, best_c5


def slsqp_polish(h, n, temp, maxiter, t_start, budget):
    if time.time() - t_start > budget - 30:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, temp),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success:
            h_t = repair_h(res.x, n)
            return h_t, compute_c5(h_t, n)
    except Exception:
        pass
    return h, compute_c5(h, n)


def cobyla_polish(h, n, t_start, budget):
    if time.time() - t_start > budget - 30:
        return h, compute_c5(h, n)
    try:
        res = minimize(smooth_obj, h, method='COBYLA', args=(n, 80),
                       options={'maxiter': 600, 'rhobeg': 0.02, 'disp': False})
        h_t = repair_h(res.x, n)
        return h_t, compute_c5(h_t, n)
    except Exception:
        return h, compute_c5(h, n)
