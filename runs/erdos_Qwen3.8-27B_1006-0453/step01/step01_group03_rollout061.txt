Terms with h[i]:
- j=i: need 0≤i+l<n. If valid, contributes t*(1-h[i+l])
- j=i-l: need 0≤i-l<n and 0≤i<n (second condition always true since 0≤i<n). If i-l valid, contributes h[i-l]*(1-t)

So C(l) = A_l + [0≤i+l<n] * t * (1-h[i+l]) + [0≤i-l<n] * h[i-l] * (1-t)

= A_l + [0≤i+l<n] * t * (1-h[i+l]) + [0≤i-l<n] * h[i-l] - [0≤i-l<n] * t * h[i-l]

= (A_l + [0≤i-l<n] * h[i-l]) + t * ([0≤i+l<n] * (1-h[i+l]) - [0≤i-l<n] * h[i-l])

Let me define:
- c_l = A_l + [0≤i-l<n] * h[i-l] (constant part)
- b_l = [0≤i+l<n] * (1-h[i+l]) - [0≤i-l<n] * h[i-l] (coefficient of t)

Then C(l) = c_l + b_l * t, and we want to minimize max_l (c_l + b_l * t) over t ∈ [0,1].

This is a 1-D convex optimization! The optimal t is where the "upper envelope" is minimized. We can find it by:
- Evaluating f(t) = max_l (c_l + b_l * t) at several points
- Or more precisely, the minimum occurs where two (or more) of the linear functions cross at the top

For 93 linear functions, the lower envelope has at most 92 breakpoints. We can find the minimum by:
1. Sort by slope b_l
2. Use a convex hull approach
Or simply:
1. Try t = 0 and t = 1
2. For each pair of functions, find their crossing point and check if it's the minimum

Actually, the simplest approach for 1-D: the function f(t) = max_l (c_l + b_l*t) is convex. We can minimize it using ternary search or by finding where the subgradient changes sign.

The subgradient at t is the set {b_l : f(t) = c_l + b_l * t} (the slopes of the active functions at the max).

The minimum is where 0 ∈ subgradient, i.e., there exist active functions with slopes on both sides of 0, or an active function with slope 0.

Simplest: just evaluate f(t) at, say, 50 evenly spaced points in [0,1] and pick the best. That's 50*93 = 4650 operations per coordinate, times 47 coordinates = 218,550 per full pass. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def soft_gradient(h, beta):
    n = len(h)
    C = np.correlate(h, 1.0 - h, mode='full')
    C_shifted = C - C.max()
    w = np.exp(beta * C_shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    mask = w > 1e-10
    if not mask.any():
        return np.zeros(n)
    active = np.where(mask)[0]
    grad = np.ones(n, dtype=float)
    for l in active:
        lag = l - (n - 1)
        wl = w[l]
        if lag == 0:
            grad -= 2.0 * wl * h
        elif lag > 0:
            grad[:n - lag] -= wl * h[lag:]
            grad[lag:] -= wl * h[:n - lag]
        else:
            p = -lag
            grad[p:] -= wl * h[:n - p]
            grad[:n - p] -= wl * h[p:]
    return grad

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = float(np.max(np.correlate(h0, 1.0 - h0, mode='full')))
    x0 = np.concatenate([h0, [t0 + 0.003]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def coord_pass(h, n_evals=25):
    n = len(h)
    ts = np.linspace(0.0, 1.0, n_evals)
    for i in range(n):
        old = h[i]
        best_t = old
        best_val = np.inf
        for t in ts:
            h[i] = t
            val = np.max(np.correlate(h, 1.0 - h, mode='full'))
            if val < best_val:
                best_val = val
                best_t = t
        h[i] = best_t
    return h

def gradient_descent(h0, dx, target_sum, max_iter=5000, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(n)
    momentum = 0.85
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        frac = it / max(max_iter, 1)
        beta = 5.0 + 70.0 * frac
        g = soft_gradient(h, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        lr = 0.025 / (1.0 + 0.003 * it)
        v = momentum * v - lr * g
        h_trial = project_sr(h + v, target_sum)
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
        else:
            h_trial2 = project_sr(h - lr * 0.05 * g, target_sum)
            c5_trial2 = compute_c5(h_trial2, dx)
            if c5_trial2 < best_c5:
                best_c5 = c5_trial2
                best_h = h_trial2.copy()
                h = h_trial2
                v = np.zeros(n)
            else:
                v *= 0.5
    return best_h, best_c5

    np.random.seed(seed)
    t0 = time.time()

    try:
        h_raw = np.array(initial_h_values, dtype=float)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def time_ok(margin=10):
        return (time.time() - t0) < budget_s - margin

    # Phase 1: SLSQP from initial
    if time_ok():
        h1 = slsqp_solve(h0, dx, target_sum, 400)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Gradient descent
    if time_ok():
        deadline = t0 + budget_s - 250
        h2, c2 = gradient_descent(best_h, dx, target_sum, 5000, deadline)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Coordinate descent passes
    while time_ok(30):
        h3 = coord_pass(best_h, n_evals=30)
        h3 = project_sr(h3, target_sum)
        c3 = compute_c5(h3, dx)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
        else:
            break

    # Phase 4: SLSQP polish
    if time_ok():
        h4 = slsqp_solve(best_h, dx, target_sum, 300)
        c4 = compute_c5(h4, dx)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Multi-start gaussian
    scales = [0.12, 0.08, 0.15, 0.05, 0.1, 0.2, 0.06, 0.18]
    for r in range(min(30, int((budget_s - (time.time() - t0) - 100) / 8) + 5)):
        if not time_ok(12):
            break

        h_r = best_h + scale * np.random.randn(n)
        h_r = project_sr(h_r, target_sum)
        h_r = slsqp_solve(h_r, dx, target_sum, 150)
        c5_r = compute_c5(h_r, dx)
        if c5_r < best_c5 and time_ok(12):
            deadline = t0 + budget_s - 80
            h_r, c5_r = gradient_descent(h_r, dx, target_sum, 1500, deadline)
            h_r = slsqp_solve(h_r, dx, target_sum, 100)
            c5_r = compute_c5(h_r, dx)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 6: Structured starts (reverse, shift-like)
    if time_ok(20):
        h_rev = best_h[::-1].copy()
        h_rev = project_sr(h_rev, target_sum)
        h_rev = slsqp_solve(h_rev, dx, target_sum, 200)
        c5_rev = compute_c5(h_rev, dx)
        if c5_rev < best_c5:
            best_h, best_c5 = h_rev, c5_rev

    # Phase 7: Random binary starts
    for r in range(6):
        if not time_ok(15):
            break
        h_b = np.zeros(n)
        pos = np.random.permutation(n)
        nb = int(target_sum)
        h_b[pos[:nb]] = 1.0
        frac = target_sum - nb
        if abs(frac) > 0.01:
            h_b[pos[nb]] = frac
        h_b = project_sr(h_b, target_sum)
        h_b = slsqp_solve(h_b, dx, target_sum, 150)
        c5_b = compute_c5(h_b, dx)
        if c5_b < best_c5 and time_ok(15):
            deadline = t0 + budget_s - 50
            h_b, c5_b = gradient_descent(h_b, dx, target_sum, 1000, deadline)
            h_b = slsqp_solve(h_b, dx, target_sum, 100)
            c5_b = compute_c5(h_b, dx)
            if c5_b < best_c5:
                best_h, best_c5 = h_b, c5_b

    # Phase 8: Final coordinate + gradient + SLSQP
    if time_ok(30):
        h8 = coord_pass(best_h, n_evals=40)
        h8 = project_sr(h8, target_sum)
        c8 = compute_c5(h8, dx)
        if c8 < best_c5:
            best_h, best_c5 = h8, c8
    if time_ok(20):
        deadline = t0 + budget_s - 20
        h8b, c8b = gradient_descent(best_h, dx, target_sum, 2000, deadline)
        if c8b < best_c5:
            best_h, best_c5 = h8b, c8b
    if time_ok(5):
        h_f = slsqp_solve(best_h, dx, target_sum, 200)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    best_h = project_sr(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
