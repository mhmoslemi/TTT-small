def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:

    return np.sum(h) - n / 2.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        per = diff / nf
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h


def sum_diff(h, n):
    return np.sum(h) - n / 2.0

    m = np.max(v)
    return m + temp * np.log(np.mean(np.exp((v - m) / temp)))

    grad = np.zeros(n)
    if shift >= 0:
        for i in range(n - shift):
            grad[i] += (1.0 - h[i + shift])
            grad[i + shift] -= h[i]
    else:
        s = -shift

            grad[i] -= h[i + s]
