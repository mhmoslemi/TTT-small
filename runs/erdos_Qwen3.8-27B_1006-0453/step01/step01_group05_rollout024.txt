import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def sum_fun(x, n):
    return np.sum(x) - n / 2.0


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = temp * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return float(np.log(np.sum(np.exp(shifted))) / temp + m)


def slsqp_opt(x0, n, bounds, temp, maxiter):
    cons = [{'type': 'eq', 'fun': sum_fun, 'args': (n,)}]
    res = minimize(smooth_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x


def pair_cd_sweep(h, n, n_sweeps=80, grid_pts=31):
    h = h.copy()
    best_val = compute_c5(h, n)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            j = (i + 1) % n
            s = h[i] + h[j]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            rng = t_hi - t_lo
            if rng < 1e-12:
                continue
            best_t = h[i]
            cur_val = best_val
            for frac in np.linspace(0, 1, grid_pts):
                t = t_lo + frac * rng
                h[i] = t
                h[j] = s - t
                val = compute_c5(h, n)
                if val < cur_val - 1e-15:
                    cur_val = val
                    best_t = t
            h[i] = best_t
            h[j] = s - best_t
            if cur_val < best_val:
                best_val = cur_val
                improved = True
        if not improved:
            break
    return h


def subgrad_step(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        j_plus = i + lag
        j_minus = i - lag
        if 0 <= j_minus < n:
            g[i] += (1.0 - h[j_minus])
        if 0 <= j_plus < n:
            g[i] -= h[j_plus]
    g *= dx
    h_new = h - alpha * g
    h_new = repair_h(h_new, n)
    return h_new


def simulate_pair_moves(h, n, n_moves=2000):
    h = h.copy()
    best_val = compute_c5(h, n)
    rng = np.random.RandomState(0)
    for _ in range(n_moves):

        j = rng.randint(n)
