import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def constraint_eq(x, target):
    return x.sum() - target

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n, t_end):
    """Coordinate descent: optimize each coordinate to minimize C5."""
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for sweep in range(200):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            others_sum = h.sum() - h[i]
            lo = max(0.0, target - others_sum)
            hi = min(1.0, target - others_sum)
            if hi - lo < 1e-10:
                continue
            cur = h[i]
            best_val = best_c5
            best_hi = cur
            # Sample candidate values
            fracs = np.array([0.0, 0.15, 0.35, 0.5, 0.65, 0.85, 1.0])
            for f in fracs:
                trial = lo + f * (hi - lo)
                if abs(trial - cur) < 1e-8:
                    trial = cur
                h[i] = trial
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_hi = trial
            h[i] = best_hi
            if best_val < best_c5 - 1e-12:
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

def interpolate_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def run_slsqp(h, n, T, maxiter):
    cons = [{'type': 'eq', 'fun': constraint_eq, 'args': (n / 2.0,)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_c5, h, args=(T,),
            method='SLSQP', bounds=bnds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
        )
        hc = project_feasible(np.clip(res.x, 0, 1), n)
        return hc, compute_c5(hc)
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def t_left(frac):
        return t0 + budget_s * frac

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(46)
        best_h[:23] = 1.0

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP refinement with increasing T (20% time)
    p1_end = t_left(0.20)
    for T in [30, 100, 300, 800]:
        if elapsed() > p1_end:
            break
        hc, c5v = run_slsqp(best_h, n, T, 500)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent (25% time)
    p2_end = t_left(0.45)
    best_h, best_c5 = coordinate_descent(best_h, n, p2_end)

    # Phase 3: SLSQP again after CD (15% time)
    p3_end = t_left(0.60)
    for T in [500, 1500, 4000]:
        if elapsed() > p3_end:
            break
        hc, c5v = run_slsqp(best_h, n, T, 600)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # Follow up with CD
        ch, cc = coordinate_descent(best_h, n, p3_end)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch

    # Phase 4: Try different resolutions (25% time)
    p4_end = t_left(0.85)
    for n_try in [46, 48, 50, 54, 64, 72]:
        if elapsed() > p4_end:
            break
        h_try = interpolate_to(best_h, n_try)
        h_try = project_feasible(h_try, n_try)
        for T in [100, 500, 1500]:
            if elapsed() > p4_end:
                break
            hc, c5v = run_slsqp(h_try, n_try, T, 400)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                n = n_try
            ch, cc = coordinate_descent(np.array(hc, copy=True), n_try, p4_end)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch
                n = n_try

    # Phase 5: Final multi-pass polish (15% time)
    p5_end = t_left(0.99)
    for round_ix in range(6):
        if elapsed() > p5_end:
            break
        T = 2000 + round_ix * 3000
        hc, c5v = run_slsqp(best_h, n, T, 400)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        ch, cc = coordinate_descent(best_h, n, p5_end)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch

    # Final feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
