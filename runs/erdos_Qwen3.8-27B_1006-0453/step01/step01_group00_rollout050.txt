def project_constraint(v, n):

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(v, n):
    target = n * 0.5
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        ni = interior.sum()
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s_star = int(np.argmax(corr))
    shift = s_star - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, shift)
    hi1 = min(n, n + shift)
    if hi1 > lo1:
        grad[lo1:hi1] += 1.0 - h[lo1 - shift:hi1 - shift]
    lo2 = max(0, -shift)
    hi2 = min(n, n - shift)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 + shift:hi2 + shift]
    return grad, corr[s_star]

def fista_solve(h0, n, max_iter=20000, t_end=None):
    t_end = t_end or (time.time() + 1100)
    h = h0.copy()
    y = h.copy()
    t_k = 1.0
    alpha0 = n / 6.0
    best_h = h.copy()
    best_val = compute_c5(h)
    stagnant = 0
    alpha = alpha0

    for it in range(max_iter):
        if time.time() > t_end:
            break
        grad, _ = subgradient(y)
        z = y - alpha * grad
        h_new = project_constraint(z, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) * 0.5
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        h = h_new
        t_k = t_new

        if (it + 1) % 500 == 0:
            val = compute_c5(h)
            if val < best_val - 1e-10:
                best_val = val
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > 3:
                    alpha *= 0.5
                    y = h.copy()
                    t_k = 1.0
                    stagnant = 0
                    if alpha < 0.01:
                        break

    val = compute_c5(h)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    return best_h, best_val

def slsqp_solve(h0, T, max_iter=600):
    n = len(h0)
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = corr.max()
        if T < 1e-15:
            return cmax * dx
        shifted = (corr - cmax) / T
        return (cmax + T * np.log(np.sum(np.exp(shifted)))) * dx
    def cons(h):
        return h.sum() - n * 0.5
    res = opt.minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
    if res.x is not None:
        h = project_constraint(np.clip(res.x, 0, 1), n)
        if compute_c5(h) < compute_c5(h0):
            return h
    return h0

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_constraint(h_new, new_n)

    np.random.seed(seed)
