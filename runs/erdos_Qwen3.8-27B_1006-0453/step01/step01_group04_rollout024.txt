Actually, you know what, let me try a completely different approach for part of the time: **gradient descent with projection**.

The subgradient of max_k c(k) is a convex combination of the gradients of the active c(k)'s. If we know which k's are active (within epsilon of the max), we can compute the subgradient and do projected gradient descent.

For a single active lag k*, the gradient of c(k*) with respect to h is:
dc/dh[i] = (1-h[i+l*]) - h[i-l*] (with boundary conditions)

This is O(n) to compute for ONE lag. So if there are only a few active lags, the gradient is cheap.

```python
def subgrad_max(h, tol=0.01):
    """Compute subgradient of max_k c(k) as average over active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    # Active lags: those within tol of max
    active = np.where(corr > cmax - tol)[0]
    # Gradient contribution from each active lag
    grad = np.zeros(n)
    for k_idx in active:
        l = k_idx - (n - 1)
        # dc(k)/dh[i] = (1-h[i+l]) if 0<=i+l<n
        #               - h[i-l] if 0<=i-l<n
        # Term 1: (1-h[i+l]) for i in [0, n-1-l] if l>=0, or [0, n-1] if l<0
        # More precisely: for each i in [0,n-1], term1 = (1-h[i+l]) if 0<=i+l<n else 0
        # Term 2: for each i in [0,n-1], term2 = -h[i-l] if 0<=i-l<n else 0
        
        # Vectorized:
        if l >= 0:
            mask1 = np.arange(n) + l < n
            grad[mask1] += (1.0 - h[np.arange(n)[mask1] + l])
            mask2 = np.arange(n) - l >= 0
            grad[mask2] -= h[np.arange(n)[mask2] - l]
        else:
            # l is negative, |l| = -l
            ml = -l
            mask1 = np.arange(n) + l < n  # i < n + l = n - ml
            # Actually i+l >= 0 means i >= -l = ml
            m1 = (np.arange(n) + l >= 0) & (np.arange(n) + l < n)
            grad[m1] += (1.0 - h[np.arange(n)[m1] + l])
            m2 = (np.arange(n) - l >= 0) & (np.arange(n) - l < n)
            grad[m2] -= h[np.arange(n)[m2] - l]
    
    # Normalize by number of active lags (subgradient = average)
    grad /= max(1, len(active))
    return grad, cmax, len(active)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    k = min(topk, len(corr))
    part = np.partition(corr, len(corr) - k)[-k:]
    cmax = np.max(part)
    # Use full log-sum-exp but shifted
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair_h(h):
    n = len(h)
    target = n * 0.5
    s = np.sum(h)
    if s < 1e-15:
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lbgfgs_round(h_in, t_deadline, schedules):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    cur = h_in.copy()
    best = compute_c5(cur)
    for (temp, pen, topk, maxit) in schedules:
        if time.time() > t_deadline:
            break
        res = minimize(smooth_obj, cur, args=(temp, pen, topk),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxls': 40})
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best:
            best = c5v
            cur = nh
    return cur, best

def coord_polish(h, n_sweeps, step_size, t_deadline):
    n = len(h)
    dx = 2.0 / n
    cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    best_c5 = cmax * dx
    for _ in range(n_sweeps):

                nv = h[i] + delta

                h[i] = nv
                c_new = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if c_new < cmax:
                    cmax = c_new
                    improved = True
                else:
                    h[i] -= delta
        if not improved:
            break
    h = repair_h(h)
    return h, compute_c5(h)

def perturb(h, sigma, rng):
    return repair_h(np.clip(h + rng.standard_normal(len(h)) * sigma, 0.0, 1.0))

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    best_h, best_c5 = h0.copy(), compute_c5(h0)
    cur = h0.copy()
    n0 = len(cur)

    # Phase 1: optimize at initial resolution
    td1 = t0 + budget_s * 0.10
    s1 = [(0.05, 500, 80, 500),
          (0.01, 2000, 40, 500),
          (0.003, 10000, 25, 400)]
    cur, lb = lbgfgs_round(cur, td1, s1)
    if lb < best_c5:
        best_c5, best_h = lb, cur.copy()
    # coord polish at n0
    cur, lb = coord_polish(cur, 5, 0.01, t0 + budget_s * 0.14)
    if lb < best_c5:
        best_c5, best_h = lb, cur.copy()

    # Phase 2: progressive upscaling
    cur_n = len(cur)
    targets = [720, 1440, 2880]
    phase2_start = time.time()
    phase2_budget = budget_s * 0.50

    for tgt in targets:
        if cur_n >= tgt:
            continue
        elapsed2 = time.time() - phase2_start
        if elapsed2 > phase2_budget:
            break
        factor = tgt // cur_n
        if factor < 1:
            continue
        cur = interp_upscale(cur, factor)
        cur = repair_h(cur)
        cur_n = len(cur)

        remain = max(budget_s * 0.03, phase2_budget - (time.time() - phase2_start))
        alloc = remain * 0.3
        td2 = time.time() + alloc

        n_it = max(40, min(600, int(400000 // cur_n)))
        s2 = [(0.04, 1000, 80, n_it),
              (0.008, 5000, 40, n_it),
              (0.001, 20000, 20, n_it // 2)]
        cur, lb = lbgfgs_round(cur, td2, s2)
        if lb < best_c5:
            best_c5, best_h = lb, cur.copy()

        # Coord polish at this level (only for manageable n)
        if cur_n <= 1440 and time.time() < t0 + budget_s * 0.68:
            tp = min(8, max(2, int(600 // (cur_n // 360))))
            stp = 0.005
            cur, lb2 = coord_polish(cur, tp, stp, t0 + budget_s * 0.72)
            if lb2 < best_c5:
                best_c5, best_h = lb2, cur.copy()

    # Phase 3: diverse restarts from best
    t3_end = t0 + budget_s * 0.87
    n_best = len(best_h)
    for ri in range(10):
        if time.time() > t3_end:
            break
        sig = 0.06 / (1.0 + 0.3 * ri)
        hp = perturb(best_h, sig, rng)
        n = len(hp)
        n_it = max(30, min(350, int(250000 // n)))
        s3 = [(0.015, 5000, 40, n_it),
              (0.003, 15000, 15, n_it // 2)]
        td3 = time.time() + budget_s * 0.045
        hp, lb = lbgfgs_round(hp, td3, s3)
        if n <= 1440:
            hp, lb = coord_polish(hp, 3, 0.004, t0 + budget_s * 0.87)
        if lb < best_c5:
            best_c5, best_h = lb, hp.copy()

    # Phase 4: final polish on best
    if time.time() < t0 + budget_s * 0.94:
        n_b = len(best_h)
        n_it = max(30, min(300, int(150000 // n_b)))
        s4 = [(0.004, 10000, 30, n_it),
              (0.001, 50000, 10, n_it // 2),
              (0.0005, 100000, 8, max(20, n_it // 3))]
        td4 = t0 + budget_s * 0.96
        best_h, lb = lbgfgs_round(best_h, td4, s4)
        if lb < best_c5:
            best_c5 = lb
        if n_b <= 1440:
            best_h, lb = coord_polish(best_h, 6, 0.003, t0 + budget_s * 0.985)
            if lb < best_c5:
                best_c5 = lb

    best_h = repair_h(best_h)
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5
    return best_h, best_c5, len(best_h)
```
