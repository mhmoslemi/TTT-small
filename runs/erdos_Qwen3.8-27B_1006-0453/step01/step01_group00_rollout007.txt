import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            ns = n - s
            grad[:ns] += (1.0 - h[s:])
            grad[s:] -= h[:ns]
    else:
        t = -s
        nt = n - t
        grad[t:] += (1.0 - h[:nt])
        grad[:nt] -= h[t:]
    return grad

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-12) & (h < 1.0 - 1e-12)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    diff2 = target - s2
    if abs(diff2) > 1e-12 and n_int > 0:
        h[interior] += diff2 / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_sum_obj(h, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = np.sort(corr)[-K:]
    return float(np.sum(topk))

def soft_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return c_max + T * np.log(s)

def soft_max_obj_scaled(h, T, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def try_slsqp(h_start, n, obj_fn, max_iter, ftol=1e-16):
    try:
        res = opt.minimize(
            obj_fn, h_start, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        return res.x
    except Exception:
        return h_start

def projected_grad_descent(h, n, K_top, lr_init, n_iter, t_deadline):
    h = h.copy()
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > t_deadline - 60:
            break
        corr = compute_corr_full(h)
        top_idx = np.argsort(corr)[-K_top:]
        grad_avg = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            grad_avg += grad_at_shift(h, s)
        grad_avg /= K_top
        gnorm = np.linalg.norm(grad_avg)
        if gnorm < 1e-14:
            break
        h_new = h - lr * grad_avg
        h_new = project_feasible(h_new, n)
        h = h_new
        if (it + 1) % 200 == 0:
            c5_curr = compute_c5(h)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h.copy()
        lr *= 0.9998
    return best_h, best_c5

def upsample_linear(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n * 0.5
    s = np.sum(h_new)
    if s > 1e-15:
        h_new *= target / s
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, new_n)

def downsample_linear(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n * 0.5
    s = np.sum(h_new)
    if s > 1e-15:
        h_new *= target / s
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, new_n)

def make_obj_T(T, dx):
    def obj(h):
        return soft_max_obj_scaled(h, T, dx)
    return obj
