import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def hard_subgrad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j + k
        if 0 <= idx1 < n:
            grad[j] += (1.0 - h[idx1])
        idx2 = j - k
        if 0 <= idx2 < n:
            grad[j] -= h[idx2]
    return grad * dx

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_to_simplex(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def gradient_descent(h0, dx, target_sum, max_iter, deadline):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = hard_subgrad(h, dx)
        # Nesterov: momentum
        mu = 0.9
        v = mu * v - 0.01 * g
        # Line search
        best_lr = 0.01
        best_val = best_c5
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = project_to_simplex(h + lr * v, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_val:
                best_val = c5_new
                best_lr = lr
                h_new_best = h_new
        if best_val < best_c5:
            best_c5 = best_val
            best_h = h_new_best.copy()
            h = h_new_best
        else:
            # Try direct gradient step
            for lr in [0.01, 0.005, 0.002, 0.001]:
                h_new = project_to_simplex(h - lr * g, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:

    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_sr(h_raw, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def tok(m=10):
        return (time.time() - t0) < budget_s - m

    # Phase 1: SLSQP
    if tok():
        h1 = slsqp(h0, dx, target_sum, 400)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Soft gradient descent (long)
    if tok(80):
        dl = t0 + budget_s * 0.40
        h2, c2 = grad_descent(best_h, dx, target_sum, dl, use_soft=True)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Hard gradient descent (refinement)
    if tok(60):
        dl = t0 + budget_s * 0.58
        h3, c3 = grad_descent(best_h, dx, target_sum, dl, use_soft=False)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: SLSQP polish
    if tok(30):
        h4 = slsqp(best_h, dx, target_sum, 500)
        c4 = compute_c5(h4, dx)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Binary swap search
    if tok(50):
        dl = t0 + budget_s * 0.75
        h5, c5 = swap_search(best_h, dx, target_sum, dl, max_rounds=3000)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5
        # SLSQP after swaps
        if tok(20):
            h5b = slsqp(best_h, dx, target_sum, 200)
            c5b = compute_c5(h5b, dx)
            if c5b < best_c5:
                best_h, best_c5 = h5b, c5b

    # Phase 6: Multi-start
    for r in range(12):
        if not tok(40):
            break
        sc = 0.04 + 0.015 * r
        hp = best_h + sc * np.random.randn(n)
        hp = project_sr(hp, target_sum)
        hp = slsqp(hp, dx, target_sum, 200)
        cp = compute_c5(hp, dx)
        if cp < best_c5:
            best_h, best_c5 = hp, cp
            if tok(30):
                dl = t0 + budget_s - 80
                hp2, cp2 = grad_descent(hp, dx, target_sum, dl)
                if cp2 < best_c5:
                    best_h, best_c5 = hp2, cp2

    # Phase 7: Random binary starts
    for r in range(6):
        if not tok(20):
            break
        hb = np.zeros(n)
        pos = np.random.permutation(n)
        nb = int(target_sum)
        hb[pos[:nb]] = 1.0
        fr = target_sum - nb
        if fr > 0.001:
            hb[pos[nb]] = fr
        hb = project_sr(hb, target_sum)
        hb = slsqp(hb, dx, target_sum, 150)
        cb = compute_c5(hb, dx)
        if cb < best_c5:
            dl = t0 + budget_s - 40
            hb2, cb2 = grad_descent(hb, dx, target_sum, dl)
            if cb2 < cb:
                cb, hb = cb2, hb2
            if cb < best_c5:
                best_h, best_c5 = hb, cb

    # Phase 8: Final hard gradient + SLSQP
    if tok(15):
        dl = t0 + budget_s - 20
        hf, cf = grad_descent(best_h, dx, target_sum, dl)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
    if tok(5):
        hf = slsqp(best_h, dx, target_sum, 300)
        cf = compute_c5(hf, dx)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    best_h = project_sr(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
