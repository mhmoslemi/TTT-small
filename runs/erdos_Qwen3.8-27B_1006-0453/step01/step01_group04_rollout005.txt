from scipy.optimize import minimize
import time

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5_val(h):
    return float(np.max(compute_c5_vals(h)))

def repair_h(h):
    n = len(h)
    integ = np.sum(h)
    target = n / 2.0
    if integ < 1e-12:
        return h
    h2 = h * (target / integ)
    return np.clip(h2, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    # Use soft threshold: sort and pick top n/2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    sm_val = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    constraint = np.sum(h) - n / 2.0
    return sm_val + 0.1 * constraint ** 2

def binary_sa(h_init, t0, time_budget, rng, initial_temp=0.05, final_temp=0.001, n_cool=500):

    h = to_binary(h_init)
    curr_c5 = compute_c5_val(h)
    best_c5 = curr_c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_iters = 100000

    stall = 0

        if time.time() - t0 > time_budget:
            break
        if every 50 iters: T *= T_factor

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        h[i] = 0; h[j] = 1

        delta = new_c5 - curr_c5

def continuous_optimize(h_init, t0, time_budget):

    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    
    best_c5 = compute_c5_val(h)

            h[i] = 0
            h[j] = 1
            nc = compute_c5_val(h)
            if nc < c5:
                c5 = nc
                ones[ii] = j
                zeros[jj] = i
                imp = True
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i] = 1
                h[j] = 0
        no_imp = 0 if imp else no_imp + 1
        if no_imp > 80:
            break
    return best_h, best_c5

def sa_search(h, t0, t_end, rng, t0_val=0.04, t1_val=0.0005, max_iters=300000):
    h = h.copy()
    c5 = compute_c5_val(h)
    best_c5 = c5
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = t0_val
    n_cool = 400
    cool_fac = (t1_val / t0_val) ** (1.0 / n_cool)
    cool_iv = max(1, max_iters // n_cool)
    stall = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        if it > 0 and it % cool_iv == 0:
            T *= cool_fac
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i, j = ones[ii], zeros[jj]
        h[i] = 0
        h[j] = 1
        nc = compute_c5_val(h)
        d = nc - c5
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-14)):
            c5 = nc
            ones[ii] = j
            zeros[jj] = i
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1
            h[j] = 0
        if stall > 20000:
            break
    return best_h, best_c5

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    sh = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(sh / temp)))
    cons = np.sum(h) - n / 2.0
    return sm + 10.0 * cons * cons

def continuous_polish(h_init, t0, t_end):
    n = len(h_init)
    h = h_init.astype(float).copy()
    bounds = [(0.0, 1.0)] * n
    best_c5 = compute_c5_val(h)
    for temp in [0.04, 0.01, 0.003, 0.0008]:
        if time.time() > t_end:
            break
        try:
            res = minimize(smooth_obj, h, args=(temp,), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-9})
            hn = np.clip(res.x, 0.0, 1.0)
            hn = repair_h(hn)
            c5 = compute_c5_val(hn)
            if c5 < best_c5:
                best_c5 = c5
                h = hn
        except Exception:
            pass
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end_total = t0 + budget_s

    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5_val(best_h)

    # Phase 1: Quick greedy on binary initial
    h_b = to_binary(h_init)
    h_b, c5 = greedy_search(h_b, t0, t0 + budget_s * 0.04, rng, 2000, 500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_b.copy()

    # Phase 2: Many SA+greedy cycles with diverse starts
    n_cycles = 25
    for cyc in range(n_cycles):
        if time.time() > t_end_total * 0.55:
            break
        cycle_end = t0 + budget_s * 0.04 + (cyc + 1) * (budget_s * 0.51 / n_cycles)

        # Create diverse start
        if cyc == 0:
            h_start = best_h.copy()
        elif cyc % 3 == 1:
            # Random binary
            h_start = rng.binomial(1, 0.5, n).astype(float)
        elif cyc % 3 == 2:
            # Perturb current best
            h_start = best_h.copy()
            nf = rng.randint(n // 15, n // 5 + 1)
            flips = rng.choice(n, nf, replace=False)
            h_start[flips] = 1.0 - h_start[flips]
            h_start = to_binary(h_start)
        else:
            # Structured: Thue-Morse-like
            h_start = np.zeros(n)
            for i in range(n):
                c = 0
                x = i
                while x > 0:
                    c ^= (x & 1)
                    x >>= 1
                h_start[i] = float(c)
            # Adjust to exactly n/2 ones
            if h_start.sum() < n // 2:
                extra = n // 2 - int(h_start.sum())
                zeros_idx = np.where(h_start == 0)[0]
                h_start[rng.choice(zeros_idx, extra, replace=False)] = 1
            elif h_start.sum() > n // 2:
                excess = int(h_start.sum()) - n // 2
                ones_idx = np.where(h_start == 1)[0]
                h_start[rng.choice(ones_idx, excess, replace=False)] = 0

        # SA
        h_sa, c5_sa = sa_search(h_start, t0, cycle_end - budget_s * 0.01, rng,
                                t0_val=0.035, t1_val=0.0004, max_iters=150000)
        # Greedy polish
        h_g, c5_g = greedy_search(h_sa, t0, cycle_end, rng, 1000, 300)

        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Thorough greedy from best
    t_greedy_end = min(t0 + budget_s * 0.8, t_end_total)
    h_fin, c5_fin = greedy_search(best_h, t0, t_greedy_end, rng, 10000, 600)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()

    # Phase 4: SA from best one more time (cooler)
    if time.time() < t0 + budget_s * 0.9:
        h_s2, c5_s2 = sa_search(best_h, t0, t0 + budget_s * 0.93, rng,
                                t0_val=0.015, t1_val=0.0002, max_iters=100000)
        h_g2, c5_g2 = greedy_search(h_s2, t0, t0 + budget_s * 0.94, rng, 500, 400)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    # Phase 5: Continuous polish
    if time.time() < t0 + budget_s * 0.95:
        h_c, c5_c = continuous_polish(best_h, t0, t0 + budget_s * 0.99)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5_val(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
