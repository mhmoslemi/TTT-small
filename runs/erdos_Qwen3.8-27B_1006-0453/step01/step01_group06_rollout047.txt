    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    return h.sum() - len(h) / 2.0

    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def active_smooth_c5(h, active_idx, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ac = corr[active_idx]
    m = np.max(ac)
    val = m + np.log(np.sum(np.exp(T * (ac - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def penalized_c5(h, penalty=20000.0):
    n = len(h)
    base = compute_c5(h)
    sp = abs(h.sum() - n / 2.0) * penalty
    bp = (np.sum(np.maximum(0.0, h - 1.0)) + np.sum(np.maximum(0.0, -h))) * penalty
    return base + sp + bp

def get_active_indices(h, top_frac=0.15):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    thresh = m - (m - np.min(corr)) * top_frac
    active = np.where(corr >= thresh)[0]
    if len(active) < 3:
        order = np.argsort(corr)[-5:]
        active = order
    return active

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T (global structure)
    for T in [5, 15, 40, 100, 250, 600, 1500]:
        if elapsed() > budget_s * 0.22:
            break
        cons = [{'type': 'eq', 'fun': sum_constraint}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_c5, best_h, args=(T,),
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-13})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Active-set focused SLSQP (iterate identifying active lags)
    for iteration in range(12):
        if elapsed() > budget_s * 0.42:
            break
        active_idx = get_active_indices(best_h, top_frac=0.12)
        T = 100 + iteration * 80
        cons = [{'type': 'eq', 'fun': sum_constraint}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(active_smooth_c5, best_h, args=(active_idx, T),
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-13})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 3: Nelder-Mead multi-start (derivative-free, handles kinks)
    n_starts = 45
    for i in range(n_starts):
        if elapsed() > budget_s * 0.62:
            break
        # Structured perturbation: focus on coordinates near 0.5
        scale = 0.08 * np.exp(-0.05 * i) + 0.004
        mask = np.abs(best_h - 0.5) < 0.35
        if np.sum(mask) < n // 4:
            mask = np.ones(n, dtype=bool)
        pert = np.zeros(n)
        pert[mask] = np.random.uniform(-scale, scale, size=np.sum(mask))
        h0 = best_h + pert
        h0 = project_feasible(h0, n)
        try:
            res = minimize(penalized_c5, h0, method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-11,
                                   'fatol': 1e-13, 'adapt': True})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 4: SLSQP refinement with high T after NM
    for T in [500, 1500, 3000]:
        if elapsed() > budget_s * 0.78:
            break
        active_idx = get_active_indices(best_h, top_frac=0.10)
        cons = [{'type': 'eq', 'fun': sum_constraint}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(active_smooth_c5, best_h, args=(active_idx, T),
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-14})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Fine Nelder-Mead with very small steps
    for i in range(35):
        if elapsed() > budget_s * 0.90:
            break
        scale = 0.025 * np.exp(-0.07 * i) + 0.001
        h0 = best_h + np.random.uniform(-scale, scale, size=n)
        h0 = project_feasible(h0, n)
        try:
            res = minimize(penalized_c5, h0, method='Nelder-Mead',
                          options={'maxiter': 400, 'xatol': 1e-12,
                                   'fatol': 1e-14, 'adapt': True})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 6: Final SLSQP polish with very high T
    for T in [5000, 10000, 25000]:
        if elapsed() > budget_s * 0.985:
            break
        active_idx = get_active_indices(best_h, top_frac=0.08)
        cons = [{'type': 'eq', 'fun': sum_constraint}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(active_smooth_c5, best_h, args=(active_idx, T),
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-15})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
