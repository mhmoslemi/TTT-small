        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        for kind in range(5):
            if budget_s - (time.time() - t0) < 100:
                break
            hs = project_h(make_structured(n_try, kind, rng), ts_t)
            cs = compute_c5(hs, dx_t)
            if cs < best_c5 + 0.03:
                hr, cr = slsqp_refine(hs, dx_t, ts_t, maxiter=250, nak=12, nout=4)
                if cr < best_c5:
                    best_h, best_c5, best_n = hr, cr, n_try
                elif cr < best_c5 + 0.005 and budget_s - (time.time() - t0) > 80:
                    hr2, cr2 = swap_local_search(hr, dx_t, ts_t, 3, 400)
                    if cr2 < best_c5:
                        best_h, best_c5, best_n = hr2, cr2, n_try

    nb = best_n
    dxb = 2.0 / nb
    tsb = nb / 2.0
    nperts = int(min(30, max(8, (budget_s - (time.time() - t0)) / 30)))
    for r in range(nperts):
        if budget_s - (time.time() - t0) < 45:
            break
        scale = 0.03 + 0.1 * (r % 4)
        hp = project_h(best_h + scale * rng.randn(nb), tsb)
        hr, cr = slsqp_refine(hp, dxb, tsb, maxiter=200, nak=10, nout=3)
        if cr < best_c5:
            best_h, best_c5 = hr, cr
        elif cr < best_c5 + 0.002 and budget_s - (time.time() - t0) > 40:
            hr2, cr2 = swap_local_search(hr, dxb, tsb, 2, 300)
            if cr2 < best_c5:
                best_h, best_c5 = hr2, cr2

    for r in range(10):
        if budget_s - (time.time() - t0) < 40:
            break
        hbv = np.zeros(nb)
        perm = rng.permutation(nb)
        nf = int(tsb)
        hbv[perm[:nf]] = 1.0
        frac = tsb - nf
        if frac > 0.01:
            hbv[perm[nf]] = frac
        hbv = project_h(hbv, tsb)
        hr, cr = slsqp_refine(hbv, dxb, tsb, maxiter=200, nak=10, nout=3)
        if cr < best_c5:
            best_h, best_c5 = hr, cr

    if budget_s - (time.time() - t0) > 100:
        hf, cf = slsqp_refine(best_h, dxb, tsb, maxiter=800, nak=20, nout=10)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    if budget_s - (time.time() - t0) > 60:
        hs, cs = swap_local_search(best_h, dxb, tsb, 5, 800)
        if cs < best_c5:
            best_h, best_c5 = hs, cs
        if cs < best_c5 + 0.001 and budget_s - (time.time() - t0) > 40:
            hf2, cf2 = slsqp_refine(best_h, dxb, tsb, maxiter=500, nak=20, nout=5)
            if cf2 < best_c5:
                best_h, best_c5 = hf2, cf2

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
