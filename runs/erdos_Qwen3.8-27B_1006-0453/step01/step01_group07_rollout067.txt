Actually, a more principled approach: **minimax via bisection on T**.
For a given T, check if there exists h with 0≤h≤1, sum(h)=n/2, and corr(h,1-h)[k] * dx ≤ T for all k. This is a feasibility problem. If feasible, decrease T; else increase T.

The feasibility problem can be written as a linear program (LP) if we relax the bilinear terms. But corr(h,1-h) = corr(h,1) - corr(h,h) = sum terms - autocorrelation. The autocorrelation is quadratic in h, so this is a QCQP (quadratic constraint QP), which is NP-hard in general.

However, for a fixed T, the constraints are:
Σ_i h[i+k](1-h[i]) ≤ T/dx for all k
Σ_i h[i] = n/2
0 ≤ h[i] ≤ 1

The first set of constraints is quadratic (bilinear). This makes it a non-convex problem in general.

But wait: h[i+k](1-h[i]) = h[i+k] - h[i+k]h[i]. The constraint is:
Σ_i h[i+k] - Σ_i h[i+k]h[i] ≤ T/dx

The term Σ_i h[i+k] is linear in h (it's just a sum of h at shifted indices). The term Σ_i h[i+k]h[i] is quadratic. So the constraint is:
(linear) - (quadratic) ≤ T/dx
i.e., (linear) - (T/dx) ≤ (quadratic)

This is a non-convex quadratic constraint (the quadratic form is h^T A h where A is a "shifted" matrix, which is not PSD in general).

So LP bisection won't work directly. But we could use SOCP or SDP relaxations... probably overkill.

import numpy as np
from scipy.optimize import minimize, L_BFGS_B
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_and_grad(c, beta):
    """Smooth max of c (vector) with temperature beta, and gradient wrt c."""
    c_scaled = c / beta
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    obj = beta * (c_max + np.log(sum_exp))
    w = exp_c / sum_exp  # softmax weights
    return obj, w

def full_objective_and_grad(h, n, beta, mu):
    """Objective with sum penalty, and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    
    obj_sm, w = smooth_max_and_grad(c, beta)
    
    # Sum penalty
    s = np.sum(h) - n / 2.0
    obj = obj_sm + mu * s * s
    
    # Gradient of smooth max
    # grad[j] = sum_m w[j-m+n-1] * (1-h[m]) - sum_m w[m-j+n-1] * h[m]
    idx1 = np.arange(n)[:, None] - np.arange(n)[None, :] + (n - 1)
    idx2 = np.arange(n)[None, :] - np.arange(n)[:, None] + (n - 1)
    M1 = w[idx1]
    M2 = w[idx2]
    grad_sm = M1 @ (1.0 - h) - M2 @ h
    
    # Gradient of penalty
    grad_pen = 2.0 * mu * s * np.ones(n)
    
    return obj, grad_sm + grad_pen

def project_h(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2 (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to adjust sum
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # Second pass
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            # Only adjust non-saturated variables
            mask = (h > 1e-9) & (h < 1.0 - 1e-9)
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_and_grad(h, n, beta, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    c_shifted = c - np.max(c)
    exp_c = np.exp(c_shifted / beta)
    sum_exp = np.sum(exp_c)
    obj_sm = beta * (np.max(c) / beta + np.log(sum_exp))
    w = exp_c / sum_exp
    s = np.sum(h) - n / 2.0
    obj = obj_sm + mu * s * s
    idx1 = np.arange(n)[:, None] - np.arange(n)[None, :] + (n - 1)
    idx2 = np.arange(n)[None, :] - np.arange(n)[:, None] + (n - 1)
    grad = dx * (w[idx1] @ (1.0 - h) - w[idx2] @ h) + 2.0 * mu * s
    return obj, grad

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v = project_h(v, new_n)
    return v

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_and_grad(h, n, beta, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / beta)
    sum_exp = np.sum(exp_c)
    obj_sm = beta * (c_max / beta + np.log(sum_exp))
    w = exp_c / sum_exp
    s = np.sum(h) - n / 2.0
    obj = obj_sm + mu * s * s
    jj = np.arange(n)
    idx1 = jj[:, None] - jj[None, :] + (n - 1)
    idx2 = jj[None, :] - jj[:, None] + (n - 1)
    grad = dx * (w[idx1] @ (1.0 - h) - w[idx2] @ h)
    grad += 2.0 * mu * s
    return obj, grad

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(4):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    return project_h(v, new_n)

def lbgfs_solve(h, n, beta, mu, maxiter, t0, budget):
    h0 = np.clip(h, 1e-9, 1.0 - 1e-9)
    bounds = [(1e-9, 1.0 - 1e-9)] * n
    try:
        res = minimize(obj_and_grad, h0, args=(n, beta, mu),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-11, 'maxls': 50,
                                'maxfun': maxiter * 60})
        return project_h(res.x, n)
    except Exception:
        return project_h(h0, n)

def gradient_local_search(h, n, t0, budget, max_iter=5000):
    cur = compute_c5(h)
    beta_small = 0.001
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for it in range(max_iter):
        if time.time() - t0 > budget - 3:
            break
        w_curr = 0.001 if it % 50 == 0 else 0.0
        _, grad = obj_and_grad(h, n, beta_small, w_curr)
        g_pos = grad.copy()
        g_pos[h >= 1.0 - 1e-7] = np.inf
        g_neg = grad.copy()
        g_neg[h <= 1e-7] = -np.inf
        j = int(np.argmax(g_neg))
        i = int(np.argmin(g_pos))
        if i == j:
            continue
        if g_neg[j] - g_pos[i] < 1e-9:
            continue
        for scale in step_sizes:
            d = scale
            h2 = h.copy()
            h2[i] = np.clip(h[i] + d, 0.0, 1.0)
            h2[j] = np.clip(h[j] - d, 0.0, 1.0)
            h2 = project_h(h2, n)
            c = compute_c5(h2)
            if c < cur - 1e-14:
                h, cur = h2, c
                break
    return h, cur

    t0 = time.time()
    rng = np.random.RandomState(seed)
    parent = np.asarray(initial_h_values, dtype=np.float64)

    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
