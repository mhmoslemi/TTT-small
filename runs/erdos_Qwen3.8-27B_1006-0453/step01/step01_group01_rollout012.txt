from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def obj_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mx = np.max(corr)
    s = t * (corr - mx)
    s = np.clip(s, -700, 700)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

    """Perturbation-based local search"""
    n = len(h)
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(n_iter):
        # Random perturbation
        idx = rng.randint(n, size=rng.randint(1, 5))
        cand = best.copy()
        cand[idx] += rng.randn(len(idx)) * step
        cand = proj(cand, n)
        c = compute_c5(cand)
        if c < best_c - 1e-10:
            best = cand
            best_c = c
    return best, best_c

    mx = np.max(corr)
    s = t * (corr - mx)
    s = np.clip(s, -700, 700)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break

            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

    """One projected subgradient step to reduce max correlation peak."""
    corr = compute_corr(h)
    k_star = int(np.argmax(corr))
    # h_star index in correlate output: k_star corresponds to lag (k_star - n + 1)
    lag = k_star - n + 1
    # Gradient of corr[k_star] w.r.t. h[j]
    # corr[k_star] = sum_i h[i] * (1-h[i+lag]) for valid (i, i+lag)
    # d/dh[j] = sum_i [(1-h[i+lag])*δ(i,j) + h[i]*(-1)*δ(i+lag,j)]
    # = (1-h[j+lag]) - h[j-lag] (where defined)
    grad = np.zeros(n)
    if lag >= 0:
        # i ranges over [0, n-1-lag]
        for j in range(n - lag):
            grad[j] += (1.0 - h[j + lag])
        for j in range(lag, n):
            grad[j] -= h[j - lag]
    else:
        l = -lag
        for j in range(l, n):
            grad[j] += (1.0 - h[j - l])
        for j in range(0, n - l):
            grad[j] -= h[j + l]
    # Project gradient onto sum=0 subspace
    grad -= np.mean(grad)
    h_new = h - alpha * grad
    return proj(h_new, n)

        if c < best_c - 1e-11:
            best = cand
            best_c = c
    return best, best_c

def anneal(h, rng, n_iter, t_init):
    n = len(h)
    best = h.copy()
    best_c = compute_c5(best)
    cur = h.copy()
    cur_c = best_c
    temp = t_init
    for _ in range(n_iter):
        temp *= 0.997
        np_ = rng.randint(1, 7)
        idx = rng.randint(n, size=np_)
        cand = cur.copy()
        cand[idx] += rng.randn(np_) * 0.02
        cand = proj(cand, n)
        c = compute_c5(cand)
        d = c - cur_c
        if d < 0 or rng.rand() < np.exp(-d / max(temp, 1e-8)):
            cur = cand
            cur_c = c
            if c < best_c:
                best = cand
                best_c = c
    return best, best_c
