import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def lse_gradient(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = corr - corr.max()
    w = np.exp(beta * cs)
    w /= w.sum()
    
    n2 = 2 * n - 1  # length of corr
    g = np.zeros(n)
    for i in range(n):
        # First term: sum_k w[k]*(1-h[i-k]) for k where 0<=i-k<n
        # i.e., k in [max(0, i-n+1), min(n2-1, i)]
        k_lo = max(0, i - n + 1)
        k_hi = min(n2 - 1, i)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            js = i - ks
            g[i] += np.dot(w[ks], 1.0 - h[js])
        # Second term: sum_k w[k]*h[i+k] for k where 0<=i+k<n
        # i.e., k in [0, min(n2-1, n-1-i)]
        k_hi2 = min(n2 - 1, n - 1 - i)
        if k_hi2 >= 0:
            ks = np.arange(0, k_hi2 + 1)
            js = i + ks
            g[i] -= np.dot(w[ks], h[js])
    
    return g * dx

def lse_value(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cs = corr - corr.max()
    return float(corr.max() + np.log(np.exp(beta * cs).sum()) / beta)

def projected_grad_descent(h0, dx, target_sum, beta_start=5, beta_end=50, 
                           n_iter=2000, lr_start=0.05, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iter):
        if t_deadline is not None and time.time() > t_deadline:
            break
        frac = it / max(n_iter, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start / (1.0 + 0.01 * it)
        
        g = lse_gradient(h, dx, beta)
        
        # Backtracking line search
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
                break
    
    return best_h, best_c5

def slsqp_solve(h0, dx, target_sum, maxiter=300):

    n2 = 2 * n - 1
    
    def obj(x):
        return x[n]
    
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    
    t0 = compute_c5(h0, dx) / dx

    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]

    try:

    except Exception:

    return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def lse_gradient(h, dx, beta):
    n = len(h)
    n2 = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = corr - corr.max()
    w = np.exp(beta * cs)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(n):
        k_lo = max(0, i - n + 1)
        k_hi = min(n2 - 1, i)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            js = i - ks
            g[i] += np.dot(w[ks], 1.0 - h[js])
        k_hi2 = min(n2 - 1, n - 1 - i)
        if k_hi2 >= 0:
            ks = np.arange(0, k_hi2 + 1)
            js = i + ks
            g[i] -= np.dot(w[ks], h[js])
    return g * dx

def lse_grad_descent(h0, dx, target_sum, beta_start=5, beta_end=60,
                     n_iter=2000, lr_start=0.04, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stagnant = 0
    for it in range(n_iter):
        if t_deadline is not None and time.time() > t_deadline:
            break
        frac = min(it / max(n_iter, 1), 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start / (1.0 + 0.02 * it)
        g = lse_gradient(h, dx, beta)
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
        if not improved:
            stagnant += 1
        else:
            stagnant = 0
        if stagnant > 200:
            break
    return best_h, best_c5

def slsqp_active(h0, dx, target_sum, maxiter=300, n_active=8, tol=0.03):
    n = len(h0)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    corr_max = corr.max()
    active = np.where(corr >= corr_max - tol)[0]
    n_c = len(active)
    obj = lambda x: x[n]
    ineq = lambda x: x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')[active]
    eqf = lambda x: np.sum(x[:n]) - target_sum
    t0 = corr_max + 0.02
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.abs(res.x[:n] - np.clip(res.x[:n],0,1)) < 1e-6):
            h_new = project_sr(res.x[:n], target_sum)
            if compute_c5(h_new, dx) <= compute_c5(h0, dx) + 1e-10:
                return h_new
    except Exception:
        pass
    return h0.copy()

def coordinate_descend(h0, dx, target_sum, n_passes=30, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    for p in range(n_passes):
        if t_deadline is not None and time.time() > t_deadline:
            break
        for i in range(n):
            h[i] = 0.0
            base = np.correlate(h, 1.0 - h, mode='full')
            base[active_idx] = base[active_idx]  # placeholder
            h[i] = 1.0
            full = np.correlate(h, 1.0 - h, mode='full')
            a_k = full - base  # linear coefficient
            b_k = base         # constant part
            xs = np.linspace(0.0, 1.0, 201)
            vals = b_k[None, :] + a_k[None, :] * xs[:, None]
            max_vals = vals.max(axis=1)
            # Also handle k=0 quadratic: c_0 = b_0 + x*(1-x) when the x term is special
            best_x = xs[np.argmin(max_vals)]
            h[i] = best_x
            h = project_sr(h, target_sum)
            c5_now = compute_c5(h, dx)
            if c5_now < best_c5:
                best_c5 = c5_now
            active_idx = None
    return h, best_c5

def coord_descend_simple(h0, dx, target_sum, n_passes=30, t_deadline=None):
    n = len(h0)
    n2 = 2 * n - 1
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    for p in range(n_passes):
        if t_deadline is not None and time.time() > t_deadline:
            break
        for i in range(n):
            h[i] = 0.0
            base = np.correlate(h, 1.0 - h, mode='full').copy()
            h[i] = 1.0
            full = np.correlate(h, 1.0 - h, mode='full')
            a_k = full - base
            xs = np.linspace(0.0, 1.0, 101)
            objective = base[None, :] + a_k[None, :] * xs[:, None]
            max_vals = objective.max(axis=1)
            best_x = float(xs[np.argmin(max_vals)])
            h[i] = best_x
            h = project_sr(h, target_sum)
    c5_final = compute_c5(h, dx)
    h_out = h if c5_final < best_c5 else h0.copy()
    c5_out = min(c5_final, best_c5)
    return h_out, c5_out

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)
