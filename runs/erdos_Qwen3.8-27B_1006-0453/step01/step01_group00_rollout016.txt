                c2 = compute_c5(h2)
                if c2 < c_curr - 1e-13:
                    h = h2
                    c_curr = c2
                    improved = True
                    break
                h[i] -= d
        if not improved:
            break
    return h

def make_diverse_start(n, variant):
    x = np.linspace(0, 2, n, endpoint=False)
    if variant == 0:
        h = 0.5 + 0.25 * np.sin(np.pi * x)
    elif variant == 1:
        h = 0.5 + 0.2 * np.cos(2 * np.pi * x)
    elif variant == 2:
        h = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(np.pi * x), 0, 1)
    else:
        h = np.clip(0.5 + 0.35 * np.exp(-2.0 * (x - 1.0) ** 2) - 0.1 * np.cos(2 * np.pi * x), 0, 1)
    return fixsum(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP at n_best with fine-grained T schedule
    for T in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
        if budget_s - (time.time() - t0) < 70:
            break
        ho = slsqp(h_best, n_best, T, 700)
        c = compute_c5(ho)
        if c < c5_best:
            h_best, c5_best = ho, c

    # Phase 2: Work at n=70 (smaller is preferred)
    n70 = 70
    h70 = downsample(h_best, n70)
    for T in [0.03, 0.01, 0.004, 0.001]:
        if budget_s - (time.time() - t0) < 50:
            break
        h70 = slsqp(h70, n70, T, 600)
    c70 = compute_c5(h70)
    h70_up = upsample(h70, n_best)
    c70up = compute_c5(h70_up)
    if c70up < c5_best:
        h_best, c5_best = h70_up, c70up
    elif c70 < c5_best:
        h_best, c5_best, n_best = h70, c70, n70

    # Phase 3: Random restarts at current best resolution
    for i in range(14):
        if budget_s - (time.time() - t0) < 130:
            break
        sig = 0.03 + 0.025 * (i % 3)
        hp = np.clip(h_best + np.random.randn(n_best) * sig, 0, 1)
        hp = fixsum(hp, n_best)
        ho = slsqp(hp, n_best, 0.01, 500)
        c = compute_c5(ho)
        if c < c5_best:
            h_best, c5_best = ho, c

    # Phase 4: Coarse resolution with diverse starts
    for nc in [20, 30, 40]:
        if budget_s - (time.time() - t0) < 120:
            break
        for var in range(4):
            if budget_s - (time.time() - t0) < 30:
                break
            hc = make_diverse_start(nc, var)
            hc = slsqp(hc, nc, 0.02, 400)
            hc = slsqp(hc, nc, 0.005, 400)
            hu = upsample(hc, n_best)
            c = compute_c5(hu)
            if c < c5_best:
                h_best, c5_best = hu, c
        # Also optimize downsample of current best
        hc = downsample(h_best, nc)
        hc = slsqp(hc, nc, 0.01, 400)
        # Random at coarse
        for i in range(3):
            if budget_s - (time.time() - t0) < 20:
                break
            hp = np.clip(hc + np.random.randn(nc) * 0.12, 0, 1)
            hp = fixsum(hp, nc)
            hc2 = slsqp(hp, nc, 0.01, 300)
            if compute_c5(hc2) < compute_c5(hc):
                hc = hc2
        hu = upsample(hc, n_best)
        c = compute_c5(hu)
        if c < c5_best:
            h_best, c5_best = hu, c

    # Phase 5: Try n=140 as an intermediate resolution
    n140 = 140
    if n140 != n_best and budget_s - (time.time() - t0) > 80:
        h140 = upsample(h_best, n140) if n_best < n140 else downsample(h_best, n140)
        for T in [0.01, 0.004, 0.001]:
            if budget_s - (time.time() - t0) < 30:
                break
            h140 = slsqp(h140, n140, T, 500)
        c140 = compute_c5(h140)
        if c140 < c5_best:
            if n140 < n_best or True:
                h_best, c5_best, n_best = h140, c140, n140

    # Phase 6: Higher resolution if time permits
    if n_best < 400 and budget_s - (time.time() - t0) > 70:
        nh = 400
        hh = upsample(h_best, nh)
        for T in [0.01, 0.003, 0.001]:
            if budget_s - (time.time() - t0) < 30:
                break
            hh = slsqp(hh, nh, T, 600)
        c_hh = compute_c5(hh)
        if c_hh < c5_best:
            h_best, c5_best, n_best = hh, c_hh, nh

    # Phase 7: Coordinate descent
    if budget_s - (time.time() - t0) > 25:
        nc_iter = 20 if n_best <= 140 else 10
        hc = coord_descend(h_best, n_best, nc_iter)
        c = compute_c5(hc)
        if c < c5_best:
            h_best, c5_best = hc, c

    # Phase 8: Final SLSQP polish
    if budget_s - (time.time() - t0) > 15:
        ho = slsqp(h_best, n_best, 0.0004, 1500)
        c = compute_c5(ho)
        if c < c5_best:
            h_best, c5_best = ho, c

    # Validate and return
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fixsum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
