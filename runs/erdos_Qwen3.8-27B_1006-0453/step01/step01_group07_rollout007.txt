        if n_try < n:
            h_start = downsample(best_h, n_try)
        else:
            h_start = upsample(best_h, n_try)
        h_opt = slsqp_optimize(h_start, n_try,
                               [0.02, 0.01, 0.005, 0.002, 0.001],
                               350, 1e-13, time_left, 30)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n_try
        if time_left() > 30:
            h_opt, c5 = local_search(h_opt, best_n, rng, time_left,
                                     min(60, time_left() - 15))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = best_n

    # Phase 7: Final polishing
    if time_left() > 20:
        hn = best_n
        h, cur = local_search(best_h.copy(), hn, rng, time_left,
                             min(60, time_left() - 5))
        if cur < best_c5:
            best_c5 = cur
            best_h = h
    if time_left() > 15:
        hn = best_n
        h, cur = block_search(best_h.copy(), hn, rng, time_left,
                             min(40, time_left() - 5))
        if cur < best_c5:
            best_c5 = cur
            best_h = h

    # Ensure exact sum and bounds
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h += (best_n / 2.0 - np.sum(best_h)) / best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
