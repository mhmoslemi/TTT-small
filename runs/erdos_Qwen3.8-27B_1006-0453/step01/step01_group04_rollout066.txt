import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = (cmax + temp * np.log(np.sum(np.exp(shifted / temp))))
    # constraint penalty
    integ = np.sum(h) * dx
    pen = 1000.0 * (integ - 1.0)**2
    return sm + pen

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def optimize_h(h_in, t0, budget_frac):
    n = len(h_in)
    cur = h_in.copy()
    best = compute_c5(cur)
    bounds = [(0.0, 1.0)] * n
    t_end = t0 + budget_frac
    
    schedules = [
        (0.05, 300, 1e-12, 1e-7),
        (0.015, 300, 1e-12, 1e-8),
        (0.005, 400, 1e-12, 1e-9),
        (0.001, 500, 1e-13, 1e-10),
    ]
    
    for temp, maxit, ftol, gtol in schedules:
        if time.time() > t_end:
            break
        res = minimize(soft_c5, cur, args=(temp,), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxit, 'ftol': ftol, 'gtol': gtol})
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best:
            best = c5v
            cur = nh
        else:
            cur = nh  # keep going from here
    
    return cur, best

def perturb_fourier(h, freq, amp, n_harmonics=1):
    n = len(h)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    noise = np.zeros(n)
    for k in range(1, n_harmonics + 1):
        m = freq * k
        phase = np.random.uniform(0, 2*np.pi)
        noise += amp * np.sin(2*np.pi*m*x + phase)
    return repair_h(h + noise)

def perturb_squeeze(h, alpha):
    return repair_h((h - 0.5) * alpha + 0.5)

def perturb_gaussian(h, sigma):
    return repair_h(h + np.random.randn(len(h)) * sigma)

    m = 2 * n - 1
    # next power of 2 >= m
    N = 1
    while N < m:
        N *= 2
    corr = np.real(np.ifft(np.fft.fft(h, N) * np.fft.fft(one_minus_h, N)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_func(h, temp, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    sh = corr - cmax
    val = cmax + temp * np.log(np.sum(np.exp(sh / temp)))
    integ = np.sum(h) * dx
    val += pen * (integ - 1.0) ** 2
    return val

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n, 0.5)
    h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def upscale(h, f):
    n = len(h)
    m = n * f
    xo = (np.arange(n) + 0.5) * 2.0 / n
    xn = (np.arange(m) + 0.5) * 2.0 / m
    return np.interp(xn, xo, h)

def optimize_full(h0, t_end):
    n = len(h0)
    cur = h0.copy()
    best_c = compute_c5(cur)
    bd = [(0.0, 1.0)] * n
    phases = [(0.03, 200, 1e-11, 1e-7), (0.008, 300, 1e-11, 1e-8),
              (0.002, 400, 1e-12, 1e-9), (0.0005, 500, 1e-12, 1e-10)]
    for temp, mi, ftol, gtol in phases:
        if time.time() >= t_end:
            break
        r = minimize(obj_func, cur, args=(temp, 200.0), method='L-BFGS-B',
                     bounds=bd, options={'maxiter': mi, 'ftol': ftol, 'gtol': gtol})
        nh = np.clip(r.x, 0.0, 1.0)
        nh = repair(nh)
        c = compute_c5(nh)
        cur = nh
        if c < best_c:
            best_c = c
    return cur, best_c

def optimize_short(h0, t_end):
    n = len(h0)
    cur = h0.copy()
    best_c = compute_c5(cur)
    bd = [(0.0, 1.0)] * n
    phases = [(0.02, 150, 1e-10, 1e-6), (0.005, 250, 1e-11, 1e-7),
              (0.001, 350, 1e-11, 1e-8)]
    for temp, mi, ftol, gtol in phases:
        if time.time() >= t_end:
            break
        r = minimize(obj_func, cur, args=(temp, 200.0), method='L-BFGS-B',
                     bounds=bd, options={'maxiter': mi, 'ftol': ftol, 'gtol': gtol})
        nh = np.clip(r.x, 0.0, 1.0)
        nh = repair(nh)
        c = compute_c5(nh)
        cur = nh
        if c < best_c:
            best_c = c
    return cur, best_c

def make_perturb(h, rng):
    n = len(h)
    r = rng.random()
    if r < 0.15:
        return repair(h + rng.randn(n) * 0.05)
    elif r < 0.30:
        x = (np.arange(n) + 0.5) * 2.0 / n
        m = rng.randint(1, 5)
        ph = rng.uniform(0, 2 * np.pi)
        return repair(h + 0.04 * np.sin(2 * np.pi * m * x + ph))
    elif r < 0.42:
        a = 1.15 + 0.3 * rng.random()
        return repair((h - 0.5) * a + 0.5)
    elif r < 0.55:
        k = rng.randint(-n // 5, n // 5 + 1)
        return repair(np.roll(h, k))
    elif r < 0.70:
        k = rng.randint(-6, 7)
        return repair(np.roll(h, k) + rng.randn(n) * 0.03)
    else:
        h2 = h.copy()
        idx = rng.choice(n, size=n // 10, replace=False)
        h2[idx] = np.clip(h2[idx] + rng.randn(len(idx)) * 0.1, 0.0, 1.0)
        return repair(h2)

def local_polish(h, t_end, sigma=0.01):
    n = len(h)
    best_c = compute_c5(h)
    cur = h.copy()
    rng = np.random.RandomState(42)
    order = rng.permutation(n)
    for i in order:
        if time.time() >= t_end:
            break
        for delta in [sigma, -sigma, 2*sigma, -2*sigma]:
            trial = cur.copy()
            trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
            c = compute_c5(trial)
            if c < best_c:
                best_c = c
                cur[i] = trial[i]
    return cur, best_c

    np.random.seed(seed)

    nW = len(bh)

        hf = upscale(bh, 2)
        hf, cf = optimize_full(hf, t0 + int(budget_s * 0.98))
        if cf < bc:

        else:
            # Keep the 720 version

    bh = repair(bh)
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    return bh, bc, len(bh)
