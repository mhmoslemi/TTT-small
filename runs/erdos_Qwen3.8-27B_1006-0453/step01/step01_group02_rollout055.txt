        if h_bin[i] > 0:
            j = rng.randint(N)
            if h_bin[j] > 0:
                continue
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        else:
            j = rng.randint(N)
            if h_bin[j] == 0:
                continue
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        c5_new = compute_c5(h_bin, dx)
        if c5_new < c5:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return best_h, best_c5

def coord_polish(h, N, target, dx, deadline, sweeps=3):
    best_c5 = compute_c5(h, dx)
    for sw in range(sweeps):
        if time.time() > deadline:
            break
        for j in range(N):
            if time.time() > deadline:
                break
            old = h[j]
            cands = [0.0, 1.0, max(0.0, old - 0.25), min(1.0, old + 0.25),
                     max(0.0, old - 0.1), min(1.0, old + 0.1)]
            best_v = old
            best_c = compute_c5(h, dx)
            for cv in cands:
                if abs(cv - old) < 1e-6:
                    continue
                h[j] = cv
                h_p = project_sum(h, target)
                hc = compute_c5(h_p, dx)
                if hc < best_c - 1e-12:
                    best_c = hc
                    best_v = cv
                    h = h_p.copy()
            h[j] = best_v
            h = project_sum(h, target)
            if best_c < best_c5:
                best_c5 = best_c
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)

    # Phase 1: Direct steepest descent (K=1)
    dl = t0 + budget_s * 0.25
    h1, c51 = optimize_sd(h0.copy(), n0, target0, dx0, dl, rng, K=1, max_iter=200000)
    if c51 < best_c5:
        best_c5, best_h = c51, h1

    # Phase 2: Steepest with K=5 (broader view)
    if time.time() < t0 + budget_s * 0.4:
        dl = t0 + budget_s * 0.42
        h2, c52 = optimize_sd(best_h.copy(), n0, target0, dx0, dl, rng, K=5, max_iter=150000)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # Phase 3: Binary local search
    if time.time() < t0 + budget_s * 0.55:
        hb = to_binary(best_h, target0)
        dl = t0 + budget_s * 0.55
        hb2, c5b = binary_local_search(hb, n0, target0, dx0, dl, rng, max_iter=800000)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb2
        # Also try binary from random init
        hrand = np.zeros(n0)
        ones = rng.choice(n0, size=int(target0), replace=False)
        hrand[ones] = 1.0
        hrand_b = to_binary(hrand, target0)
        if time.time() < t0 + budget_s * 0.55:
            hb3, c5b3 = binary_local_search(hrand_b, n0, target0, dx0, dl, rng, max_iter=400000)
            if c5b3 < best_c5:
                best_c5, best_h = c5b3, hb3

    # Phase 4: Steepest descent again from best
    if time.time() < t0 + budget_s * 0.7:
        dl = t0 + budget_s * 0.7
        h4, c54 = optimize_sd(best_h.copy(), n0, target0, dx0, dl, rng, K=1, max_iter=150000)
        if c54 < best_c5:
            best_c5, best_h = c54, h4

    # Phase 5: Coordinate polish
    if time.time() < t0 + budget_s * 0.85:
        dl = t0 + budget_s * 0.85
        h5, c55 = coord_polish(best_h.copy(), n0, target0, dx0, dl, sweeps=5)
        if c55 < best_c5:
            best_c5, best_h = c55, h5

    # Phase 6: Try N=2*n0
    N2 = 2 * n0
    if N2 <= 400 and time.time() < t0 + budget_s * 0.85:
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2i = np.interp(x2, x0, best_h)
        h2i = project_sum(h2i, N2 / 2.0)
        dl = t0 + budget_s * 0.95
        h6, c56 = optimize_sd(h2i, N2, N2 / 2.0, 2.0 / N2, dl, rng, K=3, max_iter=100000)
        if c56 < best_c5:
            best_c5, best_h = c56, h6

    # Final polish
    if time.time() < t0 + budget_s - 5:
        Nf = len(best_h)
        dx_f = 2.0 / Nf
        dl = t0 + budget_s - 2
        hf, c5f = optimize_sd(best_h.copy(), Nf, Nf / 2.0, dx_f, dl, rng, K=1, max_iter=30000)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf
        hp, c5p = coord_polish(best_h.copy(), len(best_h), len(best_h) / 2.0, 2.0 / len(best_h), dl, sweeps=3)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
