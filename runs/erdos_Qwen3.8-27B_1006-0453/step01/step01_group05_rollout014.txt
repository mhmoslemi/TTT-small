def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def logsumexp(obj, temp):
    m = np.max(obj)
    if temp < 1e-10:
        return m
    return m + temp * np.log(np.sum(np.exp((obj - m) / temp))) / len(obj)

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return dx * logsumexp(corr, temp)

def repair(h, n):

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:

        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf == 0:
            h = np.clip(h + d / n, 0, 1)
        else:

    h = h.astype(float).copy()

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if temp < 1e-10:
        return m * dx
    return (m + temp * np.log(np.mean(np.exp((corr - m) / temp)))) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf == 0:
            h = np.clip(h + d / n, 0, 1)
        else:
            h[free] = np.clip(h[free] + d / nf, 0, 1)
    return h

def project_sum_box(h, target):
    h = h.astype(float).copy()
    lo, hi = -max(np.sum(h), target) - 1, max(target, np.sum(h)) + 1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def slsqp_run(x0, n, temp, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    res = minimize(smooth_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def coord_descent(h, n, max_rounds=10):
    target = n / 2.0
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            best_v = old
            for t in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(t - old) < 1e-10:
                    continue
                h[i] = t
                h = project_sum_box(h, target)
                c = compute_c5(h, n)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = t
                    improved = True
            if best_v != old:
                h[i] = best_v
                h = project_sum_box(h, target)
        if not improved:
            break
    return h, best_c

    k_star = int(np.argmax(corr))
    grad = np.zeros(n)
    for j in range(n):

        # As h[j] factor: i+k_off=j → i=j-k_off, need 0<=j-k_off<n and 0<=j<n
