def soft_c5(h, T):

    dx = 2.0 / n

    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

            return h

    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h = h * (1.0 + diff / total)

    s = h.sum() - len(h) / 2.0

```python
def solve_1d_minimax(a, c, x_lo=0.0, x_hi=1.0, tol=1e-10):
    """Minimize max_k (a[k] + c[k]*x) for x in [x_lo, x_hi]."""
    # Golden section search for convex function
    phi = (1 + np.sqrt(5)) / 2
    inv_phi = 1.0 / phi
    inv_phi2 = inv_phi * inv_phi
    
    a_gs = x_lo + inv_phi2 * (x_hi - x_lo)
    b_gs = x_lo + inv_phi * (x_hi - x_lo)
    fa = np.max(a + c * a_gs)
    fb = np.max(a + c * b_gs)
    
    for _ in range(80):
        if x_hi - x_lo < tol:
            break
        if fa < fb:
            x_hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = x_lo + inv_phi2 * (x_hi - x_lo)
            fa = np.max(a + c * a_gs)
        else:
            x_lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = x_lo + inv_phi * (x_hi - x_lo)
            fb = np.max(a + c * b_gs)
    
    x_star = (x_lo + x_hi) / 2
    return x_star, np.max(a + c * x_star)
```
