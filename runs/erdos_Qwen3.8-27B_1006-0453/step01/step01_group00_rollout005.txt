def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50*T, 0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    # Distribute among interior points
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)

        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:

        if diff > 0:

            idx = np.argsort(h)  # smallest first
            for i in idx:

                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-12:

        else:

            idx = np.argsort(-h)  # largest first

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def do_slsqp(h, n, T, max_iter):
    try:
        res = opt.minimize(
            soft_max_obj, h.copy(), args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-16}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_nelder_mead(h, n, max_iter=300):
    def obj(x):
        x_c = fix_sum(x, n)
        return compute_c5(x_c)
    x0 = h.copy()
    res = opt.minimize(obj, x0, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'fatol': 1e-14,
                                'xatol': 1e-12, 'adaptive': True})
    if hasattr(res, 'x') and res.x is not None:
        return fix_sum(res.x, n)
    return h

def active_shift_refine(h, n, T, max_iter=500):
    """Refine by targeting the top active shifts more aggressively"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-5:]
    def topk_obj(x, _n, _Tk, _topk):
        c = np.correlate(x, 1.0 - x, mode='full')
        top_vals = c[_topk]
        t = _Tk
        s = np.exp(top_vals / t)
        sm = np.sum(s)
        return float(np.max(top_vals) + t * np.log(sm / len(top_vals)))
    res = opt.minimize(
        topk_obj, h.copy(), args=(n, T, top_k),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': 1e-15}
    )
    if hasattr(res, 'x') and res.x is not None:
        return fix_sum(res.x, n)
    return h
