            break
        h2, _ = subgradient_step(cur, lr)
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = repair_h(h2)
        c5v = compute_c5(h2)
        if c5v < best:
            best = c5v
            cur = h2
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-5:
                    break
    return cur, best

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def try_restarts(cur_h, n, best_c5, best_h, t0, budget_s, n_restarts=4):
    rng = np.random.RandomState(12345)
    sigmas = [0.2, 0.12, 0.08, 0.05]
    for ri in range(n_restarts):
        if time.time() - t0 > budget_s * 0.9:
            break
        sigma = sigmas[ri % len(sigmas)]
        noise = rng.standard_normal(n) * sigma
        h_pert = np.clip(cur_h + noise, 0.0, 1.0)
        h_pert = repair_h(h_pert)
        max_t = min(budget_s * 0.06, (budget_s * 0.88 - (time.time() - t0)))
        if max_t < 5:
            break
        h_opt, local_best = optimize_lgbfgs(h_pert, n, t0, budget_s, max_t,
                                            schedules=[(0.02, 2000.0, 300), (0.003, 10000.0, 300)])
        # Also try binary snap version
        h_bin = binary_snap(h_opt, 0.12)
        h_opt2, local_best2 = optimize_lgbfgs(h_bin, n, t0, budget_s, max_t * 0.4,
                                              schedules=[(0.01, 5000.0, 250)])
        if local_best < best_c5:
            best_c5 = local_best
            best_h = h_opt.copy()
        if local_best2 < best_c5:
            best_c5 = local_best2
            best_h = h_opt2.copy()
        cur_h = best_h if best_h is not None else cur_h
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at native resolution
    n1 = len(h_init)
    cur_h, local_best = optimize_lgbfgs(h_init, n1, t0, budget_s, budget_s * 0.18,
                                         schedules=[(0.06, 500.0, 500), (0.02, 2000.0, 600),
                                                   (0.004, 8000.0, 700), (0.001, 30000.0, 500)])
    if local_best < best_c5:
        best_c5 = local_best
        best_h = cur_h.copy()
    
    # Phase 2: Binary snap variants at native resolution
    for tau in [0.15, 0.1, 0.2]:
        if time.time() - t0 > budget_s * 0.3:
            break
        h_bin = binary_snap(cur_h, tau)
        h_opt, lb = optimize_lgbfgs(h_bin, n1, t0, budget_s, budget_s * 0.06,
                                     schedules=[(0.02, 2000.0, 400), (0.003, 8000.0, 400)])
        if lb < best_c5:
            best_c5 = lb
            best_h = h_opt.copy()
            cur_h = h_opt.copy()
    
    # Phase 3: Subgradient refinement at native resolution
    if time.time() - t0 < budget_s * 0.35:
        h_sub, lb_sub = subgradient_refine(cur_h, t0, budget_s * 0.06, n_iter=1000)
        if lb_sub < best_c5:
            best_c5 = lb_sub
            best_h = h_sub.copy()
        # Polish the subgradient result
        if time.time() - t0 < budget_s * 0.42:
            h_pol, lb_pol = optimize_lgbfgs(h_sub, n1, t0, budget_s, budget_s * 0.04,
                                             schedules=[(0.003, 8000.0, 400), (0.001, 20000.0, 300)])
            if lb_pol < best_c5:
                best_c5 = lb_pol
                best_h = h_pol.copy()
    
    # Phase 4: Upscale to 2x and optimize
    if time.time() - t0 < budget_s * 0.5:
        h_up = interp_upscale(best_h, 2)
        h_up = repair_h(h_up)
        n2 = len(h_up)
        cur2, lb2 = optimize_lgbfgs(h_up, n2, t0, budget_s, budget_s * 0.2,
                                     schedules=[(0.03, 1000.0, 400), (0.01, 5000.0, 500),
                                               (0.002, 20000.0, 400)])
        if lb2 < best_c5:
            best_c5 = lb2
            best_h = cur2.copy()
        # Binary snap at 2x
        if time.time() - t0 < budget_s * 0.65:
            h_bin2 = binary_snap(cur2, 0.1)
            cur2b, lb2b = optimize_lgbfgs(h_bin2, n2, t0, budget_s, budget_s * 0.08,
                                           schedules=[(0.01, 5000.0, 300), (0.002, 15000.0, 350)])
            if lb2b < best_c5:
                best_c5 = lb2b
                best_h = cur2b.copy()
        # Subgradient at 2x
        if time.time() - t0 < budget_s * 0.75:
            h_sub2, lb_sub2 = subgradient_refine(cur2, t0, budget_s * 0.05, n_iter=600)
            if lb_sub2 < best_c5:
                best_c5 = lb_sub2
                best_h = h_sub2.copy()
    
    # Phase 5: Random restarts at best resolution
    n_final = len(best_h)
    best_h, best_c5 = try_restarts(best_h, n_final, best_c5, best_h, t0, budget_s, n_restarts=5)
    
    # Phase 6: Final polish
    if time.time() - t0 < budget_s * 0.95:
        h_fin, lb_fin = optimize_lgbfgs(best_h, len(best_h), t0, budget_s, budget_s * 0.05,
                                         schedules=[(0.002, 10000.0, 400), (0.0005, 40000.0, 350)])
        if lb_fin < best_c5:
            best_c5 = lb_fin
            best_h = h_fin.copy()
        # Subgradient final
        if time.time() - t0 < budget_s * 0.97:
            h_sg, lb_sg = subgradient_refine(best_h, t0, budget_s * 0.02, n_iter=300)
            if lb_sg < best_c5:
                best_c5 = lb_sg
                best_h = h_sg.copy()
    
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
