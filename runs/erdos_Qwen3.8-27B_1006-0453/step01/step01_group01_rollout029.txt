import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = np.clip(t * (corr - mx), -50, 0)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(100):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def subgrad_of_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    weights = np.exp(40.0 * (corr - mx))
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n), mx
    weights /= wsum
    g = np.zeros(n)
    active = np.where(weights > 1e-5)[0]
    for ki in active:
        w = weights[ki]
        lag = ki - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += w * (1.0 - h[idx + lag])
            g[idx + lag] += -w * h[idx]
        else:
            ll = -lag
            idx = np.arange(n - ll)
            g[idx + ll] += w * (1.0 - h[idx])
            g[idx] += -w * h[idx + ll]
    return g, mx

    bh = h0.copy()
    bc = compute_c5(h0)
    n = len(bh)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}

    def elapsed():
        return time.time() - t0

    # Phase 1: Adaptive temperature annealing SLSQP
    temps = [25, 50, 90, 160, 280, 450]
    for t_val in temps:
        if elapsed() > budget_s * 0.50:
            break
        try:
            r = minimize(smooth_obj, bh, args=(t_val,), method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 1200, 'ftol': 1e-13})
            hc = proj(np.clip(r.x, 0.0, 1.0), n)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc.copy()
        except Exception:
            pass

        elif i % 4 == 2:
