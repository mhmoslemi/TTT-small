import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_h(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    return project_h(v, new_n)

def autocorr(h, n):
    """Compute full autocorrelation: ac[shift] = sum_i h[i]*h[i+shift]"""
    return np.correlate(h, h, mode="full")

def ac_grad(h, shift, n):
    """Gradient of sum_i h[i]*h[i+shift] w.r.t. h"""
    grad = np.zeros(n)
    if shift >= 0:
        a = np.arange(n - shift)
        b = a + shift
        grad[a] += h[b]
        grad[b] += h[a]
    else:
        s = -shift
        a = np.arange(s, n)
        b = a - s
        grad[a] += h[b]
        grad[b] += h[a]
    return grad

def subgrad_ascent(h, n, lr, top_k=5):
    """One subgradient ascent step to maximize min autocorrelation"""
    ac = autocorr(h, n)
    center = n - 1
    # Find top_k smallest autocorrelations
    idx = np.argsort(ac)[:top_k]
    ac_vals = ac[idx]
    # Weight by proximity to minimum (exponential)
    gaps = ac_vals - ac_vals[0]
    w = np.exp(-gaps / 0.01)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        shift = ki - center
        grad += wi * ac_grad(h, shift, n)
    h_new = h + lr * grad
    return project_h(h_new, n)

def obj_soft(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    return dx * (m + beta * np.log(np.sum(np.exp((corr - m) / beta))))
