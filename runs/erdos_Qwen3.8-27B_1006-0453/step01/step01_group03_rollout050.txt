def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0.0, 1.0, out=h)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15

            m = h > 1e-15

        if c == 0:
            break
        h[m] += d / c
    return h

def smooth_obj_grad(h, dx, target_sum, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    exps = np.exp(beta * (corr - m))
    vsum = exps.sum()
    val = m + np.log(vsum) / beta
