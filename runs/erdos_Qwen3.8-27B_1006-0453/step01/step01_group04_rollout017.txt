        cur = do_lbfgs(cur, temp, pen, nit, bounds0)
        cur = repair_h(cur)
        c5v = compute_c5(cur)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = cur.copy()

    # Phase 2: Active-set refinement at n=360
    if time.time() - t0 < budget_s * 0.48:
        for it in range(4):
            if time.time() - t0 > budget_s * 0.48:
                break
            ka = top_k_corr(cur, k=70)
            cur = do_active_lbfgs(cur, ka, 8000, 200, bounds0)
            cur = repair_h(cur)
            c5v = compute_c5(cur)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cur.copy()
            # Small perturbation to escape
            cur = block_perturb(cur, 3, 4, 0.04)

    # Phase 3: Upscale to 720
    n1 = 720
    if time.time() - t0 < budget_s * 0.75:
        cur1 = interp_upscale(best_h, n1)
        cur1 = repair_h(cur1)
        bounds1 = [(0.0, 1.0)] * n1

        schedule3 = [(0.02, 5000, 300), (0.008, 8000, 350), (0.003, 12000, 400),
                     (0.001, 15000, 450), (0.0005, 20000, 500)]
        for i, (temp, pen, nit) in enumerate(schedule3):
            if time.time() - t0 > budget_s * 0.72:
                break
            if i > 0 and i % 2 == 0:
                cur1 = point_perturb(cur1, max(10, n1 // 20), temp * 0.4)
            cur1 = do_lbfgs(cur1, temp, pen, nit, bounds1)
            cur1 = repair_h(cur1)
            c5v = compute_c5(cur1)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cur1.copy()

        # Active set at 720
        if time.time() - t0 < budget_s * 0.82:
            for it in range(3):
                if time.time() - t0 > budget_s * 0.82:
                    break
                ka1 = top_k_corr(cur1, k=90)
                cur1 = do_active_lbfgs(cur1, ka1, 10000, 200, bounds1)
                cur1 = repair_h(cur1)
                c5v = compute_c5(cur1)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = cur1.copy()
                cur1 = block_perturb(cur1, 4, 3, 0.03)

    # Phase 4: Multi-restart at best resolution
    if time.time() - t0 < budget_s * 0.92:
        wn = len(best_h)
        wb = [(0.0, 1.0)] * wn
        for r in range(8):
            if time.time() - t0 > budget_s * 0.93:
                break
            if r % 4 == 0:
                cand = block_perturb(best_h, 5, 5, 0.06)
            elif r % 4 == 1:
                cand = point_perturb(best_h, wn // 15, 0.05)
            elif r % 4 == 2:
                cand = repair_h(np.clip(best_h + np.random.randn(wn) * 0.03, 0, 1))
            else:
                cand = repair_h(np.clip(best_h * (1.0 + np.random.randn(wn) * 0.02), 0, 1))
            cand = do_lbfgs(cand, 0.002, 10000, 120, wb)
            cand = repair_h(cand)
            ka = top_k_corr(cand, k=50)
            cand = do_active_lbfgs(cand, ka, 10000, 80, wb)
            cand = repair_h(cand)
            c5v = compute_c5(cand)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cand.copy()

    # Final validation
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
