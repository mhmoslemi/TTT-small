import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            s = np.sum(h)
            if s > 1e-14:
                h = np.clip(h * (target / s), 0.0, 1.0)
            else:
                h[np.arange(0, n, 2)] = 1.0
        else:
            h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def smooth_obj(x, n, temp):
    h = repair_h(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * temp
    m = np.max(scaled)
    if m > 600:
        return m / temp * dx
    return (np.log(np.sum(np.exp(scaled - m))) + m) / temp * dx


def c5_raw(x, n):
    return compute_c5(repair_h(x, n), n)


def sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    return minimize(smooth_obj, x0, method='SLSQP', bounds=bounds,
                    constraints=cons, args=(n, temp),
                    options={'maxiter': maxiter, 'ftol': 1e-14})

    return minimize(c5_raw, x0, args=(n,), method='Nelder-Mead',

    return repair_h(h, n)
