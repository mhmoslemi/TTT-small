    m = np.max(vals)
    return m + temp * np.log(np.sum(np.exp((vals - m) / temp))) / len(vals)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if m <= 0:
        return 0.0
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))) / len(corr))

def sum_diff(x, n):
    return float(np.sum(x) - n / 2.0)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
            continue
        step = diff / nf
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    return h

def coord_opt_1d(alpha, beta, lo=0.0, hi=1.0):
    """Find x in [lo,hi] minimizing max_m (alpha[m] + beta[m]*x)."""
    # Evaluate at boundaries
    best_x = lo
    best_val = np.max(alpha + beta * lo)
    v1 = np.max(alpha + beta * hi)
    if v1 < best_val:
        best_val = v1
        best_x = hi
    # Check all pairwise crossings
    n_lines = len(alpha)
    for i in range(n_lines):
        for j in range(i + 1, n_lines):

            if abs(db) < 1e-14:
                continue
            xc = (alpha[j] - alpha[i]) / db

                val = np.max(alpha + beta * xc)
                if val < best_val:

                    best_x = xc

    pos_0 = np.where(h == 0.0)[0].copy()

import numpy as np
from scipy.optimize import minimize, basinhopping
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -700, 0)
    return float(m + temp * np.log(np.sum(np.exp(shifted))) / len(corr))

def sum_func(x, n):
    return float(np.sum(x) - n / 2.0)

def make_constraint(n):
    return {'type': 'eq', 'fun': lambda x: sum_func(x, n), 'args': ()}

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
            continue
        step = diff / nf
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, temp, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: sum_func(x, n), 'args': ()}
    try:
        res = minimize(smooth_obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=[cons], args=(n, temp),
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair_h(res.x, n)
    except Exception:
        return repair_h(h0, n)

def local_coord_search(h, n, n_passes=3, deltas=None):
    if deltas is None:
        deltas = [0.05, 0.02, 0.01, 0.08, 0.03]
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for _ in range(n_passes):
        for i in range(n):
            for d in deltas:
                for sign in [1, -1]:
                    h[i] = np.clip(h[i] + sign * d, 0.0, 1.0)
                    h = repair_h(h, n)
                    c5_try = compute_c5(h, n)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                    else:
                        h[i] = np.clip(h[i] - sign * d, 0.0, 1.0)
                        h = repair_h(h, n)
    return h, best_c5

def ba_sa(n, t_hot, t_cold, n_iter, rng):
    half = n // 2
    h = np.zeros(n)
    choices = rng.choice(n, half, replace=False)
    h[choices] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    curr_c5 = best_c5
    pos_1 = np.where(h == 1.0)[0].copy()
    pos_0 = np.where(h == 0.0)[0].copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        t = t_hot * (t_cold / t_hot) ** frac
        i1 = int(rng.integers(len(pos_1)))
        i0 = int(rng.integers(len(pos_0)))
        p1, p0 = int(pos_1[i1]), int(pos_0[i0])
        h[p1], h[p0] = 0.0, 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-14)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            pos_1[i1], pos_0[i0] = p0, p1
        else:
            h[p1], h[p0] = 1.0, 0.0
    return best_h, best_c5

def make_bh_obj(h, n, temp):
    def bh_obj(x):
        xr = repair_h(x, n)
        return smooth_obj(xr, n, temp)
    return bh_obj

def make_bh_acc(n):
    def bh_acc(x, f, T):
        return np.random.uniform(-0.05, 0.05, n)
    return bh_acc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def tl():
        return budget_s - 25.0 - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n0 = len(h0)
    h0 = repair_h(h0, n0)
    best_h, best_c5, best_n = h0.copy(), compute_c5(h0, n0), n0

    # Phase 1: SLSQP annealing from initial
    for temp in [80, 40, 20, 10, 5, 2, 1]:
        if tl() < 20:
            break
        h_t = slsqp_opt(best_h, best_n, temp, 400)
        c5_t = compute_c5(h_t, best_n)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 2: Local coordinate search (fast, no gradients)
    if tl() > 30:
        h_loc, c5_loc = local_coord_search(best_h, best_n, n_passes=5)
        if c5_loc < best_c5:
            best_c5, best_h = c5_loc, h_loc
        h_t = slsqp_opt(best_h, best_n, 5, 200)
        c5_t = compute_c5(h_t, best_n)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 3: Bang-bang SA for even n values
    for n_try in [44, 48, 46, 50]:
        if tl() < 100:
            break
        n_iter = min(100000, max(20000, int(tl() * 0.15 * 1000)))
        h_bb, c5_bb = ba_sa(n_try, 0.04, 0.0003, n_iter, rng)
        if c5_bb < best_c5:
            best_c5, best_h, best_n = c5_bb, h_bb.copy(), n_try
        # Refine bang-bang solution continuously
        if tl() > 20:
            h_cont = slsqp_opt(h_bb, n_try, 10, 200)
            c5_cont = compute_c5(h_cont, n_try)
            if c5_cont < best_c5:
                best_c5, best_h, best_n = c5_cont, h_cont, n_try
            h_loc2, c5_loc2 = local_coord_search(h_bb, n_try, n_passes=3)
            if c5_loc2 < best_c5:
                best_c5, best_h, best_n = c5_loc2, h_loc2, n_try

    # Phase 4: Basinhopping on best solution
    if tl() > 60:
        n_bh = best_n
        obj_bh = make_bh_obj(None, n_bh, 8.0)
        acc_bh = make_bh_acc(n_bh)
        temp_filt = [30.0, 15.0]
        try:
            res_bh = basinhopping(
                lambda x: smooth_obj(repair_h(x, n_bh), n_bh, 8.0),
                best_h.copy(),
                niter=int(min(30, tl() / 5)),
                stepsize=0.06,
                minimizer_kwargs=dict(
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * n_bh,
                    constraints=[{'type': 'eq', 'fun': lambda x: sum_func(x, n_bh), 'args': ()}],
                    options={'maxiter': 150, 'ftol': 1e-12}
                ),
                disp=False
            )
            h_bh = repair_h(res_bh.x, n_bh)
            c5_bh = compute_c5(h_bh, n_bh)
            if c5_bh < best_c5:
                best_c5, best_h = c5_bh, h_bh
        except Exception:
            pass

    # Phase 5: Multi-start with diverse perturbations
    for trial in range(50):
        if tl() < 15:
            break
        if trial % 5 == 4:
            # Bang-bang restart
            h_start = (best_h > 0.5).astype(float)
            h_start = repair_h(h_start, best_n)
        elif trial % 2 == 0:
            scale = 0.12 * np.exp(-trial / 25.0) + 0.01
            h_start = best_h + rng.uniform(-scale, scale, best_n)
            h_start = repair_h(h_start, best_n)
        else:
            # Random smooth pattern
            k = rng.integers(2, 7)
            freq = k * np.pi
            xg = np.linspace(0, 2, best_n, endpoint=False)
            a = rng.uniform(-0.3, 0.3)
            b = rng.uniform(-0.3, 0.3)
            h_start = 0.5 + a * np.cos(freq * xg) + b * np.sin(freq * xg)
            h_start = repair_h(h_start, best_n)

        h_t = slsqp_opt(h_start, best_n, 10, 200)
        c5_t = compute_c5(h_t, best_n)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t
            h_loc3, c5_loc3 = local_coord_search(best_h, best_n, n_passes=2,
                                                 deltas=[0.03, 0.015, 0.05])
            if c5_loc3 < best_c5:
                best_c5, best_h = c5_loc3, h_loc3

    # Phase 6: Final polish
    if tl() > 8:
        for temp in [3, 1, 0.5]:
            if tl() < 5:
                break
            h_t = slsqp_opt(best_h, best_n, temp, 500)
            c5_t = compute_c5(h_t, best_n)
            if c5_t < best_c5:
                best_c5, best_h = c5_t, h_t
        h_loc4, c5_loc4 = local_coord_search(best_h, best_n, n_passes=4,
                                             deltas=[0.01, 0.005, 0.02])
        if c5_loc4 < best_c5:
            best_c5, best_h = c5_loc4, h_loc4

    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
