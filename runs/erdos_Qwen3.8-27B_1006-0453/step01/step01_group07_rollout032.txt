    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr

def soft_max(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def subgrad_step(h, n, alpha, top_k=1):
    """Take a sub-gradient step on the top-k active correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top-k indices
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in idx:
        d = m - (n - 1)  # lag
        # c[n-1+d] = sum_{i valid} h[i]*(1-h[i+d])
        # d c/dc[h[k]] = [k valid as i]*(1-h[k+d]) - [k-d valid as i]*h[k-d]
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)

        # d/dh[k]: if k == i: 1 - h[k+d]
        #           if k == i+d: -h[i] = -h[k-d]
        if 0 <= k < n:  # k as i
            if i_lo <= k <= i_hi:
                j = k + d
                if 0 <= j < n:
                    grad[k] += (1.0 - h[j])
        if 0 <= k - d < n:  # k-d as i, k as i+d
            i = k - d
            if i_lo <= i <= i_hi and 0 <= k < n:
                grad[k] -= h[i]

    return float(np.max(corr)) * dx, corr

def soft_max_val(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(corr))
    shifted = (corr - m) / beta
    s = float(np.sum(np.exp(np.clip(shifted, -60, 60))))
    return dx * (m + beta * np.log(s))

def fix_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

def subgrad_descend(h, n, iterations, alpha0):
    """Sub-gradient descent on the max correlation."""
    h = h.copy()
    alpha = alpha0
    for it in range(iterations):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_star = int(np.argmax(corr))
        d = m_star - (n - 1)
        # Compute gradient of c[n-1+d] w.r.t. h
        grad = np.zeros(n)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)

        s1b = min(i_hi, n - 1)

            jb = s1b + d

                grad[s1a:s1b + 1] += (1.0 - h[ja:jb + 1])

            ib = s2b - d

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr


def soft_max_val(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(corr))
    shifted = (corr - m) / beta
    s = float(np.sum(np.exp(np.clip(shifted, -60.0, 60.0))))
    return dx * (m + beta * np.log(s))


def fix_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

    old_n = len(parent)

    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)


def subgrad_descend(h, n, iterations, alpha0, momentum=0.85):
    h = h.copy()
    alpha = alpha0
    v = np.zeros(n)
    for it in range(iterations):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_star = int(np.argmax(corr))
        d = m_star - (n - 1)
        grad = np.zeros(n)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        s1a = max(i_lo, 0)
        s1b = min(i_hi, n - 1)
        if s1a <= s1b:
            ja = s1a + d
            jb = s1b + d
            if 0 <= ja and jb < n:
                grad[s1a:s1b + 1] += (1.0 - h[ja:jb + 1])
        s2a = max(i_lo + d, 0)
        s2b = min(i_hi + d, n - 1)
        if s2a <= s2b:
            ia = s2a - d
            ib = s2b - d
            grad[s2a:s2b + 1] -= h[ia:ib + 1]
        grad -= np.mean(grad)
        v = momentum * v + (1.0 - momentum) * grad
        h = h - alpha * v
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum(h, n)
        if it % 400 == 399:
            alpha *= 0.82
    return h


def slsqp_obj(x, n, beta):
    return soft_max_val(x, n, beta)


def slsqp_eq(x, n):
    return np.sum(x) - n / 2.0


def slsqp_eq_jac(x, n):
    return np.ones(len(x))


def slsqp_run(h, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, beta), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return fix_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        return h


def lag_obj(x, n, m_star):
    d = m_star - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    s = 0.0
    for i in range(i_lo, i_hi + 1):
        s += x[i] * (1.0 - x[i + d])
    return s


def lag_obj_jac(x, n, m_star):
    d = m_star - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    grad = np.zeros(n)
    s1a = max(i_lo, 0)
    s1b = min(i_hi, n - 1)
    if s1a <= s1b:
        ja = s1a + d
        jb = s1b + d
        if 0 <= ja and jb < n:
            grad[s1a:s1b + 1] += (1.0 - x[ja:jb + 1])
    s2a = max(i_lo + d, 0)
    s2b = min(i_hi + d, n - 1)
    if s2a <= s2b:
        ia = s2a - d
        ib = s2b - d
        grad[s2a:s2b + 1] -= x[ia:ib + 1]
    return grad


def alt_lag_opt(h, n, max_rounds=25):
    h = h.copy()
    best_c5 = compute_c5(h)[0]
    for r in range(max_rounds):
        _, corr = compute_c5(h)
        m_star = int(np.argmax(corr))
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}]
        try:
            res = minimize(lag_obj, h, args=(n, m_star), jac=lag_obj_jac,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-12})
            h_new = fix_sum(np.clip(res.x, 0, 1), n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
            else:
                lag_val = lag_obj(h_new, n, m_star)
                if lag_val < lag_obj(h, n, m_star) - 1e-8:
                    h = h_new
                    c5_chk, _ = compute_c5(h)
                    if c5_chk < best_c5:
                        best_c5 = c5_chk
        except Exception:
            pass
    return h


def coord_descent(h, n, rng, iterations, max_step=0.03):
    h = h.copy()
    cur, _ = compute_c5(h)
    for _ in range(iterations):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-max_step, max_step)
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h[i] = hi
        h[j] = hj
        c_new, _ = compute_c5(h)
        if c_new < cur - 1e-15:
            cur = c_new
        else:
            h[i] -= delta
            h[j] += delta
    return h, cur


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_c5, _ = compute_c5(parent)
    best_h = parent.copy()
    best_n = pn

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP soft-max at parent resolution
    n = pn
    h = parent.copy()
    for beta in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if rem() < 35:
            break
        h = slsqp_run(h, n, beta, maxiter=350)
        c5, _ = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 2: Alternating worst-lag SLSQP
    if rem() > 80:
        h_alt = alt_lag_opt(best_h, best_n, max_rounds=20)
        c5_a, _ = compute_c5(h_alt)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_alt.copy()

    # Phase 3: Sub-gradient descent with momentum
    if rem() > 50:
        h_sg = subgrad_descend(best_h, best_n, iterations=5000, alpha0=0.012, momentum=0.88)
        c5_s, _ = compute_c5(h_sg)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_sg.copy()

    # Phase 4: Coordinate descent refinement
    if rem() > 15:
        h_cd, c5_c = coord_descent(best_h, best_n, rng, 10000, max_step=0.025)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_cd.copy()

    # Phase 5: Higher resolution (2x parent)
    if rem() > 160:
        n2 = min(pn * 2, 560)
        h2 = upsample_to(best_h, n2)
        for beta in [0.01, 0.003, 0.001]:
            if rem() < 35:
                break
            h2 = slsqp_run(h2, n2, beta, maxiter=250)
            c5_2, _ = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2
        if rem() > 50:
            h2 = alt_lag_opt(best_h, best_n, max_rounds=10)
            c5_2, _ = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
        if rem() > 40:
            h2 = subgrad_descend(best_h, best_n, iterations=3000, alpha0=0.008, momentum=0.85)
            c5_2, _ = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
        if rem() > 15:
            h2, c5_2 = coord_descent(best_h, best_n, rng, 5000, max_step=0.015)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 6: Final sub-gradient + coordinate polish on best
    if rem() > 25:
        h_f = subgrad_descend(best_h, best_n, iterations=2000, alpha0=0.004, momentum=0.9)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if rem() > 10:
        h_f, c5_f = coord_descent(best_h, best_n, rng, 5000, max_step=0.01)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if rem() > 20:
        h_f = slsqp_run(best_h, best_n, 0.0003, maxiter=400)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    return best_h, best_c5, best_n
