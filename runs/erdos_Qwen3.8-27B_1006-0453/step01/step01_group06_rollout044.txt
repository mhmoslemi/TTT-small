        try:
            res = minimize(
                lambda x: smooth_c5(x, T), best_h,
                method='SLSQP', bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                options={'maxiter': 400, 'ftol': 1e-14}
            )
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 3: Subgradient descent with backtracking
    if elapsed() < budget_s * 0.55:
        lr = 0.05
        for it in range(5000):
            if elapsed() > budget_s * 0.55:
                break
            grad = smooth_grad(best_h, 500)
            # Backtracking
            for trial in range(20):
                h_new = project_feasible(best_h - lr * grad, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    break
                lr *= 0.5
            else:
                lr = max(lr, 1e-8) * 0.5
                continue
            best_c5 = c5_new
            best_h = h_new
            lr = min(lr * 1.05, 0.1)
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5(best_h)

    # Phase 4: Try different n values via interpolation + optimization
    for n_try in [80, 100, 120, 50, 70]:
        if elapsed() > budget_s * 0.88:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        c5v = compute_c5(h_interp)

        if c5v < best_c5 * 1.02:
            for T, lam in [(100, 50), (500, 20), (2000, 5)]:
                if elapsed() > budget_s * 0.88:
                    break
                try:
                    res = minimize(
                        smooth_pen_grad, h_interp, args=(T, lam),
                        jac=True, method='L-BFGS-B',
                        bounds=[(0.0, 1.0)] * n_try,
                        options={'maxiter': 600, 'ftol': 1e-14, 'gtol': 1e-12}
                    )
                    hc = project_feasible(np.clip(res.x, 0, 1), n_try)
                    c5v2 = compute_c5(hc)
                    if c5v2 < best_c5:
                        best_c5 = c5v2
                        best_h = hc
                        n = n_try
                except Exception:
                    pass

    # Phase 5: Nelder-Mead local perturbations on best
    if elapsed() < budget_s * 0.95:
        n_starts = 25
        for i in range(n_starts):
            if elapsed() > budget_s * 0.95:
                break
            scale = 0.08 * np.exp(-0.06 * i) + 0.01
            h0 = best_h + np.random.uniform(-scale, scale, size=n)
            h0 = project_feasible(h0, n)
            try:
                res = minimize(
                    lambda x: smooth_pen(x, 1000, 50), h0,
                    method='Nelder-Mead',
                    options={'maxiter': 800, 'xatol': 1e-10, 'fatol': 1e-13,
                             'adapt': True}
                )
                hc = project_feasible(np.clip(res.x, 0, 1), n)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 6: Final high-T polish
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.99:
        for T, lam in [(5000, 10), (20000, 5)]:
            if elapsed() > budget_s * 0.995:
                break
            try:
                res = minimize(
                    smooth_pen_grad, best_h, args=(T, lam),
                    jac=True, method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n_cur,
                    options={'maxiter': 800, 'ftol': 1e-15, 'gtol': 1e-13}
                )
                hc = project_feasible(np.clip(res.x, 0, 1), n_cur)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
