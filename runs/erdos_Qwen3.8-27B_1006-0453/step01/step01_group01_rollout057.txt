```python
def compute_corr_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    mx = np.max(corr)
    s = t * (corr - mx)
    exp_s = np.exp(s)
    p = exp_s / np.sum(exp_s)  # softmax weights, length 2n-1
    
    # Compute gradient
    # For lag k (indexed 0 to 2n-2, actual lag = k - (n-1)):
    # d(corr[k])/d(h[m]) = (1-h[m+lag]) * [valid] - h[m-lag] * [valid]
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-15:
            continue
        lag = k - (n - 1)
        w = p[k]
        # First term: h[m] * (1-h[m+lag]) for m where 0<=m<n and 0<=m+lag<n
        # m ranges from max(0, -lag) to min(n-1, n-1-lag)
        m_lo = max(0, -lag)
        m_hi = min(n-1, n-1-lag)
        if m_lo <= m_hi:
            grad[m_lo:m_hi+1] += w * (1.0 - h[m_lo+lag:m_hi+1+lag])
        # Second term: -h[m] * h[m-lag] for m where 0<=m<n and 0<=m-lag<n
        # m ranges from max(0, lag) to min(n-1, n-1+lag)
        m_lo2 = max(0, lag)
        m_hi2 = min(n-1, n-1+lag)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2+1] -= w * h[m_lo2-lag:m_hi2+1-lag]
    
    return corr * dx, mx * dx, grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_gn = 280

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = float(np.sum(m))
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def sg_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def sg_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()
    e = np.exp(s)
    p = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lag = k - (n - 1)
        w = pk * dx
        lo1 = max(0, -lag)
        hi1 = min(n - 1, n - 1 - lag)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += w * (1.0 - h[lo1 + lag:hi1 + 1 + lag])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= w * h[lo2 - lag:hi2 + 1 - lag]
    return g

def pgrad(h0, t, n_it):
    h = h0.copy()
    n = len(h)
    lr = 1.0 / t
    for it in range(n_it):
        f = sg_obj(h, t)
        g = sg_grad(h, t)
        s = lr
        h_new = h
        for _ in range(60):
            h_try = project(h - s * g, n)
            f_try = sg_obj(h_try, t)
            if f_try < f - 1e-16:
                h_new = h_try
                break
            s *= 0.5
        h = h_new
        if it % 500 == 499:
            lr = np.clip(s * 1.4, 0.05 / t, 50.0 / t)
    return h

def _ceq(x):
    return np.array([np.sum(x) - _gn / 2.0])

def _ceqj(x):
    return np.ones((1, _gn))

def slsqp_polish(h0, t, nmax):
    global _gn
    _gn = len(h0)
    bnds = [(0.0, 1.0)] * _gn
    cons = {'type': 'eq', 'fun': _ceq, 'jac': _ceqj}
    try:
        r = minimize(sg_obj, h0, args=(t,), jac=lambda x: sg_grad(x, t),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': nmax, 'ftol': 1e-14})
        return project(np.clip(r.x, 0.0, 1.0), _gn)
    except Exception:
        return h0.copy()

def slsqp_nojac(h0, t, nmax):
    global _gn
    _gn = len(h0)
    bnds = [(0.0, 1.0)] * _gn
    cons = {'type': 'eq', 'fun': _ceq, 'jac': _ceqj}
    try:
        r = minimize(sg_obj, h0, args=(t,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': nmax, 'ftol': 1e-14})
        return project(np.clip(r.x, 0.0, 1.0), _gn)
    except Exception:
        return h0.copy()

def multi_start_pgrad(h0, t, n_it, n_start, rng):
    n = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(h0)
    for i in range(n_start):
        if i == 0:
            h_try = h0.copy()
        else:
            scale = 0.01 * (1.0 + 0.5 * i)
            h_try = project(h0 + scale * rng.randn(n), n)
        h = pgrad(h_try, t, n_it)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        hb = project(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    except Exception:
        hb = project(0.5 + 0.05 * rng.randn(280), 280)

    n = len(hb)
    cb = compute_c5(hb)

    # Phase 1: Projected gradient with increasing temperature
    temp_schedule = [(30, 3000), (100, 3000), (300, 2000), (800, 1500)]
    for t, nit in temp_schedule:
        if time.time() > t0 + 0.35 * budget_s:
            break
        h = pgrad(hb, t, nit)
        c = compute_c5(h)
        if c < cb:
            cb, hb = c, h

    # Phase 2: SLSQP polish
    if time.time() < t0 + budget_s - 200:
        for t in [300, 800, 1500]:
            if time.time() > t0 + 0.65 * budget_s:
                break
            h = slsqp_polish(hb, t, 500)
            c = compute_c5(h)
            if c < cb:
                cb, hb = c, h

    # Phase 3: Multi-restart with perturbation + gradient
    if time.time() < t0 + budget_s - 100:
        for t, nit in [(500, 800), (1500, 600)]:
            if time.time() > t0 + 0.85 * budget_s:
                break
            h, c = multi_start_pgrad(hb, t, nit, 4, rng)
            if c < cb:
                cb, hb = c, h

    # Phase 4: Final SLSQP at high temperature
    if time.time() < t0 + budget_s - 30:
        h = slsqp_polish(hb, 2000, 600)
        c = compute_c5(h)
        if c < cb:
            cb, hb = c, h

    # Phase 5: Very final projected gradient refinement
    if time.time() < t0 + budget_s - 10:
        h = pgrad(hb, 3000, 500)
        c = compute_c5(h)
        if c < cb:
            cb, hb = c, h

    return hb, compute_c5(hb), n
```
