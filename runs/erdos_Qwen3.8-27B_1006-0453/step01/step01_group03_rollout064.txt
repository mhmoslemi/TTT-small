    for j in range(n):
        i_vals = np.arange(n)
        lags1 = j - i_vals
        idx1 = off + lags1
        mask1 = (idx1 >= 0) & (idx1 < 2 * n - 1)
        if np.any(mask1):
            grad[j] += np.dot(w[idx1[mask1]], 1.0 - h[i_vals[mask1]])
        lags2 = np.arange(-(n - 1), n)

        mask2 = (hj >= 0) & (hj < n)
        if np.any(mask2):
            grad[j] -= np.dot(w[off + lags2[mask2]], h[hj[mask2]])

        lr = 0.025 / (1.0 + 0.002 * it)

def coord_descent(h0, dx, target_sum, rounds=30, deadline=None):

    h = h0.copy()
    best_c5 = compute_c5(h, dx)

                    else:

                    if len(cands) == 0:

                    break
                else:

        starts.append(h)

    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

    return float(np.sum(x[:n]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def slsqp(h0, dx, ts, mit=250):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.003]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:

                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-12})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 + 0.01:

    cr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (cr - cr.max()))

    if s < 1e-30:

            g[j] += np.dot(w[o + k], 1.0 - h[j - k])

        if kl2 <= kh2:

            g[j] -= np.dot(w[o + k], h[j + k])
