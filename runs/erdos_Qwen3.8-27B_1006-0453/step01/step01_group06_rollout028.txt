import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint_x(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint_x}]
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def erdos_ineq(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def slsqp_erdos(h, maxiter=300):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.02
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = np.clip(res.x[:n], 0, 1)
        return project_feasible(h_opt, n)
    except Exception:
        return h

def nm_polish(h, maxiter=300):
    n = len(h)
    def penalized(x):
        base = compute_c5(x)
        sp = abs(x.sum() - n / 2.0) * 100.0
        bp = (np.sum(np.maximum(0.0, x - 1.0)) + np.sum(np.maximum(0.0, -x))) * 100.0
        return base + sp + bp
    try:
        res = minimize(penalized, h, method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Smooth SLSQP warmup at parent resolution
    for T in [10, 50, 200, 800, 3000]:
        if elapsed() > budget_s * 0.08:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Direct constraint-based SLSQP (no smoothing)
    if elapsed() < budget_s * 0.18:
        h_try = slsqp_erdos(best_h, maxiter=500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: Multi-resolution basin hopping
    resolutions = sorted(set([best_n, 48, 54, 60, 66, 72, 80, 90, 100]))
    for n_try in resolutions:
        if elapsed() > budget_s * 0.62:
            break
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_feasible(h_base, n_try)

        n_trials = 7 if n_try <= 72 else 5
        for trial in range(n_trials):
            if elapsed() > budget_s * 0.62:
                break
            scale = 0.12 * (1.0 + 0.4 * np.sin(trial * 2.3 + 0.7))
            h_pert = h_base + np.random.randn(n_try) * scale
            h_pert = project_feasible(h_pert, n_try)
            # Two-stage: smooth then direct
            h_opt = slsqp_smooth(h_pert, 300, maxiter=200)
            if n_try <= 90:
                h_opt = slsqp_erdos(h_opt, maxiter=250)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
                h_base = h_opt.copy()

    # Phase 4: Cyclic rotation perturbations
    if elapsed() < budget_s * 0.75:
        n_cur = len(best_h)
        for shift in [1, 2, 3, 5, 8, 13, 21]:
            if elapsed() > budget_s * 0.75:
                break
            h_rot = np.roll(best_h, shift)
            h_opt = slsqp_erdos(h_rot, maxiter=250)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:

                best_h = h_opt.copy()

    # Phase 5: Intensive fine basin hopping
    n_cur = len(best_h)
    n_fine = max(8, int(budget_s * 0.18 / 8))
    for i in range(min(n_fine, 35)):
        if elapsed() > budget_s * 0.90:

        h_pert = project_feasible(h_pert, n_cur)
        h_opt = slsqp_smooth(h_pert, 800, maxiter=300)
        h_opt = slsqp_erdos(h_opt, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Nelder-Mead polish + final SLSQP
    if elapsed() < budget_s * 0.95:
        h_nm = nm_polish(best_h, maxiter=400)
        c5_nm = compute_c5(h_nm)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm

    if elapsed() < budget_s * 0.98:
        h_final = slsqp_erdos(best_h, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final

        h_smooth = slsqp_smooth(best_h, 10000, maxiter=500)
        c5_smooth = compute_c5(h_smooth)

            best_c5 = c5_smooth
            best_h = h_smooth

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
