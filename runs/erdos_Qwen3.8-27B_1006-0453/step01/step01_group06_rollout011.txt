import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def slsqp(h, n, T, maxiter=500, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def binary_sa(h0, n, t_start=1.0, t_end=0.01, n_iter=5000, seed=42):
    """Simulated annealing on binary patterns (flip pairs to maintain sum)."""
    rng = np.random.RandomState(seed)

    # Make it truly binary with exactly n/2 ones
    n_ones = int(round(h.sum()))

        # Fix: set top n/2 values to 1

        h = np.zeros(n)
        h[idx] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = t_start
    
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / n_iter)
        # Propose: flip one 1 to 0 and one 0 to 1

        h_new[j] = 1.0
        c5_new = compute_c5(h_new)
        delta = c5_new - compute_c5(h)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return best_h, best_c5
