    es = np.exp(s - s.max())
    p = es / es.sum()
    A = np.where(va, 1.0 - h[idx_b], 0.0)
    B = np.where(vb, h[idx_a], 0.0)
    grad = (A - B) @ p
    kmax = int(np.argmax(corr))
    kl = kmax - (n - 1)
    j = np.arange(n)
    ia, ib = j + kl, j - kl
    va2 = (ia >= 0) & (ia < n)
    vb2 = (ib >= 0) & (ib < n)
    am = np.where(vb2, 1.0 - h[ib.clip(0, n - 1)], 0.0)
    bm = np.where(va2, h[ia.clip(0, n - 1)], 0.0)
    grad += am - bm
    return dx * grad, corr, float(mx * dx)

def obj_slsqp(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    s = _GT * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s - s.max()))) / _GT) * dx

def grad_slsqp(x):
    g, _, _ = grad_lse(x, _GT, _IDX_A, _IDX_B, _VA, _VB)
    return g

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def run(seed=42, budget_s=1100, **kwargs):
    global _GT, _IDX_A, _IDX_B, _VA, _VB
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(280, 0.5)
    n = len(h)
    h = project(h, n)
    bh = h.copy()
    bc = compute_c5(h)

    _IDX_A, _IDX_B, _VA, _VB = setup_indices(n)

    # Phase 1: Annealed projected gradient with momentum
    v = np.zeros(n)
    lr = 0.4
    beta = 0.82
    no_improve = 0
    best_hr = bc

    schedule = [
        (15, 4000), (30, 4000), (60, 4000), (120, 3500),
        (250, 3000), (500, 2500), (1000, 2000), (2000, 1500),
        (4000, 1000), (8000, 800),
    ]

    for t_val, steps in schedule:
        if time.time() > t0 + budget_s - 80:
            break
        v *= 0.3
        for s in range(steps):
            if time.time() > t0 + budget_s - 80:
                break
            g, corr, cur_c = grad_lse(h, t_val, _IDX_A, _IDX_B, _VA, _VB)
            v = beta * v + g
            h_new = project(h + lr * v, n)
            c_new = np.max(corr_new)  # placeholder
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            c_new = float(np.max(corr_new) * (2.0 / n))
            if c_new < bc:
                bc = c_new
                bh = h_new.copy()
                no_improve = 0
                lr = min(1.5, lr * 1.01)
            else:
                no_improve += 1
                lr *= 0.995
                if no_improve > 800:
                    v = np.zeros(n)
                    h = project(bh + rng.randn(n) * 0.008, n)
                    no_improve = 0
                    lr = 0.3
            h = h_new
        v *= 0.2

    # Phase 2: Local SLSQP polish with high-t LSE
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    for t_polish in [3000, 8000, 20000]:
        if time.time() > t0 + budget_s - 10:
            break
        _GT = float(t_polish)
        try:
            r = minimize(obj_slsqp, bh, jac=grad_slsqp, method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-15})
            hc = project(np.clip(r.x, 0.0, 1.0), n)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc
        except Exception:
            pass

    # Phase 3: Additional gradient sweep from best
    if time.time() < t0 + budget_s - 20:
        h = bh.copy()
        v = np.zeros(n)
        lr = 0.1
        beta = 0.7
        for t_val in [10000, 30000, 100000]:
            steps = 1000
            dl = t0 + budget_s - 10
            for s in range(steps):
                if time.time() > dl:
                    break
                g, corr, _ = grad_lse(h, t_val, _IDX_A, _IDX_B, _VA, _VB)
                v = beta * v + g
                h_new = project(h + lr * v, n)
                c_new = compute_c5(h_new)
                lr = lr * 0.997
                if c_new < bc:
                    bc = c_new
                    bh = h_new.copy()
                h = h_new
            v *= 0.3

    # Final validation
    bh = project(bh, n)
    final_c5 = compute_c5(bh)
    return bh, final_c5, n
