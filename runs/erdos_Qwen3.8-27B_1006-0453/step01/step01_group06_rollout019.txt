            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        best_h = coord_sweep(best_h, best_n, max_passes=20)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 3: Perturb + coordinate descent
    n_perturb = 50
    for i in range(n_perturb):
        if elapsed() > budget_s * 0.50:
            break
        scale = 0.08 * np.exp(-0.05 * i) + 0.015
        h0 = best_h + np.random.uniform(-scale, scale, best_n)
        h0 = project_feasible(h0, best_n)
        h0 = coord_sweep(h0, best_n, max_passes=15)
        h0 = project_feasible(h0, best_n)
        c5v = compute_c5(h0)
        if c5v < best_c5:
            best_c5, best_h = c5v, h0

    # Phase 4: Multi-resolution exploration
    for n_try in [80, 100, 120, 150, 200]:
        if elapsed() > budget_s * 0.78:
            break
        for trial in range(3):
            if elapsed() > budget_s * 0.78:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h0 = np.interp(x_new, x_old, best_h)
            h0 = project_feasible(h0, n_try)
            h0 = coord_sweep(h0, n_try, max_passes=12)
            h0 = project_feasible(h0, n_try)
            c5v = compute_c5(h0)
            if c5v < best_c5:
                best_c5, best_h, best_n = c5v, h0, n_try
        # Also try a random binary init at this resolution
        if elapsed() < budget_s * 0.78:
            h0 = (np.random.rand(n_try) < 0.5).astype(float)
            h0 = project_feasible(h0, n_try)
            h0 = coord_sweep(h0, n_try, max_passes=12)
            h0 = project_feasible(h0, n_try)
            c5v = compute_c5(h0)
            if c5v < best_c5:
                best_c5, best_h, best_n = c5v, h0, n_try

    # Phase 5: SLSQP + coordinate descent refinement
    if elapsed() < budget_s * 0.90:
        for T in [80, 300, 800]:
            if elapsed() > budget_s * 0.92:
                break
            hc = slsqp_polish(best_h, T, 500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        best_h = coord_sweep(best_h, best_n, max_passes=25)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 6: Fine-scale perturbations
    if elapsed() < budget_s * 0.96:
        n_fine = 50
        for i in range(n_fine):
            if elapsed() > budget_s * 0.97:
                break
            scale = 0.02 * np.exp(-0.04 * i) + 0.003
            h0 = best_h + np.random.uniform(-scale, scale, best_n)
            h0 = project_feasible(h0, best_n)
            h0 = coord_sweep(h0, best_n, max_passes=8)
            h0 = project_feasible(h0, best_n)
            c5v = compute_c5(h0)
            if c5v < best_c5:
                best_c5, best_h = c5v, h0

    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
