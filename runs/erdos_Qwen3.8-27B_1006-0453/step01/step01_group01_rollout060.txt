def smoothed_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s_max = np.max(s)
    exp_s = np.exp(s - s_max)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    # For each lag m, compute contribution to gradient
    # k = m - (n-1)
    
    # Optimization: group by positive and negative lags
    # For lag +k (k=1..n-1), weight is w[n-1+k]
    # For lag -k (k=1..n-1), weight is w[n-1-k]
    # For lag 0, weight is w[n-1]
    
    # Lag 0: contribution is w[n-1] * (1-2h[j]) * dx
    grad += w[n-1] * (1 - 2*h)
    
    # For k = 1 to n-1:
    # Positive lag +k (weight wp = w[n-1+k]):
    #   d/dh[j] = (1-h[j-k])*I(j>=k) - h[j+k]*I(j+k<n)
    # Negative lag -k (weight wn = w[n-1-k]):
    #   d/dh[j] = (1-h[j+k])*I(j+k<n) - h[j-k]*I(j>=k)
    
    for k in range(1, n):
        wp = w[n-1+k]
        wn = w[n-1-k]
        
        # Positive lag: (1-h[j-k]) for j>=k, -h[j+k] for j+k<n
        if wp != 0:
            # (1-h[j-k]) for j=k..n-1
            grad[k:] += wp * (1 - h[:n-k])  # h[j-k] for j=k..n-1 is h[0..n-1-k]
            # -h[j+k] for j=0..n-1-k
            grad[:n-k] -= wp * h[k:]  # h[j+k] for j=0..n-1-k is h[k..n-1]
        
        # Negative lag: (1-h[j+k]) for j+k<n, -h[j-k] for j>=k
        if wn != 0:
            # (1-h[j+k]) for j=0..n-1-k
            grad[:n-k] += wn * (1 - h[k:])  # h[j+k] for j=0..n-1-k
            # -h[j-k] for j=k..n-1
            grad[k:] -= wn * h[:n-k]  # h[j-k] for j=k..n-1
    
    return dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def smooth_gradient(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    for k in range(1, n):
        wp = w[n-1+k]
        wn = w[n-1-k]
        if wp > 1e-15:
            grad[k:] += wp * (1.0 - h[:n-k])
            grad[:n-k] -= wp * h[k:]
        if wn > 1e-15:
            grad[:n-k] += wn * (1.0 - h[k:])
            grad[k:] -= wn * h[:n-k]
    
    return dx * grad

def projected_gradient(h, t, steps, alpha_init, alpha_decay):
    n = len(h)
    h_c = h.copy()
    alpha = alpha_init
    best_h = h_c.copy()
    best_f = smooth_objective(h_c, t)
    
    for i in range(steps):
        g = smooth_gradient(h_c, t)
        h_new = proj(h_c - alpha * g, n)
        f_new = smooth_objective(h_new, t)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        h_c = h_new
        alpha *= alpha_decay
    
    return best_h

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Start from initial solution
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)

    h = proj(h, n)
    bh = h.copy()
    bc = compute_c5(h)
    
    # Phase 1-4: Annealing with projected gradient
    phases = [
        (30, 2000, 0.5, 0.999),
        (60, 2000, 0.5, 0.999),
        (150, 2000, 0.3, 0.9995),
        (400, 3000, 0.1, 0.9998),
    ]
    
    for t_val, steps, a_init, a_dec in phases:
        if time.time() > t0 + budget_s * 0.7:
            break
        h = projected_gradient(h, t_val, steps, a_init, a_dec)
        c5 = compute_c5(h)
        if c5 < bc:
            bc = c5
            bh = h.copy()
        # Small perturbation
        h = proj(h + rng.randn(n) * 0.01, n)
    
    # Phase 5: SLSQP with analytical gradient for final polishing
    remaining = t0 + budget_s - 5 - time.time()
    if remaining > 30:
        n = len(bh)
        bnds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
        
        for t_val in [500, 1500, 4000]:
            if time.time() > t0 + budget_s - 10:

            try:
                r = minimize(lambda x: smooth_objective(x, t_val),
                           bh, jac=lambda x: smooth_gradient(x, t_val),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-14})
                hc = proj(np.clip(r.x, 0.0, 1.0), n)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
            except Exception:
                pass
    
    return bh, compute_c5(bh), len(bh)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[n - 1 + k]
        wn = w[n - 1 - k]
        if wp > 1e-14:
            grad[k:] += wp * (1.0 - h[:n - k])
            grad[:n - k] -= wp * h[k:]
        if wn > 1e-14:
            grad[:n - k] += wn * (1.0 - h[k:])
            grad[k:] -= wn * h[:n - k]
    return dx * grad

def cg_steps(h, t, alpha, n_steps, n):
    h_c = h.copy()
    m = len(h_c)
    if m != n:
        return h_c
    for i in range(n_steps):
        g = smooth_grad(h_c, t)
        h_c = proj(h_c - alpha * g, n)
    return h_c

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def slsqp_solve(h, t, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    r = minimize(
        lambda x: smooth_obj(x, t),
        h,
        jac=lambda x: smooth_grad(x, t),
        method='SLSQP',
        bounds=bnds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    return proj(np.clip(r.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 280
        h = np.full(n, 0.5)
    h = proj(h, n)
    bh = h.copy()
    bc = compute_c5(h)

    def elapsed():
        return time.time() - t0

    def remaining_frac():
        return (budget_s - elapsed()) / budget_s

    # Phase 1: Projected gradient exploration (fast)
    if elapsed() < budget_s * 0.15:
        for alpha, steps in [(5.0, 500), (2.0, 800), (1.0, 1000)]:
            if elapsed() > budget_s * 0.15:
                break
            h = cg_steps(bh, 40.0, alpha, steps, n)
            c5 = compute_c5(h)
            if c5 < bc:
                bc = c5
                bh = h.copy()
        h = cg_steps(bh, 100.0, 1.0, 1000, n)
        c5 = compute_c5(h)
        if c5 < bc:
            bc = c5
            bh = h.copy()

    # Phase 2-5: SLSQP with annealing
    phases = [
        (50.0, 1500, 1e-12),
        (150.0, 2000, 1e-13),
        (500.0, 3000, 1e-14),
        (2000.0, 3000, 1e-14),
    ]
    time_per_phase = budget_s * 0.2

    for idx, (t_val, maxit, ftol) in enumerate(phases):
        phase_deadline = t0 + budget_s * 0.15 + (idx + 1) * time_per_phase
        if elapsed() > phase_deadline - 5:
            continue
        # Try up to 2 perturbations per phase
        for p in range(2):
            if time.time() > phase_deadline - 10:
                break
            try:
                hc = slsqp_solve(bh, t_val, maxit, ftol)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
            except Exception:
                pass
            # Perturb for next attempt
            if p == 0 and time.time() < phase_deadline - 30:
                bh_try = proj(bh + rng.randn(n) * 0.02, n)
                # Quick check if perturbation helps
                hc2 = slsqp_solve(bh_try, t_val, maxit // 2, ftol)
                c5_2 = compute_c5(hc2)
                if c5_2 < bc:
                    bc = c5_2
                    bh = hc2.copy()

    # Phase 6: Final fine polishing
    if time.time() < t0 + budget_s - 15:
        for t_val in [5000, 10000]:
            if time.time() > t0 + budget_s - 10:

            try:
                hc = slsqp_solve(bh, t_val, 5000, 1e-15)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
            except Exception:

    # Phase 7: Last-resort micro-optimization with projected gradient at high t
    if time.time() < t0 + budget_s - 5:

        c5_fin = compute_c5(h_fin)
        if c5_fin < bc:
            bc = c5_fin
            bh = h_fin.copy()

    return bh, compute_c5(bh), n
