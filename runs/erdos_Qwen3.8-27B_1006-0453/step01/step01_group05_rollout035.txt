import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        for i in np.argsort(np.abs(h - 0.5))[::-1]:
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h


def lse_obj(x, n, temp):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    s = c / temp
    m = s.max()
    return temp * (m + np.log(np.mean(np.exp(s - m))))


def sum_con(x, n):
    return np.sum(x) - n / 2.0


def slsqp_solve(x0, n, temp, maxiter):
    res = minimize(lse_obj, x0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                   args=(n, temp), options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x


def nm_fun(x, n):
    x = repair_h(x, n)
    return compute_c5(x, n)


def nm_solve(x0, n, maxiter):
    res = minimize(nm_fun, x0, args=(n,), method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-13,
                            'fatol': 1e-15, 'adaptive': True})
    return repair_h(res.x, n)


def coord_descent(h, n, max_rounds):
    best = compute_c5(h, n)
    for _ in range(max_rounds):
        imp = False
        for i in range(n):
            for d in (0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001):
                h2 = h.copy()
                v = h2[i] + d
                if 0.0 < v < 1.0:
                    h2[i] = v
                    h2 = repair_h(h2, n)
                    c5 = compute_c5(h2, n)
                    if c5 < best - 1e-14:
                        h = h2
                        best = c5
                        imp = True
                        break
        if not imp:
            break
    return h, best


def time_left(t0, budget, margin):
    return max(0.0, budget - margin - (time.time() - t0))
