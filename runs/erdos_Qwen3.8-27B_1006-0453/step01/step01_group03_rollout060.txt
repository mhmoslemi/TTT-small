    exp_c = np.exp((corr - max_c) / tau)
    sum_exp = exp_c.sum()

    # Wait: LSE = tau * log(sum(exp(c/tau))) = tau * log(sum(exp((c-max_c)/tau) * exp(max_c/tau)))
    # = tau * (max_c/tau + log(sum(exp((c-max_c)/tau))))
    # = max_c + tau * log(sum(exp((c-max_c)/tau)))
    
    w = exp_c / sum_exp  # softmax weights
    lags = np.arange(2*n-1) - (n-1)  # L_0, L_1, ..., L_{2n-2}
    
    grad = np.zeros(n)

        t2 = 0.0

            # h[j-L]

            if 0 <= idx1 < n:

            # h[j+L]

            if 0 <= idx2 < n:
                t2 += w[k] * h[idx2]

So for coordinate descent, the optimal x for a given coordinate i is:

where a_k = 1-h[i-L_k]-h[i+L_k] and b_k = B_k + h[i+L_k]

This is a one-dimensional piecewise-linear minimization problem! We can solve it exactly:
- The minimum of max_k (a_k * x + b_k) over x∈[0,1] occurs at either x=0, x=1, or at a point where two of the lines intersect at the maximum.

This gives an exact coordinate descent, which is guaranteed to converge (though possibly slowly).

Let me implement this:

```python
def coordinate_step(h, i, n):
    """Optimal value of h[i] given all other coordinates fixed."""
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(2*n-1) - (n-1)
    
    # For each lag k, compute the coefficient of h[i] and the constant term
    # c[k] = const_k + h[i] * coeff_k
    # where coeff_k = (1-h[i-L_k]) - h[i+L_k] (with 0 for out-of-range)
    # and const_k = c[k] - h[i] * coeff_k
    
    x = h[i]
    coeffs = np.zeros(2*n-1)
    consts = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        L = lags[k]
        idx1 = i - L
        idx2 = i + L
        c1 = h[idx1] if 0 <= idx1 < n else 0.0
        c2 = h[idx2] if 0 <= idx2 < n else 0.0
        coeff = 1.0 - c1 - c2
        consts[k] = corr_full[k] - x * coeff
        coeffs[k] = coeff
    
    # Now minimize max_k (coeffs[k]*x + consts[k]) over x in [0,1]
    # This is a 1D piecewise linear minimization
    # The maximum of linear functions is the upper envelope
    # The minimum of the upper envelope occurs at:
    # - x=0, x=1, or at intersection points of adjacent lines on the upper envelope
    
    # Simple approach: the optimum is where the "active" lines balance
    # For two lines a1*x+b1 and a2*x+b2, they intersect at x* = (b2-b1)/(a1-a2)
    # At the optimum, the max is minimized, so the active constraint changes
    
    # Numerical approach: check x=0, x=1, and all pairwise intersections in [0,1]
    best_x = x
    best_val = np.max(consts)  # at x=0
    
    for x_candidate in [0.0, 1.0]:
        val = np.max(coeffs * x_candidate + consts)
        if val < best_val:
            best_val = val
            best_x = x_candidate
    
    # Check pairwise intersections
    n_corr = 2*n-1
    for k1 in range(n_corr):
        for k2 in range(k1+1, n_corr):
            da = coeffs[k1] - coeffs[k2]
            if abs(da) < 1e-14:
                continue
            x_int = (consts[k2] - consts[k1]) / da
            if 0 <= x_int <= 1:
                val = np.max(coeffs * x_int + consts)
                if val < best_val:
                    best_val = val
                    best_x = x_int
    
    return best_x
```

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def lse_obj_grad(h, n, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    exp_c = np.exp((corr - max_c) / tau)
    sum_exp = exp_c.sum()
    obj = (max_c + tau * np.log(sum_exp)) * dx
    w = exp_c / sum_exp
    lags = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        t1 = 0.0
        t2 = 0.0
        for k in range(2*n-1):
            L = lags[k]
            i1 = j - L
            if 0 <= i1 < n:
                t1 += w[k] * h[i1]
            i2 = j + L
            if 0 <= i2 < n:
                t2 += w[k] * h[i2]
        grad[j] = (1.0 - t1 - t2) * dx
    return obj, grad

def coord_descent(h0, n, dx, max_passes=200, time_limit=None, t_start=None):
    """Exact coordinate descent with box constraints + periodic projection."""
    target_sum = n / 2.0
    h = h0.copy()
    n_corr = 2 * n - 1
    xs = np.linspace(0, 1, 101)
    best_c5 = compute_c5(h, dx)
    
    for p in range(max_passes):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            # Compute coefficients
            corr = np.correlate(h, 1.0 - h, mode='full')
            lags = np.arange(n_corr) - (n-1)
            coeffs = np.zeros(n_corr)
            consts = np.zeros(n_corr)
            xi = h[i]
            for k in range(n_corr):
                L = lags[k]
                i1 = i - L
                i2 = i + L
                c1 = h[i1] if 0 <= i1 < n else 0.0
                c2 = h[i2] if 0 <= i2 < n else 0.0
                coeff = 1.0 - c1 - c2
                consts[k] = corr[k] - xi * coeff
                coeffs[k] = coeff
            # Minimize max over x in [0,1]
            vals = np.max(coeffs[:, None] * xs[None, :] + consts[:, None], axis=0)
            best_x = xs[np.argmin(vals)]
            if abs(best_x - xi) > 1e-10:

            best_c5 = c5
        if not improved:
            break
