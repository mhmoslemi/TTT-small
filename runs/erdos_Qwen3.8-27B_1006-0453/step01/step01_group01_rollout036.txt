import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_lse_hess_free(h, t):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s = np.clip(s, None, 50.0)
    exp_s = np.exp(s)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    f = (mx + np.log(sum_exp) / t) * dx
    grad = np.zeros(n)
    for i in range(len(corr)):
        wi = w[i]
        if wi < 1e-13:
            continue
        l = i - (n - 1)
        if l >= 0:
            lo1 = 0
            hi1 = n - l
            lo2 = l
            hi2 = n
        else:
            lo1 = -l
            hi1 = n
            lo2 = 0
            hi2 = n + l
        if hi1 > lo1:
            grad[lo1:hi1] += wi * b[lo1 + l:hi1 + l]
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - l:hi2 - l]
    grad *= dx
    return f, grad

def pen_fg(h, t, n, mu, nu):
    val, grad = obj_lse_hess_free(h, t)
    s = np.sum(h) - n / 2.0
    f = val + 0.5 * mu * s * s + nu * s
    grad = grad + (mu * s + nu) * np.ones(n)
    return f, grad

def alm_optimize(h0, n, t, bnds, max_outer=6, max_inner=200):
    mu = 1.0
    nu = 0.0
    h = h0.copy()
    for _ in range(max_outer):
        r = minimize(pen_fg, h, args=(t, n, mu, nu), jac=True,
                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': max_inner, 'gtol': 1e-10})
        h = np.clip(r.x, 0.0, 1.0)
        s = np.sum(h) - n / 2.0
        nu = nu + mu * s
        mu = mu * 3.0
        if abs(s) < 1e-12 and abs(nu) < 1e-8:
            break
    return proj(h, n)

def slsqp_optimize(h0, n, t, bnds, cons, maxiter=3000, ftol=1e-15):
    def f_only(h):
        return obj_lse_hess_free(h, t)[0]
    r = minimize(f_only, h0, args=(), method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def make_ceq(n):
    return lambda x: np.array([np.sum(x) - n / 2.0])

def make_ceqj(n):
    return lambda x: np.ones((1, n))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = 280
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
        h_init = proj(h_init, n)
    except Exception:
        h_init = np.full(280, 0.5) + rng.randn(280) * 0.05
        h_init = proj(h_init, 280)
        n = 280

    bh = h_init.copy()
    bc = compute_c5(bh)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': make_ceq(n), 'jac': make_ceqj(n)}

    temps = [80, 200, 500, 1200, 3000]
    fracs = [0.15, 0.30, 0.50, 0.72, 0.88]

    for ti, (t, frac) in enumerate(zip(temps, fracs)):
        deadline = t0 + budget_s * frac
        restarts = 6 if ti == 0 else 3
        if ti == 2:
            restarts = 4
        starts = [bh.copy()]
        for a in [0.01, 0.02, 0.04, 0.07, 0.1, 0.15][:restarts - 1]:
            starts.append(proj(bh + rng.randn(n) * a, n))

        for hc0 in starts:

                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
            except Exception:
                pass

    # SLSQP polish phase
    deadline_sl = t0 + budget_s * 0.96
    if time.time() < deadline_sl:

                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
            except Exception:
                pass

    # Binary refinement
    if time.time() < t0 + budget_s - 15:

            if time.time() > t0 + budget_s - 10:

            hb = bh.copy()
            low_m = hb < thresh
            high_m = hb > 1.0 - thresh
            hb[low_m] = 0.0
            hb[high_m] = 1.0
            hb = proj(hb, n)
            c5 = compute_c5(hb)
            if c5 < bc:
                bc = c5
                bh = hb.copy()

    # Final tight SLSQP
    if time.time() < t0 + budget_s - 3:

            hc = slsqp_optimize(bh, n, 5000, bnds, cons, maxiter=1500, ftol=1e-17)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc.copy()
        except Exception:
            pass

    return bh, compute_c5(bh), len(bh)
