import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(h, tau):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(corr.max())
    exps = np.exp(tau * (corr - max_c))
    s = float(exps.sum())
    F = max_c + np.log(s) / tau
    w = exps / s
    grad = np.zeros(N)
    thresh = 1.0 / (len(corr) * 5.0)
    active = np.where(w > thresh)[0]
    for k in active:
        L = abs(int(k) - (N - 1))
        wi = w[k]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
    return F, grad, corr

def adam_optimize(h, N, target, dx, t0, deadline, tau0=3.0, lr0=0.04, max_iter=3000):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    tau = tau0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        F, grad, corr = smooth_max_and_grad(h, tau)
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        it1 = it + 1
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** it1)
        v_hat = v_buf / (1.0 - beta2 ** it1)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        if compute_c5(h_new, dx) < c5_val:
            h = h_new
        else:
            lr *= 0.75
            if lr < 1e-9:
                break
        if (it + 1) % 120 == 0:
            tau = min(tau * 1.7, 8000.0)
            lr *= 0.9
    return best_h, best_c5

def sa_optimize(h, N, target, dx, t0, deadline, rng, T0=0.015, T1=0.00005):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5
    t_start = time.time()
    total_time = max(deadline - t_start, 1.0)
    no_improve = 0
    stuck_count = 0

    while time.time() < deadline:
        frac = min((time.time() - t_start) / total_time, 1.0)
        T = T0 * (T1 / T0) ** frac
        d_scale = 1.0 - 0.6 * frac

        old_h = cur_h.copy()
        move_type = rng.random()
        if move_type < 0.55:
            i = rng.randint(N)
            j = rng.randint(N)
            if i == j:
                continue
            if cur_h[i] >= cur_h[j]:
                d = min(cur_h[i], 1.0 - cur_h[j]) * rng.uniform(0.03, 0.35) * d_scale
                if d < 1e-4:
                    continue
                cur_h[i] -= d
                cur_h[j] += d
            else:
                d = min(cur_h[j], 1.0 - cur_h[i]) * rng.uniform(0.03, 0.35) * d_scale
                if d < 1e-4:
                    continue
                cur_h[j] -= d
                cur_h[i] += d
        elif move_type < 0.80:
            block = rng.randint(2, min(8, N // 4))
            i = rng.randint(N - block)
            j = (i + block + rng.randint(1, N - 2 * block)) % (N - block)
            j = min(j, N - block)
            if j < i:
                j = i + block
                j = min(j, N - block)
            total = cur_h[i:i + block].sum()
            if total < 1e-4 or (N - j - block) < 0:
                continue
            avg = total / block
            cur_h[i:i + block] -= avg * rng.uniform(0.3, 1.0)
            cur_h[j:j + block] += avg * rng.uniform(0.3, 1.0)
        else:
            idx = rng.choice(N, size=rng.randint(2, 6), replace=False)
            for ii in idx:
                cur_h[ii] += rng.randn() * 0.08 * d_scale

        cur_h = project_sum(cur_h, target)
        new_c5 = compute_c5(cur_h, dx)

        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            no_improve += 1
            delta = new_c5 - cur_c5
            if T > 1e-10 and rng.random() < np.exp(-min(delta / T, 60.0)):
                cur_c5 = new_c5
            else:
                cur_h = old_h

        if no_improve > 3000 and stuck_count < 10:
            stuck_count += 1
            no_improve = 0
            cur_h = best_h + rng.randn(N) * 0.02
            cur_h = project_sum(cur_h, target)
            cur_c5 = compute_c5(cur_h, dx)

    return best_h, best_c5

def try_size(h_best, c5_best, N_new, t0, deadline, rng):
    if N_new == len(h_best) or time.time() > deadline - 30:
        return h_best, c5_best
    dx_old = 2.0 / len(h_best)
    x_old = np.linspace(0.0, 2.0, len(h_best), endpoint=False)
    x_new = np.linspace(0.0, 2.0, N_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_best)
    target = N_new / 2.0
    h_interp = project_sum(h_interp, target)
    dx_new = 2.0 / N_new
    h_try, c5_try = adam_optimize(h_interp, N_new, target, dx_new, t0, deadline - 20, tau0=4, lr0=0.03, max_iter=400)
    if c5_try < c5_best:
        return h_try, c5_try
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0

    # Phase 1: Quick Adam warmup from parent
    h_curr, c5_curr = adam_optimize(h0.copy(), n0, target0, dx0, t0, deadline - 300, tau0=3.0, lr0=0.04, max_iter=400)

    # Phase 2: Multi-restart SA
    sa_budget = (deadline - 60) - time.time()
    n_restarts = 3
    sa_deadline = t0 + budget_s - 80
    for r in range(n_restarts):
        rem = sa_deadline - time.time()
        if rem < 30:
            break
        sa_frac = rem / sa_budget
        h_curr, c5_s = sa_optimize(h_curr.copy(), n0, target0, dx0, t0, sa_deadline - (n_restarts - 1 - r) * 60, rng, T0=0.012 * (0.7 ** r), T1=0.00004)
        if c5_s < c5_curr:
            c5_curr = c5_s
        else:
            h_curr = h_curr  # keep previous best

    # Phase 3: Try different sizes
    for N_try in [n0 * 2, n0 // 2 + 1, n0 + 10]:
        if N_try < 20 or N_try > 900 or N_try % 2 != 0:
            continue
        h_curr, c5_curr = try_size(h_curr, c5_curr, N_try, t0, deadline - 40, rng)

    # Phase 4: Final polish with tight Adam
    if time.time() < deadline - 10:
        N_f = len(h_curr)
        dx_f = 2.0 / N_f
        target_f = N_f / 2.0
        h_pol, c5_pol = adam_optimize(h_curr.copy(), N_f, target_f, dx_f, t0, deadline - 2, tau0=30.0, lr0=0.008, max_iter=2000)
        if c5_pol < c5_curr:
            h_curr = h_pol
            c5_curr = c5_pol

    c5_final = compute_c5(h_curr, 2.0 / len(h_curr))
    return h_curr, c5_final, len(h_curr)
