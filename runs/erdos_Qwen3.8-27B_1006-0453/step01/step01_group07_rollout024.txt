    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        delta = n / 2.0 - np.sum(h)
        if abs(delta) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += delta / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    v = np.interp(nx, ox, parent)
    return project(v, new_n)

def _cons_fun(x):
    return np.sum(x) - _N / 2.0

def _cons_jac(x):
    return np.ones(_N)

def _obj_slsqp(x, beta):
    n = _N
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = np.max(c)
    sh = np.clip((c - mx) / beta, -100.0, 100.0)
    return (2.0 / n) * (mx + beta * np.log(np.sum(np.exp(sh))))

def _obj_lfgs(x, beta):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    cs = corr / beta
    cs_max = np.max(cs)
    exp_cs = np.exp(cs - cs_max)
    log_sum = cs_max + np.log(np.sum(exp_cs))
    obj = dx * beta * log_sum
    w = exp_cs / np.sum(exp_cs)
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        m = s + n - 1
        wm = w[m]
        if wm < 1e-12:
            continue
        if s >= 0:
            grad[s:] += wm * (1.0 - h_slice(h, n, 0, n - s))
            grad[:n - s] -= wm * h_slice(h, n, s, n)
        else:
            t = -s
            grad[:n - t] += wm * (1.0 - h_slice(x, n, t, n))
            grad[t:] -= wm * h_slice(x, n, 0, n - t)
    return obj, dx * grad

def h_slice(arr, n, lo, hi):
    return arr[lo:hi]

def _obj_lfgs_aug(x, state):
    n = _N
    beta = state[0]
    lam = state[1]
    obj, grad = _obj_lfgs(x, beta)
    penalty = lam * (np.sum(x) - n / 2.0)
    grad_pen = np.full(n, lam)
    return obj + penalty, grad + grad_pen

def _grad_single_shift(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        t = -shift
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    t0 = time.time()
    rng = np.random.RandomState(seed)
    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP at parent resolution
    n = pn
    _N = n
    h = best_h.copy()
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    for beta in [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003]:
        if tl() < 25:
            break
        try:
            res = minimize(_obj_slsqp, h, args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project(np.clip(res.x, 0, 1), n)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    best_n = n
                h = h_new
        except Exception:
            pass

    # Phase 2: L-BFGS-B with augmented Lagrangian
    if tl() > 40:
        h = best_h.copy()
        n = best_n
        _N = n
        bounds = list(zip(np.zeros(n), np.ones(n)))
        lam = 0.0
        for outer in range(30):
            if tl() < 15:
                break
            beta = 0.0008
            state = (beta, lam)
            try:
                res = minimize(_obj_lfgs_aug, h, args=(state,), method='L-BFGS-B',
                             bounds=bounds, jac=True,
                             options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9,
                                      'maxfun': 5000})
                h_new = project(np.clip(res.x, 0, 1), n)
                viol = np.sum(h_new) - n / 2.0
                lam = np.clip(lam + 0.05 * viol, -2.0, 2.0)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    best_n = n
                h = h_new
                if abs(viol) < 1e-12:
                    break
            except Exception:
                break

    # Phase 3: Active shift cycling
    if tl() > 30:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        for rnd in range(300):
            if tl() < 8:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = np.argmax(corr)
            s = k_star - (n - 1)
            g = _grad_single_shift(h, n, s)
            lr = 0.003
            improved = False
            for step in range(100):
                if tl() < 5:
                    break
                h_try = project(h - lr * g, n)
                c_try = np.correlate(h_try, 1.0 - h_try, mode="full")[k_star]
                if c_try < corr[k_star] - 1e-12:
                    h = h_try
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    g = _grad_single_shift(h, n, s)
                    improved = True
                    lr = min(lr * 1.05, 0.02)
                else:
                    lr *= 0.7
                    if lr < 1e-7:
                        break
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
                best_n = n
            if not improved and rnd > 20:
                # Try next worst shift
                idx = np.argsort(corr)[-3:]
                found = False
                for ki in idx:
                    if ki == k_star:
                        continue
                    s2 = ki - (n - 1)
                    g2 = _grad_single_shift(h, n, s2)
                    lr2 = 0.002
                    for step2 in range(50):
                        h_try = project(h - lr2 * g2, n)
                        c_try = np.correlate(h_try, 1.0 - h_try, mode="full")[ki]
                        if c_try < corr[ki] - 1e-12:
                            h = h_try
                            lr2 = min(lr2 * 1.05, 0.015)
                            found = True
                        else:
                            lr2 *= 0.7
                            break
                    if found:
                        break

    # Phase 4: Gradient-informed coordinate descent
    if tl() > 15:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        max_iters = int(min(80000, (tl() - 5) * 150))
        for it in range(max_iters):
            if tl() < 4:
                break
            # Compute gradient of worst shift
            corr = np.correlate(h, 1.0 - h, mode="full") if it % 50 == 0 else None
            if corr is None:
                i = rng.randint(n)
                j = rng.randint(n)
            else:
                k_worst = np.argmax(corr)
                s_worst = k_worst - (n - 1)
                g = _grad_single_shift(h, n, s_worst)
                # Pick i from top positive grad, j from top negative grad
                pos_idx = np.where(g > np.median(g))[0]
                neg_idx = np.where(g < np.median(g))[0]
                if len(pos_idx) > 0 and len(neg_idx) > 0:
                    i = pos_idx[rng.randint(len(pos_idx))]
                    j = neg_idx[rng.randint(len(neg_idx))]
                else:
                    i = rng.randint(n)
                    j = rng.randint(n)
            if i == j:
                continue
            d = rng.choice([0.001, 0.002, 0.004, 0.008, 0.0005])
            if rng.rand() < 0.5:
                di, dj = d, -d
            else:
                di, dj = -d, d
            hi_new = h[i] + di
            hj_new = h[j] + dj
            if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
                continue
            h[i] = hi_new
            h[j] = hj_new
            c = compute_c5(h)
            if c < cur:
                cur = c
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    best_n = n
            else:
                h[i] -= di
                h[j] -= dj

    # Phase 5: Try upsampled resolution if time allows
    if tl() > 60:
        n2 = min(560, pn * 2)
        _N = n2
        h2 = upsample(best_h, n2)
        bounds2 = list(zip(np.zeros(n2), np.ones(n2)))
        cons2 = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        for beta in [0.003, 0.001, 0.0005]:
            if tl() < 20:
                break
            try:
                res = minimize(_obj_slsqp, h2, args=(beta,), method='SLSQP',
                             bounds=bounds2, constraints=cons2,
                             options={'maxiter': 300, 'ftol': 1e-13})
                if res.x is not None:
                    h2 = project(np.clip(res.x, 0, 1), n2)
                    c5 = compute_c5(h2)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h2.copy()
                        best_n = n2
            except Exception:
                break

    # Final polish: tiny random perturbations
    if tl() > 5:
        h = best_h.copy()
        n = best_n
        cur = best_c5
        for _ in range(int(min(20000, (tl() - 2) * 100))):
            if tl() < 2:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.0002, 0.003) * rng.choice([-1.0, 1.0])
            if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                h[i] += d
                h[j] -= d
                c = compute_c5(h)
                if c < cur:
                    cur = c
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        best_n = n
                else:
                    h[i] -= d
                    h[j] += d

    _N = best_n
    return best_h, best_c5, best_n
