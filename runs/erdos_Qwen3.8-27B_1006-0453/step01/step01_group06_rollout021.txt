    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    shifted = np.clip(corr - m, -400.0, 0.0)
    val = m + float(np.log(np.sum(np.exp(T * shifted)))) / T
    return val * dx

def sum_func(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)
    h = np.clip(h, 0.0, 1.0)
    # Final tiny correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_objective(x, T):
    return smooth_c5(x, T)

def slsqp_optimize(h, T, maxiter=400):
    n = len(h)
    cons = [{'type': 'eq', 'fun': sum_func}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(slsqp_objective, h, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_feasible(np.clip(res.x, 0.0, 1.0), n)

def pair_swap_optimize(h, n, n_passes=20, rng=None):
    if rng is None:
        rng = np.random
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(n_passes):
        improved = False
        for _ in range(250):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            t_min = max(-hi, 1.0 - hj)
            t_max = min(1.0 - hi, hj)
            if t_min >= t_max - 1e-10:
                continue
            h_mod = h.copy()
            h_mod[i] += 1.0
            h_mod[j] -= 1.0
            delta = np.correlate(h_mod, 1.0 - h_mod, mode='full') - corr
            a, b = t_min, t_max
            for _ in range(35):
                t1 = a + (b - a) / 3.0
                t2 = b - (b - a) / 3.0
                if float(np.max(corr + t1 * delta)) < float(np.max(corr + t2 * delta)):
                    b = t2
                else:
                    a = t1
            t_opt = (a + b) / 2.0
            new_corr = corr + t_opt * delta
            if float(np.max(new_corr)) < float(np.max(corr)) - 1e-11:
                h[i] += t_opt
                h[j] -= t_opt
                corr = new_corr
                improved = True
        if not improved:
            break
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=1.0)
    return project_feasible(h_new, n_new)

def thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        c = 0
        v = i
        while v > 0:
            c ^= (v & 1)
            v >>= 1
        t[i] = float(c)
    # Force exactly n/2 ones
    idx = np.argsort(-t)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 1)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = thue_morse(100)

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing temperature on initial resolution
    for T in [15, 60, 250, 1000, 4000]:
        if elapsed() > budget_s * 0.22:
            break
        try:
            hc = slsqp_optimize(best_h, T, maxiter=350)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Pair-swap local optimization at initial resolution
    if elapsed() < budget_s * 0.32:
        best_h = pair_swap_optimize(best_h, n, n_passes=25, rng=rng)
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5(best_h)

    # Phase 2: Multi-resolution refinement
    for n_try in [80, 100, 120, 150, 200]:
        if elapsed() > budget_s * 0.72:
            break
        h_work = upsample_h(best_h, n_try)
        for T in [40, 250, 1200]:
            if elapsed() > budget_s * 0.72:
                break
            try:
                hc = slsqp_optimize(h_work, T, maxiter=350)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n = n_try
                    h_work = hc
            except Exception:
                pass
        if elapsed() < budget_s * 0.74:
            h_work = pair_swap_optimize(h_work, n_try, n_passes=15, rng=rng)
            h_work = project_feasible(h_work, n_try)
            c5v = compute_c5(h_work)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_work
                n = n_try

    # Phase 3: Structured restarts at various sizes
    for n_try in [60, 80, 100, 120, 150]:
        if elapsed() > budget_s * 0.86:
            break
        h0 = thue_morse(n_try)
        h0 = h0 + rng.uniform(-0.15, 0.15, n_try)
        h0 = project_feasible(h0, n_try)
        for T in [80, 600]:
            if elapsed() > budget_s * 0.86:
                break
            try:
                hc = slsqp_optimize(h0, T, maxiter=300)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n = n_try
            except Exception:
                pass
        if elapsed() < budget_s * 0.87:
            hc = pair_swap_optimize(h0, n_try, n_passes=10, rng=rng)
            hc = project_feasible(hc, n_try)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                n = n_try

    # Phase 4: Aggressive final polish
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.94:
        for T in [5000, 15000]:
            if elapsed() > budget_s * 0.94:
                break
            try:
                hc = slsqp_optimize(best_h, T, maxiter=500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass
    if elapsed() < budget_s * 0.97:
        best_h = pair_swap_optimize(best_h, n_cur, n_passes=30, rng=rng)
        best_h = project_feasible(best_h, n_cur)
        best_c5 = compute_c5(best_h)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
