import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(400):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def slsqp_solve(h0, dx, target_sum, maxiter=300, tol=1e-13):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.05, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': tol, 'iprint': 0})
        if res.x is not None and res.fun < 1e8:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def slsqp_flatten(h0, dx, target_sum, maxiter=200, lam=0.1):
    """Minimize t + lam*sum(c_k^2) to flatten correlation profile"""
    n = len(h0)
    def obj(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return x[n] + lam * np.sum(corr**2)
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.05, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
        if res.x is not None and res.x[n] < 1e8:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def subgradient_topk(h, dx, top_k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / top_k
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for s in range(-(n-1), n):
            if 0 <= j + s < n:
                g += w[s + n - 1] * (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= w[s + n - 1] * h[j - s]
        grad[j] = g
    return grad * dx

def projected_gradient(h0, dx, target_sum, n_iter=2000, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        lr = 0.04 / (1.0 + 0.008 * it)
        top_k = min(3 + it // 80, 8)
        g = subgradient_topk(h, dx, top_k)
        best_local = (best_c5, best_h.copy())
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_local[0]:
                best_local = (c5_new, h_new)
        if best_local[0] < best_c5:
            best_c5 = best_local[0]
            best_h = best_local[1].copy()
            h = best_local[1]
        else:
            h = best_h + 0.008 * np.random.randn(n)
            h = project_sr(h, target_sum)
    return best_h, best_c5

def pair_local_search(h, dx, target_sum, n_rounds=80, partners=15):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for rd in range(n_rounds):
        improved = False
        order = np.random.permutation(n)
        for j in order:
            done = False
            for delta in [0.015, 0.01, 0.02, -0.01, -0.015]:
                hj = h[j] + delta
                if hj < -1e-12 or hj > 1.0 + 1e-12:
                    continue
                best_try = best_c5
                best_jp = -1
                for jp in list(order[:partners]) + list(order[-partners:]):
                    if jp == j:
                        continue
                    hjp = h[jp] - delta
                    if hjp < -1e-12 or hjp > 1.0 + 1e-12:
                        continue
                    h[j], h[jp] = hj, hjp
                    c = compute_c5(h, dx)
                    h[j], h[jp] = hj - delta, hjp + delta
                    if c < best_try:
                        best_try = c
                        best_jp = jp
                if best_jp >= 0 and best_try < best_c5 - 1e-10:
                    h[j] = hj
                    h[best_jp] = h[best_jp] - delta
                    best_c5 = best_try
                    improved = True
                    done = True
                    break
            if done:
                break
        if not improved:
            break
    return h, best_c5
