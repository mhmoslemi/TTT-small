import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_smoothed(h, T):
    n = len(h)
    overlap = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(overlap)
    exps = np.exp((overlap - m) / T)
    p = exps / exps.sum()
    
    grad = np.ones(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] -= w * h[s:]
            grad[s:] -= w * h[:n - s]
        else:
            ss = -s
            grad[ss:] -= w * h[:n - ss]
            grad[:n - ss] -= w * h[ss:]
    return grad

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to increase sum: add to points not at 1
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
        else:
            # Need to decrease sum: subtract from points not at 0
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = h.sum()
    if abs(s - target) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = interior.sum()
        if cnt > 0:
            h[interior] += (target - s) / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def line_search(h, grad, T, n, max_steps=50):
    """Backtracking line search"""
    f_curr = compute_c5(h)
    alpha = 0.1
    best_h = h.copy()
    best_f = f_curr
    for _ in range(max_steps):
        h_try = project(h - alpha * grad, n)
        f_try = compute_c5(h_try)
        if f_try < best_f:
            best_h = h_try
            best_f = f_try
        alpha *= 0.7
        if alpha < 1e-10:
            break
    return best_h, best_f

def optimize_gradient(h0, n, T_list, max_iter_per_T, time_limit):
    h = h0.copy()
    t_start = time.time()
    for T in T_list:
        for it in range(max_iter_per_T):
            if time.time() - t_start > time_limit:
                return h
            grad = grad_smoothed(h, T)
            h, _ = line_search(h, grad, T, n)
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    lo, hi = -h.max() - 0.5, 1.5 - h.min()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_try = np.clip(h + mid, 0.0, 1.0).sum()
        if s_try > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = interior.sum()
        if cnt > 0:
            h[interior] += diff2 / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_smoothed(h, T):
    n = len(h)
    overlap = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(overlap)
    if T < 1e-10:
        p = np.zeros(2 * n - 1)
        p[np.argmax(overlap)] = 1.0
    else:
        exps = np.exp((overlap - m) / T)
        p = exps / exps.sum()
    grad = np.ones(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-11:
            continue
        s = k - n1
        if s >= 0:
            if s < n:
                grad[:n - s] -= w * h[s:]
            if s > 0:
                grad[s:] -= w * h[:n - s]
        else:
            ss = -s
            if ss < n:
                grad[ss:] -= w * h[:n - ss]
            if ss > 0:
                grad[:n - ss] -= w * h[ss:]
    return grad

def adam_refine(h, n, T, lr, n_iter, t0, budget_frac):
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    f_curr = compute_c5(h)
    for it in range(n_iter):
        if time.time() - t0 > budget_frac:
            break
        g = gradient_smoothed(h, T)
        t_it = it + 1
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        mc = m_v / (1 - b1 ** t_it)
        vc = v_v / (1 - b2 ** t_it)
        step_lr = lr / (1.0 + 0.005 * t_it)
        h_try = project_h(h - step_lr * mc / (np.sqrt(vc) + eps), n)
        f_try = compute_c5(h_try)
        if f_try < f_curr:
            h = h_try
            f_curr = f_try
    return h, f_curr

def upsample_h(h, new_n):

    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, new_n)
