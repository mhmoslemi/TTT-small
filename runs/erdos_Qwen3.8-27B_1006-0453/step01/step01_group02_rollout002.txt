    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_grad(h, N, dx, tau):
    """Returns (objective_value, gradient) using smooth-max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = float(np.max(corr) * dx)
    
    # Softmax weights
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    
    grad = np.zeros(N)
    # Only process significant weights
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for idx in active:
        lag = int(idx) - (N - 1)
        wi = w[idx]
        if lag >= 0 and lag < N:
            grad[lag:] += wi * (1.0 - h[:N - lag])
            grad[:N - lag] -= wi * h[lag:]
        elif lag < 0:
            l = -lag
            if l < N:
                grad[:N - l] += wi * (1.0 - h[l:])
                grad[l:] -= wi * h[:N - l]
    grad *= dx
    return obj, grad

    """Main optimizer with Nesterov + adaptive tau + restarts."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    sigma = np.zeros(N)  # momentum
    t = 0
    
    phases = [
        (lr, tau, n_iter) for lr, tau, n_iter in [
            (0.03, 3.0, 3000),
            (0.02, 8.0, 3000),
            (0.01, 20.0, 3000),
            (0.005, 50.0, 3000),
            (0.002, 100.0, 2000),
        ]
    ]
    
    for lr, tau, n_iter in phases:
        if time.time() > deadline - 10:
            break
        for it in range(n_iter):
            if time.time() > deadline - 10:
                break
            obj, grad = obj_grad(h, N, dx, tau)
            if obj < best_c5:
                best_c5 = obj
                best_h = h.copy()
            t += 1
            # Nesterov
            sigma = 0.9 * sigma - lr * grad
            h_new = project_sum(h + sigma, target)
            h = h_new
            # Occasionally reset momentum to help escape
            if it % 1000 == 999:
                sigma *= 0.5
    
    return best_h, best_c5
