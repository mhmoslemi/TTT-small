def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def lse_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = np.max(corr)
    return float((np.log(np.sum(np.exp((corr - M) / temp))) * temp + M) * dx)


def lse_grad(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = np.max(corr)
    w = np.exp((corr - M) / temp)
    w_sum = w.sum()
    w = w / w_sum
    offsets = np.arange(2 * n - 1) - (n - 1)
    grad = np.ones(n)
    for j in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            off = offsets[k]
            c = 0.0
            if 0 <= j + off < n:
                c += x[j + off]
            if 0 <= j - off < n:
                c += x[j - off]
            s += w[k] * c
        grad[j] -= s
    return grad * dx

            h = np.clip(h, 0, 1)
            continue

    return h

            if c_a < best_c5:

                best_h = h_a
                n = alt_n

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
