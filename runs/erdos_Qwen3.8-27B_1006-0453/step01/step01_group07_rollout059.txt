    best_h = h.copy()
    deltas = [0.008, -0.008, 0.004, -0.004, 0.002, -0.002]
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() >= t_end:
                break
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h[i] + d, 0, 1)
                h_new = project_sum(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    h = h_new
                    cur_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    parent = np.asarray(initial_h_values, dtype=np.float64)
    n = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = n

    def remaining():
        return t0 + budget_s - time.time()

    # Phase 1: Main subgradient + annealing at parent resolution
    t1_end = t0 + budget_s * 0.50
    h1, c1 = subgrad_anneal(parent, n, rng, t0, t1_end, 0.05, 0.004, 8, 0.9997)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
        best_n = n

    # Phase 1b: Second run with different parameters from best
    if remaining() > 60:
        t1b_end = min(t0 + budget_s * 0.68, time.time() + min(remaining(), budget_s * 0.18))
        h1b, c1b = subgrad_anneal(best_h, n, rng, t0, t1b_end, 0.03, 0.002, 5, 0.9998)
        if c1b < best_c5:
            best_c5 = c1b
            best_h = h1b.copy()
            best_n = n

    # Phase 2: Fine-tuning
    if remaining() > 30:
        t2_end = min(t0 + budget_s * 0.82, time.time() + min(remaining(), budget_s * 0.14))
        h2, c2 = fine_tune(best_h, n, t2_end)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
            best_n = n

    # Phase 3: Coordinate descent polish
    if remaining() > 15:
        t3_end = min(t0 + budget_s * 0.92, time.time() + min(remaining(), budget_s * 0.10))
        h3, c3 = coord_polish(best_h, n, t3_end)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
            best_n = n

    # Phase 4: Higher resolution attempt
    if remaining() > 20:
        n2 = min(n * 2, 560)
        h_up = upsample_arr(best_h, n2)
        c_up = compute_c5(h_up)
        t4_end = t0 + budget_s - 5
        h4, c4 = subgrad_anneal(h_up, n2, rng, t0, t4_end, 0.02, 0.001, 5, 0.9999)
        h4, c4_final = fine_tune(h4, n2, t0 + budget_s - 2)
        if c4_final < best_c5:
            best_c5 = c4_final
            best_h = h4.copy()
            best_n = n2

    # Final validation
    best_c5 = compute_c5(best_h)
    assert best_h.shape[0] == best_n
    assert abs(best_h.sum() - best_n / 2.0) < 1e-6, f"sum={best_h.sum()}, target={best_n/2.0}"
    return best_h, best_c5, best_n
