import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def softmax_obj_grad(h, dx, beta):
    n = len(h)
    corr_dx = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr_dx)
    shifted = beta * (corr_dx - max_c)
    exps = np.exp(shifted)
    s = exps.sum()
    w = exps / s
    obj = (np.log(s) + beta * max_c) / beta
    grad = np.zeros(n)
    for j in range(n):
        gj = 0.0
        for k_idx in range(2*n-1):
            k = k_idx - (n-1)
            contrib = 0.0
            if 0 <= j - k < n:
                contrib += (1.0 - h[j - k])
            if 0 <= j + k < n:
                contrib -= h[j + k]
            gj += w[k_idx] * contrib * dx
        grad[j] = gj
    return obj, grad

def lp_active_step(h, dx, active_lags, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(active_lags)
    # Objective coefficients: sum over active lags of grad_k . h
    c = np.zeros(n)
    for k_idx in active_lags:
        k = k_idx - (n-1)
        for j in range(n):
            val = 0.0
            if 0 <= j - k < n:
                val += (1.0 - h[j-k])
            if 0 <= j + k < n:
                val -= h[j+k]
            c[j] += val
    c = c / m * dx
    # Equality: sum h = target_sum
    A_eq = np.ones((1, n))
    b_eq = np.array([target_sum])
    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=[(0,1)]*n, method='highs')
    if res.status == 0:
        return project_sr(res.x, target_sum)
    return h.copy()

def anneal_lbfgs(h0, dx, target_sum, beta_start=5, beta_end=80, n_beta=20, time_limit=60):
    n = len(h0)
    h = h0.copy()
    t_start = time.time()
    for bi in range(n_beta):
        if time.time() - t_start > time_limit:
            break
        beta = beta_start * (beta_end / beta_start) ** (bi / n_beta)
        def obj(x):
            return softmax_obj_grad(x, dx, beta)[0]
        def full_obj_grad(x):
            o, g = softmax_obj_grad(x, dx, beta)
            return o, g
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(full_obj_grad, h, method='L-BFGS-B',
                      bounds=[(0,1)]*n, constraints=cons,
                      options={'maxiter': 200, 'gtol': 1e-8})
        if res.x is not None:
            h_new = project_sr(res.x, target_sum)
            if compute_c5(h_new, dx) <= compute_c5(h, dx) + 1e-10:
                h = h_new
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def softmax_obj_grad(h, dx, beta):
    n = len(h)
    corr_dx = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr_dx)
    exps = np.exp(beta * (corr_dx - max_c))
    s = exps.sum()
    w = exps / s
    obj = (np.log(s) + beta * max_c) / beta
    grad = np.zeros(n)
    for j in range(n):
        gj = 0.0
        for k_idx in range(2 * n - 1):
            k = k_idx - (n - 1)
            contrib = 0.0
            if 0 <= j - k < n:
                contrib += (1.0 - h[j - k])
            if 0 <= j + k < n:
                contrib -= h[j + k]
            gj += w[k_idx] * contrib * dx
        grad[j] = gj
    return obj, grad

def lp_active_step(h, dx, active_lags, target_sum):
    n = len(h)
    m = len(active_lags)
    c = np.zeros(n)
    for k_idx in active_lags:
        k = k_idx - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j - k < n:
                val += (1.0 - h[j - k])
            if 0 <= j + k < n:
                val -= h[j + k]
            c[j] += val
    c = c / m * dx
    A_eq = np.ones((1, n))
    b_eq = np.array([target_sum])
    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=[(0.0, 1.0)] * n, method='highs')
    if res.status == 0:
        return project_sr(res.x, target_sum)
    return h.copy()

def slsqp_smooth(h0, dx, target_sum, beta, maxiter=200):
    n = len(h0)
    def obj_and_grad(x):
        o, g = softmax_obj_grad(np.clip(x, 0, 1), dx, beta)
        g = g + 50.0 * (np.sum(x) - target_sum) * np.ones(n)
        return o, g
    x0 = h0.copy()
    try:
        res = minimize(obj_and_grad, x0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x, target_sum)
    except Exception:
        pass
    return h0.copy()

def optimize_single(h0, dx, target_sum, deadline):
    n = len(h0)
    best_h = project_sr(h0, target_sum)
    best_c5 = compute_c5(best_h, dx)
    betas = [4, 8, 15, 30, 60, 120]
    for b in betas:
        if time.time() > deadline:
            break
        iters = max(50, min(250, int((deadline - time.time()) / 2.5)))
        h_new = slsqp_smooth(best_h, dx, target_sum, b, iters)
        c_new = compute_c5(h_new, dx)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
    for _ in range(15):
        if time.time() > deadline or (deadline - time.time()) < 1.0:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        for n_active in [3, 5, 8]:
            top_idx = np.argsort(corr)[-n_active:]
            h_lp = lp_active_step(best_h, dx, top_idx, target_sum)
            c_lp = compute_c5(h_lp, dx)
            if c_lp < best_c5:
                best_h, best_c5 = h_lp, c_lp
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    return best_h, best_c5
