            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_og(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    s = t * (corr - mx)
    e = np.exp(s)
    w = e / e.sum()
    obj = mx + np.log(e.sum()) / t
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wk = w[i]
        if wk < 1e-18:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            ll = -lag
            grad[:n - ll] += wk * (1.0 - h[ll:])
            grad[ll:] -= wk * h[:n - ll]
    grad *= dx
    return obj, grad

def _cef(h):
    return np.sum(h) - len(h) / 2.0

def _cej(h):
    return np.ones(len(h))

def _penalty_func(h, t, lam, n):
    f, g = softmax_og(h, t)
    cv = np.sum(h) - n / 2.0
    f += 0.5 * lam * cv * cv
    g = g + lam * cv
    return f, g

def _coord_refine(h, n, n_passes=200):
    c5 = compute_c5(h)
    for _ in range(n_passes):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        active = (corr > 0.95 * mx)
        improved = False
        for i in range(n):
            for delta in [0.005, -0.005, 0.002, -0.002]:
                h2 = h.copy()
                h2[i] += delta
                if h2[i] < 0 or h2[i] > 1:
                    continue
                h2 = proj(h2, n)
                c52 = compute_c5(h2)
                if c52 < c5 - 1e-9:
                    h = h2
                    c5 = c52
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(280, 0.5)
    n = len(h_best)
    h_best = proj(h_best, n)
    c5_best = compute_c5(h_best)

    bnds = [(0.0, 1.0)] * n
    constraints = {'type': 'eq', 'fun': _cef, 'jac': _cej}

    # Phase 1: L-BFGS-B with penalty, multiple temperatures
    dl1 = t0 + budget_s * 0.40
    for t in [30, 100, 300, 800]:
        if time.time() > dl1 - 5:
            break
        lam = 300.0
        args = (t, lam, n)
        res = minimize(_penalty_func, h_best.copy(), method='L-BFGS-B',
                      jac=True, bounds=bnds, args=args,
                      options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = proj(np.clip(res.x, 0, 1), n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 2: SLSQP exact constraint, analytical gradient
    dl2 = t0 + budget_s * 0.75
    for t in [200, 600, 2000]:
        if time.time() > dl2 - 10:
            break
        res = minimize(lambda x: softmax_og(x, t)[0], h_best.copy(),
                      jac=lambda x: softmax_og(x, t)[1],
                      method='SLSQP', bounds=bnds, constraints=constraints,
                      options={'maxiter': 3000, 'ftol': 1e-15})
        h_new = proj(np.clip(res.x, 0, 1), n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 3: Very high temperature SLSQP for sharp max
    dl3 = t0 + budget_s * 0.88
    for t in [5000, 20000]:
        if time.time() > dl3 - 5:
            break
        res = minimize(lambda x: softmax_og(x, t)[0], h_best.copy(),
                      jac=lambda x: softmax_og(x, t)[1],
                      method='SLSQP', bounds=bnds, constraints=constraints,
                      options={'maxiter': 3000, 'ftol': 1e-16})
        h_new = proj(np.clip(res.x, 0, 1), n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 4: Coordinate refinement on active correlations
    if time.time() < t0 + budget_s - 30:
        h_ref, c5_ref = _coord_refine(h_best, n, 100)
        if c5_ref < c5_best:
            c5_best = c5_ref
            h_best = h_ref

    # Phase 5: Final random perturbation restarts
    dl5 = t0 + budget_s - 15
    for r in range(8):
        if time.time() > dl5:
            break
        h_try = proj(h_best + rng.randn(n) * 0.005, n)
        t = 1000.0
        res = minimize(lambda x: softmax_og(x, t)[0], h_try,
                      jac=lambda x: softmax_og(x, t)[1],
                      method='SLSQP', bounds=bnds, constraints=constraints,
                      options={'maxiter': 2000, 'ftol': 1e-14})
        h_new = proj(np.clip(res.x, 0, 1), n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    h_best = proj(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
