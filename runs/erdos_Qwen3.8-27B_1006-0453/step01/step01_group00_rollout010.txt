        if n_try < 100:
            continue
        h_curr = upsample(h_best, n_try) if n_try != n_best else h_best.copy()
        for T in [0.002, 0.0008]:
            if budget_s - elapsed() < 50:
                break
            h_opt = slsqp_opt(h_curr, n_try, soft_max_obj, (n_try, T), 400)
            h_cand = fix_sum(np.clip(h_opt, 0.0, 1.0), n_try)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                h_best = h_cand
                c5_best = c5_cand
                n_best = n_try
                h_curr = h_best.copy()

    # Phase 7: Final aggressive polish
    if budget_s - elapsed() > 35:
        for T in [0.0005, 0.0002]:
            if budget_s - elapsed() < 20:
                break
            h_opt = slsqp_opt(h_best, n_best, soft_max_obj, (n_best, T), 700)
            h_cand = fix_sum(np.clip(h_opt, 0.0, 1.0), n_best)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                h_best = h_cand
                c5_best = c5_cand

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, n_best)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n_best)
