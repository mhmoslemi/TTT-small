**Idea: Use the subgradient of the TRUE max (not smoothed) with averaging**

At each iteration:
1. Compute corr = correlate(h, 1-h, 'full')
2. Find the set of active lags: A = {k : corr[k] > corr[k] - ε} for small ε
3. Compute the averaged subgradient: grad = (1/|A|) * sum_{k in A} ∂corr[k]/∂h
4. Take a projected gradient step: h_new = proj(h - alpha * grad)
5. Use adaptive step size (backtracking or Barzilai-Borwein)

This avoids the smoothing approximation entirely and directly optimizes the max.

The Barzilai-Borwein step size: alpha_k = <s_{k-1}, y_{k-1}> / <y_{k-1}, y_{k-1}>
where s_{k-1} = h_k - h_{k-1} and y_{k-1} = grad_k - grad_{k-1}.

This is typically very efficient for smooth problems but can be less stable for non-smooth ones. However, with the averaged subgradient, it might work well.

```python
def subgradient_step(h, n, alpha=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    # Find active lags (within tolerance of max)
    tol = 0.01 * n  # small tolerance in corr units
    active = np.where(corr > mx - tol)[0]
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n-1, n-1+lag)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += (1.0 - h[lo1-lag:hi1+1-lag])
        lo2 = max(0, -lag)
        hi2 = min(n-1, n-1-lag)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= h[lo2+lag:hi2+1+lag]
    
    grad /= len(active)
    new_h = h - alpha * grad * dx
    return proj(new_h, n)
```

This is O(n * |active|) per step. If |active| is small (say 5-20 out of 559), this is very fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(2000):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    e = np.exp(s)
    e_sum = e.sum()
    if e_sum < 1e-300:
        w = np.zeros_like(s)
        w[np.argmax(s)] = 1.0
    else:
        w = e / e_sum
    val = (mx + np.log(e_sum) / t) * dx
    nlags = 2 * n - 1
    grad = np.zeros(n)
    for k in range(nlags):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wk * (1.0 - h[lo1 - lag:hi1 + 1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= wk * h[lo2 + lag:hi2 + 1 + lag]
    return val, grad * dx

    return smooth_obj_and_grad(x, _T[0])[0]

    return smooth_obj_and_grad(x, _T[0])[1]

    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = 0.5
    h_prev = h.copy()
    g_prev = None

        mx = np.max(corr)
        tol = max(0.5, 0.005 * n)
        active = np.where(corr > mx - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            lo1 = max(0, lag)
            hi1 = min(n - 1, n - 1 + lag)
            if lo1 <= hi1:
                grad[lo1:hi1 + 1] += (1.0 - h[lo1 - lag:hi1 + 1 - lag])
            lo2 = max(0, -lag)
            hi2 = min(n - 1, n - 1 - lag)
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] -= h[lo2 + lag:hi2 + 1 + lag]
        if len(active) > 0:

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:

        if g_prev is not None:

            y_vec = grad - g_prev
            sy = np.dot(s_vec, y_vec)

        g_prev = grad.copy()
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
