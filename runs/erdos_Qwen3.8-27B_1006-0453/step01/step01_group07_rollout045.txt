    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    shifted = (c - m) / beta
    # Prevent overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = m + beta * np.log(np.sum(exp_shifted))
    return dx * log_sum

def softmax_grad(h, beta):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def softmax_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    sh = np.clip((c - m) / beta, -500.0, 0.0)
    return dx * (m + beta * np.log(np.sum(np.exp(sh))))

def softmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    sh = np.clip((c - m) / beta, -500.0, 0.0)
    exp_s = np.exp(sh)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        a1, b1 = max(0, -s), min(n, n - s)
        if a1 < b1:
            grad[a1:b1] += wk * (1.0 - h[a1 + s:b1 + s])
        a2, b2 = max(0, s), min(n, n + s)
        if a2 < b2:
            grad[a2:b2] -= wk * h[a2 - s:b2 - s]
    return dx * grad

def project_box_sum(v, n, target_sum):
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum} via bisection on shift."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        c = np.clip(v - t, 0.0, 1.0)
        if np.sum(c) > target_sum:
            lo = t
        else:
            hi = t
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 1.0, old_n, endpoint=False)
    nx = np.linspace(0.0, 1.0, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v = project_box_sum(v, new_n, new_n / 2.0)
    return v

def fix_sum(h, n):
    return project_box_sum(h, n, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = 1e9
    best_h = None
    best_n = 0

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    n_list = sorted(set([pn, pn * 2, min(pn * 3, 600), 200, 350, 450]))
    n_list = [n for n in n_list if 80 <= n <= 700]

    betas = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]

    for n in n_list:
        if budget_s - (time.time() - t0) < 60:
            break
        h0 = upsample(parent, n) if n != pn else parent.copy()
        n_tries = 2 if n > 400 else 3
        for trial in range(n_tries):
            if budget_s - (time.time() - t0) < 25:
                break
            h_init = h0.copy()
            if trial > 0:
                h_init = fix_sum(h0 + rng.randn(n) * 0.015, n)
            bounds = list(zip(np.zeros(n), np.ones(n)))
            cons = [{'type': 'eq',
                     'fun': lambda x: np.sum(x) - n / 2.0,
                     'jac': lambda x: np.ones(n)}]
            for beta in betas:
                if budget_s - (time.time() - t0) < 12:
                    break
                try:
                    res = minimize(softmax_val, h_init.copy(), jac=softmax_grad,
                                  args=(beta,), method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 400, 'ftol': 1e-13})
                    h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        best_n = n
                    h_init = h_new
                except Exception:
                    pass

    # Projected gradient refinement on best solution
    if budget_s - (time.time() - t0) > 20:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        lr = 0.02
        beta_pg = 0.003
        t_rem = budget_s - (time.time() - t0) - 5
        n_steps = int(t_rem * 50)
        for it in range(max(1, n_steps)):
            if budget_s - (time.time() - t0) < 4:
                break
            g = softmax_grad(h, beta_pg)
            h_try = h - lr * g
            h_try = project_box_sum(h_try, n, n / 2.0)
            c_new = compute_c5(h_try)
            if c_new < cur - 1e-15:
                h = h_try
                cur = c_new
                lr = min(lr * 1.05, 0.05)
            else:
                lr *= 0.7
                if lr < 1e-5:
                    break
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            best_n = n

    # Fine coordinate descent
    if budget_s - (time.time() - t0) > 5:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        t_rem = budget_s - (time.time() - t0) - 2
        n_steps = int(t_rem * 40)
        for it in range(max(1, n_steps)):
            if budget_s - (time.time() - t0) < 2:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.0005, 0.015) * rng.choice([-1.0, 1.0])
            save_i, save_j = h[i], h[j]
            h[i] = np.clip(save_i + d, 0.0, 1.0)
            h[j] = np.clip(save_j - d, 0.0, 1.0)
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
            else:
                h[i], h[j] = save_i, save_j
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            best_n = n

    return best_h, best_c5, best_n
