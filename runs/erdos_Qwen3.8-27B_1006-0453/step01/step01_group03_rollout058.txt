def compute_c5(h, dx):

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def active_set_slsqp(h0, dx, target_sum, max_iter=100, top_m=5):
    n = len(h0)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = corr.max() + 0.01
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    
    best_h = h0.copy()
    best_c5 = np.max(corr) * dx
    
    for outer in range(5):

        c5_vals = corr * dx
        max_val = c5_vals.max()
        # Get top m lag indices
        idx = np.argsort(-c5_vals)[:top_m]
        
        def ineq(x):
            c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
            return x[n] - c[idx]
        
        def eqf(x):
            return np.sum(x[:n]) - target_sum
        
        x0 = np.concatenate([best_h, [max_val + 0.005]])

def smooth_min_autocorr_grad(h, beta):
    n = len(h)
    # Autocorrelation
    R = np.correlate(h, h, mode='full')  # length 2n-1, index n-1 is lag 0
    
    # Softmin weights
    neg_R = -R.astype(float)
    neg_R -= neg_R.max()  # for numerical stability
    w = np.exp(beta * neg_R)
    w /= w.sum()
    
    # Gradient (direction to ASCEND = maximize min autocorr)
    g = np.zeros(n)
    for idx in range(len(R)):
        wk = w[idx]
        if wk < 1e-8:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # h[j+k] for j in [0, n-1-k]
            if n-1-k >= 0:
                g[:n-k] += wk * h[k:]  # h[j+k] for j=0..n-1-k → h[k], h[k+1], ...
            # h[j-k] for j in [k, n-1]
            g[k:] += wk * h[:n-k]  # h[j-k] for j=k..n-1 → h[0], h[1], ...
        else:
            m = -k  # m > 0
            # h[j+k] = h[j-m] for j in [m, n-1]
            g[m:] += wk * h[:n-m]  # h[j-m] for j=m..n-1 → h[0], h[1], ...
            # h[j-k] = h[j+m] for j in [0, n-1-m]
            g[:n-m] += wk * h[m:]  # h[j+m] for j=0..n-1-m → h[m], h[m+1], ...
    
    return g  # This is the direction of increase for min k R(k)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_vals = corr * dx
    return float(np.max(c5_vals)), corr, c5_vals

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def smooth_min_grad(h, beta):
    """Gradient direction to maximize min_k R(k) where R(k)=autocorrelation at lag k."""
    n = len(h)
    R = np.correlate(h, h, mode='full').astype(float)
    # Softmin weights
    neg_R = -R
    neg_R -= neg_R.max()
    w = np.exp(beta * neg_R)
    w /= w.sum()
    
    g = np.zeros(n)
    for idx in range(len(R)):
        wk = w[idx]
        if wk < 1e-8:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[:n-k] += wk * h[k:]
            g[k:] += wk * h[:n-k]
        else:
            m = -k
            g[m:] += wk * h[:n-m]
            g[:n-m] += wk * h[m:]
    return g

def grad_ascent(h0, dx, target_sum, n_iter=5000, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5, _, _ = compute_c5_full(h, dx)
    best_it = 0
    stagnation = 0
    
    for it in range(n_iter):
        if t_deadline is not None and time.time() > t_deadline:
            break
        
        # Adaptive beta: start moderate, increase
        beta = 5.0 + 30.0 * (it / max(n_iter, 1))
        # Learning rate: decreasing
        lr = 0.02 / (1.0 + 0.003 * it)
        
        g = smooth_min_grad(h, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        
        # Try several step sizes
        improved = False
        for shrink in [1.0, 0.5, 0.2, 0.1, 0.05]:
            h_new = project_sr(h + lr * shrink * g, target_sum)
            c5_new, _, _ = compute_c5_full(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                best_it = it
                improved = True
                stagnation = 0
                break
            else:
                h_new = None
        
        if not improved:
            stagnation += 1
            # Restart from best
            h = best_h.copy()
            if stagnation > 50:
                break
    
    return best_h, best_c5

def active_set_slsqp(h0, dx, target_sum, top_m=5, max_iter=200):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    
    best_h = h0.copy()
    c5_best, _, c5_vals = compute_c5_full(best_h, dx)
    
    for outer in range(4):
        # Active lags: top m + lag 0 (always active)
        _, _, c5_vals = compute_c5_full(best_h, dx)
        idx_sorted = np.argsort(-c5_vals)
        active_idx = set(idx_sorted[:top_m].tolist())
        
        def ineq(x):
            c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
            return x[n] - c[list(active_idx)]
        
        def eqf(x):
            return np.sum(x[:n]) - target_sum
        
        t0 = c5_best / dx + 0.003
        x0 = np.concatenate([best_h, [t0]])
        try:
            res = minimize(lambda x: x[n], x0, method='SLSQP', bounds=bnds,
                         constraints=[{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}],
                         options={'maxiter': max_iter, 'ftol': 1e-12})
            if res.x is not None:
                h_new = project_sr(res.x[:n], target_sum)
                c5_new, _, _ = compute_c5_full(h_new, dx)
                if c5_new < c5_best:
                    c5_best = c5_new
                    best_h = h_new
        except Exception:
            pass
        top_m = min(top_m + 3, 2 * n - 1)
    
    return best_h, c5_best

def coordinate_descent(h0, dx, target_sum, n_rounds=50, t_deadline=None):
    """For each coordinate, find optimal value that minimizes max correlation."""
    n = len(h0)
    h = h0.copy()
    best_c5, _, _ = compute_c5_full(h, dx)
    best_h = h.copy()
    
    for rnd in range(n_rounds):
        if t_deadline is not None and time.time() > t_deadline:
            break
        improved = False
        for j in range(n):
            save = h[j]
            # Try to find best value for h[j]
            # The effect on correlations is linear in h[j] (since h[j] appears
            # in h[j]*(1-h[j+k]) and h[j+k]*(1-h[j]))
            # Actually it's not quite linear because of the sum constraint.
            # Let's just try a few values.
            for trial in [0.0, 0.25, 0.5, 0.75, 1.0, save + 0.1, save - 0.1]:
                trial = np.clip(trial, 0.0, 1.0)
                h[j] = trial
                h_adj = project_sr(h, target_sum)
                c5_trial, _, _ = compute_c5_full(h_adj, dx)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_adj.copy()
                    h = h_adj.copy()
                    improved = True
            if h[j] == save:
                h[j] = save
            # After projection, h[j] might have changed
            # Re-project and check
        if not improved:
            break
    
    return best_h, best_c5

def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    pos = rng.permutation(n)
    n_full = int(target_sum)
    frac = target_sum - n_full
    h[pos[:n_full]] = 1.0
    if frac > 0.01:
        h[pos[n_full]] = frac
    return project_sr(h, target_sum)

def make_stepped_start(n, target_sum, rng):
    """Create a step-like function with random breakpoints."""
    n_breaks = rng.randint(3, 8)
    breaks = sorted(rng.choice(range(1, n), size=min(n_breaks, n-1), replace=False))
    h = np.zeros(n)
    cum = 0
    for i, b in enumerate(breaks + [n]):
        length = b - (breaks[i-1] if i > 0 else 0)
        if i % 2 == 0:
            # set to some value
            val = np.clip(target_sum / n * (1 + rng.randn() * 0.5), 0, 1)
            h[(breaks[i-1] if i > 0 else 0):b] = val
            cum += val * length
        else:
            h[(breaks[i-1] if i > 0 else 0):b] = 0
    return project_sr(h, target_sum)

For the soft-max of c(k) over k (we want to minimize the max):
f = (1/beta) * log(sum_k exp(beta*c(k)))

df/dh[j] = sum_k w_k * dc(k)/dh[j] where w_k = exp(beta*c(k)) / sum exp(beta*c(k))

This is concentrated on the lags where c(k) is largest.

Let me rewrite this in vectorized form:

For k >= 0 (corresponding to idx = n-1+k in the full correlation array):
  dc(k)/dh[j] = (1-h[j-k]) for j in [k, n-1]  [I(j>=k)]
               + (-h[j+k]) for j in [0, n-1-k]  [I(j<=n-1-k)]

Vectorized:
  g[k:] += wk * (1 - h[:n-k])    # (1-h[j-k]) for j=k..n-1
  g[:n-k] += wk * (-h[k:])       # -h[j+k] for j=0..n-1-k

For k < 0, m = -k (corresponding to idx = n-1-k = n-1+m):
  dc(k)/dh[j] = (1-h[j+m]) for j in [0, n-1-m]  [I(j<=n-1-m)]
               + (-h[j-m]) for j in [m, n-1]     [I(j>=m)]

Vectorized:
  g[:n-m] += wk * (1 - h[m:])    # (1-h[j+m]) for j=0..n-1-m
  g[m:] += wk * (-h[:n-m])       # -h[j-m] for j=m..n-1

Let me verify with the small example n=3, h=[a,b,c]:

c(lag=1) = b(1-a) + c(1-b)
dc/dh[0] = dc/da = -b (from b*(1-a), derivative w.r.t. a is -b)
dc/dh[1] = dc/db = (1-a) + (-c) = 1-a-c
dc/dh[2] = dc/dc = 1-b

Using my formula for k=1 (>=0):
  g[1:] += wk * (1-h[:2]) = wk*[1-a, 1-b] → g[1] += wk*(1-a), g[2] += wk*(1-b)
  g[:2] += wk * (-h[1:]) = wk*[-b, -c] → g[0] += wk*(-b), g[1] += wk*(-c)

So g[0] = -wk*b, g[1] = wk*(1-a) - wk*c = wk*(1-a-c), g[2] = wk*(1-b)

This matches! ✓

OK so my gradient formula is correct. Let me clean up the code.

Now, a key question: should I minimize the softmax of c(k) or maximize the softmin of R(k)?

Since c(k) = n/2 - sum_{first k or last k} - R(k), and the boundary terms depend on k differently, the softmax of c(k) is the correct objective to minimize.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_vals = corr * dx
    return float(np.max(c5_vals)), corr, c5_vals

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def softmax_grad(h, beta):
    """Gradient of softmax-max of c(k) = correlate(h, 1-h) components.
    We want to MINIMIZE this, so we descend in this direction."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c_shift = c - c.max()
    w = np.exp(beta * c_shift)
    w /= w.sum()
    
    g = np.zeros(n)
    for idx in range(len(c)):
        wk = w[idx]
        if wk < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += wk * (1.0 - h[:n-k])
            g[:n-k] += wk * (-h[k:])
        else:
            m = -k
            g[:n-m] += wk * (1.0 - h[m:])
            g[m:] += wk * (-h[:n-m])
    return g  # grad of smooth max; descend in this direction

def grad_descent(h0, dx, target_sum, n_iter=5000, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5, _, _ = compute_c5_full(h, dx)
    best_iter = 0
    stall = 0
    
    for it in range(n_iter):
        if t_deadline is not None and time.time() > t_deadline:
            break
        beta = 10.0 + 50.0 * (it / max(n_iter, 1))
        lr = 0.03 / (1.0 + 0.005 * it)
        
        g = softmax_grad(h, beta)
        improved = False
        for shrink in [1.0, 0.5, 0.2, 0.1, 0.05, 0.01]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new, _, _ = compute_c5_full(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                best_iter = it
                improved = True
                break
        if not improved:
            stall += 1
            h = best_h.copy()
            if stall > 100:
                break
    
    return best_h, best_c5

def slsqp_active(h0, dx, target_sum, top_m=6, max_iter=200):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    best_h = h0.copy()
    c5_best, _, c5_vals = compute_c5_full(best_h, dx)
    
    for outer in range(4):
        c5_best, _, c5_vals = compute_c5_full(best_h, dx)
        idx_sorted = np.argsort(-c5_vals)
        active = list(idx_sorted[:top_m])
        
        def ineq(x):
            c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
            return x[n] - c[active]
        
        def eqf(x):
            return np.sum(x[:n]) - target_sum
        
        x0 = np.concatenate([best_h, [c5_best / dx + 0.005]])
        try:
            res = minimize(lambda x: x[n], x0, method='SLSQP', bounds=bnds,
                         constraints=[{'type': 'ineq', 'fun': ineq},
                                      {'type': 'eq', 'fun': eqf}],
                         options={'maxiter': max_iter, 'ftol': 1e-12})
            if res.x is not None:
                h_new = project_sr(res.x[:n], target_sum)
                c5_new, _, _ = compute_c5_full(h_new, dx)
                if c5_new < c5_best:
                    c5_best = c5_new
                    best_h = h_new
        except Exception:
            pass
        top_m = min(top_m + 3, 2 * n - 1)
    
    return best_h, c5_best

def coord_descent(h0, dx, target_sum, rounds=20, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_c5, _, _ = compute_c5_full(h, dx)
    best_h = h.copy()
    
    for rnd in range(rounds):
        if t_deadline is not None and time.time() > t_deadline:
            break
        any_improve = False
        for j in range(n):
            for delta in [0.05, -0.05, 0.1, -0.1, 0.2, -0.2, 0.35, -0.35, 0.5, -0.5]:
                h[j] = np.clip(h[j] + delta, 0.0, 1.0)
                h_adj = project_sr(h, target_sum)
                c5_t, _, _ = compute_c5_full(h_adj, dx)
                if c5_t < best_c5 - 1e-10:
                    best_c5 = c5_t
                    best_h = h_adj.copy()
                    h = h_adj.copy()
                    any_improve = True
                    break
            # restore if no improvement
            h = best_h.copy() if not any_improve else h
        if not any_improve:
            break
    return best_h, best_c5

def make_binary(n, target, rng):
    h = np.zeros(n)
    pos = rng.permutation(n)
    nf = int(target)
    fr = target - nf
    h[pos[:nf]] = 1.0
    if fr > 0.01:
        h[pos[nf]] = fr
    return project_sr(h, target)
