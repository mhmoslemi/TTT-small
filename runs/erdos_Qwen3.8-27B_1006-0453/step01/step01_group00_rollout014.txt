import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smax_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    n = len(h)
    dx = 2.0 / n
    if T < 1e-10:
        return c * dx
    sh = corr - c
    s = np.sum(np.exp(sh / T))
    return (c + T * np.log(max(s, 1e-300))) * dx

def fix_sum(h, n):
    target = n * 0.5
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-6) & (h < 1 - 1e-6)
    ni = int(np.sum(interior))
    if ni > 0:
        h[interior] += diff / ni
    else:
        h += diff / n
    return np.clip(h, 0, 1)

def grad_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] -= h[lag:]
            grad[lag:] += (1.0 - h[:m])
    else:
        l = -lag
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad

def subgradient_opt(h, n, n_iters, alpha0=2.0):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = grad_for_lag(h_curr, lag)
        alpha = alpha0 * 2.0 / (it + 50)
        h_new = h_curr - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c5

def slsqp_polish(h0, n, T, iters):
    def f(x):
        return smax_obj(x, T)
    def con(x):
        return np.sum(x) - n * 0.5
    r = opt.minimize(f, h0, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': con}],
                     options={'maxiter': iters, 'ftol': 1e-15})
    out = r.x
    out = fix_sum(out, n)
    return out
