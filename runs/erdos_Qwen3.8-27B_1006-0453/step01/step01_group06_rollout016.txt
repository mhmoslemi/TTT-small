import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        T = 1
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coordinate_descent_polish(h, n, max_rounds=30, time_limit=30.0):
    t0 = time.time()
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            # Ternary search for optimal h[i]
            a, b = 0.0, 1.0
            old_val = best[i]
            for _ in range(30):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                best[i] = m1
                f1 = compute_c5(best)
                best[i] = m2
                f2 = compute_c5(best)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            opt_val = (a + b) / 2.0
            best[i] = opt_val
            f_new = compute_c5(best)
            if f_new < best_c5 - 1e-12:
                best_c5 = f_new
                improved = True
            else:
                best[i] = old_val
        if not improved:
            break
    return best, best_c5

def slsqp_smooth(h, n, T, maxiter=500):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_c5, h, args=(T,),
            method='SLSQP', bounds=bnds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-13}
        )
        hc = project_feasible(np.clip(res.x, 0, 1), n)
        return hc
    except Exception:
        return h

    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        scale = 1.0 + diff / total
        h *= scale
    return np.clip(h, 0.0, 1.0)

def coordscent(h, max_rounds=30, time_limit=20.0, ternary_iters=20):
    n = len(h)
    t0 = time.time()
    best_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            return best_h, best_max * (2.0 / n)
        cur_max = best_max
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                return best_h, best_max * (2.0 / n)
            a, b = 0.0, 1.0
            old = h[i]
            for _ in range(ternary_iters):
                m1 = a + 0.3333 * (b - a)
                m2 = b - 0.3333 * (b - a)
                h[i] = m1
                f1 = np.max(np.correlate(h, 1.0 - h, mode='full'))
                h[i] = m2
                f2 = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            h[i] = 0.5 * (a + b)
            f_new = np.max(np.correlate(h, 1.0 - h, mode='full'))
            if f_new < cur_max - 1e-11:
                cur_max = f_new
                improved = True
            else:
                h[i] = old
        if improved and cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
        elif not improved:
            h = best_h.copy()
            break
    return best_h, best_max * (2.0 / n)

def slsqp_optimize(h, n, T, maxiter=400, topk=None):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        if topk is not None:
            res = minimize(smooth_c5_topk, h, args=(T, topk),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
        else:
            res = minimize(smooth_c5, h, args=(T,),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
        hc = project_feasible(np.clip(res.x, 0, 1), n)
        return hc
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP + coordscent from initial
    for T in [30, 100, 400, 1200, 3000]:
        if elapsed() > budget_s * 0.15:
            break
        h_new = slsqp_optimize(best_h, best_n, T, maxiter=500, topk=8)
        h_new, cv = coordscent(h_new, max_rounds=15, time_limit=budget_s * 0.05, ternary_iters=22)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_new

    # Phase 2: Heavy coordinate descent
    if elapsed() < budget_s * 0.30:
        best_h, best_c5 = coordscent(best_h, max_rounds=60, time_limit=budget_s * 0.12, ternary_iters=25)

    # Phase 3: Multi-start with structured perturbations
    n_starts = 40
    for i in range(n_starts):
        if elapsed() > budget_s * 0.72:
            break
        if i % 4 == 0:
            scale = 0.08 * np.exp(-0.04 * i) + 0.015
            h0 = best_h + np.random.normal(0, scale, size=best_n)
        elif i % 4 == 1:
            h0 = np.roll(best_h, np.random.randint(-best_n // 5, best_n // 5 + 1))
        elif i % 4 == 2:
            h0 = best_h.copy()
            bsz = max(3, best_n // 6)
            a_idx = np.random.randint(0, best_n - bsz)
            b_idx = np.random.randint(0, best_n - bsz)
            if abs(a_idx - b_idx) > bsz:
                h0[a_idx:a_idx + bsz], h0[b_idx:b_idx + bsz] = h0[b_idx:b_idx + bsz].copy(), h0[a_idx:a_idx + bsz].copy()
        else:
            h0 = 1.0 - best_h
            h0 = project_feasible(h0, best_n)

        h0 = project_feasible(h0, best_n)
        T_list = [200, 800] if i % 2 == 0 else [500]
        for T in T_list:
            if elapsed() > budget_s * 0.72:
                break
            h_new = slsqp_optimize(h0, best_n, T, maxiter=300, topk=6)
            h_new, cv = coordscent(h_new, max_rounds=12, time_limit=budget_s * 0.03, ternary_iters=20)
            if cv < best_c5:
                best_c5 = cv
                best_h = h_new

    # Phase 4: Try different n via interpolation
    for n_try in [70, 90, 110]:
        if elapsed() > budget_s * 0.88:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]])))
        h_interp = project_feasible(h_interp, n_try)
        for T in [300, 1000]:
            if elapsed() > budget_s * 0.88:
                break
            h_new = slsqp_optimize(h_interp, n_try, T, maxiter=400, topk=8)
            h_new, cv = coordscent(h_new, max_rounds=20, time_limit=budget_s * 0.04, ternary_iters=20)
            if cv < best_c5:
                best_c5 = cv
                best_h = h_new
                best_n = n_try

    # Phase 5: Final heavy polish
    if elapsed() < budget_s * 0.96:
        h_pol = slsqp_optimize(best_h, best_n, 5000, maxiter=600, topk=5)
        h_pol, cv = coordscent(h_pol, max_rounds=80, time_limit=budget_s * 0.04, ternary_iters=25)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_pol

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h))
    target = len(best_h) / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - best_h) if diff > 0 else np.argmin(best_h)
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
