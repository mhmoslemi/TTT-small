def project_onto_feasible(y, n):
    """Project y onto {0≤x≤1, sum(x)=n/2}"""
    target = n * 0.5
    # Bisection on shift t: x_j = clamp(y_j - t, 0, 1)
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2, 0, 1)
    # Fix any remaining sum error
    diff = target - x.sum()
    if abs(diff) > 1e-12:

        ni = interior.sum()
        if ni > 0:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n * 0.5
    lo, hi = float(y.min()) - 2.0, float(y.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-14:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        ni = int(mask.sum())
        if ni > 0:
            x[mask] += diff / ni
    return np.clip(x, 0.0, 1.0)

def build_coeff(fixed, n, opt_h):
    n_lags = 2 * n - 1
    C = np.zeros((n_lags, n))
    for m in range(n_lags):
        lag = m - (n - 1)
        if opt_h:
            lo = max(0, -lag)
            hi = min(n, n - lag)
        else:
            lo = max(0, lag)
            hi = min(n, n + lag)
        if lo < hi:
            j = np.arange(lo, hi)
            if opt_h:
                C[m, lo:hi] = fixed[j + lag]
            else:
                C[m, lo:hi] = fixed[j - lag]
    return C

def lp_solve(fixed, n, opt_h):
    C = build_coeff(fixed, n, opt_h)
    n_lags = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([C, -np.ones((n_lags, 1))])
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def alternating_lp(h0, n, max_rounds=500):
    h = h0.copy()
    t_prev = np.inf
    for r in range(max_rounds):
        v = np.clip(1.0 - h, 0.0, 1.0)
        v = project_feasible(v, n)
        h_new, t_h = lp_solve(v, n, True)
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        v_new = np.clip(1.0 - h_new, 0.0, 1.0)
        v_new = project_feasible(v_new, n)
        v_opt, t_v = lp_solve(h_new, n, False)
        if v_opt is None:
            h = h_new
            continue
        h_next = np.clip(1.0 - v_opt, 0.0, 1.0)
        h_next = project_feasible(h_next, n)
        if np.max(np.abs(h_next - h)) < 1e-10:
            h = h_next
            break
        h = h_next
        t_curr = max(t_h, t_v)
        if abs(t_curr - t_prev) < 1e-12 and r > 20:
            break
        t_prev = t_curr
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    return (cmax + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def grad_descent_polish(h0, n, T0, steps, t_budget):
    h = h0.copy()
    T = T0
    t0 = time.time()
    lr = 0.01
    for _ in range(steps):

        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        w = np.exp((corr - cmax) / T)

        for m in range(n_lags):
            lag = m - (n - 1)
            if abs(w[m]) < 1e-15:
                continue
            if lag >= 0:
                idx = np.arange(0, n - lag)
                lo = max(0, -lag)
                hi = min(n, n - lag)
                j = np.arange(hi - lo) + lo if lo < hi else np.array([], dtype=int)
                # d/dh[j] of sum_i h[i]*v[i+lag] for term w[m]
                # = w[m]*(1-h[j+lag])*1[valid] - w[m]*h[j-lag]*1[valid]

                    grad[jj] += w[m] * (1.0 - h[jj + lag])
