```python
def binary_local_search(h, N, max_iters, t0, budget, rng, cool_factor=0.9999):
    """Local search / SA for binary sequence."""
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full')
    c5 = float(np.max(corr))
    best_c5 = c5
    best_h = h.copy()
    temp = 1.0
    
    for it in range(max_iters):
        if time.time() - t0 > budget - 5:
            break
        # Pick random 1 and 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Swap
        h[i] = 0
        h[j] = 1
        
        corr = np.correlate(h, 1-h, mode='full')
        new_c5 = float(np.max(corr))
        
        if new_c5 <= c5 or rng.random() < np.exp((c5 - new_c5) / temp):
            c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1
            h[j] = 0
        
        temp *= cool_factor
    
    return best_h, best_c5 * dx
```

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, weights, N, dx):
    grad = np.zeros(N)
    N1 = N - 1
    for i in range(2 * N - 1):
        w = weights[i]
        if abs(w) < 1e-14:
            continue
        k = i - N1
        if k >= 0:
            nv = N - k
            grad[k:] += dx * w * (1.0 - h[:nv])
            grad[:nv] -= dx * w * h[k:]
        else:
            m = -k
            nv = N - m
            grad[:nv] += dx * w * (1.0 - h[m:])
            grad[m:] -= dx * w * h[:nv]
    return grad

def adam_optimize(h, N, target, dx, t0, budget, rng, lr_init=0.05, tau_init=5.0, max_iter=3000):
    best_c5 = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
    best_h = h.copy()
    lr = lr_init
    tau = tau_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 5:
            break
        corr = np.correlate(h, 1-h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Softmax weights
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(2*N-1) / (2*N-1)
        else:
            w /= ws
        
        grad = grad_smooth_max(h, w, N, 1.0)  # dx factored into the objective
        
        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)
        
        if (it + 1) % 300 == 0:
            lr *= 0.8
            tau *= 1.5
        
        if lr < 1e-9:
            break
    
    return best_h, best_c5

def grad_c5(h, weights, N):
    """Gradient of max(corr) approximated by softmax, w.r.t. h. Returns gradient of sum w_k*corr_k."""
    grad = np.zeros(N)
    N1 = N - 1

        w = weights[i]
        if abs(w) < 1e-15:

        k = i - N1

            grad[k:] += w * (1.0 - h[:nv])
            grad[:nv] -= w * h[k:]
        else:
            m = -k
            nv = N - m
            grad[:nv] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:nv]
    return grad

    bh = h.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    mb = np.zeros(N)
    vb = np.zeros(N)
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        corr = np.correlate(h, 1-h, 'full')
        cv = float(np.max(corr) * dx)
        if cv < best:
            best = cv
            bh = h.copy()
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        g = grad_c5(h, w, N)
        g /= (np.linalg.norm(g) + 1e-10)  # normalize
