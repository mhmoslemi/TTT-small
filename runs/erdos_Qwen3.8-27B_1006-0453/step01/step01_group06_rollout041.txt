    n_full = 2*n - 1
    c = np.correlate(h, 1.0-h, mode='full').copy()

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(h):
    return h.sum() - len(h) / 2.0

def project_sum(h):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)

        lo, hi = ts[best_idx - 1], ts[best_idx + 1]

        diffs2 = (ts2[:, None] - old) * coefs[None, :]
        c_mat2 = c[None, :] + diffs2
        c_mat2[:, n-1] = c[n-1] + ts2 * (1.0 - ts2) - old * (1.0 - old)
        maxes2 = np.max(c_mat2, axis=1)
        idx2 = int(np.argmin(maxes2))
        best_t = ts2[idx2]
        best_val = maxes2[idx2]
    
    h[i] = best_t
    return h

def cd_sweep(h, order):
    n = len(h)
    for i in order:
        h = cd_update_coord(h, i, n_grid=200)
    return h

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(h):

def project_sum(h):
    n = len(h)
