import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_single_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        ka = -k
        g[:n - ka] += (1.0 - h[ka:])
        g[n - ka:] -= h[:ka]
    return g

def grad_smooth(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_c = np.exp(beta * (corr - m))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        grad += w[k_idx] * grad_single_lag(h, lag)
    return grad * 2.0 / n

def slsqp_minimax(h0, dx, target_sum, maxiter=300):
    n = len(h0)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t_init = float(corr0.max())

    def obj(x):
        return x[n]

    def ineq(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - c

    def eqf(x):
        return np.sum(x[:n]) - target_sum

    x0 = np.concatenate([h0, [t_init + 0.002]])
    bnds = [(0.0, 1.0)] * n + [(0.0, t_init + 0.05)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def lkb_penalty(h0, dx, target_sum, maxiter=300, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 5.0
    for it in range(maxiter):
        if deadline is not None and time.time() > deadline:
            break
        beta = 50.0 + 100.0 * (it / maxiter)
        g = grad_smooth(h, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        for trial in range(10):
            step = lr * (0.5 ** trial)
            h_new = project_sr(h - step * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.3, 2.0)
                improved = True
                break
        if not improved:
            lr = max(lr * 0.5, 1e-4)
    return best_h, best_c5

def pair_perturb(h, target, scale, rng):
    n = len(h)
    d = np.zeros(n)
    for _ in range(max(3, n // 4)):
        i, j = rng.choice(n, 2, replace=False)
        d[i] += scale
        d[j] -= scale
    return project_sr(h + d, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5

    n = len(h_raw)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_sr(h_raw, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def time_ok(margin=5):
        return (time.time() - t_start) < budget_s - margin

    def polish(h, dx, target, deadline):
        for _ in range(3):
            if not time_ok(10):
                break
            hs = slsqp_minimax(h, dx, target, 200)
            cs = compute_c5(hs, dx)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs
                h = hs
        return h

    # Phase 1: SLSQP from initial
    if time_ok(30):
        h1 = slsqp_minimax(h0, dx, target, 500)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: L-BFGS penalty refinement
    if time_ok(60):
        deadline = t_start + budget_s - 300
        h2, c2 = lkb_penalty(best_h, dx, target, 3000, deadline)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP polish
    if time_ok(30):
        best_h = polish(best_h, dx, target, t_start + budget_s - 100)

    # Phase 4: Multi-start with structured perturbations + alternate solvers
    n_starts = 30
    for r in range(n_starts):
        if not time_ok(35):
            break
        scale = 0.02 + 0.06 * rng.random()
        hr = pair_perturb(best_h, target, scale, rng)
        # Alternate between SLSQP and L-BFGS
        if r % 2 == 0:
            hr = slsqp_minimax(hr, dx, target, 250)
        else:
            if time_ok(20):
                deadline = t_start + budget_s - 60
                hr, _ = lkb_penalty(hr, dx, target, 800, deadline)
            hr = slsqp_minimax(hr, dx, target, 150)
        cr = compute_c5(hr, dx)
        if cr < best_c5:
            if time_ok(25):
                deadline = t_start + budget_s - 50
                hr, cr = lkb_penalty(hr, dx, target, 600, deadline)
            hr = slsqp_minimax(hr, dx, target, 100)
            cr = compute_c5(hr, dx)
        if cr < best_c5:
            best_h, best_c5 = hr, cr

    # Phase 5: Try slightly different resolutions
    for n_alt in [49, 51, 53]:
        if not time_ok(60):
            break
        dx_a = 2.0 / n_alt
        target_a = n_alt / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        h_up = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
        h_up = project_sr(h_up, target_a)
        h_up = slsqp_minimax(h_up, dx_a, target_a, 250)
        c_up = compute_c5(h_up, dx_a)
        if c_up < best_c5:
            if time_ok(30):
                deadline = t_start + budget_s - 50
                h_up, c_up = lkb_penalty(h_up, dx_a, target_a, 600, deadline)
            h_up = slsqp_minimax(h_up, dx_a, target_a, 150)
            c_up = compute_c5(h_up, dx_a)
            if c_up < best_c5:
                best_h, best_c5 = h_up, c_up
                n = n_alt
                dx = dx_a
                target = target_a

    # Phase 6: Binary-like starts
    for r in range(8):
        if not time_ok(25):
            break
        hb = np.zeros(n)
        pos = rng.permutation(n)
        hb[pos[:int(target)]] = 1.0
        frac = target - int(target)
        if frac > 0.01:
            hb[pos[int(target)]] = frac
        hb = project_sr(hb, target)
        hb = slsqp_minimax(hb, dx, target, 150)
        cb = compute_c5(hb, dx)
        if cb < best_c5 and time_ok(20):
            deadline = t_start + budget_s - 40
            hb, cb = lkb_penalty(hb, dx, target, 400, deadline)
            hb = slsqp_minimax(hb, dx, target, 100)
            cb = compute_c5(hb, dx)
            if cb < best_c5:
                best_h, best_c5 = hb, cb

    # Phase 7: Final polish with tight SLSQP
    if time_ok(15):
        best_h = slsqp_minimax(best_h, dx, target, 300)
        best_c5 = compute_c5(best_h, dx)

    best_h = project_sr(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
