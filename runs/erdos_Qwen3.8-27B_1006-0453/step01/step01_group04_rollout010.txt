def interp_up(h, factor):
    n = len(h)
    m = n * factor
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(xn, xo, h)

def optimize_stages(h0, temps, pens, maxits, t0, deadline):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cur = h0.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    for i in range(len(temps)):
        if time.time() >= deadline:
            break
        res = minimize(obj_lse, cur, args=(temps[i], pens[i]),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxits[i], 'ftol': 1e-14, 'gtol': 1e-10})
        nh = project_h(res.x)
        c5v = compute_c5(nh)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = nh.copy()
            cur = nh
        else:
            cur = best_h.copy()
    return best_h, best_c5

def mutate_struct(h, rng):
    n = len(h)
    h2 = h.copy()
    t = rng.integers(5)
    if t == 0:
        s = rng.integers(0, n // 2)
        L = rng.integers(n // 15, n // 4)
        e = min(s + L, n)
        h2[s:e] = 1.0 - h2[s:e]
    elif t == 1:
        c = rng.uniform(0.1, 0.9) * n
        w = rng.uniform(n / 25, n / 8)
        a = rng.uniform(-0.4, 0.4)
        x = np.arange(n)
        h2 = h + a * np.exp(-0.5 * ((x - c) / w) ** 2)
    elif t == 2:
        s = rng.integers(0, n // 4)
        L = rng.integers(n // 8, n // 3)
        sh = rng.integers(-n // 12, n // 12 + 1)
        e = min(s + L, n)
        seg = h2[s:e].copy()
        h2[s:e] = np.roll(seg, sh)
    elif t == 3:
        h2 = h + rng.normal(0, 0.08, n)
    else:
        th = np.median(h)
        h2 = h + 0.2 * (h - th) + rng.normal(0, 0.03, n)
    return project_h(h2)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def rem(frac):
        return (t0 + budget_s * frac) - time.time()

    # Stage 1: Optimize at n=360 with fine temperature ladder
    cur = h_init.copy()
    temps1 = [1.0, 0.4, 0.2, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001]
    pens1 = [500, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000, 1000000]
    its1 = [400, 400, 400, 300, 300, 300, 250, 250, 200, 200]
    dl = t0 + budget_s * 0.30
    cur, lc = optimize_stages(cur, temps1, pens1, its1, t0, dl)
    if lc < best_c5:
        best_c5 = lc
        best_h = cur.copy()

    # Stage 2: Upscale to 720
    if rem(0.65) > 30:
        cur = project_h(interp_up(cur, 2))
        temps2 = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
        pens2 = [5000, 10000, 20000, 50000, 100000, 200000, 500000, 1000000, 2000000]
        its2 = [250, 250, 250, 200, 200, 200, 150, 150, 100]
        dl = t0 + budget_s * 0.62
        cur, lc = optimize_stages(cur, temps2, pens2, its2, t0, dl)
        if lc < best_c5:
            best_c5 = lc
            best_h = cur.copy()

    # Stage 3: Structural restarts (up to 6)
    n_cur = len(cur)
    for ri in range(6):
        if rem(0.12) < 20:
            break
        h_p = mutate_struct(cur, rng)
        temps3 = [0.05, 0.02, 0.008, 0.003, 0.001]
        pens3 = [50000, 100000, 300000, 500000, 1000000]
        its3 = [150, 150, 120, 100, 80]
        dl = t0 + budget_s * (0.92 - 0.015 * (6 - ri))
        h_p, lc = optimize_stages(h_p, temps3, pens3, its3, t0, dl)
        if lc < best_c5:
            best_c5 = lc
            best_h = h_p.copy()
        cur = h_p

    # Stage 4: Final ultra-fine polish on best
    if rem(0.02) > 10:
        temps4 = [0.0005, 0.0002, 0.0001, 0.00005]
        pens4 = [500000, 1000000, 2000000, 5000000]
        its4 = [200, 200, 150, 150]
        dl = t0 + budget_s * 0.98
        best_h, lc = optimize_stages(best_h, temps4, pens4, its4, t0, dl)
        if lc < best_c5:
            best_c5 = lc

    best_h = project_h(best_h)
    fc = compute_c5(best_h)
    if fc < best_c5:
        best_c5 = fc

    return best_h, best_c5, len(best_h)
