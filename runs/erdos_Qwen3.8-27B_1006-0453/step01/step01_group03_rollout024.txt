def slsqp_with_jac(h0, dx, target_sum, maxiter=500):
    n = len(h0)
    n_corr = 2*n - 1
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def constr_jac(x, k):
        # Jacobian of corr[k]*dx - t w.r.t. x
        h = x[:n]
        delta = k - n + 1
        row = np.zeros(n+1)
        for j in range(n):
            idx1 = j + delta
            if 0 <= idx1 < n:
                row[j] += dx * (1.0 - h[idx1])
            idx2 = j - delta
            if 0 <= idx2 < n:
                row[j] -= dx * h[idx2]
        row[n] = -1.0
        return row
    
    constraints = []
    for k in range(n_corr):
        # We only need the binding (or near-binding) constraints
        # But for correctness, include all
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, k=k: np.correlate(x[:n], 1-x[:n], mode='full')[k] * dx - x[n],
            'jac': lambda x, k=k: constr_jac(x, k)
        })
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - target_sum,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    x0 = np.concatenate([h0, [compute_c5(h0, dx) + 0.01]])
    bnds = [(0,1)]*n + [(0, n)]
    
    res = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bnds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.success or res.x is not None:
        return np.clip(res.x[:n], 0, 1)
    return h0.copy()

            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
    return h

def corr_k_val(h, k, n):
    d = k - n + 1
    if d >= 0:
        m = n - d
        return float(np.dot(h[:m], 1.0 - h[d:]))
    else:
        m = -d
        return float(np.dot(h[m:], 1.0 - h[:n - m]))

def corr_k_grad(h, k, n):
    d = k - n + 1
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        for j in range(m):
            grad[j] += (1.0 - h[j + d])
        for j in range(d, n):
            grad[j] -= h[j - d]
    else:
        m = -d
        for j in range(m, n):
            grad[j] += (1.0 - h[j - m])
        for j in range(n - m):
            grad[j] -= h[j + m]
    return grad

def slsqp_solve(h0, dx, target, maxiter=500, top_k=None, ftol=1e-14):
    n = len(h0)
    ncorr = 2 * n - 1
    c_all = np.correlate(h0, 1.0 - h0, mode='full')
    if top_k and top_k < ncorr:
        klist = sorted(int(x) for x in np.argsort(c_all)[-top_k:])
    else:
        klist = list(range(ncorr))
    t0 = float(c_all.max()) * dx + 0.02
    x0 = np.concatenate([h0, [t0]])

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = []
    for k in klist:
        def _cf(x, kk=k):
            return corr_k_val(x[:n], kk, n) * dx - x[n]
        def _cj(x, kk=k):
            h = x[:n]
            row = dx * corr_k_grad(h, kk, n)
            row = np.append(row, -1.0)
            return row
        cons.append({'type': 'ineq', 'fun': _cf, 'jac': _cj})

    cons.append({'type': 'eq', 'fun': lambda x: float(np.sum(x[:n]) - target),
                 'jac': lambda x: np.concatenate([np.ones(n), [0.0]])})
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_sum(res.x[:n], target)
    except Exception:
        pass
    return h0.copy()

def pair_search(h0, dx, target, max_sweeps=200, deadline=None, eps_list=None):
    n = len(h0)
    if eps_list is None:
        eps_list = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    h = h0.copy()
    best = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        found = False
        for j in range(n):
            for i in range(j + 1, n):
                for eps in eps_list:
                    if h[j] + eps <= 1.0 + 1e-10 and h[i] - eps >= -1e-10:
                        h[j] += eps
                        h[i] -= eps
                        c = compute_c5(h, dx)
                        if c < best - 1e-14:
                            best = c
                            found = True
                        else:
                            h[j] -= eps
                            h[i] += eps
                    if h[j] - eps >= -1e-10 and h[i] + eps <= 1.0 + 1e-10:
                        h[j] -= eps
                        h[i] += eps
                        c = compute_c5(h, dx)
                        if c < best - 1e-14:
                            best = c
                            found = True
                        else:
                            h[j] += eps
                            h[i] -= eps
                h = project_sum(h, target)
        if not found:
            break
    return h, best

def gradient_step(h0, dx, target, beta=80.0, max_iter=3000, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        w = np.exp(np.clip(beta * (corr - max_c), -50, 0))
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            grad += w[k] * corr_k_grad(h, k, n)
        grad *= dx
        lr = 0.08 / (1.0 + 0.02 * it)
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project_sum(h - lr * shrink * grad, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
                break
        else:
            if it > 200:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5

    n = len(h_raw)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_sum(h_raw, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    best_n = n

    def tl(margin=10):
        return (time.time() - t0) < budget_s - margin

    # Phase 1: SLSQP with all constraints
    if tl(80):
        h1 = slsqp_solve(h0, dx, target, maxiter=800)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Gradient descent with soft-max
    if tl(200):
        deadline = t0 + budget_s - 200
        h2, c2 = gradient_step(best_h, dx, target, beta=60, max_iter=4000, deadline=deadline)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Pair local search (coarse)
    if tl(200):
        deadline = t0 + budget_s - 200
        h3, c3 = pair_search(best_h, dx, target, max_sweeps=150, deadline=deadline,
                             eps_list=[0.01, 0.02, 0.05, 0.1, 0.2])
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: SLSQP polish (top-k)
    if tl(60):
        h4 = slsqp_solve(best_h, dx, target, maxiter=600, top_k=12)
        c4 = compute_c5(h4, dx)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Pair local search (fine)
    if tl(200):
        deadline = t0 + budget_s - 200
        h5, c5v = pair_search(best_h, dx, target, max_sweeps=300, deadline=deadline,
                              eps_list=[0.001, 0.002, 0.005, 0.01, 0.02])
        if c5v < best_c5:
            best_h, best_c5 = h5, c5v

    # Phase 6: Multi-start with perturbations
    for r in range(25):
        if not tl(25):
            break
        scale = 0.02 + 0.03 * (r % 3)
        h_r = best_h + scale * np.random.randn(best_n)
        h_r = project_sum(h_r, target)
        h_r = slsqp_solve(h_r, target * 0 + dx, target, maxiter=300, top_k=10)
        c_r = compute_c5(h_r, dx)
        if c_r < best_c5 and tl(15):
            h_r = pair_search(h_r, dx, target, max_sweeps=100, deadline=t0 + budget_s - 15)
            c_r = compute_c5(h_r, dx)
        if c_r < best_c5:
            best_h, best_c5 = h_r, c_r

    # Phase 7: Try different n values
    for n_try in [49, 50, 51, 53, 55]:
        if not tl(40):
            break
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        h_t = np.zeros(n_try)
        h_t[:int(target_t)] = 1.0
        fr = target_t - int(target_t)
        if fr > 0.01:
            h_t[int(target_t)] = fr
        h_t = slsqp_solve(h_t, dx_t, target_t, maxiter=400, top_k=15)
        c_t = compute_c5(h_t, dx_t)
        if tl(20):
            h_t2 = pair_search(h_t, dx_t, target_t, max_sweeps=50, deadline=t0 + budget_s - 20,
                               eps_list=[0.005, 0.01, 0.02, 0.05])
            c_t = compute_c5(h_t2, dx_t)
            if c_t < compute_c5(h_t, dx_t):
                h_t = h_t2
        if c_t < best_c5:
            best_h, best_c5, best_n = h_t, c_t, n_try

    # Phase 8: Final SLSQP + fine local search
    if tl(30):
        dx_f = 2.0 / best_n
        target_f = best_n / 2.0
        h_f = slsqp_solve(best_h, dx_f, target_f, maxiter=400, top_k=8)
        c_f = compute_c5(h_f, dx_f)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f
    if tl(60):
        dx_f = 2.0 / best_n
        target_f = best_n / 2.0
        h_f, c_f = pair_search(best_h, dx_f, target_f, max_sweeps=200,
                               deadline=t0 + budget_s - 60,
                               eps_list=[0.0005, 0.001, 0.002, 0.005])
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    # Final projection and evaluation
    dx_final = 2.0 / best_n
    target_final = best_n / 2.0
    best_h = project_sum(best_h, target_final)
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, best_n
