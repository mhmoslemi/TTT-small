def proj_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)

            break

            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def pgrad_obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    s = t * (corr - cmax)
    exp_s = np.exp(s)
    w = exp_s / np.exp_s.sum()
    # Softmax objective value
    obj = (cmax + np.log(np.exp_s.sum()) / t) * dx
    # Gradient
    w_rev = w[::-1]
    conv1 = np.convolve(h, w, 'full')
    conv2 = np.convolve(h, w_rev, 'full')
    idx = np.arange(n) + n - 1
    term1 = conv1[idx]
    term2 = conv2[idx]
    grad = dx * (1.0 - term1 - term2)
    return obj, grad

    obj, grad = pgrad_obj_and_grad(h, t)
    h_new = h - lr * grad
    h_new = proj_constraint(h_new, n)
    return h_new, obj
