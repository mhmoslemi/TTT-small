import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = h[free] + diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, sweeps=200):
    h = h.copy()
    lo_obj = None
    for sw in range(sweeps):
        obj = compute_c5(h, n)
        if lo_obj is not None and abs(obj - lo_obj) < 1e-12:
            break
        lo_obj = obj
        for i in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            a = np.zeros(2 * n - 1)
            for s in range(-(n - 1), n):
                idx = s + (n - 1)
                c = 0.0
                if 0 <= i + s < n:
                    c += 1.0 - h[i + s]
                if 0 <= i - s < n:
                    c -= h[i - s]
                a[idx] = c
            old = h[i]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo_d, hi_d = -old, 1.0 - old
            if hi_d - lo_d < 1e-14:
                continue
            dadp = (1.0 - gr) * (hi_d - lo_d)
            xa = lo_d + dadp
            xb = hi_d - dadp
            fa = np.max(corr + a * xa)
            fb = np.max(corr + a * xb)
            for _ in range(50):
                if fa < fb:
                    hi_d = xb; xb = xa; fb = fa
                    dadp = (1.0 - gr) * (hi_d - lo_d)
                    xa = lo_d + dadp; fa = np.max(corr + a * xa)
                else:
                    lo_d = xa; xa = xb; fa = fb
                    dadp = gr * (hi_d - lo_d)
                    xb = hi_d - dadp; fb = np.max(corr + a * xb)
            h[i] = np.clip(old + (lo_d + hi_d) / 2.0, 0.0, 1.0)
    return repair_h(h, n)

def smooth_c5(params, n, temp):
    dx = 2.0 / n
    corr = np.correlate(params, 1.0 - params, mode='full')
    m = np.max(corr)
    return (np.log(np.sum(np.exp(temp * (corr - m)))) / temp + m / temp) * dx

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def slsqq(x0, n, temp, maxiter):
    cons = {'type': 'eq', 'fun': sum_con, 'args': (n,)}
    res = minimize(smooth_c5, x0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[cons], args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair_h(res.x, n)

def make_inits(n, rng):
    inits = []
    # Uniform
    inits.append(np.full(n, 0.5))
    # Two blocks
    h = np.zeros(n)
    h[:n//4] = 1.0; h[n//2:3*n//4] = 1.0
    inits.append(h)
    # Random
    for _ in range(4):

    # Sinusoidal
    x = np.linspace(0, 2, n)
    for freq, amp in [(1, 0.4), (2, 0.4), (3, 0.3)]:
        h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0)
        inits.append(h)
    return inits

    t0 = time.time()
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Use initial_h_values if available

        if n_init not in n_list:

        if time.time() - t0 > budget_s - 60:

        inits = make_inits(n, rng)
        # If initial_h_values matches this n, use it
        if 'initial_h_values' in globals() and initial_h_values is not None:
            if len(initial_h_values) == n:
                inits.insert(0, np.array(initial_h_values, dtype=float).copy())
        
        for h0 in inits:
            if time.time() - t0 > budget_s - 40:
                break
            h0 = repair_h(h0, n)
            
            # Phase A: Coordinate descent
            h_cd = coord_descent(h0, n, sweeps=100)
            c5_cd = compute_c5(h_cd, n)
            
            # Phase B: SLSQP polish
            remaining = budget_s - 40 - (time.time() - t0)
            if remaining > 15:
                maxit = min(300, max(50, int(remaining * 3)))
                for temp in [20, 80]:
                    if time.time() - t0 > budget_s - 20:

                    h_sl = slsqq(h_cd, n, temp, maxit)
                    c5_sl = compute_c5(h_sl, n)
                    if c5_sl < c5_cd:

                        c5_cd = c5_sl
                # One more coord descent pass

            if c5_cd < best_c5:

                best_h = h_cd.copy()
                best_n = n
    
    # Final: if we have time, do more starts from best

                best_h = h_cd.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = h[free] + diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, sweeps=150):
    h = h.copy()
    prev_obj = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sw in range(sweeps):
        for i in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            a = np.zeros(2 * n - 1)
            for s in range(-(n - 1), n):
                idx = s + (n - 1)
                c = 0.0
                if 0 <= i + s < n:
                    c += 1.0 - h[i + s]
                if 0 <= i - s < n:
                    c -= h[i - s]
                a[idx] = c
            old = h[i]
            lo_d, hi_d = -old, 1.0 - old
            if hi_d - lo_d < 1e-14:
                continue
            dadp = (1.0 - gr) * (hi_d - lo_d)
            xa = lo_d + dadp
            xb = hi_d - dadp
            fa = np.max(corr + a * xa)
            fb = np.max(corr + a * xb)
            for _ in range(45):
                if fa < fb:
                    hi_d = xb
                    xb = xa
                    fb = fa
                    dadp = (1.0 - gr) * (hi_d - lo_d)
                    xa = lo_d + dadp
                    fa = np.max(corr + a * xa)
                else:
                    lo_d = xa
                    xa = xb
                    fa = fb
                    dadp = gr * (hi_d - lo_d)
                    xb = hi_d - dadp
                    fb = np.max(corr + a * xb)
            h[i] = np.clip(old + (lo_d + hi_d) * 0.5, 0.0, 1.0)
        cur = compute_c5(h, n)
        if cur >= prev_obj - 1e-12:
            break
        prev_obj = cur
    return repair_h(h, n)

def smooth_c5(params, n, temp):
    dx = 2.0 / n
    corr = np.correlate(params, 1.0 - params, mode='full')
    m = np.max(corr)
    return (np.log(np.sum(np.exp(temp * (corr - m)))) / temp + m / temp) * dx

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def slsqp_polish(x0, n, temp, maxiter):
    cons = {'type': 'eq', 'fun': sum_con, 'args': (n,)}
    try:
        res = minimize(smooth_c5, x0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[cons], args=(n, temp),
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair_h(res.x, n)
    except Exception:
        return x0

def make_inits(n, rng):
    inits = []
    inits.append(np.full(n, 0.5))
    # Two-block: 1 on [0,.5]-[1,1.5]
    h2 = np.zeros(n)
    h2[:n // 4] = 1.0
    h2[n // 2: 3 * n // 4] = 1.0
    inits.append(h2)
    # Three-block
    h3 = np.zeros(n)
    h3[:n // 6] = 1.0
    h3[n // 3: n // 2] = 1.0
    h3[2 * n // 3: 5 * n // 6] = 1.0
    inits.append(h3)
    # Sinusoidal patterns
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.45]:
            inits.append(0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0))
    # Random
    for _ in range(3):
        inits.append(rng.uniform(0.0, 1.0, n))
    return inits

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    n_list = [30, 36, 40, 45, 50, 54, 60]
    use_initial = ('initial_h_values' in globals() and initial_h_values is not None)
    if use_initial:
        n_init = len(initial_h_values)
        if n_init not in n_list:
            n_list.insert(0, n_init)
        # Put initial n first
        n_list.remove(n_init)
        n_list.insert(0, n_init)

    for n in n_list:
        if time.time() - t0 > budget_s - 80:
            break
        inits = make_inits(n, rng)
        if use_initial and len(initial_h_values) == n:
            inits.insert(0, np.array(initial_h_values, dtype=float).copy())

        for h0 in inits:
            if time.time() - t0 > budget_s - 50:
                break
            h0 = repair_h(h0, n)
            # Coordinate descent
            h_cd = coord_descent(h0, n, sweeps=120)
            c5_v = compute_c5(h_cd, n)
            # SLSQP polish
            rem = budget_s - 30 - (time.time() - t0)
            if rem > 20:
                mi = min(250, max(40, int(rem * 2)))
                for temp in [15, 60]:
                    if time.time() - t0 > budget_s - 15:
                        break
                    h_s = slsqp_polish(h_cd, n, temp, mi)
                    c5_s = compute_c5(h_s, n)
                    if c5_s < c5_v:
                        h_cd, c5_v = h_s, c5_s
                h_cd = coord_descent(h_cd, n, sweeps=60)
                c5_v = compute_c5(h_cd, n)
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h_cd.copy()
                best_n = n

    # Perturbation restarts from best
    if time.time() - t0 < budget_s - 10 and best_h is not None:
        n = best_n
        for trial in range(15):
            if time.time() - t0 > budget_s - 5:
                break
            scale = 0.08 + 0.12 / (trial + 1)
            cand = best_h + rng.uniform(-scale, scale, n)
            cand = repair_h(cand, n)
            hc = coord_descent(cand, n, sweeps=80)
            c5c = compute_c5(hc, n)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
            # Quick SLSQP if improved
            if time.time() - t0 < budget_s - 10:
                hs = slsqp_polish(best_h, n, 30, 80)
                c5s = compute_c5(hs, n)
                if c5s < best_c5:
                    best_c5 = c5s
                    best_h = hs.copy()
                best_h = coord_descent(best_h, n, sweeps=40)
                bc = compute_c5(best_h, n)
                if bc < best_c5:
                    best_c5 = bc

    final_c5 = compute_c5(best_h, best_n)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
