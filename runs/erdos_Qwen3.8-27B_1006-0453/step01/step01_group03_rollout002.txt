```python
    t0 = compute_c5(h0, dx)
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': cons_ineq, 'jac': cons_ineq_jac},
        {'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, jac=obj_grad,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_res = project(res.x[:n], target_sum)
        return h_res
    except Exception:
        return h0.copy()
```

Let me also add a "coordinate descent" polish step: after SLSQP, iterate over each coordinate and try to optimally set it (given all others fixed) to minimize the max correlation. This is like a cyclic coordinate descent.

For coordinate i, the correlation at lag k as a function of h[i]:
- If i is in the valid range for lag k, then corr[k] is linear in h[i]
- So corr[k] = a_k * h[i] + b_k for some a_k, b_k
- We want to minimize max_k (a_k * h[i] + b_k) * dx subject to 0 ≤ h[i] ≤ 1 and sum constraint

The sum constraint means: h[i] = target_sum - sum(h[j] for j≠i) + ... wait, no. The other coordinates are fixed, so h[i] is determined by the sum constraint. Unless we're allowed to violate it slightly and project.

Actually, in the projection approach, after changing h[i], we need to re-project to satisfy the sum constraint. This makes coordinate descent harder.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def _corr_jac_block(h, lag):
    n = len(h)
    grad = np.zeros(n)
    i_all = np.arange(n)
    m1 = (i_all + lag >= 0) & (i_all + lag < n)
    m2 = (i_all - lag >= 0) & (i_all - lag < n)
    grad[m1] += (1.0 - h[i_all[m1] + lag])
    grad[m2] += (-h[i_all[m2] - lag])
    return grad

def slsqp_analytic(h0, dx, target_sum, maxiter=600):
    n = len(h0)
    n_corr = 2 * n - 1

    def obj(v):
        return v[n]

    def obj_grad(v):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def cons_ineq(v):
        corr = np.correlate(v[:n], 1.0 - v[:n], mode='full')
        return v[n] - corr * dx

    def cons_ineq_jac(v):
        h = v[:n]
        J = np.ones((n_corr, n + 1)) * np.zeros(1)
        J[:, n] = 1.0
        for ki in range(n_corr):
            lag = ki - (n - 1)
            J[ki, :n] = -_corr_jac_block(h, lag) * dx
        return J

    def cons_eq(v):
        return np.sum(v[:n]) - target_sum

    def cons_eq_jac(v):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J

    t0 = compute_c5(h0, dx)
    x0 = np.concatenate([h0, [t0 + 0.02]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': cons_ineq, 'jac': cons_ineq_jac},
        {'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, jac=obj_grad,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def topm_subgrad(h0, dx, target_sum, n_iter=4000, top_m=3, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-top_m:]
        c_top = corr[idx]
        beta = 10.0 + 40.0 * (it / max(n_iter, 1))
        w = np.exp(beta * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(n)
        for ti, ki in enumerate(idx):
            lag = ki - (n - 1)
            grad += w[ti] * _corr_jac_block(h, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        lr = 0.06 / (1.0 + 0.008 * it)
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            h_new = project(h - lr * shrink * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
                improved = True
                break
    return best_h, best_c5

def coord_polish(h, dx, target_sum, n_rounds=15):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    trials = np.linspace(0.0, 1.0, 11)
    for rd in range(n_rounds):
        improved = False
        for i in range(n):
            cur = best[i]
            for trial in trials:
                if abs(trial - cur) < 0.005:
                    continue
                test = best.copy()
                test[i] = trial
                test = project(test, target_sum)
                c5_t = compute_c5(test, dx)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best = test
                    improved = True
        if not improved:
            break
    return best, best_c5

def pair_swap(h, dx, target_sum, n_trials=200, n_rounds=10):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(12345)
    for rd in range(n_rounds):
        improved = False
        for _ in range(n_trials):
            i, j = rng.choice(n, size=2, replace=False)
            if i == j:
                continue
            for delta in [0.05, 0.1, 0.15, 0.2]:
                for sign in [1, -1]:
                    d = sign * delta
                    hi, hj = best[i] + d, best[j] - d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        test = best.copy()
                        test[i], test[j] = hi, hj
                        c5_t = compute_c5(test, dx)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best = test
                            improved = True
                            break
                if improved:
                    break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5

    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project(h_raw, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def time_ok(margin=10):
        return (time.time() - t0) < budget_s - margin

    # Phase 1: SLSQP with analytical Jacobian
    if time_ok():
        h1 = slsqp_analytic(h0, dx, target_sum, 600)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Top-m subgradient
    if time_ok(80):
        deadline = t0 + budget_s - 150
        h2, c2 = topm_subgrad(best_h, dx, target_sum, 5000, top_m=3, deadline=deadline)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP polish
    if time_ok():
        h3 = slsqp_analytic(best_h, dx, target_sum, 500)
        c3 = compute_c5(h3, dx)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Coordinate descent polish
    if time_ok(20):
        h4, c4 = coord_polish(best_h, dx, target_sum, 12)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Pair swap
    if time_ok(15):
        h5, c5 = pair_swap(best_h, dx, target_sum, 300, 8)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: Multi-restart with SLSQP
    for r in range(25):
        if not time_ok(25):
            break
        scale = 0.03 + 0.12 * (r / 25.0)
        h_r = best_h + scale * np.random.randn(n)
        h_r = project(h_r, target_sum)
        h_r = slsqp_analytic(h_r, dx, target_sum, 300)
        c5_r = compute_c5(h_r, dx)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
            if time_ok(20):
                h_r2, c5_r2 = coord_polish(best_h, dx, target_sum, 8)
                if c5_r2 < best_c5:
                    best_h, best_c5 = h_r2, c5_r2

    # Phase 7: Try higher resolution n
    if time_ok(60):
        for n_trial in [59, 67]:
            if not time_ok(45):
                break
            dx_t = 2.0 / n_trial
            ts = n_trial / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_trial, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project(h_t, ts)
            h_t = slsqp_analytic(h_t, dx_t, ts, 400)
            c5_t = compute_c5(h_t, dx_t)
            if c5_t < best_c5:
                best_h, best_c5, n, dx = h_t, c5_t, n_trial, dx_t
                target_sum = ts

    # Phase 8: Final SLSQP + coord polish
    if time_ok(5):
        h_f = slsqp_analytic(best_h, dx, target_sum, 400)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f
        if time_ok(3):
            best_h, best_c5 = coord_polish(best_h, dx, target_sum, 6)

    best_h = project(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
```
