def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        return c_max * dx
    shifted = np.clip((corr - c_max) / T, -500, 0)
    s = np.sum(np.exp(shifted))
    if s < 1e-300:
        return c_max * dx
    return (c_max + T * np.log(s)) * dx

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        mask = corr >= c_max - 1e-10
        w_full = np.zeros(2 * n - 1)
        w_full[mask] = 1.0 / np.sum(mask)
    else:
        shifted = np.clip((corr - c_max) / T, -500, 0)
        w_full = np.exp(shifted)
        tw = np.sum(w_full)
        if tw < 1e-300:
            w_full = np.zeros(2 * n - 1)
            w_full[np.argmax(corr)] = 1.0
        else:
            w_full /= tw
    active = np.where(w_full > 1e-12)[0]
    grad = np.zeros(n)
    for ki in active:
        d = ki - (n - 1)
        w = w_full[ki]
        if d >= 0:
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] += -w * h[:n - d]
        else:
            pd = -d
            grad[pd:] += w * (1.0 - h[:n + d])
            grad[:n + d + 1] += -w * h[pd:]
    return dx * grad

            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    x = np.clip(y + lam, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        interior = (x > 1e-8) & (x < 1.0 - 1e-8)
        ni = np.sum(interior)
        if ni > 0:
            x[interior] += diff / ni
            x = np.clip(x, 0.0, 1.0)

    hn = np.interp(xn, xo, h, period=2.0)
    return project_box_sum(hn, new_n * 0.5)
