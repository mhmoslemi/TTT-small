def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0

        return val, grad

    def obj(hc):
        t = 200.0
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        mx = np.max(corr)
        s = t * (corr - mx)
        s = np.clip(s, -700, 700)
        return (mx + np.log(np.sum(np.exp(s))) / t) * (2.0 / n)
    
    def obj_grad(hc):
        t = 200.0
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        mx = np.max(corr)
        s = t * (corr - mx)

    bnds = [(0.0, 1.0)] * n
