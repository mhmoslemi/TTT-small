    dx = 2.0 / n

    s = np.sum(np.exp(shifted / T))
    soft_max = c_max + T * np.log(s)

import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def objective_fn(params, T, n, active_ids):
    h = params
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active_ids is not None:
        corr_use = np.concatenate([corr[active_ids], np.array([np.max(corr)])])
    else:
        corr_use = corr
    c_max = np.max(corr_use)
    shifted = corr_use - c_max
    s = np.sum(np.exp(shifted / T))
    soft = c_max + T * np.log(s)
    sum_pen = 0.05 * (np.sum(h) - 0.5 * n) ** 2
    return soft * dx + sum_pen

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-9) & (h < 1.0 - 1e-9)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    # Second pass for edge cases
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        nearest = np.argsort(np.abs(h - 0.5))[:max(1, n_int // 4)]
        h[nearest] += (target - s2) / len(nearest)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)

    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)

def structured_start(n, rng):
    """Generate diverse structured initial guesses."""
    starts = []
    # Block pattern: alternating 1s and 0s with variable block size
    for block in [2, 3, 4, 5, 7]:
        pattern = np.zeros(n)
        for i in range(0, n, block * 2):
            pattern[i:i + block] = 1.0
        s = np.sum(pattern)
        if s > 0:
            pattern *= (n * 0.5) / s
        pattern = np.clip(pattern, 0, 1)
        starts.append(pattern)
    # Fourier-based
    x = np.linspace(0, 2.0, n, endpoint=False)
    for k in [1, 2, 3, 4]:
        for amp in [0.3, 0.4, 0.5]:
            h_f = 0.5 + amp * np.cos(2 * np.pi * k * x / 2.0)
            h_f = np.clip(h_f, 0, 1)
            h_f = fix_sum(h_f, n)
            starts.append(h_f)
    # Random with structure
    for _ in range(4):
        base = np.random.uniform(0, 1, n)
        # Smooth it
        for _s in range(3):
            base = 0.5 * base + 0.25 * np.roll(base, 1) + 0.25 * np.roll(base, -1)
        base = np.clip(base, 0, 1)
        base = fix_sum(base, n)
        starts.append(base)
    return starts

def coordinate_descent(h, n, n_steps=200):
    """Try small perturbations to individual coordinates."""
    dx = 2.0 / n
    c5 = compute_c5(h)
    for step in range(n_steps):
        i = np.random.randint(n)
        old = h[i]
        if old <= 1e-8 or old >= 1.0 - 1e-8:
            continue
        for delta in [0.02, -0.02, 0.05, -0.05]:
            h[i] = old + delta
            if h[i] < 0 or h[i] > 1:
                h[i] = old
                continue
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-10:
                c5 = new_c5
            else:
                h[i] = old
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Downsample to n=70 and do thorough optimization
    n1 = 70
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    x_b = np.linspace(0.0, 2.0, n_best, endpoint=False)
    h1 = np.interp(x1, x_b, h_best)
    h1 = fix_sum(h1, n1)
    c1 = compute_c5(h1)
    if c1 < c5_best:
        c5_best = c1
        h_best = h1
        n_best = n1

    # Multi-start at n=70
    starts = structured_start(n1, rng)
    starts.append(h1)
    # Also add smoothed version of original
    h_smooth = h1.copy()
    for _s in range(5):
        h_smooth = 0.6 * h_smooth + 0.2 * np.roll(h_smooth, 1) + 0.2 * np.roll(h_smooth, -1)
    h_smooth = fix_sum(np.clip(h_smooth, 0, 1), n1)
    starts.append(h_smooth)

    for h_s in starts:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 120:
            break
        h_s = np.clip(h_s, 0, 1)
        h_s = fix_sum(h_s, n1)
        for T in [0.05, 0.01, 0.003, 0.001]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 60:
                break
            active = get_active_ids(h_s)
            h_opt = lgbfs_pass(h_s, n1, T, active, max_iter=2000)
            h_opt = fix_sum(h_opt, n1)
            c_opt = compute_c5(h_opt)
            if c_opt < c5_best:
                c5_best = c_opt
                h_best = h_opt
                n_best = n1
                h_s = h_opt.copy()

    # Phase 2: Progressive upsample and refine
    res_list = [140, 280, 400]
    for n_try in res_list:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 200:
            break
        h_curr = upsample(h_best, n_try)
        for T in [0.03, 0.01, 0.003, 0.001, 0.0003]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 80:
                break
            active = get_active_ids(h_curr)
            h_opt = lgbfs_pass(h_curr, n_try, T, active, max_iter=2500)
            h_opt = fix_sum(h_opt, n_try)
            c_opt = compute_c5(h_opt)
            if c_opt < c5_best:
                c5_best = c_opt
                h_best = h_opt
                n_best = n_try
                h_curr = h_opt.copy()
                active = get_active_ids(h_curr)
        # SLSQP polish at this resolution
        elapsed = time.time() - t_start
        if budget_s - elapsed > 60:
            active = get_active_ids(h_curr)
            h_opt = slsqp_pass(h_curr, n_try, 0.001, active, max_iter=200)
            h_opt = fix_sum(h_opt, n_try)
            c_opt = compute_c5(h_opt)
            if c_opt < c5_best:
                c5_best = c_opt
                h_best = h_opt
                n_best = n_try
                h_curr = h_opt.copy()

    # Phase 3: Random perturbations to escape local optima
    elapsed = time.time() - t_start
    n_pert = 8
    for i in range(n_pert):
        if budget_s - elapsed < 100:
            break
        scale = 0.03 + 0.02 * (i < n_pert // 2)
        noise = rng.randn(n_best) * scale
        h_pert = np.clip(h_best + noise, 0, 1)
        h_pert = fix_sum(h_pert, n_best)
        for T in [0.01, 0.003]:
            if budget_s - elapsed < 50:
                break
            active = get_active_ids(h_pert)
            h_opt = lgbfs_pass(h_pert, n_best, T, active, max_iter=1500)
            h_opt = fix_sum(h_opt, n_best)
            c_opt = compute_c5(h_opt)
            if c_opt < c5_best:
                c5_best = c_opt
                h_best = h_opt
                h_pert = h_opt.copy()
        elapsed = time.time() - t_start

    # Phase 4: Final refinement - very low temperature + coordinate descent
    elapsed = time.time() - t_start
    if budget_s - elapsed > 60:
        active = get_active_ids(h_best)
        h_opt = lgbfs_pass(h_best.copy(), n_best, 0.0002, active, max_iter=3000)
        h_opt = fix_sum(h_opt, n_best)
        c_opt = compute_c5(h_opt)
        if c_opt < c5_best:
            c5_best = c_opt
            h_best = h_opt

    elapsed = time.time() - t_start
    if budget_s - elapsed > 30:
        h_best, c5_cd = coordinate_descent(h_best.copy(), n_best, n_steps=300)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_best.copy()

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    n_final = len(h_best)
    return (h_best, c5_final, n_final)
