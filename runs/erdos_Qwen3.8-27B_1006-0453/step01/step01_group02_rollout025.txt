def optimize_subgrad(h, N, target, dx, t0, budget_s, lr_init=0.02, n_top=3, momentum=0.9):
    """Subgradient with momentum and top-K lags."""
    best_h = h.copy()
    best_c5, _ = compute_c5(h, dx)
    lr = lr_init
    vel = np.zeros(N)
    stagnant = 0
    
    while time.time() - t0 < budget_s - 8:
        c5_val, corr = compute_c5(h, dx)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Top-K
        k_eff = min(n_top, len(corr))
        top_idx = np.argsort(corr)[-k_eff:]
        
        # Gradient sum for top lags
        grad = np.zeros(N)
        for idx in top_idx:
            lag = int(idx) - (N - 1)
            grad += grad_lag(h, lag, N)
        grad /= k_eff
        
        # Normalized momentum step
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1 - momentum) * grad / gn
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        
        h_new = h - lr * vel / vn
        h_new = project_sum(h_new, target)
        
        # Accept/reject
        c5_new, _ = compute_c5(h_new, dx)
        if c5_new < c5_val:
            h = h_new
        else:
            lr *= 0.9
        
        # Stagnation handling
        if stagnant > 200:
            # Inject noise and reset
            h = best_h + rng_global * 0.02 * np.random.randn(N)
            h = project_sum(h, target)
            stagnant = 0
            lr = max(lr, 0.01)
            vel = np.zeros(N)
        
        if (stagnant % 500 == 0) and stagnant > 0 and stagnant < 200:
            lr *= 0.95
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += (1.0 - h[:N - lag])
            g[:N - lag] -= h[lag:]
    else:
        l = -lag
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def subgrad_optimize(h, N, target, dx, t0, budget_s, lr_init=0.02, n_top=3, mom=0.85, rng=None):

    best_c5, _ = compute_c5(h, dx)
