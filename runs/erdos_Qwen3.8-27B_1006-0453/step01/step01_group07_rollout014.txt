def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

    dx = 2.0 / n
    m = np.max(corr)
    s = np.minimum((corr - m) / beta, 0.0)
    return dx * (m + beta * np.log(np.sum(np.exp(s))))

def upsample(parent, new_n):
    old_n = len(parent)
    if new_n <= old_n:
        return parent.copy()
    ox = np.linspace(0.0, 1.0, old_n, endpoint=False)
    nx = np.linspace(0.0, 1.0, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

def grad_sm(h, n, beta):
    corr = get_corr(h)
    m = np.max(corr)
    s = np.minimum((corr - m) / beta, 0.0)
    w = np.exp(s)
    w /= w.sum()
    N = len(corr)
    grad = np.zeros(n)
    off = n - 1
    for k in range(N):
        if w[k] < 1e-14:
            continue
        sh = k - off
        if sh >= 0:
            idx = np.arange(n - sh)
            grad[idx + sh] += w[k] * (1.0 - h[idx])
            grad[idx] -= w[k] * h[idx + sh]
        else:
            sa = -sh
            idx = np.arange(sa, n)
            grad[idx] += w[k] * (1.0 - h[idx - sa])
            grad[idx - sa] -= w[k] * h[idx]
    return (2.0 / n) * grad
