import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
    return h

def grad_softmax(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    if cmax < 1e-30:
        return np.zeros(n)
    w = np.exp(beta * (corr - cmax))
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for lag in range(-(n-1), n):
            idx = lag + (n - 1)
            if j - lag >= 0 and j - lag < n:
                s += w[idx] * (1.0 - h[j-lag])
            if j + lag >= 0 and j + lag < n:
                s -= w[idx] * h[j+lag]
        grad[j] = s * dx
    return grad

def coordinate_descent(h, dx, target_sum, n_passes=3, t_end=None):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for p in range(n_passes):
        if t_end is not None and time.time() > t_end:
            break
        for j in np.random.permutation(n):
            if t_end is not None and time.time() > t_end:
                break
            old = h[j]
            for trial in [0.0, 1.0, old + 0.1, old - 0.1, old + 0.05, old - 0.05]:
                if trial < 0 or trial > 1:
                    continue
                h[j] = trial
                h2 = project_feasible(h, target_sum)
                c5 = compute_c5(h2, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    h = h2
                else:
                    h[j] = old
                    h = project_feasible(h, target_sum)
            h[j] = old
            h = project_feasible(h, target_sum)
    return h, best_c5

def projected_gradient(h0, dx, target_sum, n_iter, t_deadline):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iter):
        if time.time() > t_deadline:

        beta = 20.0 + 80.0 * (it / max(n_iter - 1, 1))
        lr = 0.06 / (1.0 + 0.01 * it)
        g = grad_softmax(h, dx, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:

    n = len(h0)
    def obj(x):

def project_feasible(h, target_sum):

        h[m] += d / m.sum()

def grad_topk(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    if cmax < 1e-30:
        return np.zeros(n)
    w = np.exp(beta * (corr - cmax))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for lag in range(-(n - 1), n):
            idx = lag + (n - 1)
            if j - lag >= 0 and j - lag < n:
                s += w[idx] * (1.0 - h[j - lag])
            if j + lag >= 0 and j + lag < n:
                s -= w[idx] * h[j + lag]
        grad[j] = s * dx
    return grad

    def obj_fun(x):
        return x[n]
    def ineq_fun(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    t_init = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t_init + 0.003]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq_fun}, {'type': 'eq', 'fun': eq_fun}]
    try:
        res = minimize(obj_fun, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})

            return project_feasible(res.x[:n], target_sum)
        if res.x is not None:
            hx = project_feasible(res.x[:n], target_sum)
            if compute_c5(hx, dx) < compute_c5(h0, dx):

    except Exception:
        pass

                h0, c5 = h3, c3
        if time_ok(20):
            h4, c4 = coord_local(h0, dx, ts, t0 + budget_s * 0.45, n_sweeps=3)
            if c4 < c5:
                h0, c5 = h4, c4
        if c5 < best_c5:
            best_h, best_c5, best_n = h0, c5, n

    # Sweep over n values
    n_list = []
    if h_raw is not None:
        n_list = [x for x in [45, 47, 49, 51, 53, 55, 57, 59, 61, 63, 65, 67, 69] if x != len(h_raw)]
    else:
        n_list = [47, 49, 51, 53, 55, 57, 59, 61, 63, 65]

    n_list.sort(key=lambda x: abs(x - best_n) if best_n > 0 else x)

    for n in n_list:
        if not time_ok(50):
            break
        dx = 2.0 / n
        ts = n / 2.0
        starts = make_starts(n, dx, ts, rng)
        # Also add slightly perturbed version of best if available
        if best_h is not None and time_ok(20):
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 2, best_n, endpoint=False) + (2.0 / best_n) / 2
            x_new = np.linspace(0, 2, n, endpoint=False) + dx / 2
            f = interp1d(x_old, best_h, kind='linear', fill_value=(best_h[0], best_h[-1]), bounds_error=False)
            h_interp = np.clip(f(x_new), 0, 1)
            h_interp = project_feasible(h_interp, ts)
            starts.append(h_interp)

        for hs in starts:
            if not time_ok(25):
                break
            c5s = compute_c5(hs, dx)
            if c5s >= best_c5 + 0.005:
                continue
            if time_ok(20):
                hs1 = slsqp_solve(hs, dx, ts, 150)
                c5s1 = compute_c5(hs1, dx)
                if c5s1 < c5s:
                    hs, c5s = hs1, c5s1
            if c5s < best_c5 and time_ok(40):
                dl = t0 + budget_s - 100
                h2, c2 = gradient_descent(hs, dx, ts, 2000, dl)
                if c2 < c5s:
                    hs, c5s = h2, c2
            if c5s < best_c5 and time_ok(15):
                hs3 = slsqp_solve(hs, dx, ts, 150)
                c5s3 = compute_c5(hs3, dx)
                if c5s3 < c5s:
                    hs, c5s = hs3, c5s3
            if c5s < best_c5:
                best_h, best_c5, best_n = hs, c5s, n

    # Final polish on best
    if best_h is not None and time_ok(15):
        dx = 2.0 / best_n
        ts = best_n / 2.0
        hf = slsqp_solve(best_h, dx, ts, 200)
        cf = compute_c5(hf, dx)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    if best_h is not None and time_ok(10):
        dx = 2.0 / best_n
        ts = best_n / 2.0
        hcl, ccl = coord_local(best_h, dx, ts, t0 + budget_s - 3, n_sweeps=2)
        if ccl < best_c5:
            best_h, best_c5 = hcl, ccl

    # Final verification
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
