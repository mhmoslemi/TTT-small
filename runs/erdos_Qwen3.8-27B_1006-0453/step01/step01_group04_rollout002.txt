def smooth_obj(h, temp, pen):
    n = len(h)
    dx = 2.0 / n

    cmax = np.max(corr)
    sm = cmax + temp * np.log(np.sum(np.exp((corr - cmax) / temp)))
    integral = np.sum(h) * dx

    m = n * f
    xo = (np.arange(n) + 0.5) * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sm = cmax + temp * np.log(np.sum(np.exp((corr - cmax) / temp)))
    integral = np.sum(h) * dx
    return sm * dx + pen * (integral - 1.0) ** 2

def interp_up(h, f):
    n = len(h)
    m = n * f
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(xn, xo, h)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)

    s = np.sum(h) * (2.0 / n)

        return h
    return np.clip(h / s, 0.0, 1.0)

    return np.clip(res.x, 0.0, 1.0)
