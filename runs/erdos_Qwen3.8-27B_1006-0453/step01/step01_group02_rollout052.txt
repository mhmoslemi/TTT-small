def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Use binary search on the offset
    lo, hi = -2.0, 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def grad_lag(h, d, N):
    g = np.zeros(N)
    if d >= 0:
        if d < N:
            nr = np.arange(0, N - d)
            g[nr] += (1.0 - h[nr + d])
            g[nr + d] -= h[nr]
    else:
        l = -d
        if l < N:
            nr = np.arange(l, N)
            g[nr] += (1.0 - h[nr - l])
            g[nr - l] -= h[nr]
    return g

    grad = np.zeros(N)
    active = np.where(weights > 1e-10)[0]

        d = int(idx) - (N - 1)
        wi = float(weights[idx])

                grad[nr] += wi * (1.0 - h[nr + d])
                grad[nr + d] -= wi * h[nr]

                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Final polish with tighter parameters
    N_best = len(best_h)
    dx_best = 2.0 / N_best
    if time.time() - t0 < budget_s - 12:
        h_fin, c5_fin = optimize_core(best_h.copy(), N_best, N_best / 2.0, dx_best,
                                      t0, budget_s, lr_init=0.005, tau_init=8.0, binarize=0.003)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
        h_ls, c5_ls = binary_local_search(best_h.copy(), N_best, N_best / 2.0, dx_best,
                                          t0, budget_s, rng, n_trials=200)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
