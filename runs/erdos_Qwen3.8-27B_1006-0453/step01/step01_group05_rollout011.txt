import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n, max_iter=300):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nm = np.sum(mask)
        if nm == 0:
            h += diff / n
            h = np.clip(h, 0, 1)
        else:
            h[mask] = np.clip(h[mask] + diff / nm, 0, 1)
    return h


def obj_lse(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(corr - m)
    return (m + np.log(np.sum(e)) / T) * dx


def obj_topk(h, n, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(corr)[::-1]
    return np.mean(corr[idx[:k]])


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def slsqp_step(x0, n, T, maxiter, bounds):
    res = minimize(obj_lse, x0, args=(n, T), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x


def lfbgb_step(x0, n, T, lam, maxiter, bounds):
    def pen_obj(x):
        hx = np.clip(x, 0, 1)
        base = obj_lse(hx, n, T)
        return base + lam * (np.sum(hx) - n / 2.0) ** 2
    res = minimize(pen_obj, x0, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})

        h_t = repair_h(h_t, n)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
            cand = h_t.copy()
    return best_h, best_c5
