from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    # Project h to [0,1]^n with sum = target_sum
    h = np.asarray(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return h

def slsqp_optimize(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return compute_c5(x, dx)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_out = project_feasible(res.x, target_sum)
            if compute_c5(h_out, dx) <= compute_c5(h0, dx) + 1e-12:
                return h_out
    except Exception:
        pass
    return h0.copy()

def logsumexp_gradient(h, dx, beta=100.0):
    n = len(h)
    c = corr_values(h) * dx
    m = c.max()
    exp_c = np.exp(beta * (c - m))
    w = exp_c / exp_c.sum()
    # Gradient of log-sum-exp wrt h[i]
    # ∂c[lag]/∂h[i]: h[i] appears in c as both a and b terms
    g = np.zeros(n)
    # c[m] = sum_j h[j + m - n + 1] * (1 - h[j])
    # ∂c[m]/∂h[i] = (1-h[i-m+n-1])*[valid] - h[i+m-n+1]*[valid]
    for i in range(n):
        s = 0.0
        # h[i] on a-side: j+m-n+1 = i => j = i-m+n-1, contribution (1-h[j])
        for m in range(2*n-1):
            j = i - m + n - 1
            if 0 <= j < n:
                s += w[m] * (1.0 - h[j])
        # h[i] on b-side (1-h): j = i, contribution -h[i+m-n+1]
        for m in range(2*n-1):
            j = i + m - n + 1
            if 0 <= j < n:
                s -= w[m] * h[j]
        g[i] = s * dx
    return g

    best_c5 = compute_c5(h, dx)
    t0 = time.time()

        if time.time() - t0 > time_limit:
            break

            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
                break
    return best_h, best_c5

        frac = target_sum - ho
        if frac > 0.005:
            h[pos[ho]] = frac
        starts.append(project_feasible(h, target_sum))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 1)
    t_start = time.time()

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = None

    best_h = None
    best_c5 = float('inf')

    def t_left(margin=10):
        return budget_s - (time.time() - t_start) - margin

    n_values = [47, 51, 55]

    for n in n_values:
        if t_left(80) <= 0:
            break
        dx = 2.0 / n
        target_sum = n / 2.0
        starts = make_starts(n, target_sum, h_parent, rng)

        for si, h0 in enumerate(starts):
            if t_left(30) <= 0:
                break
            # Stage 1: SLSQP
            h1 = slsqp_optimize(h0, dx, target_sum, 200)
            c5_1 = compute_c5(h1, dx)
            # Stage 2: Gradient descent
            if t_left(25) > 5:
                h2, c5_2 = grad_descent(h1, dx, target_sum, 300, t_limit=18)
                if c5_2 < c5_1:
                    h1, c5_1 = h2, c5_2
            # Stage 3: Pairwise
            if t_left(20) > 5:
                h3, c5_3 = pairwise_descent(h1, dx, target_sum, 35, t_limit=15)
                if c5_3 < c5_1:
                    h1, c5_1 = h3, c5_3
            # Stage 4: SLSQP polish
            if t_left(15) > 5:
                h4 = slsqp_optimize(h1, dx, target_sum, 150)
                c5_4 = compute_c5(h4, dx)
                if c5_4 < c5_1:
                    h1, c5_1 = h4, c5_4
            # Stage 5: Final pairwise
            if t_left(10) > 3:
                h5, c5_5 = pairwise_descent(h1, dx, target_sum, 20, t_limit=8)
                if c5_5 < c5_1:
                    h1, c5_1 = h5, c5_5
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()

    # Refinement loop on best
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    for r in range(8):
        if t_left(25) <= 0:
            break
        h_p = best_h + 0.04 * rng.randn(n)
        h_p = project_feasible(h_p, target_sum)
        h_p = slsqp_optimize(h_p, dx, target_sum, 150)
        h_p, c_p = grad_descent(h_p, dx, target_sum, 200, t_limit=12)
        h_p, c_p = pairwise_descent(h_p, dx, target_sum, 30, t_limit=10)
        if c_p < best_c5:
            best_c5 = c_p
            best_h = h_p.copy()

    # Also try structured perturbations of best
    for r in range(4):
        if t_left(20) <= 0:
            break
        h_s = best_h.copy()
        idx = rng.choice(n, size=6, replace=False)
        for i in idx:
            h_s[i] = rng.random()
        h_s = project_feasible(h_s, target_sum)
        h_s = slsqp_optimize(h_s, dx, target_sum, 150)
        h_s, c_s = pairwise_descent(h_s, dx, target_sum, 30, t_limit=12)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()

    # Final polish
    if t_left(10) > 5:
        h_f = slsqp_optimize(best_h, dx, target_sum, 100)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f
        h_f, c_f = pairwise_descent(best_h, dx, target_sum, 15, t_limit=5)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
