def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(1000):
        d = tgt - np.sum(h)
        if abs(d) < 1e-16:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        step = d / cnt
        # Don't overshoot
        if d > 0:
            step = min(step, 1.0 - h[mask].max() if cnt > 0 else 0)
        else:
            step = max(step, h[mask].min() if cnt > 0 else 0)
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, temp, n, lambda_pen=1e4):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    val = (mx + np.log(np.sum(np.exp(s))) / temp) * dx
    val += lambda_pen * (np.sum(h) - n/2.0)**2
    return val

def lse_grad(h, temp, n, lambda_pen=1e4):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    p = np.exp(s - np.max(s))
    p /= np.sum(p)
    grad = dx * weighted_corr_grad(h, p, n)
    grad += 2 * lambda_pen * (np.sum(h) - n/2.0)
    return grad

def weighted_corr_grad(h, p, n):
    grad = np.zeros(n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-18:
            continue
        s = k - (n-1)

            grad[:n-s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n-s]
        else:
            ss = -s
            grad[ss:] += pk * (1.0 - h[:n-ss])
            grad[:n-ss] -= pk * h[ss:]
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    val = np.dot(p, corr)
    val += lambda_pen * (np.sum(h) - n/2.0)**2
    return val

    grad = weighted_corr_grad(h, p, n)
    grad += 2 * lambda_pen * (np.sum(h) - n/2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(2000):
        d = tgt - np.sum(h)
        if abs(d) < 1e-16:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def wcorr_grad(h, p, n):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n - s]
        else:
            ss = -s
            grad[ss:] += pk * (1.0 - h[:n - ss])
            grad[:n - ss] -= pk * h[ss:]
    return grad

def lse_val(h, temp, n, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    s -= np.max(s)
    val = (mx + np.log(np.sum(np.exp(s))) / temp) * dx
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def lse_grad_fn(h, temp, n, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    s -= np.max(s)
    p = np.exp(s)
    p /= np.sum(p)
    g = dx * wcorr_grad(h, p, n)
    g += 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

def wobj_fun(h, p, n, lam):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.dot(p, corr) + lam * (np.sum(h) - n / 2.0) ** 2

def wobj_grad(h, p, n, lam):
    g = wcorr_grad(h, p, n)
    g += 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

def seq_fun(x, n):
    return np.sum(x) - n / 2.0

def seq_jac(x, n):
    return np.ones(n)

def make_wfun(p, n, lam):
    def f(hv):
        return wobj_fun(hv, p, n, lam)
    return f

def make_wgrad(p, n, lam):
    def g(hv):
        return wobj_grad(hv, p, n, lam)
    return g

def make_lse_f(temp, n, lam):
    def f(hv):
        return lse_val(hv, temp, n, lam)
    return f

def make_lse_g(temp, n, lam):
    def g(hv):
        return lse_grad_fn(hv, temp, n, lam)
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = 280
    bnds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        if len(h) != n:
            xs = np.arange(len(h)) * 2.0 / len(h)
            xd = np.arange(n) * 2.0 / n
            h = np.interp(xd, xs, h)
        h = proj_sum(h, n)
    except Exception:
        h = np.full(n, 0.5)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Subgradient / cutting-plane with L-BFGS-B inner solves
    t1_end = t0 + 0.32 * budget_s
    for it in range(80):
        if time.time() > t1_end - 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        temp_pg = 25.0 + 15.0 * min(it, 20)
        s_arr = temp_pg * (corr - mx)
        s_arr -= np.max(s_arr)
        p_w = np.exp(s_arr)
        p_w /= np.sum(p_w)

        lam = 1e5
        try:
            res = minimize(make_wfun(p_w, n, lam), h,
                          jac=make_wgrad(p_w, n, lam),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9})

            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass

            if c5_new < best_c5:

                h = h_new

            break
        try:
            res = minimize(lambda x: lse_val(x, temp, n, 0.0), h,
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-17})
            h_new = proj_sum(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    # Phase 4: Coordinate descent polish
    deltas = [0.002, -0.002, 0.005, -0.005, 0.01, -0.01, 0.001, -0.001]
    for _ in range(5):
        if time.time() > t0 + budget_s - 8:
            break
        h_cur = best_h.copy()
        improved_any = False
        for i in range(n):
            if time.time() > t0 + budget_s - 5:
                break
            old_v = h_cur[i]
            for dlr in deltas:
                nv = old_v + dlr
                if nv < 0 or nv > 1:
                    continue
                h_cur[i] = nv
                d_sum = n / 2.0 - np.sum(h_cur)
                if abs(d_sum) > 1e-14:
                    others = np.ones(n, dtype=bool)
                    others[i] = False
                    if d_sum > 0:
                        others &= (h_cur < 1.0)
                    else:
                        others &= (h_cur > 0.0)
                    cnt_o = np.count_nonzero(others)
                    if cnt_o == 0:
                        h_cur[i] = old_v
                        continue
                    h_cur[others] += d_sum / cnt_o
                    h_cur = np.clip(h_cur, 0.0, 1.0)
                c5_try = compute_c5(h_cur)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_cur.copy()
                    improved_any = True
                    break
            h_cur[i] = old_v
        if not improved_any:
            break

    best_h = proj_sum(best_h, n)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
