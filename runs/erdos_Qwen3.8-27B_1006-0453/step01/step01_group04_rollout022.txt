def power_smooth(h, p, pen):
    """Power-mean smooth max of correlation values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Shift to positive
    cmin = np.min(corr)
    shifted = corr - cmin + 0.001
    power_mean = (np.sum(shifted**p) / len(shifted))**(1.0/p) + cmin - 0.001
    integ = np.sum(h) * dx - 1.0
    return power_mean + pen * integ * integ

def logsumexp_smooth(h, temp, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integ = np.sum(h) * dx - 1.0
    return sm * dx + pen * integ * integ

def project_feasible(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    diff = np.sum(h) - target
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(h)[::-1]
        for idx in order:
            if diff <= 0:
                break
            room = h[idx]
            step = min(room, diff / max(1, np.sum(h >= h[idx] - 1e-12)))
            h[idx] -= step
            diff -= step
    else:
        order = np.argsort(h)
        for idx in order:
            if diff >= 0:
                break
            room = 1.0 - h[idx]
            h[idx] += min(room, -diff)
            diff += min(room, -diff)
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):

    n = len(h)

    sm = cmax + temp * np.log(np.sum(np.exp(s / temp)))
    integ = np.sum(h) * dx - 1.0

def power_smooth(h, p, pen):
    n = len(h)
    dx = 2.0 / n

    pm = (np.sum(shifted ** p) / len(shifted)) ** (1.0 / p) + cmin - 1e-6
