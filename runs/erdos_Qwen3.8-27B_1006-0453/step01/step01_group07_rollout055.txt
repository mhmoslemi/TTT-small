    """Project h onto {0<=h<=1, sum(h)=target_sum}."""

    # Bisection on tau: find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(60):

        h_proj = np.clip(h - tau, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix any numerical issues
    h_proj += (target_sum - np.sum(h_proj)) / n
    return np.clip(h_proj, 0, 1)

            if budget_s - (time.time() - t0) < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            improved = False
            for d in [0.015, -0.015, 0.008, -0.008, 0.025, -0.025]:
                h[i] += d
                h[j] -= d
                if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                    c = compute_c5(h)
                    if c < cur - 1e-13:
                        cur = c
                        improved = True
                        break
                    h[i] -= d
                    h[j] += d
            if not improved:
                h[i] -= d
                h[j] += d
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
    
    # Ensure exact constraint satisfaction
    n = len(best_h)
    best_h += (n / 2.0 - np.sum(best_h)) / n
    best_h = np.clip(best_h, 0, 1)
    # Final correction for rounding
    best_h += (n / 2.0 - np.sum(best_h)) / n
    best_h = np.clip(best_h, 0, 1)
    
    return best_h, best_c5, len(best_h)
