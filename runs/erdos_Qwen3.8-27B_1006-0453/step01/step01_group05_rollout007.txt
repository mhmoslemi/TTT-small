import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def sum_diff(x):
    return x.sum() - x.size / 2.0


def sum_jac(x):
    return np.ones_like(x)


def lse_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = (corr - m) / temp
    return dx * (m / temp + np.log(np.mean(np.exp(s))))


def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(mask.sum())
        if nf == 0:
            h = np.clip(h + d / n, 0.0, 1.0)
        else:
            h[mask] = np.clip(h[mask] + d / nf, 0.0, 1.0)
    return h


def interp_h(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def slsqp_opt(h0, n, temp, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_diff, 'jac': sum_jac}]
    res = minimize(lse_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x


def subgrad_refine(h, n, n_steps, step_size):
    dx = 2.0 / n
    h = h.copy()
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s_shift = k_star - (n - 1)
        grad = np.zeros(n)
        if s_shift >= 0:
            for i in range(n - s_shift):
                j = i + s_shift
                grad[j] += (1.0 - h[i])
                grad[i] -= h[j]
        else:
            ss = -s_shift
            for i in range(ss, n):
                j = i - ss
                grad[i] += (1.0 - h[j])
                grad[j] -= h[i]
        lr = step_size / (1.0 + step * 0.05)
        h = h - lr * grad
        h = repair(h, n)
    return h

    return lse_obj(h, n, temp)

                   options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14})
    return res.x
