One more idea: **use the gradient of the true max (not smooth max) for the final few iterations.** At very low temperatures, the smooth max ≈ true max, and the gradient is well-defined (it's the gradient of the argmax term). I can compute this directly:

```python
def exact_max_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    # Gradient of corr[k_max] with respect to h
    # corr[k] = sum_i h[i] * (1 - h[i+k]) for valid i
    # For k >= 0: corr[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    #   d/dh[j]: (1-h[j+k]) for j <= n-1-k, minus h[j-k] for j >= k
    # For k < 0: corr[k] = sum_{i=-k}^{n-1} h[i]*(1-h[i+k])
    #   Let j = -k >= 0: corr[-j] = sum_{i=j}^{n-1} h[i]*(1-h[i-j])
    #   d/dh[m]: (1-h[m-j]) for m >= j, minus h[m+j] for m <= n-1-j
    
    grad = np.zeros(n)
    # offset in np.correlate: full mode, so index k_max corresponds to lag...
    # np.correlate(a, v, 'full')[k] = sum_i a[i+k] * v[i]
    # Wait, let me check: np.correlate(h, 1-h, 'full')[m]
    # This is sum_i h[i] * (1-h)[i + m - (n-1)]
    # Actually the convention: for 'full', output has length 2n-1
    # corr[m] = sum_i h[i] * (1-h)[i + m - (n-1)] for appropriate range
    # Hmm, let me just use: corr[k] = sum_{i} h[i+k] * (1-h)[i] where indices are valid
    # For k in [-(n-1), n-1]
    # In 'full' mode, index m corresponds to k = m - (n-1)
    
    k = k_max - (n - 1)  # actual lag
    
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} (1-h)[i] * h[i+k]
        # d/dh[j] = sum over contributions where h[j] appears
        # h[j] appears as h[i+k] when i = j-k, so i in [0, n-1-k], j in [k, n-1]
        # Also h[j] appears in (1-h)[i] when i = j, so (1-h[j])*h[j+k] for j in [0, n-1-k-1]... 
        # Wait: (1-h)[i] = 1 - h[i], so d(1-h[i])/dh[j] = -delta(i,j)
        # d corr/dh[j] = (1-h[j]) * (contrib from h[j+k] term) + (-h[j+k]) * (contrib from (1-h[j]) term)
        # h[j] appears in h[i+k] when i+k = j, i.e., i = j-k. Valid if 0 <= j-k <= n-1-k, i.e., k <= j <= n-1.
        #   Contribution: (1-h[j-k])
        # h[j] appears in (1-h[i]) when i = j. Valid if 0 <= j <= n-1-k.
        #   Contribution: -h[j+k]
        
        for j in range(k, n):
            grad[j] += (1.0 - h[j-k])
        for j in range(0, n-k):
            grad[j] -= h[j+k]
    else:
        j = -k  # positive shift
        # corr = sum_{i=j}^{n-1} (1-h)[i] * h[i-j]
        # h[j] appears in h[i-j] when i-j = j, i = 2j... no wait.
        # h[m] appears in h[i-j] when i-j = m, i = m+j. Valid if j <= m+j <= n-1, i.e., m in [0, n-1-j].
        #   Contribution: (1-h[m+j])
        # h[m] appears in (1-h[i]) when i = m. Valid if j <= m <= n-1.
        #   Contribution: -h[m-j]
        
        for m in range(0, n-j):
            grad[m] += (1.0 - h[m+j])
        for m in range(j, n):
            grad[m] -= h[m-j]
    
    return grad * dx
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_fn(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr / n
    cmax = np.max(c)
    shifted = (c - cmax) / temp
    shifted -= np.max(shifted)
    sm = (cmax + temp * np.log(np.sum(np.exp(shifted)))) * dx
    integral = np.sum(h) * dx
    return sm + penalty * (integral - 1.0) ** 2

def repair_h(h):
    n = len(h)
    target = n / 2.0
    for _ in range(10):
        integ = np.sum(h)
        if integ < 1e-14:
            return np.full(n, 0.5)
        h = h * (target / integ)
        h = np.clip(h, 0.0, 1.0)
    current = np.sum(h)
    diff = target - current
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) / n * 2.0
    x_new = (np.arange(m) + 0.5) / m * 2.0
    return np.interp(x_new, x_old, h)

def optimize_block(h_in, t0, time_budget, schedule):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    cur = h_in.copy()
    best_h = cur.copy()
    best_c5 = compute_c5(cur)
    for temp, pen, maxit in schedule:
        if time.time() - t0 > t0 + time_budget:
            break
        try:
            res = minimize(obj_fn, cur, args=(temp, pen),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
            nh = np.clip(res.x, 0.0, 1.0)
            nh = repair_h(nh)
            c5v = compute_c5(nh)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = nh.copy()
            cur = nh
        except Exception:
            pass
    return best_h, best_c5

def perturb_h(h, mode, rng):
    n = len(h)
    if mode == 0:
        noise = rng.randn(n) * 0.08
    elif mode == 1:
        noise = rng.randn(n) * 0.05
        noise[np.where(h > 0.7)[0]] *= 2.0
        noise[np.where(h < 0.3)[0]] *= 2.0
    elif mode == 2:
        shift = rng.randint(2, max(3, n // 80))
        noise = 0.25 * (np.roll(h, shift) - h)
    else:
        block = max(3, n // 50)
        noise = np.zeros(n)
        for start in range(0, n, block):
            val = (rng.rand() - 0.5) * 0.3
            end = min(start + block, n)
            noise[start:end] = val
        noise = val * 0
        for start in range(0, n, block):
            val = (rng.rand() - 0.5) * 0.25
            end = min(start + block, n)
            noise[start:end] = val
    return repair_h(np.clip(h + noise, 0.0, 1.0))

    h_init = np.array(initial_h_values, dtype=float).copy()
