        mx = np.max(vals)
        # subgradient: any a[k] where k is argmax
        # For bisection, use the average of extreme subgradients
        mask = np.isclose(vals, mx, atol=1e-12)

            s_lo = np.min(a[mask])
            s_hi = np.max(a[mask])
        else:
            idx = np.argmax(vals)
            s_lo = s_hi = a[idx]
        return mx, s_lo, s_hi

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_sm(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceq_jac(x):
    return np.ones((1, len(x)))

def opt_1d_max(a, b):
    """Minimize max_k(a[k]*x + b[k]) for x in [0,1] via bisection on subgradient."""
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        mx = np.max(vals)
        mask = vals >= mx - 1e-10
        if np.any(mask):
            s_lo = np.min(a[mask])
            s_hi = np.max(a[mask])
        else:
            idx = np.argmax(vals)
            s_lo = s_hi = a[idx]
        if s_lo > 1e-12:
            hi = mid
        elif s_hi < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

            c1 = np.correlate(h1, 1.0 - h1, mode='full')
            a = c1 - c0
            b = c0
            h[i] = opt_1d_max(a, b)
        h = project(h, n)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_sm(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceq_jac(x):
    return np.ones((1, len(x)))

def opt_1d(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        mx = np.max(vals)
        mask = vals >= mx - 1e-10
        if np.any(mask):
            s_lo = np.min(a[mask])
            s_hi = np.max(a[mask])
        else:
            idx = np.argmax(vals)
            s_lo = s_hi = a[idx]
        if s_lo > 1e-12:
            hi = mid
        elif s_hi < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

            c1 = np.correlate(h1, 1.0 - h1, mode='full')
            h[i] = opt_1d(c1 - c0, c0)
