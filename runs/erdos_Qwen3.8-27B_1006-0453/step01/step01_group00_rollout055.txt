def softmax_max(corr, T):
    """Smooth max of corr using softmax"""
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max
    shifted = corr - c_max
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50*T, 0)
    s = np.sum(np.exp(shifted / T))
    return c_max + T * np.log(s)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_weights(corr, T):
    c_max = np.max(corr)
    shifted = corr - c_max
    shifted = np.clip(shifted, -50*T, 0)
    w = np.exp(shifted / T)
    s = np.sum(w)
    return w / s

def softmax_max_val(corr, T):
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max
    shifted = corr - c_max
    shifted = np.clip(shifted, -50*T, 0)
    return c_max + T * np.log(np.sum(np.exp(shifted / T)))

def project_constraint(h, n):

        return h

        h = np.clip(h, 0.0, 1.0)
    return h

        if obj_try < obj_val:
            h_new = h_try
            break
        alpha *= 0.5
