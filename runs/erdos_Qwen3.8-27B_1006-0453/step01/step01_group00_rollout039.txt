            h_idx_end = i2_end + lag

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-15:
        return cmax * dx
    shifted = corr - cmax
    s = np.sum(np.exp(shifted / T))
    return (cmax + T * np.log(s)) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exps = np.exp(shifted / T)
    s = np.sum(exps)
    w = exps / s  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        wm = w[m]
        if wm < 1e-14:
            continue
        lag = m - (n-1)
        # Valid j: j_min to j_max
        if lag >= 0:
            j_min = 0
            j_max = n-1-lag
        else:
            j_min = -lag
            j_max = n-1
        
        # Term 1: +1 for i in [j_min, j_max]
        grad[j_min:j_max+1] += wm
        
        # Term 2: -h[i+lag] for i in [j_min, j_max] where i+lag in [0, n-1]
        # i+lag in [0,n-1] means -lag <= i <= n-1-lag
        i2_lo = max(j_min, -lag, 0)
        i2_hi = min(j_max, n-1-lag, n-1)
        if i2_hi >= i2_lo:
            grad[i2_lo:i2_hi+1] -= wm * h[i2_lo+lag : i2_hi+lag+1]
        
        # Term 3: -h[i-lag] for i in [j_min, j_max+lag] where i-lag in [j_min, j_max] and 0<=i<n
        # i-lag in [j_min, j_max] means i in [j_min+lag, j_max+lag]
        # and 0 <= i < n
        i3_lo = max(j_min + lag, 0)
        i3_hi = min(j_max + lag, n-1)
        if i3_hi >= i3_lo:
            grad[i3_lo:i3_hi+1] -= wm * h[i3_lo-lag : i3_hi-lag+1]
    
    grad *= dx
    return grad
