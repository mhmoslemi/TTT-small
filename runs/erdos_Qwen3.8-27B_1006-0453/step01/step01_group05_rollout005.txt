import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] = np.clip(h[free] + diff / np.sum(free), 0.0, 1.0)
        else:
            h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) * temp
    val = m + np.log(np.sum(np.exp(shifted))) / temp
    pen = 100.0 * (np.sum(x) - n / 2.0) ** 2
    return val + pen


def nonsmooth_obj(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - h_complement(x), mode='full') * dx
    pen = 20.0 * (np.sum(x) - n / 2.0) ** 2
    return np.max(corr) + pen


def h_complement(x):
    return 1.0 - x


def run_slsqp(x0, n, temp, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x


def run_nm(x0, n, maxiter):
    res = minimize(nonsmooth_obj, x0, method='Nelder-Mead', args=(n,),
                   options={'maxiter': maxiter, 'adaptive': True,
                            'fatol': 1e-11, 'xatol': 1e-11})
    return res.x


def run_powell(x0, n, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(nonsmooth_obj, x0, method='Powell', bounds=bounds,
                   args=(n,), options={'maxiter': maxiter,
                                       'fatol': 1e-11, 'xatol': 1e-11})
    return res.x


def local_search_bb(h, n, rng, max_iter=20000, window=300):
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    c5 = compute_c5(h, n)
    stall = 0
    for it in range(max_iter):
        improved = False
        ntries = min(window, max(50, len(ones) * len(zeros) // 10))
        for _ in range(ntries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            nc = compute_c5(h, n)
            if nc < c5 - 1e-13:
                c5 = nc
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            stall += 1
        else:
            stall = 0
        if stall > 20:
            break
    return h, c5


def make_start(n, seed, mode):
    rng = np.random.RandomState(seed)
    if mode == 0:
        x = np.full(n, 0.5)
    elif mode == 1:
        x = rng.uniform(0.2, 0.8, n)
    elif mode == 2:
        x = np.zeros(n)
        x[::2] = 1.0
    elif mode == 3:
        x = np.zeros(n)
        x[:n // 2] = 1.0
    elif mode == 4:
        x = 0.5 + 0.4 * np.sin(2 * np.pi * np.arange(n) * 3.0 / n)
    elif mode == 5:
        x = 0.5 + 0.35 * np.cos(2 * np.pi * np.arange(n) * 2.0 / n) + 0.15 * np.sin(2 * np.pi * np.arange(n) * 7.0 / n)
    elif mode == 6:
        k = n // 2
        first = np.arange(k)
        x = np.zeros(n)
        x[first] = 1.0
        rng.shuffle(first)
        x[first] = 1.0
    else:
        x = rng.uniform(0.0, 1.0, n)
    return repair_h(x, n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(45, 0.5)

    best_h = h_init.copy()
    best_n = len(h_init)
    best_c5 = compute_c5(best_h, best_n)

    n_candidates = [28, 30, 32, 34, 36, 38, 40, 42, 44, 45, 46, 48,
                    50, 52, 54, 56, 60, 64, 70, 72]

    for n in n_candidates:
        if time.time() - t_start > budget_s - 80:
            break
        if n == best_n and (time.time() - t_start) > 60:
            continue

        starts = []
        # Resample from initial
        x_r = np.interp(np.linspace(0, n - 1, n),
                        np.linspace(0, len(h_init) - 1, len(h_init)), h_init)
        starts.append(repair_h(x_r, n))
        # Structured starts
        for m in range(7):
            starts.append(make_start(n, seed + n * 10 + m, m))
        # From best if same n
        if n == best_n:
            starts.append(best_h.copy())

        for si, x0_s in enumerate(starts):
            if time.time() - t_start > budget_s - 40:
                break
            cur = x0_s.copy()
            cur_c5 = compute_c5(cur, n)

            # Round 1: Nelder-Mead
            try:
                x1 = run_nm(cur, n, maxiter=2500)
                x1 = repair_h(x1, n)
                c5_1 = compute_c5(x1, n)
                if c5_1 < best_c5:
                    best_c5 = c5_1
                    best_h = x1.copy()
                    best_n = n
                cur = x1 if c5_1 < cur_c5 else cur
                cur_c5 = min(c5_1, cur_c5)
            except Exception:
                pass

            # Round 2: Powell
            if time.time() - t_start < budget_s - 25:
                try:
                    x2 = run_powell(cur, n, maxiter=1500)
                    x2 = repair_h(x2, n)
                    c5_2 = compute_c5(x2, n)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = x2.copy()
                        best_n = n
                    cur = x2 if c5_2 < cur_c5 else cur
                    cur_c5 = min(c5_2, cur_c5)
                except Exception:
                    pass

            # Round 3: SLSQP with log-sum-exp
            if time.time() - t_start < budget_s - 15:
                for temp in [30, 100, 300]:
                    if time.time() - t_start > budget_s - 10:
                        break
                    try:
                        x3 = run_slsqp(cur, n, temp, maxiter=400)
                        x3 = repair_h(x3, n)
                        c5_3 = compute_c5(x3, n)
                        if c5_3 < best_c5:
                            best_c5 = c5_3
                            best_h = x3.copy()
                            best_n = n
                        cur = x3 if c5_3 < cur_c5 else cur
                        cur_c5 = min(c5_3, cur_c5)
                    except Exception:
                        pass

            # Round 4: Bang-bang local search (even n only)
            if n % 2 == 0 and time.time() - t_start < budget_s - 10:
                h_bb = np.where(cur > 0.5, 1.0, 0.0)
                h_bb, c5_bb = local_search_bb(h_bb, n, rng, max_iter=5000, window=200)
                if c5_bb < best_c5:
                    best_c5 = c5_bb
                    best_h = h_bb.copy()
                    best_n = n
                # Also try from random bang-bang
                h_bb2 = make_start(n, seed * 7 + n, 6)
                h_bb2, c5_bb2 = local_search_bb(h_bb2, n, rng, max_iter=5000, window=200)
                if c5_bb2 < best_c5:
                    best_c5 = c5_bb2
                    best_h = h_bb2.copy()
                    best_n = n

    # Final polish on best solution
    if time.time() - t_start < budget_s - 10:
        for temp in [50, 150, 400]:
            if time.time() - t_start > budget_s - 5:
                break
            try:
                xp = run_slsqp(best_h, best_n, temp, maxiter=600)
                xp = repair_h(xp, best_n)
                c5p = compute_c5(xp, best_n)
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = xp.copy()
            except Exception:
                pass

    # Final NM polish
    if time.time() - t_start < budget_s - 5:
        try:
            xn = run_nm(best_h, best_n, maxiter=3000)
            xn = repair_h(xn, best_n)
            c5n = compute_c5(xn, best_n)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = xn.copy()
        except Exception:
            pass

    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
