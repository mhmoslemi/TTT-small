import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    omh = 1.0 - h
    for p in range(2 * n - 1):
        if w[p] < 1e-12:
            continue
        shift = p - (n - 1)
        wp = w[p] * dx
        if shift == 0:
            grad += wp * (1.0 - 2.0 * h)
        elif shift > 0:
            grad[:n - shift] += wp * omh[shift:]
            grad[shift:] += wp * (-h[:n - shift])
        else:
            s = -shift
            grad[s:] += wp * omh[:n - s]
            grad[:n - s] += wp * (-h[s:])
    return grad

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, t=100.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * (2.0 / n)

def obj_smooth_hi(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    t = 350.0
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * (2.0 / n)

def obj_bottleneck(h, m=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-m:]
    return np.mean(corr[idx]) * (2.0 / n)

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def subgradient_phase(h, n, t_s, lr0, n_iters, momentum, t_end):
    v = np.zeros(n)
    lr = lr0
    for it in range(n_iters):
        objs = get_overlaps(h)
        mx = np.max(objs)
        s = t_s * (objs - mx)
        e = np.exp(s)
        w = e / e.sum()
        g = compute_grad(h, w)
        v = momentum * v - lr * g
        h = project_h(h + v, n)
        lr *= 0.998
        t_s = min(t_s * 1.001, t_end)
    return h

                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        return h
