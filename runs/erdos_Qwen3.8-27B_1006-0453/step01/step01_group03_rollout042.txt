import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def softmax_slsqp(h0, dx, target_sum, beta, maxiter=300):
    n = len(h0)
    K = 2 * n - 1
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        shift = c - c.max()
        return np.log(np.sum(np.exp(beta * shift))) / beta
    def grad_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        shift = c - c.max()
        w = np.exp(beta * shift)
        w /= w.sum()
        g = np.zeros(n)
        for j in range(K):
            s = j - (n - 1)
            if abs(w[j]) < 1e-15:
                continue
            lo = max(0, -s)
            hi = min(n, n - s)
            for i in range(lo, hi):
                ip = i + s
                g[i] += w[j] * ((1.0 - x[ip]) - x[i])
        return g * dx
    def eq(x):
        return np.sum(x) - target_sum
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq, 'jac': lambda x: np.ones(n)}]
    x0 = h0.copy()
    res = minimize(obj, x0, jac=grad_obj, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    if res.x is not None:
        return project_sr(res.x, target_sum)
    return h0.copy()

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)

        if c == 0:

def softmax_grad(h, dx, w):

    g = np.zeros(n)
    for j in range(2*n-1):
        s = (n-1) - j
        wi = w[j] * dx
        if abs(wi) < 1e-18:
            continue
        lo1 = max(0, -s)
        hi1 = min(n-1, n-1-s)
        if lo1 <= hi1:

        if lo2 <= hi2:

        cmax = c.max()
        s = c - cmax
        return float(np.log(np.sum(np.exp(beta*s))) / beta)
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        cmax = c.max()
        w = np.exp(beta * (c - cmax))
        w /= w.sum()
        return softmax_grad(x, dx, w)

        return np.sum(x) - target_sum
    def eq_jac(x):
        return np.ones(n)

        if res.x is not None:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_sr(h, target):
    h = np.array(h, dtype=float)
    for _ in range(300):
        np.clip(h, 0.0, 1.0, out=h)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def softmax_grad(h, dx, w):
    n = len(h)
    g = np.zeros(n)
    for j in range(2 * n - 1):
        s = (n - 1) - j
        wi = w[j] * dx
        if abs(wi) < 1e-18:
            continue
        lo1, hi1 = max(0, -s), min(n - 1, n - 1 - s)
        if lo1 <= hi1:
            g[lo1:hi1+1] += wi * (1.0 - h[lo1+s:hi1+s+1])
        lo2, hi2 = max(0, s), min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= wi * h[lo2-s:hi2-s+1]
    return g

def slsqp_soft(h0, dx, target, beta, maxiter=400):
    n = len(h0)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cm = c.max()
        return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cm = c.max()
        w = np.exp(beta * (c - cm))
        w /= w.sum()
        return softmax_grad(x, dx, w)
    def eq(x):
        return np.sum(x) - target
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_sr(res.x, target)
    except Exception:
        pass
    return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_sr(h, target):
    h = np.array(h, dtype=float)
    for _ in range(300):
        np.clip(h, 0.0, 1.0, out=h)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def softmax_grad(h, dx, w):
    n = len(h)
    g = np.zeros(n)
    for j in range(2 * n - 1):
        s = (n - 1) - j
        wi = w[j] * dx
        if abs(wi) < 1e-18:
            continue
        lo1, hi1 = max(0, -s), min(n - 1, n - 1 - s)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += wi * (1.0 - h[lo1 + s:hi1 + s + 1])
        lo2, hi2 = max(0, s), min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= wi * h[lo2 - s:hi2 - s + 1]
    return g

def slsqp_soft(h0, dx, target, beta, maxiter=400):
    n = len(h0)

    def sv_eq(x):
        return np.sum(x) - target

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cm = c.max()
        return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cm = c.max()
        w = np.exp(beta * (c - cm))
        w /= w.sum()
        return softmax_grad(x, dx, w)

    cons = [{'type': 'eq', 'fun': sv_eq}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_sr(res.x, target)
    except Exception:
        pass
    return h0.copy()

def slsqp_minmax(h0, dx, target, maxiter=500):
    n = len(h0)
    c0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
    x0 = np.concatenate([h0, [c0.max() + 0.01]])

    def mm_obj(x):
        return x[n]

    def mm_ineq(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full') * dx
        return x[n] - c

    def mm_eq(x):
        return np.sum(x[:n]) - target

    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': mm_ineq}, {'type': 'eq', 'fun': mm_eq}]
    try:

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_sr(res.x[:n], target)
    except Exception:
        pass
    return h0.copy()
