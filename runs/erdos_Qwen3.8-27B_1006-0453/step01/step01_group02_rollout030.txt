def compute_c5(h, dx):

    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    # Bisection for the shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_max_corr(h, N, tau):
    """Compute gradient of soft-max approximation of max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w_sum = w.sum()
    w /= w_sum
    
    # Compute weighted gradient
    grad = np.zeros(N)
    center = N - 1
    # Threshold to skip tiny weights
    thresh = 0.001
    for idx in range(len(w)):
        if w[idx] < thresh:
            continue
        L = idx - center  # lag
        wi = w[idx]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:

        else:

            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, float(np.max(corr))

    best_c5 = compute_c5(h, dx)
    lr = 0.02

            best_h = h.copy()

        v_hat = v_buf / (1 - beta2 ** (it + 1))

        elif it == 2 * n_iter // 3:

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(h, N, tau):
    """Compute (weighted_gradient, true_c5) using soft-max weights."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = 0.001
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def adam_opt(h, N, target, dx, deadline, tau0, tau1, lr0, decay):
    """Adam with exponential tau ramp and geometric lr decay."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / 2000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad, c5_val = smax_grad(h, N, tau)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 50 == 0:
            lr *= decay
        if lr < 1e-9:
            break
    return best_h, best_c5
