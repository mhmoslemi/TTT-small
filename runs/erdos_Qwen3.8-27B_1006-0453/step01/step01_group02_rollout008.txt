import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, tau):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau)

def c5_grad(h, N, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = tau * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(N)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-13:
            continue
        lag = idx - (N - 1)
        if lag >= 0:
            grad[lag:] += wi * (1.0 - h[:N - lag])
            grad[:N - lag] -= wi * h[lag:]
        else:
            l = -lag
            grad[:N - l] += wi * (1.0 - h[l:])
            grad[l:] -= wi * h[:N - l]
    return grad * dx

def obj_func(h, N, tau, dx):
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    return smooth_max_val(corr, tau) * dx

def grad_func(h, N, tau, dx):
    hc = np.clip(h, 0.0, 1.0)
    return c5_grad(hc, N, tau, dx)

def lrob_phase(h, N, dx, target, bounds, tau_list, max_iter, t0, budget):
    best_local = compute_c5(h, dx)
    h_cur = h.copy()
    for tau in tau_list:
        if budget - (time.time() - t0) < 25:
            break

                       jac=grad_func, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_try = project_sum(res.x, target)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_local:
            best_local = c5_try
            h_cur = h_try.copy()
    return h_cur, best_local
