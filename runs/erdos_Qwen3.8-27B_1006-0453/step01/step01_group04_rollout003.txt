import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def obj_func(h, temp, pen, tlags, tw):
    cv = get_corr(h)
    cmax = np.max(cv)
    sh = cv - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(sh / temp)))
    integ = np.sum(h) * (2.0 / len(h))
    out = sm + pen * (integ - 1.0) ** 2
    if tw > 0 and tlags is not None and len(tlags) > 0:
        idx = np.array(tlags)
        out += tw * np.sum(cv[idx] ** 2)
    return out

def repair_h(h):
    n = len(h)
    target = n / 2.0
    h2 = np.clip(h, 0.0, 1.0)
    s = np.sum(h2)
    if s < 1e-12:
        return np.full(n, 0.5)
    h2 = h2 * (target / s)
    h2 = np.clip(h2, 0.0, 1.0)
    s2 = np.sum(h2)
    diff = target - s2
    if abs(diff) > 1e-8:
        mask = (h2 > 1e-4) & (h2 < 1.0 - 1e-4)
        nm = np.sum(mask)
        if nm > 0:
            h2[mask] += diff / nm
            h2 = np.clip(h2, 0.0, 1.0)
    return h2

def interp_up(h, f):
    n = len(h)
    m = n * f
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(xn, xo, h)

def lfsb(h, temp, pen, mit, bd, tl=None, tw=0.0):
    r = minimize(obj_func, h, args=(temp, pen, tl, tw),
                 method='L-BFGS-B', bounds=bd,
                 options={'maxiter': mit, 'ftol': 1e-14, 'gtol': 1e-10})
    return np.clip(r.x, 0.0, 1.0)

def get_top_lags(cv, num=8, width=2):
    top = np.argsort(cv)[-num:]
    tset = set()
    for i in top:
        for d in range(-width, width + 1):
            j = int(i) + d
            if 0 <= j < len(cv):
                tset.add(j)
    return sorted(tset)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Work at 720 points
    cur = interp_up(h0, 2)
    cur = repair_h(cur)
    n = len(cur)
    bd = [(0.0, 1.0)] * n

    # Phase 1: Coarse optimization (30% budget)
    p1_sched = [(0.05, 500, 150), (0.01, 2000, 150), (0.003, 8000, 200)]
    for temp, pen, mit in p1_sched:
        if time.time() - t0 > budget_s * 0.30:
            break
        cur = lfsb(cur, temp, pen, mit, bd)
        cur = repair_h(cur)
        c5v = compute_c5(cur)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = cur.copy()

    # Phase 2: Targeted max-chasing (key improvement, 45% budget)
    n_target_iters = 8
    for it in range(n_target_iters):
        if time.time() - t0 > budget_s * 0.75:
            break
        cv = get_corr(cur)
        tl = get_top_lags(cv, num=8, width=2)
        tw = 0.3 + it * 0.25

        sub_sched = [(0.008, 5000, 120), (0.002, 12000, 150)]
        for temp, pen, mit in sub_sched:
            if time.time() - t0 > budget_s * 0.75:
                break
            cur = lfsb(cur, temp, pen, mit, bd, tl, tw)
            cur = repair_h(cur)
            c5v = compute_c5(cur)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cur.copy()

    # Phase 3: Final polish with very low temperature (10% budget)
    if time.time() - t0 < budget_s * 0.88:
        for temp in [0.001, 0.0003]:
            if time.time() - t0 > budget_s * 0.90:
                break
            cv = get_corr(cur)
            tl = get_top_lags(cv, num=5, width=1)
            cur = lfsb(cur, temp, 20000, 200, bd, tl, 1.5)
            cur = repair_h(cur)
            c5v = compute_c5(cur)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cur.copy()

    # Phase 4: Random restarts (5% budget)
    if time.time() - t0 < budget_s * 0.87:
        for ri in range(3):
            if time.time() - t0 > budget_s * 0.90:
                break
            sig = 0.1 / (ri + 1)
            hp = np.clip(cur + np.random.randn(n) * sig, 0.0, 1.0)
            hp = repair_h(hp)
            cv = get_corr(hp)
            tl = get_top_lags(cv, num=6, width=1)
            hp = lfsb(hp, 0.004, 5000, 80, bd, tl, 0.5)
            hp = repair_h(hp)
            hp = lfsb(hp, 0.001, 15000, 100, bd, tl, 1.0)
            hp = repair_h(hp)
            c5v = compute_c5(hp)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hp.copy()

    # Final repair and evaluation
    best_h = repair_h(best_h)
    fc = compute_c5(best_h)
    if fc < best_c5:
        best_c5 = fc

    return best_h, best_c5, len(best_h)
