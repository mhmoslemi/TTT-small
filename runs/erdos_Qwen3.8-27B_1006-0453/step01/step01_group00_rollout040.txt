    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:

    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n * 0.5
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute to interior points first
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = int(np.sum(interior))
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
        # If still off, adjust all
        diff2 = target_sum - np.sum(h)
        if abs(diff2) > 1e-14:
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):

    return fix_sum(h_new)

def pgd_step(h, T, lr):

        idx = np.where(corr >= c_max - 1e-10)[0]

        weights[idx] = 1e0 / len(idx)
    else:
        shifted = np.clip((corr - c_max) / T, -80, 0)
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= ws
    
    forward = np.zeros(n)
    backward = np.zeros(n)

            forward[:n-l] += w_l * h[l:]
        else:
            forward[-l:] += w_l * h[:n+l]
