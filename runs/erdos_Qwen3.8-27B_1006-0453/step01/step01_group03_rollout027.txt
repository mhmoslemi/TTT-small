from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_c5(h, dx, beta=80.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-15:
            continue
        k = ki - (n - 1)
        if k >= 0:
            if n - k > 0:

        else:
            kk = -k
            if n - kk > 0:
                grad[:n-kk] += wi * one_minus_h[kk:]
                grad[kk:] -= wi * h[:n-kk]

def slsqp_refine(h0, dx, target_sum, maxiter=500):
    n = len(h0)
    def obj(x):
        return x[n]
    def con_fun(x):
        # Returns array of correlation values - x[n] (all should be >= 0)
        corr = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return corr - x[n]
    def con_jac(x):
        # Jacobian: for each correlation k, and each h[j]
        # This is (2n-1) x (n+1)
        J = np.zeros((2*n-1, n+1))
        for ki in range(2*n-1):
            k = ki - (n-1)
            J[ki, n] = -1.0
            if k >= 0:
                for j in range(k, n):
                    J[ki, j] += (1.0 - x[:n][j-k])
                for j in range(0, n-k):
                    J[ki, j] -= x[:n][j+k]
            else:
                kk = -k
                for j in range(0, n-kk):
                    J[ki, j] += (1.0 - x[:n][j+kk])
                for j in range(kk, n):
                    J[ki, j] -= x[:n][j-kk]
        return J

        {'type': 'eq', 'fun': eq_fun}

        elif res.x is not None:
            return project_h(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

    x0 = np.concatenate([h0, [c5_init + 0.005]])
    
    def obj(x):
        return x[n]
    
    def ineq(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return c[idx_sorted] - x[n]
    
    def ineq_jac(x):
        Jc = corr_jac(x[:n])
        Js = Jc[idx_sorted, :]
        Jf = np.zeros((len(idx_sorted), n + 1))
        Jf[:, :n] = Js
        Jf[:, n] = -1.0
        return Jf
    
    def eqf(x):
        return x[:n].sum() - target_sum
    
    def eqf_jac(x):
        J = np.zeros(1 + n)
        J[:n] = 1.0
        return J
    
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eqf, 'jac': eqf_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_h(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def local_search(h0, dx, target_sum, n_trials=200, deadline=None):
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(best, dx)
    for trial in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        h = best.copy()
        i, j = np.random.choice(n, 2, replace=False)
        delta = 0.05 * (np.random.rand() - 0.5)
        h[i] += delta
        h[j] -= delta
        h = project_h(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best = h.copy()
            best_c5 = c5
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5
    
    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_h(h_raw, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    
    def time_left(margin=5):
        return (time.time() - t_start) < (budget_s - margin)
    
    # Phase 1: SLSQP on parent
    if time_left(60):
        h1 = slsqp_refine(h0, dx, target_sum, 500)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    
    # Phase 2: Gradient from best
    if time_left(100):
        dl = t_start + budget_s - 100
        h2, c2 = gradient_optimize(best_h, dx, target_sum, 4000, dl)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
    
    # Phase 3: SLSQP + local search
    if time_left(60):
        h3 = slsqp_refine(best_h, dx, target_sum, 300)
        c3 = compute_c5(h3, dx)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
        if time_left(30):
            dl = t_start + budget_s - 30
            h3b, c3b = local_search(best_h, dx, target_sum, 300, dl)
            if c3b < best_c5:
                best_h, best_c5 = h3b, c3b
    
    # Phase 4: Diverse starts
    starts = []
    starts.append(best_h + 0.04 * np.random.randn(n))
    tme = np.zeros(n)
    tme[0] = 1.0
    for i in range(1, n):
        tme[i] = 1.0 - tme[i // 2]
    starts.append(project_h(tme, target_sum))
    hr = np.zeros(n)
    perm = np.random.permutation(n)
    half = int(target_sum)
    hr[perm[:half]] = 1.0
    fr = target_sum - half
    if fr > 0.01:
        hr[perm[half]] = fr
    starts.append(project_h(hr, target_sum))
    xa = np.linspace(0, 2, n)
    hs = 0.5 + 0.35 * np.sin(3.0 * np.pi * xa)
    starts.append(project_h(hs, target_sum))
    starts.append(project_h(np.random.rand(n), target_sum))
    
    for h_s in starts:
        if not time_left(40):
            break
        h_s = slsqp_refine(h_s, dx, target_sum, 150)
        c_s = compute_c5(h_s, dx)
        if time_left(30):
            dl = t_start + budget_s - 30
            h_s2, c_s2 = gradient_optimize(h_s, dx, target_sum, 2000, dl)
            if c_s2 < c_s:
                h_s, c_s = h_s2, c_s2
        if c_s < best_c5:
            best_h, best_c5 = h_s, c_s
    
    # Phase 5: Random restarts around best
    for r in range(30):
        if not time_left(12):
            break
        h_r = best_h + 0.02 * np.random.randn(n)
        h_r = project_h(h_r, target_sum)
        h_r = slsqp_refine(h_r, dx, target_sum, 80)
        c_r = compute_c5(h_r, dx)
        if c_r < best_c5 and time_left(20):
            dl = t_start + budget_s - 20
            h_r2, c_r2 = gradient_optimize(h_r, dx, target_sum, 800, dl)
            if c_r2 < c_r:
                c_r, h_r = c_r2, h_r2
        if c_r < best_c5:
            best_h, best_c5 = h_r, c_r
    
    # Phase 6: Try nearby n values
    for tn in [n + 1, n - 1, n + 2, n + 3]:
        if not time_left(30) or tn < 30:
            break
        dxt = 2.0 / tn
        tst = tn / 2.0
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, tn)
        ht = np.interp(xn, xo, best_h)
        ht = project_h(ht, tst)
        ht = slsqp_refine(ht, dxt, tst, 200)
        ct = compute_c5(ht, dxt)
        if ct < best_c5 and time_left(25):
            dl = t_start + budget_s - 25
            ht2, ct2 = gradient_optimize(ht, dxt, tst, 1500, dl)
            if ct2 < ct:
                ct, ht = ct2, ht2
            if ct < best_c5:
                best_h, best_c5, n, dx, target_sum = ht, ct, tn, dxt, tst
    
    # Phase 7: Local search + final polish
    if time_left(30):
        dl = t_start + budget_s - 15
        best_h, best_c5 = local_search(best_h, dx, target_sum, 400, dl)
    if time_left(10):
        h_f = slsqp_refine(best_h, dx, target_sum, 100)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f
    
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, len(best_h)
