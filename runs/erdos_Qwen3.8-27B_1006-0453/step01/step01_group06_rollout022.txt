                else:
                    lo = c1p
                    c1p = d1p
                    fc = fd
                    d1p = lo + (hi - lo) * inv_phi
                    fd = np.max(a * d1p + b)
            h[j] = (lo + hi) * 0.5
        h = project_feasible(h, n)
    return h

def sa_optimize(h, n, temperature, n_iter, rng):
    h = h.copy()
    cur = compute_c5(h)
    best = h.copy()
    best_val = cur
    for i in range(n_iter):
        T = temperature * np.exp(-4.0 * i / max(n_iter, 1))
        use_wide = rng.random() < 0.25
        mag = 0.01 + 0.15 * np.exp(-3.0 * i / max(n_iter, 1))
        if use_wide:
            mag += 0.1
        j = int(rng.integers(0, n))
        m = int(rng.integers(0, n))
        attempts = 0
        while j == m and attempts < 10:
            m = int(rng.integers(0, n))
            attempts += 1
        if j == m:
            continue
        delta = float(rng.uniform(-mag, mag))
        old_j, old_m = h[j], h[m]
        new_j = old_j + delta
        new_m = old_m - delta
        if new_j < 0.0 or new_j > 1.0 or new_m < 0.0 or new_m > 1.0:
            continue
        h[j] = new_j
        h[m] = new_m
        new_val = compute_c5(h)
        accept = new_val < cur or (rng.random() < np.exp(-(new_val - cur) / max(T, 1e-12)))
        if accept:
            cur = new_val
            if new_val < best_val:
                best_val = new_val
                best = h.copy()
        else:
            h[j] = old_j
            h[m] = old_m
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t_start

    # Phase 1: Coordinate descent to reach local optimum
    p1_end = t_start + budget_s * 0.12
    while time.time() < p1_end:
        h_cd = coord_swift(best_h, n, 25)
        c5v = compute_c5(h_cd)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best_h = h_cd

    # Phase 2: Simulated annealing with cooling schedule
    p2_end = t_start + budget_s * 0.65
    # Estimate iterations based on available time (~15 microsecond per iter)
    sa_time = max(p2_end - time.time(), 1.0)
    temps = [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    n_per_temp = max(int(sa_time * 120000 / len(temps)), 50000)
    for T in temps:
        if time.time() > p2_end:
            break
        h_sa, c5_sa = sa_optimize(best_h, n, T, n_per_temp, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # Quick CD between SA phases
        if time.time() < p2_end - 2.0:
            h_cd = coord_swift(best_h, n, 15)
            c5v = compute_c5(h_cd)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_cd

    # Phase 3: More CD for fine-tuning
    p3_end = t_start + budget_s * 0.85
    while time.time() < p3_end:
        h_cd = coord_swift(best_h, n, 40)
        c5v = compute_c5(h_cd)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h_cd
        else:
            break

    # Phase 4: Fine SA with very low temperature
    if time.time() < t_start + budget_s * 0.93:
        fine_time = min(t_start + budget_s * 0.93 - time.time(), 30.0)
        n_fine = max(int(fine_time * 150000), 100000)
        h_sa, c5_sa = sa_optimize(best_h, n, 0.0001, n_fine, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        h_cd = coord_swift(best_h, n, 30)
        c5v = compute_c5(h_cd)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_cd

    # Phase 5: Try different grid sizes
    p5_end = t_start + budget_s * 0.98
    for n_try in [50, 70, 80, 100, 120, 150]:
        if time.time() > p5_end:
            break
        x_old = np.linspace(0.0, 2.0, n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try)
        h_new = coord_swift(h_new, n_try, 25)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new
            n = n_try

    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
