import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, k, N):
    g = np.zeros(N)
    if 0 <= k < N:
        g[:N-k] += 1.0 - h[k:]
        g[k:] -= h[:N-k]
    elif -N < k < 0:
        l = -k
        g[l:] += 1.0 - h[:N-l]
        g[:N-l] -= h[l:]
    return g

def smooth_grad(h, corr, tau, N):
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    grad = np.zeros(N)
    thresh = 0.005 / N
    for idx in np.where(w > thresh)[0]:
        lag = int(idx) - (N - 1)
        grad += w[idx] * grad_lag(h, lag, N)
    return grad

def adam_opt(h, N, dx, target, t0, budget, rng, lr=0.02, tau=5.0, max_iter=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 12:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(corr.max() * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        grad = smooth_grad(h, corr, tau, N)
        
        # Cosine LR
        lr_it = lr * 0.5 * (1 + np.cos(np.pi * it / max_iter))
        lr_it = max(lr_it, 1e-7)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** (it + 1))
        vh = v / (1 - b2 ** (it + 1))
        
        h = h - lr_it * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        # Sharpen max
        if (it + 1) % 300 == 0 and tau < 100:
            tau *= 1.6
    
    return best_h, best_c5

def local_search(h, N, dx, target, t0, budget, rng, n_rounds=50):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    
    for rnd in range(n_rounds):
        if time.time() - t0 > budget - 10:
            break
        improved = False
        order = rng.permutation(N)

                        best_c5 = c52
                        best_h = h.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5

    g = np.zeros(N)
    if 0 <= k < N:
        g[:N-k] += 1.0 - h[k:]
        g[k:] -= h[:N-k]
    elif -N < k < 0:

        g[l:] += 1.0 - h[:N-l]
