import numpy as np
from scipy.optimize import minimize, linprog
import time

_GN = 280

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * 2.0 / n

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_constraint(h, n):
    h = h.copy()
    h += (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def upsample(parent, new_n):
    ox = np.arange(len(parent), dtype=np.float64) / len(parent)
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

def soft_max_val(corr, beta):
    m = np.max(corr)
    return m + beta * np.log(np.sum(np.exp((corr - m) / beta)))

def obj_func(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode="full")
    return (2.0 / n) * soft_max_val(c, beta)

def con_fun(x):
    return np.sum(x) - _GN / 2.0

def con_jac(x):
    return np.ones(_GN)

def compute_grad_matrix(h):
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k_idx in range(n_lags):
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            G[k_idx, lag:] += (1.0 - h[:m])
            G[k_idx, :m] -= h[lag:]
        else:
            l = -lag
            m = n - l
            G[k_idx, :m] += (1.0 - h[l:])
            G[k_idx, l:] -= h[:m]
    return G
