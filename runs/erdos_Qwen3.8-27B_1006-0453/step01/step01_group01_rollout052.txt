        starts.append(trial)

    # Phase 1: FISTA with temperature annealing (65% of budget)
    fista_deadline = t0 + 0.62 * budget_s
    temps_full = [200.0, 100.0, 50.0, 25.0, 12.0, 6.0]
    temps_short = [50.0, 25.0, 12.0, 6.0]

    n_starts = len(starts)
    for si, h_start in enumerate(starts):
        if time.time() > fista_deadline:
            break
        remaining = fista_deadline - time.time()
        time_per = remaining / max(1, n_starts - si)
        est_iter = max(200, int(time_per * 800))
        if si == 0:
            h_res = fista_step(h_start, temps_full, est_iter, lr0=3.0,
                               time_limit=time.time() + time_per)
        else:
            h_res = fista_step(h_start, temps_short, est_iter, lr0=3.0,
                               time_limit=time.time() + time_per)
        c5 = compute_c5(h_res)
        if c5 < bc:
            bc = c5
            bh = h_res.copy()

    # Phase 2: SLSQP polish (30% of budget)
    slsqp_deadline = t_end - 4.0
    if time.time() < slsqp_deadline and bh is not None:
        n = len(bh)
        bnds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': _ceq, 'jac': _ceqj}

        for t in [30.0, 15.0, 8.0, 5.0, 3.0]:
            if time.time() > slsqp_deadline - 2:
                break
            try:
                r = minimize(_obj_arg, bh, args=(t,), method='SLSQP',
                             bounds=bnds, constraints=cons,
                             options={'maxiter': 1200, 'ftol': 1e-13})
                hc = proj(np.clip(r.x, 0.0, 1.0), n)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc
            except Exception:
                pass

    # Phase 3: Final FISTA refinement at very low temperature
    if time.time() < t_end - 3 and bh is not None:
        h_res = fista_step(bh, [8.0, 4.0, 2.0], 3000, lr0=2.0,
                           time_limit=t_end - 3.0)
        c5 = compute_c5(h_res)
        if c5 < bc:
            bc = c5
            bh = h_res.copy()

    return bh, compute_c5(bh), len(bh)
