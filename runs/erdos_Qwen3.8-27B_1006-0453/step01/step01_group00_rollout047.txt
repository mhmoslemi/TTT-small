import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-10:
        return cmax * dx
    s = np.sum(np.exp(np.clip(T * (corr - cmax), -50, 50)))
    return (cmax + np.log(s) / T) * dx

def penalty(h, n, lam):
    return lam * (np.sum(h) - 0.5 * n) ** 2

def fix_sum(h, n):
    target = 0.5 * n
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = interior.sum()
    if cnt > 0:
        h[interior] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        h += diff2 / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(xn, xo, h, period=2.0), new_n)

def lbfgsb_opt(h0, n, T, lam, maxiter):
    def f(x):
        xc = np.clip(x, 0, 1)
        return smooth_obj(xc, T) + penalty(xc, n, lam)
    r = opt.minimize(f, h0, method='L-BFGS-B',
                     bounds=[(0, 1)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return r.x

def slsqp_opt(h0, n, T, maxiter):
    def f(x):
        return smooth_obj(np.clip(x, 0, 1), T)
    def sc(x):
        return np.sum(x) - 0.5 * n
    r = opt.minimize(f, h0, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': sc}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
