import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        return c_max * dx
    shifted = np.clip((corr - c_max) / T, -500, 0)
    s = np.sum(np.exp(shifted))
    return (c_max + T * np.log(s)) * dx

def eq_constraint(h):
    return np.sum(h) - len(h) * 0.5

def project_to_feasible(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Adjust interior points
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.any(interior):
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
        # If still off, adjust all
        if abs(np.sum(h) - target) > 1e-10:
            h += (target - np.sum(h)) / n
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project(h):
    n = len(h)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        interior = (h > 1e-5) & (h < 1.0 - 1e-5)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_obj_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    g = 1.0 - 2.0 * h
    sum_viol = np.sum(h) - 0.5 * n
    if T < 1e-10:
        j_max = np.argmax(corr)
        grad = np.zeros(n)
        for m in range(n):
            k = j_max - m
            if 0 <= k < n:
                grad[m] = dx * (1.0 - 2.0 * h[k])
        val = c_max * dx
    else:
        shifted = np.clip((corr - c_max) / T, -400, 0)
        exps = np.exp(shifted)
        S = np.sum(exps)
        w = exps / S
        val = (c_max + T * np.log(S)) * dx
        grad = dx * np.convolve(w, g, mode='valid')
    # Add penalty gradient
    val += mu * sum_viol * sum_viol
    grad += 2.0 * mu * sum_viol
    return val, grad

def proj_feasible(h):
    n = len(h)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        interior = (h > 1e-5) & (h < 1.0 - 1e-5)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    shifted = np.clip((corr - c_max) / T, -400, 0)
    s = np.sum(np.exp(shifted))
    return (c_max + T * np.log(s)) * dx

    return np.sum(h) - 0.5 * len(h)
