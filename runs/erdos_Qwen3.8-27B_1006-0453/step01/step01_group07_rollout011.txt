import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_smooth_max(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = np.clip((corr - m) / beta, -50, 0)
    return dx * (m + beta * np.log(np.sum(np.exp(scaled))))

def grad_smooth_max(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = np.clip((corr - m) / beta, -50, 0)
    w = np.exp(scaled)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-11:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            i1 = np.arange(n - lag)
            grad[i1] += wk * (1.0 - h[i1 + lag])
            i2 = np.arange(lag, n)
            grad[i2] -= wk * h[i2 - lag]
        else:
            l = -lag
            i1 = np.arange(l, n)
            grad[i1] += wk * (1.0 - h[i1 - l])
            i2 = np.arange(n - l)
            grad[i2] -= wk * h[i2 + l]
    return grad

def project_sum(h, target):
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            return h
        h = h + d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(parent, nn):
    no = len(parent)
    if no == nn:
        return parent.copy()
    r = np.linspace(0, no - 1, nn)
    idx = np.minimum(r.astype(int), no - 2)
    fr = r - idx
    v = parent[idx] * (1 - fr) + parent[idx + 1] * fr
    return project_sum(v, nn / 2.0)

def slsqp_step(h, n, beta, maxiter=200):
    bounds = list(zip(np.zeros(n), np.ones(n)))
    eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
          'jac': lambda x: np.ones(n)}
    res = minimize(obj_smooth_max, h, args=(n, beta), method='SLSQP',
                   bounds=bounds, constraints=[eq],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_sum(np.clip(res.x, 0, 1), n / 2.0)

def grad_descent(h, n, betas, max_iter=400):
    target = n / 2.0
    for beta in betas:
        lr = 0.2
        for it in range(max_iter):
            if time.time() - getattr(grad_descent, '_t0', time.time()) > 1e9:
                break
            g = grad_smooth_max(h, beta)
            c_old = compute_c5(h)
            h_try = project_sum(h - lr * g, target)
            c_new = compute_c5(h_try)
            if c_new < c_old - 1e-15:
                h = h_try
                lr = min(lr * 1.05, 1.0)
            else:
                lr *= 0.6
                if lr < 1e-8:
                    break
    return h

def coord_descent(h, n, n_passes=60):
    target = n / 2.0
    cur = compute_c5(h)
    for p in range(n_passes):
        improved = False
        for i in range(n):
            # 1D optimization: vary h[i], adjust one other coordinate to fix sum
            # Find j with most room for adjustment
            others = np.arange(n)
            others = others[others != i]
            # Try both directions
            for direction in [1, -1]:
                if h[i] + direction * 0.5 > 1 or h[i] + direction * 0.5 < 0:
                    continue
                # Try moving h[i] by small amount, distribute to others
                for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
                    h[i] += direction * step
                    h[i] = min(max(h[i], 0), 1.0)
                    diff = target - np.sum(h)
                    if abs(diff) < 1e-14:
                        h = project_sum(h, target)
                    else:
                        h = project_sum(h, target)
                    c = compute_c5(h)
                    if c < cur - 1e-14:
                        cur = c
                        improved = True
                        break
                    else:
                        h[i] -= direction * step
                        h = project_sum(h, target)
        if not improved:
            break
    return h, cur
