import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = float(h.sum()) - target
    if abs(diff) < 1e-12:
        return h
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, m, N):
    """Gradient of correlation at lag m (m = idx - (N-1)) w.r.t. h."""
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += (1.0 - h[:N - m])
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def negate_lag_grad(h, m, N):
    """-grad of correlation at lag m (for minimization)."""
    return -grad_lag(h, m, N)

def optimize_adam_topk(h, N, target, dx, t0, budget_s, rng, lr_init=0.03, K=15):
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    
    for it in range(3000):
        if time.time() - t0 > budget_s - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local:
            best_local = c5_val
            best_h = h.copy()
        
        # Top-K lags
        top_k_idx = np.argsort(corr)[-K:]
        w = np.zeros(len(corr))
        w[top_k_idx] = 1.0 / K
        
        # Compute gradient
        grad = np.zeros(N)
        for idx in top_k_idx:
            lag = int(idx) - (N - 1)
            grad += w[idx] * grad_lag(h, lag, N)
        
        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)
        
        # Anneal
        if (it + 1) % 300 == 0:
            lr *= 0.88
            K = min(K + 5, int(N * 0.3))
        if lr < 1e-9:
            break
    return best_h, best_local

def optimize_nesterov(h, N, target, dx, t0, budget_s, rng, lr_init=0.02, K=10):
    """Nesterov accelerated projected gradient for top-K lags."""
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    x = h.copy()
    y = h.copy()
    lr = lr_init
    t_nest = 1.0
    
    for it in range(2000):
        if time.time() - t0 > budget_s - 10:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local:
            best_local = c5_val
            best_h = y.copy()
        
        top_k_idx = np.argsort(corr)[-K:]
        grad = np.zeros(N)
        for idx in top_k_idx:
            lag = int(idx) - (N - 1)
            grad += (1.0 / K) * grad_lag(y, lag, N)
        
        t_np1 = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0

        y_new = project_sum(y_new, target)
        
        alpha = (1.0 - t_nest) / t_np1
        x_new = y_new + alpha * (y_new - x)

        y = y_new
        t_nest = t_np1

def make_binary_init(N, rng, method='golden'):
    """Create a binary initialization with exactly N/2 ones."""
    h = np.zeros(N)
    if method == 'golden':
        alpha = (np.sqrt(5.0) - 1.0) / 2.0
        for i in range(N):
            h[i] = 1.0 if (i * alpha) % 1.0 < 0.5 else 0.0
    elif method == 'thue_morse':
        for i in range(N):
            h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    else:
        ones = rng.choice(N, size=N // 2, replace=False)
        h[ones] = 1.0
    
    # Adjust to get exactly N/2 ones
    n_ones = int(h.sum())
    target = N // 2
    if n_ones != target:

            zeros_to_flip = np.where(h == 0)[0]
            flip = rng.choice(zeros_to_flip, size=target - n_ones, replace=False)
            h[flip] = 1.0
    return h

        # grad[i] < 0 means increasing h[i] decreases corr, so we increase h[i]

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = float(h.sum()) - target
    if abs(diff) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += (1.0 - h[:N - m])
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize亚当(h, N, target, dx, t0, budget_s, rng, lr_init=0.025, K_init=10, max_iters=8000):
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    K = K_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)

    for it in range(max_iters):
        if time.time() - t0 > budget_s - 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v < best_local:
            best_local = c5v
            best_h = h.copy()
        top_k = np.argsort(corr)[-K:]
        grad = np.zeros(N)
        for idx in top_k:
            lag = int(idx) - (N - 1)
            grad += grad_lag(h, lag, N)
        grad /= K
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** (it + 1))
        vh = v_buf / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if (it + 1) % 400 == 0:
            lr *= 0.9
            K = min(K + 4, min(N, 60))
        if (it + 1) % 800 == 0:
            h = h + rng.randn(N) * 0.005
            h = project_sum(h, target)
        if lr < 1e-10:
            break
    return best_h, best_local

def optimize_nesterov(h, N, target, dx, t0, budget_s, lr_init=0.012, K=8, max_iters=5000):
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    x = h.copy()
    y = h.copy()
    lr = lr_init
    t_n = 1.0
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 8:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v < best_local:
            best_local = c5v
            best_h = y.copy()
        top_k = np.argsort(corr)[-K:]
        grad = np.zeros(N)
        for idx in top_k:
            lag = int(idx) - (N - 1)
            grad += grad_lag(y, lag, N)
        grad /= K
        t_np = (1.0 + np.sqrt(1.0 + 4.0 * t_n * t_n)) / 2.0
        y_new = project_sum(x - lr * grad, target)
        alpha = (1.0 - t_n) / t_np
        x = project_sum(y_new + alpha * (y_new - x), target)
        y = y_new
        t_n = t_np
        if (it + 1) % 300 == 0:
            lr *= 0.93
        if lr < 1e-11:
            break
    return best_h, best_local

def make_binary(N, rng, method=0):
    h = np.zeros(N)
    if N % 2 != 0:
        h = np.zeros(N + 1)
        N += 1
    target = N // 2
    if method == 0:
        alpha = (np.sqrt(5.0) - 1.0) / 2.0
        vals = (np.arange(N) * alpha) % 1.0
        h = (vals < 0.5).astype(float)
    elif method == 1:
        h = np.array([1.0 if bin(i).count('1') % 2 == 1 else 0.0 for i in range(N)])
    else:
        idx = rng.choice(N, size=target, replace=False)
        h[idx] = 1.0
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1.0)[0]
        flip = rng.choice(ones, size=n_ones - target, replace=False)
        h[flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        flip = rng.choice(zeros, size=target - n_ones, replace=False)
        h[flip] = 1.0
    return h
