import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nfree = np.sum(free)
        if nfree == 0:
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            per = diff / nfree
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def nm_obj(x, n):
    f = compute_c5(np.clip(x, 0, 1), n)
    pen = 500.0 * (np.sum(x) - n / 2.0) ** 2
    return f + pen

def nm_run(x0, n, maxiter=2000):
    res = minimize(nm_obj, x0, args=(n,), method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-13, 'adaptive': True})
    return res.x

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * temp
    m = np.max(scaled)
    scaled = scaled - m
    return float((np.log(np.sum(np.exp(scaled))) / temp + m / temp) * dx)

def slsqp_run(x0, n, temp, maxiter=400):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x

def bang_bang_refine(h, n, sweeps=10):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for target in [0.0, 1.0]:
                old = h[i]
                h[i] = target
                h = repair_h(h, n)
                c = compute_c5(h, n)
                if c < best - 1e-14:
                    best = c
                    improved = True
                else:
                    h[i] = old
                    h = repair_h(h, n)
        if not improved:
            break
    return h, best

def de_global(n, t_start, budget_s, margin):
    def de_obj(x):
        x = np.clip(x, 0, 1)
        f = compute_c5(x, n)
        pen = 200.0 * (np.sum(x) - n / 2.0) ** 2
        return f + pen
    res = differential_evolution(de_obj, [(0, 1)] * n, seed=7,
                                 maxiter=80, popsize=8, tol=1e-10,
                                 mutation=(0.3, 1.0), rebin=True,
                                 polish=False, init='sobol',
                                 updating='immediate')
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 45
        h0 = np.full(n, 0.5)
    n = len(h0)
    h0 = repair_h(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Nelder-Mead from parent with true max objective
    if time.time() - t_start < budget_s - 400:
        alloc = min(120, budget_s - 400 - (time.time() - t_start))
        maxit = max(500, int(alloc * 20))
        x = nm_run(best_h, n, maxit)
        x = repair_h(x, n)
        c = compute_c5(x, n)
        if c < best_c5:
            best_c5, best_h = c, x

    # Phase 2: Differential Evolution for global search
    if time.time() - t_start < budget_s - 300:
        x_de = de_global(n, t_start, budget_s, 300)
        x_de = repair_h(x_de, n)
        c = compute_c5(x_de, n)
        if c < best_c5:
            best_c5, best_h = c, x_de
        # NM polish on DE result
        x = nm_run(x_de, n, 1500)
        x = repair_h(x, n)
        c = compute_c5(x, n)
        if c < best_c5:
            best_c5, best_h = c, x

    # Phase 3: Multiple diverse NM restarts
    diverse_starts = [best_h.copy()]
    # Fourier-based starts
    for freq_set in [(1, 2), (1, 3), (2, 3), (1, 4), (3, 5)]:
        f1, f2 = freq_set

        x = repair_h(x, n)
        diverse_starts.append(x)
    # Bang-bang random

        idx = np.random.choice(n, n // 2, replace=False)

        x = repair_h(x, n)
        diverse_starts.append(x)

        maxit = max(800, min(2500, int(3000 / (i + 1))))

        x = repair_h(x, n)
        c = compute_c5(x, n)
        if c < best_c5:
            best_c5, best_h = c, x

    # Phase 4: SLSQP polish with decreasing temperature
    for temp in [200, 100, 50, 25, 10]:
        if time.time() - t_start > budget_s - 60:
            break
        x = slsqp_run(best_h, n, temp, 500)
        x = repair_h(x, n)
        c = compute_c5(x, n)
        if c < best_c5:
            best_c5, best_h = c, x

    # Phase 5: Bang-bang refinement
    if time.time() - t_start < budget_s - 40:
        best_h, best_c5 = bang_bang_refine(best_h, n, sweeps=5)

    # Phase 6: Final NM + SLSQP
    if time.time() - t_start < budget_s - 15:
        maxit = max(500, int((budget_s - 15 - (time.time() - t_start)) * 15))
        x = nm_run(best_h, n, maxit)
        x = repair_h(x, n)
        c = compute_c5(x, n)
        if c < best_c5:
            best_c5, best_h = c, x
    if time.time() - t_start < budget_s - 5:
        x = slsqp_run(best_h, n, 8, 300)
        x = repair_h(x, n)
        c = compute_c5(x, n)
        if c < best_c5:
            best_c5, best_h = c, x

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
