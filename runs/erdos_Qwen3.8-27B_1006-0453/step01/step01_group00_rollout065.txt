    n = len(h)

        return c_max * dx
    shifted = corr - c_max
    w = np.exp(shifted / T)
    s = np.sum(w)
    return (c_max + T * np.log(s)) * dx

def grad_fn(h):
    T = _T
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    if T < 1e-10:
        k_max = int(np.argmax(corr))
        g = np.zeros(n)
        for j in range(n):
            k = k_max
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= j <= hi:
                idx2 = j - k + n - 1
                g[j] += (1.0 - h[idx2])
            i2 = j + k - n + 1
            if 0 <= i2 < n and lo <= i2 <= hi:
                g[j] += -h[i2]
        return g * dx
    
    shifted = corr - c_max
    w = np.exp(shifted / T)
    s = np.sum(w)
    p = w / s
    
    g = np.zeros(n)
    cumsum_p = np.concatenate([[0.0], np.cumsum(p)])
    for j in range(n):
        # Term 1
        k1_lo = j
        k1_hi = min(2 * n - 2, j + n - 1)
        term1 = (1.0 - h[j]) * (cumsum_p[k1_hi + 1] - cumsum_p[k1_lo])
        
        # Term 2

            term2 = -np.dot(p[k2_lo:k2_hi + 1], h[m_vals])
        else:

        else:

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Core optimization with decreasing temperature
    T_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for T in T_list:
        if time_left() < 80:
            break
        h_opt = run_slsqp(h_best, T, n_best, max_iter=6000, ftol=1e-16)
        c5_cand = compute_c5(h_opt)
        if c5_cand < c5_best:
            h_best = h_opt
            c5_best = c5_cand

    # Phase 2: Perturbation-based restarts for escaping local optima
    for i in range(12):
        if time_left() < 100:
            break
        scale = 0.015 + 0.01 * (i % 3)
        noise = np.random.randn(n_best) * scale
        h_pert = np.clip(h_best + noise, 0.0, 1.0)
        h_pert = fix_sum(h_pert, n_best)
        T_try = [0.02, 0.008][i % 2]
        h_opt = run_slsqp(h_pert, T_try, n_best, max_iter=3000, ftol=1e-16)
        c5_cand = compute_c5(h_opt)
        if c5_cand < c5_best:
            h_best = h_opt
            c5_best = c5_cand

    # Phase 3: Explore lower resolutions for different basins
    for n_try in [100, 140, 200]:
        if time_left() < 130:
            break
        h_curr = downsample_h(h_best, n_try)
        for T in [0.04, 0.01]:
            if time_left() < 60:
                break
            h_opt = run_slsqp(h_curr, T, n_try, max_iter=4000, ftol=1e-16)
            c5_cand = compute_c5(h_opt)
            if c5_cand < c5_best:
                h_best = h_opt
                c5_best = c5_cand
                n_best = n_try
                h_curr = h_best.copy()

    # Phase 4: Try upsampling a good lower-res solution back to high res
    if time_left() > 100:
        h_up = upsample_h(h_best, max(n_best, 280))
        for T in [0.01, 0.003]:
            if time_left() < 60:
                break
            h_opt = run_slsqp(h_up, T, len(h_up), max_iter=5000, ftol=1e-16)
            c5_cand = compute_c5(h_opt)
            if c5_cand < c5_best:
                h_best = h_opt
                c5_best = c5_cand
                n_best = len(h_opt)

    # Phase 5: Final fine-tuning with very small T
    if time_left() > 40:
        h_opt = run_slsqp(h_best, 0.0005, n_best, max_iter=10000, ftol=1e-18)
        c5_cand = compute_c5(h_opt)
        if c5_cand < c5_best:
            h_best = h_opt
            c5_best = c5_cand

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    n_final = len(h_best)
    return (h_best, c5_final, n_final)
