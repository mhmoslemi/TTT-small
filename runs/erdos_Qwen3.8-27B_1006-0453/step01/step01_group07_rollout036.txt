from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.clip((corr - m) / beta, -40, 0)
    return dx * (m + beta * np.log(np.sum(np.exp(s))))

def fix_sum(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h += (n / 2.0 - h.sum()) / n
    return np.clip(h, 0.0, 1.0)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n) / old_n
    nx = np.arange(new_n) / new_n
    v = np.interp(nx, ox, parent)
    return fix_sum(v, new_n)

def slsqp(h_init, n, beta, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h_init, args=(n, beta),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return fix_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        return h_init

def subgrad(h, n, n_iter, step0=0.008):
    cur = compute_c5(h)
    step = step0
    for _ in range(n_iter):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        ks = np.argmax(corr)
        lag = ks - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            rng_i = range(lag, n)
        else:
            rng_i = range(0, n + lag)
        for i in rng_i:
            j1 = i
            j2 = i - lag
            grad[j1] += 1.0 - h[j2]
            grad[j2] -= h[j1]
        h_try = fix_sum(h - step * grad, n)
        c_try = compute_c5(h_try)
        if c_try < cur - 1e-15:
            h = h_try
            cur = c_try
            step = min(step * 1.01, 0.03)
        else:
            step *= 0.9
    return h, cur

    cur = compute_c5(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)

        h2 = fix_sum(h2, n)
        c = compute_c5(h2)
        if c < cur - 1e-14:
            h = h2
            cur = c

import numpy as np
from scipy.optimize import minimize
import time

_NG = 280

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h += (n / 2.0 - h.sum()) / n
    return np.clip(h, 0.0, 1.0)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n) / old_n
    nx = np.arange(new_n) / new_n
    v = np.interp(nx, ox, parent)
    return fix_sum(v, new_n)

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.clip((corr - m) / beta, -40, 0)
    return dx * (m + beta * np.log(np.sum(np.exp(s))))

def smooth_max_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.clip((corr - m) / beta, -40, 0)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        a1 = max(0, lag)
        b1 = min(n, n + lag)
        if a1 < b1:
            grad[a1:b1] += wk * (1.0 - x[a1 - lag:b1 - lag])
        a2 = max(0, -lag)
        b2 = min(n, n - lag)
        if a2 < b2:
            grad[a2:b2] += wk * (-x[a2 + lag:b2 + lag])
    grad *= dx
    return grad

def proj_grad(h, n, beta, n_iter, alpha0):
    cur = smooth_max_obj(h, n, beta)
    alpha = alpha0
    for _ in range(n_iter):
        g = smooth_max_grad(h, n, beta)
        h_new = h - alpha * g
        for _ in range(10):
            h_new = np.clip(h_new, 0, 1)
            d = n / 2.0 - h_new.sum()
            if abs(d) < 1e-12:
                break
            h_new += d / n
        h_new = np.clip(h_new, 0, 1)
        c = smooth_max_obj(h_new, n, beta)
        if c < cur - 1e-16:
            h = h_new
            cur = c
            alpha = min(alpha * 1.004, 0.015)
        else:
            alpha *= 0.93
            if alpha < 1e-7:
                break
    return h, compute_c5(h)

def cons_eq(x):
    return np.sum(x) - _NG / 2.0

def cons_eq_jac(x):
    return np.ones(_NG)

def slsqp_opt(h_init, n, beta, maxiter):
    global _NG
    _NG = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    try:
        res = minimize(smooth_max_obj, h_init, args=(n, beta),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return fix_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        return h_init

def subgrad_descend(h, n, n_iter, step0):
    cur = compute_c5(h)
    step = step0
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        ks = np.argmax(corr)
        lag = ks - (n - 1)
        grad = np.zeros(n)
        a = max(0, lag)
        b = min(n, n + lag)
        for i in range(a, b):
            grad[i] += 1.0 - h[i - lag]
            grad[i - lag] -= h[i]
        dx = 2.0 / n
        h_try = fix_sum(h - step * dx * grad, n)
        c_try = compute_c5(h_try)
        if c_try < cur - 1e-15:
            h = h_try
            cur = c_try
            step = min(step * 1.006, 0.025)
        else:
            step *= 0.94
            if step < 1e-6:
                break
    return h, cur

def coord_descent(h, n, n_iter, rng):

    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)

        h2 = fix_sum(h2, n)
        c = compute_c5(h2)
        if c < cur - 1e-14:
            h = h2
            cur = c
    return h, cur
