import numpy as np
from scipy.optimize import minimize
import time

_SM_T = [200.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(h):
    t = _SM_T[0]
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s_max = np.max(s)
    vals = np.exp(s - s_max)
    logsumexp = mx + (s_max + np.log(np.sum(vals))) / t
    return logsumexp * (2.0 / n)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return proj(np.interp(xd, xs, h, left=h[0], right=h[-1]), nd)

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def subgrad_step(h, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(corr))
    lag = m_star - (n - 1)
    g = np.zeros(n)
    if lag > 0:
        idx1 = np.arange(n - lag)
        g[idx1] += (1.0 - h[idx1 + lag])
        idx2 = np.arange(lag, n)
        g[idx2] -= h[idx2 - lag]
    elif lag < 0:
        l = -lag
        idx1 = np.arange(l, n)
        g[idx1] += (1.0 - h[idx1 - l])
        idx2 = np.arange(n - l)
        g[idx2] -= h[idx2 + l]
    else:
        g[:] = 1.0 - 2.0 * h
    h_new = h - lr * g
    return proj(h_new, n)

def subgrad_polish(h, niter=500, lr0=0.002):
    h = h.copy()
    n = len(h)
    lr = lr0
    for it in range(niter):
        h = subgrad_step(h, lr)
        if (it + 1) % 100 == 0:
            lr *= 0.75
    return h

def slsqp_solve(h, temp, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    _SM_T[0] = temp
    try:
        r = minimize(smooth_max_obj, h, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return proj(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        return h
