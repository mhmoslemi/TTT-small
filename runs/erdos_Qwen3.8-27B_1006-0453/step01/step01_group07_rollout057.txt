    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0, 1, old_n, endpoint=False)
    nx = np.linspace(0, 1, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v = project_feasible(v, new_n)
    return v

def softmax_gradient(h, n, beta):
    """Gradient of soft-max of corr values."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / beta, -50, 0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    n_lags = 2 * n - 1
    center = n - 1
    j = np.arange(n)
    
    for idx in range(n_lags):
        wi = w[idx]
        if wi < 1e-10:
            continue
        t = idx - center
        jp = j + t
        jm = j - t
        mp = (jp >= 0) & (jp < n)
        mm = (jm >= 0) & (jm < n)
        grad[mp] += wi * (1.0 - h[jp[mp]])
        grad[mm] -= wi * h[jm[mm]]
    
    return grad, m + beta * np.log(exp_s.sum())

def nesterov_descent(h0, n, beta_schedule, max_iter, rng, time_limit, t_start):
    """Nesterov momentum gradient descent on soft-max objective."""
    h = h0.copy().astype(np.float64)
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    alpha = 0.3 / n  # initial step size
    
    for i, beta in enumerate(beta_schedule):
        for it in range(max_iter):
            if time.time() - t_start > time_limit:
                break
            # Nesterov look-ahead
            y = h + mu * v
            grad, f_val = softmax_gradient(y, n, beta)
            v = mu * v - alpha * grad
            h_new = project_feasible(y + v, n)
            
            # Check improvement
            c5_new, _ = compute_c5(h_new)
            c5_old, _ = compute_c5(h)
            # Accept if better or first few iterations
            if c5_new < c5_old + 1e-15 or it < 10:
                h = h_new
            else:
                # Reduce step size
                alpha *= 0.7
                v = v * 0.5
            if alpha < 1e-8 / n:
                break
        # Decay momentum slightly
        mu = max(0.5, mu * 0.95)
    
    return h

def subgradient_polish(h, n, n_iter, rng, time_limit, t_start):
    """Polish using exact subgradient of max."""
    h = h.copy().astype(np.float64)
    c5, corr = compute_c5(h)
    dx = 2.0 / n
    center = n - 1
    j = np.arange(n)
    
    step = 0.05 / n
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        # Find top-k lags
        top_k = min(10, 2 * n - 1)
        top_indices = np.argsort(corr)[-top_k:]
        # Weighted average gradient (weights proportional to corr values near max)
        max_val = corr.max()
        top_corr = corr[top_indices]
        w = (top_corr - max_val + 0.01)  # small offset for numerical stability
        w = np.maximum(w, 0.001)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k, idx in enumerate(top_indices):
            t = idx - center
            jp = j + t
            jm = j - t
            mp = (jp >= 0) & (jp < n)
            mm = (jm >= 0) & (jm < n)
            grad[mp] += w[k] * (1.0 - h[jp[mp]])
            grad[mm] -= w[k] * h[jm[mm]]
        
        h_new = project_feasible(h - step * grad, n)
        c5_new, corr_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            corr = corr_new
            step = min(step * 1.1, 0.5 / n)
        else:
            step *= 0.7
            if step < 1e-10:
                break
    return h, c5

def random_local_search(h, n, n_iter, rng, time_limit, t_start):

    h = h.copy().astype(np.float64)

        c = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(c)
        s = c - m
        s = np.clip(s, -50, 0)
        return m + 0.003 * float(np.log(np.sum(np.exp(s)) + 1e-300))
    def eq_f(x):
        return np.sum(x) - n / 2.0
    def eq_j(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_f, 'jac': eq_j}],
                       options={'maxiter': 300, 'ftol': 1e-12})
        h_new = project_feasible(np.clip(res.x, 0, 1), n)
        return h_new
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5, _ = compute_c5(parent)
    best_n = pn

    resolutions = [pn]
    for rn in [int(pn * 1.25), int(pn * 1.5), int(pn * 2), 320, 350, 380, 400, 420, 450, 480]:
        if pn <= rn <= 500 and rn not in resolutions:
            resolutions.append(rn)
    resolutions = sorted(set(resolutions))
    resolutions = [r for r in resolutions if r >= pn]

    beta_schedules = [
        [0.04, 0.02, 0.01, 0.005, 0.003, 0.0015, 0.001, 0.0005],
        [0.03, 0.015, 0.007, 0.003, 0.001, 0.0005],
    ]

    for n in resolutions:
        if time.time() - t_start > budget_s - 80:
            break
        h0 = parent.copy() if n == pn else upsample(parent, n)

        n_starts = 5
        for s in range(n_starts):
            rem = budget_s - (time.time() - t_start)
            if rem < 40:
                break
            if s == 0:
                h_start = h0.copy()
            else:
                scale = 0.008 + 0.006 * s
                h_start = project_feasible(h0 + rng.randn(n) * scale, n)

            bs = beta_schedules[s % len(beta_schedules)]
            t_remain = min(120, (rem - 30) / (n_starts - s))
            h1 = pgd_optimize(h_start, n, bs, max_iters=int(t_remain * 4),
                              t_start=t_start, time_limit=t_remain)
            c5_1 = compute_c5(h1)[0]
            if c5_1 < best_c5:
                best_c5, best_h, best_n = c5_1, h1.copy(), n

            rem = budget_s - (time.time() - t_start)
            if rem < 25:
                continue
            t_polish = min(60, (rem - 15) // 2)
            h2, c5_2 = subgrad_polish(h1, n, n_iter=int(t_polish * 8),
                                       t_start=t_start, time_limit=t_polish)
            if c5_2 < best_c5:
                best_c5, best_h, best_n = c5_2, h2.copy(), n

            rem = budget_s - (time.time() - t_start)
            if rem < 20:
                continue
            t_ls = min(40, (rem - 10) // 2)
            h3, c5_3 = local_search(best_h, best_n, n_iter=int(t_ls * 25),
                                     rng=rng, t_start=t_start, time_limit=t_ls)
            if c5_3 < best_c5:
                best_c5, best_h, best_n = c5_3, h3, best_n

            rem = budget_s - (time.time() - t_start)
            if rem < 30:
                continue
            h4 = slsqp_polish(best_h, best_n, t_start=t_start,
                              time_limit=min(50, rem - 10))
            c5_4 = compute_c5(h4)[0]
            if c5_4 < best_c5:
                best_c5, best_h, best_n = c5_4, h4, best_n

    rem = budget_s - (time.time() - t_start)
    if rem > 20:
        hf, cf = subgrad_polish(best_h, best_n, n_iter=int(rem * 6),
                                 t_start=t_start, time_limit=rem - 5)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    rem = budget_s - (time.time() - t_start)
    if rem > 10:
        h5, c5_5 = local_search(best_h, best_n, n_iter=int(rem * 20),
                                 rng=rng, t_start=t_start, time_limit=rem - 3)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5

    return best_h, best_c5, best_n
