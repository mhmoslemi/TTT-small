            hr2, cr2 = greedy_maxtrack(hr, nt, bt, t0, budget_s, 10)
            if cr2 < best_c5:
                best_c5, best_h, best_n = cr2, hr2, nt
            elif cr < best_c5:
                best_c5, best_h, best_n = cr, hr, nt

    # Phase 4: Multi-start perturbations from best
    bb = [(0.0, 1.0)] * best_n
    for tr in range(60):
        if time.time() - t0 > budget_s - 50:
            break
        sc = 0.10 * np.exp(-tr / 10.0) + 0.01
        cand = np.clip(best_h + rng.randn(best_n) * sc, 0, 1)
        cand = repair_h(cand, best_n)
        hr, cr = opt_chain(cand, best_n, bb, t0, budget_s,
                           [50, 20, 8, 4], 40)
        hr2, cr2 = greedy_maxtrack(hr, best_n, bb, t0, budget_s, 8)
        if cr2 < best_c5:
            best_c5, best_h = cr2, hr2
            bb = [(0.0, 1.0)] * best_n
        elif cr < best_c5:
            best_c5, best_h = cr, hr
            bb = [(0.0, 1.0)] * best_n

    # Phase 5: Nelder-Mead polish on the specific max terms
    if time.time() - t0 < budget_s - 20:
        # Get top 3 lags
        dx = 2.0 / best_n
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_lags = np.argsort(corr)[::-1][:3]

        def top3_obj(x, n, lags):
            dx = 2.0 / n
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.max(c[lags]) * dx

        res = minimize(top3_obj, best_h, method='Nelder-Mead',
                       args=(best_n, top_lags),
                       options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
        hx = repair_h(res.x, best_n)
        c5x = compute_c5(hx, best_n)
        if c5x < best_c5:
            best_c5, best_h = c5x, hx

    # Phase 6: Final deep polish
    if time.time() - t0 < budget_s - 5:
        rem = budget_s - 5 - (time.time() - t0)
        mi = max(80, int(rem * 8))
        bb = [(0.0, 1.0)] * best_n
        for temp in [10, 5, 2, 1]:
            if time.time() - t0 > budget_s - 2:
                break
            try:
                hx = slsqp_solve(best_h, best_n, bb, temp, mi)
                hx = repair_h(hx, best_n)
                c = compute_c5(hx, best_n)
                if c < best_c5:
                    best_c5, best_h = c, hx
            except Exception:
                pass
        # One more greedy pass
        best_h, c_g = greedy_maxtrack(best_h, best_n, bb, t0, budget_s, 15)
        if c_g < best_c5:
            best_c5 = c_g

    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
