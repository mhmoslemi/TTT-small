def project_fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
            diff2 = n / 2.0 - h.sum()
            if abs(diff2) > 1e-12:
                mask2 = (h > 1e-12) & (h < 1.0 - 1e-12)
                if mask2.sum() > 0:
                    h[mask2] += diff2 / mask2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h

def grad_sm(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()
    w = np.exp((corr - cmax) / beta)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(w)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:

        grad[k:] += wk * (1.0 - h[:n - k])

    t1_budget = int(budget_s * 0.2)
    n_steps = 500
    for step in range(n_steps):
        if time.time() - t0 > t0 + t1_budget:
            break
        beta = max(0.0003, 0.004 * np.exp(-step / 150.0))
        lr = 0.03 * np.exp(-step / 300.0)
        grad = grad_sm(h, beta)
        h, m, v = adam_update(h, grad, m, v, step, lr)
        h = project_fix(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_h = h.copy()
            best_c5 = c5

    # Phase 2: Second Adam pass with even lower temperature
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        m2 = np.zeros(n)
        v2 = np.zeros(n)
        h2 = best_h.copy()
        n_steps2 = 300
        for step in range(n_steps2):
            if time.time() - t0 > t0 + remaining - 20:
                break
            beta = max(0.0001, 0.001 * np.exp(-step / 100.0))
            lr = 0.008 * np.exp(-step / 200.0)
            grad = grad_sm(h2, beta)
            h2, m2, v2 = adam_update(h2, grad, m2, v2, step, lr)
            h2 = project_fix(h2, n)
            c5 = compute_c5(h2)
            if c5 < best_c5 - 1e-13:
                best_h = h2.copy()
                best_c5 = c5

    # Phase 3: Coordinate descent with line search (main workhorse)
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        n_cd = int(min(50000, remaining * 25))
        best_h, best_c5 = coord_descent(best_h, n, n_cd, t0, budget_s, rng)

    # Phase 4: Second coordinate descent pass with smaller steps
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        n_cd2 = int(min(30000, remaining * 20))
        best_h, best_c5 = coord_descent(best_h, n, n_cd2, t0, budget_s, rng)

    return best_h, best_c5, n
