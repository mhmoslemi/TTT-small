            y = h + ((tau - 1.0) / t_new) * (h - h_old)
            if np.linalg.norm(h - h_old) < 1e-14:
                break
            tau = t_new
        c5_now = compute_c5(bh)
        c5_try = compute_c5(h)
        if c5_try < c5_b or smooth_obj(h, t) < smooth_obj(bh, t):
            c5_b = compute_c5(h)
            if c5_b < bc:
                bc = c5_b
                bh = h.copy()
    return bh, bc

def slsqp_polish(h_init, n, t_val, maxiter, ftol, time_limit):
    bnds = [(0.0, 1.0)] * n

    def obj_w(x):
        return smooth_obj(x, t_val)

    def grad_w(x):
        return smooth_grad(x, t_val)

    def ceq(x):
        return np.array([np.sum(x) - len(x) / 2.0])

    def ceqj(x):
        return np.ones((1, len(x)))

    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    try:
        r = minimize(obj_w, h_init, jac=grad_w, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        hc = proj_box_sum(np.clip(r.x, 0.0, 1.0), n / 2.0)
        return hc
    except Exception:
        return h_init.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial point
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        h0 = proj_box_sum(h0, len(h0) / 2.0)
    except Exception:
        pass
    if h0 is None:
        n0 = 280
        h0 = np.full(n0, 0.5) + rng.randn(n0) * 0.1
        h0 = proj_box_sum(h0, n0 / 2.0)

    n = len(h0)
    bh = h0.copy()
    bc = compute_c5(h0)

    # Phase 1: FISTA at current resolution with increasing temperature
    t_vals_1 = [40, 80, 160, 320]
    time_limit_1 = t0 + 0.35 * budget_s
    bh, bc = fista_optimize(bh, n, t_vals_1, 2000, time_limit_1)

    # Phase 2: SLSQP with analytical gradient
    time_limit_2 = t0 + 0.60 * budget_s
    if time.time() < time_limit_2:
        hc = slsqp_polish(bh, n, 200, 3000, 1e-15, time_limit_2)
        c5_try = compute_c5(hc)
        if c5_try < bc:
            bc = c5_try
            bh = hc.copy()

    # Phase 3: High-temperature SLSQP polish
    time_limit_3 = t0 + 0.80 * budget_s
    if time.time() < time_limit_3:
        hc = slsqp_polish(bh, n, 500, 5000, 1e-16, time_limit_3)
        c5_try = compute_c5(hc)
        if c5_try < bc:
            bc = c5_try
            bh = hc.copy()

    # Phase 4: Random restarts with quick FISTA
    time_limit_4 = t0 + budget_s - 30
    for trial in range(20):
        if time.time() > time_limit_4:
            break
        noise_scale = 0.03 + 0.02 * trial
        h_pert = proj_box_sum(bh + rng.randn(n) * noise_scale, n / 2.0)
        h_try, c5_try = fista_optimize(h_pert, n, [100, 400], 800, time_limit_4)
        if c5_try < bc:
            bc = c5_try
            bh = h_try.copy()

    # Phase 5: Multi-resolution: downsample, optimize, upsample
    if time.time() < t0 + budget_s - 20:
        n2 = n // 2
        h_down = np.zeros(n2)
        for i in range(n2):
            h_down[i] = 0.5 * (bh[2*i] + bh[2*i+1])
        h_down = proj_box_sum(h_down, n2 / 2.0)
        h_opt_down, _ = fista_optimize(h_down, n2, [50, 150, 400], 1500, t0 + budget_s - 20)
        h_up = xinterp(h_opt_down, n)
        c5_try = compute_c5(h_up)
        if c5_try < bc:
            # Refine the upsampled version
            hc = slsqp_polish(h_up, n, 300, 3000, 1e-15, t0 + budget_s - 10)
            c5_try2 = compute_c5(hc)
            if c5_try2 < bc:
                bc = c5_try2
                bh = hc.copy()

    # Final projection and return
    bh = proj_box_sum(bh, len(bh) / 2.0)
    return bh, compute_c5(bh), len(bh)
