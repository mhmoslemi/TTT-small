import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def topk_subgrad(h, n, topk=5, beta_lag=0.1):
    """Weighted subgradient from top lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Softmax weights over top lags
    weights = np.exp((corr - m) / beta_lag)
    # Normalize
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= w_sum
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    threshold = 0.01 / len(corr)
    active = np.where(weights > threshold)[0]
    
    for idx in active:
        w = weights[idx]
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n-l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n-l]
    
    return grad * dx, m * dx

def objective_smooth(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    return dx * (m + beta * np.log(np.sum(np.exp((corr - m) / beta))))

def fix_sum(h, n):
    s = np.sum(h) - n / 2.0
    if abs(s) < 1e-12:
        return h
    h = h.copy()
    h -= s / n
    h = np.clip(h, 0, 1)
    s2 = np.sum(h) - n / 2.0
    free = (h > 1e-10) & (h < 1 - 1e-10)
    if np.sum(free) > 0 and abs(s2) > 1e-12:
        h[free] -= s2 / np.sum(free)
        h = np.clip(h, 0, 1)
    return h

def coordinate_descent(h, n, rng, n_iter, t0, budget_s):
    h = h.copy()
    cur = compute_c5(h)
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.005, 0.05) * (1 if rng.rand() < 0.5 else -1)
        trial = h.copy()
        vi = np.clip(h[i] + d, 0, 1)
        vj = np.clip(h[j] - d, 0, 1)
        if vi == h[i] and vj == h[j]:
            continue
        # Check sum preservation
        delta = (vi - h[i]) + (vj - h[j])
        if abs(delta) > 1e-10:
            # Try to fix
            diff = -delta
            # Distribute to free variables
            free = np.where((trial > 0) & (trial < 1))[0]
            free = free[free != i]
            free = free[free != j]
            if len(free) == 0:
                continue
            idx = rng.choice(free)
            trial[i] = vi
            trial[j] = vj
            trial[idx] = np.clip(trial[idx] + diff, 0, 1)
            if abs(np.sum(trial) - n/2) > 1e-8:
                continue
        else:
            trial[i] = vi
            trial[j] = vj
        
        c = compute_c5(trial)
        if c < cur - 1e-15:
            h = trial
            cur = c
    return h, cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def topk_subgrad(h, n, beta_lag):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp((corr - m) / beta_lag)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= w_sum
    grad = np.zeros(n)
    threshold = 1e-4 / len(corr)
    active = np.where(weights > threshold)[0]
    for idx in active:
        w = weights[idx]
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
    return grad * dx, m * dx

def fix_sum(h, n):
    s = np.sum(h) - n / 2.0
    if abs(s) < 1e-12:
        return h
    h = h.copy()
    h -= s / n
    h = np.clip(h, 0, 1)
    s2 = np.sum(h) - n / 2.0
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.sum(free) > 0 and abs(s2) > 1e-12:
        h[free] -= s2 / np.sum(free)
        h = np.clip(h, 0, 1)
    return h

def obj_smooth(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    return dx * (m + beta * np.log(np.exp((corr - m) / beta).sum()))

def eq_fun(x, n):
    return np.array([np.sum(x) - n / 2.0])

def eq_jac(x, n):
    return np.array([np.ones(n)])

def obj_smooth_args(args):
    x, n, beta = args
    return obj_smooth(x, n, beta)

def coord_descent(h, n, rng, t0, deadline):
    h = h.copy()
    cur = compute_c5(h)
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.004, 0.04) * (1.0 if rng.random() < 0.5 else -1.0)
        vi = np.clip(h[i] + d, 0.0, 1.0)
        vj = np.clip(h[j] - d, 0.0, 1.0)
        if vi == h[i] and vj == h[j]:
            continue
        trial = h.copy()
        trial[i] = vi
        trial[j] = vj
        delta = (vi - h[i]) + (vj - h[j])
        if abs(delta) > 1e-10:
            free = np.where((trial > 1e-6) & (trial < 1.0 - 1e-6))[0]
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            mask[j] = False
            free = free[mask]
            if len(free) == 0:
                continue
            k = free[rng.randint(len(free))]
            trial[k] = np.clip(trial[k] - delta, 0.0, 1.0)
            if abs(np.sum(trial) - n / 2.0) > 1e-8:
                continue
        c = compute_c5(trial)
        if c < cur - 1e-15:
            h = trial
            cur = c
    return h, cur
