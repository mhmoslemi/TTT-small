import time
import math

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target):
    """Project h onto {0 <= h_i <= 1, sum(h) = target}."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        clipped = np.clip(h - mid, 0.0, 1.0)
        if clipped.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lag_grad(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:N] += 1.0 - h[0:N-m]
            g[0:N-m] -= h[m:N]
    else:
        l = -m
        if l < N:
            g[0:N-l] += 1.0 - h[l:N]
            g[l:N] -= h[0:N-l]
    return g

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    momentum = np.zeros(N)
    lr = 0.05
    decay = 0.999
    noise_period = 200
    noise_scale = 0.02
    k_start = min(3, 2*N-1)
    k_end = min(20, 2*N-1)

            best_h = h.copy()

        top_idx = np.argsort(corr)[-k:]

        if time.time() - t0 < budget_s - 60:
            h_pert = h_init + rng.randn(N) * 0.06
            h_pert = project_simplex(h_pert, target)
            h_opt2, c5_opt2 = optimize_gradient(h_pert, N, target, dx, t0, budget_s, rng,
                                                tau_init=10.0, lr_init=0.03)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
            if time.time() - t0 < budget_s - 20:
                h_lr2, c5_lr2 = local_refine(h_opt2.copy(), N, target, dx, t0, budget_s)
                if c5_lr2 < best_c5:
                    best_c5 = c5_lr2
                    best_h = h_lr2.copy()

        # Optimization run 3: anti-structured init
        if time.time() - t0 < budget_s - 60:
            h_anti = 0.5 + 0.5 * np.sin(3.0 * np.pi * xN)
            h_anti = project_simplex(h_anti, target)
            h_opt3, c5_opt3 = optimize_gradient(h_anti, N, target, dx, t0, budget_s, rng,
                                                tau_init=5.0, lr_init=0.04)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()

    # Final local refinement on best
    if time.time() - t0 < budget_s - 15:
        Nf = len(best_h)
        dxf = 2.0 / Nf
        tf = Nf / 2.0
        h_fin, c5_fin = local_refine(best_h.copy(), Nf, tf, dxf, t0, budget_s)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    if time.time() - t0 < budget_s - 10:
        Nf = len(best_h)
        dxf = 2.0 / Nf
        tf = Nf / 2.0
        h_ps2, c5_ps2 = pair_swap_refine(best_h.copy(), Nf, tf, dxf, t0, budget_s, rng)
        if c5_ps2 < best_c5:
            best_c5 = c5_ps2
            best_h = h_ps2.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
