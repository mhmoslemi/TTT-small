    """Compute correlation and gradient at specific lag index k_idx (in full correlation)."""
    n = len(h)
    l = k_idx - (n - 1)  # actual lag
    one_minus_h = 1.0 - h
    corr_full = np.correlate(h, one_minus_h, mode='full')
    
    # Gradient of corr_full[k_idx] w.r.t. h
    # corr_full[n-1+l] = sum_{i: 0<=i<n, 0<=i+l<n} h[i]*(1-h[i+l])
    # d/dh[m] = (1-h[m+l]) * 1_{0<=m+l<n} - h[m] * 1_{0<=m-l<n}

    if l >= 0:

        valid1 = np.arange(n) < (n - l)
        # m-l >= 0 means m >= l
        valid2 = np.arange(n) >= l
        grad[valid1] += (1.0 - h[valid1 + l])
        grad[valid2] -= h[valid2]
    else:

        q = -l
        # m+l >= 0 means m >= q

        # m-l < n means m < n - q
        valid2 = np.arange(n) < (n - q)
        grad[valid1] += (1.0 - h[valid1 + l])  # h[m-l] = h[m-q]
        grad[valid2] -= h[valid2]

    for s in range(steps):
        if s % 150 == 0 and s > 0:
            lr *= 0.65
        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        k_idx = np.argmax(corr)
        l = k_idx - (n - 1)
        grad = np.zeros(n)
        if l >= 0:
            m1 = idx < (n - l)
            m2 = idx >= l
            grad[m1] += (1.0 - h_c[m1 + l])
            grad[m2] -= h_c[m2]
        else:
            q = -l
            m1 = idx >= q
            m2 = idx < (n - q)
            grad[m1] += (1.0 - h_c[m1 - q])
            grad[m2] -= h_c[m2]
        v = mom * v - lr * grad
        h_c = np.clip(h_c + v, 0.0, 1.0)
        h_c = fix_sum(h_c, n)
        c5 = compute_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
    return best_h

def peaking_slsqp(h, n, top_m=8, p=64, max_iter=400):
    """Focus on worst correlations via additional inequality constraints."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    target = np.max(corr) * dx * 1.001
    target_corr = target / dx
    top_idx = np.argsort(corr)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    def make_ineq(ki):
        def f(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return c[ki] - target_corr
        return f
    for ki in top_idx:
        cons.append({'type': 'ineq', 'fun': make_ineq(ki)})
    res = opt.minimize(power_max_obj, h, args=(p,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
    if res.fun < np.inf:
        return fix_sum(res.x, n)
    return h

def perturb_top(h, n, scale=0.03):
    """Perturb in direction correlated with worst lags."""
    np.random.seed(np.random.randint(100000))
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-5:]
    noise = np.zeros(n)
    for ki in top_idx:
        l = ki - (n - 1)
        direction = np.zeros(n)
        if l >= 0:
            direction[:n-l] += 1.0
            direction[l:] -= 1.0
        else:
            q = -l
            direction[q:] += 1.0
            direction[:n-q] -= 1.0
        noise += np.random.randn(n) * 0.3 * direction
    h_p = h + scale * noise / np.linalg.norm(noise + 1e-10)
    return fix_sum(np.clip(h_p, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Power-sum SLSQP at initial resolution with increasing p
    for p in [16, 32, 64, 128, 256]:
        el = time.time() - t0
        if budget_s - el < 100:
            break
        h_opt = slsqp_opt(h_best.copy(), n_best, p, max_iter=300)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 2: Subgradient exploration
    el = time.time() - t0
    if budget_s - el > 180:
        h_opt = subgrad_search(h_best, n_best, steps=1200, lr_init=0.04, mom=0.90)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 3: Peaking SLSQP
    el = time.time() - t0
    if budget_s - el > 120:
        h_opt = peaking_slsqp(h_best, n_best, top_m=6, p=128, max_iter=500)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 4: Perturbation + re-optimize
    el = time.time() - t0
    if budget_s - el > 200:
        for attempt in range(3):
            el = time.time() - t0
            if budget_s - el < 80:
                break
            h_p = perturb_top(h_best, n_best, scale=0.04)
            h_opt = slsqp_opt(h_p, n_best, 64, max_iter=250)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best = h_opt, c

    # Phase 5: Upsample to higher resolution
    for n_try in [420, 560]:
        el = time.time() - t0
        if budget_s - el < 220:
            break
        h_up = upsample(h_best, n_try)
        for p in [32, 64, 128]:
            el = time.time() - t0
            if budget_s - el < 100:
                break
            h_opt = slsqp_opt(h_up, n_try, p, max_iter=250)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best, n_best = h_opt, c, n_try
                h_up = h_best.copy()

    # Phase 6: Final subgradient polish
    el = time.time() - t0
    if budget_s - el > 100:
        h_opt = subgrad_search(h_best, n_best, steps=600, lr_init=0.008, mom=0.92)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 7: Final SLSQP with high p
    el = time.time() - t0
    if budget_s - el > 50:
        h_opt = slsqp_opt(h_best.copy(), n_best, 256, max_iter=500)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
