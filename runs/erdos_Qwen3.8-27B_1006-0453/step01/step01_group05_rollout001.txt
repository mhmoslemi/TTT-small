    for trial_n in [40, 44, 46, 48, 50, 55, 60, 36]:
        if elapsed() > budget_s - 100:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, trial_n, endpoint=False)
        h_new = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
        h_new = repair_h(h_new, trial_n)
        bn = h_new.copy()
        bc5n = compute_c5(bn, trial_n)
        bn_bounds = [(0.0, 1.0)] * trial_n
        for temp in [40, 15, 7, 3]:
            if elapsed() > budget_s - 40:
                break
            h_t = slsqp_solve(bn, trial_n, bn_bounds, temp, min(200, max(40, int((budget_s - 40 - elapsed()) * 3))))
            h_t = repair_h(h_t, trial_n)
            c5_t = compute_c5(h_t, trial_n)
            if c5_t < bc5n:
                bc5n = c5_t
                bn = h_t.copy()
        if bc5n < best_c5:
            best_c5 = bc5n
            best_h = bn
            n = trial_n
            bounds = [(0.0, 1.0)] * n

    # Phase 4: Active-set refinement on top lags
    if elapsed() < budget_s - 40:
        for _ in range(30):
            if elapsed() > budget_s - 20:
                break
            dx = 2.0 / n
            corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
            top_k = int(np.argmax(corr))
            # Gradient descent step on the top lag
            g = active_lag_gradient(best_h, n, top_k)
            # Project gradient onto feasible direction (maintain sum)
            g -= np.mean(g)
            step = 0.01
            for _inner in range(20):
                cand = best_h - step * g
                cand = repair_h(cand, n)
                c5_t = compute_c5(cand, n)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = cand.copy()
                    break
                step *= 0.7
            # Also do SLSQP polish
            h_t = slsqp_solve(best_h, n, bounds, 5, 100)
            h_t = repair_h(h_t, n)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 5: COBYLA derivative-free for different landscape
    if elapsed() < budget_s - 30:
        res = minimize(smooth_obj, best_h, method='COBYLA',
                       args=(n, 8),
                       options={'maxiter': 600, 'rhobeg': 0.008, 'disp': False})
        h_t = repair_h(res.x, n)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 6: Final ultra-fine polish
    if elapsed() < budget_s - 5:
        remaining = max(3, budget_s - 5 - elapsed())
        for temp in [2, 1.5, 1, 0.5]:
            if elapsed() > budget_s - 3:
                break
            maxiter = min(300, max(40, int(remaining * 10)))
            h_t = slsqp_solve(best_h, n, bounds, temp, maxiter)
            h_t = repair_h(h_t, n)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
