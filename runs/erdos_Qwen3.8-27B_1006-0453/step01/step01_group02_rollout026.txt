                a[k_lo:k_hi+1] += 1.0 - h[j+m_lo:j+m_hi+1]
            
            # Second part: h[j] as (1-h[n+m]), n+j... 
            # (1-h[j-m]) where j-m is the "m-shifted" index
            # Term: h[j-m] * (-1) for h[j] as the -h in (1-h[j])
            # m ranges from max(j-(N-1), -(N-1)) to min(j, N-1)
            m_lo2 = max(j-(N-1), -(N-1))
            m_hi2 = min(j, N-1)
            if m_lo2 <= m_hi2:
                k_lo2 = m_lo2 + N - 1
                k_hi2 = m_hi2 + N - 1
                # h[j-m] for m = m_lo2..m_hi2
                # j-m for m=m_lo2: j-m_lo2 (largest)
                # j-m for m=m_hi2: j-m_hi2 (smallest)
                idxs = j - np.arange(m_lo2, m_hi2+1)
                a[k_lo2:k_hi2+1] -= h[idxs]
            
            # b[k] = corr[k] - old * a[k]
            b = corr - old * a
            
            # Add penalty for sum constraint
            # f(k) = b[k] + a[k]*t
            # Penalized: f(k) + penalty * (t - old) for maintaining sum
            # Actually, let's just do the soft constraint at the end
            
            # Minimize g(t) = max_k (b[k] + a[k]*t) over t in [0,1]
            # with g(t) convex piecewise-linear
            # Use subgradient: g'(t) = a[argmax_k(b[k]+a[k]*t)]
            # At optimum, 0 is in the subdifferential
            
            bisection_lo = 0.0
            bisection_hi = 1.0
            
            for _ in range(35):
                t_mid = (bisection_lo + bisection_hi) / 2.0
                vals = b + a * t_mid
                idx_max = np.argmax(vals)
                # Subgradient at t_mid is a[idx_max]
                # If a[idx_max] > 0, we want smaller t
                # If a[idx_max] < 0, we want larger t
                if a[idx_max] > 0:
                    bisection_hi = t_mid
                elif a[idx_max] < 0:
                    bisection_lo = t_mid
                else:
                    break
            t_opt = (bisection_lo + bisection_hi) / 2.0
