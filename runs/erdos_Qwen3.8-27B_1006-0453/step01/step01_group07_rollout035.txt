    # Get top-k indices
    idx = np.argsort(g)[-top_k:]
    # Average gradients at these indices
    grad = np.zeros(n)
    for k in idx:
        for j in range(n):
            # ∂g[k]/∂h[j]
            # g[k] = sum_i h[i] * (1-h[i-(k-n+1)])

            if 0 <= j - shift < n:

            # Second term: -h[j+shift] if j+shift in [0,n-1]
            if 0 <= j + shift < n:
                val -= h[j + shift]
            grad[j] += val
    grad /= top_k
    return grad * (2.0 / n)  # multiply by dx

            jb = j + s

                val += 1.0 - h[jf]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def subgradient_vec(h, top_k=10):
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(g)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        s = k - (n - 1)
        if s == 0:
            grad += 1.0 - 2.0 * h
        elif s > 0:
            a = np.zeros(n)
            a[s:] = 1.0 - h[:n-s]
            b = np.zeros(n)
            b[:n-s] = h[s:]
            grad += a - b
        else:
            ss = -s
            a = np.zeros(n)
            a[:n-ss] = 1.0 - h[ss:]
            b = np.zeros(n)
            b[ss:] = h[:n-ss]
            grad += a - b
    grad *= dx / top_k
    return grad

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        xp = np.clip(x - mid, 0.0, 1.0)
        s = xp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    xp = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    xp += (target_sum - xp.sum()) / n
    return np.clip(xp, 0.0, 1.0)

def adam_grad(h0, n, target, lr, top_k, n_iter, time_limit, beta1=0.9, beta2=0.999):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    t = 0
    while t < n_iter:
        if time.time() - t0 > time_limit:
            break
        grad = subgradient_vec(h, top_k)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        mhat = m / (1.0 - beta1 ** t)
        vhat = v / (1.0 - beta2 ** t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
        h = project_feasible(h_new, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

                h[i], h[j] = hi, hj
        if cur < best_c5:
            best_c5 = cur
            best_h = h
    
    return best_h, best_c5, best_n

        elif s > 0:
            a = np.zeros(n)
            a[s:] = 1.0 - h[:n - s]
            b = np.zeros(n)
            b[:n - s] = h[s:]
            grad += a - b
        else:
            ss = -s
            a = np.zeros(n)
            a[:n - ss] = 1.0 - h[ss:]
            b = np.zeros(n)
            b[ss:] = h[:n - ss]
            grad += a - b
    grad *= dx / top_k
    return grad

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        xp = np.clip(x - mid, 0.0, 1.0)
        s = xp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    xp = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    xp += (target_sum - xp.sum()) / n
    return np.clip(xp, 0.0, 1.0)

def adam_grad(h0, n, target, lr, top_k, n_iter, time_limit, beta1=0.9, beta2=0.999):
    h = h0.copy()
    mvec = np.zeros(n)
    vvec = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    t = 0
    patience = 0
    while t < n_iter:
        if time.time() - t0 > time_limit:
            break
        grad = subgradient_vec(h, top_k)
        t += 1
        mvec = beta1 * mvec + (1.0 - beta1) * grad
        vvec = beta2 * vvec + (1.0 - beta2) * (grad * grad)
        mhat = mvec / (1.0 - beta1 ** t)
        vhat = vvec / (1.0 - beta2 ** t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
        h = project_feasible(h_new, target)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > 20000:
            break
    return best_h, best_c5

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

_eq_target = 0.0

def _eq_fun(x):
    return np.sum(x) - _eq_target

def _eq_jac(x):
    return np.ones(len(x))

def _slsqp_obj(x, n, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def _slsqp_jac(x, n, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    p = np.exp((corr - m) / beta)
    p /= p.sum()
    nn = len(x)
    grad = np.zeros(nn)
    for k in range(len(corr)):
        if p[k] < 1e-14:
            continue
        s = k - (nn - 1)
        if s == 0:
            grad += p[k] * (1.0 - 2.0 * x)
        elif s > 0:
            a = np.zeros(nn)
            a[s:] = 1.0 - x[:nn - s]
            b = np.zeros(nn)
            b[:nn - s] = x[s:]
            grad += p[k] * (a - b)
        else:
            ss = -s
            a = np.zeros(nn)
            a[:nn - ss] = 1.0 - x[ss:]
            b = np.zeros(nn)
            b[ss:] = x[:nn - ss]
            grad += p[k] * (a - b)
    return grad * dx

def _slsqp_wrapper_slsqp(x):
    return _slsqp_obj(x, len(x), 2.0 / len(x), 0.002)

def _slsqp_wrapper_jac(x):
    return _slsqp_jac(x, len(x), 2.0 / len(x), 0.002)

def coordinate_descent(h0, n, target, rng, t_end):
    h = h0.copy()
    cur = compute_c5(h)
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.002, 0.025) * rng.choice([-1.0, 1.0])
        hi, hj = h[i], h[j]
        h[i] = np.clip(hi + d, 0.0, 1.0)
        h[j] = np.clip(hj - d, 0.0, 1.0)
        if abs(h.sum() - target) > 1e-7:
            h[i], h[j] = hi, hj
            continue
        c = compute_c5(h)
        if c < cur - 1e-14:
            cur = c
        else:
            h[i], h[j] = hi, hj
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    global _eq_target
    t0 = time.time()
    rng = np.random.RandomState(seed)

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    # Phase 1: Adam projected subgradient at parent resolution
    p1_configs = [
        (0.012, 8, 80000, 90),
        (0.008, 15, 80000, 80),
        (0.005, 5, 80000, 70),
        (0.015, 12, 60000, 60),
        (0.01, 25, 60000, 50),
    ]
    for lr, tk, nit, tl in p1_configs:
        rem = budget_s - (time.time() - t0)
        if rem < tl + 60:
            break
        h_new, c_new = adam_grad(best_h, pn, pn / 2.0, lr, tk, nit, tl)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
            best_n = pn

    # Phase 2: Random restarts and perturbation at parent res
    for restart in range(3):
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        perturbed = best_h + rng.randn(pn) * 0.003
        perturbed = project_feasible(perturbed, pn / 2.0)
        h_new, c_new = adam_grad(perturbed, pn, pn / 2.0, 0.01, 10, 40000, min(50, rem - 40))
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
            best_n = pn

    # Phase 3: Higher resolutions
    for n in [350, 420, 500]:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        h0 = upsample(best_h, n)
        target = n / 2.0
        for lr, tk, tl in [(0.008, 10, 50), (0.005, 20, 40)]:
            if budget_s - (time.time() - t0) < tl + 40:
                break
            h_new, c_new = adam_grad(h0, n, target, lr, tk, 25000, tl)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
                best_n = n

    # Phase 4: SLSQP refinement with analytical gradient
    rem = budget_s - (time.time() - t0)
    if rem > 70:
        n = best_n
        _eq_target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        tl_slsqp = min(55, rem - 30)
        if tl_slsqp > 15:
            try:
                import threading
                stop_flag = [False]
                def _timeout():
                    time.sleep(tl_slsqp)
                    stop_flag[0] = True
                th = threading.Thread(target=_timeout, daemon=True)
                th.start()
                res = minimize(_slsqp_wrapper_slsqp, best_h, jac=_slsqp_wrapper_jac,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-14})
                stop_flag[0] = True
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new += (n / 2.0 - h_new.sum()) / n
                h_new = np.clip(h_new, 0.0, 1.0)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new
                    best_n = n
            except Exception:
                pass

    # Phase 5: Second SLSQP pass with tighter beta
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        n = best_n
        _eq_target = n / 2.0
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        tl2 = min(30, rem - 15)
        if tl2 > 10:
            try:
                def _obj2(x):
                    corr = np.correlate(x, 1.0 - x, mode="full")
                    m = np.max(corr)
                    b = 0.0005
                    s = np.sum(np.exp((corr - m) / b))
                    return dx * (m + b * np.log(s))
                def _jac2(x):
                    corr = np.correlate(x, 1.0 - x, mode="full")
                    m = np.max(corr)
                    p = np.exp((corr - m) / 0.0005)
                    p /= p.sum()
                    nn = len(x)
                    grad = np.zeros(nn)
                    for k in range(len(corr)):
                        if p[k] < 1e-14:
                            continue
                        s = k - (nn - 1)
                        if s == 0:
                            grad += p[k] * (1.0 - 2.0 * x)
                        elif s > 0:
                            aa = np.zeros(nn)
                            aa[s:] = 1.0 - x[:nn - s]
                            bb = np.zeros(nn)
                            bb[:nn - s] = x[s:]
                            grad += p[k] * (aa - bb)
                        else:
                            ss = -s
                            aa = np.zeros(nn)
                            aa[:nn - ss] = 1.0 - x[ss:]
                            bb = np.zeros(nn)
                            bb[ss:] = x[:nn - ss]
                            grad += p[k] * (aa - bb)
                    return grad * dx
                res = minimize(_obj2, best_h, jac=_jac2, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-15})
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new += (n / 2.0 - h_new.sum()) / n
                h_new = np.clip(h_new, 0.0, 1.0)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new
                    best_n = n
            except Exception:
                pass

    # Phase 6: Coordinate descent fine-tuning
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        n = best_n
        target = n / 2.0
        t_end = t0 + budget_s - 2
        h_cd, c_cd = coordinate_descent(best_h, n, target, rng, t_end)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd
            best_n = n

    # Final feasibility check
    n = best_n
    if abs(best_h.sum() - n / 2.0) > 0.5:
        best_h = project_feasible(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
