import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0 and m < N:
        g[:N - m] += (1.0 - h[m:])
        g[m:] -= h[:N - m]
    elif m < 0:
        l = -m
        if l < N:
            g[l:] += (1.0 - h[:N - l])
            g[:N - l] -= h[l:]
    return g

def adam_opt(h, N, target, dx, t0, deadline, rng, lr=0.015, max_iter=2500, top_k=3):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    b1, b2, eps = 0.8, 0.999, 1e-8
    mb = np.zeros(N)
    vb = np.zeros(N)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        top_idx = np.argsort(corr)[-top_k:][::-1]
        k_actual = int(np.sum(corr[np.argsort(corr)[-top_k:]] > 0.01 * np.max(corr)))
        k_actual = max(1, min(k_actual, top_k))
        top_idx = top_idx[:k_actual]
        grad = np.zeros(N)
        for ii in range(len(top_idx)):
            lag = int(top_idx[ii]) - (N - 1)
            grad += grad_lag(h, lag, N)
        grad /= len(top_idx)
        mb = b1 * mb + (1.0 - b1) * grad
        vb = b2 * vb + (1.0 - b2) * (grad * grad)
        mh = mb / (1.0 - b1 ** (it + 1))
        vh = vb / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if (it + 1) % 600 == 0:
            lr *= 0.75
    return best_h, best_c5

def binary_search_pairs(h0, N, dx, rng, max_iter=3000, t0=None, deadline=None):
    h = h0.astyped(float).copy() if hasattr(h0, 'astyped') else h0.astype(float).copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    for it in range(max_iter):
        if t0 is not None and time.time() > deadline:
            break
        i1 = rng.choice(len(ones))
        i0 = rng.choice(len(zeros))
        p, q = ones[i1], zeros[i0]
        h[p] = 0.0
        h[q] = 1.0
        ones = np.delete(ones, i1)
        zeros = np.delete(zeros, i0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v <= best_c5 + 1e-14:
            best_c5 = c5v
        else:
            h[p] = 1.0
            h[q] = 0.0
            ones = np.insert(ones, i1, p)
            zeros = np.insert(zeros, i0, q)
    return h, best_c5

def make_binary(N, rng, style='random'):
    h = np.zeros(N)
    half = N // 2
    if style == 'random':
        idx = rng.choice(N, size=half, replace=False)
        h[idx] = 1.0
    elif style == 'block':
        h[:half] = 1.0
    elif style == 'alt':
        h[0::2] = 1.0
        h[1::2] = 0.0
    elif style == 'block2':
        q = N // 4
        h[:q] = 1.0
        h[2*q:3*q] = 1.0
    elif style == 'peri4':
        for i in range(N):
            if i % 4 < 2:
                h[i] = 1.0
    if float(h.sum()) != half:
        diff = half - int(h.sum())
        if diff > 0:
            z = np.where(h == 0)[0]
            h[rng.choice(z, diff, replace=False)] = 1.0
        elif diff < 0:
            o = np.where(h == 1)[0]
            h[rng.choice(o, -diff, replace=False)] = 0.0
    return h

    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    N_par = len(h_par)
    dx_par = 2.0 / N_par
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, dx_par)

    N = N_par
    dx = 2.0 / N
    target = N / 2.0
    x_par = np.linspace(0.0, 2.0, N_par, endpoint=False)

    # Strategy 1: Multiple Adam restarts from diverse inits at N_par
    inits = []
    inits.append(('parent', h_par.copy()))
    for s in ['random', 'alt', 'block2', 'peri4']:

    for i in range(4):
        noisy = h_par + rng.randn(N) * 0.08

    for name, h_init in inits:

        h_opt, c5_opt = adam_opt(h_init.copy(), N, target, dx, t0, deadline, rng, lr=0.02, max_iter=2000, top_k=3)
        if c5_opt < best_c5:

            best_h = h_opt.copy()

    # Strategy 2: Binary local search from rounded best
    h_round = (best_h > 0.5).astype(float)

        h_round = (h_round > 0.5).astype(float)
        if abs(h_round.sum() - target) > 0.5:

    h_bin, c5_bin = binary_search_pairs(h_round, N, dx, rng, max_iter=5000, t0=t0, deadline=deadline)
    if c5_bin < best_c5:

        styles = ['random', 'alt', 'block2', 'peri4', 'random']

            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Strategy 4: Adam from binary local search results (continuous refinement)
    h_ref = (best_h > 0.5).astype(float)
    h_ref = project_sum(h_ref, target)
    if time.time() < deadline:
        h_opt, c5_opt = adam_opt(h_ref.copy(), N, target, dx, t0, deadline, rng, lr=0.01, max_iter=2000, top_k=2)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Strategy 5: Interpolate to 2N and refine

        h2_init = np.interp(x2, x_par, best_h)
        h2_init = project_sum(h2_init, target2)
        h2_opt, c5_2 = adam_opt(h2_init, N2, target2, dx2, t0, deadline, rng, lr=0.01, max_iter=3000, top_k=3)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt.copy()
            N = N2
            dx = dx2
            target = target2

    # Strategy 6: Final polish with very small lr
    if time.time() < deadline:
        h_pol, c5_pol = adam_opt(best_h.copy(), N, N/2.0, 2.0/N, t0, deadline, rng, lr=0.005, max_iter=2000, top_k=2)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Strategy 7: One more binary round from final
    if time.time() < deadline:
        h_r2 = (best_h > 0.5).astype(float)
        h_r2, c5_r2 = binary_search_pairs(h_r2, N, 2.0/N, rng, max_iter=3000, t0=t0, deadline=deadline)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
