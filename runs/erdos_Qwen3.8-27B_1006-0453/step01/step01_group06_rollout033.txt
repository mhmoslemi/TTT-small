def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hc = np.clip(h + mid, 0.0, 1.0)
        if hc.sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:

        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

    if d > 1e-12:
        m = h < 1.0
        if m.any():
            h[m] = np.minimum(1.0, h[m] + d / m.sum())
    elif d < -1e-12:
        m = h > 0.0
        if m.any():
            h[m] = np.maximum(0.0, h[m] + d / m.sum())
    return h

def corr_subgrad(h, k):
    n = len(h)
    sg = np.zeros(n)
    a, b = max(0, k), min(n, n + k)
    if a < b:
        sg[a:b] += 1.0 - h[a-k:b-k]
    a, b = max(0, -k), min(n, n - k)
    if a < b:
        sg[a:b] -= h[a+k:b+k]
    return sg

def slsqp_smooth(h, n, T, maxiter=400):
    global _g_n
    _g_n = n
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(smooth_c5, h, (T,), 'SLSQP', bnds,
                    [{'type': 'eq', 'fun': _sum_cst}],
                    {'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(r.x, n)
    except Exception:
        return h

def subgrad_loop(h, n, max_iters, deadline, n_active, gap_frac):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        if time.time() > deadline:
            break

        cmax = corr.max()
        c5c = cmax * (2.0 / n)
        if c5c < best_c5:

        sg /= n_active
        nsq = float(sg @ sg)
        if nsq < 1e-18:

        tgt = cmax - gap_frac
        step = max(cmax - tgt, 0.0) / nsq
        step = min(max(step, 5e-4), 0.5)
        h2 = project_feasible(best - step * sg, n)
        c52 = compute_c5(h2)
        if c52 <= c5c + 1e-12:
            best = h2
            best_c5 = min(best_c5, c52)
        if it % 500 == 499:
            best = project_feasible(best + 0.02 * np.random.randn(n), n)
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 60

def _sum_cst(x):
    return x.sum() - _g_n / 2.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_c5(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float((m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    d = target - h.sum()
    if d > 1e-12:
        m = h < 1.0
        if m.any():
            h[m] = np.minimum(1.0, h[m] + d / m.sum())
    elif d < -1e-12:
        m = h > 0.0
        if m.any():
            h[m] = np.maximum(0.0, h[m] + d / m.sum())
    return h
