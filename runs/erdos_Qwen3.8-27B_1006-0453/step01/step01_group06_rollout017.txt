import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    w = exp_t / exp_t.sum()
    fval = (m + np.log(exp_t.sum()) / T) * dx
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            li = i - k + (n - 1)
            ri = i + k - (n - 1)
            if 0 <= li < n:
                s += w[k] * (1.0 - h[li])
            if 0 <= ri < n:
                s -= w[k] * h[ri]
        grad[i] = dx * s
    return fval, grad

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 600, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def lgbfgs_smooth(h, T, n):
    def obj(x):
        v, g = smooth_c5_and_grad(x, T)
        pen = 50.0 * (x.sum() - n / 2.0) ** 2
        gp = np.full(n, 100.0 * (x.sum() - n / 2.0))
        return v + pen, g + gp
    try:
        res = minimize(obj, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 400, 'gtol': 1e-9, 'factr': 1e8})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def pair_cd(h, n, max_rounds=3):
    best = h.copy()
    best_c5 = compute_c5(best)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _round in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                s = best[i] + best[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                a, b = lo, hi
                c = b - gr * (b - a)
                d = a + gr * (b - a)
                fc = compute_c5(np.array(best.tolist()[:i] + [c] + best.tolist()[i+1:j] + [s-c] + best.tolist()[j+1:])) if False else None
                # Efficient: modify in place
                for _ in range(25):
                    if b - a < 1e-11:
                        break
                    best[i], best[j] = c, s - c
                    fc = compute_c5(best)
                    best[i], best[j] = d, s - d
                    fd = compute_c5(best)
                    if fc < fd:
                        b = d
                    else:
                        a = c
                    c = b - gr * (b - a)
                    d = a + gr * (b - a)
                t_opt = (a + b) / 2.0
                best[i] = t_opt
                best[j] = s - t_opt
                cv = compute_c5(best)
                if cv < best_c5:
                    best_c5 = cv
                    improved = True
        if not improved:
            break
    return best, best_c5
