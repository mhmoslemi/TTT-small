        # Binary starts
        starts = []
        starts.append(make_binary(h_interp, N, target, rng))
        h_rand = np.zeros(N)
        h_rand[rng.choice(N, size=int(target), replace=False)] = 1.0
        starts.append(h_rand)
        for hs in starts:
            if time.time() - t0 > budget_s - 30:
                break
            # Simulated annealing
            h_cs, c5_cs = combinatorial_sa(hs, N, 200, t0, budget_s, rng)
            if c5_cs < best_c5:
                best_c5 = c5_cs
                best_h = h_cs.copy()
            # Targeted lag search
            if time.time() - t0 > budget_s - 25:
                break
            h_ts, c5_ts = target_lag_search(h_cs, N, 200, t0, budget_s, rng)
            if c5_ts < best_c5:
                best_c5 = c5_ts
                best_h = h_ts.copy()
        # Gradient from interpolated parent
        if time.time() - t0 > budget_s - 25:
            continue
        h_gp, c5_gp = gradient_opt(h_interp, N, target, dx, t0, budget_s, rng, lr_init=0.02, n_iters=1500)
        if c5_gp < best_c5:
            best_c5 = c5_gp
            best_h = h_gp.copy()
        # Gradient from best binary at this N
        if time.time() - t0 > budget_s - 20:
            continue
        h_bg, c5_bg = gradient_opt(best_h.copy() if len(best_h) == N else h_interp, N, target, dx, t0, budget_s, rng, lr_init=0.01, n_iters=1000)
        if c5_bg < best_c5:
            best_c5 = c5_bg
            best_h = h_bg.copy()
    # Final polish
    if time.time() - t0 < budget_s - 10:
        Nb = len(best_h)
        dxb = 2.0 / Nb
        tb = Nb / 2.0
        h_f, c5_f = gradient_opt(best_h.copy(), Nb, tb, dxb, t0, budget_s, rng, lr_init=0.005, n_iters=1000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
