import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_N = 280

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def soft_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / beta, -60, 0)
    s = np.sum(np.exp(shifted))
    return dx * (m + beta * np.log(s))

def eq_fun(x):
    return np.sum(x) - _GLOBAL_N / 2.0

def eq_jac(x):
    return np.ones(_GLOBAL_N)

def fix_sum(h, n):
    h = np.array(h, dtype=np.float64)
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def resample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 1.0, old_n, endpoint=False)
    nx = np.linspace(0.0, 1.0, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    return fix_sum(v, new_n)

def coord_descent(h, n, rng, max_iter, step_lo, step_hi, deadline):
    cur = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(step_lo, step_hi) * rng.choice([-1.0, 1.0])
        hi = min(1.0, h[i] + d)
        lo = max(0.0, h[j] - d)
        if abs(hi - h[i] - (1.0 - h[j] - lo)) > 1e-14:
            continue
        h_tr = h.copy()
        h_tr[i] = hi
        h_tr[j] = lo
        if abs(np.sum(h_tr) - n / 2.0) > 1e-8:
            continue
        c = compute_c5(h_tr)
        if c < cur - 1e-14:
            h = h_tr
            cur = c
    return h, cur

def top_k_descend(h, n, rng, max_iter, step_lo, step_hi, deadline):
    cur = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = np.argmax(corr)
        lag = k_max - (n - 1)
        # Perturb h at the lag to reduce correlation there
        i = rng.randint(n)
        j = (i + lag) % n
        if j < 0:
            j += n
        if i >= n or j >= n or i == j:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        d = rng.uniform(step_lo, step_hi) * rng.choice([-1.0, 1.0])
        h_tr = h.copy()
        h_tr[i] = np.clip(h[i] + d, 0, 1)
        h_tr[j] = np.clip(h[j] - d, 0, 1)
        if abs(np.sum(h_tr) - n / 2.0) > 1e-8:
            continue
        c = compute_c5(h_tr)
        if c < cur - 1e-14:
            h = h_tr
            cur = c
    return h, cur
