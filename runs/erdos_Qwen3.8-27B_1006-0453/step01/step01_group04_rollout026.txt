import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, temp, penalty=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h, 0.0, 1.0)
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h = h / integ
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, lr, n_steps):
    """Reduce the worst lag using subgradient descent"""
    n = len(h)
    dx = 2.0 / n
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        d = idx - (n - 1)  # the shift
        # gradient of corr[idx] w.r.t. h
        grad = np.zeros(n)
        # d(corr[idx])/dh[j] = (1-h[j-d]) - h[j+d] (with range checks)
        # For shift d: correlation at idx = sum_i h[i]*(1-h[i-d])
        # So grad[j] = (1-h[j-d]) if 0<=j-d<n, else 0
        #              - h[j+d] if 0<=j+d<n, else 0
        
        for j in range(n):
            g = 0.0
            if 0 <= j - d < n:
                g += (1.0 - h[j - d])
            if 0 <= j + d < n:
                g -= h[j + d]
            grad[j] = g
        
        h = np.clip(h - lr * grad, 0.0, 1.0)
        h = repair_h(h)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_obj(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def repair_h(h):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    integ = np.sum(h) * (2.0 / n)
    if integ < 1e-12:
        return h
    h = h / integ
    return np.clip(h, 0.0, 1.0)

def active_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        grad[:n-d] += 1.0 - h[d:]
        grad[d:] -= h[:n-d]
    else:
        dd = -d
        grad[dd:] += 1.0 - h[:n-dd]
        grad[:n-dd] -= h[dd:]
    return grad

def subgrad_descent(h, lr, n_iter, check_every=50):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        d = idx - (n - 1)
        g = active_grad(h, d)
        h = np.clip(h - lr * g, 0.0, 1.0)
        h = repair_h(h)
        if (i + 1) % check_every == 0:
            c = compute_c5(h)
            if c < best:
                best = c
    return h, best

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

    bounds = [(0.0, 1.0)] * n

    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    
    # Multiple temperature levels
    for temp, maxit in [(0.05, 150), (0.02, 150), (0.005, 150), (0.001, 100)]:
        if time.time() > deadline:
            break
        res = minimize(
            smooth_c5_obj, cur, args=(temp, 1e3),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-9}
        )
        nh = repair_h(np.clip(res.x, 0.0, 1.0))
        c5v = compute_c5(nh)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = nh.copy()
        cur = nh
    return best_h, best_c5
