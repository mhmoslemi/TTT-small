            valid = np.arange(n - shift)
            grad[valid] += (1.0 - h[valid + shift]) * dx

        T = T0 * (T1 / T0) ** (float(it) / max(float(niter - 1), 1.0))
        bn = b.copy()
        mt = rng.randint(4)
        on = np.where(bn == 1)[0]
        zo = np.where(bn == 0)[0]
        if len(on) == 0 or len(zo) == 0:
            continue
        if mt == 0:
            i = int(on[rng.randint(len(on))])
            j = int(zo[rng.randint(len(zo))])
            bn[i] = 0.0
            bn[j] = 1.0
        elif mt == 1:
            bl = rng.randint(2, max(3, n // 6))
            st = rng.randint(0, max(1, n - bl))
            bn[st:st + bl] = 1.0 - bn[st:st + bl]
            s = int(np.sum(bn))
            d = tgt - s
            if d > 0:
                zo2 = np.where(bn == 0)[0]
                for _ in range(min(d, len(zo2))):
                    j = int(zo2[rng.randint(len(zo2))])
                    bn[j] = 1.0
                    zo2 = np.where(bn == 0)[0]
            elif d < 0:
                on2 = np.where(bn == 1)[0]
                for _ in range(min(-d, len(on2))):
                    i = int(on2[rng.randint(len(on2))])
                    bn[i] = 0.0
                    on2 = np.where(bn == 1)[0]
        elif mt == 2:
            i1 = int(on[rng.randint(len(on))])
            j1 = int(zo[rng.randint(len(zo))])
            i2 = int(on[rng.randint(len(on))])
            j2 = int(zo[rng.randint(len(zo))])
            bn[i1] = 0.0; bn[j1] = 1.0
            bn[i2] = 0.0; bn[j2] = 1.0
        else:
            sl = rng.randint(3, max(4, n // 4))
            st = rng.randint(0, max(1, n - sl))
            bn[st:st + sl] = bn[st:st + sl][::-1]
        en = compute_c5(bn)
        if en < ec or (T > 1e-13 and rng.random() < np.exp(-(en - ec) / T)):
            b = bn
            ec = en
            if en < eb:
                eb = en
                bb = bn.copy()
    return bb, eb

def worst_shift_perturb(h, n, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    kstar = int(np.argmax(corr))
    shift = kstar - (n - 1)
    grad = np.zeros(n)
    s = abs(shift)
    if s >= n:
        return h
    if shift >= 0:
        for i in range(n - s):
            grad[i] += (1.0 - h[i + s]) * dx
            grad[i + s] -= h[i] * dx
    else:
        for i in range(n - s):
            grad[i + s] += (1.0 - h[i]) * dx
            grad[i] -= h[i + s] * dx
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: SA on binary version
    b_sa, e_sa = sa_binary(h_best, n_best, 0.004, 0.00003, 25000, seed)
    if e_sa < c5_best:
        c5_best = e_sa
        h_best = b_sa.copy()
    # Also SLSQP from the binary SA result
    h_try = fix_sum(b_sa.copy(), n_best)
    for T in [0.01, 0.002, 0.0005]:
        if time.time() - t_start > budget_s - 200:
            break
        h_opt = slsqp_solve(h_try, n_best, T, 600)
        h_opt = fix_sum(h_opt, n_best)
        c = compute_c5(h_opt)
        if c < c5_best:
            c5_best = c
            h_best = h_opt.copy()
            h_try = h_opt.copy()

    # Phase 2: SLSQP from parent with multiple T
    for T in [0.05, 0.01, 0.003, 0.001, 0.0003]:
        if time.time() - t_start > budget_s - 180:
            break
        h_opt = slsqp_solve(h_best.copy(), n_best, T, 800)
        h_opt = fix_sum(h_opt, n_best)
        c = compute_c5(h_opt)
        if c < c5_best:
            c5_best = c
            h_best = h_opt.copy()

    # Phase 3: Block coordinate descent
    for bsz in [8, 12, 6]:
        if time.time() - t_start > budget_s - 150:
            break
        h_tmp, c_tmp = block_cd(h_best.copy(), n_best, bsz, 2, 0.001, t_start, budget_s)
        if c_tmp < c5_best:
            c5_best = c_tmp
            h_best = h_tmp.copy()

    # Phase 4: Worst-shift guided descent
    for lr in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() - t_start > budget_s - 120:
            break
        h_tmp = h_best.copy()
        for _ in range(2000):
            h_tmp = worst_shift_perturb(h_tmp, n_best, lr)
            c = compute_c5(h_tmp)
            if c < c5_best:
                c5_best = c
                h_best = h_tmp.copy()
        # SLSQP polish from guided result
        h_opt = slsqp_solve(h_best.copy(), n_best, 0.001, 400)
        h_opt = fix_sum(h_opt, n_best)
        c = compute_c5(h_opt)
        if c < c5_best:
            c5_best = c
            h_best = h_opt.copy()

    # Phase 5: Try different resolutions
    res_list = [200, 320, 360, 420]
    for n_try in res_list:
        if time.time() - t_start > budget_s - 100:
            break
        h_up = upsample(h_best, n_try)
        for T in [0.01, 0.003, 0.001]:
            if time.time() - t_start > budget_s - 60:
                break
            h_opt = slsqp_solve(h_up, n_try, T, 500)
            h_opt = fix_sum(h_opt, n_try)
            c = compute_c5(h_opt)
            if c < c5_best:
                c5_best = c
                h_best = h_opt.copy()
                n_best = n_try
                h_up = h_opt.copy()

    # Phase 6: Final SLSQP polish
    if time.time() - t_start < budget_s - 30:
        for T in [0.0005, 0.0001]:
            if time.time() - t_start > budget_s - 15:
                break
            h_opt = slsqp_solve(h_best.copy(), n_best, T, 1000)
            h_opt = fix_sum(h_opt, n_best)
            c = compute_c5(h_opt)
            if c < c5_best:
                c5_best = c
                h_best = h_opt.copy()

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    n_final = len(h_best)
    if c5_final > c5_best:
        c5_final = c5_best
    return (h_best, c5_final, n_final)
