    return np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet sum constraint
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute adjustment
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # If clipping broke it, do a more careful fix
        if abs(np.sum(h) - n/2.0) > 1e-10:
            # Iterate
            for _ in range(10):
                diff = n/2.0 - np.sum(h)
                mask_below = h < 1.0
                mask_above = h > 0.0
                if diff > 0:
                    idx = np.where(mask_below)[0]
                    if len(idx) == 0: continue
                    adj = diff / len(idx)
                    h[idx] = np.minimum(h[idx] + adj, 1.0)
                else:
                    idx = np.where(mask_above)[0]
                    if len(idx) == 0: continue
                    adj = diff / len(idx)
                    h[idx] = np.maximum(h[idx] + adj, 0.0)
    return h

def downsample(h, new_n):
    old_n = len(h)
    if new_n >= old_n:
        return h
    # Use block averaging
    ratio = old_n / new_n
    idx = (np.arange(new_n) * ratio).astype(int)
    idx2 = (np.arange(new_n) * ratio + ratio).astype(int)
    idx2 = np.minimum(idx2, old_n)
    h2 = np.array([h[idx[i]:idx2[i]].mean() if idx2[i] > idx[i] else h[idx[i]] for i in range(new_n)])
    return fix_sum(h2, new_n)

    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    v = np.interp(nx, ox, h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_obj(x, n, beta):
    corr = corr_full(x)
    m = corr.max()
    p = np.exp((corr - m) / beta)
    f = (2.0/n) * (m + beta * np.log(p.sum()))
    diff = np.sum(x) - n/2.0
    return f + 1e4 * diff * diff

def smax_grad(x, n, beta):
    corr = corr_full(x)
    m = corr.max()
    p = np.exp((corr - m) / beta)
    w = p / p.sum()
    f = (2.0/n) * (m + beta * np.log(p.sum()))
    lags = np.arange(-(n-1), n)
    dx = 2.0/n
    grad = np.zeros(n)
    for j in range(n):
        gi = 0.0
        for si in range(len(lags)):
            s = lags[si]
            wi = w[si]
            if 0 <= j - s < n:
                gi += wi * (1.0 - x[j-s])
            if 0 <= j + s < n:
                gi -= wi * x[j+s]
        grad[j] = dx * gi
    diff = np.sum(x) - n/2.0
    grad += 2e4 * diff
    return f, grad
