def project_feasible(h, n):

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-13:

            if total > 1e-13:

        else:
            total = h.sum()
            if total > 1e-13:

    m = np.max(corr)

    w = np.exp(T * (corr - m))
    w = w / w.sum()

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-12:
                h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot > 1e-12:
                h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    c = corr_values(h)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cd = c * dx
    m = np.max(cd)
    f = m + np.log(np.sum(np.exp(T * (cd - m)))) / T
    w = np.exp(T * (cd - m))
    w = w / w.sum()

    # dc_k/dh[m] = (1-h[m+k]) - h[m-k] (with valid index checks)
    # df/dh[m] = sum_{k_idx} w[k_idx] * dx * [(1-h[m+k]) - h[m-k]]

        wk = w[ki] * dx
        if wk < 1e-16:
            continue
        k = ki - (n - 1)
        # (1 - h[m+k]) for m where 0<=m<n and 0<=m+k<n

        hi2 = min(n, n + k)
        if hi2 > lo2:

    return f, grad

def aug_obj(hh, T, nu, mu, n):
    s = hh.sum() - n / 2.0
    return smooth_c5(hh, T) + nu * s + 0.5 * mu * s * s

def aug_obj_grad(hh, T, nu, mu, n):
    s = hh.sum() - n / 2.0
    f, g = smooth_c5_grad(hh, T)
    g = g + (nu + mu * s) * np.ones(n)
    return f, g

def sum_constraint_fun(h):
    return h.sum() - len(h) / 2.0

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) + 0.5
    x_new = np.arange(n_new) + 0.5
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    return np.clip(f(x_new), 0.0, 1.0)

```python
def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = np.max(c)
    f = m + np.log(np.sum(np.exp(T * (c - m)))) / T
    w = np.exp(T * (c - m))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki] * dx
        if wk < 1e-16:
            continue
        k = ki - (n - 1)
        # Term 1: (1 - h[m-k]) for m in [max(0,k), min(n-1, n-1+k)]
        lo1 = max(0, k)
        hi1 = min(n, n + k)  # exclusive upper bound
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - k:hi1 - k])
        # Term 2: -h[m+k] for m in [max(0,-k), min(n-1, n-1-k)]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 + k:hi2 + k]
    return f, grad
```
