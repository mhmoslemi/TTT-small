    """Local search on binary sequences with swap moves."""

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    corr = np.correlate(h, 1.0 - h, mode='full')

        j = zeros[np.random.randint(len(zeros))]

                other = j + n - 1 - k
                if 0 <= other < n and other != j:
                    grad[j] += w[k] * (1.0 - h[other])
            # Second: k such that 0<=k<=2n-2 and 0<=j+n-1-k<n => k in [n-1, n-1+j]
            lo2 = max(0, n - 1)
            hi2 = min(2 * n - 2, n - 1 + j)
            for k in range(lo2, hi2 + 1):
                other = j + n - 1 - k
                if 0 <= other < n and other != j:
                    grad[j] -= w[k] * h[other]
        h = project_sr(h - lr * grad * dx, target_sum)
        c5_new = compute_c5(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h0, dx, target_sum, n_sweeps=80, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h, dx)
    for sweep in range(n_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        for i in range(n):
            old = h[i]
            # Try setting to 0 and 1, and current ± small
            for trial in [0.0, 1.0, old + 0.05, old - 0.05, old + 0.02, old - 0.02]:
                h2 = h.copy()
                h2[i] = np.clip(trial, 0.0, 1.0)
                h2 = project_sr(h2, target_sum)
                c2 = compute_c5(h2, dx)
                if c2 < best_c5 - 1e-14:
                    h = h2
                    best_c5 = c2
                    best_h = h.copy()
            # If none improved, keep old
            h[i] = old
            h = project_sr(h, target_sum)
    return best_h, best_c5

def make_binary_init(n, rng):
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        h[perm[n // 2]] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.array([1.0] * 23 + [0.5] + [0.0] * 23, dtype=float)

    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_sr(h_raw, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def tok(margin=10):
        return (time.time() - t0) < budget_s - margin

    # Phase 1: SLSQP from parent
    if tok(200):
        h1 = slsqp(h0, dx, target_sum, 500)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: L-BFGS smoothed gradient
    if tok(150):
        h2, c2 = lgbfsmooth(best_h, dx, target_sum, 3000, 20.0, 60.0, 0.015)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP polish
    if tok(120):
        h3 = slsqp(best_h, dx, target_sum, 400)
        c3 = compute_c5(h3, dx)
        if c3 < best_c5:
            best_h, best_c5 = h3, c1 if c3 < best_c5 else best_c5
        best_c5 = compute_c5(best_h, dx)

    # Phase 4: Coordinate descent
    if tok(120):
        h4, c4 = coord_descent(best_h, dx, target_sum, 60, t0 + budget_s - 120)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: SLSQP after coord descent
    if tok(80):
        h5 = slsqp(best_h, dx, target_sum, 300)
        c5v = compute_c5(h5, dx)
        if c5v < best_c5:
            best_h, best_c5 = h5, c5v

    # Phase 6: Many small perturbations + SLSQP
    for r in range(40):
        if not tok(50):
            break
        scale = [0.03, 0.05, 0.08, 0.12][r % 4]
        h_r = best_h + scale * rng.standard_normal(n)
        h_r = project_sr(h_r, target_sum)
        h_r = slsqp(h_r, dx, target_sum, 150)
        c5_r = compute_c5(h_r, dx)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 7: L-BFGS from best
    if tok(60):
        h7, c7 = lgbfsmooth(best_h, dx, target_sum, 4000, 30.0, 100.0, 0.01)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7
            h7 = slsqp(h7, dx, target_sum, 200)
            c7 = compute_c5(h7, dx)
            if c7 < best_c5:
                best_h, best_c5 = h7, c7

    # Phase 8: Try other n values
    for n_try in [48, 50, 52, 54, 60, 66]:
        if not tok(100):
            break
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        for attempt in range(3):
            if not tok(60):
                break
            if attempt == 0:
                h_t = make_binary_init(n_try, rng)
            elif attempt == 1:
                h_t = np.tile([1.0, 0.0], n_try // 2)
                h_t = h_t[:n_try]
                if n_try % 2:
                    h_t[-1] = 0.5
            else:
                h_t = np.zeros(n_try)
                for i in range(n_try):
                    h_t[i] = 1.0 if (i * 7 + attempt * 3) % n_try < n_try // 2 else 0.0
            h_t = project_sr(h_t, target_t)
            h_t = slsqp(h_t, dx_t, target_t, 250)
            c5_t = compute_c5(h_t, dx_t)
            if c5_t < best_c5:
                best_h, best_c5, n, dx, target_sum = h_t, c5_t, n_try, dx_t, target_t

    # Final polish
    if tok(10):
        hf = slsqp(best_h, dx, target_sum, 200)
        cf = compute_c5(hf, dx)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    best_h = project_sr(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, len(best_h)
