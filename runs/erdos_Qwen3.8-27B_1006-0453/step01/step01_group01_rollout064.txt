```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        step = d / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def grad_smooth(h, t):
    n = len(h)
    corr = corr_full(h)
    mx = np.max(corr)
    w = np.exp(t * (corr - mx))
    w /= w.sum()
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        jlo = max(0, n - 1 - k)
        jhi = min(n - 1, 2 * n - 2 - k)
        if jlo <= jhi:
            idx = np.arange(jlo, jhi + 1) + k - (n - 1)
            g[jlo:jhi + 1] += wk * omh[idx]
        jlo2 = max(0, k - (n - 1))
        jhi2 = min(n - 1, k)
        if jlo2 <= jhi2:
            idx2 = np.arange(jlo2, jhi2 + 1) - k + (n - 1)
            g[jlo2:jhi2 + 1] -= wk * h[idx2]
    return g

def smooth_obj(h, t):
    corr = corr_full(h)
    mx = np.max(corr)
    return (mx + np.log(np.sum(np.exp(t * (corr - mx)))) / t)

def projected_grad_descent(h0, t, n_steps, t_end, step0):
    """Projected gradient descent with adaptive step size"""
    n = len(h0)
    h = h0.copy()
    step = step0
    for i in range(n_steps):
        # Anneal temperature
        tt = t * (t_end / t) ** (i / max(1, n_steps - 1))
        # Actually, let's keep t constant or use a schedule
        tt = t + (t_end - t) * i / max(1, n_steps - 1)
        g = grad_smooth(h, tt)
        # Backtracking line search
        f0 = smooth_obj(h, tt)
        alpha = step
        for _ in range(50):
            h_new = proj_feasible(h - alpha * g, n)
            f1 = smooth_obj(h_new, tt)
            if f1 < f0 - 1e-4 * alpha * np.sum(g * (h_new - h)):
                h = h_new
                break
            alpha *= 0.5
        else:
            h = proj_feasible(h - alpha * g, n)
        # Slight increase in step size if we didn't backtrack much
        if alpha == step:
            step = min(step * 1.05, 0.1)
        else:
            step = alpha * 1.5
    return h

def slsqp_refine(h0, t, maxiter):
    n = len(h0)
    def obj(h):
        return smooth_obj(h, t)
    def obj_grad(h):
        return grad_smooth(h, t)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}
    try:
        r = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        return proj_feasible(r.x, n)
    except Exception:
        return h0

def interp_to(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = np.arange(n) * 2.0 / n
    xd = np.arange(nd) * 2.0 / nd
    return proj_feasible(np.interp(xd, xs, h), nd)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    bh = None
    try:
        bh = np.array(initial_h_values, dtype=float).copy()
        bh = proj_feasible(bh, len(bh))
    except Exception:
        bh = np.full(280, 0.5)
    
    bc = compute_c5(bh)
    
    def elapsed():
        return time.time() - t0
    
    def time_left():
        return budget_s - elapsed()
    
    # Phase 1: Refine at current resolution (280)
    n = len(bh)
    
    # Gradient descent phase
    if time_left() > 200:
        bh = projected_grad_descent(bh, t=80.0, n_steps=500, t_end=300.0, step0=0.01)
        bh = proj_feasible(bh, n)
        bc = compute_c5(bh)
    
    # SLSQP refinement
    if time_left() > 150:
        bh2 = slsqp_refine(bh, t=200.0, maxiter=1000)
        c2 = compute_c5(bh2)
        if c2 < bc:
            bc = c2
            bh = bh2
        bh = slsqp_refine(bh, t=400.0, maxiter=2000)
        c2 = compute_c5(bh)
        if c2 < bc:
            bc = c2
    
    # Phase 2: Basin hopping
    for it in range(30):
        if time_left() < 60:
            break
        # Random perturbation
        eps = 0.1 * max(0.1, 0.5 * elapsed() / budget_s)
        h_pert = proj_feasible(bh + eps * rng.randn(n), n)
        # Optimize
        h_opt = projected_grad_descent(h_pert, t=100.0, n_steps=200, t_end=300.0, step0=0.008)
        h_opt = slsqp_refine(h_opt, t=300.0, maxiter=500)
        c_opt = compute_c5(h_opt)
        if c_opt < bc:
            bc = c_opt
            bh = h_opt
    
    # Phase 3: Multi-resolution
    # Try coarse grid, then refine
    if time_left() > 200:
        for nc in [70, 140]:
            if time_left() < 80:
                break
            hc = interp_to(bh, nc)
            # Optimize on coarse grid
            hc = projected_grad_descent(hc, t=100.0, n_steps=300, t_end=400.0, step0=0.01)
            hc = slsqp_refine(hc, t=300.0, maxiter=1000)
            # Interpolate back
            h_fine = interp_to(hc, n)
            h_fine = projected_grad_descent(h_fine, t=200.0, n_steps=200, t_end=500.0, step0=0.005)
            h_fine = slsqp_refine(h_fine, t=500.0, maxiter=1000)
            c_fine = compute_c5(h_fine)
            if c_fine < bc:
                bc = c_fine
                bh = h_fine
    
    # Final polish
    if time_left() > 30:
        bh = slsqp_refine(bh, t=500.0, maxiter=3000)
        bc = compute_c5(bh)
    
    return bh, compute_c5(bh), len(bh)
```

This is a linear constraint in (x, t).

```python
def solve_cp_iteration(h, top_k, n):
    """One iteration of the cutting plane / linearization method"""
    corr = np.correlate(h, 1-h, 'full')
    
    # Build constraint matrix
    # Variables: x (n) + t (1) = n+1
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = []  # each row: grad_k @ x - t <= rhs
    b_ub = []
    
    for k in top_k:
        ik = k - (n - 1)
        gk = np.zeros(n)
        for j in range(n):
            # First factor: (1-h[j+ik]) if 0 <= j+ik < n
            if 0 <= j + ik < n:
                gk[j] += (1 - h[j + ik])
            # Second factor: -h[j-ik] if 0 <= j-ik < n
            if 0 <= j - ik < n:
                gk[j] -= h[j - ik]
        
        A_row = np.zeros(n + 1)
        A_row[:n] = gk
        A_row[n] = -1.0
        A_ub.append(A_row)
        
        rhs = gk @ h - corr[k]
        b_ub.append(rhs)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds on x: 0 <= x <= 1, t is free (but will be bounded)
    bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return h
    
    return proj_feasible(res.x[:n], n)
```

This is O(n * m) per iteration where m = len(top_k). For n=280, m=20, that's 5600 entries per constraint row, 20 rows. Very manageable.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(1000):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + d / cnt, 0.0, 1.0)
    return h

def grad_smooth(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    mx = np.max(corr)
    w = np.exp(t * (corr - mx))
    w /= w.sum()
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        ik = k - (n - 1)
        jlo = max(0, -ik)
        jhi = min(n, n - ik)
        if jlo < jhi:
            idx = np.arange(jlo, jhi) + ik
            g[jlo:jhi] += wk * omh[idx]
        jlo2 = max(0, ik)
        jhi2 = min(n, n + ik)
        if jlo2 < jhi2:
            idx2 = np.arange(jlo2, jhi2) - ik
            g[jlo2:jhi2] -= wk * h[idx2]
    return g

def smooth_obj(h, t):
    corr = np.correlate(h, 1.0 - h, 'full')
    mx = np.max(corr)
    return float(mx + np.log(np.sum(np.exp(t * (corr - mx)))) / t)

def lp_iterate(h, n, n_active, max_sub):
    corr0 = np.correlate(h, 1.0 - h, 'full')
    top_idx = np.argsort(corr0)[-n_active:]
    A_rows = []
    b_vals = []
    for k in top_idx:
        ik = k - (n - 1)
        gk = np.zeros(n)
        jlo = max(0, -ik)
        jhi = min(n, n - ik)
        if jlo < jhi:
            gk[jlo:jhi] += 1.0 - h[np.arange(jlo, jhi) + ik]
        jlo2 = max(0, ik)
        jhi2 = min(n, n + ik)
        if jlo2 < jhi2:
            gk[jlo2:jhi2] -= h[np.arange(jlo2, jhi2) - ik]
        row = np.zeros(n + 1)
        row[:n] = gk
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(gk @ h - corr0[k])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    for _ in range(max_sub):
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            break
        h_new = proj_feasible(res.x[:n], n)
        if np.max(np.abs(h_new - h)) < 1e-9:
            h = h_new
            break
        h = h_new
        corr0 = np.correlate(h, 1.0 - h, 'full')
        top_idx = np.argsort(corr0)[-n_active:]
        A_rows = []
        b_vals = []
        for k in top_idx:
            ik = k - (n - 1)
            gk = np.zeros(n)
            jlo = max(0, -ik)
            jhi = min(n, n - ik)
            if jlo < jhi:
                gk[jlo:jhi] += 1.0 - h[np.arange(jlo, jhi) + ik]
            jlo2 = max(0, ik)
            jhi2 = min(n, n + ik)
            if jlo2 < jhi2:
                gk[jlo2:jhi2] -= h[np.arange(jlo2, jhi2) - ik]
            row = np.zeros(n + 1)
            row[:n] = gk
            row[n] = -1.0
            A_rows.append(row)
            b_vals.append(gk @ h - corr0[k])
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
    return h

def pgd(h0, n, n_steps, t_start, t_end, step0):
    h = h0.copy()
    step = step0
    for i in range(n_steps):
        tt = t_start + (t_end - t_start) * i / max(1, n_steps - 1)
        g = grad_smooth(h, tt)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        f0 = smooth_obj(h, tt)
        alpha = step
        accepted = False
        for _ in range(40):
            h_try = proj_feasible(h - alpha * g, n)
            f1 = smooth_obj(h_try, tt)
            if f1 < f0 - 1e-5 * alpha * g_norm * (np.linalg.norm(h_try - h) / max(g_norm, 1e-15)):
                h = h_try
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            h = proj_feasible(h - alpha * g, n)
        if alpha >= step * 0.9:
            step = min(step * 1.03, 0.05)
        else:
            step = max(alpha * 1.2, 1e-5)
    return h

def slsqp_refine(h0, n, t, maxiter):
    bnds = [(0.0, 1.0)] * n
    def objf(x):
        return smooth_obj(x, t)
    def objg(x):
        return grad_smooth(x, t)
    def consf(x):
        return np.array([x.sum() - n / 2.0])
    def consj(x):
        return np.array([np.ones(n)])
    try:
        r = minimize(objf, h0, jac=objg, method='SLSQP',
                     bounds=bnds,
                     constraints={'type': 'eq', 'fun': consf, 'jac': consj},
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return proj_feasible(r.x, n)
    except Exception:
        return h0

def interp_to(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = np.arange(n) * 2.0 / n
    xd = np.arange(nd) * 2.0 / nd
    return proj_feasible(np.interp(xd, xs, h), nd)

        return time.time() - t0

        return budget_s - elapsed()

        c_pd = compute_c5(bh)

            bc = c_pd

    for tv in [200.0, 500.0, 1000.0]:

        c2 = compute_c5(bh2)

    base_eps = 0.12

        h_opt = pgd(h_pert, n, 150, t_start=100.0, t_end=400.0, step0=0.006)
```
