The correlation at lag k (where k = idx - (n-1)):
c(k) = sum_{i} h[i] * (1 - h[i+k]) where the sum is over valid i (0 ≤ i < n and 0 ≤ i+k < n)

∂c(k)/∂h[j] = sum over valid i of:
- (1 - h[i+k]) if i = j (j is the first index)
- (-h[i]) if i+k = j, i.e., i = j-k (j is the second index)

So: ∂c(k)/∂h[j] = (1 - h[j+k]) * [j+k is valid] + (-h[j-k]) * [j-k is valid]

With the dx factor: dc/dh[j] = dx * [(1-h[j+k])*(0≤j+k<n) - h[j-k]*(0≤j-k<n)]

Now, the subgradient of f(h) = max_k c(k)*dx is:
g = dx * ∇c(k*) for the k* that achieves the max.

More generally, it's in the convex hull of {dx * ∇c(k) : k in active set}.

For the top-k approach, I'll compute the average gradient over the top-k active shifts. This should give a better descent direction.

Let me also think about step size for subgradient descent. A standard choice is step_t = c / sqrt(t) or c/t. But since we have box constraints and a linear equality constraint, we need to be more careful.

Actually, for this problem, since the feasible set is compact and the objective is Lipschitz, subgradient descent with step = c/sqrt(t) should converge. But the convergence rate is O(1/sqrt(t)), which is slow.

A better approach: use the subgradient direction but with a line search (backtracking) to find the best step size. This is the "projected subgradient with line search" method.

def subgradient_step(h, n, top_k=5):

    grad = grad_c5_active(h, top_k)
    # Line search: try different step sizes
    best_h_step = h
    best_c5_step = compute_c5(h)

        h_trial = project_feasible(h - alpha * grad, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5_step:
            best_c5_step = c5_trial
            best_h_step = h_trial
    return best_h_step, best_c5_step

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h = h * (target / total)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    diff2 = target - s2
    if abs(diff2) > 1e-10:
        mask = (h > 0.02) & (h < 0.98)
        if mask.sum() > 0:
            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_c5_active(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        for i in range(n):
            j2 = i + k
            if 0 <= j2 < n:
                grad[i] += (1.0 - h[j2])
                grad[j2] += (-h[i])
    return grad * dx

def slsqp_run(h, T, n, maxiter=600):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def nm_run(h, n, maxiter=3000):
    def obj(x):
        pen = (abs(x.sum() - n / 2.0) * 5000.0 +
               np.sum(np.maximum(0, x - 1)) * 5000.0 +
               np.sum(np.maximum(0, -x)) * 5000.0)
        return compute_c5(x) + pen
    try:
        res = minimize(obj, h, method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up (8%)
    for T in [20, 60, 200, 600]:
        if elapsed() > budget_s * 0.08:
            break
        hc = slsqp_run(best_h, T, n, 700)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Subgradient with line search (28%)
    sg_end = budget_s * 0.36
    step_base = 0.02
    for it in range(8000):
        if elapsed() > sg_end:
            break
        grad = grad_c5_active(best_h, top_k=5)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad = grad / gn
        best_trial = best_c5
        best_trial_h = best_h
        for alpha in [0.002, 0.005, 0.01, 0.02, 0.035, 0.06]:
            h_t = project_feasible(best_h - alpha * grad * step_base / gn * 10, n)
            c5_t = compute_c5(h_t)
            if c5_t < best_trial - 1e-15:
                best_trial = c5_t
                best_trial_h = h_t
        if best_trial < best_c5:
            best_c5 = best_trial
            best_h = best_trial_h
            step_base = min(step_base * 1.02, 0.1)
        else:
            step_base = max(step_base * 0.92, 0.005)
        if it % 300 == 0 and elapsed() < sg_end:
            hc = slsqp_run(best_h, 500, n, 400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Binary pair-swap local search (15%)
    ps_end = budget_s * 0.52
    h_bin = np.round(best_h).astype(float)
    while h_bin.sum() > n / 2:
        idx = np.where(h_bin == 1)[0]
        h_bin[rng.choice(idx)] = 0
    while h_bin.sum() < n / 2:
        idx = np.where(h_bin == 0)[0]
        h_bin[rng.choice(idx)] = 1
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    improved = True
    while improved and elapsed() < ps_end:
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(ones)
        for i in ones:
            if elapsed() > ps_end:
                break
            zeros_shuffled = zeros.copy()
            rng.shuffle(zeros_shuffled)
            for j in zeros_shuffled[:25]:
                h_bin[i] = 0
                h_bin[j] = 1
                c5_new = compute_c5(h_bin)
                if c5_new < c5_bin - 1e-14:
                    c5_bin = c5_new
                    improved = True
                    if c5_bin < best_c5:
                        best_c5 = c5_bin
                        best_h = h_bin.copy()
                    zeros = np.where(h_bin == 0)[0]
                    ones = np.where(h_bin == 1)[0]
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0

    # Phase 4: SLSQP refinement from best (15%)
    p4_end = budget_s * 0.70
    for T in [300, 800, 2000, 5000]:
        if elapsed() > p4_end:
            break
        hc = slsqp_run(best_h, T, n, 900)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Try other resolutions (15%)
    p5_end = budget_s * 0.87
    for n_try in [54, 58, 62, 66, 72]:
        if elapsed() > p5_end:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [200, 1000]:
            if elapsed() > p5_end:
                break
            hc = slsqp_run(h_interp, T, n_try, 700)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                n = n_try

    # Phase 6: Nelder-Mead fine tune (10%)
    if elapsed() < budget_s * 0.97:
        hc = nm_run(best_h, n, 4000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final SLSQP
    if elapsed() < budget_s * 0.99:
        hc = slsqp_run(best_h, 3000, n, 500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
