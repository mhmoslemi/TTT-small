                          options={'maxiter': niter, 'ftol': 1e-15, 'gtol': 1e-11})
            h1 = normalize_h(np.clip(res.x, 0.0, 1.0))
            c5 = compute_c5(h1)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h1.copy()
        # Coordinate descent at n=720
        if time.time() - t0 < budget_s * 0.88:
            h1, c5 = coordinate_descent(h1, n_passes=2, step=0.008)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h1.copy()

    # Phase 5: Random restarts (small perturbations)
    if time.time() - t0 < budget_s * 0.92:
        n_f = 2 * n if time.time() - t0 > budget_s * 0.55 else n
        h_f = h1.copy() if n_f == 2 * n else h_cur.copy()
        b_f = [(0.0, 1.0)] * n_f
        for r in range(8):
            if time.time() - t0 > budget_s * 0.94:
                break
            sigma = 0.025 * np.exp(-0.35 * r)
            hp = np.clip(h_f + np.random.randn(n_f) * sigma, 0.0, 1.0)
            hp = normalize_h(hp)
            res = minimize(smooth_max_obj, hp, args=(0.002, 5000, 5, 80),
                          method='L-BFGS-B', bounds=b_f,
                          options={'maxiter': 80, 'ftol': 1e-14, 'gtol': 1e-10})
            hp = normalize_h(np.clip(res.x, 0.0, 1.0))
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()
            h_f = hp

    # Final validation
    best_h = normalize_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
