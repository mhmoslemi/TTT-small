        else:
            break
    return (lo + hi) / 2.0

def coord_descent_phase(h, N, target, dx, t0, budget_s, rng, n_sweeps):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    stuck_count = 0

    for sweep in range(n_sweeps):
        if time.time() - t0 > budget_s - 12:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(N)
        max_change = 0.0

        for i in order:
            a = compute_coeff(h, i, N)
            b = corr - a * h[i]
            new_val = find_optimal_x(a, b)
            delta = new_val - h[i]
            if abs(delta) > 1e-11:
                h[i] = new_val
                corr += delta * a
                if abs(delta) > max_change:
                    max_change = abs(delta)

        h = project_sum(h, target)

        if max_change < 1e-8:
            stuck_count += 1
        else:
            stuck_count = 0

        if sweep % 100 == 99:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        if stuck_count > 200 and sweep % 50 == 49:
            n_pert = int(rng.randint(1, 6))
            ip = rng.choice(N, size=n_pert, replace=False)
            h[ip] += rng.randn(n_pert) * 0.04
            h = project_sum(h, target)
            stuck_count = 0

    c5 = compute_c5(h, dx)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)

    # Phase 1: Coordinate descent from parent
    h1 = project_sum(h0.copy(), target0)
    h1, c5_1 = coord_descent_phase(h1, n0, target0, dx0, t0, budget_s, rng, 5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Noisy restarts from best
    for trial in range(4):
        if time.time() - t0 > budget_s - 40:
            break
        scale = 0.03 + 0.02 * trial
        h2 = best_h + rng.randn(n0) * scale
        h2 = project_sum(h2, target0)
        h2, c5_2 = coord_descent_phase(h2, n0, target0, dx0, t0, budget_s, rng, 2000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Try doubled resolution
    if time.time() - t0 < budget_s - 80:
        N2 = 2 * n0
        dx2 = 2.0 / N2
        target2 = N2 / 2.0
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2 = np.interp(x2, x0, best_h, period=2.0)
        h2 = project_sum(h2, target2)
        h2, c5_2 = coord_descent_phase(h2, N2, target2, dx2, t0, budget_s, rng, 2000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 4: Binary-like restart
    if time.time() - t0 < budget_s - 40:
        hB = np.zeros(n0)
        ones = rng.choice(n0, size=int(target0), replace=False)
        hB[ones] = 1.0
        hB, c5_B = coord_descent_phase(hB, n0, target0, dx0, t0, budget_s, rng, 3000)
        if c5_B < best_c5:
            best_c5 = c5_B
            best_h = hB

    # Phase 5: Final polish
    if time.time() - t0 < budget_s - 10:
        Nf = len(best_h)
        dxf = 2.0 / Nf
        tf = Nf / 2.0
        best_h, best_c5 = coord_descent_phase(best_h.copy(), Nf, tf, dxf, t0, budget_s, rng, 3000)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
