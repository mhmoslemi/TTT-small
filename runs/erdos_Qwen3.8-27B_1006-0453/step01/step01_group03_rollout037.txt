import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_simplex(h, target_sum):
    """Project h onto {x: 0 ≤ x ≤ 1, sum(x) = target_sum}."""
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        count = mask.sum()
        if count == 0:
            break
        h[mask] += d / count
    return h

def subgradient_obj(h, dx):
    """Compute C5 and subgradient direction (negative)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    max_val = c.max()
    
    # Subgradient of max: average over active constraints
    tol = 1e-8 * max_val + 1e-12
    active = c >= max_val - tol
    n_active = active.sum()
    
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        d = m - (n - 1)
        if d >= 0:
            # c_d = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            if d > 0:
                grad[:n-d] += (1.0 - h[d:])
                grad[d:] -= h[:n-d]
        else:
            # d < 0, m' = -d
            mp = -d
            # c = sum_{i=0}^{n-1-mp} h[i+mp]*(1-h[i])
            grad[mp:] += (1.0 - h[:n-mp])
            grad[:n-mp] -= h[mp:]
    
    grad /= n_active
    return max_val, grad * dx

def projected_subgrad(h0, dx, target_sum, n_iter=5000, deadline=None, verbose=False):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    c5, g = subgradient_obj(h, dx)
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Adaptive step size
        lr = 0.1 / (1.0 + 0.01 * it)
        # Subgradient descent: move in direction -g
        h_new = project_onto_simplex(h - lr * g, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            h = h_new
            c5, g = subgradient_obj(h, dx)
        else:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                h_new = project_onto_simplex(h - lr * shrink * g, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    h = h_new
                    break
            else:
                h = h_new  # still move to new point (for exploration)
                c5, g = subgradient_obj(h, dx)
    
    return best_h, best_c5

2. Solve the resulting LP (linear constraints, linear objective)
3. Update and repeat

        n2 = 63
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        x_old = (np.arange(n) + 0.5) * dx
        x_new = (np.arange(n2) + 0.5) * dx2
        f = interp1d(x_old, best_h, kind='cubic', bounds_error=False, fill_value='extrapolate')
        h2 = np.clip(f(x_new), 0.0, 1.0)
        h2 = project_h(h2, ts2)
        h2, c2 = slsqp_opt(h2, n2, ts2, 200)
        if c2 < best_c5 and time_ok(50):
            h2, c2 = local_search(h2, dx2, ts2, 2000, rng)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 7: Try even finer n=95
    if time_ok(180) and best_n < 63:
        n3 = 95
        dx3 = 2.0 / n3
        ts3 = n3 / 2.0
        x_a = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        x_b = (np.arange(n3) + 0.5) * dx3
        f2 = interp1d(x_a, best_h, kind='cubic', bounds_error=False, fill_value='extrapolate')
        h3 = np.clip(f2(x_b), 0.0, 1.0)
        h3 = project_h(h3, ts3)
        h3, c3 = slsqp_opt(h3, n3, ts3, 150)
        if c3 < best_c5:
            best_h, best_c5, best_n = h3, c3, n3

    # Final polish
    nf = len(best_h)
    if time_ok(20):
        best_h, best_c5 = slsqp_opt(best_h, nf, nf / 2.0, 100)
    if time_ok(15):
        dx_f = 2.0 / nf
        best_h, best_c5 = local_search(best_h, dx_f, nf / 2.0, 2000, rng)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
