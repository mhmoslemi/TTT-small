```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_grid(h_src, n_dst):
    n_src = len(h_src)
    xs = np.arange(n_src) * 2.0 / n_src
    xd = np.arange(n_dst) * 2.0 / n_dst
    return proj(np.interp(xd, xs, h_src), n_dst)

def obj_smooth(h, temp=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / temp) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def coord_descent(h, n, dx, rng, deadline):
    """Coordinate descent to minimize C5."""
    best = compute_c5(h)
    no_improve = 0
    while time.time() < deadline:
        old_c5 = best
        # Random coordinate order
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old_a = h[j]
            # Golden section / bounded minimization
            def f(a):
                h[j] = a
                corr = np.correlate(h, 1.0 - h, mode='full')
                return np.max(corr) * dx
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            h[j] = res.x
        h = proj(h, n)
        c5 = compute_c5(h)
        if c5 < best:
            best = c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb and restart
                h = proj(h + rng.randn(n) * 0.01, n)
                no_improve = 0
    return h, best

def slsqp_polish(h, n, t0, budget_s):
    """Polish with SLSQP."""
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    deadline = t0 + budget_s - 5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for temp in [50.0, 100.0, 200.0, 400.0]:
        if time.time() > deadline:
            break
        def obj(xx):
            return obj_smooth(xx, temp)
        try:
            r = minimize(obj, best_h, method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 2000, 'ftol': 1e-12})
            hc = proj(np.clip(r.x, 0.0, 1.0), n)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except:
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_grid(h_src, n_dst):
    n_src = len(h_src)
    xs = np.arange(n_src) * 2.0 / n_src
    xd = np.arange(n_dst) * 2.0 / n_dst
    return proj(np.interp(xd, xs, h_src), n_dst)

def obj_softmax(h, temp=200.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    s -= np.max(s)
    return (mx + np.log(np.sum(np.exp(s))) / temp) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def coord_descent(h, n, dx, rng, deadline, max_sweeps=100000):
    best = compute_c5(h)
    no_improve = 0
    sweeps = 0
    h_work = h.copy()
    while sweeps < max_sweeps and time.time() < deadline:
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            def f_obj(a):
                h_work[j] = a
                corr = np.correlate(h_work, 1.0 - h_work, mode='full')
                return np.max(corr) * dx
            try:
                res = minimize_scalar(f_obj, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-11})
                h_work[j] = res.x
            except Exception:
                pass
        h_work = proj(h_work, n)
        c5 = compute_c5(h_work)
        if c5 < best - 1e-14:
            best = c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                h_work = proj(h_work + rng.randn(n) * 0.008, n)
                no_improve = 0
        sweeps += 1
    return h_work, best

def slsqp_polish(h, n, t0, budget_s):
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    deadline = t0 + budget_s - 5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for temp in [80.0, 200.0, 500.0, 1200.0]:
        if time.time() > deadline:
            break
        def obj_fn(xx):
            return obj_softmax(xx, temp)
        try:
            r = minimize(obj_fn, best_h, method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 3000, 'ftol': 1e-14})
            hc = proj(np.clip(r.x, 0.0, 1.0), n)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass
    return best_h, best_c5
```
