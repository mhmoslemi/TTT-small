    return float(np.max(corr) * dx), corr

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        return float(m * dx)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    expTerms = np.exp(T * (corr - m))
    Z = np.sum(expTerms)
    
    # Compute sum_k expTerms[k] * d(corr_k)/d(h[i]) for each i
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = expTerms[k]
        if w < 1e-18:
            continue
        # d(corr_k)/d(h[i]) = (1-h[i+lag]) * 1[valid(i+lag)] - h[i-lag] * 1[valid(i-lag)]
        # Part 1: (1-h[i+lag]) for i where 0<=i+lag<n
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:

            grad[idx] += w * (1.0 - h[idx + lag])

        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= w * h[idx - lag]
    
    grad *= dx / Z
    return grad

    diff = target - s

        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)
