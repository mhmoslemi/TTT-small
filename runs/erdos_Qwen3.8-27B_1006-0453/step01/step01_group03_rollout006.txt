import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def logsumexp_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return m + np.log(np.exp(beta * (corr - m)).sum()) / beta

def grad_lse(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        lo1 = max(0, j)
        hi1 = min(2 * n - 2, j + n - 1)
        if lo1 <= hi1:
            kv = np.arange(lo1, hi1 + 1)
            iv = j - kv + n - 1
            grad[j] += np.dot(w[lo1:hi1 + 1], 1.0 - h[iv])
        lo2 = max(0, n - 1 - j)
        hi2 = min(2 * n - 2, 2 * n - 2 - j)
        if lo2 <= hi2:
            kv2 = np.arange(lo2, hi2 + 1)
            iv2 = j + kv2 - n + 1
            grad[j] -= np.dot(w[lo2:hi2 + 1], h[iv2])
    return grad * dx

def lbslb_smooth(h0, dx, target_sum, beta, maxiter=500, deadline=None):
    n = len(h0)
    def obj(x):
        hx = np.clip(x, 0, 1)
        return logsumexp_c5(hx, dx, beta)
    def jac(x):
        hx = np.clip(x, 0, 1)
        return grad_lse(hx, dx, beta)
    bnds = [(0, 1)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        if res.x is not None:
            return project_sr(res.x, target_sum)
    except Exception:
        pass
    return h0.copy()

def slsqp_solve(h0, dx, target_sum, maxiter=400):
    n = len(h0)
    t_init = compute_c5(h0, dx) / dx
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    x0 = np.concatenate([h0, [t_init + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def sa_lag_aware(h0, dx, target_sum, max_iter=6000, T_init=0.04, T_end=0.0003, deadline=None):

    cur_c5 = best_c5

    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:

        T *= cool
        if stall > 800 and stall % 400 == 0:

        stall += 1
        r = np.random.rand()

            k = np.random.randint(n)

            h_new[j] += delta
            h_new[k] -= delta

        elif r < 0.8:

            cur_c5 = c5_new

    best_c5 = compute_c5(h0, dx)
