from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    """Project x onto {0<=y<=1, sum(y)=n/2} using bisection."""
    target = n / 2.0
    x = x.astype(float)
    lo = x.min() - 1.5
    hi = x.max() + 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    diff = target - np.sum(y)
    if abs(diff) > 1e-10:
        mask = (y > 1e-6) & (y < 1.0 - 1e-6)
        if np.sum(mask) > 0:
            y[mask] += diff / np.sum(mask)
            np.clip(y, 0.0, 1.0, out=y)
    return y

def compute_subgrad(h, lag):
    """Subgradient of corr at given lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        g[:n+lag] += (1.0 - h[-lag:])
        g[-lag:] -= h[:n+lag]
    return g

def multi_subgrad(h, n_top=3):
    """Average subgradient from top n_top lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_top:]
    g = np.zeros(n)
    for m in top_idx:
        lag = m - (n - 1)
        g += compute_subgrad(h, lag)
    g /= n_top
    return g, np.max(corr)

    """Subgradient method with projection."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # Step size schedule
    g_norm_sqr = np.zeros(max_iter)
    
    for it in range(max_iter):
        if time.time() - t0 > total_budget * 0.95:
            break
        g, cmax = multi_subgrad(h, n_top)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-12:
            break
        alpha = 1.0 / np.sqrt(it + 1)

        h_new = project_feasible(h_new, n)

def smooth_c5(h, temp=0.05):
    """Smooth approximation of C5."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))

        order = np.argsort(-h[idx])[:target - num_ones]

def project_feasible(x, n):
    target = float(n) / 2.0
    x = np.asarray(x, dtype=float)
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        mask = (y > 1e-8) & (y < 1.0 - 1e-8)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            np.clip(y, 0.0, 1.0, out=y)
    return y

        hb[ix[np.argsort(-h[ix])[:tg - k]]] = 1
    return hb

def bsearch(h, t0, bdx, mi):
    n = len(h)
    hb = bround(h)
    bh, bc = hb.copy(), compute_c5(hb)
    for _ in range(mi):
        if time.time() - t0 > bdx * 0.93:
            break
        i = np.random.randint(n)
        hb[i] = 1.0 - hb[i]
        v = compute_c5(hb)
        if v < bc - 1e-14:
            bc, bh = v, hb.copy()
        else:
            hb[i] = 1.0 - hb[i]
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h0 = np.array(initial_h_values, dtype=float).copy()
    h0 = project_feasible(h0, len(h0))
    bh, bc = h0.copy(), compute_c5(h0)

    # Phase 1: subgradient at initial resolution
    ch, v = subgrad_run(h0, t0, budget_s, 2500, 5)
    if v < bc:
        bc, bh = v, ch.copy()
    ch = bh.copy()

    # Alternate: try random binary start
    if time.time() - t0 < budget_s * 0.35:
        n0 = len(h0)
        idx = np.random.permutation(n0)
        hr = np.zeros(n0)
        hr[idx[:n0 // 2]] = 1.0
        hr, v = subgrad_run(hr, t0, budget_s, 1000, 5)
        if v < bc:
            bc, bh = v, hr.copy()
        ch = bh.copy()

    # Phase 2: L-BFGS polish at initial resolution
    if time.time() - t0 < budget_s * 0.42:
        ch, v = lfgs(ch, t0, budget_s, 400)
        if v < bc:
            bc, bh = v, ch.copy()

    # Phase 3: upscale to 2x
    if time.time() - t0 < budget_s * 0.60:
        ch = project_feasible(upscale(bh, 2), len(bh) * 2)
        ch, v = subgrad_run(ch, t0, budget_s, 1500, 5)
        if v < bc:
            bc, bh = v, ch.copy()
        if time.time() - t0 < budget_s * 0.58:
            ch, v = lfgs(ch, t0, budget_s, 200)
            if v < bc:
                bc, bh = v, ch.copy()

    # Phase 4: upscale to 4x
    if time.time() - t0 < budget_s * 0.75:
        ch = project_feasible(upscale(bh, 2), len(bh) * 2)
        ch, v = subgrad_run(ch, t0, budget_s, 1000, 5)
        if v < bc:
            bc, bh = v, ch.copy()
        if time.time() - t0 < budget_s * 0.73:
            ch, v = lfgs(ch, t0, budget_s, 150)
            if v < bc:
                bc, bh = v, ch.copy()

    # Phase 5: binary local search
    if time.time() - t0 < budget_s * 0.80:
        bb, bv = bsearch(bh, t0, budget_s, 40000)
        if bv < bc:
            bc, bh = bv, bb.copy()

    # Phase 6: continuous polish around best solution
    if time.time() - t0 < budget_s * 0.90:
        hs = bh + np.random.randn(len(bh)) * 0.08
        hs = project_feasible(hs, len(hs))
        hs, v = lfgs(hs, t0, budget_s, 300)
        if v < bc:
            bc, bh = v, hs.copy()

    # Final projection and evaluation
    bh = project_feasible(bh, len(bh))
    fc = compute_c5(bh)
    if fc < bc:
        bc = fc
    return bh, bc, len(bh)
