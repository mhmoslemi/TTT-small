        h = project_sum(h, target)
        if it % 400 == 0 and it > 1000:
            h += 0.01 * (2.0 * h - 1.0) * (1.0 - 4.0 * h * (1.0 - h))
            h = project_sum(h, target)
        if it > 0 and it % 1200 == 0:
            lr *= 0.8
    return best_h, best_c

def binary_local_search(h, N, target, dx, t0, budget, rng, max_rounds=4):
    hb = (h > 0.5).astype(float)
    hb = project_sum(hb, target)
    bc = compute_c5(hb, dx)
    for rnd in range(max_rounds):
        if time.time() - t0 > budget - 15:
            break
        improved = False
        for j in range(N):
            if time.time() - t0 > budget - 15:
                break
            h_trial = hb.copy()
            h_trial[j] = 1.0 - h_trial[j]
            s = float(h_trial.sum())
            if abs(s - target) > 1e-9:
                continue
            c = compute_c5(h_trial, dx)
            if c < bc - 1e-14:
                hb = h_trial
                bc = c
                improved = True
        if not improved:
            break
    return hb, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, dx0)
    x0 = np.linspace(0, 2.0, n0, endpoint=False)

    # Phase 1: Direct optimization at native N
    h1, c1 = optimize_adagre(h0.copy(), n0, n0 / 2.0, dx0, t0, budget_s, rng,
                             n_iter=4000, top_k=3, lr0=0.04)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Binary local search on best
    if time.time() - t0 < budget_s - 60:
        hb, cb = binary_local_search(best_h, n0, n0 / 2.0, dx0, t0, budget_s, rng)
        if cb < best_c:
            best_c, best_h = cb, hb

    # Phase 3: Try N=380 from interpolated parent
    if time.time() - t0 < budget_s - 80:
        N2 = 2 * n0
        dx2 = 2.0 / N2
        x2 = np.linspace(0, 2.0, N2, endpoint=False)
        h_interp = np.interp(x2, x0, h0)
        h_interp = project_sum(h_interp, N2 / 2.0)
        h2, c2 = optimize_adagre(h_interp, N2, N2 / 2.0, dx2, t0, budget_s, rng,
                                 n_iter=4000, top_k=3, lr0=0.03)
        if c2 < best_c:
            best_c, best_h = c2, h2
        if time.time() - t0 < budget_s - 60:
            hb2, cb2 = binary_local_search(best_h, len(best_h), len(best_h) / 2.0,
                                           2.0 / len(best_h), t0, budget_s, rng)
            if cb2 < best_c:
                best_c, best_h = cb2, hb2

    # Phase 4: Perturbed restarts at native N
    if time.time() - t0 < budget_s - 50:
        for scale in [0.06, 0.12]:
            if time.time() - t0 > budget_s - 30:
                break
            hd = best_h.copy() + rng.randn(n0) * scale
            hd = project_sum(hd, n0 / 2.0)
            hh, cc = optimize_adagre(hd, n0, n0 / 2.0, dx0, t0, budget_s, rng,
                                     n_iter=3000, top_k=3, lr0=0.035)
            if cc < best_c:
                best_c, best_h = cc, hh

    # Phase 5: Final polish with smaller steps
    if time.time() - t0 < budget_s - 15:
        Nb = len(best_h)
        hh, cc = optimize_adagre(best_h.copy(), Nb, Nb / 2.0, 2.0 / Nb, t0, budget_s,
                                 rng, n_iter=2000, top_k=3, lr0=0.015)
        if cc < best_c:
            best_c, best_h = cc, hh
        if time.time() - t0 < budget_s - 10:
            hb, cb = binary_local_search(best_h, Nb, Nb / 2.0, 2.0 / Nb, t0, budget_s, rng)
            if cb < best_c:
                best_c, best_h = cb, hb

    final_c = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c, len(best_h)
