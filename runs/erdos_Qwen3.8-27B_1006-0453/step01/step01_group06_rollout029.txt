    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:

        h = h + room * (diff / total)
    else:

        if total < 1e-14:
            return h
        h = h * (target / total)
    h = np.clip(h, 0.0, 1.0)
    # Fix sum exactly if needed
    s = h.sum()
    if abs(s - target) > 1e-12:
        # Distribute remaining difference to elements with room
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 1e-14:
                h += room * (diff / total)
        else:
            total = h.sum()
            if total > 1e-14:
                h += h * (diff / total)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    return x.sum() - len(x) / 2.0

def slsqp_optimize(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(smooth_c5, h, args=(T,),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    return res.x

    """Update corr when h[i] changes to new_val, return new corr vector"""
    corr = get_corr(h)  # O(n²) - do once per pass
    delta = new_val - h[i]
    if abs(delta) < 1e-14:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h = h * (target / total)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    if abs(s2 - target) > 1e-10:
        d2 = target - s2
        if d2 > 0:
            room = 1.0 - h
            t = room.sum()
            if t > 1e-14:
                h += room * (d2 / t)
        else:
            t = h.sum()
            if t > 1e-14:
                h += h * (d2 / t)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(smooth_c5, h, args=(T,),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
    return res.x

def coord_descent(h, n, n_passes=15):
    for _ in range(n_passes):
        improved = False
        corr = get_corr(h)
        for i in range(n):
            best_val = np.max(corr)
            best_v = h[i]
            old = h[i]
            candidates = np.unique(np.concatenate([[0.0, 1.0, 0.5],
                                                   [old + d for d in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]],
                                                   [max(0, min(1, old + d)) for d in np.linspace(-0.4, 0.4, 9)]]))
            candidates = candidates[(candidates >= 0) & (candidates <= 1)]
            for v in candidates:
                if abs(v - old) < 1e-6:

                c = get_corr(h_p)

                    corr = c

            h[i] = best_v
