    if T < 0.01:

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        for trial in range(4):
            if elapsed() > budget_s * 0.89:
                break
            h0 = h_interp + np.random.uniform(-0.06, 0.06, n_try)
            h0 = project_feasible(h0, n_try)
            for T in [80, 350, 1200]:
                if elapsed() > budget_s * 0.89:
                    break
                try:
                    hc = slsqp_optimize(h0, T, n_try, maxiter=600, ftol=1e-14)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    # Phase 5: Local search - targeted coordinate adjustments
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.96:
        for iteration in range(300):
            if elapsed() > budget_s * 0.96:
                break
            improved = False
            # Find worst correlation shift
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            k_max = np.argmax(corr)
            # Try small adjustments
            for j in np.random.choice(n_cur, size=min(20, n_cur), replace=False):
                for delta in [0.03, 0.06, 0.1, -0.03, -0.06, -0.1]:
                    h_try = best_h.copy()
                    h_try[j] += delta
                    h_try = project_feasible(h_try, n_cur)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved and iteration > 50:
                break

    # Phase 6: Final high-T SLSQP polish
    n_cur = len(best_h)
    for T in [3000, 8000, 20000]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_optimize(best_h, T, n_cur, maxiter=1000, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
