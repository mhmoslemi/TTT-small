import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = temp * corr
    m = vals[0]
    for i in range(1, len(vals)):
        if vals[i] > m:
            m = vals[i]
    e = np.exp(vals - m)
    return (np.log(np.sum(e)) + m) / temp * dx

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            mask = (h < 1.0) if diff > 0 else (h > 0.0)
            nm = int(np.sum(mask))
            if nm == 0:
                break
            h[mask] += diff / nm
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        for i in range(n):
            if diff > 0.0 and h[i] < 1.0 - abs(diff):
                h[i] += diff
                break
            if diff < 0.0 and h[i] > abs(diff):
                h[i] += diff
                break
    return h

def slsqp_opt(h0, n, temp, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                   args=(n, temp), options={'maxiter': maxiter, 'ftol': 1e-10})
    return res.x

def eval_pair(h, ia, ib, t, n):
    c = h[ia] + h[ib]
    ht = h.copy()
    ht[ia] = t
    ht[ib] = c - t
    dx = 2.0 / n
    corr = np.correlate(ht, 1.0 - ht, mode='full')
    return float(np.max(corr) * dx)

def opt_pair(h, ia, ib, n, niter=40):
    c = h[ia] + h[ib]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-12:
        return h, False
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if eval_pair(h, ia, ib, m1, n) <= eval_pair(h, ia, ib, m2, n):
            hi = m2
        else:
            lo = m1
    bt = (lo + hi) / 2.0
    old_c5 = compute_c5(h, n)
    ht = h.copy()
    ht[ia] = bt
    ht[ib] = c - bt
    new_c5 = compute_c5(ht, n)
    if new_c5 < old_c5 - 1e-15:
        return ht, True
    return h, False

def coord_desc(h, n, nsweeps, seed, pair_iters):
    rng = np.random.RandomState(seed)
    h = h.copy()
    for _ in range(nsweeps):
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for a in range(n):
            ia = order[a]
            for b in range(a + 1, n):
                ib = order[b]
                h, did = opt_pair(h, ia, ib, n, pair_iters)
                if did:
                    improved = True
        if not improved:
            break
    h = repair_h(h, n)
    return h, compute_c5(h, n)

def make_structured(n, period, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) == 0 else 0.0
    h = repair_h(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    tlim = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    for p in [2, 3, 4, 5, 7]:
        if tlim - time.time() < 300:
            break
        ss = make_structured(n, p, p)
        ss_t = slsqp_opt(ss, n, 30, maxiter=150)
        ss_t = repair_h(ss_t, n)
        ss_t, ss_c5 = coord_desc(ss_t, n, 2, p * 100, 30)
        if ss_c5 < best_c5:
            best_c5 = ss_c5
            best_h = ss_t.copy()

    # Phase 1: SLSQP annealing
    for temp in [80, 30, 12, 6, 3, 2, 1.2]:
        if tlim - time.time() < 250:
            break
        h_t = slsqp_opt(best_h, n, temp, maxiter=250)
        h_t = repair_h(h_t, n)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
        best_h, best_c5 = coord_desc(best_h, n, 2, 777 + int(temp), 35)

    # Phase 2: Perturbation search
    for trial in range(200):
        if tlim - time.time() < 100:
            break
        scale = 0.02 + 0.10 * np.exp(-trial / 50.0)
        cand = best_h + np.random.uniform(-scale, scale, n)
        cand = repair_h(cand, n)
        h_t = slsqp_opt(cand, n, 10, maxiter=80)
        h_t = repair_h(h_t, n)
        h_t, c5_t = coord_desc(h_t, n, 1, trial + 500, 30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 3: Deep coordinate descent
    if tlim - time.time() > 30:
        best_h, best_c5 = coord_desc(best_h, n, 15, 4242, 50)

    # Phase 4: Final SLSQP
    if tlim - time.time() > 15:
        h_t = slsqp_opt(best_h, n, 1.0, maxiter=400)
        h_t = repair_h(h_t, n)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 5: Final coord descent
    if tlim - time.time() > 10:

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = temp * corr
    m = np.max(vals)
    return (np.log(np.sum(np.exp(vals - m))) + m) / temp * dx


def sum_con(x, n):
    return np.sum(x) - n / 2.0


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            mask = (h < 1.0) if diff > 0 else (h > 0.0)
            nm = int(np.sum(mask))
            if nm == 0:
                break
            h[mask] += diff / nm
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        for i in range(n):
            if diff > 0.0 and h[i] < 1.0 - abs(diff):
                h[i] += diff
                break
            if diff < 0.0 and h[i] > abs(diff):
                h[i] += diff
                break
    return h

                   constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                   args=(n, temp), options={'maxiter': maxiter, 'ftol': 1e-10})

    ht[ia] = t
