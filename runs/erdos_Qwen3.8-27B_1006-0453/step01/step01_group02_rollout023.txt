        if time.time() - t0 > budget_s - 80:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_interp = project_sum(np.interp(xN, x0, h0), target)
        st = h_interp.copy()
        if N != n0:
            st2 = h_interp + rng.randn(N) * 0.04
            st2 = project_sum(st2, target)
        else:
            st2 = project_sum(h_interp + rng.randn(N) * 0.03, target)
        for hs in [st, st2]:
            if time.time() - t0 > budget_s - 40:
                break
            ho, c5o = _optimize(hs, N, dx, target, t0, budget_s, rng)
            if c5o < best_c5:
                best_c5, best_h = c5o, ho.copy()

    Nb, dxb = len(best_h), 2.0 / len(best_h)
    if time.time() - t0 < budget_s - 30:
        hp, cp = _optimize(best_h.copy(), Nb, dxb, Nb / 2.0, t0, budget_s, rng)
        if cp < best_c5:
            best_c5, best_h = cp, hp
    return best_h, compute_c5(best_h, 2.0 / len(best_h)), len(best_h)
