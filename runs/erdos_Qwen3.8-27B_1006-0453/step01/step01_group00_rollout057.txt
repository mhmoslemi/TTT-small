def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    c = np.max(vals)
    if T < 1e-10:
        return c
    s = np.sum(np.exp((vals - c) / T))
    return c + T * np.log(s)

def fix_sum(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
