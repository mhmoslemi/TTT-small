import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx


def active_c5(h, T, active_idx):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = corr[active_idx]
    m = np.max(a)
    val = m + np.log(np.sum(np.exp(T * (a - m)))) / T
    return val * dx


def sum_constraint(h):
    return h.sum() - len(h) / 2.0


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)


def get_active(h, frac):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return np.where(corr >= m * (1.0 - frac))[0]
