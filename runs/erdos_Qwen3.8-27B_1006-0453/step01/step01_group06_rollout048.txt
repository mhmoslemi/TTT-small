For lag ≥ 0:
- 0≤m+lag<n → 0≤m<n-lag
- 0≤m-lag<n → lag≤m<n (i.e., m≥lag)

So for lag ≥ 0:
∂corr[k]/∂h[m] = (1-h[m+lag])·1[m<n-lag] - h[m-lag]·1[m≥lag]

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_box_sum(p, S):
    n = len(p)
    lo, hi = p.min() - 1.5, p.max() + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(p - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(p - 0.5 * (lo + hi), 0, 1)
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def smooth_c5(h, T):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    w = exp_t / exp_t.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[:n - lag] += 1.0 - h[lag:]
            g[lag:] += -h[:n - lag]
        else:
            L = -lag
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] += -h[L:]
        grad += w[k] * g
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Projected subgradient descent with softmax smoothing
    T_start, T_end = 20.0, 2000.0
    n_iter = 15000
    iter_budget = budget_s * 0.35
    for it in range(n_iter):
        if elapsed() > t0 + iter_budget:
            break
        T = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        g = grad_smooth_c5(best_h, T)
        # Polyak-like step
        f_cur = smooth_c5(best_h, T)
        if best_c5 > 0:
            step = min(0.5, (f_cur - 0.35) / (np.linalg.norm(g) ** 2 + 1e-10))
        else:
            step = 0.1 / (1.0 + it * 0.001)
        step = max(step, 0.01)
        c = best_h - step * g
        best_h = project_box_sum(c, n / 2.0)
        c5_val = compute_c5(best_h)
        if c5_val < best_c5:
            best_c5 = c5_val

    # Phase 2: SLSQP polish with high-T smoothing
    time_mark = budget_s * 0.55
    for T in [500, 2000, 8000]:
        if elapsed() > t0 + time_mark:
            break
        try:
            res = minimize(
                smooth_c5, best_h, args=(T,),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 500, 'ftol': 1e-14}
            )
            hc = project_box_sum(np.clip(res.x, 0, 1), n / 2.0)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 3: Pair-swap coordinate descent
    time_mark = budget_s * 0.82
    n_sweeps = 12
    for sweep in range(n_sweeps):
        if elapsed() > t0 + time_mark:
            break
        improved = False
        for i in range(n):
            if elapsed() > t0 + time_mark:
                break
            for j in range(i + 1, n):
                # Try transferring delta from j to i
                max_d = min(1.0 - best_h[i], best_h[j])
                if max_d < 1e-8:
                    continue
                # Binary search for optimal delta
                lo, hi = 0.0, max_d
                for _ in range(30):
                    mid = 0.5 * (lo + hi)
                    h_trial = best_h.copy()
                    h_trial[i] += mid
                    h_trial[j] -= mid
                    c5_trial = compute_c5(h_trial)
                    # Also check negative direction handled by swap
                    if c5_trial < compute_c5(best_h):
                        lo = mid
                    else:
                        hi = mid
                d_opt = 0.5 * (lo + hi)
                if d_opt > 1e-6:
                    best_h[i] += d_opt
                    best_h[j] -= d_opt
                    best_h = project_box_sum(best_h, n / 2.0)
                    c5_new = compute_c5(best_h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        improved = True
                # Try reverse direction
                max_d2 = min(1.0 - best_h[j], best_h[i])
                if max_d2 < 1e-8:
                    continue
                lo, hi = 0.0, max_d2
                for _ in range(30):
                    mid = 0.5 * (lo + hi)
                    h_trial = best_h.copy()
                    h_trial[j] += mid
                    h_trial[i] -= mid
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < compute_c5(best_h):
                        lo = mid
                    else:
                        hi = mid
                d_opt = 0.5 * (lo + hi)
                if d_opt > 1e-6:
                    best_h[j] += d_opt
                    best_h[i] -= d_opt
                    best_h = project_box_sum(best_h, n / 2.0)
                    c5_new = compute_c5(best_h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        improved = True
        if not improved and sweep > 2:
            break

    # Phase 4: Try n=46 from interpolation of best
    time_mark = budget_s * 0.93
    if elapsed() < t0 + time_mark:
        n2 = 46
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False),
                       best_h)
        h2 = project_box_sum(h2, n2 / 2.0)
        c5_2 = compute_c5(h2)
        for T in [200, 1000, 5000]:
            if elapsed() > t0 + time_mark:
                break
            try:
                res = minimize(
                    smooth_c5, h2, args=(T,),
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * n2,
                    constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n2 / 2.0,
                                  'jac': lambda x: np.ones(n2)}],
                    options={'maxiter': 400, 'ftol': 1e-14}
                )
                hc2 = project_box_sum(np.clip(res.x, 0, 1), n2 / 2.0)
                c5v2 = compute_c5(hc2)
                if c5v2 < c5_2:
                    c5_2 = c5v2
                    h2 = hc2
            except Exception:
                pass
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2

    # Phase 5: Nelder-Mead fine-tune on small neighborhood
    if elapsed() < t0 + budget_s * 0.98:
        n_cur = len(best_h)
        for attempt in range(5):
            if elapsed() > t0 + budget_s * 0.98:
                break
            scale = 0.05 * (1.0 - attempt * 0.15)
            h0 = best_h + np.random.uniform(-scale, scale, n_cur)
            h0 = project_box_sum(h0, n_cur / 2.0)
            try:
                res = minimize(
                    lambda x: smooth_c5(x, 3000.0),
                    h0, method='Nelder-Mead',
                    options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12}
                )
                hc = project_box_sum(np.clip(res.x, 0, 1), n_cur / 2.0)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Final check
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
