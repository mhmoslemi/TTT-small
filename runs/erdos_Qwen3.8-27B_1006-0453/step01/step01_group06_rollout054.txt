import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-13:
            h = h + room * (diff / total)
    else:
        total = h.sum()
        if total > 1e-13:
            h = h - h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, n_iters=50, n_grid=20):
    """Coordinate descent with grid search for each coordinate"""
    dx = 2.0 / n
    h = h.copy()
    for _ in range(n_iters):
        # Random order
        order = np.random.permutation(n)
        for i in order:
            best_t = h[i]
            best_val = compute_c5(h)
            # Try grid of values
            for t in np.linspace(0, 1, n_grid):
                if abs(t - h[i]) < 1e-6:

                h[i] = t

                val = compute_c5(h)

                    best_t = t
            h[i] = best_t

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
