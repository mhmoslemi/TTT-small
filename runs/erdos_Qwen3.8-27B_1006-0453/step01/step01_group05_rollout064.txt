import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(free.sum())
        if nf == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

def lse_obj(x, n, temp):
    corr = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(temp*(corr-m))))/temp

def eq_con(x, n):
    return np.sum(x) - n/2.0

def powell_obj(x, n):
    h = repair_h(x, n)
    return compute_c5(h, n)

def subgrad_step(h, n, beta, step):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    m = corr.max()
    w = np.exp(beta*(corr-m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2*n-1):
        wk = w[idx]
        if wk < 1e-14: continue
        k = idx - (n-1)
        if k >= 0:
            grad[k:] += wk*dx*(1.0-h[:n-k])
            grad[:n-k] -= wk*dx*h[k:]
        else:
            m2 = -k
            grad[:n-m2] += wk*dx*(1.0-h[m2:])
            grad[m2:] -= wk*dx*h[:n-m2]
    return h - step*grad, m

    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)]*n

    # Phase 1: SLSQP multi-temperature
