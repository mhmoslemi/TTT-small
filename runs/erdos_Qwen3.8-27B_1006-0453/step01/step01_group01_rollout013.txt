        h70_best = h70.copy()
        c70_best = compute_c5(h70)

        for ot, p1, mi in [('softmax', 80.0, 350), ('pnorm', 8.0, 350),
                           ('topm', 6.0, 350), ('softmax', 150.0, 350)]:
            if time.time() > T_LIMIT[0] - 250:
                break
            hc, c5 = do_slsqp(h70_best, 70, bnds70, cons70, ot, p1, mi, 1e-12)
            if hc is not None and c5 < c70_best:
                c70_best = c5
                h70_best = hc
            h_up = xinterp(h70_best, n)
            c_up = compute_c5(h_up)
            if c_up < bc:
                bc = c_up
                bh = h_up.copy()

    # Phase 2: Fine grid SLSQP - softmax with increasing temperature
    for t_val in [150.0, 300.0, 600.0, 1000.0]:
        if time.time() > T_LIMIT[0] - 80:
            break
        hc, c5 = do_slsqp(bh, n, bnds, cons, 'softmax', t_val, 2000, 1e-14)
        if hc is not None and c5 < bc:
            bc = c5
            bh = hc.copy()

    # Phase 3: Fine grid SLSQP - p-norm objectives
    for p_val in [6.0, 12.0, 24.0, 48.0]:
        if time.time() > T_LIMIT[0] - 80:
            break
        hc, c5 = do_slsqp(bh, n, bnds, cons, 'pnorm', p_val, 1200, 1e-13)
        if hc is not None and c5 < bc:
            bc = c5
            bh = hc.copy()

    # Phase 4: Fine grid SLSQP - top-m objectives
    for m_val in [4.0, 8.0, 12.0]:
        if time.time() > T_LIMIT[0] - 60:
            break
        hc, c5 = do_slsqp(bh, n, bnds, cons, 'topm', m_val, 1200, 1e-13)
        if hc is not None and c5 < bc:
            bc = c5
            bh = hc.copy()

    # Phase 5: Random restarts at fine grid
    n_restarts = 8
    for i in range(n_restarts):
        if time.time() > T_LIMIT[0] - 40:
            break
        scale = 0.005 + 0.01 * (i % 3)
        h_pert = proj(bh + rng.randn(n) * scale, n)
        hc, c5 = do_slsqp(h_pert, n, bnds, cons, 'softmax', 400.0, 800, 1e-13)
        if hc is not None and c5 < bc:
            bc = c5
            bh = hc.copy()

    # Phase 6: Coordinate descent polishing
    if time.time() < T_LIMIT[0] - 15:
        bh, bc = coord_descent(bh, n, max_rounds=15)

    # Phase 7: Subgradient polishing
    if time.time() < T_LIMIT[0] - 5:
        bh, bc = subgrad_polish(bh, n, n_iter=100)

    # Final: additional SLSQP polish if time remains
    if time.time() < T_LIMIT[0] - 5:
        hc, c5 = do_slsqp(bh, n, bnds, cons, 'softmax', 800.0, 2000, 1e-15)
        if hc is not None and c5 < bc:
            bc = c5
            bh = hc.copy()
            bh, bc = coord_descent(bh, n, max_rounds=5)

    bh = proj(bh, n)
    bc = compute_c5(bh)
    return bh, bc, n
