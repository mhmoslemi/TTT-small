            mi = min(300, max(60, int(rem * 3)))
            h_t = do_slsqp(h180, n180, bounds180, 20, mi)
            c5_t = compute_c5(h_t, n180)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t
                best_n = n180
                rem2 = budget_s - 30 - (time.time() - t0)
                if rem2 > 5:
                    h_t2 = do_nm(best_h, n180, min(1000, max(100, int(rem2 * 5))), 0.02)
                    c5_t2 = compute_c5(h_t2, n180)
                    if c5_t2 < best_c5:
                        best_c5 = c5_t2
                        best_h = h_t2

    # Phase 6: Final aggressive polish on best solution
    if time.time() - t0 < budget_s - 5:
        bn = best_n
        bnds = [(0.0, 1.0)] * bn
        rem = budget_s - 5 - (time.time() - t0)
        mi = max(50, int(rem * 10))
        h_t = do_slsqp(best_h, bn, bnds, 1, mi)
        c5_t = compute_c5(h_t, bn)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
        if time.time() - t0 < budget_s - 2:
            h_t2 = do_nm(best_h, bn, 500, 0.01)
            c5_t2 = compute_c5(h_t2, bn)
            if c5_t2 < best_c5:
                best_c5 = c5_t2
                best_h = h_t2
        # One more SLSQP at very low temp
        if time.time() - t0 < budget_s - 1:
            h_t3 = do_slsqp(best_h, bn, bnds, 0.3, 200)
            c5_t3 = compute_c5(h_t3, bn)
            if c5_t3 < best_c5:
                best_c5 = c5_t3
                best_h = h_t3

    best_h = repair(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
