                h[i] = nv
                hf = fix_sum(h, n)
                c5_n = compute_c5(hf)
                if c5_n < best_c5:
                    best_c5 = c5_n
                    best_val = nv
            h = fix_sum(h, n)
            h[i] = best_val
            if best_val != old_val:
                h = fix_sum(h, n)
                c5 = best_c5
                improved = True
        if not improved:
            break
    return h, c5

def pair_adjust(h, n, lag, t_start, time_limit):
    c5 = compute_c5(h)
    jj = abs(lag)
    deltas = np.array([0.015, 0.008, -0.015, -0.008, 0.004, -0.004])
    while time.time() - t_start < time_limit:
        improved = False
        for i in range(n - jj):
            if time.time() - t_start > time_limit:
                return h, c5
            for d in deltas:
                v1 = h[i] - d
                v2 = h[i + jj] + d
                if v1 < 0 or v1 > 1 or v2 < 0 or v2 > 1:
                    continue
                h[i] = v1
                h[i + jj] = v2
                c5_n = compute_c5(h)
                if c5_n < c5 - 1e-12:
                    c5 = c5_n
                    improved = True
                else:
                    h[i] = h[i] + d
                    h[i + jj] = h[i + jj] - d
                    break
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Continuation SLSQP with soft-max
    T_list = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0005]
    for T in T_list:
        if time.time() - t_start > budget_s - 400:
            break
        h_cand = slsqp_h(h_best.copy(), n_best, T, maxiter=1500)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand

    # Phase 2: Alternating single-lag minimization
    h_amm = alt_min_max(h_best.copy(), n_best, 30, t_start, budget_s - 300)
    c5_amm = compute_c5(h_amm)
    if c5_amm < c5_best:
        h_best = h_amm
        c5_best = c5_amm

    # Phase 3: Coordinate descent (fine)
    time_left = budget_s - (time.time() - t_start) - 120
    if time_left > 30:
        h_best, c5_best = coord_descent(h_best, n_best, t_start, time.time() + time_left)

    # Phase 4: Pair adjustment on critical lag
    if time.time() - t_start < budget_s - 60:
        corr = compute_all_corr(h_best)
        j_star = int(np.argmax(corr))
        lag_critical = j_star - (n_best - 1)
        time_left = max(0, budget_s - (time.time() - t_start) - 30)
        if time_left > 10:
            h_best, c5_best = pair_adjust(h_best, n_best, lag_critical, t_start, time.time() + time_left)

    # Phase 5: Another round of alternating min-max + coord descent
    if time.time() - t_start < budget_s - 80:
        h_amm = alt_min_max(h_best.copy(), n_best, 15, t_start, budget_s - 60)
        c5_amm = compute_c5(h_amm)
        if c5_amm < c5_best:
            h_best = h_amm
            c5_best = c5_amm
        time_left = max(0, budget_s - (time.time() - t_start) - 20)
        if time_left > 5:
            h_best, c5_best = coord_descent(h_best, n_best, t_start, time.time() + time_left)

    # Phase 6: Quick SLSQP polish
    if time.time() - t_start < budget_s - 20:
        h_cand = slsqp_h(h_best.copy(), n_best, 0.0005, maxiter=500)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, n_best)
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    return (h_best, c5_best, n_best)
