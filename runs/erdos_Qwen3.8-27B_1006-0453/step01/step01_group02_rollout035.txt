    h_r, c5_r = optimize_adam(h_p, n0, target0, dx0, t0, d3, lr=0.02, tau=40.0, niter=12000)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r

    # Phase 4: Second perturb from best (different direction)
    h_p2 = best_h - rng.randn(n0) * 0.015
    h_p2 = project_sum(h_p2, target0)
    d4 = t0 + budget_s * 0.74
    h_r, c5_r = optimize_adam(h_p2, n0, target0, dx0, t0, d4, lr=0.015, tau=50.0, niter=10000)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r

    # Phase 5: Upsample to 2N, optimize, downsample
    if time.time() < t0 + budget_s * 0.86:
        n2 = 2 * n0
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x2 = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x2, x0, best_h)
        h_up = project_sum(h_up, target2)
        d5 = t0 + budget_s * 0.86
        h_uo, _ = optimize_adam(h_up, n2, target2, dx2, t0, d5, lr=0.02, tau=45.0, niter=8000)
        h_dn = np.interp(x0, x2, h_uo)
        h_dn = project_sum(h_dn, target0)
        c5_dn = compute_c5(h_dn, dx0)
        if c5_dn < best_c5:
            best_c5, best_h = c5_dn, h_dn

    # Phase 6: Local swap search
    if time.time() < t0 + budget_s - 40:
        d6 = t0 + budget_s - 40
        best_h, best_c5 = local_swap_search(best_h, n0, target0, dx0, t0, d6, rng, niter=60000)

    # Phase 7: Final fine polish (very high tau, small lr)
    if time.time() < t0 + budget_s - 5:
        d7 = t0 + budget_s - 5
        h_f, c5_f = optimize_adam(best_h.copy(), n0, target0, dx0, t0, d7, lr=0.008, tau=80.0, niter=8000)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    # Phase 8: Tiny final local search
    if time.time() < t0 + budget_s - 2:
        best_h, best_c5 = local_swap_search(best_h, n0, target0, dx0, t0, t0 + budget_s - 2, rng, niter=20000)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
