    return np.sum(x) - n * 0.5

def sum_grad(x, n):
    return np.ones(n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-7) & (h < 1.0 - 1e-7)
    ni = np.sum(interior)
    if ni > 0:
        h[interior] += diff / ni
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)
        else:
            idx = np.argsort(h)[-1::-1]
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
            else:
                a = max(diff, -(h[i]))
            h[i] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

def coord_descent(h, n, n_sweeps=3, lr=0.006):
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        any_improve = False
        for j in range(n):
            old = h[j]
            for delta in [lr, -lr, 2*lr, -2*lr, 0.4*lr, -0.4*lr, 0.15*lr, -0.15*lr]:
                nv = old + delta
                if nv < 0 or nv > 1:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = project(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    any_improve = True
                    break
        if not any_improve:
            break
    return h, best_c5

def remaining_time(t0, budget):
    return budget - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n = len(h_best)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_func, 'jac': sum_grad, 'args': (n,)}]

    # Phase 1: SLSQP T-sweep from warm start
    T_list_1 = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0005]
    for T in T_list_1:
        if remaining_time(t0, budget_s) < 160:
            break
        res = opt.minimize(
            smooth_obj, h_best.copy(), args=(T,),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 800, 'ftol': 1e-15, 'iprint': 0}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_c = project(res.x, n)
            c = compute_c5(h_c)
            if c < c5_best:
                h_best = h_c
                c5_best = c

    # Phase 2: Diverse multi-start with varied perturbation patterns
    n_restarts = 12
    for r in range(n_restarts):
        if remaining_time(t0, budget_s) < 300:
            break
        scale = 0.02 + 0.012 * (r % 6)
        kind = r % 4
        if kind == 0:
            noise = np.random.randn(n) * scale
        elif kind == 1:
            noise = np.random.uniform(-scale, scale, n)
        elif kind == 2:
            noise = np.zeros(n)
            nblocks = 3 + (r % 4)
            for _ in range(nblocks):
                a = np.random.randint(0, n - 8)
                b = min(a + 6 + np.random.randint(0, n // 6), n)
                noise[a:b] = np.random.uniform(-scale, scale)
        else:
            f = 1.0 + np.random.uniform(0, 4.0)
            p = np.random.uniform(0, 2 * np.pi)
            noise = scale * np.sin(2 * np.pi * f * np.arange(n) / n + p)

        h_p = project(h_best + noise, n)
        T_use = 0.01 if r < 6 else 0.004
        res = opt.minimize(
            smooth_obj, h_p, args=(T_use,),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-15, 'iprint': 0}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_c = project(res.x, n)
            c = compute_c5(h_c)
            if c < c5_best:
                h_best = h_c
                c5_best = c

    # Phase 3: Fine-T refinement on current best
    T_list_3 = [0.001, 0.0005, 0.0002, 0.0001]
    for T in T_list_3:
        if remaining_time(t0, budget_s) < 80:
            break
        res = opt.minimize(
            smooth_obj, h_best.copy(), args=(T,),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 1200, 'ftol': 1e-16, 'iprint': 0}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_c = project(res.x, n)
            c = compute_c5(h_c)
            if c < c5_best:
                h_best = h_c
                c5_best = c

    # Phase 4: Additional restarts with very small perturbations (fine-tune escapes)
    for r in range(6):
        if remaining_time(t0, budget_s) < 100:
            break
        noise = np.random.randn(n) * 0.008
        h_p = project(h_best + noise, n)
        res = opt.minimize(
            smooth_obj, h_p, args=(0.001,),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 600, 'ftol': 1e-16, 'iprint': 0}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_c = project(res.x, n)
            c = compute_c5(h_c)
            if c < c5_best:
                h_best = h_c
                c5_best = c

    # Phase 5: Coordinate descent polishing
    if remaining_time(t0, budget_s) > 30:
        h_best, c5_cd = coord_descent(h_best, n, n_sweeps=3, lr=0.005)
        if c5_cd < c5_best:
            c5_best = c5_cd

    # Final validation and return
    h_best = project(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
