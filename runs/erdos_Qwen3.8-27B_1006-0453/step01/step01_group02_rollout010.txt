    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += (1.0 - h[:N - lag])
            g[:N - lag] -= h[lag:]
    else:
        l = -lag
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def weighted_grad(h, N, w, topk):
    grad = np.zeros(N)
    idx = np.argsort(w)[::-1][:topk]
    for i in idx:
        wi = w[i]
        if wi < 1e-10:
            break
        lag = int(i) - (N - 1)
        if -N < lag < N:
            grad += wi * grad_lag(h, lag, N)
    return grad

def optimize_refined(h, N, target, dx, t0, time_limit, rng, max_iters=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    corr = np.correlate(h, 1.0 - h, mode='full')

    lr = 0.015
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    stagnation = 0
    max_no_improve = 300

    for it in range(max_iters):
        if time.time() - t0 > time_limit - 3:
            break
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        progress = it / max(float(max_iters), 1)
        tau = 5.0 * (1.0 + 40.0 * progress ** 2)
        topk = max(10, int(50 * (1.0 - progress) + 10))

        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones_like(corr) / len(corr)
        else:
            w /= ws

        grad = weighted_grad(h, N, w, topk)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))

        h_new = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        mx_new = float(np.max(corr_new))
        mx_old = float(np.max(corr))

        if mx_new > mx_old * 1.02:
            h_new = project_sum(h - 0.3 * lr * m_hat / (np.sqrt(v_hat) + eps), target)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            mx_new = float(np.max(corr_new))

        if mx_new <= mx_old * 1.01:
            h = h_new
            corr = corr_new
        else:
            lr *= 0.92

        if stagnation > max_no_improve:
            m_buf[:] = 0.0
            v_buf[:] = 0.0
            lr = 0.02
            stagnation = 0
            max_no_improve = int(max_no_improve * 0.7)

        if lr < 1e-9:
            break

    return best_h, best_c5

def binary_snap(h, N, target):
    h2 = h.copy()
    h2[h2 < 0.08] = 0.0
    h2[h2 > 0.92] = 1.0
    h2 = project_sum(h2, target)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    sizes = [n0]
    if n0 * 2 <= 800:
        sizes.append(n0 * 2)
    if n0 * 4 <= 1000 and n0 * 2 in sizes:
        sizes.append(n0 * 4)
    sizes = sorted(set(s for s in sizes if s <= 1000 and s >= 90))

    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    for N in sizes:
        if time.time() - t0 > budget_s - 50:
            break
        dx = 2.0 / N
        target = N / 2.0

        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_interp = np.interp(xN, x0, best_h)
        h_interp = project_sum(h_interp, target)

        n_restarts = 3 if N <= 400 else 2
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 25:
                break
            h_try = h_interp.copy()
            if r == 1:
                h_try = h_try + rng.randn(N) * 0.03
                h_try = project_sum(h_try, target)
            elif r == 2:
                h_try = h_try + rng.randn(N) * 0.06
                h_try = project_sum(h_try, target)

            max_it = 2500 if N <= 400 else 1500
            h_opt, c5_opt = optimize_refined(h_try, N, target, dx, t0, budget_s - 10, rng, max_iters=max_it)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        # Binary snap attempt
        if time.time() - t0 < budget_s - 30:
            h_snap = binary_snap(best_h[0:N] if len(best_h) == N else np.interp(xN, x0, best_h), N, target)
            h_snap = project_sum(h_snap, target)
            h_s, c5_s = optimize_refined(h_snap, N, target, dx, t0, budget_s - 10, rng, max_iters=1000)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Final polish with more iters
    N_b = len(best_h)
    dx_b = 2.0 / N_b
    tgt_b = N_b / 2.0
    if time.time() - t0 < budget_s - 15:
        h_pol, c5_pol = optimize_refined(best_h.copy(), N_b, tgt_b, dx_b, t0, budget_s, rng, max_iters=1500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Tiny local search: nudge individual entries
    if time.time() - t0 < budget_s - 8:
        for trial in range(50):
            if time.time() - t0 > budget_s - 5:
                break
            i = rng.randint(N_b)
            for delta in [0.02, -0.02, 0.01, -0.01]:
                h_try = best_h.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                h_try = project_sum(h_try, tgt_b)
                c5_try = compute_c5(h_try, dx_b)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    break

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
