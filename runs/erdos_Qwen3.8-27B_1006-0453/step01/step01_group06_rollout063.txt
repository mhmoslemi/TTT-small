import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1e-3:
        return m * dx
    shifted = np.clip(T * (corr - m), -600, 0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return float(val * dx)

def sum_func(h):
    return float(h.sum() - len(h) / 2.0)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
        else:
            room = h.copy()
            total = room.sum()
        if total < 1e-15:
            break
        h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    n = len(h)
    lag = k - (n - 1)
    lo = max(0, -lag)
    hi = min(n - 1, n - 1 - lag)
    grad = np.zeros(n)
    for i in range(lo, hi + 1):
        jp = i + lag
        if 0 <= jp < n:
            grad[i] += (1.0 - h[jp])
        if lag != 0 and 0 <= jp < n:
            grad[jp] -= h[i]
    return grad

def soft_gradient(h, T_sub):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.clip(T_sub * (corr - m), -600, 0)
    w = np.exp(w)
    w /= w.sum()
    neg_inf = np.where(corr < m - 50.0 / T_sub, False, True)
    grad = np.zeros(n)
    for k in np.where(neg_inf)[0]:
        gk = lag_gradient(h, k)
        grad += w[k] * gk
    return grad, float(np.max(corr))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(60, 0.5)
        best_h += 0.05 * np.random.randn(60)

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with smooth objective, increasing T
    for T in [30, 100, 500, 2000, 8000, 30000]:
        if elapsed() > budget_s * 0.22:
            break
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_func}]
        try:
            res = minimize(smooth_obj, best_h.copy(), args=(T,),
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-15, 'iprint': 0})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Soft subgradient descent with line search
    n_sg = 800
    for it in range(n_sg):
        if elapsed() > budget_s * 0.42:
            break
        grad, _ = soft_gradient(best_h, 200.0)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_act = int(np.argmax(corr))
        grad_exact = lag_gradient(best_h, k_act)
        grad = 0.3 * grad_exact + 0.7 * grad
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        base_alpha = 0.06 * (0.995 ** it) + 0.002
        improved = False
        for trial in range(20):
            a = base_alpha * (0.65 ** trial)
            ht = project_feasible(best_h - a * grad, n)
            v = compute_c5(ht)
            if v < best_c5 - 1e-13:
                best_c5 = v
                best_h = ht
                improved = True
                break
        if not improved and it % 150 == 149:
            pert = np.random.randn(n) * 0.015
            ht = project_feasible(best_h + pert, n)
            if compute_c5(ht) < best_c5:
                best_c5 = compute_c5(ht)
                best_h = ht

    # Phase 3: Multi-start local optimization
    n_starts = 50
    for i in range(n_starts):
        if elapsed() > budget_s * 0.70:
            break
        scale = 0.07 * (0.94 ** i) + 0.004
        h0 = best_h + np.random.randn(n) * scale
        h0 = project_feasible(h0, n)
        for T in [300, 3000]:
            if elapsed() > budget_s * 0.70:
                break

                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    h0 = hc.copy()
            except Exception:
                pass

    # Phase 4: Try other resolutions
    for n_try in [50, 56, 64, 72, 80, 90]:
        if elapsed() > budget_s * 0.87:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try)
        for T in [500, 4000]:
            if elapsed() > budget_s * 0.87:
                break
            bnds2 = [(0.0, 1.0)] * n_try
            cons2 = [{'type': 'eq', 'fun': sum_func}]
            try:
                res = minimize(smooth_obj, h_new.copy(), args=(T,),
                              method='SLSQP', bounds=bnds2, constraints=cons2,
                              options={'maxiter': 500, 'ftol': 1e-14, 'iprint': 0})
                hc = project_feasible(np.clip(res.x, 0, 1), n_try)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 5: Final subgradient polish
    n_cur = len(best_h)
    for it in range(600):
        if elapsed() > budget_s * 0.97:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_act = int(np.argmax(corr))
        grad = lag_gradient(best_h, k_act)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        a = 0.004 * (0.997 ** it) + 0.0005
        ht = project_feasible(best_h - a * grad, n_cur)
        v = compute_c5(ht)
        if v < best_c5 - 1e-14:
            best_c5 = v
            best_h = ht

    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s * 0.995:
        n_cur = len(best_h)
        bnds = [(0.0, 1.0)] * n_cur
        cons = [{'type': 'eq', 'fun': sum_func}]
        try:
            res = minimize(smooth_obj, best_h.copy(), args=(80000,),
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
            hc = project_feasible(np.clip(res.x, 0, 1), n_cur)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
