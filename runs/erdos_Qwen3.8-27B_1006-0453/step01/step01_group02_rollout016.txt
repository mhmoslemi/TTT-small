import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, lag, N):
    g = np.zeros(N)
    if 0 <= lag < N:
        g[:N - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:N - lag]
    elif -N < lag < 0:
        l = -lag
        g[l:] += (1.0 - h[:N - l])
        g[:N - l] -= h[l:]
    return g

def optimize_phase(h, N, target, dx, t0, budget_s, lr, tau, topk, n_iter):
    best = h.copy()
    bc = compute_c5(best, dx)
    b1, b2, eps = 0.8, 0.999, 1e-8
    mb = np.zeros(N)
    vb = np.zeros(N)
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 12:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v < bc:
            bc = c5v
            best = h.copy()
        cmax = corr.max()
        idx = np.argsort(corr)[-topk:]
        sc = tau * (corr[idx] - cmax)
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(idx)) / len(idx)
        else:
            w /= ws
        grad = np.zeros(N)
        for j in range(len(idx)):
            if w[j] < 1e-7:
                continue
            lag = int(idx[j]) - (N - 1)
            grad += w[j] * grad_lag(h, lag, N)
        gn = np.linalg.norm(grad)
        if gn > 1e-12:
            grad /= gn
        mb = b1 * mb + (1.0 - b1) * grad
        vb = b2 * vb + (1.0 - b2) * grad * grad
        mh = mb / (1.0 - b1 ** (it + 1))
        vh = vb / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best, bc

def multi_phase_opt(h, N, target, dx, t0, budget_s):
    # Phase 1: Explore
    h1, c1 = optimize_phase(h, N, target, dx, t0, budget_s, 0.04, 5.0, 10, 2000)
    # Phase 2: Exploit
    h2, c2 = optimize_phase(h1, N, target, dx, t0, budget_s, 0.015, 25.0, 4, 2500)
    # Phase 3: Polish top-1
    h3, c3 = optimize_phase(h2, N, target, dx, t0, budget_s, 0.004, 80.0, 2, 3000)
    return h3, c3

def local_search(h, N, target, dx, t0, budget_s, rng, n_trials=400):
    best = h.copy()
    bc = compute_c5(best, dx)
    for _ in range(n_trials):
        if time.time() - t0 > budget_s - 14:
            break
        s = 0.003 + 0.015 * rng.random()
        ht = best + rng.randn(N) * s
        ht = project_sum(ht, target)
        c = compute_c5(ht, dx)
        if c < bc - 3e-7:
            best, bc = ht, c
    return best, bc
