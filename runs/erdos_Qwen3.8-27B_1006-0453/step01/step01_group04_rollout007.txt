        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # Distribute uniformly to those not at bounds
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(free):
            # Relax bounds slightly
            free = (h > 1e-15) & (h < 1.0 - 1e-15)
            if not np.any(free):
                free = (h > 0) | (h < 1)
        if not np.any(free):
            break
        adj = diff / np.sum(free)
        h[free] = h[free] + adj
        h = np.clip(h, 0.0, 1.0)

def subgradient_step(h, alpha, top_m=20):
    """One projected subgradient step for the max-of correlations"""
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Find top M lags
    idx = np.argpartition(corr, -top_m)[-top_m:]
    idx = idx[np.argsort(corr[idx])]
    
    # Weighted subgradient
    weights = np.exp(corr[idx] - corr[idx][-1])  # softmax-like weights
    weights /= np.sum(weights)
    
    # Compute subgradient (gradient of max over active lags)
    # For lag k (0-indexed in full correlate, where n-1 is zero lag):
    # d/dh[i] of corr[n-1+k] involves h[i] and h[i ± k]
    grad = np.zeros(n)
    
    for m in range(len(idx)):
        k = idx[m] - (n - 1)  # actual shift
        w = weights[m]
        if k >= 0:
            # corr at this lag: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[i] for 0<=i<n-k: (1-h[i+k])
            # d/dh[i] for k<=i<n: h[i-k]
            for i in range(n - k):
                grad[i] += w * (1.0 - h[i + k])
                grad[i + k] += w * h[i]
        else:
            k2 = -k
            for i in range(n - k2):
                grad[i + k2] += w * (1.0 - h[i])
                grad[i] += w * h[i + k2]
    
    # Projected gradient step
    h_new = h - alpha * grad
    h_new = project_onto_box_sum(h_new, target)
    return h_new
