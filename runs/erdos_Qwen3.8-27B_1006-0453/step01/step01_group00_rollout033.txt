def compute_grad_softmax(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    w = np.exp(shifted / T)
    sw = np.sum(w)
    w = w / sw  # normalized weights, sum to 1
    
    grad = np.zeros(n)
    for idx, wi in enumerate(w):
        if wi < 1e-15:
            continue
        m = idx - (n - 1)  # lag
        # Term 1: (1 - h[j-m]) for valid j where j-m in [0, n-1]
        # j range: m to m+n-1, intersected with [0, n-1]
        j1_lo = max(0, m)
        j1_hi = min(n-1, m + n - 1)
        if j1_lo <= j1_hi:
            h_idx = np.arange(j1_lo, j1_hi + 1) - m  # = h[j-m] indices
            grad[j1_lo:j1_hi+1] += wi * (1.0 - h[h_idx])
        # Term 2: -h[j+m] for valid j where j+m in [0, n-1]
        # j range: -m to n-1-m, intersected with [0, n-1]
        j2_lo = max(0, -m)
        j2_hi = min(n-1, n - 1 - m)
        if j2_lo <= j2_hi:
            h_idx = np.arange(j2_lo, j2_hi + 1) + m  # = h[j+m] indices
            grad[j2_lo:j2_hi+1] += wi * (-h[h_idx])
    
    return grad

**Coordinate descent with line search:**
For each coordinate j (cycle through all):
- Let current h[j] = x
- The objective as a function of x (with all other h fixed) is:
  C(x) = max_k f_k(x) where f_k is the corr value at lag k as a function of x
- f_k(x) is linear in x for each k (since h[j] appears linearly in each corr term)
  Wait, no! h[j] * (1-h[j+k]) is quadratic in h if j+k is also j... but since we're fixing all other h, h[j+k] is fixed, so h[j]*(1-h[j+k]) is linear in h[j]. Similarly h[j-k]*(1-h[j]) is linear in h[j].
  
  So YES, each f_k(x) is LINEAR in x (when all other coordinates are fixed)!
  
  Therefore C(x) = max of linear functions of x, which is a convex piecewise-linear function of x.
  
  To minimize it: find the point where the maximum of the active linear functions is minimized. This is where the "upper envelope" has its minimum.
  
  For a set of linear functions a_k * x + b_k, the max is minimized at the intersection point of the two "extreme" lines (the one with largest positive slope and the one with largest negative slope among the active constraints).

This is great! For each coordinate j, the optimal value is where the max of the active linear functions is minimized, which can be found by:
1. Computing all the (a_k, b_k) pairs for the relevant k's
2. Finding the point where the upper envelope is minimized

The upper envelope of lines is minimized at a "valley" point, which is where two adjacent lines in the sorted-by-slope order intersect below all others.

This is essentially a 1D linear programming problem: minimize t subject to a_k * x + b_k ≤ t for all k, 0 ≤ x ≤ 1.

The optimal t is min_x max_k (a_k * x + b_k).

For efficiency, we can just evaluate at the intersection points of consecutive lines (sorted by slope) and at the endpoints 0 and 1.

def coord_descent_step(h, j, corr):
    n = len(h)
    x = h[j]
    # For each lag m, compute a_m * x + b_m where corr[m] = a_m * h[j] + b_m
    # corr_full[i] where i = m + (n-1):
    # corr_full[i] = sum_{n_valid} h[n+m] * (1-h[n])
    # The term involving h[j]:
    # If j = n+m (n = j-m valid): contributes x * (1-h[j-m])
    # If j = n (and j+m valid): contributes h[j+m] * (1-x)
    # So corr_full[i] = a_m * x + b_m where:
    # a_m = (1-h[j-m]) * I(0≤j-m≤n-1) + (-h[j+m]) * I(0≤j+m≤n-1)
    # b_m = (corr_full[i] - a_m * x)  [current value minus the x-dependent part]
    
    # Actually: corr_full[i] currently = a_m * x + b_m
    # So b_m = corr_full[i] - a_m * x
    
    # We want to minimize max_m (a_m * x + b_m) over x in [0,1]
    
    a_list = []
    b_list = []
    for i in range(2*n-1):
        m = i - (n-1)
        a = 0.0
        if 0 <= j-m <= n-1:
            a += (1.0 - h[j-m])
        if 0 <= j+m <= n-1:
            a -= h[j+m]
        b = corr[i] - a * x
        a_list.append(a)
        b_list.append(b)
    
    a_arr = np.array(a_list)
    b_arr = np.array(b_list)
    
    # Minimize max_k (a_k * x + b_k) for x in [0,1]
    # The maximum of lines is convex in x. Find minimum.
    # Sort by slope, find intersection points
    
    order = np.argsort(a_arr)
    a_sorted = a_arr[order]
    b_sorted = b_arr[order]
    
    best_x = x
    best_val = np.max(a_arr * x + b_arr)
    
    # Check x=0 and x=1
    val_0 = np.max(b_arr)
    if val_0 < best_val:
        best_val = val_0
        best_x = 0.0
    val_1 = np.max(a_arr + b_arr)
    if val_1 < best_val:
        best_val = val_1
        best_x = 1.0
    
    # Check intersection points of consecutive lines (sorted by slope)
    for k in range(len(a_sorted)-1):
        if a_sorted[k+1] == a_sorted[k]:
            continue
        x_int = (b_sorted[k] - b_sorted[k+1]) / (a_sorted[k+1] - a_sorted[k])
        if 0 <= x_int <= 1:
            val = np.max(a_arr * x_int + b_arr)
            if val < best_val:
                best_val = val
                best_x = x_int
    
    return best_x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_obj(h, T):
    corr = get_corr(h)
    cmax = np.max(corr)
    if T < 1e-15:
        return cmax
    s = np.sum(np.exp((corr - cmax) / T))
    return cmax + T * np.log(s)

def softmax_grad(h, T):
    n = len(h)
    corr = get_corr(h)
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        return np.zeros(n)
    w = w / sw
    
    grad = np.zeros(n)
    for idx in range(2*n-1):
        if w[idx] < 1e-10:
            continue
        m = idx - (n-1)
        # d corr[idx]/d h[j] for all j:
        # = (1-h[j-m]) * I(0≤j-m≤n-1) - h[j+m] * I(0≤j+m≤n-1)
        # Vectorize:
        # Term 1: for j in [max(0,m), min(n-1, m+n-1)]: (1-h[j-m])
        j1_lo = max(0, m)
        j1_hi = min(n-1, m+n-1)
        if j1_lo <= j1_hi:
            idx1 = np.arange(j1_lo, j1_hi+1) - m
            grad[j1_lo:j1_hi+1] += w[idx] * (1.0 - h[idx1])
        # Term 2: for j in [max(0,-m), min(n-1, n-1-m)]: -h[j+m]
        j2_lo = max(0, -m)
        j2_hi = min(n-1, n-1-m)
        if j2_lo <= j2_hi:
            idx2 = np.arange(j2_lo, j2_hi+1) + m
            grad[j2_lo:j2_hi+1] += w[idx] * (-h[idx2])
    
    return grad

    target = n * 0.5
    s = np.sum(h)

        return h
    h = h.copy()
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n * 0.5
    s = np.sum(h_new)
    if s > 1e-15:
        h_new = h_new * (target / s)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-14:
        interior = (h_new > 1e-6) & (h_new < 1.0 - 1e-6)
        n_int = np.sum(interior)
        if n_int > 0:
            h_new[interior] += diff / n_int
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def coord_descent(h, n_passes=50, top_k=20):
    n = len(h)
    h = h.copy()
    for p in range(n_passes):
        improved = False
        corr = get_corr(h)
        max_corr = np.max(corr)
        # Get indices of top_k largest corr values
        top_idx = np.argsort(corr)[-top_k:]

                b = corr[i] - a * x
                a_vals.append(a)
                b_vals.append(b)
            a_arr = np.array(a_vals)
            b_arr = np.array(b_vals)
            
            # Find x* in [0,1] minimizing max(a*x+b)
            # Try grid
            xs = np.linspace(0, 1, 51)
            vals = np.max(a_arr[:, None] * xs[None, :] + b_arr[:, None], axis=0)
            x_new = xs[np.argmin(vals)]
            
            if x_new != x:
                h[j] = x_new
                improved = True

        if not improved:

def softmax_grad(h, T):

    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        return np.zeros(N)
    w = w / sw
    
    grad = np.zeros(N)
    for i in range(2*N-1):
        wi = w[i]
        if wi < 1e-12:
            continue

        j1_hi = min(N-1+k, N-1)
        if j1_lo <= j1_hi:
            # j-k ranges from j1_lo-k to j1_hi-k

            grad[j1_lo:j1_hi+1] += wi * (1.0 - h[idx1])
        
        # Second term: -h[j+k] for j in [max(0,-k), min(N-1, N-1-k)]

            idx2 = np.arange(j2_lo, j2_hi+1) + k
            grad[j2_lo:j2_hi+1] += wi * (-h[idx2])
    
    return grad

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_obj(h, T):
    corr = get_corr(h)
    cmax = np.max(corr)
    if T < 1e-15:
        return cmax
    s = np.sum(np.exp((corr - cmax) / T))
    return cmax + T * np.log(s)

def softmax_grad(h, T):
    N = len(h)
    corr = get_corr(h)
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        return np.zeros(N)
    w = w / sw
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = i - (N - 1)
        j1_lo = max(k, 0)
        j1_hi = min(N - 1 + k, N - 1)
        if j1_lo <= j1_hi:
            idx1 = np.arange(j1_lo, j1_hi + 1) - k
            grad[j1_lo:j1_hi + 1] += wi * (1.0 - h[idx1])
        j2_lo = max(0, -k)
        j2_hi = min(N - 1, N - 1 - k)
        if j2_lo <= j2_hi:
            idx2 = np.arange(j2_lo, j2_hi + 1) + k
            grad[j2_lo:j2_hi + 1] += wi * (-h[idx2])
    return grad

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def penalty_obj(h, T, mu):
    base = softmax_obj(h, T)
    s = np.sum(h) - len(h) * 0.5
    return base + mu * s * s

def penalty_grad(h, T, mu):
    bg = softmax_grad(h, T)
    s = np.sum(h) - len(h) * 0.5
    return bg + 2.0 * mu * s

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n * 0.5
    s = np.sum(h_new)
    if s > 1e-15:
        h_new = h_new * (target / s)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-14:
        interior = (h_new > 1e-6) & (h_new < 1.0 - 1e-6)
        n_int = np.sum(interior)
        if n_int > 0:
            h_new[interior] += diff / n_int
    return np.clip(h_new, 0.0, 1.0)

def coord_descent(h, n_passes=100, top_k=30):
    n = len(h)
    h = h.copy()
    for p in range(n_passes):
        corr = get_corr(h)
        max_before = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for j in range(n):
            x = h[j]
            a_vals = np.zeros(len(top_idx))
            b_vals = np.zeros(len(top_idx))
            for ii, i in enumerate(top_idx):
                m = i - (n - 1)
                a = 0.0
                if 0 <= j - m <= n - 1:
                    a += (1.0 - h[j - m])
                if 0 <= j + m <= n - 1:
                    m_j = j + m
                    if m_j >= 0 and m_j < n:
                        a -= h[m_j]
                b_vals[ii] = corr[i] - a * x
                a_vals[ii] = a
            xs = np.linspace(0.0, 1.0, 41)
            vals = np.max(a_vals[:, None] * xs[None, :] + b_vals[:, None], axis=0)
            x_new = float(xs[np.argmin(vals)])
            if abs(x_new - x) > 1e-10:
                h[j] = x_new
                improved = True
        if not improved:
            break
    h = fix_sum(h, n)
    return h

def run_seed_opt(h_start, n, T, max_iter, method='SLSQP'):
    if method == 'SLSQP':
        res = opt.minimize(
            softmax_obj, h_start, args=(T,),
            jac=softmax_grad, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-15}

    else:
        res = opt.minimize(
            penalty_obj, h_start, args=(T, 500.0),

        )
    if not hasattr(res, 'x') or res.x is None:
        return h_start
    h_out = np.clip(res.x, 0.0, 1.0)
    return fix_sum(h_out, n)
