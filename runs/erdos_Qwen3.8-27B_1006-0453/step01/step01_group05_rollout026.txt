def pair_c5_eval(h, n, i, j, x_i):
    """Compute C5 with h[i]=x_i and h[j]=h[i]+h[j]-x_i, maintaining sum."""
    h_new = h.copy()
    h_new[i] = x_i
    h_new[j] = h[i] + h[j] - x_i
    return compute_c5(h_new, n)

def pair_optimize(h, n, i, j):
    """Optimize h[i] (and h[j] to maintain sum). Returns new h and C5."""
    s_ij = h[i] + h[j]
    # x_i must satisfy: 0 <= x_i <= 1, 0 <= s_ij - x_i <= 1
    # So: max(0, s_ij-1) <= x_i <= min(1, s_ij)
    lo = max(0.0, s_ij - 1.0)
    hi = min(1.0, s_ij)
    if hi - lo < 1e-10:
        return h, compute_c5(h, n)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    f_c = pair_c5_eval(h, n, i, j, c)
    f_d = pair_c5_eval(h, n, i, j, d)
    
    for _ in range(50):
        if f_c < f_d:
            b = d
            d = c
            f_d = f_c
            c = b - gr * (b - a)
            f_c = pair_c5_eval(h, n, i, j, c)
        else:
            a = c
            c = d
            f_c = f_d
            d = a + gr * (b - a)
            f_d = pair_c5_eval(h, n, i, j, d)
    
    x_opt = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] = x_opt
    h_new[j] = s_ij - x_opt
    return h_new, compute_c5(h_new, n)

- If j = i+shift: term is h[i]*(1-h[j]) = x*(1-(s-x)) = x - sx + x² (quadratic, convex)
- If j = i-shift: term is h[j+shift]*(1-h[j]) = h[i]*(1-h[j]) = same thing

- h[i]*(1-h[m]) for m ≠ j: linear in x (coefficient -(h[m]) is constant... wait, no: x*(1-h[m]) is linear in x)
- h[m]*(1-h[i]) for m+shift = i, i.e., m = i-shift: h[m]*(1-x) is linear in x

So each corr[k] is a sum of:
- Linear terms in x (from most interactions)
- At most one quadratic term in x (if j is at the right offset from i for that shift k)

The quadratic term has coefficient +1 (since x*(1-s+x) = x - sx + x²). So each corr[k] is convex in x (a sum of linear and a convex quadratic term).

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):

        diff = target - s

            h = np.clip(h + diff / n, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)

    m = np.max(corr)
    z = temp * (corr - m)
    zm = np.max(z)

    fc = float(np.max(np.correlate(h2, 1.0 - h2, mode='full'))) * (2.0 / n)
    h2[i] = d; h2[j] = s_ij - d
    fd = float(np.max(np.correlate(h2, 1.0 - h2, mode='full'))) * (2.0 / n)
    for _ in range(iters):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            h2[i] = c; h2[j] = s_ij - c
            fc = float(np.max(np.correlate(h2, 1.0 - h2, mode='full'))) * (2.0 / n)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            h2[i] = d; h2[j] = s_ij - d
            fd = float(np.max(np.correlate(h2, 1.0 - h2, mode='full'))) * (2.0 / n)
    x_opt = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] = x_opt
    h_new[j] = s_ij - x_opt
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, compute_c5(h_new, n)

def pair_cd_round(h, n, order, max_updates, t_start, deadline):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    updates = 0
    for (i, j) in order:
        if updates >= max_updates:
            break
        if time.time() > deadline:
            break
        h_new, c5_new = pair_optimize(best, n, i, j, iters=28)
        if c5_new < best_c5 - 1e-13:
            best = h_new
            best_c5 = c5_new
            updates += 1
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    h0 = repair_h(h0, n)
    bounds = [(0.0, 1.0)] * n

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    # Phase 1: SLSQP polynomial polish
    for temp in [10, 50, 200]:
        if time.time() - t_start > budget_s - 400:
            break
        h_t = slsqp_opt(best_h, n, bounds, temp, 200)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 2: Iterated pair coordinate descent with random restarts
    deadline_cd = t_start + budget_s * 0.55
    stall_count = 0
    for sweep in range(300):
        remaining = deadline_cd - time.time()
        if remaining < 2:
            break
        order = all_pairs[:]
        rng.shuffle(order)
        max_up = max(50, int(remaining * 80))
        prev_c5 = best_c5
        best_h, best_c5 = pair_cd_round(best_h, n, order, max_up, t_start, deadline_cd)
        if best_c5 < prev_c5 - 1e-10:
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 12:
                # Kick: random perturbation then continue
                kick_sigma = 0.08
                cand = best_h + rng.normal(0, kick_sigma, n)
                cand = repair_h(cand, n)
                cand = slsqp_opt(cand, n, bounds, 80, 100)
                c5_kick = compute_c5(cand, n)
                if c5_kick < best_c5 + 0.002:
                    best_h = cand
                    best_c5 = c5_kick
                stall_count = 0

    # Phase 3: Simulated annealing with SLSQP local solver
    deadline_sa = t_start + budget_s * 0.85
    current = best_h.copy()
    sa_temp = 0.4
    for it in range(300):
        if time.time() > deadline_sa:
            break
        sigma = 0.12 * sa_temp + 0.02
        cand = current + rng.normal(0, sigma, n)
        cand = repair_h(cand, n)
        h_opt = slsqp_opt(cand, n, bounds, 100, 80)
        c5_opt = compute_c5(h_opt, n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            current = h_opt
        else:
            cur_c5 = compute_c5(current, n)
            if rng.random() < np.exp(-(c5_opt - cur_c5) / sa_temp):
                current = h_opt
        sa_temp = max(0.03, sa_temp * 0.985)

    # Phase 4: Final pair CD + SLSQP polish
    deadline_final = t_start + budget_s - 5
    order = all_pairs[:]
    rng.shuffle(order)
    best_h, best_c5 = pair_cd_round(best_h, n, order, 300, t_start, deadline_final)

    if time.time() < deadline_final:
        for temp in [30, 150, 600]:
            if time.time() > deadline_final:
                break
            h_t = slsqp_opt(best_h, n, bounds, temp, 400)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    best_h = repair_h(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
