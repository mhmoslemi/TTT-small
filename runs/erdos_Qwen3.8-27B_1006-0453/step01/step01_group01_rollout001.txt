import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def topm_grad(h, m=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr, kind='stable')[::-1][:m]
    grad = np.zeros(n)
    j = np.arange(n)
    for p in idx:
        k = p - (n - 1)
        a = j - k
        b = j + k
        ma = (a >= 0) & (a < n)
        mb = (b >= 0) & (b < n)
        ga = np.where(ma, 1.0 - h[a], 0.0)
        gb = np.where(mb, h[b], 0.0)
        grad += ga - gb
    return grad / m

def pgd(h0, max_iter=8000, step_init=0.004, m=30, time_limit=100.0, decay=0.9995):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_init
    t_start = time.time()
    prev_c = best_c
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        c_curr = compute_c5(h)
        if c_curr < best_c - 1e-10:
            best_c = c_curr
            best_h = h.copy()
        g = topm_grad(h, m)
        c_curr = compute_c5(h)
        for _ in range(20):
            h_new = proj(h - step * g, n)
            c_new = compute_c5(h_new)
            if c_new <= c_curr + 1e-13:
                break
            step *= 0.5
        else:
            step *= 0.5
            h_new = proj(h - step * g, n)
            c_new = compute_c5(h_new)
        if c_new < c_curr - 1e-10:
            step = min(step * 1.25, 0.04)
        h = h_new
        step *= (1.0 - decay + 0.9995)
    return best_h, best_c

def smooth_obj(h, t=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def smooth_obj_grad(h, t=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    w = np.exp(s)
    w /= np.sum(w)
    val = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    j = np.arange(n)
    grad = np.zeros(n)
    for p in range(len(corr)):
        wp = w[p]
        if wp < 1e-12:
            continue
        k = p - (n - 1)
        a = j - k
        b = j + k
        ma = (a >= 0) & (a < n)
        mb = (b >= 0) & (b < n)
        ga = np.where(ma, 1.0 - h[a], 0.0)
        gb = np.where(mb, h[b], 0.0)
        grad += wp * (ga - gb)
    return val, grad * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)
