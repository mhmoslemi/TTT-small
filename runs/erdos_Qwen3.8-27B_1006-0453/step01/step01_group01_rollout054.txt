def smooth_max_obj(h, t):

    dx = 2.0 / n
    corr = get_correlations(h)
    c_max = np.max(corr)
    s = t * (corr - c_max)
    s -= np.max(s)
    val = (c_max + np.log(np.sum(np.exp(s))) / t) * dx

def proj_280(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            h = np.clip(h + d / n, 0.0, 1.0)
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_subgradient(h, t, steps, lr_init):
    """Projected subgradient descent on the smoothed max."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()

        s -= np.max(s)
        w = np.exp(s) / np.sum(np.exp(s))

        m_idx = np.arange(2 * n - 1)
        L = m_idx - (n - 1)
        L_col = L[None, :]
        
        plus_idx = j_idx + L_col
        valid1 = (plus_idx >= 0) & (plus_idx < n)
        term1 = (1 - h[np.clip(plus_idx, 0, n-1)]) * valid1
        
        minus_idx = j_idx - L_col

        h = h_new
    
    return best_h, best_c

    return proj(np.interp(xd, xs, h), nd)

def psgd(h, t, steps, lr0, momentum):
    n = len(h)
    dx = 2.0 / n
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr0
    for step in range(steps):
        corr = get_corr(h)
        cm = np.max(corr)
        c5 = cm * dx
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        s = t * (corr - cm)
        s -= np.max(s)
        w = np.exp(s) / np.sum(np.exp(s))
        jx = np.arange(n)[:, None]
        Lx = (np.arange(2 * n - 1) - (n - 1))[None, :]
        pi = jx + Lx
        v1 = (pi >= 0) & (pi < n)
        t1 = (1.0 - h[np.clip(pi, 0, n - 1)]) * v1
        mi = jx - Lx
        v2 = (mi >= 0) & (mi < n)
        t2 = h[np.clip(mi, 0, n - 1)] * v2
        g = dx * ((t1 - t2) @ w)
        v = momentum * v + (1.0 - momentum) * g
        h = proj(h - lr * v, n)
        if step % 300 == 299:
            lr *= 0.92
    return best_h, best_c

def coord_descend(h, t, n_passes):
    n = len(h)
    dx = 2.0 / n
    best_c = compute_c5(h)
    for _ in range(n_passes):
        for j in range(n):
            old = h[j]
            # Try small perturbations
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                trial = h.copy()
                trial[j] = np.clip(old + delta, 0.0, 1.0)
                trial = proj(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c:
                    best_c = c5
                    h = trial
                    old = trial[j]
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _opt_t
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n_target = 280

    h0 = None
    try:
        if len(initial_h_values) == n_target:
            h0 = np.array(initial_h_values, dtype=float).copy()
            h0 = proj(h0, n_target)
    except Exception:
        pass
    if h0 is None:
        h0 = np.full(n_target, 0.5)
        h0 = proj(h0, n_target)

    bh = h0.copy()
    bc = compute_c5(h0)
    bnds = [(0.0, 1.0)] * n_target
    cons = {'type': 'eq', 'fun': _ceq, 'jac': _ceqj}

    # Phase 1: SLSQP at n=280, increasing temperature
    for tv in [80.0, 200.0, 500.0]:
        _opt_t = tv
        if time.time() > t0 + 320:
            break
        try:
            r = minimize(_obj, bh, jac=_jacobian, method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 800, 'ftol': 1e-13})
            hc = proj(np.clip(r.x, 0.0, 1.0), n_target)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc.copy()
        except Exception:
            pass

    # Phase 2: PSGD with momentum
    t2_end = t0 + 520
    if time.time() < t2_end:
        for tv in [300.0, 800.0, 2000.0]:
            if time.time() > t2_end - 30:
                break
            hs, cs = psgd(bh.copy(), tv, 4000, 0.004, 0.6)
            c5 = compute_c5(hs)
            if c5 < bc:
                bc = c5
                bh = hs.copy()

    # Phase 3: Coarse grid multi-start (n=70)
    t3_end = t0 + 820
    if time.time() < t3_end:
        n_c = 70
        h_c = interp_h(bh, n_c)
        bnds_c = [(0.0, 1.0)] * n_c
        xg = np.arange(n_c) * 2.0 / n_c
        starts_c = [h_c.copy()]
        for k in [2, 3, 5, 7, 11]:
            starts_c.append(proj(0.5 + 0.3 * np.sin(2 * np.pi * k * xg + rng.rand() * 6.28), n_c))
        starts_c.append(proj(0.5 + 0.15 * rng.randn(n_c), n_c))
        starts_c.append(proj(0.5 + 0.08 * np.cos(4 * np.pi * xg) + 0.1 * rng.randn(n_c), n_c))

        for tv in [200.0, 800.0]:
            _opt_t = tv
            for s_c in starts_c:
                if time.time() > t3_end - 15:
                    break
                try:
                    r = minimize(_obj, s_c, jac=_jacobian, method='SLSQP',
                                 bounds=bnds_c, constraints=cons,
                                 options={'maxiter': 600, 'ftol': 1e-13})
                    hc = proj(np.clip(r.x, 0.0, 1.0), n_c)
                    h_f = interp_h(hc, n_target)
                    c5 = compute_c5(h_f)
                    if c5 < bc:
                        bc = c5
                        bh = h_f.copy()
                except Exception:
                    pass

    # Phase 4: Coordinate descent fine-tuning
    t4_end = t0 + budget_s - 80
    if time.time() < t4_end:
        mixed = 0.7 * bh + 0.3 * interp_h(bh, 140) if len(bh) != 140 else bh.copy()
        mixed = proj(mixed, n_target)
        for _ in range(3):
            if time.time() > t4_end:
                break
            hs, cs = coord_descend(bh.copy(), 500.0, 8)
            c5 = compute_c5(hs)
            if c5 < bc:
                bc = c5
                bh = hs.copy()

    # Phase 5: Final SLSQP polish
    if time.time() < t0 + budget_s - 5:
        for tv in [2000.0, 6000.0]:
            _opt_t = tv
            if time.time() > t0 + budget_s - 5:
                break
            try:
                r = minimize(_obj, bh, jac=_jacobian, method='SLSQP',
                             bounds=bnds, constraints=cons,
                             options={'maxiter': 1500, 'ftol': 1e-15})
                hc = proj(np.clip(r.x, 0.0, 1.0), n_target)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
            except Exception:
                pass

    # Final nudge with tiny coordinate adjustments
    if time.time() < t0 + budget_s - 2:
        bh, _ = coord_descend(bh, 1000.0, 3)

    bh = proj(np.clip(bh, 0.0, 1.0), n_target)
    bc = compute_c5(bh)
    return bh, bc, n_target
