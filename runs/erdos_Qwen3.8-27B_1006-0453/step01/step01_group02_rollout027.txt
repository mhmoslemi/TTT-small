```python
def pair_swap_search(h, N, dx, t0, budget_s, rng, max_iters=100000):
    """Local search via pair swaps with simulated annealing."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    T = 0.005  # initial temperature
    
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 5:
            break
        # Pick two random positions
        i, j = rng.choice(N, size=2, replace=False)
        if i == j:
            continue
        
        # Compute new C5
        h_new = h.copy()
        h_new[i], h_new[j] = h[j], h[i]
        new_c5 = compute_c5(h_new, dx)
        
        # Accept/reject
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        
        # Cool down
        T *= 0.9999
    
    return best_h, best_c5
```

```python
def grad_lag(h, k, N):
    """Gradient of corr[k] w.r.t. h, where k is the lag (can be negative)."""
    g = np.zeros(N)
    if k >= 0:
        if k < N:
            n = N - k
            g[:n] += 1.0 - h[k:k+n]
            g[k:] -= h[:n]
    else:
        l = -k
        if l < N:
            n = N - l
            g[l:] += 1.0 - h[:n]
            g[:n] -= h[l:l+n]
    return g

def adam_optimize(h, N, target, dx, t0, time_limit, rng, lr0=0.05, tau0=3.0, n_iter=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    tau = tau0
    b1, b2, eps = 0.8, 0.999, 1e-8
    mbuf = np.zeros(N)
    vbuf = np.zeros(N)
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1-h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Softmax weights (sharp)
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w /= ws
        
        # Threshold for active lags
        thresh = 1.0 / (2 * N)
        active = np.where(w > thresh)[0]
        
        grad = np.zeros(N)
        for idx in active:
            lag = int(idx) - (N - 1)
            grad += w[idx] * grad_lag(h, lag, N)
        
        # Adam update
        t = it + 1
        mbuf = b1 * mbuf + (1 - b1) * grad
        vbuf = b2 * vbuf + (1 - b2) * grad * grad
        mhat = mbuf / (1 - b1**t)
        vhat = vbuf / (1 - b2**t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + eps)

        if c5_new <= c5 + 1e-10:
            h = h_new
        else:
            lr *= 0.9

            lr *= 0.85
            tau *= 1.5
        if (it + 1) % 1000 == 0:
            tau *= 2.0
        
        if lr < 1e-9:

    return best_h, best_c5

def pair_swap(h, N, target, dx, t0, time_limit, rng, n_swaps=200000):
```

```python
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        if it % 2000 == 0:
            T *= 0.92
    return best_h, best_c5

def guided_swap(h, N, target, dx, t0, t_limit, rng, n_max=200000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    T = 0.002
    it = 0
    while time.time() < t_limit and it < n_max:
        it += 1
        if rng.random() < 0.4:
            corr = np.correlate(h, 1 - h, mode='full')
            k_star = int(np.argmax(corr)) - (N - 1)
            if k_star >= 0 and k_star < N:
                mask = (h[:N - k_star] > 0.3) & (h[k_star:] < 0.7)
            elif -k_star < N:
                l = -k_star
                mask = np.zeros(N, dtype=bool)
                mask[l:] = (h[l:] > 0.3) & (h[:N - l] < 0.7)
            else:
                mask = None
            if mask is not None and mask.any():
                i = rng.choice(np.where(mask)[0])
                j = rng.randint(N)
                if i == j:
                    j = (j + 1) % N
            else:
                i, j = rng.randint(0, N, 2)
                if i == j:
                    continue
        else:
            i, j = rng.randint(0, N, 2)
            if i == j:
                continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - curr_c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        if it % 3000 == 0:
            T *= 0.9
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    def rem():
        return budget_s - (time.time() - t0)

    sizes = sorted(set([n0, 96, 192, 384]))

    for N in sizes:
        if rem() < 40:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        if N == n0:
            h_init = h0.copy()
        else:
            h_init = np.interp(xN, x0, h0)
            h_init = project_sum(h_init, target)
        alloc = rem() * 0.75
        # Phase 1: Adam
        tl1 = t0 + alloc * 0.35
        h1, c1 = adam_optimize(h_init.copy(), N, target, dx, t0, tl1, rng, lr0=0.04, tau0=3.0, topk=5)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
        if rem() < 30:
            break
        # Phase 2: Pair swap
        tl2 = t0 + alloc * 0.60
        h2, c2 = pair_swap(h1.copy(), N, target, dx, t0, tl2, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
        if rem() < 30:
            break
        # Phase 3: Guided swap
        source = h2 if c2 <= c1 else h1
        tl3 = t0 + alloc * 0.80
        h3, c3 = guided_swap(source.copy(), N, target, dx, t0, tl3, rng)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
        if rem() < 25:
            break
        # Phase 4: Final Adam polish
        tl4 = t0 + alloc * 0.95
        polish_src = h3 if c3 <= c2 else h2
        h4, c4 = adam_optimize(polish_src.copy(), N, target, dx, t0, tl4, rng, lr0=0.02, tau0=6.0, topk=3)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()

    # Final restarts on best N
    N_best = len(best_h)
    dx_best = 2.0 / N_best
    target_best = N_best / 2.0
    restarts = 3 if rem() > 60 else (1 if rem() > 25 else 0)
    for r in range(restarts):
        if rem() < 15:
            break
        h_r = best_h + rng.randn(N_best) * 0.025
        h_r = project_sum(h_r, target_best)
        tl = t0 + rem() * 0.45
        h_opt, c_opt = adam_optimize(h_r, N_best, target_best, dx_best, t0, tl, rng, lr0=0.03, tau0=4.0, topk=4)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
        if rem() < 12:
            break
        tl2 = t0 + rem() * 0.55
        h_s, c_s = pair_swap(h_opt.copy(), N_best, target_best, dx_best, t0, tl2, rng, n_max=100000)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
```
