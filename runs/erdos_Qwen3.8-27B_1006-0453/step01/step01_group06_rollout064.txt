```python
def topk_smooth_c5(h, T, k):
```

```python
def binary_local_search(h, max_iters=5000, timeout=30):
    """Swap-based local search on binary sequence"""
    n = len(h)
    h = h.copy()
    # Ensure binary
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) != n//2:
        # Make it binary with n/2 ones
        # ... handle this
        return h
    
    best_c5 = compute_c5(h)
    no_improve = 0
    t_start = time.time()
    
    for it in range(max_iters):
        if time.time() - t_start > timeout:
            break
        # Pick random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        # Perform swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            no_improve = 0
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
            no_improve += 1
            if no_improve > 200:
                break
    
    # Convert to continuous for further optimization
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def topk_smooth(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k:]
    top = corr[idx]
    m = top[-1]
    val = m + np.log(np.sum(np.exp(T * (top - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def analytic_subgrad(h, tol_frac=0.05):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = tol_frac * max_val if max_val > 0 else 1e-6
    active_idx = np.where(corr >= max_val - tol)[0]
    lags = active_idx - (n - 1)
    
    if len(lags) == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for lag in lags:
        # Term 1: (1-h[j-lag]) for valid j (j-lag in [0,n-1])
        j_start = max(0, lag)
        j_end = min(n, n + lag)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            grad[js] += (1.0 - h[js - lag])
        
        # Term 2: -h[j+lag] for valid j (j+lag in [0,n-1])
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)
        if j_start2 < j_end2:
            js2 = np.arange(j_start2, j_end2)
            grad[js2] -= h[js2 + lag]
    
    grad /= len(lags)
    return grad

def binary_sa(h, n, max_iters=10000, t_start_sa=0.1, t_end_sa=0.001, timeout=30.0):
    """Simulated annealing on binary configurations"""
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n//2:
        excess = ones[len(n//2):]
        h[excess] = 0
        zeros = np.where(h == 0)[0]
    elif len(ones) < n//2:
        deficit = zeros[:n//2 - len(ones)]
        h[deficit] = 1
        ones = np.where(h == 1)[0]
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    t0 = time.time()
    
    for it in range(max_iters):
        if time.time() - t0 > timeout:
            break
        # Adaptive temperature
        frac = it / max_iters
        T = t_start_sa * (t_end_sa / t_start_sa) ** frac
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        
        if new_c5 < cur_c5 or np.random.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(int(seed))
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    def remaining_frac():
        return max(0, 1.0 - elapsed() / budget_s)
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
    else:
        init = None
    
    best_h = None
    best_c5 = np.inf
    
    # Try multiple n values
    n_list = [60, 80, 100, 120, 150]
    if init is not None:
        n_list = [len(init), 80, 100, 120, 150]
        n_list = sorted(set(n_list))
    
    for n in n_list:
        if elapsed() > budget_s * 0.85:
            break
        
        # Multiple starts for this n
        n_starts = 3 if n <= 80 else 2
        
        for s in range(n_starts):
            if elapsed() > budget_s * 0.85:
                break
            
            # Create initial solution
            if s == 0 and init is not None and n == len(init):
                h0 = project_feasible(init.copy(), n)
            elif s == 0 and init is not None:
                # Interpolate
                x_old = np.linspace(0, 2, len(init))
                x_new = np.linspace(0, 2, n)
                h0 = np.interp(x_new, x_old, init)
                h0 = project_feasible(h0, n)
            else:
                # Random binary
                h0 = np.zeros(n)
                h0[np.random.choice(n, n//2, replace=False)] = 1.0
            
            # Phase A: Binary SA (short)
            time_per_sa = min(15.0, budget_s * remaining_frac() * 0.15)
            h_bin, c5_bin = binary_sa(h0, n, max_iters=20000, timeout=time_per_sa)
            
            # Phase B: Subgradient descent (continuous)
            h_sg = h_bin.astype(float)
            h_sg = project_feasible(h_sg, n)
            step = 0.1
            for sg_it in range(500):
                if elapsed() > budget_s * 0.85:
                    break
                grad = analytic_subgrad(h_sg, tol_frac=0.05)
                gn = np.linalg.norm(grad)
                if gn < 1e-10:
                    break
                h_new = h_sg - step * grad / gn
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h_sg) - 1e-10:
                    h_sg = h_new
                else:
                    step *= 0.8
                step = max(step, 0.001)
            
            # Phase C: SLSQP with top-k smoothing
            for T in [20, 50, 150]:
                if elapsed() > budget_s * 0.85:
                    break
                bnds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': sum_constraint}]
                k_smooth = min(10, 2*n-1)
                try:
                    res = minimize(
                        topk_smooth, h_sg.copy(), args=(T, k_smooth),
                        method='SLSQP', bounds=bnds, constraints=cons,
                        options={'maxiter': 300, 'ftol': 1e-13}
                    )
                    h_cand = project_feasible(np.clip(res.x, 0, 1), n)
                    c5_cand = compute_c5(h_cand)
                    if c5_cand < compute_c5(h_sg):
                        h_sg = h_cand
                except:
                    pass
            
            # Phase D: Nelder-Mead polish
            if elapsed() < budget_s * 0.85:
                try:
                    res = minimize(
                        compute_c5, h_sg.copy(),
                        method='Nelder-Mead',
                        options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12}
                    )
                    h_cand = project_feasible(np.clip(res.x, 0, 1), n)
                    c5_cand = compute_c5(h_cand)
                    if c5_cand < compute_c5(h_sg):
                        h_sg = h_cand
                except:
                    pass
            
            c5_final = compute_c5(h_sg)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_sg.copy()
    
    # Final polish
    if best_h is not None and elapsed() < budget_s * 0.95:
        n_cur = len(best_h)
        for T in [300, 800]:
            if elapsed() > budget_s * 0.97:
                break
            bnds = [(0.0, 1.0)] * n_cur
            cons = [{'type': 'eq', 'fun': sum_constraint}]
            try:
                res = minimize(
                    topk_smooth, best_h.copy(), args=(T, 5),
                    method='SLSQP', bounds=bnds, constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-14}
                )
                h_cand = project_feasible(np.clip(res.x, 0, 1), n_cur)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand
            except:
                pass
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_lse(h, T, k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k is not None and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        top = corr[idx]
    else:
        top = corr
    m = float(np.max(top))
    val = m + np.log(np.sum(np.exp(T * (top - m)))) / T
    return float(val * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = float(h.sum())
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = float(room.sum())
        if total < 1e-13:
            return h
        h += room * (diff / total)
    else:
        total = float(h.sum())
        if total < 1e-13:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_eq(x):
    return float(x.sum()) - len(x) / 2.0

def subgrad_max(h, tol_frac=0.05):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    tol = tol_frac * mx if mx > 0 else 1e-6
    active = np.where(corr >= mx - tol)[0]
    lags = active - (n - 1)
    if len(lags) == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for lag in lags:
        # (1-h[j-lag]) where 0<=j-lag<n
        js1 = np.arange(max(0, lag), min(n, n + lag))
        if len(js1) > 0:
            grad[js1] += (1.0 - h[js1 - lag])
        # -h[j+lag] where 0<=j+lag<n
        js2 = np.arange(max(0, -lag), min(n, n - lag))
        if len(js2) > 0:
            grad[js2] -= h[js2 + lag]
    grad /= len(lags)
    return grad
```
