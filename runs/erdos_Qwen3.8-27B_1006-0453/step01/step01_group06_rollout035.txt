import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / len(h))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, np.concatenate([h, [h[0]]]))

def coord_step(h, j, iters=55):
    n = len(h)
    dx = 2.0 / n
    hf = h.copy()
    hf[j] = 0.0
    c0 = np.correlate(hf, 1.0 - hf, mode='full') * dx
    hf[j] = 1.0
    c1 = np.correlate(hf, 1.0 - hf, mode='full') * dx
    sl = c1 - c0
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c0 + m1 * sl) < np.max(c0 + m2 * sl):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def slsqp_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_eq_jac(x):
    return np.ones(len(x))

def make_binary_start(n, order):
    h = np.zeros(n)
    half = n // 2
    if order == 0:
        h[:half] = 1.0
    elif order == 1:
        idx = np.arange(0, n, 2)[:half]
        h[idx] = 1.0
    elif order == 2:
        idx = np.arange(0, n, 3)[:half]
        h[idx] = 1.0
    else:
        perm = np.random.permutation(n)
        h[perm[:half]] = 1.0
    return h
