                pass

    n_list = sorted(set([best_n, pn, max(50, best_n // 2), min(560, best_n * 2), 140, 400]))
    n_list = [n for n in n_list if 32 <= n <= 600]
    n_list.sort(key=lambda n: (abs(n - best_n), n))

    for n in n_list:
        if time_left() < 40:
            break
        h_work = parent.copy() if n == pn else upsample_linear(best_h, n)
        bounds = list(zip(np.zeros(n), np.ones(n)))
        constr = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]

        for beta in [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]:
            if time_left() < 12:
                break
            iters = min(600, max(100, int(time_left() * 2.5)))
            try:
                res = minimize(obj_smooth, h_work, args=(n, beta),
                              method='SLSQP', bounds=bounds, constraints=constr,
                              options={'maxiter': iters, 'ftol': 1e-13})
                h_work = project_onto_constraint(np.clip(res.x, 0.0, 1.0), n)
            except Exception:
                pass
        c5_w = compute_c5(h_work)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n

        # Perturbed restart at same n
        if time_left() > 35:
            h_p = project_onto_constraint(h_work + rng.randn(n) * 0.008, n)
            for beta in [0.005, 0.001, 0.0003]:
                if time_left() < 12:
                    break
                try:
                    res_p = minimize(obj_smooth, h_p, args=(n, beta),
                                    method='SLSQP', bounds=bounds, constraints=constr,
                                    options={'maxiter': 400, 'ftol': 1e-13})
                    h_p = project_onto_constraint(np.clip(res_p.x, 0.0, 1.0), n)
                except Exception:
                    pass
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                best_n = n

    # Phase 3: Peak-targeting local search
    if time_left() > 15:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        target = n / 2.0
        steps_budget = int(time_left() * 35)

        for it in range(steps_budget):
            if time_left() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            found = False
            for d in [0.025, 0.01, 0.005, 0.002, 0.001, 0.0004]:
                for sgn in [1, -1]:
                    h_tr = h.copy()
                    h_tr[i] = np.clip(h[i] + sgn * d, 0.0, 1.0)
                    actual_d = h_tr[i] - h[i]
                    if abs(actual_d) < 1e-14:
                        continue
                    h_tr[j] = np.clip(h[j] - actual_d, 0.0, 1.0)
                    if abs(np.sum(h_tr) - target) > 1e-9:
                        continue
                    c = compute_c5(h_tr)
                    if c < cur - 1e-14:
                        h = h_tr
                        cur = c
                        found = True
                        break
                if found:
                    break
            # Occasionally try 3-index move
            if not found and it % 500 == 499 and time_left() > 10:
                idx3 = rng.choice(n, 3, replace=False)
                for d1 in [0.02, -0.02, 0.008, -0.008]:
                    for d2 in [0.02, -0.02, 0.008]:
                        h_tr = h.copy()
                        h_tr[idx3[0]] = np.clip(h[idx3[0]] + d1, 0.0, 1.0)
                        h_tr[idx3[1]] = np.clip(h[idx3[1]] + d2, 0.0, 1.0)
                        rem = -(h_tr[idx3[0]] - h[idx3[0]]) - (h_tr[idx3[1]] - h[idx3[1]])
                        h_tr[idx3[2]] = np.clip(h[idx3[2]] + rem, 0.0, 1.0)
                        if abs(np.sum(h_tr) - target) > 1e-9:
                            continue
                        c3 = compute_c5(h_tr)
                        if c3 < cur - 1e-14:
                            h = h_tr
                            cur = c3
                            break
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()

    # Phase 4: Final SLSQP polish
    if time_left() > 10:
        n = best_n
        bounds = list(zip(np.zeros(n), np.ones(n)))
        constr = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
        try:
            res_f = minimize(obj_smooth, best_h, args=(n, 0.00005),
                            method='SLSQP', bounds=bounds, constraints=constr,
                            options={'maxiter': 400, 'ftol': 1e-14})
            h_pol = project_onto_constraint(np.clip(res_f.x, 0.0, 1.0), n)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
                best_n = n
        except Exception:
            pass

    return best_h, best_c5, best_n
