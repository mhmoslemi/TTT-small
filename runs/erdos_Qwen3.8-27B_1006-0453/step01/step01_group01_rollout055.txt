    best = h.copy()
    best_c = compute_c5(best)

        corr = np.correlate(best, 1.0 - best, mode='full')
        worst_idx = np.argmax(corr)
        # Compute gradient of c[worst_idx] specifically
        k = worst_idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:n] += (1 - h[:n-k])
            g[:n-k] -= h[k:n]
        else:

            g[:n-p] += (1 - h[p:n])

        h_new = proj(best - 0.01 * g, n)

        if c_new < best_c:
            best = h_new
            best_c = c_new

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)

        else:

        h = np.clip(h, 0.0, 1.0)
