def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h = h + room * (diff / total)
    else:
        total = h.sum()
        if total > 1e-12:
            h = h * (1.0 + diff / total)
    return np.clip(h, 0.0, 1.0)

def active_grad_sum(h, k_active):

    im = i - s
    v1 = (ip >= 0) & (ip < n)
    v2 = (im >= 0) & (im < n)
    grad = np.zeros(n)
    grad[v1] += (1.0 - h[ip[v1]])
    grad[v2] -= h[im[v2]]
    return grad

def subgrad_local(h, n, max_iter=5000, init_step=0.01, tol=1e-10):

    step = init_step

        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

        thr = np.max(corr) * 0.98
        active = np.where(corr >= thr)[0]
        grad = np.zeros(n)
        for ka in active:
            grad += active_grad_sum(h, ka)

        h_new = h - step * grad
        h_new = project_feasible(h_new, n)

            h = h_new
            step = min(step * 1.02, 0.05)
        else:

            if step < 1e-8:
                break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w = w / w.sum()
    grad = np.zeros(n)
    i = np.arange(n)
    for k in range(len(corr)):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        ip = i + s
        im = i - s
        v1 = (ip >= 0) & (ip < n)
        v2 = (im >= 0) & (im < n)
        grad[v1] += w[k] * (1.0 - h[ip[v1]])
        grad[v2] -= w[k] * h[im[v2]]
    return grad * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h = h + room * (diff / total)
    else:
        total = h.sum()
        if total > 1e-12:
            h = h * (1.0 + diff / total)
    # Iterate projection a few times for tight feasibility
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 1e-14:
                h += room * (diff / total)
        else:
            total = h.sum()
            if total > 1e-14:
                h *= (1.0 + diff / total)
    return np.clip(h, 0.0, 1.0)

def _sum_fun(x):
    return x.sum() - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, T, n, maxiter=300):
    def f(x):
        return smooth_c5(x, T)
    def g(x):
        return smooth_c5_grad(x, T)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def active_grad_sum(h, k_list):
    n = len(h)
    i = np.arange(n)
    grad = np.zeros(n)
    for k in k_list:
        s = k - (n - 1)
        ip = i + s
        im = i - s
        v1 = (ip >= 0) & (ip < n)
        v2 = (im >= 0) & (im < n)
        grad[v1] += (1.0 - h[ip[v1]])
        grad[v2] -= h[im[v2]]
    return grad / len(k_list)

def subgrad_local(h, n, max_iter=5000, init_step=0.008):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    step = init_step
    stall = 0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = np.max(corr) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        thr = np.max(corr) * 0.97
        active = np.where(corr >= thr)[0]
        grad = active_grad_sum(h, active)
        h_new = project_feasible(h - step * grad, n)
        if compute_c5(h_new) <= c5_now + 1e-13:
            h = h_new
            step = min(step * 1.03, 0.04)
        else:
            step *= 0.75
            if step < 1e-9:
                break
    return best, best_c5

def penalized_c5(h, penalty=3000.0):
    n = len(h)
    c5 = compute_c5(h)
    c5 += penalty * abs(h.sum() - n / 2.0)
    c5 += penalty * (np.sum(np.maximum(0.0, h - 1.0)) + np.sum(np.maximum(0.0, -h)))
    return c5

def nm_optimize(h, n, maxiter=600):
    try:
        res = minimize(penalized_c5, h, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11, 'maxfev': maxiter * 3})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def upsample(h, factor):
    return project_feasible(np.repeat(h, factor), len(h) * factor)
