import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = np.sum(free)
        if nf == 0:
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    scaled = (corr - m) / temp
    return (m + temp * np.log(np.sum(np.exp(scaled)))) * dx


def sum_con(x, n):
    return np.sum(x) - n / 2.0


def slsqp(x0, n, bounds, temp, maxiter=200):
    cons = {'type': 'eq', 'fun': sum_con, 'args': (n,)}
    res = minimize(smooth_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=[cons], args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x


def subgrad_polish(h, n, steps=500):
    """Direct subgradient on the max, with projection."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = 0.05
    for t in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5val = np.max(corr) * dx
        if c5val < best_c5:
            best_c5 = c5val
            best_h = h.copy()
        active = np.where(corr >= np.max(corr) - 1e-10)[0]
        grad = np.zeros_like(h)
        for k in active:
            lag = int(k)
            if lag >= 0:
                # corr[lag] = sum_i h[i] * (1-h[i+lag])
                for i in range(n - lag):
                    grad[i] += (1.0 - h[i + lag])
                    grad[i + lag] -= h[i]
            else:
                lag2 = -lag
                for i in range(lag2, n):
                    grad[i] += (1.0 - h[i - lag2])
                    grad[i - lag2] -= h[i]
        grad *= dx * len(active)
        lr_t = lr / (1.0 + 0.001 * t)
        h = h - lr_t * grad
        h = repair_h(h, n)
    return best_h, best_c5


def fourier_init(n, seed, n_modes=5):
    rng = np.random.RandomState(seed)
    x = np.zeros(n)
    freqs = np.arange(1, min(n // 2, n_modes) + 1)
    for f in freqs:
        amp = rng.uniform(0.02, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        x += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    h = 0.5 + x
    return repair_h(h, n)


def structured_init(n, seed):
    rng = np.random.RandomState(seed)
    pattern_len = rng.choice([3, 4, 5, 6, 9, 10, 15])
    base = rng.uniform(0.2, 0.8, pattern_len)
    h = np.tile(base, n // pattern_len + 1)[:n]
    noise = rng.normal(0, 0.05, n)
    h = h + noise
    return repair_h(h, n)
