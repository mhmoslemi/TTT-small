    return float(m + np.log(np.exp(s - m).sum()) / temp)

def subgrad_step(h, n, alpha, momentum, v):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    active = np.where(corr >= max_c - 1e-10)[0]
    grad = np.zeros(n)
    for m in active:
        p = m - (n - 1)
        # Efficient gradient computation
        for j in range(n):
            g = 0.0
            jp = j - p
            if 0 <= jp < n:
                g += (1.0 - h[jp])
            jp2 = j + p
            if 0 <= jp2 < n:
                g -= h[jp2]
            grad[j] += g
    grad /= len(active) * dx  # scale by dx since objective includes dx
    v = momentum * v + (1.0 - momentum) * grad
    h_new = h - alpha * v
    h_new = project_onto_feasible(h_new, n)
    return h_new, v

Wait, I realize the subgradient computation in a Python loop over j would be slow. Let me vectorize it.

For a given lag p (which is m - (n-1)):
∂corr[m]/∂h[j] = (1-h[j-p]) * 1[0≤j-p<n] - h[j+p] * 1[0≤j+p<n]

Let me write this vectorized:
```python
def corr_gradient(h, n, p):
    """Gradient of corr at lag p (where p = m - (n-1)) w.r.t. h"""
    grad = np.zeros(n)
    # First term: (1-h[j-p]) for j-p in [0, n-1]
    if p >= 0:
        # j-p >= 0 → j >= p; j-p < n → j < n+p (always true for j<n)
        j_start = p
        grad[j_start:] += (1.0 - h[:n - p])
    else:
        # j-p >= 0 → j >= p (always true since p<0 means j>=0>-p... wait)
        # j - p = j + |p|, which is in [0, n-1] when j in [0, n-1-|p|]
        j_end = n + p  # = n - |p|
        grad[:j_end] += (1.0 - h[-p:])  # h[0:p] reversed... no
        # h[j-p] = h[j+|p|], for j in [0, n-1-|p|]
        grad[:n - abs(p)] += (1.0 - h[abs(p):])
    
    # Second term: -h[j+p] for j+p in [0, n-1]
    if p >= 0:
        # j+p < n → j < n-p
        grad[:n - p] -= h[p:]
    else:
        # j+p >= 0 → j >= -p
        grad[-p:] -= h[:p]  # h[j+p]
