        grad[lag:] += (1.0 - h[:m]) * dx
        grad[:m] -= h[lag:] * dx
    else:
        # lag < 0, let a = -lag
        a = -lag
        # g = sum_{n=a}^{n-1} h[n]*(1-h[n-a]) * dx
        # = sum_{i=0}^{n-1-a} h[i+a]*(1-h[i]) * dx  (same as positive lag a!)
        m = n - a
        grad[a:] += (1.0 - h[:m]) * dx
        grad[:m] -= h[a:] * dx
    return grad

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):

    h_new = np.interp(x_new, x_old, h, period=2.0)

import numpy as np
import time
import math
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = 0.5 * n
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
    h = np.clip(h, 0, 1)
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.max(corr)
    if T < 1e-12:
        return c
    shifted = corr - c
    m = np.log(np.sum(np.exp(shifted / T)))
    return c + T * m

def grad_at_lag(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    a = abs(lag)
    m = n - a
    if m <= 0:
        return grad
    grad[a:] += (1.0 - h[:m]) * dx
    grad[:m] -= h[a:] * dx
    return grad

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.max(corr)
    idx_max = int(np.argmax(corr))
    if T < 1e-10:
        lag = idx_max - (n - 1)
        return grad_at_lag(h, lag, dx)
    shifted = corr - c
    weights = np.exp(shifted / T)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        if weights[k_idx] < 1e-8:
            continue
        lag = k_idx - (n - 1)
        grad += weights[k_idx] * grad_at_lag(h, lag, dx)
    return grad

def upsample_fn(h, new_n):
    old_n = len(h)

    return fix_sum(h_new, new_n)

def downsample_fn(h, new_n):
    return upsample_fn(h, new_n)

def lbfbs_optimize(h, n, T, maxiter):
    res = opt.minimize(
        soft_max_obj, h.copy(),
        args=(T,),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9}
    )
    return fix_sum(res.x, n)

def subgradient_loop(h, n, steps, alpha0):
    h = h.copy()
    alpha = alpha0
    c5 = compute_c5(h)
    for i in range(steps):
        grad = soft_max_grad(h, 1e-6)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            alpha *= 0.97

            below = ~above
            h[above] += 0.008 * (1.0 - h[above])
            h[below] -= 0.008 * h[below]

            h = fix_sum(h, n)
