        if np.any(mask):
            h[mask] += err / (np.sum(mask) * dx)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_tk(h, temp, ntop):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if ntop >= len(corr):
        cmax = np.max(corr)
        sm = cmax + temp * np.log(np.sum(np.exp((corr - cmax) / temp)))
    else:
        idx = np.argpartition(corr, -ntop)[-ntop:]
        topk = corr[idx]
        cmax = np.max(topk)
        sm = cmax + temp * np.log(np.sum(np.exp((topk - cmax) / temp)))
    pen = (np.sum(h) * dx - 1.0) ** 2
    return sm + 0.3 * pen

def repair(h):
    n = len(h)
    dx = 2.0 / n
    s = np.sum(h) * dx
    if s > 1e-15:
        h = h / s
    h = np.clip(h, 0.0, 1.0)
    err = 1.0 - np.sum(h) * dx
    if abs(err) > 1e-12:
        mask = (h < 1.0) if err > 0 else (h > 0.0)
        if np.any(mask):
