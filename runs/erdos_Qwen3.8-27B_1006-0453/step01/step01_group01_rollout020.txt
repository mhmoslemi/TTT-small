def grad_smooth_max(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = t * (corr - m)
    exp_s = np.exp(s)
    sum_exp = exp_s.sum()
    weights = exp_s / sum_exp
    obj = (m + np.log(sum_exp) / t) * dx
    
    one_h = 1.0 - h
    r_idx = np.arange(2 * n - 1)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j + r_idx - (n - 1)
        v1 = (idx1 >= 0) & (idx1 < n)
        if v1.any():
            grad[j] = np.dot(weights[v1], one_h[idx1[v1]])
        idx2 = j + (n - 1) - r_idx
        v2 = (idx2 >= 0) & (idx2 < n)
        if v2.any():
            grad[j] -= np.dot(weights[r_idx[v2]], h[idx2[v2]])
    grad *= dx
    return obj, grad

def coord_optimize(h, n):
    """One sweep of coordinate descent."""
    n_lag = 2 * n - 1
    dx = 2.0 / n
    improvement = 0.0
    for j in range(n):
        # Compute d_r for coordinate j
        r_idx = np.arange(n_lag)
        idx1 = j + r_idx - (n - 1)
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + (n - 1) - r_idx
        v2 = (idx2 >= 0) & (idx2 < n)
        
        d = np.zeros(n_lag)
        if v1.any():
            d[v1] = 1.0 - h[idx1[v1]]
        if v2.any():
            d[v2] -= h[idx2[v2]]
        
        # Current corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        a = corr - h[j] * d
        
        # Evaluate f(h') = max(a + h' * d) at a few points
        # Find optimal h'[j] in [0, 1]
        # f is convex piecewise linear
        # Check boundaries
        f0 = np.max(a)  # h'=0
        f1 = np.max(a + d)  # h'=1
        
        # Check if interior optimum exists
        # The optimum is where the "top" line changes from positive slope to negative
        # Simple approach: ternary search
        if f0 <= f1 and f0 <= np.max(a + 0.5 * d):
            new_val = 0.0
        elif f1 <= f0 and f1 <= np.max(a + 0.5 * d):
            new_val = 1.0
        else:
            # Binary search for the minimum
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid1 = lo + (hi - lo) / 3
                mid2 = hi - (hi - lo) / 3
                if np.max(a + mid1 * d) < np.max(a + mid2 * d):
                    hi = mid2
                else:
                    lo = mid1
            new_val = (lo + hi) / 2
        
        if abs(new_val - h[j]) > 1e-10:
            old_corr_max = np.max(corr)
            h[j] = new_val
            # Update corr efficiently (but for simplicity, recompute is O(n^2))
            # Actually, we can update: corr_new[r] = a[r] + new_val * d[r]
            corr = a + new_val * d
            improvement += max(0, old_corr_max - np.max(corr))
    
    return h, improvement

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return project_h(np.interp(xd, xs, h), nd)

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))
