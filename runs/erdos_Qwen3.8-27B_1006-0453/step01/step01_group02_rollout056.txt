import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_grad(h, w, N):
    grad = np.zeros(N)
    for p in range(2 * N - 1):
        wp = w[p]
        if wp < 1e-14:
            continue
        shift = p - (N - 1)
        if shift >= 0:
            nv = N - shift
            if nv > 0:
                grad[:nv] += wp * (1.0 - h[shift:shift + nv])
                grad[shift:] -= wp * h[:nv]
        else:
            L = -shift
            nv = N - L
            if nv > 0:
                grad[L:] += wp * (1.0 - h[:nv])
                grad[:nv] -= wp * h[L:L + nv]
    return grad

    v_buf = np.zeros(N)

            break

            best_c5 = c5

            last_improve = it

        sc -= sc.max()
        w = np.exp(sc)
        wsum = w.sum()
        if wsum > 1e-30:

        else:
