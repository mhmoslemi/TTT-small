But I need to be careful about numerical stability. The standard trick is:

= T * log(Σ exp(c_i/T)) - m + m

    best_c5 = compute_c5(best_h, best_n)

    # Phase 1: LSE temperature annealing (decreasing temp = sharpening)
    for temp in [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05]:
        if time.time() - t0 > budget_s - 350:
            break
        rem = budget_s - 350 - (time.time() - t0)
        mi = min(500, max(100, int(rem * 5)))
        ht = slsqp_lse(best_h, best_n, temp, mi)
        ht = repair_h(ht, best_n)
        c5t = compute_c5(ht, best_n)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 2: Power-sum refinement
    for p in [50, 100, 200, 500, 1000]:
        if time.time() - t0 > budget_s - 250:
            break
        rem = budget_s - 250 - (time.time() - t0)
        mi = min(400, max(80, int(rem * 4)))
        ht = slsqp_psum(best_h, best_n, p, mi)
        ht = repair_h(ht, best_n)
        c5t = compute_c5(ht, best_n)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 3: Multi-start diverse initializations
    n_trials = 45
    for trial in range(n_trials):
        if time.time() - t0 > budget_s - 150:
            break
        mode = trial % 5
        if mode == 0:
            sc = 0.1 * np.exp(-trial / 18.0) + 0.015
            cand = best_h + np.random.uniform(-sc, sc, best_n)
        elif mode == 1:
            sl = np.random.randint(4, 12)
            st = np.random.randint(0, best_n - sl)
            cand = best_h.copy()
            cand[st:st + sl] = np.clip(1.0 - cand[st:st + sl], 0.0, 1.0)
        elif mode == 2:
            cand = fourier_init(best_n, rng, 2 + trial % 5)
        elif mode == 3:
            cand = np.zeros(best_n)
            idx = np.random.choice(best_n, best_n // 2, replace=False)
            cand[idx] = 1.0
        else:
            cand = np.zeros(best_n)
            nblk = 3 + trial % 4
            blk = best_n // nblk
            for bi in range(nblk):
                if bi % 2 == 0:
                    cand[bi * blk:(bi + 1) * blk] = 1.0

        cand = repair_h(cand, best_n)
        c5c = compute_c5(cand, best_n)
        if c5c < best_c5:
            best_c5, best_h = c5c, cand.copy()

        for (use_lse, temp, p) in [(True, 1.0, 0), (False, 0, 150)]:
            if time.time() - t0 > budget_s - 30:
                break
            rem = budget_s - 30 - (time.time() - t0)
            mi = min(200, max(50, int(rem * 3)))
            if use_lse:
                ht = slsqp_lse(cand, best_n, temp, mi)
            else:
                ht = slsqp_psum(cand, best_n, p, mi)
            ht = repair_h(ht, best_n)
            c5t = compute_c5(ht, best_n)
            if c5t < best_c5:
                best_c5, best_h = c5t, ht.copy()
            if c5t < c5c:
                cand, c5c = ht, c5t

    # Phase 4: Simulated annealing fine-tuning
    if time.time() - t0 < budget_s - 80:
        sa_limit = time.time() + min(120, budget_s - 100 - (time.time() - t0))
        sa_limit = min(sa_limit, t0 + budget_s - 40)
        ht, c5s = sa_refine(best_h, best_n, sa_limit)
        if c5s < best_c5:
            best_c5, best_h = c5s, ht

    # Phase 5: Nelder-Mead polish
    if time.time() - t0 < budget_s - 30:
        rem = budget_s - 30 - (time.time() - t0)
        mi = min(3000, max(1000, int(rem * 20)))
        ht = nm_opt(best_h, best_n, 500, mi)
        ht = repair_h(ht, best_n)
        c5t = compute_c5(ht, best_n)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 6: Higher resolution (90 points)
    if time.time() - t0 < budget_s - 40:
        n2 = 90
        xp = np.linspace(0.0, 1.0, best_n, endpoint=False)
        xn = np.linspace(0.0, 1.0, n2, endpoint=False)
        hi = repair_h(np.interp(xn, xp, best_h), n2)
        c5i = compute_c5(hi, n2)
        for p in [100, 300]:
            if time.time() - t0 > budget_s - 10:
                break
            rem = budget_s - 10 - (time.time() - t0)
            mi = min(250, max(60, int(rem * 2)))
            ht = slsqp_psum(hi, n2, p, mi)
            ht = repair_h(ht, n2)
            c5t = compute_c5(ht, n2)
            if c5t < best_c5:
                best_c5, best_h, best_n = c5t, ht, n2
            if c5t < c5i:
                hi, c5i = ht, c5t

    # Final SA polish
    if time.time() - t0 < budget_s - 5:
        sa_lim = t0 + budget_s - 1
        ht, c5s = sa_refine(best_h, best_n, sa_lim)
        if c5s < best_c5:
            best_c5, best_h = c5s, ht

    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
