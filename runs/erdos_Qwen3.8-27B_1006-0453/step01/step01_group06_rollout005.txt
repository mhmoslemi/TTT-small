    return h.sum() - len(h) / 2.0

def slsqp_optimize(h, T, maxiter=500):
    n = len(h)
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5, h, args=(T,),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})

    c5v = compute_c5(hc)
    return hc, c5v

        # Quick hill climb on best from this start
        if elapsed() < t_end4:
            try:
                h_new, c5_new = hill_climb(h0, 2000, 0.03, rng)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 5: Final high-T SLSQP polish (12% budget)
    t_end5 = t0 + budget_s * 0.97
    for T in [1000, 3000, 8000]:
        if elapsed() > t_end5:
            break
        try:
            h_new, c5_new = slsqp_opt(best_h, T, 800)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Final micro hill-climb
    if elapsed() < budget_s * 0.98:
        try:
            h_new, c5_new = hill_climb(best_h, 2000, 0.005, rng)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
