import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-12:
            return h
        h = h + room * (diff / total)
    else:
        total = s
        if total < 1e-12:
            return h
        h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')

def sum_con(h):
    return h.sum() - len(h) / 2.0

def local_search(h, n, n_iter, rng, deltas=None):
    if deltas is None:
        deltas = [0.1, -0.1, 0.2, -0.2, 0.25, -0.25, 0.5, -0.5]
    bc = compute_c5(h)
    hc = h.copy()
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        oi, oj = hc[i], hc[j]
        bd, bcn = 0.0, bc + 1e-12
        for d in deltas:
            hi, hj = oi + d, oj - d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                hc[i], hc[j] = hi, hj
                c = compute_c5(hc)
                if c < bcn:
                    bcn, bd = c, d
        if bd != 0.0:
            hc[i], hc[j] = oi + bd, oj - bd
            bc = bcn
        else:
            hc[i], hc[j] = oi, oj
    return hc, bc

def slsqp_polish(h, T, maxiter=400):
    n = len(h)
    try:
        r = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': sum_con}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = project_feasible(np.clip(r.x, 0.0, 1.0), n)
        return hc, compute_c5(hc)
    except Exception:
        return h, compute_c5(h)

def make_binary_start(n, rng):
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def make_quadratic_start(n, p=None, rng=None):
    if p is None:
        primes = [53, 61, 67, 73, 79, 97, 113, 127, 151]
        p = primes[min(n, 512) % len(primes)]
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i * i) % p) < p // 2 else 0.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_binary_start(60, rng)
    n0 = len(best_h)
    best_h = project_feasible(best_h, n0)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Heavy local search on initial (coarse + fine)
    h1, c1 = local_search(best_h, n0, 60000, rng, [0.2, -0.2, 0.4, -0.4, 0.6, -0.6, 1.0, -1.0])
    h1, c1 = local_search(h1, n0, 60000, rng, [0.05, -0.05, 0.1, -0.1, 0.15, -0.15])
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    if elapsed() < budget_s * 0.15:
        for T in [200, 1000, 5000]:
            if elapsed() > budget_s * 0.2:
                break
            hp, c5p = slsqp_polish(best_h, T, 500)
            if c5p < best_c5:
                best_h, best_c5 = hp, c5p

    # Phase 3: Multi-n exploration with diverse starts
    n_list = [60, 64, 72, 80, 96, 120, 160]
    for n_try in n_list:
        if elapsed() > budget_s * 0.70:
            break
        starts = [make_binary_start(n_try, rng), make_binary_start(n_try, rng)]
        if n_try % 2 == 0:
            starts.append(make_quadratic_start(n_try, rng=rng))
        for h0 in starts:
            if elapsed() > budget_s * 0.70:
                break
            ht, ct = local_search(h0, n_try, 20000, rng, [0.1, -0.1, 0.2, -0.2, 0.5, -0.5, 1.0, -1.0])
            ht, ct = local_search(ht, n_try, 20000, rng, [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2])
            if elapsed() < budget_s * 0.72:
                hp, c5p = slsqp_polish(ht, 500, 300)
                if c5p < ct:
                    ht, ct = hp, c5p
                hp, c5p = slsqp_polish(ht, 3000, 300)
                if c5p < ct:
                    ht, ct = hp, c5p
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 4: Deep local search on best solution
    nc = len(best_h)
    if elapsed() < budget_s * 0.80:
        h2, c2 = local_search(best_h, nc, 100000, rng, [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.25, -0.25])
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
        h2, c2 = local_search(best_h, nc, 100000, rng, [0.02, -0.02, 0.05, -0.05, 0.08, -0.08, 0.1, -0.1])
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 5: Perturbation restarts from best
    if elapsed() < budget_s * 0.88:
        for restart in range(5):
            if elapsed() > budget_s * 0.88:
                break
            scale = 0.3 * (1.0 - 0.12 * restart)
            h_pert = best_h + rng.uniform(-scale, scale, size=nc)
            h_pert = project_feasible(h_pert, nc)
            ht, ct = local_search(h_pert, nc, 30000, rng, [0.1, -0.1, 0.2, -0.2, 0.5, -0.5])
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 6: Final SLSQP polish sequence
    if elapsed() < budget_s * 0.94:
        for T in [100, 500, 2000, 10000]:
            if elapsed() > budget_s * 0.96:
                break
            hp, c5p = slsqp_polish(best_h, len(best_h), T, 600)
            if c5p < best_c5:
                best_h, best_c5 = hp, c5p
        # One more local search after SLSQP
        if elapsed() < budget_s * 0.97:
            h2, c2 = local_search(best_h, len(best_h), 50000, rng, [0.02, -0.02, 0.05, -0.05, 0.1, -0.1])
            if c2 < best_c5:
                best_h, best_c5 = h2, c2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
