def project(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Bisection on lambda: x_i = clip(h_i - lambda, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - mid, 0, 1)
    # Fix any remaining error
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute among interior points
        interior = (x > 0.001) & (x < 0.999)
        if np.sum(interior) > 0:

    best = cur
    best_h = h.copy()
    step_sizes = [0.005, 0.01, 0.02, 0.002, 0.03, 0.008]

    for step in range(max_steps):
        if time.time() - t0 > time_limit:
            break
        # Informed move: target the max lag
        if step % 3 == 0:
            corr = compute_corr(h)
            k_max = np.argmax(corr)
            s = k_max - (n - 1)
            if s >= 0 and s < n:
                # Find i where h[i+s]*(1-h[i]) is large
                valid = np.arange(n - s)
                contrib = h[s:s+n-s] * (1.0 - h[:n-s])
                i = valid[np.argmax(contrib)]
                j = i + s
            elif s < 0:
                t = -s
                valid = np.arange(t, n)
                contrib = h[:n-t] * (1.0 - h[t:])
                i = valid[np.argmax(contrib)]
                j = i - t
            else:
                i = rng.randint(n)
                j = rng.randint(n)
            if i == j:
                i = (i + 1) % n
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue

        found = False
        for d in step_sizes:
            for sign in (1, -1):
                h_new = h.copy()
                vi = h[i] + sign * d
                vj = h[j] - sign * d
                if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                    continue
                h_new[i] = vi
                h_new[j] = vj
                c = compute_c5(h_new)
                if c < cur - 1e-13:
                    h = h_new
                    cur = c
                    if cur < best:
                        best = cur
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    _N = pn
    n = pn
    best_h = parent.copy()
    best_c5 = compute_c5(parent)

    # Phase 1: SLSQP with progressive sharpening
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for beta in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() - t0 > budget_s * 0.28:
            break
        try:
            res = minimize(soft_max_obj, best_h, args=(beta,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-14})
            h_new = np.clip(res.x, 0, 1)
            h_new += (n / 2.0 - np.sum(h_new)) / n
            h_new = np.clip(h_new, 0, 1)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 2: Subgradient descent with momentum
    if time.time() - t0 < budget_s * 0.60:
        best_h, best_c5 = subgradient_refine(
            best_h, n, 8000, budget_s * 0.60, t0, rng)

    # Phase 3: Second SLSQP pass after subgradient
    if time.time() - t0 < budget_s * 0.72 and best_n_check_ok(best_h, n):
        _N = n
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        for beta in [0.001, 0.0005]:
            if time.time() - t0 > budget_s * 0.72:
                break
            try:
                res = minimize(soft_max_obj, best_h, args=(beta,),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-14})
                h_new = np.clip(res.x, 0, 1)
                h_new += (n / 2.0 - np.sum(h_new)) / n
                h_new = np.clip(h_new, 0, 1)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 4: Aggressive coordinate descent
    if time.time() - t0 < budget_s * 0.95:
        n = len(best_h)
        remaining_frac = (budget_s * 0.95 - (time.time() - t0)) / max(budget_s, 1)
        max_steps = int(remaining_frac * 50000)
        best_h, best_c5 = coord_descent(best_h, n, rng, max_steps,
                                         budget_s * 0.95, t0)

    # Final sum correction
    n = len(best_h)
    best_h += (n / 2.0 - np.sum(best_h)) / n
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n

def best_n_check_ok(h, n):
    return abs(np.sum(h) - n / 2.0) < 1e-4
