import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_step(h, j, lam=500.0):
    """Exact 1D optimization for coordinate j, with sum penalty."""
    n = len(h)
    h0 = h.copy()
    h1 = h.copy()
    h0[j] = 0.0
    h1[j] = 1.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    c1 = np.correlate(h1, 1.0 - h1, mode='full')
    a = c1 - c0
    b = c0
    # minimize max_k(a[k]*x + b[k]) + lam*(x - h[j])^2  over x in [0,1]
    # This is convex, use ternary search
    lo, hi = 0.0, 1.0
    old = h[j]
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a * m1 + b) + lam * (m1 - old) ** 2
        f2 = np.max(a * m2 + b) + lam * (m2 - old) ** 2
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, n_sweeps=20, lam=500.0, time_limit=60.0, t0=0.0):
    """Full coordinate descent with periodic projection."""
    best = h.copy()
    best_val = compute_c5(h)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            old_val = h[j]
            new_val = coord_step(h, j, lam)
            if abs(new_val - old_val) > 1e-10:
                h[j] = new_val
                improved = True
        h = project_feasible(h, n)
        val = compute_c5(h)
        if val < best_val - 1e-12:
            best_val = val
            best = h.copy()
        if not improved:
            break
    return best, best_val

def active_set_slqp(h, n, top_k=5, T=200.0, maxiter=300):
    """SLSQP minimizing smooth max of top_k correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c[idx]
        mx = np.max(m)
        val = mx + np.log(np.sum(np.exp(T * (m - mx)))) / T
        return val * dx
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slqp_smooth(h, n, T, maxiter=400):

    except Exception:

Let me also think about whether the coordinate descent will actually help. The key insight is that the coordinate step is EXACT - it finds the optimal value for that coordinate given all others fixed. This is much better than a gradient step. With lam controlling the sum constraint, as lam→∞, we get exact coordinate descent on the feasible set (approximately, since we also project).

Actually, there's a subtlety. The lam penalty approach doesn't exactly enforce the sum constraint during the coordinate step. After each full sweep, we project. This means during the sweep, the sum might drift, and the coordinate steps are optimizing a slightly wrong objective. As lam gets large, this becomes less of an issue.

An alternative: after changing h[j], immediately adjust another coordinate to maintain the sum. But this breaks the 1D exactness.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_step(h, j, lam=5000.0):
    n = len(h)
    h0 = h.copy()
    h1 = h.copy()
    h0[j] = 0.0
    h1[j] = 1.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    c1 = np.correlate(h1, 1.0 - h1, mode='full')
    a = c1 - c0
    b = c0
    old = h[j]
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a * m1 + b) + lam * (m1 - old) ** 2
        f2 = np.max(a * m2 + b) + lam * (m2 - old) ** 2
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, n_sweeps=30, lam=5000.0, time_limit=60.0, t0=0.0):
    best = h.copy()
    best_val = compute_c5(h)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            old_val = h[j]
            new_val = coord_step(h, j, lam)
            if abs(new_val - old_val) > 1e-11:
                h[j] = new_val
                improved = True
        h = project_feasible(h, n)
        val = compute_c5(h)
        if val < best_val - 1e-13:
            best_val = val
            best = h.copy()
        if not improved:
            break
    return best, best_val

def active_set_obj(x, idx, dx, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c[idx]
    mx = np.max(m)
    val = mx + np.log(np.sum(np.exp(T * (m - mx)))) / T
    return val * dx

def active_set_slqp(h, n, top_k=5, T=200.0, maxiter=400):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(active_set_obj, h, args=(idx, dx, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slqp_smooth(h, n, T, maxiter=400):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    h = np.zeros(n)
    h[:n // 2] = 1.0
    return project_feasible(h, n)

def random_feasible(n, scale=0.3):
    h = 0.5 + scale * np.random.randn(n)
    return project_feasible(h, n)
