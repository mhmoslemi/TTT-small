import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def p_norm_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (np.sum(corr**p) / len(corr))**(1.0/p) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        idx = min(int(i * old_n / new_n), old_n - 1)
        h_new[i] = h[idx]
    return proj_constraint(h_new, new_n)

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def do_pnorm_slsqp(h0, n, p, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        p_norm_obj, h0, args=(p,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def sa_refine(h0, n, t_start, budget_s, max_steps, seed):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T_sa = 0.0005
    for step in range(max_steps):
        if time.time() - t_start > budget_s - 60:
            break
        n_blk = rng.randint(1, 6)
        idxs = rng.choice(n, size=n_blk, replace=False)
        h_try = h_cur.copy()
        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_blk) * 0.04, 0, 1)
        h_try = proj_constraint(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T_sa):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T_sa *= 0.997
    return h_best, c_best
