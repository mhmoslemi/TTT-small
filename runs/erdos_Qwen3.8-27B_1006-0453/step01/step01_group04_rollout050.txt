import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if np.sum(mask) == 0:
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def lbfgs_refine(h, t0, time_limit, temps=(0.04, 0.012, 0.003, 0.0008)):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    best_c5 = compute_c5(h)
    cur = h.copy()
    for temp in temps:
        if time.time() - t0 > time_limit:
            break
        remaining = time_limit - (time.time() - t0)
        if remaining < 3:
            break
        maxit = max(30, min(400, int(remaining / 0.05)))
        res = minimize(
            smooth_c5, cur, args=(temp, 500.0),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': maxit * 8}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        cur = nh
        if c5v < best_c5:
            best_c5 = c5v
    return cur, best_c5

def perturb_shift(h, rng):
    n = len(h)
    k = rng.randint(-n // 3, n // 3 + 1)
    return repair_h(np.roll(h, k))

def perturb_reflect(h):
    return repair_h(h[::-1].copy())

def perturb_block(h, rng):
    n = len(h)
    nb = rng.randint(3, 10)
    bl = n // nb
    if bl < 2:
        return repair_h(h)
    idx = np.arange(n)
    np.random.shuffle(idx[:nb * bl])
    h2 = h[:nb * bl][idx[:nb * bl]]
    h2 = np.concatenate([h2, h[nb * bl:]])
    return repair_h(h2)

def perturb_gaussian(h, sigma, rng):
    h2 = np.clip(h + sigma * rng.standard_normal(len(h)), 0.0, 1.0)
    return repair_h(h2)

def perturb_mix(h, rng):
    a = repair_h(1.0 - h)
    b = perturb_reflect(h)
    alpha = rng.uniform(0.2, 0.5)
    return repair_h(alpha * a + (1 - alpha) * b)
