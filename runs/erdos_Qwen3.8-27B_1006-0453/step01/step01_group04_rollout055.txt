import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_lse(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(corr - cmax, -500.0, 500.0)
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def obj_pnorm(h, p, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    pos = np.maximum(corr, 0.0)
    pnorm = np.sum(pos ** p) ** (1.0 / p)
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return pnorm * dx + penalty * (integral - 1.0) ** 2

def repair_h(h):
    n = len(h)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return h
    h2 = h * (target / s)
    h2 = np.clip(h2, 0.0, 1.0)
    diff = target - np.sum(h2)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h2 < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                inc = min(diff / cnt, 1.0)
                h2[mask] = np.minimum(1.0, h2[mask] + inc)
        else:
            mask = h2 > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                dec = min(-diff / cnt, 1.0)
                h2[mask] = np.maximum(0.0, h2[mask] - dec)
    return h2

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def opt_lse_pass(h, deadline, temps, maxits):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    best = compute_c5(cur)
    for i in range(len(temps)):
        if time.time() > deadline:
            break
        temp = temps[i]
        mi = maxits[i] if i < len(maxits) else 150
        res = minimize(
            obj_lse, cur, args=(temp, 1000.0),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': mi, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best:
            best = c5v
        cur = nh
    return cur, best

def opt_pnorm_pass(h, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    best = compute_c5(cur)
    pvals = [16, 64, 256, 1024, 4096]
    for p in pvals:
        if time.time() > deadline:
            break
        res = minimize(
            obj_pnorm, cur, args=(p, 1000.0),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best:
            best = c5v
        cur = nh
    return cur, best
