def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0

        if cnt == 0:

            # d_k = (1-h[j-lag]) - h[j+lag] (with boundary handling)

            # d_k for this j
            # d_k = (1-h[j-lags]) for valid (j-lags) - h[j+lags] for valid (j+lags)
            d = np.zeros(2*n-1)

                # (1-h[j-lag]) valid when 0 <= j-lag < n
                if 0 <= j - lag < n:
                    d[idx] += (1.0 - h[j - lag])
                # -h[j+lag] valid when 0 <= j+lag < n  
                if 0 <= j + lag < n:

            c = corr - d * h[j]
            
            # Minimize max_k (c_k + d_k * t) over t in [0,1]
            # Ternary search
            a, b = 0.0, 1.0

                f1 = np.max(c + d * m1)

∂g_k/∂h[j] = (1-h[j-lag]) for 0≤j-lag<n minus h[j+lag] for 0≤j+lag<n

This can be computed as:
grad[j] = (1-h[j-lag])*I(0≤j-lag<n) - h[j+lag]*I(0≤j+lag<n)

For a fixed lag, this is O(n) using array slicing.

So the gradient of g_k for a specific k is O(n). If we use the subgradient (gradient of the active constraint), the step is:
1. Find k* = argmax g_k: O(n) (well, O(n^2) to compute all g_k, but O(1) to find the max)
2. Compute ∇g_k*: O(n)
3. Take a step: h ← h - α * ∇g_k*, then project

But wait, we need to handle the constraint sum(h) = n/2. The projection handles this.

The issue with pure subgradient is that it can be slow (O(1/ε) iterations for O(ε) accuracy). But combined with a decreasing step size and Nesterov acceleration, it can be competitive.

Let me try a different approach: **convex optimization with CVXPY**.

The problem is: minimize t subject to:
- For all k: sum_i h[i+k-(n-1)] * (1-h[i]) ≤ t (for valid i)
- 0 ≤ h ≤ 1
- sum(h) = n/2

But this has O(n^2) constraints (one for each k) and each constraint is bilinear in h. So it's NOT a convex problem! The product h[i+k]*(1-h[i]) is not convex in h.

So CVXPY won't directly work. But we could relax: since h ∈ [0,1], we have h*(1-h) ≤ 1/4 and the bilinear term makes things complex.

Let me implement the analytical gradient for the soft-max objective.

For f(h) = (max_k g_k + (1/t) log sum_k exp(t(g_k - max_j g_j))) * dx

Let w_k = exp(t(g_k - max)) / sum_j exp(t(g_j - max)) (softmax weights)

Then ∂f/∂h[j] = dx * sum_k w_k * ∂g_k/∂h[j]

And ∂g_k/∂h[j] = (1-h[j-lag_k]) * I(0≤j-lag_k<n) - h[j+lag_k] * I(0≤j+lag_k<n)

where lag_k = k - (n-1) for k = 0, ..., 2n-2.

This can be computed efficiently:
- For each k, the contribution to the gradient is w_k * [(1-h[j-lag_k]) - h[j+lag_k]] (with boundary)
- Sum over k with significant weight

If we only keep the top few weights (those above 1e-6), this is fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()
    exps = np.exp(s)
    w = exps / np.sum(exps)
    val = (mx + np.log(np.sum(exps)) / t) * dx
    
    # Compute gradient: sum_k w_k * d(g_k)/d(h[j])
    # d(g_k)/d(h[j]) = (1-h[j-lag_k]) - h[j+lag_k] (with bounds)
    # lag_k = k - (n-1)
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)
    
    # For efficiency, only process k with w_k > threshold
    thresh = 1e-8
    active = w > thresh
    if not np.any(active):
        w_active = w
        lag_active = lags
    else:
        w_active = w[active]
        lag_active = lags[active]
    
    for idx in range(len(w_active)):
        lag = lag_active[idx]
        w_k = w_active[idx]
        # (1-h[j-lag]) for 0 <= j-lag < n → j from max(0,lag) to min(n-1, n-1+lag)
        j_lo = max(0, lag)
        j_hi = min(n-1, n-1+lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w_k * (1.0 - h[j_lo-lag:j_hi+1-lag])
        # -h[j+lag] for 0 <= j+lag < n → j from max(0,-lag) to min(n-1, n-1-lag)
        j_lo2 = max(0, -lag)
        j_hi2 = min(n-1, n-1-lag)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] += w_k * (-h[j_lo2+lag:j_hi2+1+lag])
    
    grad *= dx
    return val, grad

def ceq(x):
    return np.array([np.sum(x) - len(x)/2.0])

def ceqj(x):
    return np.ones((1, len(x)))

                bc = c5

            pass

import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_val(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()
    return float((mx + np.log(np.sum(np.exp(s))) / t) * dx)

def obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()
    exps = np.exp(s)
    w = exps / np.sum(exps)
    val = (mx + np.log(np.sum(exps)) / t) * dx
    grad = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    active = w > 1e-9
    wa = w[active]
    la = lags[active]
    for idx in range(len(wa)):
        lag = la[idx]
        wk = wa[idx]
        j1lo = max(0, lag)
        j1hi = min(n - 1, n - 1 + lag)

            grad[j2lo:j2hi+1] += wk * (-h[j2lo+lag:j2hi+1+lag])

    return float(val), grad

    return np.ones((1, len(x)))

    def f(x):
        return obj_val(x, t)

    def g(x):
        return obj_grad(x, t)[1]

    r = minimize(f, h0, jac=g, method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    hc = proj(np.clip(r.x, 0.0, 1.0), n)
    return hc

    h = h0.copy()

    lags = np.arange(2 * n - 1) - (n - 1)

        h_try = proj(base + rng.randn(n) * noise_scale, n)
        try:
            hc = slsqp_optimize(h_try, 300, maxiter=400, ftol=1e-13)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc
                base = bh.copy()
        except Exception:
            pass

    # Phase 4: Final high-temperature refinement
    if time.time() < t0 + budget_s - 10:
        try:
            hc = slsqp_optimize(bh, 2000, maxiter=500, ftol=1e-15)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc
        except Exception:
            pass

    # Final subgradient pass
    if time.time() < t0 + budget_s - 3:
        bh, bc2 = subgrad_polish(bh, max_iter=2000, time_limit=t0 + budget_s - 3)
        if bc2 < bc:
            bc = bc2

    bh = proj(bh, n)
    return bh, compute_c5(bh), n
