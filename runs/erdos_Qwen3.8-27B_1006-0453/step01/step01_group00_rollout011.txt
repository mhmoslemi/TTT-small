import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        return c_max * dx
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def objective_penalized(h, T, penalty):
    n = len(h)
    f = smooth_max_objective(h, T)
    s = np.sum(h) - n * 0.5
    f += penalty * s * s
    return f

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

**Idea: Subgradient descent on the max**

After L-BFGS-B converges, the solution is at a local minimum of the smooth approximation. But the true objective is the max, which is non-smooth. We can do:
1. Compute corr = np.correlate(h, 1-h, 'full')
2. Find the active set: k values where corr[k] is close to max
3. Compute the "average gradient" over active k's
4. Take a small step in the negative gradient direction
5. Project back to feasible set (clip to [0,1], fix sum)

This is like a projected subgradient method on the max function.

**Idea: Coordinate descent on individual points**

For each point h[j], the correlation values that depend on h[j] are:
- corr[m] for m where h[j] appears as h[i+m-(n-1)] or in (1-h[i])

We could greedily update each h[j] to minimize the max correlation. This is like a greedy coordinate descent.

**Idea: Use the structure of the problem**

Notice that corr[k] = Σ_i h[i] * (1-h[i+k-(n-1)]). If h is a step function (values 0 or 1), then:
corr[k] = |A ∩ (A+k)^c| * dx = |A|*dx - |A ∩ (A+k)| * dx

where A = {i : h[i] = 1}. Since |A|*dx = 1 (the integral constraint), we have:
corr[k] = 1 - |A ∩ (A+k)| * dx

So minimizing max_k corr[k] is equivalent to maximizing min_k |A ∩ (A+k)| * dx.

For a two-valued function, this becomes a combinatorial problem. But h can take values in [0,1], giving more flexibility.

Instead of h[i] directly, use h[i] = sigmoid(β * v[i]) where v is a free vector and β controls sharpness. This ensures h ∈ (0,1) automatically. The integral constraint can be enforced by adjusting β or adding a DC offset.

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-12:

    sh = corr - cmax
    return (cmax + T * np.log(np.sum(np.exp(sh / T)))) * dx

    f = smooth_obj(h, T)
    s = np.sum(h) - n * 0.5
    return f + pen * s * s

    if abs(s - t) < 1e-10:

    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-12:
        return cmax * dx
    sh = corr - cmax
    return (cmax + T * np.log(np.sum(np.exp(sh / T)))) * dx

    n = len(h)
    f = smooth_obj(h, T)
    s = np.sum(h) - n * 0.5
    return f + pen * s * s

def fix_sum(h, n):
    t = n * 0.5
    s = np.sum(h)
    if abs(s - t) < 1e-12:
        return h
    h = h.copy()
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += (t - s) / cnt
    h = np.clip(h, 0, 1)

                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return fix_sum(res.x, n)

        return np.sum(x) - nn * 0.5

        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 2: Subgradient refinement
    if budget_s - (time.time() - t0) > 100:
        h_sg, c_sg = subgrad_refine(h_best, n_best, steps=150, lr=0.004, n_active=3)
        if c_sg < c5_best:
            h_best, c5_best = h_sg, c_sg
        h_opt = opt_lbfgs(h_best.copy(), n_best, 0.003, max_iter=200)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 3: Random restarts with varying noise
    for i in range(12):
        if budget_s - (time.time() - t0) < 180:
            break
        rng = np.random.RandomState(seed + 9000 + i * 137)
        sigma = 0.06 + 0.04 * (i % 4)
        h_p = np.clip(h_best + rng.randn(n_best) * sigma, 0, 1)
        h_p = fix_sum(h_p, n_best)
        h_opt = opt_lbfgs(h_p, n_best, 0.012, max_iter=150)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c
            # Quick subgradient from new best
            h_sg2, c_sg2 = subgrad_refine(h_best, n_best, steps=50, lr=0.003, n_active=3)
            if c_sg2 < c5_best:
                h_best, c5_best = h_sg2, c_sg2

    # Phase 4: SLSQP refinement at current resolution
    if budget_s - (time.time() - t0) > 80:
        h_opt = opt_slsqp(h_best.copy(), n_best, 0.004, max_iter=300)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 5: Multi-resolution search
    res_list = [80, 120, 160, 200, 320]
    for n_try in res_list:
        if budget_s - (time.time() - t0) < 100:
            break
        h_c = interp_h(h_best, n_try)
        for T in [0.01, 0.004]:
            if budget_s - (time.time() - t0) < 50:
                break
            h_opt = opt_lbfgs(h_c.copy(), n_try, T, max_iter=200)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best, n_best = h_opt, c, n_try
                h_c = h_best.copy()
        # Try a perturbation at this resolution
        if budget_s - (time.time() - t0) > 60:
            rng = np.random.RandomState(seed + 20000 + n_try)
            h_p = np.clip(h_c + rng.randn(n_try) * 0.07, 0, 1)
            h_p = fix_sum(h_p, n_try)
            h_opt = opt_lbfgs(h_p, n_try, 0.01, max_iter=150)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best, n_best = h_opt, c, n_try

    # Phase 6: Final polish - high penalty L-BFGS-B
    if budget_s - (time.time() - t0) > 50:
        h_opt = opt_lbfgs(h_best.copy(), n_best, 0.001, max_iter=500, pen=1e8)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 7: Snap near-boundary values and re-optimize
    if budget_s - (time.time() - t0) > 30:
        for p in [0.03, 0.08]:
            if budget_s - (time.time() - t0) < 20:
                break
            h_s = h_best.copy()
            h_s[h_s < p] = 0.0
            h_s[h_s > 1.0 - p] = 1.0
            h_s = fix_sum(h_s, n_best)
            h_opt = opt_lbfgs(h_s, n_best, 0.003, max_iter=200)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best = h_opt, c

    # Final subgradient squeeze
    if budget_s - (time.time() - t0) > 20:
        h_sg, c_sg = subgrad_refine(h_best, n_best, steps=80, lr=0.002, n_active=2)
        if c_sg < c5_best:
            h_best, c5_best = h_sg, c_sg

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
