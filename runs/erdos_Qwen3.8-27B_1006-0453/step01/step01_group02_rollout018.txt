def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    lo, hi = -h.max(), h.max()
    for _ in range(80):
        mid = (lo + hi) / 2
        s = float(np.clip(h - mid, 0, 1).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
