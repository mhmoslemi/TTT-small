import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(float(h.sum()) - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, dx, tau):
    N = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    overlap = corr * dx
    mx = overlap.max()
    shifted = np.clip(tau * (overlap - mx), -500, 500)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    val = mx + np.log(exps.sum()) / tau
    grad = np.zeros(N)
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    for idx in active:
        m = int(idx) - (N - 1)
        wi = w[idx]
        if m >= 0:
            if m < N:
                grad[m:] += wi * omh[:N - m]
                if N - m > 0:
                    grad[:N - m] -= wi * h[m:]
        else:
            l = -m
            if l < N:
                L = N - l
                grad[:L] += wi * omh[l:]
                grad[l:] -= wi * h[:L]
    grad *= dx
    return val, grad, w
