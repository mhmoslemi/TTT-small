def project_feasible(h, S):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = S - h.sum()
        if abs(d) < 1e-15:
            break

        h[m] += d / c
    return h

def grad_smooth(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for i in range(2 * n - 1):
            lag = i - (n - 1)
            n_lo = max(0, -lag)
            n_hi = min(n - 1, n - 1 - lag)
            if n_lo > n_hi:
                continue
            nn = j - lag
            if n_lo <= nn <= n_hi:
                s += w[i] * (1.0 - h[nn])
            if n_lo <= j <= n_hi:
                s -= w[i] * h[j + lag]
        grad[j] = s
    return grad * dx

def projected_grad(h0, dx, S, beta, max_iter, deadline):

    best_h = h.copy()
