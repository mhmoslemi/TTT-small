```python
def lbgfgs_optimize(h_start, n, T, alpha_sum=5000.0, max_iter=300):
    dx = 2.0 / n
    target = n * 0.5
    
    def obj(h):
        N = 1
        while N < 2*n - 1:
            N <<= 1
        H = np.fft.rfft(h, N)
        G = np.fft.rfft(1.0 - h, N)
        corr = np.fft.irfft(H * np.conj(G), N)
        c_max = np.max(corr)
        if T < 1e-12:
            val = c_max * dx
        else:
            shifted = corr - c_max
            val = (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx
        s = np.sum(h)
        val += alpha_sum * (s - target)**2
        return val
    
    res = opt.minimize(obj, h_start, method='L-BFGS-B',
                       bounds=[(0,1)]*n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8})
    return res.x
```

```python
def objective_smooth(h, T, alpha_sum, n):
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    c_max = np.max(corr)
    if T < 1e-12:
        val = c_max * dx
    else:
        shifted = corr - c_max
        val = (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx
    s = np.sum(h)
    target = n * 0.5
    val += alpha_sum * (s - target)**2
    return val

def lbgfgs_optimize(h_start, n, T, alpha_sum=5000.0, max_iter=200):
    h_start = h_start.copy()
    res = opt.minimize(
        objective_smooth, h_start,
        args=(T, alpha_sum, n),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10}
    )
    return res.x

def project_fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Adjust the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_fix_sum(h_new, new_n)
```
