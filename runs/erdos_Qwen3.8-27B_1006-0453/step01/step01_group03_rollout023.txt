        return x[n]

        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return x[n] - corr

        return x[:n].sum() - target

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.exp(beta * (corr - m)).sum()) / beta) * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        for j in range(n):
            t = 0.0
            if 0 <= j + lag < n:
                t += 1.0 - h[j + lag]
            if 0 <= j - lag < n:
                t -= h[j - lag]
            if t != 0.0:
                grad[j] += wk * t
    return grad * dx

def lbfgs_solve(h0, dx, target, beta=40.0, max_iter=800):
    n = len(h0)
    def obj_grad(hv):
        s = hv.sum() - target
        val = smooth_max_val(hv, dx, beta) + 500.0 * s * s
        g = smooth_max_grad(hv, dx, beta) + 1000.0 * s
        return val, g
    try:
        res = minimize(obj_grad, h0, jac=True, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxfun': max_iter * 3})
        if res.x is not None:
            return project_box(res.x, target)
    except Exception:
        pass
    return h0.copy()

def slsqp_smooth(h0, dx, target, maxiter=300, beta=50.0):
    n = len(h0)
    t_init = compute_c5(h0, dx) / dx + 0.02
    def obj(x):
        return x[n]
    def ineq_con(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        sm = m + np.log(np.exp(beta * (corr - m)).sum()) / beta
        return x[n] - sm
    def eq_con(x):
        return x[:n].sum() - target
    x0 = np.concatenate([h0, [t_init]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq_con}, {'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_box(res.x[:n], target)
    except Exception:
        pass
    return h0.copy()

def coord_descent(h0, dx, target, n_iter, time_limit):
    n = len(h0)
    h = h0.copy()
    best = compute_c5(h, dx)
    deltas = [0.01, 0.02, 0.04, 0.06, 0.1, 0.15]
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        i = it % n
        j = (it * 13 + 7) % n
        if i == j:
            j = (i + n // 2) % n
        improved = False
        for d in deltas:
            for sgn in [1, -1]:
                h2 = h.copy()
                h2[i] -= sgn * d
                h2[j] += sgn * d
                h2 = project_box(h2, target)
                c = compute_c5(h2, dx)
                if c < best - 1e-13:
                    h, best, improved = h2, c, True
                    break
            if improved:
                break
        # Also try swap
        h2 = h.copy()
        h2[i], h2[j] = h2[j], h2[i]
        h2 = project_box(h2, target)
        c = compute_c5(h2, dx)
        if c < best - 1e-13:
            h, best = h2, c
    return h, best

def anneal_lbfgs(h0, dx, target, time_limit):
    h = h0.copy()
    best_c = compute_c5(h, dx)
    for beta in [20.0, 35.0, 50.0, 80.0]:
        if time.time() > time_limit:
            break
        h1 = lbfgs_solve(h, dx, target, beta=beta, max_iter=500)
        c1 = compute_c5(h1, dx)
        if c1 < best_c:
            h, best_c = h1, c1
    return h, best_c

def make_dispersed(n, rng):
    target = n / 2.0
    h = np.zeros(n)
    m = int(target)
    frac = target - m
    golden = 0.618033988749895
    positions = []
    for k in range(m):
        pos = int((k * golden * n) % n)
        while pos in positions:
            pos = (pos + 1) % n
        positions.append(pos)
        h[pos] = 1.0
    if frac > 0.01:
        for pos in range(n):
            if h[pos] < 0.5:
                h[pos] = frac
                break
    h += 0.03 * rng.standard_normal(n)
    return project_box(h, target)

def make_binary_spread(n, rng):
    target = n / 2.0
    m = int(target)
    frac = target - m
    h = np.zeros(n)
    step = n / m if m > 0 else n
    for k in range(m):
        pos = min(int(k * step + rng.rand() * 0.5), n - 1)
        h[pos] = 1.0
    if frac > 0.01:
        for pos in range(n):
            if h[pos] < 0.5:
                h[pos] = frac
                break
    h += 0.02 * rng.standard_normal(n)
    return project_box(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        h_init = None
        n_init = 47

    best_h = None
    best_c5 = float('inf')
    best_n = n_init

    def remaining():
        return budget_s - (time.time() - t0)

    n_list = [n_init]
    for cand in [47, 63, 67, 91, 127]:
        if cand not in n_list and cand <= 127:
            n_list.append(cand)

    for n in n_list:
        if remaining() < 100:
            break
        if n >= 100 and remaining() < 250:
            continue
        if n >= 90 and remaining() < 180:
            continue

        dx = 2.0 / n
        target = n / 2.0

        starts = []
        if n == n_init and h_init is not None:
            starts.append(project_box(h_init, target))
        starts.append(make_dispersed(n, rng))
        starts.append(make_binary_spread(n, rng))
        if n <= 70:
            h_r = 0.5 + 0.3 * np.sin(np.arange(n) * 0.7) * rng.rand(n)
            starts.append(project_box(h_r, target))

        for h_start in starts:
            if remaining() < 50:
                break
            h = h_start.copy()

            # Phase 1: Annealed L-BFGS-B
            h, c = anneal_lbfgs(h, dx, target, t0 + budget_s - 60)
            if c < best_c5 and n == best_n:
                best_c5, best_h = c, h.copy()

            # Phase 2: SLSQP with smooth constraint
            if remaining() > 30:
                h2 = slsqp_smooth(h, dx, target, maxiter=250, beta=60.0)
                c2 = compute_c5(h2, dx)
                if c2 < c:
                    h, c = h2, c2

            # Phase 3: L-BFGS-B with high beta
            if remaining() > 30:
                h3 = lbfgs_solve(h, dx, target, beta=80.0, max_iter=400)
                c3 = compute_c5(h3, dx)
                if c3 < c:
                    h, c = h3, c3

            # Phase 4: Coordinate descent
            if remaining() > 20:
                h4, c4 = coord_descent(h, dx, target, 8000, t0 + budget_s - 20)
                if c4 < c:
                    h, c = h4, c4

            # Phase 5: Final SLSQP
            if remaining() > 25:
                h5 = slsqp_smooth(h, dx, target, maxiter=150, beta=80.0)
                c5 = compute_c5(h5, dx)
                if c5 < c:
                    h, c = h5, c5

            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

    # Final polish
    if best_h is not None and remaining() > 20:
        dx = 2.0 / best_n
        target = best_n / 2.0
        hf = slsqp_smooth(best_h, dx, target, maxiter=200, beta=100.0)
        cf = compute_c5(hf, dx)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        best_h = project_box(best_h, target)
        best_c5 = compute_c5(best_h, dx)

    if best_h is None:
        n = n_init
        dx = 2.0 / n
        best_h = make_dispersed(n, rng)
        best_n = n
        best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, best_n
