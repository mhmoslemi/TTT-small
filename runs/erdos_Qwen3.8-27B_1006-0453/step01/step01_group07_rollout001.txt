                            best_c5 = c5
                            best_h = h_new.copy()
                            best_n = n
                    except Exception:
                        pass

    # Phase 4: Try lower resolutions for potentially better structure
    if remaining() > 100:
        for n2 in [140, 70, 210]:
            if remaining() < 50:
                break
            if n2 >= pn:
                continue
            h0 = upsample(parent, n2)
            bounds2 = list(zip(np.zeros(n2), np.ones(n2)))
            for beta in [0.003, 0.001]:
                if remaining() < 20:
                    break
                try:
                    res = minimize(
                        softmax_obj, h0, args=(n2, beta),
                        method='L-BFGS-B', bounds=bounds2,
                        options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10}
                    )
                    h_new = fix_sum(np.clip(res.x, 0, 1), n2)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        best_n = n2
                except Exception:
                    pass

    # Phase 5: Fine local search
    if remaining() > 15:
        h, cur = local_search(best_h, best_n, rng, remaining() - 10, step_scale=1.0)
        if cur < best_c5:
            best_c5 = cur
            best_h = h

    # Phase 6: Focused search on active lags
    if remaining() > 10:
        h, cur = focused_local_search(best_h, best_n, rng, remaining() - 5)
        if cur < best_c5:
            best_c5 = cur
            best_h = h

    # Phase 7: Final ultra-fine refinement
    if remaining() > 5:
        h, cur = local_search(best_h, best_n, rng, remaining() - 2, step_scale=0.3)
        if cur < best_c5:
            best_c5 = cur
            best_h = h

    return best_h, best_c5, best_n
