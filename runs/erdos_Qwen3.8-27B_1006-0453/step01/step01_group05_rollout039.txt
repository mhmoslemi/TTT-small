    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    # corr[k] = sum_j h[j + k - (n-1)] * v[j] where 0 <= j < n and 0 <= j+k-(n-1) < n

    # = sum_j h[j+lag] * (1-h[j])

    # d/dh[m] of corr[k]:
    # Term 1: when j+lag = m, i.e., j = m-lag: contributes v[m-lag] = 1-h[m-lag]
    #   valid when 0 <= m-lag < n
    # Term 2: when j = m: contributes -h[m+lag] * v... wait
    #   corr = sum_j h[j+lag]*(1-h[j])
    #   d/dh[m] = (1-h[m-lag]) * [valid j=m-lag] + sum_j h[j+lag] * (-1) * [j=m]
    #   = (1-h[m-lag]) * [0<=m-lag<n] - h[m+lag] * [0<=m+lag<n]

    lag = k_star - (n - 1)

    # i.e., max(0,lag) <= m < min(n, n+lag)

    return corr, k_star, lag, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = temp * (corr - m)
    return (m + np.log(np.sum(np.exp(s))) / temp) * dx

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def slsqp_opt(x0, n, temp, maxiter=200):
    bds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, x0, method='SLSQP',
                       bounds=bds,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
                       args=(n, temp),
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return x0

def compute_grad_worst(h, n):
    """Compute gradient of the worst correlation lag w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, lag)
    hi1 = min(n, n + lag)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[0:hi1 - lo1])
    lo2 = max(0, -lag)
    hi2 = min(n, n - lag)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 + lag:hi2 + lag]
    return grad, k_star

def guided_local_search(h, n, t0, budget_s, margin=15):
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    step_base = 0.01
    no_improve = 0
    while time.time() - t0 < budget_s - margin:
        t_batch = time.time()
        for _ in range(2000):
            if time.time() - t0 > budget_s - margin:
                break
            grad, _ = compute_grad_worst(h, n)
            top_k = 3 + np.random.randint(0, 4)
            idx = np.argsort(np.abs(grad))[::-1][:top_k]
            h2 = h.copy()
            scale = step_base * (0.5 + np.random.rand())
            h2[idx] -= scale * grad[idx]
            h2 = np.clip(h2, 0.0, 1.0)
            diff = n / 2.0 - np.sum(h2)
            free = (h2 > 1e-8) & (h2 < 1.0 - 1e-8)
            nf = int(np.sum(free))
            if nf > 0:
                h2[free] += diff / nf
                h2 = np.clip(h2, 0.0, 1.0)
            h2 += (n / 2.0 - np.sum(h2)) / n
            h2 = np.clip(h2, 0.0, 1.0)
            c2 = compute_c5(h2, n)
            if c2 < best_c5 - 1e-15:
                best_c5 = c2
                h = h2
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > 4000:
            step_base *= 0.5
            no_improve = 0
    return h, best_c5

def random_local_search(h, n, t0, budget_s, margin=10):
    best_c5 = compute_c5(h, n)
    step = 0.03
    while time.time() - t0 < budget_s - margin:
        for _ in range(3000):
            if time.time() - t0 > budget_s - margin:
                break
            h2 = h.copy()
            npert = 1 + np.random.randint(0, 4)
            idx = np.random.choice(n, npert, replace=False)
            h2[idx] += np.random.uniform(-step, step, npert)
            h2 = np.clip(h2, 0.0, 1.0)
            diff = n / 2.0 - np.sum(h2)
            free = (h2 > 1e-10) & (h2 < 1.0 - 1e-10)
            nf = int(np.sum(free))
            if nf > 0:
                h2[free] += diff / nf
            h2 = np.clip(h2, 0.0, 1.0)
            h2 += (n / 2.0 - np.sum(h2)) / n
            h2 = np.clip(h2, 0.0, 1.0)
            c2 = compute_c5(h2, n)
            if c2 < best_c5 - 1e-15:
                best_c5 = c2
                h = h2
    return h, best_c5

def periodic_start(n, period, seed):
    rng = np.random.RandomState(seed)
    base = rng.uniform(0.1, 0.9, period)
    rep = n // period
    if rep == 0:
        return None
    h = np.tile(base, rep)
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        h = np.pad(h, (0, n - len(h)), mode='wrap')
    return h
