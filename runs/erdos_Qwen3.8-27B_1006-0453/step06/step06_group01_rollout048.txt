def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, nh):

            # d/dh[i] for i in [shift, N-1]: -h[i-shift]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def thue_morse(N):
    t = np.zeros(N, dtype=int)
    for i in range(N):
        n = i
        c = 0
        while n:
            c ^= (n & 1)
            n >>= 1
        t[i] = c
    return t.astype(float)

def beatty_init(N, alpha):
    x = np.arange(N) * alpha
    return ((x - np.floor(x)) < 0.5).astype(float)

def van_der_corput(N):
    # Van der Corput sequence (first N terms)
    v = np.zeros(N)
    for i in range(1, N+1):
        f, d = 0.0, 1.0
        n = i
        while n > 0:
            d *= 2
            f += (n & 1) / d
            n >>= 1
        v[i-1] = f
    return v  # values in [0,1)

def sa_binary(h_in, N, deadline, t_init=1.0, t_final=0.01, max_iters=50000, multi=1):
    """Simulated annealing with single or multi swaps."""
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    stall = 0
    
    while it < max_iters and time.time() < deadline:
        it += 1
        T = t_init * (t_final / t_init) ** (float(it) / max_iters)
        
        # Select positions to swap
        if multi == 1:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            ps, qs = [p], [q]
        else:
            ps = list(np.random.choice(ones, multi, replace=False))
            qs = list(np.random.choice(zeros, multi, replace=False))
        
        old_c = float(corr.max())
        
        # Apply swap
        for p, q in zip(ps, qs):
            h[p], h[q] = 0.0, 1.0
        corr_new = fft_corr(h)
        new_c = float(corr_new.max())
        
        delta = new_c - old_c
        if delta < 0 or (T > 1e-10 and np.random.random() < np.exp(-delta / T)):
            corr = corr_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            for p, q in zip(ps, qs):
                h[p], h[q] = 1.0, 0.0
            stall += 1
        
        if stall > max_iters * 0.1:
            # Restart temperature
            T = t_init * 0.5
            stall = 0
    
    return best_h, best_c * (2.0 / N)

def targeted_sa(h_in, N, deadline, top_k=10):
    """SA that focuses swaps on positions affecting the top-k worst shifts."""
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Find top-k worst shifts
        top_indices = np.argsort(corr)[-top_k:]
        
        # For each top shift, find positions that contribute most
        # and prefer swapping those
        hot_positions = set()
        for t in top_indices:
            k = int(t - (N - 1))
            if k >= 0:
                # corr at lag k: sum_{i=0}^{N-1-k} h[i]*(1-h[i+k])
                for i in range(N - k):
                    if h[i] == 1 and h[i+k] == 0:
                        hot_positions.add(i)
                        hot_positions.add(i+k)
            else:
                M = -k
                for i in range(M, N):
                    if h[i] == 1 and h[i-M] == 0:
                        hot_positions.add(i)
                        hot_positions.add(i-M)
        
        # Pick swap from hot positions if possible
        hot_ones = list(ones & hot_positions)
        hot_zeros = list(zeros & hot_positions)
        
        if hot_ones and hot_zeros:
            p = int(np.random.choice(hot_ones))
            q = int(np.random.choice(hot_zeros))
        else:
            p = int(np.random.choice(list(ones)))
            q = int(np.random.choice(list(zeros)))
        
        h[p], h[q] = 0.0, 1.0
        corr = fft_corr(h)
        c = float(corr.max())
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
            ones.discard(p)
            ones.add(q)
            zeros.discard(q)
            zeros.add(p)
        else:
            # Revert with some probability (greedy with occasional acceptance)
            if np.random.random() < 0.1:
                ones.discard(p)
                ones.add(q)
                zeros.discard(q)
                zeros.add(p)
            else:
                h[p], h[q] = 1.0, 0.0
    
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h_in, N, deadline, top_k=50, lr0=0.015):
    h = h_in.astype(float).copy()
    target = N / 2.0
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max()) * (2.0 / N)
        if pk < best_c5:
            best_c5 = pk
            best_h = h.copy()
        top = np.argpartition(corr, -top_k)[-top_k:]
        top = np.sort(top)
        grad = np.zeros(N)
        for t in top:
            shift = int(t - (N - 1))
            if shift == 0:
                continue
            elif shift > 0:
                grad[:N - shift] += (1.0 - y[shift:])
                grad[shift:] -= y[:N - shift]
            else:
                M = -shift
                grad[M:] += (1.0 - y[:N - M])
                grad[:N - M] -= y[M:]
        grad /= max(1, len(top))
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.97
            beta = min(0.97, 0.82 + it * 0.0001)
    return best_h, best_c5
