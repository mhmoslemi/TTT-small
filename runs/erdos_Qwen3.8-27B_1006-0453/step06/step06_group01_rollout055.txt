import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def binary_anneal(h_in, N, deadline, T0=5.0, Tend=0.01, cool=0.99995, perturb_every=30000):

    best_c = float(corr.max())
    best_h = h.copy()
    cur_c = best_c
    T = T0
    cnt = 0
    no_impr = 0
    ones = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    
    while time.time() < deadline:
        cnt += 1
        if cnt % perturb_every == 0 and no_impr > perturb_every // 2:
            # Perturb
            for _ in range(8):
                pp = int(np.random.randint(len(ones)))
                qq = int(np.random.randint(len(zeros_arr)))
                p, q = int(ones[pp]), int(zeros_arr[qq])
                h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
            cur_c = float(corr.max())
            ones = np.where(h == 1)[0].copy()
            zeros_arr = np.where(h == 0)[0].copy()
            no_impr = 0
            T = max(T, T0 * 0.3)
            continue
        
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros_arr[np.random.randint(len(zeros_arr))])

        lp = N - p
        sq = N - 1 - q
        lq = N - q
        
        dp = -(1.0 - h[:lp])
        dq = (1.0 - h[:lq])
        
        corr[sp:sp+lp] += dp
        corr[sq:sq+lq] += dq
        h[p] = 0.0
        h[q] = 1.0
        new_c = float(corr.max())
        
        if new_c < cur_c or (T > Tend and np.random.random() < np.exp(-(new_c - cur_c) / T)):
            cur_c = new_c
            # Update lists

            zeros_arr = np.where(h == 0)[0].copy()
            if cur_c < best_c:

                no_impr = 0

def nesterov_opt(h_in, N, deadline, k=40, lr0=0.01):
    h = h_in.astype(float).copy()
    target = N / 2.0
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        pk = float(corr_y.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr_y, -k)[-k:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.97
    return best_h, best_pk * (2.0 / N)

    h_new = np.interp(x_new, x_old, h_src)
    return h_new
