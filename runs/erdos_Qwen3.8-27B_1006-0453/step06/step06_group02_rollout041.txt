import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_delta(h, N, p, q):
    """O(N) computation of correlation change from flipping p(1->0) and q(0->1)."""
    s = np.arange(-(N-1), N)
    delta = np.zeros(2*N-1)
    ps = p - s
    m1 = (ps >= 0) & (ps < N)
    delta[m1] -= (1.0 - h[ps[m1]])
    psp = p + s
    m2 = (psp >= 0) & (psp < N)
    delta[m2] += h[psp[m2]]
    qs = q - s
    m3 = (qs >= 0) & (qs < N)
    delta[m3] += (1.0 - h[qs[m3]])
    qsp = q + s
    m4 = (qsp >= 0) & (qsp < N)
    delta[m4] -= h[qsp[m4]]
    return delta

def binary_sa(h, N, deadline, n_candidates=20, T0=0.01, T1=0.001):
    """Simulated annealing on binary states with O(N) delta evaluation."""
    h = h.astype(float).copy()
    c = fft_corr(h)
    current_max = float(c.max())
    best_max = current_max
    best_h = h.copy()
    
    n1 = int(h.sum())
    n0 = N - n1
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_steps = 0
    t_start = time.time()
    
    while time.time() < deadline:
        # Temperature schedule
        elapsed = time.time() - t_start
        progress = min(1.0, elapsed / 100.0)
        T = T0 * (T1/T0) ** progress
        
        # Pick n_candidates random pairs
        best_delta_max = current_max
        best_pair = None
        best_new_c = c
        
        for _ in range(n_candidates):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            if p == q:
                continue
            d = compute_delta(h, N, int(p), int(q))
            new_c = c + d
            new_max = float(new_c.max())
            if new_max < best_delta_max:
                best_delta_max = new_max
                best_pair = (int(p), int(q))
                best_new_c = new_c
        
        if best_pair is None:
            break
        
        # Accept/reject
        if best_delta_max < current_max or np.random.random() < np.exp(-(best_delta_max - current_max) / max(T, 1e-8)):
            p, q = best_pair
            h[p] = 0.0
            h[q] = 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = best_new_c
            current_max = best_delta_max
            if current_max < best_max - 1e-12:
                best_max = current_max
                best_h = h.copy()
        
        n_steps += 1
        if n_steps % 500 == 0 and time.time() < deadline:
            # Periodic re-evaluation
            c = fft_corr(h)
            current_max = float(c.max())
    
    return best_h, best_max * (2.0 / N)

def continuous_opt(h, N, deadline, n_top=20):
    """Projected subgradient descent for continuous h."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    
    lr = 0.05
    beta = 0.9
    m = np.zeros(N)
    
    while time.time() < deadline:
        c = fft_corr(h)
        pk = float(c.max()) * (2.0/N)
        if pk < best_c5 - 1e-10:
            best_c5 = pk
            best_h = h.copy()
        
        # Top-K subgradient
        top_k = np.argpartition(c, -n_top)[-n_top:]
        s_vals = top_k - (N - 1)
        grad = np.zeros(N)
        for s in s_vals:
            s = int(s)
            # ∂r/∂h[i] = (1-h[i-s])*[0≤i-s<N] - h[i+s]*[0≤i+s<N]
            if s >= 0:
                # i-s valid: i >= s, i < N+s → i in [s, N-1]
                grad[s:] += (1.0 - h[:N-s])
                # i+s valid: i < N-s → i in [0, N-s-1]
                grad[:N-s] -= h[s:]
            else:
                ss = -s
                # i-s = i+ss valid: i < N-ss → i in [0, N-ss-1]
                grad[:N-ss] += (1.0 - h[ss:])
                # i+s = i-ss valid: i >= ss → i in [ss, N-1]
                grad[ss:] -= h[:N-ss]
        
        grad /= n_top
        m = beta * m - lr * grad
        h = project_sum(h + m, target)
        lr *= 0.999
        beta = min(0.95, 0.85 + 0.001 * (n_steps if (n_steps := 0) else 0))
    
    return best_h, best_c5
