import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:2*N-1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5*(lo+hi), 0, 1)
    return h

def fix_binary(h, N):
    target = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[target:]] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target-n1]] = 1
    return h

def fast_local_search(h, N, deadline, max_sweeps=3000, sample_p=180, sample_q=90):
    h = fix_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    no_improve = 0
    center = N - 1
    shifts = np.arange(2*N-1) - center
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Weight sampling: prefer positions contributing to peak correlations
        top_shifts = np.argsort(c)[-8:]
        hot = np.zeros(N)
        for ai in top_shifts:
            L = int(ai - center)
            if L >= 0:
                valid = np.arange(0, N-L)
                hot[valid] += (1.0 - h[valid+L])
            elif L < 0:
                M = -L
                valid = np.arange(0, N-M)
                hot[valid+M] += h[valid]
        hot[ones] += best_c * 0.5
        hot[zeros] = 0
        
        if len(ones) > sample_p:
            probs = hot[ones]
            probs = np.maximum(probs, 0.01)
            probs /= probs.sum()
            p_idx = np.random.choice(len(ones), sample_p, replace=False, p=probs)
        else:
            p_idx = np.arange(len(ones))
        p_list = ones[p_idx]
        
        if len(zeros) > sample_q:
            q_idx = np.random.choice(len(zeros), sample_q, replace=False)
        else:
            q_idx = np.arange(len(zeros))
        q_list = zeros[q_idx]

        best_pq = None

            ps = p + shifts
            pm = p - shifts
            dp = np.zeros(2*N-1)
            mp = (ps >= 0) & (ps < N)
            mn = (pm >= 0) & (pm < N)
            dp[mp] -= (1.0 - h[ps[mp]])
            dp[mn] += h[pm[mn]]
            base = c + dp
            base_max = base.max()
            if base_max >= best_score:

        top_k = np.argpartition(corr, -30)[-30:]

            beta = min(0.95, 0.85 + it * 0.00001)
    
    return best_h, best_pk * (2.0 / N)
