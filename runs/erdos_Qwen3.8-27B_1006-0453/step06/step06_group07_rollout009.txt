import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(42)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def two_bit_swap(h, N, deadline, n_tries=200):
    """Try 2-bit swaps: flip 2 ones to 0 and 2 zeros to 1 simultaneously."""
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    D = compute_D(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rng = np.random.RandomState(777)
    
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        D = compute_D(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Find top peaks
        top_k = np.argpartition(c, -5)[-5:]
        
        # For each top peak, find candidate ones/zeros
        best_swap = None
        best_new_P = P
        
        for kp in top_k:
            # Top ones by D[:, kp]
            D_ones = D[ones, kp]
            top_ones_local = ones[np.argsort(D_ones)[-4:]]
            D_zeros = D[zeros, kp]
            top_zeros_local = zeros[np.argsort(D_zeros)[-4:]]
            
            # Try all pairs
            for i1 in range(len(top_ones_local)):
                for i2 in range(i1+1, len(top_ones_local)):
                    p1, p2 = top_ones_local[i1], top_ones_local[i2]
                    for j1 in range(len(top_zeros_local)):
                        for j2 in range(j1+1, len(top_zeros_local)):
                            q1, q2 = top_zeros_local[j1], top_zeros_local[j2]
                            # Approximate new correlation
                            nc = c - D[p1] - D[p2] - D[q1] - D[q2]
                            np_val = float(nc.max())
                            if np_val < best_new_P:
                                best_new_P = np_val
                                best_swap = (int(p1), int(p2), int(q1), int(q2))
        
        if best_swap is None:
            # Try random 2-bit swaps
            for _ in range(20):
                p1, p2 = rng.choice(ones, 2, replace=False)
                q1, q2 = rng.choice(zeros, 2, replace=False)
                nc = c - D[p1] - D[p2] - D[q1] - D[q2]
                np_val = float(nc.max())
                if np_val < best_new_P:
                    best_new_P = np_val
                    best_swap = (int(p1), int(p2), int(q1), int(q2))
        
        if best_swap is None:
            break
        
        p1, p2, q1, q2 = best_swap
        h[p1] = h[p2] = 0.0
        h[q1] = h[q2] = 1.0
    
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=50, lr0=0.018):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

    return np.interp(x_new, x_old, h_old)
