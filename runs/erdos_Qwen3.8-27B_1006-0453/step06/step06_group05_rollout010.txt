def make_binary(h, N):
    h = h.astype(float).copy()
    nh = N // 2
    if h.sum() == nh:
        h = (h >= 0.5).astype(float)
    else:
        h[:] = 0.0
        idx = np.argsort(-h)
        h[idx[:nh]] = 1.0
    return h

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)

    hp = np.where(vp, h[pp_c], 0.0)

        if new_P <= P or np.random.random() < np.exp(-(new_P - P) / max(T, 0.01)):
            c = new_c
            h[p] = 0.0
            h[q] = 1.0
        T *= 0.997
        if T < 0.05:
            T = 0.05
    return best_h, best_c

def perturb_binary(h, N, n_flips):
    h = h.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_flips):
        p = int(np.random.randint(len(ones)))
        q = int(np.random.randint(len(zeros)))
        h[ones[p]] = 0.0
        h[zeros[q]] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = make_binary(h0, N0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Single-swap + 2-opt from initial
    d1 = min(deadline - 600, t0 + 150)
    hs, cs = swap_search(best_h, N0, d1, max_iters=5000, stall_limit=80, two_opt=True, top_k=5)
    if cs * (2.0/N0) < best_c5:
        best_c5 = cs * (2.0/N0)
        best_h = hs

    # Phase 2: Anneal to escape local minima
    if time.time() < deadline - 400:
        d2 = min(deadline - 300, time.time() + 200)
        ha, ca = anneal_search(best_h, N0, d2, max_iters=4000, T0=8.0)
        if ca * (2.0/N0) < best_c5:
            best_c5 = ca * (2.0/N0)
            best_h = ha
        # Polish after anneal
        if time.time() < deadline - 280:
            d2b = min(deadline - 280, time.time() + 80)
            hp2, cp2 = swap_search(best_h, N0, d2b, max_iters=3000, stall_limit=60, two_opt=True, top_k=4)
            if cp2 * (2.0/N0) < best_c5:
                best_c5 = cp2 * (2.0/N0)
                best_h = hp2

    # Phase 3: Restart with perturbation
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 120)
        hp3 = perturb_binary(best_h, N0, 8)
        hs3, cs3 = swap_search(hp3, N0, d3, max_iters=3000, stall_limit=70, two_opt=True, top_k=5)
        if cs3 * (2.0/N0) < best_c5:
            best_c5 = cs3 * (2.0/N0)
            best_h = hs3
        # Second perturbation
        if time.time() < deadline - 180:
            d3b = min(deadline - 180, time.time() + 80)
            hp3b = perturb_binary(best_h, N0, 5)
            hs3b, cs3b = swap_search(hp3b, N0, d3b, max_iters=2000, stall_limit=50, two_opt=False)
            if cs3b * (2.0/N0) < best_c5:
                best_c5 = cs3b * (2.0/N0)
                best_h = hs3b

    # Phase 4: Try different N values
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [360, 400, 380]:
        if time.time() > deadline - 100:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == N0:
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_try = make_binary(h_interp, N_try)
        d_try = min(deadline - 60, time.time() + 80)
        ht, ct = swap_search(h_try, N_try, d_try, max_iters=3000, stall_limit=60, two_opt=True, top_k=4)
        c5_try = ct * (2.0/N_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = ht
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 5: Final anneal + polish on best
    if time.time() < deadline - 60:
        d5 = min(deadline - 40, time.time() + 100)
        h5, c5 = anneal_search(best_h, len(best_h), d5, max_iters=3000, T0=4.0)
        if c5 * (2.0/len(best_h)) < best_c5:
            best_c5 = c5 * (2.0/len(best_h))
            best_h = h5

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        Nm = len(best_h)
        hf, cf = swap_search(best_h, Nm, deadline - 2, max_iters=5000, stall_limit=150, two_opt=True, top_k=6)
        if cf * (2.0/Nm) < best_c5:
            best_c5 = cf * (2.0/Nm)
            best_h = hf

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
