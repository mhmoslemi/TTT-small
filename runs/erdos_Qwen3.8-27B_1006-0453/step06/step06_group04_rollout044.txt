def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_at_lag(y, L, N):
    """Compute gradient contribution at lag L (L = index - (N-1))"""
    grad = np.zeros(N)
    if L == 0:
        grad[:] += (1.0 - 2.0 * y)
    elif L > 0 and L < N:
        seg = N - L
        grad[L:] += (1.0 - y[:seg])
        grad[:seg] -= y[L:]
    elif L < 0 and L > -N:
        M = -L
        seg = N - M
        grad[:seg] += (1.0 - y[M:])
        grad[M:] -= y[:seg]
    return grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_grad(y, corr, top, center, N):
    cmax = corr[top].max()
    w = np.exp((corr[top] - cmax) / 0.5)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(len(top)) / len(top)
    else:
        w = w / ws
    grad = np.zeros(N)
    for wi, ai in zip(w, top):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif 0 < L < N:
            seg = N - L
            grad[L:] += wi * (1.0 - y[:seg])
            grad[:seg] -= wi * y[L:]
        elif -N < L < 0:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - y[M:])
            grad[M:] -= wi * y[:seg]
    return grad

def continuous_opt(h0, N, deadline, k=80, lr0=0.022, temp0=0.003, beta_start=0.9):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    lr = lr0
    beta = beta_start
    temp = temp0
    stall = 0
    it = 0
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2 = 0.9, 0.999
    center = N - 1
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        kk = min(k, len(corr) - 1)
        top = np.argpartition(corr, -kk)[-kk:]
        grad = make_grad(y, corr, top, center, N)
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        step = lr * mh / (np.sqrt(vh) + 1e-8)
        h = project_sum(y - step, target)
        if it % 250 == 0:
            lr = max(0.002, lr * 0.993)
            temp = max(0.0003, temp * 0.99)
            beta = min(0.97, beta_start + it * 0.00008)
        if stall > 250:
            scale = 0.02 + 0.03 * min(stall / 500.0, 2.0)
            h = best_h + np.random.randn(N) * scale
            h = project_sum(h, target)
            h_prev = h.copy()
            m[:] = 0.0
            v[:] = 0.0
            lr = max(lr, lr0 * 0.85)
            it = 0
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_opt(h0, N, deadline, n_tries=18, kick_freq=60):
    h = (h0 > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - nh]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h[idx] = 1
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sw = int(c.max())
        best_sh = None
        best_sc = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            ct = fft_corr(h)
            cti = int(ct.max())
            if cti < best_sw:
                best_sw = cti
                best_sh = h.copy()
                best_sc = ct.copy()
            h[p] = 1
            h[q] = 0
        if best_sh is not None:
            h = best_sh
            c = best_sc
            stall = 0
            if best_sw < best_int:
                best_int = best_sw
                best_h = h.copy()
        else:
            stall += 1
            if stall > kick_freq and len(ones) >= 2 and len(zeros) >= 2:
                i1 = np.random.choice(len(ones), 2, replace=False)
                i2 = np.random.choice(len(zeros), 2, replace=False)
                h[ones[i1[0]]] = 0
                h[ones[i1[1]]] = 0
                h[zeros[i2[0]]] = 1
                h[zeros[i2[1]]] = 1
                c = fft_corr(h)
                if int(c.max()) < best_int:
                    best_int = int(c.max())
                    best_h = h.copy()
                stall = 0
    return best_h, best_int * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
