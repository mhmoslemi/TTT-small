import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h -= (s - target) / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    s = h.sum()
    if s != target:
        diff = target - s
        idx = np.argsort(h)[::-1] if diff > 0 else np.argsort(h)
        for i in idx:
            add = min(max(diff, 0), 1 - h[i]) if diff > 0 else min(max(-diff, 0), h[i])
            h[i] += add if diff > 0 else -add
            diff -= add if diff > 0 else -(-add)
            if abs(diff) < 1e-12:
                break
    return h

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv  # j - k: index for h[j-k]
    pp = pos + sv  # j + k: index for h[j+k]
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[np.clip(pm, 0, N-1)], 0.0)
    hp = np.where(vp, h[np.clip(pp, 0, N-1)], 0.0)
    return (1.0 - hp) * vp.astype(float) - hm * vm.astype(float)

def binary_local_search(h, N, deadline, max_sweeps=500, anneal=0.0):

    n1 = int(h.sum())
    nh = N // 2

        zi = np.where(h == 0)[0]
        h[:] = 0
        if n1 > nh:
            h[oi[:nh]] = 1
        else:
            h[oi] = 1
            h[zi[:nh - n1]] = 1
    c = fft_corr(h)

    stall = 0

        P = best_c
        ones = np.where(h == 1)[0]

        if len(ones) < 1 or len(zeros) < 1:

        D = compute_D(h, N)
        bd, bpq = 0.0, None
        for p in ones:

            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)

    target = N / 2.0

    beta = beta0

            stall = 0

        # Softmax weights

        w = np.exp(sc * beta)

        D = compute_D(h, N)

        grad -= grad.mean()
        bc1 = 1 - 0.9**t

        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        lr = lr0 / (1 + 0.01 * t)
        h = project_sum(h + lr * m / (np.sqrt(v / bc2) + 1e-8), target)
        if t % 200 == 0:

    return best_h, best_pk * (2.0 / N)

    pos = np.arange(N)[:, None]

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def c5_val(h):
    N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def compute_D(h, N):
    sh = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = sh[None, :]
    pm, pp = pos - sv, pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    return (1.0 - np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)) * vp \
         - np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0) * vm

        else: break

    return h, bc

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def compute_D(h, N):
    sh = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = sh[None, :]
    pm, pp = pos - sv, pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    rhm = np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    rhp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    return (1.0 - rhp) * vp - rhm * vm

def bsearch(h, N, dl, mx=500, ann=0.0):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        h[:] = 0.0
        if n1 > nh:
            h[oi[:nh]] = 1.0
        else:
            h[oi] = 1.0
            h[zi[:nh - n1]] = 1.0
    c = fft_corr(h)
    bc = float(c.max())
    st = 0
    for _ in range(mx):
        if time.time() > dl:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        P = float(c.max())
        D = compute_D(h, N)
        bd, bq = 0.0, None
        for p in ones:
            base = c - D[p]
            nk = (base[None, :] - D[zeros]).max(axis=1)
            mi = int(nk.argmin())
            d = float(nk[mi]) - P
            if d < bd:
                bd, bq = d, (int(p), int(zeros[mi]))
        if bq is not None and bd < -1e-12:
            h[bq[0]], h[bq[1]] = 0.0, 1.0
            c = fft_corr(h)
            if c.max() < bc:
                bc = float(c.max())
                st = 0
            else:
                st += 1
        elif ann > 0 and np.random.random() < ann:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            if c.max() < bc:
                bc = float(c.max())
            st = 0
        else:
            break
        if st > 50:
            break
    return h, bc

def smax_opt(h, N, dl, b0=5, bmax=60, lr0=0.04, pat=100):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    bh, bk = h.copy(), float(fft_corr(h).max())
    m, v = np.zeros(N), np.zeros(N)
    t, b, st, lr = 0, b0, 0, lr0
    while time.time() < dl:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < bk - 1e-12:
            bk, bh, st = pk, h.copy(), 0
        else:
            st += 1
        if st >= pat:
            b = min(b * 1.4, bmax)
            lr *= 0.7
            st = 0
            if b >= bmax and lr < 5e-4:
                break
        w = np.exp((c - c.max()) * b)
        w /= w.sum()
        D = compute_D(h, N)
        g = -(D @ w) * (2.0 / N)
        g -= g.mean()
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        m2 = m / (1 - 0.9 ** t)
        v2 = v / (1 - 0.999 ** t)
        h = project_sum(h + lr * m2 / (np.sqrt(v2) + 1e-8), N / 2.0)
        if t % 400 == 0:
            b = min(b * 1.05, bmax)
    return bh, bk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    bh, bc = h0.copy(), c5_val(h0)

    # P1: Binary local search from initial
    d1 = min(dl - 680, t0 + 100)
    h1, c1 = bsearch(h0, N0, d1, 800)
    if c1 < bc:
        bc, bh = c1, h1.copy()

    # P2: Two rounds of soft-max continuous optimization
    for ns in [0.03, 0.08]:
        if time.time() > dl - 430:
            break
        d2 = min(dl - 420, time.time() + 150)
        hh, c2 = smax_opt(bh + np.random.randn(N0) * ns, N0, d2, 4, 50, 0.035, 120)
        if c2 < bc:
            bc, bh = c2, hh

    # P3: Binary local search from continuous result
    if time.time() < dl - 250:
        d3 = min(dl - 160, time.time() + 130)
        h3, c3 = bsearch(bh, N0, d3, 700)
        if c3 < bc:
            bc, bh = c3, h3.copy()

    # P4: Focused continuous optimization with higher beta
    if time.time() < dl - 130:
        d4 = min(dl - 60, time.time() + 130)
        h4, c4 = smax_opt(bh + np.random.randn(N0) * 0.015, N0, d4, 15, 90, 0.012, 80)
        if c4 < bc:
            bc, bh = c4, h4

    # P5: Multi-resolution
    for Nt in [382, 378, 384]:
        if time.time() > dl - 30:
            break
        if Nt == len(bh):
            continue
        xb = np.linspace(0, 2, len(bh), endpoint=False)
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, xb, bh)
        ht = (hi > 0.5).astype(float)
        nh_t, n1_t = Nt // 2, int(ht.sum())
        if n1_t > nh_t:
            ix = np.where(ht == 1)[0]
            ht[ix[np.argsort(hi[ix])[:n1_t - nh_t]]] = 0
        elif n1_t < nh_t:
            ix = np.where(ht == 0)[0]
            ht[ix[np.argsort(-hi[ix])[:nh_t - n1_t]]] = 1
        dr = min(dl - 12, time.time() + 28)
        hr, cr = bsearch(ht, Nt, dr, 250)
        cr = c5_val(hr)
        if cr < bc:
            bc, bh = cr, hr

    # P6: Final annealed binary search
    if time.time() < dl - 12:
        h6, c6 = bsearch(bh, len(bh), dl - 8, 200, 0.1)
        if c6 < bc:
            bc, bh = c6, h6.copy()

    # P7: Final continuous polish
    if time.time() < dl - 2:
        hf, cf = smax_opt(bh + np.random.randn(len(bh)) * 0.01, len(bh), dl - 0.5, 25, 80, 0.008, 30)
        if cf < bc:
            bc, bh = cf, hf

    return bh, c5_val(bh), len(bh)
