import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft(np.flip(1.0 - h.astype(float)), L)
    return np.real(np.fft.irfft(H*G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_feasible(h, N):
    """Ensure sum(h) = N/2 and 0<=h<=1 (binary)."""
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1
    return h

def efficient_local_search(h, N, deadline, M=30):
    h = ensure_feasible(h.copy(), N)
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        
        top_idx = np.argpartition(corr, -M)[::-1][:M]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]  # sort descending
        top_vals = corr[top_idx]
        # Check gap: if (M+1)-th value + 2 < pk, we're safe
        if M < len(corr) - 1:
            rest = np.delete(corr, top_idx)
            next_val = float(rest.max()) if len(rest) > 0 else 0
            if next_val + 2 < pk - 1:
                pass  # Safe
            # else: increase M in next iteration (but for now, M=30 should be enough)
        
        lags = top_idx.astype(int)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Precompute delta components
        # For p in ones: A[p_idx, lag_idx] = -(1-h[p+lag]) if valid
        #                B[p_idx, lag_idx] = h[p-lag] if valid
        # For q in zeros: C[q_idx, lag_idx] = (1-h[q+lag]) if valid
        #                D[q_idx, lag_idx] = -h[q-lag] if valid
        
        ones_2d = ones[:, None]  # (n_ones, 1)
        zeros_2d = zeros[:, None]  # (n_zeros, 1)
        lags_1d = lags[None, :]  # (1, M)
        
        ip = ones_2d + lags_1d  # (n_ones, M)
        im = ones_2d - lags_1d

        vpm = (im >= 0) & (im < N)
        A = np.where(vpp, -(1.0 - h[np.clip(ip, 0, N-1)]), 0.0)
        B = np.where(vpm, h[np.clip(im, 0, N-1)], 0.0)
        
        iq = zeros_2d + lags_1d
        imq = zeros_2d - lags_1d
        vqp = (iq >= 0) & (iq < N)
        vqm = (imq >= 0) & (imq < N)
        C = np.where(vqp, (1.0 - h[np.clip(iq, 0, N-1)]), 0.0)
        D = np.where(vqm, -h[np.clip(imq, 0, N-1)], 0.0)
        
        # Combine: delta[i,j,k] = A[i,k] + B[i,k] + C[j,k] + D[j,k]
        # new_val[i,j,k] = top_vals[k] + delta[i,j,k]

        best_pair = None
        
        for i_start in range(0, n_ones, batch):

            A_sub = A[i_start:i_end] + B[i_start:i_end]  # (batch, M)
            # delta for all j: A_sub[:,None,:] + C[None,:,:] + D[None,:,:]
            delta = A_sub[:, None, :] + C[None, :, :] + D[None, :, :]  # (batch, n_zeros, M)
            new_vals = top_vals[None, None, :] + delta  # (batch, n_zeros, M)
            new_maxes = np.max(new_vals, axis=2)  # (batch, n_zeros)
            
            # Also check non-top lags: they can increase by at most 2
            # If next_val + 2 > new_max, we need to be more careful
            # For now, M=30 should capture the relevant lags
            
            global_idx = new_maxes.argmin()
            local_i, local_j = divmod(int(global_idx), n_zeros)
            val = float(new_maxes[local_i, local_j])
            if val < best_new_pk - 1e-12:
                best_new_pk = val
                best_pair = (i_start + local_i, local_j)
        
        if best_pair is None:
            break
        
        p = int(ones[best_pair[0]])
        q = int(zeros[best_pair[1]])
        h[p] = 0.0
        h[q] = 1.0
        
        actual_pk = float(fft_corr(h).max())
        if actual_pk < best_pk:
            best_pk = actual_pk
            best_h = h.copy()
    
    return best_h, best_pk

def efficient_local_search(h, N, deadline, M=30, verify_k=15):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h==1)[0]
        np.random.shuffle(idx)
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h==0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1
    
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        
        top_M_idx = np.argpartition(corr, -M)[::-1][:M]
        top_M_idx = top_M_idx[np.argsort(corr[top_M_idx])[::-1]]
        top_M_vals = corr[top_M_idx]
        lags = top_M_idx - (N - 1)  # convert to lag values
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones < 1 or n_zeros < 1:
            break
        
        # Base delta (ignoring collision corrections)
        ones_2d = ones[:, None].astype(int)
        zeros_2d = zeros[:, None].astype(int)
        lags_1d = lags[None, :].astype(int)
        
        ip = ones_2d + lags_1d
        im = ones_2d - lags_1d
        iq = zeros_2d + lags_1d
        imq = zeros_2d - lags_1d
        
        v_p = (ip >= 0) & (ip < N)
        v_m = (im >= 0) & (im < N)
        v_q = (iq >= 0) & (iq < N)
        v_mq = (imq >= 0) & (imq < N)
        
        # Base terms
        t1 = np.where(v_p, -(1.0 - h[np.clip(ip, 0, N-1)]), 0.0)  # -(1-h[p-m])
        t2 = np.where(v_m, h[np.clip(im, 0, N-1)], 0.0)  # h[p+m]
        t3 = np.where(v_q, (1.0 - h[np.clip(iq, 0, N-1)]), 0.0)  # (1-h[q-m])
        t4 = np.where(v_mq, -h[np.clip(imq, 0, N-1)], 0.0)  # -h[q+m]
        
        # Combined: delta_base[i,j,k] = t1[i,k] + t2[i,k] + t3[j,k] + t4[j,k]
        AB = t1 + t2  # (n_ones, M)
        CD = t3 + t4  # (n_zeros, M)
        
        delta = AB[:, None, :] + CD[None, :, :]  # (n_ones, n_zeros, M)
        new_vals = top_M_vals[None, None, :] + delta
        new_maxes = np.max(new_vals, axis=2)  # (n_ones, n_zeros)
        
        # Find top verify_k candidates
        flat = new_maxes.flatten()
        kval = verify_k if verify_k < len(flat) else len(flat)
        top_k_flat = np.argpartition(flat, kval)[:kval]
        cand_pairs = [np.unravel_index(int(i), new_maxes.shape) for i in top_k_flat]
        cand_pairs.sort(key=lambda x: new_maxes[x[0], x[1]])
        
        best_improved = False
        for (ip_idx, iq_idx) in cand_pairs:
            p = int(ones[ip_idx])
            q = int(zeros[iq_idx])
            h[p] = 0.0
            h[q] = 1.0
            actual_pk = float(np.max(fft_corr(h)))
            h[p] = 1.0
            h[q] = 0.0
            if actual_pk < pk - 1e-12:
                h[p] = 0.0
                h[q] = 1.0
                best_improved = True
                if actual_pk < best_pk:
                    best_pk = actual_pk
                    best_h = h.copy()
                break
        
        if not best_improved:
            break
    
    return best_h, best_pk

Can we achieve max_corr = 72 with N=380? The average corr value over all lags is:
Sum over all lags of corr[lag] = sum_i h[i] * sum_s (1-h[i+s]) = sum_i h[i] * (N - sum_s h[i+s])

For binary h with 190 ones and 190 zeros:
sum_s (1-h[i+s]) over all valid s = number of zeros in the full range = 190 (since for each i, as s ranges over all valid values, i+s ranges over all of [0,N))

Wait, that's not quite right. For a specific i, the valid s range is -(N-1) to (N-1), but i+s must be in [0,N-1]. So the number of valid s for position i is i+1 + (N-1-i) = N. No wait, i+s ranges from max(0, i-(N-1)) to min(N-1, i+(N-1)), which for i in [0,N-1] is just [0,N-1], giving N values of s. But s itself ranges from -i to N-1-i, which is N values.

So for each i, sum over valid s of (1-h[i+s]) = number of positions j in [0,N-1] where h[j]=0 = N-190 = 190.

Total sum over all i and s: sum_i h[i] * 190 = 190 * 190 = 36100.

Number of lags: 2N-1 = 759.
Average corr per lag: 36100/759 ≈ 47.56.

So the average is about 47.6, and we're trying to get the max down to 72. That's not too extreme.

For a truly uniform distribution (which is impossible for binary), the max would be close to the mean. The question is whether 72 is achievable.
