def nesterov_subgrad(h, N, deadline, top_k=15, lr0=0.05):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    center = N - 1
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        c5 = pk * dx
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        
        # Subgradient from top lags
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for mi in top_idx:
            k = int(mi - center)
            if k == 0:
                grad += (1.0 - 2.0 * y)
            elif k > 0:
                grad[k:] += (1.0 - y[:N - k])
                grad[:N - k] -= y[k:]
            else:
                kk = -k
                grad[:N - kk] += (1.0 - y[kk:])
                grad[kk:] -= y[:N - kk]

        m = beta * (h_new - h)
        h = h_new
        
        if it % 500 == 0:
            lr *= 0.98
            beta = min(0.95, 0.85 + it * 0.00005)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):

            lo = mid
        else:
            hi = mid

        h_int = np.interp(x_new, x_old, best_h)
        hb = ensure_binary(h_int, N_try)
        d6 = min(deadline - 40, time.time() + 45)
        h6, _ = binary_search(hb, N_try, d6, max_iters=60000, double_swap=False, anneal=False)
        c5_6 = compute_c5(h6, N_try)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final Nesterov polish
    if time.time() < deadline - 15:
        hF = best_h + np.random.randn(len(best_h)) * 0.015
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 10, top_k=30, lr0=0.02)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Phase 8: Final binary cleanup
    if time.time() < deadline - 1:
        hB = ensure_binary(best_h, len(best_h))
        hB, _ = binary_search(hB, len(best_h), deadline - 0.5, max_iters=30000, double_swap=False, anneal=False)
        c5B = compute_c5(hB, len(best_h))
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    # Final validation
    final_h = ensure_binary(best_h, len(best_h))
    final_c5 = compute_c5(final_h, len(final_h))
    return final_h, final_c5, len(final_h)
