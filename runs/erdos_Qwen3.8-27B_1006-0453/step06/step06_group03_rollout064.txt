def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def swap_delta(h, p, q):
    N = len(h)
    c = N - 1
    d = np.zeros(2 * N - 1)
    # p: 1->0
    if p > 0:
        d[c+1:c+1+p] -= (1.0 - h[p-1::-1])
    if N-1-p > 0:
        d[c+1:c+1+N-1-p] += h[p+1:p+N]
    if N-1-p > 0:
        d[c-1:c-(N-1-p):-1] -= (1.0 - h[p+1:p+N])
    if p > 0:
        d[c-1:c-p:-1] += h[p-1::-1]
    # q: 0->1 (negate, swap p/q)
    if q > 0:
        d[c+1:c+1+q] += (1.0 - h[q-1::-1])
    if N-1-q > 0:
        d[c+1:c+1+N-1-q] -= h[q+1:q+N]
    if N-1-q > 0:
        d[c-1:c-(N-1-q):-1] += (1.0 - h[q+1:q+N])
    if q > 0:
        d[c-1:c-q:-1] -= h[q-1::-1]
    # Correction
    ad = abs(p - q)
    if 0 < ad < N:
        d[c+ad] += 1.0
        d[c-ad] += 1.0
    return d

    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def swap_delta(h, p, q):
    N = len(h)
    c = N - 1
    d = np.zeros(2 * N - 1)
    r = N - 1
    if p > 0:
        d[c+1:c+1+p] -= (1.0 - h[p-1::-1])
        d[c-1:c-p:-1] += h[p-1::-1]
    np_ = r - p
    if np_ > 0:
        d[c+1:c+1+np_] += h[p+1:p+N]
        d[c-1:c-np_:-1] -= (1.0 - h[p+1:p+N])
    if q > 0:
        d[c+1:c+1+q] += (1.0 - h[q-1::-1])
        d[c-1:c-q:-1] -= h[q-1::-1]
    nq = r - q
    if nq > 0:
        d[c+1:c+1+nq] -= h[q+1:q+N]
        d[c-1:c-nq:-1] += (1.0 - h[q+1:q+N])
    ad = abs(p - q)
    if 0 < ad < N:
        d[c+ad] += 1.0
        d[c-ad] += 1.0
    return d

def make_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    t = N // 2
    n1 = int(h.sum())
    if n1 > t:
        idx = np.where(h == 1)[0]
        h[idx[rng.permutation(len(idx))[t:]]] = 0
    elif n1 < t:
        idx = np.where(h == 0)[0]
        h[idx[rng.permutation(len(idx))[:t-n1]]] = 1
    return h

def local_search(h, N, deadline, max_sweeps=2000, sn=100):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        s = min(sn, len(ones), len(zeros))
        pl = np.random.choice(ones, s, replace=False)
        ql = np.random.choice(zeros, s, replace=False)
        bn, bpq = c, None
        for p in pl:
            for q in ql:
                dd = swap_delta(h, int(p), int(q))
                nm = float(np.max(corr + dd))
                if nm < bn:
                    bn, bpq = nm, (int(p), int(q))
        if bpq is not None and bn < c:
            h[bpq[0]], h[bpq[1]] = 0.0, 1.0
            corr = fft_corr(h)
            stall = 0 if bn < best_pk else stall + 1
            if bn < best_pk:
                best_pk, best_h = bn, h.copy()
        else:
            stall += 1
            if np.random.random() < min(0.3, 0.03 + stall * 0.005):
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                corr = fft_corr(h)
            elif stall > 500:
                break
    return best_h, best_pk

def golden_h(N, seed):
    rng = np.random.RandomState(seed)
    phi = (1.0 + np.sqrt(5)) / 2.0
    h = np.zeros(N)
    t = N // 2
    vals = (np.arange(t) * phi) % 1.0
    idx = (vals * N).astype(int) % N
    h[idx] = 1.0
    return make_binary(h, N, rng)

def run(seed=42, budget_s=1100, **kwargs):

        hn[fi], hn[fj] = 0.0, 1.0

        if c5 < best_c5:

        if time.time() > dl - 20:

        if c5 < best_c5:
