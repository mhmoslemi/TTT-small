def cont_opt(h, N, deadline, top_k=40, lr0=0.005):
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1
    lr = lr0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx_k = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx_k:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N - s] += (1.0 - h[s:])
                grad[s:] -= h[:N - s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N - s2])
                grad[:N - s2] -= h[s2:]
        grad /= top_k
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 400 == 0:
            lr *= 0.96
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(fft_corr(h0))) * (2.0 / N0)

    # Phase 1: Quick continuous to find good binarization landscape
    d1 = min(deadline - 500, t0 + 90)
    hc = h0 + np.random.randn(N0) * 0.03
    hc = proj_sum(hc, N0 / 2.0)
    hc2, pkC = cont_opt(hc, N0, d1, top_k=50, lr0=0.005)
    c5C = pkC * (2.0 / N0)
    if c5C < best_c5:
        best_c5 = c5C
        best_h = hc2.copy()

    # Phase 2: Long binary anneal with 2-swap
    hB = binarize(best_h, N0)
    d2 = min(deadline - 280, time.time() + 250)
    hb, pkB = anneal(hB, N0, d2, T0=0.08, cool=0.9994, reheat_T=0.025, max_stall=500, p_2swap=0.3)
    c5B = pkB * (2.0 / N0)
    if c5B < best_c5:
        best_c5 = c5B
        best_h = hb.copy()

    # Phase 3: Perturbed restart anneal
    if time.time() < deadline - 180:
        d3 = min(deadline - 160, time.time() + 140)
        hB2 = best_h.copy()
        ones = np.where(hB2 == 1)[0]
        zeros = np.where(hB2 == 0)[0]
        nflip = 12
        for _ in range(nflip):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            hB2[p] = 0
            hB2[q] = 1
            ones = np.where(hB2 == 1)[0]
            zeros = np.where(hB2 == 0)[0]
        hb2, pkB2 = anneal(hB2, N0, d3, T0=0.05, cool=0.9996, reheat_T=0.015, max_stall=400, p_2swap=0.2)
        c5B2 = pkB2 * (2.0 / N0)
        if c5B2 < best_c5:
            best_c5 = c5B2
            best_h = hb2.copy()

    # Phase 4: Multi-N exploration
    for Nt in [220, 280, 340, 420, 480, 560, 640, 740]:
        if time.time() > deadline - 50:
            break
        if Nt % 2:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = binarize(hi, Nt)
        sub = min(deadline - 3, time.time() + 22)
        hrt, pkr = anneal(hbt, Nt, sub, T0=0.04, cool=0.9997, reheat_T=0.01, max_stall=250, p_2swap=0.15)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hrt.copy()

    # Phase 5: Final polish on best
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hf = binarize(best_h, Nf)
        hfp, pkf = anneal(hf, Nf, deadline - 1, T0=0.02, cool=0.9998, reheat_T=0.005, max_stall=200, p_2swap=0.1)
        c5f = pkf * (2.0 / Nf)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hfp.copy()

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
