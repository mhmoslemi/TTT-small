        lr *= 0.9998

            h = project_sum(h, target)
            v[:] = 0
            lr = 0.04
            beta = 50.0
            stall = 0
    
    return best, best_val*(2.0/N)

def binary_opt(h_init, N, deadline):

    corr = fft_corr(h)
    best_int = int(corr.max())
    best_h = h.copy()
    stall = 0
    it = 0

        zeros = np.where(h==0)[0]

        # Try multiple random candidates, keep best

        best_p, best_q = -1, -1
        best_c = None

        for _ in range(n_try):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            c_new = fft_corr(h)
            mx = int(c_new.max())

            h[best_q] = 1

                best_h = h.copy()

                h[qs] = 1
                h = fix_binary(h, N/2.0)
                corr = fft_corr(h)
                stall = 0

def interp(h, N_new):

    x_new = np.linspace(0, 2, N_new, endpoint=False)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fix_binary(h, target):
    n1 = int(h.sum())
    tgt = int(target)
    if n1 > tgt:
        idx = np.where(h == 1)[0]
        h[np.random.permutation(idx)[:n1 - tgt]] = 0
    elif n1 < tgt:
        idx = np.where(h == 0)[0]
        h[np.random.permutation(idx)[:tgt - n1]] = 1
    return h

def build_idx(N):
    a = np.arange(N)
    d = a[None, :] - a[:, None]
    return (N - 1 + d).astype(np.int64), (N - 1 - d).astype(np.int64)

def softgrad(corr, h, N, w1i, w2i, beta):
    sh = beta * (corr - corr.max())
    w = np.exp(sh)
    w /= w.sum()
    return w[w1i] @ (1.0 - h) - w[w2i] @ h

def smooth_opt(h0, N, dl, w1i, w2i, b0=40.0, lr0=0.05):
    tgt = N / 2.0
    h = project_sum(h0.copy(), tgt)
    best = h.copy()
    bv = fft_corr(h).max()
    v = np.zeros(N)
    beta, lr, mu, stall = b0, lr0, 0.85, 0
    while time.time() < dl:
        corr = fft_corr(h)
        if corr.max() < bv:
            bv = corr.max()
            best = h.copy()
            stall = 0
        else:
            stall += 1
        g = softgrad(corr, h, N, w1i, w2i, beta)
        v = mu * v - lr * g
        h = project_sum(h + v, tgt)
        beta = min(800.0, beta * 1.008)
        lr = max(0.003, lr * 0.9998)
        if stall > 350:
            h = best + np.random.randn(N) * 0.035
            h = project_sum(h, tgt)
            v[:] = 0.0
            lr, beta, stall = lr0 * 0.9, b0, 0
    return best, bv * (2.0 / N)

def binary_opt(h0, N, dl, ntry=35):
    h = (h0 > 0.5).astype(float).copy()
    h = fix_binary(h, N / 2.0)
    corr = fft_corr(h)
    bi = int(corr.max())
    bh = h.copy()
    stall = 0
    while time.time() < dl:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        bs, bp, bq, bc = bi, -1, -1, None
        for _ in range(ntry):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            cn = fft_corr(h)
            mx = int(cn.max())
            if mx < bs:
                bs, bp, bq, bc = mx, p, q, cn.copy()
            h[p], h[q] = 1.0, 0.0
        if bp >= 0:
            h[bp], h[bq] = 0.0, 1.0
            corr = bc
            if bs < bi:
                bi, bh, stall = bs, h.copy(), 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 90:
                k = min(3, len(ones) // 2, len(zeros) // 2)
                ps = np.random.choice(ones, k, replace=False)
                qs = np.random.choice(zeros, k, replace=False)
                h[ps] = 0
                h[qs] = 1
                h = fix_binary(h, N / 2.0)
                corr = fft_corr(h)
                stall = 0
    return bh, bi * (2.0 / N)

def interp(h, Nn):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, Nn, endpoint=False)
    return np.interp(xn, xo, h)

    for Nt in [190, 760, 380]:
        if time.time() > dl - 50:
            break
        if Nt % 2 != 0:
            Nt += 1
        ht = interp(best_h, Nt)
        ht_bin = fix_binary((ht > 0.5).astype(float), Nt / 2.0)
        c5t = compute_c5(ht_bin, Nt)
        if time.time() < dl - 30:

            cr2 = compute_c5(hr, Nt)
            sub2 = min(dl - 20, time.time() + 30)

        elif c5t < best_c5:
            best_c5, best_h = c5t, ht_bin

    # Final Nesterov polish
    if time.time() < dl - 3:
        Nf = len(best_h)
        wf1, wf2 = build_idx(Nf)

        if cf < best_c5:

    best_h = project_sum(best_h, len(best_h) / 2.0)
    return best_h, compute_c5(best_h, len(best_h)), len(best_h)
