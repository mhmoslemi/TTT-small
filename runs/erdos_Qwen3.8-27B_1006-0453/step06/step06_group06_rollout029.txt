import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def grad_at_lag(h, lag):
    """Compute gradient of corr[lag+(N-1)] with respect to h."""
    N = len(h)
    grad = np.zeros(N)
    # corr[m] = sum_n h[n] * (1-h[n+m-(N-1)])
    # = sum_n h[n] * (1-h[n+lag]) where lag = m-(N-1)
    # d/dh[j]: 
    #   from h[j] in first factor: (1-h[j+lag]) if j+lag in [0,N)
    #   from h[j] in second factor (n+lag=j, n=j-lag): -h[j-lag] if j-lag in [0,N)
    
    # First term: (1-h[j+lag]) for valid j+lag
    if lag >= 0:
        grad[:N-lag] += (1.0 - h[lag:])
    else:
        grad[-lag:] += (1.0 - h[:N+lag])
    
    # Second term: -h[j-lag] for valid j-lag
    if lag >= 0:
        grad[lag:] -= h[:N-lag]
    else:
        grad[:N+lag] -= h[-lag:]
    
    return grad

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def grad_at_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    if lag >= 0:
        g[:N-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:N-lag]
    else:
        g[-lag:] += (1.0 - h[:N+lag])
        g[:N+lag] -= h[-lag:]
    return g

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h

            idx = np.argpartition(corr, -n_lags)[-n_lags:]
            g = np.zeros(N)
            for ai in idx:
                g += grad_at_lag(h, int(ai)-(N-1))
            g /= n_lags
        h = proj_sum(h - lr*g, N/2.0)
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        t += 1
        if t % 300 == 0:
            lr *= cool
    return best_h, best_pk

def bin_anneal(h, N, deadline, T0=0.05, cool=0.9995, max_no_imp=2500, restarts=2):
    h = fix_bin(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    s_arr = np.arange(-(N-1), N)
    for r in range(restarts):
        T = T0 * (0.4 ** r)
        no_imp = 0
        if r == 0:
            h = h.copy()
            corr = fft_corr(h)
        else:
            h = best_h.copy()
            n_flip = max(1, N//30)
            idx_f = np.random.choice(N, n_flip, replace=False)
            h[idx_f] = 1.0 - h[idx_f]
            corr = fft_corr(h)
        current_pk = float(corr.max())
        while time.time() < deadline and T > 0.0005 and no_imp < max_no_imp:
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            ps = p + s_arr; pm = p - s_arr
            qs = q + s_arr; qm = q - s_arr
            vps = (ps>=0)&(ps<N); vpm = (pm>=0)&(pm<N)
            vqs = (qs>=0)&(qs<N); vqm = (qm>=0)&(qm<N)
            d = np.zeros(2*N-1)
            d[vps] -= (1.0 - h[np.clip(ps,0,N-1)[vps]])
            d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
            d[vqs] += (1.0 - h[np.clip(qs,0,N-1)[vqs]])
            d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
            nc = corr + d
            npk = float(nc.max())
            delta = npk - current_pk
            if delta < 0 or np.random.random() < np.exp(-delta/max(T,1e-10)):
                h[p] = 0; h[q] = 1
                corr = nc
                current_pk = npk
                if npk < best_pk:
                    best_pk = npk
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
            T *= cool
    return best_h, best_pk

def best_improve_bin(h, N, n_trials=300):
    h = h.copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    s_arr = np.arange(-(N-1), N)
    improved = False
    for _ in range(n_trials):
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        ps = p + s_arr; pm = p - s_arr
        qs = q + s_arr; qm = q - s_arr
        vps = (ps>=0)&(ps<N); vpm = (pm>=0)&(pm<N)
        vqs = (qs>=0)&(qs<N); vqm = (qm>=0)&(qm<N)
        d = np.zeros(2*N-1)
        d[vps] -= (1.0 - h[np.clip(ps,0,N-1)[vps]])
        d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
        d[vqs] += (1.0 - h[np.clip(qs,0,N-1)[vqs]])
        d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
        nc = corr + d
        npk = float(nc.max())
        if npk < best_pk:
            h[p] = 0; h[q] = 1
            corr = nc
            best_pk = npk
            improved = True
    return h, best_pk, improved

def resample_h(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
