def compute_delta_c(h, a, b, N):
    """Compute change in correlation when h[a]: 1->0, h[b]: 0->1"""
    # k ranges from -(N-1) to N-1
    # delta_c[k] = h[a+k] + h[a-k] - h[b+k] - h[b-k] (out of bounds = 0)
    # We only need the max of (curr_c + delta_c)
    k_vals = np.arange(-(N-1), N)
    # For each k, a+k is valid if 0 <= a+k < N, i.e., -a <= k < N-a
    # a-k valid if 0 <= a-k < N, i.e., a-N+1 <= k <= a
    # b+k valid if -b <= k < N-b
    # b-k valid if b-N+1 <= k <= b
    
    delta = np.zeros(2*N-1)
    idx = k_vals + (N-1)  # convert to 0-indexed
    
    # h[a+k]: valid for -a <= k < N-a
    lo = max(idx[0], -a + (N-1))
    hi = min(idx[-1] + 1, N - a + (N-1))
    k_range = np.arange(lo - (N-1), hi - (N-1))
    delta[lo:hi] += h[a + k_range]
    
    # h[a-k]: valid for a-N+1 <= k <= a
    lo = max(idx[0], a - N + 1 + (N-1))
    hi = min(idx[-1] + 1, a + 1 + (N-1))
    k_range = np.arange(lo - (N-1), hi - (N-1))
    delta[lo:hi] += h[a - k_range]
    
    # h[b+k]: valid for -b <= k < N-b
    lo = max(idx[0], -b + (N-1))
    hi = min(idx[-1] + 1, N - b + (N-1))
    k_range = np.arange(lo - (N-1), hi - (N-1))
    delta[lo:hi] -= h[b + k_range]
    
    # h[b-k]: valid for b-N+1 <= k <= b

    hi = min(idx[-1] + 1, b + 1 + (N-1))
    k_range = np.arange(lo - (N-1), hi - (N-1))
    delta[lo:hi] -= h[b - k_range]
    
    return delta

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(1.0 - h[::-1], L)))

def c5(h):
    return float(np.max(fft_corr(h))) * 2.0 / len(h)

def proj01(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - tgt) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        m = (lo + hi) / 2
        if np.clip(h - m, 0, 1).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) / 2, 0, 1)

def delta_corr(h, a, b, N):
    c = N - 1
    kmin = max(-c, -a, a - c, -b, b - c)
    kmax = min(c, c - a, a, c - b, b)
    if kmin > kmax:
        return np.empty(0), np.empty(0, dtype=int)
    ks = np.arange(kmin, kmax + 1)
    d = np.zeros(len(ks))
    m = (a + ks >= 0) & (a + ks < N); d[m] += h[a + ks[m]]
    m = (a - ks >= 0) & (a - ks < N); d[m] += h[a - ks[m]]
    m = (b + ks >= 0) & (b + ks < N); d[m] -= h[b + ks[m]]
    m = (b - ks >= 0) & (b - ks < N); d[m] -= h[b - ks[m]]
    return d, ks + c

def sa_binary(h, N, deadline, tgt_max, n_iter, T0, cool, tgt_frac=0.7):
    h = h.copy()
    C = fft_corr(h)
    cmax = float(np.max(C))
    bh, bc = h.copy(), cmax
    T = T0
    c = N - 1
    rng = np.random
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 400 == 0:
            T = max(T * cool, 0.005)
        if cmax <= tgt_max:
            break
        if rng.random() < tgt_frac:
            kstar = int(np.argmax(C)) - c
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if kstar >= 0:
                va = ones[(ones + kstar >= N) | (h[np.clip(ones + kstar, 0, N-1)] == 0)]
                vb = zeros[(zeros + kstar >= N) | (h[np.clip(zeros + kstar, 0, N-1)] == 1)]
            else:
                ks = -kstar
                va = ones[ones >= ks]
                va = va[(va - ks < 0) | (h[va - ks] == 0)]
                vb = zeros[zeros >= ks]
                vb = vb[(vb - ks < 0) | (h[vb - ks] == 1)]
            if len(va) > 0 and len(vb) > 0:
                a = int(rng.choice(va)); b = int(rng.choice(vb))
            else:
                a = int(rng.choice(ones)); b = int(rng.choice(zeros))
        else:
            a = int(rng.choice(np.where(h == 1)[0]))
            b = int(rng.choice(np.where(h == 0)[0]))
        d, idx = delta_corr(h, a, b, N)
        cmax_new = cmax
        C_new = C
        if len(d) > 0:
            C_new = C.copy()
            C_new[idx] += d
            cmax_new = float(np.max(C_new))
        acc = cmax_new < cmax or (cmax_new == cmax and rng.random() < 0.4)
        if not acc and cmax_new > cmax:
            acc = rng.random() < np.exp(-(cmax_new - cmax) / max(T, 1e-9))
        if acc:
            h[a], h[b] = 0.0, 1.0
            C = C_new
            cmax = cmax_new
            if cmax < bc:
                bc, bh = cmax, h.copy()
    return bh, bc

def nesterov(h, N, deadline, k=60, lr0=0.015):
    h = h.copy()
    target = N / 2.0
    bx, bp = h.copy(), float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr, beta, it = lr0, 0.9, 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < bp:
            bp, bx = pk, h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in top:
            L = int(ai - ctr)
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += 1.0 - y[:N - L]
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += 1.0 - y[M:]
                    grad[M:] -= y[:N - M]
        grad /= k
        h = proj01(y - lr * grad, target)
        m = beta * (h - y + (y - m))
        it += 1
        if it % 200 == 0:
            lr *= 0.995
            beta = min(0.97, 0.88 + it * 0.0002)
    return bx, bp * (2.0 / N)
