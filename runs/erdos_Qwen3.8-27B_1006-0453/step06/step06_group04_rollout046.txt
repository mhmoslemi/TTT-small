import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_grad(h, N, beta):
    """Gradient of LogSumExp(beta * corr) w.r.t. h, scaled by dx."""
    corr = fft_corr(h)
    mx = corr.max()
    w = np.exp((corr - mx) * beta)
    w /= w.sum()
    dx = 2.0 / N
    grad = np.zeros(N)
    # L=0
    if w[N-1] > 1e-15:
        grad += w[N-1] * (1.0 - 2.0 * h) * dx
    for L in range(1, N):
        wt = w[N-1+L] + w[N-1-L]
        if wt < 1e-15:
            continue
        seg = N - L
        grad[:seg] += wt * (1.0 - h[L:]) * dx
        grad[L:] -= wt * h[:seg] * dx
    return grad, float(np.max(corr)) * dx

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_gradient(h, N, beta):
    corr = fft_corr(h)
    mx = corr.max()
    w = np.exp((corr - mx) * beta)
    w /= w.sum()
    dx = 2.0 / N
    grad = np.zeros(N)
    w0 = w[N - 1]
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h) * dx
    for L in range(1, N):
        wt = w[N - 1 + L] + w[N - 1 - L]
        if wt < 1e-14:
            continue
        seg = N - L
        grad[:seg] += wt * (1.0 - h[L:]) * dx
        grad[L:] -= wt * h[:seg] * dx
    return grad

    # Phase 4: Multi-restart binary (150s)
    d4 = max(d3, t0) + 150
    d4 = min(deadline - 50, d4)
    for _ in range(3):
        if time.time() > d4 - 10:
            break
        hR = (best_h + np.random.randn(N0) * 0.15 > 0.5).astype(float)
        hR = fix_binary_sum(hR, N0)
        dR = min(d4, time.time() + 45)
        hR, cR = binary_opt(hR, N0, dR, n_inner=18)
        if cR < best_c5:
            best_c5, best_h = cR, hR.copy()
    # Phase 5: Final LSE + binary polish (50s)
    if time.time() < deadline - 10:
        hF = best_h + np.random.randn(N0) * 0.025
        hF = project_sum(hF, N0 / 2.0)
        hF, cF = lse_adam(hF, N0, deadline - 15, beta_init=120.0, lr0=0.015)
        hFb = fix_binary_sum((hF > 0.5).astype(float), N0)
        cFb = compute_c5(hFb, N0)
        if cFb < best_c5:
            best_c5, best_h = cFb, hFb.copy()
    # Final binary check
    hb = fix_binary_sum((best_h > 0.5).astype(float), N0)
    cb = compute_c5(hb, N0)
    if cb < best_c5:
        best_c5, best_h = cb, hb.copy()
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
