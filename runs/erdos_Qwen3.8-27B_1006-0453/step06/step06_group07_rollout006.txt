def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D_all(h, N, ones):
    """Compute D matrix for given one positions."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = ones[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D_all(h, N, ones)
        Dz = compute_D_all(h, N, zeros)
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            pi = int(np.where(ones == p)[0][0])
            base = c - D[pi, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def sa_search(h, N, deadline, T0=0.003, T_end=0.00005, two_bit_prob=0.2, rng=None):
    """Simulated annealing with 1-bit and 2-bit moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_h = h.copy()
    best_P = float(np.max(fft_corr(h)))
    t_start = time.time()
    total_est = 300  # estimate total sweeps
    
    while time.time() < deadline:
        elapsed = time.time() - t_start
        frac = min(1.0, elapsed / total_est)
        T = T0 * (T_end / T0) ** frac
        
        c = fft_corr(h)
        P = float(c.max())
        if P < best_P - 1e-14:
            best_P = P
            best_h = h.copy()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        if rng.random() < two_bit_prob:
            # 2-bit move: pick 2 random ones, 2 random zeros
            p1, p2 = rng.choice(ones, 2, replace=False)
            q1, q2 = rng.choice(zeros, 2, replace=False)
            h_try = h.copy()
            h_try[p1] = h_try[p2] = 0.0
            h_try[q1] = h_try[q2] = 1.0
            new_P = float(np.max(fft_corr(h_try)))
            delta = new_P - P
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                h = h_try
        else:
            # 1-bit move: greedy best or random
            D = compute_D_all(h, N, ones)
            Dz = compute_D_all(h, N, zeros)
            # Pick a random subset of ones for speed
            if len(ones) > 60:
                p_list = rng.choice(ones, 40, replace=False)
            else:
                p_list = ones
            best_new_P = P
            best_pq = None
            for p in p_list:
                pi = int(np.where(ones == p)[0][0])
                if pi >= len(D):
                    continue
                base = c - D[pi, :]
                cand = base[None, :] - Dz
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
                if npk[mi] < best_new_P:
                    best_new_P = float(npk[mi])
                    best_pq = (int(p), int(zeros[mi]))
            if best_pq is not None:
                delta = best_new_P - P
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    h[best_pq[0]] = 0.0
                    h[best_pq[1]] = 1.0
            else:
                # No improving move; try random move
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
    
    return best_h, best_P * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def nesterov_opt(h, N, deadline, k=50, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it * 0.0001)
    return best_h, best_pk * dx
