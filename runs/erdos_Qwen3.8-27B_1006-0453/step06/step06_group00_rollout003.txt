import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def annil_search(h, N, deadline, max_iters=20000, block_sizes=(1,1,1,2,2,3,5),
                 T0=0.05, T1=1e-5, use_blocks=True):
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = T0
    cool = (T1 / T0) ** (1.0 / max(1, int(max_iters)))
    no_improve = 0
    stall_limit = 3000

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        L = int(np.random.choice(block_sizes))
        if L == 1:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            if oi == zi:
                continue
            h[oi] = 0
            h[zi] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < -1e-14 or (T > 1e-7 and np.random.random() < np.exp(-dE / T)):
                cur_max = new_max
                corr = new_corr
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[oi] = 1
                h[zi] = 0
        else:
            if not use_blocks:
                continue
            one_blocks = []
            zero_blocks = []
            i = 0
            while i < N:
                if h[i] == 1:
                    j = i
                    while j < N and h[j] == 1:
                        j += 1
                    if j - i >= L:
                        one_blocks.append(i)
                    i = j
                else:
                    j = i
                    while j < N and h[j] == 0:
                        j += 1
                    if j - i >= L:
                        zero_blocks.append(i)
                    i = j
            if not one_blocks or not zero_blocks:
                continue
            ob = int(np.random.choice(one_blocks))
            zb = int(np.random.choice(zero_blocks))
            if abs(ob - zb) < L:
                continue
            h[ob:ob+L] = 0
            h[zb:zb+L] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < -1e-14 or (T > 1e-7 and np.random.random() < np.exp(-dE / T)):
                cur_max = new_max
                corr = new_corr
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[ob:ob+L] = 1
                h[zb:zb+L] = 0
        T *= cool
        if no_improve > stall_limit:
            # Restart with perturbation
            for _ in range(5):
                oi = int(np.random.choice(np.where(h == 1)[0]))
                zi = int(np.random.choice(np.where(h == 0)[0]))
                h[oi], h[zi] = h[zi], h[oi]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, 0.01)
            no_improve = 0
    return best_h, best_c

def continuous_refine(h, N, deadline, lr0=0.006):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    soft_temp = 0.03
    it = 0
    while time.time() < deadline - 2:
        it += 1
        corr = fft_corr(h)
        peak = np.max(corr)
        w = np.exp((corr - peak) / soft_temp)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        significant = np.where(w > 0.0005)[0]
        for ki in significant:
            L = int(ki - center)
            wt = w[ki]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wt * (1.0 - h[:N-L])
                grad[:N-L] -= wt * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += wt * (1.0 - h[M:])
                    grad[M:] -= wt * h[:N-M]
        h_new = np.clip(h - lr * grad, 0, 1)
        lo, hi = -5.0, 5.0
        for _ in range(45):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        c_new = compute_c5(h_new, N)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 300 == 0:
            lr *= 0.985
            soft_temp *= 0.96
    return best_h, best_c

def structured_init(N, variant):
    np.random.seed(7777 + variant * 131)
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[np.random.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        h = (np.arange(N) % 2).astype(float)
        h = fix_sum(h, nh)
    elif variant == 2:
        x = np.arange(N)
        v = np.sin(2*np.pi*x*5.0/N) + 0.7*np.sin(2*np.pi*x*11.0/N) + 0.5*np.cos(2*np.pi*x*3.0/N)
        h = (v > np.median(v)).astype(float)
        h = fix_sum(h, nh)
    elif variant == 3:
        h = ((np.arange(N)*7) % 11 < 5).astype(float)
        h = fix_sum(h, nh)
    else:
        x = np.arange(N)
        v = 0.0
        for f in [3, 7, 13, 21]:
            v += np.sin(2*np.pi*f*x/N + 0.3*f)
        h = (v > np.median(v)).astype(float)
        h = fix_sum(h, nh)
    return h
