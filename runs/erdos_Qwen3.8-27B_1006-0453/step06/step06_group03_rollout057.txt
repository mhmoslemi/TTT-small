def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))

    lo, hi = -1.0, 1.0

        else:

            if q < N - M:

    return out

c[N-1-M] = sum_{n=0}^{N-1-M} h[n]*(1-h[n+M])

p changes 1->0:
- p appears as n (the h side): valid if 0 <= p <= N-1-M, i.e., p < N-M
  Old contribution: h[p]*(1-h[p+M]) = 1*(1-h[p+M])
  New: 0*(1-h[p+M]) = 0
  Delta: -(1-h[p+M])
- p appears as n+M (the v=1-h side): valid if n = p-M >= 0, i.e., p >= M, and n <= N-1-M, i.e., p-M <= N-1-M, i.e., p <= N-1 (always true)
  Old contribution: h[p-M]*(1-h[p]) = h[p-M]*0 = 0
  New: h[p-M]*(1-0) = h[p-M]
  Delta: +h[p-M]

q changes 0->1:
- q appears as n: valid if q < N-M
  Old: 0*(1-h[q+M]) = 0
  New: 1*(1-h[q+M])

- q appears as n+M: valid if q >= M
  Old: h[q-M]*(1-0) = h[q-M]
  New: h[q-M]*(1-1) = 0
  Delta: -h[q-M]

def softmax_grad(h, beta, N):
    """Gradient of log-sum-exp(beta * c_k) / beta"""
    corr = fft_corr(h)
    # Softmax weights
    max_c = np.max(corr)
    weights = np.exp(beta * (corr - max_c))  # for numerical stability
    weights /= weights.sum()
    
    # Gradient of c_k w.r.t. h[i]
    # c_k = sum_j h[j] * (1-h[j-k]) for appropriate k
    # This is the same as the subgradient but weighted
    
    grad = np.zeros(N)
    center = N - 1
    for ai in range(len(corr)):
        w = weights[ai]
        if w < 1e-10:
            continue
        L = int(ai - center)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N-L])
            grad[:N-L] -= w * h[L:]
        else:
            M = -L
            grad[:N-M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:N-M]
    return grad / beta

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def swap_delta_top(c, h, p, q, top_idx, N):
    center = N - 1
    out = np.empty(len(top_idx))
    for i in range(len(top_idx)):
        ai = top_idx[i]
        L = int(ai - center)
        if L == 0:
            out[i] = c[ai]
            continue
        d = 0.0
        if L > 0:
            if p >= L:
                d -= (1.0 - h[p - L])
            if p < N - L:
                d += h[p + L]
            if q >= L:
                d += (1.0 - h[q - L])
            if q < N - L:
                d -= h[q + L]
        else:
            M = -L
            if p < N - M:
                d -= (1.0 - h[p + M])
            if p >= M:
                d += h[p - M]
            if q < N - M:
                d += (1.0 - h[q + M])
            if q >= M:
                d -= h[q - M]
        out[i] = c[ai] + d
    return out

def subgrad_vec(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    if top_k >= len(corr):
        top = np.arange(len(corr))
    else:
        top = np.argpartition(corr, -top_k)[-top_k:]
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    grad /= len(top)
    return grad

def nesterov_opt(h, N, deadline, k=60, lr0=0.02, beta0=0.90):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_vec(y, corr, k, N)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.99
            beta = min(0.97, beta0 + it * 0.0002)
            if it % 300 == 0:
                lr = max(0.003, lr * 0.9)
    return best_h, best_pk * (2.0 / N)

def local_search(h, N, deadline, top_m=25, sample_n=50, max_patience=120, anneal_rate=0.3):
    h = h.astype(float).copy()
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 != nh:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if n1 > nh:
            idx = np.random.choice(ones, n1 - nh, replace=False)
            h[idx] = 0
        else:
            idx = np.random.choice(zeros, nh - n1, replace=False)
            h[idx] = 1
    best_c = float(np.max(fft_corr(h)))
    patience = 0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            patience = 0
        else:
            patience += 1
        if patience > max_patience:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argsort(c)[-top_m:]
        best_new_P = P
        best_move = None
        n_analyze = min(10, len(top_idx))
        for ai in top_idx[:n_analyze]:
            L = int(ai - (N - 1))
            if L == 0:
                continue
            aL = abs(L)
            if L > 0:
                bo = np.where((h[:N-aL] == 1) & (h[aL:] == 0))[0]
                bz = np.where((h[:N-aL] == 0) & (h[aL:] == 1))[0]
            else:
                bo = np.where((h[aL:] == 1) & (h[:N-aL] == 0))[0] + aL
                bz = np.where((h[aL:] == 0) & (h[:N-aL] == 1))[0] + aL
            if len(bo) < 1 or len(bz) < 1:
                continue
            sp = min(sample_n, len(bo))
            sq = min(sample_n, len(bz))
            pc = np.random.choice(bo, sp, replace=False)
            qc = np.random.choice(bz, sq, replace=False)
            for p in pc:
                for q in qc:
                    if p == q:
                        continue
                    nv = swap_delta_top(c, h, int(p), int(q), top_idx, N)
                    nm = float(np.max(nv))
                    if nm < best_new_P:
                        best_new_P = nm
                        best_move = (int(p), int(q))
        nr = min(8, len(ones), len(zeros))
        for _ in range(nr):
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            nv = swap_delta_top(c, h, p, q, top_idx, N)
            nm = float(np.max(nv))
            if nm < best_new_P:
                best_new_P = nm
                best_move = (p, q)
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
        else:
            if np.random.random() < anneal_rate * (0.5 + 0.5 * patience / max_patience):
                if len(ones) >= 2 and len(zeros) >= 2 and np.random.random() < 0.3:
                    p1, p2 = np.random.choice(ones, 2, replace=False)
                    q1, q2 = np.random.choice(zeros, 2, replace=False)
                    if len(set([p1, p2, q1, q2])) == 4:
                        h[p1], h[q1] = 0.0, 1.0
                        h[p2], h[q2] = 0.0, 1.0
                        np2 = float(np.max(fft_corr(h)))
                        if np2 >= best_c:
                            h[p1], h[q1] = 1.0, 0.0
                            h[p2], h[q2] = 1.0, 0.0
                        else:
                            patience = 0
                else:
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p], h[q] = 0.0, 1.0
            patience += 2
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

        c5b = compute_c5(cs)
        if c5b < best_c5:

    # Phase 4: Try nearby N values
    res_list = [378, 382, 384, 386, 390, 400]
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in res_list:
        if time.time() > deadline - 80:
            break
        if Nt % 2 != 0:
            Nt += 1
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_best, best_h)
        hb2 = (hi > 0.5).astype(float)
        nh2 = Nt // 2
        n1 = int(hb2.sum())
        if n1 != nh2:
            ones2 = np.where(hb2 == 1)[0]

        sub_dl = min(deadline - 40, time.time() + 50)
        hr, _ = local_search(hb2, Nt, sub_dl, top_m=24, sample_n=35, max_patience=60, anneal_rate=0.2)
        c5r = compute_c5(hr)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_best = np.linspace(0, 2, Nt, endpoint=False)

    # Phase 5: Final Nesterov

    target = len(best_h) / 2.0
    s = float(best_h.sum())

    final_c5 = compute_c5(best_h)
