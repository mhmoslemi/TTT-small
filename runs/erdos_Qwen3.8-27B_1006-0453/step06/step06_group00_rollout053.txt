        D = compute_D(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1_count = len(ones)
        n0_count = len(zeros)
        
        if n1_count < 1 or n0_count < 1:
            break
        
        # Find best swap: minimize max_k (corr[k] - D[a,k] + D[b,k])
        # Batch over ones
        D_ones = D[ones]  # (n1_count, 2N-1)
        D_zeros = D[zeros]  # (n0_count, 2N-1)
        
        best_improve = P + 1e-12
        best_a, best_b = -1, -1
        
        batch_sz = 30
        for start in range(0, n1_count, batch_sz):
            end = min(start + batch_sz, n1_count)
            Do = D_ones[start:end]  # (bs, 2N-1)
            # temp[i, j, k] = corr[k] - Do[i,k] + D_zeros[j,k]
            temp = corr[None, None, :] - Do[:, None, :] + D_zeros[None, :, :]  # (bs, n0, 2N-1)

                best_a = ones[start + mi[0]]
                best_b = zeros[mi[1]]
            del temp

            break
        
        h[best_a] = 0
        h[best_b] = 1

def nesterov_subgrad(h, N, deadline, k_top=80, lr0=0.015):
    """Subgradient descent on max correlation with projection."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        
        top = np.argpartition(corr, -k_top)[-k_top:]
        # Subgradient: average of gradients of top-k correlations
        # grad of corr[k] w.r.t. y[i]:
        # corr[k] = sum_j y[j]*(1-y[j-k])
        # d corr[k] / d y[i] = (1-y[i-k])[valid] - y[i+k][valid]
        # which is exactly -D[i, k_idx] where D is defined as above!
        # Wait: D[i, k] = (1-h[i-k]) - h[i+k]
        # d corr[k]/d y[i] = (1-y[i-k])*valid(i-k) - y[i+k]*valid(i+k)
        # That's the same as D[i, k]! But D was defined for the current h, and here we use y.
        # Close enough.
        
        # grad[i] = (1/k_top) * sum_{k in top} D_y[i, k]
        # where D_y is computed with y instead of h
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                # corr[0] = sum y[i]*(1-y[i]) = sum y[i] - y[i]^2
                # d/dy[i] = 1 - 2*y[i]
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                # corr[L] = sum_{i=0}^{N-1-L} y[i]*(1-y[i+L])
                # d/dy[i] = (1-y[i+L]) for i <= N-1-L (as the y[i] factor)
                #         + y[i-L] for i >= L (as the (1-y[i-L]) factor... wait)
                # Actually: d/dy[i] of y[i]*(1-y[i+L]) = (1-y[i+L])
                # d/dy[i] of y[i-L]*(1-y[i]) = -y[i-L]
                # So: d/dy[i] corr[L] = (1-y[i+L])*[i<=N-1-L] - y[i-L]*[i>=L]
                if L < N:
                    grad[:N-L] += (1.0 - y[L:])  # i from 0 to N-1-L: (1-y[i+L])
                    grad[L:] -= y[:N-L]  # i from L to N-1: -y[i-L]
            else:
                M = -L
                if M < N:
                    # corr[-M] = sum_{i=M}^{N-1} y[i]*(1-y[i-M])
                    # = sum_{j=0}^{N-1-M} y[j+M]*(1-y[j])
                    # d/dy[i]: 
                    # As y[i] factor: (1-y[i-M]) for i >= M
                    # As (1-y[i]) factor: -y[i+M] for i <= N-1-M
                    grad[M:] += (1.0 - y[:N-M])  # i from M to N-1: (1-y[i-M])
                    grad[:N-M] -= y[M:]  # i from 0 to N-1-M: -y[i+M]
        
        grad /= k_top
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.997
            beta = min(0.97, 0.88 + it * 0.0001)
        if it % 500 == 0 and lr < 0.002:
            break
    return best_h, best_pk * dx

def multi_start_binary(h_init, N, deadline, n_starts=5, perturb=[3,5,8,12,20]):
    """Multi-start binary IL with perturbations of different sizes."""
    best_h = h_init.copy()
    best_c = compute_c5(h_init, N)
    
    for i in range(n_starts):

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

        perturbs = [3, 5, 8, 12, 20, 30]
    best_h = h_init.astype(float).copy()
    best_c = compute_c5(h_init, N)
    per_time = (deadline - time.time()) / n_starts
    for i in range(n_starts):
        if time.time() > deadline - 30:
            break
        p = perturbs[i % len(perturbs)]
        h_pert = h_init.astype(float).copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        fn = min(p, len(ones), len(zeros))
        if fn < 1:
            continue
        fa = np.random.choice(ones, fn, replace=False)
        fb = np.random.choice(zeros, fn, replace=False)
        h_pert[fa] = 0
        h_pert[fb] = 1
        sub_dl = min(deadline - 30, time.time() + per_time)
        h_s, c_s = binary_il_full(h_pert, N, sub_dl, max_rounds=150)
        c5_s = compute_c5(h_s, N)
        if c5_s < best_c:
            best_c = c5_s
            best_h = h_s.copy()
            h_init = h_s.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    target = N0 // 2
    n1 = int(h_bin.sum())
    if n1 != target:
        if n1 > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[np.random.choice(idx, n1 - target, replace=False)] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[np.random.choice(idx, target - n1, replace=False)] = 1
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    d1 = min(deadline - 500, t0 + 180)
    h_s, _ = binary_il_full(best_h, N0, d1, max_rounds=400)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    d2 = min(deadline - 350, t0 + 450)
    h_ms, c_ms = multi_start_binary(best_h, N0, d2, n_starts=7, perturbs=[2, 4, 6, 10, 15, 25, 40])
    if c_ms < best_c5:
        best_c5, best_h = c_ms, h_ms.copy()

    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 130)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_01(hc, N0 / 2.0)
        hn, c5n = nesterov_subgrad(hc, N0, d3, k_top=120, lr0=0.013)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 100)
        h4 = (best_h > 0.5).astype(float)
        h_s4, _ = binary_il_full(h4, N0, d4, max_rounds=300)
        c5_4 = compute_c5(h_s4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_s4.copy()

    if time.time() < deadline - 80:
        np.random.seed(seed + 200)
        d5 = min(deadline - 60, time.time() + 70)
        h_ms2, c_ms2 = multi_start_binary(best_h, N0, d5, n_starts=5, perturbs=[2, 3, 5, 8, 12])
        if c_ms2 < best_c5:
            best_c5, best_h = c_ms2, h_ms2.copy()

    if time.time() < deadline - 40:
        d6 = min(deadline - 20, time.time() + 50)
        h6 = (best_h > 0.5).astype(float) + np.random.randn(N0) * 0.03
        h6 = project_01(h6, N0 / 2.0)
        h6, c5_6 = nesterov_subgrad(h6, N0, d6, k_top=150, lr0=0.008)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    if time.time() < deadline - 5:
        hB = (best_h > 0.5).astype(float)
        hB, _ = binary_il_full(hB, N0, deadline - 1, max_rounds=100)
        c5B = compute_c5(hB, N0)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    final_h = (best_h > 0.5).astype(float)
    n1f = int(final_h.sum())
    if n1f != N0 // 2:
        if n1f > N0 // 2:
            idx = np.where(final_h == 1)[0]
            final_h[np.random.choice(idx, n1f - N0 // 2, replace=False)] = 0
        else:
            idx = np.where(final_h == 0)[0]
            final_h[np.random.choice(idx, N0 // 2 - n1f, replace=False)] = 1
    c5_final = compute_c5(final_h, N0)
    c5_cont = compute_c5(np.clip(best_h, 0, 1), N0)
    if c5_cont < c5_final:
        return np.clip(best_h, 0, 1), c5_cont, N0
    return final_h, c5_final, N0
