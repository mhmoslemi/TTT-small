def nesterov_smooth(h, N, deadline, top_k=40, T=12.0, lr0=0.03, beta0=0.85):

def compute_D_matrix(h, N):
    """D[i, center+s] = (1-h[i+s])*1{0<=i+s<N} - h[i-s]*1{0<=i-s<N}"""
    L = 2*N - 1
    center = N - 1
    D = np.zeros((N, L))
    one_minus_h = 1.0 - h
    for s in range(-(N-1), N):
        # Term 1: (1-h[i+s]) for valid i (0<=i<N and 0<=i+s<N)
        # i ranges over max(0,-s) to min(N-1, N-1-s)
        i_lo = max(0, -s)
        i_hi = min(N-1, N-1-s)
        if i_lo <= i_hi:
            D[i_lo:i_hi+1, center+s] += one_minus_h[i_lo+s:i_hi+s+1]
        # Term 2: -h[i-s] for valid i (0<=i<N and 0<=i-s<N)
        # i-s ranges over [0, N-1], so i ranges over s to s+N-1, intersect [0,N-1]
        i_lo2 = max(0, s)

        if i_lo2 <= i_hi2:
            D[i_lo2:i_hi2+1, center+s] -= h[i_lo2-s:i_hi2-s+1]
    return D

def nesterov_smooth(h, N, deadline, top_k=50, T=12.0, lr0=0.025, beta0=0.83):
    h = h.astype(float).copy()
    target = N / 2.0

    m = np.zeros(N)
    best_h = h.copy()
    best_v = float(fft_corr(h).max())
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        corr_y = fft_corr(y)
        idx = np.argpartition(corr_y, -top_k)[-top_k:]
        top = corr_y[idx]
        w = np.exp(T * (top - top.max()))
        w /= w.sum()
        smax = float(w @ top)
        if smax < best_v:
            best_v = smax
            best_h = h.copy()
        grad = np.zeros(N)

            s = N - 1 - int(idx[ki])
            wm = w[ki]

                grad[:N - s] += wm * (1.0 - y[s:])
                grad[s:] -= wm * y[:N - s]

            no_improve += 1
        omh = 1.0 - h
        D = np.zeros((N, 2 * N - 1))
        center = N - 1
        for s in range(-(N - 1), N):
            il = max(0, -s)
            ihi = min(N - 1, N - 1 - s)
            if il <= ihi:
                D[il:ihi + 1, center + s] += omh[il + s:ihi + s + 1]
            il2 = max(0, s)
            ihi2 = min(N - 1, s + N - 1)
            if il2 <= ihi2:
                D[il2:ihi2 + 1, center + s] -= h[il2 - s:ihi2 - s + 1]
        n_ps = min(len(ones), sample_n)
        p_sel = ones[np.random.choice(len(ones), n_ps, replace=False)]
        best_score = cur_max + 1.0
        best_pq = None
        qD = D[zeros]
        for p in p_sel:
            base = corr + D[p]
            maxes = (base[None, :] - qD).max(axis=1)
            mi = int(np.argmin(maxes))
            ml = float(maxes[mi])
            if ml < best_score:
                best_score = ml
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None or (no_improve > 100 and np.random.random() < 0.12):
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
        else:
            p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        if no_improve > 800:
            break
    return h, float(np.max(fft_corr(h))) * (2.0 / N)

def two_two_exchange(h, N, deadline, n_tries=500):
    h = h.astype(float).copy()
    best_c = float(fft_corr(h).max())
    target = N // 2
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        p1, p2 = np.random.choice(ones, 2, replace=False)
        q1, q2 = np.random.choice(zeros, 2, replace=False)
        h_t = h.copy()
        h_t[p1] = 0.0
        h_t[p2] = 0.0
        h_t[q1] = 1.0
        h_t[q2] = 1.0
        nv = float(fft_corr(h_t).max())
        if nv < best_c - 1e-14:
            best_c = nv
            h = h_t
    return h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of(best_h)
    # Phase 1: Nesterov continuous
    d1 = min(deadline - 500, t0 + 160)
    hs1, cv1 = nesterov_smooth(best_h, N0, d1, top_k=60, T=10.0, lr0=0.02, beta0=0.80)
    if cv1 < best_c5:
        best_c5, best_h = cv1, hs1.copy()
    # Phase 2: Binary local search
    d2 = min(deadline - 350, time.time() + 350)
    hb2 = (best_h > 0.5).astype(float)
    hs2, cv2 = binary_search(hb2, N0, d2, max_sweeps=5000, sample_n=150)
    if cv2 < best_c5:
        best_c5, best_h = cv2, hs2.copy()
    # Phase 3: 2-2 exchanges
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 80)
        hs3, cv3 = two_two_exchange(best_h, N0, d3, n_tries=3000)
        if cv3 < best_c5:
            best_c5, best_h = cv3, hs3.copy()
    # Phase 4: Short Nesterov from binary
    if time.time() < deadline - 150:
        d4 = min(deadline - 130, time.time() + 60)
        hc4 = best_h + np.random.randn(N0) * 0.02
        hc4 = project_sum(hc4, N0 / 2.0)
        hs4, cv4 = nesterov_smooth(hc4, N0, d4, top_k=80, T=8.0, lr0=0.015, beta0=0.88)
        if cv4 < best_c5:
            best_c5, best_h = cv4, hs4.copy()
    # Phase 5: Final binary polish
    if time.time() < deadline - 60:
        d5 = min(deadline - 30, time.time() + 120)
        hb5 = (best_h > 0.5).astype(float)
        hs5, cv5 = binary_search(hb5, N0, d5, max_sweeps=3000, sample_n=200)
        if cv5 < best_c5:
            best_c5, best_h = cv5, hs5.copy()
    # Phase 6: Upscale to 2x
    N_up = N0 * 2
    if time.time() < deadline - 30:
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_up = (h_interp > 0.5).astype(float)
        n1 = int(h_up.sum())
        tgt = N_up // 2
        if n1 > tgt:
            oz = np.where(h_up == 1)[0]
            h_up[oz[np.random.choice(len(oz), n1 - tgt, replace=False)]] = 0
        elif n1 < tgt:
            zz = np.where(h_up == 0)[0]
            h_up[zz[np.random.choice(len(zz), tgt - n1, replace=False)]] = 1
        d6 = min(deadline - 5, time.time() + 100)
        hs6, cv6 = binary_search(h_up, N_up, d6, max_sweeps=1000, sample_n=200)
        if cv6 < best_c5:
            best_c5, best_h = cv6, hs6.copy()
    final_c5 = c5_of(best_h)
    return best_h, final_c5, len(best_h)
