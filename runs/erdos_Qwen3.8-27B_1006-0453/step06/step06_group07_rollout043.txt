def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return np.max(fft_corr(h)) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(30):

        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0, 1)

def ensure_binary(h, N):
    target = N // 2
    h_bin = (h >= 0.5).astype(float)

    if n1 == target:
        return h_bin
    if n1 > target:
