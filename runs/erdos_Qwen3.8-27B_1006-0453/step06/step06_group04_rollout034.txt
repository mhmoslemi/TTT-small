    h2s = proj(bh + np.random.randn(N0) * 0.035, N0 / 2.0)
    h2, c2 = cont_opt(h2s, d2, T0=25.0, lr0=0.011, mom=0.88)
    if c2 < bc:
        bc, bh = c2, h2.copy()

    # Phase 3: Second continuous from perturbed best (180s)
    d3 = min(dl - 180, max(d2, t0) + 180)
    h3s = proj(bh + np.random.randn(N0) * 0.02, N0 / 2.0)
    h3, c3 = cont_opt(h3s, d3, T0=60.0, lr0=0.007, mom=0.91)
    if c3 < bc:
        bc, bh = c3, h3.copy()

    # Phase 4: Random binary polish (120s)
    d4 = min(dl - 100, max(d3, t0) + 120)
    h4, c4 = bin_random(bh, d4, ntr=25)
    if c4 < bc:
        bc, bh = c4, h4.copy()

    # Phase 5: Informed binary on best (100s)
    d5 = min(dl - 50, max(d4, t0) + 100)
    h5, c5 = bin_informed(bh, d5, ntr=15)
    if c5 < bc:
        bc, bh = c5, h5.copy()

    # Phase 6: Multi-resolution attempts
    for Nt in [570, 760, 950]:
        if time.time() > dl - 35:
            break
        Nt += Nt % 2
        hi = interp(bh, Nt)
        hbt = fix_ones(hi, Nt)
        sd = min(dl - 15, time.time() + 35)
        ht, ct = bin_informed(hbt, sd, ntr=10)
        if ct < bc:
            bc, bh = ct, ht.copy()

    # Phase 7: Final continuous squeeze
    if time.time() < dl - 3:
        hF = proj(bh + np.random.randn(len(bh)) * 0.012, len(bh) / 2.0)
        hF, cF = cont_opt(hF, dl - 2, T0=150.0, lr0=0.004, mom=0.93)
        if cF < bc:
            bc, bh = cF, hF.copy()

    final_c5 = c5_of(bh)
    return bh, final_c5, len(bh)
