def fft_corr(h):
    N = len(h)

            p1 = int(ones[np.random.randint(len(ones))])
            p2 = int(ones[np.random.randint(len(ones))])
            q1 = int(zeros[np.random.randint(len(zeros))])
            q2 = int(zeros[np.random.randint(len(zeros))])
            if p1 == p2 or q1 == q2 or {p1,p2} & {q1,q2}:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                d = move_delta(h, N, p, q, s_arr)
                nc = corr + d
                npk = float(nc.max())
                delta = npk - current_pk
                if delta < 0 or np.random.random() < np.exp(-delta / max(T0, 1e-10)):
                    h[p] = 0; h[q] = 1; corr = nc; current_pk = npk
                    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                    if npk < best_pk: best_pk = npk; best_h = h.copy(); no_imp = 0
                    else: no_imp += 1
                else: no_imp += 1
                T0 *= cool
            else:
                d1 = move_delta(h, N, p1, q1, s_arr)
                h[p1] = 0; h[q1] = 1
                d2 = move_delta(h, N, p2, q2, s_arr)
                h[p1] = 1; h[q1] = 0
                d = d1 + d2
                nc = corr + d
                npk = float(nc.max())
                delta = npk - current_pk
                if delta < 0 or np.random.random() < np.exp(-delta / max(T0, 1e-10)):
                    h[p1] = 0; h[q1] = 1; h[p2] = 0; h[q2] = 1
                    corr = nc; current_pk = npk
                    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                    if npk < best_pk: best_pk = npk; best_h = h.copy(); no_imp = 0
                    else: no_imp += 1
                else: no_imp += 1
                T0 *= cool
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            d = move_delta(h, N, p, q, s_arr)
            nc = corr + d
            npk = float(nc.max())
            delta = npk - current_pk
            if delta < 0 or np.random.random() < np.exp(-delta / max(T0, 1e-10)):
                h[p] = 0; h[q] = 1; corr = nc; current_pk = npk
                ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                if npk < best_pk: best_pk = npk; best_h = h.copy(); no_imp = 0
                else: no_imp += 1
            else: no_imp += 1
            T0 *= cool
        if no_imp > 4000:
            h = best_h.copy(); corr = fft_corr(h); current_pk = best_pk
            ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
            no_imp = 0; T0 = max(T0, 0.02)
    return best_h, best_pk

def cont_opt(h, N, deadline, T_soft=0.015, lr0=0.004):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    center = N - 1
    s_arr = np.arange(-(N - 1), N)
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        w = np.exp((corr - pk) / T_soft)
        ws = w.sum()
        if ws > 0:
            w /= ws
        else:
            w = np.zeros(2*N - 1); w[center] = 1.0
        grad = np.zeros(N)
        for si in range(2*N - 1):
            wi = w[si]
            if wi < 1e-13:
                continue
            s = s_arr[si]
            if s >= 0:
                v1 = np.arange(s, N); v2 = np.arange(0, N - s)
                grad[v1] += wi * (1.0 - h[v1 - s])
                grad[v2] -= wi * h[v2 + s]
            else:
                ss = -s
                v1 = np.arange(0, N - ss); v2 = np.arange(ss, N)
                grad[v1] += wi * (1.0 - h[v1 + ss])
                grad[v2] -= wi * h[v2 - ss]
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        lr = lr0 * (0.9997 ** (t // 50))
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(fft_corr(h0).max()) * (2.0 / N0)

    # Phase 1: Continuous optimization from perturbed start (80s)
    d1 = min(deadline - 280, t0 + 80)
    hc = h0 + np.random.randn(N0) * 0.03
    hc = proj_sum(hc, N0 / 2.0)
    hC, pkC = cont_opt(hc, N0, d1, T_soft=0.015, lr0=0.004)
    c5C = pkC * (2.0 / N0)
    if c5C < best_c5:
        best_c5, best_h = c5C, hC.copy()

    # Phase 2: Binary local search (220s)
    d2 = min(deadline - 200, time.time() + 220)
    hb = fix_bin((best_h > 0.5).astype(float), N0)
    hB, pkB = binary_search(hb, N0, d2, T0=0.07, cool=0.9998)
    c5B = pkB * (2.0 / N0)
    if c5B < best_c5:
        best_c5, best_h = c5B, hB.copy()

    # Phase 3: Continuous polish (50s)
    if time.time() < deadline - 130:
        d3 = min(deadline - 100, time.time() + 50)
        hc2 = best_h.astype(float) + np.random.randn(len(best_h)) * 0.02
        hc2 = proj_sum(hc2, len(best_h) / 2.0)
        hC2, pkC2 = cont_opt(hc2, len(best_h), d3, T_soft=0.008, lr0=0.003)
        c5C2 = pkC2 * (2.0 / len(best_h))
        if c5C2 < best_c5:
            best_c5, best_h = c5C2, hC2.copy()

    # Phase 4: Multi-resolution exploration
    for Nt in [240, 300, 380, 440, 500, 580, 680, 780]:
        if time.time() > deadline - 30:
            break
        if Nt == N0:
            continue
        if Nt % 2 != 0:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_bin((hi > 0.5).astype(float), Nt)
        sub = min(deadline - 10, time.time() + 30)
        hrt, pkr = binary_search(hbt, Nt, sub, T0=0.04, cool=0.9998)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hrt.copy()

    # Phase 5: Final binary polish on best
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hf = fix_bin((best_h > 0.5).astype(float), Nf)
        hfp, pkf = binary_search(hf, Nf, deadline - 2, T0=0.015, cool=0.9999)
        c5f = pkf * (2.0 / Nf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hfp.copy()

    final_c5 = float(fft_corr(best_h).max()) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
