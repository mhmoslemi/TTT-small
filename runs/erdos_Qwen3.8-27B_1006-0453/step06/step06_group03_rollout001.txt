import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, corr, t, N, k=40):
    cmax = corr.max()
    shifted = t * (corr - cmax)
    shifted = np.clip(shifted, -50, 0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(N)
    w /= wsum
    top_idx = np.argsort(w)[-k:]
    grad = np.zeros(N)
    center = N - 1
    for idx in top_idx:
        wi = w[idx]
        if wi < 1e-8:
            continue
        L = int(idx - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[:N-L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:N-L]
        elif -N < L < 0:
            M = -L
            grad[M:] += wi * (1.0 - h[:N-M])
            grad[:N-M] -= wi * h[M:]
    return grad
