def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

        else:
            hi = mid

def make_binary(h, N):

        h = np.zeros(N)
        h[order[:nh]] = 1

    T = T0

            it += 1

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def enforce_sum(h, N):
    nh = N // 2
    n1 = int(h.sum())
    if n1 == nh:
        return h
    order = np.argsort(-h)
    out = np.zeros(N)
    out[order[:nh]] = 1
    return out

def compute_C(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2*N-1)[None, :] - (N-1)
    imk = pos - k_vals
    ipk = pos + k_vals
    A = np.where((imk >= 0) & (imk < N), 1.0 - h[np.clip(imk, 0, N-1)], 0.0)
    B = np.where((ipk >= 0) & (ipk < N), h[np.clip(ipk, 0, N-1)], 0.0)
    return A - B

def il_search_kick(h, N, deadline, sample_sz=80, kick_every=100, kick_sz=3):
    h = enforce_sum(h.astype(float), N).copy()
    best_h = h.copy()
    best_p = float(np.max(fft_corr(h)))
    stagnant = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_p - 1e-12:
            best_p = P
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= kick_every:
            for _ in range(kick_sz):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                a = int(np.random.choice(ones))
                b = int(np.random.choice(zeros))
                h[a], h[b] = 0, 1
            stagnant = 0
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        C = compute_C(h, N)
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        new = corr[None, None, :] + C[os_][None, :, :] - C[zs][:, None, :]
        newP = new.max(axis=2)
        flat_count = np.sum(newP >= P - 1e-12)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-12:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
        elif flat_count < ns*nz*0.3 and np.random.random() < 0.1:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
    return best_h, best_p * (2.0 / N)

def sa_block(h, N, deadline, T0=0.5, cool=0.997, max_block=5):
    h = enforce_sum(h.astype(float), N).copy()
    best_h = h.copy()
    best_p = float(np.max(fft_corr(h)))
    T = T0
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_p - 1e-12:

            best_h = h.copy()
        if np.random.random() < 0.65:

            zeros = np.where(h == 0)[0]
            a = int(np.random.choice(ones))
            b = int(np.random.choice(zeros))
            h[a], h[b] = 0, 1
            Pn = float(np.max(fft_corr(h)))

            i = np.random.randint(0, N-bsz)
            j = np.random.randint(0, N-bsz)

                T *= cool; it += 1; continue
            bi = np.arange(i, i+bsz)
            bj = np.arange(j, j+bsz)
            if np.array_equal(h[bi], h[bj]):
                T *= cool; it += 1; continue

        T *= cool
        it += 1
        if it % 3000 == 0:

def nesterov_cont(h, N, deadline, k_top=120, lr0=0.025):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_01(h, target)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        hp = project_01(h + m, target)
        pkh = float(np.max(fft_corr(hp)))
        if pkh < best_pk:
            best_pk = pkh
            best_h = hp.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= k_top
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            lr *= 0.985
            beta = min(0.94, 0.82 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: IL search with kicks on initial binary (long)
    if time.time() < deadline - 300:
        hb = (h0 > 0.5).astype(float)
        d1 = min(deadline - 290, t0 + 220)
        hs, cs = il_search_kick(hb, N0, d1, sample_sz=90, kick_every=80, kick_sz=4)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 2: Continuous Nesterov
    if time.time() < deadline - 220:
        d2 = min(deadline - 210, time.time() + 170)
        hc = best_h + np.random.randn(N0) * 0.35
        hc = project_01(hc, N0 / 2.0)
        hn, cn = nesterov_cont(hc, N0, d2, k_top=140, lr0=0.022)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 3: SA with block moves (escape local minima)
    if time.time() < deadline - 150:
        h3 = (best_h > 0.5).astype(float)
        d3 = min(deadline - 140, time.time() + 160)
        hs3, cs3 = sa_block(h3, N0, d3, T0=0.7, cool=0.9965, max_block=6)
        if cs3 < best_c5:
            best_c5, best_h = cs3, hs3

    # Phase 4: IL search on best binary (fine-tune)
    if time.time() < deadline - 100:
        h4 = (best_h > 0.5).astype(float)
        d4 = min(deadline - 90, time.time() + 150)
        hs4, cs4 = il_search_kick(h4, N0, d4, sample_sz=100, kick_every=60, kick_sz=3)
        if cs4 < best_c5:
            best_c5, best_h = cs4, hs4

    # Phase 5: Multi-resolution
    for N_try in [360, 370, 390, 400, 420, 440]:
        if time.time() > deadline - 50:
            break
        if N_try == N0:
            continue
        xb = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xb, best_h)
        hb = enforce_sum((hi > 0.5).astype(float), N_try)
        sub_dl = min(deadline - 15, time.time() + 45)
        hr, cr = il_search_kick(hb, N_try, sub_dl, sample_sz=70, kick_every=100, kick_sz=3)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        h6 = (best_h > 0.5).astype(float)
        d6 = deadline - 1
        hs6, cs6 = il_search_kick(h6, len(best_h), d6, sample_sz=100, kick_every=50, kick_sz=2)
        if cs6 < best_c5:
            best_c5, best_h = cs6, hs6

    h_final = enforce_sum((best_h > 0.5).astype(float), len(best_h))
    final_c5 = compute_c5(h_final, len(h_final))
    return h_final, final_c5, len(h_final)
