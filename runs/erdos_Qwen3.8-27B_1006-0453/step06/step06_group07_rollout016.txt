def nesterov_softmax(h, N, deadline, k_top=80, temp_init=2.0, temp_decay=0.998, lr0=0.02):
    """Nesterov with soft-max gradient."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    temp = temp_init
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        # Soft-max gradient: use top-k with softmax weighting
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        top_corr = corr[top_idx]
        w = np.exp((top_corr - top_corr.max()) / temp)
        w /= w.sum()
        
        grad = np.zeros(N)
        center = N - 1
        for i, ai in enumerate(top_idx):
            L = int(ai - center)
            wi = w[i]
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            temp *= temp_decay
            lr *= 0.995
            beta = min(0.97, 0.85 + it * 0.00005)
    
    return best_h, best_pk * dx

    # For binary h[i]=0: D[i,k] = +(1-h[i+k-center]) - h[i-k+center]
    # Combined: D[i,k] = (1-2*h[i]) * ((1-h[i+k-center])*valid1 - h[i-k+center]*valid2)

    ip2 = pos - ks  # i - k (for the h side)
    
    v1 = (ip1 >= 0) & (ip1 < N)
    v2 = (ip2 >= 0) & (ip2 < N)
    
    c1 = np.clip(ip1, 0, N-1)
    c2 = np.clip(ip2, 0, N-1)

    term1 = np.where(v1, 1.0 - h[c1], 0.0)
    # h[ip2] where valid, else 0
    term2 = np.where(v2, h[c2], 0.0)
    
    f = term1 - term2
    D = (1.0 - 2.0 * h[:, None]) * f

            mi = int(np.argmin(npk))
            if float(npk[mi]) < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        t_frac = max(0.0, 1.0 - sweep / (max_sweeps * 0.7))
        temp = sa_temp * t_frac * t_frac
        if best_pq is not None and best_new_P < P - 1e-14:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        elif best_pq is not None and temp > 1e-10:
            delta = max(0.0, best_new_P - P)
            if rng.random() < math.exp(-delta / temp):
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
        else:
            if stall > 30:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
    return best_h, best_c * (2.0 / N)

def nesterov_sm(h, N, deadline, k_top=100, temp_init=3.0, temp_decay=0.997, lr0=0.02, beta0=0.9, noise=0.04):
    h = h.astype(float).copy() + np.random.randn(len(h)) * noise
    h = project_sum(h, N / 2.0)
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    temp = temp_init
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        tc = corr[top]
        w = np.exp((tc - tc.max()) / temp)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for i, ai in enumerate(top):
            L = int(ai - center)
            wi = w[i]
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            temp = max(0.1, temp * temp_decay)
            lr *= 0.995
            beta = min(0.97, beta0 + it * 0.00003)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Thorough binary SA at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 450, t0 + 150)
    h1, c5_1 = binary_sa(h_bin, N0, d1, max_sweeps=10000, sample_frac=1.0, sa_temp=0.05, seed_rng=42)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Second binary SA with different seed
    d2 = min(deadline - 350, time.time() + 80)
    h2, c5_2 = binary_sa(ensure_sum(best_h, N0), N0, d2, max_sweeps=6000, sample_frac=1.0, sa_temp=0.04, seed_rng=777)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Continuous Nesterov (softmax) at N0
    if time.time() < deadline - 280:
        d3 = min(deadline - 230, time.time() + 70)
        h3, c5_3 = nesterov_sm(best_h, N0, d3, k_top=80, temp_init=2.5, lr0=0.02, noise=0.05)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary SA after continuous
    if time.time() < deadline - 180:
        d4 = min(deadline - 130, time.time() + 60)
        h4, c5_4 = binary_sa(ensure_sum(best_h, N0), N0, d4, max_sweeps=8000, sample_frac=1.0, sa_temp=0.03, seed_rng=314)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Nesterov with different params
    if time.time() < deadline - 130:
        d5 = min(deadline - 100, time.time() + 40)
        h5, c5_5 = nesterov_sm(best_h, N0, d5, k_top=120, temp_init=1.5, lr0=0.01, noise=0.02)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Upscale to 2*N0
    N1 = 2 * N0
    if time.time() < deadline - 100:
        h_up = ensure_sum(upsample_fn(best_h, N1), N1)
        d6 = min(deadline - 60, time.time() + 70)
        h6, c5_6 = binary_sa(h_up, N1, d6, max_sweeps=2500, sample_frac=0.55, sa_temp=0.04, seed_rng=99)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Upscale to 4*N0
    N2 = 4 * N0
    if N2 <= 1600 and time.time() < deadline - 60:
        h_up2 = ensure_sum(upsample_fn(best_h, N2), N2)
        d7 = min(deadline - 30, time.time() + 40)
        h7, c5_7 = binary_sa(h_up2, N2, d7, max_sweeps=1000, sample_frac=0.3, sa_temp=0.03, seed_rng=55)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final binary polish at best resolution
    N_fin = len(best_h)
    sf = 1.0 if N_fin <= 400 else (0.6 if N_fin <= 800 else 0.35)
    hF = ensure_sum(best_h, N_fin)
    hF2, c5F = binary_sa(hF, N_fin, deadline - 0.5, max_sweeps=5000, sample_frac=sf, sa_temp=0.02, seed_rng=271)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
