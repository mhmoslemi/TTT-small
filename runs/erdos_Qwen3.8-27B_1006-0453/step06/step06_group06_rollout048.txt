import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h, N):
    target = N // 2
    h = h.copy()
    n1 = int(np.sum(h > 0.5))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        rem = np.random.choice(idx, n1 - target, replace=False)
        h[rem] = 0
    elif n1 < target:
        idx = np.where(h <= 0.5)[0]
        add = np.random.choice(idx, target - n1, replace=False)
        h[add] = 1
    return h.astype(float)

def do_swap(h, corr, N, p, q, s_arr):
    ps = p + s_arr
    pm = p - s_arr
    qs = q + s_arr
    qm = q - s_arr
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2*N - 1)
    d[vps] -= (1.0 - h[ps[vps]])
    d[vpm] += h[pm[vpm]]
    d[vqs] += (1.0 - h[qs[vqs]])
    d[vqm] -= h[qm[vqm]]
    nc = corr + d
    return nc, d

def anneal_binary(h, N, deadline, T0=0.1, cool=0.9996, max_no_imp=2000, use_blocks=True):
    h = fix_count(h.astype(float), N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    T = T0
    no_imp = 0
    step = 0
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    while time.time() < deadline and T > 0.003:
        step += 1
        do_block = use_blocks and (step % 4000) < 800 and len(ones) >= 2 and len(zeros) >= 2
        if not do_block:
            do_block = (step % 2000) < 500 and len(ones) >= 2 and len(zeros) >= 2
        
        if do_block:
            p1 = int(ones[np.random.randint(len(ones))])
            p2 = int(ones[np.random.randint(len(ones))])
            if p1 == p2:
                p2 = int(ones[np.random.randint(len(ones))])
            if p1 == p2:
                continue
            q1 = int(zeros[np.random.randint(len(zeros))])
            q2 = int(zeros[np.random.randint(len(zeros))])
            if q1 == q2:
                q2 = int(zeros[np.random.randint(len(zeros))])
            if q1 == q2:
                continue
            d_total = np.zeros(2*N - 1)
            d_total += do_swap(h, corr, N, p1, q1, s_arr)[1]
            d_total += do_swap(h, corr, N, p2, q2, s_arr)[1]
            nc = corr + d_total
            npk = float(nc.max())
            delta = npk - cur_pk
            if delta < -1e-12 or (delta < 0.01 and np.random.random() < np.exp(-delta / max(T, 1e-8))):
                h[p1] = 0; h[q1] = 1; h[p2] = 0; h[q2] = 1
                corr = nc; cur_pk = npk
                if npk < best_pk - 1e-12:
                    best_pk = npk; best_h = h.copy(); no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            nc, d = do_swap(h, corr, N, p, q, s_arr)
            npk = float(nc.max())
            delta = npk - cur_pk
            if delta < -1e-12 or np.random.random() < np.exp(-max(delta, 0) / max(T, 1e-8)):

    return best_h, best_pk

def adam_opt(h, N, deadline, top_k=40, lr0=0.004):
    h = proj_sum(h.astype(float).copy(), N / 2.0)
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    center = N - 1
    lr = lr0
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk; best_h = h.copy()
        
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += (1.0 - h[s:])
                grad[s:] -= h[:N-s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N-s2])
                grad[:N-s2] -= h[s2:]
        grad /= top_k
        
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 400 == 0:
            lr *= 0.94
    
    return best_h, best_pk
