        zs = np.random.choice(zeros, nz, replace=False)
        C = build_C(h, N)
        newP = (corr[None, None, :] + C[os_][None, :, :] - C[zs][:, None, :]).max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        delta = mv - P
        if delta < -1e-12 or (delta < 5.0 and np.random.random() < np.exp(-delta / max(T, 1e-8))):
            h[os_[mi[1]]] = 0.0
            h[zs[mi[0]]] = 1.0
        T *= cool
        if T < T_min and P >= best_c:
            break
    return best_h, best_c * (2.0 / N)

def nesterov_weighted(h, N, deadline, k=100, lr0=0.012, Tg=3.0):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top_idx = np.argpartition(corr, -k)[-k:]
        pk = float(np.mean(corr[top_idx]))
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        w = np.exp(corr[top_idx] / Tg)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for j, ai in enumerate(top_idx):
            L = int(ai - center)
            wt = w[j]
            if L == 0:
                grad += wt * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wt * (1.0 - y[:N - L])
                grad[:N - L] -= wt * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wt * (1.0 - y[M:])
                    grad[M:] -= wt * y[:N - M]
        h_old = h.copy()
        h = project_01(y - lr * grad, target)
        m = beta * (h - h_old)
        it += 1
        if it % 250 == 0:
            lr *= 0.996
            beta = min(0.97, 0.86 + it * 0.0002)
            Tg *= 0.97
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Aggressive annealed IL from initial (300s)
    d1 = min(deadline - 450, t0 + 300)
    h_bin = (h0 > 0.5).astype(float)
    h1, c1 = annealed_il(h_bin, N0, d1, max_iters=8000, s_sz=75, T0=5.0, T_min=0.05, cool=0.9996, patience=500)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Finer annealed IL (180s)
    d2 = min(deadline - 250, time.time() + 180)
    h2, c2 = annealed_il(best_h, N0, d2, max_iters=6000, s_sz=90, T0=1.5, T_min=0.02, cool=0.9998, patience=300)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 3: Try alternate N values (150s)
    for N_try in [N0 + 4, N0 + 8, N0 + 12]:
        if time.time() > deadline - 200:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1 - nh, replace=False)] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh - n1, replace=False)] = 1
        sub_dl = min(deadline - 180, time.time() + 70)
        h_rt, c_rt = annealed_il(h_bt, N_try, sub_dl, max_iters=3000, s_sz=65, T0=2.0, T_min=0.03, cool=0.9997, patience=350)
        if c_rt < best_c5:
            best_c5, best_h = c_rt, h_rt.copy()

    # Phase 4: Continuous Nesterov with weighted gradient (120s)
    d4 = min(deadline - 80, time.time() + 120)
    hc = best_h + np.random.randn(len(best_h)) * 0.05
    hc = project_01(hc, len(best_h) / 2.0)
    hn, cn = nesterov_weighted(hc, len(best_h), d4, k=100, lr0=0.01, Tg=4.0)
    if cn < best_c5:
        best_c5, best_h = cn, hn.copy()

    # Phase 5: Final binary polish (80s)
    d5 = min(deadline - 10, time.time() + 80)
    hF = (best_h > 0.5).astype(float)
    hF, cF = annealed_il(hF, len(best_h), d5, max_iters=4000, s_sz=85, T0=0.5, T_min=0.01, cool=0.9999, patience=250)
    if cF < best_c5:
        best_c5, best_h = cF, hF.copy()

    # Phase 6: Final micro-polish
    if time.time() < deadline - 1:
        hB = (best_h > 0.5).astype(float)
        hB, cB = annealed_il(hB, len(best_h), deadline - 0.5, max_iters=500, s_sz=90, T0=0.1, T_min=0.005, cool=0.9999, patience=100)
        if cB < best_c5:
            best_c5, best_h = cB, hB.copy()

    # Final: ensure binary and correct count
    best_h = (best_h > 0.5).astype(float)
    nh = len(best_h) // 2
    n1 = int(best_h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(best_h == 1)[0]
            best_h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(best_h == 0)[0]
            best_h[np.random.choice(idx, nh - n1, replace=False)] = 1
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
