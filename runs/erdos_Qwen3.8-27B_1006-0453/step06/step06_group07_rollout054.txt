import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def greedy_improve(h, N, deadline, max_iters=500):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Compute effect of each swap
        # For swap (p, q): corr changes by -contrib_p + contrib_q at each lag
        # Use the D matrix approach from previous code
        D = compute_D(h, N)
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def simulated_annealing(h, N, deadline, T0=3.0, cool_rate=0.9995, max_iters=100000, kick_interval=3000):

    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    current_c = best_c
    T = T0
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Adaptive cooling
        if it % 100 == 0:
            T *= cool_rate
        
        # Pick random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        
        # Try swap
        h[p], h[q] = 0.0, 1.0
        new_c = float(np.max(fft_corr(h)))
        delta = new_c - current_c
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current_c = new_c
            if new_c < best_c - 1e-13:
                best_c = new_c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[p], h[q] = 1.0, 0.0  # revert
            stall += 1
        
        # Kick if stalled
        if stall > kick_interval:
            n_kick = 3 + np.random.randint(3)
            for _ in range(n_kick):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                pk = ones[np.random.randint(len(ones))]
                qk = zeros[np.random.randint(len(zeros))]
                h[pk], h[qk] = 0.0, 1.0
            current_c = float(np.max(fft_corr(h)))
            T = max(T, T0 * 0.5)
            stall = 0
    
    return best_h, best_c * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def project_simplex(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())

        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:

        else:
            hi = mid

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix rounding to get exact sum
    diff = target - r.sum()
    if abs(diff) > 1e-10:
        r[r.argmax()] += diff
        r = np.clip(r, 0.0, 1.0)
    return r

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

    h = h_init.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr_cur = lr
    center = N - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
        
        # Softmax weights (numerically stable)
        c_shifted = beta * (c - c.max())
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2*N-1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum
        
        # Only use top significant weights for gradient
        w_idx = np.argsort(w)[::-1]
        n_sig = min(15, int(np.sum(w > 0.01 / (2*N-1))))
        n_sig = max(n_sig, 3)
        
        grad = np.zeros(N)
        for idx in w_idx[:n_sig]:
            L = int(idx) - center
            wv = float(w[idx])
            if wv < 1e-6:

            elif 0 < L < N:
                grad[L:] += wv * (1.0 - h[:N-L])

                M = -L
                grad[:N-M] += wv * (1.0 - h[M:])
                grad[M:] += (-wv * h[:N-M])
        
        m = rho * m + grad
        h = project_sum(h - lr_cur * m, target)
