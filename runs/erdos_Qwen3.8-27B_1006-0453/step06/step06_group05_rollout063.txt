- First factor: h[n+0] = h[n], when n=p: h[p] changes 1→0, term h[p]*(1-h[p]) changes from 1*0=0 to 0*1=0. No change!
- Second factor: (1-h[n]), when n=p: (1-h[p]) changes 0→1, term h[p]*(1-h[p]) changes from 1*0=0 to 0*1=0. No change!

So for shift=0, the delta is always 0 for binary h. The formula h[p+0] - (1-h[p-0]) = h[p] - (1-h[p]) = 1 - 0 = 1 is wrong because it double-counts.

When shift=0, p appears in BOTH the first and second factor simultaneously. The first-factor change is -(1-h[p]) = 0, and the second-factor change is h[p] = 1. But the actual change is 0 because the term h[p]*(1-h[p]) is 0 both before and after.

So I need to handle the self-interaction: when p+shift = p (i.e., shift=0), the two effects cancel because the term is h[p]*(1-h[p]) which is 0 for binary.

More generally, if p+shift = q (for the second factor) or p-shift = q (for the first factor), there are cross-interactions.

This is getting complicated. For the swap case, if p-shift = q or p+shift = q, we need to handle the interaction.

For simplicity, let me just compute the correlation from scratch after each swap. With FFT, it's O(N log N) which is fast enough for N=380.

- The new max is at most max(P, max over affected indices of old_corr + delta)

The delta is non-zero only at a limited set of indices (those where p or q appears). For a specific p:

- p-shift is valid: shift ∈ [p-N+1, N-1-p], same range

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def round_to_binary(h, N):
    """Round continuous h to binary with exactly N/2 ones."""
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    target = N // 2
    if n1 == target:
        return h_b
    if n1 > target:
        ones = np.where(h_b == 1)[0]
        # Remove the ones with lowest h values
        to_remove = ones[np.argsort(h[ones])[:n1 - target]]
        h_b[to_remove] = 0
    else:
        zeros = np.where(h_b == 0)[0]
        to_add = zeros[np.argsort(-h[zeros])[:target - n1]]
        h_b[to_add] = 1
    return h_b

        #   Second factor: n = p, gain h[p+shift] [if valid]
        #   Self: if shift=0, both apply to same term, net effect = 0 for binary

def make_binary(h, N):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    t = N // 2
    if n1 > t:
        oi = np.where(hb==1)[0]
        hb[oi[np.argsort(h[oi])[:n1-t]]] = 0
    elif n1 < t:
        zi = np.where(hb==0)[0]
        hb[zi[np.argsort(-h[zi])[:t-n1]]] = 1

def local_search(h, N, deadline, max_iter=2000, rng=None):

    if rng is None:
        rng = np.random.RandomState()
    h = h.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    shifts = np.arange(2*N-1) - (N-1)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        
        # Compute E matrix: E[p,k] = effect of h[p]: 1->0 on corr[k]
        pos = np.arange(N)[:, None]
        sv = shifts[None, :]
        pm = pos + sv
        pp = pos - sv
        vm = (pm>=0) & (pm<N)
        vp = (pp>=0) & (pp<N)
        pm_c = np.clip(pm, 0, N-1)

        base = corr_t[None,:] + E1  # (n1, top_n)

    corr = fft_corr(h)

        ones = np.where(h==1)[0]

        h[q] = 1  # ensure

        else:

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def make_binary(h, N):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    t = N // 2
    if n1 > t:
        oi = np.where(hb==1)[0]
        hb[oi[np.argsort(h[oi])[:n1-t]]] = 0
    elif n1 < t:
        zi = np.where(hb==0)[0]
        hb[zi[np.argsort(-h[zi])[:t-n1]]] = 1
    return hb

def compute_E(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv
    pp = pos - sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    E = np.where(vm, h[np.clip(pm, 0, N-1)], 0.0) - np.where(vp, 1.0-h[np.clip(pp, 0, N-1)], 0.0)
    E[:, shifts == 0] = 0
    return E, shifts

            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if dE < 0 or np.log(rng.random() + 1e-15) > -dE / (T + 1e-15):
            corr = nc
        else:
            h[p], h[q] = 1, 0
        it += 1
    return h, best_c, best_h

def multi_anneal(h, N, deadline, rng, n_restarts=3):
    bh = h.copy()
    bc = compute_c5(bh)
    per_dl = max(10, (deadline - time.time()) / n_restarts - 5)
    for r in range(n_restarts):
        cur = bh.copy()
        nf = rng.randint(5, 20)
        oi, zi = np.where(cur == 1)[0], np.where(cur == 0)[0]
        for _ in range(nf):
            cur[oi[rng.randint(len(oi))]], cur[zi[rng.randint(len(zi))]] = 0, 1
            oi, zi = np.where(cur == 1)[0], np.where(cur == 0)[0]
        sd = min(deadline - 5, time.time() + per_dl)
        _, c, bh2 = anneal_search(cur, N, sd, rng, t0=5.0, t1=0.05)
        if c < bc:
            bc, bh = c, bh2
        _, c, bh2 = greedy_search(bh, N, min(deadline - 2, time.time() + 30), max_iter=800, rng=rng)
        if c < bc:
            bc, bh = c, bh2
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    hb = make_binary(h0, N0)
    best_h = hb.copy()
    best_c5 = compute_c5(best_h)

    d1 = min(deadline - 500, t0 + 280)
    _, c1, bh1 = greedy_search(hb, N0, d1, max_iter=6000, rng=rng, top_n=12)
    if c1 < best_c5:
        best_c5, best_h = c1, bh1

    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 220)
        _, c2, bh2 = anneal_search(best_h, N0, d2, rng, t0=5.0, t1=0.08)
        if c2 < best_c5:
            best_c5, best_h = c2, bh2

    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 180)
        _, c3, bh3 = greedy_search(best_h, N0, d3, max_iter=4000, rng=rng, top_n=14)
        if c3 < best_c5:
            best_c5, best_h = c3, bh3

    for att in range(10):
        if time.time() > deadline - 120:
            break
        hp = best_h.copy()
        nf = rng.randint(10, 30)
        oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
        for _ in range(nf):
            hp[oi[rng.randint(len(oi))]], hp[zi[rng.randint(len(zi))]] = 0, 1
            oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
        sd = min(deadline - 100, time.time() + 50)
        _, c4, bh4 = greedy_search(hp, N0, sd, max_iter=800, rng=rng, top_n=10)
        if c4 < best_c5:
            best_c5, best_h = c4, bh4

    if time.time() < deadline - 80:
        d5 = min(deadline - 60, time.time() + 150)
        bh5, c5 = multi_anneal(best_h, N0, d5, rng, n_restarts=3)
        if c5 < best_c5:
            best_c5, best_h = c5, bh5

    for Nt in [382, 378, 384, 386, 390, 376]:
        if time.time() > deadline - 35:
            break
        xb = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, xb, best_h)
        hbt = make_binary(hi, Nt)
        sd = min(deadline - 20, time.time() + 45)
        _, c6, bh6 = greedy_search(hbt, Nt, sd, max_iter=1500, rng=rng, top_n=10)
        if c6 < best_c5:
            best_c5, best_h = c6, bh6

    if time.time() < deadline - 8:
        d7 = min(deadline - 3, time.time() + 50)
        _, c7, bh7 = greedy_search(best_h, len(best_h), d7, max_iter=3000, rng=rng, top_n=16)
        if c7 < best_c5:
            best_c5, best_h = c7, bh7
    if time.time() < deadline - 2:
        _, c8, bh8 = anneal_search(best_h, len(best_h), deadline - 1, rng, t0=1.5, t1=0.02)
        if c8 < best_c5:
            best_c5, best_h = c8, bh8

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
