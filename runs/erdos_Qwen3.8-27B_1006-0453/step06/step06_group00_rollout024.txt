Let me also think about the continuous optimization. For the continuous case:

The objective is C₅(h) = max_m (Σ_i h[i]*(1-h[i+k]) * dx) where k = m-(N-1).

Using smooth max: f(h) = (1/β) log(Σ_m exp(β * corr_m(h)))

The gradient: ∂f/∂h[i] = (1/β) * Σ_m w_m * ∂corr_m/∂h[i] where w_m = exp(β*corr_m)/Σ exp(β*corr_m')

And ∂corr_m/∂h[i]:
- As first position at lag k=m-(N-1): if i+k valid, ∂corr/∂h[i] = (1-h[i+k])
- As second position at lag k: if i-k valid, ∂corr/∂h[i] = -h[i-k]

So ∂corr_m/∂h[i] = (1-h[i+k]) * 1[valid] - h[i-k] * 1[valid']

This can be computed efficiently for the weighted sum.

Actually, for the smooth max gradient, we need:
∂f/∂h[i] = Σ_m w_m * [(1-h[i+k_m]) * 1[i+k_m valid] - h[i-k_m] * 1[i-k_m valid]]

where k_m = m - (N-1).

This is a weighted sum of shifted versions of (1-h) and h. Can be computed with convolutions!

Let me define:
- A[i] = Σ_m w_m * (1-h[i+k_m]) * 1[0≤i+k_m<N]
- B[i] = Σ_m w_m * h[i-k_m] * 1[0≤i-k_m<N]

Then ∂f/∂h[i] = A[i] - B[i]

A[i] = Σ_{m: 0≤i+k_m<N} w_m * (1-h[i+k_m])

def smooth_max_gradient(h, beta, weights=None):
    N = len(h)
    corr = get_corr(h)
    # Softmax weights
    c = corr - corr.max()
    exp_c = np.exp(beta * c)
    w = exp_c / exp_c.sum()
    
    # Gradient: grad[i] = sum_m w[m] * dcorr_m/dh[i]
    # dcorr_m/dh[i] = (1-h[i+k])*1[valid] - h[i-k]*1[valid] where k = m-(N-1)
    
    grad = np.zeros(N)
    k_vals = np.arange(2*N-1) - (N-1)  # -(N-1) to N-1
    
    # For each lag k with weight w_k:
    # dcorr/dh[i] includes (1-h[i+k]) if 0<=i+k<N, and -h[i-k] if 0<=i-k<N
    # So: grad[i] += w_k * (1-h[i+k]) for valid i+k
    #     grad[i] -= w_k * h[i-k] for valid i-k
    
    # Vectorized: for each k, the contribution to grad is:
    # (1-h) shifted by -k for the first term, -h shifted by +k for the second
    
    # Actually let's just do it in a loop over k (only 759 iterations)
    for k, wk in zip(k_vals, w):
        if wk < 1e-15:
            continue
        if k >= 0:
            # i+k valid for i in [0, N-1-k]
            grad[:N-k] += wk * (1.0 - h[k:N])
            # i-k valid for i in [k, N-1]
            grad[k:] -= wk * h[0:N-k]
        else:
            M = -k  # M > 0
            # i+k = i-M valid for i in [M, N-1]
            grad[M:] += wk * (1.0 - h[0:N-M])
            # i-k = i+M valid for i in [0, N-1-M]
            grad[:N-M] -= wk * h[M:]
    
    return grad

        for ai in top_a:
            base = corr + chg_a[ai]
            maxes = (base[None, :] + chg_b[top_b]).max(axis=1)
            bi = int(np.argmin(maxes))
            if maxes[bi] < best_val - 1e-12:
                best_val = float(maxes[bi])
                best_pair = (int(sa[ai]), int(sz[top_b[bi]]))
        if best_pair is not None:
            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
        else:
            if stagnant > 30:
                break
            ra = int(np.random.choice(sa))
            rb = int(np.random.choice(sz))
            h[ra] = 0
            h[rb] = 1
    return best_h, best_c * (2.0 / N)

def nesterov_smooth(h, N, deadline, beta_init=25, beta_inc=0.002, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    m = np.zeros(N)
    beta = float(beta_init)
    lr = lr0
    bmom = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        grad = smooth_max_grad(y, beta)
        h_new = project_01(y - lr * grad, target)
        m = bmom * (h_new - h)
        h = h_new
        it += 1
        if it % 40 == 0:
            cv = compute_c5(h)
            if cv < best_val:
                best_val = cv
                best_h = h.copy()
            beta += beta_inc
            lr *= 0.9985
            bmom = min(0.97, 0.87 + it * 0.0003)
            if it > 800:
                break
    cv = compute_c5(h)
    if cv < best_val:
        best_val = cv
        best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    # Phase 1: Binary optimization
    d1 = min(deadline - 320, t0 + 180)
    h1 = (h0 > 0.5).astype(float)
    h1, c1 = binary_opt(h1, N0, d1, max_crit=5, top_k=25, sample_sz=90)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    # Phase 2: Continuous smooth-max Nesterov
    if time.time() < deadline - 260:
        d2 = min(deadline - 240, time.time() + 150)
        hc = best_h + np.random.randn(N0) * 0.07
        hc = project_01(hc, N0 / 2.0)
        h2, c2 = nesterov_smooth(hc, N0, d2, beta_init=30, beta_inc=0.003, lr0=0.010)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    # Phase 3: Deeper binary search
    if time.time() < deadline - 160:
        d3 = min(deadline - 150, time.time() + 200)
        h3 = (best_h > 0.5).astype(float)
        h3, c3 = binary_opt(h3, N0, d3, max_crit=8, top_k=30, sample_sz=100)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
    # Phase 4: Multi-resolution
    for N_try in [400, 440, 500]:
        if time.time() > deadline - 70:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh_t = N_try // 2
        n1_t = int(h_bt.sum())
        if n1_t > nh_t:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1_t - nh_t, replace=False)] = 0
        elif n1_t < nh_t:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh_t - n1_t, replace=False)] = 1
        sub_dl = min(deadline - 30, time.time() + 70)
        h_try, c_try = binary_opt(h_bt, N_try, sub_dl, max_crit=5, top_k=20, sample_sz=80)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try
    # Phase 5: Final Nesterov polish
    if time.time() < deadline - 8:
        hF = best_h + np.random.randn(len(best_h)) * 0.025
        hF = project_01(hF, len(best_h) / 2.0)
        hF, cF = nesterov_smooth(hF, len(best_h), deadline - 2, beta_init=50, beta_inc=0.005, lr0=0.007)
        if cF < best_c5:
            best_c5, best_h = cF, hF
    # Final binary cleanup
    if time.time() < deadline - 1:
        hB = (best_h > 0.5).astype(float)
        hB, cB = binary_opt(hB, len(best_h), deadline - 0.5, max_crit=6, top_k=30, sample_sz=100)
        if cB < best_c5:
            best_c5, best_h = cB, hB
    hB2 = (best_h > 0.5).astype(float)
    cB2 = compute_c5(hB2)
    if cB2 < best_c5:
        best_h, best_c5 = hB2, cB2
    return best_h, best_c5, len(best_h)
