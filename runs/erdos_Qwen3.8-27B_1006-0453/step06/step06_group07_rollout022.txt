import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum_binary(h, N):

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def nesterov_topk(h, N, deadline, k=50, lr0=0.02, beta=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.00005)
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum_binary(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def focused_binary_search(h, N, deadline, top_M=40, perturb_every=40, sa_prob=0.0):
    h = h.astype(float).copy()
    dx = 2.0 / N
    rng = np.random.RandomState(7)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argpartition(corr, -top_M)[-top_M:]
        offsets = top_idx.astype(int) - (N - 1)
        p_list = ones
        q_list = zeros
        pm = p_list[:, None] - offsets[None, :]
        pp = p_list[:, None] + offsets[None, :]
        qm = q_list[:, None] - offsets[None, :]
        qp = q_list[:, None] + offsets[None, :]
        vm_p = (pm >= 0) & (pm < N)
        vp_p = (pp >= 0) & (pp < N)
        vm_q = (qm >= 0) & (qm < N)
        vp_q = (qp >= 0) & (qp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        qm_c = np.clip(qm, 0, N - 1)
        qp_c = np.clip(qp, 0, N - 1)
        A = -(1.0 - h[pm_c]) * vm_p + h[pp_c] * vp_p
        B = (1.0 - h[qm_c]) * vm_q - h[qp_c] * vp_q
        best_new = np.inf
        best_pi, best_mj = -1, -1
        for pi in range(len(p_list)):
            Cp = corr[top_idx] + A[pi, :]
            costs = np.max(Cp[None, :] + B, axis=1)
            mq = int(np.argmin(costs))
            if costs[mq] < best_new:
                best_new = float(costs[mq])
                best_pi, best_mj = pi, mq
        if best_pi < 0:
            break
        if sa_prob > 0 and rng.random() < sa_prob:
            best_pi = rng.randint(len(p_list))
            best_mj = rng.randint(len(q_list))
        p_idx = int(p_list[best_pi])
        q_idx = int(q_list[best_mj])
        h[p_idx] = 0.0
        h[q_idx] = 1.0
        corr = fft_corr(h)
        cur_max = float(corr.max())
        if cur_max < best_c - 1e-12:
            best_c = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > perturb_every and perturb_every > 0:
            ones_p = np.where(h == 1)[0]
            zeros_p = np.where(h == 0)[0]
            for _ in range(2 + stall // perturb_every):
                p = int(ones_p[rng.randint(len(ones_p))])
                q = int(zeros_p[rng.randint(len(zeros_p))])
                h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
            stall = 0
    return best_h, best_c * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def nesterov_topk(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.00005)
    return best_h, best_pk * dx
