```python
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = np.round(h).astype(float)
    h = np.clip(h, 0.0, 1.0)
    n1 = int(h.sum())
    if n1 > nh:
        ones = np.where(h == 1)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h == 0)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    return h

def greedy_best_swap(h, N, deadline, max_iters=50000, n_trial=10):
    """Greedy search: at each step, try n_trial random swaps, take the best."""
    h = fix_binary(h, N // 2).copy()

    best_c = float(corr.max()) * (2.0 / N)
    best_h = h.copy()
    
    while time.time() < deadline:

        zeros = np.where(h == 0)[0]

        best_p, best_q = -1, -1
        
        for _ in range(n_trial):
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p], h[q] = 0.0, 1.0
            new_corr = fft_corr(h)
            c = float(new_corr.max()) * (2.0 / N)
            if c < best_trial_c:

                best_p, best_q = p, q
            h[p], h[q] = 1.0, 0.0

        h[best_p], h[best_q] = 0.0, 1.0
        best_c = best_trial_c
        best_h = h.copy()

        zeros = np.where(h == 0)[0]
```


This sets deltas[i] to s[i] repeated in positions i to i+N-1. But the actual delta of the correlation when flipping position i should be related to (1-h) at shifted positions, not s at shifted positions. This could be causing the previous code to make suboptimal swaps.

Let me also add a "geometric" local search: instead of just swapping one 1 with one 0, try a "rotation" - shift a block of the binary string by 1 position. This is a different kind of move that might escape local optima.

Actually, for a binary string with exactly N/2 ones, a "rotation" of a block means:
- Find a contiguous block [a, b] where h[a]=1, h[b+1]=0 (or vice versa)
- Shift the boundary: h[a] = 0, h[b+1] = 1 (if h[a] was 1 and h[b+1] was 0)

This is actually just a swap of two adjacent boundary positions. Not very different from a regular swap.
