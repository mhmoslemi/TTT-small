import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D_subset(h, N, idx):
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(np.float64) + hp * vp.astype(np.float64)

        h[h <= 0.5] = 0.0

        eps = 1.0
        active = np.where(c >= pk - eps)[0]
        if len(active) > 8:
            active = np.argpartition(c, -8)[-8:]
        
        D_act = compute_D_subset(h, N, active)
        grad = -D_act.mean(axis=1)
        
        # Normalize
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        
        h_new = h + lr * grad
        h_new = project_sum(h_new, target)
        
        c_new = float(np.max(fft_corr(h_new)))
        if c_new < pk:
            h = h_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.6
            if lr < 0.0005:
                break
        
        # Periodic binarization nudge
        if t % 300 == 0:
            hb = binarize(h_new, N)
            h = 0.5 * h + 0.5 * hb
            h = project_sum(h, target)
    
    return best_h, best_c

def anneal_binary(h, N, deadline, T0=2.0, T1=0.05, max_iter=30000):
    h = binarize(h, N)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    n = max_iter
    T = T0
    
    for itr in range(n):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (itr / n)
        
        c = fft_corr(h)
        P = float(c.max())
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        p = int(np.random.randint(len(ones)))
        q = int(np.random.randint(len(zeros)))
        ip, iq = int(ones[p]), int(zeros[q])
        
        # Compute delta
        h[ip], h[iq] = 0.0, 1.0
        c_new = fft_corr(h)
        P_new = float(c_new.max())
        delta = P_new - P
        h[ip], h[iq] = 1.0, 0.0
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h[ip] = 0.0
            h[iq] = 1.0
            if P_new < best_c - 1e-12:
                best_c = P_new
                best_h = h.copy()
    
    return best_h, best_c
