def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft(np.flip(1.0 - h.astype(float)), L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)
