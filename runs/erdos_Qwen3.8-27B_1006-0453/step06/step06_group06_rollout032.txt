def continuous_refine(h, N, deadline, top_k=40, lr0=0.003):
    """Adam-based continuous optimization."""
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    center = N - 1
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Focus gradient on top-k correlation values
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s_val = int(ai - center)
            if s_val == 0:
                grad += (1.0 - 2.0 * h)
            elif s_val > 0:
                grad[:N-s_val] += (1.0 - h[s_val:])
                grad[s_val:] -= h[:N-s_val]
            else:
                s2 = -s_val
                grad[s2:] += (1.0 - h[:N-s2])
                grad[:N-s2] -= h[s2:]
        grad /= top_k
        
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        lr = lr0 * (1.0 / (1.0 + t * 0.001))
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
    
    return best_h, best_pk

def resample(h, N_new):
    """Resample h to N_new points."""
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    hi = np.interp(x_new, x_old, h)
    return hi

                    continue

                h[p] = 0; h[q] = 1
                corr = nc; current_pk = npk
                ones = np.where(h==1)[0]
                zeros = np.where(h==0)[0]
                tabu.append(p)
                if npk < best_pk:
                    best_pk = npk; best_h = h.copy(); no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
            T *= cool
            if no_imp > 4000:
                break
        # Perturb
        if r < n_restart - 1:
            h = best_h.copy()
            corr = fft_corr(h)
            current_pk = best_pk
            nf = max(5, int(N * perturb_frac * (0.5 + r*0.3)))
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            idx1 = np.random.choice(len(ones), nf, replace=False)
            idx0 = np.random.choice(len(zeros), nf, replace=False)
            h[ones[idx1]] = 0
            h[zeros[idx0]] = 1
            corr = fft_corr(h)
            current_pk = float(corr.max())
    
    return best_h, best_pk
