import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, N):
    """Project to binary with exactly N/2 ones"""
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        # Flip the "least confident" ones (closest to 0.5 in original)
        order = np.argsort(h[idx])  # most likely to flip
        h[idx[order[:n1-nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.argsort(-h[idx])  # most likely to flip to 1
        h[idx[order[:nh-n1]]] = 1
    return h

def local_search(h, N, deadline, max_sweeps=2000, top_k=15, patience=500):
    """Binary local search with smart move selection"""
    h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    no_improve = 0
    T = 0.01  # temperature for accepting worse moves
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            break
        
        # Find top peaks
        center = N - 1
        top_idx = np.argsort(c)[-top_k:]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Greedy: find best single swap
        best_delta = P  # We want to reduce below P
        best_pq = None
        
        # Only check a subset for speed
        n_check = min(len(ones), 60)
        n_check_z = min(len(zeros), 60)
        
        # Prioritize positions near the peaks
        # Positions p where h[p]=1 contribute to peaks
        # We want to move 1s away from peak-contributing positions
        scored_ones = []
        scored_zeros = []
        for p in ones[:n_check]:
            score = 0.0
            for ki in top_idx:
                m = int(ki - center)
                if 0 <= p-m < N and h[p-m] == 0:
                    score += 1
                if 0 <= p+m < N and h[p+m] == 1:
                    score += 1
            scored_ones.append((score, p))
        scored_ones.sort(reverse=True)
        
        for q in zeros[:n_check_z]:
            score = 0.0
            for ki in top_idx:
                m = int(ki - center)
                if 0 <= q-m < N and h[q-m] == 1:
                    score += 1
            scored_zeros.append((score, q))
        scored_zeros.sort(reverse=True)
        
        p_candidates = [p for _, p in scored_ones[:30]]
        q_candidates = [q for _, q in scored_zeros[:30]]
        
        for p in p_candidates:
            for q in q_candidates:
                if p == q:
                    continue
                # Evaluate swap
                h[p] = 0
                h[q] = 1
                new_c = fft_corr(h)
                new_P = float(new_c.max())
                h[p] = 1
                h[q] = 0
                
                if new_P < best_delta:
                    best_delta = new_P
                    best_pq = (p, q)
        
        if best_pq is not None:
            h[best_pq[0]] = 0
            h[best_pq[1]] = 1
        else:
            # Try random move (annealing)
            if np.random.random() < T:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p] = 0
                h[q] = 1
                new_P = float(np.max(fft_corr(h)))
                if new_P > P + 5:  # Too bad, revert
                    h[p] = 1
                    h[q] = 0
            T *= 0.995
    
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    """Convert to binary with exactly N/2 ones"""
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        order = np.argsort(h[idx])
        h[idx[order[:n1-nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.argsort(-h[idx])
        h[idx[order[:nh-n1]]] = 1
    return h

                    best_pq = (int(p), int(q))
                    c = new_c  # Keep for next iteration if accepted
        
        if best_pq is not None:
            h[best_pq[0]] = 0
            h[best_pq[1]] = 1
            c = fft_corr(h)
    
    return best_h, best_c

def crossover(h1, h2, N):
    """Crossover two binary solutions"""
    nh = N // 2
    mask = np.random.random(N) > 0.5
    child = np.where(mask, h1, h2).astype(float)
    n1 = int(child.sum())
    if n1 > nh:
        idx = np.where(child == 1)[0]
        idx = idx[np.random.choice(len(idx), n1-nh, replace=False)]
        child[idx] = 0
    elif n1 < nh:
        idx = np.where(child == 0)[0]
        idx = idx[np.random.choice(len(idx), nh-n1, replace=False)]
        child[idx] = 1
    return child

        idx_zeros = np.where(pop[i] == 0)[0]

            q = idx_zeros[np.random.randint(len(idx_zeros))]
            pop[i][p], pop[i][q] = 0, 1
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    for gen in range(gen_max):
        if time.time() > deadline:
            break

        for i in range(pop_size):
            if scores[i] < best_c5:
                best_c5 = scores[i]
                best_h = pop[i].copy()
        
        # Select top half, breed new half
        order = np.argsort(scores)
        parents = [pop[order[i]] for i in range(pop_size // 2)]
        
        new_pop = []
        for i in range(pop_size // 2):
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            child = crossover(p1, p2, N)
            # Brief local search
            d_sub = min(deadline, time.time() + 15)
            child, _ = local_search_binary(child, N, d_sub, top_k=top_k, patience=50, max_sweeps=200)
            new_pop.append(child)
        
        pop = [pop[order[i]] for i in range(pop_size // 2)] + new_pop
    
    return best_h, best_c5

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        order = np.argsort(h[idx])
        h[idx[order[:n1 - nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.argsort(-h[idx])
        h[idx[order[:nh - n1]]] = 1
    return h

def score_positions(h, c, top_idx, N, top_k):
    center = N - 1
    s1 = np.zeros(N)
    s0 = np.zeros(N)
    for ki in top_idx:
        m = int(ki - center)
        if m > 0:
            v = np.arange(0, N - m)
            s1[v[(h[v] == 1) & (h[v + m] == 0)]] += 1
            v2 = np.arange(m, N)
            s0[v2[(h[v2 - m] == 1) & (h[v2] == 0)]] += 1
        elif m < 0:
            mm = -m
            v = np.arange(mm, N)
            s1[v[(h[v] == 1) & (h[v - mm] == 0)]] += 1
            v2 = np.arange(0, N - mm)
            s0[v2[(h[v2 + mm] == 1) & (h[v2] == 0)]] += 1
    return s1, s0

def local_search_binary(h, N, deadline, top_k=12, patience=300, max_sweeps=5000, cand_n=25):

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            break

        top_idx = np.argsort(c)[-top_k:]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        s1, s0 = score_positions(h, c, top_idx, N, top_k)

        tc1 = np.where(s1 > 0)[0]
        tc0 = np.where(s0 > 0)[0]
        if len(tc1) > cand_n:
            tc1 = tc1[np.argsort(-s1[tc1])[:cand_n]]
        else:
            tc1 = np.append(tc1, ones[:cand_n - len(tc1)])
        if len(tc0) > cand_n:
            tc0 = tc0[np.argsort(-s0[tc0])[:cand_n]]
        else:
            tc0 = np.append(tc0, zeros[:cand_n - len(tc0)])

        best_new_P = P
        best_pq = None
        for p in tc1:
            for q in tc0:
                if p == q:
                    continue
                h[p] = 0
                h[q] = 1

                h[p] = 1
                h[q] = 0
                if nP < best_new_P:
