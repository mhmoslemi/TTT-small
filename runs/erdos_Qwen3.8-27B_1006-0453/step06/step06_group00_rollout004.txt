import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient_step(grad, h, lr):
    """One projected subgradient step for the max-correlation problem."""
    N = len(h)
    # grad is the subgradient of max_k R(k)
    h_new = h - lr * grad
    h_new = project_01(h_new, N / 2.0)
    return h_new

def compute_subgrad(h, corr, top_k):
    """Compute subgradient of max_k R(k) using top-k lags."""
    N = len(h)
    center = N - 1
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    # Weight by proximity to max
    vals = corr[top_idx]
    max_val = vals.max()
    weights = np.exp(50 * (vals - max_val))  # soft weighting
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for w, ai in zip(weights, top_idx):
        L = int(ai - center)
        g = np.zeros(N)
        if L == 0:
            continue  # R(0) = 0 for valid h, gradient is tricky
        elif L > 0:
            # ∂R(k)/∂h[j] = (1-h[j+k]) for j < N-k, plus (-h[j-k]) for j >= k
            g[:N-L] += (1.0 - h[L:])
            g[L:] += (-h[:N-L])
        else:
            M = -L
            # For negative lag: R(-M) = sum_i h[i]*(1-h[i-M])
            # ∂/∂h[j] = (1-h[j-M]) for j>=M, plus (-h[j+M]) for j < N-M
            g[M:] += (1.0 - h[:N-M])
            g[:N-M] += (-h[M:])
        grad += w * g
    return grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def compute_subgrad(h, corr, top_k):
    N = len(h)
    center = N - 1
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[top_idx]
    mval = vals.max()
    weights = np.exp(40.0 * (vals - mval))
    s = weights.sum()
    if s < 1e-10:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights /= s
    grad = np.zeros(N)
    for w, ai in zip(weights, top_idx):
        L = int(ai - center)
        if L == 0:
            continue
        elif L > 0:
            grad[:N - L] += w * (1.0 - h[L:])
            grad[L:] += w * (-h[:N - L])
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:N - M])
            grad[:N - M] += w * (-h[M:])
    return grad

def continuous_opt(h_init, N, deadline, k_top=30):
    h = h_init.astype(float).copy()
    h = project_01(h, N / 2.0)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    C_lb = 0.0
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = compute_subgrad(h, corr, k_top)
        gnorm = np.sqrt(np.sum(grad * grad))
        if gnorm < 1e-12:
            break
        step = (pk - C_lb + 1e-6) / (gnorm * gnorm + 1e-12)
        step = min(step, 1.5)
        h_new = project_01(h - step * grad, N / 2.0)
        corr_new = fft_corr(h_new)
        pk_new = float(np.max(corr_new))
        if pk_new <= pk * 1.002:
            h = h_new
        else:
            h = project_01(h - step * 0.1 * grad, N / 2.0)
        it += 1
        if it % 200 == 0:
            C_lb = max(C_lb, pk - 0.02)
    return best_h, best_pk

        h[rng.choice(idx, nh - n1, replace=False)] = 1
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    curr_c = best_c
    temp = t_init
    ones = np.where(h == 1)[0]

    for it in range(max_iters):
