import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    # (1-h) reversed for correlation
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def swap_impact_all_peaks(h, p, q, peak_lags, N):
    """Compute change in overlap at given lags for swap p(1->0), q(0->1)"""
    center = N - 1
    changes = np.zeros(len(peak_lags))
    for idx, lag in enumerate(peak_lags):
        # Delta = h[p+lag] - h[q+lag] + h[p-lag] - h[q-lag]
        ip = p + lag
        iq = q + lag
        ip2 = p - lag
        iq2 = q - lag
        v = 0.0
        if 0 <= ip < N: v += h[ip]
        if 0 <= iq < N: v -= h[iq]
        if 0 <= ip2 < N: v += h[ip2]
        if 0 <= iq2 < N: v -= h[iq2]
        changes[idx] = v
    return changes

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0 - h), L)))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[idx[rng.permutation(len(idx))[:n1 - nh]]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[rng.permutation(len(idx))[:nh - n1]]] = 1
    return h

def eval_swap_vals(h, p, q, peak_idx, N):

    c = lambda x: h[np.clip(x, 0, N - 1)]
    m = lambda x: (x >= 0) & (x < N)
    v = np.zeros(len(lags))
    ip = p + lags; iq = q + lags; ip2 = p - lags; iq2 = q - lags
    v[c(ip)[m(ip)]] = 1  # placeholder, do it properly below

         + np.where(m(ip2), c(ip2), 0) - np.where(m(iq2), c(iq2), 0))
    return v

def continuous_polish(h, N, deadline, lr0=0.012, k=50):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        # Project to sum
        s = h_new.sum()
        if abs(s - target) > 1e-8:
            lo, hi = -0.5, 0.5
            for _ in range(40):
                mid = (lo + hi) * 0.5
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo + hi) * 0.5, 0, 1)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.993
            beta = min(0.96, beta + 0.0002)
    return best_h, best_pk
