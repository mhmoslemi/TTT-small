import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def interp_to(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def continuous_opt(h0, N, deadline, k=40, lr0=0.02, temp_start=0.008, temp_end=0.0001, beta0=0.85):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    lr = lr0
    beta = beta0
    stall = 0
    it = 0
    center = N - 1
    t_start = time.time()
    dur = max(deadline - t_start, 1.0)
    while time.time() < deadline:
        prog = min(1.0, (time.time() - t_start) / dur)
        temp = temp_start * (temp_end / temp_start) ** (prog * 1.5)
        temp = max(temp, temp_end)
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        k_eff = min(k + stall // 25, N - 1)
        top = np.argpartition(corr, -k_eff)[-k_eff:]
        cm = corr[top].max()
        w = np.exp((corr[top] - cm) / max(temp, 1e-10))
        w /= w.sum()
        grad = np.zeros(N)
        for wi, ai in zip(w, top):
            L = int(round(ai - center))
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif 0 < L < N:
                seg = N - L
                grad[L:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:]
            elif -N < L < 0:
                M = -L
                seg = N - M
                grad[:seg] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:seg]
        h_new = project_sum(y - lr * grad, target)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr = max(0.0008, lr * 0.994)
            beta = min(0.94, beta0 + it * 0.00004)
        if stall > 150:
            scale = 0.015 + 0.01 * (stall // 300)
            h = best_h + np.random.randn(N) * scale
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = max(lr, lr0 * 0.6)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_opt(h0, N, deadline, n_tries=12, kick_freq=50):

                    h[pp] = 0
                for qq in zeros[i2]:
                    h[qq] = 1

    return best_h, best_int * (2.0 / N)
