import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0-h, L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def compute_dp(h, N):
    dp = np.zeros((N, 2*N-1))
    for i in range(N):
        n1 = N - i
        dp[i, :n1] += h[i:]
        n2 = i + 1
        dp[i, :n2] -= (1.0 - h[i::-1])
    return dp

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

        idx = np.where(h_bin==0)[0]
        selecting = np.random.choice(len(idx), target-n1, replace=False)
