def nesterov_continuous(h, N, deadline, k0=50, k_min=5, lr0=0.02):
    h = project_sum(h.copy(), N/2.0)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    dx = 2.0/N
    m = np.zeros(N)  # momentum
    beta = 0.9
    lr = lr0
    k = k0
    it = 0
    stall = 0
    
    while time.time() < deadline:
        it += 1
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_c - 1e-12:
            best_c = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptively reduce k
        k = max(k_min, int(k0 * max(0.5, 1.0 - it/500.0)))
        
        # Get top-k lags
        top_idx = np.argpartition(c, -k)[-k:]
        
        # Compute exact gradient
        center = N - 1
        grad = np.zeros(N)
        for ai in top_idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0*y)
            elif s > 0:
                grad[:N-s] += (1.0 - y[s:])
                grad[s:] -= y[:N-s]
            else:
                t = -s
                if t < N:
                    grad[t:] += (1.0 - y[:N-t])
                    grad[:N-t] -= y[t:]
        grad /= k
        
        # Nesterov update
        h_new = project_sum(y - lr * grad, N/2.0)
        m = beta * (h_new - h)
        h = h_new
        
        # Adapt parameters
        if it % 200 == 0:
            lr *= 0.995
            beta = min(0.97, 0.85 + it * 0.00005)
        if stall > 100:
            # Perturb
            h = project_sum(h + np.random.randn(N) * 0.02, N/2.0)
            m = np.zeros(N)
            stall = 0
    
    return best_h, best_c * dx

For shift s (positive or negative), c[s] involves h[i] and h[i-s].
∂c[s]/∂h[j] = 1 - h[j-s] * I(j-s valid) - h[j+s] * I(j+s valid)

For s = 0: ∂c[0]/∂h[j] = 1 - 2*h[j]

For s > 0:
- h[j-s] valid when j >= s: contributes h[j-s]
- h[j+s] valid when j < N-s: contributes h[j+s]
- So: grad[j] += 1 - (h[j-s] if j>=s else 0) - (h[j+s] if j<N-s else 0)

- h[j-s] = h[j+t], valid when j+t < N, i.e., j < N-t: contributes h[j+t]
- h[j+s] = h[j-t], valid when j-t >= 0, i.e., j >= t: contributes h[j-t]

        delta[v1] += h[ps_p[v1]]
        v2 = (pm_p >= 0) & (pm_p < N)
        delta[v2] += h[pm_p[v2]]
        v3 = (ps_q >= 0) & (ps_q < N)
        delta[v3] -= h[ps_q[v3]]
        v4 = (pm_q >= 0) & (pm_q < N)
        delta[v4] -= h[pm_q[v4]]
        c_new = c + delta
        new_max = float(c_new.max())
        cur_max = float(c.max())
        delta_e = new_max - cur_max
        if delta_e < 0 or np.random.random() < np.exp(-delta_e / t):
            c = c_new
            h[p] = 0.0
            h[q] = 1.0
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            stall += 1
        if stall >= kick_every and len(ones) > 4 and len(zeros) > 4:
            for _ in range(3):
                pp = int(ones[np.random.randint(len(ones))])
                qq = int(zeros[np.random.randint(len(zeros))])
                if pp != q:
                    h[pp], h[qq] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = fft_corr(h)
            stall = 0
        if it % 5000 == 0:
            t = max(t * cool ** 5000, t_min)
    return best_h, best_c * (2.0 / N)

def nesterov_cont(h, N, deadline, k0=40, k_min=5, lr0=0.015):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = 0.88
    lr = lr0
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_c - 1e-13:
            best_c = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k = max(k_min, int(k0 * (1.0 - 0.4 * it / max(1, it + 100))))
        top_idx = np.argpartition(c, -k)[-k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top_idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * y)
            elif s > 0 and s < N:
                grad += 1.0
                grad[s:] -= y[:N - s]
                grad[:N - s] -= y[s:]
            else:
                tval = -s
                if tval < N:
                    grad += 1.0
                    grad[:N - tval] -= y[tval:]
                    grad[tval:] -= y[:N - tval]
        grad /= k
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        if it % 300 == 0:
            lr *= 0.996
            beta = min(0.96, 0.84 + it * 0.00003)
        if stall > 150:
            h = project_sum(h + np.random.randn(N) * 0.015, N / 2.0)
            m = np.zeros(N)
            stall = 0
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N)

    # Phase 1: Greedy from initial
    h_g, c_g = greedy_binary(ensure_sum(h0, N), N, min(deadline - 700, t0 + 25), max_sweeps=400)
    if c_g < best_c5:
        best_c5, best_h = c_g, h_g.copy()

    # Phase 2: SA warm start (from best greedy)
    d2 = min(deadline - 500, t0 + 120)
    h_sa1, c_sa1 = sim_anneal(ensure_sum(best_h, N), N, d2, t_start=2.0, t_min=0.05, max_iter=350000)
    if c_sa1 < best_c5:
        best_c5, best_h = c_sa1, h_sa1.copy()

    # Phase 3: SA cold start (random binary for diversity)
    rng = np.random.RandomState(seed + 7)
    h_rand = np.zeros(N)
    perm = rng.permutation(N)
    h_rand[perm[:N // 2]] = 1.0
    d3 = min(deadline - 380, time.time() + 120)
    h_sa2, c_sa2 = sim_anneal(h_rand, N, d3, t_start=4.0, t_min=0.1, max_iter=350000, kick_every=6000)
    if c_sa2 < best_c5:
        best_c5, best_h = c_sa2, h_sa2.copy()

    # Phase 4: SA from initial with second temp schedule
    d4 = min(deadline - 280, time.time() + 120)
    h_sa3, c_sa3 = sim_anneal(ensure_sum(h0, N), N, d4, t_start=1.5, t_min=0.03, max_iter=350000)
    if c_sa3 < best_c5:
        best_c5, best_h = c_sa3, h_sa3.copy()

    # Phase 5: SA refinement on current best
    d5 = min(deadline - 200, time.time() + 100)
    h_sa4, c_sa4 = sim_anneal(ensure_sum(best_h, N), N, d5, t_start=0.8, t_min=0.02, max_iter=300000, kick_every=10000)
    if c_sa4 < best_c5:
        best_c5, best_h = c_sa4, h_sa4.copy()

    # Phase 6: Continuous Nesterov
    if time.time() < deadline - 120:
        hc = project_sum(ensure_sum(best_h, N) + np.random.randn(N) * 0.02, N / 2.0)
        d6 = min(deadline - 100, time.time() + 70)
        h6, c6 = nesterov_cont(hc, N, d6, k0=50, k_min=5, lr0=0.012)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: SA on continuous result (binarized)
    if time.time() < deadline - 60:
        h_bin7 = ensure_sum(best_h, N)
        d7 = min(deadline - 40, time.time() + 45)
        h7, c7 = sim_anneal(h_bin7, N, d7, t_start=0.5, t_min=0.01, max_iter=200000)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Phase 8: Final greedy polish
    if time.time() < deadline - 5:
        hF, cF = greedy_binary(ensure_sum(best_h, N), N, deadline - 1, max_sweeps=300, sample_frac=1.0)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    final_c5 = compute_c5(best_h, N)
    return best_h, final_c5, N
