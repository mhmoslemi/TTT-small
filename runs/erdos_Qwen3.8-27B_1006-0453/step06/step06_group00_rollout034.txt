For the best possible binary sequence, we want to minimize the maximum overlap between h and 1-h at any shift.

Let me think about the structure. If h has N/2 ones, then at shift 0, the overlap is count of positions where h=1 and 1-h=1, which is 0 (since they're disjoint). Wait no, h and 1-h are complementary, so at shift 0, h(x) * (1-h(x)) = 0 for all x. So the max shift probably isn't 0.

Actually wait, np.correlate(h, 1-h, mode="full") - let me think about what this computes. For arrays a and b, correlate(a, b)[k] = sum_i a[i] * b[i+k-shift] for appropriate shifts.

Actually, np.correlate(a, v, mode='full')[k] = sum_n a[n+k] * conj(v[n]) where the sum is over valid n.

So if a = h and v = 1-h, then correlate(h, 1-h, mode='full')[k] = sum_n h[n+k] * (1-h[n])

This is the overlap between h shifted by k and (1-h) unshifted. For k=0, this is sum h[n](1-h[n]) = 0 for binary h. For other k, this counts positions where h has a 1 at position n+k AND (1-h) has a 1 at position n, i.e., h[n] = 0.

So C₅/h = max_k sum_n h[n+k](1-h[n]) * dx

This is the maximum number of "1s in h that land on 0s of h" at any circular-ish shift.

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_c5_direct(h, N):
    # Use np.correlate directly for verification
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    """Project h to have sum exactly target, keeping values in [0,1] approximately."""
    h = h.copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    # Adjust by spreading the difference
    n = len(h)
    h += diff / n
    return np.clip(h, 0, 1)

def tabu_search(h, N, deadline, max_iters=2000, tabu_size=50):
    """Tabu search with single swaps."""
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    
    best_c = np.inf
    best_h = h.copy()
    tabu = {}  # (i,j) -> iteration when it becomes untabu
    it = 0
    ns_sample = 40
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        it += 1
        
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        ns = min(len(ones), ns_sample)
        nz = min(len(zeros), ns_sample)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        
        L = 2 * N - 1
        co = np.zeros((ns, L))
        for idx_i, i in enumerate(os_):
            ms = max(0, N - 1 - i)
            me = min(L - 1, 2 * N - 2 - i)
            if ms <= me:
                idx_m = np.arange(ms, me + 1)
                co[idx_i, idx_m] = 1.0 - h[i + idx_m - (N - 1)]
        
        cz = np.zeros((nz, L))
        for idx_j, j in enumerate(zs):
            ms = max(0, N - 1 - j)
            me = min(L - 1, 2 * N - 2 - j)
            if ms <= me:
                idx_m = np.arange(ms, me + 1)
                cz[idx_j, idx_m] += 1.0 - h[j + idx_m - (N - 1)]
            ms2 = max(0, j)
            me2 = min(L - 1, j + N - 1)
            if ms2 <= me2:
                idx_m2 = np.arange(ms2, me2 + 1)
                cz[idx_j, idx_m2] -= h[j + (N - 1) - idx_m2]
        
        new_corr = corr[None, None, :] - co[None, :, :] + cz[:, None, :]
        newP = new_corr.max(axis=2)
        
        # Find best non-tabu move
        best_val = P  # don't move if no improvement
        best_move = None
        for a in range(nz):
            for b in range(ns):
                if (os_[b], zs[a]) in tabu and tabu[(os_[b], zs[a])] > it:
                    continue
                if newP[a, b] < best_val:
                    best_val = float(newP[a, b])
                    best_move = (a, b)
        
        if best_move is not None:
            a, b = best_move
            h[os_[b]] = 0
            h[zs[a]] = 1
            tabu[(os_[b], zs[a])] = it + tabu_size
        
        # Occasionally allow a worst move to escape
        if it % 100 == 0 and P > best_c:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p], h[q] = 0.0, 1.0
            tabu[(p, q)] = it + tabu_size
    
    return best_h, best_c

def anneal_search(h, N, deadline, max_iters=1500, T0=0.3, T1=0.01):
    """Simulated annealing with swaps."""
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    
    best_c = np.inf
    best_h = h.copy()
    curr_c = float(np.max(fft_corr(h)))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        T = T0 * (T1 / T0) ** (it / max_iters)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        p = int(np.random.choice(ones))
        q = int(np.random.choice(zeros))
        
        # Compute delta
        corr = fft_corr(h)
        P = float(np.max(corr))
        
        h[p], h[q] = 0.0, 1.0
        new_corr = fft_corr(h)
        newP = float(np.max(new_corr))
        
        delta = newP - P
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            if newP < curr_c:
                curr_c = newP
            if newP < best_c:
                best_c = newP
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
    
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def build_f_vectors(positions, h, N):
    """Build f(pos) vectors: f[pos][m] = 2*h[pos+m-(N-1)]-1 for |m-(N-1)-pos| <= pos, else 0."""
    c = N - 1
    L = 2 * N - 1
    ns = len(positions)
    F = np.zeros((ns, L))
    for idx, pos in enumerate(positions):
        lo = c - pos
        hi = c + pos
        F[idx, lo:hi+1] = 2.0 * h[:2*pos+1] - 1.0
    return F
