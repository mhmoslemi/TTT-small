    if time.time() < deadline - 250:
        d3 = min(deadline - 150, time.time() + 130)
        h3 = make_binary(best_h, N0)
        h3b, _ = bin_search(h3, N0, d3, max_iters=600, anneal=0.0, rng=rng)
        c3 = compute_c5(h3b)
        if c3 < best_c5:
            best_c5, best_h = c3, h3b.copy()
    
    # Phase 4: Annealed binary search
    if time.time() < deadline - 130:
        d4 = min(deadline - 80, time.time() + 60)
        h4 = make_binary(best_h, N0)
        h4b, _ = bin_search(h4, N0, d4, max_iters=300, anneal=0.15, rng=rng)
        c4 = compute_c5(h4b)
        if c4 < best_c5:
            best_c5, best_h = c4, h4b.copy()
    
    # Phase 5: Multi-resolution exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [400, 420, 390, 440, 380]:
        if time.time() > deadline - 40:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == len(best_h):
            N_try += 2
            if N_try % 2 != 0:
                N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = make_binary(h_interp, N_try)
        sub_dl = min(deadline - 15, time.time() + 40)
        h_rt, _ = bin_search(h_bt, N_try, sub_dl, max_iters=250, anneal=0.0, rng=rng)
        c5_rt = compute_c5(h_rt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)
    
    # Phase 6: Perturbation restarts
    if time.time() < deadline - 30:
        d6 = deadline - 5
        h_b6 = make_binary(best_h, len(best_h))
        for attempt in range(6):
            if time.time() > d6:
                break
            hp = h_b6.copy()
            nf = rng.randint(2, 6)
            oi = np.where(hp == 1)[0].copy()
            zi = np.where(hp == 0)[0].copy()
            for _ in range(nf):
                if len(oi) == 0 or len(zi) == 0:
                    break
                p = int(rng.randint(len(oi)))
                q = int(rng.randint(len(zi)))
                hp[oi[p]], hp[zi[q]] = 0, 1
                oi = np.where(hp == 1)[0]
                zi = np.where(hp == 0)[0]
            sub_dl = min(d6, time.time() + 12)
            h_pt, _ = bin_search(hp, len(hp), sub_dl, max_iters=100, anneal=0.0, rng=rng)
            c5_pt = compute_c5(h_pt)
            if c5_pt < best_c5:
                best_c5, best_h = c5_pt, h_pt.copy()
    
    # Phase 7: Final binary polish
    if time.time() < deadline - 2:
        hF = make_binary(best_h, len(best_h))
        hF, _ = bin_search(hF, len(best_h), deadline - 0.5, max_iters=80, anneal=0.0, rng=rng)
        cF = compute_c5(hF)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
