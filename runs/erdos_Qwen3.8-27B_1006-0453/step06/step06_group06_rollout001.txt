        T *= cool
        if no_imp > max_no_imp:
            no_imp = 0
            r = np.random.random()
            if r < 0.4:
                n_p = max(2, N//40)
                pi = np.random.choice(len(ones), n_p, replace=False)
                qi = np.random.choice(len(zeros), n_p, replace=False)
                h[ones[pi]]=0; h[zeros[qi]]=1
                corr=fft_corr(h); current_pk=float(corr.max())
                ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
            else:
                h=best_h.copy(); corr=fft_corr(h); current_pk=best_pk
                ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
                T=max(T, T0*0.4)
    return best_h, best_pk

def directed_improve(h, N, corr, max_iters=200):
    center = N-1
    ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
    for _ in range(max_iters):
        if len(ones)==0 or len(zeros)==0: break
        pk = float(corr.max())
        peak_idx = int(np.argmax(corr))
        shift = peak_idx - center
        # Find best swap targeting the peak
        ns_p = min(len(ones), 50); ns_q = min(len(zeros), 50)
        p_cands = ones[np.random.choice(len(ones), ns_p, replace=False)]
        q_cands = zeros[np.random.choice(len(zeros), ns_q, replace=False)]
        best_np, best_p, best_q = pk+1, -1, -1
        best_d_arr = None
        for p in p_cands:
            ps, pm = p+np.arange(-(N-1),N), p-np.arange(-(N-1),N)
            vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
            pc=np.clip(ps,0,N-1)[vps]; pmc=np.clip(pm,0,N-1)[vpm]
            dp = np.zeros(2*N-1)
            dp[vps] -= (1.0-h[pc]); dp[vpm] += h[pmc]
            for q in q_cands:
                qs, qm = q+np.arange(-(N-1),N), q-np.arange(-(N-1),N)
                vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
                qsc=np.clip(qs,0,N-1)[vqs]; qmc=np.clip(qm,0,N-1)[vqm]
                dq = np.zeros(2*N-1)
                dq[vqs] += (1.0-h[qsc]); dq[vqm] -= h[qmc]
                nc = corr + dp + dq
                npk = float(nc.max())
                if npk < best_np:
                    best_np = npk; best_p = p; best_q = q; best_d_arr = dp+dq
        if best_p >= 0 and best_np < pk - 0.5:
            h[best_p]=0; h[best_q]=1
            corr += best_d_arr
            ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
        else:
            break
    return h, float(corr.max())

def continuous_opt(h, N, deadline, top_k=40, lr0=0.004):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1,b2,eps = 0.82,0.999,1e-8
    t=0; best_pk=float(np.max(fft_corr(h))); best_h=h.copy()
    center=N-1; lr=lr0
    while time.time() < deadline:
        corr=fft_corr(h)
        pk=float(corr.max())
        if pk<best_pk: best_pk=pk; best_h=h.copy()
        idx=np.argpartition(corr,-top_k)[-top_k:]
        grad=np.zeros(N)
        for ai in idx:
            s=int(ai-center)
            if s==0: grad += (1.0-2.0*h)
            elif s>0:
                grad[:N-s]+=(1.0-h[s:]); grad[s:]-=h[:N-s]
            else:
                s2=-s; grad[s2:] +=(1.0-h[:N-s2]); grad[:N-s2] -= h[s2:]
        grad/=top_k
        t+=1
        m=b1*m+(1-b1)*grad; v=b2*v+(1-b2)*grad*grad
        mh=m/(1-b1**t); vh=v/(1-b2**t)
        h=proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        if t%250==0: lr*=0.95
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0=time.time(); deadline=t0+budget_s-3
    np.random.seed(seed)
    h0=np.array(initial_h_values, dtype=float)
    N0=len(h0)
    best_h=h0.copy(); best_c5=float(np.max(fft_corr(h0)))*(2.0/N0)

    # Phase 1: Continuous warm-up (short)
    d1=min(deadline-600, t0+80)
    hc=h0+np.random.randn(N0)*0.06
    hc=proj_sum(hc, N0/2.0)
    hC,pkC=continuous_opt(hc,N0,d1,top_k=45,lr0=0.005)
    c5C=pkC*(2.0/N0)
    if c5C<best_c5: best_c5,best_h=c5C,hC.copy()

    # Phase 2: Main binary peaking search (long)
    d2=min(deadline-350, time.time()+280)
    hb=fix_count((best_h>0.5).astype(float), N0)
    hB,pkB=peaking_search(hb,N0,d2,T0=0.065,cool=0.99925,max_no_imp=900,n_tries=3)
    c5B=pkB*(2.0/N0)
    if c5B<best_c5: best_c5,best_h=c5B,hB.copy()

    # Phase 3: Directed improvement on current best
    if time.time() < deadline-250:
        hcur=(best_h>0.5).astype(float); fix_count(hcur, len(hcur))
        Nc=len(hcur)
        corr=fft_corr(hcur)
        hcur2,pk2=directed_improve(hcur.copy(), Nc, corr.copy(), max_iters=300)
        c5d=pk2*(2.0/Nc)
        if c5d<best_c5: best_c5,best_h=c5d,hcur2.copy()
        # Continue peaking from directed-improved
        d3=min(deadline-250, time.time()+60)
        hB2,pkB2=peaking_search(hcur2,Nc,d3,T0=0.03,cool=0.9994,max_no_imp=400,n_tries=2)
        c5B2=pkB2*(2.0/Nc)
        if c5B2<best_c5: best_c5,best_h=c5B2,hB2.copy()

    # Phase 4: Multi-N exploration
    N_targets = [360, 400, 440, 480, 520, 560, 640, 720, 800]
    elapsed4 = time.time() - t0
    time_left4 = deadline - elapsed4 - 40
    if time_left4 > 60:
        per_N = max(40, int(time_left4 / len(N_targets)))
        for Nt in N_targets:
            if time.time() > deadline - 30: break
            if Nt == N0: continue
            x_old=np.linspace(0,2,len(best_h),endpoint=False)
            x_new=np.linspace(0,2,Nt,endpoint=False)
            hi=np.interp(x_new,x_old,best_h)
            hbt=fix_count((hi>0.5).astype(float),Nt)
            sub=min(deadline-10, time.time()+per_N)
            hrt,pkr=peaking_search(hbt,Nt,sub,T0=0.04,cool=0.99945,max_no_imp=450,n_tries=2)
            c5r=pkr*(2.0/Nt)
            if c5r<best_c5: best_c5,best_h=c5r,hrt.copy()

    # Phase 5: Final polish on best solution
    if time.time() < deadline-2:
        Nf=len(best_h)
        hf=fix_count((best_h>0.5).astype(float),Nf)
        d5=deadline-1
        hfp,pkf=peaking_search(hf,Nf,d5,T0=0.012,cool=0.9996,max_no_imp=300,n_tries=2)
        c5f=pkf*(2.0/Nf)
        if c5f<best_c5: best_c5,best_h=c5f,hfp.copy()

    final_c5=float(np.max(fft_corr(best_h)))*(2.0/len(best_h))
    return best_h, final_c5, len(best_h)
