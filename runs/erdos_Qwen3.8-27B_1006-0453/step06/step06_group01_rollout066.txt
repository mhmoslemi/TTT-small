import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def weighted_grad(h, N, tau=2.0):
    """Compute gradient of max corr using soft-max weighting."""
    corr = fft_corr(h)
    mx = float(corr.max())
    # Soft weights: w[m] = exp((corr[m] - mx) / tau)
    w = np.exp((corr - mx) / tau)
    w_sum = w.sum()
    if w_sum < 1e-10:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        w_sum = 1.0
    w /= w_sum
    
    grad = np.zeros(N)
    # For each shift m with weight w[m], add w[m] * d(corr[m])/dh
    # d(corr[m])/dh[i] = (1-h[i-L])·[valid] - h[i+L]·[valid] where L = m-(N-1)
    # Vectorize: for each m, the gradient contribution is a "convolution-like" pattern
    
    # Alternative: use the fact that corr = ifft(fft(h) * fft(flip(1-h)))
    # d(corr)/dh[i] can be computed via the correlation structure
    
    # For efficiency, only process shifts with significant weight
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    
    for m in active:
        L = int(m - (N - 1))
        wm = w[m]
        if L == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wm * (1.0 - h[:N-L])
            grad[:N-L] -= wm * h[L:]
        else:
            M = -L
            grad[:N-M] += wm * (1.0 - h[M:])
            grad[M:] -= wm * h[:N-M]
    
    return grad, mx, corr

def adam_continuous(h_in, N, deadline, lr=0.01, tau=2.0, beta1=0.9, beta2=0.999):
    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    while time.time() < deadline:
        t += 1
        grad, mx, corr = weighted_grad(h, N, tau)
        c_val = mx * (2.0 / N)
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h.copy()
        
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad * grad
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        
        lr_t = lr * (1.0 / (1.0 + 0.001 * t))
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_sum(h_new, target)
        h = h_new
    
    return best_h, best_c

def binary_swap_search(h_in, N, deadline, max_rounds=2000, n_p=20):
    """Binary local search: try swap pairs, compute new max efficiently."""
    h = h_in.astype(float).copy()
    h = (h > 0.5).astype(float)
    # Fix sum
    n1 = int(h.sum())
    tgt = N // 2
    if n1 > tgt:
        ones = np.where(h == 1)[0]
        h[np.random.choice(ones, n1 - tgt, replace=False)] = 0.0
    elif n1 < tgt:
        zeros = np.where(h == 0)[0]
        h[np.random.choice(zeros, tgt - n1, replace=False)] = 1.0
    
    corr = fft_corr(h).copy()
    best_c = float(corr.max())
    best_h = h.copy()
    no_imp = 0
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        improved = False
        n_p_sample = min(n_p, len(ones))
        p_list = np.random.choice(ones, n_p_sample, replace=False)
        
        for p in p_list:
            # Compute delta_1to0 for position p
            # Δ[m] = -(1-h[p-L])·[0≤p-L<N] + h[p+L]·[0≤p+L<N] where L=m-(N-1)
            L_all = np.arange(2*N-1) - (N-1)  # shape (2N-1,)
            d = np.zeros(2*N-1)
            
            # Term 1: -(1-h[p-L]) where 0 <= p-L < N, i.e., L in (p-N, p]
            idx1 = np.where((p - L_all >= 0) & (p - L_all < N))[0]
            if len(idx1) > 0:
                d[idx1] -= (1.0 - h[p - L_all[idx1]])
            
            # Term 2: +h[p+L] where 0 <= p+L < N, i.e., L in [-p, N-1-p]
            idx2 = np.where((p + L_all >= 0) & (p + L_all < N))[0]
            if len(idx2) > 0:
                d[idx2] += h[p + L_all[idx2]]
            
            # Now for each q in zeros, compute new_corr = corr + d + delta_0to1[q]
            # and find q that minimizes max(new_corr)
            # delta_0to1[q, m] = (1-h[q-L])·[0≤q-L<N] - h[q+L]·[0≤q+L<N]
            
            # Vectorize over q:
            # For each q, the delta_0to1 is:
            #   (1-h[q-L_m]) where valid, minus h[q+L_m] where valid
            # This is a shifted version of h.
            
            # For efficiency, compute for all q at once using matrix ops
            n_q = len(zeros)
            if n_q > 200:
                q_sample = np.random.choice(zeros, 200, replace=False)
            else:
                q_sample = zeros
            
            # q_sample + L_all and q_sample - L_all
            # delta_0to1[q, m] = (1-h[q-L_m])·[valid] - h[q+L_m]·[valid]
            # For all q in q_sample:
            qL = q_sample[:, None] + L_all[None, :]  # (n_q, 2N-1)
            qmL = q_sample[:, None] - L_all[None, :]
            
            mask_p = (qL >= 0) & (qL < N)
            mask_m = (qmL >= 0) & (qmL < N)
            
            # h[q+L] and h[q-L]
            h_qL = h[np.minimum(qL, N-1)] * mask_p  # careful with indexing
            h_qmL = h[np.minimum(qmL, N-1)] * mask_m
            
            # Actually need to handle boundaries: where mask is False, value is 0
            h_qL = np.where(mask_p, h[np.clip(qL, 0, N-1)], 0.0)
            h_qmL = np.where(mask_m, h[np.clip(qmL, 0, N-1)], 0.0)
            
            d01 = (1.0 - h_qmL) * mask_m - h_qL * mask_p  # (n_q, 2N-1)
            
            new_corrs = corr[None, :] + d[None, :] + d01  # (n_q, 2N-1)
            new_maxes = new_corrs.max(axis=1)  # (n_q,)
            
            best_qi = int(new_maxes.argmin())
            if new_maxes[best_qi] < best_c - 1e-10:
                q = int(q_sample[best_qi])
                h[p], h[q] = 0.0, 1.0
                corr = fft_corr(h)  # Recompute for safety
                best_c = float(corr.max())
                best_h = h.copy()
                no_imp = 0
                improved = True
                break
        
        if not improved:
            no_imp += 1
            if no_imp > 50:
                break
    
    return best_h, best_c * (2.0 / N)
