def fft_corr(h):
    """Compute cross-correlation of h with 1-h using FFT."""
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h, N):

def compute_grad(h, corr, N, weights):

def swap_delta(h, N, p, q):
    s = np.arange(-(N - 1), N)
    ps, pm = p + s, p - s
    qs, qm = q + s, q - s
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)

        sub_c = min(deadline - 60, time.time() + 22)
        hct, pkt = cont_opt(hbt.astype(float), Nt, sub_c, tau0=0.003, lr0=0.18)
        c5t = pkt * dt; bestNt = hct.copy(); c5Nt = c5t
        sub_b = min(deadline - 45, time.time() + 38)
        hb2 = fix_count((bestNt > 0.5).astype(float), Nt)
        hb2, pkb2 = bin_anneal(hb2, Nt, sub_b, T0=0.04, cool=0.9995, max_stall=800)
        c5b2 = pkb2 * dt
        if c5b2 < c5Nt: c5Nt, bestNt = c5b2, hb2.copy()
        if c5Nt < best_c5: best_c5, best_h = c5Nt, bestNt.copy()

    # Phase 5: Final polish
    if time.time() < deadline - 8:
        Nf = len(best_h)
        hf = fix_count((best_h > 0.5).astype(float), Nf)
        hfp, pkf = bin_anneal(hf, Nf, deadline - 2, T0=0.012, cool=0.9997, max_stall=800)
        c5f = pkf * (2.0 / Nf)
        if c5f < best_c5: best_c5, best_h = c5f, hfp.copy()

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
