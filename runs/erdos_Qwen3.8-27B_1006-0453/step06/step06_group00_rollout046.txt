import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgrad_step(h, N, k_active=100, lr=0.01):
    corr = fft_corr(h)
    top = np.argpartition(corr, -k_active)[-k_active:]
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0*h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N-L])
            grad[:N-L] -= h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += (1.0 - h[M:])
                grad[M:] -= h[:N-M]
    grad /= k_active
    return h - lr * grad

def nesterov_cont(h, N, deadline, k_active=100, lr0=0.015, beta0=0.9):

    best_h = h.copy()

    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        c = float(np.max(corr))
        if c < best_c:
            best_c = c

        top = np.argpartition(corr, -k_active)[-k_active:]

            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])

        h = h_new
        it += 1

            lr *= 0.99
            beta = min(0.98, 0.85 + it * 0.0001)
    return best_h, best_c

def block_move_search(h, N, deadline, max_iters=500, block_sz=3):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h))) * (2.0/N)
    best_h = h.copy()

            break

        # Need total = N/2, so we need to adjust by (N/2 - rest_1 - n0_block) ones elsewhere

        if abs(diff) > min(rest_0, rest_1):
            continue  # can't maintain constraint
        
        h_new = h.copy()
        h_new[start:start+b] = 1.0 - block
        if diff > 0:
            # Need to set 'diff' zeros in the rest to 1
            zero_idx = np.where((h_new == 0) & (np.arange(N) < start | np.arange(N) >= start+b))[0]
            if len(zero_idx) < diff:

                zero_idx = zero_idx[zero_idx != np.arange(start, start+b)]

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def sa_search(h, N, deadline, t_start=0.1, t_end=0.001, max_iters=2000):

    best_c = compute_c5(h, N)
    best_h = h.copy()

            break
        T = t_start * (t_end/t_start) ** (it / max(1, n_iter_target))
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, N)
        if c_new < cur_c:

            if c_new < best_c:

                cur_c = c_new
            else:
                h[i], h[j] = h[j], h[i]
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_grad(h, N, tau):
    corr = fft_corr(h)
    mx = float(np.max(corr))
    w = np.exp(tau * (corr - mx))
    sw = w.sum()
    w /= sw
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-8
    for k in range(2 * N - 1):
        if w[k] < thresh:
            continue
        L = k - center
        if L == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w[k] * (1.0 - h[:N - L])
            grad[:N - L] -= w[k] * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += w[k] * (1.0 - h[M:])
                grad[M:] -= w[k] * h[:N - M]
    return grad, mx

def nesterov_smooth(h, N, deadline, tau0=15.0, lr0=0.02, beta0=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    tau = tau0
    it = 0
    while time.time() < deadline:
        y = h + m
        grad, pk = smoothed_grad(y, N, tau)
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        h_new = project_01(np.clip(y - lr * grad, 0.0, 1.0), target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            lr *= 0.992
            beta = min(0.98, 0.85 + it * 0.00008)
            tau = min(tau * 1.02, 80.0)
    return best_h, best_c

def pairwise_il(h, N, deadline, max_iters=1000, sample_sz=55):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        center = N - 1
        # Compute correlation change for each swap
        # corr_new = corr + delta where delta comes from changing h[i]=1->0, h[j]=0->1
        # delta(k) = -[(1-h[k+i]) * h[i] + ... ] complicated; use vectorized approach
        # For swap (i in ones -> 0, j in zeros -> 1):
        # Effect on corr at lag L: 
        #   old contribution from i: h[i]*(1-h[i+L]) if i+L valid
        #   new: 0
        #   old contribution from j: h[j]*(1-h[j+L]) = 0 (since h[j]=0)  
        #   new: 1*(1-h[j+L]) if j+L valid
        # This is complex for linear correlation; use approximation via FFT
        # Simpler: just evaluate a small number of candidates
        candidates = []
        for a in range(ns):
            for b in range(nz):
                i, j = os_[a], zs_[b]
                # Quick estimate: check a few key lags
                candidates.append((i, j))
        idx_all = np.random.choice(len(candidates), min(80, len(candidates)), replace=False)
        best_mv = P
        best_pair = None
        for ci in idx_all:
            i, j = candidates[ci]
            h[i], h[j] = 0.0, 1.0
            c_new = float(np.max(fft_corr(h)))
            h[i], h[j] = 1.0, 0.0
            if c_new < best_mv:
                best_mv = c_new
                best_pair = (i, j)
        if best_pair is not None:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
        else:
            break
    return best_h, best_c

def sa_refine(h, N, deadline, t_start=0.04, t_end=0.0005):
    h = h.astype(float).copy()
    best_c = compute_c5(h, N)
    best_h = h.copy()
    cur_c = best_c
    total_iters = 5000
    it = 0
    while time.time() < deadline and it < total_iters:
        T = t_start * (t_end / t_start) ** (it / max(1, total_iters))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h, N)
        if c_new <= cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            if np.random.random() < np.exp(-(c_new - cur_c) / max(T, 1e-10)):
                cur_c = c_new
            else:
                h[i], h[j] = h[j], h[i]
        it += 1
    return best_h, best_c
