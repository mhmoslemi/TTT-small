import numpy as np
import time

def project_01(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def subgrad_lag(h, lag, N):
    """Subgradient of corr at a given lag (0-indexed, lag in [-(N-1), N-1])."""
    grad = np.zeros(N)
    if lag >= 0:
        # d/dh[j]: (1-h[j-lag]) if j-lag valid, plus -h[j+lag] if j+lag valid
        # j-lag valid: j >= lag; j+lag valid: j < N-lag
        idx_high = np.arange(lag, N)
        grad[idx_high] += (1.0 - h[idx_high - lag])
        idx_low = np.arange(0, N - lag)
        grad[idx_low] -= h[idx_low + lag]
    else:
        M = -lag
        idx_high = np.arange(M, N)
        grad[idx_high] += (1.0 - h[idx_high - M])
        idx_low = np.arange(0, N - M)
        grad[idx_low] -= h[idx_low + M]
    return grad

def smooth_subgrad_step(h, N, deadline, lam_init=50.0, lam_end=300.0,
                        lr0=0.015, lr_decay=0.9995, min_lr=0.001):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    it = 0
    center = N - 1
    while time.time() < deadline and lr > min_lr:
        corr = fft_corr(h)
        c_val = float(np.max(corr))
        c5_val = c_val * (2.0 / N)
        if c5_val < best_c:
            best_c = c5_val
            best_h = h.copy()
        # Softmax weights over lags
        lam = lam_init + (lam_end - lam_init) * min(1.0, it / 2000.0)
        shifted = corr - np.max(corr)
        w = np.exp(lam * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * N - 1)
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        # Weighted subgradient
        grad = np.zeros(N)
        active = np.where(w > 1e-10)[0]
        for m in active:
            lag = m - center
            grad += w[m] * subgrad_lag(h, lag, N)
        h_new = project_01(h - lr * grad, target)
        if not np.all(np.isfinite(h_new)):
            break
        h = h_new
        it += 1
        lr = max(min_lr, lr * lr_decay)
    return best_h, best_c

            for bi in range(nz):
                b = zs[bi]
                # Quick: just evaluate the max change using the vector
                h[a] = 0.0
                h[b] = 1.0
                c_new = float(np.max(fft_corr(h)))
                h[a] = 1.0
                h[b] = 0.0
                d = c_new - P
                if d < best_delta:
                    best_delta = d
                    best_a, best_b = a, b
        if best_a >= 0 and best_delta < -1e-14:
            h[best_a] = 0.0
            h[best_b] = 1.0
    return best_h, best_c * (2.0 / N)

def fast_binary_search(h, N, deadline, max_iters=5000, sample_sz=50):
    """Faster binary search using precomputed C-matrix for O(1) swap evaluation."""
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1.0)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1.0
    best_c = np.inf
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        # Precompute contribution vectors
        # For index i, define v_i[m] = h[i]*(1-h[i-((m-center))]) contribution
        # When we set h[i]=0, we remove its contributions; h[j]=1, we add.
        # Use the full correlation difference
        center = N - 1
        # Vector for "setting h[i]=1" contribution to corr: e_i convolved with (1-h) pattern
        # Simpler: corr_new = corr - contrib_removed + contrib_added
        # contrib of h[i] to corr: for each valid lag, h[i]*(1-h[i-lag])
        # This is equivalent to: h[i] * correlated_vector
        # Let's use: corr = sum_i h[i] * v_i where v_i[m] = (1-h[m-center-i]) if valid
        # Setting h[a]:0 removes h[a]*v_a; setting h[b]:1 adds 1*v_b
        # v_a[m]: 1-h[m-(N-1)-a] if 0<=m-(N-1)-a<N else 0
        # = 1-h[m - center - a] valid when 0 <= m-center-a < N
        # v_b[m]: 1-h[m - center - b] valid when 0 <= m-center-b < N
        va = np.zeros(2 * N - 1)
        lo_a = center + a - (N - 1)
        hi_a = center + a
        lo_a = max(0, lo_a)
        hi_a = min(2 * N - 2, hi_a + 1)
        # v_a[m] = 1-h[m-center-a] for 0<=m-center-a<N
        # m ranges from center+a to center+a+N-1
        m_start = max(0, center + a)
        m_end = min(2 * N - 1, center + a + N)
        if m_end > m_start:
            idx = np.arange(m_start, m_end)
            va[idx] = 1.0 - h[idx - (center + a)]
        vb = np.zeros(2 * N - 1)
        m_start_b = max(0, center + b) if False else max(0, center + b)
        m_start_b = max(0, center + b)
        m_end_b = min(2 * N - 1, center + b + N)
        if m_end_b > m_start_b:
            idx_b = np.arange(m_start_b, m_end_b)
            vb[idx_b] = 1.0 - h[idx_b - (center + b)]
        # New corr = corr - va + vb (since h[a] was 1, becomes 0; h[b] was 0, becomes 1)
        # But also need to account for the (1-h) part changing:
        # When h[a] changes 1->0, the (1-h[a]) in other terms changes 0->1
        # When h[b] changes 0->1, the (1-h[b]) in other terms changes 1->0
        # So: new_corr = corr - h[a]*v_a + h[b]*v_b + (1-h[a])_new*(contributions where a is in 1-h) - ...
        # This is getting complex. Let me just use the simpler approach for small sample sizes.
        best_delta = 0.0
        best_ai, best_bi = -1, -1
        for ai in range(ns):
            a = int(os_[ai])
            for bi in range(nz):
                b = int(zs[bi])
                h[a] = 0.0
                h[b] = 1.0
                c_new = float(np.max(fft_corr(h)))
                h[a] = 1.0
                h[b] = 0.0
                d = c_new - P
                if d < best_delta:
                    best_delta = d
                    best_ai, best_bi = ai, bi
        if best_ai >= 0 and best_delta < -1e-14:
            h[os_[best_ai]] = 0.0
            h[zs[best_bi]] = 1.0
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    def remaining():
        return deadline - time.time()

    # Phase 1: Smoothed subgradient (continuous)
    d1 = min(deadline - 400, t0 + 200)
    hc = best_h.copy()
    hc[~np.isfinite(hc)] = 0.5
    hc = project_01(hc, N0 / 2.0)
    hs, c5s = smooth_subgrad_step(hc, N0, d1, lam_init=30.0, lam_end=200.0,
                                   lr0=0.02, lr_decay=0.9997)
    if c5s < best_c5:
        best_c5, best_h = c5s, hs.copy()

    # Phase 2: Binary fast search
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 130)
        hb = (best_h > 0.5).astype(float)
        hs2, c5s2 = fast_binary_search(hb, N0, d2, max_iters=4000, sample_sz=40)
        if c5s2 < best_c5:
            best_c5, best_h = c5s2, hs2.copy()

    # Phase 3: Second subgradient pass with different params
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 100)
        hc3 = best_h + np.random.randn(N0) * 0.02
        hc3 = project_01(hc3, N0 / 2.0)
        hs3, c5s3 = smooth_subgrad_step(hc3, N0, d3, lam_init=80.0, lam_end=400.0,
                                         lr0=0.01, lr_decay=0.9998)
        if c5s3 < best_c5:
            best_c5, best_h = c5s3, hs3.copy()

    # Phase 4: Deeper binary search
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 150)
        hb4 = (best_h > 0.5).astype(float)
        hs4, c5s4 = fast_binary_search(hb4, N0, d4, max_iters=6000, sample_sz=60)
        if c5s4 < best_c5:
            best_c5, best_h = c5s4, hs4.copy()

    # Phase 5: Try slightly different N values
    for N_try in [378, 382, 384, 386]:
        if time.time() > deadline - 60:
            break
        if N_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1.0)[0]
            h_bt[np.random.choice(idx, n1 - nh, replace=False)] = 0.0
        elif n1 < nh:
            idx = np.where(h_bt == 0.0)[0]
            h_bt[np.random.choice(idx, nh - n1, replace=False)] = 1.0
        sub_dl = min(deadline - 10, time.time() + 50)
        h_rt, c5_rt = fast_binary_search(h_bt, N_try, sub_dl, max_iters=2000, sample_sz=50)
        c5_rt_check = compute_c5(h_rt, N_try)
        if c5_rt_check < best_c5:
            best_c5, best_h = c5_rt_check, h_rt.copy()

    # Phase 6: Final subgradient polish
    if time.time() < deadline - 8:
        hF = best_h.copy()
        hF = project_01(hF, len(hF) / 2.0)
        hF, c5F = smooth_subgrad_step(hF, len(hF), deadline - 2,
                                       lam_init=150.0, lam_end=500.0,
                                       lr0=0.005, lr_decay=0.9999)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Final: ensure binary and check
    hB = (best_h > 0.5).astype(float)
    n1 = int(hB.sum())
    nh = len(hB) // 2
    if n1 != nh:
        if n1 > nh:
            idx = np.where(hB == 1.0)[0]
            hB[np.random.choice(idx, n1 - nh, replace=False)] = 0.0
        else:
            idx = np.where(hB == 0.0)[0]
            hB[np.random.choice(idx, nh - n1, replace=False)] = 1.0
    cB = compute_c5(hB, len(hB))
    cC = compute_c5(best_h, len(best_h))
    if cB <= cC:
        best_h, best_c5 = hB, cB
    else:
        best_c5 = cC
    return best_h, best_c5, len(best_h)
