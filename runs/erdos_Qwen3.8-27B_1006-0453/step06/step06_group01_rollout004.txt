import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        perm = np.random.permutation(len(ones))
        for i in range(n1 - nh):
            h[ones[perm[i]]] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        perm = np.random.permutation(len(zeros))
        for i in range(nh - n1):
            h[zeros[perm[i]]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def targeted_local_search(h, N, deadline, max_evals=100000, block_max=3):
    h = h.copy()
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    eval_count = 0
    no_improve = 0
    rng = np.random.RandomState(42)

    while eval_count < max_evals and time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        improved = False
        n_tries = 40
        for _ in range(n_tries):
            # Targeted: focus on worst lag
            worst_k = int(np.argmax(corr))
            d = int(worst_k - (N - 1))
            if d >= 0 and d < N:
                # Positions i where h[i]=1, h[i+d]=0 contribute to this lag
                cands_p = np.array([i for i in ones if i + d < N and h[i + d] == 0])
                cands_q = np.array([i for i in zeros if i + d < N and h[i + d] == 1])
            elif d < 0:
                m = -d
                cands_p = np.array([i for i in ones if i >= m and h[i - m] == 0])
                cands_q = np.zeros(0, dtype=int)
                cands_q = np.array([i for i in zeros if i >= m and h[i - m] == 1])
            else:
                cands_p = ones
                cands_q = zeros

            if len(cands_p) > 0 and len(cands_q) > 0:
                p = int(cands_p[rng.randint(len(cands_p))])
                q = int(cands_q[rng.randint(len(cands_q))])
            else:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])

            h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
            c_new = float(np.max(corr))
            eval_count += 1
