import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_gradient(h, top_idx, weights, N):
    """Compute weighted gradient of correlation values at top_idx."""
    grad = np.zeros(N)
    center = N - 1
    for wi, ai in zip(weights, top_idx):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad

            h = best_h + np.random.randn(N) * 0.03

            exp_c = np.exp((corr_top - max_c) / T)

                    grad[:seg] += wi * (1.0 - z[M:])

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_flip_deltas(h, N):
    """Compute delta to corr vector for each position flipping 1→0. Shape (N, 2N-1)."""
    L = 2 * N - 1
    idx_fwd = np.arange(N)[:, None] + np.arange(N)[None, :]  # (N,N)
    idx_bwd = np.arange(N)[:, None] - np.arange(N)[None, :]  # (N,N)
    valid_fwd = idx_fwd < N
    valid_bwd = idx_bwd >= 0
    fwd = np.zeros((N, N))

            idx = idx[np.random.permutation(len(idx))[:n1 - target]]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))[:target - n1]]
            h[idx] = 1
    
    L = 2 * N - 1
    corr = fft_corr(h)
    current_max = float(corr.max())
    best_h = h.copy()

            break
        
        # Compute flip deltas
        deltas = compute_flip_deltas(h, N)  # (N, L)
        
        # Find best swap
        best_val = current_max
        best_p, best_q = -1, -1
        batch_size = 10
        for start in range(0, len(ones), batch_size):
            p_b = ones[start:start+batch_size]
            # new_corr[p_batch_idx, q_idx, lag] = corr + delta[p] - delta[q]
            dp = deltas[p_b][None, :, :]  # (batch, 1, L)
            dq = deltas[zeros][:, None, :]  # (n_zeros, 1, L) → need (1, n_zeros, L)
            dq = deltas[zeros][None, :, :]  # (1, n_zeros, L)
            new_batch = corr[None, None, :] + dp - dq  # (batch, n_zeros, L)
            maxes = new_batch.max(axis=2)  # (batch, n_zeros)
            flat_min = maxes.min()
            if flat_min < best_val:
                best_val = float(flat_min)
                min_idx = maxes.argmin()
                bi, qi = divmod(min_idx, len(zeros))
                best_p = int(p_b[bi])
                best_q = int(zeros[qi])
            del new_batch
        
        if best_p >= 0:
            h[best_p] = 0.0
            h[best_q] = 1.0
            corr = fft_corr(h)
            current_max = float(corr.max())
            stall = 0
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                # Perturb: flip 5 random pairs
                n_flip = min(5, len(ones) // 2, len(zeros) // 2)
                fp = ones[np.random.choice(len(ones), n_flip, replace=False)]
                fq = zeros[np.random.choice(len(zeros), n_flip, replace=False)]
                h[fp] = 0.0
                h[fq] = 1.0
                corr = fft_corr(h)
                current_max = float(corr.max())
                stall = 0
    
    return best_h, best_max * (2.0 / N)

def continuous_opt(h0, N, deadline, k=80, lr0=0.02, temp0=0.02, beta0=0.9):
    """Adam-style continuous optimization with log-sum-exp."""
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    
    m = np.zeros(N)  # Adam first moment
    v = np.zeros(N)  # Adam second moment
    t_adam = 0
    lr = lr0
    temp = temp0
    beta = beta0
    stall = 0
    h_prev = h.copy()
    
    while time.time() < deadline:
        y = h + beta * (h - h_prev)  # Nesterov
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Top-k lags with softmax weighting
        top_k = np.argpartition(corr, -k)[-k:]
        corr_top = corr[top_k]
        max_c = corr_top.max()
        w = np.exp((corr_top - max_c) / max(temp, 1e-10))
        w /= w.sum()
        
        # Gradient
        grad = np.zeros(N)
        center = N - 1
        for wi, ai in zip(w, top_k):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                seg = N - L
                grad[L:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:]
            else:
                M = -L
                seg = N - M
                grad[:seg] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:seg]
