import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_p(h, N, tau):
    corr = fft_corr(h)
    mx = np.max(corr)
    e = np.exp(tau * (corr - mx))
    S = e.sum()
    if S < 1e-300:
        return corr, np.full(len(corr), 1.0 / len(corr)), float(mx)
    return corr, e / S, float(mx)

def lse_grad(h, N, p):
    L = 2 * N - 1
    L_fft = 4 * N
    P_f = np.fft.rfft(p, L_fft)
    H_f = np.fft.rfft(h, L_fft)
    conv = np.fft.irfft(P_f * H_f, L_fft)
    A = conv[N - 1:2 * N - 1]
    P_rev_f = np.fft.rfft(p[::-1], L_fft)
    conv2 = np.fft.irfft(P_rev_f * H_f, L_fft)
    B = conv2[N - 1:2 * N - 1]
    return 1.0 - A - B

def continuous_opt(h0, N, deadline, tau0=20.0, tau1=200.0, lr0=0.04, k_mom=0.85):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    v = np.zeros(N)
    lr = lr0
    tau = tau0
    stall = 0
    it = 0
    total_iters = max(1, int((time.time() < deadline) * 999999))
    while time.time() < deadline:
        progress = min(1.0, it / 800.0)
        tau = tau0 + (tau1 - tau0) * progress
        corr, p, mx = lse_p(h, N, tau)
        if mx < best_pk - 1e-14:
            best_pk = mx
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        g = lse_grad(h, N, p)
        v = k_mom * v - lr * g
        h_new = h + v
        h_new = project_sum(h_new, target)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.97
            tau = min(tau * 1.05, tau1 * 3)
        if stall > 500:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            v *= 0.3
            lr = max(lr, lr0 * 0.8)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_opt(h0, N, deadline, n_tries=15, kick_freq=60):
    h = (h0 > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)
            h[idx] = 0
        else:
            idx = np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)
            h[idx] = 1
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sweep_int = int(c.max())
        best_sweep_h = None
        best_sweep_c = None
        pk_lag = int(np.argmax(c))
        pk_lag_n = pk_lag - (N - 1)
        for _ in range(n_tries):
            if np.random.random() < 0.5 and abs(pk_lag_n) < N:
                sh = abs(pk_lag_n)
                if pk_lag_n >= 0:
                    cands = np.where((h[:N - sh] == 1) & (h[sh:] == 0))[0]
                else:
                    cands = np.where((h[-sh:] == 1) & (h[:N - sh] == 0))[0]
                if len(cands) > 0:

                    q = p + pk_lag_n if pk_lag_n > 0 else p - pk_lag_n

                    q = zeros[np.random.randint(len(zeros))]

            h[p] = 0
            h[q] = 1

            h[p] = 1
            h[q] = 0
        if best_sweep_h is not None:

            c = best_sweep_c

                    qs = zeros[np.random.choice(len(zeros), 3, replace=False)]
                    h[ps] = 0
                    h[qs] = 1
                    c = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.arange(N_old) / N_old * 2.0
    x_new = np.arange(N_new) / N_new * 2.0
    return np.interp(x_new, x_old, h)

def ensure_sum(h, N):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:

    lo, hi = -1.0, 1.0
