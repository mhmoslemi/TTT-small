def nesterov_opt(h, N, deadline, k=60, lr0=0.015, beta0=0.90, use_soft=False, soft_beta=50.0):
    """Nesterov with optional log-sum-exp for smoother gradient"""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    center = N - 1
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Compute subgradient
        if use_soft:
            # Log-sum-exp gradient
            max_c = corr.max()
            shifted = np.exp((corr - max_c) * soft_beta)
            weights = shifted / shifted.sum()
            grad = np.zeros(N)
            # For each lag with non-zero weight, add contribution
            # This is expensive for large N with many lags
            # Use top-k approximation
            top_k = min(k, 2 * N - 1)
            top = np.argpartition(corr, -top_k)[-top_k:]
            w_top = np.exp((corr[top] - max_c) * soft_beta)
            w_top /= w_top.sum()
            for ai in range(len(top)):
                ti = top[ai]
                L = int(ti - center)
                w = w_top[ai]
                if L == 0:
                    grad += w * (1.0 - 2.0 * y)
                elif L > 0:
                    grad[L:] += w * (1.0 - y[:N - L])
                    grad[:N - L] -= w * y[L:]
                else:
                    M = -L
                    if M < N:
                        grad[:N - M] += w * (1.0 - y[M:])
                        grad[M:] -= w * y[:N - M]
        else:
            # Hard max subgradient (top-k average)
            top_k = min(k, 2 * N - 1)
            top = np.argpartition(corr, -top_k)[-top_k:]
            grad = np.zeros(N)
            for ai in top:
                L = int(ai - center)
                if L == 0:
                    grad += (1.0 - 2.0 * y)
                elif L > 0:
                    grad[L:] += (1.0 - y[:N - L])
                    grad[:N - L] -= y[L:]
                else:
                    M = -L
                    if M < N:
                        grad[:N - M] += (1.0 - y[M:])
                        grad[M:] -= y[:N - M]
            grad /= len(top)
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, beta0 + it * 0.0001)
    
    return best_h, best_pk * (2.0 / N)

- h is a step function on [0,2] with integral 1 (so average value 1/2)
- We want to minimize the maximum overlap ∫ h(x)(1-h(x+k)) dx

For a binary function with exactly N/2 ones, the overlap at lag k is:
number of i where h[i]=1 and h[i+k]=0 (for lag k)

The maximum of this over all k is what we're minimizing.

A key insight: if h is a "balanced" binary sequence (like a Thue-Morse sequence or a van der Corput sequence), the autocorrelation properties might be good. But we're looking at cross-correlation between h and 1-h, which is different.

Actually, ∫ h(x)(1-h(x+k)) dx for binary h is:
= sum_i h[i](1-h[i+k])
= (number of 1s in h) - sum_i h[i]h[i+k]
= N/2 - autocorrelation at lag k

So minimizing the max of ∫h(x)(1-h(x+k)) is equivalent to maximizing the min of the autocorrelation!

So C5 = N/2 - min_k A[k].

To minimize C5, we want to maximize min_k A[k], i.e., make the autocorrelation as "flat" as possible (high minimum).

This is related to the construction of sequences with low Peak-to-Average Power Ratio (PAPR) or "Golay sequences" type objects.

For a binary sequence of length N with N/2 ones, the autocorrelation A[0] = N/2 (when lag=0, sum h[i]^2 = N/2 for binary). For other lags, A[k] = sum h[i]h[i+k].

We want to make A[k] as large as possible for the minimum k. Ideally, all A[k] for k≠0 would be close to N/4 (the expected value for a random sequence). But we can do better.

def eval_swaps_vectorized(c, h, p, q_list, top_idx, N):
    """Evaluate 1-swap of h[p]:1->0 with each q in q_list: 0->1"""
    center = N - 1
    Ls = top_idx.astype(int) - center  # shape (top_m,)
    
    # For position p (1->0):
    # delta_p at each lag L:
    #   -(1-h[p+L]) if 0<=p+L<N
    #   +h[p-L] if 0<=p-L<N
    ip = p + Ls  # p+L for each lag
    ip2 = p - Ls  # p-L for each lag
    dp = np.zeros(len(Ls))
    valid1 = (ip >= 0) & (ip < N)
    dp[valid1] -= (1.0 - h[ip[valid1]])
    valid2 = (ip2 >= 0) & (ip2 < N)
    dp[valid2] += h[ip2[valid2]]
    
    # For each q (0->1):
    # delta_q at each lag L:
    #   +(1-h[q+L]) if 0<=q+L<N
    #   -h[q-L] if 0<=q-L<N
    # Shape: (len(q_list), top_m)
    iq = q_list[:, None] + Ls[None, :]  # (nq, top_m)
    iq2 = q_list[:, None] - Ls[None, :]
    dq = np.zeros((len(q_list), len(Ls)))
    v1 = (iq >= 0) & (iq < N)
    dq[v1] += (1.0 - h[iq[v1]])
    v2 = (iq2 >= 0) & (iq2 < N)
    dq[v2] -= h[iq2[v2]]
    
    # Total delta: (nq, top_m)
    total_delta = dq + dp[None, :]
    
    # New correlation values at top positions: (nq, top_m)
    new_vals = c[top_idx][None, :] + total_delta
    
    # Max over top positions for each q
    new_maxes = new_vals.max(axis=1)  # (nq,)
    
    return new_maxes

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

        # Precompute delta_q: (nsq, top_m)
        iq = q_list[:, None].astype(np.int64) + Ls[None, :]
        iq2 = q_list[:, None].astype(np.int64) - Ls[None, :]
        dq = np.zeros((nsq, top_m))
        v1 = (iq >= 0) & (iq < N)
        dq[v1] += (1.0 - h[iq[v1]])
        v2 = (iq2 >= 0) & (iq2 < N)
        dq[v2] -= h[iq2[v2]]
        # Precompute delta_p: (nsp, top_m)
        ip_all = p_list[:, None].astype(np.int64) + Ls[None, :]
        ip2_all = p_list[:, None].astype(np.int64) - Ls[None, :]
        dp_all = np.zeros((nsp, top_m))
        w1 = (ip_all >= 0) & (ip_all < N)
        dp_all[w1] -= (1.0 - h[ip_all[w1]])
        w2 = (ip2_all >= 0) & (ip2_all < N)
        dp_all[w2] += h[ip2_all[w2]]
        # Full 1-swap evaluation: (nsp, nsq, top_m)
        nv = c_top[None, None, :] + dp_all[:, None, :] + dq[None, :, :]
        nm = nv.max(axis=2)
        gi = np.unravel_index(np.argmin(nm), nm.shape)
        best_new_P = float(nm[gi])
        best_move = ('1s', int(p_list[gi[0]]), int(q_list[gi[1]]))
        # 2-swap moves
        if use_2swap and nsp >= 4 and nsq >= 4:
            for _k in range(min(n_2swap, nsp // 2)):
                pa, pb = rng.choice(nsp, 2, replace=False)
                qa, qb = rng.choice(nsq, 2, replace=False)
                d2 = dp_all[pa] + dp_all[pb] + dq[qa] + dq[qb]
                nv2 = (c_top + d2).max()
                if nv2 < best_new_P:
                    best_new_P = float(nv2)
                    best_move = ('2s', int(p_list[pa]), int(p_list[pb]), int(q_list[qa]), int(q_list[qb]))
        if best_new_P < P:
            if best_move[0] == '1s':
                h[best_move[1]] = 0.0
                h[best_move[2]] = 1.0
            else:
                h[best_move[1]] = 0.0
                h[best_move[2]] = 0.0
                h[best_move[3]] = 1.0
                h[best_move[4]] = 1.0
    return h, best_c

def nesterov_opt(h, N, deadline, k=60, lr0=0.015, soft_beta=40.0):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.90
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        max_c = corr.max()
        top_k = min(k, 2 * N - 1)
        top = np.argpartition(corr, -top_k)[-top_k:]
        w = np.exp((corr[top] - max_c) * soft_beta)
        w /= w.sum()
        grad = np.zeros(N)
        for ii in range(len(top)):
            L = int(top[ii] - center)
            wi = w[ii]
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.994
            beta = min(0.97, 0.90 + it * 0.00012)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    hp = np.array(initial_h_values, dtype=float)
    N0 = len(hp)
    best_h = hp.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: Vectorized binary search on parent
    d1 = min(deadline - 780, t0 + 200)
    h1, _ = vec_binary_search(hp, N0, d1, rng, max_sweeps=800, sample_n=110, top_m=16, use_2swap=True, n_2swap=15)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()
    # Phase 2: Multi-resolution exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    res_list = [360, 380, 400, 420, 440, 460, 500, 540, 600, 650, 700]
    d2_end = min(deadline - 500, time.time() + 220)
    for Nt in res_list:
        if time.time() > d2_end - 25:
            break
        if Nt == len(best_h):
            continue
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh = Nt // 2
        n1c = int(h_bt.sum())
        if n1c > nh:
            idx = np.where(h_bt == 1)[0]
            h_bt[idx[rng.permutation(len(idx))[nh:]]] = 0
        elif n1c < nh:
            idx = np.where(h_bt == 0)[0]
            h_bt[idx[rng.permutation(len(idx))[:nh - n1c]]] = 1
        sub_dl = min(d2_end - 10, time.time() + 35)
        h_rt, _ = vec_binary_search(h_bt, Nt, sub_dl, rng, max_sweeps=200, sample_n=90, top_m=14, use_2swap=True, n_2swap=10)
        c5_rt = compute_c5(h_rt, Nt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, Nt, endpoint=False)
    # Phase 3: Extended binary search with 2-swaps on best
    if time.time() < deadline - 280:
        d3 = min(deadline - 280, time.time() + 250)
        h3, _ = vec_binary_search(best_h, len(best_h), d3, rng, max_sweeps=2000, sample_n=130, top_m=20, use_2swap=True, n_2swap=25)
        c3 = compute_c5(h3, len(best_h))
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
    # Phase 4: Continuous Nesterov polish
    if time.time() < deadline - 120:
        d4 = min(deadline - 120, time.time() + 130)
        hc = best_h + rng.randn(len(best_h)) * 0.03
        hc = project_sum(hc, len(best_h) / 2.0)
        hn, c4 = nesterov_opt(hc, len(best_h), d4, k=80, lr0=0.010, soft_beta=50.0)
        if c4 < best_c5:
            best_c5, best_h = c4, hn.copy()
    # Phase 5: Final binary polish
    if time.time() < deadline - 8:
        d5 = min(deadline - 3, time.time() + 120)
        h5, _ = vec_binary_search(best_h, len(best_h), d5, rng, max_sweeps=1500, sample_n=160, top_m=22, use_2swap=True, n_2swap=30)
        c5 = compute_c5(h5, len(best_h))
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
