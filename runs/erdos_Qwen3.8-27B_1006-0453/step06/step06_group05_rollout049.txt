        c3 = compute_c5(h3, N0)

            best_c5, best_h = c3, h3.copy()

    # Phase 4: Try nearby N values
    for Ntry in [382, 378, 384, 376]:
        if time.time() > deadline - 80:
            break
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Ntry, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = project_binary(h_interp, Ntry, h_interp)
        if time.time() > deadline - 60:
            break
        sub_dl = min(deadline - 50, time.time() + 50)
        h_rt, _ = bin_search_opt(h_bt, Ntry, sub_dl, max_sweeps=2000, tabu_len=20)
        c5_rt = compute_c5(h_rt, Ntry)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 5: Another round of binary search on best
    if time.time() < deadline - 40:
        d5 = min(deadline - 20, time.time() + 40)
        h5, _ = bin_search_opt(best_h, len(best_h), d5, max_sweeps=5000, tabu_len=10)
        c5 = compute_c5(h5, len(best_h))
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Multi-restart on best N
    if time.time() < deadline - 30:
        N_cur = len(best_h)
        for attempt in range(3):
            if time.time() > deadline - 10:
                break
            hp = best_h.copy()
            n_flip = np.random.randint(2, 6)
            oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            for _ in range(n_flip):
                if len(oi) < 1 or len(zi) < 1:
                    break
                p = int(np.random.randint(len(oi)))
                q = int(np.random.randint(len(zi)))
                hp[oi[p]], hp[zi[q]] = 0, 1
                oi = np.where(hp == 1)[0]
                zi = np.where(hp == 0)[0]
            sub_dl = min(deadline - 8, time.time() + 12)
            h_pt, _ = bin_search_opt(hp, N_cur, sub_dl, max_sweeps=800, tabu_len=15)
            c5_pt = compute_c5(h_pt, N_cur)
            if c5_pt < best_c5:
                best_c5, best_h = c5_pt, h_pt.copy()

    # Final verify
    final_c5 = compute_c5(best_h, len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
