    # w has length 2N-1, w[j] is the weight for correlation entry j
    # s_j = j - (N-1)

    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1).astype(np.intp)
    pp_c = np.clip(pp, 0, N-1).astype(np.intp)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0-hm)*vm.astype(np.float64) + hp*vp.astype(np.float64)

def binary_search(h, N, deadline, max_sweeps=2000, anneal=0.0):
    h = h.astype(np.float64).copy()
    nh = N // 2
    h = (h > 0.5).astype(np.float64)
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            oi = np.where(h==1)[0]
            h[oi[n1-nh:]] = 0
        else:
            zi = np.where(h==0)[0]
            h[zi[:nh-n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    h_best = h.copy()
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            h_best = h.copy()
        stall = 0 if P < best_c else stall + 1
        if stall > (50 if anneal==0 else 25):
            if anneal == 0 or np.random.random() > anneal:
                break
            oi, zi = np.where(h==1)[0], np.where(h==0)[0]
            h[oi[np.random.randint(len(oi))]] = 0
            h[zi[np.random.randint(len(zi))]] = 1
            stall = 0
            continue
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        D = compute_D(h, N)
        bd, bpq = 0.0, None
        for p in ones:
            base = c - D[p]
            cand = base[None,:] - D[zeros,:]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:

        if bpq is not None and bd < -1e-13:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal <= 0:
            break
    return h_best, best_c

def smooth_subgrad(h, N, deadline, lr0=0.05, alpha0=10.0, alpha_max=80.0):

    target = N / 2.0
    dx = 2.0 / N

    mom = np.zeros(N)
    beta = 0.85
    n_corr = 2*N - 1
    shifts = np.arange(n_corr) - (N-1)

            break
        c = fft_corr(h)
        P = float(c.max()) * dx
        if P < best_c5:
            best_c5 = P
            best_h = h.copy()
        
        alpha = min(alpha_max, alpha0 * math.exp(0.005*t))
        w = np.exp(c * dx * alpha - c.max() * dx * alpha)
        w /= w.sum()
        
        # Gradient: O(N*(2N-1)) vectorized
        pos = np.arange(N)[:, None]
        sv = shifts[None, :]
        pp = pos + sv
        pm = pos - sv
        vp = (pp >= 0) & (pp < N)
        vm = (pm >= 0) & (pm < N)
        pp_c = np.clip(pp, 0, N-1).astype(np.intp)
        pm_c = np.clip(pm, 0, N-1).astype(np.intp)
        term1 = (w[None,:] * (1-h[pp_c]) * vp).sum(axis=1)
        term2 = (w[None,:] * h[pm_c] * vm).sum(axis=1)
        grad = (term1 - term2) * dx
        
        lr = lr0 / (1.0 + 0.005*t)
        mom = beta * mom + (1-beta) * grad
        h = project_sum(h - lr * mom, target)
        
        # Periodic check on binary
        if t % 300 == 0:
            hb = (h > 0.5).astype(np.float64)
            n1 = int(hb.sum())

                    zi = np.where(hb==0)[0]
                    hb[zi[:N//2-n1]] = 1
            c5b = float(np.max(fft_corr(hb))) * dx
            if c5b < best_c5:

def resample_binary(h_old, N_old, N_new, np_random):
    """Resample a binary solution to a new resolution."""
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    # Use step interpolation
    h_interp = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    h_new = (h_interp > 0.5).astype(np.float64)
    nh = N_new // 2
    n1 = int(h_new.sum())
    if n1 != nh:
        if n1 > nh:
            oi = np.where(h_new==1)[0]
            order = oi[np.argsort(h_interp[oi])]
            h_new[order[-(n1-nh):]] = 0
        else:
            zi = np.where(h_new==0)[0]
            order = zi[np.argsort(-h_interp[zi])]
            h_new[order[:nh-n1]] = 1
    return h_new

def interpolate_continuous(h_old, N_old, N_new):
    """Interpolate a continuous solution to new resolution."""
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return h_interp

                h[oi[np.random.randint(len(oi))]] = 0
                h[zi[np.random.randint(len(zi))]] = 1
                stall = 0
                continue
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if sample_n > 0 and len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        D = compute_D(h, N)
        bd, bpq = 0.0, None
        for p in p_list:
            base = c - D[p]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-13:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal <= 0:
            break
    return h_best, best_c

def smooth_opt(h, N, deadline, lr0=0.04, alpha_growth=0.004, alpha_max=60.0):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = float(np.max(fft_corr(h))) * dx
    mom = np.zeros(N)
    beta = 0.82
    for t in range(50000):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(np.max(c)) * dx
        if P < best_c5:
            best_c5 = P
            best_h = h.copy()
        alpha = min(alpha_max, 5.0 * math.exp(alpha_growth * t))
        cs = c * dx * alpha
        w = np.exp(cs - cs.max())
        w /= w.sum()
        grad = compute_grad(h, w, N) * dx
        lr = lr0 / (1.0 + 0.003 * t)
        mom = beta * mom + (1 - beta) * grad
        h = project_sum(h - lr * mom, target)
        if t % 200 == 0:
            hb = fix_count(h, N // 2)
            c5b = float(np.max(fft_corr(hb))) * dx
            if c5b < best_c5:
                best_c5 = c5b
                best_h = hb.copy()
    return best_h, best_c5

def resample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_i = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    h_new = (h_i > 0.5).astype(np.float64)
    nh = N_new // 2
    n1 = int(h_new.sum())
    if n1 > nh:
        oi = np.where(h_new == 1)[0]
        h_new[oi[np.argsort(h_i[oi])[:n1 - nh]]] = 0
    elif n1 < nh:
        zi = np.where(h_new == 0)[0]
        h_new[zi[np.argsort(-h_i[zi])[:nh - n1]]] = 1
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=np.float64)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    best_N = N0

    # Phase 1: Binary search n=380
    d1 = min(deadline - 650, t0 + 280)
    h1, _ = binary_search(h0, N0, d1, max_sweeps=5000, anneal=0.0)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h, best_N = c1, h1.copy(), N0

    # Phase 2: Smooth optimization n=380
    if time.time() < deadline - 400:
        d2 = min(deadline - 350, time.time() + 220)
        h2 = best_h.copy()
        if best_h.min() >= 0 and best_h.max() <= 1 and (best_h.max() - best_h.min()) < 0.1:
            h2 = h2 + np.random.randn(N0) * 0.04
            h2 = project_sum(h2, N0 / 2.0)
        h2s, c2 = smooth_opt(h2, N0, d2, lr0=0.035, alpha_growth=0.003, alpha_max=50.0)
        if c2 < best_c5:
            best_c5, best_h, best_N = c2, h2s.copy(), N0

    # Phase 3: Binary polish after smooth
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 130)
        h3 = fix_count(best_h, best_N // 2)
        h3s, _ = binary_search(h3, best_N, d3, max_sweeps=3000, anneal=0.0)
        c3 = compute_c5(h3s, best_N)
        if c3 < best_c5:
            best_c5, best_h, best_N = c3, h3s.copy(), best_N

    # Phase 4: Multi-resolution n=500
    N4 = 500
    if time.time() < deadline - 200:
        d4 = min(deadline - 150, time.time() + 120)
        h4 = resample(best_h, best_N, N4)
        h4s, _ = binary_search(h4, N4, d4, max_sweeps=2000, anneal=0.0)
        c4 = compute_c5(h4s, N4)
        if c4 < best_c5:
            best_c5, best_h, best_N = c4, h4s.copy(), N4

    # Phase 5: Smooth at n=500
    if best_N == 500 and time.time() < deadline - 130:
        d5 = min(deadline - 90, time.time() + 90)
        h5 = best_h + np.random.randn(500) * 0.025
        h5 = project_sum(h5, 250.0)
        h5s, c5v = smooth_opt(h5, 500, d5, lr0=0.03, alpha_growth=0.003, alpha_max=45.0)
        if c5v < best_c5:
            best_c5, best_h, best_N = c5v, h5s.copy(), 500

    # Phase 6: Binary polish
    if time.time() < deadline - 60:
        d6 = min(deadline - 35, time.time() + 80)
        h6 = fix_count(best_h, best_N // 2)
        h6s, _ = binary_search(h6, best_N, d6, max_sweeps=2000, anneal=0.06)
        c6 = compute_c5(h6s, best_N)
        if c6 < best_c5:
            best_c5, best_h, best_N = c6, h6s.copy(), best_N

    # Phase 7: Try n=420
    N7 = 420
    if time.time() < deadline - 30 and N7 != best_N:
        d7 = min(deadline - 10, time.time() + 35)
        h7 = resample(best_h, best_N, N7)
        h7s, _ = binary_search(h7, N7, d7, max_sweeps=800, anneal=0.0)
        c7 = compute_c5(h7s, N7)
        if c7 < best_c5:
            best_c5, best_h, best_N = c7, h7s.copy(), N7

    # Phase 8: Final annealed polish
    if time.time() < deadline - 5:
        d8 = deadline - 1
        h8 = fix_count(best_h, best_N // 2)
        h8s, _ = binary_search(h8, best_N, d8, max_sweeps=500, anneal=0.1)
        c8 = compute_c5(h8s, best_N)
        if c8 < best_c5:
            best_c5, best_h = c8, h8s.copy()

    # Final: ensure valid output
    final_h = fix_count(best_h, best_N // 2)
    final_c5 = compute_c5(final_h, best_N)
    return final_h, final_c5, best_N
