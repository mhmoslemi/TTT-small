```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

        else:
            hi = mid
```

```python
def swap_delta(h, p, q, N):
    """Compute change in correlation array when flipping h[p] 1->0 and h[q] 0->1"""
    delta = np.zeros(2*N-1)
    center = N - 1
    for L in range(-N+1, N):
        if p == q:
            break
        if L > 0:
            d = 0.0
            if p <= N-1-L:
                d -= (1-h[p+L])
            if p >= L:
                d += h[p-L]
            if q <= N-1-L:
                d += (1-h[q+L])
            if q >= L:
                d -= h[q-L]
            delta[L+center] += d
        elif L < 0:
            M = -L
            d = 0.0
            if p >= M:
                d -= (1-h[p-M])
            if p+M < N:
                d += h[p+M]
            if q >= M:
                d += (1-h[q-M])
            if q+M < N:
                d -= h[q+M]
            delta[L+center] += d
        # L=0: no change for binary
    return delta
```

But this is O(N) per swap which is fine. However, doing this in a Python loop over L is slow. Let me vectorize:

Actually for a single swap (p, q), the delta is nonzero only at specific lags:
- For L > 0: delta at L depends on h[p+L], h[p-L], h[q+L], h[q-L]
- These are specific values of h

So delta[L] for L > 0:
= -(1-h[p+L])*[p+L<N and p<N-L] + h[p-L]*[p-L>=0] + (1-h[q+L])*[q+L<N and q<N-L] - h[q-L]*[q-L>=0]

Let me vectorize this over L:
L ranges from 1 to N-1

For p (flipping 1→0):
- term1: -(1-h[p+1:p+1+(N-1-p)]) for L where p+L < N, i.e., L < N-p, so L in [1, N-p-1]
- term2: +h[max(0,p-(N-1)):p] for L where p-L >= 0, i.e., L <= p, so L in [1, p]

For q (flipping 0→1):
- term3: +(1-h[q+1:q+1+(N-1-q)]) for L in [1, N-q-1]
- term4: -h[max(0,q-(N-1)):q] for L in [1, q]

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -4.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgrad_corr(h, N, top_k=80):
    corr = fft_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(N)
    center = N - 1
    for ai in idx:
        L = int(ai - center)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[:N-L] += (1.0 - h[L:])
            g[L:] -= h[:N-L]
        else:
            M = -L
            g[M:] += (1.0 - h[:N-M])
            g[:N-M] -= h[M:]
    g /= top_k
    return g, float(np.max(corr))

def nesterov_opt(h, N, deadline, top_k=80, lr0=0.014, beta0=0.88):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        g, pk = subgrad_corr(y, N, top_k)
        h_new = project_01_sum(y - lr * g, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 5 == 0:
            val = float(np.max(fft_corr(h)))
            if val < best_val - 1e-9:
                best_val = val
                best_h = h.copy()
        if it % 300 == 0:
            lr *= 0.996
            beta = min(0.96, beta0 + it * 0.00004)
            if lr < 0.001:
                break
    return best_h, best_val * dx

def swap_delta_vec(h, p, q, N):
    L = np.arange(1, N)
    delta = np.zeros(2 * N - 1)
    center = N - 1
    for sign in [1, -1]:
        idx = L if sign == 1 else -L
        h_pl = np.where(p + idx < N, h[np.clip(p + idx, 0, N-1)], 0.0)
        h_pL = np.where(p - idx >= 0, h[np.clip(p - idx, 0, N-1)], 0.0)
        h_ql = np.where(q + idx < N, h[np.clip(q + idx, 0, N-1)], 0.0)
        h_qL = np.where(q - idx >= 0, h[np.clip(q - idx, 0, N-1)], 0.0)
        valid_pl = (p + idx >= 0) & (p + idx < N)
        valid_pL = (p - idx >= 0) & (p - idx < N)
        valid_ql = (q + idx >= 0) & (q + idx < N)
        valid_qL = (q - idx >= 0) & (q - idx < N)
        if sign == 1:
            dp = valid_pl * (-(1 - h_pl)) + valid_pL * h_pL
            dq = valid_ql * (1 - h_ql) + valid_qL * (-h_qL)
        else:
            M = idx
            h_pM = np.where(p + M < N, h[p + M], 0.0)
            h_pM2 = np.where(p - M >= 0, h[p - M], 0.0)
            h_qM = np.where(q + M < N, h[q + M], 0.0)
            h_qM2 = np.where(q - M >= 0, h[q - M], 0.0)
            dp = (p + M >= N) * 0 + (p - M >= 0) * (-(1 - h_pM2)) + (p + M < N) * h_pM
            # Fix: for L=-M, p contributes -(1-h[p-M])*[p>=M] + h[p+M]*[p+M<N]
            dp = (p >= M) * (-(1 - h_pM2)) + (p + M < N) * h_pM
            dq = (q >= M) * (1 - h_qM2) + (q + M < N) * (-h_qM)
        delta_val = dp + dq
        positions = idx + center
        delta[positions] += delta_val
    return delta

def binary_hill_climb(h, N, deadline, top_k=50, trials=40, max_iters=2000):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        pk = float(np.max(corr))
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        improved = False
        best_delta = 0
        best_pq = None
        ns = min(len(ones), 120)
        nzs = min(len(zeros), 120)
        os_ = np.random.choice(ones, ns, replace=False) if ns < len(ones) else ones
        zs = np.random.choice(zeros, nzs, replace=False) if nzs < len(zeros) else zeros
        for trial in range(trials):
            p = os_[np.random.randint(ns)]
            q = zs[np.random.randint(nzs)]
            delta = swap_delta_vec(h, int(p), int(q), N)
            new_corr = corr + delta
            new_pk = float(np.max(new_corr))
            d = new_pk - pk
            if d < best_delta:
                best_delta = d
                best_pq = (int(p), int(q), delta)
        if best_pq is not None and best_delta < -1e-12:
            h[best_pq[0]] = 0
            h[best_pq[1]] = 1
            corr = corr + best_pq[2]
            improved = True
        elif stall > 100 and np.random.random() < 0.1:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p] = 0
            h[q] = 1
            corr = fft_corr(h)
            stall //= 2
    corr = fft_corr(best_h)
    best_c = float(np.max(corr))
    return best_h, best_c
```
