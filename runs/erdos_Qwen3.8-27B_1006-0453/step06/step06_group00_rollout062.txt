import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def corr_lags(h, L_vals):
    """Compute correlation only at specific lags."""
    N = len(h)
    # Use full FFT and index into it
    full = fft_corr(h)
    center = N - 1
    return [full[int(l) + center] for l in L_vals]

def compute_C_mat(h, N):
    """C[i, j] = change in max-corr if we set h[i]=1, h[j]=0 simultaneously (swap i,j where h[i]=1, h[j]=0)"""
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def binary_il_full(h, N, deadline, max_iters=5000, block_sizes=(1,), restart_every=200):
    """Binary IL search with full neighborhood evaluation."""
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.round(h.sum()))
    if n1 != target:
        if n1 > target:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - target, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, target - n1, replace=False)] = 1
    
    best_c = np.inf
    best_h = h.copy()
    stall = 0
    max_stall = restart_every
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= max_stall:
            # Perturb: random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p], h[q] = 0.0, 1.0
            stall = 0
            continue
        
        C = compute_C_mat(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Full neighborhood: try all (one_pos, zero_pos) pairs
        # new_corr_at_lag = corr + C[one_pos, lag] - C[zero_pos, lag]
        # new_max = max_lag (corr[lag] + C[one_pos, lag] - C[zero_pos, lag])
        # We want to minimize new_max over all pairs
        
        # This is O(N^3) naively. Let's be smarter.
        # For each pair (o, z), the new correlation is corr + C[o] - C[z]
        # We want min_{o,z} max_lag (corr[lag] + C[o,lag] - C[z,lag])
        
        # Vectorized: new_corr[o,z,lag] = corr[lag] + C[o,lag] - C[z,lag]
        # new_max[o,z] = max_lag new_corr[o,z,lag]
        # This is O(N^2 * L) = O(N^3) which for N=380, L=759 is 380^2 * 759 ≈ 110M. Too slow per iteration.
        
        # Better: sample approach but with larger sample
        ns = min(len(ones), 100)
        nz = min(len(zeros), 100)
        os_ = np.random.choice(ones, ns, replace=False) if ns < len(ones) else ones
        zs = np.random.choice(zeros, nz, replace=False) if nz < len(zeros) else zeros
        
        Co = C[os_]  # (ns, L)
        Cz = C[zs]   # (nz, L)
        new = corr[None, None, :] + Co[None, :, :] - Cz[:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        
        if mv < P - 1e-12:
            h[os_[mi[1]]] = 0
            h[zs_[mi[0]]] = 1
        elif mv < P + 0.5 and np.random.random() < 0.1:
            # Accept slightly worse with low probability (simulated annealing)
            h[os_[mi[1]]] = 0
            h[zs_[mi[0]]] = 1
        else:
            # Try block swap
            if it % 5 == 0:
                for bl in block_sizes:
                    if bl > 1:
                        found = False
                        for _ in range(10):
                            # Find a random block of 1s and block of 0s of length bl
                            o_start = int(np.random.randint(0, N - bl + 1))
                            z_start = int(np.random.randint(0, N - bl + 1))
                            if np.all(h[o_start:o_start+bl] == 1) and np.all(h[z_start:z_start+bl] == 0):
                                new_h = h.copy()
                                new_h[o_start:o_start+bl] = 0
                                new_h[z_start:z_start+bl] = 1
                                new_c = compute_c5(new_h)
                                if new_c < best_c:
                                    h = new_h
                                    found = True

                        if found:

            if not (mv < P - 1e-12):

                    # Force a random move to escape
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p], h[q] = 0.0, 1.0
    
    return best_h, best_c

def subgradient_opt(h, N, deadline, k_top=30, max_iters=2000):
    """Subgradient method on continuous relaxation with projection."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    
    # Momentum
    v = np.zeros(N)
    beta = 0.9
    lr = 0.05
    it = 0
    
    while time.time() < deadline and it < max_iters:
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        # Find active lags (top k)
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        center = N - 1
        
        # Compute subgradient
        grad = np.zeros(N)
        for ai in top_idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - h[M:])
                    grad[M:] -= h[:N - M]
        grad /= k_top
        
        # Update with momentum
        v = beta * v - lr * grad
        h_new = h + v
        
        # Project
        h_new = project_01(h_new, target)
        
        # Check for stalling
        if np.allclose(h_new, h, atol=1e-10):
            lr *= 0.95
            beta *= 0.99
        if it % 100 == 0 and it > 0:
            lr *= 0.97
            beta = min(0.95, beta + 0.001)
        
        h = h_new
        it += 1
    
    return best_h, best_pk * dx
