import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, target):
    h = h.copy()
    s = np.round(h).astype(int)
    s = np.clip(s, 0, 1)
    cnt = s.sum()
    if cnt > target:
        ones = np.where(s == 1)[0]
        order = np.argsort(h[ones])[:cnt - target]
        s[ones[order]] = 0
    elif cnt < target:
        zeros = np.where(s == 0)[0]
        order = np.argsort(-h[zeros])[:target - cnt]
        s[zeros[order]] = 1
    return s.astype(float)

    h = h.copy()

    if cnt > target:
        idx = np.where(s == 1)[0]
        order = np.argsort(h[idx])[:cnt - target]
        s[idx[order]] = 0
    elif cnt < target:
        idx = np.where(s == 0)[0]
        order = np.argsort(-h[idx])[:target - cnt]
        s[idx[order]] = 1
    return s.astype(float)

def sa_search(h_bin, N, deadline, T0=0.09, Tmin=0.003):
    h = h_bin.copy()
    best_c = float(fft_corr(h).max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    no_imp = 0
    while time.time() < deadline and no_imp < 6000:
        if np.random.random() < 0.12 and len(ones) >= 3 and len(zeros) >= 3:
            bc = best_c + 1.0
            bp, bq = int(ones[0]), int(zeros[0])
            for _ in range(4):
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p] = 0.0; h[q] = 1.0
                nc = float(np.max(fft_corr(h)))
                h[p] = 1.0; h[q] = 0.0
                if nc < bc:
                    bc = nc; bp, bq = p, q
            p, q = bp, bq
            h[p] = 0.0; h[q] = 1.0
            new_c = bc
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p] = 0.0; h[q] = 1.0
            new_c = float(np.max(fft_corr(h)))
        d = new_c - best_c
        if d < -1e-14 or np.random.random() < np.exp(-d / max(T, 1e-10)):
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[p] = 1.0; h[q] = 0.0
            no_imp += 1
        T = max(Tmin, T * 0.99998)
    return best_h, best_c

def adam_cont(h_init, N, deadline, k=25, lr0=0.02):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    lr = lr0
    n_it = 0
    while time.time() < deadline:
        n_it += 1
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        top = np.sort(top)
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * h
            elif L > 0:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            else:
                M = -L
                grad[:N - M] += 1.0 - h[M:]
                grad[M:] -= h[:N - M]
        grad /= float(len(top))
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * (grad * grad)
        m1h = m1 / (1 - b1**n_it)
        m2h = m2 / (1 - b2**n_it)
        h_new = project_sum(h - lr * m1h / (np.sqrt(m2h) + eps), target)
        h = h_new
        if n_it % 100 == 0:
            lr *= 0.985
    return best_h, best_pk * (2.0 / N)

def nesterov_cont(h_init, N, deadline, k=18, lr0=0.014):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    n_it = 0
    while time.time() < deadline:
        n_it += 1
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        top = np.sort(top)
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += 1.0 - y[M:]
                grad[M:] -= y[:N - M]
        grad /= float(len(top))
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if n_it % 80 == 0:
            lr *= 0.99
            beta = min(0.94, 0.80 + n_it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary SA
    d1 = min(deadline - 400, t0 + 180)
    h_b = to_binary(h0, N0 // 2)
    h1, c1 = sa_search(h_b, N0, d1, T0=0.11, Tmin=0.004)
    c1s = c1 * (2.0 / N0)
    if c1s < best_c5:
        best_c5, best_h = c1s, h1.copy()

    # Phase 2: Adam continuous (primary refiner)
    if time.time() < deadline - 250:
        d2 = min(deadline - 180, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.02
        hc = project_sum(hc, N0 / 2.0)
        ha, ca = adam_cont(hc, N0, d2, k=30, lr0=0.018)
        if ca < best_c5:
            best_c5, best_h = ca, ha.copy()

    # Phase 3: Nesterov continuous (complementary)
    if time.time() < deadline - 200:
        d3 = min(deadline - 140, time.time() + 80)
        hc2 = best_h + np.random.randn(N0) * 0.015
        hc2 = project_sum(hc2, N0 / 2.0)
        hn, cn = nesterov_cont(hc2, N0, d3, k=20, lr0=0.012)
        if cn < best_c5:
            best_c5, best_h = cn, hn.copy()

    # Phase 4: Binary SA from improved (with perturbation for diversity)
    if time.time() < deadline - 150:
        d4 = min(deadline - 80, time.time() + 180)
        h4b = to_binary(best_h, len(best_h) // 2)
        ones4 = np.where(h4b == 1)[0]
        zeros4 = np.where(h4b == 0)[0]
        for _ in range(3):
            pp = int(ones4[np.random.randint(len(ones4))])
            qq = int(zeros4[np.random.randint(len(zeros4))])
            h4b[pp], h4b[qq] = 0.0, 1.0
        h4, c4 = sa_search(h4b, len(best_h), d4, T0=0.09, Tmin=0.003)
        c4s = c4 * (2.0 / len(best_h))
        if c4s < best_c5:
            best_c5, best_h = c4s, h4.copy()

    # Phase 5: Fine Adam on best
    if time.time() < deadline - 100:
        d5 = min(deadline - 60, time.time() + 80)
        hc5 = best_h + np.random.randn(len(best_h)) * 0.01
        hc5 = project_sum(hc5, len(best_h) / 2.0)
        h5, c5v = adam_cont(hc5, len(best_h), d5, k=40, lr0=0.008)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()

    # Phase 6: Other resolutions
    for N_try in [360, 400, 440]:
        if time.time() > deadline - 50:
            break
        if N_try == len(best_h) or N_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_b6 = to_binary(h_i, N_try // 2)
        sub_d = min(deadline - 25, time.time() + 40)
        h6, c6 = sa_search(h_b6, N_try, sub_d, T0=0.08, Tmin=0.003)
        c6s = c6 * (2.0 / N_try)
        if c6s < best_c5:
            best_c5 = c6s
            best_h = h6.copy()
            # Quick Adam refinement at new resolution
            if time.time() < deadline - 20:
                d62 = min(deadline - 15, time.time() + 30)
                hc6 = h6 + np.random.randn(N_try) * 0.01
                hc6 = project_sum(hc6, N_try / 2.0)
                h6a, c6a = adam_cont(hc6, N_try, d62, k=25, lr0=0.01)
                if c6a < best_c5:
                    best_c5, best_h = c6a, h6a.copy()

    # Phase 7: Final micro-Adam
    if time.time() < deadline - 3:
        d7 = deadline - 2
        hF = best_h + np.random.randn(len(best_h)) * 0.005
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cF = adam_cont(hF, len(best_h), d7, k=50, lr0=0.005)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
