import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    full = np.fft.irfft(H * G, L)
    return full

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = h.astype(float)
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target_ones:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target_ones:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target_ones, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target_ones - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D_fixed(h, N):
    """Compute D matrix: D[p, s] = change in correlation at shift s when h[p] flips 1->0."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    # D[p, s] = -(1-h[p-shift])*[0<=p-shift<N] + h[p+shift]*[0<=p+shift<N]
    # FIX: at shift=0, the correct value for binary h is 0 (not 2h[p]-1)
    # Build using broadcasting
    pos = np.arange(N)
    D = np.zeros((N, 2*N-1))
    for shift in shifts:
        pm = pos - shift  # p - shift
        pp = pos + shift  # p + shift
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N-1)
        pp_c = np.clip(pp, 0, N-1)
        hm = np.where(vm, h[pm_c], 0.0)

def binary_search_improved(h, N, deadline, rng, max_sweeps=500, sample_frac=1.0, 
                           sa_temp=0.0, sa_cool=0.995, perturb_every=50):
    h = h.astype(float).copy()
    target = N // 2
    if int(h.sum()) != target:

    stall = 0
    temp = sa_temp
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        zeros = np.where(h == 0)[0]

        else:
            p_list = ones

        best_pq = None
        second_best = P
        second_pq = None

                second_pq = best_pq
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is not None:
            delta = best_new_P - P
            if delta < 0 or (sa_temp > 0 and rng.random() < np.exp(-delta / (sa_temp * (2.0/N)))):
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
            elif second_pq is not None and sa_temp > 0:
                delta2 = second_best - P
                if delta2 < 0 or rng.random() < np.exp(-delta2 / (sa_temp * (2.0/N))):
                    h[second_pq[0]] = 0.0
                    h[second_pq[1]] = 1.0

            break
        
        temp *= sa_cool
    
    return best_h, best_c * (2.0 / N)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = h.astype(float)
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target_ones:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target_ones:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target_ones, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target_ones - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = -(1.0 - hm) * vm + hp * vp
    D[:, N-1] = 0.0
    return D

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
