def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binary_search_improved(h, N, deadline, max_stall=50):
    h = h.astype(float).copy()
    c = fft_corr(h)
    stall = 0
    iters = 0
    while time.time() < deadline:
        max_c = float(np.max(c))
        # For binary h, max_c should be integer (or very close)
        max_c_int = int(round(max_c))
        relevant = np.where(c >= max_c_int - 1)[0]
        relevant_c = c[relevant].astype(float)
        n_rel = len(relevant)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_new_max = max_c_int + 1
        best_pq = None
        lat_pq = None
        
        for p in ones:
            mp = relevant - p
            vp = (mp >= 0) & (mp < N)
            dp = np.zeros(n_rel)
            dp[vp] = 1.0 - h[mp[vp]]
            
            mq = relevant[:, None] - zeros[None, :]
            vq = (mq >= 0) & (mq < N)
            hmq = h[np.clip(mq, 0, N - 1)]
            dq = np.where(vq, 1.0 - hmq, 0.0)
            
            nc = relevant_c[:, None] + dq - dp[:, None]
            nm = nc.max(axis=0).astype(int)
            
            mn = int(nm.min())
            if mn < best_new_max:
                best_new_max = mn
                best_pq = (int(p), int(zeros[np.argmin(nm)]))
            elif mn == max_c_int and lat_pq is None:
                lat_pq = (int(p), int(zeros[np.argmin(nm)]))
        
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            stall = 0
        elif lat_pq is not None and iters > 0:
            h[lat_pq[0]] = 0.0
            h[lat_pq[1]] = 1.0
            stall += 1
        else:
            break
        
        if stall > max_stall:
            break
        c = fft_corr(h)
        iters += 1
    
    return h, compute_c5(h)

def nesterov(h, N, deadline, k=80, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    bh = h.copy()
    bpk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < bpk:
            bpk = pk
            bh = h.copy()
        tk = min(k, 2 * N - 1)
        top = np.argpartition(corr, -tk)[-tk:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai) - center
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= len(top)
        hn = project_sum(y - lr * grad, target)
        m = beta * (hn - h)
        h = hn
        it += 1
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.97, 0.9 + it * 0.0001)
    return bh, bpk * (2.0 / N)

def make_golden(N):
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    h = np.zeros(N)
    nh = N // 2
    pos = set()
    for n in range(N * 5):
        p = int(n * 0.5 * phi + n * 0.5 * phi * phi) % N  
        # Better: use Kronecker sequence
        p = int(n * phi) % N
        if p not in pos:
            h[p] = 1.0
            pos.add(p)
            if len(pos) >= nh:
                break
    if len(pos) < nh:
        rem = np.array([i for i in range(N) if i not in pos])
        np.random.shuffle(rem)
        h[rem[:nh - len(pos)]] = 1.0
    return h

def to_binary_fixed(h, N, seed=0):
    h_b = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h_b.sum())
    rng = np.random.RandomState(seed)
    if n1 > nh:
        idx = np.where(h_b == 1)[0]
        sel = rng.choice(idx, n1 - nh, replace=False)
        h_b[sel] = 0
    elif n1 < nh:
        idx = np.where(h_b == 0)[0]
        sel = rng.choice(idx, nh - n1, replace=False)
        h_b[sel] = 1
    return h_b

        # Also try golden start

        Nc = len(best_h)
        hc = best_h + np.random.randn(Nc) * 0.04
        hc = project_sum(hc, Nc / 2.0)
        d3 = min(deadline - 30, time.time() + 100)

        if cn < best_c5:

    # Phase 4: Binary cleanup
    if time.time() < deadline - 5:
        hb = to_binary_fixed(best_h, len(best_h))
        d4 = deadline - 2
        h4, c4 = binary_search_improved(hb, len(hb), d4)
        if c4 < best_c5:
            best_c5, best_h = c4, h4
    
    # Final: try Nesterov one more time on binary result

        hf, cf = nesterov(hf, Nf, deadline - 0.5, k=100, lr0=0.008)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
