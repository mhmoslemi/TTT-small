```python
    Returns the max of the new corr."""
```

```python
    """Try n_random random swaps, return best one."""

    zeros = np.where(h == 0)[0]

    idx_b = np.random.randint(n0, n_random)
```

```python
def eval_swap_max(h, corr, a, b):
    """Evaluate max corr after swapping h[a]:1→0, h[b]:0→1 without modifying h or corr."""
    N = len(h)
    # The change at lag n (in full correlate indexing 0..2N-2):
    # Δ(n) = -(1-h[a-(N-1)+n])·[0≤a-(N-1)+n≤N-1] 
    #        + h[a+(N-1)-n]·[0≤a+(N-1)-n≤N-1]
    #        + (1-h[b-(N-1)+n])·[0≤b-(N-1)+n≤N-1]
    #        - h[b+(N-1)-n]·[0≤b+(N-1)-n≤
    #
    # Effect: corr_new[n] = corr[n] + Δ(n)
    # We want max of corr_new.
    #
    # The affected n ranges:
    # R1: a-(N-1)+n ∈ [0,N-1] → n ∈ [N-1-a, 2N-2-a]
    # R2: a+(N-1)-n ∈ [0,N-1] → n ∈ [a, a+N-1]
    # R3: b-(N-1)+n ∈ [0,N-1] → n ∈ [N-1-b, 2N-2-b]
    # R4: b+(N-1)-n ∈ [0,N-1] → n ∈ [b, b+N-1]
    
    # Unaffected lags keep their value. The max of unaffected is:
    # We need the max of corr outside the union of R1,R2,R3,R4.
    # And the max of (corr + Δ) inside the union.
    
    # For simplicity, let's just compute the new max over the affected regions
    # and compare with the max over unaffected.
    
    # Create a "mask" of affected indices
    L = 2*N - 1
    affected = np.zeros(L, dtype=bool)
    r1_start, r1_end = N-1-a, 2*N-2-a
    r2_start, r2_end = a, a+N-1
    r3_start, r3_end = N-1-b, 2*N-2-b
    r4_start, r4_end = b, b+N-1
    affected[r1_start:r1_end+1] = True
    affected[r2_start:r2_end+1] = True
    affected[r3_start:r3_end+1] = True
    affected[r4_start:r4_end+1] = True
    
    # Max over unaffected
    max_unaffected = np.max(corr[~affected]) if np.any(~affected) else -np.inf
    
    # Max over affected: corr[n] + Δ(n)
    aff_idx = np.where(affected)[0]
    new_vals = corr[aff_idx].copy()
    
    # Apply Δ
    n1 = aff_idx - (N-1)  # this is the "shift" k = n - (N-1)
    # Actually let me just directly compute Δ for each affected n
    
    # R1 contribution: -(1-h[a-(N-1)+n]) for n in R1
    mask_r1 = (aff_idx >= r1_start) & (aff_idx <= r1_end)
    j = a - (N-1) + aff_idx[mask_r1]  # indices into h
    new_vals[mask_r1] -= (1.0 - h[j])
    
    # R2 contribution: +h[a+(N-1)-n] for n in R2
    mask_r2 = (aff_idx >= r2_start) & (aff_idx <= r2_end)
    j2 = a + (N-1) - aff_idx[mask_r2]
    new_vals[mask_r2] += h[j2]
    
    # R3 contribution: +(1-h[b-(N-1)+n]) for n in R3
    mask_r3 = (aff_idx >= r3_start) & (aff_idx <= r3_end)
    j3 = b - (N-1) + aff_idx[mask_r3]
    new_vals[mask_r3] += (1.0 - h[j3])
    
    # R4 contribution: -h[b+(N-1)-n] for n in R4
    mask_r4 = (aff_idx >= r4_start) & (aff_idx <= r4_end)
    j4 = b + (N-1) - aff_idx[mask_r4]
    new_vals[mask_r4] -= h[j4]
    
    max_affected = np.max(new_vals)
    return max(max_unaffected, max_affected)
```

```python
import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(H * np.conj(G), L)
    return np.concatenate([c[N-1::-1], c[2*N-2:N-1:-1]])

def c5_of(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_onto_box_sum(h, target):
    """Project h onto {0≤h≤1, sum(h)=target}."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s_mid = np.clip(h - mid, 0, 1).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lse_smooth(corr, tau):
    """Log-sum-exp of corr values, scaled by 1/tau."""
    c = corr / tau
    c_max = c.max()
    return c_max + np.log(np.exp(c - c_max).sum())

def grad_lse(corr, tau, N):
    """Gradient of lse_smooth w.r.t. h, given corr = correlate(h, 1-h, full)."""
    # d/dh[j] of sum_n exp(corr[n]/tau) / (tau * sum exp(corr[n]/tau))
    # corr[n] = sum_i h[i] * (1-h[i-(N-1)+n])
    # d corr[n] / d h[j] = (1-h[j-(N-1)+n]) * [0≤j-(N-1)+n≤N-1]
    #                      + h[j+(N-1)-n] * ... wait
    # Actually: corr[n] = sum_i h[i](1-h[i-(N-1)+n])
    # d corr[n]/d h[j] (treating h as a function):
    #   Term where i=j: (1-h[j-(N-1)+n]) if j-(N-1)+n in [0,N-1]
    #   Term where i-(N-1)+n = j, i=j+(N-1)-n: h[j+(N-1)-n] * d(1-h[j])/dh[j] = -h[j+(N-1)-n] if j+(N-1)-n in [0,N-1]
    
    # This gives a "convolution-like" gradient. For efficiency, use the weight vector.
    w = np.exp(corr / tau - corr.max() / tau)
    w /= w.sum()
    
    # grad[j] = sum_n w[n] * (d corr[n]/d h[j])
    # = sum_n w[n] * [(1-h[j-(N-1)+n])·[valid] - h[j+(N-1)-n]·[valid]]
    
    # This can be computed as a convolution:
    # First term: for each n, contributes w[n] to indices j where j = i and i-(N-1)+n in [0,N-1]
    #   i.e., j-(N-1)+n in [0,N-1] → j in [(N-1)-n, (N-1)-n+N-1] = [(N-1)-n, 2N-2-n]
    #   and contributes w[n]*(1-h[j-(N-1)+n]) at j
    # This is: grad1[j] = sum_{n: j-(N-1)+n in [0,N-1]} w[n] * (1-h[j-(N-1)+n])
    # = sum_{m in [0,N-1]: m=j-(N-1)+n, n=j-(N-1)-m, n in [0,2N-2]}
    #   w[j-(N-1)-m] * (1-h[m])
    # = sum_m w[j-N+1-m] * (1-h[m]) for valid m and valid n=j-N+1-m in [0,2N-2]
    
    # This is a cross-correlation of w with (1-h)!
    # grad1 = correlate(w, 1-h, "full") appropriately indexed...
    
    # Actually, it's simpler to just use FFT:
    # grad1[j] = sum_m (1-h[m]) * w[j-(N-1)-m] for valid j-m and valid w-index
    # Let n' = j-(N-1)-m. Then grad1[j] = sum_m (1-h[m]) * w'[j-m] where w'[n] corresponds to w at index n...
    
    # This is getting complicated. Let me just do it directly for N=380 (it's fast enough).
    
    grad = np.zeros(N)
    center = N - 1
    for n in range(2*N-1):
        if w[n] < 1e-15:
            continue
        # d corr[n] / d h[j] for all j:
        # (1-h[j-center+n]) for j-center+n in [0,N-1] → j in [center-n, center-n+N-1]
        # -h[j+center-n] for j+center-n in [0,N-1] → j in [n-center, n-1]
        lo1 = max(0, center - n)
        hi1 = min(N-1, center - n + N - 1)
        if lo1 <= hi1:
            m = np.arange(lo1, hi1+1) - center + n
            grad[lo1:hi1+1] += w[n] * (1.0 - h[m])
        
        lo2 = max(0, n - center)
        hi2 = min(N-1, n - 1)
        if lo2 <= hi2:
            m2 = np.arange(lo2, hi2+1) + center - n
            grad[lo2:hi2+1] -= w[n] * h[m2]
    
    return grad
```

Wait, this grad_lse function is O(N²) because of the loop over n. For N=380, that's 759 iterations, each doing O(N) work, so O(N²) ≈ 285K operations. That's fine!

But actually, I can compute this more efficiently using the correlation structure. The gradient of the correlation w.r.t. h at each point is itself a correlation-like quantity.

Let me think again. corr[n] = sum_i h[i](1-h[i-(N-1)+n]).

d corr[n] / d h[j] = (1-h[j-(N-1)+n])·[0≤j-(N-1)+n≤N-1] + h[j+(N-1)-n]·(-1)·[0≤j+(N-1)-n≤N-1]

Wait: d/dh[j] of sum_i h[i](1-h[i-(N-1)+n]):
- If i = j: d/dh[j] of h[j](1-h[j-(N-1)+n]) = (1-h[j-(N-1)+n]) (assuming j-(N-1)+n ≠ j, i.e., n≠N-1)
- If i-(N-1)+n = j: d/dh[j] of h[i](1-h[j]) = -h[i] = -h[j+(N-1)-n] (assuming i ≠ j, i.e., n≠N-1)

At n = N-1 (center): corr[N-1] = sum_i h[i](1-h[i]) = 0 for binary, or sum h[i]-h[i]² for continuous.
d/dh[j] of sum_i h[i](1-h[i]) = 1-2h[j].

So the gradient is:
grad[j] = sum_n w[n] * [(1-h[j-(N-1)+n])·[valid] - h[j+(N-1)-n]·[valid]] + w[N-1]*(1-2h[j])

The first sum is a cross-correlation! Let me define:
A = (1-h) (length N)
B = h (length N)
w (length 2N-1)

grad[j] = sum_n w[n] * A[j-(N-1)+n] (with valid index) - sum_n w[n] * B[j+(N-1)-n] (with valid index)

The first sum: for each n, A is accessed at index j-(N-1)+n as j varies. This is like sliding A over a range and weighting by w[n].
= sum_n w[n] * A[(j-(N-1)+n)] where (j-(N-1)+n) ∈ [0, N-1]

Let k = j-(N-1)+n, so n = j-(N-1)-k. The constraint k∈[0,N-1] and n∈[0,2N-2] means:
j-(N-1)-k ∈ [0, 2N-2] → k ∈ [j-(N-1)-(2N-2), j-(N-1)] = [j-N+1, j] ∩ [0, N-1]

So first_sum[j] = sum_{k∈[max(0,j-N+1), min(N-1,j)]} w[j-(N-1)-k] * A[k]

This is a convolution/correlation of w with A! Specifically:
first_sum = correlate(w, A, "full") indexed appropriately.

Actually, let me just use FFT for this:
first_sum[j] = sum_k A[k] * w[j-(N-1)-k] for valid indices.
Let m = j-(N-1)-k, so k = j-(N-1)-m.
first_sum[j] = sum_m A[j-(N-1)-m] * w[m] for j-(N-1)-m ∈ [0,N-1] and m ∈ [0,2N-2].

This is the cross-correlation of A and w, evaluated at position (N-1-j)... hmm, it's getting confusing with indices.

Let me just do it with a direct FFT:
F = rfft(A, L) where L = 2N-1 or larger
W = rfft(w, L)
corr_AW = irfft(F * conj(W), L)

Then first_sum[j] = corr_AW[(N-1-j) mod L]... I'll just compute it numerically and verify.

Actually, you know what, for N=380, the O(N²) direct computation takes 759*380 ≈ 288K operations per gradient evaluation. At ~10 billion ops/sec in numpy (conservative), that's 30μs. Very fast!

So let me just use the direct approach. The total cost per iteration is:
- Compute corr: O(N log N) via FFT ≈ 7K ops
- Compute weights w: O(N)
- Compute gradient: O(N²) ≈ 288K ops
- Update h: O(N)
- Project: O(N log N) for bisection

Total: ~300K ops per iteration. At 300K * 100ns (conservative) = 30μs per iteration. In 700s, we can do ~23M iterations. That's more than enough.

But wait, the gradient I computed is for the LSE-of-corr objective. Let me double-check the sign. We want to MINIMIZE C₅ = max_k corr[k]. The LSE approximation: f(h) = (1/τ) * log(sum_k exp(τ*corr[k])).

df/dh[j] = (1/τ) * sum_k exp(τ*corr[k]) * τ * (d corr[k]/d h[j]) / sum_k exp(τ*corr[k])
= sum_k w_k * (d corr[k]/d h[j])

where w_k = exp(τ*corr[k]) / sum exp(τ*corr[k]).

And d corr[k] / d h[j] as derived above.

To minimize f, we do h ← h - lr * grad, then project.

This is a standard approach. Let me implement it.

One issue: the LSE approximation is only close to the max when τ is large. But for large τ, the weights w are concentrated on the max, and the gradient is noisy. A good strategy is to start with small τ (smooth) and increase it (sharpen).

Let me now write the full algorithm:

```python
import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(H * np.conj(G), L)
    return np.concatenate([c[N-1::-1], c[2*N-2:N-1:-1]])

def project_onto_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo + hi), 0.0, 1.0)

def lse_grad(h, corr, tau):
    N = len(h)
    center = N - 1
    c_scaled = corr / tau
    c_max = c_scaled.max()
    w = np.exp(c_scaled - c_max)
    w_sum = w.sum()
    w /= w_sum
    
    grad = np.zeros(N)
    for n in range(2*N-1):
        wn = w[n]
        if wn < 1e-18:
            continue
        # First part: (1-h[j-center+n]) for j-center+n in [0,N-1]
        # j in [center-n, center-n+N-1]
        lo1 = max(0, center - n)
        hi1 = min(N-1, center - n + N - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1+1) - center + n
            grad[lo1:hi1+1] += wn * (1.0 - h[idx])
        # Second part: -h[j+center-n] for j+center-n in [0,N-1]
        # j in [n-center, N-1+center-n]... wait:
        # j+center-n in [0,N-1] → j in [n-center, N-1-center+n]... 
        # Actually: j+(center-n) in [0,N-1] → j in [-(center-n), N-1-(center-n)] = [n-center, N-1-center+n]
        lo2 = max(0, n - center)
        hi2 = min(N-1, N-1 - center + n)  # = min(N-1, n-1) since N-1-center = 0
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2+1) + center - n
            grad[lo2:hi2+1] -= wn * h[idx2]
    
    return grad
```
