    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def fix_sum_binary(h, N):
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        oi = np.where(h > 0.5)[0]
        zi = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            keep = np.argsort(-h[oi])[:nh]
            h[oi[keep]] = 1.0
        else:
            h[oi] = 1.0
            h[zi[:nh-n1]] = 1.0
    return h

    stall = 0
    temp = anneal

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if pk < best_c - 1e-12:

        else:

                    q = int(np.random.randint(len(zeros)))

                    zeros = np.where(h == 0)[0]

                stall = 0
            else:
                break
        
        D = compute_D(h, N)

        # For each p in ones: new_c for all q = (c - D[p,:])[None,:] + D[zeros,:]
        best_d = 0.0
        best_pq = None
        
        # If too many, sample

            mi = int(np.argmin(npk))
            d = float(npk[mi]) - pk
            if d < best_d:
                best_d = d
                best_pq = (int(p), int(zeros[mi]))

def cont_optimize(h0, N, deadline, npeaks=5, lr_init=0.3):
    h = h0.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c = float(fft_corr(h).max())
    t = 0
    lr = lr_init
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        
        D = compute_D(h, N)
        # Subgradient from top peaks
        top_idx = np.argpartition(-c, npeaks)[:npeaks]
        w = c[top_idx] - c[top_idx].min() + 0.1
        w /= w.sum()
        grad = (D[:, top_idx] * w[None, :]).sum(axis=1)
        
        # Adam update
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h_new = h + lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h_new, target)
        
        if t % 200 == 0:
            lr *= 0.95
            # Periodic binarization
            hb = (h > 0.5).astype(float)
            hb = fix_sum_binary(hb, N)
            h = 0.4 * h + 0.6 * hb
    
    return best_h, best_c

    # c: (2N-1,) correlation values

    w /= w.sum()
    return (D * w[None, :]).sum(axis=1)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)
