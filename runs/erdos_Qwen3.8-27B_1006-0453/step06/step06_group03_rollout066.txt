```python
def best_swap_for_lag(h, N, m, ones, zeros):
    """Find best (p, q) swap to reduce correlation at lag m."""
    # p should be in ones (h[p]=1), q in zeros (h[q]=0)
    # Δ_p(m) = -(1-h[p+m])*I[p+m<N] + h[p-m]*I[p-m>=0]
    # Δ_q(m) = (1-h[q+m])*I[q+m<N] - h[q-m]*I[q-m>=0]
    
    p_scores = np.zeros(len(ones))
    for i, p in enumerate(ones):
        s = 0.0
        if p + m < N:
            s -= (1.0 - h[p+m])
        if p - m >= 0:
            s += h[p-m]
        p_scores[i] = s
    
    q_scores = np.zeros(len(zeros))
    for i, q in enumerate(zeros):
        s = 0.0
        if q + m < N:
            s += (1.0 - h[q+m])
        if q - m >= 0:
            s -= h[q-m]
        q_scores[i] = s
    
    best_pi = np.argmin(p_scores)
    best_qi = np.argmin(q_scores)
    
    return ones[best_pi], zeros[best_qi], p_scores[best_pi] + q_scores[best_qi]
```

    #   At positions i where i+m=p, i.e., i=p-m: gain of h[p-m]*(1-0) - h[p-m]*(1-1) = h[p-m]
    # In the full correlation array (index = m + N - 1):
    #   The "p+m" part: for m = 0, 1, ..., N-1-p: d[N-1+m] -= (1-h[p+m])
    #   The "p-m" part: for m = 0, 1, ..., p: d[N-1-m] += h[p-m]  (note: m=0 gives d[N-1] += h[p])
    
    # Actually let me think of it differently.
    # c[k] = sum_i h[i] * (1-h[k-i+N-1 - (N-1)]) ... this is getting confusing.
    
    # Let me use a different indexing. The full correlation out has length 2N-1.
    # out[j] for j=0,...,2N-2 represents the correlation at lag (j - (N-1)).

    # p-m in [0,N): m in [p-N+1, p]

    return np.real(np.fft.ifft(H * G))

```python
def smooth_max_grad(h, N, tau=50.0):
    """Compute gradient of smooth max of correlation."""
    c = fft_corr(h)
    # Softmax weights
    tc = tau * c
    tc -= tc.max()  # numerical stability
    w = np.exp(tc)
    w /= w.sum()
    # Gradient: w @ dc/dh
    grad = np.zeros(N)
    center = N - 1
    for m_offset in range(-(N-1), N):
        wi = w[center + m_offset]
        if wi < 1e-10:
            continue
        m = m_offset
        # dc[m]/dh[i] = (1-h[i+m])*I[valid] - h[i-m]*I[valid]
        # For fixed m:
        #   i+m in [0,N): i in [max(0,-m), min(N, N-m))
        #   i-m in [0,N): i in [max(0,m), min(N, N+m))
        i_lo = max(0, -m)
        i_hi = min(N, N - m)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wi * (1.0 - h[i_lo + m:i_hi + m])
        i_lo2 = max(0, m)
        i_hi2 = min(N, N + m)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] -= wi * h[i_lo2 - m:i_hi2 - m]
    return grad
```

This is O(N²) per gradient computation (N iterations over m, each O(N)). For N=380: 380*380 = 144K ops. At maybe 1M ops/s, that's 0.14s per gradient. Over 150s, that's about 1000 iterations. Decent.

Actually, I can vectorize this more. For a fixed m, the gradient contribution is:
grad[i] += w[m] * (1-h[i+m]) - w[m] * h[i-m]

This is a shifted copy of (1-h) and h, weighted by w[m]. I can do this more efficiently by precomputing.

Actually, the key insight is that the gradient at position i is:
grad[i] = Σ_m w[m] * [(1-h[i+m])*I[0≤i+m<N] - h[i-m]*I[0≤i-m<N]]
= Σ_m w[m]*(1-h[i+m])*I[0≤i+m<N] - Σ_m w[m]*h[i-m]*I[0≤i-m<N]

Let j = i+m in the first sum: = Σ_j w[j-i]*(1-h[j])*I[0≤j<N] (where j-i is valid lag)
= convolution of w with (1-h), evaluated at i.

Second sum: let j = i-m: = Σ_j w[i-j]*h[j]*I[0≤j<N]
= convolution of w with h, evaluated at i.

So grad = conv(w, 1-h) - conv(w, h) at the appropriate indices!

This can be computed with FFT in O(N log N)!

```python

    c = fft_corr(h)
    tc = tau * c
    tc -= tc.max()
    w = np.exp(tc)
    w /= w.sum()
    # w is length 2N-1, centered at N-1
    # grad[i] = Σ_m w[m+center] * (1-h[i+m]) - Σ_m w[m+center] * h[i-m]
    # First term: for each i, sum over valid m of w[i+m+center]*(1-h[i+m])
    #   Let j = i+m: sum over j of w[j+center]*(1-h[j]) where j=i+m and m is valid
    #   m = j-i, valid when m in [-(N-1), N-1] and j in [0,N-1]
    #   So: first_term[i] = Σ_{j=0}^{N-1} w[j-i+N-1]*(1-h[j]) I[0≤j-i+N-1<2N-1]
    #   This is a cross-correlation of (1-h) with w!
    
    # Actually let me just do it with the loop for clarity, N=380 is small enough
    grad = np.zeros(N)
    center = N - 1
    for i in range(N):
        s1 = 0.0
        s2 = 0.0
        # m ranges: i+m in [0,N) → m in [-i, N-1-i]
        # and i-m in [0,N) → m in [i-N+1, i]
        for m in range(-i, N - i):
            s1 += w[center + m] * (1.0 - h[i + m])
        for m in range(i - N + 1, i + 1):
            s2 += w[center + m] * h[i - m]
        grad[i] = s1 - s2
    return grad
```

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def delta_vec(h, N, p, q):
    d = np.zeros(2 * N - 1)
    c = N - 1
    m1 = np.arange(-p, N - p)
    d[c + m1] -= (1.0 - h[p + m1])
    m2 = np.arange(p - N + 1, p + 1)
    d[c + m2] += h[p - m2]
    m3 = np.arange(-q, N - q)
    d[c + m3] += (1.0 - h[q + m3])
    m4 = np.arange(q - N + 1, q + 1)
    d[c + m4] -= h[q - m4]
    return d

def to_binary(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, N, tau):
    c = fft_corr(h)
    tc = tau * c
    tc -= tc.max()
    w = np.exp(tc)
    w /= w.sum()
    center = N - 1
    grad = np.zeros(N)
    for i in range(N):
        m1 = np.arange(-i, N - i)
        grad[i] += np.dot(w[center + m1], 1.0 - h[i + m1])
        m2 = np.arange(i - N + 1, i + 1)
        grad[i] -= np.dot(w[center + m2], h[i - m2])
    return grad

def targeted_search(h, N, deadline, K=25, perturb_n=5, stuck_tol=40):
    h = to_binary(h, N)
    c = fft_corr(h)
    best_P = float(c.max())
    stuck = 0
    while time.time() < deadline:
        top_k = np.argsort(c)[-K:]
        best_new_P = best_P
        best_pq = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for lag_idx in top_k:
            m = int(lag_idx - (N - 1))
            if m == 0:
                continue
            pm = ones + m
            pmv = (pm >= 0) & (pm < N)
            pmm = ones - m
            pmmv = (pmm >= 0) & (pmm < N)
            ps = np.zeros(len(ones))
            ps[pmv] -= (1.0 - h[pm[pmv]])
            ps[pmmv] += h[pmm[pmmv]]
            qm = zeros + m
            qmv = (qm >= 0) & (qm < N)
            qmm = zeros - m
            qmmv = (qmm >= 0) & (qmm < N)
            qs = np.zeros(len(zeros))
            qs[qmv] += (1.0 - h[qm[qmv]])
            qs[qmmv] -= h[qmm[qmmv]]
            pi = int(np.argmin(ps))
            qi = int(np.argmin(qs))
            p_cv = int(ones[pi])
            q_cv = int(zeros[qi])
            dv = delta_vec(h, N, p_cv, q_cv)
            nP = float((c + dv).max())
            if nP < best_new_P:
                best_new_P = nP
                best_pq = (p_cv, q_cv)
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            best_P = float(c.max())
            stuck = 0
        else:
            stuck += 1
            if stuck >= stuck_tol:
                for _ in range(perturb_n):
                    pr = int(ones[np.random.randint(len(ones))])
                    qr = int(zeros[np.random.randint(len(zeros))])
                    dv = delta_vec(h, N, pr, qr)
                    if float((c + dv).max()) < best_P + 0.01:
                        h[pr] = 0.0
                        h[qr] = 1.0
                        c = fft_corr(h)
                        best_P = float(c.max())
                        stuck = 0
                        break
                stuck = 0
    return h, best_P

def nesterov_smooth(h, N, deadline, tau0=80.0, tau1=40.0, lr0=0.012, beta0=0.85):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    m_vec = np.zeros(N)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    lr = lr0
    beta = beta0
    it = 0
    total_time = max(1.0, deadline - time.time())
    while time.time() < deadline:
        t_frac = min(1.0, it / max(1, int(total_time / 0.05)))
        tau = tau0 + (tau1 - tau0) * t_frac
        y = h + m_vec
        g = smooth_grad(y, N, tau)
        h_new = project_sum(y - lr * g, target)
        m_vec = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            pc = float(np.max(fft_corr(h)))
            if pc < best_c:
                best_c = pc
                best_h = h.copy()
            lr *= 0.995
    pc = float(np.max(fft_corr(h)))
    if pc < best_c:
        best_c = pc
        best_h = h.copy()
    return best_h, best_c * (2.0 / N)
