For a swap (p: 1→0, q: 0→1), the change in corr[lag] is:

For binary h, h[p±lag] is either 0 or 1, so this is just:

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def c5_of(h):

    return float(np.max(fft_corr(h))) * 2.0 / N

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def subgrad_h(h, corr, N, K):
    grad = np.zeros(N)
    center = N - 1
    idx = np.argpartition(corr, -K)[-K:]
    for ai in idx:
        lag = int(ai - center)
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            grad[lag:] += 1.0 - h[:N - lag]
            grad[:N - lag] -= h[lag:]
        else:
            M = -lag
            grad[:N - M] += 1.0 - h[M:]
            grad[M:] -= h[:N - M]
    return grad / K

def swap_delta(c, h, p, q, top_idx, N):
    center = N - 1
    best = 0.0
    for ai in top_idx:
        lag = int(ai - center)
        d = 0.0
        if 0 <= p - lag < N:
            d -= 1.0 - h[p - lag]
        if 0 <= p + lag < N:
            d += h[p + lag]
        if 0 <= q - lag < N:
            d += 1.0 - h[q - lag]
        if 0 <= q + lag < N:
            d -= h[q + lag]
        nv = c[ai] + d
        if nv > best:
            best = nv
    return best

    n1 = int(h.sum())

        if time.time() > deadline:
            break
        c = fft_corr(h)
        pk = float(c.max())

        if len(ones) > n_samp:

        else:

        if len(zeros) > n_samp:

    it = 0
