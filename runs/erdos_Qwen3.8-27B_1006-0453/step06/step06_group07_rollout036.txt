import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_binary(h, N, seed=7):
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    nh = N // 2
    if n1 != nh:
        rng = np.random.RandomState(seed)
        if n1 > nh:
            idx = rng.choice(np.where(h==1)[0], n1-nh, replace=False)
            h[idx] = 0
        else:
            idx = rng.choice(np.where(h==0)[0], nh-n1, replace=False)
            h[idx] = 1
    return h

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def nesterov_cont(h, N, deadline, k=80, lr0=0.02, iters=None):

    m = np.zeros(N)

    for it in range(iters if iters else 100000):
        if time.time() > deadline:
            break

            best_pk = pk
            best_h = h.copy()

        else:

        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)

                grad[L:] += (1.0 - y[:N-L])

        # We want to go AWAY from 0.5, so subtract this from grad

        # We want to minimize F (which is negative, more negative in middle)

def upsample(h_old, N_new):
