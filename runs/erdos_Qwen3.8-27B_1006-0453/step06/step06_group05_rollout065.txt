def cont_grad_opt(h, N, deadline, beta=30.0, lr0=0.01, iters=3000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h))) * dx
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    while time.time() < deadline and t < iters:
        t += 1
        c = fft_corr(h)
        cur = float(c.max()) * dx
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
        # log-sum-exp weights
        cm = c - c.max()
        w = np.exp(beta * cm)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(c)); w[np.argmax(c)] = 1.0
        else:
            w /= ws
        # gradient: grad = -(D @ w) * dx
        D = compute_D(h, N)
        grad = -(D @ w) * dx
        grad -= grad.mean()  # enforce sum constraint approximately
        bc1 = 1.0 - b1 ** t
        bc2 = 1.0 - b2 ** t
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        h = project_sum(h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps), target)
        if t % 500 == 0:
            lr *= 0.85
    return best_h, best_val

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def make_binary(h, N):
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())

            h[oi[:N//2]] = 1.0
        else:
            h[oi] = 1.0
            h[zi[:N//2 - n1]] = 1.0
    return h

        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        ns = max(30, int(len(ones) * sample_frac))
        if ns < len(ones):
            p_list = np.random.choice(ones, ns, replace=False)
        else:
            p_list = ones
        bd, bpq = 0.0, None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0:
            temp = anneal * max(0.05, 1.0 - sweep / max_sweeps)
            if np.random.random() < temp:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
        else:
            break
    return h, best_c

def cont_grad_opt(h, N, deadline, beta=20.0, lr0=0.012, iters=5000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h))) * dx
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    while time.time() < deadline and t < iters:
        t += 1
        c = fft_corr(h)
        cur = float(c.max()) * dx
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
        cm = c - c.max()
        e = np.exp(beta * cm)
        es = e.sum()
        if es < 1e-30:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / es
        D = compute_D(h, N)
        grad = -(D @ w) * dx
        grad -= grad.mean()
        bc1 = 1.0 - b1 ** t
        bc2 = 1.0 - b2 ** t
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        h = project_sum(h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps), target)
        if t % 400 == 0:
            lr *= 0.87
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Fixed-sign binary search (largest chunk)
    d1 = t0 + 320
    h_b = make_binary(h0, N0)
    h_s, _ = bin_search(h_b, N0, d1, max_sweeps=3000, anneal=0.0, sample_frac=0.8)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Continuous log-sum-exp optimization
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 220)
        hc = best_h.astype(float).copy()
        hc += np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = cont_grad_opt(hc, N0, d2, beta=20.0, lr0=0.014, iters=8000)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search from continuous result
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 160)
        h_b3 = make_binary(best_h, N0)
        h_s3, _ = bin_search(h_b3, N0, d3, max_sweeps=2000, anneal=0.0, sample_frac=1.0)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Second continuous pass with higher beta
    if time.time() < deadline - 180:
        d4 = min(deadline - 130, time.time() + 100)
        hc4 = make_binary(best_h, N0).astype(float)
        hc4 += np.random.randn(N0) * 0.02
        hc4 = project_sum(hc4, N0 / 2.0)
        hn4, c5n4 = cont_grad_opt(hc4, N0, d4, beta=50.0, lr0=0.008, iters=6000)
        if c5n4 < best_c5:
            best_c5, best_h = c5n4, hn4.copy()

    # Phase 5: Annealed binary search
    if time.time() < deadline - 120:
        d5 = min(deadline - 70, time.time() + 90)
        h_b5 = make_binary(best_h, N0)
        h_s5, _ = bin_search(h_b5, N0, d5, max_sweeps=600, anneal=0.1, sample_frac=0.6)
        c5_5 = compute_c5(h_s5, N0)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h_s5.copy()

    # Phase 6: Multi-start perturbations
    if time.time() < deadline - 60:
        d6 = min(deadline - 30, time.time() + 55)
        for attempt in range(4):
            if time.time() > d6:
                break
            hp = make_binary(best_h, N0).copy()
            n_flip = np.random.randint(5, 14)
            for _ in range(n_flip):
                oi = np.where(hp == 1)[0]
                zi = np.where(hp == 0)[0]
                if len(oi) < 1 or len(zi) < 1:
                    break
                p = int(np.random.randint(len(oi)))
                q = int(np.random.randint(len(zi)))
                hp[oi[p]], hp[zi[q]] = 0.0, 1.0
            sub_dl = min(d6, time.time() + 10)
            h_pt, _ = bin_search(hp, N0, sub_dl, max_sweeps=250, anneal=0.05, sample_frac=0.9)
            c5_pt = compute_c5(h_pt, N0)
            if c5_pt < best_c5:
                best_c5, best_h = c5_pt, h_pt.copy()

    # Phase 7: Multi-resolution
    if time.time() < deadline - 30:
        d7 = min(deadline - 12, time.time() + 25)
        for N_try in [400, 360, 420]:
            if time.time() > d7:
                break
            if N_try % 2 != 0:
                N_try += 1
            if N_try == N0:
                continue
            x_old = np.linspace(0, 2, N0, endpoint=False)
            x_new = np.linspace(0, 2, N_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_bt = (h_interp > 0.5).astype(float)
            nh2 = N_try // 2
            n1 = int(h_bt.sum())
            if n1 > nh2:
                idx = np.where(h_bt == 1)[0]
                h_bt[idx[np.argsort(h_interp[idx])[:n1 - nh2]]] = 0.0
            elif n1 < nh2:
                idx = np.where(h_bt == 0)[0]
                h_bt[idx[np.argsort(-h_interp[idx])[:nh2 - n1]]] = 1.0
            sub_dl = min(d7, time.time() + 8)
            h_rt, _ = bin_search(h_bt, N_try, sub_dl, max_sweeps=400, anneal=0.0, sample_frac=1.0)
            c5_rt = compute_c5(h_rt, N_try)
            if c5_rt < best_c5:
                best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 8: Final binary polish
    if time.time() < deadline - 3:
        hF = make_binary(best_h, len(best_h))
        hF, _ = bin_search(hF, len(best_h), deadline - 1, max_sweeps=150, anneal=0.0, sample_frac=1.0)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
