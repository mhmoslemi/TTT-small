        if L == 0:

        else:

        delta += h[p + L]
    # i=q+L contribution (i-L = q)
    if L != 0 and 0 <= q + L < N and q + L != p:
        delta -= h[q + L]
    return delta

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        sel = idx[rng.permutation(len(idx))[:n1 - nh]]
        h[sel] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        sel = idx[rng.permutation(len(idx))[:nh - n1]]
        h[sel] = 1
    return h

def lsearch_swap(h, N, deadline, n_cand=400, top_lags=15, anneal_p=0.25):

        if len(ones) < 1 or len(zeros) < 1:
