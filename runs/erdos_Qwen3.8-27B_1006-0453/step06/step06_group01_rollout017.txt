    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, n_ones):
    h = h.astype(float).copy()
    s = int(np.round(h.sum()))
    if s > n_ones:
        idx = np.where(h > 0.5)[0]
        if len(idx) >= s - n_ones:
            h[np.random.choice(idx, s - n_ones, replace=False)] = 0.0
    elif s < n_ones:
        idx = np.where(h < 0.5)[0]
        if len(idx) >= n_ones - s:
            h[np.random.choice(idx, n_ones - s, replace=False)] = 1.0
    return (h > 0.5).astype(float)

def sa_optimize(h_init, N, deadline, T0=1.2, T1=0.002, n_ref=6000, hybrid=0.25):
    h = make_binary(h_init, N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    no_imp = 0
    step = 0
    while time.time() < deadline:
        T = T0 * (T1 / T0) ** (step / n_ref)
        step += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        use_pair = len(ones) >= 2 and len(zeros) >= 2 and (np.random.random() < 0.12)
        if use_pair:
            i1, i2 = np.random.randint(len(ones), 2)
            j1, j2 = np.random.randint(len(zeros), 2)
            if i1 == i2 or j1 == j2:
                continue
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
            nc = float(fft_corr(h).max())
            if nc <= best_c or np.random.random() < np.exp(-(nc - best_c) / max(T, 1e-12)):
                corr = fft_corr(h)
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_imp = 0
                for arr in (ones, zeros):
                    for p in (p1, p2):
                        if p in arr:
                            arr[:] = arr[arr != p]
                    for q in (q1, q2):
                        if q not in arr:
                            arr = np.append(arr, q)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[p1] = h[p2] = 1.0
                h[q1] = h[q2] = 0.0
        elif np.random.random() < hybrid and len(ones) >= 5:
            # Best-improvement: sample a few p, find best q
            n_s = min(12, len(ones))
            p_sample = np.random.choice(ones, n_s, replace=False)
            nn = len(ones) * len(zeros)
            best_nc, best_p, best_q = 1e9, -1, -1
            for p in p_sample:
                h[int(p)] = 0.0
                sub = fft_corr(h)
                # find best q to flip to 1: minimize max corr
                for q in zeros:
                    h[int(q)] = 1.0
                    c = float(sub.max())
                    if c < best_nc:
                        best_nc = c
                        best_p, best_q = int(p), int(q)
                    h[int(q)] = 0.0
                h[int(p)] = 1.0
            if best_p >= 0:
                h[best_p] = 0.0
                h[best_q] = 1.0
                corr = fft_corr(h)
                nc = float(corr.max())
                if nc <= best_c or np.random.random() < np.exp(-(nc - best_c) / max(T, 1e-12)):
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                        no_imp = 0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p] = 0.0
            h[q] = 1.0
            nc = float(fft_corr(h).max())
            if nc <= best_c or np.random.random() < np.exp(-(nc - best_c) / max(T, 1e-12)):
                corr = fft_corr(h)
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_imp = 0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[p] = 1.0
                h[q] = 0.0
        no_imp += 1
        if no_imp > 8000:
            # Block perturbation
            blen = np.random.randint(3, 11)
            if blen < N:
                st = np.random.randint(0, N - blen)
                h[st:st+blen] = 1.0 - h[st:st+blen]
                n1 = int(h.sum())
                tgt = N // 2
                if n1 > tgt:
                    oz = np.where(h == 1)[0]
                    h[np.random.choice(oz, n1 - tgt, replace=False)] = 0.0
                elif n1 < tgt:
                    oz = np.where(h == 0)[0]
                    h[np.random.choice(oz, tgt - n1, replace=False)] = 1.0
                corr = fft_corr(h)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                no_imp = 0
    return best_h, best_c * (2.0 / N)

def cont_opt(h_init, N, deadline, k=50):
    h = h_init.astype(float).copy()
    h = project_sum(h, N / 2.0)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = 0.012
    beta = 0.82
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for t in idx:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(idx)
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.97
            beta = min(0.95, 0.80 + it * 0.0004)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SA binary (core search)
    d1 = min(deadline - 400, t0 + 280)
    h1, c1 = sa_optimize((h0 > 0.5).astype(float), N, d1, T0=1.0, T1=0.003, n_ref=5000)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Continuous polish
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + 130)
        hc = best_h + np.random.randn(N) * 0.03
        hc = project_sum(hc, N / 2.0)
        h2, c2 = cont_opt(hc, N, d2, k=50)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: SA from improved (lower temp)
    if time.time() < deadline - 150:
        d3 = min(deadline - 100, time.time() + 130)
        h3, c3 = sa_optimize((best_h > 0.5).astype(float), N, d3, T0=0.4, T1=0.001, n_ref=4000, hybrid=0.35)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Try other N values
    for Nt in [360, 400, 420, 460]:
        if time.time() > deadline - 40:
            break
        x_old = np.linspace(0, 2, N, endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = make_binary(hi, Nt // 2)
        d4 = min(deadline - 25, time.time() + 35)
        h4, c4 = sa_optimize(hbt, Nt, d4, T0=0.6, T1=0.002, n_ref=3000)
        if c4 < best_c5:
            best_c5, best_h, N = c4, h4, Nt

    # Phase 5: Final continuous micro-polish
    if time.time() < deadline - 3:
        hF = best_h + np.random.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cF = cont_opt(hF, len(best_h), deadline - 1, k=70)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
