import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    N = len(h)
    return float(compute_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def build_A(h):
    """Build A[p,j] = h[p+sh_j] + h[p-sh_j] with boundary=0"""
    N = len(h)
    L = 2 * N - 1
    center = N - 1
    sh = np.arange(L) - center
    p = np.arange(N)[:, None]
    ps = p + sh[None, :]
    pm = p - sh[None, :]
    A = np.where((ps >= 0) & (ps < N), h[np.clip(ps, 0, N-1)], 0.0) + \
        np.where((pm >= 0) & (pm < N), h[np.clip(pm, 0, N-1)], 0.0)
    return A

def nesterov_smooth(h_init, N, deadline, lam0=15, lam1=150, topk=60):
    h = h_init.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(N)
    beta = 0.85
    lr = 0.04
    it = 0
    target = N / 2.0
    while time.time() < deadline:
        frac = min(1.0, it / 300.0)
        lam = lam0 + (lam1 - lam0) * frac
        y = h + m
        c = compute_corr(y)
        cm = c.max()
        w = np.exp(lam * (c - cm))
        w /= w.sum()
        c5_y = compute_c5(y)
        if c5_y < best_c5:
            best_c5 = c5_y
            best_h = h.copy()
        top = np.argsort(w)[-topk:]
        center = N - 1
        grad = np.zeros(N)
        for j in top:
            wt = w[j]
            if wt < 1e-14:
                continue
            sh = j - center
            if sh == 0:
                grad += wt * (1.0 - 2.0 * y)
            elif 0 < sh < N:
                g = np.ones(N)
                g[:N - sh] -= y[sh:]
                g[sh:] -= y[:N - sh]
                grad += wt * g
            elif -N < sh < 0:
                s2 = -sh
                g = np.ones(N)
                g[s2:] -= y[:N - s2]
                g[:N - s2] -= y[s2:]
                grad += wt * g
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.95
        if it % 300 == 0 and best_c5 < 0.381:
            lr *= 0.9
    return best_h, best_c5

def binary_search(h_init, N, deadline, max_sweeps=5000, n_sample=200, patience=600):
    h = (h_init > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[nh:]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1.0
    best_c = float(compute_corr(h).max())
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        P = c.max()
        if P < best_c - 1e-14:
            best_c = P
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > patience:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        A = build_A(h)
        # Sample p's, prioritizing high-contribution positions
        if len(ones) > n_sample:
            # Prioritize: larger A[p, argmax(c)] means higher risk
            am = int(c.argmax())
            risk = A[ones, am]
            perm = np.argsort(-risk)[:n_sample // 2]
            perm2 = np.random.choice(len(ones), n_sample - n_sample // 2, replace=False)
            p_list = np.concatenate([ones[perm], ones[perm2]])
        else:
            p_list = ones
        best_score = P + 1.0
        best_pq = None
        zero_A = A[zeros, :]
        for p in p_list:
            base = c + A[p, :]
            diffs = base[None, :] - zero_A
            new_maxes = diffs.max(axis=1)
            mi = int(new_maxes.argmin())
            if new_maxes[mi] < best_score:
                best_score = float(new_maxes[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None and best_score < P - 1e-12:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            # Random kick with simulated annealing
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            # Check if random move is better
            new_c = c + A[p, :] - A[q, :]
            if new_c.max() < P + 0.5:  # accept if not too bad
                h[p], h[q] = 0.0, 1.0
    final_c5 = best_c * (2.0 / N)
    # Return the h that achieves best_c
    return h, final_c5

    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def build_A(h):
    N = len(h)
    L = 2 * N - 1
    center = N - 1
    sh = np.arange(L) - center
    p = np.arange(N)[:, None]
    ps = p + sh[None, :]
    pm = p - sh[None, :]
    A = np.where((ps >= 0) & (ps < N), h[np.clip(ps, 0, N - 1)], 0.0) + \
        np.where((pm >= 0) & (pm < N), h[np.clip(pm, 0, N - 1)], 0.0)
    return A

def grad_component(h, sh, N):
    g = np.zeros(N)
    if sh == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < sh < N:
        g[:N - sh] += (1.0 - h[sh:])
        g[sh:] -= h[:N - sh]
    elif -N < sh < 0:
        s2 = -sh
        g[s2:] += (1.0 - h[:N - s2])
        g[:N - s2] -= h[s2:]
    return g
