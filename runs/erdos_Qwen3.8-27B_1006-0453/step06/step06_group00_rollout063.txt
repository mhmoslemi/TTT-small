def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0/N)

def project_01(h, target):

        return h
    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

    if L == 0:
        g[:] = 1.0 - 2.0*h
    elif 0 < L < N:
        g[:N-L] += 1.0 - h[L:]
        g[L:] -= h[:N-L]
    elif -N < L < 0:

        g[M:] += 1.0 - h[:N-M]
        g[:N-M] -= h[M:]
