    S = 2 * N - 1
    shifts = np.arange(S) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv  # p - t
    pp = pos + sv  # p + t
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * np.conj(G), L)

def compute_c5(h, N):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * np.conj(G), L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    S = 2 * N - 1
    shifts = np.arange(S) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
    D[np.arange(N), N - 1] = 0.0
    return D

def bin_search(h, N, deadline, max_sweeps=500, sample_n=None, anneal=0.0, seed_rng=None):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[ones_i[:nh]] = 1
        else:
            h[ones_i] = 1
            h[zeros_i[:nh - n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = seed_rng or np.random
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 100 and anneal == 0:
            break
        if stall > 50 and anneal > 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        p_list = ones if not sample_n or len(ones) <= sample_n else rng.choice(ones, sample_n, replace=False)
        bd, bpq = 0.0, None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P

        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0

        elif anneal > 0 and rng.random() < anneal:

def continuous_opt(h, N, deadline, tau0=50.0, lr0=0.01, topk=10):

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    tau = tau0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        # Top-k weighting
        if topk < len(c):
            top_idx = np.argpartition(c, -topk)[-topk:]
            w = np.zeros(len(c))
            w[top_idx] = 1.0 / topk
        else:
            w = np.exp(tau * (c - c.max()))
            w /= w.sum()
        D = compute_D(h, N)
        grad = -(D.T @ w) * dx
        # Center the gradient (for sum constraint)
        grad -= grad.mean()
        bc1 = 1 - b1 ** t
        bc2 = 1 - b2 ** t
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        h_new = h + lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = project_sum(h_new, target)
        if t % 200 == 0:
            lr *= 0.9
            tau *= 0.85
            # Periodic binarize-blend
            hb = (h > 0.5).astype(float)
