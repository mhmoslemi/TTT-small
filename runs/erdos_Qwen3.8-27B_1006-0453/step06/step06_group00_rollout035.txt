import numpy as np
import time

def compute_overlap_full(h):
    """Full overlap vector for h vs 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    N = len(h)
    return float(compute_overlap_full(h).max()) * (2.0 / N)

def project_binary_target(h, target_sum):
    """Project h to binary with exactly target_sum ones."""
    h_sorted_idx = np.argsort(h)[::-1]
    result = np.zeros(len(h))
    result[h_sorted_idx[:int(target_sum)]] = 1.0
    return result

def anneal_binary(h, N, deadline, max_iters=5000, tabu_len=50, T0=1.0, T1=0.001):
    """Simulated annealing on binary h with exactly N/2 ones."""
    target = N // 2
    h = h.copy()
    # Ensure correct number of ones
    n_ones = int(h.sum())
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:n_ones-target]] = 0
        else:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target-n_ones]] = 1
    
    best_h = h.copy()
    best_c = c5_of(h)
    current_c = best_c
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    tabu = []
    tabu_set = set()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0) ** (it / max_iters)
        
        # Sample candidate move
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        
        key = (min(a,b), max(a,b))
        if key in tabu_set:
            # Try another
            continue
        
        # Swap
        h[a], h[b] = 0, 1
        new_c = c5_of(h)
        delta = new_c - current_c
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            current_c = new_c
            # Update tracking
            ones[a], zeros[b] = 0, 1  # a was 1, now 0; b was 0, now 1
            # Actually need to update the sets properly
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            
            # Add to tabu
            tabu.append(key)
            tabu_set.add(key)
            if len(tabu) > tabu_len:
                old = tabu.pop(0)
                tabu_set.discard(old)
        else:
            # Reject, swap back
            h[a], h[b] = 1, 0
    
    return best_h, best_c

    """Compute change in full correlation vector when h[a]:1→0, h[b]:0→1."""
    # For lag k (in full mode, k ranges from -(N-1) to N-1, stored as 0 to 2N-2)
    # corr_full[k] = sum_i h[i] * (1-h[i+k]) where valid index range depends on k

    # output[k] = sum_{i=max(0,k-(N-1))}^{min(N-1,k)} h[i] * (1-h[i+k-(N-1)])

    # represents sum_i h[i] * (1-h[i-(k-(N-1))]) adjusted for bounds
    
    # Simpler: let lag = k - (N-1), ranging from -(N-1) to N-1
    # corr[lag] = sum_i h[i] * (1-h[i+lag]) for valid i
    
    # Delta from changing h[a]: 1→0:
    #   -1 * (1-h[a+lag]) for term where i=a (valid if 0<=a+lag<N)
    #   +h[a-lag] * 1 for term where i+lag=a, i=a-lag (valid if 0<=a-lag<N)
    #   Wait: the term at i=a-lag was h[a-lag]*(1-h[a])=0, now h[a-lag]*(1-0)=h[a-lag]
    #   So delta is +h[a-lag] (if valid)
    
    # Delta from changing h[b]: 0→1:
    #   +1 * (1-h[b+lag]) for term where i=b (valid if 0<=b+lag<N)
    #   -h[b-lag] * 1 for term where i+lag=b, i=b-lag (valid if 0<=b-lag<N)
    #   Was h[b-lag]*(1-0)=h[b-lag], now h[b-lag]*(1-1)=0
    #   So delta is -h[b-lag] (if valid)
    
    # Total delta for lag l:
    # delta[l] = -(1-h[a+l]) + h[a-l] + (1-h[b+l]) - h[b-l]
    # where each term is 0 if index out of [0,N)
    
    lags = np.arange(-(N-1), N)  # shape (2N-1,)
    
    # Term 1: -(1-h[a+l]) valid when 0<=a+l<N
    t1 = np.zeros(2*N-1)
    valid1 = (a + lags >= 0) & (a + lags < N)
    t1[valid1] = -(1.0 - h[a + lags[valid1]])
    
    # Term 2: +h[a-l] valid when 0<=a-l<N
    t2 = np.zeros(2*N-1)
    valid2 = (a - lags >= 0) & (a - lags < N)
    t2[valid2] = h[a - lags[valid2]]
    
    # Term 3: +(1-h[b+l]) valid when 0<=b+l<N
    t3 = np.zeros(2*N-1)
    valid3 = (b + lags >= 0) & (b + lags < N)
    t3[valid3] = (1.0 - h[b + lags[valid3]])
    
    # Term 4: -h[b-l] valid when 0<=b-l<N
    t4 = np.zeros(2*N-1)
    valid4 = (b - lags >= 0) & (b - lags < N)
    t4[valid4] = -h[b - lags[valid4]]
    
    return t1 + t2 + t3 + t4

import numpy as np
import time

def overlap_full(h):
    """Full cross-correlation of h and 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def c5_from_overlap(ov, N):
    return float(ov.max()) * (2.0 / N)

def swap_delta(ov, h, a, b, N):
    """Compute new overlap after swapping h[a]:1->0, h[b]:0->1. O(N) operation."""
    # lags in 'full' mode: index k corresponds to lag d = k - (N-1)
    # cov[k] = sum_i h[i] * (1-h[i+d]) where d = k-(N-1)
    # After swap, for each lag d:
    # delta = -(1-h[a+d]) + h[a-d] + (1-h[b+d]) - h[b-d]  (out-of-range terms are 0)
    
    d = np.arange(-(N-1), N)  # shape (2N-1,)
    # This is 2N-1 values from -(N-1) to N-1
    
    delta = np.zeros(2*N-1)
    
    # -(1-h[a+d]) where 0<=a+d<N
    va = (a + d >= 0) & (a + d < N)
    delta[va] += -(1.0 - h[a + d[va]])
    
    # +h[a-d] where 0<=a-d<N  (i.e., d is such that a-d is valid)
    vb = (a - d >= 0) & (a - d < N)
    delta[vb] += h[a - d[vb]]
    
    # +(1-h[b+d]) where 0<=b+d<N
    vc = (b + d >= 0) & (b + d < N)
    delta[vc] += (1.0 - h[b + d[vc]])
    
    # -h[b-d] where 0<=b-d<N
    vd = (b - d >= 0) & (b - d < N)
    delta[vd] += -h[b - d[vd]]
    
    return ov + delta

def anneal_focused(h, N, deadline, max_iters=20000, focus_lags=30, tabu_size=100):
    """Focused simulated annealing on binary h."""
    target = N // 2
    h = h.copy()
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:n1-target]] = 0
        else:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target-n1]] = 1
    
    ov = overlap_full(h)
    current_max = float(ov.max())
    best_ov = ov.copy()
    best_h = h.copy()
    best_max = current_max
    
    ones = np.where(h == 1)[0].tolist()  # Use lists for fast add/remove
    zeros = np.where(h == 0)[0].tolist()
    
    tabu = []
    tabu_set = set()
    
    # Focus on lags near the center (small |d|)
    N1 = N - 1
    d_range = np.arange(-(N-1), N)
    # Weight: focus on |d| <= focus_lags
    weights = np.ones(2*N-1)
    focus_mask = np.abs(d_range) <= focus_lags
    weights[focus_mask] = 5.0  # Extra weight on focused lags
    
    T = 0.5
    T_min = 0.0001
    cool = (T_min / T) ** (1.0 / max(max_iters, 1))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = max(T * cool, T_min)
        
        # Generate candidate: pick a from ones, b from zeros
        # Bias: pick a from positions that are "bad" (h[a]=1 but many neighbors are 0)
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        
        if a == b:
            continue
        
        key = (min(a,b), max(a,b))
        if key in tabu_set:
            continue
        
        # Compute new overlap
        new_ov = swap_delta(ov, h, a, b, N)
        new_max = float(np.max(new_ov * weights))  # Weighted max for focus
        
        # Actually, let's just use the true max for acceptance,
        # but use weighted for candidate selection
        true_new_max = float(new_ov.max())
        delta = true_new_max - current_max
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            # Accept
            h[a], h[b] = 0, 1
            ov = new_ov
            current_max = true_new_max
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
            
            if true_new_max < best_max:
                best_max = true_new_max
                best_h = h.copy()
                best_ov = ov.copy()
            
            tabu.append(key)
            tabu_set.add(key)
            if len(tabu) > tabu_size:
                old = tabu.pop(0)
                tabu_set.discard(old)
        # else: reject, do nothing
    
    return best_h, best_max * (2.0 / N)

    """Compute scores for biased move generation."""
    # For positions with h=1: score = number of zeros at distance <= K
    # For positions with h=0: score = number of ones at distance <= K
    scores = np.zeros(N)
    for i in range(N):
        # Count neighbors within K positions
        lo = max(0, i - K)
        hi = min(N, i + K + 1)
        neighborhood = h[lo:hi]
        if h[i] == 1:

        else:
            scores[i] = neighborhood.sum()  # h[i]=0, so no need to exclude
    
    return scores

        new_maxes = new_ovs.max(axis=1)  # (nz,)
        
        best_local = new_maxes.argmin()
        if new_maxes[best_local] < best_new_max:
            best_new_max = new_maxes[best_local]
            best_a = a
            best_b = zeros[best_local]
    
    return best_a, best_b, best_new_max

Actually, let me think about the continuous problem more carefully.

We want to minimize max_k overlap_k(h) where overlap_k = Σ_i h[i](1-h[i+k]).

This is a DC (Difference of Convex) problem since overlap_k = (N/2) - Σ_i h[i]h[i+k] and the second term is not concave in general.

But we can write it as a min-max:
min_t { t : overlap_k(h) ≤ t ∀k, 0≤h≤1, Σh=N/2 }

For a fixed t, this is a feasibility check. The constraints overlap_k ≤ t are:
Σ_i h[i] - Σ_i h[i]h[i+k] ≤ t
Σ_i h[i]h[i+k] ≥ N/2 - t

The constraint Σ h[i]h[i+k] ≥ c is a **quadratic** constraint (not linear, not convex in general).

So this isn't straightforwardly solvable by LP or SOCP.

Alternative: **Benders decomposition** or **column generation** approach. Or just use the projected gradient / Nesterov.

Let me try a different continuous approach: **Coordinate descent**

For each coordinate h[j], optimize it while keeping others fixed:
∂(max_k overlap_k)/∂h[j] - this is complex because of the max.

Simpler: minimize Σ_k (overlap_k)² (sum of squares) as a proxy for max. This is smooth and we can use gradient descent.

∂(Σ_k overlap_k²)/∂h[j] = 2 * Σ_k overlap_k * ∂overlap_k/∂h[j]

∂overlap_k/∂h[j] = (1-h[j+k]) - h[j-k] (with bounds, i.e., terms are 0 if index out of range)

This is a smooth optimization problem! And the sum of squares penalizes the peaks.

But it's not exactly the same as minimizing the max. The max is more important for our objective.

Compromise: minimize Σ_k max(overlap_k - t, 0)² + λ*max_k(overlap_k)
This penalizes both the tail (above t) and the max.

Or: use the **log-sum-exp** approximation:
max_k overlap_k ≈ (1/β) * log(Σ_k exp(β * overlap_k))

For large β, this approximates the max. And it's smooth!

Gradient: ∂/∂h[j] [(1/β) log Σ_k exp(β*overlap_k)] = (1/Σ_k exp(β*overlap_k)) * Σ_k exp(β*overlap_k) * β * ∂overlap_k/∂h[j]

This puts weight on the large overlaps, which is what we want.

But computing this for all k (759 of them) at each step is expensive. We can approximate with the top-k.
