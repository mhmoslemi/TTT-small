def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / N)

def project_01(h, target):
    """Project h onto [0,1]^N ∩ {sum = target}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient(h, m=20):
    """Compute subgradient of max_k f_k(h) using top-m active constraints"""
    N = len(h)
    corr = fft_corr(h)
    top = np.argpartition(corr, -m)[-m:]
    grad = np.zeros(N)
    center = N - 1
    for idx in top:
        L = int(idx - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
    grad /= m
    return grad, corr

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def grad_smoothed_max(h, m=25, T=40):
    N = len(h)
    corr = fft_corr(h)
    top = np.argpartition(corr, -m)[-m:]
    f_a = corr[top]
    f_s = f_a - f_a.max()
    w = np.exp(T * f_s)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for wi, idx in zip(w, top):
        k = int(idx - center)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wi * (1.0 - h[:N - k])
            grad[:N - k] -= wi * h[k:]
        else:
            M = -k
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad, corr

def cont_opt(h, N, deadline, lr0=0.04, m=25, Ts=40):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h, N)
    lr = lr0
    momentum = np.zeros(N)
    beta = 0.90
    it = 0
    stall = 0
    while time.time() < deadline:
        grad, corr = grad_smoothed_max(h, m, Ts)
        c5_cur = float(np.max(corr)) * (2.0 / N)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 150:
            h = project_01(h + np.random.randn(N) * 0.02, target)
            stall = 0
            momentum *= 0.3
        y = h + momentum
        lr_try = lr
        h_new = project_01(y - lr_try * grad, target)
        for _ in range(6):
            c_new = float(np.max(fft_corr(h_new))) * (2.0 / N)
            if c_new <= c5_cur + 1e-12:
                break
            lr_try *= 0.5
            h_new = project_01(y - lr_try * grad, target)
        momentum = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            beta = min(0.96, 0.88 + 0.0006 * it)
        if it % 500 == 0:
            lr = max(lr * 1.005, lr0 * 0.4)
    return best_h, best_c5

def bin_sa(h, N, deadline, max_iters=20000, T0=0.025):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h >= 0.5)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h < 0.5)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    h = (h > 0.5).astype(float)
    best_c = np.inf
    best_h = h.copy()
    T = T0
    T_min = 1e-7
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        c5_cur = float(np.max(corr)) * (2.0 / N)
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        p = int(np.random.choice(ones))
        q = int(np.random.choice(zeros))
        h[p], h[q] = 0.0, 1.0
        c5_new = float(np.max(fft_corr(h))) * (2.0 / N)
        if c5_new < c5_cur:
            T = max(T * 0.9995, T_min)
        elif np.random.random() < np.exp(-(c5_new - c5_cur) / max(T, T_min)):
            T = max(T * 0.999, T_min)
        else:
            h[p], h[q] = 1.0, 0.0
            T = max(T * 0.999, T_min)
            if T < T_min:
                for _ in range(10):
                    ones2 = np.where(h == 1)[0]
                    zeros2 = np.where(h == 0)[0]
                    p2 = int(np.random.choice(ones2))
                    q2 = int(np.random.choice(zeros2))
                    h[p2], h[q2] = 0.0, 1.0
                    ct = float(np.max(fft_corr(h))) * (2.0 / N)
                    if ct < c5_cur:
                        c5_cur = ct
                        if ct < best_c:
                            best_c = ct
                            best_h = h.copy()
                    else:
                        h[p2], h[q2] = 1.0, 0.0
    return best_h, best_c

def make_binary(hf, N):
    nh = N // 2
    h = (hf > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary SA on initial (40s)
    d1 = min(deadline - 500, t0 + 40)
    h_bin = make_binary(h0, N0)
    hb, cb = bin_sa(h_bin, N0, d1, max_iters=30000, T0=0.03)
    if cb < best_c5:
        best_c5, best_h = cb, hb.copy()

    # Phase 2: Continuous proximal gradient (280s)
    if time.time() < deadline - 300:
        d2 = min(deadline - 260, time.time() + 280)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_01(hc, N0 / 2.0)
        hp, cp = cont_opt(hc, N0, d2, lr0=0.05, m=25, Ts=35)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()

    # Phase 3: Second continuous pass, different hyperparams (120s)
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 120)
        hc2 = best_h + np.random.randn(N0) * 0.01
        hc2 = project_01(hc2, N0 / 2.0)
        hp2, cp2 = cont_opt(hc2, N0, d3, lr0=0.02, m=15, Ts=60)
        if cp2 < best_c5:
            best_c5, best_h = cp2, hp2.copy()

    # Phase 4: Binary SA polish (60s)
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 60)
        h4 = make_binary(best_h, N0)
        h4b, c4b = bin_sa(h4, N0, d4, max_iters=50000, T0=0.01)
        if c4b < best_c5:
            best_c5, best_h = c4b, h4b.copy()

    # Phase 5: Multi-resolution search
    for N_try in [300, 440, 540, 680]:
        if time.time() > deadline - 50:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = make_binary(h_int, N_try)
        sub_dl = min(deadline - 30, time.time() + 35)
        hr, cr = bin_sa(h_bt, N_try, sub_dl, max_iters=20000, T0=0.02)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()
        if time.time() < deadline - 25 and N_try >= 400:
            d_c = min(deadline - 20, time.time() + 25)
            hc3 = best_h + np.random.randn(N_try) * 0.015
            hc3 = project_01(hc3, N_try / 2.0)
            hp3, cp3 = cont_opt(hc3, N_try, d_c, lr0=0.03, m=20, Ts=30)
            if cp3 < best_c5:
                best_c5, best_h = cp3, hp3.copy()

    # Phase 6: Final continuous polish
    if time.time() < deadline - 3:
        hF = best_h + np.random.randn(len(best_h)) * 0.008
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = cont_opt(hF, len(best_h), deadline - 1, lr0=0.012, m=20, Ts=50)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
