import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    hp = np.where(vp, h[pp_c], 0.0)
    hm = np.where(vm, h[pm_c], 0.0)
    D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
    D[:, N - 1] = 0.0  # fix k=0 double-counting
    return D

def bin_search(h, N, deadline, max_sweeps=500, anneal=0.0, sample_frac=1.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[ones_i[:nh]] = 1
        else:
            h[ones_i] = 1
            h[zeros_i[:nh - n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-15:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 120:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        ns = max(1, int(len(ones) * sample_frac))
        p_list = np.random.choice(ones, ns, replace=False) if ns < len(ones) else ones
        bd, bpq = 0.0, None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

        tau = tau0 * (0.97 ** (t * 0.01)) + 0.5
        c = fft_corr(h)
        mc = c.max() + 1e-10
        w = np.exp((c - mc) / tau)
        w = w / w.sum()
        D = compute_D(h, N)
        grad = (D.T @ w) * dx
        grad -= grad.mean()
        lr = lr0 / (1.0 + 0.03 * t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        hc1 = m / (1 - beta1 ** t)
        vc1 = v / (1 - beta2 ** t)
        h = project_sum(h - lr * hc1 / (np.sqrt(vc1) + eps), target)
        c5_now = compute_c5(h, N)
        if c5_now < best_val:
