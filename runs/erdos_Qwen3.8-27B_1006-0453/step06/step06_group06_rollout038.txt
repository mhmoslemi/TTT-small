import numpy as np

def fft_corr(h):

    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def evaluate(h):
    N = len(h)
    dx = 2.0 / N
    corr = fft_corr(h)
    return float(corr.max()) * dx

def proj_binary(h, N):
    """Project to binary with exactly N//2 ones"""
    h = np.round(h).astype(float)
    n1 = int(h.sum())
    target = N // 2
    if n1 > target:
        idx1 = np.where(h == 1)[0]
        np.random.shuffle(idx1)
        h[idx1[:n1-target]] = 0
    elif n1 < target:
        idx0 = np.where(h == 0)[0]
        np.random.shuffle(idx0)
        h[idx0[:target-n1]] = 1
    return h

def compute_swap_delta(h, N, p, q):
    """Compute delta in corr when h[p]:1->0 and h[q]:0->1"""
    s = np.arange(-(N-1), N)
    ps, pm = p + s, p - s
    qs, qm = q + s, q - s
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    
    d = np.zeros(2*N-1)
    pc = np.clip(ps, 0, N-1)
    pmc = np.clip(pm, 0, N-1)
    qsc = np.clip(qs, 0, N-1)
    qmc = np.clip(qm, 0, N-1)
    
    # p goes 1->0
    d[vps] -= (1.0 - h[pc[vps]])
    d[vpm] += h[pmc[vpm]]
    # q goes 0->1
    d[vqs] += (1.0 - h[qsc[vqs]])
    d[vqm] -= h[qmc[vqm]]
    return d

                    valid = np.arange(-s, N)
                contrib = valid[(h[valid] == 1)]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_binary(h, N):
    h = np.clip(np.round(h), 0, 1).astype(float)
    n1 = int(h.sum())
    target = N // 2
    if n1 > target:
        idx = np.where(h==1)[0]
        h[idx[np.random.choice(len(idx), n1-target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(h==0)[0]
        h[idx[np.random.choice(len(idx), target-n1, replace=False)]] = 1
    return h

def hill_climb(h, N, deadline, kick_size=10, kick_patience=500):
    h = h.astype(float).copy()
    h = proj_binary(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    
    s_arr = np.arange(-(N-1), N)
    no_improve = 0
    total_iter = 0
    
    while time.time() < deadline:
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        
        # Smart selection
        # Focus on shifts with high correlation
        top_5 = np.argsort(corr)[-5:]
        p = None
        for ti in top_5:
            s = int(ti - (N-1))
            if s > 0:
                valid = np.arange(0, N-s)
            elif s < 0:
                valid = np.arange(-s, N)
            else:
                continue
            contrib = valid[h[valid]==1]
            if len(contrib) > 0:
                p = contrib[np.random.randint(len(contrib))]
                break
        if p is None:
            p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        
        # Compute delta
        ps = p + s_arr
        pm = p - s_arr
        qs = q + s_arr
        qm = q - s_arr
        vps = (ps>=0)&(ps<N)
        vpm = (pm>=0)&(pm<N)
        vqs = (qs>=0)&(qs<N)
        vqm = (qm>=0)&(qm<N)
        
        d = np.zeros(2*N-1)
        pc = np.clip(ps, 0, N-1)
        pmc = np.clip(pm, 0, N-1)
        qsc = np.clip(qs, 0, N-1)
        qmc = np.clip(qm, 0, N-1)
        d[vps] -= (1.0 - h[pc[vps]])
        d[vpm] += h[pmc[vpm]]
        d[vqs] += (1.0 - h[qsc[vqs]])
        d[vqm] -= h[qmc[vqm]]
        
        nc = corr + d
        npk = float(nc.max())
        delta = npk - current_pk
        
        if delta < 0:
            h[p] = 0; h[q] = 1
            corr = nc
            current_pk = npk
            no_improve = 0
            total_iter += 1
            if npk < best_pk:
                best_pk = npk
                best_h = h.copy()
        else:
            total_iter += 1
            no_improve += 1
        
        if no_improve >= kick_patience:
            no_improve = 0
            # Kick: flip random pairs
            for _ in range(kick_size):
                o = np.where(h==1)[0]
                z = np.where(h==0)[0]
                pp = o[np.random.randint(len(o))]
                qq = z[np.random.randint(len(z))]
                h[pp], h[qq] = h[qq], h[pp]
            corr = fft_corr(h)
            current_pk = float(corr.max())
    
    return best_h, best_pk

def continuous_refine(h, N, deadline, top_k=40, lr0=0.003):

            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0*h)
            elif s > 0:

                grad[s:] -= h[:N-s]
            else:

                else:
                    hi = mid
            h_new = np.clip(h_new-(lo+hi)*0.5, 0, 1)
        h = h_new
        
        if t % 300 == 0:
            lr_scale = 0.98 ** (t//300)
            # Reduce lr over time

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def proj_binary(h, N):
    h = np.clip(np.round(h), 0, 1).astype(float)
    n1 = int(h.sum())
    target = N // 2
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[idx[np.random.choice(len(idx), n1 - target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[idx[np.random.choice(len(idx), target - n1, replace=False)]] = 1
    return h

def proj_continuous(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def compute_delta(h, N, p, q, s_arr):
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2*N - 1)
    pc = np.clip(ps, 0, N-1); pmc = np.clip(pm, 0, N-1)
    qsc = np.clip(qs, 0, N-1); qmc = np.clip(qm, 0, N-1)
    d[vps] -= (1.0 - h[pc[vps]])
    d[vpm] += h[pmc[vpm]]
    d[vqs] += (1.0 - h[qsc[vqs]])
    d[vqm] -= h[qmc[vqm]]
    return d

    h = proj_binary(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    no_improve = 0
    center = N - 1
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Smart p selection: bias toward positions in high-corr shifts
        p = None
        if top_shifts > 0:
            top_idx = np.argpartition(corr, -top_shifts)[-top_shifts:]
            for ti in top_idx:

                if s > 0:

                else:

        if p is None:
