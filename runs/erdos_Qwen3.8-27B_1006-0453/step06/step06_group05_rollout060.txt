```python
def soft_max_gradient(h, N, tau):
    L = 2 * N - 1
    c = full_corr(h, N)  # length L
    max_c = c.max()
    w = c * np.exp(tau * (c - max_c))
    w /= w.sum()
    
    # g[i] = sum_k w[k] * ((1-h[i-k]) - h[i+k])
    # For each k (lag), the contribution to g[i] is:
    #   w[k] * ((1-h[i-k]) - h[i+k])
    # This is a weighted sum of shifted (1-h) and -shifted h
    
    # Vectorize using numpy:
    # For lag k (index in w is k+N-1):
    #   (1-h[i-k]) for i-k in [0,N): i in [k, k+N) → shift (1-h) by k
    #   h[i+k] for i+k in [0,N): i in [-k, N-k) → shift h by -k
    
    g = np.zeros(N)
    for lag in range(-(N-1), N):
        wval = w[lag + N - 1]
        if abs(wval) < 1e-15:
            continue
        # (1-h)[i-lag]: valid when i-lag in [0,N), i.e., i in [lag, lag+N)
        i_lo = max(0, lag)
        i_hi = min(N, lag + N)
        src = slice(i_lo - lag, i_hi - lag)  # indices in (1-h)
        g[i_lo:i_hi] += wval * (1 - h[src])
        # -h[i+lag]: valid when i+lag in [0,N), i.e., i in [-lag, N-lag)
        i_lo2 = max(0, -lag)
        i_hi2 = min(N, N - lag)
        src2 = slice(i_lo2 + lag, i_hi2 + lag)
        g[i_lo2:i_hi2] -= wval * h[src2]
    
    return g, c
```

import time

def full_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(full_corr(h).max()) * (2.0 / N)

def soft_gradient(h, N, tau):
    L = 2 * N - 1
    c = full_corr(h)
    mc = c.max()
    exp_c = np.exp(tau * (c - mc))
    w = c * exp_c
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(L, 1.0 / L)
    else:
        w /= ws
    g = np.zeros(N)
    for lag in range(-(N-1), N):
        wv = w[lag + N - 1]
        if abs(wv) < 1e-14:
            continue
        lo1 = max(0, lag)
        hi1 = min(N, lag + N)
        if lo1 < hi1:
            g[lo1:hi1] += wv * (1.0 - h[lo1-lag:hi1-lag])
        lo2 = max(0, -lag)
        hi2 = min(N, N - lag)
        if lo2 < hi2:
            g[lo2:hi2] -= wv * h[lo2+lag:hi2+lag]
    return g, c

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0, 1)

def to_binary(h, N):
    """Project to binary with exact N/2 ones."""
    hb = h.copy()
    n1 = 0
    # Greedy: fill from highest values
    order = np.argsort(-hb)
    hb[:] = 0
    hb[order[:N//2]] = 1
    return hb

def guided_binary(h, N, deadline, top_m=80, max_sweeps=10000, tau_g=4.0):
    h = h.astype(float).copy()
    best_c = float(full_corr(h).max())
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = full_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        g, _ = soft_gradient(h, N, tau=tau_g)
        m = min(top_m, len(ones), len(zeros))
        tp = ones[np.argsort(g[ones])[:m]]
        tq = zeros[np.argsort(-g[zeros])[:m]]
        # Precompute shift contributions
        L = 2 * N - 1
        lags = np.arange(L) - (N - 1)
        A = np.zeros((m, L))
        B = np.zeros((m, L))
        for idx in range(m):
            p = tp[idx]
            for ki in range(L):
                k = lags[ki]
                v = 0.0
                a = p - k
                if 0 <= a < N:
                    v += h[a]
                b = p + k
                if 0 <= b < N:
                    v += h[b]
                A[idx, ki] = v
        for idx in range(m):
            q = tq[idx]
            for ki in range(L):
                k = lags[ki]
                v = 0.0
                a = q - k

                b = q + k

                    v += h[b]

        # Shape: (m, m, L)

        new_max = np.max(c[None, None, :] + diffs, axis=2)  # (m, m)
        deltas = new_max - P
        min_idx = np.unravel_index(np.argmin(deltas), deltas.shape)
