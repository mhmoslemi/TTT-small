def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

    it = 0

        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -top_k)[-top_k:]

        h = h_new
        it += 1

            lr *= 0.995
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= top_k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.994
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.random.permutation(len(idx))[:n1 - nh]]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        add = idx[np.random.permutation(len(idx))[:nh - n1]]
        h[add] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Greedy binary (main exploitation)
    d1 = min(deadline - 420, t0 + 190)
    h_bin = (h0 > 0.5).astype(float)
    h_s, _ = greedy_binary(h_bin, N0, d1, max_sweeps=900, sample_n=200, flat_lambda=0.03)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Simulated annealing for exploration
    if time.time() < deadline - 280:
        d2 = min(deadline - 260, time.time() + 100)
        h_sa = (best_h > 0.5).astype(float)
        h_sa, c5_sa = sa_binary(h_sa, N0, d2, T0=0.2, T1=0.003, max_steps=2000)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()
        # Second SA run from perturbed state
        if time.time() < deadline - 220:
            h_pert = (best_h > 0.5).astype(float)
            idx_flip = np.random.choice(len(h_pert), size=15, replace=False)
            h_pert[idx_flip] = 1.0 - h_pert[idx_flip]
            h_pert = fix_sum_binary(h_pert, N0)
            d2b = min(deadline - 200, time.time() + 50)
            h_sa2, c5_sa2 = sa_binary(h_pert, N0, d2b, T0=0.15, T1=0.002, max_steps=1000)
            if c5_sa2 < best_c5:
                best_c5, best_h = c5_sa2, h_sa2.copy()

    # Phase 3: Nesterov continuous optimization
    if time.time() < deadline - 160:
        d3 = min(deadline - 140, time.time() + 70)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d3, top_k=40, lr0=0.014, beta=0.9)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 4: Greedy binary from best state
    if time.time() < deadline - 80:
        d4 = min(deadline - 60, time.time() + 100)
        h_bin4 = (best_h > 0.5).astype(float)
        h_s4, _ = greedy_binary(h_bin4, N0, d4, max_sweeps=500, sample_n=200, flat_lambda=0.05)
        c5_4 = compute_c5(h_s4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_s4.copy()

    # Phase 5: Upscale to 2x resolution
    N_up = N0 * 2
    if time.time() < deadline - 40:
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_up = (h_interp > 0.5).astype(float)
        h_up = fix_sum_binary(h_up, N_up)
        d5 = min(deadline - 15, time.time() + 55)
        h_rt, _ = greedy_binary(h_up, N_up, d5, max_sweeps=250, sample_n=100, flat_lambda=0.03)
        c5_rt = compute_c5(h_rt, N_up)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final Nesterov polish
    Nf = len(best_h)
    if time.time() < deadline - 4:
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.015
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_opt(hF, Nf, deadline - 1.5, top_k=60, lr0=0.008, beta=0.92)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 7: Final binary cleanup
    if deadline - 1 > time.time() + 4:
        hF2 = (best_h > 0.5).astype(float)
        hF2 = fix_sum_binary(hF2, len(hF2))
        hF2, _ = greedy_binary(hF2, len(hF2), deadline - 0.5, max_sweeps=30, sample_n=100, flat_lambda=0.03)
        c5F2 = compute_c5(hF2, len(hF2))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
