import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        np.random.shuffle(ones)
        h[ones[:n1-nh]] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        np.random.shuffle(zeros)
        h[zeros[:nh-n1]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def sa_search(h_in, N, deadline, T0=0.4, alpha=0.9998, max_no_improve=500):
    """Simulated annealing for binary swaps with FFT-based correlation."""
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    cur_c = best_c
    T = T0
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    no_improve = 0
    
    while time.time() < deadline and no_improve < max_no_improve:
        ip = np.random.randint(len(ones))
        iq = np.random.randint(len(zeros))
        p, q = int(ones[ip]), int(zeros[iq])
        
        h[p], h[q] = 0.0, 1.0
        new_corr = fft_corr(h)
        new_c = float(new_corr.max())
        
        if new_c <= cur_c:
            # Accept
            corr = new_corr
            # Update sets
            ones[ip] = q  # p leaves ones, q joins
            # Wait, need to be careful. ones and zeros are index arrays.
            # p was in ones at position ip, q was in zeros at position iq.
            # After swap: p is now 0, q is now 1.
            # ones: replace p with q → ones[ip] = q
            # zeros: replace q with p → zeros[iq] = p
            ones[ip] = q
            zeros[iq] = p
            cur_c = new_c
            no_improve = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Metropolis
            delta = new_c - cur_c
            if np.random.rand() < np.exp(-delta / max(T, 1e-12)):
                corr = new_corr
                ones[ip] = q
                zeros[iq] = p
                cur_c = new_c
                no_improve += 1
            else:
                h[p], h[q] = 1.0, 0.0
                no_improve += 1
        
        T *= alpha
        if no_improve > 200:
            T = max(T, 0.05)
            no_improve = 0
    
    return best_h, best_c * (2.0 / N)

def greedy_sweep(h_in, N, deadline):

    h = fix_binary(h_in.copy(), N // 2)

        idx = np.random.choice(len(ones), n_sample, replace=False)
        sample_ones = ones[idx]
        
        best_delta = best_c + 1.0
        best_p, best_q = -1, -1
        
        for p in sample_ones:
            # For each zero q, compute max(corr + deltas[p] - deltas[q])
            # = max over k of (corr[k] + deltas[p,k] - deltas[q,k])

            # Now subtract each deltas[q] and find min of max

                best_q = int(zeros[mi])
