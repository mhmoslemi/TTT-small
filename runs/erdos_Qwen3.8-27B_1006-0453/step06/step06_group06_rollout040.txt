import numpy as np
import time

def compute_c5(h):
    N = len(h)
    corr = np.correlate(h.astype(float), 1.0-h, mode="full")
    return float(corr.max()) * (2.0/N)

def proj_to_constraint(h, target):
    """Project to [0,1] with sum = target (bisection)."""
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def binary_swap_delta(h, N, p, q, s_arr, corr):
    """Compute new corr after swapping h[p]:1->0, h[q]:0->1."""
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps = (ps>=0)&(ps<N); vpm = (pm>=0)&(pm<N)
    vqs = (qs>=0)&(qs<N); vqm = (qm>=0)&(qm<N)
    d = np.zeros(2*N-1)
    d[vps] -= (1.0 - h[np.clip(ps,0,N-1)[vps]])
    d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
    d[vqs] += (1.0 - h[np.clip(qs,0,N-1)[vqs]])
    d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
    return corr + d

def greedy_binary(h, N, deadline, sample_frac=0.3):
    """Greedy local search: repeatedly find best improving swap."""
    h = h.copy()
    s_arr = np.arange(-(N-1), N)
    corr = np.correlate(h.astype(float), 1.0-h, mode="full")
    curr_max = float(corr.max())
    best_h = h.copy(); best_max = curr_max
    no_improve = 0
    
    while time.time() < deadline:
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        improved = False
        
        # Sample a subset of swaps
        n_try = max(20, int(len(ones)*len(zeros)*sample_frac))
        for _ in range(n_try):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            new_corr = binary_swap_delta(h, N, p, q, s_arr, corr)
            new_max = float(new_corr.max())
            if new_max < curr_max:
                h[p]=0; h[q]=1; corr=new_corr; curr_max=new_max
                improved = True
                if curr_max < best_max:
                    best_max = curr_max; best_h = h.copy()
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
        if no_improve > 50:
            break
    
    return best_h, best_max

def anneal_binary(h, N, deadline, T0=0.04, cool=0.9997):

            corr = np.correlate(h.astype(float), 1.0-h, mode="full")

    return best_h, best_max

def subgrad_continuous(h, N, deadline, beta=50.0, lr0=0.01):
    """Subgradient method with soft-max weighting."""
    h = proj_to_constraint(h.copy(), N/2.0)
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1.0-h, mode="full")))
    lr = lr0
    center = N-1
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0-h, mode="full")
        curr_val = float(corr.max())
        if curr_val < best_val:
            best_val = curr_val; best_h = h.copy()
        
        # Soft-max weights
        w = np.exp(beta * (corr - curr_val))
        w /= w.sum()
        
        # Gradient
        grad = np.zeros(N)
        # Vectorized: for each lag, add weighted gradient
        active = w > 1e-10
        for k in np.where(active)[0]:
            s = int(k - center)
            if s == 0:
                grad += w[k] * (1.0 - 2.0*h)
            elif s > 0:
                grad[:N-s] += w[k] * (1.0 - h[s:])
                grad[s:] += w[k] * (-h[:N-s])
            else:
                s2 = -s
                grad[s2:] += w[k] * (1.0 - h[:N-s2])
                grad[:N-s2] += w[k] * (-h[s2:])
        
        h = proj_to_constraint(h - lr * grad, N/2.0)
        lr *= 0.999
        
        if lr < 0.0001:
            break
    
    return best_h, best_val * (2.0/N)

    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

- For each top lag ℓ, this constrains p and q to certain positions
- Enumerate feasible (p,q) pairs that reduce at least one top lag

Let me think about this more carefully. For lag ℓ:
delta_p[ℓ] = -(1-h[p+ℓ]) + h[p-ℓ] (with boundary conditions)
delta_q[ℓ] = +(1-h[q+ℓ]) - h[q-ℓ]

So corr_new[ℓ] = corr[ℓ] - (1-h[p+ℓ]) + h[p-ℓ] + (1-h[q+ℓ]) - h[q-ℓ]

For this to decrease: (1-h[p+ℓ]) - h[p-ℓ] - (1-h[q+ℓ]) + h[q-ℓ] > 0

def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * (2.0 / N)

def proj_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def swap_corr(h, N, p, q, corr, s_arr):
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2 * N - 1)
    if vps.any(): d[vps] -= (1.0 - h[np.clip(ps, 0, N-1)[vps]])
    if vpm.any(): d[vpm] += h[np.clip(pm, 0, N-1)[vpm]]
    if vqs.any(): d[vqs] += (1.0 - h[np.clip(qs, 0, N-1)[vqs]])
    if vqm.any(): d[vqm] -= h[np.clip(qm, 0, N-1)[vqm]]
    return corr + d

    h = proj_constraint(h, N / 2.0)
    best_h = h.copy()

k=-1: a[1]*v[0] + a[2]*v[1] = 0*0 + 0*0 = 0
k=0: a[0]*v[0] + a[1]*v[1] + a[2]*v[2] = 0+0+0 = 0
k=1: a[0]*v[1] + a[1]*v[2] = 0+0 = 0
k=2: a[0]*v[2] = 1*1 = 1

So np.correlate([1,0,0], [0,0,1], "full") = [0, 0, 0, 0, 1]

The "center" (lag 0) is at index len(v)-1 = 2. So:
corr[0] corresponds to lag -(len(v)-1) = -2
corr[2] corresponds to lag 0
corr[4] corresponds to lag 2

And corr[lag] = sum_n a[n] * v[n+lag] for valid n.

In our problem: a = h, v = 1-h.
corr[lag] = sum_n h[n] * (1-h[n+lag]) for valid n, n+lag.

This matches what I had before. The "full" array has 2N-1 elements, indexed 0 to 2N-2, with lag = index - (N-1).

def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * (2.0 / N)

def proj_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def make_binary(h, N):
    hb = (h >= 0.5).astype(float)
    n1 = int(hb.sum())
    target = N // 2
    if n1 > target:
        ones = np.where(hb == 1)[0]
        # Remove ones that are "least committed" (closest to 0.5)
        order = np.argsort(h[ones][:n1-target]) if n1-target <= len(ones) else np.random.choice(ones, n1-target, replace=False)
        hb[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(hb == 0)[0]
        hb[np.random.choice(zeros, target - n1, replace=False)] = 1
    return hb

def resample(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

    T = T0
    no_imp = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]

            q = int(zeros[np.random.randint(len(zeros))])
            ps = p + s_arr; pm = p - s_arr
            qs = q + s_arr; qm = q - s_arr
            vps = (ps >= 0) & (ps < N)
            vpm = (pm >= 0) & (pm < N)
            vqs = (qs >= 0) & (qs < N)
            vqm = (qm >= 0) & (qm < N)
            d = np.zeros(2 * N - 1)

            d[vqs] += (1.0 - h[qs[vqs]])
            d[vqm] -= h[qm[vqm]]
            new_corr = corr + d
            new_max = float(new_corr.max())
            delta = new_max - curr_max
            
            if delta < -1e-9:
                accept = True
            elif use_anneal and T > 0.003:

            else:
                accept = False

                corr = new_corr; curr_max = new_max
                improved_this_round = True
                if new_max < best_max - 1e-12:

    best_h = h.copy()

    lr = lr0
    center = N - 1
    t_start = time.time()

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        curr_val = float(corr.max())
        if curr_val < best_val - 1e-10:
            best_val = curr_val; best_h = h.copy()
        
        frac = min((time.time() - t_start) / dur, 1.0)
        beta = beta0 * (beta1 / beta0) ** frac
        shifted = corr - curr_val
        w = np.exp(beta * shifted)
        wsum = w.sum()
        if wsum < 1e-30: break
        w /= wsum
        
        grad = np.zeros(N)
        active = w > (w.max() * 0.005)
        for k in np.where(active)[0]:
            s = int(k - center)
            if s == 0:
                grad += w[k] * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += w[k] * (1.0 - h[s:])
                grad[s:] += w[k] * (-h[:N-s])

        lr *= 0.9997
    
    return best_h, best_val * (2.0 / N)
