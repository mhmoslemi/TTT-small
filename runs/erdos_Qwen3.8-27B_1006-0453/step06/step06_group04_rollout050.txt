    if c5n < best_c5:
        best_c5, best_h = c5n, hn.copy()
    # Also try binary version of continuous result
    hb_c = binarize(best_h, N0)
    c5_bc = compute_c5(hb_c, N0)
    if c5_bc < best_c5:
        best_c5, best_h = c5_bc, hb_c.copy()

    # Phase 3: Multi-resolution - up to 760 (200s)
    N_high = 760
    d3 = max(d2, t0) + 200
    d3 = min(deadline - 200, d3)
    if N_high > N0:
        h_hi = interp_to(best_h, N_high)
        h_hi_b = binarize(h_hi, N_high)
        sub_dl1 = min(d3, time.time() + 90)
        h_hi_b, c5_hi1 = binary_opt(h_hi_b, N_high, sub_dl1, n_tries=14, kick_freq=60)
        h_hi_c = h_hi_b + np.random.randn(N_high) * 0.03
        h_hi_c = project_sum(h_hi_c, N_high / 2.0)
        h_hi_c, c5_hi2 = nesterov_opt(h_hi_c, N_high, d3, k_top=150, lr0=0.02, temp0=0.003)
        c5_hi = min(compute_c5(h_hi_b, N_high), c5_hi2)
        if c5_hi < best_c5:
            if c5_hi2 < compute_c5(h_hi_b, N_high):
                best_h_hi = h_hi_c
            else:
                best_h_hi = h_hi_b
            best_c5 = c5_hi
            # Interpolate back to N0
            best_h = interp_to(best_h_hi, N0)
            best_h_b = binarize(best_h, N0)
            c5_back = min(compute_c5(best_h, N0), compute_c5(best_h_b, N0))
            if compute_c5(best_h_b, N0) < compute_c5(best_h, N0):
                best_h = best_h_b
            best_c5 = compute_c5(best_h, N0)

    # Phase 4: Binary polish at N0 (140s)
    d4 = max(d3, t0) + 140
    d4 = min(deadline - 80, d4)
    hb4, cb4 = binary_opt(best_h, N0, d4, n_tries=25, kick_freq=90)
    c5b4 = compute_c5(hb4, N0)
    if c5b4 < best_c5:
        best_c5, best_h = c5b4, hb4.copy()

    # Phase 5: Final continuous polish (remaining)
    d5 = deadline - 3
    if time.time() < d5 - 10:
        hF = best_h + np.random.randn(N0) * 0.02
        hF = project_sum(hF, N0 / 2.0)
        hF, c5F = nesterov_opt(hF, N0, d5, k_top=250, lr0=0.012, temp0=0.002,
                               beta0=0.93, beta_max=0.98, restart_each=150)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
        hbF = binarize(best_h, N0)
        c5_hbF = compute_c5(hbF, N0)
        if c5_hbF < best_c5:
            best_c5, best_h = c5_hbF, hbF.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
