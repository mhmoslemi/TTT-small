import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft(1.0 - h.astype(float), L)
    return np.fft.irfft(H * G.conj(), L)[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def fix_binary(h, N):
    h = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[idx[np.random.choice(len(idx), n1 - target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[idx[np.random.choice(len(idx), target - n1, replace=False)]] = 1
    return h

def move_delta(h, N, p, q, s_arr, center):
    d = np.zeros(2 * N - 1)
    ps, pm = p + s_arr, p - s_arr
    qs, qm = q + s_arr, q - s_arr
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    if np.any(vpm):
        d[center + s_arr[vpm]] -= (1.0 - h[pm[vpm]])
    if np.any(vqm):
        d[center + s_arr[vqm]] += (1.0 - h[qm[vqm]])
    if np.any(vps):
        d[center + s_arr[vps]] += h[ps[vps]]
    if np.any(vqs):
        d[center + s_arr[vqs]] -= h[qs[vqs]]
    return d

def anneal(h, N, deadline, T0=0.05, cool=0.9995, evals_per_iter=1, max_no_improve=2000):

    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    center = N - 1
    T = T0
    no_imp = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline and T > 0.0005:
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        if p == q:
            T *= cool
            continue
        d = move_delta(h, N, p, q, s_arr, center)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - current_pk

            h[p] = 0
            h[q] = 1
            corr = nc
            current_pk = npk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if npk < best_pk:

                best_h = h.copy()
                no_imp = 0
            else:

        else:
