    h = np.clip(h, 0.0, 1.0)

    # D[p, l] = effect on c[l] of changing h[p] from 1 to 0

    vp = (pp >= 0) & (pp < N)

    c = fft_corr(h)
    # softmax weights
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    s = w.sum()
    w = w / s

    D = compute_D(h, N)
    grad = D.T @ w  # (N,)
    return grad, c.max()

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def softmax_optimize(h, N, deadline, beta0=8.0, beta_max=80.0, lr0=0.012, b1=0.85, b2=0.998):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    eps = 1e-8
    while time.time() < deadline:
        beta = min(beta0 * (1.0 + 0.05 * t), beta_max)
        t += 1
        lr = lr0 / (1.0 + 0.005 * t)
        c = fft_corr(h)
        cs = c - c.max()
        w = np.exp(beta * cs)
        sw = w.sum()
        if sw > 0:
            w = w / sw
        D = compute_D(h, N)
        grad = D.T @ w
        grad -= (grad * h).sum() / max(h.sum(), 1)
        bc1 = 1 - b1 ** t
        bc2 = 1 - b2 ** t
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        h = project_sum(h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps), target)
        c5n = compute_c5(h, N)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h.copy()
    return best_h, best_c5

def binary_search(h, N, deadline, max_sweeps=600, anneal=0.0, sample_n=None):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        oi = np.where(h > 0.5)[0]
        zi = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[oi[:nh]] = 1
        else:
            h[oi] = 1
            h[zi[:nh - n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > (100 if anneal == 0 else 60):
            if anneal == 0:
                break
            if stall > 150:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        if sample_n and len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        bd, bpq = 0.0, None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = max(0, stall - 20)
        else:

    return h, best_c

def to_binary(h_cont, N):
    hb = (h_cont > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == N // 2:
        return hb
    hb[:] = 0.0
    oi = np.where(h_cont > 0.5)[0]
    zi = np.where(h_cont <= 0.5)[0]
    if n1 > N // 2:
        order = np.argsort(-h_cont[oi])
        hb[oi[order[:N//2]]] = 1
    else:
        hb[oi] = 1
        order = np.argsort(-h_cont[zi])
        hb[zi[order[:N//2 - n1]]] = 1
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Softmax continuous optimization (multiple beta starts)
    d1 = min(deadline - 500, t0 + 180)
    for beta_s in [6.0, 15.0]:
        if time.time() > d1:
            break
        hc = best_h.copy() + np.random.randn(N0) * 0.015
        hc = project_sum(hc, N0 / 2.0)
        sub_d = min(d1, time.time() + 80)
        h_c, c5_c = softmax_optimize(hc, N0, sub_d, beta0=beta_s, beta_max=beta_s * 8, lr0=0.010)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()

    # Phase 2: Binary refinement from best continuous
    h_b = to_binary(best_h, N0)
    d2 = min(deadline - 380, time.time() + 160)
    h_b, _ = binary_search(h_b, N0, d2, max_sweeps=1500, anneal=0.0)
    c5_b = compute_c5(h_b, N0)
    if c5_b < best_c5:
        best_c5, best_h = c5_b, h_b.copy()

    # Phase 3: Second round softmax with perturbation
    if time.time() < deadline - 280:
        d3 = min(deadline - 260, time.time() + 100)
        hc = to_binary(best_h, N0) + np.random.randn(N0) * 0.025
        hc = project_sum(hc, N0 / 2.0)
        h_c, c5_c = softmax_optimize(hc, N0, d3, beta0=12.0, beta_max=120.0, lr0=0.008)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()
        h_b3 = to_binary(best_h, N0)
        d3b = min(deadline - 250, time.time() + 60)
        h_b3, _ = binary_search(h_b3, N0, d3b, max_sweeps=500, anneal=0.0)
        c5_b3 = compute_c5(h_b3, N0)
        if c5_b3 < best_c5:
            best_c5, best_h = c5_b3, h_b3.copy()

    # Phase 4: Annealed binary search
    if time.time() < deadline - 200:
        h_b4 = to_binary(best_h, N0)
        d4 = min(deadline - 180, time.time() + 100)
        h_b4, _ = binary_search(h_b4, N0, d4, max_sweeps=400, anneal=0.06)
        c5_4 = compute_c5(h_b4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_b4.copy()

    # Phase 5: Multi-restart perturbation binary search
    if time.time() < deadline - 120:
        d5 = min(deadline - 100, time.time() + 100)
        h_base = to_binary(best_h, N0)
        for attempt in range(8):
            if time.time() > d5:
                break
            hp = h_base.copy()
            n_flip = np.random.randint(2, 7)
            oi = np.where(hp == 1)[0]
            zi = np.where(hp == 0)[0]
            for _ in range(n_flip):
                p = int(np.random.randint(len(oi)))
                q = int(np.random.randint(len(zi)))
                hp[oi[p]], hp[zi[q]] = 0.0, 1.0
                oi = np.where(hp == 1)[0]
                zi = np.where(hp == 0)[0]
            sub_dl = min(d5, time.time() + 12)
            h_pt, _ = binary_search(hp, N0, sub_dl, max_sweeps=120, anneal=0.03)
            c5_pt = compute_c5(h_pt, N0)
            if c5_pt < best_c5:
                best_c5, best_h = c5_pt, h_pt.copy()
                h_base = h_pt.copy()

    # Phase 6: Multi-resolution (try 760)
    if time.time() < deadline - 60:
        N1 = 760
        x0 = np.linspace(0, 2, N0, endpoint=False)
        x1 = np.linspace(0, 2, N1, endpoint=False)
        h_in = np.interp(x1, x0, best_h)
        h_b1 = to_binary(h_in, N1)
        d6 = min(deadline - 40, time.time() + 50)
        h_b1, _ = binary_search(h_b1, N1, d6, max_sweeps=200, sample_n=600, anneal=0.0)
        c5_760 = compute_c5(h_b1, N1)
        if c5_760 < best_c5:
            best_c5, best_h = c5_760, h_b1.copy()

    # Phase 7: Final softmax polish on best
    if time.time() < deadline - 15:
        Nf = len(best_h)
        hf = best_h.copy() + np.random.randn(Nf) * 0.008
        hf = project_sum(hf, Nf / 2.0)
        d7 = min(deadline - 8, time.time() + 30)
        hf, c5_f = softmax_optimize(hf, Nf, d7, beta0=20.0, beta_max=200.0, lr0=0.005)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, hf.copy()

    # Phase 8: Final binary cleanup
    if time.time() < deadline - 2:
        hF = to_binary(best_h, len(best_h))
        d8 = deadline - 1
        hF, _ = binary_search(hF, len(best_h), d8, max_sweeps=80, anneal=0.0)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
