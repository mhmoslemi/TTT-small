def lp_phase_a(v, N):
    # Given v (fixed), find h to minimize max_k sum_i v_shifted_k[i] * h[i]
    h = cp.Variable(N)
    t = cp.Variable()
    
    # Constraints: for each k, sum_i P_k[i] * h_i <= t
    # P_k[i] = v[i + k - (N-1)] for valid i
    # This is the cross-correlation structure
    
    # For lag j = k-(N-1):
    # If j >= 0: sum_{i=0}^{N-1-j} v[i+j] * h[i] <= t
    # If j < 0: sum_{i=-j}^{N-1} v[i+j] * h[i] <= t

from scipy.optimize import linprog

def lp_phase_a_csc(v, N):
    # Build sparse constraint matrix for Phase A
    # Variables: [h_0, ..., h_{N-1}, t] (N+1 vars)
    # Minimize: t → c = [0,...,0, 1]
    # Inequality: A_ub @ x <= b_ub
    #   For each lag j: P_j^T h <= t → P_j^T h - t <= 0
    # Equality: A_eq @ x == b_eq
    #   sum(h) = N/2, 1*t = 0
    # Bounds: 0 <= h_i <= 1, t free (or t >= 0)
    
    from scipy.sparse import csr_matrix, eye as speye
    
    n_vars = N + 1
    n_ineq = 2*N - 1
    
    # Build A_ub: (2N-1) × (N+1)
    # Row for lag j: [P_j, -1]
    
    rows = []
    for j in range(-(N-1), N):
        row = np.zeros(N+1)
        row[-1] = -1.0  # -t
        if j >= 0:
            row[:N-j] = v[j:]  # P_j[i] = v[i+j] for i=0..N-1-j
        else:
            m = -j
            row[m:] = v[:N-m]  # P_j[i] = v[i-m] for i=m..N-1
        rows.append(row)
    A_ub = np.array(rows)
    b_ub = np.zeros(n_ineq)
    
    c = np.zeros(N+1)
    c[-1] = 1.0
    
    # Equality: sum(h) = N/2, t unconstrained by equality
    A_eq = np.zeros((1, N+1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([N/2.0])
    
    bounds = [(0, 1)] * N + [(0, None)]  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return res.x[:N]

def fft_corr(h):
    N = len(h)
    L = 2*N
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)
