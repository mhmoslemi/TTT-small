    corr = fft_corr(h)

def nesterov_opt(h, N, deadline, k=80, lr0=0.016):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    L = 2 * N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            Lg = int(ai - center)
            if Lg == 0:
                grad += (1.0 - 2.0 * y)
            elif Lg > 0:
                grad[Lg:] += (1.0 - y[:N - Lg])
                grad[:N - Lg] -= y[Lg:]
            else:
                M = -Lg
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        s = float(h_new.sum())
        if abs(s - target) > 1e-10:
            lo, hi = -1.0, 1.0

                    lo = mid
                else:
                    hi = mid

        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it * 0.00008)
