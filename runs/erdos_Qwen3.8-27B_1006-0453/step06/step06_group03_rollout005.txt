import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def full_swap_search(h, N, top_m=25, max_iters=3000, deadline=None, t0=None, anneal_T=0.0):
    h = h.astype(float).copy()
    center = N - 1
    dx = 2.0 / N
    best_c5 = float(fft_corr(h).max()) * dx
    it = 0
    
    while True:
        if deadline is not None and time.time() > deadline:
            break
        if it >= max_iters:
            break
        it += 1
        
        c = fft_corr(h)
        P = float(c.max()) * dx
        if P < best_c5:
            best_c5 = P
        
        # Get top correlations
        top_idx = np.argsort(c)[-top_m:]
        Ls = (top_idx - center).astype(int)
        c_top = c[top_idx]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Precompute vectors
        # For each one position p: vec1[i,j] = h[clip(ones[i]+Ls[j])] + h[clip(ones[i]-Ls[j])]
        n1, n0 = len(ones), len(zeros)
        m = top_m
        
        # Build index arrays with clipping
        idx1_plus = np.clip(ones[:, None] + Ls[None, :], 0, N-1)
        idx1_minus = np.clip(ones[:, None] - Ls[None, :], 0, N-1)
        idx0_plus = np.clip(zeros[:, None] + Ls[None, :], 0, N-1)
        idx0_minus = np.clip(zeros[:, None] - Ls[None, :], 0, N-1)
        
        # But we need to handle out-of-bounds correctly (contribution should be 0)
        valid1_plus = (ones[:, None] + Ls[None, :]) >= 0
        valid1_plus &= (ones[:, None] + Ls[None, :]) < N
        valid1_minus = (ones[:, None] - Ls[None, :]) >= 0
        valid1_minus &= (ones[:, None] - Ls[None, :]) < N
        valid0_plus = (zeros[:, None] + Ls[None, :]) >= 0
        valid0_plus &= (zeros[:, None] + Ls[None, :]) < N
        valid0_minus = (zeros[:, None] - Ls[None, :]) >= 0
        valid0_minus &= (zeros[:, None] - Ls[None, :]) < N
        
        h1p = np.where(valid1_plus, h[idx1_plus], 0.0)
        h1m = np.where(valid1_minus, h[idx1_minus], 0.0)
        h0p = np.where(valid0_plus, h[idx0_plus], 0.0)
        h0m = np.where(valid0_minus, h[idx0_minus], 0.0)
        
        vec1 = h1p + h1m  # shape (n1, m)
        vec0 = h0p + h0m  # shape (n0, m)

        new_max = new_corr.max(axis=2)  # shape (n1, n0)

    final_c5 = compute_c5(h_best)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def fix_sum_binary(h, N):
    h = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0.0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1.0
    return h
