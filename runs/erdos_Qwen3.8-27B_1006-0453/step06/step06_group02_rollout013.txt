def fft_corr(h):

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary_sum(h, N):
    nh = N // 2
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[:n1 - nh]]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[:nh - n1]]] = 1.0
    return h

def delta_minus(h, p, N):
    """Delta in corr when h[p] goes 1->0."""
    d = np.zeros(2 * N - 1)
    k_start, k_end = p, N - 1 + p
    v_idx = N - 1 + p - np.arange(k_start, k_end + 1)
    d[k_start:k_end + 1] -= (1.0 - h[v_idx])
    k2_start, k2_end = N - 1 - p, 2 * N - 2 - p
    h_idx = p - N + 1 + np.arange(k2_start, k2_end + 1)
    d[k2_start:k2_end + 1] -= h[h_idx]
    d[N - 1] += 1.0
    return d

def delta_plus(h, q, N):
    """Delta in corr when h[q] goes 0->1."""
    d = np.zeros(2 * N - 1)
    k_start, k_end = q, N - 1 + q
    v_idx = N - 1 + q - np.arange(k_start, k_end + 1)
    d[k_start:k_end + 1] += (1.0 - h[v_idx])
    k2_start, k2_end = N - 1 - q, 2 * N - 2 - q
    h_idx = q - N + 1 + np.arange(k2_start, k2_end + 1)
    d[k2_start:k2_end + 1] += h[h_idx]
    d[N - 1] -= 1.0
    return d

def local_search(h, N, deadline, max_iters=200000, sample_p=40, sample_q=40):
    h = h.astype(float).copy()
    target = N / 2.0
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    no_improve = 0
    temp = 0.05
    t0 = time.time()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cur_max = float(np.max(corr))
        if cur_max < best_c - 1e-12:
            best_c = cur_max
            no_improve = 0
        else:
            no_improve += 1
            temp = max(0.5 * temp, 0.001)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        np_s = min(sample_p, len(ones))
        nq_s = min(sample_q, len(zeros))
        p_candidates = ones[np.random.choice(len(ones), np_s, replace=False)]
        q_candidates = zeros[np.random.choice(len(zeros), nq_s, replace=False)]
        
        best_score = cur_max + 1e-6
        best_pq = None
        
        for p in p_candidates:
            dm = delta_minus(h, p, N)
            corr_minus = corr + dm
            for q in q_candidates:
                if q == p:
                    continue
                dp = delta_plus(h, q, N)
                new_max = float(np.max(corr_minus + dp))
                if new_max < best_score:
                    best_score = new_max
                    best_pq = (int(p), int(q))
        
        # Annealing: accept worse moves with probability
        if best_pq is None:
            if np.random.random() < min(temp, 0.3) and no_improve > 20:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                if p != q:
                    h[p], h[q] = 0.0, 1.0
                    corr = fft_corr(h)
            if no_improve > 500:
                break
            continue
        
        p, q = best_pq
        h[p], h[q] = 0.0, 1.0
        corr = fft_corr(h)
    
    return h, best_c

def nesterov_opt(h, N, deadline, k_top=40, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.993
            beta = min(0.95, 0.85 + it * 0.0004)
    return best_h, best_pk * (2.0 / N)

def upscale(h, factor):
    N0 = len(h)
    N1 = N0 * factor
    x0 = np.linspace(0, 2, N0, endpoint=False)
    x1 = np.linspace(0, 2, N1, endpoint=False)
    return np.interp(x1, x0, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Intensive binary local search from initial
    d1 = min(deadline - 500, t0 + 250)
    h_bin = fix_binary_sum(h0, N0)
    h_s, _ = local_search(h_bin, N0, d1, max_iters=200000, sample_p=50, sample_q=50)
    c5_s = compute_c5(h_s)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    
    # Phase 2: Nesterov continuous optimization
    if time.time() < deadline - 350:
        d2 = min(deadline - 330, time.time() + 120)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k_top=50, lr0=0.02)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    
    # Phase 3: Binary search from continuous result
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 160)
        h_bin3 = fix_binary_sum(best_h, N0)
        h_s3, _ = local_search(h_bin3, N0, d3, max_iters=150000, sample_p=60, sample_q=60)
        c5_3 = compute_c5(h_s3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()
    
    # Phase 4: Upscale to 2x resolution
    N_up = N0 * 2
    if time.time() < deadline - 100:
        h_up = upscale(best_h, 2)
        h_bt = fix_binary_sum(h_up, N_up)
        d4 = min(deadline - 60, time.time() + 100)
        h_rt, _ = local_search(h_bt, N_up, d4, max_iters=80000, sample_p=35, sample_q=35)
        c5_rt = compute_c5(h_rt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
    
    # Phase 5: Final Nesterov polish at best resolution
    Nf = len(best_h)
    if time.time() < deadline - 10:
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.015
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_opt(hF, Nf, deadline - 3, k_top=60, lr0=0.01)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    
    # Phase 6: Final binary cleanup
    if time.time() < deadline - 1:
        Nf2 = len(best_h)
        hF2 = fix_binary_sum(best_h, Nf2)
        hF2, _ = local_search(hF2, Nf2, deadline - 0.5, max_iters=20000, sample_p=30, sample_q=30)
        c5F2 = compute_c5(hF2)
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2.copy()
    
    # Ensure valid output
    Nf = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Fix sum if slightly off
    target = Nf / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 0.01:
        best_h = project_sum(best_h, target)
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, Nf
