import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        if len(idx) >= nh - n1:
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def binary_search(h, N, deadline, max_iters=5000, sample_sz=150, temp=0.0, kick_thresh=60, kick_sz=4):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    best_c = np.inf
    best_h = h.copy()
    C = compute_C_mat(h, N)
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        new = corr[None, None, :] + Co[None, :, :] + Cz[:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-14:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
            C[os_[mi[1]], :] = compute_C_single(h, N, os_[mi[1]])
            C[zs[mi[0]], :] = compute_C_single(h, N, zs[mi[0]])
        elif temp > 0 and np.random.random() < temp * np.exp(-stuck * 0.005):
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p] = 0
            h[q] = 1
            C[p, :] = compute_C_single(h, N, p)
            C[q, :] = compute_C_single(h, N, q)
        elif stuck >= kick_thresh:
            for _ in range(kick_sz):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1

                C[q, :] = compute_C_single(h, N, q)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stuck = 0
        else:
            break
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=80, lr0=0.012):

    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.993
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)
