def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(corr.max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, nh):

def compute_deltas(h, s, N):

    target = N / 2.0
    h = project_sum(h, target)
    
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.8
    it = 0
    stale = 0
    
    while time.time() < deadline:
        y = h + m
        y = np.clip(y, 0.0, 1.0)
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = np.clip(h, 0, 1).copy()
            stale = 0
        else:
            stale += 1
        
        # Gradient from top-k shifts
        top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for t in top_idx:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top_idx)
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        
        if it % 50 == 0:
            lr *= 0.97
            beta = min(0.95, 0.75 + it * 0.0005)
        
        if stale > 300:
            break
    
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def sa_binary(h_in, N, deadline, T0=0.05, cool=0.9997, max_sweeps=10000, n_sample=12):
    h = h_in.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        h = make_binary(h, nh)
    corr = fft_corr(h)
    s = 2.0 * h - 1.0
    best_c = float(corr.max())
    best_h = h.copy()
    cur_c = best_c
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    ndl = 2 * N - 1
    deltas = np.zeros((N, ndl))
    for i in range(N):
        deltas[i, i:i+N] = s[i]
    T = T0
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        T = T0 * (cool ** sweep)
        if T < 1e-5:
            T = 1e-5
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(n_sample, len(ones))
        p_list = np.random.choice(ones, ns, replace=False)
        zq = len(zeros)
        chunk = 4
        best_nc = cur_c
        best_p, best_q = int(p_list[0]), int(zeros[0])
        for ci in range(0, ns, chunk):
            dc = min(ci + chunk, ns)
            dp = deltas[p_list[ci:dc]].reshape(-1, 1, 1)
            dq = deltas[zeros].reshape(1, -1, 1)
            vals = corr[None, None, :] + dp - dq
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            nc = float(nm[mi])
            if nc < best_nc:
                best_nc = nc
                best_p = int(p_list[mi[0] + ci])
                best_q = int(zeros[mi[1]])
        dc_c = best_nc - cur_c
        accept = dc_c < 0 or (np.random.rand() < np.exp(-dc_c / max(T, 1e-10)))
        if accept:
            h[best_p], h[best_q] = 0.0, 1.0
            corr += deltas[best_p].copy()
            corr -= deltas[best_q].copy()
            s[best_p] = -s[best_p]
            s[best_q] = -s[best_q]
            deltas[best_p, :] = 0.0
            deltas[best_p, best_p:best_p+N] = s[best_p]
            deltas[best_q, :] = 0.0
            deltas[best_q, best_q:best_q+N] = s[best_q]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_c = float(corr.max())
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 150:
            np_ = min(4, len(ones), len(zeros))
            for _ in range(np_):
                pp = int(np.random.choice(ones))
                qq = int(np.random.choice(zeros))
                h[pp], h[qq] = 0.0, 1.0
                corr += deltas[pp].copy() - deltas[qq].copy()
                s[pp] = -s[pp]
                s[qq] = -s[qq]
                deltas[pp, :] = 0.0; deltas[pp, pp:pp+N] = s[pp]
                deltas[qq, :] = 0.0; deltas[qq, qq:qq+N] = s[qq]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_c = float(corr.max())
            no_improve = 0
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h_in, N, deadline, k=40, lr0=0.015):
    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.82
    it = 0
    stale = 0
    while time.time() < deadline:
        y = np.clip(h + m, 0.0, 1.0)
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = np.clip(h, 0, 1).copy()
            stale = 0
        else:
            stale += 1
        top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for t in top_idx:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top_idx)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.96
            beta = min(0.94, 0.78 + it * 0.0004)
        if stale > 400:
            break
    return best_h, best_pk * (2.0 / N)

def golden_h(N):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.array([int((i * phi) % N) for i in range(N // 2)])
    h = np.zeros(N)
    h[positions] = 1.0
    return h

def interleaved_h(N):
    h = np.zeros(N)
    h[::2] = 1.0
    n1 = int(h.sum())
    if n1 > N // 2:
        extra = h[::2][-(n1 - N//2):]
        for v in extra:
            idx = np.where(h == 1)[0]
            h[idx[0]] = 0.0
    return h
