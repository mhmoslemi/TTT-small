    """Best-of-batch binary local search."""
    c = fft_corr(h)
    best_int = int(c.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    
    while time.time() < deadline:
        improved = False
        for _ in range(n_batch):
            if len(ones) < 1 or len(zeros) < 1:
                break
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]

                best_h = h.copy()
                ones = np.where(h == 1)[0]

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    temp = temp0

    stall = 0

        h_prev = h
        corr = fft_corr(y)
        val = float(np.max(corr))

            best_val = val
            best_h = y.copy()

        else:

                if len(ones) >= nk and len(zeros) >= nk:
                    ps = np.random.choice(len(ones), nk, replace=False)
                    qs = np.random.choice(len(zeros), nk, replace=False)
                    for i in ps:
                        h[ones[i]] = 0
                    for i in qs:
                        h[zeros[i]] = 1
                    c = fft_corr(h)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if int(c.max()) < best_int:
                        best_int = int(c.max())
                        best_h = h.copy()
                stall = 0
    return best_h, best_int * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search at N0 (280s)
    d1 = min(deadline - 550, t0 + 280)
    hb, cb = binary_opt(h0, N0, d1, n_tries=30, kick_size=3)
    if cb < best_c5:
        best_c5, best_h = cb, hb.copy()

    # Phase 2: Continuous fine-tune at N0 (250s)
    d2 = max(d1, t0) + 250
    d2 = min(deadline - 350, d2)
    hc = best_h + np.random.randn(N0) * 0.02
    hc = project_sum(hc, N0 / 2.0)
    hn, c5n = continuous_opt(hc, N0, d2, k=80, lr0=0.012, temp0=0.0012, beta0=0.93)
    if c5n < best_c5:
        best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary at N=760 (180s)
    N1 = 760
    d3 = max(d2, t0) + 180
    d3 = min(deadline - 200, d3)
    h_interp = interp_to_N(best_h, N1)
    h_b1 = (h_interp > 0.5).astype(float)
    nh1 = N1 // 2
    n1c = int(h_b1.sum())
    if n1c > nh1:
        idx = np.where(h_b1 == 1)[0]
        idx = idx[np.random.permutation(len(idx))][:n1c - nh1]
        h_b1[idx] = 0
    elif n1c < nh1:
        idx = np.where(h_b1 == 0)[0]
        idx = idx[np.random.permutation(len(idx))][:nh1 - n1c]
        h_b1[idx] = 1
    hb1, cb1 = binary_opt(h_b1, N1, d3, n_tries=25, kick_size=4)
    if cb1 < best_c5:
        best_c5, best_h = cb1, hb1.copy()

    # Phase 4: Continuous at current best N (150s)
    d4 = max(d3, t0) + 150
    d4 = min(deadline - 80, d4)
    Nc = len(best_h)
    hc4 = best_h + np.random.randn(Nc) * 0.015
    hc4 = project_sum(hc4, Nc / 2.0)
    hn4, c5n4 = continuous_opt(hc4, Nc, d4, k=100, lr0=0.008, temp0=0.0008, beta0=0.94)
    if c5n4 < best_c5:
        best_c5, best_h = c5n4, hn4.copy()

    # Phase 5: Binary polish at N0 (80s)
    d5 = max(d4, t0) + 80
    d5 = min(deadline - 20, d5)
    best_at_N0 = interp_to_N(best_h, N0) if len(best_h) != N0 else best_h
    hb5, cb5 = binary_opt(best_at_N0, N0, d5, n_tries=35, kick_size=2)
    c5b5 = compute_c5(hb5)
    if c5b5 < best_c5:
        best_c5, best_h = c5b5, hb5.copy()

    # Phase 6: Final ultra-fine continuous (remaining time)
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hf = best_h + np.random.randn(Nf) * 0.008
        hf = project_sum(hf, Nf / 2.0)
        hf, c5f = continuous_opt(hf, Nf, deadline - 2, k=120, lr0=0.005, temp0=0.0005, beta0=0.95)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
