import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_C_mat(h, N):
    """C[i, k] = contribution of position i to correlation at lag k"""
    # corr[k] = sum_i h[i] * (1-h[i-k-shift])
    # Changing h[i] from v to w changes corr by (w-v)*(1-h_at_lag) - (v-w)*(h_at_lag)
    # = (w-v) * (1 - 2*h_at_lag) for that position
    # Actually for the full correlation:
    # corr_full[t] = sum_i h[i] * (1-h[i - (t - (N-1))])
    # If we change h[p] from 1 to 0: subtract (1-h[p - (t-(N-1))]) from corr[t] for valid t
    #   i.e., t such that p-(t-(N-1)) is in [0,N-1]
    #   t = p + N - 1 - q where q = p-(t-(N-1)) in [0,N-1]
    #   so t ranges from p+1 to p+2N-2 (clamped to [0, 2N-2])
    # If we change h[q] from 0 to 1: add (1-h[q - (t-(N-1))]) to corr[t]
    #   which is: add contribution at position q
    
    # Let's define: effect of changing h[i] from 0 to 1 on corr[t]:
    #   delta[i, t] = 1 - h[t - (N-1) - i] if 0 <= t-(N-1)-i < N else 0
    #   Wait, that's the contribution of h[i]=1 to corr[t]: h[i]*(1-h[i-k]) where k = t-(N-1)
    #   = 1 - h[i-k] if i-k in [0,N-1], else 0
    #   = 1 - h[i - (t-(N-1))] if 0 <= i-(t-(N-1)) < N, else 0
    
    # So the C matrix: C[i, t] = 1 if pos i contributes when h[i]=1 at lag t
    #   C[i, t] = 1 - h[i - (t - (N-1))] if 0 <= i - (t-(N-1)) < N
    #            = 0 otherwise
    
    # For a pair flip (p: 1->0, q: 0->1):
    #   new_corr[t] = corr[t] - C[p,t] + C[q,t]
    #   (because removing h[p]=1 subtracts its contribution, adding h[q]=1 adds its contribution)
    
    pos = np.arange(N)
    center = N - 1
    t_vals = np.arange(2 * N - 1)
    # C[i, t] = 1 - h[i - (t - center)] if 0 <= i - (t-center) < N else 0
    # i - (t - center) = i + center - t
    idx = pos[:, None] + center - t_vals[None, :]
    valid = (idx >= 0) & (idx < N)
    C = np.where(valid, 1.0 - h[np.clip(idx, 0, N-1)], 0.0)
    return C, t_vals

        pk = float(np.max(corr))

        # corr(k) = sum_i h[i] * (1-h[i-k]) for valid i
        # d corr(k) / d h[j] = sum over valid (i,k) pairs where j is involved
        #   If j = i: (1-h[j-k])
        #   If j = i-k: -h[i] = -h[j+k]
        
        top = np.argpartition(corr, -k_peaks)[-k_peaks:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                g = 1.0 - 2.0 * h
            elif L > 0 and L < N:

                g[L:] += (1.0 - h[:N-L])  # h[i] contribution where i-k valid

                g[:N-L] -= h[L:]  # derivative of -h[i]*h[i-k] w.r.t. h[i-k] where i-k = j => i = j+k
            else:
                M = -L
                if M < N:
                    g = np.zeros(N)
                    g[:N-M] += (1.0 - h[M:])
                    g[M:] -= h[:N-M]
            grad += g
        grad /= k_peaks
        
        # Project gradient step
        lr = lr0 / (1 + it * 0.01)
        h_new = project_01(h - lr * grad, target)
        
        # Polyak averaging
        h_avg += h_new
        h = h_new
    
    # Final: use average
    h_avg /= n_averages  # approximate
    h_avg = project_01(h_avg, target)
    avg_c = float(np.max(fft_corr(h_avg))) * dx
    pk_c = float(np.max(fft_corr(h))) * dx
    if avg_c < pk_c:
        return h_avg, avg_c
    return h, pk_c

            h[z2] = 1
            nc = fft_corr(h)
            nP = float(np.max(nc))
            if nP - P < temp:
                corr = nc
            else:
                h[o2] = 1
                h[z2] = 0
        else:
            ns = min(len(ones), sample_sz)
            nz = min(len(zeros), sample_sz)
            os_ = np.random.choice(ones, ns, replace=False)
            zs = np.random.choice(zeros, nz, replace=False)
            C_all = np.where(valid_mask, 1.0 - h[clip_idx], 0.0)
            Co = C_all[os_]
            Cz = C_all[zs]
            new = corr[None, None, :] + Cz[:, None, :] - Co[None, :, :]
            newP = new.max(axis=2)
            mi = np.unravel_index(np.argmin(newP), newP.shape)
            mv = float(newP[mi])
            delta = mv - P
            if delta < 0 or (temp > 1e-6 and np.random.random() < np.exp(-delta / temp)):
                h[os_[mi[1]]] = 0
                h[zs_[mi[0]]] = 1
                corr = fft_corr(h)
        temp *= cool
    return best_h, best_c

def cont_momentum(h, N, deadline, k_peaks=60, lr0=0.02, beta=0.75):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    v = np.zeros(N)
    center = N - 1
    for it in range(int(1e7)):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        c = float(np.max(corr))
        if c < best_c:
            best_c = c
            best_h = h.copy()
        top = np.argpartition(corr, -k_peaks)[-k_peaks:]
        grad = np.zeros(N)
        for k in top:
            L = int(k - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            elif 0 < -L < N:
                M = -L
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
        grad /= k_peaks
        lr = lr0 / (1.0 + 0.004 * it)
        v = beta * v + (1.0 - beta) * grad
        h = project_01(h - lr * v, target)
    return best_h, best_c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: SA binary search from initial
    d1 = min(deadline - 350, t0 + 220)
    h_bin = (h0 > 0.5).astype(float)
    h_s, _ = sa_binary(h_bin, N0, d1, sample_sz=100, temp0=0.35, cool=0.9996, two_pair_interval=300)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Continuous momentum optimization
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_01(hc, N0 / 2.0)
        hn, c5n = cont_momentum(hc, N0, d2, k_peaks=60, lr0=0.018, beta=0.78)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Deeper SA binary (lower temp, larger sample)
    if time.time() < deadline - 180:
        d3 = min(deadline - 120, time.time() + 200)
        h3 = (best_h > 0.5).astype(float)
        h_s3, _ = sa_binary(h3, N0, d3, sample_sz=120, temp0=0.12, cool=0.9998, two_pair_interval=600)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Second continuous pass from best binary
    if time.time() < deadline - 100:
        d4 = min(deadline - 60, time.time() + 100)
        hc2 = best_h.copy()
        hc2, c5c2 = cont_momentum(hc2, N0, d4, k_peaks=80, lr0=0.012, beta=0.82)
        if c5c2 < best_c5:
            best_c5, best_h = c5c2, hc2.copy()

    # Phase 5: Multi-resolution
    for N_try in [240, 320, 440, 520]:
        if time.time() > deadline - 35:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1 - nh, replace=False)] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh - n1, replace=False)] = 1
        sub_dl = min(deadline - 15, time.time() + 35)
        h_rt, _ = sa_binary(h_bt, N_try, sub_dl, sample_sz=80, temp0=0.25, cool=0.9995, two_pair_interval=200)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final SA polish (very low temp, high sample)
    if time.time() < deadline - 3:
        hB = (best_h > 0.5).astype(float)
        dB = deadline - 1
        hB, _ = sa_binary(hB, len(hB), dB, sample_sz=130, temp0=0.05, cool=0.9999, two_pair_interval=99999)
        c5B = compute_c5(hB, len(hB))
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    # Ensure final answer is valid
    h_final = (best_h > 0.5).astype(float)
    c5_final = compute_c5(h_final, len(h_final))
    if c5_final < best_c5:
        best_h, best_c5 = h_final, c5_final
    else:
        best_c5 = compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
