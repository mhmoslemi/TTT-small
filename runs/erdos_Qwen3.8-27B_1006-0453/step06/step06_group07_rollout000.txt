    best_h = h.copy()

    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        # Softmax approximation of max
        c_max = float(corr.max())
        weights = np.exp(alpha * (corr - c_max))
        weights /= weights.sum()
        # Gradient of weighted sum of correlations
        pk = float(np.sum(weights * corr))
        if pk - 1.0/alpha < best_pk - 1e-6:  # rough comparison
            if c_max < best_pk:
                best_pk = c_max
                best_h = h.copy()
        top = np.where(weights > weights.max() * 0.01)[0]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]

        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, beta0 + it * 0.0002)

        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def simulated_anneal(h, N, deadline, t_start=0.04, t_end=0.0003, cool=0.996):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    dx = 2.0 / N
    temp = t_start
    max_iter = 80000
    it = 0
    while time.time() < deadline and it < max_iter:
        if temp < t_end:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        h[p], h[q] = 0.0, 1.0
        c_new = fft_corr(h)
        P_new = float(c_new.max())
        delta = (P_new - P) * dx
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-12)):
            temp *= cool
        else:
            h[p], h[q] = 1.0, 0.0
            temp *= cool
        it += 1
    return best_h, best_c * dx

def multi_swap(h, N, deadline, n_bits=2, n_attempts=30):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    dx = 2.0 / N
    max_sweeps = 500
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < n_bits or len(zeros) < n_bits:
            break
        improved = False
        for _ in range(n_attempts):
            ps = np.random.choice(ones, n_bits, replace=False)
            qs = np.random.choice(zeros, n_bits, replace=False)
            ht = h.copy()
            for pp, qq in zip(ps, qs):
                ht[pp] = 0.0
                ht[qq] = 1.0
            Pt = float(np.max(fft_corr(ht)))
            if Pt < P - 1e-13:
                h = ht
                improved = True
                break
        if not improved and np.random.random() < 0.15:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
    return best_h, best_c * dx

def softmax_nesterov(h, N, deadline, lr0=0.012, alpha=180):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        c_max = float(corr.max())
        if c_max < best_pk:
            best_pk = c_max
            best_h = h.copy()
        w = np.exp(alpha * (corr - c_max))
        w /= w.sum()
        top = np.where(w > w.max() * 0.02)[0]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.985
            beta = min(0.96, 0.85 + it * 0.0003)
    return best_h, best_pk * dx

def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Thorough greedy at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 500, t0 + 100)
    h1, c5_1 = greedy_search(h_bin, N0, d1, max_sweeps=1200, sample_frac=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Simulated annealing to escape local minima
    d2 = min(deadline - 420, time.time() + 130)
    h2, c5_2 = simulated_anneal(best_h, N0, d2, t_start=0.035, cool=0.9965)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Multi-bit swap (2 bits)
    d3 = min(deadline - 340, time.time() + 90)
    h3, c5_3 = multi_swap(best_h, N0, d3, n_bits=2, n_attempts=25)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Greedy again after SA/multi-swap
    d4 = min(deadline - 280, time.time() + 70)
    h_bin4 = ensure_sum(best_h, N0)
    h4, c5_4 = greedy_search(h_bin4, N0, d4, max_sweeps=800, sample_frac=1.0)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Upsample to 2*N0, greedy
    N1 = 2 * N0
    if time.time() < deadline - 220:
        h_up = ensure_sum(upsample(best_h, N1), N1)
        d5 = min(deadline - 180, time.time() + 80)
        h5, c5_5 = greedy_search(h_up, N1, d5, max_sweeps=350, sample_frac=0.55)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Softmax Nesterov continuous at N0
    if time.time() < deadline - 130:
        Nc = N0
        hc = best_h[:Nc].copy() if len(best_h) == Nc else upsample(best_h, Nc)
        hc = hc + np.random.randn(Nc) * 0.025
        hc = project_sum(hc, Nc / 2.0)
        d6 = min(deadline - 100, time.time() + 55)
        h6, c5_6 = softmax_nesterov(hc, Nc, d6, lr0=0.01, alpha=200)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: SA at best resolution
    N_cur = len(best_h)
    if time.time() < deadline - 70:
        d7 = min(deadline - 50, time.time() + 45)
        h7, c5_7 = simulated_anneal(best_h, N_cur, d7, t_start=0.02, cool=0.997)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: 3-bit multi-swap
    if time.time() < deadline - 40:
        d8 = min(deadline - 25, time.time() + 30)
        h8, c5_8 = multi_swap(best_h, len(best_h), d8, n_bits=3, n_attempts=20)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    # Phase 9: Final greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 3:
        hF = ensure_sum(best_h, N_fin)
        sf = 1.0 if N_fin <= 500 else 0.6
        hF2, c5F = greedy_search(hF, N_fin, deadline - 0.5, max_sweeps=300, sample_frac=sf, perturb_every=35)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
