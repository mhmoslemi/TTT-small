import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N, rng):
    nb = (h + rng.rand(N) * 0.01 > 0.5).astype(float)
    n1 = int(nb.sum())
    t = N // 2
    if n1 > t:
        ones = np.where(nb == 1)[0]

        nb[sel] = 1
    return nb

    ip = p + km
    iq = q + km

        ps = rng.choice(ones, min(sn, len(ones)), replace=False) if len(ones) > sn else ones
        qs = rng.choice(zeros, min(sn, len(zeros)), replace=False) if len(zeros) > sn else zeros
        be = P
        bp = None
        for p in ps:
            for q in qs:
                e = swap_est(c, h, p, q, top_idx, N)
                if e < be:
                    be = e
                    bp = (int(p), int(q))
        if bp is not None:
            h[bp[0]] = 0.0
            h[bp[1]] = 1.0
            c = fft_corr(h)
        else:
            if T > 0.001 and rng.random() < T:
                p = int(rng.choice(ones))
                q = int(rng.choice(zeros))
                h[p] = 0.0
                h[q] = 1.0
                c = fft_corr(h)
            T = max(0.001, T * 0.999)
        it += 1
    return best_h, best_c
