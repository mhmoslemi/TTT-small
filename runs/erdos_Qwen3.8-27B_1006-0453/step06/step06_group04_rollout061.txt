import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, corr, beta, k_top):
    N = len(h)
    center = N - 1
    if k_top < len(corr):
        top = np.argpartition(corr, -k_top)[-k_top:]
    else:
        top = np.arange(len(corr))
    w = np.exp((corr[top] - corr[top].max()) / max(beta, 1e-8))
    w /= w.sum()
    grad = np.zeros(N)
    for wi, ai in zip(w, top):
        L = center - int(ai)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:seg]
    return grad

def adam_opt(h0, N, deadline, k_top=80, lr0=0.015, beta0=0.004):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_corr = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr0
    beta = beta0
    mom = 0.6
    h_prev = h.copy()
    corr = fft_corr(h)
    stall = 0
    while time.time() < deadline:
        t += 1
        y = h + mom * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_corr - 1e-13:
            best_corr = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        grad = compute_grad(y, corr, beta, k_top)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        mc = m / (1.0 - b1 ** t)
        vc = v / (1.0 - b2 ** t)
        step = lr * mc / (np.sqrt(vc) + eps)
        h = project_sum(y - step, target)
        if t % 400 == 0:
            lr = max(0.002, lr * 0.996)
            beta = max(0.0004, beta * 0.993)
            mom = min(0.94, 0.6 + t * 0.00006)
        if stall > 600:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            h_prev = h.copy()
            m = np.zeros(N)
            v = np.zeros(N)
            t = 0
            lr = lr0 * 0.7
            beta = max(beta0 * 0.5, 0.001)
            stall = 0
    return best_h, best_corr * (2.0 / N)

def binary_search(h0, N, deadline, n_trials=25, multi=1):

    corr = fft_corr(h)

    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < multi or len(zeros) < multi:
            break
        best_t = int(np.max(corr))
        best_th = None
        best_tc = None
        for _ in range(n_trials):
            if multi == 1:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p] = 0; h[q] = 1
            else:

                qs = np.random.choice(len(zeros), multi, replace=False)
                for pi in ps: h[ones[pi]] = 0
                for qi in qs: h[zeros[qi]] = 1
            c_t = fft_corr(h)
            ct = int(np.max(c_t))
            if ct < best_t:

                best_tc = c_t.copy()
            if multi == 1:

            else:
                for pi in ps: h[ones[pi]] = 1
                for qi in qs: h[zeros[qi]] = 0
        if best_th is not None:
            h = best_th
            corr = best_tc
            stall = 0
            if best_t < best_int:
                best_int = best_t
                best_h = h.copy()
        else:
            stall += 1
            if stall > 120:
                nf = max(2, N // 30)

                qs = np.random.choice(len(zeros), nf, replace=False)
                for pi in ps: h[ones[pi]] = 0
                for qi in qs: h[zeros[qi]] = 1
                corr = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)

def snap_to_binary(h, N):
    nh = N // 2
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 > nh:
        vals = h[hb == 1]
        idx = np.where(hb == 1)[0]
        order = np.argsort(vals)[:n1 - nh]
        hb[idx[order]] = 0
    elif n1 < nh:
        vals = h[hb == 0]
        idx = np.where(hb == 0)[0]
        order = np.argsort(vals)[::-1][:nh - n1]
        hb[idx[order]] = 1
    return hb

    if c5n < best_c5:
        best_c5, best_h = c5n, hn.copy()
    hb_snap = snap_to_binary(best_h, N0)
    c5s = compute_c5(hb_snap, N0)
    if c5s < best_c5:
        best_c5, best_h = c5s, hb_snap

    # Phase 2: Binary local search single-bit (180s)
    d2 = min(t0 + 560, deadline - 400)
    hb, cb = binary_search(best_h, N0, d2, n_trials=30, multi=1)
    if cb < best_c5:
        best_c5, best_h = cb, hb.copy()

    # Phase 3: Second Adam, lower temp (150s)
    d3 = min(t0 + 710, deadline - 250)
    hc3 = best_h + np.random.randn(N0) * 0.03
    hc3 = project_sum(hc3, N0 / 2.0)
    hn3, c5n3 = adam_opt(hc3, N0, d3, k_top=120, lr0=0.01, beta0=0.002)
    if c5n3 < best_c5:
        best_c5, best_h = c5n3, hn3.copy()
    hb_s3 = snap_to_binary(best_h, N0)
    c5s3 = compute_c5(hb_s3, N0)
    if c5s3 < best_c5:
        best_c5, best_h = c5s3, hb_s3

    # Phase 4: Binary multi-bit (100s)
    d4 = min(t0 + 810, deadline - 180)
    hb4, cb4 = binary_search(best_h, N0, d4, n_trials=20, multi=2)
    if cb4 < best_c5:
        best_c5, best_h = cb4, hb4.copy()

    # Phase 5: Try N=760 (120s)
    if time.time() < deadline - 130:
        N2 = 760
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N2, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h2 = snap_to_binary(h_int, N2)
        d5 = min(t0 + 930, deadline - 60)
        hb5, cb5 = binary_search(h2, N2, d5, n_trials=20, multi=1)
        if cb5 < best_c5:
            best_c5, best_h = cb5, hb5.copy()

    # Phase 6: Final Adam polish (remaining)
    if time.time() < deadline - 10:
        hF = best_h + np.random.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = adam_opt(hF, len(best_h), deadline - 3, k_top=150, lr0=0.006, beta0=0.001)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
        hbF = snap_to_binary(best_h, len(best_h))
        c5Fb = compute_c5(hbF, len(best_h))
        if c5Fb < best_c5:
            best_c5, best_h = c5Fb, hbF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
