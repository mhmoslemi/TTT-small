            grad[:N - L] -= wj * h[L:]
        elif -N < L < 0:

            grad[:N - M] += wj * (1.0 - h[M:])

            best_c5, best_h = c5_rt, h_rt.copy()
            Ncur = Ntry

    if time.time() < deadline - 10:
        d6 = deadline - 3
        hb, c5b = binary_local(best_h, len(best_h), d6, max_sweeps=500, top_m=30, sample_n=90, T0=0.05, T_decay=0.999, rng=rng)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
