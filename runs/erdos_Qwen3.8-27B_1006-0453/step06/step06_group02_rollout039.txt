```python
import numpy as np
import time

def compute_corr(h):
    """Compute np.correlate(h, 1-h, mode='full')"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    corr = compute_corr(h)
    return float(corr.max()) * (2.0 / N)

def build_G(h, N):
    """Build the G matrix: G[i, k] = effect of decreasing h[i] by 1 on corr[k]
    G[i, k] = h[i+k]*I(0<=i+k<N) - (1-h[i-k])*I(0<=i-k<N)
    """
    # k ranges from -(N-1) to (N-1), indexed as 0 to 2N-2
    # corr index j corresponds to lag = j - (N-1)
    G = np.zeros((N, 2*N-1))
    idx = np.arange(N)
    for lag in range(-(N-1), N):
        j = lag + (N-1)
        # h[i+lag] for i=0..N-1: valid when 0 <= i+lag < N, i.e., -lag <= i < N-lag
        # (1-h[i-lag]): valid when 0 <= i-lag < N, i.e., lag <= i < N+lag
        i_plus = np.clip(idx + lag, 0, N-1)
        valid_plus = (idx + lag >= 0) & (idx + lag < N)
        i_minus = np.clip(idx - lag, 0, N-1)
        valid_minus = (idx - lag >= 0) & (idx - lag < N)
        G[:, j] = np.where(valid_plus, h[i_plus], 0.0) - np.where(valid_minus, (1.0 - h[i_minus]), 0.0)
    return G
```

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(compute_corr(h).max()) * (2.0 / N)

def build_G(h, N):
    i = np.arange(N)[:, None]
    lag = np.arange(-(N-1), N)[None, :]
    ip = i + lag
    im = i - lag
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    hp = np.where(vp, h[np.clip(ip, 0, N-1)], 0.0)
    hm = np.where(vm, 1.0 - h[np.clip(im, 0, N-1)], 0.0)
    return hp - hm

def create_initial(N, mode='alternating'):
    h = np.zeros(N)
    if mode == 'alternating':
        h[::2] = 1.0
    elif mode == 'golden':
        phi = (1 + np.sqrt(5)) / 2
        positions = np.floor(np.arange(N/2) * phi / N * 2 * N / 2).astype(int) % N
        h[positions] = 1.0
    elif mode == 'block':
        h[:N//4] = 1.0
        h[N//2:N*3//4] = 1.0
    elif mode == 'random':
        idx = np.random.permutation(N)[:N//2]
        h[idx] = 1.0
    return h

def fix_sum_binary(h, N):
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        remove = np.random.choice(ones, n1 - target, replace=False)
        h[remove] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        add = np.random.choice(zeros, target - n1, replace=False)
        h[add] = 1
    return h
