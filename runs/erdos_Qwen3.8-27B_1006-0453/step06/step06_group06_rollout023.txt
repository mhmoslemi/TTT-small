            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N - s] += (1.0 - h[s:])
                grad[s:] -= h[:N - s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N - s2])
                grad[:N - s2] -= h[s2:]
        grad /= top_k
        t += 1
        step = lr / (1.0 + 0.0005 * t)
        h = proj_sum(h - step * grad, N / 2.0)
    return best_h, best_c

def delta_swap(h, N, p, q, s_arr):
    d = np.zeros(2 * N - 1)
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
    pc = np.clip(ps, 0, N-1)[vps]; pmc = np.clip(pm, 0, N-1)[vpm]
    qsc = np.clip(qs, 0, N-1)[vqs]; qmc = np.clip(qm, 0, N-1)[vqm]
    d[vps] -= (1.0 - h[pc])
    d[vpm] += h[pmc]
    d[vqs] += (1.0 - h[qsc])
    d[vqm] -= h[qmc]
    return d

def anneal(h, N, deadline, T0=0.06, cool=0.9994, restarts=3):
    h = to_binary(h, N).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    s_arr = np.arange(-(N - 1), N)
    for r in range(restarts):
        T = T0 * (0.4 ** r)
        no_imp = 0
        while time.time() < deadline and T > 0.0004:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if np.random.random() < 0.65:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                d = delta_swap(h, N, p, q, s_arr)
                nc = corr + d
                npk = float(nc.max())
                if npk < cur_pk or np.random.random() < np.exp(-(npk - cur_pk) / max(T, 1e-9)):
                    h[p] = 0; h[q] = 1
                    corr = nc; cur_pk = npk
                    if npk < best_pk:
                        best_pk = npk; best_h = h.copy(); no_imp = 0
                    else:
                        no_imp += 1
            else:
                if len(ones) < 2 or len(zeros) < 2:
                    T *= cool; continue
                i1, i2 = np.random.choice(len(ones), 2, replace=False)
                j1, j2 = np.random.choice(len(zeros), 2, replace=False)
                p1, q1 = int(ones[i1]), int(zeros[j1])
                p2, q2 = int(ones[i2]), int(zeros[j2])
                if p1 == p2 or q1 == q2:
                    T *= cool; continue
                d = delta_swap(h, N, p1, q1, s_arr) + delta_swap(h, N, p2, q2, s_arr)
                nc = corr + d
                npk = float(nc.max())
                if npk < cur_pk or np.random.random() < np.exp(-(npk - cur_pk) / max(T, 1e-9)):
                    h[p1] = 0; h[p2] = 0
                    h[q1] = 1; h[q2] = 1
                    corr = nc; cur_pk = npk
                    if npk < best_pk:
                        best_pk = npk; best_h = h.copy(); no_imp = 0
                    else:
                        no_imp += 1
            T *= cool
            if no_imp > 500:
                h = best_h.copy(); corr = fft_corr(h); cur_pk = best_pk; no_imp = 0
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(fft_corr(h0).max()) * (2.0 / N0)

    # Phase 1: Continuous subgradient from perturbed start
    d1 = min(deadline - 450, t0 + 200)
    h_c = h0 + np.random.randn(N0) * 0.06
    h_c = proj_sum(h_c, N0 / 2.0)
    h_cont, pk_cont = subgrad_opt(h_c, N0, d1, top_k=40, lr0=0.007)
    c5_cont = pk_cont * (2.0 / N0)
    if c5_cont < best_c5:
        best_c5, best_h = c5_cont, h_cont.copy()

    # Phase 2: Binary annealing
    d2 = min(deadline - 250, time.time() + 200)
    h_bin = to_binary(best_h, N0)
    h_ann, pk_ann = anneal(h_bin, N0, d2, T0=0.07, cool=0.9993, restarts=3)
    c5_ann = pk_ann * (2.0 / N0)
    if c5_ann < best_c5:
        best_c5, best_h = c5_ann, h_ann.copy()

    # Phase 3: Continuous polish around best
    if time.time() < deadline - 180:
        d3 = min(deadline - 130, time.time() + 130)
        h_p = best_h.astype(float) + np.random.randn(len(best_h)) * 0.04
        h_p = proj_sum(h_p, len(best_h) / 2.0)
        h_pol, pk_pol = subgrad_opt(h_p, len(best_h), d3, top_k=60, lr0=0.004)
        c5_pol = pk_pol * (2.0 / len(best_h))
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol.copy()

    # Phase 4: Multi-N exploration
    for Nt in [280, 340, 400, 460, 520, 600, 700, 840]:
        if time.time() > deadline - 80:
            break
        if Nt % 2 != 0:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hb = to_binary(hi, Nt)
        sub = min(deadline - 5, time.time() + 35)
        h_r, pk_r = anneal(hb, Nt, sub, T0=0.05, cool=0.9996, restarts=2)
        c5_r = pk_r * (2.0 / Nt)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()

    # Phase 5: Final binary + continuous refinement
    if time.time() < deadline - 10:
        hf = to_binary(best_h, len(best_h))
        d5a = min(deadline - 5, time.time() + 30)
        h_fb, pk_fb = anneal(hf, len(best_h), d5a, T0=0.02, cool=0.9997, restarts=1)
        c5_fb = pk_fb * (2.0 / len(best_h))
        if c5_fb < best_c5:
            best_c5, best_h = c5_fb, h_fb.copy()
        # Tiny continuous nudge
        if time.time() < deadline - 2:
            h_f = best_h.astype(float) + np.random.randn(len(best_h)) * 0.015
            h_f = proj_sum(h_f, len(best_h) / 2.0)
            h_fin, pk_fin = subgrad_opt(h_f, len(best_h), deadline - 1, top_k=100, lr0=0.002)
            c5_fin = pk_fin * (2.0 / len(best_h))
            if c5_fin < best_c5:
                best_c5, best_h = c5_fin, h_fin.copy()

    # Final evaluation using np.correlate to match harness
    final_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full"))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
