def sa_phase(h, N, T_start, T_end, n_iter, deadline, p2swap=0.1):
    """Simulated annealing with 1-swap and 2-swap moves."""
    h = h.copy()
    corr = init_corr(h)
    curr_max = float(np.max(corr))
    best_c = curr_max
    best_h = h.copy()
    
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    ones_list = list(ones)
    zeros_list = list(zeros)

    alpha = (T_end / T_start) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        if np.random.random() < p2swap and len(ones) >= 2 and len(zeros) >= 2:
            # 2-swap
            p1, p2 = (np.random.choice(ones_list, 2, replace=False))
            q1, q2 = (np.random.choice(zeros_list, 2, replace=False))
            # Apply 4 changes
            do_swap(h, corr, int(p1), int(q1))
            do_swap(h, corr, int(p2), int(q2))
            new_max = float(np.max(corr))
            if new_max <= curr_max or np.random.random() < np.exp(-(new_max - curr_max) / T):
                curr_max = new_max
                if curr_max < best_c:
                    best_c = curr_max
                    best_h = h.copy()
                ones_list.remove(int(p1)); zeros_list.append(int(p1))
                zeros_list.remove(int(q1)); ones_list.append(int(q1))
                ones_list.remove(int(p2)); zeros_list.append(int(p2))
                zeros_list.remove(int(q2)); ones_list.append(int(q2))
            else:
                undo_swap(h, corr, int(p2), int(q2))
                undo_swap(h, corr, int(p1), int(q1))
        else:
            # 1-swap
            p = ones_list[np.random.randint(len(ones_list))]
            q = zeros_list[np.random.randint(len(zeros_list))]
            do_swap(h, corr, int(p), int(q))
            new_max = float(np.max(corr))
            if new_max <= curr_max or np.random.random() < np.exp(-(new_max - curr_max) / T):
                curr_max = new_max
                if curr_max < best_c:
                    best_c = curr_max
                    best_h = h.copy()
                ones_list.remove(int(p)); zeros_list.append(int(p))
                zeros_list.remove(int(q)); ones_list.append(int(q))
            else:
                undo_swap(h, corr, int(p), int(q))
        
        T = max(T_end, T * alpha)
    
    return best_h, best_c * (2.0 / N)

def greedy_polish(h, N, deadline, max_iter=100000):
    """Greedily find the best single swap at each step."""
    h = h.copy()
    corr = init_corr(h)
    curr_max = float(np.max(corr))
    best_c = curr_max
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Sample a few random swaps, pick the best
        n_try = 50
        best_delta = 0
        best_p, best_q = -1, -1
        for _ in range(n_try):
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            do_swap(h, corr, p, q)
            nm = float(np.max(corr))
            delta = nm - curr_max
            if delta < best_delta:
                best_delta = delta
                best_p, best_q = p, q
            undo_swap(h, corr, p, q)
        
        if best_p >= 0 and best_delta < -1e-10:
            p, q = best_p, best_q
            do_swap(h, corr, p, q)
            curr_max += best_delta  # approximately
            curr_max = float(np.max(corr))  # exact
            if curr_max < best_c:
                best_c = curr_max
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            break
    
    return best_h, best_c * (2.0 / N)

def init_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def do_swap(h, corr, p, q):
    N = len(h)
    lo = max(0, N - 1 - p)
    hi = min(2 * N - 2, N - 1 + p)
    js = np.arange(lo + p - N + 1, hi + p - N + 2)
    corr[lo:hi+1] -= (1.0 - h[js])
    lo2 = max(0, N - 1 - q)
    hi2 = min(2 * N - 2, N - 1 + q)
    js2 = np.arange(lo2 + q - N + 1, hi2 + q - N + 2)
    corr[lo2:hi2+1] += (1.0 - h[js2])
    h[p] = 0.0
    h[q] = 1.0

def undo_swap(h, corr, p, q):

import numpy as np

def init_corr(h):
    N = len(h)
    L = 2 * N - 1
    # correlate(h, 1-h, full) via FFT
    G = np.fft.rfft(np.flip(1.0 - h), L)
    H = np.fft.rfft(h, L)
    return np.fft.irfft(H * G, L)

def run_sa(h, N, deadline, T0, T1, p2=0.15, n_restarts=3, stuck_limit=50000):
    """Simulated annealing with restarts."""
    best_h = h.copy()
    best_raw = float(np.max(init_corr(h)))
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        h = best_h.copy() if restart > 0 else h.copy()
        corr = init_corr(h)
        curr = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        T = T0 if restart == 0 else T0 * 0.5
        alpha = (T1 / T0) ** (1.0 / max(1, stuck_limit * 3))
        no_improve = 0
        
        while time.time() < deadline:
            it = 1
            # Choose move
            if np.random.random() < p2 and len(ones) >= 2 and len(zeros) >= 2:
                idx_p = np.random.choice(len(ones), 2, replace=False)
                idx_q = np.random.choice(len(zeros), 2, replace=False)
                p1, p2_ = int(ones[idx_p[0]]), int(ones[idx_p[1]])
                q1, q2_ = int(zeros[idx_q[0]]), int(zeros[idx_q[1]])
                # Apply 2-swap
                h[p1], h[q1] = 0.0, 1.0
                h[p2_], h[q2_] = 0.0, 1.0
                corr = init_corr(h)
                nm = float(np.max(corr))
                if nm <= curr or np.random.random() < np.exp(-(nm - curr) / max(T, 1e-10)):
                    curr = nm
                    no_improve = 0
                    if curr < best_raw:
                        best_raw = curr

                    ones = np.where(h == 1)[0]
