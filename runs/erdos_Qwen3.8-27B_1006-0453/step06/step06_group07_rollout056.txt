def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def D_row(h, N, p):
    L = 2 * N - 1
    s = np.arange(L) - (N - 1)
    ps = p + s
    pm = p - s
    vp = (ps >= 0) & (ps < N)
    vm = (pm >= 0) & (pm < N)
    d = np.zeros(L)
    if vp.any():
        d[vp] = h[ps[vp]]
    if vm.any():
        d[vm] -= (1.0 - h[pm[vm]])
    return d

def compute_D_full(h, N):
    """Compute full D matrix (N x (2N-1))."""
    return np.array([D_row(h, N, p) for p in range(N)])

def ensure_sum(h, N):

    h = h.astype(float).copy()

    n1 = int(h_bin.sum())

def greedy_search(h, N, deadline, max_sweeps=500):

    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]

        if len(ones) < 1 or len(zeros) < 1:

        D = compute_D_full(h, N)
        Dz = D[zeros, :]

                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:

        h[best_pq[0]] = 0.0

    return best_h, best_c * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0

        else:

def adam_opt(h, N, deadline, lr=0.005, k_top=30):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m_v = np.zeros(N)
    v_v = np.zeros(N)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    s_arr = np.arange(2*N-1) - (N-1)
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        cur_c = float(corr.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # Find top-k shifts
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        # Compute gradient
        grad = np.zeros(N)
        for ai in top_idx:
            L_shift = int(ai - (N - 1))
            for j in range(N):
                val = 0.0
                jp = j + L_shift
                jm = j - L_shift
                if 0 <= jp < N:
                    val += (1.0 - h[jp])
                if 0 <= jm < N:
                    val -= h[jm]
                grad[j] += val
        grad /= k_top
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * (grad ** 2)
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
    return best_h, best_c * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def D_row(h, N, p, shifts):
    L = 2 * N - 1
    pm = p - shifts
    pp = p + shifts
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    d = np.zeros(L)
    if vm.any():
        d[vm] = -(1.0 - h[pm[vm]])
    if vp.any():
        d[vp] += h[pp[vp]]
    return d

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def greedy_search(h, N, deadline, max_sweeps=400):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    shifts = np.arange(2 * N - 1) - (N - 1)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = np.array([D_row(h, N, int(p), shifts) for p in range(N)])
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        for p in ones:
            base = c - D[int(p), :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def sa_search(h, N, deadline, t_init=0.004, t_min=1e-5, cool=0.99992, p_2swap=0.25, max_iter=500000):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    shifts = np.arange(2 * N - 1) - (N - 1)
    c_corr = fft_corr(h)
    cur_c = float(c_corr.max())
    best_h = h.copy()
    best_c = cur_c
    L = 2 * N - 1
    rng = np.random.RandomState(42)
    T = t_init
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        if it % 2000 == 0:
            T = max(t_min, T * cool)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        if p_2swap > 0 and rng.random() < p_2swap and len(ones) >= 2 and len(zeros) >= 2:
            p1, p2 = rng.choice(ones, 2, replace=False)
            q1, q2 = rng.choice(zeros, 2, replace=False)
            delta = -D_row(h, N, int(p1), shifts) - D_row(h, N, int(p2), shifts) + D_row(h, N, int(q1), shifts) + D_row(h, N, int(q2), shifts)
            new_c = float(np.max(c_corr + delta))
            h[p1];h[p2]=0.0,0.0
            h[q1];h[q2]=1.0,1.0
            c_corr = c_corr + delta
            if new_c > cur_c and rng.random() > np.exp(-(new_c - cur_c) / T):
                h[p1];h[p2]=1.0,1.0
                h[q1];h[q2]=0.0,0.0
                c_corr = c_corr - delta
            else:
                cur_c = new_c
        else:
            p = int(rng.choice(ones))
            q = int(rng.choice(zeros))
            delta = -D_row(h, N, p, shifts) + D_row(h, N, q, shifts)
            new_c = float(np.max(c_corr + delta))
            h[p] = 0.0
            h[q] = 1.0
            c_corr = c_corr + delta
            if new_c > cur_c and rng.random() > np.exp(-(new_c - cur_c) / T):
                h[p] = 1.0
                h[q] = 0.0
                c_corr = c_corr - delta
            else:
                cur_c = new_c
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if it % 4000 == 0:
            c_corr = fft_corr(h)
            cur_c = float(c_corr.max())
    return best_h, best_c * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def adam_opt(h, N, deadline, lr=0.004, k_top=25):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m_v = np.zeros(N)
    v_v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        cur_c = float(corr.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for ai in top_idx:
            s = int(ai - (N - 1))
            if s > 0:
                grad[:N - s] += (1.0 - h[s:])
                grad[s:] -= h[:N - s]
            elif s < 0:
                M = -s
                if M < N:
                    grad[M:] += (1.0 - h[:N - M])
                    grad[:N - M] -= h[M:]
            else:
                grad += (1.0 - 2.0 * h)
        grad /= k_top
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * (grad ** 2)
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
    return best_h, best_c * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
