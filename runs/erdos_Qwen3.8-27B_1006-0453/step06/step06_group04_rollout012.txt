def fft_corr(h):

    L = 2 * N - 1

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

def grad_c5_smooth(h, N, tau, topk=200):
    """Compute smooth gradient using softmax weighting over top-k correlations."""
    corr = fft_corr(h)
    center = N - 1
    idx = np.argpartition(corr, -topk)[-topk:]
    c_topk = corr[idx]
    weights = np.exp(tau * (c_topk - c_topk.max()))
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for wi, ai in zip(weights, idx):
        if wi < 1e-10:
            continue
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:seg]
    return grad, corr, float(corr.max()) * (2.0 / N)

def adam_step(params, grad, Adam, step):
    Adam['m'] = 0.9 * Adam['m'] + 0.1 * grad
    Adam['v'] = 0.999 * Adam['v'] + 0.001 * grad**2
    m_hat = Adam['m'] / (1 - 0.9**step)
    v_hat = Adam['v'] / (1 - 0.999**step)
    params -= Adam['lr'] * m_hat / (np.sqrt(v_hat) + 1e-8)
