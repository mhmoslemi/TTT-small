import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h)[::-1]  # wait, need to be careful
    # Actually for cross-correlation: corr(a,b)[k] = sum a[i] b[k-i]
    # np.correlate(a, b, 'full')[k] = sum a[i] b[i+k-n] ... 
    # Let me just use the same formula as before
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    """C[i,k] = change in corr[k] when h[i] is flipped (1→0 or 0→1)."""
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    # corr[k] = sum_i h[i] * (1-h[i-k]) for valid indices
    # dcorr[k]/dh[i] = (1-h[i-k]) - h[i+k] (for valid indices)
    i_mk = pos - k_vals  # i - k
    i_pk = pos + k_vals  # i + k
    term1 = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    term2 = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N-1)], 0.0)
    return term1 - term2

def fix_sum(h, N):
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def binary_search(h, N, deadline, max_rounds=1000, batch=80, sa_temp=0.0):
    h = h.astype(float).copy()
    fix_sum(h, N)
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    no_improve = 0
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        
        C = compute_C_mat(h, N)
        P = float(np.max(corr))
        
        if P < best_c:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        ns = min(len(ones), batch)
        nz = min(len(zeros), batch)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        
        Co = C[os_]
        Cz = C[zs]
        newP = (corr[None, None, :] + Co[None, :, :] - Cz[:, None, :]).max(axis=2)
        
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        
        if mv < P - 1e-12:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
            corr = fft_corr(h)
        elif sa_temp > 0 and np.random.random() < sa_temp:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p] = 0
            h[q] = 1
            corr = fft_corr(h)
            sa_temp *= 0.995
        else:
            if no_improve > 50:
                # Perturb and restart
                for _ in range(10):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0
                    h[q] = 1
                    fix_sum(h, N)
                corr = fft_corr(h)
                no_improve = 0
            else:
                break
    
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def fix_sum(h, N):
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def binary_search(h, N, deadline, batch=70, max_rounds=2000, sa_prob=0.0, perturb_every=100):
    h = h.astype(float).copy()
    fix_sum(h, N)
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stuck = 0
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        C = compute_C_mat(h, N)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), batch)
        nz = min(len(zeros), batch)
        os_ = np.random.choice(ones, ns, replace=False)

        Co = C[os_]
        Cz = C[zs]

            h[zs[mi[0]]] = 1
            corr = fft_corr(h)
