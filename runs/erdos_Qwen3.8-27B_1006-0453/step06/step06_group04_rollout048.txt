    hb, c5b = binary_pair_opt(best_h, N0, d2, n_tries=28, kick_freq=65)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 3: Second Adam with different params (200s)
    d3 = max(d2, t0) + 200
    d3 = min(deadline - 220, d3)
    hc3 = best_h + np.random.randn(N0) * 0.03
    hc3 = project_sum(hc3, N0 / 2.0)
    hn3, c5n3 = adam_opt(hc3, N0, d3, k=90, lr=0.016, temp0=0.0028)
    if c5n3 < best_c5:
        best_c5, best_h = c5n3, hn3.copy()

    # Phase 4: Binary polish (130s)
    d4 = max(d3, t0) + 130
    d4 = min(deadline - 90, d4)
    hb4, c5b4 = binary_pair_opt(best_h, N0, d4, n_tries=45, kick_freq=45)
    if c5b4 < best_c5:
        best_c5, best_h = c5b4, hb4.copy()

    # Phase 5: Try doubled resolution (80s)
    if time.time() < deadline - 40:
        N2 = N0 * 2
        hi = interp_to_N(best_h, N2)
        h_bt = (hi > 0.5).astype(float)
        nh2 = N2 // 2
        n1 = int(h_bt.sum())
        if n1 > nh2:
            idx = np.where(h_bt == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - nh2]
            h_bt[idx] = 0
        elif n1 < nh2:
            idx = np.where(h_bt == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh2 - n1]
            h_bt[idx] = 1
        d5 = min(deadline - 25, time.time() + 70)
        h_rt, c5_rt = adam_opt(h_bt, N2, d5, k=80, lr=0.012, temp0=0.002)
        c5_rt = compute_c5(h_rt, N2)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final Adam polish (remaining time)
    if time.time() < deadline - 3:
        hF = best_h + np.random.randn(len(best_h)) * 0.012
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = adam_opt(hF, len(best_h), deadline - 2, k=100, lr=0.007, temp0=0.001)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
