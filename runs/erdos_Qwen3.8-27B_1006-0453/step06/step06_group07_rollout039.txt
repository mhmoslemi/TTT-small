    """Improved Nesterov with soft top-k gradient."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta_start
    it = 0
    best_k = k
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        # Soft top-k: use softmax weights
        # Focus gradient on the top correlations
        top_idx = np.argpartition(corr, -best_k)[-best_k:]
        # Compute gradient for top-k average
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]

Actually, I think the issue is that the two changes are NOT independent when p and q interact. If p+s = q or q+s = p, then the two changes interfere.

In my example: p=0, q=1. For shift s=-1: p+s = 0+(-1) = -1 (invalid). q+s = 1+(-1) = 0 = p. So h[p] appears in the "second factor" for position q at shift -1.

This means the linear approximation (sum of individual changes) fails when the two positions interact!

The correction term: when we flip both p and q, the cross term h[p]*(1-h[q]) or h[q]*(1-h[p]) at the interacting shift needs adjustment.

Specifically, for shift s where q+s = p (i.e., s = p-q):
- Before: h[q]*(1-h[p]) = 0*0 = 0
- After: h[q]'=(1), h[p]'=0: 1*(1-0) = 1
- The individual calculation: D[q] accounts for h[q] going 0→1 as first factor (adds 1-h[q+s] = 1-h[p] = 0), and D[p] accounts for h[p] going 1→0 as second factor (adds h[q] = 0).
- But the actual change at this shift is +1, while the sum of individuals gives 0+0=0.

So there's a missing cross term of +1 at shift s=p-q (and by symmetry, at shift s=q-p for the other direction).

This means the code's formula `c - D[p] - D[q]` is an APPROXIMATION that ignores the cross term. For most p,q pairs, the cross term is at a specific shift and the effect is small. But it means the algorithm might occasionally make a suboptimal choice.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_binary(h, N, seed=0):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 == nh:
        return h
    rng = np.random.RandomState(seed + N)
    if n1 > nh:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h[idx] = 0
    else:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h[idx] = 1
    return h

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c].astype(float), 0.0)
    hp = np.where(vp, h[pp_c].astype(float), 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k=50, lr0=0.015, decays_per=100, decay=0.993,
                 beta0=0.9, beta1=0.97):

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    k_cur = k
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        top_idx = np.argpartition(corr, -k_cur)[-k_cur:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L

                    grad[:N - M] += (1.0 - y[M:])

        m = beta * (h_new - h)
        h = h_new

        if it % decays_per == 0:
