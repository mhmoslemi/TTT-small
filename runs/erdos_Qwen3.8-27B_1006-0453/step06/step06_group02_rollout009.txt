    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def nesterov_smooth_max(h, N, deadline, k_top=30, lr0=0.02):
    """Nesterov with smooth max approximation"""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    temp = 1.0  # temperature for softmax
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        
        # Smooth max: use softmax
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient of smooth max
        exp_c = np.exp(corr / temp - np.max(corr) / temp)
        softmax_w = exp_c / exp_c.sum()
        # Grad of corr[k] w.r.t. h[i]
        # corr[k] = sum_i h[i]*(1-h[i+k])
        # d/dh[i] corr[k] = (1-h[i+k]) - h[i]*(d/dh[i] h[i+k])
        # For k=0: d/dh[i] = (1-h[i]) - h[i] = 1-2h[i]... 
        # Actually for the linear correlation (not circular):
        # corr[k] = sum_{i: i,i+k valid} h[i]*(1-h[i+k])
        # d/dh[j] corr[k] = (1-h[j+k])*I(j+k valid) - h[j-k]*I(j-k valid) for the k where j is involved
        
        # This is complex to compute in full. Use the top-k approximation.
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        
        # Weight by softmax
        grad_w = np.zeros(N)
        center = N - 1
        for t, ai in enumerate(top):
            w = softmax_w[np.argpartition(corr, -k_top)[-k_top:][t] if False else list(corr).index(corr[ai])]  # messy
            # Actually, let me just weight by the softmax value at that shift
            L = int(ai - center)
            w = softmax_w[t]  # approximate
            if L == 0:
                grad_w += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad_w[L:] += w * (1.0 - y[:N - L])
                grad_w[:N - L] -= w * y[L:]
            else:
                M = -L
                grad_w[:N - M] += w * (1.0 - y[M:])
                grad_w[M:] -= w * y[:N - M]
        
        # Actually the gradient of smooth_max is softmax-weighted sum of individual gradients
        # Let me just use the unweighted top-k for simplicity
        grad = grad_w / k_top
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.95, 0.88 + it * 0.0002)
            temp = max(0.1, temp * 0.95)
    
    return best_h, best_pk * dx

def multi_start_binary(N, deadline, n_starts=3, top_k=5, max_sweeps=500):
    """Try multiple starting points"""
    best_h = None
    best_c = 1.0
    
    for s in range(n_starts):
        t_sub = deadline - (n_starts - 1 - s) * (deadline - time.time()) / n_starts
        # Different starting patterns
        if s == 0:
            h = np.zeros(N)
            h[::2] = 1.0  # alternating
        elif s == 1:
            h = np.random.rand(N)
            h = (h > 0.5).astype(float)
        else:
            h = np.zeros(N)
            # Block pattern
            block = N // 4
            h[0:block] = 1.0
            h[2*block:3*block] = 1.0
        
        h = fix_sum_binary(h, N)
        h_sub, c_sub = binary_search_minmax(h, N, t_sub, max_sweeps=max_sweeps, top_k=top_k)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub.copy()
    
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[np.random.permutation(len(idx))][:n1 - nh]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[np.random.permutation(len(idx))][:nh - n1]] = 1.0
    return h

def binary_minmax(h, N, deadline, max_sweeps=1000, sample_n=200, top_k=5):
    h = h.astype(float).copy()
    weights = np.array([1.0, 0.4, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.01, 0.005][:top_k])
    c = fft_corr(h)
    best_pk = float(np.max(c))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(-c[top_idx])]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if len(ones) > sample_n:
            p_list = ones[np.random.permutation(len(ones))[:sample_n]]
        else:
            p_list = ones
        if len(zeros) > sample_n * 2:
            q_pool = zeros[np.random.permutation(len(zeros))[:sample_n * 2]]
        else:
            q_pool = zeros
        best_score = best_pk + 1.0
        best_pq = None
        for p in p_list:
            dp = np.zeros(top_k)
            for j in range(top_k):
                k = int(top_idx[j])
                if 0 <= p + k < N:
                    dp[j] -= (1.0 - h[p + k])
                if 0 <= p - k < N:
                    dp[j] += h[p - k]
            dq = np.zeros((len(q_pool), top_k))
            for j in range(top_k):
                k = int(top_idx[j])
                qk = q_pool + k
                qm = q_pool - k
                vp = (qk >= 0) & (qk < N)
                vm = (qm >= 0) & (qm < N)
                dq[:, j] += np.where(vp, 1.0 - h[qk[vp]], 0.0)
                dq[:, j] -= np.where(vm, h[qm[vm]], 0.0)
            new_vals = c[top_idx][None, :] + dp[None, :] + dq
            sorted_v = np.sort(new_vals, axis=1)[:, ::-1]
            scores = sorted_v @ weights
            mi = int(np.argmin(scores))
            if scores[mi] < best_score:
                best_score = float(scores[mi])
                best_pq = (int(p), int(q_pool[mi]))
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            new_pk = float(np.max(c))

                no_improve = 0

                else:

                    c = fft_corr(h)
            no_improve += 1
        if no_improve > 500:
            break
    return h, best_pk * (2.0 / N)

def nesterov_h(h, N, deadline, k_top=40, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        top_sorted = top[np.argsort(-corr[top])]
        shifted = corr[top_sorted] - corr[top_sorted][0]
        w = np.exp(shifted * 5.0)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for t in range(k_top):
            L = int(top_sorted[t] - center)
            wt = w[t]
            if L == 0:
                grad += wt * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wt * (1.0 - y[:N - L])
                grad[:N - L] -= wt * y[L:]
            else:
                M = -L
                grad[:N - M] += wt * (1.0 - y[M:])
                grad[M:] -= wt * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.992
            beta = min(0.96, 0.88 + it * 0.0003)
    return best_h, best_pk * dx
