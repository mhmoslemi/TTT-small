def project_to_binary(h):
    h = h.copy()
    # Clipping enforces point-wise [0,1]
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    desired = len(h) // 2  # dx=2/n so integral is (sum/2)*dx = 1
    # Determine direction to adjust
    if total > desired:
        # Remove excess by converting some 1s to 0s
        selection = np.random.choice(np.where(h>=0.5)[0], size=total-desired, replace=False)
        h[selection] = 0.0
    elif total < desired:
        # Introduce more 1s
        l0, l1 = np.where(h<0.5)[0], np.where(h>=0.5)[0]
        candidates = l0 if len(l0)>=len(l1) else l1
        while total < desired:
            if not candidates:
                # Fallback: convert zeros to ones (this should not happen but...)
                idx = np.argmax(1-h)
                h[idx] = 1.0
                candidates = [idx]
            else:
                while len(candidates) > 0 and total < desired:
                    idx = np.random.choice(candidates)
                    h[idx] = 1.0
                    total += 1
                    candidates = np.delete(candidates, np.flatnonzero(candidates==idx)[0])
    return h.astype(float)
