    L = 2 * N - 1
    center = N - 1
    effects = np.zeros((N, L))
    for p in range(N):
        dv = 1.0 - 2.0 * h[p]
        # Part (a)
        ka_lo = max(p - (N-1), -(N-1))
        ka_hi = min(p, N-1)
        if ka_lo <= ka_hi:
            ks = np.arange(ka_lo, ka_hi+1)
            effects[p, ks+center] += dv * (1.0 - h[p-ks])
        # Part (b)
        kb_lo = max(-p, -(N-1))
        kb_hi = min(N-1-p, N-1)
        if kb_lo <= kb_hi:
            ks = np.arange(kb_lo, kb_hi+1)
            effects[p, ks+center] -= dv * h[p+ks]

        effects[p, center] -= dv * dv
