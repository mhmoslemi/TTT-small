```python
def swap_effect(corr, h, p, q, N):
    """Compute new max correlation after swapping h[p]:1->0, h[q]:0->1"""
    center = N - 1
    # For h[p] 1->0:
    #   lag k = j - p (where j = p+k, i.e., k = j-p): +h[j] for j in [p+1, N-1]
    #   lag k = p - j (where j = p-k, i.e., k = p-j): -(1-h[j]) for j in [0, p-1]
    # For h[q] 0->1:
    #   lag k = j - q: -h[j] for j in [q+1, N-1]
    #   lag k = q - j: +(1-h[j]) for j in [0, q-1]
    
    delta = np.zeros(2*N-1)
    # h[p] 1->0: for j > p, lag j-p (index j-p+center): +h[j]
    for j in range(p+1, N):
        delta[j - p + center] += h[j]
    for j in range(p):
        delta[p - j + center] -= (1 - h[j])
    # h[q] 0->1: for j > q, lag j-q: -h[j]
    for j in range(q+1, N):
        delta[j - q + center] -= h[j]
    for j in range(q):
        delta[q - j + center] += (1 - h[j])
    
    new_corr = corr + delta
    return float(new_corr.max())
```

For step 3: At lag k (result index k+center), the correlation is sum of h[n]*(1-h[n+k]) over valid n. To reduce this, we want to:
- Turn off a 1 at position n+k where h[n]=0 (reduce by 1)
- Turn on a 0 at position n where h[n+k]=1 (reduce by 1)

So good swaps are: find p with h[p]=1, q with h[q]=0 such that for the top lag k, p and q are "aligned" (p = n+k, q = n or vice versa for the top lag).

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def local_search_binary(h, N, deadline, max_sweeps=2000, top_lags=15, candidates_per_lag=50):
    h = (h > 0.5).astype(float).copy()
    # Ensure sum = N/2
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        idx = idx[np.random.permutation(len(idx))][:n1 - nh]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        idx = idx[np.random.permutation(len(idx))][:(nh - n1)]
        h[idx] = 1
    
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
        
        # Find top lag indices
        center = N - 1
        top_idx = np.argsort(corr)[-top_lags:]
        
        # For each top lag, find candidate swaps
        best_new_P = P
        best_pq = None
        evaluated = set()
        
        for ti in top_idx:
            k = int(ti - center)  # lag
            # For this lag, the correlation is sum of h[n]*(1-h[n+k]) 
            # We want to find a 1 at position n+k and 0 at position n (for k>0)
            # Or more generally, find contributing pairs
            
            if k > 0:
                # corr[k] = sum_{n=0}^{N-1-k} h[n+k]*(1-h[n])
                # Contributing pairs: n where h[n]=0 and h[n+k]=1
                # Swap candidate: turn h[n+k]=1 to 0, and turn some other h[q]=0 to 1
                # Best: turn h[n+k] to 0 (reduces this lag by 1)
                #         and turn h[q] to 1 where q is NOT in a contributing position for top lags
                contributing_1s = [n + k for n in range(N - k) if h[n + k] == 1 and h[n] == 0]
                # For each contributing 1, find a good 0 to swap with
                np.random.shuffle(contributing_1s[:min(len(contributing_1s), candidates_per_lag)])
                for p in contributing_1s[:candidates_per_lag]:
                    # Try swapping p (1→0) with a random zero
                    for _ in range(3):  # try a few random zeros
                        q = zeros[np.random.randint(len(zeros))]
                        if q == p or (p, q) in evaluated:
                            continue
                        evaluated.add((p, q))
                        # Quick check: does this reduce the top lag?
                        new_corr_t = corr[ti] - 1  # at minimum this lag decreases
                        # But other lags might increase. Quick upper bound:
                        # The new max is at most max(corr - 1 at ti, Other top lags + 1)
                        if new_corr_t + 1 >= P:  # probably not an improvement
                            continue
                        # More careful: compute exact new max
                        h[p], h[q] = 0.0, 1.0
                        new_P = float(np.max(fft_corr(h)))
                        h[p], h[q] = 1.0, 0.0
                        if new_P < best_new_P:
                            best_new_P = new_P
                            best_pq = (p, q)
            elif k < 0:
                m = -k
                # corr[-m] = sum_{n=m}^{N-1} h[n-m]*(1-h[n])
                # Contributing: h[n-m]=1 and h[n]=0
                contributing_1s = [n - m for n in range(m, N) if h[n - m] == 1 and h[n] == 0]
                np.random.shuffle(contributing_1s[:min(len(contributing_1s), candidates_per_lag)])
                for p in contributing_1s[:candidates_per_lag]:
                    for _ in range(3):
                        q = zeros[np.random.randint(len(zeros))]
                        if q == p or (p, q) in evaluated:
                            continue
                        evaluated.add((p, q))
                        h[p], h[q] = 0.0, 1.0
                        new_P = float(np.max(fft_corr(h)))
                        h[p], h[q] = 1.0, 0.0
                        if new_P < best_new_P:
                            best_new_P = new_P
                            best_pq = (p, q)
        
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    
    return best_h, best_c * (2.0 / N)
```

```python
    #                         or 0 - (1-h_old[q]) if p-lag == q, but h_old[q]=0 so = 0-1 = -1
    #   Hmm, this is the old value. Let me be more careful.
    
    # delta[m] = sum_n (h'[n]*(1-h'[n-lag]) - h[n]*(1-h[n-lag]))
    # Non-zero terms: n where h'[n]≠h[n] (i.e., n=p or n=q) 
    #                 OR n-lag where h'[n-lag]≠h[n-lag] (i.e., n-lag=p or n-lag=q, so n=p+lag or n=q+lag)
    
    center = N - 1
    delta = np.zeros(2*N-1)
    for m in range(2*N-1):
        lag = m - center
        d = 0.0
        # n = p: h'[p]=0, h[p]=1
        #   term_new = 0 * (1-h'[p-lag]) = 0
        #   term_old = 1 * (1-h[p-lag]) (if p-lag in [0,N-1])
        if 0 <= p - lag < N:
            old_val = h[p-lag]
            # Actually term_old = 1*(1-h[p-lag]) = 1-h[p-lag]
            # term_new = 0
            d += -(1.0 - old_val)
            # BUT if p-lag == q, then in the new array h'[q]=1, but we're computing old vs new
            # The old array has h[q]=0, so 1-h_old[p-lag] = 1-h_old[q] = 1-0 = 1
            # So d += -1 in this case (which is what -(1-0) gives)
            # The new term is 0 regardless. So this is correct.
        
        # n = q: h'[q]=1, h[q]=0
        #   term_new = 1*(1-h'[q-lag]) (if q-lag in [0,N-1])
        #   term_old = 0*(...) = 0
        if 0 <= q - lag < N:
            new_val = h[q-lag]  # h'[q-lag] = h[q-lag] unless q-lag==p, in which case h'[p]=0
            if q - lag == p:
                new_val = 0.0  # h'[p] = 0
            d += (1.0 - new_val)
        
        # n = p+lag: (where n-lag = p)
        #   term_new = h'[p+lag]*(1-0) = h'[p+lag] (since h'[p]=0)
        #   term_old = h[p+lag]*(1-1) = 0 (since h[p]=1)
        if 0 <= p + lag < N:
            new_val = h[p+lag]  # h'[p+lag] = h[p+lag] unless p+lag==q, then h'[q]=1
            if p + lag == q:
                new_val = 1.0
            d += new_val  # new - old = new_val - 0 = new_val
        
        # n = q+lag: (where n-lag = q)
        #   term_new = h'[q+lag]*(1-1) = 0 (since h'[q]=0... wait h'[q]=1)
        #   Wait: h'[q] = 1. So (1-h'[q]) = 0.
        #   term_new = h'[q+lag] * 0 = 0
        #   term_old = h[q+lag] * (1-h[q]) = h[q+lag] * 1 = h[q+lag] (since h[q]=0)
        if 0 <= q + lag < N:
            old_val = h[q+lag]
            d += -(old_val)  # new - old = 0 - old_val
        
        delta[m] = d
    
    return delta
```

This is O(N) per swap (iterating over all m), but with small constant. For N=380, this is about 759 iterations per swap. With 10000 swaps to evaluate, that's 7.6M operations - very fast.

Actually, the non-zero entries are at:

Actually, a much simpler approach: since the delta is O(N) to compute and we just need the max of (corr + delta), we can do:

```python
def eval_swap(corr, h, p, q, N):
    """Return new max correlation after swap p(1->0), q(0->1)"""
    center = N - 1
    new_corr = corr.copy()
    for m in range(2*N-1):
        lag = m - center
        d = 0.0
        if 0 <= p - lag < N:
            d -= (1.0 - h[p-lag])
        if 0 <= q - lag < N:
            nv = h[q-lag]
            if q - lag == p: nv = 0.0
            d += (1.0 - nv)
        if 0 <= p + lag < N:
            nv = h[p+lag]
            if p + lag == q: nv = 1.0
            d += nv
        if 0 <= q + lag < N:
            d -= h[q+lag]
        if d != 0:
            new_corr[m] += d
    return float(new_corr.max())
```

This is O(N) per evaluation. For N=380, about 759 iterations. If we evaluate 5000 swaps per sweep, that's 3.8M iterations per sweep, which takes maybe 10-20ms in Python (or faster with numpy). Actually, a Python loop over 759 elements × 5000 = 3.8M iterations would take about 1-2 seconds. That's too slow.

Let me vectorize this. The key insight is that for a given (p, q), the delta is non-zero only at specific lags. Let me figure out which:

The delta is non-zero when:
1. p - lag ∈ [0, N-1] → lag ∈ [p-N+1, p]
2. q - lag ∈ [0, N-1] → lag ∈ [q-N+1, q]  
3. p + lag ∈ [0, N-1] → lag ∈ [-p, N-1-p]
4. q + lag ∈ [0, N-1] → lag ∈ [-q, N-1-q]

So the union of affected lags is: [min(p-N+1, q-N+1, -p, -q), max(p, q, N-1-p, N-1-q)]

In the worst case, this covers all lags from -(N-1) to N-1, i.e., all 2N-1 entries. But in practice, for a given (p,q), it's a subset.

For vectorization, I can precompute the delta for a given (p,q) as:

```python
def eval_swap_fast(corr, h, p, q, N):
    center = N - 1
    L = 2*N - 1
    new_corr = corr.copy()
    
    # Contribution 1: for m in [p, min(2p+center, L-1)]: new_corr[m] += h[center+p-m] - 1
    lo1 = max(p, 0)
    hi1 = min(2*p + center, L - 1)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        h_idx = center + p - idx
        new_corr[idx] += h[h_idx] - 1.0
    
    # Contribution 2: for m in [q, min(2q+center, L-1)]: new_corr[m] += 1 - h'[center+q-m]
    lo2 = max(q, 0)
    hi2 = min(2*q + center, L - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        h_idx = center + q - idx
        vals = 1.0 - h[h_idx]
        # Fix where h_idx == p (h'[p]=0)
        mask = (h_idx == p)
        vals[mask] = 1.0  # 1 - 0
        new_corr[idx] += vals
    
    # Contribution 3: for m in [center-p, min(center+N-1-p, L-1)]: new_corr[m] += h'[m-(center-p)]
    lo3 = max(center - p, 0)
    hi3 = min(center + N - 1 - p, L - 1)
    if lo3 <= hi3:
        idx = np.arange(lo3, hi3 + 1)
        h_idx = idx - (center - p)
        vals = h[h_idx]
        # Fix where h_idx == q (h'[q]=1)
        mask = (h_idx == q)
        vals = vals.copy()
        vals[mask] = 1.0
        new_corr[idx] += vals
    
    # Contribution 4: for m in [center-q, min(center+N-1-q, L-1)]: new_corr[m] -= h[m-(center-q)]
    lo4 = max(center - q, 0)
    hi4 = min(center + N - 1 - q, L - 1)
    if lo4 <= hi4:
        idx = np.arange(lo4, hi4 + 1)
        h_idx = idx - (center - q)
        new_corr[idx] -= h[h_idx]
    
    return float(new_corr.max())
```
