    N = len(h)

    return np.real(np.fft.ifft(H * G))

        else:

def compute_D(h, N):
    """Compute D[j, k] = effect of changing h[j] on correlation at full index k"""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]

    pp = pos + sv

def smooth_max_grad(h, N, mu=0.01, top_k=100):
    """Compute gradient of smooth max of correlations"""
    corr = fft_corr(h)
    # Smooth max: (1/mu) * log(sum exp(mu * c_k))
    # But for numerical stability, use top-k
    # Weight by softmax of top-k
    c_max = corr.max()
    # Weights: exp((c_k - c_max) / mu) for top k
    idx = np.argsort(corr)[-top_k:]
    w = np.exp((corr[idx] - c_max) / mu)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for i, ai in enumerate(idx):
        L = int(ai - center)
        if L == 0:
            grad += w[i] * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w[i] * (1.0 - h[:N-L])
            grad[:N-L] -= w[i] * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += w[i] * (1.0 - h[M:])
                grad[M:] -= w[i] * h[:M]
    return grad, corr.max()

def grad_smooth_max(h, N, mu, top_k):
    corr = fft_corr(h)
    c_max = corr.max()
    idx = np.argsort(corr)[-top_k:]
    w = np.exp((corr[idx] - c_max) / mu)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for i, ai in enumerate(idx):
        L = int(ai - center)
        wi = w[i]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad, float(c_max)

    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        grad, pk = grad_smooth_max(y, N, mu, top_k)
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.995
            beta = min(0.97, 0.88 + it * 0.0001)
            mu = max(0.005, mu * 0.95)
    return best_h, best_pk * dx

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    shifts = np.arange(2 * N - 1) - (N - 1)
