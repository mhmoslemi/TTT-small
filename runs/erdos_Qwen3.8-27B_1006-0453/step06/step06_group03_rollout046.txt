    c = fft_corr(h)

    log_sum = cmax + np.log(np.exp(c/T - cmax/T).sum())

        else:
            M = -Lag
            grad[:N-M] += wm * (1.0 - h[M:])
            grad[M:] -= wm * h[:N-M]
    return obj, grad

    best_c = float(np.max(fft_corr(h)))

        b01 = list(np.where((h == 0) & (hp1 == 1))[0])

        if len(b10) < 2 or len(b01) < 2:
            break

        # Try moving a block: shift b10 forward AND b01 forward simultaneously
        n_try = min(40, len(b10) * len(b01))
        best_delta = 1e-14
        best_move = None
        attempts = 0
        for i in b10:
            for j in b01:
                if attempts >= n_try:
                    break
                p1, q1 = i, (i + 1) % N
                p2, q2 = (j - 1) % N, j
                if p1 == p2 or p1 == q2 or q1 == p2 or q1 == q2:

                h[p1] -= 1.0
                h[q1] += 1.0
                h[p2] -= 1.0

                new_P = float(np.max(fft_corr(h)))
