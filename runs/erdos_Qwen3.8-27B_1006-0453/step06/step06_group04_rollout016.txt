import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_gradient(corr, y, N, k, temp):
    center = N - 1
    upper = min(k, 2 * N - 2)
    top = np.argpartition(corr, -upper)[-upper:]
    w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-10))
    ws = w.sum()
    if ws < 1e-15:
        return np.zeros(N)
    w = w / ws
    grad = np.zeros(N)
    for wi, ai in zip(w, top):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif L > 0:
            seg = N - L
            if seg > 0:
                grad[L:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:seg]
    return grad

def adam_subgrad(h0, N, deadline, k_start=50, k_end=140, lr0=0.012, temp0=0.004):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m_v = np.zeros(N)
    m_sq = np.zeros(N)
    b1, b2 = 0.85, 0.998
    t = 0
    lr = lr0
    temp = temp0
    beta = 0.88
    h_prev = h.copy()
    stall = 0
    total_time = deadline - time.time()
    while time.time() < deadline:
        t += 1
        progress = min(1.0, t / max(1.0, total_time * 12))
        k_cur = k_start + int(progress * (k_end - k_start))
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        grad = topk_gradient(corr, y, N, k_cur, temp)
        m_v = b1 * m_v + (1 - b1) * grad
        m_sq = b2 * m_sq + (1 - b2) * grad * grad
        m_hat = m_v / (1 - b1 ** t)
        sq_hat = m_sq / (1 - b2 ** t)
        h_new = y - lr * m_hat / (np.sqrt(sq_hat) + 1e-7)
        h = project_sum(h_new, target)
        if t % 400 == 0:

            m_v = np.zeros(N)
            m_sq = np.zeros(N)
