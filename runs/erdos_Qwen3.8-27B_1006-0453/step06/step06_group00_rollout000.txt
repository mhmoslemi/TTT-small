    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def corr_values(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def project_binary(h, target_ones):
    """Project to binary with exact target_ones ones."""
    h = (h > 0.5).astype(float)
    current = int(h.sum())
    if current == target_ones:
        return h
    if current > target_ones:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:current - target_ones]] = 0
    else:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target_ones - current]] = 1
    return h

def single_swap_eval(h, corr, i, j, N):
    """Evaluate C5 after swapping h[i]=1→0, h[j]=0→1. Return new max correlation."""
    # Instead of recomputing full correlation, use incremental update
    # correlate(h, 1-h)[k] changes when we flip positions
    # This is complex; for simplicity, recompute
    new_h = h.copy()
    new_h[i] = 0
    new_h[j] = 1
    new_corr = np.correlate(new_h, 1.0 - new_h, mode="full")
    return float(np.max(new_corr))
    
def anneal_binary(h, N, deadline, max_temp=2.0, cool_rate=0.9997, max_iters=50000):
    """Simulated annealing on binary h."""
    h = h.copy()
    target = N // 2
    best_h = h.copy()
    best_c = compute_c5(h, N)
    curr_c = best_c
    temp = max_temp
    center = N - 1  # center of correlate 'full' output
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Propose a swap: pick one 1 and one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Compute new C5
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c = compute_c5(new_h, N)
        
        delta = new_c - curr_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):
            h = new_h
            curr_c = new_c
            if curr_c < best_c:
                best_c = curr_c
                best_h = h.copy()
        
        temp *= cool_rate
        if it % 1000 == 0:
            # Periodic reheating
            if it % 5000 == 0 and it > 0:
                temp = max(temp, max_temp * 0.1)
                
    return best_h, best_c

def gradient_refine(h, N, deadline, n_iter=5000):
    """Subgradient-style refinement for continuous h in [0,1]."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = corr_values(h)
        curr_max = float(np.max(corr))
        if curr_max * dx < best_c:
            best_c = curr_max * dx
            best_h = h.copy()
        
        # Find top contributing shifts
        top_k = np.argsort(corr)[-10:]
        
        # Subgradient: for each top shift k, gradient of corr[k] w.r.t. h
        # corr[k] = sum_n h[n+k-center]*(1-h[n]) for the appropriate range
        # d/dh[m] of corr[k] depends on whether m is in the "a" or "b" part
        grad = np.zeros(N)
        for k in top_k:
            d = k - (N - 1)  # actual shift
            if d == 0:
                # corr[center] = sum h[n]*(1-h[n]), grad = 1-2h
                grad += (1 - 2 * h)
            elif d > 0:
                # sum_{n=0}^{N-d-1} h[n+d]*(1-h[n])
                # d/dh[n] = -(h>n+d exists? 1) * h[n+d] for n < N-d
                # d/dh[n] = (1-h[n-d]) for n >= d
                for n in range(N):
                    if n < N - d:
                        grad[n] += -h[n + d]  # h[n] is in (1-h[n]) part... wait
                    if n >= d:
                        grad[n] += (1 - h[n - d])  # h[n] is in h[n] part
        
        # This is O(N * top_k) per iteration which is fine for N=380
        # Normalize and step
        grad /= len(top_k)
        lr = 0.1 / (1 + it * 0.001)
        h_new = h - lr * grad
        # Project to [0,1] with sum constraint
        h_new = np.clip(h_new, 0, 1)
        # Adjust sum
        s = h_new.sum()
        if abs(s - target) > 1e-8:
            # Simple projection
            h_new = h_new - (s - target) / N
            h_new = np.clip(h_new, 0, 1)
            # Re-adjust
            s = h_new.sum()
            if s > target:
                excess = s - target
                # Reduce largest values
                order = np.argsort(h_new)[::-1]
                for idx in order:
                    reduce = min(excess, h_new[idx])
                    h_new[idx] -= reduce
                    excess -= reduce
                    if excess < 1e-10:
                        break
            elif s < target:
                deficit = target - s
                order = np.argsort(h_new)
                for idx in order:
                    add = min(deficit, 1 - h_new[idx])
                    h_new[idx] += add
                    deficit -= add
                    if deficit < 1e-10:
                        break
        h = h_new
    
    return best_h, best_c
