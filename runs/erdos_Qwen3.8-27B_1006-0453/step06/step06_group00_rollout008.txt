def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient(h, y, k=100):
    """Compute subgradient of max_k corr_k w.r.t. h, given evaluation at y"""
    corr = fft_corr(y)
    N = len(y)
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    return grad / k, corr

    best_corr = fft_corr(h)

    k = k_start

    while time.time() < deadline:
        y = h + m

        pk = float(np.max(corr))

            best_h = h.copy()
            stall = 0
