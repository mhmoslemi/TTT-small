        h[idx] = 1

def binary_opt(h0, N, deadline, n_tries=30, directed=True, multi_period=35):
    h = _fix_binary(h0.copy(), N)
    c = fft_corr(h)
    best_int = int(c.max())
    best_h = h.copy()
    stall = 0
    it = 0
    center = N - 1
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sweep_int = best_int
        best_sweep_h = None
        best_sweep_c = None
        p_probs = None
        q_probs = None
        if directed:
            top_k = min(40, len(c) - 1)
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            tv = c[top_idx]
            wm = tv - tv.max() + 1.0
            os_ = np.full(len(ones), 0.05)
            zs_ = np.full(len(zeros), 0.05)
            for t in range(top_k):
                lag = top_idx[t] - center
                w = wm[t]
                if 0 < lag < N:
                    v1 = (ones - lag >= 0) & (ones - lag < N)
                    os_[v1] += w * (1.0 - h[ones[v1] - lag])
                    v2 = (zeros + lag >= 0) & (zeros + lag < N)
                    zs_[v2] += w * h[zeros[v2] + lag]
                elif -N < lag < 0:
                    M = -lag
                    v1 = (ones + M >= 0) & (ones + M < N)
                    os_[v1] += w * (1.0 - h[ones[v1] + M])
                    v2 = (zeros - M >= 0) & (zeros - M < N)
                    zs_[v2] += w * h[zeros[v2] - M]
            p_probs = os_ / os_.sum()
            q_probs = zs_ / zs_.sum()
        for _ in range(n_tries):
            if p_probs is not None:
                p = ones[np.random.choice(len(ones), p=p_probs)]
                q = zeros[np.random.choice(len(zeros), p=q_probs)]
            else:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = fft_corr(h)
            ct = int(c_t.max())
            if ct < best_sweep_int:
                best_sweep_int = ct
                best_sweep_h = h.copy()
                best_sweep_c = c_t.copy()
            h[p] = 1
            h[q] = 0
        if it % multi_period == multi_period - 1 and len(ones) >= 2 and len(zeros) >= 2:
            pm = ones[np.random.choice(len(ones), 2, replace=False)]
            qm = zeros[np.random.choice(len(zeros), 2, replace=False)]
            h[pm] = 0
            h[qm] = 1
            c_t = fft_corr(h)
            ct = int(c_t.max())
            if ct < best_sweep_int:
                best_sweep_int = ct
                best_sweep_h = h.copy()
                best_sweep_c = c_t.copy()
            h[pm] = 1
            h[qm] = 0
        if best_sweep_h is not None:
            h = best_sweep_h
            c = best_sweep_c
            if best_sweep_int < best_int:
                best_int = best_sweep_int
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        it += 1
        if stall > 70:
            h = best_h.copy()
            nf = np.random.randint(3, 7)
            fo = np.where(h == 1)[0]
            fz = np.where(h == 0)[0]
            for _ in range(nf):
                p = fo[np.random.randint(len(fo))]
                q = fz[np.random.randint(len(fz))]
                h[p] = 0
                h[q] = 1
            c = fft_corr(h)
            stall = 0
    return best_h, best_int * (2.0 / N)

def continuous_opt(h0, N, deadline, k=60, lr0=0.02, temp0=0.003):
    h = h0.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    lr = lr0
    beta = 0.9
    temp = temp0
    stall = 0
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        top = np.argpartition(corr, -k)[-k:]
        w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-10))
        w /= w.sum()
        grad = np.zeros(N)
        for wi, ai in zip(w, top):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                seg = N - L
                grad[L:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    seg = N - M
                    grad[:seg] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:seg]
        h = project_sum(y - lr * grad, target)
        it += 1
        if it % 200 == 0:
            lr = max(0.003, lr * 0.995)
            temp = max(0.0005, temp * 0.993)
            beta = min(0.97, 0.88 + it * 0.0002)
        if stall > 300:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = max(lr, lr0 * 0.8)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_N = N0
    best_c5 = compute_c5(best_h, N0)

    d1 = min(deadline - 620, t0 + 370)
    hb, cb = binary_opt(h0, N0, d1, n_tries=35, directed=True, multi_period=30)
    if cb < best_c5:
        best_c5, best_h, best_N = cb, hb.copy(), N0

    d2 = max(d1, t0) + 240
    d2 = min(deadline - 320, d2)
    hc = best_h + np.random.randn(len(best_h)) * 0.04
    hc = project_sum(hc, len(best_h) / 2.0)
    hn, c5n = continuous_opt(hc, N0, d2, k=60, lr0=0.02, temp0=0.003)
    if c5n < best_c5:
        best_c5, best_h, best_N = c5n, hn.copy(), N0

    d3 = max(d2, t0) + 240
    d3 = min(deadline - 150, d3)
    hb3, cb3 = binary_opt(best_h, N0, d3, n_tries=40, directed=True, multi_period=25)
    if cb3 < best_c5:
        best_c5, best_h, best_N = cb3, hb3.copy(), N0

    N2 = 760
    if time.time() < deadline - 130:
        d4 = min(deadline - 40, max(d3, t0) + 120)
        hi = interp_to_N(best_h, N2)
        hb2 = _fix_binary(hi, N2)
        h_rt, c5_rt = binary_opt(hb2, N2, d4, n_tries=30, directed=True, multi_period=30)
        c5_rt_v = compute_c5(h_rt, N2)
        if c5_rt_v < best_c5:
            best_c5, best_h, best_N = c5_rt_v, h_rt.copy(), N2
        hd = interp_to_N(h_rt, N0)
        hd = _fix_binary(hd, N0)
        c5_d = compute_c5(hd, N0)
        if c5_d < best_c5:
            best_c5, best_h, best_N = c5_d, hd.copy(), N0

    if time.time() < deadline - 5:
        hF = best_h + np.random.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = continuous_opt(hF, best_N, deadline - 4, k=80, lr0=0.008, temp0=0.001)
        if c5F < best_c5:
            best_c5, best_h, best_N = c5F, hF.copy(), best_N
        if time.time() < deadline - 2:
            hbp, c5bp = binary_opt(best_h, best_N, deadline - 2, n_tries=25, directed=True, multi_period=20)
            if c5bp < best_c5:
                best_c5, best_h, best_N = c5bp, hbp.copy(), best_N

    final_c5 = compute_c5(best_h, best_N)
    return best_h, final_c5, best_N
