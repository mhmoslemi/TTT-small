- h[j] appears as h[i+M] when i=j-M, if 0 <= j-M <= N-1-M (i.e., M <= j <= N-1): contributes (1-h[j-M])
- h[j] appears as h[i] when i=j, if 0 <= j <= N-1-M: contributes -h[j+M]

So: dr(-M)/dh[j] = (1-h[j-M]) * [M <= j <= N-1] - h[j+M] * [0 <= j <= N-1-M]

Let me verify with a simple example. h = [a, b, c], N=3.
r(shift=1) = h[0]*(1-h[1]) + h[1]*(1-h[2]) = a(1-b) + b(1-c)
dr/dh[0] = (1-b) ✓ (j=0, s=1: j<=N-1-s=1, and s<=j is 1<=0 which is false, so just (1-h[1]))
dr/dh[1] = (1-c) - h[0] = (1-c) - a ✓ (j=1, s=1: j<=1 so (1-h[2]), and s<=j<=N-1 so -h[0])
dr/dh[2] = -h[1] = -b ✓ (j=2, s=1: j<=1 is false, s<=j<=2 so -h[1])

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    tol = 1e-9
    if abs(s - target) < tol:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_weighted_grad(y, N, w):
    """Compute sum_k w[k] * d(corr[k])/dh[j] for all j."""
    grad = np.zeros(N)
    center = N - 1
    L = len(w)
    for k in range(L):
        if w[k] < 1e-15:
            continue
        wi = w[k]
        s = k - center  # shift
        if s == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif s > 0:
            seg = N - s
            if seg > 0:
                grad[:seg] += wi * (1.0 - y[s:])
                grad[s:] += -wi * y[:seg]
        else:
            M = -s
            seg = N - M
            if seg > 0:
                grad[M:] += wi * (1.0 - y[:seg])
                grad[:seg] += -wi * y[M:]
    return grad

    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = float(np.max(fft_corr(h))) * (2.0 / N)
    h_prev = h.copy()
    beta = 0.85
    lr = lr0
    it = 0
    stall = 0
    n_total = len(h)
    
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        y = np.clip(y, 0.0, 1.0)
        
        corr = fft_corr(y)
        c5_curr = float(np.max(corr)) * (2.0 / N)
        
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = y.copy()
            stall = 0
        else:
            stall += 1

        T = T_start + (T_end - T_start) * min(1.0, it / 8000.0)
        c_max = corr.max()
        exp_arg = T * (corr - c_max)
        # Clip to avoid overflow
        exp_arg = np.clip(exp_arg, -50, 50)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum < 1e-300:
            idx = np.argmax(corr)
            w = np.zeros(len(corr))
            w[idx] = 1.0
        else:
            w /= w_sum
        
        grad = compute_weighted_grad(y, N, w)
        
        h_new = project_sum(y - lr * grad, target)
        h = h_new

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    c = fft_corr(h)
    return float(c.max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def wgrad(y, N, w):
    """Weighted gradient of cross-correlation."""
    grad = np.zeros(N)
    c = N - 1
    for k in range(len(w)):
        wi = w[k]
        if wi < 1e-14:
            continue
        s = k - c
        if s == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif s > 0:
            m = N - s
            if m > 0:
                grad[:m] += wi * (1.0 - y[s:m+s])
                if s < N:
                    grad[s:] += -wi * y[:m]
        else:
            M = -s
            m = N - M
            if m > 0:
                if M < N:
                    grad[M:] += wi * (1.0 - y[:m])
                    grad[:m] += -wi * y[M:M+m]
    return grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def wgrad(y, N, w):
    grad = np.zeros(N)
    c = N - 1
    Lw = len(w)
    for k in range(Lw):
        wi = w[k]
        if wi < 1e-14:
            continue
        s = k - c
        if s == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif s > 0:
            m = N - s
            if m > 0:
                grad[:m] += wi * (1.0 - y[s:s + m])
                grad[s:] += -wi * y[:m]
        else:
            M = -s
            m = N - M
            if m > 0:
                if M < N:
                    grad[M:] += wi * (1.0 - y[:m])
                    grad[:m] += -wi * y[M:M + m]
    return grad

def cont_opt(h0, N, deadline, T0=200, T1=5000, lr0=0.018):
    h = h0.astype(np.float64).copy()
    tgt = N / 2.0
    bh = h.copy()
    bc = float(fft_corr(h).max()) * (2.0 / N)
    hp = h.copy()
    beta = 0.84
    lr = lr0
    it = 0
    stall = 0
    while time.time() < deadline:
        y = h + beta * (h - hp)
        hp = h
        y = np.clip(y, 0.0, 1.0)
        corr = fft_corr(y)
        cv = float(corr.max()) * (2.0 / N)
        if cv < bc - 1e-13:
            bc = cv
            bh = y.copy()
            stall = 0

        cm = corr.max()

        w = np.exp(ea)

        if ws < 1e-300:

            w[np.argmax(corr)] = 1.0
        else:

        g = wgrad(y, N, w)
        h = project_sum(y - lr * g, tgt)

        if stall > 350:

            h = project_sum(h, tgt)
            hp = h.copy()
            lr = max(lr, lr0 * 0.7)

    return bh, bc

def bin_opt(h0, N, dl, ntr=15, kf=40):

    nh = N // 2
    n1 = int(h.sum())

        h[idx] = 0
    elif n1 < nh:

        idx = idx[np.random.permutation(len(idx))][:nh - n1]
        h[idx] = 1
    c = fft_corr(h)
    bi = float(c.max())
    bh = h.copy()
    st = 0

        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        bs = float(c.max())
        bh2 = None
        bc2 = None
        do2 = (len(ones) >= 2 and len(zeros) >= 2 and np.random.rand() < 0.25)
        for _ in range(ntr):

                i2 = np.random.randint(len(zeros))

                rem_o = np.delete(ones, i1, 0)
                rem_z = np.delete(zeros, i2, 0)
                if len(rem_o) < 1 or len(rem_z) < 1:
                    continue
                p2 = rem_o[np.random.randint(len(rem_o))]
                q2 = rem_z[np.random.randint(len(rem_z))]
                if p2 == p1 or q2 == q1:
                    continue
                h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                ct = fft_corr(h)
                cv2 = float(ct.max())
                if cv2 < bs:
                    bs = cv2; bh2 = h.copy(); bc2 = ct.copy()
                h[p1] = 1; h[p2] = 1; h[q1] = 0; h[q2] = 0
            else:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]

                    h[ps] = 0; h[qs] = 1
                    c = fft_corr(h)
                st = 0
    return bh, bi * (2.0 / N)

    No = len(h)
    xo = np.linspace(0, 2, No, endpoint=False)
    xn = np.linspace(0, 2, Nn, endpoint=False)
    return np.interp(xn, xo, h)
