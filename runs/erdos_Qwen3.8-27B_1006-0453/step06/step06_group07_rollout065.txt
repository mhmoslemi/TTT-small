import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = -(1.0 - hm) * vm + hp * vp
    D[:, N - 1] = 0.0
    return D

def ensure_sum(h, N):

    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    rng = np.random.RandomState(54321 + N)
    if n1 != nh:

        else:

    best_c = float(np.max(fft_corr(h)))

    rng = np.random.RandomState(seed)

    T = T0

        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        T = max(T, 0.001)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        D = compute_D(h, N)

    best_h = h.copy()

    for rnd in range(max_rounds):

            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        D = compute_D(h, N)
        n_try = min(30, len(ones))
        improved = False
        for _ in range(n_try):
            p1 = int(ones[rng.randint(len(ones))])
            p2 = int(ones[rng.randint(len(ones))])
            if p1 == p2:
                continue
            q1 = int(zeros[rng.randint(len(zeros))])
            q2 = int(zeros[rng.randint(len(zeros))])
            if q1 == q2:
                continue
            c_new = c + D[p1] + D[p2] - D[q1] - D[q2]
            if c_new.max() < P - 1e-10:
                h[p1] = 0.0
                h[p2] = 0.0
                h[q1] = 1.0
                h[q2] = 1.0
                improved = True
                break
        if not improved:
            break
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=80, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        weights = np.exp(5.0 * (corr[top] - corr[top].max()) / max(corr[top].max(), 1e-10))
        weights /= weights.sum()
        grad = np.zeros(N)
        center = N - 1
        for idx, ai in enumerate(top):
            L = int(ai - center)
            w = weights[idx]
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w * (1.0 - y[:N - L])
                grad[:N - L] -= w * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - y[M:])
                    grad[M:] -= w * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.985
            beta = min(0.97, 0.88 + it * 0.00008)
    return best_h, best_pk * dx
