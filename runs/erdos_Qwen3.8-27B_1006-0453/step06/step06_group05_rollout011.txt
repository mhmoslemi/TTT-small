    h = h.astype(float).copy()
    target = N / 2.0

```python
def compute_grad(h, weights):
    N = len(h)
    w = weights  # length 2N-1
    L = 1  # enough for linear conv/correlation
    # Need L >= N + (2N-1) - 1 = 3N-2
    while L < 3*N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    W = np.fft.fft(w, L)
    conv = np.real(np.fft.ifft(H * W))  # linear convolution h*w, length N+2N-2
    xcorr = np.real(np.fft.ifft(H * np.conj(W)))  # cross-correlation
    
    # A[j] = conv[j + N - 1] for j = 0..N-1
    # B[j] = xcorr[(j - (N-1)) mod L] for j = 0..N-1
    # For j-(N-1): ranges from -(N-1) to N-1
    # In FFT indexing: lag d maps to index d if d>=0, or L+d if d<0
    idx_B = np.arange(N) - (N-1)  # -(N-1), ..., 0, ..., N-1
    idx_B = idx_B % L  # convert to FFT indices
    idx_A = np.arange(N) + (N-1)  # N-1, ..., 2N-2
    
    A = conv[idx_A]
    B = xcorr[idx_B]
    return 1.0 - A - B
```
