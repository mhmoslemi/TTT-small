    # Shift so index 0 corresponds to s=0

    C_s = C[center - (N-1) : center + N]  # indices 0..2N-2, representing s = -(N-1)..N-1
    # Actually C is already length 2N-1, C[i] corresponds to some shift
    # Let's just work with C directly

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01_sum(v, target):
    v = np.clip(v, 0, 1)
    s = v.sum()
    if abs(s - target) < 1e-14:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def smooth_max_gradient(h, N, tau):
    C = fft_corr(h)
    center = N - 1
    logw = tau * C - tau * C.max()
    w = np.exp(logw)
    w /= w.sum()
    grad = np.zeros(N)
    thresh = w.max() * 0.005
    active = np.where(w > thresh)[0]
    for i in active:
        s = int(i - center)
        wi = float(w[i])
        idx = np.arange(N) + s
        m = (idx >= 0) & (idx < N)
        grad[m] += wi * (1.0 - h[idx[m]])
        idx2 = np.arange(N) - s
        m2 = (idx2 >= 0) & (idx2 < N)
        grad[m2] -= wi * h[idx2[m2]]
    return grad

def adam_opt(h, N, deadline, tau=80, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    while time.time() < deadline:
        t += 1
        grad = smooth_max_gradient(h, N, tau)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_01_sum(h_new, target)
        h = h_new
        if t % 50 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            # Anneal tau
            tau = min(200, tau * 1.02)
            lr *= 0.998
    c = compute_c5(h)
    if c < best_c:
        best_h, best_c = h.copy(), c
    return best_h, best_c

def binary_flip_search(h, N, deadline, max_iters=200, n_trials=30):
    h = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        C = fft_corr(h)
        pk = float(C.max())
        if pk * (2.0/N) < best_c:
            best_c = pk * (2.0/N)
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        improved = False
        for _ in range(n_trials):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i] = 0
            h[j] = 1
            new_pk = float(fft_corr(h).max())
            if new_pk <= pk:
                pk = new_pk
                improved = True
            else:
                h[i] = 1
                h[j] = 0
        if not improved:
            # Try random restart
            if np.random.random() < 0.1:
                for _ in range(5):
                    i = np.random.choice(np.where(h == 1)[0])
                    j = np.random.choice(np.where(h == 0)[0])
                    h[i] = 0
                    h[j] = 1
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-14:
        return v.copy()
    lo, hi = -0.5, 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_gradient(h, N, tau):
    C = fft_corr(h)
    center = N - 1
    logw = tau * C - tau * C.max()
    w = np.exp(logw)
    w /= w.sum()
    grad = np.zeros(N)
    thresh = w.max() * 0.003
    active = np.where(w > thresh)[0]
    for i in active:
        s = int(i - center)
        wi = float(w[i])
        idx = np.arange(N) + s
        m1 = (idx >= 0) & (idx < N)
        grad[m1] += wi * (1.0 - h[idx[m1]])
        idx2 = np.arange(N) - s
        m2 = (idx2 >= 0) & (idx2 < N)
        grad[m2] -= wi * h[idx2[m2]]
    return grad
