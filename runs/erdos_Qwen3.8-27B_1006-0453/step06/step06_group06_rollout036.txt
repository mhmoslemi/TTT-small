import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0-h, L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    # Binary search for threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def to_binary(h, N):
    target = N // 2
    h_bin = h.copy()
    # Sort by value, set top N/2 to 1, rest to 0
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(N)
    h_bin[idx[:target]] = 1.0
    return h_bin

def compute_grad_topk(h, corr, N, top_k):
    center = N - 1
    idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(N)
    for ai in idx:
        s = int(ai - center)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += (1.0 - h[s:])
            grad[s:] -= h[:N-s]
        else:
            s2 = -s
            grad[s2:] += (1.0 - h[:N-s2])
            grad[:N-s2] -= h[s2:]
    grad /= top_k
    return grad

def adam_optimize(h, N, deadline, top_k=30, lr0=0.005):

    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    patience = 0

        if pk < best_pk - 1e-9:
            best_pk = pk
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 3000:

        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)

        lr = lr0 / (1.0 + t * 0.0001)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        corr = fft_corr(h)
    return best_h, best_pk

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def to_binary(h, N):
    target = N // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(N)
    h_bin[idx[:target]] = 1.0
    return h_bin

def grad_topk(h, corr, N, top_k):
    center = N - 1
    idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(N)
    for ai in idx:
        s = int(ai - center)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += (1.0 - h[s:])
            grad[s:] -= h[:N-s]
        else:
            s2 = -s
            grad[s2:] += (1.0 - h[:N-s2])
            grad[:N-s2] -= h[s2:]
    grad /= top_k
    return grad

def adam_opt(h, N, deadline, top_k=40, lr0=0.004):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        pk = float(corr.max())
        if pk < best_pk - 1e-9:
            best_pk = pk; best_h = h.copy(); stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
        grad = grad_topk(h, corr, N, top_k)
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        lr = lr0 / (1 + 0.0002*t)
        h = proj_sum(h - lr*m/(1-b1**t) / np.sqrt(v/(1-b2**t) + eps), N/2.0)
        corr = fft_corr(h)
    return best_h, best_pk

def bin_anneal(hb, N, deadline, T0=0.06, cool=0.9994, stall_max=600, heats=3):
    h = hb.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    s_arr = np.arange(-(N-1), N)
    for heat in range(heats):
        T = T0 * (0.5 ** heat)
        stall = 0
        while time.time() < deadline and T > 0.003:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            ps = p + s_arr; pm = p - s_arr
            qs = q + s_arr; qm = q - s_arr
            vps = (ps>=0)&(ps<N); vpm = (pm>=0)&(pm<N)
            vqs = (qs>=0)&(qs<N); vqm = (qm>=0)&(qm<N)
            d = np.zeros(2*N-1)
            d[vps] -= (1.0-h[np.clip(ps,0,N-1)[vps]])
            d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
            d[vqs] += (1.0-h[np.clip(qs,0,N-1)[vqs]])
            d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
            nc = corr + d
            npk = float(nc.max())
            cur_pk = float(corr.max())
            delta = npk - cur_pk

            if stall > stall_max:
                h = best_h.copy(); corr = fft_corr(h)

                stall = 0
    return best_h, best_pk

def grad_logsumexp(h, corr, N, beta):
    center = N - 1
    # Softmax weights
    shifted = corr - corr.max()  # numerical stability
    weights = np.exp(beta * shifted)
    total = weights.sum()
    if total < 1e-300:
        # Fallback to argmax
        ai = np.argmax(corr)
        s = int(ai - center)
        if s == 0:
            return 1.0 - 2.0*h
        elif s > 0:
            g = np.zeros(N)
            g[:N-s] += (1.0-h[s:])
            g[s:] -= h[:N-s]
            return g
        else:
            s2 = -s
            g = np.zeros(N)
            g[s2:] += (1.0-h[:N-s2])
            g[:N-s2] -= h[s2:]
            return g
    
    # Weighted sum of gradients - but this requires summing over ALL k
    # which is expensive. Instead, use top effective-K.
    # Approximate: take top K by weight
    sorted_w = np.sort(weights)[::-1]
    cumsum = np.cumsum(sorted_w)
    K = int(np.searchsorted(cumsum, 0.99 * total)) + 1  # 99% of weight
    K = min(K, N)
    
    top_idx = np.argsort(weights)[::-1][:K]
    grad = np.zeros(N)
    for ai in top_idx:
        w = weights[ai] / total
        s = int(ai - center)

            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:N-s]
        else:
            s2 = -s
            grad[s2:] += w * (1.0 - h[:N-s2])

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def to_binary(h, N):
    target = N // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(N)
    hb[idx[:target]] = 1.0
    return hb

def single_grad(h, s, N):
    g = np.zeros(N)
    if s == 0:
        return 1.0 - 2.0 * h
    elif s > 0:
        g[:N-s] += (1.0 - h[s:])
        g[s:] -= h[:N-s]
    else:
        s2 = -s
        g[s2:] += (1.0 - h[:N-s2])
        g[:N-s2] -= h[s2:]
    return g

def lse_grad(h, corr, N, beta):
    center = N - 1
    sh = corr - corr.max()
    w = np.exp(beta * sh)
    tot = w.sum()
    if tot < 1e-300:
        return single_grad(h, int(np.argmax(corr) - center), N)
    sorted_idx = np.argsort(w)[::-1]
    cum = np.cumsum(w[sorted_idx])
    K = int(np.searchsorted(cum, 0.98 * tot)) + 1
    K = min(K, 2*N - 1)
    top = sorted_idx[:K]
    grad = np.zeros(N)
    for ai in top:
        wt = w[ai] / tot
        s = int(ai - center)
        grad += wt * single_grad(h, s, N)
    return grad

def adam_lse(h, N, deadline, beta0=8.0, beta1=60.0, lr0=0.004):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    corr = fft_corr(h)
    best_pk = float(corr.max()); best_h = h.copy()
    stall = 0
    total_time = max(deadline - time.time(), 1.0)
    while time.time() < deadline:
        frac = (time.time() - (deadline - total_time)) / total_time
        beta = beta0 + (beta1 - beta0) * frac ** 0.7
        pk = float(corr.max())
        if pk < best_pk - 1e-9:
            best_pk = pk; best_h = h.copy(); stall = 0
        else:
            stall += 1
            if stall > 8000:
                break
        grad = lse_grad(h, corr, N, beta)
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        lr = lr0 / (1.0 + 0.0003*t)
        h = proj_sum(h - lr * m/(1-b1**t) / np.sqrt(v/(1-b2**t) + eps), N/2.0)
        corr = fft_corr(h)
    return best_h, best_pk
