            continue
        d3 = min(deadline - 120, time.time() + 80)
        h_rt, c5_rt = try_resolution(best_h, len(best_h), N_try, d3, max_sweeps=3000, top_m=28)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 4: More binary on best with annealing
    if time.time() < deadline - 100:
        d4 = min(deadline - 80, time.time() + 120)
        h4, _ = vect_binary_search(best_h, len(best_h), d4, max_sweeps=6000, top_m=32, anneal_prob=0.12, stuck_limit=100)
        c4 = compute_c5(h4, len(best_h))
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Final Nesterov
    if time.time() < deadline - 30:
        d5 = min(deadline - 10, time.time() + 60)
        hF = best_h + np.random.randn(len(best_h)) * 0.02
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), d5, k=100, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 6: Final binary cleanup
    if time.time() < deadline - 2:
        hF2 = (best_h > 0.5).astype(float)
        hF2, _ = vect_binary_search(hF2, len(best_h), deadline - 1, max_sweeps=2000, top_m=35)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
