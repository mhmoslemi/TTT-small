import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def bin_search_sa(h, N, deadline, max_sweeps=500, anneal=0.0, sample_frac=1.0):
    """Binary hill climbing with optional simulated annealing."""
    h = h.astype(float).copy()
    nh = N // 2
    # Ensure exactly N/2 ones
    n1 = int(round(h.sum()))
    if n1 != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            keep = np.argsort(-h[ones_i])[:nh]
            h[ones_i[keep]] = 1
        else:
            h[ones_i] = 1
            extra = zeros_i[:nh - n1]
            h[extra] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    step = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 100:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        # Sample a subset of ones for speed
        n_sample = max(1, int(len(ones) * sample_frac))
        if n_sample < len(ones):
            p_idx = np.random.choice(len(ones), n_sample, replace=False)
            p_list = ones[p_idx]
        else:
            p_list = ones
        bd, bpq = 0.0, None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:

        # Accept move
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
            step += 1
        elif anneal > 0:

            T = anneal * max(0.01, 1.0 - step / max_sweeps)

                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                step += 1
    return best_h, best_c

            ht, ct = hill_climb(best_h, N_try, dl, max_iters=400, anneal=0.04)
            c5_t = ct * 2.0 / N_try
            if c5_t < best_c5:
                best_c5, best_h, best_N = c5_t, ht, N_try
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x0, best_h)
        h_bin = make_binary(h_interp, N_try)
        # Also try a perturbed version
        h_pert = h_interp + np.random.randn(N_try) * 0.05
        h_pert_bin = make_binary(h_pert, N_try)
        dl = min(deadline - 380, time.time() + 30)
        ht, ct = hill_climb(h_bin, N_try, dl, max_iters=300)
        c5_t = ct * 2.0 / N_try
        if c5_t < best_c5:
            best_c5, best_h, best_N = c5_t, ht, N_try
        if time.time() < dl:
            ht2, ct2 = hill_climb(h_pert_bin, N_try, min(deadline - 380, time.time() + 15), max_iters=150, anneal=0.06)
            c5_t2 = ct2 * 2.0 / N_try
            if c5_t2 < best_c5:
                best_c5, best_h, best_N = c5_t2, ht2, N_try

    # Phase 3: Long hill climb on best (200s)
    if time.time() < deadline - 250:
        dl3 = min(deadline - 200, time.time() + 200)
        h3, c3 = hill_climb(best_h, best_N, dl3, max_iters=3000, anneal=0.02, sample_frac=0.8)
        c5_3 = c3 * 2.0 / best_N
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 4: Genetic algorithm (120s)
    if time.time() < deadline - 180:
        dl4 = min(deadline - 150, time.time() + 120)
        pop_size = 12
        pop = [best_h.copy()]
        for _ in range(pop_size - 1):
            hp = best_h.copy()
            n_fl = np.random.randint(4, 12)
            oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            for _ in range(n_fl):
                p = int(np.random.randint(len(oi)))
                q = int(np.random.randint(len(zi)))
                hp[oi[p]], hp[zi[q]] = 0.0, 1.0
                oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            pop.append(hp)
        h4, c4 = ga_phase(pop, best_N, dl4, n_gen=20)
        c5_4 = c4 * 2.0 / best_N
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Phase 5: Continuous optimization + binary (150s)
    if time.time() < deadline - 120:
        N_c = best_N
        h_c = best_h.copy() + np.random.randn(N_c) * 0.02
        dl_c = min(deadline - 80, time.time() + 100)
        h_cont, c_cont = cont_optimize(h_c, N_c, dl_c)
        h_bin_c = make_binary(h_cont, N_c)
        dl_b = min(deadline - 50, time.time() + 35)
        h_rb, c_rb = hill_climb(h_bin_c, N_c, dl_b, max_iters=300, anneal=0.03)
        c5_rb = c_rb * 2.0 / N_c
        if c5_rb < best_c5:
            best_c5, best_h, best_N = c5_rb, h_rb, N_c

    # Phase 6: Multi-start from best (100s)
    if time.time() < deadline - 60:
        for att in range(10):
            if time.time() > deadline - 60:
                break
            hp = best_h.copy()
            n_fl = np.random.randint(2, 6)
            oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            for _ in range(n_fl):
                p = int(np.random.randint(len(oi)))
                q = int(np.random.randint(len(zi)))
                hp[oi[p]], hp[zi[q]] = 0.0, 1.0
                oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            dl_s = min(deadline - 60, time.time() + 8)
            hs, cs = hill_climb(hp, best_N, dl_s, max_iters=100, anneal=0.04)
            c5_s = cs * 2.0 / best_N
            if c5_s < best_c5:
                best_c5, best_h = c5_s, hs

    # Phase 7: Final polish (50s)
    if time.time() < deadline - 3:
        hF = make_binary(best_h, best_N)
        hF, cF = hill_climb(hF, best_N, deadline - 1, max_iters=200)
        c5F = cF * 2.0 / best_N
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, best_N)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_N
