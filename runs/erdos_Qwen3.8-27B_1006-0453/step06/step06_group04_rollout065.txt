def compute_gradient(h, N, w):
    """Compute gradient of smooth max of correlation."""
    grad = np.zeros(N)
    L = 2 * N - 1
    for i in range(N):
        for l in range(-N+1, N):
            k = N - 1 + l  # index into w
            if w[k] < 1e-15:

            # Term from (1-h[i-l]) if i-l is valid

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_gradient(h, beta):
    N = len(h)
    corr = fft_corr(h)
    cm = corr.max()
    w = np.exp(beta * (corr - cm))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    # Gradient via convolution
    s1 = fftconvolve(w, 1.0 - h, mode='full')
    s2 = fftconvolve(h, w[::-1], mode='full')
    grad = s1[N-1:2*N-1] - s2[N-1:2*N-1]
    return grad, w, corr

def smooth_obj(h, beta):
    corr = fft_corr(h)
    cm = corr.max()
    return float(np.log(np.exp(beta * (corr - cm)).sum()) / beta + cm)
