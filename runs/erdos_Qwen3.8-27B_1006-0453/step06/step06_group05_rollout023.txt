            best_c5, best_h = c5l_dx, hl.copy()
        hr = round_to_binary(hl, N0)
        hb, _ = binary_optimize(hr, N0, min(deadline - 240, time.time() + 60), max_sweeps=500)
        c5b = compute_c5(hb, N0)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()

    # Phase 3: Second LSE with different params from best binary
    if time.time() < deadline - 250:
        d3 = min(deadline - 180, time.time() + 120)
        Nf = len(best_h)
        hc3 = round_to_binary(best_h, Nf).astype(float) + np.random.randn(Nf) * 0.02
        hc3 = project_sum(hc3, Nf / 2.0)
        hl3, c5l3 = lse_optimize(hc3, Nf, d3, alpha0=100.0, alpha1=2000.0, lr0=0.025)
        c5l3_dx = c5l3 * (2.0 / Nf)
        if c5l3_dx < best_c5:
            best_c5, best_h = c5l3_dx, hl3.copy()
        hr3 = round_to_binary(hl3, Nf)
        hb3, _ = binary_optimize(hr3, Nf, min(deadline - 170, time.time() + 50), max_sweeps=400)
        c5b3 = compute_c5(hb3, Nf)
        if c5b3 < best_c5:
            best_c5, best_h = c5b3, hb3.copy()

    # Phase 4: Binary polish
    if time.time() < deadline - 120:
        Nf = len(best_h)
        d4 = min(deadline - 80, time.time() + 80)
        hb4 = round_to_binary(best_h, Nf)
        hs4, _ = binary_optimize(hb4, Nf, d4, max_sweeps=800)
        c54 = compute_c5(hs4, Nf)
        if c54 < best_c5:
            best_c5, best_h = c54, hs4.copy()

    # Phase 5: Annealed binary
    if time.time() < deadline - 80:
        Nf = len(best_h)
        d5 = min(deadline - 50, time.time() + 50)
        hb5 = round_to_binary(best_h, Nf)
        hs5, _ = binary_optimize(hb5, Nf, d5, max_sweeps=400, anneal=0.12)
        c55 = compute_c5(hs5, Nf)
        if c55 < best_c5:
            best_c5, best_h = c55, hs5.copy()

    # Phase 6: Multi-resolution
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [200, 260, 300, 340, 420, 480, 540, 600]:
        if time.time() > deadline - 30:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(x_new, x_best, best_h)
        ht = round_to_binary(hi, N_try)
        sub_dl = min(deadline - 15, time.time() + 30)
        hrt, _ = binary_optimize(ht, N_try, sub_dl, max_sweeps=300)
        c5rt = compute_c5(hrt, N_try)
        if c5rt < best_c5:
            best_c5, best_h = c5rt, hrt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 7: Final LSE polish
    if time.time() < deadline - 15:
        Nf = len(best_h)
        hc7 = best_h.astype(float) + np.random.randn(Nf) * 0.015
        hc7 = project_sum(hc7, Nf / 2.0)
        hl7, c57 = lse_optimize(hc7, Nf, deadline - 3, alpha0=200.0, alpha1=3000.0, lr0=0.015)
        c57_dx = c57 * (2.0 / Nf)
        if c57_dx < best_c5:
            best_c5, best_h = c57_dx, hl7.copy()
        else:
            hr7 = round_to_binary(hl7, Nf)
            c5r7 = compute_c5(hr7, Nf)
            if c5r7 < best_c5:
                best_c5, best_h = c5r7, hr7.copy()

    # Final binary cleanup
    if time.time() < deadline - 1:
        Nf = len(best_h)
        hf = round_to_binary(best_h, Nf)
        hf, _ = binary_optimize(hf, Nf, deadline - 0.5, max_sweeps=40)
        c5f = compute_c5(hf, Nf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf

    final_c5 = compute_c5(best_h, len(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, final_c5, len(best_h)
