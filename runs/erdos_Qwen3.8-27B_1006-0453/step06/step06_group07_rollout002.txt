    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    tau = tau0
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        c_scaled = corr / tau
        c_max = c_scaled.max()
        w = np.exp(c_scaled - c_max)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            wgt = w[ai]
            if L == 0:
                grad += wgt * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wgt * (1.0 - y[:N - L])
                grad[:N - L] -= wgt * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wgt * (1.0 - y[M:])
                    grad[M:] -= wgt * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.985
            tau = max(8.0, tau * 0.97)
            beta = min(0.95, 0.85 + it * 0.0002)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    if N_old == N_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Thorough binary search at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 450, t0 + 90)
    h1, c5_1 = binary_search(h_bin, N0, d1, max_sweeps=1200, sample_frac=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Softmax Nesterov at N0, then rasterize + binary search
    if time.time() < deadline - 350:
        hc = best_h + np.random.randn(N0) * 0.06
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 300, time.time() + 90)
        h2, c5_2 = softmax_nesterov(hc, N0, d2, k=100, lr0=0.022, tau0=35.0)
        h2b = ensure_sum(h2, N0)
        d2b = min(deadline - 250, time.time() + 50)
        h2b2, c5_2b = binary_search(h2b, N0, d2b, max_sweeps=400, sample_frac=1.0)
        if c5_2b < best_c5:
            best_c5, best_h = c5_2b, h2b2.copy()

    # Phase 3: Another continuous pass with different init
    if time.time() < deadline - 250:
        hc2 = best_h * 0.7 + np.random.rand(N0) * 0.3
        hc2 = project_sum(hc2, N0 / 2.0)
        d3 = min(deadline - 200, time.time() + 70)
        h3, c5_3 = softmax_nesterov(hc2, N0, d3, k=120, lr0=0.015, tau0=50.0)
        h3b = ensure_sum(h3, N0)
        d3b = min(deadline - 150, time.time() + 35)
        h3b2, c5_3b = binary_search(h3b, N0, d3b, max_sweeps=250, sample_frac=1.0)
        if c5_3b < best_c5:
            best_c5, best_h = c5_3b, h3b2.copy()

    # Phase 4: Upscale to 2*N0, binary search with restarts
    N1 = 2 * N0
    for restart in range(3):
        if time.time() > deadline - 180:
            break
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        if restart > 0:
            n_flip = restart
            ones = np.where(h_up == 1)[0]
            zeros = np.where(h_up == 0)[0]
            for _ in range(n_flip):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h_up[p], h_up[q] = 0.0, 1.0
        d4 = min(deadline - 130 + restart * 15, time.time() + 80)
        sf = 0.75 if restart == 0 else 0.55
        h4, c5_4 = binary_search(h_up, N1, d4, max_sweeps=250, sample_frac=sf)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Continuous at N1
    if time.time() < deadline - 120:
        h_res = upsample(best_h, len(best_h), N1) if len(best_h) != N1 else best_h.copy()
        hc4 = h_res + np.random.randn(N1) * 0.025
        hc4 = project_sum(hc4, N1 / 2.0)
        d5 = min(deadline - 80, time.time() + 40)
        h5, c5_5 = softmax_nesterov(hc4, N1, d5, k=150, lr0=0.01, tau0=60.0)
        h5b = ensure_sum(h5, N1)
        d5b = min(deadline - 40, time.time() + 25)
        h5b2, c5_5b = binary_search(h5b, N1, d5b, max_sweeps=150, sample_frac=0.5)
        if c5_5b < best_c5:
            best_c5, best_h = c5_5b, h5b2.copy()

    # Phase 6: Final binary polish at N0 resolution (most likely best quantization)
    N_fin = N0
    if time.time() < deadline - 10:
        hF = ensure_sum(upsample(best_h, len(best_h), N_fin) if len(best_h) != N_fin else best_h, N_fin)
        d6 = deadline - 2
        h6, c5_6 = binary_search(hF, N_fin, d6, max_sweeps=600, sample_frac=1.0)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Try N0/2 resolution if even
    if N0 % 2 == 0 and time.time() < deadline - 5:
        N_half = N0 // 2
        h_half = upsample(best_h, len(best_h), N_half)
        h_half_b = ensure_sum(h_half, N_half)
        h7, c5_7 = binary_search(h_half_b, N_half, deadline - 1, max_sweeps=300, sample_frac=1.0)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
