import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def ensure_sum_binary(h, N, seed=42):
    h_bin = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1-target, replace=False)
        h_bin[idx] = 0.0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target-n1, replace=False)
        h_bin[idx] = 1.0
    return h_bin

def compute_D(h, N):
    center = N-1
    shifts = np.arange(2*N-1) - center
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0-hm)*vm + hp*vp

    c = fft_corr(h)

    stall = 0

            break

            beta = min(0.98, 0.85 + it*0.00005)
