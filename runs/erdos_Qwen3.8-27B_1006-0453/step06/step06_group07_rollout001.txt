    e = np.exp(c)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h, N, k_active):
    """Compute gradient of soft-max of correlation values."""
    corr = fft_corr(h)
    # Find top k_active lags
    if k_active >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -k_active)[-k_active:]
    
    # Softmax weights over active lags
    c_sub = corr[top_idx]
    c_sub = c_sub - c_sub.max()
    w = np.exp(c_sub)
    w /= w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for idx in range(len(top_idx)):
        ai = top_idx[idx]
        L = ai - center
        wi = w[idx]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            # corr = sum_{i=0}^{N-1-L} h[i+L]*(1-h[i])
            # d/dh[j] = (1-h[j-L])*[j>=L] - h[j+L]*[j<=N-1-L]
            if L < N:
                grad[L:] += wi * (1.0 - h[:N-L])
                grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            # corr = sum_{i=0}^{N-1-M} h[i]*(1-h[i+M])
            # d/dh[j] = (1-h[j+M])*[j<=N-1-M] - h[j-M]*[j>=M]
            if M < N:
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
    
    return grad * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h, N, k_active):
    corr = fft_corr(h)
    if k_active >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -k_active)[-k_active:]
    c_sub = corr[top_idx].copy()
    c_sub -= c_sub.max()
    w = np.exp(c_sub)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for idx in range(len(top_idx)):
        ai = int(top_idx[idx])
        L = ai - center
        wi = w[idx]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad * (2.0 / N)

def nesterov_opt(h, N, deadline, k_init=100, k_max=None, lr0=0.02, beta0=0.9):
    if k_max is None:
        k_max = 2 * N
    h = h.astype(np.float64).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        k_active = min(k_max, k_init + it * 3)
        grad = compute_gradient(y, N, k_active)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 5 == 0:
            pk = float(np.max(fft_corr(h)))
            if pk < best_pk:
                best_pk = pk
                best_h = h.copy()
        if it % 250 == 0:
            lr *= 0.96
            beta = min(0.975, 0.88 + it * 0.00004)
    pk = float(np.max(fft_corr(h)))
    if pk < best_pk:
        best_pk = pk
        best_h = h.copy()
    return best_h, best_pk * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, max_sweeps=200, sample_frac=1.0):

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()

        zeros = np.where(h == 0)[0]

        n_s = max(30, int(len(ones) * sample_frac)) if sample_frac < 1.0 else len(ones)
        p_list = np.random.choice(ones, n_s, replace=False) if n_s < len(ones) else ones
        best_new_P = float(c.max())

        for p in p_list:

            cand = base[None, :] - Dz

            mi = int(np.argmin(npk))

                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:

            if best_new_P < best_c - 1e-14:
                best_c = best_new_P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
