import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def c5_value(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(fft_corr(h))) * dx

def project_binary(h, target):
    """Project continuous h to binary with exactly target ones."""
    N = len(h)
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    oi = np.where(h_bin == 1)[0]
    zi = np.where(h_bin == 0)[0]
    if n1 > target:
        # Remove (n1-target) ones: keep highest h values
        sort_idx = oi[np.argsort(h[oi])[:-(n1-target)]]  # lowest h values among ones
        h_bin[sort_idx] = 0
    else:
        # Add (target-n1) zeros: add highest h values among zeros
        sort_idx = zi[np.argsort(-h[zi])[:target-n1]]  # highest h values among zeros
        h_bin[sort_idx] = 1
    return h_bin

def enumerate_periods(n_points, max_period=15):
    """Try all periodic patterns with period <= max_period."""
    best_h = None
    best_c = 1.0
    x = np.linspace(0, 2, n_points, endpoint=False)
    for p in range(2, max_period+1):
        for n_ones in range(1, p):
            # Generate all combinations
            from itertools import combinations
            for onefs in combinations(range(p), n_ones):
                pattern = np.zeros(p)
                pattern[list(onefs)] = 1
                # Tile to fill n_points
                repeats = n_points // p
                h = np.tile(pattern, repeats)
                h = np.concatenate([h, pattern[:n_points % p]])
                # Check sum
                s = int(h.sum())
                dx = 2.0 / n_points
                if abs(s * dx - 1.0) > 0.5 * dx:
                    continue
                c = float(np.max(fft_corr(h))) * dx
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    return best_h, best_c

import numpy as np
import time
from itertools import combinations

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def c5_val(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def to_binary(h, target):
    N = len(h)
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == target:
        return hb
    oi = np.where(hb == 1)[0]
    zi = np.where(hb == 0)[0]
    if n1 > target:
        rm = oi[np.argsort(h[oi])[:n1 - target]]
        hb[rm] = 0.0
    else:
        ad = zi[np.argsort(-h[zi])[:target - n1]]
        hb[ad] = 1.0
    return hb

def swap_search(h, deadline, max_sweeps=500, anneal=0.0, use_top=200):
    N = len(h)
    P_cur = float(np.max(fft_corr(h)))
    best_c = P_cur
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 60 and anneal == 0:
            break
        if stall > 30 and anneal > 0:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        # Compute D matrix efficiently
        shifts = np.arange(2 * N - 1) - (N - 1)
        pos = np.arange(N)[:, None]
        sv = shifts[None, :]
        pm = pos - sv
        pp = pos + sv
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pmc = np.clip(pm, 0, N - 1)
        ppc = np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pmc], 0.0)
        hp = np.where(vp, h[ppc], 0.0)
        D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
        # Focus on near-active shifts for speed
        if use_top and (2 * N - 1) > use_top:
            thresh = np.partition(c, -(use_top // 2))[-use_top // 2]
            active = np.where(c >= thresh - 0.5)[0]
        else:
            active = np.arange(len(c))
        Da = D[:, active]
        ca = c[active]
        bd, bpq = 0.0, None
        Dones = Da[ones, :]
        Dzeros = Da[zeros, :]
        # Vectorized: for each q in zeros, find best p
        # new_max[p] = max_k (ca[k] - Dones[p,k] + Dzeros[q,k])
        # = max_k ((ca + Dzeros[q]) - Dones[p,k])
        for qi in range(len(zeros)):
            A = ca + Dzeros[qi, :]  # (n_active,)
            nm = np.max(A[None, :] - Dones, axis=1)  # (n_ones,)
            mi = int(np.argmin(nm))
            d = float(nm[mi]) - P
            if d < bd:
                bd = d
                bpq = (int(ones[mi]), int(zeros[qi]))
        if bpq is not None and bd < -1e-10:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

def smooth_opt(h, deadline, k=8, lr0=0.03, max_iter=5000):
    N = len(h)
    target = N / 2.0
    dx = 2.0 / N
    h = h.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = lr0
    while time.time() < deadline and t < max_iter:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(c, -k)[-k:]
        shifts = np.arange(2 * N - 1) - (N - 1)
        grad = np.zeros(N)
        for kk in top:
            sh = shifts[kk]
            d1 = np.zeros(N)
            d2 = np.zeros(N)
            idx = np.arange(N)
            im = idx - sh
            ip = idx + sh
            vm = (im >= 0) & (im < N)
            vp = (ip >= 0) & (ip < N)
            d1[idx[vm]] = (1.0 - h[im[vm]])
            d2[idx[vp]] = h[ip[vp]]
            grad += d1 - d2
        grad = grad / k * dx
        grad -= grad.mean()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        h = h + lr * (m / (1 - b1 ** t)) / (np.sqrt(v / (1 - b2 ** t)) + eps)
        # Project to [0,1] and sum constraint
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) > 1e-6:
            for _ in range(30):
                h = np.clip(h - (s - target) / N, 0.0, 1.0)
                s = h.sum()
                if abs(s - target) < 1e-6:
                    break
        if t % 200 == 0:
            lr *= 0.92
            hb = to_binary(h, N // 2)
            h = 0.6 * h + 0.4 * hb
    return best_h, best_pk * dx

def enumerate_small(n_pts):
    best_h, best_c = None, 1.0
    for p in range(4, 11):
        if n_pts % p != 0:
            continue
        n_per = n_pts // p
        need = n_per * p // 2
        ones_pp = need // n_per
        if ones_pp * n_per != need:
            continue
        if p // 2 != ones_pp:
            continue
        for onefs in combinations(range(p), ones_pp):
            pat = np.zeros(p)
            pat[list(onefs)] = 1.0
            h = np.tile(pat, n_per)
            c = c5_val(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c
