    h = h.astype(float).copy()
    L = 2 * N - 1
    center = N - 1
    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    best_c = c.max()
    best_h = h.copy()
    
    sweep = 0
    no_improve = 0
    while sweep < max_sweeps:
        if deadline and time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        cur_max = c.max()
        
        best_delta = 0  # We want negative delta (decrease in max)
        best_pq = None
        
        for p in ones:
            hp = 1.0  # h[p] is 1
            # Precompute the effect of removing p
            # Δc from p: -(1-h[p+k]) for valid p+k, +h[p-k] for valid p-k
            for q in zeros:
                # Compute new max
                dc = np.zeros(L)
                # p in a: -(1-h[p+k])
                lo = max(0, center - p)
                hi = min(L, center + N - p)
                if lo < hi:
                    dc[lo:hi] -= (1.0 - h[p + (np.arange(lo, hi) - center)])
                # p in b: +h[p-k]  
                lo = max(0, center + p - N + 1)
                hi = min(L, center + p + 1)
                if lo < hi:
                    dc[lo:hi] += h[p - (np.arange(lo, hi) - center)]
                # q in a: +(1-h[q+k])
                lo = max(0, center - q)
                hi = min(L, center + N - q)
                if lo < hi:
                    dc[lo:hi] += (1.0 - h[q + (np.arange(lo, hi) - center)])
                # q in b: -h[q-k]
                lo = max(0, center + q - N + 1)
                hi = min(L, center + q + 1)
                if lo < hi:
                    dc[lo:hi] -= h[q - (np.arange(lo, hi) - center)]
                
                nm = (c + dc).max()
                delta = nm - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_pq = (p, q)
                elif anneal_temp > 0 and delta < 0.01:
                    # Occasionally accept non-improving moves
                    if np.random.random() < np.exp(-delta / anneal_temp):
                        best_pq = (p, q)
                        best_delta = 0  # Mark as accepted
        
        if best_pq is None:
            no_improve += 1
            if no_improve > 50:
                break
            if anneal_temp > 0.01:
                anneal_temp *= 0.95
                # Random perturbation
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                best_pq = (p, q)
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = np.correlate(h, 1.0 - h, mode="full")
            if c.max() < best_c:
                best_c = c.max()
                best_h = h.copy()
            no_improve = 0
        sweep += 1
    
    return best_h, best_c
