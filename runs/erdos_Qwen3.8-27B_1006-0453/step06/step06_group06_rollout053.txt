        h4 = best_h.astype(float) + np.random.randn(len(best_h)) * 0.012
        h4 = proj_sum(h4, len(best_h) / 2.0)
        hP, pkP = soft_max_opt(h4, len(best_h), d4, alpha0=90.0, lr0=0.002, max_active=150)
        c5P = pkP * (2.0 / len(best_h))
        if c5P < best_c5:
            best_c5 = c5P; best_h = hP.copy()

    # Phase 5: Multi-resolution exploration
    for Nt in [400, 480, 560]:
        if time.time() > deadline - 40:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_bin((hi > 0.5).astype(float), Nt)
        sub = min(deadline - 10, time.time() + 35)
        hrt, pkr = anneal_binary(hbt, Nt, sub, T0=0.04, cool=0.9996, max_no_imp=1000, n_restarts=1)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5 = c5r; best_h = hrt.copy()

    # Phase 6: Final focused polish
    if time.time() < deadline - 10:
        d6 = deadline - 3
        h6 = best_h.astype(float) + np.random.randn(len(best_h)) * 0.005
        h6 = proj_sum(h6, len(best_h) / 2.0)
        hFinal, pkFinal = focused_opt(h6, len(best_h), d6, n_top=4)
        c5Final = pkFinal * (2.0 / len(best_h))
        if c5Final < best_c5:
            best_c5 = c5Final; best_h = hFinal.copy()

    final_c5 = float(fft_corr(best_h).max()) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
