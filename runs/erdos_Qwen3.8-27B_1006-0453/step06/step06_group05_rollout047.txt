import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_max_grad(c, tau):
    """Return softmax(c/tau) which is the gradient of soft_max w.r.t. c"""
    m = c.max()
    e = np.exp((c - m) / tau)
    return e / e.sum()

def compute_grad(h, N, tau):
    """Compute gradient of soft_max(correlate(h,1-h))/tau w.r.t. h"""
    c = fft_corr(h)
    w = soft_max_grad(c, tau)  # length 2N-1
    oh = 1.0 - h
    # first_sum[a] = sum_m w[m] * oh[a+m] for valid a+m
    # = correlate(oh, w, 'full')[2N-1 : 3N-1]
    L_nw = 2 * N - 1
    corr_fw = np.correlate(oh, w, 'full')  # length N + L_nw - 1 = 3N-2
    first_sum = corr_fw[L_nw : L_nw + N]  # length N
    
    # second_sum[a] = sum_k w[k] * h[a-k] for valid a-k
    # = convolve(w, h, 'full')[:N]
    conv_w = np.convolve(w, h, 'full')  # length L_nw + N - 1 = 3N-2
    second_sum = conv_w[:N]  # length N
    
    return first_sum - second_sum

def continuous_opt(h_init, N, deadline, tau_start=0.5, tau_end=0.05, 
                   lr=0.01, n_steps_max=5000):
    """Run soft-max gradient descent on continuous h"""
    h = h_init.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.time() < deadline and t < n_steps_max:
        t += 1
        # Adaptive temperature
        progress = min(t / n_steps_max, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        c = fft_corr(h)
        c5 = float(c.max()) * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_grad(h, N, tau)
        # Remove mean (preserves sum constraint to first order)
        grad -= grad.mean()
        
        # Adam update
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * grad ** 2
        mh = m1 / (1 - b1 ** t)
        vh = m2 / (1 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    
    return best_h, best_c5

def compute_D(h, N):
    """Compute the swap derivative matrix D[p, k] for binary h"""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def bin_search(h, N, deadline, max_sweeps=500, anneal=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[ones_i[:nh]] = 1
        else:
            h[ones_i] = 1
            h[zeros_i[:nh - n1]] = 1
    
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 60 and anneal == 0:
            break
        if stall > 30 and anneal > 0:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        D = compute_D(h, N)
        bd, bpq = 0.0, None
        # Vectorized: for each p in ones, find best q in zeros
        p_list = ones
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    
    return h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, N, tau):
    c = fft_corr(h)
    m = c.max()
    e = np.exp((c - m) / tau)
    w = e / e.sum()
    oh = 1.0 - h
    Lw = 2 * N - 1
    corr_fw = np.correlate(oh, w, 'full')
    first_sum = corr_fw[Lw:Lw + N]
    conv_w = np.convolve(w, h, 'full')
    second_sum = conv_w[:N]
    grad = first_sum - second_sum
    grad -= grad.mean()
    return grad

def continuous_opt(h_init, N, deadline, tau_start=0.3, tau_end=0.03, lr=0.008):
    h = h_init.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    stuck = 0
    while time.time() < deadline and t < 8000:
        t += 1
        progress = min(t / 4000.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress
        c = fft_corr(h)
        c5 = float(c.max()) * (2.0 / N)
        if c5 < best_c5 - 1e-7:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 2000:
            break
        grad = compute_grad(h, N, tau)
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * grad ** 2
        mh = m1 / (1 - b1 ** t)
        vh = m2 / (1 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def bin_search(h, N, deadline, max_sweeps=500, anneal=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        oi = np.where(h > 0.5)[0]
        zi = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[oi[:nh]] = 1
        else:
            h[oi] = 1
            h[zi[:nh - n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 60 and anneal == 0:
            break
        if stall > 30 and anneal > 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        bd, bpq = 0.0, None
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

def fix_binary(h, N):
    nb = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(nb.sum())
    if n1 != nh:
        all_idx = np.argsort(-h)
        nb[:] = 0
        nb[all_idx[:nh]] = 1
    return nb
