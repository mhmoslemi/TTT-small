        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, min(n1-nh, len(ones)), replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, min(nh-n1, len(zeros)), replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h
