import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h[::-1], L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target):
    h = (h >= 0.5).astype(float).copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def sa_optimize(h, N, deadline, T0=1.0, Tend=0.005, alpha=0.9997, pair_prob=0.2):
    h = project_binary(h, N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    T = T0
    no_improve = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        if np.random.rand() < pair_prob and len(ones) >= 2 and len(zeros) >= 2:
            # Pair swap
            p1, p2 = np.random.choice(ones, 2, replace=False)
            q1, q2 = np.random.choice(zeros, 2, replace=False)
            h[p1] = h[p2] = 0
            h[q1] = h[q2] = 1
        else:
            # Single swap
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p] = 0
            h[q] = 1
        
        new_corr = fft_corr(h)
        new_c = float(new_corr.max())
        
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
            no_improve = 0
        else:
            if np.random.rand() >= np.exp(-(new_c - best_c) / max(T, 1e-10)):
                # Reject
                if np.random.rand() < pair_prob:
                    h[p1] = h[p2] = 1
                    h[q1] = h[q2] = 0
                else:
                    h[p] = 1
                    h[q] = 0
            no_improve += 1
        
        T = max(Tend, T * alpha)
        if no_improve > 5000:
            # Reset from best
            h = best_h.copy()
            T = T0 * 0.5
            no_improve = 0
    
    return best_h, best_c * (2.0 / N)

def greedy_polish(h, N, deadline):
    h = project_binary(h, N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()

    """Gradient of R[lag] w.r.t. h, where R = correlate(h, 1-h, full) indexed as lag = -(N-1)..N-1"""
    grad = np.zeros(N)
    if lag >= 0:
        # R = sum_{i=0}^{N-1-lag} h[i+lag] * (1-h[i])
        # dR/dh[j] from first factor (j = i+lag, i = j-lag): (1-h[j-lag]) for lag <= j <= N-1
        if lag < N:

        # dR/dh[j] from second factor (j = i, i = j): -h[j+lag] for 0 <= j <= N-1-lag

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    V = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * np.conj(V)))

def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / N)

def project_cont(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def project_binary(h, target):
    h = (h >= 0.5).astype(float).copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def grad_lag(h, N, lag):
    grad = np.zeros(N)
    if lag >= 0:
        if lag < N:
            grad[lag:] += (1.0 - h[:N - lag])
            grad[:N - lag] -= h[lag:]
    else:
        l = -lag
        if l < N:
            grad[:N - l] += (1.0 - h[l:])
            grad[l:] -= h[:N - l]
    return grad

def subgrad_opt(h_init, N, deadline, lr0=0.3, top_k=3):
    h = h_init.astype(float).copy()
    h = project_cont(h, N / 2.0)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    lr = lr0
    step = 0
    mom = np.zeros(N)
    beta = 0.5
    while time.time() < deadline:
        corr = fft_corr(h)
        c = float(corr.max())
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        # Soft top-k weighting
        sorted_idx = np.argsort(corr)[::-1]
        top = sorted_idx[:top_k]
        weights = np.zeros(len(corr))
        max_vals = corr[top]
        w_sum = 0.0
        for i, idx in enumerate(top):
            weights[idx] = max_vals[i]
            w_sum += max_vals[i]
        if w_sum > 0:
            weights /= w_sum
        # Combine gradients
        g = np.zeros(N)
        for idx in top:
            w = weights[idx]
            if w < 1e-8:
                continue
            if idx < N:
                lag = idx
            else:
                lag = idx - (2 * N - 1)
            g += w * grad_lag(h, N, lag)
        # Momentum step
        mom = beta * mom - lr * g
        h = project_cont(h + mom, N / 2.0)
        step += 1
        if step % 500 == 0:
            lr *= 0.95
            beta = min(0.7, 0.5 + step * 0.0001)
    return best_h, best_c * (2.0 / N)

def sa_binary(h_init, N, deadline, T0=0.5, Tend=0.001, alpha=0.99985):
    h = project_binary(h_init, N // 2)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_ci = float(np.max(corr))
    best_h = h.copy()
    T = T0
    stuck = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        # Try pair swap with some probability
        if len(ones) >= 2 and len(zeros) >= 2 and np.random.rand() < 0.25:
            p1, p2 = np.random.choice(ones, 2, replace=False)
            q1, q2 = np.random.choice(zeros, 2, replace=False)
            h[p1] = h[p2] = 0
            h[q1] = h[q2] = 1
            new_ci = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
            if new_ci < best_ci:
                best_ci = new_ci
                best_h = h.copy()
                stuck = 0
            elif np.random.rand() > np.exp(-(new_ci - best_ci) / T):
                h[p1] = h[p2] = 1
                h[q1] = h[q2] = 0
            else:
                stuck += 1
        else:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p] = 0
            h[q] = 1
            new_ci = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
            if new_ci < best_ci:
                best_ci = new_ci
                best_h = h.copy()
                stuck = 0
            elif np.random.rand() > np.exp(-(new_ci - best_ci) / T):
                h[p] = 1
                h[q] = 0
            else:
                stuck += 1
        T = max(Tend, T * alpha)
        if stuck > 8000:
            h = best_h.copy()
            T = max(T, T0 * 0.4)
            stuck = 0
    return best_h, best_ci * (2.0 / N)

def greedy_improve(h_init, N, deadline, n_tries=5000):
    h = project_binary(h_init, N // 2)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_ci = float(np.max(corr))
    best_h = h.copy()
    iterations = 0
    while time.time() < deadline and iterations < n_tries:
        iterations += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Sample subset of ones for speed
        n_s = min(30, len(ones))
        samp = np.random.choice(ones, n_s, replace=False) if n_s < len(ones) else ones
        best_delta = 0.0
        bp, bq = -1, -1
        for p in samp:
            for q in zeros:
                if p == q:
                    continue
                h[p] = 0
                h[q] = 1
                nc = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
                if nc < best_ci + best_delta:
                    best_delta = nc - best_ci
                    bp, bq = p, q
                h[p] = 1
                h[q] = 0
        if bp == -1:
            break
        h[bp] = 0
        h[bq] = 1
        best_ci += best_delta
        best_h = h.copy()
    return best_h, best_ci * (2.0 / N)
