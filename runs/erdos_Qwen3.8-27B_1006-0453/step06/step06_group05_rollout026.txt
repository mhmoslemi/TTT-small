import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * 2.0 / N

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_correlation(h, weights):
    """Compute gradient of sum_p weights[p] * corr[p]."""
    N = len(h)
    grad = np.zeros(N)
    for p in range(2*N-1):
        w = weights[p]
        if w < 1e-10:
            continue
        lag = p - (N-1)
        d = abs(lag)
        if d == 0:
            grad += w * (1.0 - 2.0*h)
        else:
            # (1-h[j-d]) for j in [d, N-1]
            grad[d:] += w * (1.0 - h[:-d])
            # -h[j+d] for j in [0, N-d-1]
            grad[:N-d] -= w * h[d:]
    return grad

def softmax_weights(corr, alpha):
    """Compute softmax weights for correlation values."""
    c = corr - corr.max()
    w = np.exp(alpha * c)
    return w / w.sum()

def continuous_optimize(h, deadline, alpha0=50, alpha_final=200, lr0=0.02):
    """Continuous projected gradient on softmax-smoothed max correlation."""
    N = len(h)
    target = N / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        c5 = float(corr.max()) * 2.0 / N
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive alpha (increase over time for sharper max)
        alpha = alpha0 + (alpha_final - alpha0) * min(1.0, t / 500.0)
        
        # Compute gradient
        w = softmax_weights(corr, alpha)
        grad = grad_correlation(h, w)
        grad -= grad.mean()  # Preserve sum (approximately)
        
        # Adam step
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        lr = lr0 / (1 + t * 0.001)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        
        # Periodic binarization for exploration
        if t % 200 == 0:
            hb = (h > 0.5).astype(float)
            n1 = int(hb.sum())
            if n1 != N//2:
                if n1 > N//2:
                    ones = np.where(hb == 1)[0]
                    ones_sorted = ones[np.argsort(h[ones])]
                    hb[ones_sorted[:(n1 - N//2)]] = 0
                else:
                    zeros = np.where(hb == 0)[0]
                    zeros_sorted = zeros[np.argsort(-h[zeros])]
                    hb[zeros_sorted[:(N//2 - n1)]] = 1
            h = 0.7 * h + 0.3 * hb
    
    return best_h, best_c5
