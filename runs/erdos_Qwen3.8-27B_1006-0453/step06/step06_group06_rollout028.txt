```python
    mk = n - 1 + s  # correlation array indices
    d = np.zeros(2*n-1)
    
    mask_pk = (pk >= 0) & (pk < n)
    mask_qk = (qk >= 0) & (qk < n)
    
    d[mask_pk] -= (1.0 - h[pk[mask_pk]])
    d[mask_qk] += (1.0 - h[qk[mask_qk]])
    
    mask_pm = (0 <= p - s) & (p - s < n)  # p-k in range
    mask_qm = (0 <= q - s) & (q - s < n)
    
    d[mask_pm] += h[p - s[mask_pm]]
    d[mask_qm] -= h[q - s[mask_qm]]
    
    return d
```

This is fully vectorized and O(n) with very low constant. Good.

Now, for the annealing, the key operation per step is:
1. Pick random p (from ones) and q (from zeros): O(1)
2. Compute delta: O(n) vectorized
3. new_max = max(corr + delta): O(n)
4. Accept/reject: O(1)

Total: O(n) per step. For n=380, this is very fast. We can do millions of steps.

Let me also think about a **gradient-based continuous optimization**:

For the continuous case, the objective is f(h) = max_k corr(h, 1-h, 'full')[k] * dx

The subgradient at h is the average of the gradients of the active (max) components.

The gradient of corr[k] w.r.t. h[i]:
corr[k] = sum_j h[j] * (1-h[j+k])
∂corr[k]/∂h[i] = (1-h[i+k]) - h[i-k] (with boundary conditions)

So the subgradient of f is:
∇f[h_i] = (1/n_active) * sum_{k ∈ active} [(1-h[i+k]) - h[i-k]]

Then we do a projected gradient step.

```python
import numpy as np
import time

def swap_delta_vec(h, p, q, n):
    s = np.arange(-(n-1), n)
    d = np.zeros(2*n-1)
    pk = p + s
    qk = q + s
    pm = p - s
    qm = q - s
    m1 = (pk >= 0) & (pk < n)
    m2 = (qk >= 0) & (qk < n)
    m3 = (pm >= 0) & (pm < n)
    m4 = (qm >= 0) & (qm < n)
    d[m1] -= (1.0 - h[pk[m1]])
    d[m2] += (1.0 - h[qk[m2]])
    d[m3] += h[pm[m3]]
    d[m4] -= h[qm[m4]]
    return d

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones using soft-thresholding."""
    target = n // 2
    # Use water-filling: find threshold tau such that sum(clip(h - tau, 0, 1)) ~ target
    # Then set top n/2 to 1
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

    corr = np.correlate(h, 1.0 - h, mode='full')
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_imp = 0
    s = np.arange(-(n-1), n)
    
    T = T0
    while time.time() < deadline and T > 0.002:
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        
        # Compute delta
        d = np.zeros(2*n-1)
        pk = p + s; qk = q + s; pm = p - s; qm = q - s

        m3 = (pm>=0)&(pm<n); m4 = (qm>=0)&(qm<n)
        d[m1] -= (1.0-h[pk[m1]])
        d[m2] += (1.0-h[qk[m2]])
        d[m3] += h[pm[m3]]
        d[m4] -= h[qm[m4]]
        
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h[p] = 0; h[q] = 1
            corr = nc; cur_pk = npk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if npk < best_pk:

            else:
                no_imp += 1
        else:
            no_imp += 1
        
        T *= cool
        if no_imp > 2000:
            no_imp = 0
            h = best_h.copy()
            corr = np.correlate(h, 1.0-h, mode='full')
            cur_pk = best_pk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            T = max(T, T0 * 0.3)
    
    return best_h, best_pk

def cont_opt(h, n, deadline, top_k=50, lr0=0.01):
    """Continuous gradient optimization minimizing max correlation."""
    h = h.astype(float).copy()
    h = proj_sum(h, n/2.0)
    corr = np.correlate(h, 1.0-h, mode='full')
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(n); v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    center = n - 1
    lr = lr0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0-h, mode='full')
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        
        # Get top-k active lags
        idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Compute gradient: average gradient over active lags
        grad = np.zeros(n)
        for ai in idx:
            k = int(ai - center)
            if k == 0:
                grad += (1.0 - 2.0*h)
            elif k > 0:
                # ∂corr[k]/∂h[i] = (1-h[i+k]) - h[i-k]
                # For i where i+k < n: (1-h[i+k])
                # For i where i-k >= 0: -h[i-k]
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] -= h[:n-k]
            else:
                k2 = -k
                grad[k2:] += (1.0 - h[:n-k2])
                grad[:n-k2] -= h[k2:]
        
        grad /= top_k
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), n/2.0)
        
        if t % 500 == 0:
            lr *= 0.98
    
    return best_h, best_pk
```
