        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    cur_val = best_val
    T = T0
    stuck = 0
    it = 0
    while time.time() < deadline and it < max_it:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        h[i] = 0
        h[j] = 1
        new_val = float(np.max(fft_corr(h)))
        delta = new_val - cur_val
        if delta <= 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i] = 1
            h[j] = 0
        if stuck > 400:
            T = T0 * 0.4
            stuck = 0
            for _ in range(3):
                ii = int(np.random.choice(ones))
                jj = int(np.random.choice(zeros))
                h[ii], h[jj] = h[jj], h[ii]
            cur_val = float(np.max(fft_corr(h)))
        T *= cool
    return best_h, best_val * (2.0 / N)

def greedy_il(h, N, deadline, max_it=2000, sz=80):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_h = h.copy()
    best_c = P
    for it in range(max_it):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sz)
        nz = min(len(zeros), sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        N = len(h)
        center = N - 1
        best_mv = P
        best_pair = None
        for oi in os_:
            d_o = np.zeros(2 * N - 1)
            for shift in range(-N + 1, N):
                lo = max(0, -shift)
                hi = min(N, N - shift)
                valid = oi >= lo and oi < hi
                if valid:
                    k_idx = oi - shift + center
                    if h[oi] == 1:
                        d_o[k_idx] -= 1.0
                    d_o[k_idx + 1] += 1.0
            d_o = np.cumsum(d_o)
            for zi in zs:
                delta = np.zeros(2 * N - 1)
                for shift in range(-N + 1, N):
                    lo = max(0, -shift)
                    hi = min(N, N - shift)
                    if zi >= lo and zi < hi:
                        k_idx = zi - shift + center
                        delta[k_idx] += 1.0
                new_P = np.max(corr + d_o[None, :] + delta[:, None])
                if new_P < best_mv - 1e-14:
                    best_mv = new_P
                    best_pair = (oi, zi)
        if best_pair is None:
            break
        h[best_pair[0]] = 0
        h[best_pair[1]] = 1
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c:
            best_c = P
            best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def fixed_swap_il(h, N, deadline, max_it=3000, sz=60):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_h = h.copy()
    best_c = P
    old_corr = corr.copy()
    for it in range(max_it):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ns = min(len(ones), sz)
        nz = min(len(zeros), sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        N = len(h)
        center = N - 1
        best_mv = P
        best_pair = None
        for oi in os_:
            d_o = np.zeros(2 * N - 1)
            for shift in range(-N + 1, N):
                if 0 <= oi - shift < N:
                    d_o[oi - shift + center] -= 1.0
            for zi in zs:
                d_z = np.zeros(2 * N - 1)
                for shift in range(-N + 1, N):
                    if 0 <= zi - shift < N:
                        d_z[zi - shift + center] += 1.0
                new_P = float(np.max(old_corr + d_o + d_z))
                if new_P < best_mv - 1e-14:
                    best_mv = new_P
                    best_pair = (oi, zi)
        if best_pair is None:
            break
        h[best_pair[0]] = 0
        h[best_pair[1]] = 1
        old_corr = fft_corr(h)
        P = float(np.max(old_corr))
        if P < best_c:
            best_c = P
            best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Greedy IL on initial binary
    h_bin = (h0 > 0.5).astype(float)
    d1 = min(deadline - 500, t0 + 150)
    h1, c1 = fixed_swap_il(h_bin, N0, d1, max_it=3000, sz=70)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Continuous Nesterov
    if time.time() < deadline - 350:
        d2 = min(deadline - 330, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.06
        hc = project_01(hc, N0 / 2.0)
        h2, c2 = cont_opt(hc, N0, d2, k_top=55, lr0=0.016)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: SA to escape
    if time.time() < deadline - 250:
        d3 = min(deadline - 230, time.time() + 140)
        h3b = (best_h > 0.5).astype(float)
        h3, c3 = binary_sa(h3b, N0, d3, T0=0.5, cool=0.9995)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Polish with IL
    if time.time() < deadline - 180:
        d4 = min(deadline - 160, time.time() + 100)
        h4 = (best_h > 0.5).astype(float)
        h4, c4 = fixed_swap_il(h4, N0, d4, max_it=2500, sz=80)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Multi-resolution
    for N_try in [260, 320, 420, 500, 580]:
        if time.time() > deadline - 60:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(np.round(h_bt.sum()))
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1 - nh, replace=False)] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh - n1, replace=False)] = 1
        sub_dl = min(deadline - 20, time.time() + 50)
        h_rt, c5_rt = binary_sa(h_bt, N_try, sub_dl, T0=0.4, cool=0.9996)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final continuous + binary polish
    if time.time() < deadline - 15:
        hF = best_h + np.random.randn(len(best_h)) * 0.02
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = cont_opt(hF, len(best_h), deadline - 5, k_top=70, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    if time.time() < deadline - 1:
        hB = (best_h > 0.5).astype(float)
        hB, c5B = binary_sa(hB, len(hB), deadline - 0.5, T0=0.15, cool=0.999)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    final_c5 = compute_c5(best_h, len(best_h))
    hb = (best_h > 0.5).astype(float)
    c5b = compute_c5(hb, len(hb))
    if c5b < final_c5:
        best_h, final_c5 = hb, c5b
    return best_h, final_c5, len(best_h)
