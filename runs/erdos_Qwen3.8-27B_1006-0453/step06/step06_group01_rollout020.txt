import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, target):
    h = np.array(h, dtype=float)
    h = (h >= 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1.0
    return h

def init_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return deltas, s

def apply_swap(h, corr, s, deltas, p, q, N):
    h[p], h[q] = 0.0, 1.0
    corr += deltas[p] - deltas[q]
    s[p] = -s[p]
    s[q] = -s[q]
    deltas[p, :] = 0.0
    deltas[p, p:p + N] = s
    deltas[q, :] = 0.0
    deltas[q, q:q + N] = s

def find_best_swap(corr, deltas, ones, zeros):
    best_val = np.inf
    best_p, best_q = -1, -1
    z = zeros
    dz = deltas[z]
    for p in ones:
        base = corr + deltas[p]
        vals = (base[None, :] - dz).max(axis=1)
        j = int(np.argmin(vals))
        v = float(vals[j])
        if v < best_val:
            best_val = v
            best_p, best_q = int(p), int(z[j])
    return best_p, best_q, best_val

def local_search(h, deadline, max_iter, sa_temp, sa_cool, seed):
    np.random.seed(seed)
    N = len(h)
    target = N // 2
    h = make_binary(h, target)
    corr = fft_corr(h)
    deltas, s = init_deltas(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_c = float(corr.max())
    best_h = h.copy()
    temp = sa_temp
    for it in range(max_iter):
        if time.time() > deadline:
            break
        p, q, v = find_best_swap(corr, deltas, ones, zeros)
        if p < 0:
            break
        dv = v - best_c
        accept = (dv < -1e-12)
        if not accept and temp > 1e-4:
            accept = np.random.random() < np.exp(max(-40.0, -dv / temp))
        if accept:
            apply_swap(h, corr, s, deltas, p, q, N)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
        temp *= sa_cool
    return best_h, best_c * (2.0 / N)

def project_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(v - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h_in, N, deadline, k=40, lr0=0.012):
    h = np.clip(h_in.astype(float), 0.0, 1.0)
    target = N / 2.0
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        top = np.sort(top)
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.98
            beta = min(0.95, 0.80 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
