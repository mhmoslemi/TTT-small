    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    cooling = (T_end / T0) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()

            new_c = c + D[p1] + D[p2] + D[q1] + D[q2]
            new_P = float(new_c.max())
            delta = new_P - P
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def greedy_search(h, N, deadline, max_sweeps=5000, sf=1.0, perturb_every=25, seed=0):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > perturb_every * 4:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sf < 1.0 and len(ones) > 30:
            n_s = max(30, int(len(ones) * sf))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-14:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if stall >= perturb_every:
                nflip = rng.randint(2, 5)
                for _ in range(nflip):
                    if len(ones) < 1 or len(zeros) < 1:
                        break
                    p = ones[rng.randint(len(ones))]
                    q = zeros[rng.randint(len(zeros))]
                    h[p] = 0.0
                    h[q] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def sa_optimize(h, N, deadline, seed=0, T0=0.35, Tend=0.004, frac2swap=0.3):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    T = T0
    total_iters = 5000
    cooling = (Tend / T0) ** (1.0 / total_iters)
    while time.time() < deadline and T > Tend * 1.5:
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        if rng.random() < frac2swap and len(ones) >= 2 and len(zeros) >= 2:
            p1 = ones[rng.randint(len(ones))]
            p2 = ones[rng.randint(len(ones))]
            tries = 0
            while p2 == p1 and tries < 20:
                p2 = ones[rng.randint(len(ones))]
                tries += 1
            if p2 == p1:
                p2 = (p1 + 1 + rng.randint(N - 1)) % N
                while h[p2] != 1:
                    p2 = (p2 + 1) % N
            q1 = zeros[rng.randint(len(zeros))]
            q2 = zeros[rng.randint(len(zeros))]
            tries = 0
            while q2 == q1 and tries < 20:
                q2 = zeros[rng.randint(len(zeros))]
                tries += 1
            if q2 == q1:
                q2 = (q1 + 1 + rng.randint(N - 1)) % N
                while h[q2] != 0:
                    q2 = (q2 + 1) % N
            new_c = c - D[p1] - D[p2] - D[q1] - D[q2]
            new_P = float(new_c.max())
            delta = new_P - P
            if delta < 0 or (T > Tend and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-9))):
                h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                c = new_c
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            new_c = c - D[p] - D[q]
            new_P = float(new_c.max())
            delta = new_P - P
            if delta < 0 or (T > Tend and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-9))):
                h[p] = 0; h[q] = 1
                c = new_c
        T *= cooling
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def nesterov_relax(h, N, deadline, k=60, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        it += 1
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        s = float(h_new.sum())
        if abs(s - target) > 1e-6:
            lo, hi = -1.0, 1.0
            for _ in range(40):
                mid = (lo + hi) * 0.5
                if float(np.clip(h_new - mid, 0.0, 1.0).sum()) > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo + hi) * 0.5, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it * 0.00008)
    return best_h, best_pk * dx
