def fft_corr(h):
    """Compute full linear correlation of h with (1-h) via FFT."""
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    """Compute C5 = max(corr) * dx."""
    return float(np.max(fft_corr(h))) * (2.0 / N)

def swap_delta_full(c, h, p, q, N, top_m=200):
    """
    Compute the new max correlation after swapping h[p]:1->0, h[q]:0->1.
    Uses the correct delta formula.
    """
    center = N - 1
    # Get top lags for evaluation
    # We need to check all lags where the delta might push c above current max
    # But for efficiency, check top (top_m) + buffer
    idx = np.argpartition(c, -top_m)[-top_m:]
    # Also check a buffer of the next ones (in case delta pushes a non-top one up)
    # Actually, delta is bounded by 4 (max change in one term), so we need to check
    # any lag where c[m] + max_delta > current top

    p_minus = p - kvals
    q_plus = q + kvals
    q_minus = q - kvals
    
    # Delta at each lag:
    # -(1-h[p+k])*I(0<=p+k<N) + h[p-k]*I(0<=p-k<N) + (1-h[q+k])*I(0<=q+k<N) - h[q-k]*I(0<=q-k<N)

    new_vals = c[all_idx] + deltas
    return float(new_vals.max())

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def local_search(h, N, deadline, max_sweeps=800, top_m=250, sample_n=180,
                 anneal_prob=0.0, rng=None, verify=True):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.astype(float).copy()
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, nh - n1, replace=False)] = 1

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1
    n_lags = 2 * N - 1

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        n_check = min(top_m, n_lags)
        top_idx = np.argpartition(c, -n_check)[-n_check:]
        kvals = top_idx - center
        c_vals = c[top_idx]

        if len(ones) > sample_n:
            p_list = rng.choice(ones, sample_n, replace=False)
        else:
            p_list = ones.copy()
        if len(zeros) > sample_n:
            q_list = rng.choice(zeros, sample_n, replace=False)
        else:
            q_list = zeros.copy()

        n_p = len(p_list)
        n_q = len(q_list)

        # Precompute p_deltas: (n_p, n_check)
        pp = p_list[:, None] + kvals[None, :]
        pm = p_list[:, None] - kvals[None, :]
        vpp = (pp >= 0) & (pp < N)
        vpm = (pm >= 0) & (pm < N)
        ppc = np.clip(pp, 0, N - 1)
        pmc = np.clip(pm, 0, N - 1)
        p_deltas = np.zeros((n_p, n_check))
        p_deltas[vpp] = -(1.0 - h[ppc[vpp]])
        p_deltas[vpm] += h[pmc[vpm]]

        # Precompute q_deltas: (n_q, n_check)
        qp = q_list[:, None] + kvals[None, :]
        qm = q_list[:, None] - kvals[None, :]
        vqp = (qp >= 0) & (qp < N)
        vqm = (qm >= 0) & (qm < N)
        qpc = np.clip(qp, 0, N - 1)
        qmc = np.clip(qm, 0, N - 1)
        q_deltas = np.zeros((n_q, n_check))
        q_deltas[vqp] = (1.0 - h[qpc[vqp]])
        q_deltas[vqm] += -h[qmc[vqm]]

        best_nm = P
        best_pq = None
        for i in range(n_p):
            base = c_vals + p_deltas[i]
            cand = (base[:, None] + q_deltas).max(axis=1)
            j_b = int(np.argmin(cand))
            if cand[j_b] < best_nm:
                best_nm = float(cand[j_b])
                best_pq = (int(p_list[i]), int(q_list[j_b]))

        if best_pq is not None:
            if verify and n_check < n_lags:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
                c_full = fft_corr(h)
                if float(c_full.max()) >= P:
                    h[best_pq[0]] = 1.0
                    h[best_pq[1]] = 0.0
                else:
                    c = c_full
                    P = float(c.max())
                    if P < best_c:
                        best_c = P
                        best_h = h.copy()
            else:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
        else:
            if anneal_prob > 0 and rng.random() < anneal_prob:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0

    return best_h, best_c
