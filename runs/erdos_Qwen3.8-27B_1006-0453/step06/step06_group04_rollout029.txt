def compute_gradient(h, N, top_k=100, temp=0.5):
    # Compute all correlations
    c = np.correlate(h, 1-h, mode='full')  # length 2N-1
    center = N - 1
    # softmax weights
    max_c = c.max()
    w = np.exp((c - max_c) / temp)
    w /= w.sum()
    
    grad = np.zeros(N)
    # Only use top-k to save time
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    w_top = w[top_idx]
    w_top /= w_top.sum()
    
    for wi, ki in zip(w_top, top_idx):
        k = ki - center  # actual lag

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    return best_h, best_c5

def binary_targeted(h0, N, deadline, n_rand=25, kick_2swap=60):
    h = (h0 > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.random.permutation(np.where(h == 1)[0][:n1])[:n1 - nh]
            h[idx] = 0
        else:
            idx = np.random.permutation(np.where(h == 0)[0][:N - n1])[:nh - n1]
            h[idx] = 1
    c = fft_corr(h)
    best_val = int(np.ceil(c.max()))
    best_h = h.copy()
    center = N - 1
    stall = 0
    while time.time() < deadline:
        cur_max = int(np.ceil(c.max()))
        if cur_max < best_val:
            best_val = cur_max
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        max_lags = np.where(c >= c.max() - 0.5)[0]
        np.random.shuffle(max_lags)
        for li in max_lags[:4]:
            k = li - center
            bad_pos = np.empty(0, dtype=int)
            if k > 0:
                seg = N - k
                bad_pos = np.where((h[:seg] == 1) & (h[k:k + seg] == 0))[0]
            elif k < 0:
                m = -k
                seg = N - m
                bad_pos = (np.where((h[m:] == 1) & (h[:seg] == 0))[0]) + m
            else:
                continue
            if len(bad_pos) == 0:
                continue
            for _ in range(12):
                p = bad_pos[np.random.randint(len(bad_pos))]
                q = zeros[np.random.randint(len(zeros))]
                if p == q:
                    continue
                h[p], h[q] = h[q], h[p]
                c_t = fft_corr(h)
                if int(np.ceil(c_t.max())) < cur_max:
                    c = c_t
                    improved = True
                    break
                h[p], h[q] = h[q], h[p]
            if improved:
                stall = 0
                break
        if not improved:
            for _ in range(n_rand):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = h[q], h[p]
                c_t = fft_corr(h)
                if int(np.ceil(c_t.max())) <= cur_max:
                    c = c_t
                    break
                h[p], h[q] = h[q], h[p]
            else:
                stall += 1
                if stall > kick_2swap:
                    if len(ones) >= 2 and len(zeros) >= 2:
                        p1, p2 = np.random.choice(len(ones), 2, replace=False)
                        q1, q2 = np.random.choice(len(zeros), 2, replace=False)
                        hi_pos = ones[p1] + (ones[p2] << 32)
                        hi_pos = np.array([ones[p1], ones[p2]])
                        lo_pos = np.array([zeros[q1], zeros[q2]])
                        h[hi_pos] = 0
                        h[lo_pos] = 1
                        c = fft_corr(h)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    stall = 0
    return best_h, best_val * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_N = N0
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: LSE continuous at N0 (300s)
    d1 = min(deadline - 500, t0 + 300)
    h1 = best_h + np.random.randn(N0) * 0.04
    h1 = project_sum(h1, N0 / 2.0)
    hn1, c5n1 = lse_optimize(h1, N0, d1, K=250, lr0=0.035, T0=8.0, Tmin=0.3)
    if c5n1 < best_c5:
        best_c5, best_h, best_N = c5n1, hn1.copy(), N0

    # Phase 2: Targeted binary at N0 (250s)
    d2 = max(d1, t0) + 250
    d2 = min(deadline - 250, d2)
    hb2, c5b2 = binary_targeted(best_h, N0, d2, n_rand=20, kick_2swap=50)
    if c5b2 < best_c5:
        best_c5, best_h, best_N = c5b2, hb2.copy(), N0

    # Phase 3: LSE at N=760 (200s)
    N3 = 760
    d3 = max(d2, t0) + 200
    d3 = min(deadline - 100, d3)
    h3 = interp_to_N(best_h, N3)
    h3 = project_sum(h3, N3 / 2.0)
    hn3, c5n3 = lse_optimize(h3, N3, d3, K=300, lr0=0.02, T0=6.0, Tmin=0.4)
    c5n3_final = compute_c5(hn3, N3)
    if c5n3_final < best_c5:
        best_c5, best_h, best_N = c5n3_final, hn3.copy(), N3

    # Phase 4: Binary at N=760 (100s)
    d4 = max(d3, t0) + 100
    d4 = min(deadline - 30, d4)
    if best_N == N3:
        hb4, c5b4 = binary_targeted(best_h, N3, d4, n_rand=15, kick_2swap=40)
        c5b4f = compute_c5(hb4, N3)
        if c5b4f < best_c5:
            best_c5, best_h, best_N = c5b4f, hb4.copy(), N3
    else:
        # Binary at N0
        hb4, c5b4 = binary_targeted(best_h, N0, d4, n_rand=20, kick_2swap=50)
        c5b4f = compute_c5(hb4, N0)
        if c5b4f < best_c5:
            best_c5, best_h, best_N = c5b4f, hb4.copy(), N0

    # Phase 5: Final short LSE polish
    if time.time() < deadline - 5:
        hF = best_h + np.random.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = lse_optimize(hF, len(best_h), deadline - 2, K=300, lr0=0.01, T0=2.0, Tmin=0.2)
        c5Ff = compute_c5(hF, len(hF))
        if c5Ff < best_c5:
            best_c5, best_h, best_N = c5Ff, hF.copy(), len(hF)

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
