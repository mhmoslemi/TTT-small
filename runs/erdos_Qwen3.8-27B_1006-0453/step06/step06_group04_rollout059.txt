def delta_c_for_swap(h, p, q, N):
    """Compute the change in correlation vector when flipping h[p] 1->0 and h[q] 0->1."""
    # c[k] = sum_n h[n] * (1-h[n+k-(N-1)])
    # Delta from h changes:
    #   -v[p+k-(N-1)] for valid index
    #   +v[q+k-(N-1)] for valid index
    # Delta from v changes (v = 1-h, Δv[p]=+1, Δv[q]=-1):
    #   +h[n] where n+k-(N-1)=p, i.e., n=p-k+(N-1)
    #   -h[n] where n+k-(N-1)=q, i.e., n=q-k+(N-1)
    
    v = 1.0 - h
    dc = np.zeros(2*N-1)
    
    # Term 1: -v[p+k-(N-1)] for k where 0 <= p+k-(N-1) <= N-1
    k_start = max(0, N-1-p)
    k_end = min(2*N-2, 2*N-2-p)  # p+k-(N-1) <= N-1 => k <= 2N-2-p
    if k_start <= k_end:
        idx = np.arange(k_start, k_end+1)
        v_idx = np.arange(N-1-p+k_start, N-1-p+k_end+1)
        dc[idx] -= v[v_idx]
    
    # Term 2: +v[q+k-(N-1)]
    k_start = max(0, N-1-q)
    k_end = min(2*N-2, 2*N-2-q)
    if k_start <= k_end:
        idx = np.arange(k_start, k_end+1)
        v_idx = np.arange(N-1-q+k_start, N-1-q+k_end+1)
        dc[idx] += v[v_idx]
    
    # Term 3: +h[p-k+(N-1)] for k where 0 <= p-k+(N-1) <= N-1
    # p-k+(N-1) >= 0 => k <= p+N-1
    # p-k+(N-1) <= N-1 => k >= p
    k_start = max(0, p)
    k_end = min(2*N-2, p+N-1)
    if k_start <= k_end:
        idx = np.arange(k_start, k_end+1)
        h_idx = np.arange(p-k_start+N-1, p-k_end+N-1+1)
        h_idx = h_idx[::-1]  # p-k+(N-1) decreases as k increases
        dc[idx] += h[h_idx]
    
    # Term 4: -h[q-k+(N-1)]
    k_start = max(0, q)
    k_end = min(2*N-2, q+N-1)
    if k_start <= k_end:
        idx = np.arange(k_start, k_end+1)
        h_idx = np.arange(q-k_start+N-1, q-k_end+N-1+1)
        h_idx = h_idx[::-1]
        dc[idx] -= h[h_idx]
