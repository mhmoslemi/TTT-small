def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h = np.clip(h + diff / N, 0.0, 1.0)
    return h

def grad_topk(h, N, k=30):
    corr = fft_corr(h)
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
    return grad / k, corr

def nesterov_restart(h, N, deadline, n_restarts=5, k_vals=(30, 60, 100), lr0=0.02):

    best_val = compute_c5(h, N)
    target = N / 2.0
    elapsed_total = deadline - time.time()
    time_per_restart = elapsed_total / n_restarts
    dx = 2.0 / N
    for r in range(n_restarts):
        if time.time() > deadline - 5:
            break
        sub_dl = min(deadline - 3, time.time() + time_per_restart)

        if r > 0:

        else:
            h[p] = 1
            h[q] = 0

    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.-h), L)))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def c5(h, N):
    return float(np.max(fft_corr(h))) * 2.0 / N

def proj(h, tgt, N):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        h = np.clip(h + d / N, 0.0, 1.0)
    return h

def gtop(h, N, k):
    c = fft_corr(h)
    idx = np.argpartition(c, -k)[-k:]
    g = np.zeros(N)
    ctr = N - 1
    for a in idx:
        L = int(a - ctr)
        if L == 0:
            g += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:N - L]
            g[:N - L] -= h[L:]
        else:
            M = -L
            if M < N:
                g[:N - M] += 1.0 - h[M:]
                g[M:] -= h[:N - M]
    return g / k

def nest(h, N, dl, k=50, lr0=0.015, max_it=12000):
    h = h.copy()
    bh = h.copy()
    bv = c5(h, N)
    tgt = N / 2.0
    m = np.zeros(N)

    b = 0.9
    for it in range(max_it):
        if time.time() > dl:
            break
        y = h + m
        g = gtop(y, N, k)
        hn = proj(y - lr * g, tgt, N)
        m = b * (hn - h)
        h = hn
        if it % 150 == 0:
            v = c5(h, N)
            if v < bv:
                bv, bh[:] = v, h
        if it % 1500 == 0:
            lr *= 0.93
            b = min(0.97, 0.85 + 0.00012 * it)
            k = max(20, k - 3)
    v = c5(h, N)
    if v < bv:
        bv, bh[:] = v, h
    return bh, bv

    zeros = np.where(h == 0)[0]
    for it in range(max_iters):

        worst = np.argmax(corr)
        if it % 100 == 0:

                bv = float(np.max(corr)) * 2.0 / N
                bh[:] = h
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
        L = int(worst - (N - 1))
        M = abs(L)
        if M >= N:
            h[ones[0]] = 0
            h[zeros[0]] = 1
            corr = fft_corr(h)
            continue
        if L > 0:
            bad = np.where((h[:N - M] == 1) & (h[M:] == 0))[0]
            good = np.where((h[:N - M] == 0) & (h[M:] == 1))[0]
        else:
