```python
        # If j is the "second" element in term i, the derivative of h[i]*(1-h[j]) w.r.t. h[j] is -h[i]
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum_01(h, N):
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(99991 + N)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def upsample(h_old, N_new):
    x_old = np.linspace(0, 1, len(h_old), endpoint=False)
    x_new = np.linspace(0, 1, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def lse_gradient(corr, h, D, T):
    max_c = corr.max()
    exp_c = np.exp((corr - max_c) / T)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_c / s
    grad = D.T @ w
    return grad, w

def nesterov_lse(h, N, deadline, T_init=None, lr0=0.02, max_iters=100000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    L = 2 * N - 1
    if T_init is None:
        T = L / 10.0
    else:
        T = T_init
    T_min = L / 500.0
    
    while time.time() < deadline and it < max_iters:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        
        D = compute_D(y, N)
        grad, w = lse_gradient(c, y, D, T)
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.997
            beta = min(0.97, 0.85 + it * 0.0001)
            T = max(T * 0.998, T_min)
    
    return best_h, best_pk * dx

def bit_greedy(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0):
    h = ensure_sum_01(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                rng = np.random.RandomState(int(time.time()*1000) % 100000)
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def bit_anneal(h, N, deadline, T0=2.0, T1=0.01, max_iters=20000):
    h = ensure_sum_01(h, N)
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    rng = np.random.RandomState(4242)

        else:
            D = compute_D(h, N)

        q = int(zeros[qi])
```

```python
def compute_grad_continuous(h, N, w):
    # w: weights over shifts (2N-1,)
    # Returns gradient of sum_k w_k * corr_k w.r.t. h
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)  # (L,)
    # For each position p and shift s:
    # contribution = w[idx] * [(1-h[p+s])*1{0<=p+s<N} - h[p-s]*1{0<=p-s<N}]
    
    # Efficient: use the same approach as compute_D
    pos = np.arange(N)[:, None]  # (N,1)
    sv = shifts[None, :]  # (1,L)
    pm = pos - sv  # (N,L) : p - s
    pp = pos + sv  # (N,L) : p + s
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)  # h[p-s] where valid
    hp = np.where(vp, h[pp_c], 0.0)  # h[p+s] where valid
    
    # For each k: (1-hp)*vp - hm*vm
    # = (1-hp)*vp + (1-hm)*vm - hm*vm - hm*vm ... no
    # = vp - hp*vp - hm*vm
    contrib = (1.0 - hp) * vp - hm * vm  # (N, L)
    grad = contrib @ w  # (N,)
    return grad
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum_01(h, N):
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(99991 + N)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def compute_grad_lse(h, N, w):
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    contrib = (1.0 - hp) * vp - hm * vm
    return contrib @ w

def upsample(h_old, N_new):
    x_old = np.linspace(0, 1, len(h_old), endpoint=False)
    x_new = np.linspace(0, 1, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def nesterov_lse(h, N, deadline, T_init=None, lr0=0.02, max_iters=200000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    L = 2 * N - 1
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    T = T_init if T_init is not None else L / 12.0
    T_min = L / 600.0
    while time.time() < deadline and it < max_iters:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        max_c = c.max()
        exp_c = np.exp((c - max_c) / T)
        s = exp_c.sum()
        w = exp_c / s if s > 1e-300 else None
        if w is None:
            w = np.zeros(L)
            w[np.argmax(c)] = 1.0
        grad = compute_grad_lse(y, N, w)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.996
            beta = min(0.97, 0.88 + it * 0.00005)
            T = max(T * 0.997, T_min)
    return best_h, best_pk * dx

def bit_search(h, N, deadline, max_sweeps=500, sample_frac=1.0):
    h = ensure_sum_01(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if stall > 30:
                break
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def bit_anneal(h, N, deadline, T0=3.0, T1=0.005, max_iters=30000):
    h = ensure_sum_01(h, N)
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    rng = np.random.RandomState(31337)
    it = 0
    while time.time() < deadline and it < max_iters:
        T = T0 * (T1 / T0) ** (it / max(it, max_iters))
        if T < 1e-8:
            T = 1e-8
        D = compute_D(h, N)
        pi = rng.randint(len(ones))
        qi = rng.randint(len(zeros))
        p = int(ones[pi])
        q = int(zeros[qi])
        new_c = c - D[p, :] - D[q, :]
        new_P = float(new_c.max())
        delta = new_P - P
        if delta < 0 or rng.rand() < np.exp(-max(delta, 0.0) / T):
            h[p], h[q] = 0.0, 1.0
            ones[pi] = q
            zeros[qi] = p
            c = new_c
            P = new_P
            if P < best_c - 1e-12:
                best_c = P
                best_h = h.copy()
        it += 1
        if it % 8000 == 0 and P > best_c + 2:
            h = best_h.copy()
            c = fft_corr(h)
            P = float(c.max())
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
    return best_h, best_c * (2.0 / N)
```
