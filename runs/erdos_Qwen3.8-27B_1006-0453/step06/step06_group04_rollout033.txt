```python
def compute_change_vector(h, p, direction):
    """Compute the change in all correlation values when flipping h[p].
    direction: +1 means 1→0, -1 means 0→1
    Returns array of length 2N-1 (the full correlation change).
    """
    N = len(h)
    center = N - 1
    change = np.zeros(2 * N - 1)
    # For 1→0: change at lag L is h[p-L]+h[p+L]-1 (with bounds)
    # For 0→1: negative of that
    
    # For lag +L (index center + L), L = 1 to N-1:
    for L in range(1, N):
        val = -1.0
        if 0 <= p - L < N:
            val += h[p - L]
        if 0 <= p + L < N:
            val += h[p + L]
        change[center + L] = val * direction
        change[center - L] = val * direction  # same by symmetry
    
    return change
```

```python
def compute_change_vector(h, p, direction):
    N = len(h)
    center = N - 1
    # For lag L: change = h[p-L] + h[p+L] - 1
    # This is like: (h shifted right by p) + (h shifted left by p) - 1
    # In terms of the correlation array indexed 0..2N-2:
    # Index center + L corresponds to lag L
    # h[p-L] at index center+L means: element at position (p-L) shifted to position (center+L)
    # That's a shift of center+L-(p-L) = center - p + 2L... hmm this is getting confusing.
    
    # Simpler: just build it directly
    change = np.full(2*N-1, -1.0) * direction
    offsets = np.arange(1, N)
    # h[p-L] for L=1..N-1: positions p-1, p-2, ..., p-(N-1)
    pos_left = p - offsets
    valid_left = (pos_left >= 0) & (pos_left < N)
    change[center + offsets[valid_left]] += h[pos_left[valid_left]] * direction
    change[center - offsets[valid_left]] += h[pos_left[valid_left]] * direction
    
    # h[p+L] for L=1..N-1: positions p+1, p+2, ..., p+(N-1)
    pos_right = p + offsets
    valid_right = (pos_right >= 0) & (pos_right < N)
    change[center + offsets[valid_right]] += h[pos_right[valid_right]] * direction
    change[center - offsets[valid_right]] += h[pos_right[valid_right]] * direction
    
    return change
```

```python
def change_vec(h, p, is_one_to_zero=True):
    """Change in full correlation array when flipping h[p]."""
    N = len(h)
    center = N - 1
    change = np.zeros(2 * N - 1)
    
    L = np.arange(1, N)  # L = 1, 2, ..., N-1
    
    # h[p+L] valid when p+L < N, i.e., L < N-p
    valid_plus = L < (N - p)
    # h[p-L] valid when p-L >= 0, i.e., L <= p
    valid_minus = L <= p
    
    h_plus = np.where(valid_plus, h[p + L[valid_plus]], 0)
    h_minus = np.where(valid_minus, h[p - L[valid_minus]], 0)
    
    # For 1→0:
    # delta_pos[L] = h[p+L] - (1-h[p-L]) [with 0 for invalid]
    # = (h[p+L] if valid_plus else 0) - (1-h[p-L] if valid_minus else 0)
    delta_pos = np.zeros(N-1)
    delta_pos[valid_plus] += h[p + L[valid_plus]]
    delta_pos[valid_minus] -= (1 - h[p - L[valid_minus]])
    
    # delta_neg[L] = h[p-L] - (1-h[p+L]) [with 0 for invalid]
    delta_neg = np.zeros(N-1)
    delta_neg[valid_minus] += h[p - L[valid_minus]]
    delta_neg[valid_plus] -= (1 - h[p + L[valid_plus]])
    
    if not is_one_to_zero:
        delta_pos = -delta_pos
        delta_neg = -delta_neg
    
    change[center + L] = delta_pos
    change[center - L] = delta_neg
    
    return change
```

For the targeted binary optimization:
1. Compute c = correlate(h, 1-h, "full")
2. Find peak lags
3. For peak lag L*, find positions p where h[p]=1 and the flip would reduce c[L*]
   - Δc_L* = h[p+L*] - (1-h[p-L*]) < 0 means h[p+L*] < 1-h[p-L*], i.e., h[p+L*] + h[p-L*] < 1
   - For binary: h[p+L*] = 0 AND h[p-L*] = 0 (or one is out of bounds)
4. Among such positions, pick the one that minimizes the new max(c)
5. Apply the flip, update c incrementally
6. If no such position exists, try 2-swaps or random restarts
