```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(np.flip(1.0 - h), L), L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, N):
    target = N // 2
    h = h.astype(float).copy()
    order = np.argsort(-h)
    h[:] = 0.0
    h[order[:target]] = 1.0
    return h

def local_search(h, N, deadline, max_iter=100000, anneal=0.0, K_init=8, K_max=50):
    h = h.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = fft_corr(h)
    best_val = float(corr.max())
    stall = 0
    K = K_init
    max_stall = 4000 if anneal == 0 else 2500
    for it in range(max_iter):
        if time.time() > deadline:
            break
        P = float(corr.max())
        if P < best_val - 1e-12:
            best_val = P
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
        if stall > 800 and K < K_max:
            K = min(K + 2, K_max)
        n1, n0 = len(ones), len(zeros)
        if n1 < 1 or n0 < 1:
            break
        K_eff = min(K, 2*N-1)
        top_idx = np.argpartition(corr, -K_eff)[-K_eff:] if K_eff < 2*N-1 else np.arange(2*N-1)
        sv = (top_idx - (N - 1)).astype(np.int64)
        Km = len(top_idx)
        ps = ones[:, None] + sv[None, :]
        pm = ones[:, None] - sv[None, :]
        m1 = (ps >= 0) & (ps < N)
        m2 = (pm >= 0) & (pm < N)
        dp = np.zeros((n1, Km))
        if m1.any(): dp[m1] -= (1.0 - h[np.clip(ps, 0, N-1)[m1]])
        if m2.any(): dp[m2] += h[np.clip(pm, 0, N-1)[m2]]
        qs = zeros[:, None] + sv[None, :]
        qm = zeros[:, None] - sv[None, :]
        m3 = (qs >= 0) & (qs < N)
        m4 = (qm >= 0) & (qm < N)
        dq = np.zeros((n0, Km))
        if m3.any(): dq[m3] += (1.0 - h[np.clip(qs, 0, N-1)[m3]])
        if m4.any(): dq[m4] -= h[np.clip(qm, 0, N-1)[m4]]
        base = corr[top_idx]
```

```python
def continuous_opt(h, N, deadline, T=20.0, lr=0.05, steps=200):
    h = h.astype(float).copy()
    target = N / 2.0
    j = np.arange(N)
    ks = np.arange(2*N-1) - (N-1)
    jk = j[:, None] + ks[None, :]
    jm = j[:, None] - ks[None, :]
    v1 = ((jk >= 0) & (jk < N)).astype(float)
    v2 = ((jm >= 0) & (jm < N)).astype(float)
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    for _ in range(steps):
        if time.time() > deadline: break
        corr = fft_corr(h)
        w = np.exp(T * (corr - corr.max()))
        w /= w.sum()
        hjk = np.where((jk >= 0) & (jk < N), h[np.clip(jk, 0, N-1)], 0.0)

        grad -= grad.mean()

        c5 = compute_c5(h, N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.995
    return best_h, best_c5
```

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(np.flip(1.0 - h), L), L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, N):
    target = N // 2
    h = h.astype(float).copy()
    order = np.argsort(-h)
    h[:] = 0.0
    h[order[:target]] = 1.0
    return h
