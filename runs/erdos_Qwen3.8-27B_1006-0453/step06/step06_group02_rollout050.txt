def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)

    lo, hi = -1.0, 1.0

def smooth_max_grad(corr, beta):
    """Gradient of smooth max: softmax-weighted indicator of max."""
    # corr has 2N-1 entries, center is at N-1
    # For the smooth max, gradient wrt corr[k] is softmax(beta*corr)
    # But we need gradient wrt h, which involves the correlation structure
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted - np.max(shifted))
    weights = exps / exps.sum()
    return weights

def smooth_max_h_grad(h, beta, top_k=50):
    """Gradient of smooth max of correlation wrt h."""
    N = len(h)
    corr = fft_corr(h)
    center = N - 1
    weights = smooth_max_grad(corr, beta)
    # Top-k for efficiency
    top = np.argpartition(weights, -top_k)[-top_k:]
    grad = np.zeros(N)
    for ai in top:
        w = weights[ai]
        if w < 1e-8:
            continue

        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        else:
            M = -L
            if M >= N:

            grad[:N - M] += w * (1.0 - h[M:])

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_smooth_grad(h, beta, k=40):
    N = len(h)
    corr = fft_corr(h)
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= ws
    top = np.argpartition(w, -k)[-k:]
    center = N - 1
    grad = np.zeros(N)
    for ai in top:
        wi = w[ai]
        if wi < 1e-10:
            continue
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])

        elif -N < L < 0:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

                stall = 0

                p1 = int(ones[np.random.randint(len(ones))])
                p2 = int(ones[np.random.randint(len(ones))])
                if p1 == p2:
                    continue
                q1 = int(zeros[np.random.randint(len(zeros))])
                q2 = int(zeros[np.random.randint(len(zeros))])
                if q1 == q2:
                    continue
                sc2 = eval_swap(corr, h, N, p1, q1, top_idx[:4]) + eval_swap(corr, h, N, p2, q2, top_idx[:4])
                if sc2 < best_score * 0.7:
                    best_2flip = (p1, p2, q1, q2)
        if best_2flip is not None:
            p1, p2, q1, q2 = best_2flip
            h[p1], h[p2], h[q1], h[q2] = 0.0, 0.0, 1.0, 1.0
        elif best_pq is not None:
            best_pq[0], best_pq[1] = h[best_pq[0]], h[best_pq[1]]
            h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
        else:
            rp, rq = int(ones[0]), int(zeros[0])
            h[rp], h[rq] = 0.0, 1.0
        corr = fft_corr(h)
        new_c = float(corr.max())
        delta = new_c - cur_c
        accept = delta < 0 or (delta > 0 and np.random.random() < np.exp(-delta / max(temp, 1e-10)))
        if accept:
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
            cur_c = new_c
        else:
            if best_2flip is not None:
                p1, p2, q1, q2 = best_2flip
                h[p1], h[p2], h[q1], h[q2] = 1.0, 1.0, 0.0, 0.0
            elif best_pq is not None:
                h[best_pq[0]], h[best_pq[1]] = 1.0, 0.0
            else:
                h[rp], h[rq] = 1.0, 0.0
            stuck += 1
            corr = fft_corr(h)
        temp *= cool
        if stuck > 180:
            temp = max(temp0 * 0.4, temp * 2.5)
            stuck = 0
            if np.random.random() < 0.4:
                for _ in range(4):
                    rp = int(ones[np.random.randint(len(ones))])
                    rq = int(zeros[np.random.randint(len(zeros))])
                    h[rp], h[rq] = h[rq], h[rp]
                corr = fft_corr(h)
                cur_c = float(corr.max())
    return best_h, best_c * (2.0 / N)

def try_multi_thresh(hc, N, deadline):
    best_h, best_c = None, 1e9
    for thr in [0.4, 0.45, 0.5, 0.55, 0.6]:
        hb = fix_binary(hc, N, thr)
        c = compute_c5(hb)
        if c < best_c:
            best_c, best_h = c, hb.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 >= 0.5).astype(float)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive binary SA (most impactful)
    d1 = min(deadline - 350, t0 + 200)
    hs1, c5_1 = binary_sa(best_h, N0, d1, n_trails=60, temp0=0.10, cool=0.997, do_2flip=True)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, hs1.copy()

    # Phase 2: Continuous Nesterov exploration
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_phase(hc, N0, d2, beta_start=30, k=35, lr0=0.03)
        hb_n, c5_b = try_multi_thresh(hn, N0, deadline)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, hb_n.copy()
        elif c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary SA from Nesterov result
    if time.time() < deadline - 150:
        d3 = min(deadline - 130, time.time() + 100)
        hs3, c5_3 = binary_sa(best_h, N0, d3, n_trails=55, temp0=0.08, cool=0.997, do_2flip=True)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, hs3.copy()

    # Phase 4: Second Nesterov with higher beta (sharp peaks)
    if time.time() < deadline - 100:
        d4 = min(deadline - 80, time.time() + 60)
        hc2 = best_h + np.random.randn(N0) * 0.02
        hc2 = project_sum(hc2, N0 / 2.0)
        hn2, c5n2 = nesterov_phase(hc2, N0, d4, beta_start=80, k=30, lr0=0.015)
        hb2, c5_b2 = try_multi_thresh(hn2, N0, deadline)
        if c5_b2 < best_c5:
            best_c5, best_h = c5_b2, hb2.copy()

    # Phase 5: Final binary polish (low temperature)
    if time.time() < deadline - 40:
        d5 = min(deadline - 20, time.time() + 40)
        hs5, c5_5 = binary_sa(best_h, N0, d5, n_trails=50, temp0=0.04, cool=0.9985, do_2flip=False)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, hs5.copy()

    # Phase 6: Try 2x resolution
    N2 = N0 * 2
    if time.time() < deadline - 15 and N2 <= 900:
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N2, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bin = fix_binary(h_int, N2)
        d6 = min(deadline - 5, time.time() + 25)
        hs6, c5_6 = binary_sa(h_bin, N2, d6, n_trails=30, temp0=0.08, cool=0.997, do_2flip=False)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, hs6.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
