        else:
            hi = mid

def continuous_opt(h, N, deadline, top_k=50, lr0=0.003, beta_0=50):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1
    lr = lr0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        # Soft-max gradient over top_k
        idx = np.argpartition(corr, -top_k)[-top_k:]
        weights = np.exp((corr[idx] - corr[idx].max()) * beta_0)
        weights /= weights.sum()
        grad = np.zeros(N)
        for ai, w in zip(idx, weights):
            s = int(ai) - center
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:N-s]
            else:
                s2 = -s
                grad[s2:] += w * (1.0 - h[:N-s2])
                grad[:N-s2] -= w * h[s2:]
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad

        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 300 == 0:

            beta_0 *= 1.1
    return best_h, best_pk

def corr_delta_swap(h, N, p, q, s_arr):
    """Compute delta in corr after swapping h[p]:1->0 and h[q]:0->1."""
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2*N - 1)
    pc = np.clip(ps, 0, N-1); pmc = np.clip(pm, 0, N-1)
    qsc = np.clip(qs, 0, N-1); qmc = np.clip(qm, 0, N-1)
    d[vps] -= (1.0 - h[pc[vps]])
    d[vpm] += h[pmc[vpm]]
    d[vqs] += (1.0 - h[qsc[vqs]])
    d[vqm] -= h[qmc[vqm]]
    return d

    best_h = h.copy()

    lr = lr0
    beta = 30.0
    while time.time() < deadline:
        pk = float(corr.max())
        if pk < best_pk:

        idx = np.argpartition(corr, -top_k)[-top_k:]

        weights /= weights.sum()
        grad = np.zeros(N)
        for ai, w in zip(idx, weights):

            if s == 0:

            elif s > 0:
                grad[:N-s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:N-s]
            else:
                s2 = -s
                grad[s2:] += w * (1.0 - h[:N-s2])
                grad[:N-s2] -= w * h[s2:]
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad

        h = h_new; corr = corr_new

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        s = k - center
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:N-s]
        else:
            s2 = -s
            grad[s2:] += wk * (1.0 - h[:N-s2])
            grad[:N-s2] -= wk * h[s2:]
    return grad

def softmax_opt(h, N, deadline, lr0=0.003, beta0=20.0):
    """Full-softmax gradient optimization with Adam."""
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    lr = lr0
    beta = beta0
    while time.time() < deadline:
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        # Compute softmax weights over all shifts
        cm = corr.max()
        w = np.exp(beta * (corr - cm))
        w /= w.sum()
        # Full gradient
        grad = full_grad(h, w, N)
        # Adam step
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h_new = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        corr_new = fft_corr(h_new)
        pk_new = float(corr_new.max())
        if pk_new > pk + 0.05:
            lr *= 0.85
        elif t % 300 == 0:
            lr *= 1.03
        h = h_new; corr = corr_new
        if t % 400 == 0:
            beta *= 1.08
    return best_h, best_pk

def anneal(h, N, deadline, T0=0.05, cool=0.9995, max_no_improve=600):
    h = fix_binary(h.copy(), N, np.random.RandomState(7))
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = T0
    no_imp = 0
    rng = np.random.RandomState(99)
    while time.time() < deadline and T > 0.0005:
        p = int(ones[rng.randint(len(ones))])
        q = int(zeros[rng.randint(len(zeros))])
        ps = p + s_arr; pm = p - s_arr
        qs = q + s_arr; qm = q - s_arr
        vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
        vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
        d = np.zeros(2 * N - 1)
        pc = np.clip(ps, 0, N-1); pmc = np.clip(pm, 0, N-1)
        qsc = np.clip(qs, 0, N-1); qmc = np.clip(qm, 0, N-1)
        d[vps] -= (1.0 - h[pc[vps]])
        d[vpm] += h[pmc[vpm]]
        d[vqs] += (1.0 - h[qsc[vqs]])
        d[vqm] -= h[qmc[vqm]]
        nc = corr + d
        npk = float(nc.max())
        delta = npk - current_pk
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h[p] = 0; h[q] = 1
            corr = nc; current_pk = npk
            ones = list(np.where(h == 1)[0]); zeros = list(np.where(h == 0)[0])
            no_imp = 0
            if npk < best_pk:
                best_pk = npk; best_h = h.copy()
        else:
            no_imp += 1
        T *= cool
        if no_imp > max_no_improve:
            no_imp = 0
            h = best_h.copy(); corr = fft_corr(h); current_pk = best_pk
            ones = list(np.where(h == 1)[0]); zeros = list(np.where(h == 0)[0])
    return best_h, best_pk

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):

    c5C = pkC * (2.0 / N0)
    if c5C < best_c5:
        best_c5, best_h = c5C, hC.copy()

    d2 = min(deadline - 250, time.time() + 150)
    hb = fix_binary(best_h, N0, rng)

    c5B = pkB * (2.0 / N0)
    if c5B < best_c5:
        best_c5, best_h = c5B, hB.astype(float).copy()
    
    # Phase 3: Continuous polish
    if time.time() < deadline - 150:
        d3 = min(deadline - 100, time.time() + 100)

        if c5C2 < best_c5:
            best_c5, best_h = c5C2, hC2.copy()
    
    # Phase 4: Multi-N exploration
    for Nt in [300, 340, 400, 460, 520, 600, 700, 800]:
        if time.time() > deadline - 35:
            break

            Nt += 1

        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hrt.astype(float).copy()

    if time.time() < deadline - 3:

        hf = best_h.astype(float) + np.random.randn(Nf) * 0.01
        hf = proj_sum(hf, Nf / 2.0)

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
