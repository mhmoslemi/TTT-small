import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h, N):
    return float(np.max(corr_full(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, L, N):
    """Gradient of c_L (lag L) w.r.t. h"""
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        seg = N - L
        g[L:] += 1.0 - h[:seg]
        g[:seg] -= h[L:]
    else:
        M = -L
        if M < N:
            seg = N - M
            g[:seg] += 1.0 - h[M:]
            g[M:] -= h[:seg]
    return g

def continuous_opt(h0, N, deadline, k_sched=(3, 8, 20, 50), lr0=0.03, beta0=0.85):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(corr_full(h)))
    h_prev = h.copy()
    lr = lr0
    beta = beta0
    it = 0
    stall = 0
    while time.time() < deadline:
        # Nesterov step
        y = h + beta * (h - h_prev)
        h_prev = h
        y = project_sum(y, target)
        
        corr = corr_full(y)
        pk = float(np.max(corr))
        if pk < best_c - 1e-12:
            best_c = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive k based on stall
        phase = min(stall // 80, len(k_sched) - 1)
        k = k_sched[phase]
        
        # Top-k lags
        top_idx = np.argpartition(corr, -k)[-k:]
        w = np.exp((corr[top_idx] - corr[top_idx].max()) / 0.05)
        w /= w.sum()
        
        # Gradient
        g = np.zeros(N)
        for wi, ai in zip(w, top_idx):
            L = int(ai - (N - 1))
            g += wi * grad_lag(y, L, N)
        
        # Backtracking line search
        c_new = pk
        lr_try = lr
        for _ in range(12):
            h_candidate = project_sum(y - lr_try * g, target)
            c_candidate = float(np.max(corr_full(h_candidate)))
            if c_candidate < c_new:
                h = h_candidate
                c_new = c_candidate
                lr = min(lr * 1.1, 0.15)
                break
            lr_try *= 0.5
        else:
            h = project_sum(y - lr_try * g, target)
            c_new = float(np.max(corr_full(h)))
            lr = max(lr * 0.8, 0.001)
        
        it += 1
        if it % 300 == 0:
            beta = min(0.97, beta0 + it * 0.0001)
        
        if stall > 500:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = lr0 * 0.8
            stall = 0
    
    return best_h, best_c * (2.0 / N)

def binary_opt(h0, N, deadline, n_evals_per_round=30):
    h = (h0 > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - target]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:target - n1]
            h[idx] = 1
    
    corr = corr_full(h)
    best_int = int(corr.max())
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        round_best_int = best_int
        round_best_h = None
        for _ in range(n_evals_per_round):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_new = corr_full(h)
            ct = int(c_new.max())
            if ct < round_best_int:
                round_best_int = ct
                round_best_h = h.copy()
            h[p] = 1
            h[q] = 0
        
        if round_best_h is not None:

            if round_best_int < best_int:

                best_h = h.copy()
            stall = 0
        else:

                    h[zeros[i2]] = 1
                stall = 0

import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h, N):
    return float(np.max(corr_full(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, L, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        seg = N - L
        g[L:] += 1.0 - h[:seg]
        g[:seg] -= h[L:]
    else:
        M = -L
        if M < N:
            seg = N - M
            g[:seg] += 1.0 - h[M:]
            g[M:] -= h[:seg]
    return g

def continuous_opt(h0, N, deadline, k_sched, lr0, beta0):

    h_prev = h.copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(corr_full(h)))
    lr = lr0
    beta = beta0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        y = project_sum(y, target)
        corr = corr_full(y)
        pk = float(np.max(corr))
        if pk < best_c - 1e-13:
            best_c = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        phase = min(stall // 100, len(k_sched) - 1)
        k = k_sched[phase]
        top_idx = np.argpartition(corr, -k)[-k:]
        w = np.exp((corr[top_idx] - corr[top_idx].max()) / 0.04)
        w /= w.sum()
        g = np.zeros(N)
        center = N - 1

            L = int(ai - center)

        h_new = project_sum(y - lr * g, target)

            h = best_h + np.random.randn(N) * perturb
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = max(lr, lr0 * 0.7)

    return best_h, best_c * (2.0 / N)

def binary_opt(h0, N, deadline, n_round=30, kick_freq=25):

    target = N // 2
    n1 = int(h.sum())

        if n1 > target:

            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:target - n1]
            h[idx] = 1
    corr = corr_full(h)
    best_int = int(corr.max())
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        round_best = best_int
        round_h = None
        for _ in range(n_round):

            h[q] = 1
            ct = int(np.max(corr_full(h)))
            if ct < round_best:
                round_best = ct
                round_h = h.copy()
            h[p] = 1
            h[q] = 0
        if round_h is not None:
            h = round_h
            if round_best < best_int:
                best_int = round_best
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

                    h[zeros[i2]] = 1
                elif len(ones) >= 1 and len(zeros) >= 1:
                    h[ones[0]] = 0
                    h[zeros[0]] = 1
                stall = 0
    return best_h, best_int * (2.0 / N)

        idx = np.where(hb == 0)[0]
