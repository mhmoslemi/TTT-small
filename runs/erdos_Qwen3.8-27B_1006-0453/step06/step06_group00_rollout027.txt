```python
            break
        
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        center = N - 1
        
        # Generate candidates
        cands = []
        for Lp in top_idx:
            L = int(Lp - center)
            if L == 0:
                continue
            if L > 0:
                va = ones[ones + L < N]
                va = va[h[va + L] == 0]
                vb = zeros[zeros + L < N]
                vb = vb[h[vb + L] == 1]
            else:
                va = ones[ones - L >= 0]
                va = va[h[va - L] == 0]
                vb = zeros[zeros - L >= 0]
                vb = vb[h[vb - L] == 1]
            if len(va) > 0 and len(vb) > 0:
                na = min(len(va), 4)
                nb = min(len(vb), 4)
                asel = np.random.choice(va, na, replace=False)
                bsel = np.random.choice(vb, nb, replace=False)
                for a in asel:
                    for b in bsel:
                        cands.append((int(a), int(b)))
        # Random fallback
        for _ in range(8):
            cands.append((int(np.random.choice(ones)), int(np.random.choice(zeros))))
        
        cands = cands[:80]
        if not cands:
            break
        
        # Evaluate: compute delta at top_k lags for each candidate
        top_L = top_idx - center  # array of top_k L values
        arr_a = np.array([c[0] for c in cands])
        arr_b = np.array([c[1] for c in cands])
        Mc = len(cands)
        
        # Vectorized delta computation at top lags
        # For each candidate m and each top lag j:
        # delta(m,j) = -(1-h[am(m)-L(j)]) + h[am(m)+L(j)] + (1-h[bm(m)-L(j)]) - h[bm(m)+L(j)]
        # Shape: (Mc, top_k)
        amL = arr_a[:, None] - top_L[None, :]
        apL = arr_a[:, None] + top_L[None, :]
        bmL = arr_b[:, None] - top_L[None, :]
        bpL = arr_b[:, None] + top_L[None, :]
        
        # Index into h with boundary handling
        hap = np.where((amL >= 0) & (amL < N), h[np.clip(amL, 0, N-1)], 0.0)
        ham = np.where((apL >= 0) & (apL < N), h[np.clip(apL, 0, N-1)], 0.0)
        hbp = np.where((bmL >= 0) & (bmL < N), h[np.clip(bmL, 0, N-1)], 0.0)
        hbm = np.where((bpL >= 0) & (bpL < N), h[np.clip(bpL, 0, N-1)], 0.0)
        
        deltas = -(1.0 - hap) + ham + (1.0 - hbp) - hbm  # (Mc, top_k)
        
        # New max at top lags
        new_top = corr[top_idx][None, :] + deltas  # (Mc, top_k)
        new_max_top = new_top.max(axis=1)  # (Mc,)
        
        # Find best candidate
        best_li = int(np.argmin(new_max_top))
        if new_max_top[best_li] < P - 1e-10:
            h[cands[best_li][0]] = 0
            h[cands[best_li][1]] = 1
        else:
            # Stuck: try restart
            rest_count += 1
            if rest_count > 200:
                break
            if rest_count % 30 == 0 and rest_count <= restarts * 60:
                # Perturb: do 5 random swaps
                for _ in range(5):
                    a = int(np.random.choice(ones))
                    b = int(np.random.choice(zeros))
                    h[a] = 0
                    h[b] = 1
    
    return best_h, best_c
```
