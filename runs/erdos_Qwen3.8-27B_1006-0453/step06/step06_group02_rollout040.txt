```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(-(N-1), N)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    vp = (pp >= 0) & (pp < N)
    pp_c = np.clip(pp, 0, N-1)
    pm = pos - sv
    vm = (pm >= 0) & (pm < N)
    pm_c = np.clip(pm, 0, N-1)
    return (1.0 - h[pp_c]) * vp.astype(float) - h[pm_c] * vm.astype(float)

def fix_sum(h, N):
    nh = N // 2
    h = h.copy()
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-nh]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1.0
    return h

def single_swap_search(h, N, deadline, max_sweeps=800, sample_frac=1.0):
    h = h.astype(float).copy()
    c = fft_corr(h)
    D = compute_D(h, N)
    best_c = float(c.max()) * (2.0/N)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Sample ones if too many
        if len(ones) > 200:
            perm = np.random.permutation(len(ones))[:200]
            ones_s = ones[perm]
        else:
            ones_s = ones
        
        best_score = P + 1.0
        best_pq = None
        for p in ones_s:
            base = c - D[p, :]
            # For each zero q, score = max(base + D[q])
            # Vectorize: compute all scores at once
            cand = base[None, :] + D[zeros, :]  # shape (n_zeros, 2N-1)
            scores = cand.max(axis=1)
            mi = int(np.argmin(scores))
            if scores[mi] < best_score:
                best_score = float(scores[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            # Random move (annealing)
            p = int(ones[np.random.randint(len(ones))])
```

```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(-(N-1), N)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    vp = (pp >= 0) & (pp < N)
    pp_c = np.clip(pp, 0, N-1)
    pm = pos - sv
    vm = (pm >= 0) & (pm < N)
    pm_c = np.clip(pm, 0, N-1)
```
```python
    return (1.0 - h[pp_c]) * vp - h[pm_c] * vm
```

```python
def project_sum(v, target):
```
