def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Simple approach: distribute the difference
    # Binary search for the right offset
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

    """Smoothed approximation of max(corr) using top-k log-sum-exp"""
    top = np.sort(corr)[-k:]

    stall = 0
    it = 0

        h_prev = h

        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Gradient of smooth objective
        # d/dy [log(sum(exp(corr_k/T))) * T] ≈ (1/k_active) * sum of top-k gradients
        top_k = np.argpartition(corr, -k_smooth)[-k_smooth:]
        exp_vals = np.exp((corr[top_k] - corr[top_k].max()) / T)
        weights = exp_vals / exp_vals.sum()
        
        # Compute gradient
        grad = np.zeros(N)
        center = N - 1
        for wi, ci in zip(weights, top_k):
            L = int(ci - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                seg = N - L
                grad[L:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    seg = N - M
                    grad[:seg] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:seg]

        v = 0.9 * v - lr * grad
        h_new = project_sum(y + v, target)
        h = h_new
        it += 1

            T = max(0.003, T * 0.99)

    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lag_gradient(h, L, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < L < N:
        seg = N - L
        g[L:] += (1.0 - h[:seg])
        g[:seg] -= h[L:]
    elif -N < L < 0:
        M = -L
        seg = N - M
        g[:seg] += (1.0 - h[M:])
        g[M:] -= h[:seg]
    return g

def continuous_opt(h0, N, deadline, lr0=0.04, mu0=0.82, T0=0.015, k_smooth=12):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    v = np.zeros(N)
    lr = lr0
    mu = mu0
    T = T0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + mu * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        center = N - 1
        top_k = np.argpartition(corr, -k_smooth)[-k_smooth:]
        expv = np.exp((corr[top_k] - corr[top_k].max()) / max(T, 1e-8))
        wts = expv / expv.sum()
        grad = np.zeros(N)
        for wi, ci in zip(wts, top_k):
            L = int(ci - center)
            g = lag_gradient(y, L, N)
            grad += wi * g
        v = 0.88 * v - lr * grad
        h_try = project_sum(y + v, target)
        pk_try = float(np.max(fft_corr(h_try)))
        if pk_try > pk * 1.005:
            v = 0.5 * v - lr * 0.3 * grad
            h_try = project_sum(y + v, target)
        h = h_try
        it += 1
        if it % 80 == 0:
            lr = max(0.003, lr * 0.996)
            T = max(0.002, T * 0.992)
            mu = min(0.94, mu0 + it * 0.00008)
        if stall > 250:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            h_prev = h.copy()
            v = np.zeros(N)
            lr = max(lr, lr0 * 0.75)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def peak_reduction(h0, N, deadline, lr0=0.06):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    lr = lr0
    stall = 0
    it = 0
    center = N - 1
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        grad = np.zeros(N)
        near = np.where(corr > pk - 1.5)[0]
        for ki in near:
            L = int(ki - center)
            w = max(0.1, 1.0 - (pk - corr[ki]) * 0.8)
            grad += w * lag_gradient(h, L, N)
        grad /= max(1.0, len(near) * 0.5)
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_sum(h - lr * alpha * grad, target)
            pk_try = float(np.max(fft_corr(h_try)))
            if pk_try < pk:
                h = h_try
                break
        else:
            h = project_sum(h - lr * 0.05 * grad, target)
        it += 1
        if it % 60 == 0:
            lr = max(0.004, lr * 0.997)
        if stall > 200:
            h = best_h + np.random.randn(N) * 0.025
            h = project_sum(h, target)
            lr = max(lr, lr0 * 0.6)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_search(h0, N, deadline, n_tries=20):
    h = (h0 > 0.5).astype(float).copy()
    tgt = N // 2
    n1 = int(h.sum())
    if n1 != tgt:
        if n1 > tgt:
            idx = np.where(h == 1)[0]
            h[idx[np.random.permutation(len(idx))[:n1 - tgt]]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[np.random.permutation(len(idx))[:tgt - n1]]] = 1
    c = fft_corr(h)
    best_pk = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_d = 0
        best_mv = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] -= 1.0
            h[q] += 1.0
            pk = int(np.max(fft_corr(h)))
            d = pk - best_pk
            if d < best_d:
                best_d = d
                best_mv = (p, q)
            h[p] += 1.0
            h[q] -= 1.0
        if best_mv is not None:
            p, q = best_mv
            h[p] -= 1.0
            h[q] += 1.0
            c = fft_corr(h)
            pk = int(np.max(c))
            if pk < best_pk:
                best_pk = pk
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                for _ in range(np.random.randint(2, 6)):
                    p = ones[np.random.randint(len(ones))]
                    q = zeros[np.random.randint(len(zeros))]
                    h[p] -= 1.0
                    h[q] += 1.0
                one2 = np.where(h == 1)[0]
                zero2 = np.where(h == 0)[0]
                stall = 0
    return best_h, best_pk * (2.0 / N)

def interp_to_N(h, N_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: Peak reduction at N0 (250s)
    d1 = min(deadline - 600, t0 + 250)
    hn, c5n = peak_reduction(best_h, N0, d1, lr0=0.05)
    if c5n < best_c5:
        best_c5, best_h = c5n, hn.copy()
    # Phase 2: Nesterov continuous at N0 (200s)
    d2 = max(d1, t0) + 200
    d2 = min(deadline - 400, d2)
    h2 = best_h + np.random.randn(N0) * 0.03
    h2 = project_sum(h2, N0 / 2.0)
    hn2, c5n2 = continuous_opt(h2, N0, d2, lr0=0.03, T0=0.012, k_smooth=15)
    if c5n2 < best_c5:
        best_c5, best_h = c5n2, hn2.copy()
    # Phase 3: Multi-resolution continuous (250s)
    for Ns in [570, 760]:
        if time.time() > deadline - 180:
            break
        if Ns % 2 != 0:
            Ns += 1
        hi = interp_to_N(best_h, Ns)
        hi = project_sum(hi, Ns / 2.0)
        sub_dl = min(deadline - 170, time.time() + 120)
        hi2, c5i = peak_reduction(hi, Ns, sub_dl, lr0=0.04)
        if c5i < best_c5:
            best_c5 = c5i
            best_h = hi2.copy()
    # Phase 4: Binary search at best N (120s)
    Nb = len(best_h)
    if time.time() < deadline - 100:
        d4 = min(deadline - 80, time.time() + 120)
        hb, c5b = binary_search(best_h, Nb, d4, n_tries=25)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()
    # Phase 5: Final Nesterov polish
    if time.time() < deadline - 5:
        hF = best_h + np.random.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = continuous_opt(hF, len(best_h), deadline - 2, lr0=0.012, T0=0.006, k_smooth=20)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
