import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binarize(h, target_ones):
    """Convert continuous h to binary with exactly target_ones ones."""
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == target_ones:
        return hb
    if n1 > target_ones:
        oi = np.where(hb == 1)[0]
        # Remove the ones with smallest h value
        to_remove = n1 - target_ones
        idx = oi[np.argsort(h[oi])[:to_remove]]
        hb[idx] = 0
    else:
        zi = np.where(hb == 0)[0]
        to_add = target_ones - n1
        idx = zi[np.argsort(-h[zi])[:to_add]]
        hb[idx] = 1
    return hb

def swap_search(h, N, c, D, top_k=1, sample_ones=None, sample_zeros=None):
    """Find best swap reducing top-k peaks."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None, 0.0
    
    if sample_ones and len(ones) > sample_ones:
        ones_s = np.random.choice(ones, sample_ones, replace=False)
    else:
        ones_s = ones
    if sample_zeros and len(zeros) > sample_zeros:
        zeros_s = np.random.choice(zeros, sample_zeros, replace=False)
    else:
        zeros_s = zeros
    
    # Get top-k peak indices
    sorted_idx = np.argsort(c)[::-1]
    top_idx = sorted_idx[:top_k]
    
    best_delta = 0.0
    best_pair = None
    
    for p in ones_s:
        base = c - D[p, :]  # correlation after removing p
        # For top-k: compute sum of top-k after the swap
        cand = base[None, :] - D[zeros_s, :]  # (n_zeros, 2N-1)
        # Use top-k sum as objective
        if top_k == 1:
            obj = cand.max(axis=1)
        else:
            # Sum of top-k
            sorted_cand = np.sort(cand, axis=1)[:, ::-1]
            obj = sorted_cand[:, :top_k].sum(axis=1)
        mi = int(np.argmin(obj))
        d = float(obj[mi]) - float(c[top_idx].sum() if top_k > 1 else c.max())
        if d < best_delta:
            best_delta = d
            best_pair = (int(p), int(zeros_s[mi]))
    
    return best_pair, best_delta

def continuous_descent(h, N, deadline, lr=0.02, k=5):
    """Continuous gradient descent minimizing top-k peaks."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        current = float(c.max())
        if current < best_val:
            best_val = current
            best_h = h.copy()
        
        D = compute_D(h, N)
        # Gradient of top-k sum
        top_idx = np.argsort(c)[::-1][:k]
        grad = -D[:, top_idx].sum(axis=1) * dx
        grad -= grad.mean()  # maintain sum constraint
        
        bc1 = 1 - b1**t
        bc2 = 1 - b2**t
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad**2
        
        lr_t = lr / (1 + t*0.001)
        h = h + lr_t * (m/bc1) / (np.sqrt(v/bc2) + eps)
        
        # Project to [0,1] with correct sum
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) > 1e-6:
            adj = (target - s) / N
            h = h + adj
            h = np.clip(h, 0, 1)
            # Iterate to fix
            for _ in range(10):
                s = h.sum()
                if abs(s - target) < 1e-8:
                    break
                h = h + (target - s) / N
                h = np.clip(h, 0, 1)
        
        if t % 200 == 0:
            lr *= 0.9
    
    return best_h, best_val * dx

For binary h, the exact change in correlation at lag k from swapping p (1→0) and q (0→1):
- If p ≠ q (which is always the case since p is in ones and q is in zeros):

    pp = pos + sv

    vp = (pp >= 0) & (pp < N)

    pp_c = np.clip(pp, 0, N - 1)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)

    shifts = np.arange(2 * N - 1) - (N - 1)

    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv

    vp = (pp >= 0) & (pp < N)

    pp_c = np.clip(pp, 0, N - 1)

    hp = np.where(vp, h[pp_c], 0.0)

            hb, _ = optimize(best_h.copy(), N0, dB, top_k=1, stall_lim=50)
            c5b = compute_c5(hb, N0)
            if c5b < best_c5:
                best_c5, best_h = c5b, hb.copy()

    # Phase 4: Annealed search (70s)
    if time.time() < deadline - 110:
        d4 = min(deadline - 50, time.time() + 70)
        ha, _ = optimize(best_h.copy(), N0, d4, top_k=1, stall_lim=200, anneal=0.06)
        c5a = compute_c5(ha, N0)
        if c5a < best_c5:
            best_c5, best_h = c5a, ha.copy()

    # Phase 5: Multi-resolution (110s)
    xb = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [400, 360, 440, 480, 500]:
        if time.time() > deadline - 30:
            break
        if Nt % 2:
            Nt += 1
        if Nt == N0:
            continue
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, xb, best_h)
        ht = binarize(hi, Nt // 2)
        sub = min(deadline - 12, time.time() + 22)
        hr2, _ = optimize(ht, Nt, sub, top_k=1, stall_lim=40)
        c5r = compute_c5(hr2, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr2.copy()
            xb = np.linspace(0, 2, Nt, endpoint=False)

    # Phase 6: Final ILS (20s)
    if time.time() < deadline - 15:
        d6 = min(deadline - 5, time.time() + 15)
        for _ in range(10):
            if time.time() > d6:
                break
            hp = perturb(best_h, np.random.randint(2, 6))
            sub = min(d6, time.time() + 2.5)
            ho, _ = optimize(hp, len(hp), sub, top_k=1, stall_lim=20)
            c5 = compute_c5(ho, len(ho))
            if c5 < best_c5:
                best_c5, best_h = c5, ho.copy()

    # Final polish
    if time.time() < deadline - 1:
        hf = best_h.copy()
        hf, _ = optimize(hf, len(hf), deadline - 0.5, top_k=1, stall_lim=15)
        c5f = compute_c5(hf, len(hf))
        if c5f < best_c5:
            best_c5, best_h = c5f, hf

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
