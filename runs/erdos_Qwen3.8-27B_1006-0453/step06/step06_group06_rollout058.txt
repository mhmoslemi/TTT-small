```python
def compute_all_flip_deltas(h, N):
    """Compute delta for each single position flip.
    Returns: flip_one (N, 2N-1) where flip_one[p] is delta for h[p]: 1->0
              flip_zero (N, 2N-1) where flip_zero[q] is delta for h[q]: 0->1
    """
    lags = np.arange(-(N-1), N)  # length 2N-1
    # For position p, lag l: need h[p+l] and h[p-l]
    # p+l valid: 0 <= p+l < N
    # p-l valid: 0 <= p-l < N
    
    positions = np.arange(N)[:, None]  # (N, 1)
    lags_r = lags[None, :]  # (1, 2N-1)
    
    idx_plus = positions + lags_r  # (N, 2N-1)
    idx_minus = positions - lags_r
    
    valid_plus = (idx_plus >= 0) & (idx_plus < N)
    valid_minus = (idx_minus >= 0) & (idx_minus < N)
    
    # Clip for safe indexing
    ip = np.clip(idx_plus, 0, N-1)
    im = np.clip(idx_minus, 0, N-1)
    
    # delta for flip 1->0: -(1-h[p+l]) * valid_plus + h[p-l] * valid_minus
    flip_one = -(1.0 - h[ip]) * valid_plus + h[im] * valid_minus
    # delta for flip 0->1: + (1-h[q+l]) * valid_plus - h[q-l] * valid_minus
    flip_zero = (1.0 - h[ip]) * valid_plus - h[im] * valid_minus
    
    # Only keep valid positions
    flip_one = flip_one * ((h == 1)[ :, None])
    flip_zero = flip_zero * ((h == 0)[:, None])
    
    return flip_one, flip_zero
```

```python
def compute_flip_deltas(h, N):
    """Return flip_deltas (N, 2N-1) where flip_deltas[p] is the change in corr
       when h[p] goes from its current value to opposite (1->0 or 0->1)."""
    lags = np.arange(-(N-1), N)  # (2N-1,)
    pos = np.arange(N)[:, None]  # (N, 1)
    lr = lags[None, :]  # (1, 2N-1)
    
    idx_p = pos + lr  # (N, 2N-1) = p + lag
    idx_m = pos - lr  # (N, 2N-1) = p - lag
    
    vp = (idx_p >= 0) & (idx_p < N)
    vm = (idx_m >= 0) & (idx_m < N)
    
    ip = np.clip(idx_p, 0, N-1)
    im = np.clip(idx_m, 0, N-1)
    
    # For h[p]: 1->0: delta = -(1-h[p+l]) + h[p-l] (for l≠0), 0 for l=0
    d10 = -(1.0 - h[ip]) * vp + h[im] * vm
    # For h[p]: 0->1: delta = +(1-h[p+l]) - h[p-l] (for l≠0), 0 for l=0
    d01 = (1.0 - h[ip]) * vp - h[im] * vm
    
    # At lag=0 (index N-1), both should be 0
    center = N - 1
    d10[:, center] = 0.0
    d01[:, center] = 0.0
    
    # Only apply to positions with correct value
    d10 *= (h == 1)[:, None]
    d01 *= (h == 0)[:, None]
    
    return d10, d01
```

```python
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def flip_deltas(h, N):
    lags = np.arange(-(N-1), N)
    pos = np.arange(N)[:, None]
    lr = lags[None, :]
    idx_p = pos + lr
    idx_m = pos - lr
    vp = (idx_p >= 0) & (idx_p < N)
    vm = (idx_m >= 0) & (idx_m < N)
    ip = np.clip(idx_p, 0, N-1)
    im = np.clip(idx_m, 0, N-1)
    d10 = -(1.0 - h[ip]) * vp + h[im] * vm
    d01 = (1.0 - h[ip]) * vp - h[im] * vm
    c = N - 1
    d10[:, c] = 0.0
    d01[:, c] = 0.0
    d10 *= (h == 1)[:, None]
    d01 *= (h == 0)[:, None]
    return d10, d01

def local_search(h, N, deadline, max_kicks=30, kick_sizes=None):
    h = h.copy()
    corr = make_corr(h).astype(float)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    if kick_sizes is None:
        kick_sizes = [0, 2, 2, 3, 3, 4, 4, 5, 5, 6, 8, 8, 10, 12, 15, 20, 25, 35, 45, 55, 70, 90, 110, 130, 160, 200, 250, 300, 350, 400]
    max_kicks = min(max_kicks, len(kick_sizes))
    for kick_num in range(max_kicks):
        if time.time() > deadline - 3:
            break
        if kick_num > 0:
            ks = kick_sizes[kick_num]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(min(ks, len(ones))):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 1.0 - h[p], 1.0 - h[q]
            corr = make_corr(h).astype(float)
            cur_max = float(corr.max())
        while True:
            if time.time() > deadline - 3:
                return best_h, best_max
            d10, d01 = flip_deltas(h, N)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            fo = d10[ones]
            fz = d01[zeros]
            m_star = int(np.argmax(corr))
            fz_col_min = fz[:, m_star].min()
            best_val = cur_max
            best_pi = -1
            best_qi = -1
            for i in range(len(ones)):
                lb = corr[m_star] + fo[i, m_star] + fz_col_min
                if lb >= best_val - 1e-12:
                    continue
                A = corr + fo[i]
                C = A[None, :] + fz
                maxs = C.max(axis=1)
                j = int(np.argmin(maxs))
                if maxs[j] < best_val - 1e-12:
                    best_val = float(maxs[j])
                    best_pi = i
                    best_qi = j
            if best_pi < 0:
                break
            p = ones[best_pi]
            q = zeros[best_qi]
            h[p] = 0.0
            h[q] = 1.0
            corr = corr + d10[p] + d01[q]
            cur_max = best_val
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def continuous_opt(h, N, deadline, top_k=40, lr0=0.004):
    h = h.astype(float).copy()
    m_arr = np.zeros(N)
    v_arr = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(make_corr(h)))
    best_h = h.copy()
    center = N - 1
    lr = lr0
    while time.time() < deadline:
        corr = make_corr(h).astype(float)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += (1.0 - h[s:])
                grad[s:] -= h[:N-s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N-s2])
                grad[:N-s2] -= h[s2:]
        grad /= top_k
        t += 1
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * grad * grad
        mh = m_arr / (1 - b1**t)
        vh = v_arr / (1 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 800 == 0:
            lr *= 0.95
    return best_h, best_pk

def to_binary(h, N):
    hb = (h >= 0.5).astype(float)
    target = N // 2
    n1 = int(hb.sum())
    if n1 > target:
        idx = np.where(hb == 1)[0]
        sel = np.random.choice(len(idx), n1 - target, replace=False)
        hb[idx[sel]] = 0
    elif n1 < target:
        idx = np.where(hb == 0)[0]
        sel = np.random.choice(len(idx), target - n1, replace=False)
        hb[idx[sel]] = 1
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_c5 = float(np.max(make_corr(h0))) * (2.0 / N0)
    best_h = h0.copy()

    # Phase 1: Extended best-improvement local search at N0
    hb0 = to_binary(h0, N0)
    d1 = min(deadline - 300, t0 + 350)
    hb, pb = local_search(hb0, N0, d1, max_kicks=25)
    c5b = pb * (2.0 / N0)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 2: Continuous optimization from best binary
    if time.time() < deadline - 250:
        d2 = min(deadline - 250, time.time() + 150)
        hc = hb + np.random.randn(N0) * 0.03
        hc = proj_sum(hc, N0 / 2.0)
        hC, pkC = continuous_opt(hc, N0, d2, top_k=45, lr0=0.004)
        c5C = pkC * (2.0 / N0)
        if c5C < best_c5:
            best_c5, best_h = c5C, hC.copy()
        # Re-binarize and polish
        hb2 = to_binary(hC, N0)
        d2b = min(deadline - 200, time.time() + 80)
        hb3, pb3 = local_search(hb2, N0, d2b, max_kicks=10)
        c5b3 = pb3 * (2.0 / N0)
        if c5b3 < best_c5:
            best_c5, best_h = c5b3, hb3.copy()

    # Phase 3: Multi-resolution exploration
    for Nt in [360, 400, 440, 500, 560, 700]:
        if time.time() > deadline - 60:
            break
        if Nt % 2 != 0:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = to_binary(hi, Nt)
        sub = min(deadline - 5, time.time() + 50)
        hrt, pkr = local_search(hbt, Nt, sub, max_kicks=15)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hrt.copy()

    # Phase 4: Final intensive local search on best
    if time.time() < deadline - 10:
        Nf = len(best_h)
        hbf = to_binary(best_h, Nf)
        hff, pf = local_search(hbf, Nf, deadline - 2, max_kicks=20)
        c5f = pf * (2.0 / Nf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hff.copy()

    # Final verification
    Nf = len(best_h)
    final_c5 = float(np.max(make_corr(best_h))) * (2.0 / Nf)
    return best_h, final_c5, Nf
```
