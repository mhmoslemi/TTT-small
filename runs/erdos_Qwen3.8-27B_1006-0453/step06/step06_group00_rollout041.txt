        c5_val = current_max * (2.0 / N)
        if c5_val < best_c:
            best_c = c5_val
            best_h = h.copy()

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # For each p in ones, find best q in zeros
        v_ones = v[ones]   # (n1, 2N-1)
        v_zeros = v[zeros] # (n0, 2N-1)
        
        best_max = np.inf

            # For each q: max_k(c_p[k] - v_zeros[q][k])
            diffs = c_p[:, None] - v_zeros  # (2N-1, n0)
            maxes = diffs.max(axis=0)  # (n0,)

                best_q = zeros[mi]

        h[best_q] = 1

    final_c = compute_c5(h, N)
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()
    
    return best_h, best_c

    best_h = h.copy()
    best_c = compute_c5(h, N)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_curr = best_h.copy()
        for _ in range(perturb_sz):
            ones = np.where(h_curr == 1)[0]
            zeros = np.where(h_curr == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            h_curr[int(np.random.choice(ones))] = 0
            h_curr[int(np.random.choice(zeros))] = 1
        sub_dl = min(deadline - 20, time.time() + 30)
        h_new, c_new = steepest_binary(h_curr, N, sub_dl, max_iters=150)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c
