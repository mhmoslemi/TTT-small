def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def to_binary(h, N):
    n_t = N // 2
    idx = np.argsort(-h)
    r = np.zeros(N)
    r[idx[:n_t]] = 1.0
    return r

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    ip = pos + sv
    im = pos - sv
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    ipc = np.clip(ip, 0, N - 1)
    imc = np.clip(im, 0, N - 1)
    return (1.0 - h[ipc]) * vp - h[imc] * vm, vp, vm, ipc, imc

def soft_max_energy(c, T, dx):
    cc = c * dx
    cs = cc * T
    m = cs.max()
    return m / T + np.log(np.sum(np.exp(cs - m))) / T

def soft_max_w(c, T):
    cs = c * T
    m = cs.max()
    e = np.exp(cs - m)
    return e / e.sum()

def local_search(h, N, T, deadline, max_sw=500, sa_init=0.002, sa_min=0.0001):
    h = h.astype(float).copy()
    dx = 2.0 / N
    c = fft_corr(h)
    D, vp, vm, ipc, imc = compute_D(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_c5 = float(c.max())
    cur_E = soft_max_energy(c, T, dx)
    best_E = cur_E
    sa_T = sa_init
    stall = 0
    
    for sw in range(max_sw):
        if time.time() > deadline:
            break
        w = soft_max_w(c, T)
        wD = D @ w
        # Best swap: max wD in ones, min wD in zeros
        bi = ones[np.argmax(wD[ones])]
        bj = zeros[np.argmin(wD[zeros])]
        delta = dx * (wD[bj] - wD[bi])
        
        accept = delta < -1e-13 or (sa_T > sa_min and np.random.rand() < np.exp(-delta / sa_T))
        if accept:
            h[bi], h[bj] = 0.0, 1.0
            c += D[bj] - D[bi]
            D = (1.0 - h[ipc]) * vp - h[imc] * vm
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_E = soft_max_energy(c, T, dx)
            pm = c.max()
            if pm < best_c5 - 1e-14:
                best_c5 = pm
            if cur_E < best_E:
                best_E = cur_E
                stall = 0
            else:
                stall += 1
            if stall % 20 == 0 and sa_T > sa_min:
                sa_T *= 0.95
            # Random swap to escape
            if stall > 60:
                ri = ones[np.random.randint(len(ones))]
                rj = zeros[np.random.randint(len(zeros))]
                h[ri], h[rj] = 0.0, 1.0
                c = fft_corr(h)
                D = (1.0 - h[ipc]) * vp - h[imc] * vm
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                sa_T = max(sa_T * 2, 0.001)
                stall = 0
        else:
            stall += 1
            if stall > 80:
                break
    
    return h, float(c.max()) * dx

def adam_continuous(h, N, deadline, T=20, lr0=0.05, k_top=8):
    """Continuous relaxation optimization"""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = fft_corr(h).max()
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = lr0
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    ip = pos + sv
    im = pos - sv
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    ipc = np.clip(ip, 0, N - 1)
    imc = np.clip(im, 0, N - 1)
    
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = c.max()
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        w = soft_max_w(c, T)
        D = (1.0 - h[ipc]) * vp - h[imc] * vm
        grad = (D @ w) * dx
        grad -= grad.mean()
        bc1 = 1 - b1 ** t
        bc2 = 1 - b2 ** t
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        h = h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = proj_sum(h, target)
        if t % 200 == 0:
            lr *= 0.94
            # Periodic binary-ization
            hb = to_binary(h, N)
            h = 0.7 * h + 0.3 * hb
    return best_h, float(best_pk) * dx
