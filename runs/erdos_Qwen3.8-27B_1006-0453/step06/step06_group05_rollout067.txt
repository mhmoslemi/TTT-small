import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    grad = np.zeros(N)
    if top_indices is None:

        T = 50.0
        w = np.exp(T * (c - c_max))

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    m = np.zeros(N)
    v = np.zeros(N)

        c_scaled = c * dx
        pk = float(c_scaled.max())

        c_max = c.max()
        # Use points within 5% of max

        top_mask = c >= threshold

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    """Convert continuous h to binary with exactly N/2 ones."""
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    nh = N // 2
    if n1 == nh:
        return hb
    oi = np.where(hb == 1)[0]
    zi = np.where(hb == 0)[0]
    if n1 > nh:
        # Remove n1-nh ones (lowest h values)
        idx = oi[np.argsort(h[oi])[:n1-nh]]
        hb[idx] = 0
    else:
        # Add nh-n1 ones (highest h values among zeros)
        idx = zi[np.argsort(-h[zi])[:nh-n1]]
        hb[idx] = 1
    return hb

def cont_opt(h, N, deadline, lr0=0.03, topk=8):
    """Continuous subgradient optimization."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max()) * dx
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        # Top-k indices
        top_idx = np.argpartition(c, -topk)[-topk:]
        # Gradient for each top k
        grad = np.zeros(N)
        for j in top_idx:
            k = j - (N - 1)
            if k >= 0:
                grad[k:] += (1.0 - h[:N-k])
                grad[:N-k] -= h[k:]
            else:
                kp = -k
                grad[:N-kp] += (1.0 - h[kp:])
                grad[kp:] -= h[:N-kp]
        grad /= topk
        grad -= grad.mean()
        bc1 = 1 - b1**t
        bc2 = 1 - b2**t
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad**2
        h_new = h - lr * (m/bc1) / (np.sqrt(v/bc2) + eps)
        h = project_sum(h_new, target)
        if t % 200 == 0:
            lr *= 0.95
        # Periodic binary blend
        if t % 400 == 0:
            hb = to_binary(h, N)
            h = 0.6*h + 0.4*hb
            h = project_sum(h, target)
    return best_h, best_c

def bin_opt(h, N, deadline, max_sweeps=3000):
    """Binary local optimization with FFT verification."""
    h = h.astype(float).copy()
    best_c = compute_c5(h, N)
    dx = 2.0 / N
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h) * dx
        kstar = int(np.argmax(c))
        shift = kstar - (N - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Score each p (to turn off) and q (to turn on) for this shift
        if shift >= 0:
            Ap = h[np.minimum(ones+shift, N-1)].astype(float) * (ones+shift < N) 
            Ap -= (1.0 - h[np.maximum(ones-shift, 0)]) * (ones-shift >= 0)
            Bq = -h[np.minimum(zeros+shift, N-1)].astype(float) * (zeros+shift < N)
            Bq += (1.0 - h[np.maximum(zeros-shift, 0)]) * (zeros-shift >= 0)
        else:
            kp = -shift
            Ap = h[np.minimum(ones+kp, N-1)].astype(float) * (ones+kp < N)
            Ap -= (1.0 - h[np.maximum(ones-kp, 0)]) * (ones-kp >= 0)
            Bq = -(1.0 - h[np.maximum(zeros-kp, 0)]) * (zeros-kp >= 0)
            Bq += h[np.minimum(zeros+kp, N-1)].astype(float) * (zeros+kp < N)
        
        # Find best (p,q) pair: minimize Ap[p] + Bq[q]
        # Use a trick: for each p, best q is argmin Bq
        best_q = int(np.argmin(Bq))
        scores = Ap + Bq[best_q]
        best_p = int(np.argmin(scores))
        
        p_swap = ones[best_p]
        q_swap = zeros[best_q]
        h[p_swap] = 0.0
        h[q_swap] = 1.0
        
        new_c = compute_c5(h, N)
        if new_c < best_c - 1e-12:
            best_c = new_c
            stall = 0
        else:
            h[p_swap] = 1.0
            h[q_swap] = 0.0
            stall += 1
            if stall > 50:
                break
    return h, best_c

One issue with my current approach: the `bin_opt` only considers the single worst shift k*. This means it might get stuck. A better approach:

**Multi-shift binary optimization**: Consider the top-3 worst shifts, and find a swap that reduces the maximum of these three.

For a swap (p, q), the change in c_j for shift j is:
Δc_j = A_j[p] + B_j[q]

We want to minimize max_{j in {k1,k2,k3}} (c_j + Δc_j)

For each p, the best q for shift j is argmin B_j[q]. But the best q might be different for different j's.

Heuristic: For each p, compute the "worst case" over the top-3 shifts:
score_p = max_j (c_j + A_j[p] + min_q B_j[q])
Then find p that minimizes score_p, and for that p, find q that minimizes max_j(c_j + A_j[p] + B_j[q]).

This is O(N_ones * 3 + N_zeros * 3) per evaluation, which is fast.

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

        h7, c7 = bin_opt(hb7, len(best_h), d7, max_sweeps=5000, n_shifts=5)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    h_final = to_binary(best_h, len(best_h))
    final_c5 = compute_c5(h_final, len(h_final))
    return h_final, final_c5, len(h_final)
