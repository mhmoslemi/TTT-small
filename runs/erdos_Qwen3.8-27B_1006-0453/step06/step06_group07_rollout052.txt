import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, stall_limit=40):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(777)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        p_list = ones
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))

        if best_pq is None:

        h[best_pq[0]] = 0.0

def nesterov_opt(h, N, deadline, k=50, lr0=0.018):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

def simulate_anneal(h, N, deadline, n_iter=5000, T0=0.005, T1=0.0001):
    """Simulated annealing on binary strings."""
    h = h.astype(float).copy()
    h = (h > 0.5).astype(float)
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    curr_c = best_c
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_total = n_iter
    for it in range(n_total):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_total, 1))
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        h[p], h[q] = 0.0, 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        new_c = float(np.max(fft_corr(h)))
        delta = new_c - curr_c
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    return best_h, best_c * (2.0 / N)

def fft_corr(h):

    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)

    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv

    vp = (pp >= 0) & (pp < N)

    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp
