    #   valid j: max(0, N-1-m) to min(N-1, 2N-2-m)
    #   grad[j] += p[m] * (1-h)[j+m-N+1]
    
    # Let me just vectorize with broadcasting for the first term:
    # Create matrix M1[j, m] = (1-h)[j+m-(N-1)] if valid, 0 otherwise
    # Then grad_first = M1 @ p
    
    # For N=380, M1 is 380 x 759, which is 288,420 elements. Fine.
    j_idx = np.arange(N)[:, None]  # N x 1
    m_idx2 = np.arange(2*N-1)[None, :]  # 1 x (2N-1)

    valid = (pos >= 0) & (pos < N)
    pos_clipped = np.clip(pos, 0, N-1)
    M1 = np.where(valid, one_minus_h[pos_clipped], 0.0)
    grad += M1 @ p  # This is the first term
    
    # Second term: grad[j] -= sum_m p[m] * h[j-m+(N-1)] * I(0<=j-m+(N-1)<N)
    # j-m+(N-1) >= 0 → j >= m-(N-1)
    # j-m+(N-1) < N → j < N-1+m
    # So j ranges from max(0, m-(N-1)) to min(N-1, N-2+m)
    
    pos2 = j_idx - m_idx2 + (N-1)
    valid2 = (pos2 >= 0) & (pos2 < N)
    pos2_clipped = np.clip(pos2, 0, N-1)
    M2 = np.where(valid2, h[pos2_clipped], 0.0)
    grad -= M2 @ p
    
    return grad

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    a = np.zeros(L)
    b = np.zeros(L)
    a[:N] = h
    b[:N] = 1.0 - h
    H = np.fft.fft(a, L)
    G = np.fft.fft(b, L)
    return np.real(np.fft.ifft(H * G))

def project_bounds_sum(h, target):

    h = h.astype(float)

        clipped = np.clip(h - mid, 0.0, 1.0)
        if clipped.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

    """Compute gradient of log-sum-exp(β*C) w.r.t. h."""
    N = len(h)

    v2 = (pos2 >= 0) & (pos2 < N)
    M2 = np.where(v2, h[np.clip(pos2, 0, N-1)], 0.0)

            last_improve = time.time()
        
        h_new = project_bounds_sum(h - lr * grad, target)

        C_new = fft_corr(h_new)
        pk_new = float(C_new.max())
        if pk_new < pk:

            C = C_new

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    a = np.zeros(L)
    b = np.zeros(L)
    a[:N] = h
    b[:N] = 1.0 - h
    return np.real(np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))))
