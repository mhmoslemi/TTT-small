def simulate_anneal(h_bin, N, deadline, T0=5.0, T1=0.01, restarts=3):
    """SA with proper cooling schedule."""
    h = h_bin.copy()
    s = 2.0 * h - 1.0
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    T = T0

        # Pick random pair
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]

        # Flipping p: h[p] 1->0, s[p] 1->-1
        # Flipping q: h[q] 0->1, s[q] -1->1
        # New corr = corr + (s[p] * ... ) - (s[q] * ...)
        # Actually, delta in corr from flipping position i:
        # corr_new[k] = corr[k] + delta_i[k]
        # where delta_i[k] depends on s[i] and the structure

        # delta_p[k] = -s[p] * (contribution of position p at lag k)

        h[p], h[q] = 0.0, 1.0
        new_corr = fft_corr(h)
        new_c = float(new_corr.max())
        delta_c = new_c - best_c
        
        if delta_c < 0 or np.random() < math.exp(-delta_c / T):
            corr = new_corr
            s[p], s[q] = -s[p], -s[q]
            ones.remove(p); ones.append(q)
            zeros.remove(q); zeros.append(p)

                best_c = new_c
                best_h = h.copy()

            h[p], h[q] = 1.0, 0.0
        
        T *= cooling
        n_iter += 1
    
    return best_h, best_c

import numpy as np
import time
import math

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary_sum(h, N):
    h = h.astype(float).copy()
    h = np.clip(h, 0, 1)
    target = N // 2
    n1 = int(round(h.sum()))
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        np.random.shuffle(ones)
        h[ones[:n1 - target]] = 0.0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - n1]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def init_deltas(h, N):
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        lo1 = max(0, i - N + 1)
        hi1 = min(2 * N - 2, i)
        if lo1 <= hi1:
            j1 = np.arange(lo1, hi1 + 1)
            deltas[i, lo1:hi1 + 1] += 1.0 - h[j1 + N - 1 - i]
        lo2 = max(0, i)
        hi2 = min(2 * N - 2, i + N - 1)
        if lo2 <= hi2:
            j2 = np.arange(lo2, hi2 + 1)
            deltas[i, lo2:hi2 + 1] -= h[i - j2 + N - 1]
    return deltas

def sa_optimize(h_init, N, deadline, T0=0.004, alpha=0.99997, min_T=0.00005, greedy_prob=0.3, sample_size=40):
    h = fix_binary_sum(h_init, N)
    corr = fft_corr(h)
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    deltas = init_deltas(h, N)
    i_arr = np.arange(N)
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    T = T0
    no_imp = 0
    n_iter = 0
    while time.time() < deadline and T > min_T:
        if greedy_prob > 0 and np.random.rand() < T / T0 * greedy_prob:
            n_s = min(sample_size, len(ones_list), len(zeros_list))
            ps = np.random.choice(len(ones_list), n_s, replace=False)
            qs = np.random.choice(len(zeros_list), n_s, replace=False)
            best_dc = 1e9
            bp, bq = ps[0], qs[0]
            for ii in range(n_s):
                for jj in range(min(n_s, 10)):
                    p = ones_list[int(ps[ii])]
                    q = zeros_list[int(qs[jj])]
                    dc = float((corr + deltas[q] - deltas[p]).max()) - cur_c
                    if dc < best_dc:
                        best_dc = dc
                        bp, bq = p, q
            p, q = bp, bq
            delta_c = best_dc
        else:
            p = ones_list[np.random.randint(len(ones_list))]
            q = zeros_list[np.random.randint(len(zeros_list))]
            delta_c = float((corr + deltas[q] - deltas[p]).max()) - cur_c
        if delta_c <= 0 or (T > 1e-10 and np.random.rand() < math.exp(-delta_c / T)):
            h[p], h[q] = 0.0, 1.0
            corr = corr + deltas[q] - deltas[p]
            cur_c = float(corr.max())
            j1 = p - i_arr + N - 1
            m1 = (j1 >= 0) & (j1 < 2 * N - 1)

    for it in range(n_iter_max):

        corr_y = fft_corr(y)
        top = np.argpartition(corr_y, -k)[-k:]
        top = np.sort(top)
        w = np.exp(20.0 * (corr_y[top] - corr_y.max()))
        w /= w.sum()
        grad = np.zeros(N)
        for t in top:

                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                wi = w[list(top).index(t)]
                grad[:N - M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N - M]
        h_new = np.clip(y - lr * grad, 0, 1)
        s = float(h_new.sum())
        h_new = h_new - (s - target) / N
        h_new = np.clip(h_new, 0, 1)
        m_vec = beta * (h_new - h)
        h = h_new
        c_now = compute_c5(h)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
        lr *= 0.998
