import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

def make_deltas(h, s):

    for i in range(N):
        si = s[i]

        valid_h = (idx_h >= 0) & (idx_h < N)
        idx_u = i + k
        valid_u = (idx_u >= 0) & (idx_u < N)
        deltas[i, valid_h] += si * h[idx_h[valid_h]]

def smooth_max_nesterov(h_in, N, deadline, T=20.0, max_iter=3000):
    h = h_in.astype(float).copy()
    target = N / 2.0
    hr = h.copy()
    m = np.zeros(N)
    best_pk = float(np.max(fft_corr(hr)))
    best_h = h.copy()
    beta = 0.8
    lr = 0.02
    top_k = 30

    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        # Smooth max gradient
        corr_clipped = np.clip(corr, -10.0, 10.0)
        w = np.exp(T * (corr_clipped - corr_clipped.max()))
        w /= w.sum()
        top = np.where(w > w.max() * 0.01)[0]
        if len(top) > top_k:
            top = np.where(w > 0.95 * w.max())[0]
            if len(top) > top_k:
                top = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= max(1.0, len(top))
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        c = float(np.max(fft_corr(h)))
        if c < best_pk:
            best_pk = c
            best_h = h.copy()
        if it % 200 == 0:
            lr *= 0.97
            beta = min(0.97, 0.75 + it * 0.0004)
            T *= 1.05
    return best_h, best_pk * (2.0 / N)
