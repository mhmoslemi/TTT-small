def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):

    return h_bin

def compute_delta(h, s_arr, p, q, p_is_one=True):
    """Compute the delta in correlation when flipping p (1->0 or 0->1) and q (opposite)."""
    N = len(h)
    ps, pm = p+s_arr, p-s_arr
    qs, qm = q+s_arr, q-s_arr
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d = np.zeros(2*N-1)
    pc=np.clip(ps,0,N-1); pmc=np.clip(pm,0,N-1)
    qsc=np.clip(qs,0,N-1); qmc=np.clip(qm,0,N-1)
    if p_is_one:
        d[vps] -= (1.0-h[pc[vps]])
        d[vpm] += h[pmc[vpm]]
    else:
        d[vps] += (1.0-h[pc[vps]])
        d[vpm] -= h[pmc[vpm]]
    d[vqs] += (1.0-h[qsc[vqs]])
    d[vqm] -= h[qmc[vqm]]
    return d

    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    pc=np.clip(ps,0,N-1)[vps]; pmc=np.clip(pm,0,N-1)[vpm]
    qsc=np.clip(qs,0,N-1)[vqs]; qmc=np.clip(qm,0,N-1)[vqm]
    d[vps] -= (1.0 - h[pc])
    d[vpm] += h[pmc]
    d[vqs] += (1.0 - h[qsc])
    d[vqm] -= h[qmc]
    return d

def anneal_binary(h, N, deadline, T0=0.05, cool=0.9996, max_stall=600, min_T=0.003):
    h = h.astype(float).copy()
    h = fix_count(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    T = T0
    stall = 0
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    while time.time() < deadline and T > min_T:
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        d = make_delta(h, s_arr, p, q)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-9)):
            h[p] = 0.0; h[q] = 1.0
            corr = nc; cur_pk = npk
            i1 = np.where(ones != p)[0]
            ones = np.concatenate([zeros[[np.where(zeros==q)[0][0]] if len(np.where(zeros==q)[0])>0 else 0], ones[i1]])
            j1 = np.where(zeros != q)[0]
            zeros = np.concatenate([ones[[np.where(ones==p)[0][0]] if len(np.where(ones==p)[0])>0 else 0], zeros[j1]])
            # Simpler update
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if npk < best_pk:
                best_pk = npk; best_h = h.copy(); stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T *= cool
        if stall > max_stall:
            h = best_h.copy(); corr = fft_corr(h); cur_pk = best_pk
            ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
            stall = 0
            T = max(T * 0.6, min_T * 3)
    return best_h, best_pk

def hill_climb(h, N, max_iters=20000, sample_size=40):
    h = h.astype(float).copy()
    h = fix_count(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    s_arr = np.arange(-(N-1), N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_iters):
        improved = False
        for _ in range(sample_size):
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            d = make_delta(h, s_arr, p, q)
            nc = corr + d
            npk = float(nc.max())
            if npk < best_pk:
                h[p] = 0.0; h[q] = 1.0
                corr = nc; best_pk = npk
                ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                improved = True
                break
        if not improved:
            break
    return h, best_pk

def continuous_opt(h, N, deadline, top_k=25, lr0=0.003):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h))); best_h = h.copy()
    center = N - 1; lr = lr0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += (1.0 - h[s:]); grad[s:] -= h[:N-s]
            else:
                s2 = -s; grad[s2:] += (1.0 - h[:N-s2]); grad[:N-s2] -= h[s2:]
        grad /= top_k
        t += 1
        m = b1*m + (1-b1)*grad; v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**t); vh = v/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        if t % 200 == 0:
            lr *= 0.96
    return best_h, best_pk

def interpolate_to_n(h_src, N_src, N_dst):
    x_old = np.linspace(0, 2, N_src, endpoint=False)
    x_new = np.linspace(0, 2, N_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(fft_corr(h0))) * (2.0 / N0)

    # Phase 1: Continuous optimization (quick, ~90s)
    d1 = min(deadline - 500, t0 + 90)
    for pert in [0.03, 0.07]:
        if time.time() > d1:
            break
        hc = h0 + np.random.randn(N0) * pert
        hc = proj_sum(hc, N0 / 2.0)
        hC, pkC = continuous_opt(hc, N0, min(deadline - 500, time.time() + 40), top_k=30, lr0=0.004)
        c5C = pkC * (2.0 / N0)
        if c5C < best_c5:
            best_c5, best_h = c5C, hC.copy()

    # Phase 2: Binary annealing at N0 (~200s)
    if time.time() < deadline - 300:
        d2 = min(deadline - 300, time.time() + 200)
        hb = fix_count((best_h > 0.5).astype(float), N0)
        hB, pkB = anneal_binary(hb, N0, d2, T0=0.06, cool=0.9995, max_stall=800)
        c5B = pkB * (2.0 / N0)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB.copy()

    # Phase 3: Multi-N binary search (~300s total)
    n_list = [320, 360, 400, 440, 500, 560, 640]
    time_per_N = 35
    for Nt in n_list:
        if time.time() > deadline - 120:
            break
        if Nt == N0:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_count((hi > 0.5).astype(float), Nt)
        sub = min(deadline - 120, time.time() + time_per_N)
        hrt, pkr = anneal_binary(hbt, Nt, sub, T0=0.05, cool=0.9996, max_stall=500)
        # Hill climb on result
        hrt2, pkr2 = hill_climb(hrt, Nt, max_iters=3000, sample_size=50)
        c5r = pkr2 * (2.0 / Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hrt2.copy()

    # Phase 4: Focused binary annealing at best_N (~150s)
    if time.time() < deadline - 80:
        Nb = len(best_h)
        d4 = min(deadline - 80, time.time() + 150)
        hb2 = fix_count((best_h > 0.5).astype(float), Nb)
        hB2, pkB2 = anneal_binary(hb2, Nb, d4, T0=0.04, cool=0.9997, max_stall=1000)
        c5B2 = pkB2 * (2.0 / Nb)
        if c5B2 < best_c5:
            best_c5, best_h = c5B2, hB2.copy()

    # Phase 5: Final hill climb polish
    if time.time() < deadline - 5:
        hf = fix_count((best_h > 0.5).astype(float), len(best_h))
        hf2, pkf = hill_climb(hf, len(best_h), max_iters=30000, sample_size=60)
        c5f = pkf * (2.0 / len(best_h))
        if c5f < best_c5:
            best_c5, best_h = c5f, hf2.copy()

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
