def get_flip_deltas(h, N):
    """Compute flip_delta[i] for all i: change in corr when h[i] is flipped."""
    L = 2*N - 1
    base = np.zeros((N, L))
    for i in range(N):
        # term1: (1-h[i-k+N-1]) for k in [i, min(i+N-1, L-1)]
        ks = np.arange(max(i,0), min(i+N, L))
        n_idx = i - ks + N - 1
        base[i, ks] += (1.0 - h[n_idx])
        # term2: -h[i+k-N+1] for k in [max(N-1-i,0), min(2*N-2-i, L-1)]
        ks2 = np.arange(max(N-1-i, 0), min(2*N-1-i, L))
        j_idx = i + ks2 - N + 1
        base[i, ks2] -= h[j_idx]
    sig = 1.0 - 2.0*h  # +1 where h=0, -1 where h=1
    return sig * base  # flip_delta[i] = (1-2h[i]) * base[i]
