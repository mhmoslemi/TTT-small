    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_binary(h, N, rng=None):
    nh = N // 2
    h = h.astype(float).copy()

    n1 = int(h_bin.sum())

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    D = np.zeros((N, 2 * N - 1))
    for i in range(N):
        pm = i - shifts
        pp = i + shifts
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        D[i] = -(1.0 - h[np.clip(pm, 0, N-1)]) * vm + h[np.clip(pp, 0, N-1)] * vp
    return D

def greedy_search(h, N, deadline, rng, max_sweeps=2000, sample_frac=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        if sample_frac < 1.0 and len(ones) > 30:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P + 1
        best_pq = None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def simulated_annealing(h, N, deadline, rng, t_start=0.05, t_end=0.001, cool=0.995):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = t_start
    no_improve = 0
    while time.time() < deadline and T > t_end:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        p = int(rng.choice(ones))
        q = int(rng.choice(zeros))
        new_c = c + D[p, :] - D[q, :]
        new_P = float(new_c.max())
        delta = new_P - float(c.max())
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h[p], h[q] = 0.0, 1.0
            c = new_c
            if new_P < best_c:
                best_c = new_P
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool
    return best_h, best_c * (2.0 / N)
