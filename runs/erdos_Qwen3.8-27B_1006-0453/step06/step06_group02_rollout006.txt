```python
def swap_delta(h, p, q):
    """Compute change in correlation when h[p]:1->0, h[q]:0->1"""
    N = len(h)
    # shift ranges from -(N-1) to (N-1), indexed 0 to 2N-2 in full correlation
    # For a given shift s (where corr index = s + N - 1):
    # delta[s] = -(1-h[p+s]) I[0<=p+s<N] + h[p-s] I[0<=p-s<N]
    #           + (1-h[q+s]) I[0<=q+s<N] - h[q-s] I[0<=q-s<N]
    
    # Build delta array of length 2N-1
    delta = np.zeros(2*N-1)
    
    # Contribution from p going 1->0
    # -(1-h[p+s]) for s where 0<=p+s<N, i.e., -p <= s < N-p
    lo = max(-p, -(N-1))
    hi = min(N-1-p, N-1)  # p+s < N means s < N-p
    for s in range(lo, hi+1):
        delta[s + N - 1] -= (1.0 - h[p + s])
    
    # +h[p-s] for s where 0<=p-s<N, i.e., p-N < s <= p
    lo = max(-N+1, -(N-1))
    hi = min(p, N-1)
    for s in range(lo, hi+1):
        delta[s + N - 1] += h[p - s]
    
    # Contribution from q going 0->1
    # +(1-h[q+s]) for s where 0<=q+s<N
    lo = max(-q, -(N-1))
    hi = min(N-1-q, N-1)
    for s in range(lo, hi+1):
        delta[s + N - 1] += (1.0 - h[q + s])
    
    # -h[q-s] for s where 0<=q-s<N
    lo = max(-N+1, -(N-1))
    hi = min(q, N-1)
    for s in range(lo, hi+1):
        delta[s + N - 1] -= h[q - s]
    
    return delta
```

```python
                if 0 <= i + s < N:
                    grad[i] += (1.0 - y[i + s])
                if 0 <= i - s < N:
                    grad[i] -= y[i - s]
        
        grad /= k_peaks

        hx_new = y - lr * grad
        hx_new = np.clip(hx_new, 0.0, 1.0)
        # Fix sum to target
        s = hx_new.sum()
        if abs(s - target) > 1e-10:
            # Adjust: shift all values by (target - s)/N, then clip
            adj = (target - s) / N
            hx_new = np.clip(hx_new + adj, 0.0, 1.0)
            if abs(hx_new.sum() - target) > 0.5:
                # Rough projection
                hx_new = np.clip(hx_new + (target - hx_new.sum()) / N, 0.0, 1.0)
        
        m = beta * (hx_new - hx)
        hx = hx_new
        it += 1
        
        if it % 50 == 0:
            c_bin = true_c5(project_binary(hx, N))
            if c_bin < best_c:
                best_c = c_bin
                best_hx = hx.copy()
            lr *= 0.99
            beta = min(0.95, 0.9 + it * 0.0002)
    
    return project_binary(best_hx, N), best_c
```

Let me also think about what makes this problem hard. The key challenge is that C5 = max_k corr[k], and we want to minimize this max. The optimal solution should have the top peaks all at roughly the same level (like a "waterfilling" solution). If one peak is much higher than others, we can probably reduce it at the cost of slightly increasing others.

A good strategy:
1. Start with the initial construction
2. Do aggressive greedy binary search to find a good local minimum
3. Use Nesterov on the continuous relaxation to smooth out
4. Binary cleanup
5. Repeat with different random seeds/perturbations

Let me also consider: the previous code used FFT circular correlation, which is different from the true linear correlation. This means the previous optimization was optimizing the wrong objective! The "improvement" it found might not actually be better under the true metric.

Let me verify: for N=380, the circular correlation has indices 0 to 2N-2, but wraps around. The linear correlation also has 2N-1 values but the boundary effects are different.

For the circular correlation: corr_circ[k] = sum_i h[i] * (1-h[(i+k) mod N])
For the linear correlation: corr_lin[k] = sum_i h[i] * (1-h[i+k-(N-1)]) where indices must be valid

The difference is at the boundaries. For the center (k=N-1, shift=0), they're the same. For large shifts, they differ.

This is a significant bug in the previous code! The optimizer was minimizing circular correlation, but the evaluator uses linear correlation. This could explain why the previous result (0.380893) wasn't as good as it could be.

```python
def true_c5(h):

    dx = 2.0 / N

    return float(corr.max()) * dx

def project_binary(h, N):
    target = N // 2
    h = np.where(h >= 0.5, 1.0, 0.0).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:n1 - target]] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - n1]] = 1.0
    return h

def compute_delta_matrices(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    idx = np.arange(N)[:, None]
    sh = shifts[None, :]
    ps = idx + sh
    pm = idx - sh
    ms = (ps >= 0) & (ps < N)
    mm = (pm >= 0) & (pm < N)
    ps_c = np.clip(ps, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    d10 = -(1.0 - h[ps_c]) * ms.astype(float) + h[pm_c] * mm.astype(float)
    d01 = (1.0 - h[ps_c]) * ms.astype(float) - h[pm_c] * mm.astype(float)
    return d10, d01

def nesterov_continuous(h_init, N, deadline):

    hx = np.clip(hx, 0.0, 1.0)
    # Project sum
```
