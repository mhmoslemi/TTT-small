    return np.real(np.fft.irfft(H * G, L))

def adam_opt(h, N, deadline, k=50, lr0=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        # Get top-k active shifts
        top = np.argpartition(corr, -k)[-k:]
        # Weight by how close to max
        active = corr[top]
        w = ((active - (active.min() - 1e-6)) / (active.max() - active.min() + 1e-6)) ** 2
        w /= w.sum()
        # Compute gradient
        grad = np.zeros(N)
        center = N - 1
        for ai in range(len(top)):
            L = int(top[ai] - center)
            wi = w[ai]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N - L])
                grad[:N - L] -= wi * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - h[M:])
                    grad[M:] -= wi * h[:N - M]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / (1 + t * 0.001)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
        if t % 1000 == 0:
            lr0 *= 0.95
    return best_h, best_pk * dx

    if n1 > nh:

        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum_bin(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def simulated_annealing(h, N, deadline, T0=0.3, T_end=0.005):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum_bin(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    cur_P = best_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    t_start = time.time()
    remaining = deadline - t_start
    n_steps = int(remaining / 5e-4)  # estimate steps per second
    n_steps = max(n_steps, 100)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(n_steps, 1)
        T = T0 * (T_end / T0) ** frac
        pi = np.random.randint(len(ones))
        qi = np.random.randint(len(zeros))
        p = ones[pi]
        q = zeros[qi]
        h[p], h[q] = 0.0, 1.0
        new_P = float(np.max(fft_corr(h)))
        delta = new_P - cur_P
        if delta < 0 or (T > 1e-8 and np.random.random() < np.exp(-delta / T)):
            cur_P = new_P
            ones[pi] = q
            zeros[qi] = p
            if cur_P < best_c:
                best_c = cur_P
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
    return best_h, best_c * (2.0 / N)

def adam_opt(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        active = corr[top]
        rng_active = active.max() - active.min() + 1e-12
        w = ((active - active.min()) / rng_active) ** 2
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ai in range(len(top)):
            L = int(top[ai] - center)
            wi = w[ai]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N - L])
                grad[:N - L] -= wi * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - h[M:])
                    grad[M:] -= wi * h[:N - M]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 * 0.9995 ** (t // 100)
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
    return best_h, best_pk * dx

def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum_bin(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum_bin(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)
