        if best_d is not None and best_np < current_pk - 1e-10:

            h[p]=0; h[q]=1; corr = corr + best_d; current_pk = best_np
            if best_np < best_pk:
                best_pk=best_np; best_h=h.copy(); no_imp=0
            else:
                no_imp+=1
        else:
            no_imp+=1
            if no_imp > 3000:
                h = best_h.copy(); corr = fft_corr(h); current_pk = best_pk; no_imp = 0
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(fft_corr(h0)))*(2.0/N0)

    # Phase 1: Continuous LSE optimization (120s)
    d1 = min(t0+120, deadline-600)
    rng_p = np.random.RandomState(seed)
    hc = h0 + rng_p.randn(N0)*0.05
    hc = proj_sum(hc, N0/2.0)
    hC, pkC = continuous_opt(hc, N0, d1, top_k=50, lr0=0.005, beta=50)
    c5C = pkC*(2.0/N0)
    if c5C < best_c5:
        best_c5 = c5C; best_h = hC.copy()

    # Phase 2: Main binary annealing (450s)
    d2 = min(time.time()+450, deadline-200)
    hb = fix_count((best_h>0.5).astype(float), N0, np.random.RandomState(seed+1))
    hB, pkB = anneal_binary(hb, N0, d2, seed=seed)
    c5B = pkB*(2.0/N0)
    if c5B < best_c5:
        best_c5 = c5B; best_h = hB.copy()

    # Phase 3: Multi-candidate binary search (150s)
    if time.time() < deadline-100:
        d3 = min(time.time()+150, deadline-80)
        hb3 = fix_count((best_h>0.5).astype(float), N0, np.random.RandomState(seed+2))
        hM, pkM = multi_candidate_binary(hb3, N0, d3, seed=seed+3, k_cand=10)
        c5M = pkM*(2.0/N0)
        if c5M < best_c5:
            best_c5 = c5M; best_h = hM.copy()

    # Phase 4: Try different N values
    for Nt in [340, 420, 460, 520]:
        if time.time() > deadline-60: break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_count((hi>0.5).astype(float), Nt, np.random.RandomState(Nt))
        sub = min(deadline-10, time.time()+40)
        hrt, pkr = anneal_binary(hbt, Nt, sub, seed=Nt)
        c5r = pkr*(2.0/Nt)
        if c5r < best_c5:
            best_c5 = c5r; best_h = hrt.copy()

    # Phase 5: Final intensive binary polish
    if time.time() < deadline-3:
        hf = fix_count((best_h>0.5).astype(float), len(best_h), np.random.RandomState(seed+99))
        hfp, pkf = anneal_binary(hf, len(best_h), deadline-1, seed=seed+100)
        c5f = pkf*(2.0/len(best_h))
        if c5f < best_c5:
            best_c5 = c5f; best_h = hfp.copy()

    final_c5 = float(np.max(fft_corr(best_h)))*(2.0/len(best_h))
    return best_h, final_c5, len(best_h)
