def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_grad_matrix(h, N):
    grad = np.zeros((N, 2 * N - 1))
    center = N - 1
    grad[:, center] = 1.0 - 2.0 * h
    for L in range(1, N):
        col = center + L
        grad[:N - L, col] += (1.0 - h[L:])
        grad[L:, col] -= h[:N - L]
    for M in range(1, N):
        col = center - M
        grad[:N - M, col] += (1.0 - h[M:])
        grad[M:, col] -= h[:N - M]
    return grad

        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_c = np.inf
    best_h = h.copy()
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr)) * (2.0 / N)
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100 and temp == 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        # Compute gradient matrix (only for needed positions)

        # = max_k((corr[k] + gz[j,k]) - go[i,k])
        diff = corr[None, None, :] + gz[None, :, :] - go[:, None, :]
        newP = diff.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi]) * (2.0 / N)

                q = int(np.random.choice(zeros))

def nesterov_opt(h, N, deadline, k=60, lr0=0.015):

    best_pk = float(np.max(fft_corr(h)))

    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary IL with full gradient matrix (primary)
    d1 = min(deadline - 350, t0 + 220)
    h_bin = (h0 > 0.5).astype(float)
    h_s, _ = binary_il_search(h_bin, N0, d1, max_iters=600, sample_sz=90)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov continuous with jitter restart
    if time.time() < deadline - 280:
        d2 = min(deadline - 240, time.time() + 130)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_01(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=70, lr0=0.01)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Deep binary IL round 2
    if time.time() < deadline - 200:
        d3 = min(deadline - 160, time.time() + 180)
        h3 = (best_h > 0.5).astype(float)
        h_s3, _ = binary_il_search(h3, N0, d3, max_iters=500, sample_sz=110)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Simulated annealing binary (escape local optima)
    if time.time() < deadline - 120:
        d4 = min(deadline - 70, time.time() + 160)
        h4 = (best_h > 0.5).astype(float)
        h_s4, _ = binary_il_search(h4, N0, d4, max_iters=700, sample_sz=80, temp=0.06)
        c5_4 = compute_c5(h_s4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_s4.copy()

    # Phase 5: Second Nesterov from annealed result
    if time.time() < deadline - 80:
        d5 = min(deadline - 50, time.time() + 80)
        h5 = best_h + np.random.randn(N0) * 0.02
        h5 = project_01(h5, N0 / 2.0)
        hn5, c5n5 = nesterov_opt(h5, N0, d5, k=90, lr0=0.008)
        if c5n5 < best_c5:
            best_c5, best_h = c5n5, hn5.copy()

    # Phase 6: Multi-resolution exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [240, 320, 480, 580]:
        if time.time() > deadline - 50:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == N0:
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh_r = N_try // 2
        n1_r = int(h_bt.sum())
        if n1_r > nh_r:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1_r - nh_r, replace=False)] = 0
        elif n1_r < nh_r:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh_r - n1_r, replace=False)] = 1
        sub_dl = min(deadline - 15, time.time() + 50)
        h_rt, _ = binary_il_search(h_bt, N_try, sub_dl, max_iters=350, sample_sz=70)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 7: Final binary polish
    if time.time() < deadline - 3:
        hB = (best_h > 0.5).astype(float)
        hB, _ = binary_il_search(hB, len(hB), deadline - 0.5, max_iters=150, sample_sz=100)
        c5B = compute_c5(hB, len(hB))
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    final_c5 = compute_c5(best_h, len(best_h))
    hB2 = (best_h > 0.5).astype(float)
    final_c52 = compute_c5(hB2, len(best_h))
    if final_c52 < final_c5:
        best_h = hB2
        final_c5 = final_c52
    return best_h, final_c5, len(best_h)
