def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2

    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

def swap_delta(h, N, p, q):
    """Compute delta in correlation when h[p]:1->0, h[q]:0->1"""
    s_arr = np.arange(-(N-1), N)
    ps, pm = p+s_arr, p-s_arr
    qs, qm = q+s_arr, q-s_arr
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d = np.zeros(2*N-1)
    pc=np.clip(ps,0,N-1); pmc=np.clip(pm,0,N-1)
    qsc=np.clip(qs,0,N-1); qmc=np.clip(qm,0,N-1)
    # p goes 1->0: for each lag, the contribution of position p changes
    # At lag s (position p+s in corr), h[p] used to be 1, now 0:
    # OLD: h[p]*(1-h[p+s]) = 1*(1-h[p+s]) = 1-h[p+s] (when p+s valid)
    # NEW: h[p]*(1-h[p+s]) = 0
    # So delta at lag where p is the "first" index: -(1-h[p+s])
    # At lag where p is the "second" index: h[p-s] used to multiply (1-h[p])=0, now (1-h[p])=1
    # OLD: h[p-s]*(1-h[p]) = 0
    # NEW: h[p-s]*1 = h[p-s]
    # So delta at that lag: +h[p-s]
    d[vps] -= (1.0-h[pc[vps]])
    d[vpm] += h[pmc[vpm]]
    # q goes 0->1: 
    # At lag where q is "first": OLD h[q]*(1-h[q+s])=0, NEW 1*(1-h[q+s])=1-h[q+s]
    # delta: +(1-h[q+s])
    # At lag where q is "second": OLD h[q-s]*(1-h[q])=h[q-s], NEW h[q-s]*0=0
    # delta: -h[q-s]
    d[vqs] += (1.0-h[qsc[vqs]])
    d[vqm] -= h[qmc[vqm]]
    return d

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(hb, N):
    target = N // 2
    n1 = int(hb.sum())
    if n1 > target:
        idx = np.where(hb==1)[0]
        hb[idx[np.random.choice(len(idx), n1-target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(hb==0)[0]
        hb[idx[np.random.choice(len(idx), target-n1, replace=False)]] = 1
    return hb

def swap_delta(h, N, p, q):
    s_arr = np.arange(-(N-1), N)
    ps, pm = p+s_arr, p-s_arr
    qs, qm = q+s_arr, q-s_arr
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d = np.zeros(2*N-1)
    d[vps] -= (1.0-h[np.clip(ps,0,N-1)[vps]])
    d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
    d[vqs] += (1.0-h[np.clip(qs,0,N-1)[vqs]])
    d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
    return d

def binary_anneal(h, N, deadline, T0=0.055, cool=0.9994, max_no_imp=350, p2swap=0.25):

    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    no_imp = 0
    T = T0
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    s_arr = np.arange(-(N-1), N)
    while time.time() < deadline and T > 0.0002:
        if p2swap > 0 and np.random.random() < p2swap and len(ones)>=2 and len(zeros)>=2:
            i1,i2 = np.random.randint(len(ones),2)
            j1,j2 = np.random.randint(len(zeros),2)
            if i1!=i2 and j1!=j2:
                p1,p2 = int(ones[i1]), int(ones[i2])
                q1,q2 = int(zeros[j1]), int(zeros[j2])
                d = swap_delta(h,N,p1,q1) + swap_delta(h,N,p2,q2)
                nc = corr + d
                npk = float(nc.max())
                dv = npk - cur_pk
                if dv < 0 or np.random.random() < np.exp(-dv/max(T,1e-8)):
                    h[p1]=0;h[p2]=0;h[q1]=1;h[q2]=1
                    corr=nc;cur_pk=npk
                    ones=np.where(h==1)[0];zeros=np.where(h==0)[0]
                    if npk<best_pk: best_pk=npk;best_h=h.copy();no_imp=0
                    else: no_imp+=1
                    T*=cool;continue
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        d = swap_delta(h, N, p, q)
        nc = corr + d
        npk = float(nc.max())
        dv = npk - cur_pk

def grad_desc(h, N, deadline, top_k=50, lr0=0.003, decay=0.995):
    h = h.astype(float).copy()
    m=np.zeros(N); v=np.zeros(N)
    b1,b2,eps=0.85,0.999,1e-8
    t=0; best_pk=float(fft_corr(h).max()); best_h=h.copy()
    center=N-1; lr=lr0
    while time.time() < deadline:
        corr=fft_corr(h)
        pk=float(corr.max())
        if pk<best_pk: best_pk=pk; best_h=h.copy(); lr*=decay
        idx=np.argpartition(corr,-top_k)[-top_k:]
        grad=np.zeros(N)
        for ai in idx:
            s=int(ai-center)
            if s==0: grad+=(1.0-2.0*h)
            elif s>0:
                grad[:N-s]+=(1.0-h[s:]); grad[s:]-=h[:N-s]
            else:
                s2=-s; grad[s2:] +=(1.0-h[:N-s2]); grad[:N-s2]-=h[s2:]
        grad/=top_k
        t+=1
        m=b1*m+(1-b1)*grad; v=b2*v+(1-b2)*grad*grad
        mh=m/(1-b1**t); vh=v/(1-b2**t)
        h=proj_sum(h-lr*mh/np.sqrt(vh+eps), N/2.0)
    return best_h, best_pk
