    return np.real(np.fft.ifft(H*G))[:2*N-1]

def anneal_binary(h, N, deadline, T0=0.05, cool=0.9993, max_no_improve=1500, pair_prob=0.3):

                T *= cool
                no_imp += 1

            q2 = int(zeros[np.random.randint(len(zeros))])

                T *= cool
                no_imp += 1
                continue

def local_binary_search(h, N, deadline):
    """Exhaustive single-flip and 2-swap local search"""
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    s_arr = np.arange(-(N-1), N)
    improved = True
    
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        
        # Try all single swaps
        for p in ones:
            for q in zeros:
                if p == q:
                    continue
                ps = p + s_arr
                pm = p - s_arr
                qs = q + s_arr
                qm = q - s_arr
                vps = (ps>=0)&(ps<N)
                vpm = (pm>=0)&(pm<N)
                vqs = (qs>=0)&(qs<N)
                vqm = (qm>=0)&(qm<N)
                
                d = np.zeros(2*N-1)
                pc = np.clip(ps, 0, N-1)
                pmc = np.clip(pm, 0, N-1)
                qsc = np.clip(qs, 0, N-1)
                qmc = np.clip(qm, 0, N-1)
                d[vps] -= (1.0-h[pc[vps]])
                d[vpm] += h[pmc[vpm]]
                d[vqs] += (1.0-h[qsc[vqs]])
                d[vqm] -= h[qmc[vqm]]
                
                nc = corr + d
                npk = float(nc.max())
                if npk < best_pk - 1e-12:
                    h[p] = 0; h[q] = 1
                    corr = nc
                    best_pk = npk
                    improved = True
                    break
            if improved:
                break
        if time.time() > deadline:
            break
    
    return h, best_pk

            if delta < 0 or np.random.random() < np.exp(-delta/max(T,1e-8)):
                h[p]=0; h[q]=1; corr=nc; cur_pk=npk
                if npk < best_pk: best_pk=npk; best_h=h.copy(); noimp=0
                else: noimp+=1
            else: noimp+=1
        T *= cool
    return best_h, best_pk

def limited_local(h, N, deadline, max_rounds=5):
    """Try best single swap per round"""
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    sa = np.arange(-(N-1), N)
    for _ in range(max_rounds):
        if time.time() > deadline: break
        ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
        best_delta = 0; best_pq = None
        # Sample a subset for large N
        if len(ones)*len(zeros) > 50000:
            ps = np.random.choice(ones, min(len(ones), 60), replace=False)
            qs = np.random.choice(zeros, min(len(zeros), 60), replace=False)
        else:
            ps = ones; qs = zeros
        d = np.zeros(2*N-1)
        for p in ps:
            for q in qs:
                if p == q: continue
                ps_ = p+sa; pm = p-sa; qs_ = q+sa; qm = q-sa
                vp=(ps_>=0)&(ps_<N); vm=(pm>=0)&(pm<N)
                vq=(qs_>=0)&(qs_<N); vm2=(qm>=0)&(qm<N)
                pc=np.clip(ps_,0,N-1); pmc=np.clip(pm,0,N-1)
                qsc=np.clip(qs_,0,N-1); qmc=np.clip(qm,0,N-1)
                d[:] = 0
                d[vp] -= (1.0-h[pc[vp]]); d[vm] += h[pmc[vm]]
                d[vq] += (1.0-h[qsc[vq]]); d[vm2] -= h[qmc[vm2]]
                nc = corr + d; npk = float(nc.max())
                delta = npk - best_pk
                if delta < best_delta:
                    best_delta = delta; best_pq = (p,q,d)
        if best_pq is not None:
            h[best_pq[0]]=0; h[best_pq[1]]=1
            corr += best_pq[2]; best_pk += best_delta
    return h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    # Phase 1: Continuous soft-max optimization (multiple starts)
    for r in range(4):
        if time.time() > deadline - 350: break
        ns = 0.025 + 0.015*r
        hc = best_h + np.random.randn(N0)*ns
        hc = proj_sum(hc, N0/2.0)
        d = min(deadline-300, time.time()+130)
        hC, pkC = softmax_opt(hc, N0, d, a0=3.0+2*r, a1=50+15*r, lr0=0.005-0.001*r, top_k=50)
        c5C = pkC*(2.0/N0)
        if c5C < best_c5: best_c5=c5C; best_h=hC.copy()
    # Phase 2: Binary annealing
    if time.time() < deadline - 200:
        hb = fix_count((best_h>0.5).astype(float), N0)
        d = min(deadline-150, time.time()+180)
        hB, pkB = anneal_bin(hb, N0, d, T0=0.07, cool=0.9994, max_noimp=2500, pp=0.3)
        c5B = pkB*(2.0/N0)
        if c5B < best_c5: best_c5=c5B; best_h=hB.copy()
    # Phase 3: Limited local search
    if time.time() < deadline - 80:
        hf = (best_h>0.5).astype(float)
        d = min(deadline-40, time.time()+60)
        hf, pkf = limited_local(hf, N0, d, max_rounds=8)
        c5f = pkf*(2.0/N0)
        if c5f < best_c5: best_c5=c5f; best_h=hf.copy()
    # Phase 4: Multi-N exploration
    for Nt in [320, 400, 448, 512, 640, 256]:
        if time.time() > deadline - 70: break
        if Nt % 2 != 0: Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_count((hi>0.5).astype(float), Nt)
        if time.time() < deadline - 45:
            hc2 = hbt.astype(float) + np.random.randn(Nt)*0.03
            hc2 = proj_sum(hc2, Nt/2.0)
            d2 = min(deadline-35, time.time()+40)
            hC2, pkC2 = softmax_opt(hc2, Nt, d2, a0=5, a1=35, lr0=0.005, top_k=35)
            c5C2 = pkC2*(2.0/Nt)
            if c5C2 < best_c5: best_c5=c5C2; best_h=hC2.copy()
        if time.time() < deadline - 15:
            src = best_h if len(best_h)==Nt else hi
            hb2 = fix_count((src>0.5).astype(float), Nt)
            d3 = min(deadline-5, time.time()+30)
            hB2, pkB2 = anneal_bin(hb2, Nt, d3, T0=0.04, cool=0.9995, max_noimp=800, pp=0.2)
            c5B2 = pkB2*(2.0/Nt)
            if c5B2 < best_c5: best_c5=c5B2; best_h=hB2.copy()
    # Phase 5: Final polish
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hf = fix_count((best_h>0.5).astype(float), Nf)
        hfp, pkf2 = anneal_bin(hf, Nf, deadline-1, T0=0.012, cool=0.9997, max_noimp=1200, pp=0.15)
        c5f2 = pkf2*(2.0/Nf)
        if c5f2 < best_c5: best_c5=c5f2; best_h=hfp.copy()
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
