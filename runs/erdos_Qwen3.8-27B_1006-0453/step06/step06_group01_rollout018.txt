import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, target_ones):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > target_ones:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, min(n1 - target_ones, len(ones)), replace=False)
        h[rem] = 0.0
    elif n1 < target_ones:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, min(target_ones - n1, len(zeros)), replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def topk_score(corr, k):
    k = min(k, len(corr))
    top = np.partition(corr, -k)[-k:]
    return float(top[0]) + 0.4 * float(top[1:].sum()) / (k - 1)

def sa_search(h_in, N, deadline, topk=12, T0=2.0, T1=0.05, max_iter=200000):
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    score = topk_score(corr, topk)
    best_score = score
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    cool = (T1 / T0) ** (1.0 / max(1, max_iter))
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        h[p], h[q] = 0.0, 1.0
        new_corr = fft_corr(h)
        new_score = topk_score(new_corr, topk)
        delta = new_score - score
        if delta < 0 or np.log(np.random.uniform(1e-30, 1.0)) < -delta / T:
            score = new_score
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if score < best_score:
                best_score = score
                nc = float(corr.max())
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
            if it % 10000 == 9999:
                h = best_h.copy()
                corr = fft_corr(h)
                score = topk_score(corr, topk)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        else:
            h[p], h[q] = 1.0, 0.0
        T *= cool
    return best_h, best_c

def ils_search(h_in, N, deadline, n_restarts=6, iters_per=80000, topk=12):
    best_h = fix_binary(h_in.copy(), N // 2)
    best_c = compute_c5(best_h, N)
    per_budget = max(5, int((deadline - time.time()) / n_restarts))
    for r in range(n_restarts):
        sub_dl = min(deadline - 2, time.time() + per_budget)
        if time.time() > sub_dl - 1:
            break
        pert_n = 3 + r * 2
        hp = best_h.copy()
        ones = np.where(hp == 1)[0]
        zeros = np.where(hp == 0)[0]
        n_flip = min(pert_n, len(ones) // 2, len(zeros) // 2)
        if n_flip > 0:
            fi = np.random.choice(ones, n_flip, replace=False)
            fj = np.random.choice(zeros, n_flip, replace=False)
            hp[fi], hp[fj] = 0.0, 1.0
        hl, cl = sa_search(hp, N, sub_dl, topk=topk, T0=1.5 / (1 + 0.3 * r),
                           T1=0.03, max_iter=iters_per)
        cl5 = compute_c5(hl, N)
        if cl5 < best_c:
            best_c = cl5
            best_h = hl
    return best_h, best_c

def greedy_polish(h_in, N, deadline, n_cand=40, max_sweeps=8000):
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stuck = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        n_p = min(n_cand, len(ones))
        p_list = np.random.choice(ones, n_p, replace=False)
        improved = False
        for p in p_list:
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            new_corr = fft_corr(h)
            nc = float(new_corr.max())
            if nc < best_c - 1e-10:
                best_c = nc
                best_h = h.copy()
                corr = new_corr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
                stuck = 0
            else:
                h[p], h[q] = 1.0, 0.0
        if not improved:
            stuck += 1
            if stuck > 50:
                break
    return best_h, best_c * (2.0 / N)

def perturb_for_new_N(h_src, N_old, N_new):
    x_old = np.arange(N_old, dtype=float)
    x_new = np.arange(N_new, dtype=float) * (N_old / N_new)
    h_interp = np.interp(x_new, x_old, h_src)
    return (h_interp > 0.5).astype(float)
