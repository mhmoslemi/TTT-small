import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(y, target):
    y = np.clip(y, 0.0, 1.0)
    s = float(y.sum())
    if abs(s - target) < 1e-12:
        return y
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_m = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s_m > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, N):
    hb = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(hb.sum())
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        to_zero = idx[np.argsort(h[idx])[:n1 - nh]]
        hb[to_zero] = 0
    elif n1 < nh:
        idx = np.where(hb == 0)[0]
        to_one = idx[np.argsort(-h[idx])[:nh - n1]]
        hb[to_one] = 1
    return hb

def greedy_search(h, N, deadline, max_iter=3000, stall_limit=120):
    h = h.astype(float).copy()
    nh = N // 2
    if int(round(h.sum())) != nh:
        h = binarize(h, N)
    corr = fft_corr(h)
    best_c = float(corr.max())
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        P = float(corr.max())
        if P + 1e-14 < best_c:
            best_c = P
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        shifts = np.arange(2 * N - 1)
        idx_m = np.arange(N)[:, None] - shifts[None, :] + (N - 1)
        valid = (idx_m >= 0) & (idx_m < N)
        idx_cl = np.clip(idx_m, 0, N - 1)
        D = (1.0 - h[idx_cl]) * valid
        best_d, best_pq = 0.0, None
        for p in ones:
            base = corr - D[p]
            cand = base[None, :] - D[zeros]
            nm = cand.max(axis=1)
            mi = int(np.argmin(nm))
            d = float(nm[mi]) - P
            if d < best_d:
                best_d, best_pq = d, (int(p), int(zeros[mi]))
        if best_pq is not None and best_d < -1e-12:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            corr = fft_corr(h)
        else:
            break
    return h, best_c

def subgrad_dir(h, N, corr, tau=0.8):
    max_c = float(corr.max())
    w = np.exp((corr - max_c) / tau)
    ws = float(w.sum())
    if ws < 1e-15:
        w = np.zeros(2 * N - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= ws
    shifts = np.arange(2 * N - 1) - (N - 1)
    grad = np.zeros(N)
    active = np.where(w > 1e-7)[0]
    for k in active:
        s = shifts[k]
        i_s = max(0, -s)
        i_e = min(N, N - s)
        if i_e <= i_s:
            continue
        n_v = i_e - i_s
        grad[i_s:i_e] += w[k]
        grad[i_s:i_e] -= w[k] * h[i_s + s:i_e + s]
        if s > 0:
            i_ms = max(i_s, s)
            if i_ms < i_e:
                grad[i_ms:i_e] -= w[k] * h[i_ms - s:i_e - s]
        else:
            grad[i_s:i_e] -= w[k] * h[i_s - s:i_e - s]
    return grad

def subgrad_optimize(h, N, deadline, lr0=0.04, tau=0.8):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    t = 0
    while time.time() < deadline:

        corr = fft_corr(h)
        c5 = float(corr.max()) * (2.0 / N)

        grad = subgrad_dir(h, N, corr, tau)

            hb = binarize(h, N)
            h = 0.5 * h + 0.5 * hb
            h = project_box_sum(h, target)
    return best_h, best_c5
