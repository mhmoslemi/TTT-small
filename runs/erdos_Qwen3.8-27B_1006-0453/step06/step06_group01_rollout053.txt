def adam_opt(h_in, N, deadline, lr=0.01, k=50):
    h = h_in.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    beta1_t, beta2_t = 1.0, 1.0
    best_c = compute_c5(h, N)
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N-1))
            if L == 0:
                grad += 1.0 - 2.0*h
            elif L > 0:
                grad[L:] += 1.0 - h[:N-L]
                grad[:N-L] -= h[L:]
            else:
                M = -L
                grad[:N-M] += 1.0 - h[M:]
                grad[M:] -= h[:N-M]
        grad /= len(top)
        it += 1
        beta1_t *= beta1
        beta2_t *= beta2
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad**2
        m_hat = m / (1-beta1_t)
        v_hat = v / (1-beta2_t)
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 200 == 0:
            lr *= 0.95
    return best_h, best_c

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, target):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > target:
        order = np.argsort(h)[::-1]
        for i in range(min(n1 - target, len(order))):
            h[order[i]] = 0.0
    elif n1 < target:
        order = np.argsort(h)
        for i in range(min(target - n1, len(order))):
            h[order[i]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        h[ones[:n1 - target]] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - n1]] = 1.0
    return h

def build_deltas(s, N):
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return deltas

def update_deltas(deltas, s, p, q, N):
    s[p] = -s[p]
    s[q] = -s[q]
    rows = np.arange(N)
    deltas[rows, rows + p] -= 2.0
    deltas[rows, rows + q] += 2.0

def full_neighbor_search(h_in, N, deadline, max_sweeps=20000, pert_every=45, chunk=25):
    h = make_binary(h_in, N // 2)
    corr = fft_corr(h)
    s = 2.0 * h - 1.0
    deltas = build_deltas(s, N)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        n1, n0 = len(ones), len(zeros)
        if n1 < 1 or n0 < 1:
            break
        dp = deltas[ones]
        dq = deltas[zeros]
        best_new = 1e10
        best_pi, best_qi = 0, 0
        for ci in range(0, n1, chunk):
            dc = min(ci + chunk, n1)
            vals = corr[None, None, :] + dp[ci:dc, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                best_pi, best_qi = mi[0] + ci, mi[1]
        if best_new >= best_c - 1e-12:
            no_improve += 1
            if no_improve >= pert_every:
                n_pert = min(4, n1, n0)
                rp = np.random.choice(ones, n_pert, replace=False)
                rq = np.random.choice(zeros, n_pert, replace=False)
                for ii in range(n_pert):
                    pp, qq = int(rp[ii]), int(rq[ii])
                    h[pp], h[qq] = 0.0, 1.0
                    corr += deltas[pp].copy() - deltas[qq].copy()
                    update_deltas(deltas, s, pp, qq, N)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                no_improve = 0
                continue
            if no_improve > pert_every * 4:
                break
            best_pi = np.random.randint(n1)
            best_qi = np.random.randint(n0)
        no_improve = 0
        p = int(ones[best_pi])
        q = int(zeros[best_qi])
        h[p], h[q] = 0.0, 1.0
        corr += deltas[p].copy() - deltas[q].copy()
        update_deltas(deltas, s, p, q, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def adam_continuous(h_in, N, deadline, lr=0.018, k=45):
    h = project_sum(h_in.astype(float).copy(), N / 2.0)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    b1_t, b2_t = 1.0, 1.0
    best_c = compute_c5(h, N)
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * h
            elif L > 0:
                grad[L:] += 1.0 - h[:N - L]
                grad[:N - L] -= h[L:]
            else:
                M = -L
                grad[:N - M] += 1.0 - h[M:]
                grad[M:] -= h[:N - M]
        grad /= len(top)
        it += 1
        b1_t *= b1
        b2_t *= b2
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1_t)
        v_hat = v / (1 - b2_t)
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), N / 2.0)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 400 == 0:
            lr *= 0.92
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Full neighborhood binary search
    d1 = min(deadline - 450, t0 + 420)
    h1, _ = full_neighbor_search(h0, N0, d1, max_sweeps=25000, pert_every=50, chunk=30)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Continuous Adam on 380
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + 150)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hn, c2 = adam_continuous(hc, N0, d2, lr=0.015, k=50)
        if c2 < best_c5:
            best_c5, best_h = c2, hn.copy()

    # Phase 3: Binary search from continuous refinement
    if time.time() < deadline - 180:
        d3 = min(deadline - 120, time.time() + 200)
        h3, _ = full_neighbor_search(best_h, N0, d3, max_sweeps=20000, pert_every=40, chunk=30)
        c3 = compute_c5(h3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Try higher resolution
    for N_try in [400, 420]:
        if time.time() > deadline - 100:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bt = make_binary(h_interp, N_try // 2)
        sub_dl = min(deadline - 40, time.time() + 55)
        h_rt, _ = full_neighbor_search(h_bt, N_try, sub_dl, max_sweeps=10000, pert_every=35, chunk=25)
        c_rt = compute_c5(h_rt, N_try)
        if c_rt < best_c5:
            best_c5, best_h = c_rt, h_rt.copy()

    # Phase 5: Second continuous pass
    if time.time() < deadline - 40:
        d5 = min(deadline - 10, time.time() + 50)
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.02
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cF = adam_continuous(hF, len(best_h), d5, lr=0.008, k=60)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    # Phase 6: Final binary polish
    if time.time() < deadline - 3:
        d6 = deadline - 1
        h6, _ = full_neighbor_search(best_h, len(best_h), d6, max_sweeps=10000, pert_every=30, chunk=35)
        c6 = compute_c5(h6, len(best_h))
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
