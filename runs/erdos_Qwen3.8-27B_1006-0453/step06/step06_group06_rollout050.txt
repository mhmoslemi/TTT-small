def swap_delta(h, N, p, q):
    center = N - 1
    s_arr = np.arange(-(N-1), N)  # shape (2N-1,)
    d = np.zeros(2*N-1)
    
    # p goes 1->0
    n1 = p - s_arr  # for this s, h[p] contributes to (1-h[p-s]) term if p-s is valid

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(np.conj(H) * G, L)[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h_arr, N):
    target = N // 2
    h_arr = h_arr.astype(float).copy()
    n1 = int(round(h_arr.sum()))
    if n1 > target:
        idx = np.where(h_arr >= 0.5)[0]
        if len(idx) > 0:
            perm = np.random.permutation(len(idx))
            h_arr[idx[perm[:min(n1 - target, len(idx))]]] = 0.0
    elif n1 < target:
        idx = np.where(h_arr < 0.5)[0]
        need = target - n1
        if len(idx) >= need and need > 0:
            perm = np.random.permutation(len(idx))
            h_arr[idx[perm[:need]]] = 1.0
    return h_arr

def swap_delta(h, N, p, q):
    center = N - 1
    s_arr = np.arange(-(N - 1), N)
    d = np.zeros(2 * N - 1)
    n1 = p - s_arr
    m1 = (n1 >= 0) & (n1 < N)
    d[m1] -= (1.0 - h[n1[m1]])
    n2 = p + s_arr
    m2 = (n2 >= 0) & (n2 < N)
    d[m2] += h[n2[m2]]
    n3 = q - s_arr
    m3 = (n3 >= 0) & (n3 < N)
    d[m3] += (1.0 - h[n3[m3]])
    n4 = q + s_arr
    m4 = (n4 >= 0) & (n4 < N)
    d[m4] -= h[n4[m4]]
    return d

def anneal_binary(h, N, deadline, T0=0.05, cool=0.9996, max_stall=2500, reheat_T=None):
    h = fix_count(h.copy(), N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    T = T0
    if reheat_T is None:
        reheat_T = T0 * 0.4
    stall = 0
    while time.time() < deadline and T > 0.0004:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for _ in range(4):
            if time.time() >= deadline:
                break
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            if p == q:
                continue
            d = swap_delta(h, N, p, q)
            nc = corr + d
            npk = float(nc.max())
            delta = npk - cur_pk
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                h[p] = 0.0
                h[q] = 1.0
                corr = nc
                cur_pk = npk
                if npk < best_pk - 1e-12:
                    best_pk = npk
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        T *= cool
        if stall > max_stall:
            stall = 0
            h = best_h.copy()
            corr = fft_corr(h)
            cur_pk = best_pk
            T = reheat_T
    return best_h, best_pk

def continuous_opt(h, N, deadline, top_k=40, lr0=0.004):
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    center = N - 1
    lr = lr0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        top_vals = corr[idx]
        w = np.exp((top_vals - top_vals.max()) * 50.0)
        w /= w.sum()
        grad = np.zeros(N)
        for ai, wi in zip(idx, w):
            s = int(ai - center)
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N - s] += wi * (1.0 - h[s:])
                grad[s:] -= wi * h[:N - s]
            else:
                s2 = -s
                grad[s2:] += wi * (1.0 - h[:N - s2])
                grad[:N - s2] -= wi * h[s2:]
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = proj_sum(h - lr * mh / (np.sqrt(vh) + eps), N / 2.0)
        if t % 400 == 0:
            lr *= 0.96
    return best_h, best_pk

def interpolate_to(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
