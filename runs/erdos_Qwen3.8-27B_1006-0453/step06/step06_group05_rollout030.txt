import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def compute_grad(h, N, T):
    c = fft_corr(h)
    cs = c - c.max()
    p = np.exp(cs / T)
    p /= p.sum()
    m = N - 1
    j_idx = np.arange(N)[:, None]
    k_idx = np.arange(2*N-1)[None, :]
    idx1 = j_idx + k_idx - m
    idx2 = j_idx - k_idx + m
    v1 = (idx1 >= 0) & (idx1 < N)
    v2 = (idx2 >= 0) & (idx2 < N)
    i1c = np.clip(idx1, 0, N-1)
    i2c = np.clip(idx2, 0, N-1)
    t1 = (p[None,:] * (1.0 - h[i1c]) * v1).sum(axis=1)
    t2 = (p[None,:] * h[i2c] * v2).sum(axis=1)
    return (t1 - t2), c

def optimize_cont(h, N, deadline, T=3.0, lr=0.06, max_iter=8000):
    h = h.astype(float).copy()
    tgt = N / 2.0
    bh, bc = h.copy(), compute_c5(h, N)
    mu1 = np.zeros(N)
    mu2 = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    for t in range(1, max_iter+1):
        if time.time() > deadline:
            break
        g, c = compute_grad(h, N, T)
        c5 = float(c.max()) * (2.0/N)
        if c5 < bc - 1e-12:
            bc, bh = c5, h.copy()
        bc1 = 1.0 - b1**t
        bc2 = 1.0 - b2**t
        mu1 = b1*mu1 + (1-b1)*g
        mu2 = b2*mu2 + (1-b2)*g*g
        step = lr * (mu1/bc1) / (np.sqrt(mu2/bc2) + eps)
        h = project_sum(h - step, tgt)
        if t % 400 == 0:
            lr *= 0.96
    return bh, bc

def bin_local(h, N, deadline, max_it=600, anneal=0.0):

    nh = N // 2
    h = (h > 0.5).astype(float)
    ones = np.where(h == 1)[0]
    if len(ones) > nh:
        h[:] = 0.0
        h[ones[:nh]] = 1.0
    elif len(ones) < nh:
        h[:] = 0.0
        h[ones] = 1.0
        z = np.where((h0_ref > 0.5) == 0 & (h == 0))[0] if False else np.where(h == 0)[0]
        h[z[:nh-len(ones)]] = 1.0
    m = N - 1
    bc = float(np.max(fft_corr(h)))
    stall = 0
    T = anneal
    for _ in range(max_it):
        if time.time() > deadline or stall > 180:
            break
        if T > 0:
            T *= 0.9985
        c = fft_corr(h)
        P = float(c.max())
        if P < bc - 1e-10:
            bc, stall = P, 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = ns = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ns_p = min(len(ones), 100)
        ns_q = min(len(zeros), 60)
        ps = ones if ns_p == len(ones) else np.random.choice(ones, ns_p, replace=False)
        qs = zeros if ns_q == len(zeros) else np.random.choice(zeros, ns_q, replace=False)
        bd, bpq = 0.0, None
        for p in ps:
            dp = np.zeros(2*N-1)
            ks = np.arange(max(0, m-p), min(2*N-1, m-p+N))
            if len(ks):
                dp[ks] += h[p+ks-m] - 1.0
            ks2 = np.arange(max(0, m-p), min(2*N-1, 2*m-p+1))
            if len(ks2):
                dp[ks2] += h[p-m+ks2]
            base = c + dp
            for q in qs:
                dq = np.zeros(2*N-1)
                ks3 = np.arange(max(0, m-q), min(2*N-1, m-q+N))
                if len(ks3):
                    dq[ks3] += 1.0 - h[q+ks3-m]
                ks4 = np.arange(max(0, m-q), min(2*N-1, 2*m-q+1))
                if len(ks4):
                    dq[ks4] -= h[q-m+ks4]
                dm = float(np.max(base + dq)) - P
                if dm < bd:
                    bd, bpq = dm, (int(p), int(q))
        if bpq is not None:
            if bd < -1e-10:
                h[bpq[0]] = 0.0; h[bpq[1]] = 1.0
            elif T > 0 and np.random.random() < min(1.0, np.exp(bd/(T+1e-10))):
                h[bpq[0]] = 0.0; h[bpq[1]] = 1.0; stall = 0
        elif stall > 80:
            break
    return h, bc

def threshold_proj(h, N):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    nh = N // 2
    if n1 != nh:
        oi = np.where(hb == 1)[0]
        zi = np.where(hb == 0)[0]
        hb[:] = 0.0
        if n1 > nh:
            order = np.argsort(-h[oi])
            hb[oi[order[:nh]]] = 1.0
        else:
            hb[oi] = 1.0
            order = np.argsort(-h[zi])
            hb[zi[order[:nh-n1]]] = 1.0
    return hb
