import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_subgrad(h, N, k):
    """Subgradient of corr(k) w.r.t. h"""
    g = np.zeros(N)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0 and k < N:
        g[:N-k] += (1.0 - h[k:])
        g[k:] -= h[:N-k]
    elif k < 0 and k > -N:
        m = -k
        g[m:] += (1.0 - h[:N-m])
        g[:N-m] -= h[m:]
    return g

def softmax_opt(h, N, deadline, beta_start=5.0, beta_end=200.0, lr0=0.05, target=None):
    """Soft-max gradient descent with annealing."""
    if target is None:
        target = N / 2.0
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    total_time = max(1, deadline - time.time())
    # Simple linear annealing
    it = 0
    lr = lr0
    while time.time() < deadline:
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (total_time * 2))
        corr = fft_corr(h)
        # Soft-max weights
        mc = corr.max()
        weights = np.exp(beta * (corr - mc))
        weights /= weights.sum()
        # Weighted gradient
        g = np.zeros(N)
        center = N - 1
        for idx in range(len(corr)):
            if weights[idx] > 1e-10:
                k = idx - center
                g += weights[idx] * compute_subgrad(h, N, k)
        g /= weights.sum()
        h_new = project_sum(h - lr * g, target)
        c_val = float(np.max(fft_corr(h_new)))
        if c_val < best_c:
            best_c = c_val
            best_h = h_new.copy()
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.95
    return best_h, best_c * (2.0/N)

            g[:N - s] += w * (1.0 - h[s:])
            g[s:] -= w * h[:N - s]
        else:
            m = -s
            g[m:] += w * (1.0 - h[:N - m])
            g[:N - m] -= w * h[m:]
    return g

def softmax_descent(h, N, deadline, beta0=2.0, beta1=300.0, lr0=0.06):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    lr = lr0
    it = 0
    est_iters = max(1, int((deadline - time.time()) * 12))
    while time.time() < deadline:
        beta = beta0 + (beta1 - beta0) * min(1.0, float(it) / est_iters)
        corr = fft_corr(h)
        cm = corr.max()
        expo = np.exp(np.clip(beta * (corr - cm), -50, 0))
        expo[expo > 1e8] = 1e8
        w = expo / expo.sum()
        g = weighted_grad(h, N, w)
        h = project_sum(h - lr * g, target)
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        it += 1
        if it % 60 == 0:
            lr *= 0.975
    return best_h, best_pk * (2.0 / N)

def binary_local(h, N, deadline, max_sweeps=500, sample_p=100):
    h = (h > 0.5).astype(float).copy()
    n1 = int(h.sum())
    nh = N // 2
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.random.choice(len(idx), n1 - nh, replace=False)]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        add = idx[np.random.choice(len(idx), nh - n1, replace=False)]
        h[add] = 1.0
    best_pk = float(np.max(fft_corr(h)))
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_pk - 1e-12:
            best_pk = P
            no_imp = 0
        else:
            no_imp += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        n_samp = min(sample_p, len(ones))
        perm = np.random.permutation(len(ones))[:n_samp]
        p_list = ones[perm]
        best_score = P + 1.0
        best_move = None
        z_arr = D[zeros]
        for p in p_list:
            base = corr - D[p]
            cand = z_arr + base[None, :]
            peaks = cand.max(axis=1)
            mi = int(np.argmin(peaks))
            if peaks[mi] < best_score:
                best_score = float(peaks[mi])
                best_move = (int(p), int(zeros[mi]))
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
        else:
            p = int(ones[np.random.randint(len(ones))])
            z = int(zeros[np.random.randint(len(zeros))])
            h[p] = 0.0
            h[z] = 1.0
        if no_imp > 250:
            break
    return h, best_pk * (2.0 / N)

def fix_binary_sum(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.random.choice(len(idx), n1 - nh, replace=False)]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        add = idx[np.random.choice(len(idx), nh - n1, replace=False)]
        h[add] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Soft-max descent (continuous)
    d1 = t0 + 180
    if time.time() < d1:
        hc = best_h.copy() + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hs, c5s = softmax_descent(hc, N0, d1, beta0=2.0, beta1=200.0, lr0=0.06)
        if c5s < best_c5:
            best_c5, best_h = c5s, hs

    # Phase 2: Binary local search
    d2 = t0 + 370
    if time.time() < d2:
        hb = (best_h > 0.5).astype(float)
        hsb, c5b = binary_local(hb, N0, d2, max_sweeps=500, sample_p=110)
        if c5b < best_c5:
            best_c5, best_h = c5b, hsb

    # Phase 3: Multistart binary
    d3 = t0 + 560
    if time.time() < d3:
        per = max(8.0, (d3 - time.time()) / 6.0)
        for trial in range(6):
            tt = time.time() + per
            if tt > d3:
                break
            hr = np.random.randint(0, 2, N0).astype(float)
            hr = fix_binary_sum(hr, N0)
            hsr, c5r = binary_local(hr, N0, tt, max_sweeps=150, sample_p=80)
            if c5r < best_c5:
                best_c5, best_h = c5r, hsr

    # Phase 4: Soft-max refine from best binary
    d4 = t0 + 730
    if time.time() < d4:
        hc2 = best_h.copy() + np.random.randn(N0) * 0.03
        hc2 = project_sum(hc2, N0 / 2.0)
        hs2, c5s2 = softmax_descent(hc2, N0, d4, beta0=5.0, beta1=400.0, lr0=0.04)
        if c5s2 < best_c5:
            best_c5, best_h = c5s2, hs2

    # Phase 5: Binary local on refined
    d5 = t0 + 880
    if time.time() < d5:
        hb2 = (best_h > 0.5).astype(float)
        hsb2, c5b2 = binary_local(hb2, N0, d5, max_sweeps=400, sample_p=120)
        if c5b2 < best_c5:
            best_c5, best_h = c5b2, hsb2

    # Phase 6: Upscale to 2N
    N_up = N0 * 2
    if time.time() < deadline - 40:
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_up = (h_interp > 0.5).astype(float)
        h_up = fix_binary_sum(h_up, N_up)
        ht, c5t = binary_local(h_up, N_up, deadline, max_sweeps=200, sample_p=70)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
