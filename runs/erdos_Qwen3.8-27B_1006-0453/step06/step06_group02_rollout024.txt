def softmax_nesterov(h, N, deadline, t_start=5, t_end=200, n_steps=2000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = 0.9
    lr = 0.01
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Adaptive temperature
        frac = step / n_steps
        t = t_start * (t_end / t_start) ** frac
        # Or: t = t_start + (t_end - t_start) * frac**2
        
        y = h + m
        grad = softmax_grad(y, N, t)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        
        if step % 10 == 0:
            c = float(np.max(fft_corr(h)))
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
        # Decay learning rate
        lr *= 0.999
        # Adapt beta
        beta = min(0.97, 0.85 + step * 0.0005)
    
    return best_h, best_c * (2.0 / N)

            grad[M:] += w * (1.0 - h[:N - M])
            grad[:N - M] -= w * h[M:]
    
    return obj, grad

```python
def compute_D_binary(h, N):
    """D[p, k] = influence of flipping h[p] on correlation at lag k, for binary h."""
    D = np.zeros((N, 2*N-1))
    pos = np.arange(N)
    shifts = np.arange(2*N-1) - (N-1)  # L = -(N-1), ..., 0, ..., N-1
    # For each shift L:
    # D[p, L] = [h[p+L]=0 and 0<=p+L<N] - [h[p-L]=1 and 0<=p-L<N]
    for L in range(2*N-1):
        l_val = L - (N-1)
        if l_val == 0:
            D[:, L] = 1 - 2*h  # (1-h[p]) - h[p] = 1-2h[p]
        elif l_val > 0:
            # h[p+l_val] valid for p < N-l_val
            # h[p-l_val] valid for p >= l_val
            idx1 = np.arange(N - l_val)
            D[idx1, L] = 1 - h[l_val:]  # (1-h[p+L]) for valid p
            idx2 = np.arange(l_val, N)
            D[idx2, L] -= h[:N - l_val]  # -h[p-L] for valid p
        else:
            M = -l_val
            # h[p-M] valid for p >= M
            # h[p+M] valid for p < N-M
            idx1 = np.arange(M, N)
            D[idx1, L] = 1 - h[:N - M]  # (1-h[p+L]) for valid p
            idx2 = np.arange(N - M)
            D[idx2, L] -= h[M:]  # -h[p-L] for valid p
    return D
```

This is O(N² * avg_shift) which for N=380 is about 380*759 = 288K operations (vectorized per shift). Should be fast.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D_binary(h, N):
    D = np.zeros((N, 2 * N - 1))
    for L in range(2 * N - 1):
        l = L - (N - 1)
        if l == 0:
            D[:, L] = 1.0 - 2.0 * h
        elif l > 0:
            D[:N - l, L] = 1.0 - h[l:]
            D[l:, L] -= h[:N - l]
        else:
            M = -l
            D[M:, L] = 1.0 - h[:N - M]
            D[:N - M, L] -= h[M:]
    return D

def binary_local_search(h, N, deadline, max_sweeps=5000):

    h = h.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    no_improve = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D_binary(h, N)
        # For each one p, find best zero q to swap
        best_score = best_c + 1.0
        best_pq = None
        # Vectorize: for each p in ones, compute new max for all q in zeros
        # new_corr[k] = corr[k] - D[p,k] + D[q,k]
        # Score = max_k new_corr[k]
        
        # To reduce computation, sample if too many
        if len(ones) > 200:
            perm = np.random.permutation(len(ones))[:200]
            p_list = ones[perm]
        else:
            p_list = ones
        
        for p in p_list:
            # For all zeros q: change = D[q,:] - D[p,:]
            # new_corr = corr + change
            changes = D[zeros, :] - D[p, :][None, :]  # shape (n_zeros, 2N-1)
            new_maxes = (corr[None, :] + changes).max(axis=1)  # shape (n_zeros,)
            mi = int(np.argmin(new_maxes))
            score = float(new_maxes[mi])
            if score < best_score - 1e-12:
                best_score = score
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            # No improvement: try 2-for-2 swap or random move
            if np.random.random() < 0.3 or no_improve > 100:
                # 2-for-2: pick 2 ones and 2 zeros
                if len(ones) >= 2 and len(zeros) >= 2:
                    p1, p2 = np.random.choice(ones, 2, replace=False)
                    q1, q2 = np.random.choice(zeros, 2, replace=False)
                    # Check if this reduces max
                    change = -D[p1] - D[p2] + D[q1] + D[q2]
                    new_max = float((corr + change).max())
                    if new_max < best_c - 0.5:

                        corr = fft_corr(h)

                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        no_improve = 0
                        continue
            # Random single swap
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
        else:
            p, q = best_pq
            h[p] = 0.0
            h[q] = 1.0
            corr = fft_corr(h)
        
        new_c = float(corr.max())
        if new_c < best_c - 1e-12:
            best_c = new_c
            no_improve = 0
        else:
            no_improve += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if no_improve > 200:
            break
    
    return h, best_c * (2.0 / N)
