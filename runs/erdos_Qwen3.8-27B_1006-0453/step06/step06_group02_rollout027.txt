    return np.real(np.fft.ifft(H * G))

def compute_D(h, N):

    shifts = np.arange(L) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv

    vm = (pm >= 0) & (pm < N)

    pm_c = np.clip(pm, 0, N - 1)

    hm = np.where(vm, h[pm_c], 0.0)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 > 0.5).astype(float)
    best_h = fix_binary(best_h, N0)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Intensive binary search
    d1 = min(deadline - 550, t0 + 280)
    h1, _ = binary_phase(best_h, N0, d1, max_sweeps=20000, sample_q=130, pw=0.025)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Nesterov with softmax
    if time.time() < deadline - 420:
        d2 = min(deadline - 400, time.time() + 160)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.06
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_phase(hc, N0, d2, T0=0.012, lr0=0.013)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary from Nesterov
    if time.time() < deadline - 320:
        d3 = min(deadline - 300, time.time() + 220)
        h3 = (best_h > 0.5).astype(float)
        h3 = fix_binary(h3, N0)
        h3, _ = binary_phase(h3, N0, d3, max_sweeps=20000, sample_q=140, pw=0.02)
        c3 = compute_c5(h3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Upscale to 2x and binary search
    N_up = N0 * 2
    if time.time() < deadline - 180:
        x_best = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_up = (h_interp > 0.5).astype(float)
        h_up = fix_binary(h_up, N_up)
        d4 = min(deadline - 140, time.time() + 260)
        h4, _ = binary_phase(h_up, N_up, d4, max_sweeps=20000, sample_q=110, pw=0.02)
        c4 = compute_c5(h4, N_up)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Nesterov polish at best resolution
    Nf = len(best_h)
    if time.time() < deadline - 40:
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.03
        hF = project_sum(hF, Nf / 2.0)
        d5 = min(deadline - 8, time.time() + 130)
        hF, c5F = nesterov_phase(hF, Nf, d5, T0=0.008, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Phase 6: Final binary polish
    if time.time() < deadline - 3:
        hB = (best_h > 0.5).astype(float)
        hB = fix_binary(hB, len(hB))
        hB, _ = binary_phase(hB, len(hB), deadline - 0.5, max_sweeps=2000, sample_q=120, pw=0.015)
        c6 = compute_c5(hB, len(hB))
        if c6 < best_c5:
            best_c5, best_h = c6, hB

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
