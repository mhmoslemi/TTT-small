import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * np.conj(G), L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(80):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        mid = (h > 1e-9) & (h < 1.0 - 1e-9)
        nm = int(mid.sum())
        if nm == 0:
            if diff > 0:
                idx = np.argsort(-h)[:int(np.ceil(diff))]
                h[idx] = 1.0
            else:
                idx = np.argsort(h)[:int(np.ceil(-diff))]
                h[idx] = 0.0
            break
        h[mid] += diff / nm
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad(h, corr, N, n_act):
    idx = np.argsort(corr)[-n_act:]
    g = np.zeros(N)
    for k in idx:
        L = int(k) - (N - 1)
        if L == 0:
            g += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += (1.0 - h[:N - L])
            g[:N - L] -= h[L:]
        else:
            M = -L
            g[:N - M] += (1.0 - h[M:])
            g[M:] -= h[:N - M]
    g /= len(idx)
    return g

    target = N / 2.0
    corr = fft_corr(h)
    best_val = float(corr.max()) * (2.0 / N)
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    stall = 0
    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        val_y = float(corr_y.max()) * (2.0 / N)
        if val_y < best_val - 1e-10:
            best_val = val_y
            best_h = h.copy()
            stall = 0

        it += 1

        if cf < best_c5:
            best_c5, best_h = cf, hf

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
