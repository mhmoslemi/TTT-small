import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def eff_matrix(positions, h, N):
    ka = np.arange(2 * N - 1) - (N - 1)
    pos = np.asarray(positions)
    idx1 = pos[:, None] + ka[None, :]
    idx2 = pos[:, None] - ka[None, :]
    m1 = (idx1 >= 0) & (idx1 < N)
    m2 = (idx2 >= 0) & (idx2 < N)
    E = np.zeros((len(pos), 2 * N - 1))
    E += np.where(m1, h[np.clip(idx1, 0, N - 1)], 0.0)
    E += np.where(m2, h[np.clip(idx2, 0, N - 1)], 0.0)
    return E

def binary_il(h, N, deadline, max_iters=10000, sample_sz=70, temp=0.0, cool=0.9995, use_2swap=0.25):
    h = (h > 0.5).astype(float)
    tgt = N // 2
    s = int(h.sum())
    if s > tgt:
        h[np.random.choice(np.where(h == 1)[0], s - tgt, replace=False)] = 0
    elif s < tgt:
        h[np.random.choice(np.where(h == 0)[0], tgt - s, replace=False)] = 1
    best_c = np.inf
    best_h = h.copy()
    T = temp
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        pk = float(np.max(corr))
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Eo = eff_matrix(os_, h, N)
        Ez = eff_matrix(zs, h, N)
        new_c = corr[None, None, :] + Eo[:, None, :] - Ez[None, :, :]
        new_pk = new_c.max(axis=2)
        mi = np.unravel_index(np.argmin(new_pk), new_pk.shape)
        mn = float(new_pk[mi])
        accept = mn < pk - 1e-12
        action = (os_[mi[1]], zs_[mi[0]], '1for1')
        if not accept and temp > 0:
            if np.random.random() < np.exp((pk - mn) / max(T, 1e-6)):
                accept = True
            T *= cool
        if not accept and use_2swap > 0 and np.random.random() < use_2swap:
            if ns >= 2 and nz >= 2:
                i1, i2 = np.random.choice(ns, 2, replace=False)
                j1, j2 = np.random.choice(nz, 2, replace=False)
                d2 = (Eo[i1] + Eo[i2] - Ez[j1] - Ez[j2])
                mk2 = float(np.max(corr + d2))
                if mk2 < pk - 1e-12:
                    h[os_[i1]] = 0; h[os_[i2]] = 0
                    h[zs[j1]] = 1; h[zs[j2]] = 1
                    continue
        if accept:
            p, q = action[0], action[1]
            h[p] = 0
            h[q] = 1
        else:
            break
    return best_h, best_c

def cont_opt(h, N, deadline, lr0=0.4, mom=0.85, topk=3):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros(N)
    lr = lr0
    center = N - 1
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        top = np.argsort(corr)[-topk:][::-1]
        kst = int(np.random.choice(top))
        m = kst - center
        g = np.ones(N)
        idx1 = np.arange(N) + m
        m1 = (idx1 >= 0) & (idx1 < N)
        g[m1] = 1.0 - h[idx1[m1]]
        idx2 = np.arange(N) - m
        m2 = (idx2 >= 0) & (idx2 < N)
        g[m2] -= h[idx2[m2]]
        v = mom * v - lr * g
        h = project_sum(h + v, N / 2.0)
        it += 1
        if it % 50 == 0:
            c_now = compute_c5(h)
            if c_now < best_c:

            lr *= 0.999
    c_f = compute_c5(h)
    if c_f < best_c:

    return best_h, best_c
