import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0-h, L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0, 1)

def fix_count(hb, N):
    target = N // 2
    n1 = int(hb.sum())
    if n1 > target:
        idx = np.where(hb==1)[0]
        hb[idx[np.random.choice(len(idx), n1-target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(hb==0)[0]
        hb[idx[np.random.choice(len(idx), target-n1, replace=False)]] = 1
    return hb

def _continuous(h, N, deadline, best_h, best_c5, top_k=40, lr0=0.004):
    center = N - 1
    mb = np.zeros(N); vb = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0; lr = lr0
    best_pk = float(np.max(fft_corr(h)))
    best_hc = h.copy()
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_hc = h.copy()
        idx = np.argpartition(-corr, top_k)[:top_k]
        N_ = N
        grad = np.zeros(N_)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0*h)
            elif s > 0:
                grad[:N_-s] += (1.0 - h[s:])
                grad[s:] -= h[:N_-s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N_-s2])
                grad[:N_-s2] -= h[s2:]
        grad /= top_k
        t += 1
        mb = b1*mb + (1-b1)*grad
        vb = b2*vb + (1-b2)*grad*grad
        mh = mb/(1-b1**t); vh = vb/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        if t % 300 == 0: lr *= 0.97
    c5 = best_pk * (2.0/N)
    if c5 < best_c5:
        return best_hc.copy(), c5
    return best_h, best_c5

def _anneal(h, N, deadline, T0=0.06, cool=0.9993):
    h = fix_count(h.copy(), N)
    corr = fft_corr(h)
    best_pk = float(corr.max()); best_h = h.copy(); cur_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    T = T0; no_imp = 0
    ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
    while time.time() < deadline and T > 0.0005:
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        # Delta computation
        ps = p + s_arr; pm = p - s_arr
        qs = q + s_arr; qm = q - s_arr
        vps = (ps>=0)&(ps<N); vpm = (pm>=0)&(pm<N)
        vqs = (qs>=0)&(qs<N); vqm = (qm>=0)&(qm<N)
        d = np.zeros(2*N-1)
        d[vps] -= (1.0 - h[np.clip(ps,0,N-1)[vps]])
        d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
        d[vqs] += (1.0 - h[np.clip(qs,0,N-1)[vqs]])
        d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
        ncorr = corr + d
        npk = float(ncorr.max())
        delta = npk - cur_pk
        if delta < 0 or np.random.random() < np.exp(-delta/max(T,1e-8)):
            h[p]=0; h[q]=1; corr=ncorr; cur_pk=npk
            ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
            if npk < best_pk: best_pk=npk; best_h=h.copy(); no_imp=0
            else: no_imp+=1
        else:
            no_imp+=1
        T *= cool
        if no_imp > 1000:
            h=best_h.copy(); corr=fft_corr(h); cur_pk=best_pk
            ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
            no_imp=0; T=max(T0*0.3, T*0.5)
