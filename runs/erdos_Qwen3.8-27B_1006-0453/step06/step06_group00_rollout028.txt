    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return -1, -1, np.inf
    best_nm = np.inf
    best_i, best_j = -1, -1
    for i in ones:
        nm = np.max(corr + profs[i][None, :] - profs[zeros], axis=1)
        ji = np.argmin(nm)
        if nm[ji] < best_nm:
            best_nm = float(nm[ji])
            best_i, best_j = int(i), int(zeros[ji])
    return best_i, best_j, best_nm

def steepest_descent(h, N, deadline, max_iters=5000):
    h = h.astype(float).copy()
    corr = fft_corr(h).copy()
    P = float(np.max(corr))
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        profs = build_profs(h, N)
        i, j, new_P = best_swap(h, corr, profs)
        if i < 0 or new_P >= P - 1e-12:
            break
        h[i] = 0
        h[j] = 1
        corr += profs[i] - profs[j]
        P = new_P
    return h, P * (2.0 / N)

def anneal_search(h, N, deadline, t0_temp=0.0015):
    h = h.astype(float).copy()
    corr = fft_corr(h).copy()
    P = float(np.max(corr))
    best_P = P
    best_h = h.copy()
    temp = t0_temp
    stall = 0
    iter_cnt = 0
    while time.time() < deadline:
        profs = build_profs(h, N)
        i, j, new_P = best_swap(h, corr, profs)
        if i < 0:
            break
        delta = new_P - P
        accept = delta < -1e-12
        if not accept and temp > 1e-6:
            if delta > 0.5:
                p_acc = 1e-8
            else:
                p_acc = np.exp(-delta / temp)
            accept = np.random.random() < p_acc
        if accept:
            h[i] = 0
            h[j] = 1
            corr += profs[i] - profs[j]
            P = new_P
            iter_cnt += 1
            if P < best_P:
                best_P = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 80:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            n_kicks = min(stall // 40, 5)
            for _ in range(n_kicks):
                ki = int(np.random.choice(ones))
                kj = int(np.random.choice(zeros))
                h[ki] = 0
                h[kj] = 1
                corr += build_profs(h, N)[ki] - build_profs(h, N)[kj]
                h[ki] = h[ki]  # already set
            profs = build_profs(h, N)
            corr = fft_corr(h).copy()
            P = float(np.max(corr))
            if P < best_P:
                best_P = P
                best_h = h.copy()
            stall = 0
            temp = min(temp * 2.0, t0_temp)
        temp = max(temp * 0.9997, 1e-5)
    c5_best = best_P * (2.0 / N)
    c5_cur = P * (2.0 / N)
    if c5_cur < c5_best:
        return h, c5_cur
    return best_h, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h = (h0 > 0.5).astype(float)
    h = fix_sum(h, N0 // 2)

    best_h = h.copy()
    best_c5 = compute_c5(h, N0)

    # Phase 1: Steepest descent on initial
    d1 = min(deadline - 450, t0 + 100)
    h_s, c5_s = steepest_descent(h, N0, d1, max_iters=3000)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Annealing (main search)
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + 350)
        h_a, c5_a = anneal_search(best_h, N0, d2, t0_temp=0.002)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a.copy()

    # Phase 3: Second anneal from perturbed start
    if time.time() < deadline - 150:
        d3 = min(deadline - 120, time.time() + 180)
        h_p = best_h.copy()
        ones = np.where(h_p == 1)[0]
        zeros = np.where(h_p == 0)[0]
        n_pert = max(3, len(ones) // 40)
        pi = np.random.choice(ones, n_pert, replace=False)
        pj = np.random.choice(zeros, n_pert, replace=False)
        h_p[pi] = 0
        h_p[pj] = 1
        h_p2, c5_p = steepest_descent(h_p, N0, time.time() + 30, max_iters=500)
        if time.time() < d3 - 10:
            h_a2, c5_a2 = anneal_search(h_p2, N0, d3, t0_temp=0.001)
            if c5_a2 < best_c5:
                best_c5, best_h = c5_a2, h_a2.copy()
        elif c5_p < best_c5:
            best_c5, best_h = c5_p, h_p2.copy()

    # Phase 4: Multi-N exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [360, 400, 420, 440, 480, 520, 560]:
        if time.time() > deadline - 60:
            break
        if N_try < 4 or N_try % 2 != 0:
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_t = (h_int > 0.45).astype(float)
        h_t = fix_sum(h_t, N_try // 2)
        # Quick steepest + short anneal
        d_sub = min(deadline - 10, time.time() + 60)
        h_sd, c5_sd = steepest_descent(h_t, N_try, min(d_sub - 15, time.time() + 25), max_iters=1000)
        if c5_sd < best_c5:
            best_c5, best_h = c5_sd, h_sd.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)
        elif time.time() < d_sub - 10:
            h_an, c5_an = anneal_search(h_sd, N_try, d_sub, t0_temp=0.001)
            if c5_an < best_c5:
                best_c5, best_h = c5_an, h_an.copy()
                x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 5: Final steepest polish
    if time.time() < deadline - 3:
        h_f, c5_f = steepest_descent(best_h, len(best_h), deadline - 1, max_iters=2000)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    # Ensure valid
    final_c5 = compute_c5(best_h, len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
