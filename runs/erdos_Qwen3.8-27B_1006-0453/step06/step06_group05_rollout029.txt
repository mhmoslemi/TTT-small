def lp_step(h, N, deadline):

    D = compute_D(h, N)
    dx = 2.0 / N
    
    dh = cp.Variable(N)
    t = cp.Variable()
    
    constraints = [
        cp.sum(dh) == 0,
        dh >= -h,
        dh <= 1 - h,
    ]
    for k in range(2*N-1):
        constraints.append(c[k] + D[:, k] @ dh <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)  # or CBC, GLPK, etc.
    
    if dh.value is None:
        return h, float(c.max())
    
    # Line search
    best_h = h.copy()
    best_c = float(c.max())

        c_try = float(np.max(fft_corr(h_try)))
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

        for p in ones:

            cand = base[None, :] - D[zeros, :]

def lp_optimize(h, N, deadline, max_iters=200, all_k=False, topK=30, trust=0.25):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    for it in range(max_iters):
        if time.time() > deadline: break
        c = fft_corr(h); P = float(c.max())
        if P < best_pk:
            best_pk = P; best_h = h.copy()
        if all_k:
            k_sel = np.arange(2 * N - 1)
        else:
            k_sel = np.argsort(c)[::-1][:topK]
        D = compute_D(h, N)
        dh_v = cp.Variable(N)
        t_v = cp.Variable()
        cons = [
            cp.sum(dh_v) == 0,
            cp.abs(dh_v) <= trust,
            dh_v >= -h + 1e-6,
            dh_v <= 1.0 - h - 1e-6,
        ]
        for k in k_sel:
            cons.append(D[:, k] @ dh_v - t_v <= -c[k])
        prob = cp.Problem(cp.Minimize(t_v), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
        except Exception:
            try: prob.solve(solver=cp.ECOS, max_iters=10000)
            except Exception: break
        if dh_v.value is None: break
        step = np.clip(dh_v.value, -trust, trust)
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_sum(h + alpha * step, target)
            c_try = float(np.max(fft_corr(h_try)))
            if c_try < P - 1e-9:
                h = h_try; improved = True; break
        if not improved: break
    return best_h, best_pk

def subgrad_optimize(h, N, deadline, lr0=0.03, topK=8):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N); v = np.zeros(N)
    t = 0; lr = lr0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h); pk = float(c.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        top = np.argpartition(c, -topK)[-topK:]
        w = np.exp((c[top] - c[top].max()) / 0.05)
        w /= w.sum()
        D = compute_D(h, N)
        grad = (D[:, top] * w[None, :]).sum(axis=1) * dx
        grad -= grad.mean()
        m = 0.8 * m + 0.2 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        h = project_sum(h - lr * m / (np.sqrt(v) + 1e-8), target)
        if t % 200 == 0:
            lr *= 0.9
            hb = (h > 0.5).astype(float)

                oi = np.where(hb == 1)[0]; zi = np.where(hb == 0)[0]

                    hb[oi[np.argsort(h[oi])[:-(n1 - N//2)]]] = 0
                else:
                    hb[zi[np.argsort(-h[zi])[:(N//2 - n1)]]] = 1
                h = 0.5 * h + 0.5 * hb
    return best_h, best_pk * dx

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    return np.where(vp, h[pp_c], 0.0) * vp.astype(float) - (1.0 - np.where(vm, h[pm_c], 0.0)) * vm.astype(float)
