def sa_binary_search(h_init, N, deadline, max_iters=500000):

    best_h = h.copy()
    
    ones_mask = (h == 1)
    zeros_mask = (h == 0)

    zeros_list = np.where(zeros_mask)[0].tolist()
    
    T = 0.02
    T_min = 0.001
    cool_rate = 0.99995
    use_double = 0.3
    
    for it in range(max_iters):

                if q1 == q2:

                ones_list.remove(p2); ones_list.append(q2)
                zeros_list.remove(q1); zeros_list.append(p1)

                zeros_list.remove(q2); zeros_list.append(p2)
            else:

                h[p2] = 1; h[q2] = 0

            q = int(np.random.choice(zeros_list))

def compute_c5(h, N):
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def to_binary(h, n1):
    h = h.copy().astype(float)
    cur = int(np.round(h.sum()))
    if cur > n1:
        idx = np.where(h >= 0.5)[0]
        rem = np.random.choice(idx, cur - n1, replace=False)
        h[rem] = 0.0
    elif cur < n1:
        idx = np.where(h < 0.5)[0]
        add = np.random.choice(idx, n1 - cur, replace=False)
        h[add] = 1.0
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    return h

            if p1 == p2: continue

def continuous_refine(h_init, N, deadline, lr=0.02, n_top=10):
    h = h_init.astype(float).copy()
    target = N / 2.0
    # Project to [0,1] with sum constraint
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    h += diff / N
    h = np.clip(h, 0, 1)
    # Fine-tune sum
    while abs(h.sum() - target) > 1e-10:
        diff = target - h.sum()
        if diff > 0:
            room = 1.0 - h
            if room.sum() < diff: break
            idx = np.where(room > 0)[0]
            n_add = min(len(idx), int(np.ceil(diff)))
            add_idx = np.random.choice(idx, n_add, replace=False)
            h[add_idx] = np.minimum(1.0, h[add_idx] + diff / n_add)
        else:
            room = h
            if room.sum() < -diff: break
            idx = np.where(room > 0)[0]
            n_sub = min(len(idx), int(np.ceil(-diff)))
            sub_idx = np.random.choice(idx, n_sub, replace=False)
            h[sub_idx] = np.maximum(0.0, h[sub_idx] + diff / n_sub)
    
    best_mc = np.max(np.correlate(h, 1.0 - h, mode="full"))
    best_h = h.copy()
    
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argpartition(corr, -n_top)[-n_top:]
        
        # Subgradient: for each active constraint k, the gradient w.r.t. h is
        # grad[i] += (1-h[i-shift_k]) - h[i+shift_k] (approximately)
        grad = np.zeros(N)
        for k in top_k:
            s = k - (N - 1)
            if s == 0:
                grad += (1 - 2 * h)
            elif s > 0:
                grad[s:] += (1 - h[:N-s])
                grad[:N-s] -= h[s:]
            else:
                m = -s
                grad[:N-m] += (1 - h[m:])
                grad[m:] -= h[:N-m]
        grad /= len(top_k)
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        diff = target - h_new.sum()
        h_new += diff / N
        h_new = np.clip(h_new, 0, 1)
        
        mc = np.max(np.correlate(h_new, 1.0 - h_new, mode="full"))
        if mc < best_mc - 1e-12:
            best_mc = mc; best_h = h_new.copy()
            h = h_new
        elif mc < best_mc + 0.1:
            # Accept with probability (soft acceptance)
            if np.random.random() < np.exp(-(mc - best_mc) / 0.5):
                h = h_new
            else:
                h = h.copy()
        else:
            h = h.copy()
        
        it += 1
        if it % 500 == 0:
            lr *= 0.95
    
    return best_h, best_mc
