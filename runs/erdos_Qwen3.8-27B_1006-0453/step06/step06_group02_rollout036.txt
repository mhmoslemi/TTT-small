    Actually for the swap: if h[p]=1 goes to 0, effect is -D1[p,j] where D1 is 0->1 effect
    If h[q]=0 goes to 1, effect is +D1[q,j]

      wait, for lag -M: sum_{n=0}^{N-1-M} (1-h[n])*h[n+M]

      h[i] in (1-h)-slot: i = n, valid if i <= N-1-M: coefficient -h[i+M]

    center = N - 1

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    L_total = 2 * N - 1
    center = N - 1
    ii = np.arange(N)[:, None]
    lag = (np.arange(L_total) - center)[None, :]
    M = np.abs(lag)
    m_pos = (lag >= 0)
    m_neg = (lag < 0)
    p1 = ii + lag
    n1 = ii - lag
    v1 = np.where(m_pos & (p1 >= 0) & (p1 < N), 1.0 - h[np.clip(p1, 0, N-1)], 0.0)
    v2 = np.where(m_pos & (n1 >= 0) & (n1 < N), -h[np.clip(n1, 0, N-1)], 0.0)
    v3 = np.where(m_neg & (ii >= M) & (ii - M < N), 1.0 - h[np.clip(ii - M, 0, N-1)], 0.0)
    v4 = np.where(m_neg & (ii + M <= N - 1), -h[np.clip(ii + M, 0, N-1)], 0.0)
    return v1 + v2 + v3 + v4

def binary_opt(h, N, deadline, max_sweeps=5000, sp=200, sq=250, k_top=3):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_imp = 0
    tabu_q = []
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        c5 = P * 2.0 / N
        if c5 < best_c5 - 5e-7:
            best_c5 = c5
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 500:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        p_list = ones[np.random.permutation(len(ones))[:min(sp, len(ones))]]
        tset = set(tabu_q)
        zc = zeros[~np.isin(zeros, np.array(list(tset), dtype=int))] if tset else zeros
        if len(zc) < 10:
            zc = zeros
        q_list = zc[np.random.permutation(len(zc))[:min(sq, len(zc))]]
        best_new = P + 1.0
        best_pq = None
        for p in p_list:
            base = corr - D[p, :]
            nm = (base[None, :] + D[q_list, :]).max(axis=1)
            mi = int(np.argmin(nm))
            if nm[mi] < best_new:
                nc = base + D[q_list[mi], :]
                sk = float(np.sum(np.sort(nc)[-k_top:]))
                if sk < best_new:
                    best_new = sk
                    best_pq = (int(p), int(q_list[mi]))
        if best_pq is None:
            if no_imp > 100 and np.random.random() < 0.5:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
            elif no_imp > 350:
                break
            else:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
        else:
            p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        tabu_q.append(q)
        if len(tabu_q) > 250:
            tabu_q.pop(0)
    return best_h, best_c5

def lse_nesterov(h, N, deadline, k_peaks=30, lr0=0.025, beta0=0.88, burn=200):
    h = h.astype(float).copy()
    target = float(N) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline and it < 4000:
        y = np.clip(h + m, 0.0, 1.0)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        c5 = pk * 2.0 / N
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        top = np.argpartition(corr, -k_peaks)[-k_peaks:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N-L] += (1.0 - y[L:])
                grad[L:] += (-y[:N-L])
            else:
                M = -L
                grad[M:] += (1.0 - y[:N-M])
                grad[:N-M] += (-y[M:])
        grad /= k_peaks
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it == burn:
            lr *= 0.5
            beta = 0.92
        elif it % 60 == 0:
            lr *= 0.992
            beta = min(0.96, beta + 0.0003)
    return best_h, best_c5

def fix_binary(h, N):
    h = (h >= 0.5).astype(float)
    n1 = int(h.sum())
    tg = N // 2
    if n1 > tg:
        idx = np.where(h == 1)[0]
        h[idx[np.random.choice(len(idx), n1 - tg, replace=False)]] = 0.0
    elif n1 < tg:
        idx = np.where(h == 0)[0]
        h[idx[np.random.choice(len(idx), tg - n1, replace=False)]] = 1.0
    return h
