def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_step(h, y, corr, k_focus, temp, lr, target):
    N = len(h)
    center = N - 1
    # Soft-max weights
    top = np.argpartition(corr, -k_focus)[-k_focus:]
    max_c = corr[top].max()
    w = np.exp((corr[top] - max_c) / temp)
    w /= w.sum()
    
    # Compute weighted gradient
    grad = np.zeros(N)
    for wi, ai in zip(w, top):
        L = int(round(ai - center))
        if L == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - y[:seg])
            grad[:seg] -= wi * y[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:seg]
    
    h_new = project_sum(y - lr * grad, target)
    return h_new

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_for_lags(y, lags, N):
    grad = np.zeros(N)
    center = N - 1
    for L in lags:
        L = int(round(L))
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0 and L < N:
            seg = N - L
            grad[L:] += (1.0 - y[:seg])
            grad[:seg] -= y[L:]
        elif L < 0 and L > -N:
            M = -L
            seg = N - M
            grad[:seg] += (1.0 - y[M:])
            grad[M:] -= y[:seg]
    return grad

def continuous_opt(h0, N, deadline, k0=10, k1=60, lr0=0.03, temp0=0.004, temp_min=0.0003):

def binary_polish(h0, N, deadline, n_tries=15):

    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - nh]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h[idx] = 1
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sweep_int = int(c.max())
        best_sweep_h = None
        best_sweep_c = None
        # Try random swaps
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = fft_corr(h)
            ct_int = int(c_t.max())
            if ct_int < best_sweep_int:
                best_sweep_int = ct_int
                best_sweep_h = h.copy()
                best_sweep_c = c_t.copy()
            h[p] = 1
            h[q] = 0
        # Also try 2-swap moves
        if len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(n_tries // 2):
                i1 = np.random.randint(len(ones))
                i2 = np.random.randint(len(ones))
                if i1 == i2:

                j1 = np.random.randint(len(zeros))
                j2 = np.random.randint(len(zeros))

                    continue
                p1, p2 = ones[i1], ones[i2]
                q1, q2 = zeros[j1], zeros[j2]
                h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                c_t = fft_corr(h)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    beta = beta0
    temp = temp0
    stall = 0
    it = 0
    center = N - 1

        y = h + beta * (h - h_prev)
