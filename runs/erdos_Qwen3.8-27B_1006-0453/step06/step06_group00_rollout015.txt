def compute_eff_1_to_0(p, h, N):
    """Effect vector when h[p] changes from 1 to 0."""
    Nfull = 2*N - 1
    eff = np.zeros(Nfull)
    center = N - 1
    # Positive lags (index center+L, L=1..N-1)
    for L in range(1, N):
        idx = center + L
        d = 0.0
        if p <= N-1-L:
            d -= (1.0 - h[p+L])
        if p >= L:
            d += h[p-L]

        d = 0.0
        if p >= L:
            d -= (1.0 - h[p-L])
        if p+L < N:
            d += h[p+L]
        eff[idx] = d
    return eff
