import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    # Make L a power of 2 for FFT efficiency
    L2 = 1
    while L2 < L:
        L2 *= 2
    H = np.fft.fft(h, L2)
    G = np.fft.fft(np.flip(1.0 - h), L2)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h, N):
    dx = 2.0 / N
    return float(np.max(fft_corr(h))) * dx

def project_sum(h, target):
    """Project h to [0,1]^N with sum = target using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_mid = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_c5(h, N, top_k=30):
    """Compute subgradient of max_k corr[k] w.r.t. h."""
    corr = fft_corr(h)
    # Find top-k lags
    k = min(top_k, len(corr))
    top_idx = np.argpartition(corr, -k)[-k:]
    # Weight by softmax
    temp = max((corr[top_idx].max() - corr[top_idx].min()) * 0.1, 1e-8)
    w = np.exp((corr[top_idx] - corr[top_idx].max()) / temp)
    w /= w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for wi, ai in zip(w, top_idx):
        L = int(ai) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            # corr at lag L = sum_{i=L}^{N-1} h[i](1-h[i-L])
            # d/dh[i]: for i >= L: (1-h[i-L]); for i < N-L: -h[i+L]
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        elif L < 0 and L > -N:
            M = -L  # M > 0
            seg = N - M
            # corr at lag -M = sum_{i=0}^{N-1-M} h[i](1-h[i+M])
            # d/dh[i]: for i <= N-1-M: (1-h[i+M]); for i >= M: -h[i-M]
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad, corr

def continuous_opt(h0, N, deadline, k=30, lr0=0.02, temp0=0.002):

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_grad(h, T):
    """Gradient of log-sum-exp smooth max of corr components."""
    corr = fft_corr(h)
    N = len(h)
    cm = corr.max()
    exp_c = np.exp(T * (corr - cm))
    w = exp_c / exp_c.sum()
    obj = (1.0 / T) * np.log(exp_c.sum()) + cm
    grad = np.zeros(N)
    center = N - 1
    active = w > 1e-7
    idx_a = np.where(active)[0]
    for wi, ai in zip(w[idx_a], idx_a):
        L = int(ai) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad, obj, cm * (2.0 / N)

        T = T_start * (T_end / T_start) ** frac
        grad, obj, pk_c5 = smooth_max_grad(h, T)
        m = beta * m - lr * grad

            stall = 0
