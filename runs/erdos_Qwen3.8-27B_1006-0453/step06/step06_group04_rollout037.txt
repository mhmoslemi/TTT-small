    """Compute gradient of sum(weights[idx] * corr[idx]) w.r.t. h"""
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        L = idx - center
        if L == 0:

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(corr, lam):
    """Compute softmax weights and smoothed max value."""
    shifted = corr - corr.max()
    exps = np.exp(lam * shifted)
    total = exps.sum()
    weights = exps / total
    smooth_val = (1.0/lam) * (np.log(total) + lam * corr.max())
    return weights, smooth_val

def compute_grad_full(h, weights):
    """Full gradient of sum_k weights[k] * corr[k] w.r.t. h."""
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-12:
            continue
        L = int(idx - center)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[:seg] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:seg]
        else:
            M = -L
            seg = N - M
            grad[:seg] -= w * h[:seg]
            grad[M:] += w * (1.0 - h[:seg])
    return grad

def adam_opt(h0, N, deadline, lam_start=20.0, lam_end=200.0, 
             lr0=0.05, k_top=80):
    """Adam optimization with log-sum-exp smoothing."""
    target = N / 2.0
    h = h0.astype(float).copy()
    h = project_sum(h, target)
    
    m = np.zeros(N)  # Adam first moment
    v = np.zeros(N)  # Adam second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    stall = 0
    stall_count = 0
    
    while time.time() < deadline:
        t += 1
        # Adaptive lambda
        frac = min(t / max(t, 1), 1.0)  # will be refined
        lam = lam_start + (lam_end - lam_start) * min(1.0, t / 300.0)
        
        # Compute correlation and weights
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Log-sum-exp weights
        shifted = corr - corr.max()
        exps = np.exp(lam * shifted)
        total = exps.sum()
        weights = exps / total
        
        # Compute gradient
        # For efficiency, only use top k lags
        top_idx = np.argpartition(weights, -k_top)[-k_top:]
        w_top = np.zeros(len(weights))
        w_top[top_idx] = weights[top_idx]
        w_top /= w_top.sum()
        
        grad = compute_grad_full(h, w_top)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr = lr0 / np.sqrt(t + 1)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        
        # Periodic perturbation for exploration
        if stall > 300:
            h = best_h + np.random.randn(N) * (0.05 + 0.05 * stall / 300.0)
            h = project_sum(h, target)
            m = np.zeros(N)
            v = np.zeros(N)
            stall = 0
            stall_count += 1
            if stall_count > 5:
                break
        
        # Decay learning rate periodically
        if t % 500 == 0:
            lr0 *= 0.9
    
    return best_h, best_pk * (2.0 / N)

def binary_refine(h0, N, deadline, n_tries=20):
    """Binary hill-climbing refinement."""
    h = (h0 > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        idx = idx[np.random.permutation(len(idx))][:n1 - nh]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        idx = idx[np.random.permutation(len(idx))][:nh - n1]
        h[idx] = 1
    
    c = fft_corr(h)
    best_int = int(np.round(c.max()))
    best_h = h.copy()
    stall = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        improved = False
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = fft_corr(h)
            mt = int(np.round(c_t.max()))
            if mt < best_int:
                best_int = mt
                best_h = h.copy()
                improved = True
            elif mt > int(np.round(c.max())):
                h[p] = 1
                h[q] = 0
            else:
                h[p] = 1
                h[q] = 0
        c = fft_corr(h)
        if not improved:
            stall += 1
            if stall > 100:
                # Double flip kick
                if len(ones) >= 2 and len(zeros) >= 2:
                    i1 = np.random.choice(len(ones), 2, replace=False)
                    i2 = np.random.choice(len(zeros), 2, replace=False)
                    h[ones[i1]] = 0
                    h[zeros[i2]] = 1
                stall = 0
        else:
            stall = 0
    
    return best_h, best_int * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_weights(corr, lam):
    shifted = corr - corr.max()
    exps = np.exp(lam * shifted)
    total = exps.sum()
    return exps / total

def compute_grad(h, weights):
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    for idx in range(len(weights)):
        w = weights[idx]
        if abs(w) < 1e-14:
            continue
        L = int(idx - center)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[:seg] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:seg]
        else:
            M = -L
            seg = N - M
            grad[:seg] -= w * h[M:]
            grad[M:] += w * (1.0 - h[:seg])
    return grad

def adam_opt(h0, N, deadline, lam_start=15.0, lam_end=200.0,
             lr0=0.04, k_top=80, perturb_period=300):
    target = N / 2.0
    h = project_sum(h0.astype(float).copy(), target)
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    stall = 0
    
    while time.time() < deadline:
        t += 1
        lam = lam_start + (lam_end - lam_start) * min(1.0, t / 250.0)
        
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        weights = smooth_weights(corr, lam)
        # Use top-k for efficiency
        if k_top < len(weights):
            top_idx = np.argpartition(weights, -k_top)[-k_top:]
            w_sparse = np.zeros_like(weights)
            w_sparse[top_idx] = weights[top_idx]
            w_sparse /= w_sparse.sum()
        else:
            w_sparse = weights
        
        grad = compute_grad(h, w_sparse)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr = lr0 / np.sqrt(t + 1)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        
        if stall > perturb_period:
            h = best_h + np.random.randn(N) * (0.03 + 0.03 * min(stall / perturb_period, 3.0))
            h = project_sum(h, target)
            m[:] = 0
            v[:] = 0
            stall = 0
    
    return best_h, best_pk * (2.0 / N)

def binary_refine(h0, N, deadline, n_tries=20):

        if len(ones) < 1 or len(zeros) < 1:

        cur_max = float(c.max())
        for _ in range(n_tries):

            mt = float(c_t.max())

                    c = fft_corr(h)
