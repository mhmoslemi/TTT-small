import time

def compute_c5(h, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_vec(h, corr, N, top_k, temp):
    center = N - 1
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[top_idx]
    w = np.exp((top_vals - top_vals.max()) / max(temp, 1e-10))
    w /= w.sum()
    grad = np.zeros(N)
    for wi, idx in zip(w, top_idx):
        M = int(abs(idx - center))
        if M == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif M < N:
            seg = N - M
            grad[M:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[M:]
    return grad

def binary_opt(h0, N, deadline, n_tries=20, kick_freq=50):
    h = (h0 > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            idx = np.random.choice(np.where(h == 1)[0], n1 - target, replace=False)
            h[idx] = 0
        else:
            idx = np.random.choice(np.where(h == 0)[0], target - n1, replace=False)
            h[idx] = 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_max = int(np.max(corr))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sweep_max = int(np.max(corr))
        best_sweep_h = None
        best_sweep_corr = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = np.correlate(h, 1.0 - h, mode="full")
            m_t = int(np.max(c_t))
            if m_t < best_sweep_max:
                best_sweep_max = m_t
                best_sweep_h = h.copy()
                best_sweep_corr = c_t.copy()
            h[p] = 1
            h[q] = 0
        if best_sweep_h is not None:
            h = best_sweep_h
            corr = best_sweep_corr
            stall = 0
            if best_sweep_max < best_max:
                best_max = best_sweep_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > kick_freq:
                # Multi-swap kick
                k = min(3, len(ones), len(zeros))
                ps = np.random.choice(ones, k, replace=False)
                qs = np.random.choice(zeros, k, replace=False)
                h[ps] = 0
                h[qs] = 1
                corr = np.correlate(h, 1.0 - h, mode="full")
                stall = 0
    return best_h, best_max * (2.0 / N)

def continuous_opt(h0, N, deadline, k0=30, lr0=0.02, temp0=0.003):
    h = h0.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_pk = float(np.max(corr))
    best_h = h.copy()
    h_mom = h.copy()
    lr = lr0
    temp = temp0
    top_k = k0
    stall = 0
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        grad = compute_grad_vec(h, corr, N, top_k, temp)
        # Nesterov-like momentum
        h_new = h + 0.9 * (h - h_mom) - lr * grad
        h_mom = h
        h = project_sum(h_new, target)
        
        it += 1
        if it % 100 == 0:
            lr = max(0.005, lr * 0.997)
            temp = max(0.0005, temp * 0.995)
            top_k = min(80, top_k + 1)
        
        if stall > 300:
            # Restart with structured perturbation
            h = best_h + np.random.randn(N) * 0.03 * (1 + grad.std() / (1e-10 + np.abs(grad).mean()))
            h = project_sum(h, target)
            h_mom = h.copy()
            lr = max(lr, lr0 * 0.8)
            temp = min(0.01, temp * 2)
            stall = 0
            it = 0
    
    h = project_sum(best_h, target)
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def compute_c5(h, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_vec(h, corr, N, top_k, temp):
    center = N - 1
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[top_idx]
    w = np.exp((top_vals - top_vals.max()) / max(temp, 1e-10))
    w /= w.sum()
    grad = np.zeros(N)
    for wi, ii in zip(w, top_idx):
        M = int(abs(ii - center))
        if M == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif M < N:
            seg = N - M
            grad[M:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[M:]
    return grad

def binary_opt(h0, N, deadline, n_tries=30, kick_size=3, kick_freq=30):
    h = (h0 > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:n1 - target]] = 0
        else:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target - n1]] = 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_max = int(np.max(corr))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        cur_max = int(np.max(corr))
        best_try_max = cur_max
        best_try_h = None
        best_try_corr = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = np.correlate(h, 1.0 - h, mode="full")
            m_t = int(np.max(c_t))
            if m_t < best_try_max:
                best_try_max = m_t
                best_try_h = h.copy()
                best_try_corr = c_t.copy()
            h[p] = 1
            h[q] = 0
        if best_try_h is not None:
            h = best_try_h
            corr = best_try_corr
            stall = 0
            if best_try_max < best_max:
                best_max = best_try_max
                best_h = best_try_h.copy()
        else:
            stall += 1
            if stall > kick_freq:
                ks = min(kick_size, len(ones), len(zeros))
                ps = np.random.choice(ones, ks, replace=False)
                qs = np.random.choice(zeros, ks, replace=False)
                h[ps] = 0
                h[qs] = 1
                corr = np.correlate(h, 1.0 - h, mode="full")
                stall = 0
    return best_h, best_max * (2.0 / N)

def continuous_opt(h0, N, deadline, k0=35, lr0=0.022, temp0=0.003):

    stall = 0

            lr = max(lr, lr0 * 0.7)
