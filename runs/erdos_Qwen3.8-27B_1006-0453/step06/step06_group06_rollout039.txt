def full_delta(h, p, q, N):
    s = np.arange(-(N - 1), N)
    ps = p + s; pm = p - s
    qs = q + s; qm = q - s
    v1 = (ps >= 0) & (ps < N)
    v2 = (pm >= 0) & (pm < N)
    v3 = (qs >= 0) & (qs < N)
    v4 = (qm >= 0) & (qm < N)
    d = np.zeros(2 * N - 1)
    d[v1] -= (1.0 - h[np.clip(ps, 0, N - 1)[v1]])
    d[v2] += h[np.clip(pm, 0, N - 1)[v2]]
    d[v3] += (1.0 - h[np.clip(qs, 0, N - 1)[v3]])
    d[v4] -= h[np.clip(qm, 0, N - 1)[v4]]
    return d

    d = np.zeros(2 * N - 1)
    d[v1] -= (1.0 - h[np.clip(ps, 0, N - 1)[v1]])
    d[v2] += h[np.clip(pm, 0, N - 1)[v2]]
    d[v3] += (1.0 - h[np.clip(qs, 0, N - 1)[v3]])
    d[v4] -= h[np.clip(qm, 0, N - 1)[v4]]
    return d

def greedy_anneal(h, N, deadline, T0=0.05, cool=0.9996, topK=3, rand_frac=0.2):
    h = fix_count(h.astype(float), N)
    corr = fft_corr(h)
    best_h = h.copy()
    best_pk = float(corr.max())
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    no_imp = 0
    center = N - 1
    while time.time() < deadline and T > 0.0003:
        if np.random.random() < rand_frac or len(ones) == 0 or len(zeros) == 0:
            pi = int(ones[np.random.randint(len(ones))])
            qi = int(zeros[np.random.randint(len(zeros))])
        else:
            d = None
            pi = qi = -1
            top_idx = np.argpartition(corr, -topK)[-topK:]
            best_nc_max = cur_pk + 1e9
            for pm_ in top_idx:
                k = int(pm_ - center)
                a_val = np.zeros(len(ones))
                ma = (ones - k >= 0) & (ones - k < N)
                ma2 = (ones + k >= 0) & (ones + k < N)
                a_val[ma] -= (1.0 - h[ones[ma] - k])
                a_val[ma2] += h[ones[ma2] + k]
                b_val = np.zeros(len(zeros))
                mb = (zeros - k >= 0) & (zeros - k < N)
                mb2 = (zeros + k >= 0) & (zeros + k < N)
                b_val[mb] += (1.0 - h[zeros[mb] - k])
                b_val[mb2] -= h[zeros[mb2] + k]
                pi = int(ones[np.argmin(a_val)])
                qi = int(zeros[np.argmin(b_val)])
                if pi == qi:
                    pi2 = int(ones[np.argpartition(a_val, 1)[1]])
                    if pi2 != qi:
                        pi = pi2
                    else:
                        qi2 = int(zeros[np.argpartition(b_val, 1)[1]])
                        if qi2 != pi2:
                            qi = qi2
                        else:
                            continue
                dd = full_delta(h, pi, qi, N)
                nc = corr + dd
                npk_ = float(nc.max())
                if npk_ < best_nc_max:
                    best_nc_max = npk_
                    d = dd
            if d is None:
                pi = int(ones[np.random.randint(len(ones))])
                qi = int(zeros[np.random.randint(len(zeros))])
                d = full_delta(h, pi, qi, N)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or (npk > cur_pk and np.random.random() < np.exp(-delta / max(T, 1e-10))):
            h[pi] = 0
            h[qi] = 1
            corr = nc
            cur_pk = npk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if npk < best_pk:
                best_pk = npk
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cool
        if no_imp > 4000:
            h = best_h.copy()
            corr = fft_corr(h)
            cur_pk = best_pk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_imp = 0
            T = T0 * 0.4
    return best_h, best_pk

def continuous_opt(h, N, deadline, top_k=40, lr0=0.004):
    h = h.astype(float).copy()
    m = np.zeros(N)
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    t = 0
    lr = lr0
    center = N - 1
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s > 0:
                grad[s:] += (1.0 - h[:N - s])
                grad[:N - s] -= h[s:]
            elif s < 0:
                s2 = -s
                grad[:N - s2] += (1.0 - h[s2:])
                grad[s2:] -= h[:N - s2]
            else:
                grad += (1.0 - 2.0 * h)
        grad /= top_k
        t += 1
        m = 0.9 * m + 0.1 * grad
        h = proj_sum(h - lr * m, N / 2.0)
        if t % 400 == 0:
            lr *= 0.95
            if lr < 1e-7:
                break
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(fft_corr(h0))) * (2.0 / N0)
    # Phase 1: Aggressive binary greedy on N0 (aim for max corr = floor of needed)
    d1 = min(deadline - 350, t0 + 320)
    hb = fix_count(h0, N0)
    h_ga, pk_ga = greedy_anneal(hb, N0, d1, T0=0.06, cool=0.9995, topK=4, rand_frac=0.15)
    c5_ga = pk_ga * (2.0 / N0)
    if c5_ga < best_c5:
        best_c5, best_h = c5_ga, h_ga.copy()
    # Phase 2: Second binary pass on N0 with different params
    if time.time() < deadline - 280:
        d2 = min(deadline - 280, time.time() + 180)
        hb2 = fix_count(h0 + np.random.randn(N0) * 0.5, N0)
        h_ga2, pk_ga2 = greedy_anneal(hb2, N0, d2, T0=0.08, cool=0.9993, topK=3, rand_frac=0.35)
        c5_ga2 = pk_ga2 * (2.0 / N0)
        if c5_ga2 < best_c5:
            best_c5, best_h = c5_ga2, h_ga2.copy()
    # Phase 3: Continuous optimization
    if time.time() < deadline - 200:
        d3 = min(deadline - 200, time.time() + 150)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = proj_sum(hc, N0 / 2.0)
        h_co, pk_co = continuous_opt(hc, N0, d3, top_k=50, lr0=0.005)
        c5_co = pk_co * (2.0 / N0)
        if c5_co < best_c5:
            best_c5, best_h = c5_co, h_co.copy()
    # Phase 4: Multi-N binary exploration
    for Nt in [360, 400, 410, 420, 440, 460]:
        if time.time() > deadline - 60:
            break
        if Nt % 2 != 0:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_count((hi > 0.5).astype(float), Nt)
        sub = min(deadline - 5, time.time() + 35)
        hrt, pkr = greedy_anneal(hbt, Nt, sub, T0=0.05, cool=0.9995, topK=3, rand_frac=0.25)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hrt.copy()
    # Phase 5: Final polish on best
    if time.time() < deadline - 3:
        hf = fix_count(best_h, len(best_h))
        hfp, pkf = greedy_anneal(hf, len(best_h), deadline - 1, T0=0.02, cool=0.9998, topK=2, rand_frac=0.3)
        c5f = pkf * (2.0 / len(best_h))
        if c5f < best_c5:
            best_c5, best_h = c5f, hfp.copy()
    final_h = best_h
    final_c5 = float(np.max(fft_corr(final_h))) * (2.0 / len(final_h))
    return final_h, final_c5, len(final_h)
