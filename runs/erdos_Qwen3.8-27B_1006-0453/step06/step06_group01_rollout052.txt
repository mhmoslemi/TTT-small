    delta = np.zeros(2*N-1)
    # p goes 1->0

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, n_ones):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    if n1 > n_ones:
        idx = np.where(h > 0.5)[0]
        rem = np.random.choice(idx, n1 - n_ones, replace=False)
        h[rem] = 0.0
    elif n1 < n_ones:
        idx = np.where(h < 0.5)[0]
        add = np.random.choice(idx, n_ones - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def compute_D(h, N):
    D = np.zeros((N, 2 * N - 1))
    s = np.arange(N)
    t = np.arange(1, N)
    for j in range(N):
        m = s[s < N - j]
        D[j, N - 1 + m] -= (1.0 - h[j + m])
        m = s[s <= j]
        D[j, N - 1 + m] += h[j - m]

        D[j, N - 1 - m] -= (1.0 - h[j - m])
        m = t[t < N - j]
        D[j, N - 1 - m] += h[j + m]
    return D
