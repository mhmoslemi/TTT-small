The cross term:
When i(1→0) and j(0→1) interact:
- If s = i-j (so i-s=j): In the product h[i]*(1-h[i-s]) = h[i]*(1-h[j])
  Old: 1*1 = 1, New: 0*0 = 0, true change = -1
  From down[i]: at this m, the first factor term gives -(1-h[j]) = -(1-0) = -1
  From -down[j]: at this m, the second factor term gives -(h[j+s]) = -h[i] = -1... wait
  down[j] at this m: j's second factor is at k=j+s=i, value h[i] = 1, so down[j,m] += h[j+s] = h[i] = 1
  -down[j,m] contributes -1
  Total from down[i]-down[j]: -1 + (-1) = -2, but true change is -1.
  Cross = +1.

- If s = j-i (so j-s=i): In the product h[j]*(1-h[j-s]) = h[j]*(1-h[i])
  Old: 0*(1-1) = 0, New: 1*(1-0) = 1, true change = +1
  From down[i]: at this m, i's second factor: h[i+s] = h[j] = 0, so +0
  From down[j]: at this m, j's first factor: -(1-h[j-s]) = -(1-h[i]) = -(1-1) = 0
  -down[j,m] contributes -0 = 0
  Total: 0 + 0 = 0, but true change is +1.
  Cross = +1.

So cross(i,j) at shift s:
- If s = i-j and valid: cross += 1
- If s = j-i and valid: cross += 1
(special case: if i=j, which can't happen in a swap)

Note: s = i-j means m = i-j+N-1, s = j-i means m = j-i+N-1.
These are different unless i=j.
If i>j: s=i-j>0, so m1 = i-j+N-1, and s=j-i<0, m2 = j-i+N-1.
Both m1 and m2 are valid indices in [0, 2N-2] as long as |i-j| < N, which is always true.

But we also need the validity conditions:
- For s=i-j: need 0<=i-s<N and 0<=j<N (which is always true) → 0<=j<N ✓, and 0<=i-(i-j)<N → 0<=j<N ✓
  Actually the cross term in the product h[k]*(1-h[k-s]) with k=i, k-s=j needs 0<=k<N and 0<=k-s<N.
  k=i: 0<=i<N ✓
  k-s=j: 0<=j<N ✓
  So always valid.

- For s=j-i: need 0<=j<N and 0<=j-(j-i)<N → 0<=i<N ✓
  Always valid.

So the cross term is always:
cross[i,j,m] = 1 if m == i-j+N-1 OR m == j-i+N-1, else 0
(assuming i≠j)

2. For candidate swap (i,j): delta = down[i] - down[j] + cross_vector(i,j)
3. new_corr = corr + delta
4. new_max = max(new_corr)

The cross_vector is sparse (only 2 non-zero entries), so step 2-3 can be done in O(N) or even O(1) if we track the max carefully.

def local_search(h, N, deadline, active_K=10, iters_per_phase=50, phases=10):
    h = h.copy()
    nh = N // 2
    corr = np.correlate(h, 1.0-h, mode="full")
    P = corr.max()
    best_P = P
    best_h = h.copy()
    
    # Precompute down matrix
    def compute_down(h):
        shifts = np.arange(-(N-1), N)  # (2N-1,)
        p = np.arange(N)[:, None]  # (N,1)
        s = shifts[None, :]  # (1, 2N-1)
        pms = p - s  # (N, 2N-1)
        pps = p + s  # (N, 2N-1)
        v1 = (pms >= 0) & (pms < N)
        v2 = (pps >= 0) & (pps < N)
        d = np.where(v1, h[np.clip(pms, 0, N-1)] - 1.0, 0.0) + np.where(v2, h[np.clip(pps, 0, N-1)], 0.0)
        return d
    
    down = compute_down(h)
    
    for phase in range(phases):
        if time.time() > deadline:
            break
        # Find active shifts
        top_idx = np.argpartition(corr, -active_K)[-active_K:]
        active_corr = corr[top_idx]
        active_max = active_corr.max()
        
        for _ in range(iters_per_phase):
            if time.time() > deadline:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            # Sample candidates
            np.random.shuffle(ones)
            np.random.shuffle(zeros)
            ns = min(len(ones), 30)
            nz = min(len(zeros), 30)
            
            best_delta = 0
            best_i, best_j = -1, -1
            
            for i in ones[:ns]:
                for j in zeros[:nz]:
                    # Compute delta at active shifts
                    d = down[i] - down[j]
                    # Add cross terms
                    m1 = i - j + N - 1
                    m2 = j - i + N - 1
                    d = d.copy()
                    d[m1] += 1.0
                    d[m2] += 1.0
                    
                    # Check at active shifts
                    new_active = active_corr + d[top_idx]
                    na_max = new_active.max()
                    # Also check global max (approximate: check a few more shifts)
                    # For speed, just check active + the shift where d is most positive
                    pos_d = d.argmax()
                    new_global = max(na_max, corr[pos_d] + d[pos_d])
                    
                    if new_global < P - 1e-10:
                        if new_global < best_delta or best_i == -1:
                            best_delta = new_global
                            best_i, best_j = i, j
            
            if best_i >= 0:
                h[best_i] = 0
                h[best_j] = 1
                # Recompute
                corr = np.correlate(h, 1.0-h, mode="full")
                P = corr.max()
                down = compute_down(h)
                if P < best_P - 1e-10:
                    best_P = P
                    best_h = h.copy()
            else:
                # Stuck at this phase, recompute active set
                top_idx = np.argpartition(corr, -active_K)[-active_K:]
                active_corr = corr[top_idx]
    
    return best_h, best_P * (2.0/N)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    N = len(h)
    return float(compute_corr(h).max()) * (2.0 / N)

def compute_down(h, N):
    shifts = np.arange(-(N-1), N)
    p = np.arange(N)[:, None]
    s = shifts[None, :]
    pms = p - s
    pps = p + s
    v1 = (pms >= 0) & (pms < N)
    v2 = (pps >= 0) & (pps < N)
    return (np.where(v1, h[np.clip(pms, 0, N-1)] - 1.0, 0.0) +
            np.where(v2, h[np.clip(pps, 0, N-1)], 0.0))

def greedy_search(h, N, deadline, sample_sz=35, patience=8):
    h = h.astype(float).copy()
    nh = N // 2
    if int(h.sum() + 0.5) != nh:
        h = (h > 0.5).astype(float)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > nh:
            h[np.random.choice(ones, len(ones)-nh, replace=False)] = 0
        elif len(ones) < nh:
            h[np.random.choice(zeros, nh-len(ones), replace=False)] = 1
    corr = compute_corr(h)
    P = float(corr.max())
    best_P, best_h = P, h.copy()
    down = compute_down(h, N)
    stalls = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        oi = np.random.choice(ones, ns, replace=False)
        zo = np.random.choice(zeros, nz, replace=False)
        delta = down[oi][:,None,:] - down[zo][None,:,:]
        base = corr[None,None,:] + delta
        bP = base.max(axis=2)
        m1 = oi[:,None] - zo[None,:] + N - 1
        m2 = zo[None,:] - oi[:,None] + N - 1
        vm1 = np.take_along_axis(base, m1[:,:,None], 2).squeeze(-1)
        vm2 = np.take_along_axis(base, m2[:,:,None], 2).squeeze(-1)
        nP = np.maximum(bP, np.maximum(vm1+1, vm2+1))
        bi = np.unravel_index(np.argmin(nP), nP.shape)
        if nP[bi] < P - 1e-12:
            h[oi[bi[0]]] = 0
            h[zo[bi[1]]] = 1
            corr = compute_corr(h)
            P = float(corr.max())
            down = compute_down(h, N)
            stalls = 0
            if P < best_P - 1e-12:
                best_P, best_h = P, h.copy()
        else:
            stalls += 1
            if stalls >= patience:
                break
    return best_h, best_P * (2.0/N)

def nesterov_opt(h, N, deadline, k=50, lr0=0.015, max_iter=3000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h, best_pk = h.copy(), float(compute_corr(h).max())
    m = np.zeros(N)
    lr, beta = lr0, 0.9
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + m
        corr = compute_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        top = np.argpartition(corr, -k)[-k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0*y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= k
        # Project: clip to [0,1] and adjust sum
        h_new = np.clip(y - lr*grad, 0, 1)
        s = h_new.sum()
        if not np.isclose(s, target, atol=0.5):
            # Adjust mean
            h_new = h_new - (s - target)/N
            h_new = np.clip(h_new, 0, 1)
            # Water-filling to get exact sum
            cs = np.sort(h_new)
            cumsum = np.cumsum(cs)
            n = N
            while n > 0:
                t = (cs.sum() - target) / (N - n + 1) if n < N else 0
                # Simpler: just subtract a constant and clip
                break
            # Binary search for threshold
            lo, hi = h_new.min() - 1, h_new.max() + 1
            for _ in range(40):
                mid = (lo+hi)/2
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo+hi)/2, 0, 1)
        m = beta*(h_new - h)
        h = h_new
    return best_h, best_pk * dx
