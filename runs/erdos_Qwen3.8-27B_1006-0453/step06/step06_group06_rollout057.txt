    d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
    return d

def anneal_binary(h_init, N, deadline, T0=0.05, cool=0.9996, max_stall=400):
    h = fix_count(h_init, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    stall = 0
    while time.time() < deadline and T > 0.0003:
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        d = swap_delta(h, p, q, N)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-9)):
            h[p] = 0.0; h[q] = 1.0
            corr = nc; cur_pk = npk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if npk < best_pk:
                best_pk = npk; best_h = h.copy(); stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T *= cool
        if stall > max_stall:
            stall = 0
            h = best_h.copy(); corr = fft_corr(h); cur_pk = best_pk
            ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
            T = max(T * 0.5, T0 * 0.2)
    return best_h, best_pk

def targeted_anneal(h_init, N, deadline, T0=0.04, cool=0.9997):
    h = fix_count(h_init, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    stall = 0
    top_k_corr = 8
    while time.time() < deadline and T > 0.0005:
        corr = fft_corr(h)
        cur_pk = float(corr.max())
        top_lags = np.argpartition(corr, -top_k_corr)[-top_k_corr:]
        center = N - 1
        hot_ones = set()
        hot_zeros = set()
        for ai in top_lags:
            s = int(ai - center)
            if s >= 0 and s < N:
                for i in range(N-s):
                    if h[i] == 1 and h[i+s] == 0:
                        hot_ones.add(i)
                        hot_zeros.add(i+s)
            elif s < 0 and -s < N:
                s2 = -s
                for i in range(N-s2):
                    if h[i+s2] == 1 and h[i] == 0:
                        hot_ones.add(i+s2)
                        hot_zeros.add(i)
        if len(hot_ones) > 0 and len(hot_zeros) > 0:
            p = int(list(hot_ones)[np.random.randint(len(hot_ones))])
            q = int(list(hot_zeros)[np.random.randint(len(hot_zeros))])
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
        d = swap_delta(h, p, q, N)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-9)):
            h[p] = 0.0; h[q] = 1.0
            corr = nc; cur_pk = npk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if npk < best_pk:
                best_pk = npk; best_h = h.copy(); stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T *= cool
        if stall > 300:
            stall = 0
            h = best_h.copy(); corr = fft_corr(h); cur_pk = best_pk
            ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
            T = max(T * 0.6, T0 * 0.3)
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(fft_corr(h0))) * (2.0 / N0)

    # Phase 1: Careful continuous optimization from parent (180s)
    d1 = min(deadline - 500, time.time() + 150)
    hc = h0.copy()
    hc += np.random.randn(N0) * 0.03
    hc = proj_sum(hc, N0 / 2.0)
    hC, pkC = continuous_opt(hc, N0, d1, k0=25, lr0=0.004, noise_scale=0.0)
    c5C = pkC * (2.0 / N0)
    if c5C < best_c5:
        best_c5 = c5C; best_h = hC.copy()

    # Phase 2: Continuous polish with very small lr (120s)
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 100)
        hC2, pkC2 = continuous_polish(best_h, N0, d2, k=40, lr0=0.0012)
        c5C2 = pkC2 * (2.0 / N0)
        if c5C2 < best_c5:
            best_c5 = c5C2; best_h = hC2.copy()

    # Phase 3: Binary annealing (200s)
    d3 = min(deadline - 180, time.time() + 180)
    hb = fix_count(best_h, N0)
    hB, pkB = anneal_binary(hb, N0, d3, T0=0.06, cool=0.9995, max_stall=500)
    c5B = pkB * (2.0 / N0)
    if c5B < best_c5:
        best_c5 = c5B; best_h = hB.copy()

    # Phase 4: Targeted binary annealing (120s)
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 110)
        ht = fix_count(best_h, N0)
        hT, pkT = targeted_anneal(ht, N0, d4, T0=0.035, cool=0.9997)
        c5T = pkT * (2.0 / N0)
        if c5T < best_c5:
            best_c5 = c5T; best_h = hT.copy()

    # Phase 5: Multi-N exploratiion
    for Nt in [340, 400, 460, 520]:
        if time.time() > deadline - 80:
            break
        if Nt % 2 != 0:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_count(hi, Nt)
        sub = min(deadline - 20, time.time() + 35)
        hrt, pkr = anneal_binary(hbt, Nt, sub, T0=0.04, cool=0.9997, max_stall=300)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5 = c5r; best_h = hrt.copy()

    # Phase 6: Final continuous polish on best (remaining time)
    if time.time() < deadline - 10:
        d6 = deadline - 3
        hf = best_h.astype(float).copy()
        hf = proj_sum(hf, len(best_h) / 2.0)
        hf2, pkf2 = continuous_polish(hf, len(best_h), d6, k=60, lr0=0.0008)
        c5f2 = pkf2 * (2.0 / len(best_h))
        if c5f2 < best_c5:
            best_c5 = c5f2; best_h = hf2.copy()

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
