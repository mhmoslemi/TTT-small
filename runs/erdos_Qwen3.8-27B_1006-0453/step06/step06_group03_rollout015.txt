    """p_list: indices to set to 0, q_list: indices to set to 1 (same length).
    Return approximate new max of c after swap."""
    max_val = 0.0
    for L in top_lags:
        delta = 0.0
        for p in p_list:
            if 0 <= p + L < N: delta += h[p+L]
            if 0 <= p - L < N: delta += h[p-L]
        for q in q_list:
            if 0 <= q + L < N: delta -= h[q+L]
            if 0 <= q - L < N: delta -= h[q-L]
        v = c[L + N - 1] + delta  # c is indexed by m = L + N - 1
        if v > max_val: max_val = v
    return max_val

Valid i: 0≤i<N and 0≤i-M<N, i.e., M≤i<N.

d(f)/d(h[j]):
- j appears as i-M (i.e., i=j+M): valid when M≤j+M<N, i.e., 0≤j<N-M.

- j appears as i: valid when M≤j<N.
  Contribution: -h[j-M]

- For j ∈ [0, N-M): (1-h[j+M])

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        sel = np.random.choice(idx, n1 - nh, replace=False)
        h[sel] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        sel = np.random.choice(idx, nh - n1, replace=False)
        h[sel] = 1
    return h

def binary_search(h, N, deadline, top_m=40, sample_n=200, anneal_top=20):
    h = to_binary(h, N)
    center = N - 1
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    temp = 0.02
    no_improve = 0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            temp = min(0.05, temp * 1.05)
        elif no_improve == 0:
            temp = max(0.005, temp * 0.95)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argsort(c)[-top_m:]
        top_lags = (top_idx - center).astype(int)
        c_top = c[top_idx]
        ns = min(sample_n, len(ones))
        nz = min(sample_n, len(zeros))
        p_c = np.random.choice(ones, ns, replace=False)
        q_c = np.random.choice(zeros, nz, replace=False)
        tl = top_lags[None, :]
        pp = p_c[:, None] + tl
        pn = p_c[:, None] - tl
        dp = np.where((pp >= 0) & (pp < N), h[np.clip(pp, 0, N - 1)], 0.0) + \
             np.where((pn >= 0) & (pn < N), h[np.clip(pn, 0, N - 1)], 0.0)
        pq = q_c[:, None] + tl
        qn = q_c[:, None] - tl
        dq = -(np.where((pq >= 0) & (pq < N), h[np.clip(pq, 0, N - 1)], 0.0) +
               np.where((qn >= 0) & (qn < N), h[np.clip(qn, 0, N - 1)], 0.0))
        nd = dp[:, None, :] + dq[None, :, :]
        new_corr = c_top[None, None, :] + nd
        new_max = np.max(new_corr, axis=2)
        fi = int(np.argmin(new_max))
        bi, bj = divmod(fi, nz)
        best_new = float(new_max[bi, bj])
        best_p, best_q = int(p_c[bi]), int(q_c[bj])
        accept = best_new < P
        if not accept and anneal_top > 0:
            ranked = np.argsort(new_max)[:anneal_top]
            ri = int(ranked[np.random.randint(anneal_top)])
            ri_bi, ri_bj = divmod(ri, nz)
            if new_max[ri_bi, ri_bj] < P + temp:
                accept = True
                best_p, best_q = int(p_c[ri_bi]), int(q_c[ri_bj])
        if accept:
            h[best_p] = 0.0
            h[best_q] = 1.0
    return best_h, best_c

def nesterov_cont(h, N, deadline, k=50, lr0=0.015, beta0=0.92):
    h = h.astype(float).copy()
    target = N / 2.0
    center = N - 1
    best_h = h.copy()
    best_v = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_v:
            best_v = pk
            best_h = h.copy()
        top_idx = np.argpartition(corr, -k)[-k:]
        top_lags = (top_idx - center).astype(int)
        grad = np.zeros(N)
        for L in top_lags:
            if L >= 0:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.996
            beta = min(0.97, beta0 + it * 0.0001)
    return best_h, best_v * (2.0 / N)

    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)

    x_best = np.linspace(0, 2, len(best_h), endpoint=False)

        d5 = min(deadline - 50, time.time() + 70)
        h_a, _ = binary_search(best_h, len(best_h), d5, top_m=50, sample_n=180, anneal_top=30)
        c5_a = compute_c5(h_a, len(best_h))

            best_c5, best_h = c5_a, h_a.copy()

    # Phase 6: Final Nesterov
    if time.time() < deadline - 10:

        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_cont(hF, len(best_h), deadline - 5, k=70, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 7: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = to_binary(best_h, len(best_h))
        hF2, _ = binary_search(hF2, len(best_h), deadline - 0.5, top_m=40, sample_n=150)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    if final_c5 > best_c5:
        best_h = to_binary(best_h, len(best_h))
        final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
