    N = len(h)
    c = np.zeros(N)
    if shift >= 0:
        # j < N-shift: c[j] = 1 - h[j+shift]
        c[:N-shift] = 1.0 - h[shift:]
        # j >= shift: c[j] -= h[j-shift]

        c[s:] = 1.0 - h[:N-s]
        # j < N-s: c[j] -= h[j+s]
        c[:N-s] -= h[s:]
    return c

    """Gradient of C(shift) w.r.t. h, where C(s) = sum_i h[i]*(1-h[i+s])"""
    G = np.zeros(N)

        G[shift:] -= h[:N-shift]
    else:
        s = -shift
        G[s:] = 1.0 - h[:N-s]
        G[:N-s] -= h[s:]
    return G

def compute_d_flip(h, N):
    """Compute d_flip[p, s] for all p, s: change in corr[s] when h[p] goes 1→0"""
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)  # shape (L,)
    D = np.zeros((N, L))
    for p in range(N):
        # h[p+shift] valid where 0 <= p+shift < N
        idx_p_shift = p + shifts  # shape (L,)
        valid_plus = (idx_p_shift >= 0) & (idx_p_shift < N)
        idx_p_negshift = p - shifts
        valid_minus = (idx_p_negshift >= 0) & (idx_p_negshift < N)
        
        h_plus = np.where(valid_plus, h[np.clip(idx_p_shift, 0, N-1)], 0.0)
        h_minus = np.where(valid_minus, h[np.clip(idx_p_negshift, 0, N-1)], 0.0)
        
        D[p, :] = -(1.0 - h_plus) * valid_plus + h_minus * valid_minus
    return D

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_shift(h, N, shift):
    G = np.zeros(N)
    if shift >= 0:
        G[:N-shift] = 1.0 - h[shift:]
        if N > shift:
            G[shift:] -= h[:N-shift]
    else:
        s = -shift
        G[s:] = 1.0 - h[:N-s]
        G[:N-s] -= h[s:]
    return G

def compute_d_flip(h, N):
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)
    idx_p = np.arange(N)[:, None] + shifts[None, :]
    idx_m = np.arange(N)[:, None] - shifts[None, :]
    vp = (idx_p >= 0) & (idx_p < N)
    vm = (idx_m >= 0) & (idx_m < N)
    hp = np.where(vp, h[np.clip(idx_p, 0, N-1)], 0.0)
    hm = np.where(vm, h[np.clip(idx_m, 0, N-1)], 0.0)
    return -(1.0 - hp) * vp.astype(np.float64) + hm * vm.astype(np.float64)

def continuous_opt(h, N, deadline, lr=0.005):
    dx = 2.0 / N
    target = N / 2.0

    best_val = compute_c5(h)

            best_h = h.copy()
        
        # Gradient from top shifts

        top_k = np.argpartition(c, -8)[-8:]
        G = np.zeros(N)
        for sk in top_k:
            shift = sk - (N - 1)
            G += grad_shift(h, N, shift)
        G /= len(top_k)
        G *= dx
        G -= G.mean()
        
        bc1 = 1.0 - b1 ** t
        bc2 = 1.0 - b2 ** t
        m = b1 * m + (1 - b1) * G
        v = b2 * v + (1 - b2) * G * G
        h_new = h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = project_sum(h_new, target)

    # log-sum-exp

    G -= G.mean()
    return G, c.max() * (2.0/N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_shift(h, N, shift):
    G = np.zeros(N)
    if shift >= 0:
        G[:N-shift] = 1.0 - h[shift:]
        G[shift:] -= h[:N-shift]
    else:
        s = -shift
        G[s:] = 1.0 - h[:N-s]
        G[:N-s] -= h[s:]
    return G

def lse_grad(h, N, beta):
    dx = 2.0 / N
    c = fft_corr(h)
    L = len(c)
    cmax = float(c.max())
    exps = np.exp(beta * (c - cmax))
    w = exps / exps.sum()
    top = np.where(w > 1e-5)[0]
    shifts = np.arange(L) - (N - 1)
    G = np.zeros(N)
    for k in top:
        G += w[k] * grad_shift(h, N, shifts[k])
    G *= dx
    G -= G.mean()
    return G, cmax * dx

def cont_opt(h, N, deadline, lr=0.004, beta0=5.0, beta1=60.0):

def compute_d_flip(h, N):
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)
    ip = np.arange(N)[:, None] + shifts[None, :]
    im = np.arange(N)[:, None] - shifts[None, :]
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    hp = np.where(vp, h[np.clip(ip, 0, N-1)], 0.0)
    hm = np.where(vm, h[np.clip(im, 0, N-1)], 0.0)
    return -(1.0 - hp) * vp.astype(np.float64) + hm * vm.astype(np.float64)
