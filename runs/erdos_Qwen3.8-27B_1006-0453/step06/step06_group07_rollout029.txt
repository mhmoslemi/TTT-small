    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        lr *= 0.9996
        beta = min(0.97, 0.85 + it * 0.00008)
    return best_h, best_pk * dx

def upsample(h_old, N_new):
    N_old = len(h_old)
    if N_old == N_new:
        return h_old.copy()
    x_old = np.arange(N_old, dtype=float)
    x_new = np.arange(N_new, dtype=float) * N_old / N_new
    return np.interp(x_new, x_old, h_old)

def multi_nesterov(h_best, N, deadline, n_restarts=5, seed=42):
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best, N)
    rng = np.random.RandomState(seed)
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        noise_adj = 0.03 + 0.02 * r
        hc = h_best + rng.randn(N) * noise_adj
        hc = project_sum(hc, N / 2.0)
        sub_deadline = min(deadline - 5, time.time() + (deadline - time.time()) / (n_restarts - r) - 2)
        k_val = 40 + 10 * r
        lr_val = 0.014 + 0.003 * r
        h_r, c5_r = nesterov_opt(hc, N, sub_deadline, k=k_val, lr0=lr_val, seed=seed + r * 7)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary search at N0 with multiple restarts
    h_bin = ensure_sum_binary(h0, N0, seed)
    d1 = min(deadline - 500, t0 + 120)
    h1, c5_1 = binary_search(h_bin, N0, d1, max_sweeps=2000, seed=seed)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Nesterov at N0 (multi-restart)
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 180)
        h2, c5_2 = multi_nesterov(best_h, N0, d2, n_restarts=5, seed=seed)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Upscale to 2*N0 and binary search
    N1 = 2 * N0
    if N1 <= 1600 and time.time() < deadline - 200:
        h_up = ensure_sum_binary(upsample(best_h, N1), N1, seed + 1)
        d3 = min(deadline - 150, time.time() + 120)
        h3, c5_3 = binary_search(h_up, N1, d3, max_sweeps=800, seed=seed + 10)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Nesterov at N1
    if time.time() < deadline - 100:
        d4 = min(deadline - 60, time.time() + 100)
        h4, c5_4 = multi_nesterov(best_h, len(best_h), d4, n_restarts=4, seed=seed + 20)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Final binary polish at best resolution
    N_cur = len(best_h)
    if time.time() < deadline - 20:
        hF = ensure_sum_binary(best_h, N_cur, seed + 30)
        d5 = deadline - 5
        sf_extra = ""
        h5, c5_5 = binary_search(hF, N_cur, d5, max_sweeps=1000, seed=seed + 30)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final tiny Nesterov polish
    if time.time() < deadline - 3:
        N_fin = len(best_h)
        hc = best_h + np.random.randn(N_fin) * 0.015
        hc = project_sum(hc, N_fin / 2.0)
        h6, c5_6 = nesterov_opt(hc, N_fin, deadline - 1, k=60, lr0=0.008, seed=seed + 50)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, final_c5, len(best_h)
