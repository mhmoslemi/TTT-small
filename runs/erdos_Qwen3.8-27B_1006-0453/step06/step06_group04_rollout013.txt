import numpy as np
import time

def lin_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(np.max(lin_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient(y, corr, N, k_top, temp):
    k_top = min(k_top, 2 * N - 1)
    top = np.argpartition(corr, -k_top)[-k_top:]
    w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-8))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for wi, j in zip(w, top):
        d = int(j - center)
        if d == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif d > 0:
            seg = N - d
            if seg > 0:
                grad[:seg] += wi * (1.0 - y[d:])
                grad[d:] -= wi * y[:seg]
        else:
            dd = -d
            if dd < N:
                seg = N - dd
                grad[dd:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[dd:]
    return grad

def cont_opt(h0, N, deadline, k_top=60, lr0=0.02, temp0=0.025):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    h_prev = h.copy()
    lr = lr0
    beta = 0.9
    temp = temp0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = lin_corr(y)
        val = float(np.max(corr)) * (2.0 / N)
        if val < best_val - 1e-12:
            best_val = val
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        grad = subgradient(y, corr, N, k_top, temp)
        h = project_sum(y - lr * grad, target)
        it += 1
        if it % 150 == 0:
            lr = max(0.002, lr * 0.996)
            temp = max(0.008, temp * 0.996)
            beta = min(0.96, 0.85 + it * 0.00015)
        if stall > 350:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = max(lr, lr0 * 0.7)
            stall = 0
    return best_h, best_val

def bin_search(h0, N, deadline, n_rounds=50):
    h = (h0 > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            idx = np.where(h == 1)[0].copy()
            np.random.shuffle(idx)
            h[idx[:n1 - target]] = 0
        else:
            idx = np.where(h == 0)[0].copy()
            np.random.shuffle(idx)
            h[idx[:target - n1]] = 1
    best_mx = int(np.max(lin_corr(h)))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline - 2:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(n_rounds):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            mx = int(np.max(lin_corr(h)))
            if mx < best_mx:
                best_mx = mx
                best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
            else:
                h[p] = 1
                h[q] = 0
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 15:
                nf = min(5, len(ones) // 2, len(zeros) // 2)
                if nf > 0:
                    pf = np.random.choice(ones, nf, replace=False)
                    qf = np.random.choice(zeros, nf, replace=False)
                    h[pf] = 0
                    h[qf] = 1
                stall = 0
    return best_h, best_mx * (2.0 / N)

def interp_to(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

    return cands
