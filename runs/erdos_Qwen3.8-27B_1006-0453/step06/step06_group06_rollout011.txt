def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

    n1 = int(np.sum(h > 0.5))
    h = (h > 0.5).astype(float)
    if n1 > target:
        idx = np.where(h==1)[0]

        h[idx[order]] = 0
    elif n1 < target:
        idx = np.where(h==0)[0]
        order = np.random.permutation(len(idx))[:target-n1]
        h[idx[order]] = 1
    return h

def swap_delta(h, p, q, N, s_arr):
    """Compute delta in correlation when h[p]: 1->0, h[q]: 0->1"""
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    d = np.zeros(len(s_arr))
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
    # p goes 1->0: corr changes by -(1-h[p+k]) at positions where p+k valid
    #                 and +h[p-k] at positions where p-k valid
    pc = ps[vps]; pmc = pm[vpm]
    d[vps] -= (1.0 - h[pc])
    d[vpm] += h[pmc]
    # q goes 0->1: corr changes by +(1-h[q+k]) at positions where q+k valid

    qsc = qs[vqs]; qmc = qm[vqm]
    d[vqs] += (1.0 - h[qsc])
    d[vqm] -= h[qmc]
    return d

    zeros = np.where(h==0)[0]

            corr = nc; current_pk = npk

def continuous_refine(h, N, deadline, top_k=40, lr0=0.003):

        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N/2.0)
        if t % 300 == 0:

    return best_h, best_pk

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:2*N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def to_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        sel = rng.choice(len(idx), n1 - target, replace=False)
        h[idx[sel]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        sel = rng.choice(len(idx), target - n1, replace=False)
        h[idx[sel]] = 1
    return h
