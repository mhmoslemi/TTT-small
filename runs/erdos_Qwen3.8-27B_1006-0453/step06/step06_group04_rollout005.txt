def binary_sa(h, N, deadline, max_iters=100000):
    """Simulated annealing for binary h."""
    assert N == len(h)
    assert abs(h.sum() - N/2) < 0.1
    h = (h > 0.5).astype(float)
    
    corr = fft_corr(h)
    best_max = corr.max()
    best_h = h.copy()
    
    T = 0.5  # initial temperature
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    stall = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # Pick a random swap
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        
        # Evaluate swap
        h[p], h[q] = 0.0, 1.0
        new_corr = fft_corr(h)
        new_max = new_corr.max()
        
        if new_max <= best_max or np.random.random() < np.exp((best_max - new_max) / max(T, 1e-10)):
            h[p], h[q] = 0.0, 1.0
            # commit
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[p], h[q] = 1.0, 0.0
            stall += 1
        
        # Cool down
        if it % 500 == 0:
            T *= 0.95
        if stall > 2000:
            T *= 0.5
            if T < 0.01:
                T = 0.3  # reheat
                # random perturbation
                for _ in range(10):
                    p = ones[np.random.randint(len(ones))]
                    q = zeros[np.random.randint(len(zeros))]
                    h[p], h[q] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stall = 0
    
    return best_h, best_max

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_grad(h, w):
    """Gradient of sum_k w[k]*corr[k] w.r.t. h."""
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    for ki in range(len(w)):
        wk = w[ki]
        if wk < 1e-12:
            continue
        L = ki - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wk * (1.0 - h[:seg])
            grad[:seg] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:seg]
    return grad

        corr = fft_corr(y)
