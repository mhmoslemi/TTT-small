import time

def compute_c5_correlate(h, N):
    """Exact C5 using np.correlate"""
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:L]

def swap_delta(h, p, q, N):
    """Compute ΔC for swapping h[p]:1->0 and h[q]:0->1. O(N) time."""
    C = np.zeros(2 * N - 1)
    # For L >= 0 (output index k = L + N - 1, k from N-1 to 2N-2)
    # ΔC[L] for L >= 0:
    #   -(1-h[p-L]) if L <= p
    #   -h[p+L] if L <= N-1-p
    #   +(1-h[q-L]) if L <= q

    #   -h[p-M] if M <= p
    #   +(1-h[q+M]) if M <= N-1-q
    #   -h[q-M] if M <= q

        if L <= p:
            delta -= (1.0 - h[p - L])
        if L <= N - 1 - p:
            delta -= h[p + L]
        if L <= q:
            delta += (1.0 - h[q - L])
        if L <= N - 1 - q:
            delta -= h[q + L]
        C[N - 1 + L] = delta
    # L < 0: M = -L, k = N - 1 - M, M from 1 to N-1
    for M in range(1, N):
        delta = 0.0
        if M <= N - 1 - p:
            delta -= (1.0 - h[p + M])
        if M <= p:
            delta -= h[p - M]
        if M <= N - 1 - q:

        if M <= q:

def compute_swap_effect(h, p, q, N):
    """Compute the change in correlation when h[p] 1->0, h[q] 0->1."""
    # The correlation C[k] = sum_n h[n+k-(N-1)] * (1-h[n])
    # After swap, h' = h - e_p + e_q
    # ΔC[k] = C'[k] - C[k]
    # = sum_n (h'[n+k-(N-1)] - h[n+k-(N-1)]) * (1-h[n])
    #   + sum_n h[n+k-(N-1)] * ((1-h'[n]) - (1-h[n]))
    # = sum_n Δh[n+k-(N-1)] * (1-h[n]) - sum_n h[n+k-(N-1)] * Δh[n]
    
    # Δh = -e_p + e_q
    # Δh[j] = -1 if j==p, +1 if j==q, 0 otherwise

    # where C_contrib_from_p[k] = (1-h[n]) at n+k-(N-1)=p, i.e., n=p+k-(N-1)
    #   plus h[n+k-(N-1)] at n=p (i.e., the term where n=p)
    
    # This is getting complicated. Let me just use the formula directly.
    
    # For output index k (0 to 2N-2), lag L = k - (N-1):
    # C[k] = sum_valid_n h[n+L] * (1-h[n])
    
    # ΔC[k] when h[p] 1->0:
    #   - contribution where n+L=p (n=p-L): (1-h[p-L]) * [p-L valid and n+L=p means h[p]]
    #     = (1-h[p-L]) if 0<=p-L<N and 0<=p<N (always true for p)
    #     More precisely: valid if 0<=p-L<N, i.e., L<=p<N+L, i.e., -L<=p and p-L<N
    #   - contribution where n=p: h[p+L] * 1 * [p valid and n=p]
    #     = h[p+L] if 0<=p<N and 0<=p+L<N
    # ΔC[k] when h[q] 0->1:
    #   + contribution where n+L=q: (1-h[q-L]) if valid
    #   - contribution where n=q: h[q+L] * 0 (was h[q+L]*1, now h[q+L]*0) → change is -h[q+L] if valid
    #     Wait: before: h[q+L]*(1-h[q]) = h[q+L]*1 = h[q+L]
    #     After: h'[q+L]*(1-h'[q]) = h[q+L]*0 = 0

    #            + (1-h[q-L])*[valid_q_as_top] - h[q+L]*[valid_q_as_bot]
    
    # where valid_p_as_top means 0<=p-L<N (i.e., p-L in [0,N-1])
    # and valid_p_as_bot means 0<=p+L<N (i.e., p+L in [0,N-1])
    
    # Let's vectorize:
    L = np.arange(-(N-1), N)  # all lags
    k = L + (N - 1)  # output indices
    C_delta = np.zeros(2*N-1)
    
    # p as "top" (n+L=p, so n=p-L): valid if 0<=p-L<N
    mask = (p - L >= 0) & (p - L < N)
    C_delta[mask] -= (1.0 - h[np.clip(p - L[mask], 0, N-1)])
    
    # p as "bottom" (n=p, so n+L=p+L): valid if 0<=p+L<N
    mask = (p + L >= 0) & (p + L < N)
    C_delta[mask] -= h[np.clip(p + L[mask], 0, N-1)]
    
    # q as "top" (n+L=q, so n=q-L): valid if 0<=q-L<N

    mask = (q + L >= 0) & (q + L < N)
    C_delta[mask] -= h[np.clip(q + L[mask], 0, N-1)]
    
    return C_delta

def compute_swap_delta(h, p, q, N):
    L = np.arange(-(N-1), N)
    C_delta = np.zeros(2*N-1)
    
    # p top: n = p-L, valid if 0 <= p-L < N
    mask = (p - L >= 0) & (p - L < N)
    C_delta[mask] -= (1.0 - h[p - L[mask]])
    
    # p bottom: top = p+L, valid if 0 <= p+L < N
    mask = (p + L >= 0) & (p + L < N)
    C_delta[mask] -= h[p + L[mask]]
    
    # q top: n = q-L, valid if 0 <= q-L < N
    mask = (q - L >= 0) & (q - L < N)
    C_delta[mask] += (1.0 - h[q - L[mask]])
    
    # q bottom: top = q+L, valid if 0 <= q+L < N
    mask = (q + L >= 0) & (q + L < N)
    C_delta[mask] -= h[q + L[mask]]
    
    # Fix double-counting when L = p - q
    idx = (N - 1) + (p - q)
    C_delta[idx] += 1.0
    
    return C_delta
