import numpy as np

def compute_corr(h):
    """Compute np.correlate(h, 1-h, 'full')"""
    return np.correlate(h, 1.0 - h, mode='full')

def proj_sum(h, target):
    """Project h onto [0,1]^N with sum = target"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Bisection on threshold
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    h = h.astype(float).copy()
    h = proj_sum(h, N / 2.0)
    corr = compute_corr(h)
    pk = corr.max()
    best_h = h.copy()
    best_pk = pk
    center = N - 1
    lr = lr0
    m = np.zeros(N)
    t = 0
    while time.time() < deadline:
        # Compute subgradient: weighted average of gradients at top-k shifts
        idx = np.where(corr >= np.partition(corr, -top_k)[-top_k])[0]
        w = np.maximum(0, corr[idx] - corr[idx].min())
        w = w / (w.sum() + 1e-10)
        grad = np.zeros(N)

            elif s > 0 and s < N:
                grad[:N-s] += w[np.where(idx == ai)[0][0]] * (1.0 - h[s:])
                grad[s:] -= w[np.where(idx == ai)[0][0]] * h[:N-s]
            elif s < 0 and s > -N:
                s2 = -s
                grad[s2:] += w[np.where(idx == ai)[0][0]] * (1.0 - h[:N-s2])
                grad[:N-s2] -= w[np.where(idx == ai)[0][0]] * h[s2:]
        t += 1

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_binary(h, N):
    h = np.round(h)
    target = N // 2
    s = int(h.sum())
    if s > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - s]] = 1
    return h

def continuous_opt(h, N, deadline, top_k=60, lr0=0.002):
    h = h.astype(float).copy()
    h = proj_sum(h, N / 2.0)
    corr = compute_corr(h)
    pk = float(corr.max())
    best_h, best_pk = h.copy(), pk
    center = N - 1
    m, v = np.zeros(N), np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t, lr = 0, lr0
    while time.time() < deadline:
        thr = corr.max() - 1.0
        active = np.where(corr >= thr)[0]
        if len(active) > top_k:
            active = np.where(corr >= np.partition(corr, -top_k)[-top_k])[0]
        n_act = len(active)
        grad = np.zeros(N)
        for ai in active:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < s < N:
                grad[:N - s] += (1.0 - h[s:])
                grad[s:] -= h[:N - s]
            elif -N < s < 0:
                s2 = -s
                grad[s2:] += (1.0 - h[:N - s2])
                grad[:N - s2] -= h[s2:]
        grad /= n_act
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 80 == 0:
            corr = compute_corr(h)
            pk = float(corr.max())
            if pk < best_pk:
                best_pk, best_h = pk, h.copy()
            lr *= 0.99
    corr = compute_corr(h)
    if corr.max() < best_pk:
        best_pk, best_h = float(corr.max()), h.copy()
    return best_h, best_pk

def swap_delta(h, N, p, q, s_arr):
    pp, pm = p + s_arr, p - s_arr
    qq, qm = q + s_arr, q - s_arr
    vpp, vpm = (pp >= 0) & (pp < N), (pm >= 0) & (pm < N)
    vqq, vqm = (qq >= 0) & (qq < N), (qm >= 0) & (qm < N)
    d = np.zeros(2 * N - 1)
    d[vpp] -= (1.0 - h[np.clip(pp, 0, N - 1)])
    d[vpm] += h[np.clip(pm, 0, N - 1)]
    d[vqq] += (1.0 - h[np.clip(qq, 0, N - 1)])
    d[vqm] -= h[np.clip(qm, 0, N - 1)]
    return d

def binary_ls(h, N, deadline, max_stall=4000):
    h = fix_binary(h, N)
    s_arr = np.arange(-(N - 1), N)
    corr = compute_corr(h)
    pk = float(corr.max())
    best_h, best_pk = h.copy(), pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    it = 0
    while time.time() < deadline and stall < max_stall:
        if len(ones) < N // 4 or len(zeros) < N // 4:
            break
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        d = swap_delta(h, N, p, q, s_arr)
        nc = corr + d
        npk = float(nc.max())
        if npk <= pk + 1e-12:
            h[p], h[q] = 0, 1
            corr = nc
            if npk < pk:
                pk = npk
                stall = 0
                if pk < best_pk:
                    best_pk, best_h = pk, h.copy()
            else:
                stall += 1
            ones, zeros = np.where(h == 1)[0], np.where(h == 0)[0]
        else:
            stall += 1
        it += 1
    return best_h, best_pk

def block_ls(h, N, deadline, max_stall=2000):
    h = fix_binary(h, N)
    s_arr = np.arange(-(N - 1), N)
    corr = compute_corr(h)
    pk = float(corr.max())
    best_h, best_pk = h.copy(), pk
    stall = 0
    while time.time() < deadline and stall < max_stall:
        bl = int(np.random.choice([3, 4, 5, 6, 8]))
        if bl >= N // 3:
            continue
        v1, v0 = [], []
        for i in range(N - bl + 1):
            if h[i:i + bl].sum() == bl:
                v1.append(i)
            if h[i:i + bl].sum() == 0:
                v0.append(i)
        if not v1 or not v0:
            stall += 50
            continue
        p = int(np.random.choice(v1))
        q = int(np.random.choice(v0))
        if abs(p - q) < bl:
            stall += 1
            continue
        d = np.zeros(2 * N - 1)
        for off in range(bl):
            d += swap_delta(h, N, p + off, q + off, s_arr)
        nc = corr + d
        npk = float(nc.max())
        if npk <= pk + 1e-12:
            h[p:p + bl] = 0
            h[q:q + bl] = 1
            corr = nc
            pk = npk
            stall = 0
            if pk < best_pk:
                best_pk, best_h = pk, h.copy()
        else:
            stall += 1
    return best_h, best_pk
