def fft_corr(h):

    H = np.fft.fft(h, L)

def swap_delta_exact(c, h, p, q, top_idx, N):
    """Compute exact new max correlation after swapping h[p]:1→0, h[q]:0→1."""
    center = N - 1
    mags = top_idx - center  # signed lags
    new_c = np.zeros(len(mags))
    for i, m in enumerate(mags):
        m = int(m)
        cv = c[top_idx[i]]
        dc = 0.0
        if m >= 0:
            # full[m] = sum_{n=0}^{N-1-m} h[n+m]*(1-h[n])
            if m <= p <= N-1: dc -= (1-h[p-m])  # p as n+m
            if 0 <= p <= N-1-m: dc += h[p+m]  # p as n (term was 0 since h[p]=1)
            if m <= q <= N-1: dc += (1-h[q-m])  # q as n+m (term was 0 since h[q]=0)
            if 0 <= q <= N-1-m: dc -= h[q+m]  # q as n
        else:
            M = -m
            # full[-M] = sum_{n=M}^{N-1} h[n-M]*(1-h[n])
            if M <= p <= N-1: dc += (1-h[p+M])  # p as n (was h[p-M]*(1-1)=0, now ... wait)
            # Actually: h[p] appears as h[n-M] when n-M=p, i.e., n=p+M. Valid if M≤p+M≤N-1, i.e., 0≤p≤N-1-M.
            # Term was: h[p]*(1-h[p+M]) = 1*(1-h[p+M]). Now: 0. Δ: -(1-h[p+M])
            if 0 <= p <= N-1-M: dc -= (1-h[p+M])
            # h[p] appears as (1-h[n]) when n=p. Valid if M≤p≤N-1.
            # Term was: h[p-M]*(1-h[p]) = h[p-M]*0 = 0. Now: h[p-M]*(1-0) = h[p-M]. Δ: +h[p-M]
            if M <= p <= N-1: dc += h[p-M]
            # h[q]: 0→1
            # q appears as h[n-M] when n=q+M. Valid if 0≤q≤N-1-M.
            # Term was: 0*(1-h[q+M]) = 0. Now: 1*(1-h[q+M]). Δ: +(1-h[q+M])
            if 0 <= q <= N-1-M: dc += (1-h[q+M])
            # q appears as (1-h[n]) when n=q. Valid if M≤q≤N-1.
            # Term was: h[q-M]*1. Now: h[q-M]*0. Δ: -h[q-M]
            if M <= q <= N-1: dc -= h[q-M]
        new_c[i] = cv + dc

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_subgrad(h, corr, T, N, top_m=60):
    center = N - 1
    grad = np.zeros(N)
    top_idx = np.argsort(corr)[-min(top_m, len(corr)):]
    w = np.exp((corr[top_idx] - corr[top_idx].max()) / T)
    w = w / w.sum()
    for ii in range(len(top_idx)):
        lag = int(top_idx[ii] - center)
        wi = w[ii]
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:N - lag])
            grad[:N - lag] -= wi * h[lag:]
        else:
            M = -lag
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad

def nesterov_smooth(h, N, deadline, T0=8.0, T_decay=0.003, T_min=0.3, lr0=0.02, top_m=60):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.90
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        T = max(T_min, T0 * np.exp(-it * T_decay))
        grad = lse_subgrad(y, corr, T, N, top_m)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-8:
            break
        step = lr / max(gnorm, 0.01)
        h_new = project_sum(y - step * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.96
            beta = min(0.97, 0.90 + it * 0.00008)
    return best_h, best_pk * (2.0 / N)

def swap_delta_exact(c, h, p, q, top_idx, N):
    center = N - 1
    best_new = 0.0
    for ai in top_idx:
        m = int(ai - center)
        cv = c[ai]
        dc = 0.0
        if m >= 0:
            if m <= p <= N - 1:
                dc -= (1.0 - h[p - m])
            if 0 <= p <= N - 1 - m:
                dc += h[p + m]
            if m <= q <= N - 1:
                dc += (1.0 - h[q - m])
            if 0 <= q <= N - 1 - m:
                dc -= h[q + m]
        else:
            M = -m
            if 0 <= p <= N - 1 - M:
                dc -= (1.0 - h[p + M])
            if M <= p <= N - 1:
                dc += h[p - M]
            if 0 <= q <= N - 1 - M:
                dc += (1.0 - h[q + M])
            if M <= q <= N - 1:
                dc -= h[q - M]
        best_new = max(best_new, cv + dc)
    return best_new

def binary_local(h, N, deadline, max_sweeps=600, sample_n=160, top_m=20, anneal=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))][nh:]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))][:nh - n1]
        h[idx] = 1
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200 and anneal == 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argsort(c)[-top_m:]
        sn = min(sample_n, len(ones), len(zeros))
        p_list = np.random.choice(ones, sn, replace=False) if len(ones) > sn else ones
        q_list = np.random.choice(zeros, sn, replace=False) if len(zeros) > sn else zeros
        best_np = P
        best_pq = None
        for p in p_list:
            for q in q_list:
                nm = swap_delta_exact(c, h, int(p), int(q), top_idx, N)
                if nm < best_np - 1e-12:
                    best_np = nm
                    best_pq = (int(p), int(q))
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
    return h, best_c

def refine_small(h, N, deadline, n_trials=500):
    """Try small continuous perturbations."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_c = float(np.max(fft_corr(h)))
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        idx = np.random.randint(N)
        delta = np.random.uniform(-0.3, 0.3)
        h2 = h.copy()
        h2[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
        h2 = project_sum(h2, target)
        c2 = float(np.max(fft_corr(h2)))
        if c2 < best_c:
            best_c = c2
            h = h2.copy()
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
