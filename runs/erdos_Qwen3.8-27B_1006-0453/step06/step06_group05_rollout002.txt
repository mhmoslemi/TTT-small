        h[:] = 0
        idx = np.arange(N)
        vals = h[idx]
        if pattern >= 0:
            h[idx] = 1.0
            h[np.random.choice(N, N//2, replace=False)] = 1.0
        else:
            h[np.random.choice(N, N//2, replace=False)] = 1.0
    if int(h.sum()) != N // 2:
        ones_i = np.where(h == 1)[0]
        zeros_i = np.where(h == 0)[0]
        h[:] = 0
        if len(ones_i) > N//2:
            h[np.random.choice(ones_i, N//2, replace=False)] = 1
        else:
            h[ones_i] = 1
            h[np.random.choice(zeros_i, N//2 - len(ones_i), replace=False)] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Full exhaustive binary local search from initial
    d1 = min(deadline - 400, t0 + 200)
    h_b1 = (h0 > 0.5).astype(float)
    n1 = int(h_b1.sum())
    if n1 != N0 // 2:
        oi, zi = np.where(h_b1==1)[0], np.where(h_b1==0)[0]
        h_b1[:] = 0
        if n1 > N0//2:
            h_b1[np.random.choice(oi, N0//2, replace=False)] = 1
        else:
            h_b1[oi] = 1
            h_b1[np.random.choice(zi, N0//2-n1, replace=False)] = 1
    h_s1, bc1 = bin_search_full(h_b1, N0, d1, max_sweeps=10000, anneal_t0=0.0)
    c5_1 = compute_c5(h_s1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h_s1
    
    # Phase 2: Soft-max continuous optimization
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 200)
        hc = (best_h > 0.5).astype(float).copy()
        hc += np.random.randn(N0) * 0.03
        hc = project_sum(hc, N0/2.0)
        hn, c5n = soft_max_opt(hc, N0, d2, lr0=0.025, k_soft=10)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    
    # Phase 3: Binary search from continuous result
    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 150)
        h_b3 = (best_h > 0.5).astype(float)
        n3 = int(h_b3.sum())
        if n3 != N0//2:
            oi, zi = np.where(h_b3==1)[0], np.where(h_b3==0)[0]
            h_b3[:] = 0
            if n3 > N0//2:
                h_b3[np.random.choice(oi, N0//2, replace=False)] = 1
            else:
                h_b3[oi] = 1
                h_b3[np.random.choice(zi, N0//2-n3, replace=False)] = 1
        h_s3, bc3 = bin_search_full(h_b3, N0, d3, max_sweeps=10000, anneal_t0=0.0)
        c5_3 = compute_c5(h_s3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3
    
    # Phase 4: Block perturbation + local search (escape local optima)
    if time.time() < deadline - 100:
        d4 = min(deadline - 40, time.time() + 70)
        h_cur = best_h.copy()
        for attempt in range(6):
            if time.time() > d4:
                break
            hp = block_perturb((h_cur > 0.5).astype(float), N0, n_blocks=1, max_len=25)
            sub_dl = min(d4, time.time() + 12)
            h_pt, bc_pt = bin_search_full(hp, N0, sub_dl, max_sweeps=3000, anneal_t0=0.05, anneal_rate=0.98)
            c5_pt = compute_c5(h_pt)
            if c5_pt < best_c5:
                best_c5, best_h = c5_pt, h_pt
                h_cur = h_pt.copy()
    
    # Phase 5: Try different n values
    for N_try in [360, 400, 340]:
        if time.time() > deadline - 30:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == N0:
            continue
        x_best = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        nh = N_try // 2
        n1t = int(h_bt.sum())
        if n1t != nh:
            oi, zi = np.where(h_bt==1)[0], np.where(h_bt==0)[0]
            h_bt[:] = 0
            if n1t > nh:
                idx = oi[np.argsort(h_interp[oi])[:-(n1t-nh)]]
                h_bt[oi] = 1; h_bt[idx] = 0
            else:
                h_bt[oi] = 1
                h_bt[np.random.choice(zi, nh-n1t, replace=False)] = 1
        sub_dl = min(deadline - 15, time.time() + 25)
        h_rt, bc_rt = bin_search_full(h_bt, N_try, sub_dl, max_sweeps=5000, anneal_t0=0.0)
        c5_rt = compute_c5(h_rt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
    
    # Phase 6: Final soft-max polish + binary cleanup
    if time.time() < deadline - 8:
        Nf = len(best_h)
        hF = (best_h > 0.5).astype(float).copy()
        hF += np.random.randn(Nf) * 0.015
        hF = project_sum(hF, Nf/2.0)
        hF, c5F = soft_max_opt(hF, Nf, deadline - 3, lr0=0.015, k_soft=6, max_iters=2000)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    
    # Final binary snap
    hF2 = (best_h > 0.5).astype(float)
    nF2 = int(hF2.sum())
    if nF2 != len(hF2)//2:
        oi, zi = np.where(hF2==1)[0], np.where(hF2==0)[0]
        hF2[:] = 0
        if nF2 > len(hF2)//2:
            idx = oi[np.argsort(best_h[oi])[:-(nF2-len(hF2)//2)]]
            hF2[oi] = 1; hF2[idx] = 0
        else:
            hF2[oi] = 1
            hF2[np.random.choice(zi, len(hF2)//2-nF2, replace=False)] = 1
    hF2, bcF2 = bin_search_full(hF2, len(hF2), deadline - 0.5, max_sweeps=200, anneal_t0=0.0)
    c5F2 = compute_c5(hF2)
    if c5F2 < best_c5:
        best_c5, best_h = c5F2, hF2
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
