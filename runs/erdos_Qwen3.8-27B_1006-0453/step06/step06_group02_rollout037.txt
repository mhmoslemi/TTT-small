    target = N / 2.0
    dx = 2.0 / N

    best_pk = float(np.max(fft_corr(h)))

    lr = lr0
    beta = 0.9
    
    while time.time() < deadline:
        y = project_sum(h + m, target)
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Subgradient of max component
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - (N - 1))
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N - L] += (1.0 - y[L:])
                grad[L:] -= y[:N - L]
            else:

                    grad[M:] += (1.0 - y[:N - M])
                    grad[:N - M] -= y[M:]

        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        lr *= 0.998
    
    return best_h, best_pk * dx

    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_search(h, N, deadline, max_sweeps=600, sample_n=180, anneal_start=150):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        if len(ones) > sample_n:
            perm = rng.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
        else:
            p_list = ones
        best_score = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            # Try a few random alternatives for diversity
            n_rand = min(5, len(zeros) - 1)
            for _ in range(n_rand):
                rj = int(rng.integers(len(zeros)))
                if rj != mi:
                    v = float(cand[rj, :].max())
                    if v < npk[mi]:
                        mi = rj
            if npk[mi] < best_score:
                best_score = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        # Double-swap escape
        use_kick = (sweep > anneal_start and no_improve > 60 and best_pq is None) or \
                   (sweep > anneal_start and no_improve > 100 and rng.random() < 0.08)
        if best_pq is None or use_kick:
            if len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.4:
                p1 = int(ones[rng.integers(len(ones))])
                p2 = int(ones[rng.integers(len(ones))])
                q1 = int(zeros[rng.integers(len(zeros))])
                q2 = int(zeros[rng.integers(len(zeros))])
                if p1 != p2 and q1 != q2:
                    h[p1], h[p2] = 0.0, 0.0
                    h[q1], h[q2] = 1.0, 1.0
                else:
                    h[p1], h[q1] = 0.0, 1.0
            else:
                p = int(ones[rng.integers(len(ones))])
                q = int(zeros[rng.integers(len(zeros))])
                h[p], h[q] = 0.0, 1.0
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return h, best_c

def nesterov_opt(h, N, deadline, k=40, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = project_sum(h + m, target)
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - (N - 1))
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N - L] += (1.0 - y[L:])
                grad[L:] -= y[:N - L]
            else:
                M = -L
                if M < N:
                    grad[M:] += (1.0 - y[:N - M])
                    grad[:N - M] -= y[M:]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.996
            beta = min(0.96, 0.88 + it * 0.0002)
    return best_h, best_pk * dx

def fix_sum(h, N):
    h = h.copy()
    n1 = int(h.sum())
    nh = N // 2
    if n1 > nh:
        idx = np.where(h == 1)[0]
        n_rm = n1 - nh
        perm = np.random.permutation(len(idx))[:n_rm]
        h[idx[perm]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        n_add = nh - n1
        perm = np.random.permutation(len(idx))[:n_add]
        h[idx[perm]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary search at N0 (intensive)
    d1 = min(deadline - 450, t0 + 250)
    h_bin = (h0 > 0.5).astype(float)
    h_bin = fix_sum(h_bin, N0)
    h_s, _ = binary_search(h_bin, N0, d1, max_sweeps=800, sample_n=180, anneal_start=120)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov continuous exploration
    if time.time() < deadline - 350:
        d2 = min(deadline - 330, time.time() + 130)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=50, lr0=0.018)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search from continuous result
    if time.time() < deadline - 250:
        h_bin3 = make_binary(best_h, N0)
        d3 = min(deadline - 230, time.time() + 130)
        h_s3, _ = binary_search(h_bin3, N0, d3, max_sweeps=600, sample_n=200, anneal_start=100)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Coarse re-exploration at n=30 for global structure
    if time.time() < deadline - 200:
        N_c = 30
        x_0 = np.linspace(0, 2, N0, endpoint=False)
        x_c = np.linspace(0, 2, N_c, endpoint=False)
        best_c_coarse = 1.0
        best_h_coarse = None
        d4 = min(deadline - 180, time.time() + 80)
        per_trial = max(15, int((d4 - time.time()) / 4))
        for trial in range(4):
            if time.time() > d4 - 10:
                break
            h_init = np.interp(x_c, x_0, best_h)
            h_c = make_binary(h_init + np.random.randn(N_c) * 0.15, N_c)
            d_t = min(d4 - 5, time.time() + per_trial)
            h_c, _ = binary_search(h_c, N_c, d_t, max_sweeps=300, sample_n=30, anneal_start=80)
            cc = compute_c5(h_c, N_c)
            if cc < best_c_coarse:
                best_c_coarse = cc
                best_h_coarse = h_c.copy()
        if best_h_coarse is not None:
            h_up = np.interp(x_0, x_c, best_h_coarse)
            h_up_b = make_binary(h_up, N0)
            c5_up = compute_c5(h_up_b, N0)
            if c5_up < best_c5:
                best_c5, best_h = c5_up, h_up_b.copy()

    # Phase 5: Upscale to 2x (760)
    N_up = N0 * 2
    if time.time() < deadline - 100:
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        h_bt = fix_sum(h_bt, N_up)
        d5 = min(deadline - 50, time.time() + 80)
        h_rt, _ = binary_search(h_bt, N_up, d5, max_sweeps=400, sample_n=120, anneal_start=100)
        c5_rt = compute_c5(h_rt, N_up)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final Nesterov polish
    Nf = len(best_h)
    if time.time() < deadline - 15:
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.015
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_opt(hF, Nf, deadline - 8, k=60, lr0=0.01)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 7: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = make_binary(best_h, len(best_h))
        hF2, _ = binary_search(hF2, len(best_h), deadline - 0.5, max_sweeps=40, sample_n=150, anneal_start=50)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
