            break

        # B_k[q] for q in S0: B_k[q] = (1-h[q-k]) - h[q+k]

        B = np.zeros((len(ks), n0))

            vm = (pm >= 0) & (pm < N)

        term = c_top[:, None, None] + A[:, :, None] + B[:, None, :]

            # Remove q from S0, add p to S0
            S1 = np.delete(S1, p_idx)
            S1 = np.insert(S1, np.searchsorted(S1, q), q)

                c = fft_corr(h)
            else:
                break

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N=None):
    if N is None:
        N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    nh = N // 2
    idx = np.argsort(-h, kind='stable')
    out = np.zeros(N)
    out[idx[:nh]] = 1.0
    return out

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def greedy_descend(h, N, top_k=40, deadline=1e9, rng=None, max_iter=100000):
    if rng is None:
        rng = np.random.RandomState(0)
    h = make_binary(h, N)
    S1 = np.where(h == 1)[0].copy()
    S0 = np.where(h == 0)[0].copy()
    n1, n0 = len(S1), len(S0)
    c = fft_corr(h)
    best_max = float(c.max())
    best_h = h.copy()
    stall = 0
    tk = min(top_k, 2 * N - 1)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        top_idx = np.argpartition(-c, tk)[-tk:]
        ks = top_idx - (N - 1)
        A = np.empty((tk, n1))
        B = np.empty((tk, n0))
        for ki in range(tk):
            k = int(ks[ki])
            pp = S1 + k; pm = S1 - k
            A[ki] = (np.where((pp>=0)&(pp<N), h[np.clip(pp,0,N-1)], 0.) -
                     np.where((pm>=0)&(pm<N), 1.-h[np.clip(pm,0,N-1)], 0.))
            qp = S0 + k; qm = S0 - k
            B[ki] = (np.where((qm>=0)&(qm<N), 1.-h[np.clip(qm,0,N-1)], 0.) -
                     np.where((qp>=0)&(qp<N), h[np.clip(qp,0,N-1)], 0.))
        c_top = c[top_idx]
        nm = (c_top[:,None,None] + A[:,:,None] + B[:,None,:]).max(axis=0)
        fi = int(np.argmin(nm))
        pi, qi = fi // n0, fi % n0
        if nm[pi, qi] < best_max - 1e-12:
            p, q = int(S1[pi]), int(S0[qi])
            h[p], h[q] = 0., 1.
            S1 = np.delete(S1, pi)
            S1 = np.insert(S1, int(np.searchsorted(S1, q)), q)
            S0 = np.delete(S0, qi)
            S0 = np.insert(S0, int(np.searchsorted(S0, p)), p)
            c = fft_corr(h)
            cm = float(c.max())
            if cm < best_max:
                best_max, best_h = cm, h.copy()
            stall = 0
        else:
            stall += 1
            if stall < 25 and rng.random() < 0.7:
                pi2, qi2 = rng.randint(n1), rng.randint(n0)
                p, q = int(S1[pi2]), int(S0[qi2])

                S0 = np.delete(S0, qi2)
                S0 = np.insert(S0, int(np.searchsorted(S0, p)), p)
                c = fft_corr(h)
            else:
                break
    return best_h, best_max

def nesterov_opt(h, N, deadline, k=60, lr0=0.015, beta0=0.90):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr, beta, it = lr0, beta0, 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        center = N - 1
        tk = min(k, len(corr))
        top = np.argpartition(corr, -tk)[-tk:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.992
            beta = min(0.97, beta0 + it * 0.0002)
    return best_h, best_pk * dx
