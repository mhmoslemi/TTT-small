def fft_corr(h):

    while L < 2*N-1:

    G = np.fft.rfft(np.flip(1.0-h).astype(np.float64), L)
    return np.fft.irfft(H*G, L)[:2*N-1]

def proj_sum(h, target):

    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2

    G = np.fft.rfft(np.flip(1.0-h).astype(np.float64), L)

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h

    s = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    ps = p + s; pm = p - s
    qs = q + s; qm = q - s
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d[vps] -= (1.0-h[ps[vps]])
    d[vpm] += h[pm[vpm]]
    d[vqs] += (1.0-h[qs[vqs]])
    d[vqm] -= h[qm[vqm]]
    return d

            d_p[vps] -= (1.0-h[ps[vps]])
            d_p[vpm] += h[pm[vpm]]
            base = corr + d_p

            best_q = -1

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)[:2*N-1]

        mid = (lo+hi)*0.5

        else:
