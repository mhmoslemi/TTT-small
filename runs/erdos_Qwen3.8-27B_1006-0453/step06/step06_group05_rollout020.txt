import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

    best_h = h.copy()

        grad = D.T @ w  # shape (N,)
        grad -= grad.mean()
        # Adam update

        bc2 = 1 - beta2 ** t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2

            n >>= 1
        tm[i] = 1.0 if c % 2 == 0 else 0.0

    elif n1 < nh:
        zeros = np.where(tm == 0)[0]
        np.random.shuffle(zeros)
        tm[zeros[:nh-n1]] = 1
    return tm

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def fix_sum_binary(h, N):
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 == nh:
        return (h > 0.5).astype(float)
    hb = np.zeros(N)
    if n1 > nh:
        oi = np.where(h > 0.5)[0]
        hb[oi[:nh]] = 1.0
    else:
        oi = np.where(h > 0.5)[0]
        hb[oi] = 1.0
        zi = np.where(h <= 0.5)[0]
        hb[zi[:(nh - n1)]] = 1.0
    return hb

def bin_search(h, N, deadline, max_sweeps=600, anneal=0.0, jump_mag=4):
    h = h.astype(float).copy()
    h = fix_sum_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 100 and anneal == 0:
            break
        if stall > 50 and anneal > 0:
            # Jump: flip multiple random pairs
            for _ in range(jump_mag):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                pi = int(np.random.randint(len(ones)))
                qi = int(np.random.randint(len(zeros)))
                h[ones[pi]], h[zeros[qi]] = 0.0, 1.0
            stall = 0
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        bd, bpq = -1e-15, None
        sample_sz = min(len(ones), 100)
        if sample_sz < len(ones):
            p_list = np.random.choice(ones, sample_sz, replace=False)
        else:
            p_list = ones
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))

            h[bpq[1]] = 1.0

            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0

        else:

def softmax_smooth(h, N, deadline, T0=3.0, lr0=0.035, min_T=0.2):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    beta1, beta2, eps = 0.82, 0.998, 1e-8
    lr = lr0
    T_cur = T0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        pk_dx = pk * dx
        if pk_dx < best_pk:
            best_pk = pk_dx
            best_h = h.copy()
        T_cur = max(min_T, T0 * (0.997 ** t))
        c_safe = c.copy()
        c_safe[c_safe < 0] = 0
        c_norm = c_safe / T_cur
        c_max = c_norm.max()
        exp_c = np.exp(c_norm - c_max)
        w = exp_c / (exp_c.sum() + 1e-30)
        D = compute_D(h, N)
        grad = D.T @ w
        grad -= grad.mean()
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        h_new = h + lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = project_sum(h_new, target)
        if t % 150 == 0:
            lr *= 0.91
        if t % 300 == 0:
            hb = (h > 0.5).astype(float)
            n1 = int(hb.sum())
            if n1 != N // 2:
                ones_i = np.where(hb == 1)[0]
                zeros_i = np.where(hb == 0)[0]
                if n1 > N // 2:
                    worst = ones_i[np.argsort(h[ones_i])[:-(n1 - N // 2)]]
                    hb[worst] = 0
                else:
                    best = zeros_i[np.argsort(-h[zeros_i])[:(N // 2 - n1)]]
                    hb[best] = 1
            h = 0.6 * h + 0.4 * hb
    return best_h, best_pk

def thue_morse_init(N):
    tm = np.zeros(N, dtype=float)
    for i in range(N):
        cnt = bin(i).count('1')
        tm[i] = 1.0 if cnt % 2 == 0 else 0.0
    return fix_sum_binary(tm, N)

def periodic_init(N, period):
    h = np.zeros(N)
    half = period // 2
    for i in range(N):
        h[i] = 1.0 if (i % period) < half else 0.0
    return fix_sum_binary(h, N)
