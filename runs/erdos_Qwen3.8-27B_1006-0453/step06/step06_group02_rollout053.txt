def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)

def build_delta_remove(h, N):
    """Build the delta_remove matrix: effect of changing h[p] from 1 to 0 on c_full."""
    L = 2 * N - 1
    D = np.zeros((N, L))
    h_rev = h[N-1::-1]  # h reversed
    for p in range(N):
        # c_full[N-1-p : 2N-1-p] -= (1-h[0:N-p])
        D[p, N-1-p : 2*N-1-p] = -(1.0 - h[0 : N-p])
        # c_full[p : N+p] += h_reversed
        D[p, p : N+p] += h_rev
    return D

def swap_delta(D, p, q):
    """Delta for moving 1 from p to q."""
    return D[p] - D[q]
