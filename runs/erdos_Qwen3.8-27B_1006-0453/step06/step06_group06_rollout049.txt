import numpy as np
import time

def full_corr(h):
    """Compute full linear cross-correlation h ⊛ (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")

def proj_sum(h, target):
    """Project h onto {[0,1]^N : sum(h) = target}."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def fix_binary(h, N):
    """Ensure exactly N/2 ones."""
    target = N // 2
    h = (h > 0.5).astype(float).copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        chosen = np.random.choice(idx, n1 - target, replace=False)
        h[chosen] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        chosen = np.random.choice(idx, target - n1, replace=False)
        h[chosen] = 1
    return h

def grad_max_corr(h, top_k=50, weight_decay=0.7):
    """Compute subgradient of max-shift correlation, weighted by rank."""
    N = len(h)
    corr = full_corr(h)
    center = N - 1
    # Select top_k shifts
    idx = np.argpartition(corr, -top_k)[-top_k:]
    weights = np.ones(top_k)
    # Exponential weighting: worse shifts get more weight
    sorted_corr = corr[idx][np.argsort(corr[idx])[::-1]]
    weights = weight_decay ** np.arange(top_k)
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for w, ai in zip(weights, idx):
        s = int(ai) - center
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            # corr = sum_{i=0}^{N-1-s} h[i+s]*(1-h[i])
            grad[s:] += w * (1.0 - h[:N-s])  # h[j] as first factor: d = (1-h[j-s])
            grad[:N-s] -= w * h[s:]  # h[j] as second factor: d = -h[j+s]
        else:
            s2 = -s
            # corr = sum_{i=s2}^{N-1} h[i-s2]*(1-h[i])
            grad[:N-s2] += w * (1.0 - h[s2:])  # h[j] as first: d = (1-h[j+s2]) for j=0..N-1-s2
            grad[s2:] -= w * h[:N-s2]  # h[j] as second: d = -h[j-s2] for j=s2..N-1
    return grad / top_k, corr.max()

    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    corr = full_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    lr = lr0

import numpy as np
import time

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def fix_binary(h, N):
    target = N // 2
    h = (h > 0.5).astype(float).copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[idx[np.random.choice(len(idx), n1 - target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[idx[np.random.choice(len(idx), target - n1, replace=False)]] = 1
    return h

def compute_grad(h, top_k=50, wdecay=0.72):
    N = len(h)
    corr = full_corr(h)
    center = N - 1
    pk = float(corr.max())
    idx = np.argpartition(corr, -top_k)[-top_k:]
    order = np.argsort(corr[idx])[::-1]
    idx = idx[order]
    wts = wdecay ** np.arange(top_k)
    wts /= wts.sum()
    grad = np.zeros(N)
    for w, ai in zip(wts, idx):
        s = int(ai) - center
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w * (1.0 - h[:N - s])
            grad[:N - s] -= w * h[s:]
        else:
            s2 = -s
            grad[:N - s2] += w * (1.0 - h[s2:])
            grad[s2:] -= w * h[:N - s2]
    return grad / top_k, pk

def continuous_opt(h, N, deadline, top_k=45, lr0=0.006, wdecay=0.72):
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    corr = full_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    lr = lr0
    while time.time() < deadline:
        grad, pk = compute_grad(h, top_k, wdecay)
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = proj_sum(h - lr * mh / (np.sqrt(vh) + eps), N / 2.0)
        if t % 600 == 0:
            lr *= 0.95
    return best_h, best_pk

def binary_anneal(h, N, deadline, T0=0.055, cool=0.9994, max_no_imp=800, n_restarts=3):
    h = fix_binary(h, N)
    corr = full_corr(h).copy()
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    s_arr = np.arange(-(N - 1), N)
    for restart in range(n_restarts):
        T = T0 if restart == 0 else T0 * 0.45
        no_imp = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        while time.time() < deadline:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len_zeros if 'len_zeros' in dir() else len(zeros))])
            ps = p + s_arr
            pm = p - s_arr
            qs = q + s_arr
            qm = q - s_arr
            vps = (ps >= 0) & (ps < N)
            vpm = (pm >= 0) & (pm < N)
            vqs = (qs >= 0) & (qs < N)
            vqm = (qm >= 0) & (qm < N)
            d = np.zeros(2 * N - 1)
            pc = np.clip(ps, 0, N - 1)
            pmc = np.clip(pm, 0, N - 1)
            qc = np.clip(qs, 0, N - 1)
            qmc = np.clip(qm, 0, N - 1)
            d[vps] -= (1.0 - h[pc[vps]])
            d[vpm] += h[pmc[vpm]]
            d[vqs] += (1.0 - h[qc[vqs]])
            d[vqm] -= h[qmc[vqm]]
            nc = corr + d
            npk = float(nc.max())
            delta = npk - cur_pk
            if delta < 0 or (T > 1e-6 and np.random.random() < np.exp(-delta / max(T, 1e-8))):
                h[p] = 0
                h[q] = 1
                corr = nc
                cur_pk = npk
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if npk < best_pk:
                    best_pk = npk
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:

            T *= cool
            if no_imp > max_no_imp:
                if restart < n_restarts - 1:
                    break
                else:
                    h = best_h.copy()
                    corr = full_corr(h).copy()
                    cur_pk = best_pk
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    no_imp = 0
                    T = T0 * 0.3
    return best_h, best_pk
