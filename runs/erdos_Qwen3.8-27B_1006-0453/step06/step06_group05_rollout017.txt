        if not improved:
            stall += 1
    return h, float(np.max(fft_corr(h)))

def make_staircase(N, alpha, beta):
    x = np.arange(N)
    return ((x * alpha + beta) % 1.0 < 0.5).astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Continuous optimization on initial (35% budget)
    d1 = t0 + budget_s * 0.30
    hc = best_h.copy()
    hc += np.random.randn(N0) * 0.01
    hc = project_sum(hc, N0 / 2.0)
    hn, c5n = cont_opt(hc, N0, d1, base_lr=0.045)
    if c5n < best_c5:
        best_c5, best_h = c5n, hn.copy()

    # Phase 2: Binary search from continuous result (25% budget)
    d2 = t0 + budget_s * 0.52
    h_b2 = (best_h > 0.5).astype(float)
    n1 = int(h_b2.sum())
    if n1 != N0 // 2:
        oi = np.where(h_b2 == 1)[0]
        zi = np.where(h_b2 == 0)[0]
        h_b2[:] = 0
        if n1 > N0 // 2:
            h_b2[oi[np.argsort(-best_h[oi])[:N0 // 2]]] = 1
        else:
            h_b2[oi] = 1
            h_b2[zi[np.argsort(-best_h[zi])[:N0 // 2 - n1]]] = 1
    h_s2, _ = bin_search(h_b2, N0, d2, max_sweeps=2000, anneal=0.0)
    c5_2 = compute_c5(h_s2, N0)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h_s2.copy()

    # Phase 3: Continuous from binary + re-search (20% budget)
    d3 = t0 + budget_s * 0.72
    hc3 = (best_h > 0.5).astype(float).copy()
    hc3 += np.random.randn(N0) * 0.015
    hc3 = project_sum(hc3, N0 / 2.0)
    hn3, c5n3 = cont_opt(hc3, N0, d3, base_lr=0.035)
    if c5n3 < best_c5:
        best_c5, best_h = c5n3, hn3.copy()
    h_b3 = (best_h > 0.5).astype(float)
    h_s3, _ = bin_search(h_b3, N0, min(d3 + 30, deadline - 100), max_sweeps=500)
    c5_3 = compute_c5(h_s3, N0)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Block moves (10% budget)
    d4 = min(deadline - 80, t0 + budget_s * 0.88)
    h_b4 = (best_h > 0.5).astype(float)
    h_s4, _ = block_move_search(h_b4, N0, d4, max_sweeps=200)
    c5_4 = compute_c5(h_s4, N0)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h_s4.copy()

    # Phase 5: Multi-start with different N and perturbations (12% budget)
    d5 = min(deadline - 30, t0 + budget_s * 0.97)
    # Try perturbations at same N
    for attempt in range(5):
        if time.time() > d5:
            break
        np.random.seed(seed + 1000 * attempt)
        hp = (best_h > 0.5).astype(float).copy()
        n_flip = np.random.randint(4, 10)
        oi = np.where(hp == 1)[0]
        zi = np.where(hp == 0)[0]
        for _ in range(n_flip):
            p = int(np.random.randint(len(oi)))
            q = int(np.random.randint(len(zi)))
            hp[oi[p]], hp[zi[q]] = 0.0, 1.0
            oi = np.where(hp == 1)[0]
            zi = np.where(hp == 0)[0]
        sub_dl = min(d5, time.time() + 12)
        h_pt, _ = bin_search(hp, N0, sub_dl, max_sweeps=200, anneal=0.05)
        c5_pt = compute_c5(h_pt, N0)
        if c5_pt < best_c5:
            best_c5, best_h = c5_pt, h_pt.copy()

    # Try different N values
    for N_try in [400, 360, 420]:
        if time.time() > d5:
            break
        if N_try == N0 or N_try % 2 != 0:
            continue
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        nht, n1t = N_try // 2, int(h_bt.sum())
        if n1t > nht:
            idx = np.where(h_bt == 1)[0]
            idx = idx[np.argsort(h_interp[idx])[:-(n1t - nht)]]
            h_bt[idx] = 0
        elif n1t < nht:
            idx = np.where(h_bt == 0)[0]
            idx = idx[np.argsort(-h_interp[idx])[:nht - n1t]]
            h_bt[idx] = 1
        sub_dl = min(d5, time.time() + 35)
        h_rt, _ = bin_search(h_bt, N_try, sub_dl, max_sweeps=400)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final continuous polish
    if time.time() < deadline - 15:
        hF = (best_h > 0.5).astype(float).copy()
        hF += np.random.randn(len(hF)) * 0.008
        hF = project_sum(hF, len(hF) / 2.0)
        hF, c5F = cont_opt(hF, len(hF), deadline - 8, base_lr=0.02)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = (best_h > 0.5).astype(float)
        hF2, _ = bin_search(hF2, len(best_h), deadline - 0.5, max_sweeps=100)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
