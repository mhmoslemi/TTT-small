def swap_improve(h, N, max_sweeps, deadline, seed):
    """Local search by swaps for binary h."""
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full')
    best_max = corr.max()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        best_gain = 0
        best_swap = None
        
        # Sample a subset of pairs for speed
        ones_list = list(ones)
        zeros_list = list(zeros)
        np.random.shuffle(ones_list)
        np.random.shuffle(zeros_list)
        
        # Try up to 200 random pairs
        for _ in range(200):
            i = ones_list[np.random.randint(len(ones_list))]
            j = zeros_list[np.random.randint(len(zeros_list))]
            h[i], h[j] = 0, 1
            new_corr = np.correlate(h, 1-h, mode='full')
            new_max = new_corr.max()
            gain = best_max - new_max
            if gain > best_gain:
                best_gain = gain
                best_swap = (i, j)
                best_corr = new_corr
            h[i], h[j] = 1, 0  # undo
        
        if best_swap:
            i, j = best_swap
            h[i], h[j] = 0, 1
            best_max = best_max - best_gain
            ones.remove(i); ones.add(j)
            zeros.remove(j); zeros.add(i)
        else:
            # No improvement found, try a random swap (annealing)
            if np.random.random() < 0.3:
                i = ones_list[np.random.randint(len(ones_list))]
                j = zeros_list[np.random.randint(len(zeros_list))]
                h[i], h[j] = 0, 1
                best_max = np.correlate(h, 1-h, mode='full').max()
                ones.remove(i); ones.add(j)
                zeros.remove(j); zeros.add(i)
    
    return h, best_max
