def local_search(h, N, deadline, max_sweeps=5000, n_sample_p=30, n_sample_q=30):
    h = h.copy()
    c = fft_corr(h)
    best_c5 = float(c.max()) * (2.0/N)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Sample candidates
        if len(ones) > n_sample_p:
            p_cands = ones[np.random.choice(len(ones), n_sample_p, replace=False)]
        else:
            p_cands = ones
        if len(zeros) > n_sample_q:
            q_cands = zeros[np.random.choice(len(zeros), n_sample_q, replace=False)]
        else:
            q_cands = zeros
        
        best_score = c.max() + 1.0
        best_move = None
        
        for p in p_cands:
            # Compute effect of flipping p from 1 to 0
            center = N - 1
            j = np.arange(2*N-1)
            i1 = p + j - center
            i2 = p - j + center
            v1 = (i1 >= 0) & (i1 < N)
            v2 = (i2 >= 0) & (i2 < N)
            delta_p = np.zeros(2*N-1)
            delta_p[v1] -= (1.0 - h[i1[v1]])
            delta_p[v2] += h[i2[v2]]
            
            base = c + delta_p  # correlation after flipping p to 0
            
            for q in q_cands:
                i1q = q + j - center
                i2q = q - j + center
                v1q = (i1q >= 0) & (i1q < N)
                v2q = (i2q >= 0) & (i2q < N)
                delta_q = np.zeros(2*N-1)
                delta_q[v1q] += (1.0 - h[i1q[v1q]])
                delta_q[v2q] -= h[i2q[v2q]]
                
                new_c = base + delta_q
                score = new_c.max()
                if score < best_score:
                    best_score = score
                    best_move = (int(p), int(q))
        
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            c = fft_corr(h)
            if c.max() < best_c5 - 1e-10:
                best_c5 = float(c.max()) * (2.0/N)
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Random kick
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0.0
            h[q] = 1.0
            c = fft_corr(h)
            no_improve += 1
        
        if no_improve > 500:
            break
    
    return h, best_c5

        dq_mat = np.zeros((len(q_cands), 2*N-1))

            i2 = q - j + center
            v1 = (i1 >= 0) & (i1 < N)
            v2 = (i2 >= 0) & (i2 < N)

            v2 = (i2 >= 0) & (i2 < N)

            delta_p[v1] = -(1.0 - h[i1[v1]])
            delta_p[v2] += h[i2[v2]]
            
            base = c + delta_p
            new_cs = base[None, :] + dq_mat  # (n_q, 2N-1)
            scores = new_cs.max(axis=1)
            mi = int(np.argmin(scores))
            if scores[mi] < best_score:
                best_score = float(scores[mi])
                best_move = (int(p), int(q_cands[mi]))
        
        if best_move is not None and best_score < c.max():
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            c = fft_corr(h)
            if c.max() < best_val:
                best_val = float(c.max())
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Occasionally make a random move to escape
            if no_improve > 100:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p] = 0.0
                h[q] = 1.0
                c = fft_corr(h)
            else:
                no_improve += 1
                if no_improve > 500:
                    break
            no_improve += 1
    
    return h, best_val * (2.0/N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fix_binary_sum(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(np.sum(h >= 0.5))
    if n1 > nh:
        idx = np.where(h >= 0.5)[0]
        idx = idx[np.random.permutation(len(idx))][nh:]
        h[idx] = 0.0
    elif n1 < nh:
        idx = np.where(h < 0.5)[0]
        idx = idx[np.random.permutation(len(idx))]
        h[idx[:nh-n1]] = 1.0
    else:
        h[h >= 0.5] = 1.0
        h[h < 0.5] = 0.0
    return h

def flip_vec(h, N, p, sign):
    """Change in correlation when h[p] changes by `sign` (+1 or -1)."""
    # sign = -1 means h[p] goes from 1 to 0
    # sign = +1 means h[p] goes from 0 to 1
    center = N - 1
    L = 2 * N - 1
    j = np.arange(L)
    d = np.zeros(L)
    i1 = p + j - center  # where (1-h) is evaluated
    i2 = p - j + center  # where h is evaluated (the (1-h[p]) part)
    v1 = (i1 >= 0) & (i1 < N)
    v2 = (i2 >= 0) & (i2 < N)
    # h[p] changes by sign:
    # Term h[p]*(1-h[p+j-center]): changes by sign*(1-h[i1])
    # Term h[p-j+center]*(1-h[p]): changes by -sign*h[i2]
    if sign > 0:
        d[v1] += (1.0 - h[i1[v1]])
        d[v2] -= h[i2[v2]]
    else:
        d[v1] -= (1.0 - h[i1[v1]])
        d[v2] += h[i2[v2]]
    return d

def local_search(h, N, deadline, max_sweeps=20000, n_p=40, n_q=60, score_alpha=0.2):
    """Local search minimizing max + alpha * second_max of correlation."""
    h = h.astype(float).copy()
    c = fft_corr(h)
    L = 2 * N - 1
    center = N - 1
    j = np.arange(L)
    best_val = float(c.max())
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Sample candidates with some bias toward "boundary" positions
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        p_cands = ones[:n_p]
        q_cands = zeros[:n_q]
        
        # Precompute flip vectors for q candidates (0→1)
        dq_mat = np.zeros((len(q_cands), L))
        for qi in range(len(q_cands)):
            q = q_cands[qi]
            i1 = q + j - center
            i2 = q - j + center
            v1 = (i1 >= 0) & (i1 < N)
            v2 = (i2 >= 0) & (i2 < N)
            dq_mat[qi, v1] = (1.0 - h[i1[v1]])
            dq_mat[qi, v2] -= h[i2[v2]]
        
        best_score = best_val + 1.0
        best_move = None
        
        for p in p_cands:
            i1 = p + j - center
            i2 = p - j + center
            v1 = (i1 >= 0) & (i1 < N)
            v2 = (i2 >= 0) & (i2 < N)
            dp = np.zeros(L)
            dp[v1] = -(1.0 - h[i1[v1]])
            dp[v2] += h[i2[v2]]
            
            base = c + dp
            new_cs = base[None, :] + dq_mat
            # Score: max + alpha * second_max
            mx1 = new_cs.max(axis=1)
            # For second max, use top-2
            # Approximate: use mean of top 5 as proxy
            top5 = np.sort(new_cs, axis=1)[:, -5:]
            score = mx1 + score_alpha * top5.mean(axis=1)
            mi = int(np.argmin(score))
            if score[mi] < best_score:
                best_score = float(score[mi])
                best_move = (int(p), int(q_cands[mi]))
        
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            c = fft_corr(h)
            new_max = float(c.max())
            if new_max < best_val - 1e-10:
                best_val = new_max

            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 200:

                q = int(zeros[np.random.randint(len(zeros))])
                h[p] = 0.0
                h[q] = 1.0
                c = fft_corr(h)
            elif no_improve > 800:
                break
        
    return h, best_val * (2.0 / N)

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def fix_binary(h, N):
    h = h.copy()
    nh = N // 2
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1.0)[0]
        rm = idx[np.random.permutation(n1)][nh:]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0.0)[0]
        add = idx[np.random.permutation(n1)] if n1 < nh else idx
        h[idx[:nh-n1] if n1<nh else add[:nh-n1]] = 1.0
    return h

def int_local_search(h, N, deadline, max_sweeps=5000, n_p=30, n_q=50):
    h = h.astype(float).copy()
    c = fft_corr(h)
    L = 2*N-1
    center = N-1
    j = np.arange(L)
    best_val = float(c.max())
    no_imp = 0
    
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h>=0.5)[0]
        zeros = np.where(h<0.5)[0]
        if len(ones)<2 or len(zeros)<2:
            break
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        ps = ones[:n_p]
        qs = zeros[:n_q]
        
        # Precompute dq for all q
        dq = np.zeros((len(qs), L))
        for qi in range(len(qs)):
            q = qs[qi]
            i1 = q+j-center; i2 = q-j+center

            dq[qi,v1] = 1.0-h[i1[v1]]
            dq[qi,v2] -= h[i2[v2]]
        
        best_sc = best_val + 1.0
        best_mv = None
        for p in ps:
            i1 = p+j-center; i2 = p-j+center
            v1 = (i1>=0)&(i1<N); v2 = (i2>=0)&(i2<N)
            dp = np.zeros(L)
            dp[v1] = -(1.0-h[i1[v1]])
            dp[v2] += h[i2[v2]]
            base = c + dp
            nc = base[None,:] + dq
            sc = nc.max(axis=1)
            mi = int(np.argmin(sc))
            if sc[mi] < best_sc:
                best_sc = float(sc[mi])
                best_mv = (int(p), int(qs[mi]))
        
        if best_mv is not None and best_sc < c.max():
            h[best_mv[0]]=0; h[best_mv[1]]=1
            c = fft_corr(h)
            if c.max() < best_val:

                h[p]=0; h[q]=1; c=fft_corr(h)

def nesterov(h, N, deadline, k=40, lr=0.01):
    h = h.astype(float).copy()
    target = N/2.0
    c = fft_corr(h)
    best_h = h.copy(); best_pk = float(c.max())
    m = np.zeros(N)
    beta = 0.9
    for it in range(20000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N-1
        for ai in top:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0*y)
            elif s > 0 and s < N:
                grad[s:] += (1.0 - y[:N-s])
                grad[:N-s] -= y[s:]
            elif s < 0 and -s < N:
                M = -s
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= k
        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        if it % 200 == 0:
            lr *= 0.995
    return best_h, best_pk*(2.0/N)

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fix_binary(h, N):
    h = h.copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1.0)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[nh:]]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0.0)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[:nh - n1]]] = 1.0
    return h

def int_local_search(h, N, deadline, max_sweeps=5000, n_p=35, n_q=55, multi_weight=0.0):
    h = h.astype(float).copy()
    c = fft_corr(h)
    L = 2 * N - 1
    center = N - 1
    j = np.arange(L)
    best_val = float(c.max())
    no_imp = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        ps = ones[:n_p]
        qs = zeros[:n_q]
        dq = np.zeros((len(qs), L))
        for qi in range(len(qs)):
            q = qs[qi]
            i1 = q + j - center
            i2 = q - j + center
            v1 = (i1 >= 0) & (i1 < N)
            v2 = (i2 >= 0) & (i2 < N)
            dq[qi, v1] = 1.0 - h[i1[v1]]
            dq[qi, v2] -= h[i2[v2]]
        best_sc = best_val + 1.0
        best_mv = None
        for p in ps:
            i1 = p + j - center
            i2 = p - j + center
            v1 = (i1 >= 0) & (i1 < N)
            v2 = (i2 >= 0) & (i2 < N)
            dp = np.zeros(L)
            dp[v1] = -(1.0 - h[i1[v1]])
            dp[v2] += h[i2[v2]]
            base = c + dp
            nc = base[None, :] + dq
            sc = nc.max(axis=1)
            if multi_weight > 0:
                top5 = np.sort(nc, axis=1)[:, -5:]
                sc = sc + multi_weight * top5.mean(axis=1)
            mi = int(np.argmin(sc))
            if sc[mi] < best_sc:
                best_sc = float(sc[mi])
                best_mv = (int(p), int(qs[mi]))
        if best_mv is not None and best_sc < c.max() - 1e-12:
            h[best_mv[0]] = 0.0

def nesterov_opt(h, N, deadline, k=50, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    beta = 0.9
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * y)
            elif 0 < s < N:
                grad[:N - s] += (1.0 - y[s:])
                grad[s:] -= y[:N - s]
            elif -N < s < 0:
                M = -s
                grad[M:] += (1.0 - y[:N - M])
                grad[:N - M] -= y[M:]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)
