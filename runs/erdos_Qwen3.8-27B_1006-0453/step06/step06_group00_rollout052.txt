def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)

    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid

    """Compute subgradient of max_k corr[k] and take a projected step."""
    corr = fft_corr(h)
    # Find top-k peaks
    top_idx = np.argpartition(corr, -top_k)[-top_k:]

            grad[L:] += (1.0 - h[:N - L])

        elif L < 0 and -L < N:
            M = -L
            grad[:N - M] += (1.0 - h[M:])
            grad[M:] -= h[:N - M]

        hb[np.random.choice(idx, nh - n1, replace=False)] = 1
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Aggressive Nesterov from perturbed start
    if time.time() < deadline - 500:
        hc = best_h + np.random.randn(N0) * 0.1
        hc = project_01(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, min(deadline - 480, t0 + 130), k=80, lr0=0.012, beta0=0.85)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 2: Vectorized pair-swap on binary
    if time.time() < deadline - 380:
        hb = make_binary(best_h, N0)
        hb2, _ = vectorized_pair_search(hb, N0, min(deadline - 360, t0 + 260), n_samp=70, T0=0.02)
        c5b = compute_c5(hb2, N0)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb2.copy()

    # Phase 3: Second Nesterov round (different params)
    if time.time() < deadline - 260:
        hc2 = best_h + np.random.randn(N0) * 0.06
        hc2 = project_01(hc2, N0 / 2.0)
        hn2, c5n2 = nesterov_opt(hc2, N0, min(deadline - 240, t0 + 380), k=100, lr0=0.010, beta0=0.90)
        if c5n2 < best_c5:
            best_c5, best_h = c5n2, hn2.copy()

    # Phase 4: Deeper pair-swap with more samples
    if time.time() < deadline - 180:
        hb4 = make_binary(best_h, N0)
        hb4, _ = vectorized_pair_search(hb4, N0, min(deadline - 160, t0 + 520), n_samp=90, T0=0.012)
        c5_4 = compute_c5(hb4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, hb4.copy()

    # Phase 5: Multi-resolution search
    for N_try in [420, 480, 540, 600]:
        if time.time() > deadline - 120:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = make_binary(h_int, N_try)
        sub_dl = min(deadline - 60, time.time() + 70)
        h_rt, _ = vectorized_pair_search(h_bt, N_try, sub_dl, n_samp=60, T0=0.015)
        c5_rt = compute_c5(h_rt, N_try)
        # Also try Nesterov on this resolution
        if time.time() < deadline - 55:
            hc_rt = project_01(h_rt + np.random.randn(N_try) * 0.05, N_try / 2.0)
            hn_rt, c5n_rt = nesterov_opt(hc_rt, N_try, min(deadline - 50, time.time() + 30), k=60, lr0=0.01, beta0=0.88)
            if c5n_rt < c5_rt:
                c5_rt, h_rt = c5n_rt, hn_rt
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 10:
        hF = best_h + np.random.randn(len(best_h)) * 0.04
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 2, k=100, lr0=0.007, beta0=0.92)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final: try binary version
    hB = (best_h > 0.5).astype(float)
    c5B = compute_c5(hB, len(hB))
    if c5B < best_c5:
        best_h = hB
        best_c5 = c5B
    else:
        best_h = (best_h > 0.5).astype(float)
        best_c5 = compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
