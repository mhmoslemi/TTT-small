def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

    d2 = min(deadline - 570, d2)
    h_bin = (best_h > 0.5).astype(float)
    hb, c5b = binary_sa(h_bin, N0, d2, t_init=12.0, t_final=0.2, n_inner=60)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 3: Continuous from best binary (180s)
    d3 = max(d2, t0) + 180
    d3 = min(deadline - 350, d3)
    hc3 = best_h.astype(float) + np.random.randn(N0) * 0.02
    hc3 = project_sum(hc3, N0 / 2.0)
    hn3, c5n3 = continuous_opt(hc3, N0, d3, k=110, lr0=0.016, temp0=0.002)
    if c5n3 < best_c5:
        best_c5, best_h = c5n3, hn3.copy()

    # Phase 4: Binary SA round 2 (130s)
    d4 = max(d3, t0) + 130
    d4 = min(deadline - 170, d4)
    h_bin2 = (best_h > 0.5).astype(float)
    hb2, c5b2 = binary_sa(h_bin2, N0, d4, t_init=8.0, t_final=0.15, n_inner=40)
    if c5b2 < best_c5:
        best_c5, best_h = c5b2, hb2.copy()

    # Phase 5: Coordinate descent (120s)
    d5 = max(d4, t0) + 120
    d5 = min(deadline - 60, d5)
    hc5 = best_h.astype(float)
    hc5 = np.clip(hc5, 0.0, 1.0)
    hc5 = project_sum(hc5, N0 / 2.0)
    hcd, c5cd = coord_decent(hc5, N0, d5, n_sweeps=80)
    if c5cd < best_c5:
        best_c5, best_h = c5cd, hcd.copy()

    # Phase 6: Quick binary from best continuous
    if time.time() < deadline - 30:
        d6 = deadline - 5
        h_bin3 = (best_h > 0.5).astype(float)
        hb3, c5b3 = binary_sa(h_bin3, N0, d6, t_init=5.0, t_final=0.1, n_inner=30)
        if c5b3 < best_c5:
            best_c5, best_h = c5b3, hb3.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, final_c5, len(best_h)
