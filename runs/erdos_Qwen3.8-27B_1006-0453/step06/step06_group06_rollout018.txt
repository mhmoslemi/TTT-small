```python
def fft_corr(h):
    """Compute full correlation using FFT."""
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]
```

```python
def swap_delta_binary(h, N, p, q):
    s_arr = np.arange(-(N-1), N)  # s from -(N-1) to N-1
    d = np.zeros(2*N-1)
    
    # p: 1->0
    # Term -（1-h[p-s]): valid when 0<=p-s<N, i.e., p-N+1<=s<=p
    # In s_arr, position of value s is s+N-1
    # p-s ranges 0..N-1 means s ranges p-N+1..p
    # Direct: for each valid s, d[s+N-1] += -(1-h[p-s])
    # p-s in [0,N-1] means s in [p-N+1, p]
    sps = p - s_arr  # = p-s for each s in s_arr
    v_sps = (sps >= 0) & (sps < N)
    d[v_sps] -= (1.0 - h[sps[v_sps]])
    
    sps2 = p + s_arr  # = p+s for each s
    v_sps2 = (sps2 >= 0) & (sps2 < N)
    d[v_sps2] += h[sps2[v_sps2]]
    
    # q: 0->1
    sqs = q - s_arr
    v_sqs = (sqs >= 0) & (sqs < N)
    d[v_sqs] += (1.0 - h[sqs[v_sqs]])
    
    sqs2 = q + s_arr
    v_sqs2 = (sqs2 >= 0) & (sqs2 < N)
    d[v_sqs2] -= h[sqs2[v_sqs2]]
    
    return d
```
