def fft_corr(h):

    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:

        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_direction(y, corr, N, T=0.1):
    """Compute subgradient with soft-max weighting over top correlations."""
    center = N - 1
    # Use temperature-weighted selection of lags
    c_shifted = corr - corr.max()
    weights = np.exp(c_shifted / T)
    weights /= weights.sum()
    
    grad = np.zeros(N)
    # For each lag, add gradient contribution
    # corr[m] for m=0..2N-2 corresponds to lag (m - center)
    # ∂corr[m]/∂y[i]:
    # If m >= center (lag = m-center >= 0): corr = sum_i y[i+lag]*(1-y[i])
    #   ∂/∂y[i] = (1-y[i-lag]) if i>=lag else 0 [from first term]
    #            - y[i+lag] if i+lag < N else 0 [from second term]
    # If m < center (lag = m-center < 0, |lag| = center-m):
    #   Similar by symmetry
    
    # More efficient: process by lag
    max_lag = min(int(0.3 * N), N - 1)  # only process significant lags
    for d in range(-max_lag, max_lag + 1):
        w = weights[center + d]
        if w < 1e-6:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * y)
        elif d > 0:
            # corr involves y[d:] * (1-y[:N-d])
            grad[d:] += w * (1.0 - y[:N-d])
            grad[:N-d] -= w * y[d:]
        else:
            M = -d
            grad[:N-M] += w * (1.0 - y[M:])
            grad[M:] -= w * y[:N-M]
    return grad

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2 * N - 1]

def compute_c5(h, N):

def project_sum(h, target):

def subgrad_topk(y, corr, top_k, N):

            M = -L
