    lo, hi = -1.0, 1.0
    for _ in range(50):

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:

        else:

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def thue_morse_init(N):
    """Generate Thue-Morse-like binary sequence of length N with N//2 ones."""
    h = np.zeros(N)
    for i in range(N):
        # popcount mod 2
        c = 0
        x = i
        while x > 0:
            c += x & 1
            x >>= 1
        h[i] = c & 1
    return fix_binary(h, N // 2)

def anneal_search(h_in, N, deadline, max_iters=20000, double_frac=0.3):
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T0 = 0.02
    T_end = 1e-12
    thermal_iters = max_iters * 0.4
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        if it < thermal_iters:
            frac = it / thermal_iters
            T = T0 * (T_end / T0) ** frac
        else:
            T = T_end  # Greedy phase
        
        r = np.random.random()
        
        if r < double_frac:
            # Double swap
            if len(ones) < 2 or len(zeros) < 2:
                continue
            p1, p2 = np.random.choice(ones, 2, replace=False)
            q1, q2 = np.random.choice(zeros, 2, replace=False)
            
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
            
            new_corr = fft_corr(h)
            new_c = float(new_corr.max())
            
            accept = new_c < best_c or (T > T_end and np.random.random() < np.exp(-(new_c - best_c) / T))
            
            if accept:
                corr = new_corr
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[p1] = h[p2] = 1.0
                h[q1] = h[q2] = 0.0
        else:
            # Single swap
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            
            h[p] = 0.0
            h[q] = 1.0
            
            new_corr = fft_corr(h)
            new_c = float(new_corr.max())
            
            accept = new_c < best_c or (T > T_end and np.random.random() < np.exp(-(new_c - best_c) / T))
            
            if accept:
                corr = new_corr
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[p] = 1.0
                h[q] = 0.0
    
    return best_h, best_c * (2.0 / N)

def continuous_refine(h_in, N, deadline, k_lags=30, lr0=0.02):
    """Subgradient descent on continuous h to refine the solution."""
    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    momentum = np.zeros(N)
    beta = 0.9
    lr = lr0
    
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        c = float(corr.max()) * (2.0 / N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Find top lags
        top_idx = np.argpartition(corr, -k_lags)[-k_lags:]
        top_idx = np.sort(top_idx)
        
        # Compute gradient
        grad = np.zeros(N)
        for t in top_idx:
            L = int(t - (N - 1))
            if L >= 0:
                # C = sum_{i=0}^{N-1-L} h[i]*(1-h[i+L])
                # dC/dh[j] = (1-h[j+L]) for j in [0,N-1-L] - h[j-L] for j in [L,N-1]
                if N - 1 - L >= 0:
                    grad[:N-L] += (1.0 - h[L:N])  # wait, h[j+L] for j in [0,N-1-L]
                if L > 0:
                    grad[L:] -= h[:N-L]
            else:
                M = -L
                if M > 0:
                    grad[M:] += (1.0 - h[:N-M])
                if N - 1 - M >= 0:
                    grad[:N-M] -= h[M:]
        
        grad /= len(top_idx)
        
        # Momentum update
        momentum = beta * momentum - lr * grad
        h = project_sum(h + momentum, target)
        
        it += 1
        if it % 200 == 0:
            lr *= 0.97
            beta = min(0.98, 0.85 + it * 0.0001)
    
    return best_h, best_c

    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    momentum = np.zeros(N)
    beta = 0.9
    lr = lr0
    it = 0
    
    while time.time() < deadline:
        corr = fft_corr(h)
        c = float(corr.max()) * (2.0 / N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        top_idx = np.argpartition(corr, -k_lags)[-k_lags:]
        
        grad = np.zeros(N)
        for t in top_idx:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * h
            elif L > 0:
                grad[:N-L] += (1.0 - h[L:])
                grad[L:] -= h[:N-L]
            else:
                M = -L
                grad[M:] += (1.0 - h[:N-M])
                grad[:N-M] -= h[M:]
        
        grad /= k_lags
        
        momentum = beta * momentum - lr * grad
        h = project_sum(h + momentum, target)
        
        it += 1
        if it % 300 == 0:
            lr *= 0.96
            beta = min(0.98, 0.85 + it * 0.00005)
    
    return best_h, best_c

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        sel = np.random.choice(ones, n1 - nh, replace=False)
        h[sel] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        sel = np.random.choice(zeros, nh - n1, replace=False)
        h[sel] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h
