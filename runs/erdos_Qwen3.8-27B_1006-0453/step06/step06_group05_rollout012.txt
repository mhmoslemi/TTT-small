    # grad_i = sum_k c_k^(p-1) * [(1-h[i+k]) - h[i-k]]
    # where i+k and i-k are within [0,N-1]
    w = c ** (p - 1)

    # term2_i = sum_k w_k * h[i-k] for valid i-k

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    W = np.fft.fft(w, L)

    # Actually: ifft(fft(h) * conj(fft(w))) gives sum_i h[i] * w[i-k]
    # And ifft(fft(w) * conj(fft(h))) gives sum_i w[i] * h[i-k]
    
    # term2_i = sum_k w[k] * h[i-k] = correlation of w and h at lag i
    # = ifft(conj(FFT(w)) * FFT(h))[i] ... need to be careful
    
    # Let me just use the direct approach for N=380 (it's fast enough)

    # term2: for each i, sum over k where 0<=i-k<N of w[k] * h[i-k]
    
    # term2_i = sum_k w[k] * h[i-k] = (w * h)_circ[i] under appropriate indexing
    # This is the circular correlation: sum_k w[k] * conj(h[k-i])...

    term2 = term2[:N]  # circular correlation of w and h, first N elements
    
    # Hmm, I need to be more careful about the indexing

    shifts = np.arange(L) - (N - 1)  # -N+1 to N-1
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv  # i + k

    return term1 - term2

import numpy as np
import time
import math

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_lp_grad(h, c, p, N):
    # Lp gradient: grad_i = sum_k c_k^(p-1) * [(1-h[i+k]) - h[i-k]]
    w = np.maximum(c, 1e-10) ** (p - 1)
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv
    pm_c = np.clip(pm, 0, N - 1)
    vm = (pm >= 0) & (pm < N)
    term1 = (w[None, :] * (1.0 - h[pm_c]) * vm).sum(axis=1)
    pm2 = pos - sv
    pm2_c = np.clip(pm2, 0, N - 1)
    vm2 = (pm2 >= 0) & (pm2 < N)
    term2 = (w[None, :] * h[pm2_c] * vm2).sum(axis=1)
    return term1 - term2

def lp_gradient_descent(h, N, deadline, p_start=4, p_end=64, n_iter_per_p=200, lr0=0.01):

    target = N / 2.0
    dx = 2.0 / N
    best_c5 = float(np.max(fft_corr(h))) * dx
    best_h = h.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    p = p_start
    while time.time() < deadline and p <= p_end:
        for _ in range(n_iter_per_p):
            if time.time() > deadline:
                break
            t += 1
            c = fft_corr(h)
            c5 = float(c.max()) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            grad = compute_lp_grad(h, c, p, N)
            grad -= grad.mean() / N * N  # center
            # Adam update
            lr = lr0 / (1 + 0.01 * t)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            h = h + lr * m / (np.sqrt(v) + 1e-8)
            h = project_sum(h, target)
        p = min(p * 2, p_end)
    return best_h, best_c5

import numpy as np
import time
import math

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G)).clip(0, None)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv
    pp = pos - sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def greedy_search(h, N, deadline, max_sweeps=500):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        bd, bpq = 0.0, None
        for p in ones:
            base = c - D[p]
            cand = base[None, :] - D[zeros]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        else:
            break
    return h, best_c

def anneal_search(h, N, deadline, T0=0.8, T_end=0.005, max_sweeps=3000):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    n_steps = max_sweeps
    cool_rate = (T_end / T0) ** (1.0 / max(n_steps, 1))
    T = T0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T *= cool_rate
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        new_c = c - D[p] - D[q]
        newP = float(new_c.max())
        delta = newP - P
        if delta < 0 or np.random.random() < math.exp(-delta / max(T, 1e-8)):
            h[p] = 0.0
            h[q] = 1.0
    return best_h, best_c

def lp_grad(h, c, p, N):
    w = np.maximum(c, 1e-10) ** (p - 1)
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv
    pm_c = np.clip(pm, 0, N - 1)
    vm = (pm >= 0) & (pm < N)
    pm2 = pos - sv
    pm2_c = np.clip(pm2, 0, N - 1)
    vm2 = (pm2 >= 0) & (pm2 < N)
    t1 = (w[None, :] * (1.0 - h[pm_c]) * vm).sum(axis=1)
    t2 = (w[None, :] * h[pm2_c] * vm2).sum(axis=1)
    return t1 - t2

def lp_descent(h, N, deadline, p_levels=(2, 4, 8, 16, 32, 64), iters_per_p=100):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_c5 = float(np.max(fft_corr(h))) * dx
    best_h = h.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    for p in p_levels:
        for _ in range(iters_per_p):
            if time.time() > deadline:
                break
            t += 1
            c = fft_corr(h)
            c5 = float(c.max()) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            g = lp_grad(h, c, p, N)
            g = g - g.mean()
            lr = 0.02 / math.sqrt(1 + 0.05 * t)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * g * g
            h = h - lr * m / (np.sqrt(v) + 1e-8) * dx
            h = project_sum(h, target)
    return best_h, best_c5
