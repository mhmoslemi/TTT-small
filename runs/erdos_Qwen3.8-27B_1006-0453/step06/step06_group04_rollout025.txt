def swap_delta(h, p, q, N):
    """Compute Δcorr for all lags when swapping h[p]:1→0, h[q]:0→1 in binary h."""
    delta = np.zeros(2*N-1)  # full correlation array
    # For k > 0 (index k + N - 1 in full array):
    # Δcorr[k] = -(1-h[p+k])·[k≤N-1-p] + h[p-k]·[1≤k≤p] + (1-h[q+k])·[k≤N-1-q] - h[q-k]·[1≤k≤q]

            d -= 1
        if 1 <= k <= p and h[p-k] == 1:
            d += 1

            d += 1

def swap_delta_vec(h, p, q, N):
    k = np.arange(1, N)
    delta = np.zeros(2*N-1)
    
    # For positive lags k:
    term1 = np.where((k <= N-1-p) & (h[p+k] == 0), -1.0, 0.0)
    term2 = np.where((k >= 1) & (k <= p) & (h[p-k] == 1), 1.0, 0.0)  # k>=1 always true
    term3 = np.where((k <= N-1-q) & (h[q+k] == 0), 1.0, 0.0)
    term4 = np.where((k >= 1) & (k <= q) & (h[q-k] == 1), -1.0, 0.0)
    
    delta[k + N - 1] = term1 + term2 + term3 + term4
    
    # For negative lags (M = 1..N-1):
    M = k  # same array
    term1n = np.where((p-M >= 0) & (p-M <= N-1-M) & (h[p-M] == 1), -1.0, 0.0)
    term2n = np.where((p >= 0) & (p <= N-1-M) & (h[p+M] == 1), 1.0, 0.0)
    term3n = np.where((q-M >= 0) & (q-M <= N-1-M) & (h[q-M] == 1), 1.0, 0.0)
    term4n = np.where((q >= 0) & (q <= N-1-M) & (h[q+M] == 1), -1.0, 0.0)
    
    delta[N-1-M] = term1n + term2n + term3n + term4n
    
    return delta

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_mid = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad(corr, h, top_k):
    N = len(h)
    idx = np.argsort(corr)[::-1][:top_k]
    weights = np.exp((corr[idx] - corr[idx[0]]) * 50)
    weights /= weights.sum()
    grad = np.zeros(N)
    center = N - 1
    for wi, a in zip(weights, idx):
        L = int(a - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            if seg > 0:
                grad[L:] += wi * (1.0 - h[:seg])
                grad[:seg] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:seg]
    return grad

    target = N / 2.0

    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    if lr is None:
        lr = 0.02
    beta = beta0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        grad = subgrad(corr, y, top_k)
