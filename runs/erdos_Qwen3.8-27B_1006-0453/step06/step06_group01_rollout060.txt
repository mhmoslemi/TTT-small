def fix_binary(h, target_sum):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target_sum = int(target_sum)
    if n1 > target_sum:
        idx = np.where(h > 0.5)[0]
        perm = np.random.permutation(len(idx))[:n1 - target_sum]
        h[idx[perm]] = 0.0
    elif n1 < target_sum:
        idx = np.where(h < 0.5)[0]
        perm = np.random.permutation(len(idx))[:target_sum - n1]
        h[idx[perm]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def lse_optimize(h, N, deadline, tau=5.0, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = project_sum(h, target)
    corr = fft_corr(best_h)
    best_val = float(corr.max())
    lr = lr0
    momentum = np.zeros(N)
    beta = 0.9
    it = 0
    
    while time.time() < deadline:
        grad = lse_grad(best_h, tau)
        # Gradient descent with momentum
        momentum = beta * momentum + (1 - beta) * grad
        h_new = best_h - lr * momentum
        h_new = project_sum(h_new, target)
        
        corr_new = fft_corr(h_new)
        val_new = float(corr_new.max())
        
        if val_new < best_val:
            best_val = val_new
            # Update momentum toward this
            momentum = beta * momentum + (1 - beta) * grad  # already done
        else:
            # Rejected step, stay
            h_new = best_h
        
        best_h = h_new
        it += 1
        if it % 50 == 0:
            lr *= 0.97
            tau = max(3.0, tau * 0.99)  # anneal temperature
    
    return best_h, best_val * (2.0 / N)

def lse_grad_direct(h, tau):
    N = len(h)
    corr = fft_corr(h)
    cmax = corr.max()
    p = np.exp(tau * (corr - cmax))
    p /= p.sum()
    
    shifts = np.arange(-(N-1), N)  # length 2N-1
    # grad[i] = sum_j p[j] * ((1-h[i-shifts[j]]) * valid1 - h[i+shifts[j]] * valid2)
    
    grad = np.zeros(N)
    for j in range(len(p)):
        if p[j] < 1e-15:
            continue
        s = shifts[j]
        w = p[j]
        if s >= 0:
            # i-s in [0,N-1]: i in [s, N-1]
            grad[s:] += w * (1.0 - h[:N-s])
            # i+s in [0,N-1]: i in [0, N-1-s]
            grad[:N-s] -= w * h[s:]
        else:
            ss = -s
            # i-s = i+ss in [0,N-1]: i in [0, N-1-ss]
            grad[:N-ss] += w * (1.0 - h[ss:])
            # i+s = i-ss in [0,N-1]: i in [ss, N-1]
            grad[ss:] -= w * h[:N-ss]
    return grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:2 * N - 1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, target_sum):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    ts = int(target_sum)
    if n1 > ts:
        idx = np.where(h > 0.5)[0]
        perm = np.random.permutation(len(idx))[:n1 - ts]
        h[idx[perm]] = 0.0
    elif n1 < ts:
        idx = np.where(h < 0.5)[0]
        perm = np.random.permutation(len(idx))[:ts - n1]
        h[idx[perm]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def lse_grad_direct(h, tau):
    N = len(h)
    corr = fft_corr(h)
    cmax = corr.max()
    p = np.exp(tau * (corr - cmax))
    p /= p.sum()
    shifts = np.arange(-(N - 1), N)
    grad = np.zeros(N)
    for j in range(len(p)):
        if p[j] < 1e-14:
            continue
        s = shifts[j]
        w = p[j]
        if s >= 0:
            grad[s:] += w * (1.0 - h[:N - s])
            grad[:N - s] -= w * h[s:]
        else:
            ss = -s
            grad[:N - ss] += w * (1.0 - h[ss:])
            grad[ss:] -= w * h[:N - ss]
    return grad

def binary_sa(h_in, N, deadline, t_start=0.12, t_end=0.002, max_sweeps=30000, use_2swap=False):
    h = fix_binary(h_in, N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    shifts = np.arange(2 * N - 1) - (N - 1)
    temp = t_start
    n_total = max(1, max_sweeps)
    cool = (t_end / t_start) ** (1.0 / n_total)
    no_improve = 0

    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break

        if use_2swap and len(ones) >= 2 and len(zeros) >= 2 and np.random.random() < 0.3:
            i1, i2 = np.random.choice(ones, 2, replace=False)
            j1, j2 = np.random.choice(zeros, 2, replace=False)
            delta = np.zeros(2 * N - 1)
            for ii in (i1, i2):
                m1 = (ii + shifts >= 0) & (ii + shifts < N)
                m2 = (ii - shifts >= 0) & (ii - shifts < N)
                delta[m1] += h[ii + shifts[m1]]
                delta[m2] += h[ii - shifts[m2]]
            for jj in (j1, j2):
                m1 = (jj + shifts >= 0) & (jj + shifts < N)
                m2 = (jj - shifts >= 0) & (jj - shifts < N)
                delta[m1] -= h[jj + shifts[m1]]
                delta[m2] -= h[jj - shifts[m2]]
            new_max = float((corr + delta).max())
            if new_max < best_c or np.random.random() < np.exp((best_c - new_max) / max(temp, 1e-10)):
                h[i1], h[j1] = 0.0, 1.0
                h[i2], h[j2] = 0.0, 1.0
                corr += delta
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if new_max < best_c:
                    best_c = new_max
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
            m1 = (i + shifts >= 0) & (i + shifts < N)
            m2 = (i - shifts >= 0) & (i - shifts < N)
            delta = np.zeros(2 * N - 1)
            delta[m1] += h[i + shifts[m1]]
            delta[m2] += h[i - shifts[m2]]
            m3 = (j + shifts >= 0) & (j + shifts < N)
            m4 = (j - shifts >= 0) & (j - shifts < N)
            delta[m3] -= h[j + shifts[m3]]
            delta[m4] -= h[j - shifts[m4]]
            new_max = float((corr + delta).max())
            if new_max < best_c or np.random.random() < np.exp((best_c - new_max) / max(temp, 1e-10)):
                h[i], h[j] = 0.0, 1.0
                corr = new_corr = corr + delta
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if new_max < best_c:
                    best_c = new_max
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        temp *= cool
    return best_h, best_c * (2.0 / N)

def lse_optimize(h_in, N, deadline, tau0=10.0, lr0=0.015, tau_min=4.0):
    h = project_sum(h_in.astype(float).copy(), N / 2.0)
    target = N / 2.0
    corr = fft_corr(h)
    best_val = float(corr.max())
    best_h = h.copy()
    lr = lr0
    mom = np.zeros(N)
    beta = 0.85
    tau = tau0
    it = 0
    while time.time() < deadline:
        grad = lse_grad_direct(h, tau)
        mom = beta * mom + (1 - beta) * grad
        h_new = project_sum(h - lr * mom, target)
        corr_new = fft_corr(h_new)
        val_new = float(corr_new.max())
        if val_new < best_val:
            best_val = val_new
            best_h = h.copy()
            h = h_new
        else:
            h = best_h.copy()
        it += 1
        if it % 30 == 0:
            lr *= 0.975
            tau = max(tau_min, tau * 0.995)
    return best_h, best_val * (2.0 / N)
