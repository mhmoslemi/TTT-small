import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(corr_full(h))) * (2.0 / N)

def project_sum(h, target):
    """Project h onto [0,1]^N with sum = target using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -max(h.max(), target/N).1, max(1.0, target/N + 1.0)
    # Need: sum(clip(h - t, 0, 1)) = target
    # As t increases, sum decreases
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s_mid = np.clip(h - mid, 0.0, 1.0).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def grad_sum_topk(h, k_idx, N):
    """Compute sum of ∂corr[k]/∂h for k in k_idx.
    corr[k] = sum_i h[i]*(1-h[i+s]) where s = k-(N-1), valid: 0≤i<N, 0≤i+s<N
    ∂corr[k]/∂h[j] = I(valid) - h[j+s]*I(0≤j+s<N) - h[j-s]*I(0≤j-s<N)
    """
    g = np.zeros(N)
    j = np.arange(N)
    for k in k_idx:
        s = int(k) - (N - 1)
        if s >= 0:
            # valid j: 0 ≤ j < N-s
            valid = j < (N - s) if s > 0 else (j < N)
            g[valid] += 1.0
            if s > 0:
                g[:N-s] -= h[s:]  # -h[j+s]
            # -h[j-s]: valid for j >= s
            g[s:] -= h[:N-s]
        else:
            ss = -s
            # valid j: ss ≤ j < N
            valid = j >= ss
            g[valid] += 1.0
            # -h[j+s] = -h[j-ss]: valid for j >= ss
            g[ss:] -= h[:N-ss]
            # -h[j-s] = -h[j+ss]: valid for j < N-ss
            g[:N-ss] -= h[ss:]
    return g

def swap_delta(p, q, s, h, N):
    """Compute Δcorr for swap p(1→0), q(0→1) at shift s."""
    d = 0.0
    # Terms involving p:
    # i=p: was 1*(1-h[p+s]) → 0, change = -(1-h[p+s])
    ps = p + s
    if 0 <= ps < N:
        d -= (1.0 - h[ps])
    # i=p-s: was h[p-s]*(1-h[p]) = h[p-s]*0 → h[p-s]*(1-0) = h[p-s]
    ps2 = p - s
    if 0 <= ps2 < N:
        d += h[ps2]
    # Terms involving q:
    # i=q: was 0*(...) → 1*(1-h[q+s]), change = (1-h[q+s])
    qs = q + s
    if 0 <= qs < N:
        d += (1.0 - h[qs])
    # i=q-s: was h[q-s]*(1-h[q]) = h[q-s]*1 → h[q-s]*(1-1) = 0
    qs2 = q - s
    if 0 <= qs2 < N:
        d -= h[qs2]
    return d

def binary_local_search(h, N, deadline, max_iter=5000, topk=20):
    """Local search: find best swap to reduce max correlation."""
    h = h.astype(float).copy()
    nh = N // 2
    # Ensure exactly nh ones
    n1 = int(h.sum())
    if n1 != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[ones_i[:nh]] = 1.0
        else:
            h[ones_i] = 1.0
            h[zeros_i[:nh-n1]] = 1.0
    
    best = compute_c5(h, N)
    corr = corr_full(h)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Find top entries
        k_idx = np.argpartition(corr, -topk)[-topk:]
        P = float(corr.max())
        if P * (2.0/N) < best:
            best = P * (2.0/N)
            stall = 0
        else:
            stall += 1
        if stall > 200:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Get shifts for top-k
        shifts = k_idx - (N - 1)
        
        best_d = 0.0
        best_pq = None
        for p in ones:
            for q in zeros:
                new_max = 0.0
                for ki, s in zip(k_idx, shifts):
                    cd = swap_delta(p, q, int(s), h, N)
                    nm = corr[ki] + cd
                    if nm > new_max:
                        new_max = nm
                if new_max < P - 1e-12:
                    if new_max < best_d or best_pq is None:
                        best_d = new_max
                        best_pq = (int(p), int(q))
        
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            corr = corr_full(h)
        else:
            break
    
    return h, best

def continuous_opt(h, N, deadline, topk=8, lr=0.02, max_iter=2000):
    """Projected gradient descent on continuous h."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    while time.time() < deadline and t < max_iter:
        t += 1
        corr = corr_full(h)
        c = float(np.max(corr)) * (2.0/N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        k_idx = np.argpartition(corr, -topk)[-topk:]
        g = grad_sum_topk(h, k_idx, N)
        g -= g.mean()  # Preserve sum
        
        # Adam update
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g**2
        mhat = m / (1 - b1**t)
        vhat = v / (1 - b2**t)
        
        h_new = h - lr * mhat / (np.sqrt(vhat) + eps)
        h = project_sum(h_new, target)
        
        if t % 500 == 0:
            lr *= 0.95
    
    return best_h, best_c

Let me vectorize the swap evaluation. For a fixed p, I can compute the delta for all q simultaneously:

For a given shift s:
Δcorr_s(p, q) = -(1-h[p+s])*I(0≤p+s<N) + h[p-s]*I(0≤p-s<N) + (1-h[q+s])*I(0≤q+s<N) - h[q-s]*I(0≤q-s<N)

The first two terms depend only on p, the last two only on q!

So: Δcorr_s(p, q) = A_s(p) + B_s(q)
where:
A_s(p) = -(1-h[p+s])*I(0≤p+s<N) + h[p-s]*I(0≤p-s<N)
B_s(q) = (1-h[q+s])*I(0≤q+s<N) - h[q-s]*I(0≤q-s<N)

And for the top-k shifts, the new max after swap is:
max_s (corr_s + A_s(p) + B_s(q))

So I can precompute A_s for all p and s in top-k, and B_s for all q and s in top-k. Then for each (p,q) pair, the new max is max_s(corr_s + A_s[p] + B_s[q]).

This can be vectorized:

- B: shape (N, topk)
- For all pairs: corr[None,None,:] + A[:,None,:] + B[None,:,:] → shape (N, N, topk)
- new_max = max over last axis → shape (N, N)

import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def compute_c5(h, N):
    return float(np.max(corr_full(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_topk(h, k_idx, N):
    g = np.zeros(N)
    for k in k_idx:
        s = int(k) - (N - 1)
        i_lo = max(0, -s)
        i_hi = min(N, N - s)
        g[i_lo:i_hi] += 1.0
        js_lo = i_lo
        js_hi = min(i_hi, N - s)
        if js_hi > js_lo:
            g[js_lo:js_hi] -= h[js_lo + s:js_hi + s]
        jm_lo = max(0, i_lo + s)
        jm_hi = min(N, i_hi + s)
        if jm_hi > jm_lo:
            g[jm_lo:jm_hi] -= h[jm_lo - s:jm_hi - s]
    return g

def binarize(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        h[:] = 0.0
        if n1 > nh:
            h[oi[:nh]] = 1.0
        else:
            h[oi] = 1.0
            h[zi[:nh - n1]] = 1.0
    return h

def binary_local_search(h, N, deadline, topk=12):
    h = h.astype(float).copy()
    h = binarize(h, N)
    best = compute_c5(h, N)
    corr = corr_full(h)
    stall = 0
    while time.time() < deadline:
        P = float(corr.max())
        if P * (2.0 / N) < best:
            best = P * (2.0 / N)
            stall = 0
        else:
            stall += 1
        if stall > 400:
            break
        k_idx = np.argpartition(corr, -topk)[-topk:].astype(int)
        shifts = k_idx - (N - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        no, nz = len(ones), len(zeros)
        if no == 0 or nz == 0:
            break
        A = np.zeros((no, topk))
        B = np.zeros((nz, topk))
        for si in range(topk):
            s = int(shifts[si])
            if s >= 0:
                m1 = ones < (N - s)
                A[m1, si] -= (1.0 - h[ones[m1] + s])
                m2 = ones >= s
                A[m2, si] += h[ones[m2] - s]
                m1z = zeros < (N - s)
                B[m1z, si] += (1.0 - h[zeros[m1z] + s])
                m2z = zeros >= s
                B[m2z, si] -= h[zeros[m2z] - s]
            else:
                ss = -s
                m1 = ones >= ss
                A[m1, si] -= (1.0 - h[ones[m1] - ss])
                m2 = ones < (N - ss)
                A[m2, si] += h[ones[m2] + ss]
                m1z = zeros >= ss
                B[m1z, si] += (1.0 - h[zeros[m1z] - ss])
                m2z = zeros < (N - ss)
                B[m2z, si] -= h[zeros[m2z] + ss]
        cv = corr[k_idx]
        temp = A[:, :, None] + B[None, :, :] + cv[None, None, :]
        new_max = temp.max(axis=2)
        flat_i = int(np.argmin(new_max))
        pi, qi = divmod(flat_i, nz)
        if new_max[pi, qi] < P - 1e-12:
            h[ones[pi]] = 0.0
            h[zeros[qi]] = 1.0
            corr = corr_full(h)
        else:
            break
    return h, best

def continuous_opt(h, N, deadline, topk=10, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    t = 0
    while time.time() < deadline:
        t += 1
        corr = corr_full(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        k_idx = np.argpartition(corr, -topk)[-topk:].astype(int)
        g = grad_topk(h, k_idx, N)
        g -= g.mean()
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g ** 2
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + eps)
        h = project_sum(h_new, target)
        if t % 200 == 0:
            lr *= 0.95
            hb = binarize(h, N)
            h = 0.6 * h + 0.4 * hb
    return best_h, best_c

def multi_start_binary(h_best, N, n_starts, deadline):
    best_h = h_best.copy()
    best_c = compute_c5(best_h, N)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        hp = best_h.copy()
        nf = np.random.randint(2, 6)
        oi = np.where(hp == 1)[0]
        zi = np.where(hp == 0)[0]
        for _ in range(nf):
            oi = np.where(hp == 1)[0]
            zi = np.where(hp == 0)[0]
            p = oi[np.random.randint(len(oi))]
            q = zi[np.random.randint(len(zi))]
            hp[p], hp[q] = 0.0, 1.0
        sub_dl = min(deadline, time.time() + 30)
        ht, ct = binary_local_search(hp, N, sub_dl, topk=10)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    return best_h, best_c

def upscale(h, N_src, N_dst):
    x_src = np.linspace(0, 2, N_src, endpoint=False)
    x_dst = np.linspace(0, 2, N_dst, endpoint=False)
    h_interp = np.interp(x_dst, x_src, h, period=2.0)
    return h_interp
