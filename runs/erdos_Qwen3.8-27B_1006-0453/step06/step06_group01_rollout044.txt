import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def swap_delta(h, p, q, N):
    """Compute corr change for swap p(1->0), q(0->1) at all lags."""
    corr_len = 2 * N - 1
    delta = np.zeros(corr_len)
    for k in range(corr_len):
        lag = k - (N - 1)
        d = 0.0
        if 0 <= p - lag < N:
            d += h[p - lag]
        if 0 <= q - lag < N:
            d -= h[q - lag]
        if 0 <= p + lag < N:
            d += h[p + lag]
        if 0 <= q + lag < N:
            d -= h[q + lag]
        delta[k] = d
    return delta

def swap_delta_fast(h, p, q, N):
    """Vectorized swap delta."""
    corr_len = 2 * N - 1
    lags = np.arange(corr_len) - (N - 1)
    delta = np.zeros(corr_len)
    m1 = (p - lags).clip(0, N - 1)
    valid1 = (p - lags >= 0) & (p - lags < N)
    m2 = (q - lags).clip(0, N - 1)
    valid2 = (q - lags >= 0) & (q - lags < N)
    m3 = (p + lags).clip(0, N - 1)
    valid3 = (p + lags >= 0) & (p + lags < N)
    m4 = (q + lags).clip(0, N - 1)
    valid4 = (q + lags >= 0) & (q + lags < N)
    delta += np.where(valid1, h[m1], 0.0)
    delta -= np.where(valid2, h[m2], 0.0)
    delta += np.where(valid3, h[m3], 0.0)
    delta -= np.where(valid4, h[m4], 0.0)
    return delta

def targeted_binary(h, N, deadline, max_iter=8000, sa_temp=0.3):
    corr = get_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    no_improve = 0
    temp = sa_temp
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        peak_k = int(corr.argmax())
        peak_val = corr[peak_k]
        
        # Focus on ones that contribute to the peak
        lag = peak_k - (N - 1)
        if lag >= 0:
            # corr at lag: h[i+lag]=1, h[i]=0; contributing i's have h[i+lag]=1 means i+lag in ones
            contrib_p = np.intersect1d(ones - lag, np.arange(N)) if lag > 0 else ones
            contrib_q = np.intersect1d(ones, np.arange(N))  # q such that q-lag is a zero
            contrib_q = np.intersect1d(zeros, np.arange(N) - lag) if lag >= 0 else zeros
        else:
            contrib_p = np.intersect1d(ones, np.arange(N) - lag)
            contrib_q = np.intersect1d(zeros, np.arange(N) + lag)
        
        # Sample candidate p's from peak contributors + random
        n_cp = min(20, len(ones))
        if len(contrib_p) > 0 and len(contrib_p) < 50:
            p_cand = np.random.choice(contrib_p, min(n_cp, len(contrib_p)), replace=False)
        else:
            p_cand = np.random.choice(ones, n_cp, replace=False)
        
        best_new_peak = best_c
        best_p, best_q = -1, -1
        
        for p in p_cand:
            # Best q: minimize max(corr + delta)
            # Try zeros near p (local) and some random
            near_q = np.array([p - 1, p + 1, p - 2, p + 2, p - 3, p + 3])
            near_q = near_q[(near_q >= 0) & (near_q < N) & (h[near_q] == 0)]
            rand_q = np.random.choice(zeros, min(15, len(zeros)), replace=False)
            q_cand = np.unique(np.concatenate([near_q, rand_q])) if len(near_q) > 0 else rand_q
            
            for q in q_cand:

                    zeros = np.where(h == 0)[0]

                no_improve += 1
            else:
                no_improve += 1
            if no_improve > 200:
                break
            continue
        
        # Apply best swap
        p, q = best_p, best_q
        d = swap_delta_fast(h, p, q, N)
        corr += d
        h[p], h[q] = 0.0, 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if best_new_peak < best_c - 1e-12:
            best_c = best_new_peak
            best_h = h.copy()
            no_improve = 0
            temp = max(0.05, temp * 0.995)
        else:
            no_improve += 1
    
    return best_h, best_c

def double_swap_search(h, N, corr, deadline, max_iter=2000):
    """2-for-2 swaps to escape local optima."""
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        p1, p2 = np.random.choice(ones, 2, replace=False)
        q1, q2 = np.random.choice(zeros, 2, replace=False)
        d = swap_delta_fast(h, int(p1), int(q1), N) + swap_delta_fast(h, int(p2), int(q2), N)
        np.fill_diagonal  # noqa
        new_peak = float(np.max(corr + d))
        if new_peak < best_c - 1e-12:
            corr += d
            h[p1], h[q1] = 0.0, 1.0
            h[p2], h[q2] = 0.0, 1.0
            best_c = new_peak
            best_h = h.copy()
    return best_h, best_c, corr

def nesterov_polish(h_in, N, deadline, k=60, lr0=0.008):
    h = h_in.astype(float).copy()
    target = N / 2.0
    corr = get_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = get_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.97
            beta = min(0.95, 0.80 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)
