```python
def targeted_swap(h, corr, top_k=3):
    """Find a targeted swap that reduces the top correlation peaks."""
    N = len(h)
    center = N - 1
    peaks = np.argpartition(corr, -top_k)[-top_k:]
    
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    
    for p in peaks:
        lag = int(p - center)
        if lag == 0:
            continue
        if lag > 0:
            # Bad pairs: h[i]=1, h[i+lag]=0
            bad = np.where((h[:N-lag]==1) & (h[lag:]==0))[0]
        else:
            M = -lag
            # Bad pairs: h[i]=1, h[i+M]=0 (but in the "reverse" direction)
            # corr at "negative lag M" = sum h[i]*(1-h[i+M])
            # Bad: h[i]=1, h[i+M]=0
            bad = np.where((h[:N-M]==1) & (h[M:]==0))[0]
        
        if len(bad) == 0:
            continue
        
        # For each bad index, try to fix by swapping
        # If lag > 0: set h[bad+lag]=1, set some h[j]=0
        # If lag < 0: set h[bad+M]=1, set some h[j]=0
        
        if lag > 0:
            targets = bad + lag
        else:
            targets = bad + (-lag)
        
        # Find a zero-position to set to 1 (from targets that are already 0)
        # And a one-position to set to 0 that doesn't create new bad pairs
        
        # For simplicity: pick the first target, find best source
        t = int(targets[0])
        if h[t] != 0:
            continue
        
        # Find best o in ones: not in [t-lag, t+lag] range to avoid creating new bad pairs
        # Simple heuristic: pick o that's furthest from all peak lags
        peak_lags = [int(pp - center) for pp in peaks]
        scores = np.zeros(len(ones))
        for j, o in enumerate(ones):
            for pl in peak_lags:
                if abs(o - t) == abs(pl):
                    scores[j] -= 2
                if abs(int((o+t)/2*2 - (o+pl))) < 2:
                    scores[j] -= 0.5
        # Pick the one with highest score (least interference)
        o = int(ones[np.argmax(scores)])
        
        return o, t  # swap: h[o] 1->0, h[t] 0->1
    
    return None, None
```

import numpy as np
import time

def corr_full(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L <<= 1
    H = np.fft.fft(h, L)
    K = np.fft.fft(1.0-h, L)
    return np.real(np.fft.ifft(H*K))[:2*N-1]

def c5_of(h):
    return float(np.max(corr_full(h))) * 2.0 / len(h)

def proj01(h, s):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum()-s) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def bproj(h, n1):
    idx = np.argsort(h)[::-1][:n1]
    out = np.zeros(len(h))
    out[idx] = 1.0
    return out

def compute_C(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2*N-1)[None, :] - (N-1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk>=0)&(i_mk<N), 1.0-h[np.clip(i_mk,0,N-1)], 0.0)
    B = np.where((i_pk>=0)&(i_pk<N), h[np.clip(i_pk,0,N-1)], 0.0)
    return A - B

def il_cmat(h, deadline, top_k=6, kick_every=80, kick_size=5, stall_limit=400):
    N = len(h)
    nh = N//2
    h = bproj(h, nh)
    best_h = h.copy()
    best_pk = float(np.max(corr_full(h)))
    st = 0
    it = 0
    while time.time() < deadline:
        if it % kick_every == kick_every - 1 and it > 0:
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            for _ in range(kick_size):
                o = int(np.random.choice(ones))
                z = int(np.random.choice(zeros))
                h[o]=0; h[z]=1
                idx_o = np.where(ones==o)[0]
                if len(idx_o): ones = np.delete(ones, idx_o[0])
                idx_z = np.where(zeros==z)[0]
                if len(idx_z): zeros = np.delete(zeros, idx_z[0])
                ones = np.append(ones, z)
                zeros = np.append(zeros, o)
        corr = corr_full(h)
        P = float(corr.max())
        if P < best_pk - 1e-9:
            best_pk = P
            best_h = h.copy()
        C = compute_C(h, N)
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)<2 or len(zeros)<2:
            break
        ns = min(len(ones), 50)
        nz = min(len(zeros), 50)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        new = corr[None,None,:] + Co[None,:,:] - Cz[:,None,:]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-11:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
            st = 0
        else:
            st += 1
        it += 1
        if st >= stall_limit:
            break
    return best_h, best_pk*2.0/N

def fista(h, deadline, k_top=20, lr=0.012):
    N = len(h)
    target = N/2.0
    h = proj01(h, target)
    best_h = h.copy()
    best_pk = float(np.max(corr_full(h)))
    m = np.zeros(N)
    t_f = 1.0
    while time.time() < deadline:
        y = h + m
        c = corr_full(y)
        pk = float(np.max(c))
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        top = np.argpartition(c, -k_top)[-k_top:]
        center = N-1
        grad = np.zeros(N)
        for p in top:
            lag = int(p-center)
            if lag >= 0:
                nv = N-lag
                if nv > 0:
                    grad[lag:] += 1.0-y[:nv]
                    grad[:nv] -= y[lag:]
            elif lag > -N:
                M = -lag
                nv = N-M
                if nv > 0:
                    grad[:nv] += 1.0-y[M:]
                    grad[M:] -= y[:nv]
        grad /= k_top
        h_new = proj01(y-lr*grad, target)
        t_new = 0.5*(1.0+np.sqrt(1.0+4.0*t_f*t_f))
        m = (t_f-1.0)/(t_new+1.0)*(h_new-h)
        t_f = t_new
        h = h_new
    return best_h, best_pk*2.0/N
