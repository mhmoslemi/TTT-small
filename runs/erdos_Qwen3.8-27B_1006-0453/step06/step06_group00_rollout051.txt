def fft_corr(h):
    """Compute linear cross-correlation of h and 1-h using FFT."""
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection on threshold."""

        hc = project(best_h + np.random.randn(N0) * 0.08, N0 / 2.0)
        hc, c2 = adam_cont(hc, N0, d2, lr=0.003, topk=55)
        if c2 < best_c5:
            best_c5, best_h = c2, hc

    # Phase 3: Binary IL deeper
    if time.time() < deadline - 250:
        d3 = min(deadline - 230, time.time() + 200)
        h3 = fix_count((best_h > 0.5).astype(float), N0)
        h3, _ = il_search(h3, N0, d3, iters=20000, ss=55)
        c3 = c5_val(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Multi-resolution search
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [340, 400, 460, 520]:
        if time.time() > deadline - 60:
            break
        if Nt % 2:
            Nt += 1
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_best, best_h)
        hb = fix_count((hi + np.random.randn(Nt) * 0.03 > 0.5).astype(float), Nt)
        dt = min(deadline - 10, time.time() + 75)
        ht, _ = il_search(hb, Nt, dt, iters=12000, ss=45)
        ct = c5_val(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
            x_best = np.linspace(0, 2, Nt, endpoint=False)

    # Phase 5: Short Adam polish
    if time.time() < deadline - 50:
        d5 = min(deadline - 35, time.time() + 50)
        Ns = len(best_h)
        h5 = project(best_h + np.random.randn(Ns) * 0.04, Ns / 2.0)
        h5, c5a = adam_cont(h5, Ns, d5, lr=0.002, topk=40)
        if c5a < best_c5:
            best_c5, best_h = c5a, h5

    # Phase 6: Final binary IL
    if time.time() < deadline - 5:
        h6 = fix_count((best_h > 0.5).astype(float), len(best_h))
        h6, _ = il_search(h6, len(best_h), deadline - 1, iters=8000, ss=70)
        c6 = c5_val(h6)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    # Final: pick best of continuous and binary
    hfinal = best_h
    final_c5 = c5_val(hfinal)
    hb2 = fix_count((best_h > 0.5).astype(float), len(best_h))
    c5b = c5_val(hb2)
    if c5b < final_c5:
        hfinal, final_c5 = hb2, c5b
    return hfinal, final_c5, len(hfinal)
