def fft_corr(h):

    H = np.fft.fft(h, L)

    dx = 2.0 / N
    best_h = h.copy()

        grad, c, w = softmax_gradient(h, N, temp)

        # Gradient descent

        if c_new < best_c:

        h = h_new
        # Cool down
        temp = max(temp_min, temp_init * (0.995 ** t))
        # Adaptive lr
        if t % 100 == 0:

    return best_h, best_c

def compute_D_fast(h, N):
    shifts = np.arange(2*N - 1) - (N - 1)
    # Use broadcasting
    # D[p, idx] where idx corresponds to shift k
    # D[p, idx] = (1-h[p+k]) - h[p-k] (with bounds)
    
    # For (1-h[p+k]): p in 0..N-1, k in -(N-1)..(N-1)
    # p+k in 0..N-1 => p in -k..N-1-k, intersect with 0..N-1
    # p + k: for fixed p, valid k range: -(p)..(N-1-p)
    
    # Let's use a 2D approach
    p = np.arange(N)[:, None]  # N x 1
    k = shifts[None, :]  # 1 x (2N-1)
    pk = p + k  # N x (2N-1), p+k values
    pk2 = p - k  # N x (2N-1), p-k values
    
    valid1 = (pk >= 0) & (pk < N)
    valid2 = (pk2 >= 0) & (pk2 < N)
    
    pk_clipped = np.clip(pk, 0, N-1)
    pk2_clipped = np.clip(pk2, 0, N-1)
    
    D = np.where(valid1, 1.0 - h[pk_clipped], 0.0) - np.where(valid2, h[pk2_clipped], 0.0)
    return D

            h[zeros_i[:nh - n1]] = 1

            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    temp = temp_init
    lr = lr_init
    t = 0
    prev_pk = best_pk
    patience = 0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        # Softmax weights
        c_s = c - c.max()
        w = np.exp(c_s / temp)
        w /= w.sum()
        # Gradient
        D = compute_D(h, N)
        grad = D @ w
        grad -= grad.mean()  # preserve sum
        # Step

            patience = 0
        else:
            patience += 1
        h = h_new

        temp = max(temp_min, temp_init * (0.999 ** t))

            patience = 0

            lr *= 0.97

def bin_search(h, N, deadline, max_sweeps=500, anneal=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[ones_i[:nh]] = 1
        else:
            h[ones_i] = 1
            h[zeros_i[:nh - n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 90 and anneal == 0:
            break
        if stall > 50 and anneal > 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        bd, bpq = 0.0, None
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

def softmax_smooth(h, N, deadline, temp_init=0.8, temp_min=0.04, lr_init=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    temp = temp_init
    lr = lr_init
    t = 0
    patience = 0
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        c_s = c - c.max()
        w = np.exp(c_s / temp)
        w /= w.sum()
        D = compute_D(h, N)
        grad = D @ w
        grad -= grad.mean()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + eps)
        h_new = project_sum(h_new, target)
        pk = float(np.max(fft_corr(h_new)))
        if pk < best_pk:
            best_pk = pk
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new
        temp = max(temp_min, temp_init * (0.9985 ** t))
        if patience > 60:
            lr *= 0.92
            patience = 0
        if t % 300 == 0:
            lr *= 0.96
    return best_h, best_pk * dx

def block_move_search(h, N, deadline, max_iter=3000):
    h = h.astype(float).copy()
    best_c = compute_c5(h, N)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        diff = np.diff(np.concatenate([[0], h, [0]]))
        starts = np.where(diff == 1)[0]
        ends = np.where(diff == -1)[0]
        if len(starts) == 0:
            break
        moved = False
        for i in range(len(starts)):
            s, e = starts[i], ends[i]
            length = e - s
            if length < 2 or length > 12:
                continue
            for shift in [1, -1, 2, -2, 3, -3]:
                ns, ne = s + shift, e + shift
                if ns < 0 or ne > N:
                    continue
                block = h[s:e].copy()
                h[s:e] = 0.0
                if np.any(h[ns:ne] != 0):
                    h[s:e] = block
                    continue
                h[ns:ne] = block
                c_new = compute_c5(h, N)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    moved = True
                    break
                h[ns:ne] = 0
                h[s:e] = block
            if moved:
                break
        if not moved:
            break
    return h, best_c

def threshold_binarize(h, N):
    best_c5 = float('inf')
    best_binary = None
    for thr in [0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65]:
        hb = (h > thr).astype(float)
        n1 = int(hb.sum())
        nh = N // 2
        if n1 != nh:
            oi, zi = np.where(hb == 1)[0], np.where(hb == 0)[0]
            hb[:] = 0
            if n1 > nh:
                hb[oi[np.argsort(-h[oi])[:nh]]] = 1
            else:
                hb[oi] = 1
                hb[zi[np.argsort(-h[zi])[:(nh - n1)]]] = 1
        c5 = compute_c5(hb, N)
        if c5 < best_c5:
            best_c5 = c5
            best_binary = hb.copy()
    return best_binary, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 != N0 // 2:
        oi, zi = np.where(h_bin == 1)[0], np.where(h_bin == 0)[0]
        h_bin[:] = 0
        if n1 > N0 // 2:
            h_bin[oi[:N0 // 2]] = 1
        else:
            h_bin[oi] = 1
            h_bin[zi[:N0 // 2 - n1]] = 1
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: Greedy binary search
    d1 = min(deadline - 450, t0 + 180)
    h_s, _ = bin_search(h_bin, N0, d1, max_sweeps=1200, anneal=0.0)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    # Phase 2: Softmax Adam smooth optimization
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 220)
        hc = best_h.copy() + np.random.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = softmax_smooth(hc, N0, d2, temp_init=0.7, temp_min=0.04, lr_init=0.045)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    # Phase 3: Threshold binarize + binary
    if time.time() < deadline - 220:
        best_h, best_c5 = threshold_binarize(best_h, N0)
    if time.time() < deadline - 180:
        d3 = min(deadline - 150, time.time() + 130)
        h_s3, _ = bin_search(best_h, N0, d3, max_sweeps=900, anneal=0.0)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()
    # Phase 4: Block moves
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 80)
        h_blk, c5_blk = block_move_search(best_h, N0, d4)
        if c5_blk < best_c5:
            best_c5, best_h = c5_blk, h_blk.copy()
        if time.time() < deadline - 20:
            h_s4, _ = bin_search(best_h, N0, deadline - 15, max_sweeps=400, anneal=0.0)
            c5_s4 = compute_c5(h_s4, N0)
            if c5_s4 < best_c5:
                best_c5, best_h = c5_s4, h_s4.copy()
    # Phase 5: Multi-resolution
    N_cur = len(best_h)
    for N_try in [380, 400, 360, 420, 340, 440]:
        if time.time() > deadline - 40:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == N_cur:
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        x_old = np.linspace(0, 2, N_cur, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        nh, n1 = N_try // 2, int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            idx = idx[np.argsort(h_interp[idx])[:-(n1 - nh)]]
            h_bt[idx] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            idx = idx[np.argsort(-h_interp[idx])[:nh - n1]]
            h_bt[idx] = 1
        sub_dl = min(deadline - 20, time.time() + 55)
        h_rt, _ = bin_search(h_bt, N_try, sub_dl, max_sweeps=500, anneal=0.0)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h, N_cur = c5_rt, h_rt.copy(), N_try
    # Phase 6: Annealed binary escape
    if time.time() < deadline - 25:
        h_b6 = (best_h > 0.5).astype(float)
        h_s6, _ = bin_search(h_b6, len(best_h), min(deadline - 8, time.time() + 70), max_sweeps=500, anneal=0.12)
        c5_6 = compute_c5(h_s6, len(best_h))
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h_s6.copy()
    # Phase 7: Perturbation + quick search
    if time.time() < deadline - 15:
        h_p = (best_h > 0.5).astype(float)
        for _ in range(3):
            if time.time() > deadline - 15:
                break
            hp = h_p.copy()
            oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            for _ in range(4):
                hp[oi[np.random.randint(len(oi))]] = 0
                hp[zi[np.random.randint(len(zi))]] = 1
                oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            sub_dl = min(deadline - 10, time.time() + 20)
            h_pt, _ = bin_search(hp, len(best_h), sub_dl, max_sweeps=200, anneal=0.0)
            c5_pt = compute_c5(h_pt, len(best_h))
            if c5_pt < best_c5:
                best_c5, best_h = c5_pt, h_pt.copy()
                h_p = h_pt.copy()
    # Final polish
    if time.time() < deadline - 2:
        hF = (best_h > 0.5).astype(float)
        hF, _ = bin_search(hF, len(best_h), deadline - 1, max_sweeps=100, anneal=0.0)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
