import numpy as np

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(hb, N, rng):
    target = N // 2
    n1 = int(hb.sum())
    hb = hb.astype(float).copy()
    if n1 > target:
        idx = np.where(hb > 0.5)[0]
        sel = rng.choice(len(idx), n1 - target, replace=False)
        hb[idx[sel]] = 0
    elif n1 < target:
        idx = np.where(hb <= 0.5)[0]
        sel = rng.choice(len(idx), target - n1, replace=False)
        hb[idx[sel]] = 1
    return hb

def swap_delta(h, N, p, q):
    s_arr = np.arange(-(N - 1), N)
    d = np.zeros(2 * N - 1)
    ps, pm = p + s_arr, p - s_arr
    qs, qm = q + s_arr, q - s_arr
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
    pc = np.clip(ps, 0, N - 1); pmc = np.clip(pm, 0, N - 1)
    qsc = np.clip(qs, 0, N - 1); qmc = np.clip(qm, 0, N - 1)
    d[vps] -= (1.0 - h[pc[vps]])
    d[vpm] += h[pmc[vpm]]
    d[vqs] += (1.0 - h[qsc[vqs]])
    d[vqm] -= h[qmc[vqm]]
    return d

def soft_grad(h, N, beta=35.0):
    corr = fft_corr(h)
    center = N - 1
    valid = np.arange(2 * N - 1)
    shifts = valid - center
    mask = shifts != 0
    c = np.where(mask, corr, -1e10)
    cmax = c.max()
    w = np.where(mask, np.exp(beta * (c - cmax)), 0.0)
    ws = w.sum()
    if ws < 1e-30:
        return float(corr.max()) * (2.0 / N), np.zeros(N)
    w /= ws
    grad = np.zeros(N)
    active = np.where(w > 1e-12)[0]
    for si in active:
        s = si - center
        wv = w[si]
        if s > 0:
            grad[:N - s] += wv * (1.0 - h[s:])
            grad[s:] += wv * (-h[:N - s])
        elif s < 0:
            s2 = -s
            grad[s2:] += wv * (1.0 - h[:N - s2])
            grad[:N - s2] += wv * (-h[s2:])
        else:
            grad += wv * (1.0 - 2.0 * h)
    pk = float(corr.max()) * (2.0 / N)
    return pk, grad

        m2 = b2 * m2 + (1 - b2) * grad * grad

        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        if r < 0.55:
            worst_k = np.argmax(np.where(wmask, corr, -1e10))
            s = worst_k - center
            if s > 0:
                cands = np.where((h[:N - s] == 1) & (h[s:] == 0))[0]
                if len(cands) > 0:
                    p = int(cands[rng.randint(len(cands))]); q = p + s
                else:
                    p = int(ones[rng.randint(len(ones))]); q = int(zeros[rng.randint(len(zeros))])
            elif s < 0:
                s2 = -s
                cands = np.where((h[s2:] == 1) & (h[:N - s2] == 0))[0]
                if len(cands) > 0:
                    q = int(cands[rng.randint(len(cands))]); p = q + s2
                else:
                    p = int(ones[rng.randint(len(ones))]); q = int(zeros[rng.randint(len(zeros))])
            else:
                p = int(ones[rng.randint(len(ones))]); q = int(zeros[rng.randint(len(zeros))])
            d = swap_delta(h, N, p, q)
        elif r < 0.85:
            p = int(ones[rng.randint(len(ones))]); q = int(zeros[rng.randint(len(zeros))])
            d = swap_delta(h, N, p, q)
        else:
            if len(ones) < 2 or len(zeros) < 2:
                p = int(ones[rng.randint(len(ones))]); q = int(zeros[rng.randint(len(zeros))])
                d = swap_delta(h, N, p, q)
            else:
                pi = rng.choice(len(ones), 2, replace=False)
                qi = rng.choice(len(zeros), 2, replace=False)
                p1, p2 = int(ones[pi[0]]), int(ones[pi[1]])
                q1, q2 = int(zeros[qi[0]]), int(zeros[qi[1]])
                d = swap_delta(h, N, p1, q1) + swap_delta(h, N, p2, q2)
                nc = corr + d
                npk = float(nc.max())
                delta = npk - cur_pk
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                    corr = nc; cur_pk = npk
                    if npk < best_pk:
                        best_pk = npk; best_h = h.copy(); no_imp = 0
                    else:
                        no_imp += 1
                else:
                    no_imp += 1
                T *= cool
                if no_imp > max_no_imp:
                    no_imp = 0
                    h = best_h.copy(); corr = fft_corr(h); cur_pk = best_pk
                    T = T0 * 0.5
                continue
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h[p] = 0; h[q] = 1
            corr = nc; cur_pk = npk
            if npk < best_pk:
                best_pk = npk; best_h = h.copy(); no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cool
        if no_imp > max_no_imp:
            no_imp = 0
            h = best_h.copy()
            corr = fft_corr(h)
            cur_pk = best_pk
            for _ in range(rng.integers(3, 8)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    h[int(oi[rng.randint(len(oi))])] = 0
                    h[int(zi[rng.randint(len(zi))])] = 1
            corr = fft_corr(h)
            cur_pk = float(corr.max())
            T = T0 * 0.4
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(fft_corr(h0))) * (2.0 / N0)

    # Phase 1: Continuous (15%)
    d1 = t0 + 0.15 * budget_s
    hc = h0 + rng.standard_normal(N0) * 0.04
    hc = proj_sum(hc, N0 / 2.0)
    try:
        hC, c5C = continuous_opt(hc, N0, d1, rng, beta=35, lr0=0.007)
        if c5C < best_c5:
            best_c5, best_h = c5C, hC.copy()
    except Exception:
        pass

    # Phase 2: Guided binary annealing (45%)
    d2 = t0 + 0.60 * budget_s
    hb = fix_count((best_h > 0.5).astype(float), N0, rng)
    try:
        hB, pkB = guided_anneal(hb, N0, d2, T0=0.05, cool=0.9993, rng=rng)
        c5B = pkB * (2.0 / N0)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB.copy()
    except Exception:
        pass

    # Phase 3: Multi-N sweep (30%)
    elapsed = time.time() - t0
    remaining = deadline - time.time()
    n_list = [200, 240, 280, 320, 360, 400, 440, 480, 520, 560]
    n_list = [n for n in n_list if n % 2 == 0 and abs(n - N0) > 20]
    time_per = max(8.0, remaining / max(len(n_list), 1) - 2)
    x_old = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in sorted(n_list, key=lambda x: abs(x - N0)):
        if time.time() > deadline - 10:
            break
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_count((hi > 0.5).astype(float), Nt, rng)
        sub_dl = min(deadline - 5, time.time() + time_per)
        try:
            hrt, pkr = guided_anneal(hbt, Nt, sub_dl, T0=0.04, cool=0.9994, rng=rng)
            c5r = pkr * (2.0 / Nt)
            if c5r < best_c5:
                best_c5, best_h = c5r, hrt.copy()
        except Exception:
            pass

    # Phase 4: Final polish (10%)
    if time.time() < deadline - 5:
        Nb = len(best_h)
        hfp = fix_count((best_h > 0.5).astype(float), Nb, rng)
        try:
            hfin, pkf = guided_anneal(hfp, Nb, deadline - 1, T0=0.02, cool=0.9996, rng=rng)
            c5f = pkf * (2.0 / Nb)
            if c5f < best_c5:
                best_c5, best_h = c5f, hfin.copy()
        except Exception:
            pass

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
