import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_effect(h, N):
    pos = np.arange(N)[:, None]
    k_off = np.arange(2 * N - 1)[None, :] - (N - 1)
    idx1 = pos + k_off
    idx2 = pos - k_off
    A = np.where((idx1 >= 0) & (idx1 < N), 1.0 - h[np.clip(idx1, 0, N - 1)], 0.0)
    B = np.where((idx2 >= 0) & (idx2 < N), h[np.clip(idx2, 0, N - 1)], 0.0)
    return A - B

    best_c = np.inf
    best_h = h.copy()
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > patience:
            break
        C = compute_effect(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        Co = C[ones]
        Cz = C[zeros]
        new = corr[None, None, :] + (-Co)[None, :, :] + Cz[:, None, :]
        newP = new.max(axis=2)
        flat = newP.ravel()
        kk = min(topK, flat.size)

        ai, bi = np.unravel_index(chosen, newP.shape)
        h[ones[ai]] = 0
        h[zeros[bi]] = 1
        it += 1
    return best_h, best_c

def ils_search(h, N, deadline, n_restarts=8, topK=3, patience=200):
    best_h = h.copy()
    best_c = compute_c5(h, N) / (2.0 / N)
    n1 = N // 2
    t_per = (deadline - time.time()) / max(n_restarts, 1)
    for r in range(n_restarts):
        sub_dl = min(deadline, time.time() + t_per)
        cur = h.copy()
        if r > 0:
            n_flip = min(3 + r, 8)
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            for _ in range(n_flip):
                a = int(np.random.choice(ones))
                b = int(np.random.choice(zeros))

                zeros = np.where(cur == 0)[0]
        h_cur, c_cur = binary_search(cur, N, sub_dl, max_iters=5000, topK=topK, patience=patience)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
        h = best_h
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=40, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0001)
    return best_h, best_pk * dx
