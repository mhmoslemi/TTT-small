def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def nesterov_opt(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = y.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.993
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * dx

    x_f = np.linspace(0, 2, N_fine, endpoint=False)
    h_interp = np.interp(x_f, x_c, h_coarse)
    h_b = (h_interp > 0.5).astype(float)
    return fix_sum_binary(h_b, N_fine)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_N = len(best_h)
    best_c5 = compute_c5(best_h, best_N)

    # Phase 1: Exhaustive coarse search N=20
    h_c20, _ = exhaustive_coarse(20)

    # Phase 2: Upscale to N=200 and search
    h200 = upscale(h_c20, 200)
    h_improved = None
    if time.time() < deadline - 500:
        d2 = min(deadline - 480, time.time() + 80)
        h200, _ = binary_search(h200, 200, d2, max_sweeps=2000, top_k=5)
        c200 = compute_c5(h200, 200)
        if c200 < best_c5:
            best_c5, best_h, best_N = c200, h200.copy(), 200
            h_improved = h200.copy()

    # Phase 3: Search at N=380 (or best_N) from initial_h_values
    N_main = best_N if best_N >= 200 else 380
    if N_main != 380 and best_N == 380:
        N_main = 380
    h_start = (np.array(initial_h_values, dtype=float) > 0.5).astype(float)
    if len(h_start) != N_main:
        h_start = upscale(h_start, N_main)
    d3 = min(deadline - 250, time.time() + 250)
    h3, _ = binary_search(h_start, N_main, d3, max_sweeps=4000, top_k=5)
    c3 = compute_c5(h3, N_main)
    if c3 < best_c5:
        best_c5, best_h, best_N = c3, h3.copy(), N_main

    # Phase 4: Search from upscaled coarse at N_main
    if h_improved is not None and time.time() < deadline - 200:
        h_up = upscale(h_improved, N_main) if len(h_improved) != N_main else h_improved.copy()
        h_up = fix_sum_binary(h_up, N_main)
        d4 = min(deadline - 180, time.time() + 150)
        h4, _ = binary_search(h_up, N_main, d4, max_sweeps=3000, top_k=5)
        c4 = compute_c5(h4, N_main)
        if c4 < best_c5:
            best_c5, best_h, best_N = c4, h4.copy(), N_main

    # Phase 5: Second search from best at higher resolution
    N_hi = best_N * 2 if best_N <= 400 else best_N
    if N_hi != best_N and time.time() < deadline - 150:
        h_hi = upscale(best_h, N_hi)
        h_hi = fix_sum_binary(h_hi, N_hi)
        d5 = min(deadline - 130, time.time() + 120)
        h5, _ = binary_search(h_hi, N_hi, d5, max_sweeps=2000, top_k=5)
        c5_hi = compute_c5(h5, N_hi)
        if c5_hi < best_c5:
            best_c5, best_h, best_N = c5_hi, h5.copy(), N_hi

    # Phase 6: Nesterov continuous refinement
    Nf = len(best_h)
    if time.time() < deadline - 60:
        d6 = min(deadline - 40, time.time() + 80)
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.03
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_opt(hF, Nf, d6, k=60, lr0=0.015)
        if c5F < best_c5:
            best_c5, best_h, best_N = c5F, hF.copy(), Nf

    # Phase 7: Binary cleanup from continuous
    if time.time() < deadline - 10:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = fix_sum_binary(h_bin, len(h_bin))
        d7 = min(deadline - 3, time.time() + 30)
        h7, _ = binary_search(h_bin, len(h_bin), d7, max_sweeps=500, top_k=5)
        c7 = compute_c5(h7, len(h_bin))
        if c7 < best_c5:
            best_c5, best_h, best_N = c7, h7.copy(), len(h_bin)

    # Final verification
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
