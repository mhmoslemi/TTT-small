def build_A(h, N):
    # A[p][k] for p=0..N-1, k=0..2N-2
    # L_k = k - (N-1)
    # A[p][k] = h[p+L_k] (if 0<=p+L_k<N) - (1-h[p-L_k]) (if 0<=p-L_k<N)
    lags = np.arange(2*N-1) - (N-1)  # shape (2N-1,)
    # For each p, compute h[p+lags] and h[p-lags] with masking
    # p + lags: shape (N, 2N-1)
    p_idx = np.arange(N)[:, None]  # (N, 1)
    pos1 = p_idx + lags[None, :]  # (N, 2N-1): p + L_k
    pos2 = p_idx - lags[None, :]  # (N, 2N-1): p - L_k
    
    mask1 = (pos1 >= 0) & (pos1 < N)
    mask2 = (pos2 >= 0) & (pos2 < N)
    
    # Safe indexing
    pos1_safe = np.clip(pos1, 0, N-1)
    pos2_safe = np.clip(pos2, 0, N-1)
    
    A = h[pos1_safe] * mask1 - (1 - h[pos2_safe]) * mask2
    return A  # (N, 2N-1)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def build_A_active(h, N, act_lags):
    """Build A[p, j] for p in 0..N-1, j in act_lags.
    A[p, j] = h[p+L_j]*[valid] - (1-h[p-L_j])*[valid]
    where L_j = act_lags[j] - (N-1)
    """
    lags = act_lags.astype(int) - (N - 1)
    p_idx = np.arange(N)[:, None]
    pos1 = p_idx + lags[None, :]
    pos2 = p_idx - lags[None, :]
    m1 = (pos1 >= 0) & (pos1 < N)
    m2 = (pos2 >= 0) & (pos2 < N)
    pos1c = np.clip(pos1, 0, N - 1)
    pos2c = np.clip(pos2, 0, N - 1)
    return h[pos1c] * m1 - (1 - h[pos2c]) * m2

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def build_A_active(h, N, act_lags):
    lags = act_lags.astype(np.int64) - (N - 1)
    p_idx = np.arange(N)[:, None]
    pos1 = p_idx + lags[None, :]
    pos2 = p_idx - lags[None, :]
    m1 = (pos1 >= 0) & (pos1 < N)
    m2 = (pos2 >= 0) & (pos2 < N)
    pos1c = np.clip(pos1, 0, N - 1)
    pos2c = np.clip(pos2, 0, N - 1)
    return h[pos1c].astype(float) * m1 - (1.0 - h[pos2c].astype(float)) * m2

def fix_sum(h, N):
    target = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[target:]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h

def exact_ls(h, N, deadline, n_active=100, max_sweeps=500000, perturb_every=35, max_perturb=4):
    h = fix_sum(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    L_tot = 2 * N - 1
    dx = 2.0 / N
    no_improve = 0
    sweep = 0

    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_on = len(ones)
        n_zo = len(zeros)
        if n_on == 0 or n_zo == 0:
            break

        cur_max = float(c.max())
        n_act = min(n_active, L_tot)
        act_idx = np.argsort(c)[-n_act:]

        A_act = build_A_active(h, N, act_idx)
        A_o = A_act[ones]
        A_z = A_act[zeros]
        c_act = c[act_idx]

        C = c_act[None, None, :] + A_o[:, None, :] - A_z[None, :, :]
        new_maxes = C.max(axis=2)
        del C

        flat_min = int(new_maxes.argmin())
        p_i = flat_min // n_zo
        q_i = flat_min % n_zo
        quick_max = float(new_maxes[p_i, q_i])

        if quick_max < cur_max - 1e-10:
            p = int(ones[p_i])
            q = int(zeros[q_i])
            h[p] = 0
            h[q] = 1
            c = fft_corr(h)
            full_max = float(c.max())
            if full_max < cur_max - 1e-10:
                no_improve = 0
                if full_max < best_c:
                    best_c = full_max
                    best_h = h.copy()
            else:
                h[p] = 1
                h[q] = 0
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= perturb_every:
            n_pert = int(np.random.randint(1, max_perturb + 1))
            if n_pert > min(n_on, n_zo):
                n_pert = min(n_on, n_zo)
            p_sel = np.random.choice(ones, n_pert, replace=False)
            q_sel = np.random.choice(zeros, n_pert, replace=False)
            h[p_sel] = 0
            h[q_sel] = 1
            c = fft_corr(h)
            no_improve = 0

    return best_h, best_c * dx
