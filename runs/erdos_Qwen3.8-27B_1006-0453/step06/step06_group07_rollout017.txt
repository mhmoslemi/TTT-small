import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum_binary(h, N):
    """Project to exactly N/2 ones (binary)."""
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    target = N // 2
    if n1 == target:
        return h
    if n1 > target:
        # Need to set more to 0
        diff = n1 - target
        ones = np.where(h == 1)[0]
        # Set the "least important" ones to 0 (by their value in continuous h)
        if len(ones) <= diff:
            h[:] = 0
            # Set target random positions to 1
            idx = np.random.choice(N, target, replace=False)
            h[idx] = 1
        else:
            rng = np.random.RandomState(42)
            idx = rng.choice(ones, diff, replace=False)
            h[idx] = 0
    else:
        diff = target - n1
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(42)
        idx = rng.choice(zeros, diff, replace=False)
        h[idx] = 1
    return h

def compute_D(h, N):
    """Compute the change in correlation for each (p, lag) where h[p]=1 is set to 0."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def swap_search(h, N, deadline, max_sweeps=1000, verbose=False):
    """Single-swap binary search with simulated annealing."""
    h = h.astype(float).copy()
    target = N // 2
    if int(h.sum()) != target:
        h = project_sum_binary(h, N)
    
    corr = fft_corr(h)
    P = float(corr.max())
    best_c = P
    best_h = h.copy()
    stall = 0
    temp = 0.5 * P
    rng = np.random.RandomState(123)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        D = compute_D(h, N)
        Dz = D[zeros, :]
        
        # Try all ones (or sample)
        p_list = ones if len(ones) <= 100 else rng.choice(ones, 100, replace=False)
        
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = corr - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is not None:
            # Simulated annealing: accept with some probability if worse
            if best_new_P <= P:
                accept = True
            else:
                delta = best_new_P - P
                accept = rng.random() < np.exp(-delta / temp)
            
            if accept:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
                corr = fft_corr(h)
                P = float(corr.max())
                if P < best_c - 1e-13:
                    best_c = P
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        # Cooldown
        if sweep % 10 == 0:
            temp *= 0.98
            if temp < 0.01 * best_c:
                temp = 0.01 * best_c
        
        # Periodic perturbation
        if stall > 50 and len(ones) > 2 and len(zeros) > 2:
            n_flip = min(3, len(ones), len(zeros))
            for _ in range(n_flip):
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
            P = float(corr.max())
            temp *= 2.0
            stall = 0
    
    return best_h, best_c * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def c5_val(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        h = np.clip(h + d / len(h), 0.0, 1.0)
    return h

def make_binary(h, N, seed=42):
    hb = (h >= 0.5).astype(float)
    n1 = int(hb.sum())
    tgt = N // 2
    if n1 == tgt:
        return hb
    rng = np.random.RandomState(seed)
    if n1 > tgt:
        ones = np.where(hb == 1)[0]
        idx = rng.choice(ones, n1 - tgt, replace=False)
        hb[idx] = 0
    else:
        zeros = np.where(hb == 0)[0]
        idx = rng.choice(zeros, tgt - n1, replace=False)
        hb[idx] = 1
    return hb

def grad_corr(h, N, weights=None):
    """Gradient of weighted sum of correlations."""
    corr = fft_corr(h)
    if weights is None:
        i = np.argmax(corr)
        weights = np.zeros(2*N-1)
        weights[i] = 1.0
    grad = np.zeros(N)
    center = N - 1
    for m in range(2*N-1):
        w = weights[m]
        if w < 1e-12:

        L = abs(m - center)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[L:] += w * (1.0 - h[:N-L])
            grad[:N-L] -= w * h[L:]
    return grad

    tgt = N / 2.0
    best_h = h.copy()

        if softmax:
            cm = corr.max()
            w = np.exp(b * (corr - cm))
            ws = w.sum()
            if ws < 1e-30:
                w = np.zeros(2*N-1)
                w[np.argmax(corr)] = 1.0
            else:
                w /= ws
        else:

        v = c5_val(h_new, N)
        if v < best_v:
            best_v = v
            best_h = h_new.copy()
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.98
            b = min(b * 1.03, 300)
    return best_h, best_v

def swap_search(h, N, deadline, max_sweeps=500):
    h = h.astype(float).copy()
    tgt = N // 2
    if int(h.sum()) != tgt:
        h = make_binary(h, N)
    corr = fft_corr(h)
    P = float(corr.max())
    best_c = P
    best_h = h.copy()
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Compute D

        pos = np.arange(N)[:, None]
        sv = shifts[None, :]
        pm = pos - sv
        pp = pos + sv
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N-1)
        pp_c = np.clip(pp, 0, N-1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm + hp * vp
        Dz = D[zeros, :]

        for p in ones:

                stall = 0

            grad += w * (1.0 - 2.0 * h)
        else:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
    return grad

def softmax_weights(corr, beta):
    cm = corr.max()
    w = np.exp(beta * (corr - cm))
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def topk_weights(corr, k):
    k = min(k, len(corr))
    top = np.argpartition(corr, -k)[-k:]
    w = np.zeros(len(corr))
    w[top] = 1.0 / k
    return w

def cont_opt(h, N, deadline, beta=40.0, lr0=0.006, use_softmax=True, topk=5):
    h = h.astype(float).copy()
    tgt = N * 0.5
    best_h = h.copy()
    best_v = c5_val(h, N)
    lr = lr0
    b = beta
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        if use_softmax:
            w = softmax_weights(corr, b)
        else:
            w = topk_weights(corr, topk)
        g = grad_corr(h, N, w)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = proj_sum(h_new, tgt)
        v = c5_val(h_new, N)
        if v < best_v:
            best_v = v
            best_h = h_new.copy()
        h = h_new
        it += 1
        if it % 150 == 0:
            lr = max(lr * 0.985, 0.0005)
            b = min(b * 1.04, 300.0)
    return best_h, best_v

def cont_sa(h, N, deadline, beta=50.0, lr0=0.005, temp0=0.005, seed=7):
    h = h.astype(float).copy()
    tgt = N * 0.5
    best_h = h.copy()
    best_v = c5_val(h, N)
    lr = lr0
    temp = temp0
    rng = np.random.RandomState(seed)
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        w = softmax_weights(corr, beta)
        g = grad_corr(h, N, w)
        noise = rng.randn(N) * temp * 0.05
        h_new = np.clip(h - lr * g + noise, 0.0, 1.0)
        h_new = proj_sum(h_new, tgt)
        v_new = c5_val(h_new, N)
        v_cur = c5_val(h, N)
        if v_new < best_v:
            best_v = v_new
            best_h = h_new.copy()
        if v_new <= v_cur or rng.random() < np.exp(-(v_new - v_cur) / max(temp, 1e-10)):
            h = h_new
        it += 1
        if it % 100 == 0:
            temp *= 0.96
            lr = max(lr * 0.99, 0.0003)
    return best_h, best_v

def swap_search(h, N, deadline, max_sweeps=600, sample_frac=1.0, seed=123):
    h = h.astype(float).copy()
    tgt = N // 2
    if int(h.sum()) != tgt:
        h = make_binary(h, N, seed)
    corr = fft_corr(h)
    P = float(corr.max())
    best_c = P
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(seed + 999)
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        shifts = np.arange(2 * N - 1) - (N - 1)
        pos = np.arange(N)[:, None]
        sv = shifts[None, :]
        pm = pos - sv
        pp = pos + sv
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm + hp * vp
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        if sample_frac < 1.0 and len(ones) > 60:
            p_list = rng.choice(ones, max(40, int(len(ones) * sample_frac)), replace=False)
        else:
            p_list = ones
        for p in p_list:
            base = corr - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            stall += 1
            if stall > 40:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            corr = fft_corr(h)
            P = float(corr.max())
            if P < best_c - 1e-13:
                best_c = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 40:
                break
    return best_h, best_c * (2.0 / N)

def upsample(h, N_old, N_new):
    x_o = np.linspace(0.0, 2.0, N_old, endpoint=False)
    x_n = np.linspace(0.0, 2.0, N_new, endpoint=False)
    return np.interp(x_n, x_o, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    tgt = N0 * 0.5
    best_h = h0.copy()
    best_c5 = c5_val(best_h, N0)

    # Phase 1: Continuous optimization (softmax gradient)
    d1 = min(dl - 450, t0 + 120)
    h1, c1 = cont_opt(h0, N0, d1, beta=30.0, lr0=0.006, use_softmax=True)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary local search
    hb = make_binary(best_h, N0, seed)
    d2 = min(dl - 350, time.time() + 70)
    h2, c2 = swap_search(hb, N0, d2, max_sweeps=500, sample_frac=1.0, seed=seed)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 3: Continuous from binary (topk gradient)
    d3 = min(dl - 250, time.time() + 100)
    h3, c3 = cont_opt(best_h, N0, d3, beta=60.0, lr0=0.005, use_softmax=False, topk=5)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: Simulated annealing continuous
    d4 = min(dl - 180, time.time() + 100)
    h4, c4 = cont_sa(best_h, N0, d4, beta=50.0, lr0=0.004, temp0=0.004, seed=seed + 1)
    if c4 < best_c5:
        best_c5, best_h = c4, h4.copy()

    # Phase 5: ILS - perturbed continuous restarts
    ilt = 0
    while time.time() < dl - 120:
        amp = 0.02 + ilt * 0.005
        h_cur = best_h.copy() + np.random.randn(N0) * amp
        h_cur = proj_sum(h_cur, tgt)
        d5 = min(dl - 100, time.time() + 45)
        h5, c5_ = cont_opt(h_cur, N0, d5, beta=50.0, lr0=0.004, use_softmax=True)
        if c5_ < best_c5:
            best_c5, best_h = c5_, h5.copy()
            ilt = 0
        else:
            ilt += 1
        if ilt > 8:
            ilt = 0

    # Phase 6: Binary polish at N0
    if time.time() < dl - 40:
        hbp = make_binary(best_h, N0, seed + 2)
        d6 = min(dl - 25, time.time() + 30)
        h6, c6 = swap_search(hbp, N0, d6, max_sweeps=300, sample_frac=1.0, seed=seed + 2)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: Try at 2*N0 with continuous
    N1 = 2 * N0
    if N1 <= 1000 and time.time() < dl - 20:
        h_up = proj_sum(upsample(best_h, len(best_h), N1), N1 * 0.5)
        d7 = min(dl - 5, time.time() + 25)
        h7, c7 = cont_opt(h_up, N1, d7, beta=40.0, lr0=0.004, use_softmax=True)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Final: ensure valid and compute
    best_h = proj_sum(best_h, len(best_h) * 0.5)
    final_c5 = c5_val(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
