    L = 2 * N - 1
    H = np.fft.fft(h, L)

    """D[i,k] = change in corr[k] when h[i] goes 1->0"""
    shifts = np.arange(2 * N - 1) - (N - 1)  # s = -(N-1), ..., 0, ..., N-1
    pos = np.arange(N)[:, None]  # N x 1
    sv = shifts[None, :]  # 1 x (2N-1)
    pm = pos - sv  # i - s: for h[i-s] term
    pp = pos + sv  # i + s: for (1-h)[i+s] term
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)

    pp_c = np.clip(pp, 0, N - 1)

    hp_val = np.where(vp, h[pp_c], 0.0)

    return -(1.0 - hp_val) * vp.astype(float) + hm * vm.astype(float)

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hp) * vp.astype(float) + hm * vm.astype(float)

def binarize(h, N):
    """Convert to binary with exactly N/2 ones."""
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 != N // 2:
        if n1 > N // 2:
            oi = np.where(hb == 1)[0]
            # Turn off the smallest h values
            idx = oi[np.argsort(h[oi])[:n1 - N//2]]
            hb[idx] = 0
        else:
            zi = np.where(hb == 0)[0]
            idx = zi[np.argsort(-h[zi])[:N//2 - n1]]
            hb[idx] = 1
    return hb

def tabu_search(h, N, deadline, tabu_size=30, max_sweeps=5000):
    """Intensive tabu local search."""
    h = h.astype(float).copy()
    nh = N // 2
    assert int(h.sum()) == nh, f"Sum {h.sum()} != {nh}"
    
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    
    tabu = {}  # (p,q) -> iteration when it becomes non-tabu
    it = 0
    stall = 0
    
    while time.time() < deadline and it < max_sweeps:
        it += 1
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        D = compute_D(h, N)
        
        # Find best move
        best_delta = 0.0  # We want delta < 0 (improving)
        best_pq = None
        
        # Sample a subset of ones for speed
        sample_ones = ones if len(ones) <= 50 else np.random.choice(ones, 50, replace=False)
        
        for p in sample_ones:
            base = c + D[p, :]
            # For each zero q, new_max = max(base - D[q,:])
            cand = base[None, :] - D[zeros, :]  # (nz, 2N-1)
            new_maxes = cand.max(axis=1)  # (nz,)
            deltas = new_maxes - P
            mi = int(np.argmin(deltas))
            d = float(deltas[mi])
            q = int(zeros[mi])
            if d < best_delta:
                best_delta = d
                best_pq = (int(p), q)
        
        if best_pq is not None and best_delta < -1e-12:
            p, q = best_pq
            if tabu.get((p, q), 0) > it and tabu.get((q, p), 0) > it:
                # Tabu, accept anyway (aspiration: improves best)
                if P < best_c:
                    pass
                else:
                    continue
            h[p] = 0.0
            h[q] = 1.0
            tabu[(p, q)] = it + tabu_size
            tabu[(q, p)] = it + tabu_size  # Prevent undoing
            c = fft_corr(h)
        else:
            # No improving move: try random perturbation
            if stall > 50:
                n_flips = np.random.randint(3, 8)
                for _ in range(n_flips):
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                    p = int(np.random.randint(len(oi)))
                    q = int(np.random.randint(len(zi)))
                    h[oi[p]], h[zi[q]] = 0, 1
                c = fft_corr(h)
                stall = 0
            else:
                break
    
    return best_h, best_c * (2.0 / N)

If h = [1,1,...,1,0,0,...,0] (first half 1s, second half 0s):
- At shift s (1 to N/2): overlap is N/2 - s. So c[s] = N/2 - s.
- Max is at s=1: c = N/2 - 1 = 189. Terrible.

If h is a "sawtooth" or "triangle" pattern: 1,1,0,0,1,1,0,0,...
We computed earlier: max c = N/2. Also terrible.

What about a more "spread out" pattern? Like the van der Corput sequence or a Halton sequence mapped to binary?

Actually, I think the key insight is: we want h(x) to be such that for EVERY shift s, h(x) and h(x+s) are "in agreement" (both 1 or both 0) as much as possible. This means h should have a high "self-similarity" at all scales.

This is related to the concept of a "scrambled" or "equidistributed" sequence, but with the specific goal of minimizing the max deviation from the expected value.

In fact, this is very close to the problem of constructing a sequence with low "L∞ discrepancy" or "low discrepancy sequence" (Quasi-Monte Carlo). The Halton, Sobol, and other QM sequences have this property.

For a 1D Sobol sequence with N points, the binary representation of each point gives a "spread out" pattern. But I'm not sure this directly translates to our problem.

Let me try a different angle. What if I think of the problem as:
- We have N positions, each either 1 or 0
- For each pair of positions (i, j) with i ≠ j, the "contribution" to the correlation at shift |i-j| is 1 if h[i]≠h[j] and... wait, it's specifically h[i]=1 and h[j]=0.

So the correlation at shift s is: #{i : h[i]=1, h[i+s]=0}

We want to minimize the max over s.

This is equivalent to: for each shift s, the number of "1→0" transitions at distance s should be as small and as uniform as possible.

A "1→0" at distance s means: position i has a 1, position i+s has a 0.
A "0→1" at distance s means: position i has a 0, position i+s has a 1.

Note: #{1→0 at dist s} + #{0→1 at dist s} = #{disagreements at dist s}
And: #{1→0 at dist s} = #{0→1 at dist s} (by counting: total 1s = total 0s = N/2)

Wait, is that true? Let's see:
#{i: h[i]=1, h[i+s]=0} = #{i: h[i]=0, h[i+s]=1}?

The LHS counts positions where the pair (i, i+s) is (1,0).
The RHS counts positions where the pair (i, i+s) is (0,1).

These are NOT necessarily equal! They would be equal if the sequence is "balanced" in a certain sense.

Actually: #{(1,0) pairs at dist s} - #{(0,1) pairs at dist s} = sum_i (h[i] - h[i+s]) * [h[i]≠h[i+s]]
= sum_i (h[i] - h[i+s]) (since when h[i]=h[i+s], the product is 0)
= sum_i h[i] - sum_i h[i+s]
= sum_i h[i] - sum_{i+s valid} h[i+s]
= (sum of h over all) - (sum of h over all positions except the boundary)

For shift s > 0: = N/2 - (sum h[0:N-s]) = sum h[N-s:N] - (sum h[0:s]) (approximately, if the sums are balanced)

This is NOT zero in general. So the (1,0) and (0,1) counts are not equal.

def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))
