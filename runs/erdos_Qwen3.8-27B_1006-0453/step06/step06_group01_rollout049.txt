def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_F(h, N):
    """Compute flip delta matrix F of shape (N, 2N-1)."""
    L = 2*N - 1
    d = np.arange(L) - (N-1)  # shape (L,), values from -(N-1) to N-1
    i = np.arange(N)[:, None]  # shape (N, 1)
    # term1: (1-h[i+d]) where 0 <= i+d < N
    idx1 = i + d[None, :]  # shape (N, L)
    mask1 = (idx1 >= 0) & (idx1 < N)
    term1 = np.where(mask1, 1.0 - h[idx1], 0.0)
    # term2: h[i-d] where 0 <= i-d < N
    idx2 = i - d[None, :]  # shape (N, L)
    mask2 = (idx2 >= 0) & (idx2 < N)
    term2 = np.where(mask2, h[idx2], 0.0)
    return term1 - term2

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def compute_F(h, N):
    L = 2 * N - 1
    d = np.arange(L) - (N - 1)
    i = np.arange(N)
    idx1 = i[:, None] + d[None, :]
    mask1 = (idx1 >= 0) & (idx1 < N)
    idx2 = i[:, None] - d[None, :]
    mask2 = (idx2 >= 0) & (idx2 < N)
    term1 = np.where(mask1, 1.0 - h[np.clip(idx1, 0, N-1)], 0.0)
    term2 = np.where(mask2, h[np.clip(idx2, 0, N-1)], 0.0)
    return term1 - term2
