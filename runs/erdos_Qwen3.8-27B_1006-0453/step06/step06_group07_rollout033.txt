import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def softmax_grad(h, N, T):
    """Compute gradient of soft-max of correlations."""
    c = fft_corr(h)
    center = N - 1
    # Weight with soft-max
    c_shifted = c - c.max()
    w = np.exp(T * c_shifted)
    w /= w.sum()
    
    grad = np.zeros(N)
    for k in range(2*N - 1):
        L = k - center
        wk = w[k]
        if wk < 1e-8:
            continue
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            # i in [0, L-1]: (1-h[i+L])
            grad[:L] += wk * (1.0 - h[L:N])
            # i in [L, N-1-L]: (1-h[i+L]) - h[i-L]
            if L <= N-1-L:
                grad[L:N-L] += wk * ((1.0 - h[2*L:N]) - h[:N-2*L])
            # i in [N-L, N-1]: -h[i-L]
            grad[N-L:] += wk * (-h[:L])
        else:
            M = -L
            # i in [0, M-1]: -h[i+M]
            grad[:M] += wk * (-h[M:N])
            # i in [M, N-1-M]: (1-h[i-M]) - h[i+M]
            if M <= N-1-M:

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_grad(h, N, T):
    c = fft_corr(h)
    center = N - 1
    c_max = c.max()
    w = np.exp(T * (c - c_max))
    w /= w.sum()
    grad = np.zeros(N)
    for k in range(2*N - 1):
        L = k - center
        wk = w[k]

        elif L > 0:

                grad[N-M:] += wk * (1.0 - h[:M])

        zeros = np.where(hr <= 0.5)[0]
        for _ in range(nr):
            if len(ones) > 0 and len(zeros) > 0:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                hr[p], hr[q] = 0.0, 1.0
        hr = project_sum(hr, N0 / 2.0)
        d5 = min(deadline - 120 + restart * 20, time.time() + 60)
        h5, c5_5 = binary_swap(hr, N0, d5, max_it=200, sf=0.8, perturb=30)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final Nesterov at best resolution
    if time.time() < deadline - 60:
        Nf = len(best_h) if len(best_h) <= 760 else N0
        hf = best_h.copy() if Nf == len(best_h) else upsample(best_h, Nf)
        hf = project_sum(hf + np.random.randn(Nf) * 0.02, Nf / 2.0)
        d6 = min(deadline - 30, time.time() + 40)
        h6, c5_6 = nesterov_opt(hf, Nf, d6, T=50.0, lr=0.006, beta=0.92)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final binary polish
    if time.time() < deadline - 2:
        Np = len(best_h)
        h7, c5_7 = binary_swap(best_h, Np, deadline - 0.5, max_it=200,
                               sf=1.0 if Np <= 400 else 0.5, perturb=20)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
