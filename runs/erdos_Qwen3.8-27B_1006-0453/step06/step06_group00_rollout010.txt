    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

        else:

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

    h = h.copy()

        idx = np.where(h == 1)[0]

    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def greedy_binary(h, N, deadline, max_rounds=500, batch_sz=12):
    h = h.astype(float).copy()
    best_c = np.inf
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
        C = compute_C_mat(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        Co = C[ones]
        Cz = C[zeros]
        best_mv = P
        best_z = best_o = -1
        for zs in range(0, len(zeros), batch_sz):
            zb = zeros[zs:min(zs + batch_sz, len(zeros))]
            nm = (corr[None, None, :] + Cz[zb][:, None, :] - Co[None, :, :]).max(axis=2)
            mi = np.unravel_index(np.argmin(nm), nm.shape)
            mv = float(nm[mi])
            if mv < best_mv:
                best_mv = mv
                best_z = zb[mi[0]]
                best_o = ones[mi[1]]
        if best_mv >= P - 1e-12:
            break
        h[best_o] = 0
        h[best_z] = 1
    return best_h, best_c

def ils_binary(h, N, deadline, max_outer=50):
