def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softgrad_opt(h, N, deadline, T0=20.0, T1=80.0, lr0=0.02, decay=0.95):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        T = T0 + (T1 - T0) * min(1.0, t / 500.0)
        sc = T * c
        sc -= sc.max()  # numerical stability
        w = np.exp(sc)
        w /= w.sum()
        D = compute_D(h, N)
        grad = -(D * w[None, :]).sum(axis=1) * dx
        grad -= grad.mean()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h_new = h + lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h_new, target)
        if t % 200 == 0:
            lr *= decay
        val = float(np.max(c))
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val * dx
