    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def anneal(h, N, deadline, T0, cool, max_stall):
    h = fix_binary(h, N).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    stall = 0
    T = T0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    s_arr = np.arange(-(N-1), N)

    while time.time() < deadline and T > 0.001:
        r = np.random.random()
        if r < 0.72:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            ps, pm = p + s_arr, p - s_arr
            qs, qm = q + s_arr, q - s_arr
            vps = (ps >= 0) & (ps < N)
            vpm = (pm >= 0) & (pm < N)
            vqs = (qs >= 0) & (qs < N)
            vqm = (qm >= 0) & (qm < N)
            d = np.zeros(2*N - 1)
            d[vps] -= (1.0 - h[np.clip(ps, 0, N-1)[vps]])
            d[vpm] += h[np.clip(pm, 0, N-1)[vpm]]
            d[vqs] += (1.0 - h[np.clip(qs, 0, N-1)[vqs]])
            d[vqm] -= h[np.clip(qm, 0, N-1)[vqm]]
            nc = corr + d
            npk = float(nc.max())
            if npk < cur_pk or np.random.random() < np.exp(-(npk - cur_pk)/max(T, 1e-9)):
                h[p], h[q] = 0.0, 1.0
                corr = nc
                cur_pk = npk
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        elif r < 0.90:
            if len(ones) < 2 or len(zeros) < 2:
                T *= cool
                continue
            i1, i2 = np.random.randint(len(ones), 2)
            j1, j2 = np.random.randint(len(zeros), 2)
            if i1 == i2 or j1 == j2:
                T *= cool
                continue
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            d1 = _one_delta(h, N, p1, q1, s_arr)
            d2 = _one_delta(h, N, p2, q2, s_arr)
            nc = corr + d1 + d2
            npk = float(nc.max())
            if npk < cur_pk or np.random.random() < np.exp(-(npk - cur_pk)/max(T, 1e-9)):
                h[p1] = h[p2] = 0.0
                h[q1] = h[q2] = 1.0
                corr = fft_corr(h)
                cur_pk = float(corr.max())
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        else:
            nb = max(2, N // 50)
            fl = min(nb, len(ones)//2)
            ft = min(nb, len(zeros)//2)
            fi = np.random.choice(ones, fl, replace=False)
            fj = np.random.choice(zeros, ft, replace=False)
            h[fi] = 0.0
            h[fj] = 1.0
            corr = fft_corr(h)
            cur_pk = float(corr.max())
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

        if cur_pk < best_pk:
            best_pk = cur_pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        T *= cool
        if stall > max_stall:
            h = best_h.copy()
            corr = fft_corr(h)
            cur_pk = best_pk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            T = T0 * 0.45
            stall = 0
    return best_h, best_pk

def _one_delta(h, N, p, q, s_arr):
    ps, pm = p + s_arr, p - s_arr
    qs, qm = q + s_arr, q - s_arr
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2*N - 1)
    d[vps] -= (1.0 - h[np.clip(ps, 0, N-1)[vps]])
    d[vpm] += h[np.clip(pm, 0, N-1)[vpm]]
    d[vqs] += (1.0 - h[np.clip(qs, 0, N-1)[vqs]])
    d[vqm] -= h[np.clip(qm, 0, N-1)[vqm]]
    return d

def cont_opt(h, N, deadline, top_k, lr):
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    center = N - 1
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai) - center
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += (1.0 - h[s:])
                grad[s:] -= h[:N-s]
            else:
                ss = -s
                grad[ss:] += (1.0 - h[:N-ss])
                grad[:N-ss] -= h[ss:]
        grad /= top_k
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        if t % 800 == 0:
            lr *= 0.96
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = eval_c5(h0)

    # Phase 1: Continuous from perturbed start (50s)
    d1 = min(deadline - 600, t0 + 50)
    hc = h0 + np.random.randn(N0) * 0.03
    hc = proj_sum(hc, N0/2.0)
    hC, pkC = cont_opt(hc, N0, d1, 30, 0.004)
    if pkC*(2.0/N0) < best_c5:
        best_c5 = pkC*(2.0/N0)
        best_h = hC.copy()

    # Phase 2: Binary anneal on main N (250s)
    d2 = min(deadline - 400, time.time() + 250)
    hb = fix_binary(best_h, N0)
    hB, pkB = anneal(hb, N0, d2, 0.06, 0.9994, 700)
    if pkB*(2.0/N0) < best_c5:
        best_c5 = pkB*(2.0/N0)
        best_h = hB.copy()

    # Phase 3: Multi-N exploration (220s)
    n_list = [320, 360, 400, 440, 520]
    for Nt in n_list:
        if time.time() > deadline - 150:
            break
        Na = Nt if Nt % 2 == 0 else Nt + 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Na, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_binary(hi, Na)
        sub = min(deadline - 100, time.time() + 44)
        hrt, pkr = anneal(hbt, Na, sub, 0.05, 0.9995, 450)
        if pkr*(2.0/Na) < best_c5:
            best_c5 = pkr*(2.0/Na)
            best_h = hrt.copy()

    # Phase 4: Second-round binary anneal, lower T (180s)
    if time.time() < deadline - 60:
        d4 = min(deadline - 40, time.time() + 180)
        hf = fix_binary(best_h, len(best_h))
        h4, pk4 = anneal(hf, len(best_h), d4, 0.03, 0.9996, 600)
        if pk4*(2.0/len(best_h)) < best_c5:
            best_c5 = pk4*(2.0/len(best_h))
            best_h = h4.copy()

    # Phase 5: Continuous micro-polish (30s)
    if time.time() < deadline - 12:
        d5 = min(deadline - 6, time.time() + 30)
        hc2 = best_h.astype(float) + np.random.randn(len(best_h))*0.01
        hc2 = proj_sum(hc2, len(best_h)/2.0)
        hC2, pkC2 = cont_opt(hc2, len(best_h), d5, 50, 0.0015)
        if pkC2*(2.0/len(best_h)) < best_c5:
            best_c5 = pkC2*(2.0/len(best_h))
            best_h = hC2.copy()

    # Phase 6: Final binary polish at very low T (remaining)
    if time.time() < deadline - 2:
        hf2 = fix_binary(best_h, len(best_h))
        h6, pk6 = anneal(hf2, len(best_h), deadline-1, 0.008, 0.9998, 350)
        if pk6*(2.0/len(best_h)) < best_c5:
            best_c5 = pk6*(2.0/len(best_h))
            best_h = h6.copy()

    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
