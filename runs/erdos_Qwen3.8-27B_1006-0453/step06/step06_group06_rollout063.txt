def continuous_adam(h, N, deadline, lr=0.003, top_k=40):
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        
        # Compute gradient on top-k correlations
        # Use soft-max weighting for smoother gradient
        beta = 5.0 + min(20.0, t * 0.01)  # increase sharpness over time
        corr_normed = corr - corr.max()
        weights = np.exp(beta * corr_normed)
        weights /= weights.sum()
        
        # Use top-k for efficiency
        idx = np.argpartition(corr, -top_k)[-top_k:]
        w_idx = weights[idx]
        w_idx /= w_idx.sum()
        
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += w_idx[list(idx).index(ai)] * (1.0 - 2.0*h)
            elif s > 0:
                grad[:N-s] += w_idx[list(idx).index(ai)] * (1.0-h[s:])
                grad[s:] -= w_idx[list(idx).index(ai)] * h[:N-s]
            else:
                s2 = -s
                grad[s2:] += w_idx[list(idx).index(ai)] * (1.0-h[:N-s2])
                grad[:N-s2] -= w_idx[list(idx).index(ai)] * h[s2:]
        
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1-b1**t)
        vh = v / (1-b2**t)
        
        # Adaptive learning rate
        current_lr = lr * (0.5 + 0.5 * np.cos(np.pi * min(1.0, t/5000)))
        h = proj_sum(h - current_lr * mh / np.sqrt(vh + eps), N/2.0)
    
    return best_h, best_pk
```

def compute_gradient(h, corr, N, top_k=50, beta=10.0):
    center = N - 1
    idx = np.argpartition(corr, -top_k)[-top_k:]
    # Soft-max weights
    c = corr[idx]
    w = np.exp(beta * (c - c.max()))
    w /= w.sum()
    
    grad = np.zeros(N)
    for wi, ki in zip(w, idx):
        s = int(ki - center)
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:N-s]
        else:
            s2 = -s
            grad[s2:] += wi * (1.0 - h[:N-s2])
            grad[:N-s2] -= wi * h[s2:]
    return grad
```
