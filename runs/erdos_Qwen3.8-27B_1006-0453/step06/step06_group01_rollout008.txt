    N = len(h)
    # Soft-max weights
    c = corr[top_idx]
    log_sum = np.logaddexp.reduce(c / tau)  # or manual
    p = np.exp(c / tau - log_sum)
    
    grad = np.zeros(N)
    for w, idx in zip(p, top_idx):

            # -h[j+lag] for j <= N-1-lag

        d4 = min(deadline - 40, time.time() + 80)
        h4, c4 = binary_descent((best_h > 0.5).astype(float), N0, d4, max_sweeps=2000, n_sample=45)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
    # Phase 5: Final Adam micro-polish
    if time.time() < deadline - 5:
        d5 = deadline - 2
        h5 = best_h.astype(float) + np.random.randn(N0) * 0.015
        h5 = project_sum(h5, N0 / 2.0)
        h5f, c5f = adam_opt(h5, N0, d5, top_k=100, tau_init=1.5, tau_end=0.05, lr0=0.015)
        if c5f < best_c5:
            best_c5, best_h = c5f, h5f
    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
