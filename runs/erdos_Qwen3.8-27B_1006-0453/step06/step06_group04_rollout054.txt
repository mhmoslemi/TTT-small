                        h[qq] = 1
                    c2 = fft_corr(h)
                    if int(c2.max()) < best_int:
                        best_int = int(c2.max())
                        best_h = h.copy()
                        c = c2
                    else:
                        for pp in ones[i1]:
                            h[pp] = 1
                        for qq in zeros[i2]:
                            h[qq] = 0
                        c = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Coordinate descent - main optimization (500s)
    d1 = min(deadline - 550, t0 + 500)
    hc = h0 + np.random.randn(N0) * 0.02
    hc = project_sum(hc, N0 / 2.0)
    h1, cm1 = coord_descent(hc, N0, d1, prec=18)
    c5_1 = compute_c5(h1, N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Subgradient fine-tuning (180s)
    d2 = min(deadline - 350, max(d1, t0) + 180)
    hc2 = best_h + np.random.randn(N0) * 0.01
    hc2 = project_sum(hc2, N0 / 2.0)
    h2, c5_2 = subgrad_opt(hc2, N0, d2, k=100, lr0=0.01, temp0=0.002)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Directed binary optimization (200s)
    d3 = min(deadline - 150, max(d2, t0) + 200)
    h3, c5_3 = binary_directed(best_h, N0, d3, n_cand=30, kick=35)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Second coordinate descent from best (150s)
    d4 = min(deadline - 50, max(d3, t0) + 150)
    hc4 = best_h + np.random.randn(N0) * 0.008
    hc4 = project_sum(hc4, N0 / 2.0)
    h4, cm4 = coord_descent(hc4, N0, d4, prec=20)
    c5_4 = compute_c5(h4, N0)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Multi-resolution (50s)
    N2 = N0 * 2
    if time.time() < deadline - 30:
        h_hi = interp_to_N(best_h, N2)
        h_hi = project_sum(h_hi, N2 / 2.0)
        sub_dl = min(deadline - 10, time.time() + 35)
        h5, cm5 = coord_descent(h_hi, N2, sub_dl, prec=15)
        c5_5 = compute_c5(h5, N2)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final polish (remaining)
    if time.time() < deadline - 3:
        hF = best_h + np.random.randn(len(best_h)) * 0.005
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cmF = coord_descent(hF, len(best_h), deadline - 2, prec=22)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
