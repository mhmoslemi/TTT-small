import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

        y = h + m
        corr = fft_corr(y)
        pk_h = float(np.max(fft_corr(h)))
        if pk_h < best_pk:
            best_pk = pk_h
            best_h = h.copy()
        # LSE weights
        c = corr - corr.max()
        w = np.exp(T * c)
        wsum = w.sum()
        if wsum > 1e-300:
            w = w / wsum
        else:
            w = np.zeros(len(corr))
            w[int(np.argmax(corr))] = 1.0
        # Top-k
        idx = np.argpartition(w, -k)[-k:]
        idx = idx[np.argsort(w[idx])[::-1]]
        grad = np.zeros(N)
        for ii in idx:
            wi = w[ii]
            if wi < 1e-10:
                break
            L = int(ii - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N-L])
                grad[:N-L] -= wi * y[L:]
            else:
                M = -L
                grad[:N-M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N-M]
        grad /= max(1, np.sum(w[idx] > 1e-10))
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 50 == 0:
            lr *= 0.99
            beta = min(0.98, 0.92 + it * 0.0001)
    return best_h, best_pk

def swap_eval(corr, h, p, q, top_idx, N):
    """Evaluate new max if we swap h[p]<->h[q] (p is 1, q is 0)."""
    center = N - 1
    new_corr = corr.copy()
    for ai in top_idx:
        L = int(ai - center)
        # Change in c[L] from setting h[p]=0 and h[q]=1
        delta = 0.0
        # h[p] changes from 1 to 0: affects terms where p appears as h[i] or h[i+L]
        # In c[L] = sum h[m+L]*(1-h[m]):
        # h[p] as h[m+L]: m = p-L, contributes h[p]*(1-h[p-L]) → changes by -(1-h[p-L])
        if 0 <= p - L < N:
            delta -= (1.0 - h[p - L])
        # h[p] as h[m]: m = p, contributes h[p+L]*(1-h[p]) → changes by +h[p+L]  
        if 0 <= p + L < N:
            delta += h[p + L]
        # h[q] changes from 0 to 1:
        if 0 <= q - L < N:
            delta += (1.0 - h[q - L])  # was 0*(1-h[q-L]), now 1*(1-h[q-L])
        if 0 <= q + L < N:
            delta -= h[q + L]  # h[q+L]*(1-h[q]) was h[q+L]*1, now h[q+L]*0
        new_corr[ai] += delta
    return float(np.max(new_corr))

def binary_opt(h, N, deadline, sample_n=200, top_m=25, anneal=0.0):

def coord_decsweep(h, N, corr, top_idx, center):
    """One vectorized coordinate descent sweep."""
    P = len(top_idx)
    shifts = (top_idx - center).astype(int)
    h_new = h.copy()
    for j in range(N):
        # For each top peak, compute coefficient of h[j]
        valid_m = (j - shifts >= 0) & (j - shifts < N)
        valid_p = (j + shifts >= 0) & (j + shifts < N)
        a = np.zeros(P)
        a[valid_m] += (1.0 - h[j - shifts[valid_m]])
        a[valid_p] -= h[j + shifts[valid_p]]
        b = corr[top_idx] - a * h[j]
        # min_{x in [0,1]} max_i (a_i * x + b_i)
        # Try x=0, x=1, x=h[j]
        vals_0 = b  # a_i*0 + b_i = b_i
        vals_1 = a + b
        obj0 = vals_0.max()
        obj1 = vals_1.max()
        objh = float((a * h[j] + b).max())
        best_x = h[j]
        best_obj = objh
        if obj0 < best_obj:
            best_obj = obj0
            best_x = 0.0
        if obj1 < best_obj:
            best_obj = obj1
            best_x = 1.0
        # Also try the crossing point of top two
        if a.max() > 0 and a.min() < 0:
            i_max = a.argmax()
            i_min = a.argmin()
            if abs(a[i_max] - a[i_min]) > 1e-10:
                x_cross = (b[i_min] - b[i_max]) / (a[i_max] - a[i_min])
                if 0 <= x_cross <= 1:
                    obj_cross = float((a * x_cross + b).max())
                    if obj_cross < best_obj:
                        best_obj = obj_cross
                        best_x = x_cross
        h_new[j] = best_x
    return h_new

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):

    d1 = min(deadline - 500, t0 + 280)
    hc = best_h + np.random.randn(N0) * 0.03
    hc = project_sum(hc, N0 / 2.0)
    hn, c5n = nesterov_lse(hc, N0, d1, T0=2.0, T1=70.0, k=120, lr0=0.016)
    if c5n < best_c5:
        best_c5, best_h = c5n, hn.copy()

    # Phase 2: Coordinate descent refinement
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 180)
        hc2 = project_sum(best_h + np.random.randn(N0) * 0.015, N0 / 2.0)
        center = N0 - 1
        for outer in range(30):
            if time.time() > d2:
                break
            corr2 = fft_corr(hc2)
            top_idx2 = np.argpartition(corr2, -100)[-100:]
            for _ in range(4):
                hc2 = coord_sweep(hc2, N0, fft_corr(hc2), top_idx2, center)
                hc2 = project_sum(hc2, N0 / 2.0)
            c2v = float(np.max(fft_corr(hc2)))
            if c2v * (2.0 / N0) < best_c5:
                best_c5 = c2v * (2.0 / N0)
                best_h = hc2.copy()

    # Phase 3: Binary local search
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 220)
        hb, _ = binary_opt(best_h, N0, d3, sample_n=200, top_m=30, anneal=0.04, seed=seed)
        c5b = compute_c5(hb)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()

    # Phase 4: Multi-resolution exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    res_list = [360, 400, 420, 440, 480, 520, 600]
    for N_try in res_list:
        if time.time() > deadline - 40:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_i = np.interp(x_new, x_best, best_h)
        h_bt = (h_i > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        rs_r = np.random.RandomState(seed + N_try)
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            idx = idx[rs_r.permutation(len(idx))[nh:]]
            h_bt[idx] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            idx = idx[rs_r.permutation(len(idx))[:nh - n1]]
            h_bt[idx] = 1
        sub_dl = min(deadline - 20, time.time() + 55)
        hr, _ = binary_opt(h_bt, N_try, sub_dl, sample_n=150, top_m=25, anneal=0.12, seed=seed + N_try)
        c5r = compute_c5(hr)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 5: Final Nesterov polish
    if time.time() < deadline - 15:
        hF = best_h + np.random.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_lse(hF, len(best_h), deadline - 3, T0=8.0, T1=100.0, k=100, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 6: Final binary cleanup
    if time.time() < deadline - 0.5:
        hF2 = (best_h > 0.5).astype(float)
        hF2, _ = binary_opt(hF2, len(best_h), deadline - 0.3, sample_n=150, top_m=20, anneal=0.0, seed=seed + 99)
        c5F2 = compute_c5(hF2)
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
