def exact_1d_min(a, b, lo=0.0, hi_bound=1.0):
    """Find min_{x in [lo, hi_bound]} max_k (a[k] + x * b[k])"""
    # Binary search on the answer
    # Lower bound: max(a) (at x=0) or max(a + hi_bound*b) (at x=hi_bound)
    # Upper bound: min of the two
    
    def feasible(V):
        # Check if exists x in [lo, hi_bound] with a_k + x*b_k <= V for all k
        lo_x = lo
        hi_x = hi_bound
        for k in range(len(a)):
            if b[k] > 1e-15:
                hi_x = min(hi_x, (V - a[k]) / b[k])
            elif b[k] < -1e-15:
                lo_x = max(lo_x, (V - a[k]) / b[k])
            else:
                if a[k] > V:
                    return False
        return lo_x <= hi_x + 1e-14
    
    # Get bounds for V
    V_lo = min(np.max(a), np.max(a + hi_bound * b))  # actually need max, not min
    # Lower bound for V: the minimum possible max is at least max of the min possible values
    # Actually V_lo = min over x of max_k(a_k + x*b_k) >= max_k(min_x(a_k+x*b_k))
    # For each k, min over x in [lo,hi_bound] of a_k + x*b_k:
    # if b_k >= 0: a_k + lo*b_k
    # if b_k < 0: a_k + hi_bound*b_k
    # So V_lo = max_k of these
    V_lo = max(a[k] + (lo if b[k] >= 0 else hi_bound) * b[k] for k in range(len(a)))
    V_hi = min(np.max(a), np.max(a + hi_bound * b))
    
    for _ in range(60):
        V_mid = (V_lo + V_hi) / 2
        if feasible(V_mid):
            V_hi = V_mid
        else:
            V_lo = V_mid
    
    # Now find the optimal x at V_hi
    V = V_hi + 1e-15
    lo_x = lo
    hi_x = hi_bound
    for k in range(len(a)):
        if b[k] > 1e-15:
            hi_x = min(hi_x, (V - a[k]) / b[k])
        elif b[k] < -1e-15:
            lo_x = max(lo_x, (V - a[k]) / b[k])
    x_opt = (lo_x + hi_x) / 2
    return max(lo, min(hi_bound, x_opt))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

            return h
        h = h + room * (diff / tot)

            else:
                best_h[i] = old
                best_h = project(best_h, n)
        if not improved:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(160)
        h0[:80] = 1.0

    n = len(h0)
    h0 = project(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP warmup with temperature ramp (12% budget)
    t1 = t0 + budget_s * 0.12
    for T in [15, 80, 400, 2000]:
        if elapsed() > t1:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Heavy coordinate descent (43% budget)
    t2 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=5000)

    # Phase 3: High-T SLSQP polish (10% budget)
    t3 = t0 + budget_s * 0.65
    for T in [15000, 80000]:
        if elapsed() > t3:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: More coordinate descent (18% budget)
    t4 = t0 + budget_s * 0.83
    best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=5000)

    # Phase 5: Basin hopping + coordinate (8% budget)
    t5 = t0 + budget_s * 0.92
    for it in range(15):
        if elapsed() > t5:
            break
        step = 0.025 * (1.0 - it / 15.0) + 0.005
        h_pert = best_h + step * np.random.randn(n)
        h_pert = project(h_pert, n)
        hc = slsqp_opt(h_pert, 20000, n, maxiter=400)
        hc, c5v = coord_descent(hc, n, min(t0 + budget_s * 0.91, elapsed() + 40), max_sweeps=800)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Try alternate resolutions (5% budget)
    t6 = t0 + budget_s * 0.97
    for n_try in [144, 180]:
        if elapsed() > t6:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        hc = slsqp_opt(h_interp, 5000, n_try, maxiter=800)
        hc, c5v = coord_descent(hc, n_try, t0 + budget_s * 0.965, max_sweeps=2000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            n = n_try

    # Final polish
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, len(best_h), t0 + budget_s * 0.99, max_sweeps=2000)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
