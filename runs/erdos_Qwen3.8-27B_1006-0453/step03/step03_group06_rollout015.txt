            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_and_grad(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = 2 * n - 1
    if topk is not None and topk < m:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(m)
    vals = corr[idx] * dx
    mx = vals.max()
    e = np.exp(T * (vals - mx))
    s = e.sum()
    logsumexp = mx + np.log(s)
    obj = logsumexp / T
    p = e / s
    shifts = (idx - (n - 1)).astype(int)
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = p[i] * dx
        if w < 1e-16:
            continue
        sh = shifts[i]
        if sh == 0:
            grad += w * (1.0 - 2.0 * h)
        elif sh > 0:
            grad[:n - sh] += w * (1.0 - h[sh:])
            grad[sh:] -= w * h[:n - sh]
        else:
            sp = -sh
            grad[sp:] += w * (1.0 - h[:n - sp])
            grad[:n - sp] -= w * h[sp:]
    return obj, grad

def nesterov_opt(h0, T, topk, max_iter, alpha0, deadline=None):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    best = h0.copy()
    best_c5 = compute_c5(best)
    t_val = 1.0
    alpha = alpha0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        beta = (t_val - 1.0) / (t_val + 2.0) if t_val > 1.0 else 0.0
        y = project_feasible(h + beta * (h - h_prev), n)
        obj, grad = smooth_c5_and_grad(y, T, topk)
        h_new = project_feasible(y - alpha * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        h_prev = h.copy()
        h = h_new
        t_val += 1.0
        alpha *= 0.9997
    return best, best_c5

def coordinate_grind(h0, n_trials, deadline=None):
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        top = np.argsort(corr)[-3:]
        for k_sh in top:
            shift = k_sh - (n - 1)
            if shift > 0:
                region = slice(0, n - shift)
            elif shift < 0:
                region = slice(-shift, n)
            else:
                region = slice(0, n)
            j = region.start + np.random.randint(region.stop - region.start)
            for delta in [0.02, -0.02, 0.05, -0.05]:
                h_try = best.copy()
                h_try[j] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best = h_try.copy()
                    break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-resolution gradient optimization
    res_list = [60, 80, 100, 120, 160, 200, 240]
    phase1_deadline = t0 + budget_s * 0.32
    for n_try in res_list:
        if time.time() > phase1_deadline:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_cur = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_cur += np.random.uniform(-0.03, 0.03, n_try)
        h_cur = project_feasible(h_cur, n_try)
        for T in [20, 60, 200, 600, 2000]:
            if time.time() > phase1_deadline:
                break
            tk = min(5, 2 * n_try - 1)
            h_cur, c5v = nesterov_opt(h_cur, T, tk, 3000, 3.0 / n_try, phase1_deadline)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_cur.copy()

    # Phase 2: Focused high-T optimization at best resolution
    n_best = len(best_h)
    h_cur = best_h.copy()
    phase2_deadline = t0 + budget_s * 0.62
    for T in [100, 400, 1500, 5000, 15000, 50000]:
        if time.time() > phase2_deadline:
            break
        tk = min(3, 2 * n_best - 1)
        h_cur, c5v = nesterov_opt(h_cur, T, tk, 6000, 2.0 / n_best, phase2_deadline)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_cur.copy()
    # Coordinate grind
    if time.time() < phase2_deadline:
        best_h, best_c5 = coordinate_grind(best_h, 2000, phase2_deadline)

    # Phase 3: Perturbation and restart
    n_cur = len(best_h)
    h_cur = best_h.copy()
    phase3_deadline = t0 + budget_s * 0.85
    for it in range(400):
        if time.time() > phase3_deadline:
            break
        step = 0.08 * max(0.1, 1.0 - it / 400.0) + 0.01
        r = np.random.rand()
        if r < 0.4:
            h_pert = h_cur + step * np.random.randn(n_cur)
        elif r < 0.7:
            blen = np.random.randint(3, max(4, n_cur // 3))
            start = np.random.randint(0, max(1, n_cur - blen))
            h_pert = h_cur.copy()
            h_pert[start:start + blen] += step * (np.random.rand(blen) * 2.0 - 1.0)
        else:
            h_pert = h_cur.copy()
            i1 = np.random.randint(0, n_cur)
            i2 = np.random.randint(0, n_cur)
            h_pert[i1] += step * 0.5
            h_pert[i2] -= step * 0.5
        h_pert = project_feasible(h_pert, n_cur)
        T_use = int(np.random.choice([200, 600, 2000, 5000]))
        tk = min(3, 2 * n_cur - 1)
        h_opt, c5v = nesterov_opt(h_pert, T_use, tk, 600, 2.5 / n_cur, phase3_deadline)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt.copy()
            h_cur = h_opt.copy()

    # Phase 4: Try different resolutions with best pattern
    phase4_deadline = t0 + budget_s * 0.94
    for n_try in [80, 100, 140, 180, 240, 320]:
        if time.time() > phase4_deadline:
            break
        if n_try == n_cur:
            continue
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        for T in [200, 1000, 5000]:
            if time.time() > phase4_deadline:
                break
            tk = min(3, 2 * n_try - 1)
            h_try, c5v = nesterov_opt(h_try, T, tk, 3000, 2.0 / n_try, phase4_deadline)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try.copy()

    # Phase 5: Final ultra-fine polish
    n_final = len(best_h)
    h_final = best_h.copy()
    phase5_deadline = t0 + budget_s * 0.995
    for T in [30000, 100000, 300000]:
        if time.time() > phase5_deadline:
            break
        tk = min(3, 2 * n_final - 1)
        h_final, c5v = nesterov_opt(h_final, T, tk, 4000, 1.0 / n_final, phase5_deadline)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_final.copy()
    # Final coordinate grind
    if time.time() < phase5_deadline:
        best_h, best_c5 = coordinate_grind(best_h, 1500, phase5_deadline)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
