import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def compute_Q(h):
    """Compute autocorrelation Q_k = sum_i h[i]*h[i+k] for k=0..n-1"""
    n = len(h)
    # Use FFT for efficiency
    # Q_k = sum_i h[i]*h[i+k] for k >= 0
    # This is the first n values of the full autocorrelation
    # np.correlate(h, h, mode='full')[n-1:] gives Q_0, Q_1, ..., Q_{n-1}
    ac = np.correlate(h, h, mode='full')
    return ac[n-1:]  # Q_0 through Q_{n-1}

def grad_Q_at_lag(h, k):
    """Gradient of Q_k = sum_i h[i]*h[i+k] with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g = 2.0 * h
    else:
        g[:n-k] += h[k:]      # dQ_k/dh[i] = h[i+k] for i=0..n-k-1
        g[k:] += h[:n-k]      # dQ_k/dh[i] = h[i-k] for i=k..n-1
    return g

def projected_subgrad_method(h0, n, max_iter, lr_init, time_limit):
    """Projected gradient with momentum for minimizing C5"""
    h = h0.copy()
    m = np.zeros(n)  # momentum first moment
    v = np.zeros(n)  # momentum second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 1
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        # Compute Q_k
        Q = compute_Q(h)
        # We want to minimize max_k (n/2 - Q_k) = n/2 - min_k Q_k
        # So we want to maximize min_k Q_k
        min_Q = np.min(Q)
        
        # Active set: lags within tolerance of minimum
        tol = 0.02 * (n/2 - min_Q) + 1e-6
        active = np.where(Q <= min_Q + tol)[0]
        
        # Gradient of -min_k Q_k ≈ -1/|A| * sum_{k in A} grad_Q_k
        # We want to descend on f = max(c - Q_k), so gradient is -1/|A| sum grad_Q_k
        # Step: h += lr * 1/|A| * sum_{k in A} grad_Q_k (ascent on min Q)
        
        g = np.zeros(n)
        for k in active:
            g += grad_Q_at_lag(h, k)
        g /= len(active)
        
        # Adam update (we want to ascend on min_Q, so negative gradient of f)
        # f gradient = -g (since f = max(c-Q_k), grad f = -1/|A| sum grad_Q_k = -g)
        # We want to minimize f, so step = -lr * grad_f = lr * g
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        h_new = make_feasible(h_new, n)

        if c5_new < best_c5:

            best_h = h_new.copy()
        
        # Adaptive lr: reduce if not improving for a while
        # (simplified: use decay)
        lr = lr_init * (1.0 / (1.0 + 0.01 * it))
        
        h = h_new
        t += 1
    
    return best_h, best_c5
