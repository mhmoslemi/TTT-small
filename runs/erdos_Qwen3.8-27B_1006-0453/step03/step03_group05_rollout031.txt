def project_feasible(h, n):

            if total < 1e-14:

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def targeted_repair(h, n, n_worst=15, strength=0.15):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_worst:][::-1]
    h_new = h.copy()
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            j = i + lag
            if h_new[i] > 0.4 and h_new[j] < 0.6:
                delta = min((h_new[i] - 0.4) * strength, (0.6 - h_new[j]) * strength)
                if delta > 1e-7:
                    h_new[i] -= delta
                    h_new[j] += delta
            elif h_new[i] < 0.6 and h_new[j] > 0.4:
                delta = min((0.6 - h_new[i]) * strength, (h_new[j] - 0.4) * strength)
                if delta > 1e-7:
                    h_new[i] += delta
                    h_new[j] -= delta
    return project_feasible(h_new, n)

def adam_subgrad(h, n_steps=200, top_k=5, lr=0.005, decay=0.999):
    n = len(h)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h)
    beta1, beta2 = 0.8, 0.999
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        wts = corr[top_idx].astype(float)
        ws = wts.sum()
        wts = wts / ws if ws > 1e-10 else np.ones(top_k) / top_k
        grad = np.zeros(n)
        for wi, idx in zip(wts, top_idx):
            lag = int(idx) - (n - 1)
            i_min = max(0, -lag)
            i_max = min(n, n - lag)
            s1 = slice(i_min, i_max)
            s2 = slice(i_min + lag, i_max + lag)
            grad[s1] += wi * (1.0 - h[s2])
            grad[s2] -= wi * h[s1]
        grad *= (2.0 / n)
        t = step + 1
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad * grad)
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
        h = h_new
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Quick analytical-gradient SLSQP (10%)
    for T in [400, 1000]:
        if elapsed() > budget_s * 0.08:
            break
        h_new = slsqq_opt(best_h, T, 250, 1e-14)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new
    h_new = topk_opt(best_h, 4, 200, 1e-14)
    c5_new = compute_c5(h_new)
    if c5_new < best_c5:
        best_c5, best_h = c5_new, h_new

    # Phase 2: Multi-resolution exploration (22%)
    for nr in [100, 140, 180]:
        if elapsed() > budget_s * 0.30:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        h_t = np.interp(x_new, x_old, best_h)
        h_t = project_feasible(h_t, nr)
        h_t = slsqq_opt(h_t, 600, 300, 1e-13)
        h_t = topk_opt(h_t, 5, 200, 1e-14)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_h, best_n = c5_t, h_t, nr
        # Also try with small perturbation
        h_t2 = project_feasible(h_t + np.random.randn(nr) * 0.01, nr)
        h_t2 = slsqq_opt(h_t2, 800, 250, 1e-13)
        c5_t2 = compute_c5(h_t2)
        if c5_t2 < best_c5:
            best_c5, best_h, best_n = c5_t2, h_t2, nr

    # Phase 3: Diverse multi-start (15%)
    for trial in range(6):
        if elapsed() > budget_s * 0.45:
            break
        r = np.random.random()
        if r < 0.4:
            h_s = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * np.random.randn(best_n)), best_n)
        elif r < 0.7:
            h_s = np.zeros(best_n)
            block = best_n // 4
            perm = np.random.permutation(best_n)
            h_s[perm[:best_n // 2]] = 1.0
            h_s = project_feasible(h_s + np.random.randn(best_n) * 0.1, best_n)
        else:
            h_s = project_feasible(best_h + np.random.randn(best_n) * 0.08, best_n)
        h_s = slsqq_opt(h_s, 500, 250, 1e-12)
        h_s = topk_opt(h_s, 5, 150, 1e-13)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s

    # Phase 4: Intensive local refinement (30%)
    for rnd in range(8):
        if elapsed() > budget_s * 0.75:
            break
        T_use = 300 + rpc * 400
        h_new = slsqq_opt(best_h, T_use, 300, 1e-14)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new
        h_new = topk_opt(best_h, 3, 250, 1e-14)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new
        # Adam subgradient micro-steps
        h_sg, c5_sg = adam_subgrad(best_h, 150, 4, 0.004, 0.999)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg
        # Targeted repair
        h_r = targeted_repair(best_h, best_n, 12, 0.12)
        h_r = slsqq_opt(h_r, 800, 200, 1e-14)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r

    # Phase 5: Basin hopping (15%)
    for trial in range(30):
        if elapsed() > budget_s * 0.90:
            break
        sc = 0.02 + 0.03 * np.random.rand()
        h_p = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        h_p = slsqq_opt(h_p, 700, 150, 1e-13)
        h_p = topk_opt(h_p, 4, 100, 1e-14)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p

    # Phase 6: Final polish (10%)
    if elapsed() < budget_s * 0.96:
        h_f = slsqq_opt(best_h, 8000, 400, 1e-15)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        h_f = topk_opt(best_h, 2, 300, 1e-15)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        # One more targeted repair + optimize
        h_f = targeted_repair(best_h, best_n, 8, 0.08)
        h_f = slsqq_opt(h_f, 5000, 300, 1e-15)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
