def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def grad_f_d(h, d):
    """Gradient of f_d(h) = sum_{j=0}^{n-1-d} h[j+d]*(1-h[j]) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        # (1-h[i-d]) for d <= i <= n-1
        idx1 = np.arange(d, n)
        g[idx1] += (1.0 - h[idx1 - d])
        # -h[i+d] for 0 <= i <= n-1-d
        idx2 = np.arange(0, n - d)
        g[idx2] -= h[idx2 + d]
    return g

def subgradient_max(h, K=5):
    """Weighted subgradient of max_k f_k(h)"""
    corr = compute_corr_arr(h)
    n = len(h)
    # Due to symmetry, only need lags 0 to n-1
    # corr index m corresponds to lag = m - (n-1)
    # We want lags 0 to n-1, which are indices n-1 to 2n-2
    lags = np.arange(n)
    corr_lags = corr[n-1:]  # lags 0 to n-1
    # Find top K
    top_idx = np.argsort(corr_lags)[-K:]
    vals = corr_lags[top_idx]
    # Softmax weights
    w = np.exp(100 * (vals - vals.max()))
    w = w / w.sum()
    # Weighted gradient
    g = np.zeros(len(h))
    for wi, li in zip(w, top_idx):
        g += wi * grad_f_d(h, int(li))
    return g

def smooth_obj(hv, T):
    nv = len(hv)
    dx = 2.0 / nv
    corr = np.correlate(hv, 1.0 - hv, mode='full') * dx
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val

def sum_con(hv):
    return hv.sum() - len(hv) / 2.0

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                  bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

            h_p[idx] = 1.0 - h_p[idx]
        elif ptype == 3:
            # Target worst-correlation region
            corr = compute_corr_arr(bh_h)
            max_lag = abs(int(np.argmax(corr) - (n_bh - 1)))
            half = max(3, min(n_bh // 4, max_lag))
            h_p = bh_h.copy()
            h_p[:half] += 0.04 * np.random.randn(half)
            h_p[-half:] += 0.04 * np.random.randn(half)
        else:
            # Large jump for exploration
            h_p = bh_h + 0.15 * np.random.randn(n_bh)

        h_p = make_feasible(h_p, n_bh)
        T_use = 300 + 300 * (it // 15)
        try:
            h_new = slsqp_optimize(h_p, T_use, n_bh, 200, 1e-13)
            c5v = compute_c5(h_new)
        except Exception:
            h_new = h_p
            c5v = compute_c5(h_p)

        temp = 0.0015 * (1.0 - it / max(1, n_iters)) + 0.0001
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 5: Final high-T polish (10%)
    n_fin = len(best_h)
    for T in [10000, 50000, 200000]:
        if time.time() > t0 + budget_s * 0.995:
            break
        try:
            h_new = slsqp_optimize(best_h.copy(), T, n_fin, 3000, 1e-18)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    # Extra: try subgradient on final answer for micro-improvements
    if time.time() < t0 + budget_s * 0.99:
        h_final, c5_final = subgradient_phase(best_h.copy(), n_fin, t0 + budget_s * 0.99)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
