import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m <= 0:
        return 0.0
    shifted = T * (corr - m)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1500, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent_fast(h, n, t_end, max_sweeps=300):
    """Fast coordinate descent with incremental correlation updates."""
    best_h = h.copy()
    corr = get_corr(best_h).copy()
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    ncorr = len(corr)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        # Process coordinates in random order
        order = np.arange(n)
        np.random.shuffle(order)
        for j in order:
            if time.time() > t_end:
                best_h = project(best_h, n)
                best_c5 = compute_c5(best_h)
                return best_h, best_c5
            # Compute coefficient for coordinate j in each corr[k]
            # corr[k] = sum_i h[i](1-h[i+k])
            # d corr[k] / d h[j] = (1-h[j+k]) * 1[0<=j+k<n] + (-h[j-k]) * 1[0<=j-k<n]
            # For k=0: (1-h[j]) + (-h[j]) = 1 - 2h[j]
            coef = np.zeros(ncorr)
            # Vectorized: for lag k, j+k is valid when -(j) <= k <= n-1-j
            # and j-k is valid when j-n+1 <= k <= j (i.e., -(j) <= k <= j... let me just loop)
            for k_shift in range(-(n-1), n):
                c = 0.0
                jp = j + k_shift
                jm = j - k_shift
                if 0 <= jp < n:
                    c += 1.0 - best_h[jp]
                if 0 <= jm < n:
                    c -= best_h[jm]
                coef[k_shift + n - 1] = c
            
            # 1D problem: min_{δ∈[0,1]} max_k (corr[k] + δ*coef[k])
            # This is convex PL, so use bounded scalar minimization
            vals = corr + 0.0  # at δ=0
            f0 = np.max(vals)
            vals = corr + 1.0 * coef  # at δ=1
            f1 = np.max(vals)
            
            if f0 <= f1 + 1e-15:
                # Try δ near 0
                res = minimize_scalar(lambda d: np.max(corr + d * coef),
                                     bounds=(0.0, 0.5), method='bounded',
                                     options={'xatol': 1e-9})
            else:
                res = minimize_scalar(lambda d: np.max(corr + d * coef),
                                     bounds=(0.0, 1.0), method='bounded',
                                     options={'xatol': 1e-9})
            hi_new = res.x
            old_hj = best_h[j]
            delta = hi_new - old_hj
            if abs(delta) < 1e-10:
                continue
            
            # Update corr incrementally
            corr += delta * coef
            best_h[j] = hi_new
            c5_new = np.max(corr) * dx
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
        
        # End of sweep: project and recompute
        if improved:
            best_h = project(best_h, n)
            corr = get_corr(best_h)
            best_c5 = np.max(corr) * dx
    
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_cd_state = {'corr': None, 'coef': None}

def _cd_f1d(d):
    return np.max(_cd_state['corr'] + d * _cd_state['coef'])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m <= 0:
        return 0.0
    shifted = T * (corr - m)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1500, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def _compute_coef_vec(h, j, n):
    ks = np.arange(-(n - 1), n)
    jp = j + ks
    jm = j - ks
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    term1 = np.zeros(2 * n - 1)
    term2 = np.zeros(2 * n - 1)
    np.subtract(1.0, h[np.clip(jp, 0, n - 1)], out=term1)
    term1[~valid_p] = 0.0
    np.negative(h[np.clip(jm, 0, n - 1)], out=term2)
    term2[~valid_m] = 0.0
    return term1 + term2

def coord_descent_fast(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    corr = get_corr(best_h).copy()
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    ncorr = 2 * n - 1
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.arange(n)
        np.random.shuffle(order)
        for j in order:
            if time.time() > t_end:
                best_h = project(best_h, n)
                best_c5 = compute_c5(best_h)
                return best_h, best_c5
            coef = _compute_coef_vec(best_h, j, n)
            _cd_state['corr'] = corr
            _cd_state['coef'] = coef
            f0 = np.max(corr)
            f1 = np.max(corr + coef)
            if f0 <= f1 + 1e-14:
                res = minimize_scalar(_cd_f1d, bounds=(0.0, 0.6), method='bounded',
                                      options={'xatol': 1e-10})
            else:
                res = minimize_scalar(_cd_f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-10})
            hi_new = res.x
            old_hj = best_h[j]
            delta = hi_new - old_hj
            if abs(delta) < 1e-11:
                continue
            corr += delta * coef
            best_h[j] = hi_new
            c5_new = np.max(corr) * dx
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
        best_h = project(best_h, n)
        corr = get_corr(best_h)
        best_c5 = np.max(corr) * dx

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)
