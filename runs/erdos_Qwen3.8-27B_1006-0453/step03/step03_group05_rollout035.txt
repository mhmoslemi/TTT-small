def compute_c5(h):

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h += room * (diff / t)
        else:
            room = h.copy()
            t = room.sum()
            if t < 1e-15:
                break
            h += room * (diff / t)
        h = np.clip(h, 0.0, 1.0)
    return h

def ls_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_obj(h, k, n):

    idx = np.argpartition(corr, -k)[-k:]
    return float(np.mean(corr[idx]) * dx)

    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase A: L-BFGS-B with escalating temperature (fast global exploration)
    for T in [50, 200, 800, 3000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        hl = lbfgsb_opt(best_h, T, best_n, 250)
        cl = compute_c5(hl)
        if cl < best_c5:
            best_c5, best_h = cl, hl

    # Phase B: SLSQP with log-sum-exp (refined local optimization)
    for T in [500, 2000, 8000, 30000]:
        if time.time() - t0 > budget_s * 0.20:
            break
        ht = slsqp_opt(best_h, ls_obj, (T, best_n), 300, 1e-13)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase C: Weighted-max objective (targets multi-peak max)
    if time.time() - t0 < budget_s * 0.28:
        hw = slsqp_opt(best_h, weight_max_obj, (best_n,), 350, 1e-13)
        cw = compute_c5(hw)
        if cw < best_c5:
            best_c5, best_h = cw, hw

    # Phase D: 0-1 Simulated Annealing + continuous refinement
    if time.time() - t0 < budget_s * 0.55:
        h01 = (best_h > 0.4).astype(float)
        h01 = project_feasible(h01, best_n)
        for restart in range(4):
            if time.time() - t0 > budget_s * 0.50:
                break
            hsa, csa = sa_binary(h01, best_n, 60000, 0.004, 0.99996)
            if csa < best_c5:
                best_c5, best_h = csa, hsa.copy()
            hr = slsqp_opt(hsa, ls_obj, (2000, best_n), 250, 1e-13)
            cr = compute_c5(hr)
            if cr < best_c5:
                best_c5, best_h = cr, hr
            h01 = (best_h > 0.4).astype(float)
            h01 = project_feasible(h01, best_n)

    # Phase E: Basin hopping with mixed perturbations
    for trial in range(60):
        if time.time() - t0 > budget_s * 0.75:
            break
        r = np.random.random()
        if r < 0.25:
            sc = 0.05 + 0.02 * (1.0 - trial / 60.0)
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.45:
            hp = spectral_perturb(best_h, best_n, 0.02)
        elif r < 0.65:
            hp = best_h.copy()
            L = np.random.randint(3, max(4, best_n // 10))
            s = np.random.randint(0, max(1, best_n - L))
            hp[s:s+L] = 1.0 - hp[s:s+L]
            hp = project_feasible(hp, best_n)
        elif r < 0.80:
            hp = best_h.copy()
            for _ in range(np.random.randint(3, 8)):
                i, j = np.random.choice(best_n, 2, replace=False)
                hp[i], hp[j] = hp[j], hp[i]
            hp = project_feasible(hp, best_n)
        else:
            hp = (best_h > 0.5).astype(float)
            for _ in range(3):
                oi = np.where(hp > 0.5)[0]
                zi = np.where(hp < 0.5)[0]
                if len(oi) > 0 and len(zi) > 0:
                    i = oi[np.random.randint(len(oi))]
                    j = zi[np.random.randint(len(zi))]
                    hp[i], hp[j] = hp[j], hp[i]
            hp = project_feasible(hp, best_n)
        ho = slsqp_opt(hp, ls_obj, (1500, best_n), 150, 1e-12)
        ho = slsqp_opt(ho, topk_obj, (3, best_n), 150, 1e-13)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase F: Multi-resolution search
    for nr in [150, 250, 300]:
        if time.time() - t0 > budget_s * 0.87:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nr)
        hu = project_feasible(np.interp(xn, xo, best_h), nr)
        hu_p = project_feasible(hu + np.random.randn(nr) * 0.015, nr)
        cands = [hu, hu_p]
        best_hi_c5 = best_c5
        best_hi_h = best_h
        for hu_c in cands:
            hh1 = slsqp_opt(hu_c, ls_obj, (800, nr), 200, 1e-12)
            c1 = compute_c5(hh1)
            hh2 = slsqp_opt(hh1, topk_obj, (3, nr), 200, 1e-13)
            c2 = compute_c5(hh2)
            c_hi = min(c1, c2)
            h_hi = hh1 if c1 < c2 else hh2
            if c_hi < best_hi_c5:
                best_hi_c5, best_hi_h = c_hi, h_hi
        if best_hi_c5 < best_c5:
            best_c5, best_h, best_n = best_hi_c5, best_hi_h, nr

    # Phase G: Intensive final polish (very high T, top-2)
    if time.time() - t0 < budget_s * 0.96:
        for T in [10000, 50000, 200000]:
            if time.time() - t0 > budget_s * 0.95:
                break
            hf = slsqp_opt(best_h, ls_obj, (T, best_n), 500, 1e-15)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        hf2 = slsqp_opt(best_h, topk_obj, (2, best_n), 500, 1e-15)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2
        hf3 = slsqp_opt(best_h, weight_max_obj, (best_n,), 400, 1e-15)
        cf3 = compute_c5(hf3)
        if cf3 < best_c5:
            best_c5, best_h = cf3, hf3

    # Final: micro-refinement with L-BFGS-B
    if time.time() - t0 < budget_s * 0.98:
        hf4 = lbfgsb_opt(best_h, 100000, best_n, 300)
        cf4 = compute_c5(hf4)
        if cf4 < best_c5:
            best_c5, best_h = cf4, hf4

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
