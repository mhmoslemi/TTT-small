import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    # For each d from -(n-1) to n-1, weight w[d+n-1]
    # grad_d[i] = (1-h[i+d]) * 1{0<=i+d<n} - h[i-d] * 1{0<=i-d<n}
    for d in range(-(n-1), n):
        wd = w[d + n - 1]
        if wd < 1e-12:
            continue
        if d == 0:
            grad += wd * (1.0 - 2.0 * h)
        elif d > 0:
            # (1-h[i+d]) for i in [0, n-1-d]
            grad[:n-d] += wd * (1.0 - h[d:n])
            # -h[i-d] for i in [d, n-1]
            grad[d:] += -wd * h[:n-d]
        else:
            d2 = -d  # d2 > 0
            # (1-h[i-d2]) for i in [d2, n-1]
            grad[d2:] += wd * (1.0 - h[:n-d2])
            # -h[i+d2] for i in [0, n-1-d2]
            grad[:n-d2] += -wd * h[d2:n]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_proj_descent(h, n, T, t_end, lr0=0.1, lr_decay=0.99):
    """Gradient descent with projection on smooth approximation."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)[0]
    lr = lr0
    for it in range(10000):
        if time.time() > t_end:
            break
        g = smooth_grad(h_cur, T)
        # Project gradient onto constraint sum(h)=n/2
        g = g - np.mean(g)
        h_new = h_cur - lr * g
        h_new = project(h_new, n)
        c5_new = compute_c5(h_new)[0]
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= lr_decay
            if lr < 1e-8:
                break
    return h_cur, c5_cur

def coord_opt(h, n, t_end, max_sweeps=100):
    """Coordinate descent with exact 1D optimization."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)[0]
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            # Compute a_k and b_k for the 1D subproblem
            h_base = best_h.copy()
            hi = best_h[i]
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            # wrt h[i]=x: corr_k = corr_base[k] + x*(1-h[i+d]) - h[i-d]*x (for valid indices)
            # = corr_base[k] + x * ((1-h[i+d])*1{valid} - h[i-d]*1{valid})
            # So a_k = corr_base[k], b_k = (1-h[i+d])*1{0<=i+d<n} - h[i-d]*1{0<=i-d<n}
            b = np.zeros(2*n-1)
            for d in range(-(n-1), n):
                idx = d + n - 1
                coef = 0.0
                if 0 <= i + d < n:
                    coef += (1.0 - h_base[i + d])  # but h_base[i] = 0, so if d=0, this is 1
                if 0 <= i - d < n and (i - d) != i:
                    coef += (-h_base[i - d])
                elif d == 0:
                    coef += (-h_base[i])  # = 0 since h_base[i]=0
                b[idx] = coef
            dx = 2.0 / n
            # Minimize max(corr_base + x*b) * dx over x in [0,1]
            def f1d(x):
                return np.max(corr_base + x * b) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
            x_new = res.x
            if abs(x_new - hi) > 1e-10:
                h_try = best_h.copy()
                h_try[i] = x_new
                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)[0]
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5
