def _smooth_obj(x, T):
    n = len(x)

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)

    h_save = h[i]
    h[i] = 0.0

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def _smooth_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = T * (corr - m)
    mx = np.max(sh)
    return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / T) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-13:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-13:
            return h
        h += r * (d / t)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, T, n, maxiter=600, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, args=(T,),
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def _f1d(val, cb, cf, n):
    return np.max(cb + val * cf) * (2.0 / n)

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_h[i] = 0.0
            cb = np.correlate(best_h, 1.0 - best_h, mode='full')
            coeff = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                if 0 <= i + k < n:
                    c += 1.0 - best_h[i + k]
                if 0 <= i - k < n:
                    if i - k != i + k or k == 0:
                        c -= best_h[i - k]
                coeff[k + n - 1] = c
            best_h[i] = h_save
            res = minimize_scalar(_f1d, args=(cb, coeff, n),
                                 bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-11, 'maxiter': 35})
            if res.fun < best_c5 - 1e-14:
                best_h[i] = res.x
                best_c5 = res.fun
                improved = True
        if not improved:
            break
    return best_h, best_c5

def discrete_ls(h, n, t_end):
    hb = np.where(h > 0.5, 1.0, 0.0)
    ones = np.where(hb == 1.0)[0]
    if len(ones) > n // 2:
        p = np.random.permutation(ones)
        hb[p[len(ones) - n // 2:]] = 0.0
    elif len(ones) < n // 2:
        zeros = np.where(hb == 0.0)[0]
        p = np.random.permutation(zeros)
        hb[p[:n // 2 - len(ones)]] = 1.0
    bc = compute_c5(hb)
    bh = hb.copy()
    for _ in range(400):
        if time.time() > t_end:
            break
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        imp = False

            i = ones[np.random.randint(len(ones))]

                bc = c; bh = hb.copy(); imp = True
            else:
                hb[i] = 1.0; hb[j] = 0.0
        if not imp:
            break
    return bh, bc

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def _smooth_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = T * (corr - m)
    mx = np.max(sh)
    return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / T) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-13:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-13:
            return h
        h += r * (d / t)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, T, n, maxiter=600, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, args=(T,),
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def _f1d(val, cb, cf, n):
    return np.max(cb + val * cf) * (2.0 / n)

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_h[i] = 0.0
            cb = np.correlate(best_h, 1.0 - best_h, mode='full')
            coeff = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i + k
                j2 = i - k
                if 0 <= j1 < n:
                    c += 1.0 - best_h[j1]
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coeff[k + n - 1] = c
            best_h[i] = h_save
            res = minimize_scalar(_f1d, args=(cb, coeff, n),
                                 bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-11, 'maxiter': 35})
            if res.fun < best_c5 - 1e-14:
                best_h[i] = res.x
                best_c5 = res.fun
                improved = True
        if not improved:
            break
    return best_h, best_c5

def discrete_ls(h, n, t_end):
    hb = np.where(h > 0.5, 1.0, 0.0)
    ones = np.where(hb == 1.0)[0]
    if len(ones) > n // 2:
        p = np.random.permutation(ones)
        hb[p[len(ones) - n // 2:]] = 0.0
    elif len(ones) < n // 2:
        zeros = np.where(hb == 0.0)[0]
        p = np.random.permutation(zeros)
        hb[p[:n // 2 - len(ones)]] = 1.0
    bc = compute_c5(hb)
    bh = hb.copy()
    for _ in range(400):
        if time.time() > t_end:
            break
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        imp = False
        nt = min(500, len(ones) * len(zeros) // 10)
        for _t in range(nt):
            if time.time() > t_end:
                return bh, bc
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            hb[i] = 0.0
            hb[j] = 1.0
            c = compute_c5(hb)
            if c < bc - 1e-12:
                bc = c
                bh = hb.copy()
                imp = True
            else:
                hb[i] = 1.0
                hb[j] = 0.0
        if not imp:
            break
    return bh, bc

def _pair_obj(delta, hbase, i, j, hi0, hj0):
    hv = hbase.copy()
    hv[i] = hi0 + delta
    hv[j] = hj0 - delta
    if hv[i] < -1e-9 or hv[i] > 1 + 1e-9 or hv[j] < -1e-9 or hv[j] > 1 + 1e-9:
        return 1e10
    return compute_c5(hv)

def pair_cd(h, n, t_end, max_sweeps=15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        idx = np.random.permutation(n)
        for ii in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            i = idx[ii]
            j = (i + n // 2 + np.random.randint(-4, 5)) % n
            if j == i:
                continue
            lo = max(-best_h[i], best_h[j] - 1.0)
            hi = min(1.0 - best_h[i], best_h[j])
            if hi - lo < 1e-9:
                continue
            res = minimize_scalar(_pair_obj,
                                 args=(best_h, i, j, best_h[i], best_h[j]),
                                 bounds=(lo, hi), method='bounded',
                                 options={'xatol': 1e-11, 'maxiter': 25})
            if res.fun < best_c5 - 1e-13:
                best_h[i] += res.x
                best_h[j] -= res.x
                best_c5 = res.fun
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(120)
        for bl in range(12):
            best_h[bl * 10:bl * 10 + 5] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: discrete local search
    h_dis, c5_dis = discrete_ls(best_h, n, t0 + budget_s * 0.06)

    # Phase 2: SLSQP T-schedule from initial
    for T in [8, 40, 150, 500, 1500, 5000, 15000, 40000]:
        if time.time() > t0 + budget_s * 0.28:
            break
        hc = slsqp_opt(best_h, T, n, 1000, 1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5, best_h = cv, hc

    # Phase 3: exact coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.47, 50)

    # Phase 4: SLSQP from discrete solution
    if c5_dis < best_c5 + 0.005:
        for T in [100, 500, 2000, 8000, 25000]:
            if time.time() > t0 + budget_s * 0.62:
                break
            hc = slsqp_opt(h_dis, T, n, 800, 1e-14)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_c5, best_h = cv, hc

    # Phase 5: basin hopping
    wh, wc = best_h.copy(), best_c5
    for it in range(55):
        if time.time() > t0 + budget_s * 0.80:
            break
        cool = 1.0 - 0.85 * (it / 55.0)
        h_p = wh + (0.02 + 0.12 * cool) * np.random.randn(n)
        h_p = project(h_p, n)
        hc = slsqp_opt(h_p, 300 + 4000 * cool, n, 250, 1e-13)
        cv = compute_c5(hc)
        temp = 0.0025 * cool + 0.0001
        if cv < wc or np.random.rand() < np.exp(-(cv - wc) / temp):
            wh, wc = hc, cv
            if cv < best_c5:
                best_c5, best_h = cv, hc

    # Phase 6: multi-resolution
    if time.time() < t0 + budget_s * 0.88:
        for nt in [120, 200, 144]:
            if time.time() > t0 + budget_s * 0.93:
                break
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, nt, endpoint=False)
            hi = np.interp(xn, xo, best_h)
            hi = project(hi, nt)
            for T in [200, 2000]:
                if time.time() > t0 + budget_s * 0.93:
                    break
                hc = slsqp_opt(hi, T, nt, 600, 1e-14)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_c5, best_h, n = cv, hc, nt

    # Phase 7: final coordinate descent + pair exchanges
    if time.time() < t0 + budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, 20)
    if time.time() < t0 + budget_s * 0.975:
        best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.98, 8)

    # Phase 8: very high-T SLSQP polish
    if time.time() < t0 + budget_s * 0.99:
        hc = slsqp_opt(best_h, 50000, n, 1500, 1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5, best_h = cv, hc

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
