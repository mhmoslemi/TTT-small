            k_idx = np.argpartition(corr, -topk)[-topk:]
            c = corr[k_idx]
        else:

        p = e / e.sum()

            if s >= 0:

                grad[v1] += pk * (1.0 - x[v1+s]) * dx_
                v2 = np.arange(s, n_)
                grad[v2] += -pk * x[v2-s] * dx_

                v1 = np.arange(s2, n_)
                grad[v1] += pk * (1.0 - x[v1-s2]) * dx_
                v2 = np.arange(n_-s2)
                grad[v2] += -pk * x[v2+s2] * dx_

                   constraints=cons, options={'maxiter':maxiter, 'ftol':ftol})

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def proj_feasible(h):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = len(h) / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        t = room.sum()
        if t < 1e-14:
            return h
        h = h + room * (d / t)
    else:
        t = h.sum()
        if t < 1e-14:
            return h
        h = h + h * (d / t)
    return np.clip(h, 0.0, 1.0)

def lse_grad(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):

def subgradient_descent(h0, n_iter=500, topk=5, lr_init=0.01):
    h = h0.copy()
    n = len(h)
    best = h0.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        lr = lr_init / (1.0 + 0.01 * it)
        corr = compute_corr(h)
        idx = np.argpartition(corr, -topk)[-topk:]
        cv = corr[idx]
        w = np.exp(cv - np.max(cv))
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, mi in zip(w, idx):
            s = int(mi) - (n - 1)
            if s >= 0:
                v1 = np.arange(n - s)
                grad[v1] += wi * (1.0 - h[v1 + s]) * dx
                v2 = np.arange(s, n)
                grad[v2] += -wi * h[v2 - s] * dx
            else:
                s2 = -s
                v1 = np.arange(s2, n)
                grad[v1] += wi * (1.0 - h[v1 - s2]) * dx
                v2 = np.arange(n - s2)
                grad[v2] += -wi * h[v2 + s2] * dx
        h_new = proj_feasible(h - lr * grad)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new
            best_c5 = c5_new
        h = h_new
