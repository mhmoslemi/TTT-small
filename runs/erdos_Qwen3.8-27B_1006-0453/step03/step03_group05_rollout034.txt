import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, T, n):
    corr = compute_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / n)

def topk_obj(h, k, n):
    corr = compute_corr(h)
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top)) * (2.0 / n) / k

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp(h, obj, args, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun': sum_con}],
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def coord_descent(h, n_iter, n_active, alpha0, decay):
    """Coordinate descent targeting active correlations."""
    n = len(h)
    best = h.copy()
    best_c = compute_c5(h)
    alpha = alpha0
    for it in range(n_iter):
        corr = compute_corr(best)
        # Find active shifts
        active_idx = np.argpartition(corr, -n_active)[-n_active:]
        # Compute subgradient
        grad = np.zeros(n)
        weights = corr[active_idx].astype(float)
        wsum = weights.sum()
        if wsum > 0:
            weights /= wsum
        for w, ki in zip(weights, active_idx):
            s = int(ki - (n-1))
            for m in range(n):
                if 0 <= m+s < n:
                    grad[m] += w * (1.0 - best[m+s])
                if 0 <= m-s < n:
                    grad[m] += w * (-best[m-s])
        # Update
        for m in range(n):
            best[m] -= alpha * grad[m]
        best = project_feasible(best, n)
        c = compute_c5(best)
        if c < best_c:
            best_c = c
        else:
            alpha *= decay
        alpha = max(alpha, 0.0005)
    return best, best_c

Let me also add a "coordinate descent" approach that's different from subgradient - specifically, single-coordinate updates where we optimize one h[i] at a time given all others fixed.

For a single coordinate h[m], fixing everything else:
corr(s) = A_s + B_s * h[m] (linear in h[m] for each s that involves m)

So the max over s of (A_s + B_s * h[m]) is a piecewise linear function of h[m]. We can find the optimal h[m] in [0,1] by checking all breakpoints.

This is a much more principled coordinate descent! Let me implement it.

For each m:
- For each shift s, determine if m is involved:
  - If m is in the "first" position: corr(s) has term h[m]*(1-h[m+s]), so coefficient of h[m] is (1-h[m+s]), constant is the rest
  - If m is in the "second" position: corr(s) has term h[m-s]*(1-h[m]), so coefficient of h[m] is -h[m-s], constant is the rest
  - m could be in both positions for different s values

So for each s: corr(s) = C_s + D_s * h[m] where C_s and D_s don't depend on h[m].

The max over s of (C_s + D_s * h[m]) is a piecewise linear, convex function of h[m]. Its minimum over [0,1] is at 0, 1, or at an intersection of two lines.

Number of shifts: 2n-1. For n=100, that's 199 lines. Finding the minimum of the max of 199 lines over [0,1] can be done in O(n log n) by sorting slopes and finding the relevant breakpoints. But even O(n²) = 40000 is fine.

Actually, for simplicity, since it's a 1D problem:
- The function f(t) = max_s (C_s + D_s * t) is convex in t
- Minimum over [0,1] is at t=0, t=1, or where f has a kink (subgradient contains 0)
- We can find this by: evaluate f at 0, 1, and at all pairwise intersections clipped to [0,1]

But that's O(n²) intersections. For n=100, 199×199/2 ≈ 20000 evaluations, each O(n) to evaluate the max. Total O(n³) ≈ 200×100³ = 2×10⁸. That's too slow per coordinate.

Alternative: since f is convex, we can use a ternary search or binary search on the subgradient. The subgradient of f at t is the set of D_s for which C_s + D_s*t = max. So we want to find t where 0 is in the subgradient, i.e., there exist s1, s2 with D_s1 < 0 < D_s2 (or D_s = 0) and C_s1 + D_s1*t = C_s2 + D_s2*t.

1. Sort all lines by slope D_s
2. For a given t, the active line is the one with max C_s + D_s*t
3. Binary search on t where the slope crosses 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            room = h.copy()
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / n)

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top)) * (2.0 / n) / k

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp(h, obj, args, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def corr_grad(h, s):
    n = len(h)
    grad = np.zeros(n)
    a = max(0, -s)
    b = min(n, n - s)
    if b > a:
        idx = np.arange(a, b)
        grad[idx] += (1.0 - h[idx + s])
    a2 = max(0, s)
    b2 = min(n, n + s)
    if b2 > a2:
        idx = np.arange(a2, b2)
        grad[idx] -= h[idx - s]
    return grad

def subgrad_descent(h, n_steps, n_active, alpha0, decay):
    n = len(h)
    best = h.copy()
    best_c = compute_c5(h)
    alpha = alpha0
    for _ in range(n_steps):
        corr = compute_corr(best)
        active = np.argpartition(corr, -n_active)[-n_active:]
        wts = corr[active].astype(float)
        ws = wts.sum()
        if ws < 1e-15:
            wts = np.ones(n_active) / n_active
        else:
            wts /= ws
        grad = np.zeros(n)
        for w, ki in zip(wts, active):
            s = int(ki - (n - 1))
            grad += w * corr_grad(best, s)
        h_new = project_feasible(best - alpha * grad, n)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best = c, h_new.copy()
            alpha = min(alpha * 1.005, alpha0)
        else:
            alpha *= decay
        alpha = max(alpha, 0.0001)
    return best, best_c

def coord_cd(h, n_sweeps=50, seed=0):
    """Coordinate descent: optimize each h[m] via ternary search on max of lines."""
    n = len(h)
    best = h.copy()
    best_c = compute_c5(best)
    rng = np.random.RandomState(seed)
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for m in order:
            best[m] = 0.0
            corr0 = compute_corr(best)
            best[m] = 1.0
            corr1 = compute_corr(best)
            dC = corr1 - corr0
            lo, hi = 0.0, 1.0
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(corr0 + m1 * dC)
                f2 = np.max(corr0 + m2 * dC)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            best[m] = (lo + hi) / 2.0
        best = project_feasible(best, n)
        c = compute_c5(best)
        if c < best_c:
            best_c = c
    return best, best_c

            # Downsample back
            x_dn = np.linspace(0, 2, best_n)
            hd = project_feasible(np.interp(x_dn, x_new, h_cd), best_n)
            c_dn = compute_c5(hd)
            if c_dn < best_c5:
                best_c5, best_h, best_n = c_dn, hd, best_n
            elif c_cd < best_c5:
                best_c5, best_h, best_n = c_cd, h_cd, nr

    # Phase 5: Final intensive polish
    if elapsed() < budget_s * 0.95:
        for rnd in range(5):
            if elapsed() > budget_s * 0.95:
                break
            h_cd, c_cd = coord_cd(best_h, n_sweeps=100, seed=seed + 20000 + rnd)
            if c_cd < best_c5:
                best_c5, best_h = c_cd, h_cd
            ho = slsqp(best_h, smooth_max_obj, (20000, best_n), 500, 1e-14)
            c = compute_c5(ho)
            h_cd2, c_cd2 = coord_cd(ho, n_sweeps=50, seed=seed + 30000 + rnd)
            if c_cd2 < c:
                ho, c = h_cd2, c_cd2
            if c < best_c5:
                best_c5, best_h = c, ho
            ho2 = slsqp(best_h, topk_obj, (2, best_n), 500, 1e-14)
            c2 = compute_c5(ho2)
            h_cd3, c_cd3 = coord_cd(ho2, n_sweeps=50, seed=seed + 40000 + rnd)
            if c_cd3 < c2:
                ho2, c2 = h_cd3, c_cd3
            if c2 < best_c5:
                best_c5, best_h = c2, ho2

    # Final micro subgradient + coordinate
    if elapsed() < budget_s * 0.98:
        h_sg, c_sg = subgrad_descent(best_h, 80, 3, 0.002, 0.999)
        if c_sg < best_c5:
            best_c5, best_h = c_sg, h_sg
        h_cd, c_cd = coord_cd(best_h, n_sweeps=80, seed=seed + 99999)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
