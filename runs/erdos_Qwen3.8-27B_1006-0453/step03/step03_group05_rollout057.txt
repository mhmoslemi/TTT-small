    m = np.max(corr)

    sw = exps.sum()

    one_minus_h = 1.0 - h
    for k in range(len(corr)):

        if wk < 1e-14:

        else:
            p = -m
            grad[:n-p] += wk * one_minus_h[p:]
            grad[p:] -= wk * h[:n-p]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(y, target):
    n = len(y)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def lse_grad_components(h, T):
    n = len(h)
    corr = corr_full(h)
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    sw = exps.sum()
    w = exps / sw
    val = m + np.log(sw) / T
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * omh[:n - s]
            grad[:n - s] -= wk * h[s:]
        else:
            p = -s
            grad[:n - p] += wk * omh[p:]
            grad[p:] -= wk * h[:n - p]
    return val, T * grad

def fista_opt(h0, T, n_iter, lr=None):
    n = len(h0)
    if lr is None:
        lr = 1.0 / (2.0 * T)
    x = project_box_sum(h0, n / 2.0)
    y = x.copy()
    t = 1.0
    best = x.copy()
    best_c5 = compute_c5(x)
    for k in range(n_iter):
        _, g = lse_grad_components(y, T)
        z = y - lr * g
        x_new = project_box_sum(z, n / 2.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        y = project_box_sum(y, n / 2.0)
        x = x_new
        t = t_new
        if k % 200 == 0:
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best = x.copy()
    c5 = compute_c5(x)
    if c5 < best_c5:
        best_c5, best = c5, x.copy()
    return best, best_c5

def lbfgs_opt(h0, T, lam, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def obj_(x):
        val, g = lse_grad_components(x, T)
        pen = lam * (x.sum() - n / 2.0) ** 2
        return val + pen
    def grad_(x):
        _, g = lse_grad_components(x, T)
        g = g.copy()
        g += 2.0 * lam * (x.sum() - n / 2.0)
        return g
    res = minimize(obj_, h0, jac=grad_, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return project_box_sum(np.clip(res.x, 0, 1), n)

def topk_refine(h0, K, n_inner, lr, t0, budget_s):
    n = len(h0)
    h = h0.copy()
    for _ in range(n_inner):
        if time.time() - t0 > budget_s:
            break
        corr = corr_full(h)
        top_idx = np.argpartition(corr, -K)[-K:]
        s_vals = top_idx - (n - 1)
        grad = np.zeros(n)
        omh = 1.0 - h
        for s in s_vals:
            if s >= 0:
                grad[s:] += omh[:n - s]
                grad[:n - s] -= h[s:]
            else:
                p = -s
                grad[:n - p] += omh[p:]
                grad[p:] -= h[:n - p]
        h = project_box_sum(h - lr * grad, n / 2.0)
    best = h.copy()
    best_c5 = compute_c5(h)
    return best, best_c5

def basin_hopping(h0, t0, budget_s, n_rounds, seed_rng):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for r in range(n_rounds):
        if time.time() - t0 > budget_s:
            break
        rtype = seed_rng.random()
        if rtype < 0.25:
            hp = best_h.copy()
            i = seed_rng.randint(0, n - 1)
            j = seed_rng.randint(i + 1, n)
            hp[i], hp[j] = hp[j], hp[i]
        elif rtype < 0.45:
            hp = best_h.copy()
            length = seed_rng.randint(3, max(4, n // 5))
            start = seed_rng.randint(0, max(1, n - length))
            shift = seed_rng.randint(1, max(2, length))
            hp[start:start+length] = np.roll(hp[start:start+length], shift)
        elif rtype < 0.65:
            scale = 0.03 * (1.0 - 0.005 * r) + 0.005
            hp = project_box_sum(best_h + seed_rng.randn(n) * scale, n / 2.0)
        elif rtype < 0.80:
            corr = corr_full(best_h)
            s_star = int(np.argmax(corr) - (n - 1))
            hp = best_h.copy()
            for i in range(n):
                j = i + s_star
                if 0 <= j < n:
                    if hp[i] > 0.5 and hp[j] < 0.5:
                        d = min(hp[i] * 0.4, 1.0 - hp[j])
                        hp[i] -= d
                        hp[j] += d
                    elif hp[i] < 0.5 and hp[j] > 0.5:
                        d = min((1.0 - hp[i]) * 0.4, hp[j])
                        hp[i] += d
                        hp[j] -= d
            hp = project_box_sum(hp, n / 2.0)
        else:
            hp = project_box_sum(best_h + seed_rng.randn(n) * 0.08, n / 2.0)
        ho = lbfgs_opt(hp, 300.0, 500.0, 300)
        ho, c5_ho = fista_opt(ho, 500.0, 3000)
        c5_ho = compute_c5(ho)
        if c5_ho < best_c5:
            best_c5, best_h = c5_ho, ho
    return best_h, best_c5

def multi_res(h0, t0, budget_s):
    n0 = len(h0)
    best_h, best_c5 = h0.copy(), compute_c5(h0)
    for nr in [120, 150, 250, 300]:
        if time.time() - t0 > budget_s:
            break
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, nr)
        hup = np.interp(x_new, x_old, h0)
        hup = project_box_sum(hup, nr / 2.0)
        hup2 = project_box_sum(hup + np.random.randn(nr) * 0.015, nr / 2.0)
        hup = lbfgs_opt(hup, 200.0, 500.0, 400)
        hup2 = lbfgs_opt(hup2, 200.0, 500.0, 400)
        c1, c2 = compute_c5(hup), compute_c5(hup2)
        hi = hup if c1 <= c2 else hup2
        hi, c_hi = fista_opt(hi, 400.0, 5000)
        hi, c_hi = topk_refine(hi, 5, 500, 0.01, time.time(), budget_s)
        c_hi = compute_c5(hi)
        x_dn = np.linspace(0, 2, n0)
        hd = project_box_sum(np.interp(x_dn, x_new, hi), n0 / 2.0)
        c_dn = compute_c5(hd)
        if c_dn < best_c5:
            best_c5, best_h, n0 = c_dn, hd, n0
        elif c_hi < best_c5:
            best_c5, best_h, n0 = c_hi, hi, nr
    return best_h, best_c5, n0

        best_c5 = compute_c5(best_h)

    # Phase 7: Final polish - LSE very high T + FISTA
    if time.time() - t0 < budget_s * 0.95:
        ho = lbfgs_opt(best_h, 5000.0, 2000.0, 600)
        ho, c5o = fista_opt(ho, 10000.0, 10000, lr=1e-6)
        c5o = compute_c5(ho)
        if c5o < best_c5:
            best_c5, best_h = c5o, ho
        ho2, c5o2 = topk_refine(best_h, 3, 3000, 0.003, time.time(), budget_s * 0.97)
        c5o2 = compute_c5(ho2)
        if c5o2 < best_c5:
            best_c5, best_h = c5o2, ho2

    # Final project and evaluate
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
