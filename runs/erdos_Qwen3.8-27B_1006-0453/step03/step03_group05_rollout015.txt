def coord_descent(h, n, steps=200, t0=None, budget=None):
    """Coordinate descent targeting worst shifts"""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(steps):
        if t0 is not None and time.time() - t0 > budget:
            break
        corr = get_corr(h)
        worst_k = int(np.argmax(corr))
        s = worst_k - (n - 1)
        # Try adjusting pairs at the worst shift
        improved = False
        if s >= 0:
            pairs = range(n - s)
        else:
            s2 = -s
            pairs = range(s2, n)
        for i in pairs:
            if s >= 0:
                j = i + s
                a, b = i, j
            else:
                a = i + s
                b = i
            # Try moving mass from max(h[a],h[b]) to min
            if h[a] >= h[b]:
                dv = 0.05 * (h[a] - h[b])
            else:
                dv = 0.05 * (h[b] - h[a])
            h_try = h.copy()
            h_try[a] -= dv
            h_try[b] += dv
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-10:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
        if not improved and it > 50:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(100)
        best_h[:50] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Saddle-point min-max (45% budget)
    h_sp = saddle_point(best_h.copy(), best_n, t0, budget_s, max_outer=35)
    c5_sp = compute_c5(h_sp)
    if c5_sp < best_c5:
        best_c5, best_h = c5_sp, h_sp

    # Phase 2: Softmax SLSQP with increasing T (15%)
    for T in [200, 1000, 5000, 20000]:
        if time.time() - t0 > budget_s * 0.65:
            break
        _Tg[0] = T
        _ng[0] = best_n
        h_s = _solve_slsqp(best_h.copy(), _sobj, _sgrad, best_n, 300, 1e-13)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s

    # Phase 3: Saddle-point again at higher T (10%)
    if time.time() - t0 < budget_s * 0.75:
        h_sp2 = saddle_point(best_h.copy(), best_n, t0, budget_s, max_outer=20)
        c5_sp2 = compute_c5(h_sp2)
        if c5_sp2 < best_c5:
            best_c5, best_h = c5_sp2, h_sp2

    # Phase 4: Coordinate descent on worst shifts (10%)
    if time.time() - t0 < budget_s * 0.88:
        for _ in range(3):
            if time.time() - t0 > budget_s * 0.88:
                break
            h_cd, c5_cd = coord_descent(best_h.copy(), best_n, 300, t0, budget_s * 0.88)
            if c5_cd < best_c5:
                best_c5, best_h = c5_cd, h_cd
            # Softmax polish after CD
            _Tg[0] = 10000
            _ng[0] = best_n
            h_p = _solve_slsqp(best_h.copy(), _sobj, _sgrad, best_n, 200, 1e-14)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5, best_h = c5_p, h_p

    # Phase 5: Basin hopping with structured perturbations (8%)
    for trial in range(30):
        if time.time() - t0 > budget_s * 0.96:
            break
        r = np.random.random()
        if r < 0.4:
            # Fourier perturbation
            hf = np.fft.rfft(best_h)
            mask = np.ones(len(hf))
            mask[1:4] = 0.0  # Keep DC and low freq
            hf[1:] *= (1.0 + 0.03 * np.random.randn(len(hf) - 1))
            h_try = np.fft.irfft(hf, n=best_n)
        elif r < 0.7:
            # Local bump
            h_try = best_h.copy()
            c = np.random.randint(10, best_n - 10)
            width = np.random.randint(3, 15)
            idx = slice(max(0, c - width), min(best_n, c + width))
            amp = 0.05 * np.random.choice([-1, 1])
            h_try[idx] += amp * np.cos(np.pi * (np.arange(idx.stop - idx.start) / width - 0.5))
        else:
            # Gaussian noise
            h_try = best_h + np.random.randn(best_n) * 0.02
        h_try = project_feasible(h_try, best_n)
        _Tg[0] = 5000
        _ng[0] = best_n
        h_opt = _solve_slsqp(h_try, _sobj, _sgrad, best_n, 150, 1e-12)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    # Phase 6: Final intensive polish (4%)
    if time.time() - t0 < budget_s * 0.99:
        _Tg[0] = 50000
        _ng[0] = best_n
        h_f = _solve_slsqp(best_h.copy(), _sobj, _sgrad, best_n, 500, 1e-15)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        # Saddle final
        h_f2 = saddle_point(best_h.copy(), best_n, t0, budget_s, max_outer=10)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2
        h_f3, c5_f3 = coord_descent(best_h.copy(), best_n, 200)
        if c5_f3 < best_c5:
            best_c5, best_h = c5_f3, h_f3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
