        move_type = rng.randint(3)
        if move_type == 0:
            n_mod = rng.randint(1, 5)
            idx = rng.choice(N, size=n_mod, replace=False)
            trial[idx] += rng.randn(n_mod) * s
        elif move_type == 1:
            blk = rng.randint(2, 7)
            start = rng.randint(0, max(1, N - blk))
            trial[start:start + blk] += rng.randn(blk) * s * 1.5
        else:
            start = rng.randint(0, N - 2)
            length = rng.randint(3, max(4, N // 6))
            shift = rng.randint(1, 3) * rng.choice([-1, 1])
            end = min(start + length, N)
            seg = trial[start:end].copy()
            if shift > 0 and end - start > shift:
                trial[start:end] = np.concatenate([seg[shift:], seg[:shift]])
            elif shift < 0 and end - start > -shift:
                trial[start:end] = np.concatenate([seg[-shift:], seg[:-shift]])
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = trial.copy()
            corr_best = np.correlate(best_h, 1.0 - best_h, mode='full')
    return best_h, best_c5

def targeted_fix(h, N, target, dx, deadline, rng):
    """Focus specifically on the worst 3 shifts with small targeted moves"""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(5000):
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        worst_idx = np.argsort(corr)[-3:]
        trial = best_h.copy()
        for wi in worst_idx:
            L = wi - (N - 1)
            positions = []
            if L == 0:
                positions = np.arange(N)
            elif L > 0:
                positions = np.concatenate([np.arange(N - L), np.arange(L, N)])
            else:
                M = -L
                positions = np.concatenate([np.arange(N - M), np.arange(M, N)])
            if len(positions) == 0:
                continue
            n_adj = min(4, len(positions))
            adj_idx = rng.choice(positions, size=n_adj, replace=False)
            trial[adj_idx] += rng.randn(n_adj) * 0.008
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = trial.copy()
    return best_h, best_c5

def build_structural_init(N, x0, h0):
    """Try a structure: interpolate + add small periodic component"""
    xN = np.linspace(0.0, 2.0, N, endpoint=False)
    h_base = np.interp(xN, x0, h0)
    return h_base

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    x0 = np.linspace(0.0, 2.0, N0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / N0)

    N = N0
    dx = 2.0 / N
    target = N / 2.0
    h_work = best_h.copy()

    # Phase 1: Adam with moderate focus (20% budget)
    p1_dead = t0 + (deadline - t0) * 0.18
    h1, c1 = adam_opt(h_work, N, target, dx, p1_dead,
                      tau0=10.0, tau1=60.0, lr0=0.035, lr1=0.006,
                      K=40, iters_target=350)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1
    h_work = h1

    # Phase 2: Nesterov with higher focus (18% budget)
    p2_dead = t0 + (deadline - t0) * 0.36
    h2, c2 = nesterov_opt(h_work, N, target, dx, p2_dead,
                          tau0=60.0, tau1=300.0, lr0=0.012, lr1=0.002,
                          K=20, iters_target=350)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2
    h_work = h2

    # Phase 3: Local search - block moves (15% budget)
    p3_dead = t0 + (deadline - t0) * 0.51
    h3, c3 = local_search(h_work, N, target, dx, 4000, p3_dead, rng, scale=0.035)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3
    h_work = h3

    # Phase 4: Nesterov polish - very high tau (12% budget)
    p4_dead = t0 + (deadline - t0) * 0.63
    h4, c4 = nesterov_opt(h_work, N, target, dx, p4_dead,
                          tau0=300.0, tau1=1500.0, lr0=0.004, lr1=0.0003,
                          K=10, iters_target=250)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4
    h_work = h4

    # Phase 5: Targeted fix on worst shifts (12% budget)
    p5_dead = t0 + (deadline - t0) * 0.75
    h5, c5v = targeted_fix(h_work, N, target, dx, p5_dead, rng)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h5
    h_work = h5

    # Phase 6: Adam ultra-high tau (10% budget)
    p6_dead = t0 + (deadline - t0) * 0.85
    h6, c6 = adam_opt(h_work, N, target, dx, p6_dead,
                      tau0=1500.0, tau1=8000.0, lr0=0.001, lr1=0.00005,
                      K=5, iters_target=200)
    if c6 < best_c5:
        best_c5 = c6
        best_h = h6
    h_work = h6

    # Phase 7: Fine local search (10% budget)
    p7_dead = deadline - 8.0
    h7, c7 = local_search(h_work, N, target, dx, 6000, p7_dead, rng, scale=0.015)
    if c7 < best_c5:
        best_c5 = c7
        best_h = h7
    h_work = h7

    # Phase 8: Final Nesterov micro-polish
    p8_dead = deadline - 2.0
    h8, c8 = nesterov_opt(h_work, N, target, dx, p8_dead,
                          tau0=5000.0, tau1=30000.0, lr0=0.0003, lr1=0.00001,
                          K=3, iters_target=100)
    if c8 < best_c5:
        best_c5 = c8
        best_h = h8

    # Try a random restart with noise if time allows
    if time.time() < deadline - 3.0:
        hr = best_h + rng.randn(N) * 0.01
        hr = project_sum(hr, target)
        hr, cr = nesterov_opt(hr, N, target, dx, deadline - 1.0,
                              tau0=500.0, tau1=3000.0, lr0=0.002, lr1=0.0001,
                              K=8, iters_target=80)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
