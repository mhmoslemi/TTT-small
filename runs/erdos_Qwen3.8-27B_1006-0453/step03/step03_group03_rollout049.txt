import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        else:
            room = h.copy()
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, T, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    lse = m + np.log(np.sum(np.exp((corr - m) * T))) / T
    pen = lam * (h.sum() - n / 2.0) ** 2
    return lse + pen

def solve_lse(h0, T, lam, maxiter=600):
    n = len(h0)
    try:
        res = minimize(lse_obj, h0, args=(T, lam), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0

def solve_slsq_lse(h0, T, maxiter=600):
    n = len(h0)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return m + np.log(np.sum(np.exp((corr - m) * T))) / T
    def eobj(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eobj}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0

def bb_subgrad(h0, n_iters=2000, top_m=5):
    n = len(h0)
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.05
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        m_idx = np.argsort(corr)[-top_m:]
        g = np.zeros(n)
        for idx in m_idx:
            k = idx - (n - 1)
            if k >= 0:
                g[:n - k] += (1.0 - h_cur[k:])
                g[k:] += -h_cur[:n - k]
            else:
                kp = -k
                g[kp:] += (1.0 - h_cur[:n - kp])
                g[:n - kp] += -h_cur[kp:]
        g /= len(m_idx)
        h_new = project_h(h_cur - lr * g, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_new2 = project_h(h_cur - lr * 0.5 * g, n)
            if compute_c5(h_new2)[0] < c5_new:
                lr *= 1.2
            else:
                lr *= 1.05
            h_cur = h_new
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
    return best_h, best_c5

def binary_flip_search(h0, max_passes=150):
    n = len(h0)
    h_cur = h0.copy()
    best_c5, _ = compute_c5(h_cur)
    for _ in range(max_passes):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for tgt in (1.0, 0.0):
                if abs(h_cur[i] - tgt) < 0.01:
                    continue
                h_try = h_cur.copy()
                h_try[i] = tgt
                h_try = project_h(h_try, n)
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h_cur = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h_cur, best_c5

def pairwise_flip(h0, max_passes=200):
    n = len(h0)
    h_cur = h0.copy()
    best_c5, _ = compute_c5(h_cur)
    ones = np.where(h_cur > 0.5)[0]
    zeros = np.where(h_cur < 0.5)[0]
    for _ in range(max_passes):
        improved = False
        for i in np.random.choice(ones, size=min(20, len(ones)), replace=False):
            for j in np.random.choice(zeros, size=min(20, len(zeros)), replace=False):
                h_try = h_cur.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h_cur = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur < 0.5)[0]
        if not improved:
            break
    return h_cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_h(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP with escalating log-sum-exp
    for T in [300, 2000, 10000, 40000, 150000]:
        if elapsed() > budget_s * 0.12:
            break
        h_try = solve_slsq_lse(best_h, T, maxiter=800)
        h_try = project_h(h_try, best_n)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: L-BFGS-B with penalty (different trajectory)
    for T, lam in [(5000, 200), (30000, 800)]:
        if elapsed() > budget_s * 0.22:
            break
        h_try = solve_lse(best_h, T, lam, maxiter=700)
        h_try = project_h(h_try, best_n)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: BB subgradient on true objective
    if elapsed() < budget_s * 0.38:

        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    # Phase 4: SLSQP again (refined)
    if elapsed() < budget_s * 0.48:
        h_try = solve_slsq_lse(best_h, 200000, maxiter=1000)
        h_try = project_h(h_try, best_n)

        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_base = np.interp(x_new, x_old, best_h)
        h_bin = np.where(h_base >= 0.5, 1.0, 0.0)

        h_opt = solve_slsq_lse(h_bin, 20000, maxiter=600)
        h_opt = project_h(h_opt, n_try)
