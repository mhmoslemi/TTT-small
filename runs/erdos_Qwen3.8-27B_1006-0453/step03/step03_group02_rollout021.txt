```python
            # Minimize max_k (A_k * t + B_k) over t in [0,1]
            # Ternary search (convex function)
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                m1 = lo_t + (hi_t - lo_t) / 3
                m2 = hi_t - (hi_t - lo_t) / 3
                f1 = np.max(A * m1 + B)
                f2 = np.max(A * m2 + B)
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1
            t_star = (lo_t + hi_t) / 2
            
            # Check if change is beneficial
            if abs(t_star - h[i]) > 1e-10:
                h_new = h.copy()
                h_new[i] = t_star
                h_new = project_sum(h_new, target)
                c_new = compute_c5(h_new, 2.0/N)
                c_old = compute_c5(h, 2.0/N)
                if c_new < c_old:
                    h = h_new
                    improved = True
        
        if not improved and p > 10:
            break
    
    return h, compute_c5(h, 2.0/N)
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_gradient(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    nw = len(w)
    for ix in range(nw):
        wi = w[ix]
        if wi < 1e-9:
            continue
        L = ix - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def nesterov_optimize(h, N, target, dx, deadline, tau_start, tau_end, lr_start, lr_end):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    beta = 0.85
    tau = tau_start
    lr = lr_start
    it = 0
    total_est = max(int((deadline - time.time()) / 0.005), 1)
    while time.time() < deadline:
        frac = min(it / total_est, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** (0.7 * frac)
        grad, c5_val = lse_gradient(h, N, tau)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        v = beta * v + grad
        h_new = h - lr * v
        h_new = project_sum(h_new, target)
        h = h_new
        it += 1
    return best_h, best_c5

def coord_descent(h, N, target, dx, max_passes, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    ncorr = 2 * N - 1
    for p in range(max_passes):
        if time.time() > deadline:
            break
        for i in range(N):
            if time.time() > deadline:
                break
            A = np.zeros(ncorr)
            k_lo = max(0, i - N + 1)
            k_hi = min(ncorr - 1, i)
            if k_lo <= k_hi:
                idx = np.arange(k_lo, k_hi + 1)
                A[idx] += (1.0 - h[i - idx])
            k_lo2 = max(0, -i)
            k_hi2 = min(ncorr - 1, N - 1 - i)
            if k_lo2 <= k_hi2:
                idx2 = np.arange(k_lo2, k_hi2 + 1)
                A[idx2] -= h[i + idx2]
            corr = np.correlate(h, 1.0 - h, mode='full')
            B = corr - A * h[i]
            lo_t, hi_t = 0.0, 1.0
            for _ in range(35):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = np.max(A * m1 + B)
                f2 = np.max(A * m2 + B)
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1
            t_star = (lo_t + hi_t) / 2.0
            if abs(t_star - h[i]) > 1e-10:
                h[i] = t_star
                h = project_sum(h, target)
        c5_cur = compute_c5(h, dx)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def local_search(h, N, target, dx, deadline, scale=0.04):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(777)
    while time.time() < deadline:
        trial = best_h.copy()
        n_mod = rng.randint(1, 5)
        idx = rng.choice(N, size=n_mod, replace=False)
        trial[idx] += rng.randn(n_mod) * scale
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = trial.copy()
        scale = max(0.003, scale * 0.998)
    return best_h, best_c5

def single_element_opt(h, N, target, dx, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    ncorr = 2 * N - 1
    order = list(range(N))
    np.random.RandomState(99).shuffle(order)
    while time.time() < deadline:
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            A = np.zeros(ncorr)
            k_lo = max(0, i - N + 1)
            k_hi = min(ncorr - 1, i)
            if k_lo <= k_hi:
                idx = np.arange(k_lo, k_hi + 1)
                A[idx] += (1.0 - h[i - idx])
            k_lo2 = max(0, -i)
            k_hi2 = min(ncorr - 1, N - 1 - i)
            if k_lo2 <= k_hi2:
                idx2 = np.arange(k_lo2, k_hi2 + 1)
                A[idx2] -= h[i + idx2]
            corr = np.correlate(h, 1.0 - h, mode='full')
            B = corr - A * h[i]
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                if np.max(A * m1 + B) < np.max(A * m2 + B):
                    hi_t = m2
                else:
                    lo_t = m1
            t_star = (lo_t + hi_t) / 2.0
            if abs(t_star - h[i]) > 1e-9:
                h_old = h.copy()
                h[i] = t_star
                h = project_sum(h, target)
                c_new = compute_c5(h, dx)
                if c_new < best_c5 - 1e-14:
                    best_c5 = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h = h_old
        if not improved:
            break
    return best_h, best_c5
```
