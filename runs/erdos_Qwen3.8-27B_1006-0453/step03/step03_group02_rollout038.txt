    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_gradient(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr

        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

    v = np.zeros(N)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1

        vh = v / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)

        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])

    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    top_idx = np.argsort(corr)[-K:]
    c_top = corr[top_idx]
    w = c_top / c_top.sum()
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-7:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

    v = np.zeros(N)

        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if use_subgrad:
            grad, c5_val = subgrad_max(h, N, K)
        else:
            grad, c5_val = topk_gradient(h, N, tau, K)
        if c5_val < best_c5 - 1e-13:

            best_h = h.copy()

        vh = v / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_sum(h_new, target)
        h = h_new

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline and it < max_it:
        f = min(it / 200.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        g, cv = weighted_grad(h, N, K, tau)
        if cv < best_c - 1e-13:
            best_c = cv
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        it += 1
        mhat = m / (1.0 - b1 ** it)
        vhat = v / (1.0 - b2 ** it)
        h = project_sum(h - lr * mhat / (np.sqrt(vhat) + eps), target)
    return best_h, best_c

def cluster_search(h, N, target, dx, dl, nt, sc):
    bh = h.copy()
    bc = compute_c5(h, dx)
    r = np.random.RandomState(42)
    for _ in range(nt):
        if time.time() > dl:
            break
        t = bh.copy()
        s = r.randint(0, N)
        l = r.randint(1, 6)
        e = min(s + l, N)
        t[s:e] += r.randn(e - s) * sc
        t = project_sum(t, target)
        c = compute_c5(t, dx)
        if c < bc - 1e-13:
            bc = c
            bh = t
    return bh, bc

def coord_desc(h, N, target, dx, dl, nr):
    bh = h.copy()
    bc = compute_c5(h, dx)
    dls = [0.003, 0.007, 0.015, 0.04]
    for _ in range(nr):
        if time.time() > dl:
            break
        imp = False
        for i in range(N):
            for d in dls:
                for sg in (1.0, -1.0):
                    t = bh.copy()
                    t[i] += sg * d
                    t = project_sum(t, target)
                    c = compute_c5(t, dx)
                    if c < bc - 1e-13:
                        bc = c
                        bh = t
                        imp = True
        if not imp:
            break
    return bh, bc

def pal_drop(h):
    return (h + h[::-1]) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    for N in [150, 200]:
        rem = dl - time.time()
        if rem < 90:
            break
        dx = 2.0 / N
        tgt = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        hi = project_sum(np.interp(xN, x0, h0), tgt)
        ch = hi.copy()
        cc = compute_c5(ch, dx)

        # Phase A: Wide softmax, broad exploration
        ch, cc = adam_opt(ch, N, tgt, dx, time.time() + rem * 0.20,
                          12.0, 50.0, 0.05, 0.012, min(50, 2 * N - 1), 10**6)
        # Phase B: Narrower focus
        ch, cc = adam_opt(ch, N, tgt, dx, time.time() + (dl - time.time()) * 0.22,
                          50.0, 250.0, 0.012, 0.002, min(15, 2 * N - 1), 10**6)
        # Phase C: Very focused (near-subgradient on top-5)
        ch, cc = adam_opt(ch, N, tgt, dx, time.time() + (dl - time.time()) * 0.20,
                          250.0, 2000.0, 0.004, 0.0004, min(5, 2 * N - 1), 10**6)
        # Fourier smooth to remove high-freq noise, then re-optimize
        nk = max(6, N // 20)
        H = np.fft.rfft(ch)
        H[nk:] = 0.0
        hs = project_sum(np.fft.irfft(H, N), tgt)
        cs = compute_c5(hs, dx)
        if cs < cc:
            ch, cc = hs, cs
        ch, cc = adam_opt(ch, N, tgt, dx, time.time() + (dl - time.time()) * 0.12,
                          100.0, 800.0, 0.003, 0.0003, min(8, 2 * N - 1), 10**6)
        # Try palindromic projection
        hp = project_sum(pal_drop(ch), tgt)
        cp = compute_c5(hp, dx)
        if cp < cc:
            ch, cc = hp, cp
        # Random restart perturbation + short re-opt
        if time.time() < dl - 60:
            cr = ch + rng.randn(N) * 0.04
            cr = project_sum(cr, tgt)
            cr, cr_c = adam_opt(cr, N, tgt, dx, time.time() + (dl - time.time()) * 0.08,
                                20.0, 200.0, 0.01, 0.001, min(10, 2 * N - 1), 10**6)
            if cr_c < cc:
                ch, cc = cr, cr_c
        # Cluster local search
        if time.time() < dl - 15:
            ch, cc = cluster_search(ch, N, tgt, dx, dl - 8, 10000, 0.012)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch.copy()

    # Coordinate descent fine-tuning
    Nb = len(best_h)
    if time.time() < dl - 10:
        best_h, best_c5 = coord_desc(best_h, Nb, Nb // 2, 2.0 / Nb, dl - 5, 60)
    # Final ultra-fine Adam pass
    if time.time() < dl - 4:
        hf, cf = adam_opt(best_h.copy(), Nb, Nb // 2, 2.0 / Nb, dl - 1,
                          1000.0, 8000.0, 0.002, 0.0002, min(4, 2 * Nb - 1), 10**6)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf
    fc = compute_c5(best_h, 2.0 / len(best_h))
    if fc < best_c5:
        best_c5 = fc
    return best_h, best_c5, len(best_h)
