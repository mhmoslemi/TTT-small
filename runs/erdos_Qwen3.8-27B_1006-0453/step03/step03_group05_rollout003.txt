```python
def project_box_sum(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()

            if tot < 1e-14:
```

c_s = sum_{n=lo}^{hi-1} h[n+s] * (1-h[n])

- h[i] appears in (1-h[n]) when n = i. This requires lo <= i <= hi-1.
  The derivative contribution is -h[n+s] = -h[i+s].

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            mask = h > 0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += h[mask] * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def analytic_grad_weighted(h, shifts, weights):
    n = len(h)
    grad = np.zeros(n)
    for s, w in zip(shifts, weights):
        if abs(w) < 1e-15:
            continue
        lo = max(0, -s)
        hi = min(n, n - s)
        r1_lo = max(0, lo + s)
        r1_hi = min(n, hi + s)
        if r1_lo < r1_hi:
            idx = np.arange(r1_lo, r1_hi)
            grad[idx] += w * (1.0 - h[idx - s])
        r2_lo = max(0, lo)
        r2_hi = min(n, hi)
        if r2_lo < r2_hi:
            idx = np.arange(r2_lo, r2_hi)
            grad[idx] -= w * h[idx + s]
    return grad

def weighted_obj(h, shifts, weights):
    n = len(h)
    c = get_corr(h)
    val = 0.0
    for s, w in zip(shifts, weights):
        idx = s + (n - 1)
        if 0 <= idx < len(c):
            val += w * c[idx]
    return val

def coordinated_slq(h, n, max_outer=8, maxiter=400, topk=6):
    """Iteratively reweighted SLSQP with analytic gradients."""
    best = h.copy()
    best_c5 = compute_c5(h)
    shifts = np.arange(-10, 11)
    weights = np.ones(len(shifts)) / len(shifts)
    for it in range(max_outer):
        corr = get_corr(h)
        topk_idx = np.argpartition(corr, -topk)[-topk:]
        topk_shifts = topk_idx - (n - 1)
        topk_corr = corr[topk_idx]
        w_max = np.max(topk_corr)
        w_min = np.min(topk_corr)
        rng = max(w_max - w_min, 1e-10)
        new_w = np.exp(3.0 * (topk_corr - w_max) / rng)
        new_w /= new_w.sum()
        
        obj_func = lambda x: weighted_obj(x, topk_shifts, new_w)
        grad_func = lambda x: analytic_grad_weighted(x, topk_shifts, new_w)
        
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            h_new = h
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
        elif it > 2:
            h = project_feasible(0.7 * h + 0.3 * h_new, n)
    return best, best_c5

def coord_descent(h, n, n_sweeps=5, step=0.05):
    """Coordinate descent: try adjusting each h[i] by +-step."""
    best = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            for delta in [step, -step, step * 0.5, -step * 0.5]:
                h_try = best.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                # Restore sum constraint by adjusting a random other element
                diff = n / 2.0 - h_try.sum()
                if abs(diff) > 1e-14:
                    others = np.arange(n) != i
                    room = np.where(h_try[others] < 1.0, 1.0 - h_try[others], 0.0)
                    low = np.where(h_try[others] > 0.0, h_try[others], 0.0)
                    if diff > 0:
                        tt = room.sum()
                    else:
                        tt = low.sum()
                    if tt > 1e-14:
                        if diff > 0:
                            h_try[others] += room * (diff / tt)
                        else:
                            h_try[others] += low * (diff / tt)
                        h_try = np.clip(h_try, 0, 1)
                    else:
                        continue
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved and sweep > 1:
            break
        step *= 0.7
    return best, best_c5

def initial_construction(n, seed=0):
    """Create diverse starting points."""
    rng = np.random.RandomState(seed)
    if seed == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif seed == 1:
        h = np.sin(np.linspace(0, 2, n) * np.pi) ** 2
        h = project_feasible(h, n)
    elif seed == 2:
        h = np.zeros(n)
        for i in range(0, n, 4):
            h[i:i + 2] = 1.0
        h = project_feasible(h, n)
    elif seed == 3:
        h = rng.chisquare(2, n)
        h = h / h.sum() * (n / 2.0)
        h = project_feasible(h, n)
    else:
        h = rng.beta(2, 2, n)
        h = h / h.sum() * (n / 2.0)
        h = project_feasible(h, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = initial_construction(200, 0)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Iteratively reweighted SLSQP on parent
    if elapsed() < budget_s * 0.30:
        h1, c5_1 = coordinated_slq(best_h, best_n, max_outer=10, maxiter=350, topk=8)
        if c5_1 < best_c5:
            best_c5, best_h, best_n = c5_1, h1, best_n

    # Phase 2: Different starting points at smaller n
    if elapsed() < budget_s * 0.50:
        for n_alt in [100, 150]:
            if elapsed() > budget_s * 0.50:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_alt)
            h_start = np.interp(x_new, x_old, best_h)
            h_start = project_feasible(h_start, n_alt)
            ha, ca = coordinated_slq(h_start, n_alt, max_outer=8, maxiter=300, topk=5)
            if ca < best_c5:
                best_c5, best_h, best_n = ca, ha, n_alt

    # Phase 3: Fresh constructions
    if elapsed() < budget_s * 0.62:
        for s in [1, 2, 3, 4]:
            if elapsed() > budget_s * 0.62:
                break
            hw = initial_construction(best_n, s)
            hw = project_feasible(hw + np.random.randn(best_n) * 0.05, best_n)
            hw = project_feasible(hw, best_n)
            ha, ca = coordinated_slq(hw, best_n, max_outer=6, maxiter=250, topk=6)
            ca2 = compute_c5(ha)
            if ca2 < best_c5:
                best_c5, best_h = ca2, ha

    # Phase 4: Higher resolution
    if elapsed() < budget_s * 0.78:
        nr = best_n + 50
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)

        hu = project_feasible(hu + np.random.randn(nr) * 0.015, nr)
```
