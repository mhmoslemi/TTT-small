def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    """L-BFGS-B optimization of smooth_c5 with equality constraint handled by projection."""
    def obj(x):

        target = n / 2.0
        penalty = 1e4 * (x.sum() - target) ** 2
        return smooth_c5(x, T) + penalty

                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return project(np.clip(res.x, 0, 1), n)

Actually, `minimize_scalar` with bounded method does a ternary/golden section search, which takes about 50-100 function evaluations. Each evaluation is O(n) (just computing max of a vector). So it's about 50*160 = 8000 operations per coordinate. For 160 coordinates: 1.28M per sweep. At Python speed, maybe 0.5-1 second per sweep. We can afford 100-200 sweeps.

Let me just go with `minimize_scalar` for simplicity and focus on making the overall algorithm more effective.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def l_bfgs_smooth(h, T, n, maxiter=400):
    target = n / 2.0
    def obj(x):
        return smooth_c5(x, T) + 5e4 * (x.sum() - target) ** 2
    res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
    return project(np.clip(res.x, 0, 1), n)

def coord_opt_1d(corr_base, coef):
    res = minimize_scalar(lambda t: np.max(corr_base + t * coef),
                          bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = project(h.copy(), n)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            old_val = best_h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                j1 = i - k
                j2 = i + k
                c = 0.0
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c += (-best_h[j2])
                coef[k + n - 1] = c
            corr_base = corr - old_val * coef
            new_val = coord_opt_1d(corr_base, coef)
            if abs(new_val - old_val) > 1e-10:
                delta = new_val - old_val
                corr += delta * coef
                best_h[i] = new_val
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    improved = True
        best_h = project(best_h, n)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
        if not improved:
            break
    return best_h, best_c5

def basin_step(work_h, work_c5, n, cool, t0, budget_s):
    step_size = 0.02 + 0.10 * cool
    # Targeted perturbation: boost directions that increase min autocorrelation
    corr = np.correlate(work_h, 1.0 - work_h, mode='full')
    m = np.max(corr)
    active = np.where(corr > m - 0.05 * m)[0]
    pert = step_size * np.random.randn(n)
    # Add a small targeted push
    idx_shift = np.random.randint(n - 1, size=min(3, n - 1))
    for sh in idx_shift:
        pert[:n - sh] += 0.01 * np.random.choice([-1, 1])
        pert[n - sh:] -= 0.01 * np.random.choice([-1, 1])
    h_pert = work_h + pert
    h_pert = project(h_pert, n)
    T_bh = 1500
    hc = l_bfgs_smooth(h_pert, T_bh, n, maxiter=200)
    c5v = compute_c5(hc)
    temp = 0.003 * cool + 0.0005
    accept = c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp)
    if accept:
        return hc, c5v
    return work_h, work_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 120
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B T-schedule
    for T in [15, 50, 200, 800, 3000, 10000]:
        if elapsed() > budget_s * 0.18:
            break
        hc = l_bfgs_smooth(best_h.copy(), T, n, maxiter=350)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent
    t_cd1 = t0 + budget_s * 0.42
    best_h, best_c5 = coord_descent(best_h, n, t_cd1, max_sweeps=400)

    # Phase 3: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(80):
        if elapsed() > budget_s * 0.62:
            break
        cool = 1.0 - 0.92 * (it / 80.0)
        work_h, work_c5 = basin_step(work_h, work_c5, n, cool, t0, budget_s)
        if work_c5 < best_c5:
            best_c5 = work_c5
            best_h = work_h.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Coordinate descent again
    t_cd2 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descent(best_h, n, t_cd2, max_sweeps=400)

    # Phase 5: Multi-resolution
    for n_try in [100, 120, 144, 160, 200]:
        if elapsed() > budget_s * 0.90:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project(h_int, n_try)
        for T in [500, 3000]:
            if elapsed() > budget_s * 0.91:
                break
            hc = l_bfgs_smooth(h_int, T, n_try, maxiter=200)
            hc, cv = coord_descent(hc, n_try, t0 + budget_s * 0.92, max_sweeps=60)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc

    # Phase 6: Final polish
    n_final = len(best_h)
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n_final, t0 + budget_s * 0.96, max_sweeps=200)
    if elapsed() < budget_s * 0.98:
        hc = l_bfgs_smooth(best_h, 20000, n_final, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
