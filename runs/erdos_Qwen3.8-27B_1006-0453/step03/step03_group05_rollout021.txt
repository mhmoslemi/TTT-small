    h = np.clip(h, 0.0, 1.0).copy()

    """Differentiable approximation of max(corr) * dx"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    # Log-sum-exp for numerical stability
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    top = np.partition(corr, -k)[-k:]

        grad = np.zeros(n)

                g = 0.0
