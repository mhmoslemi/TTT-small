    target = n / 2.0
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14: break
            h += r * (d / t)
        else:
            t = h.sum()
            if t < 1e-14: break
            h += h * (d / t)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_obj(x, K, n):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(np.sort(corr)[::-1][:K]) * (2.0/n) / K)

def topk_grad(x, K, n):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -K)[-K:]
    grad = np.zeros(n)
    for ki in idx:
        s = int(ki - (n - 1))
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            s2 = -s
            grad[s2:] += (1.0 - h[:n-s2])
            grad[:s2] -= h[s2:]
    return grad * (2.0 / n) / K

def smax_obj(x, T, n):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0/n)

def smax_grad(x, T, n):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wt = w[k]
        if wt < 1e-10: continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += wt * (1.0 - h[s:])
            grad[s:] -= wt * h[:n-s]
        else:
            s2 = -s
            grad[s2:] += wt * (1.0 - h[:n-s2])
            grad[:s2] -= wt * h[s2:]
    return grad * (2.0 / n)

def sum_eq(x):
    return float(x.sum() - len(x)/2.0)

def optimize_lbfgsb(h0, obj, grad, args, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    def objf(x):
        return obj(np.clip(x, 0, 1), *args)
    def gradf(x):
        return grad(np.clip(x, 0, 1), *args)
    try:
        res = minimize(objf, h0, jac=gradf, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h0

def optimize_slsqp(h0, obj, args, n, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(obj, h0, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h0

def shift_repair(h, n, n_rounds=10):
    """Targeted repair: reduce overlap at worst shifts by moving mass."""
    h = h.copy()
    for _ in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top 3 shifts
        top_idx = np.argpartition(corr, -3)[-3:]
        for ki in top_idx:
            s = int(ki - (n - 1))
            if s == 0: continue
            if s >= 0:
                # Conflict: h[j] high, h[j+s] low
                for j in range(n - s):
                    if h[j] > 0.4 and h[j+s] < 0.6:
                        delta = min((h[j] - 0.5) * 0.2, 0.1)
                        if delta > 0.001:
                            h[j] -= delta
                            h[j+s] += delta
                    elif h[j] < 0.6 and h[j+s] > 0.4:
                        # This is fine (contributes to 1-h at shift)
                        pass
            else:
                s2 = -s
                for j in range(s2):
                    # h[j+s2] high, h[j] low at shift s
                    if h[j+s2] > 0.4 and h[j] < 0.6:
                        delta = min((h[j+s2] - 0.5) * 0.2, 0.1)
                        if delta > 0.001:
                            h[j+s2] -= delta
                            h[j] += delta
        h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    # Choose n/2 positions to set to 1
    positions = rng.choice(n, n//2, replace=False)
    h[positions] = 1.0
    # Randomize slightly
    h = project_feasible(h + rng.randn(n) * 0.1, n)
    return h

def momentum_grad(h, grad_func, args, n, n_steps, lr0, momentum, decay):

def smax_obj_slsqp(x, T, n):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def sum_eq(x):
    return float(x.sum() - len(x) / 2.0)

def opt_lbfgsb(h0, obj, grad, args, n, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, args=args, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 50})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def opt_slsqp(h0, obj, args, n, maxiter=400, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(obj, h0, args=args, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def momentum_grad(h0, grad_func, args, n, n_steps, lr0, momentum, decay):
    h = h0.copy()
    v = np.zeros(n)
    lr = lr0
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_steps):
        g = grad_func(h, *args)
        v = momentum * v - lr * g
        h = project_feasible(h + v, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= decay
    return best, best_c5

def shift_repair(h, n, n_rounds=8):
    h = h.copy()
    for _ in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top3 = np.argpartition(corr, -3)[-3:]
        for ki in top3:
            s = int(ki - (n - 1))
            if s == 0:
                continue
            if s >= 0:
                for j in range(n - s):
                    if h[j] > 0.45 and h[j + s] < 0.55:
                        delta = min((h[j] - 0.5) * 0.15, 0.08)
                        if delta > 0.005:
                            h[j] -= delta
                            h[j + s] += delta
            else:
                s2 = -s
                for j in range(s2):
                    if h[j + s2] > 0.45 and h[j] < 0.55:
                        delta = min((h[j + s2] - 0.5) * 0.15, 0.08)
                        if delta > 0.005:
                            h[j + s2] -= delta
                            h[j] += delta
        h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def rand_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return project_feasible(h + rng.randn(n) * 0.05, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(100)
        best_h[:50] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    lam = 500.0

    # Phase 1: L-BFGS-B with penalty from parent (different optimizer)
    if time.time() - t0 < budget_s * 0.12:
        for K in [2, 3, 5, 8]:
            if time.time() - t0 > budget_s * 0.09:
                break
            h1 = opt_lbfgsb(best_h, topk_obj_pen, topk_grad_pen, (K, best_n, lam), best_n, 350)
            c1 = compute_c5(h1)
            if c1 < best_c5:
                best_c5, best_h = c1, h1
        for T in [500, 2000, 8000]:
            if time.time() - t0 > budget_s * 0.12:
                break
            h1 = opt_lbfgsb(best_h, smax_obj_pen, smax_grad_pen, (T, best_n, lam), best_n, 350)
            c1 = compute_c5(h1)
            if c1 < best_c5:
                best_c5, best_h = c1, h1

    # Phase 2: Diverse starts at multiple resolutions
    for nr in [72, 96, 120, 150, 300]:
        if time.time() - t0 > budget_s * 0.35:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        starts = [np.interp(x_new, x_old, best_h)]
        starts.append(rand_binary_init(nr, seed + nr))
        for h_init in starts:
            if time.time() - t0 > budget_s * 0.35:
                break
            h_init = project_feasible(h_init, nr)
            ho = opt_lbfgsb(h_init, topk_obj_pen, topk_grad_pen, (3, nr, lam), nr, 250)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h, best_n = co, ho, nr
            ho2 = opt_lbfgsb(ho, smax_obj_pen, smax_grad_pen, (1000, nr, lam), nr, 200)
            co2 = compute_c5(ho2)
            if co2 < best_c5:
                best_c5, best_h, best_n = co2, ho2, nr

    # Phase 3: Momentum gradient refinement
    if time.time() - t0 < budget_s * 0.45:
        for K in [3, 5]:
            if time.time() - t0 > budget_s * 0.42:
                break
            hm, cm = momentum_grad(best_h, topk_grad_pen, (K, best_n, lam), best_n, 500, 0.02, 0.9, 0.998)
            if cm < best_c5:
                best_c5, best_h = cm, hm
        hm2, cm2 = momentum_grad(best_h, smax_grad_pen, (2000, best_n, lam), best_n, 400, 0.01, 0.85, 0.999)
        if cm2 < best_c5:
            best_c5, best_h = cm2, hm2

    # Phase 4: Basin hopping with structured moves
    elapsed = time.time() - t0
    remaining = budget_s * 0.68 - elapsed
    n_trials = max(20, int(remaining / 7.0))
    for trial in range(n_trials):
        if time.time() - t0 > budget_s * 0.68:
            break
        r = np.random.random()
        if r < 0.22:
            hp = shift_repair(best_h, best_n, 4 + trial % 8)
        elif r < 0.42:
            sc = 0.025 * (1.0 - 0.004 * trial) + 0.005
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.58:
            hp = best_h.copy()
            bl = max(3, best_n // 12)
            is_ = np.random.randint(0, max(1, best_n - 2 * bl))
            hp[is_:is_ + bl] = np.roll(hp[is_:is_ + bl], np.random.randint(1, bl))
            hp = project_feasible(hp, best_n)
        elif r < 0.72:
            hp = best_h.copy()
            frac = np.random.uniform(0.2, 0.4)
            nmod = max(3, int(n * frac)) if (n := best_n) else 3
            mask = np.zeros(best_n, dtype=bool)
            mask[np.random.choice(best_n, nmod, replace=False)] = True
            hp[~mask] = 0.5 + 0.5 * np.random.randn(best_n - mask.sum()) * 0.3
            hp = project_feasible(hp, best_n)
        else:
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            ki = int(np.argmax(corr))
            s = ki - (best_n - 1)
            hp = best_h.copy()
            if s != 0 and abs(s) < best_n // 2:
                seg = max(2, best_n // 15)
                st = np.random.randint(0, max(1, best_n - seg * 2))
                if s > 0:
                    mv = np.clip(hp[st:st+seg] * 0.25, 0, 1.0 - hp[st+s:st+s+seg])
                    hp[st:st+seg] -= mv
                    hp[st+s:st+s+seg] += mv
                else:
                    s2 = -s
                    mv = np.clip(hp[st+s2:st+s2+seg] * 0.25, 0, 1.0 - hp[st:st+seg])
                    hp[st+s2:st+s2+seg] -= mv
                    hp[st:st+seg] += mv
            hp = project_feasible(hp, best_n)

        ho = opt_lbfgsb(hp, topk_obj_pen, topk_grad_pen, (3, best_n, lam), best_n, 180)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho
        ho2 = opt_lbfgsb(best_h, smax_obj_pen, smax_grad_pen, (3000, best_n, lam), best_n, 150)
        co2 = compute_c5(ho2)
        if co2 < best_c5:
            best_c5, best_h = co2, ho2

    # Phase 5: SLSQP with equality constraint (tight)
    if time.time() - t0 < budget_s * 0.84:
        for K in [2, 3, 4, 5]:
            if time.time() - t0 > budget_s * 0.82:
                break
            hs = opt_slsqp(best_h, topk_obj_slsqp, (K, best_n), best_n, 500, 1e-15)
            cs = compute_c5(hs)
            if cs < best_c5:
                best_c5, best_h = cs, hs
        for T in [5000, 20000, 80000]:
            if time.time() - t0 > budget_s * 0.84:
                break
            hs = opt_slsqp(best_h, smax_obj_slsqp, (T, best_n), best_n, 400, 1e-15)
            cs = compute_c5(hs)
            if cs < best_c5:
                best_c5, best_h = cs, hs

    # Phase 6: Nearby resolution exploration
    for dr in [-3, -2, -1, 1, 2, 3, 5, 10]:
        if time.time() - t0 > budget_s * 0.93:
            break
        nr = best_n + dr
        if nr < 48 or nr > 400:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        ho = opt_lbfgsb(hu, topk_obj_pen, topk_grad_pen, (3, nr, lam), nr, 200)
        ho = opt_slsqp(ho, smax_obj_slsqp, (10000, nr), nr, 250, 1e-15)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h, best_n = co, ho, nr
        else:
            x_dn = np.linspace(0, 2, best_n)
            hd = np.interp(x_dn, x_new, ho)
            hd = project_feasible(hd, best_n)
            cd = compute_c5(hd)
            if cd < best_c5:
                best_c5, best_h = cd, hd

    # Phase 7: Final intense polish
    if time.time() - t0 < budget_s * 0.98:
        for _ in range(4):
            if time.time() - t0 > budget_s * 0.98:
                break
            hf = opt_lbfgsb(best_h, topk_obj_pen, topk_grad_pen, (2, best_n, lam), best_n, 400)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf
            hf2 = opt_slsqp(best_h, smax_obj_slsqp, (200000, best_n), best_n, 350, 1e-16)
            cf2 = compute_c5(hf2)
            if cf2 < best_c5:
                best_c5, best_h = cf2, hf2
            hf3 = opt_slsqp(best_h, topk_obj_slsqp, (3, best_n), best_n, 350, 1e-16)
            cf3 = compute_c5(hf3)
            if cf3 < best_c5:
                best_c5, best_h = cf3, hf3

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
