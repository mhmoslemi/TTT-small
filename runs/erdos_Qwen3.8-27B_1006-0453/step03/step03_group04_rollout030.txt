The objective is max_k sum_i h[i] * (1-h[i+k]). If we fix all but one h[j], the objective becomes a function of h[j]. We could do coordinate descent: for each j, find the optimal h[j] given all others.

For a fixed h[j], the correlation at lag L involves terms h[j]*(1-h[j+L]) and h[j-L]*(1-h[j]).
= h[j] - h[j]*h[j+L] + h[j-L] - h[j]*h[j-L]
= h[j] * (1 - h[j+L] - h[j-L]) + h[j-L]

This is linear in h[j]! So the optimal h[j] is:
- If (1 - h[j+L] - h[j-L]) > 0: h[j] = 1
- If (1 - h[j+L] - h[j-L]) < 0: h[j] = 0
- If = 0: any value

def grad_topk(h, N, k, one_minus_h):

    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    center = N - 1
    wk = 1.0 / k
    grad = np.zeros(N)

            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):

    best_peak = float(np.max(compute_corr(h)))
    one_minus_h = 1.0 - h
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k, one_minus_h)
        if peak < best_peak:

            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr0 * (decay ** (it // 150)) * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        one_minus_h = 1.0 - h
    return best_h, best_peak * (2.0 / N)

            zeros = np.where(h_b == 0)[0]
            # pick from highest h values
            h_zeros = h[zeros]
            top_zeros = zeros[np.argsort(-h_zeros)[:n_need]]
            h_b[top_zeros] = 1

    cur_pk = best_pk

    T0, T1 = 4.0, 0.002

    zeros_list = np.where(h_b == 0)[0]

        h_b[i], h_b[j] = h_b[j], h_b[i]

        delta = new_pk - cur_pk

            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
    return best, best_pk * dx

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, omh, N, k):
    corr = np.correlate(h, omh, mode='full')
    peak = float(np.max(corr))
    if k >= len(corr):
        top = np.arange(len(corr))
    else:
        top = np.argpartition(corr, -k)[-k:]
    center = N - 1
    wk = 1.0 / k
    grad = np.zeros(N)
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    omh = 1.0 - h
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, omh, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr0 * (decay ** (it // 100)) * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        omh = 1.0 - h
    return best_h, best_peak * (2.0 / N)

def anneal_binary(h, N, deadline, rng):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    n_half = N // 2
    # Ensure exactly n_half ones
    s = int(h_b.sum())
    if s != n_half:
        if s > n_half:
            # Remove excess ones (lowest h values among ones)
            one_positions = np.where(h_b == 1)[0]
            h_vals = h[one_positions]
            remove = one_positions[np.argsort(h_vals)[:s - n_half]]
            h_b[remove] = 0
        else:
            zero_positions = np.where(h_b == 0)[0]
            h_vals = h[zero_positions]
            add = zero_positions[np.argsort(-h_vals)[:n_half - s]]
            h_b[add] = 1
    
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    
    corr = compute_corr(h_b)
    best = h_b.copy()
    best_pk = float(np.max(corr))
    cur_pk = best_pk
    
    t_start = time.time()
    t_total = deadline - t_start
    T0, T1 = 6.0, 0.001
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    while time.time() < deadline:
        T = T0 * (T1 / T0) ** ((time.time() - t_start) / t_total)
        ii = rng.randint(n_ones)
        jj = rng.randint(n_zeros)
        i = ones[ii]
        j = zeros[jj]
        # Swap: i was 1→0, j was 0→1
        h_b[i] = 0.0
        h_b[j] = 1.0
        ones[ii] = j
        zeros[jj] = i
        
        corr = compute_corr(h_b)
        new_pk = float(np.max(corr))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            # Revert
            h_b[i] = 1.0
            h_b[j] = 0.0
            ones[ii] = i
            zeros[jj] = j
    
    return best, best_pk * dx

def local_search(h_b, N, iterations):
    """Try all single swaps and keep improvements."""
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    n_half = N // 2
    
    for _ in range(iterations):
        improved = False
        for ii in range(n_half):
            if improved:
                break
            i = ones[ii]
            for jj in range(n_half):
                j = zeros[jj]
                h_b[i], h_b[j] = h_b[j], h_b[i]
                corr = compute_corr(h_b)
                pk = float(np.max(corr))
                if pk < best_pk:
                    best_pk = pk
                    best = h_b.copy()
                    ones[ii] = j
                    zeros[jj] = i
                    improved = True
                else:
                    h_b[i], h_b[j] = h_b[j], h_b[i]
    return best, best_pk * (2.0 / N)

            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    omh = 1.0 - h
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, omh, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr0 * (decay ** (it // 100)) * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        omh = 1.0 - h
    return best_h, best_peak * (2.0 / N)

def anneal_binary(h, N, deadline, rng):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    n_half = N // 2
    s = int(h_b.sum())
    if s != n_half:
        if s > n_half:
            op = np.where(h_b == 1)[0]
            hv = h[op]
            h_b[op[np.argsort(hv)[:s - n_half]]] = 0.0
        else:
            zp = np.where(h_b == 0)[0]
            hv = h[zp]
            h_b[zp[np.argsort(-hv)[:n_half - s]]] = 1.0
    ones = np.where(h_b == 1)[0].copy()
    zeros = np.where(h_b == 0)[0].copy()
    corr = compute_corr(h_b)
    best = h_b.copy()
    best_pk = float(np.max(corr))
    cur_pk = best_pk
    t_start = time.time()
    t_total = max(1.0, deadline - t_start)
    T0, T1 = 6.0, 0.0005
    while time.time() < deadline:
        frac = (time.time() - t_start) / t_total
        frac = min(frac, 1.0)
        T = T0 * (T1 / T0) ** frac
        ii = rng.randint(n_half)
        jj = rng.randint(n_half)
        i = ones[ii]
        j = zeros[jj]
        h_b[i] = 0.0
        h_b[j] = 1.0
        ones[ii] = j
        zeros[jj] = i
        corr = compute_corr(h_b)
        new_pk = float(np.max(corr))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.002 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
            ones[ii] = i
            zeros[jj] = j
    return best, best_pk * dx
