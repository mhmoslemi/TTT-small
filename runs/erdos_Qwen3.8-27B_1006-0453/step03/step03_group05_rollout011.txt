    h = np.clip(h, 0.0, 1.0).copy()

    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

                       options={'maxiter': maxiter, 'ftol': ftol})

                    best_c5, best_h, best_n = c2, b2, nr
            else:
                best_c5, best_h, best_n = c2, b2, nr

    # Phase 3: SLSQP polish
    if time.time() - t0 < budget_s * 0.85:
        for k in [3, 5, 8]:
            if time.time() - t0 > budget_s * 0.85:
                break
            hp = slsqp_solve(best_h.copy(), topk_obj, (k, best_n), 400, 1e-13)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5, best_h = cp, hp

    # Phase 4: CD again on polished solution
    if time.time() - t0 < budget_s * 0.92:
        b3, c3, _ = cd_full(best_h, best_n, rng, max_sweeps=5000, patience=30)
        if c3 < best_c5:
            best_c5, best_h = c3, b3

    # Phase 5: Perturbation + CD from multiple starts
    if time.time() - t0 < budget_s * 0.96:
        for attempt in range(5):
            if time.time() - t0 > budget_s * 0.96:
                break
            scale = 0.03 * (1 - attempt * 0.15)
            hp = project_feasible(best_h + rng.randn(best_n) * scale, best_n)
            b4, c4, _ = cd_full(hp, best_n, rng, max_sweeps=2000, patience=30)
            if c4 < best_c5:
                best_c5, best_h = c4, b4

    # Phase 6: Final SLSQP
    if time.time() - t0 < budget_s * 0.98:
        hf = slsqp_solve(best_h.copy(), topk_obj, (3, best_n), 500, 1e-14)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hf2 = slsqp_solve(best_h.copy(), softmax_obj, (5000, best_n), 400, 1e-14)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
