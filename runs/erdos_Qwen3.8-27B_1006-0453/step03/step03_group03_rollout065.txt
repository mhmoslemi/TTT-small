import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_from_continuous(h, n):
    # Sort by value, set top n/2 to 1
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def hill_climb_binary(h, n, max_iters=5000):
    """Pair-flip hill climbing for binary sequences."""
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    for it in range(max_iters):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        improved = False
        for i in ones:
            for j in zeros:
                new_h = best_h.copy()
                new_h[i] = 0
                new_h[j] = 1
                c5_new, _ = compute_c5(new_h)
                if c5_new < best_c5 - 1e-14:

                    best_c5 = c5_new
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def simulated_annealing(h, n, t_start=0.01, t_end=0.0001, n_iter=20000):
    """Simulated annealing with pair flips."""
    cur_h = h.copy()
    cur_c5, _ = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    alpha = (t_end / t_start) ** (1.0 / n_iter)
    T = t_start

        T *= alpha

            cur_c5 = c5_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            room = h.copy()
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h

def to_binary(h, n):
    order = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[order[:n // 2]] = 1.0
    return h_b

def hill_climb(h, n, max_rounds=200):
    h_cur = h.copy()
    c5_cur, _ = compute_c5(h_cur)
    for _ in range(max_rounds):
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        best_c5 = c5_cur
        best_ij = None
        for i in ones:
            for j in zeros:
                nh = h_cur.copy()
                nh[i] = 0.0
                nh[j] = 1.0
                c, _ = compute_c5(nh)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_ij = (i, j)
                    h_cur = nh
        if best_ij is None:
            break
        c5_cur = best_c5
    return h_cur, c5_cur

def sa_search(h, n, rng, n_iter=10000, t_hi=0.02, t_lo=0.0005):
    cur = h.copy()
    c5_cur, _ = compute_c5(cur)
    best = cur.copy()
    c5_best = c5_cur
    alpha = (t_lo / t_hi) ** (1.0 / max(n_iter, 1))
    T = t_hi
    for _ in range(n_iter):
        T *= alpha
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        nh = cur.copy()
        nh[i] = 0.0
        nh[j] = 1.0
        c, _ = compute_c5(nh)
        d = c - c5_cur
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur = nh
            c5_cur = c
            if c < c5_best:
                c5_best = c
                best = cur.copy()
    return best, c5_best

    hk = h.copy()

        if len(ones) < 1 or len(zeros) < 1:

        j = zeros[rng.integers(len(zeros))]
        hk[i] = 0.0
        hk[j] = 1.0
    return hill_climb(hk, n, max_rounds)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        room = (1.0 - h) if diff > 0 else h.copy()
        tot = room.sum()
        if tot < 1e-14:
            break
        h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def _sum_eq(x):
    return x.sum() - len(x) / 2.0

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    try:
        res = minimize(_smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h

    h_b[order[:n // 2]] = 1.0
    return h_b
