def grad_topk(h, N, k):
    """Compute gradient for top-k peaks of correlation."""
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def binary_local_search(h_b, N, deadline, rng, max_iter_per_pass=50000):
    """Intensive local search with multiple move types."""
    dx = 2.0 / N
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur = h_b.copy()
    cur_pk = best_pk
    
    T = 5.0
    cooling = 0.9995
    T_min = 0.01
    
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    it = 0
    stuck = 0
    while time.time() < deadline:
        # Choose move type
        move_type = rng.randint(3)
        
        if move_type == 0:  # Single swap
            if len(ones) < 1 or len(zeros) < 1:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            cur[i], cur[j] = cur[j], cur[i]
            new_pk = float(np.max(compute_corr(cur)))
            delta = new_pk - cur_pk
            if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
                cur_pk = new_pk
                # Update ones/zeros
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = cur.copy()
                    stuck = 0
                stuck = 0
            else:
                cur[i], cur[j] = cur[j], cur[i]
                stuck += 1
        
        elif move_type == 1:  # Block swap (swap block around i with block around j)
            L = rng.randint(2, 6)
            i = rng.randint(0, N - L)
            j = rng.randint(0, N - L)
            if abs(i - j) < L:
                continue
            # Check both blocks have same number of 1s
            block_i = cur[i:i+L].copy()
            block_j = cur[j:j+L].copy()
            if block_i.sum() != block_j.sum():
                continue
            cur[i:i+L] = block_j
            cur[j:j+L] = block_i
            new_pk = float(np.max(compute_corr(cur)))
            delta = new_pk - cur_pk
            if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = cur.copy()
                    stuck = 0
                stuck = 0
            else:
                cur[i:i+L] = block_i
                cur[j:j+L] = block_j
                stuck += 1
        
        else:  # Multi-swap (swap 2 ones with 2 zeros)
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(ones, size=2, replace=False)
            j1, j2 = rng.choice(zeros, size=2, replace=False)
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            cur[i1], cur[j1] = cur[j1], cur[i1]
            cur[i2], cur[j2] = cur[j2], cur[i2]
            new_pk = float(np.max(compute_corr(cur)))
            delta = new_pk - cur_pk
            if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = cur.copy()
                    stuck = 0
                stuck = 0
            else:
                cur[i1], cur[j1] = cur[j1], cur[i1]
                cur[i2], cur[j2] = cur[j2], cur[i2]
                stuck += 1
        
        it += 1
        if it % 1000 == 0:
            T *= cooling
            if stuck > 5000:
                # Restart with perturbation
                for _ in range(20):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    cur[i], cur[j] = cur[j], cur[i]
                # Recompute ones/zeros
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                cur_pk = float(np.max(compute_corr(cur)))
                stuck = 0
    
    return best, best_pk * dx

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def binarize_fix(h_cont, N, rng):
    target = N // 2
    h_b = (h_cont > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        h_b[rng.choice(idx, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        h_b[rng.choice(idx, target - n_ones, replace=False)] = 1
    return h_b

def sa_binary(cur, N, t_end, rng, T0=5.0, T1=0.005):
    dx = 2.0 / N
    best = cur.copy()
    best_pk = float(np.max(compute_corr(cur)))
    cur_pk = best_pk
    t_start = time.time()
    total = max(t_end - t_start, 1.0)
    no_improve = 0

    while time.time() < t_end:
        frac = min((time.time() - t_start) / total, 1.0)
        T = T0 * (T1 / T0) ** frac
        r = rng.random()

        if r < 0.65:
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            cur[i] = 0; cur[j] = 1
            new_pk = float(np.max(compute_corr(cur)))
            d = new_pk - cur_pk
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                cur_pk = new_pk; no_improve = 0
                if new_pk < best_pk:
                    best_pk = new_pk; best = cur.copy()
            else:
                cur[i] = 1; cur[j] = 0
                no_improve += 1
        elif r < 0.85:
            L = rng.randint(2, 5)
            i = rng.randint(0, N - L)
            j = rng.randint(0, N - L)
            if abs(i - j) < L:
                continue
            bi = cur[i:i+L].copy(); bj = cur[j:j+L].copy()
            if bi.sum() != bj.sum():
                continue
            cur[i:i+L] = bj; cur[j:j+L] = bi
            new_pk = float(np.max(compute_corr(cur)))
            d = new_pk - cur_pk
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                cur_pk = new_pk; no_improve = 0
                if new_pk < best_pk:
                    best_pk = new_pk; best = cur.copy()
            else:
                cur[i:i+L] = bi; cur[j:j+L] = bj
                no_improve += 1
        else:
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            cur[i1], cur[j1] = cur[j1], cur[i1]
            cur[i2], cur[j2] = cur[j2], cur[i2]
            new_pk = float(np.max(compute_corr(cur)))
            d = new_pk - cur_pk
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                cur_pk = new_pk; no_improve = 0
                if new_pk < best_pk:
                    best_pk = new_pk; best = cur.copy()
            else:
                cur[i1], cur[j1] = cur[j1], cur[i1]
                cur[i2], cur[j2] = cur[j2], cur[i2]
                no_improve += 1

        if no_improve > 5000:
            for _ in range(rng.randint(3, 10)):
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                cur[ii], cur[jj] = 0, 1
            cur_pk = float(np.max(compute_corr(cur)))
            no_improve = 0

    return best, best_pk * dx

def targeted_polish(cur, N, t_end, rng):
    dx = 2.0 / N
    best = cur.copy()
    best_pk = float(np.max(compute_corr(cur)))
    cur_pk = best_pk

    while time.time() < t_end:
        corr = compute_corr(cur)
        peak = float(np.max(corr))
        if peak < best_pk:
            best_pk = peak; best = cur.copy()

        peak_idx = np.where(corr >= peak - 0.01)[0]
        improved = False

        for pi in peak_idx[:4]:
            L = pi - (N - 1)
            if L == 0:
                contrib = np.where(cur == 1)[0]
            elif L > 0 and L < N:
                contrib = np.where((cur[:N-L] == 1) & (cur[L:N] == 0))[0]
            elif L < 0 and -L < N:
                M = -L
                contrib = np.where((cur[M:N] == 1) & (cur[:N-M] == 0))[0]
            else:
                continue
            if len(contrib) == 0:
                continue
            zeros = np.where(cur == 0)[0]
            for _ in range(30):
                i = contrib[rng.randint(len(contrib))]
                j = zeros[rng.randint(len(zeros))]
                if i == j:
                    continue
                cur[i] = 0; cur[j] = 1
                new_pk = float(np.max(compute_corr(cur)))
                if new_pk < cur_pk:
                    cur_pk = new_pk; improved = True
                    break
                cur[i] = 1; cur[j] = 0
            if improved:
                break

        if not improved:
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            n_pert = rng.randint(2, 6)
            for _ in range(n_pert):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                cur[ii], cur[jj] = 0, 1
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
            cur_pk = float(np.max(compute_corr(cur)))

    return best, best_pk * dx

    w = [2.5, 2.0, 2.0, 1.8, 1.5, 1.5]

    total_w = sum(w)
    total_time = deadline - time.time() - 25

    for idx, N in enumerate(N_list):

            break
        alloc = total_time * w[idx] / total_w
        if alloc < 20:
            alloc = 20
        if time.time() + alloc > deadline - 25:
            alloc = deadline - 25 - time.time()
        if alloc < 15:
            break

        x_src = np.linspace(0.0, 2.0, n0, endpoint=False)
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        target = N // 2

        h_interp = np.interp(xN, x_src, h0)
        h_b = binarize_fix(h_interp, N, rng)

        h_rand = np.zeros(N)
        h_rand[rng.choice(N, target, replace=False)] = 1.0

        pk1 = float(np.max(compute_corr(h_b)))
        pk2 = float(np.max(compute_corr(h_rand)))
        h_start = h_b if pk1 <= pk2 else h_rand

        c5_s = compute_c5(h_start, N)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()

        t_sa_end = time.time() + alloc * 0.55
        h_sa, c5_sa = sa_binary(h_start, N, t_sa_end, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        t_tp_end = time.time() + alloc * 0.35
        if t_tp_end < deadline - 15:
            h_tp, c5_tp = targeted_polish(h_sa, N, t_tp_end, rng)
            if c5_tp < best_c5:
                best_c5 = c5_tp
                best_h = h_tp.copy()

        t_gl_end = min(time.time() + alloc * 0.15, deadline - 15)
        if t_gl_end > time.time() + 3:
            h_gl, c5_gl = targeted_polish(best_h.copy() if len(best_h) == N else h_tp, N, t_gl_end, rng)
            if c5_gl < best_c5:
                best_c5 = c5_gl
                best_h = h_gl.copy()

    if time.time() < deadline - 4:
        Nb = len(best_h)
        h_f, c5_f = targeted_polish(best_h, Nb, deadline - 2, rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    final_c5 = float(compute_c5(best_h, len(best_h)))
    return best_h, final_c5, len(best_h)
