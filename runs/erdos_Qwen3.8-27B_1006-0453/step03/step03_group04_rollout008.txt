    """Compute the delta when swapping h[i]=1 to 0 and h[j]=0 to 1.
    Returns the delta vector to add to the correlation vector."""
    # Full correlation has length 2N-1, index N-1+s corresponds to shift s
    # Removing h[i]=1: c[s] -= (1-h[i-s]) for s in [-(N-1-i), i]
    # Adding h[j]=1: c[s] += (1-h[j-s]) for s in [-(N-1-j), j]

```python
def build_patterns(h, N):
    """Build the (N, 2N-1) pattern matrix.
    P[p, N-1+s] = 1-h[p-s] for valid s, 0 otherwise."""
    center = N - 1
    P = np.zeros((N, 2*N - 1))
    for p in range(N):
        # s ranges from -(N-1-p) to p
        s_vals = np.arange(-(N-1-p), p+1)
        idx = p - s_vals  # = p - s, should be in [0, N-1]
        valid = (idx >= 0) & (idx < N)
        P[p, center + s_vals[valid]] = 1.0 - h[idx[valid]]
    return P
```
