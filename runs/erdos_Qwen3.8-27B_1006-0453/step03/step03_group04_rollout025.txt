def compute_soft_grad(h, N, weights):
    """Compute weighted sum of gradients of all correlation terms."""
    N = len(h)
    grad = np.zeros(N)
    for k in range(-N+1, N):
        if weights is not None and abs(weights[k+N-1]) < 1e-10:
            continue
        w = weights[k+N-1] if weights is not None else 1.0
        if k >= 0:
            # corr = sum_{i=0}^{N-1-k} h[i]*(1-h[i+k])
            # d/dh[j]: (1-h[j+k]) for j in [0, N-1-k] minus h[j-k] for j in [k, N-1]
            if k == 0:
                continue  # corr[0] = 0 always
            # (1-h[j+k]) term: j in [0, N-1-k]
            grad[:N-k] += w * (1 - h[k:])
            # -h[j-k] term: j in [k, N-1]
            grad[k:] -= w * h[:N-k]
        else:
            # k < 0, let m = -k > 0
            # corr = sum_{i=m}^{N-1} h[i]*(1-h[i-m])
            # d/dh[j]: (1-h[j-m]) for j in [m, N-1] minus h[j+m] for j in [0, N-1-m]
            m = -k
            grad[m:] += w * (1 - h[:N-m])
            grad[:N-m] -= w * h[m:]
    return grad * (2.0/N)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    corr = compute_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def project_binary(h, N):
    """Project to binary with exactly N/2 ones."""
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        # Remove excess ones (preferentially those with lowest "importance")
        idx = np.where(h_b == 1)[0]
        # Remove the ones with lowest h values (closest to 0.5)
        vals = h[idx]
        order = np.argsort(vals)[:n_ones - N//2]
        h_b[idx[order]] = 0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        vals = h[idx]
        order = np.argsort(-vals)[:N//2 - n_ones]
        h_b[idx[order]] = 1
    return h_b

def soft_subgrad(h, N, tau, weights=None):
    """Compute soft subgradient of max_k corr[k]."""
    corr = compute_corr(h)
    # Soft weights
    if weights is None:
        w = np.exp(corr / tau)
        w /= w.sum()
    else:

def project_sum(h, target):
    """Project h to [0,1] with sum = target."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())

        else:

def to_binary(h, N):
    """Convert to binary with exactly N//2 ones."""
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    half = N // 2
    if n_ones > half:
        idx = np.where(h_b == 1)[0]
        # Remove ones with lowest h value
        order = np.argsort(h[idx])[:n_ones - half]
        h_b[idx[order]] = 0
    elif n_ones < half:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])[:half - n_ones]
        h_b[idx[order]] = 1
    return h_b

def soft_subgrad(h, N, tau):
    """Compute soft subgradient with temperature tau."""
    corr = compute_corr(h)
    peak = np.max(corr)
    w = np.exp((corr - peak) / tau)  # shifted for numerical stability
    w /= w.sum()
    dx = 2.0 / N

            continue

            grad[:N-k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:N-k]
        else:
            m = -k
            grad[m:] += wi * (1.0 - h[:N-m])
            grad[:N-m] -= wi * h[m:]

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    half = N // 2
    if n_ones > half:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])[:n_ones - half]
        h_b[idx[order]] = 0
    elif n_ones < half:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])[:half - n_ones]
        h_b[idx[order]] = 1
    return h_b

def cont_opt(h, N, target, dur, tau0=0.4):
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    lr = 0.4
    b1, b2, eps = 0.8, 0.93, 1e-7
    m_arr = np.zeros(N)
    v_arr = np.zeros(N)
    it = 0
    t_end = time.time() + dur
    n_top = min(120, 2 * N - 1)
    while time.time() < t_end:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_pk:
            best_pk = peak
            best_h = h.copy()
        if peak < 1e-6:
            break
        top_idx = np.argpartition(corr, -n_top)[-n_top:]
        ct = corr[top_idx]
        ws = np.exp((ct - ct.max()) / tau0)
        ws /= ws.sum()
        center = N - 1
        grad = np.zeros(N)
        for ii in range(n_top):
            ki = top_idx[ii]
            wi = ws[ii]
            if wi < 1e-9:
                continue
            k = ki - center
            if k == 0:
                continue
            if k > 0:
                grad[:N - k] += wi * (1.0 - h[k:])
                grad[k:] -= wi * h[:N - k]
            else:
                m_ = -k
                grad[m_:] += wi * (1.0 - h[:N - m_])
                grad[:N - m_] -= wi * h[m_:]
        grad *= dx
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * grad * grad
        it += 1
        h = h - lr * (m_arr / (1 - b1 ** it)) / (np.sqrt(v_arr / (1 - b2 ** it)) + eps)
        h = project_sum(h, target)
        if it % 300 == 0:
            lr *= 0.92
    return best_h, best_pk * dx

def local_search(h, N, dur, rng):
    dx = 2.0 / N
    h = to_binary(h, N).copy()
    corr = compute_corr(h)
    best_pk = float(np.max(corr))
    best_h = h.copy()
    cur_pk = best_pk
    t_end = time.time() + dur
    stall = 0
    while time.time() < t_end:
        corr = compute_corr(h)
        cur_pk = float(np.max(corr))
        if cur_pk < best_pk:
            best_pk = cur_pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        made = False
        # Targeted moves on top-4 shifts
        topN = min(4, 2 * N - 1)
        top_idx = np.argsort(corr)[-topN:]
        for ki in top_idx:
            k = ki - (N - 1)
            if k == 0:
                continue
            if k > 0:
                mask = (h[:N - k] > 0.5) & (h[k:] < 0.5)
                witnesses = np.where(mask)[0]
            else:
                m_ = -k
                mask = (h[m_:] > 0.5) & (h[:N - m_] < 0.5)
                witnesses = np.where(mask)[0] + m_
            if len(witnesses) == 0:
                continue
            n_try = min(10, len(zeros))
            picks = rng.choice(len(zeros), size=n_try, replace=False)
            for _ in range(n_try):
                j = zeros[picks[_]]
                i = witnesses[rng.randint(len(witnesses))]
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                npk = float(np.max(compute_corr(h)))
                if npk < cur_pk:
                    cur_pk = npk
                    made = True
                    break
                h[i], h[j] = h[j], h[i]
            if made:
                break
        if not made:
            if stall > 300:
                # Perturb and restart
                n_pert = rng.randint(5, 15)
                for _ in range(n_pert):
                    p = ones[rng.randint(len(ones))]
                    q = zeros[rng.randint(len(zeros))]
                    if p != q:
                        h[p], h[q] = h[q], h[p]
                stall = 0
                continue
            # Annealing move
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = h[q], h[p]
            npk = float(np.max(compute_corr(h)))
            delta = (npk - cur_pk)
            T = max(0.3, 2.5 * (1.0 + best_pk - cur_pk) * (1.0 - stall / 400.0))
            if npk < cur_pk or (T > 0.1 and rng.random() < np.exp(-delta / T)):
                cur_pk = npk
            else:
                h[p], h[q] = h[q], h[p]
    return best_h, best_pk * dx

def block_swap_search(h, N, dur, rng):
    """Try block swaps to escape local minima."""
    dx = 2.0 / N
    h = h.copy()
    corr = compute_corr(h)
    best_pk = float(np.max(corr))
    best_h = h.copy()
    cur_pk = best_pk
    t_end = time.time() + dur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    while time.time() < t_end:
        # Pick block sizes
        bl = rng.randint(2, 5)
        if len(ones) < bl or len(zeros) < bl:
            break
        p_start = ones[rng.randint(max(0, len(ones) - bl))]
        q_start = zeros[rng.randint(max(0, len(zeros) - bl))]
        # Check blocks don't overlap
        p_block = range(p_start, p_start + bl)
        q_block = range(q_start, q_start + bl)
        if any(pp in q_block for pp in p_block):
            continue
        # Perform block swap
        h[list(p_block)] = 0
        h[list(q_block)] = 1
        npk = float(np.max(compute_corr(h)))
        if npk < cur_pk:
            cur_pk = npk
            if npk < best_pk:
                best_pk = npk
                best_h = h.copy()
        else:
            h[list(p_block)] = 1
            h[list(q_block)] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return best_h, best_pk * dx
