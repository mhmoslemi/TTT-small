import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_avg_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def slsqp(h, obj_func, args, maxiter=300, ftol=1e-12):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

        b[rm] = 0
    elif n_ones < target:
        zeros = np.where(b == 0)[0]
        add = rng.choice(zeros, target - n_ones, replace=False)
        b[add] = 1
    return b

    cur = h_bin.copy()
    cc = bc
    T = t0

    while time.time() < t_deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        nc = float(np.max(np.correlate(cur, 1.0 - cur, mode='full'))) / n
        d = nc - cc
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-8)):

                best = cur.copy()
                restarts = 0

            cc = bc
