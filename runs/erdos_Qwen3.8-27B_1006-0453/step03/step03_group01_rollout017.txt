import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        c = float(np.clip(h - mid, 0.0, 1.0).sum())
        if c > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_softmax(h, tau, topK):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    nc = len(corr)
    sc = tau * (corr - corr.max())
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx = np.argpartition(corr, -k)[-k:]
    w_top = np.zeros(nc)
    w_top[idx] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 15.0) / 60.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, float(np.max(corr)) * (2.0 / N)
