def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float(dx * (m + np.log(np.sum(np.exp(T * (corr - m)))) / T))

def softmax_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            idx1 = np.arange(n - s)
            grad[idx1] += p[k] * (1.0 - h[idx1 + s])
            idx2 = np.arange(s, n)
            grad[idx2] -= p[k] * h[idx2 - s]
        else:
            a = -s
            idx1 = np.arange(a, n)
            grad[idx1] += p[k] * (1.0 - h[idx1 + s])
            idx2 = np.arange(n - a)
            grad[idx2] -= p[k] * h[idx2 - s]

        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    return np.ones(len(x))

def vec_subgrad(h, n_steps, top_k, alpha0, decay):
    n = len(h)
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        wts = corr[idx].astype(float)
        ws = wts.sum()
        wts = wts / ws if ws > 0 else np.ones(top_k) / top_k
        grad = np.zeros(n)
        for w, ki in zip(wts, idx):
            s = int(ki - (n - 1))
            if s >= 0:
                idx1 = np.arange(n - s)
                grad[idx1] += w * (1.0 - h[idx1 + s])
                idx2 = np.arange(s, n)
                grad[idx2] -= w * h[idx2 - s]
            else:

                idx1 = np.arange(a, n)
                grad[idx1] += w * (1.0 - h[idx1 + s])
                idx2 = np.arange(n - a)
                grad[idx2] -= w * h[idx2 - s]
        h = project_feasible(h - alpha * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        alpha *= decay
    return best, best_c5

def cheb_perturb(h, n_coeffs):
    """Perturb h using low-frequency Chebyshev modes."""
    n = len(h)
    x = np.linspace(-1, 1, n)
    delta = np.zeros(n)

        delta += c * np.cos(k * np.arccos(x))
    return project_feasible(h + delta, n)

def block_reshape(h, n):
    """Try a block-structured perturbation."""
    h_new = h.copy()
    n_blocks = np.random.randint(3, 8)
    block_sizes = np.random.randint(n // n_blocks // 2, n // n_blocks, n_blocks)
    total = block_sizes.sum()
    if total > n:

        block_sizes[-1] = n - block_sizes[:-1].sum()
        total = n
    if total <= 0:

    offsets = np.cumsum(np.concatenate([[0], block_sizes[:-1]]))
    for i in range(len(block_sizes)):
        start = offsets[i]
        end = start + block_sizes[i]
        if end > n:
            end = n
        shift = np.random.randint(-3, 4)
        h_new[start:end] = np.roll(h_new[start:end], shift)
    return h_new

    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]

                best_c5, best_h = cf, hf

    # Phase 6: Micro subgradient + final SLSQP
    if time.time() - t0 < budget_s * 0.98:
        hs, cs = vec_subgrad(best_h, 300, 3, 0.0015, 0.999)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        if time.time() - t0 < budget_s * 0.99:
            hf = slsqp_analytic(best_h, 200000, best_n, 500, 1e-15)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
