    k = abs(k_star)  # k_star can be positive or negative depending on correlation index
    # Actually, np.correlate(h, 1-h, mode='full') index k corresponds to lag k-(n-1)

    #          + (-h[j-lag]) if 0<=j-lag<n and j-lag is in the sum range
    # More precisely, for lag > 0:
    # f = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
    # d/dh[j]: from first factor: (1-h[j+lag]) for j in [0, n-1-lag]
    #          from second factor: -h[j-lag] for j in [lag, n-1]
    # For lag < 0, let l = -lag:
    # f = sum_{i=l}^{n-1} h[i]*(1-h[i-l])
    # d/dh[j]: first factor (1-h[j-l]) for j in [l, n-1]
    #          second factor -h[j+l] for j in [0, n-1-l]

    # f = sum_i h[i]*(1-h[i])
    # d/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]
    
    grad = np.zeros(n)
    if lag > 0:
        # First part: j from 0 to n-1-lag
        grad[0:n-lag] += (1-h[lag:n])
        # Second part: j from lag to n-1
        grad[lag:n] -= h[0:n-lag]

        l = -lag
        grad[l:n] += (1-h[0:n-l])
        grad[0:n-l] -= h[l:n]
    else:
        grad[:] = 1.0 - 2.0*h
    return grad

    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
    return np.clip(hp, 0.0, 1.0)

def subgrad_at(h, idx_max):
    n = len(h)
    lag = idx_max - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-lag] += (1.0 - h[lag:]) * dx
        grad[lag:] -= h[:n-lag] * dx
    elif lag < 0:
        l = -lag
        grad[l:] += (1.0 - h[:n-l]) * dx
        grad[:n-l] -= h[l:] * dx
    else:
        grad[:] = (1.0 - 2.0 * h) * dx
    return grad

def smooth_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

    return h.sum() - len(h) / 2.0

    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

    return compute_c5(np.clip(h, 0, 1))

                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(np.clip(res.x, 0, 1), n)
