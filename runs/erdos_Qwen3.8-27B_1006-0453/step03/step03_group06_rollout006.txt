                shift = k_sh - (n_cur - 1)
                for j in range(n_cur):
                    for delta in [0.03, -0.03, 0.06, -0.06]:
                        h_try = best_h.copy()
                        h_try[j] += delta
                        if h_try[j] < 0 or h_try[j] > 1:
                            continue
                        h_try = proj_feasible(h_try, n_cur)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    # Phase 5: Final high-T polish
    n_cur = len(best_h)
    for T in [8000, 20000, 50000]:
        if time.time() - t0 > budget_s * 0.98:
            break
        for k in [None, 2, 3, 5]:
            if time.time() - t0 > budget_s * 0.98:
                break
            try:
                if k is None:
                    hc = slsqp_run(best_h, n_cur, lse_c5, (T,), 2000, 1e-15)
                else:
                    hc = slsqp_run(best_h, n_cur, topk_lse_c5, (T, k), 2000, 1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass
    # Deflation final
    if time.time() - t0 < budget_s * 0.99:
        for k_defl in [3, 5]:
            for lam in [1.0, 2.0, 3.0]:
                if time.time() - t0 > budget_s * 0.99:
                    break
                try:
                    hc = slsqp_run(best_h, n_cur, defl_c5, (10000, k_defl, lam), 1000, 1e-15)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    best_h = proj_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
