    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

def smooth_max_corr(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n

        if cr < best_c5:
            best_c5, best_h = cr, hr
        hd = slsqp_opt(best_h, topk_obj, (2, best_n), 400, 1e-15)
        cd = compute_c5(hd)
        if cd < best_c5:
            best_c5, best_h = cd, hd

    # Phase 6: Multi-resolution
    for nr in [100, 240, 300]:
        if time.time() - t0 > t0 + budget_s * 0.84:
            break
        if nr == best_n or nr % 2 != 0:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, nr, endpoint=False)
        hu = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
        hu = project_feasible(hu, nr)
        hh = slsqp_opt(hu, smooth_max_corr, (200, nr), 250, 1e-12)
        hh = slsqp_opt(hh, topk_obj, (3, nr), 250, 1e-14)
        c5_hi = compute_c5(hh)
        x_dn = np.linspace(0, 2, best_n, endpoint=False)
        hd = np.interp(x_dn, x_new, np.concatenate([hh, [hh[0]]]))
        hd = project_feasible(hd, best_n)
        c5_dn = compute_c5(hd)
        if c5_dn < best_c5:
            best_c5, best_h = c5_dn, hd
        elif c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, hh, nr
            m = best_n // 2

    # Phase 7: Symmetric polish at final resolution
    if time.time() - t0 < t0 + budget_s * 0.92:
        m2 = best_n // 2
        h2 = best_h.copy()
        half2 = 0.5 * (h2[:m2] + (1.0 - h2[m2:]))
        half2 = np.clip(half2, 0, 1)
        hs2 = slsqp_sym(half2, sym_topk, (2, best_n), 500, 1e-15)
        c5s2 = sym_c5(hs2, best_n)
        if c5s2 < best_c5:
            best_c5 = c5s2
            best_h = np.concatenate([hs2, 1.0 - hs2])

    # Phase 8: Final intensive local search
    if time.time() - t0 < t0 + budget_s * 0.97:
        for _ in range(4):
            if time.time() - t0 > t0 + budget_s * 0.96:
                break
            hd = slsqp_opt(best_h, topk_obj, (2, best_n), 300, 1e-15)
            cd = compute_c5(hd)
            if cd < best_c5:
                best_c5, best_h = cd, hd
            hr, cr = shift_repair(best_h, best_n, 20)
            if cr < best_c5:
                best_c5, best_h = cr, hr

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
