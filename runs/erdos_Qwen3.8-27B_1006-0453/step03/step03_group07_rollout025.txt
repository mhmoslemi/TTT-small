    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_c = np.max(corr)
    exp_vals = np.exp(T * (corr - max_c))
    p = exp_vals / np.exp_vals.sum()  # p has length 2n-1
    
    # Gradient: G[j] = dx * sum_k p[k] * dc[k]/dh[j]
    G = np.zeros(n)

        # dc[k]/dh[j] = (1-h[j+d]) if 0<=j+d<n, minus h[j-d] if 0<=j-d<n

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)

    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ev = np.exp(T * (corr - m))
    Z = ev.sum()
    p = ev / Z
    obj = dx * (m + np.log(Z) / T)
    G = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        pk = p[k]
        if pk < 1e-18:
            continue
        if d >= 0:
            if n - d > 0:
                G[0:n-d] += pk * (1.0 - h[d:n])
            G[d:n] -= pk * h[0:n-d]
        else:
            dd = -d
            G[dd:n] += pk * (1.0 - h[0:n-dd])
            G[0:n-dd] -= pk * h[dd:n]
    G *= dx
    s = h.sum() - n / 2.0
    obj += lam * s * s
    G += 2.0 * lam * s
    return obj, G

def lbfgs_optimize(h0, T, lam, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_and_grad, h0, args=(T, lam), jac=True,
                   method='L-BFGS-B', bounds=bnds, maxiter=maxiter,
                   options={'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40})
    return make_feasible(res.x, n)

        best_h = np.zeros(n0)

        best_h = make_feasible(best_h, n0)
    n = len(best_h)
    best_c5 = compute_c5(best_h)
