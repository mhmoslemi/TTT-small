import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_sym(params, n):
    h = np.empty(n)
    half = n // 2
    h[:half] = params
    h[half:] = 1.0 - params[::-1]
    return h

def to_params(h, n):
    half = n // 2
    p = 0.5 * (h[:half] + 1.0 - h[n - 1:n - 1 - half:-1])
    return np.clip(p, 0.0, 1.0)

def sym_obj(params, T, n, topk):
    h = make_sym(params, n)
    corr = get_corr(h)
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        vals = corr
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(T * (vals - m)))) / T) * (2.0 / n)

def full_obj(hv, T, n, topk):
    c = get_corr(hv)
    if topk is not None and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        vals = c[idx]
    else:
        vals = c
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(T * (vals - m)))) / T) * (2.0 / n)

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h = h + room * (diff / total)
    else:
        total = h.sum()
        if total > 1e-12:
            h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sym_optimize(p0, n, T, k, maxiter):
    half = n // 2
    res = minimize(sym_obj, p0, args=(T, n, k),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * half,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
    return np.clip(res.x, 0.0, 1.0)

def full_optimize(h0, n, T, k, maxiter):
    res = minimize(full_obj, h0, args=(T, n, k),
                   method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': sum_eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_h(res.x, n)

def coord_search(h, n, max_iter, time_limit, t0):
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        improved = False
        corr = get_corr(h)
        top_idx = np.argsort(corr)[-4:]
        for ki in top_idx:
            shift = ki - (n - 1)
            for j in range(n):
                for d in [0.03, -0.03]:
                    ht = h.copy()
                    ht[j] += d
                    ht = project_h(ht, n)
                    ct = compute_c5(ht)
                    if ct < best_c5 - 1e-15:
                        best_c5 = ct
                        h = ht
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and it > 80:
            break
    return h, best_c5
