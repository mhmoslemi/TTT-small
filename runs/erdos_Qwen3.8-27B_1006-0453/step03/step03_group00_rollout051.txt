        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def get_weights(corr, sharpness):
    m = np.max(corr)
    spread = m - np.min(corr) + 1e-10
    w = np.exp(sharpness * (corr - m) / spread)
    w /= w.sum()
    return w

def compute_grad_weighted(h, w, n):
    """Compute gradient of sum_k w[k]*corr[k] w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[n-1+s]
        if ws < 1e-10:
            continue
        # First term: (1-h[j-s]) for valid j
        j_lo = max(0, s)
        j_hi = min(n-1, n-1+s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += ws * (1.0 - h[j_lo-s:j_hi+1-s])
        # Second term: -h[j+s] for valid j
        j_lo2 = max(0, -s)
        j_hi2 = min(n-1, n-1-s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] += ws * (-h[j_lo2+s:j_hi2+1+s])
    return grad * dx

def subgrad_opt(h, n, t_end, lr0=0.05, momentum=0.85, sharpness=80):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    stall = 0
    
    while time.time() < t_end:
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = get_weights(corr, sharpness)
        grad = compute_grad_weighted(h, w, n)
        
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            lr *= 0.97
            if stall > 100:
                h = best_h.copy()
                v = np.zeros(n)
                lr = max(lr0 * 0.4, 0.005)
                sharpness = min(sharpness * 1.5, 500)
                stall = 0
        h = h_new
    
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    from scipy.optimize import minimize_scalar
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Compute 1D subproblem
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            coef = np.zeros(2*n-1)
            for k in range(-(n-1), n):
                c = 0.0
                j1 = i + k
                j2 = i - k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c += (-best_h[j2])
                coef[k+n-1] = c
            def f1d(hi):
                return np.max(corr_base + hi * coef) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            h_try = best_h.copy()
            h_try[i] = res.x
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

    best_h, best_c5 = active_set_slsqp(best_h, n, t3, topk=6, inner_iters=4)

    # Phase 4: SLSQP polish
    for T in [800, 3000, 8000]:
        if elapsed() > budget_s * 0.42:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Multi-resolution exploration
    for n_try in [80, 100, 128, 144, 192]:
        if elapsed() > budget_s * 0.58:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_pert = h_interp + 0.06 * np.random.randn(n_try)
        h_pert = project_feasible(h_pert, n_try)
        for T in [150, 1500]:
            if elapsed() > budget_s * 0.58:
                break
            try:
                hc = slsqp_opt(h_pert, T, n_try, maxiter=500, ftol=1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    hc_full = interpolate_h(hc, n)
                    c5v2 = compute_c5(hc_full)
                    if c5v2 < best_c5:
                        best_c5 = c5v2
                        best_h = hc_full
            except Exception:
                pass

    # Phase 6: Second subgradient pass with higher sharpness
    if elapsed() < budget_s * 0.60:
        t6 = t0 + budget_s * 0.72
        best_h, best_c5 = subgrad_opt(best_h, n, t6, lr0=0.02, momentum=0.88, sharpness0=120)

    # Phase 7: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(35):
        if elapsed() > budget_s * 0.84:
            break
        cool = 1.0 - 0.85 * (it / 35.0)
        step = 0.03 + 0.10 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        try:
            hc = slsqp_opt(h_pert, 600, n, maxiter=250, ftol=1e-13)
            c5v = compute_c5(hc)
            temp = 0.003 * cool + 0.0002
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 8: Coordinate descent fine-tuning
    if elapsed() < budget_s * 0.85:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=20)

    # Phase 9: Final high-T SLSQP polish
    for T in [15000, 50000]:
        if elapsed() > budget_s * 0.985:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
