import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    s = x.sum() - n / 2.0
    return val * dx + lam * s * s

def lbfgsb_opt(h0, T, n, lam=50.0, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T, lam), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return project_feasible(res.x, n)

def coord_search(h, n, n_rounds=200, base_lr=0.015, time_limit=None, t0=None):
    best = h.copy()
    best_c5 = compute_c5(best)
    for r in range(n_rounds):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        lr = base_lr * (1.0 - 0.3 * r / max(1, n_rounds))
        order = np.random.permutation(n)
        improved = False
        for j in order:
            deltas = [lr, -lr, lr*0.5, -lr*0.5, lr*2.0, -lr*2.0, lr*0.25, -lr*0.25]
            for d in deltas:
                hv = best.copy()
                hv[j] = np.clip(hv[j] + d, 0.0, 1.0)
                hv = project_feasible(hv, n)
                c5v = compute_c5(hv)
                if c5v < best_c5 - 1e-14:
                    best_c5 = c5v
                    best = hv
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def rounding_pass(h, n, t0=None, budget_frac=None, budget_s=1100):
    """Try setting individual coordinates to 0, 0.5, or 1."""
    best = h.copy()
    best_c5 = compute_c5(best)
    improved_any = True
    while improved_any:
        if t0 is not None and time.time() - t0 > budget_frac * budget_s:
            break
        improved_any = False
        for j in range(n):
            for v in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(best[j] - v) < 1e-6:
                    continue
                hv = best.copy()
                hv[j] = v
                hv = project_feasible(hv, n)
                c5v = compute_c5(hv)
                if c5v < best_c5 - 1e-14:
                    best_c5 = c5v
                    best = hv
                    improved_any = True
                    break
    return best, best_c5

def multi_start_lbf(h0, n, T, n_starts, t0, budget_frac, budget_s):
    best = h0.copy()
    best_c5 = compute_c5(best)
    for s in range(n_starts):
        if time.time() - t0 > budget_frac * budget_s:
            break
        r = np.random.rand()
        if r < 0.4:
            h0s = best_h + np.random.normal(0, 0.05, n)
        elif r < 0.7:
            blen = np.random.randint(3, n // 5 + 1)
            start = np.random.randint(0, n - blen)
            h0s = best_h.copy()
            h0s[start:start+blen] += np.random.uniform(-0.1, 0.1, blen)
        else:
            h0s = best_h.copy()
            idx = np.random.choice(n, size=n//4, replace=False)
            h0s[idx] += np.random.uniform(-0.08, 0.08, len(idx))
        h0s = project_feasible(h0s, n)
        try:
            hc = lbfgsb_opt(h0s, T, n, lam=80.0, maxiter=1500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best = hc
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 60
        best_h = np.zeros(n0)
        best_h[:n0//2] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Multi-start L-BFGS-B at moderate T
    for T in [100, 500]:
        if time.time() - t0 > budget_s * 0.25:
            break
        best_h, best_c5 = multi_start_lbf(best_h, n, T, 4, t0, budget_s * 0.25, budget_s)

    # Phase 2: Fine coordinate descent
    if time.time() - t0 < budget_s * 0.45:
        best_h, best_c5 = coord_search(best_h, n, n_rounds=150, base_lr=0.012,
                                       time_limit=budget_s * 0.45, t0=t0)

    # Phase 3: Rounding pass to exploit structure
    if time.time() - t0 < budget_s * 0.55:
        best_h, best_c5 = rounding_pass(best_h, n, t0, 0.55, budget_s)

    # Phase 4: High-T L-BFGS-B polish
    for T in [2000, 10000, 50000]:
        if time.time() - t0 > budget_s * 0.72:
            break
        try:
            hc = lbfgsb_opt(best_h, T, n, lam=200.0, maxiter=5000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Multi-resolution exploration
    best_n = len(best_h)
    for n_try in [80, 100, 120, 200]:
        if time.time() - t0 > budget_s * 0.88:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        if h_interp.sum() * (2.0/n_try) < 0.99:
            h_interp = np.clip(h_interp, 0, 1)
        for T in [300, 3000]:
            if time.time() - t0 > budget_s * 0.89:
                break
            try:
                hc = lbfgsb_opt(h_interp, T, n_try, lam=100.0, maxiter=3000)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    best_n = n_try
            except Exception:
                pass

    # Phase 6: Final coordinate descent with very fine steps
    n = len(best_h)
    if time.time() - t0 < budget_s * 0.96:
        best_h, best_c5 = coord_search(best_h, n, n_rounds=100, base_lr=0.005,
                                       time_limit=budget_s * 0.96, t0=t0)

    # Phase 7: Ultra-fine rounding + L-BFGS-B
    if time.time() - t0 < budget_s * 0.98:
        best_h, best_c5 = rounding_pass(best_h, n, t0, 0.98, budget_s)
        try:
            hc = lbfgsb_opt(best_h, 100000, n, lam=500.0, maxiter=5000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final SLSQP for precision
    n = len(best_h)
    if time.time() - t0 < budget_s * 0.995:
        try:
            def slq_obj(x, T=100000):
                dx_v = 2.0 / n
                corr = np.correlate(x, 1.0 - x, mode='full')
                m = np.max(corr)
                val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
                return val * dx_v + 1000.0 * (x.sum() - n/2.0)**2
            res = minimize(slq_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 3000, 'ftol': 1e-16})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
