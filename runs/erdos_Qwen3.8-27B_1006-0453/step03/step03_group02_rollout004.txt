def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, L, N):
    """Gradient of corr at lag L (shift = L - (N-1)) w.r.t. h"""
    grad = np.zeros(N)
    shift = L - (N - 1)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        n_valid = N - shift
        grad[:n_valid] += (1.0 - h[shift:])
        grad[shift:] -= h[:n_valid]
    else:
        s = -shift
        n_valid = N - s
        grad[s:] += (1.0 - h[:n_valid])
        grad[:n_valid] -= h[s:]
    return grad

    c5 = float(np.max(corr)) * (2.0 / N)
    top_idx = np.argsort(corr)[-K:]
    c_top = corr[top_idx]
    sc = tau * c_top - tau * c_top.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(K) / K
    else:
        w /= ws
    return w, top_idx, c5

            best_c5 = c5_try
            return best_h, best_c5, lr

    return best_h, best_c5, lr0

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0

        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, L, N):
    grad = np.zeros(N)
    shift = L - (N - 1)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        nv = N - shift
        grad[:nv] += (1.0 - h[shift:shift+nv])
        grad[shift:shift+nv] -= h[:nv]
    else:
        s = -shift
        nv = N - s
        grad[s:s+nv] += (1.0 - h[:nv])
        grad[:nv] -= h[s:s+nv]
    return grad

def weighted_grad(h, N, K, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    top_idx = np.argsort(corr)[-K:]
    c_top = corr[top_idx]
    sc = tau * (c_top - c_top.max())
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(K) / K
    else:
        w /= ws
    grad = np.zeros(N)
    for ix in range(K):
        if w[ix] > 1e-8:
            grad += w[ix] * grad_lag(h, int(top_idx[ix]), N)
    return grad, c5

def grad_phase(h, N, target, dx, deadline, Ks, Ke, ts, te, lrs, lre, n_iter):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline and it < n_iter:
        f = min(it / max(n_iter, 1), 1.0)
        K = max(1, int(round(Ks + (Ke - Ks) * f)))
        tau = ts * (te / ts) ** f
        lr = lrs * (lre / lrs) ** f
        grad, c5v = weighted_grad(h, N, K, tau)
        it += 1

        v = 0.999 * v + 0.001 * grad * grad
        mh = m / (1 - 0.9 ** it)
        vh = v / (1 - 0.999 ** it)
        h_new = project_sum(h - lr * mh / (np.sqrt(vh) + 1e-8), target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coord_refine(h, N, target, dx, deadline):

def swap_search(h, N, target, dx, deadline, n_trials=5000):

                            lrs=0.001, lre=0.00005, n_iter=300)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    Nb = len(best_h)
    final_h = project_sum(best_h, Nb // 2)
    final_c5 = compute_c5(final_h, 2.0 / Nb)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h

    return best_h, best_c5, len(best_h)
