import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def thue_morse_init(N):
    t = np.zeros(N, dtype=float)
    for i in range(1, N):
        t[i] = t[i >> 1] ^ (i & 1)
    # Adjust to have exactly N/2 ones
    n_ones = int(t.sum())
    if n_ones > N // 2:
        idx = np.where(t == 1)[0]
        t[idx[N//2:]] = 0
    elif n_ones < N // 2:
        idx = np.where(t == 0)[0]
        t[idx[:N//2 - n_ones]] = 1
    return t

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    wk = 1.0 / k
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak - 1e-10:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)

    h_b = (h > 0.5).astype(float)
    nn = int(h_b.sum())
    if nn > N // 2:
        idx = np.where(h_b == 1)[0]
        h_b[idx[N//2:]] = 0
    elif nn < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N//2 - nn]] = 1
    
    dx = 2.0 / N
    best = h_b.copy()
    best_val = float(np.max(compute_corr(h_b))) * dx
    cur = h_b.copy()
    cur_val = float(np.max(compute_corr(cur))) * dx
    
    no_improve = 0
    max_stuck = 3000
    
    while time.time() < deadline:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        if r < 0.6:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            cur[i] = 0.0; cur[j] = 1.0
            nv = float(np.max(compute_corr(cur))) * dx
            if nv < cur_val - 1e-14:
                cur_val = nv
                no_improve = 0
                if nv < best_val:
                    best_val = nv
                    best = cur.copy()
            else:
                cur[i] = 1.0; cur[j] = 0.0
                no_improve += 1
        elif r < 0.85:
            # Double swap
            i1, i2 = ones[rng.randint(len(ones))], ones[rng.randint(len(ones))]
            j1, j2 = zeros[rng.randint(len(zeros))], zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                no_improve += 1
                continue
            cur[i1] = 0; cur[i2] = 0; cur[j1] = 1; cur[j2] = 1
            nv = float(np.max(compute_corr(cur))) * dx
            if nv < cur_val - 1e-14:
                cur_val = nv
                no_improve = 0
                if nv < best_val:
                    best_val = nv
                    best = cur.copy()
            else:
                cur[i1] = 1; cur[i2] = 1; cur[j1] = 0; cur[j2] = 0
                no_improve += 1
        else:
            # Triple flip (maintain count)
            nf = 3
            if len(ones) < nf or len(zeros) < nf:
                no_improve += 1
                continue
            fi = rng.choice(len(ones), nf, replace=False)
            fj = rng.choice(len(zeros), nf, replace=False)
            a = ones[fi]; b = zeros[fj]
            cur[a] = 0

            cur_val = float(np.max(compute_corr(cur))) * dx
            no_improve = 0
    
    return best, best_val

def lag_focused_move(cur, N, rng, top_lags=3):
    corr = compute_corr(cur)
    center = N - 1
    # Get top lags (excluding center which is 0 for binary)
    top_k = np.argpartition(corr, -top_lags)[-top_lags:]
    
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    best_move = None
    best_new_val = float(np.max(compute_corr(cur)))
    
    for lag_idx in top_k:
        lag = lag_idx - center
        if lag == 0:
            continue
        # Find confluent pairs at this lag
        if lag > 0:
            # Pairs (i, i+lag) where cur[i]=1, cur[i+lag]=0
            valid_i = np.arange(max(0, -lag), min(N-lag, N))
            confluent = valid_i[(cur[valid_i] == 1) & (cur[valid_i + lag] == 0)]
        else:
            m = -lag
            valid_i = np.arange(0, N-m)
            confluent = valid_i[(cur[valid_i] == 1) & (cur[valid_i + m] == 0)]
        
        if len(confluent) == 0:
            continue
        
        # Pick a random confluent position
        i = confluent[rng.randint(len(confluent))]
        if lag > 0:
            partner = i + lag  # cur[partner] = 0
        else:
            partner = i - lag  # cur[partner] = 0
        
        # Option 1: flip cur[i] to 0, put 1 at random zero
        if len(zeros) > 1:
            j = zeros[rng.randint(len(zeros))]
            if j != partner:
                cur[i] = 0; cur[j] = 1
                nv = float(np.max(compute_corr(cur)))
                if nv < best_new_val:
                    best_new_val = nv
                    best_move = (i, j, 'flip1')
                # Revert
                cur[i] = 1; cur[j] = 0
        
        # Option 2: flip cur[partner] to 1, put 0 at random one
        if len(ones) > 1:
            j = ones[rng.randint(len(ones))]
            if j != i:
                cur[partner] = 1; cur[j] = 0
                nv = float(np.max(compute_corr(cur)))
                if nv < best_new_val:
                    best_new_val = nv
                    best_move = (partner, j, 'flip2')
                # Revert
                cur[partner] = 0; cur[j] = 1
    
    if best_move is not None:
        i, j, typ = best_move
        if typ == 'flip1':
            cur[i] = 0; cur[j] = 1
        else:
            cur[i] = 1; cur[j] = 0
        return True, best_new_val
    return False, best_new_val

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def thue_morse_init(N):
    t = np.zeros(N, dtype=float)
    for i in range(1, N):
        t[i] = t[i >> 1] ^ (i & 1)
    n_ones = int(t.sum())
    if n_ones > N // 2:
        idx = np.where(t == 1)[0]
        t[idx[N // 2:]] = 0
    elif n_ones < N // 2:
        idx = np.where(t == 0)[0]
        t[idx[:N // 2 - n_ones]] = 1
    return t

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    wk = 1.0 / k
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak - 1e-10:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)

def lag_focused_ls(h, N, deadline, rng):
    """Lag-focused iterated local search with multi-point moves."""
    h_b = (h > 0.5).astype(float)
    nn = int(h_b.sum())
    if nn > N // 2:
        idx = np.where(h_b == 1)[0]
        h_b[idx[N // 2:]] = 0
    elif nn < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N // 2 - nn]] = 1
    dx = 2.0 / N
    best = h_b.copy()
    best_val = float(np.max(compute_corr(h_b))) * dx
    cur = h_b.copy()
    cur_val = float(np.max(compute_corr(cur))) * dx
    no_improve = 0
    max_stuck = 4000
    center = N - 1
    while time.time() < deadline:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        r = rng.random()
        if r < 0.55:
            # Lag-focused: target worst lag
            corr = compute_corr(cur)
            top_idx = np.argpartition(corr, -3)[-3:]
            for li in top_idx:
                lag = li - center
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= N:
                    continue
                if m < N:
                    vi = np.arange(max(0, -lag), N - m) if lag > 0 else np.arange(0, N - m)
                    if lag > 0:
                        conf = vi[(cur[vi] == 1) & (cur[vi + m] == 0)]
                    else:
                        conf = vi[(cur[vi] == 1) & (cur[vi - lag] == 0)]
                    if len(conf) == 0:
                        continue
                    n_try = min(2, len(conf))
                    picks = rng.choice(len(conf), n_try, replace=False)
                    for p in picks:
                        i = conf[picks[p]] if False else conf[p]
                        if lag > 0:
                            partner = i + m
                        else:
                            partner = i - lag
                        # Flip i from 1 to 0, put 1 at random zero
                        if len(zeros) > 1 and partner not in zeros[:0]:
                            j = zeros[rng.randint(len(zeros))]
                            if j != partner:
                                cur[i] = 0.0; cur[j] = 1.0
                                nv = float(np.max(compute_corr(cur))) * dx
                                if nv < cur_val:
                                    cur_val = nv
                                    improved = True
                                    if nv < best_val:
                                        best_val = nv
                                        best = cur.copy()
                                else:
                                    cur[i] = 1.0; cur[j] = 0.0
                            else:
                                # Flip partner from 0 to 1, put 0 at random one
                                j = ones[rng.randint(len(ones))]
                                if j != i:
                                    cur[partner] = 1.0; cur[j] = 0.0
                                    nv = float(np.max(compute_corr(cur))) * dx
                                    if nv < cur_val:
                                        cur_val = nv
                                        improved = True
                                        if nv < best_val:
                                            best_val = nv
                                            best = cur.copy()
                                    else:
                                        cur[partner] = 0.0; cur[j] = 1.0
        elif r < 0.85:
            # Random single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            cur[i] = 0.0; cur[j] = 1.0
            nv = float(np.max(compute_corr(cur))) * dx
            if nv < cur_val:
                cur_val = nv
                improved = True
                if nv < best_val:
                    best_val = nv
                    best = cur.copy()
            else:
                cur[i] = 1.0; cur[j] = 0.0
        else:
            # Double swap
            if len(ones) < 2 or len(zeros) < 2:
                no_improve += 1
                continue
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                no_improve += 1
                continue
            cur[i1] = 0; cur[i2] = 0; cur[j1] = 1; cur[j2] = 1
            nv = float(np.max(compute_corr(cur))) * dx
            if nv < cur_val:
                cur_val = nv
                improved = True
                if nv < best_val:
                    best_val = nv
                    best = cur.copy()
            else:
                cur[i1] = 1; cur[i2] = 1; cur[j1] = 0; cur[j2] = 0
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_stuck:
            npairs = rng.randint(3, 6)
            for _ in range(npairs):
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                cur[ii] = 0; cur[jj] = 1
            cur_val = float(np.max(compute_corr(cur))) * dx
            no_improve = 0
    return best, best_val
