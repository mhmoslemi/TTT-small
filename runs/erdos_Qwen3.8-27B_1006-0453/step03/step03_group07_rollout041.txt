```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        idx = np.where(room > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += room[idx] * (diff / room[idx].sum())
    else:
        idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += h[idx] * (diff / h[idx].sum())
    h = np.clip(h, 0.0, 1.0)
    # Fix any residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        h[0] = np.clip(h[0] + residual, 0, 1)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def get_subgrad(h, lag):
    """Compute subgradient of the corr at a specific lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        # corr = sum_{i=lag}^{n-1} h[i]*(1-h[i-lag])
        g[lag:] += (1.0 - h[:-lag]) if lag > 0 else (1.0 - h)
        g[:n-lag] -= h[lag:] if lag > 0 else h
    else:
        m = -lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:-m]
    return g

def slsqp_opt(h0, T, n, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def targeted_fix(h, n, top_k=5, steps=20):
    """Targeted coordinate descent on worst lags."""
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[::-1][:top_k]
        lags = idx - (n - 1)
        grad = np.zeros(n)
        for lag in lags:
            grad += get_subgrad(h, lag)
        # Normalized subgradient descent
        gnorm = np.sqrt(np.sum(grad**2))
        if gnorm < 1e-12:
            break
        lr = 0.02
        h_new = h - lr * grad / gnorm
        h_new = make_feasible(h_new, n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
        else:
            lr *= 0.5
            h_new = h - lr * grad / gnorm
            h_new = make_feasible(h_new, n)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        m = room.sum()
        if m > 1e-12:
            h += room * (diff / m)
    else:
        m = h.sum()
        if m > 1e-12:
            h += h * (diff / m)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_eq(h, n):
    return h.sum() - n / 2.0

def do_slsqp(h0, T, n, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def get_subgrad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def subgrad_step(h, n, top_k, lr):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:top_k]
    lags = idx - (n - 1)
    grad = np.zeros(n)
    for lag in lags:
        grad += get_subgrad(h, int(lag))
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h.copy()
    h_new = h - lr * grad / gn
    return make_feasible(h_new, n)

def pair_adjust(h, n, lag, strength):
    h_new = h.copy()
    if lag > 0:
        pairs_a = h_new[lag:]
        pairs_b = h_new[:-lag]
    else:
        m = -lag
        pairs_a = h_new[:n - m]
        pairs_b = h_new[m:]
    idx = np.where((pairs_a > 0.3) & (pairs_b < 0.7))[0]
    if len(idx) == 0:
        return h_new
    delta = strength * (pairs_a[idx] + pairs_b[idx] - 1.0)
    delta = np.clip(np.abs(delta) * np.sign(0.5 - (pairs_a[idx] - pairs_b[idx])), -0.05, 0.05)
    if lag > 0:
        h_new[lag:][idx] -= delta
        h_new[:-lag][idx] += delta
    else:
        m = -lag
        h_new[:n - m][idx] -= delta
        h_new[m:][idx] += delta
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float).copy()
    else:
        n = 200
        best_h = np.zeros(n)
        best_h[::3] = 1.0
        best_h = make_feasible(best_h, n)

    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing temperature (20% budget)
    t1 = t0 + budget_s * 0.20
    for T in [50, 200, 800, 3000, 10000]:
        if time.time() > t1:
            break
        try:
            h_try = do_slsqp(best_h.copy(), T, n, 1200, 1e-15)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try
        except Exception:
            pass

    # Phase 2: Subgradient descent on worst lags (30% budget)
    t2 = t0 + budget_s * 0.50
    sg_h = best_h.copy()
    sg_c5 = best_c5
    no_improve = 0
    while time.time() < t2:
        top_k = 3 + (no_improve // 30)
        lr = 0.015 * (1.0 + 0.5 * np.sin(no_improve * 0.1))
        sg_h = subgrad_step(sg_h, n, top_k, lr)
        # Pair adjustment at worst lag
        corr = np.correlate(sg_h, 1.0 - sg_h, mode='full')
        worst_lag = int(np.argmax(corr) - (n - 1))
        sg_h = pair_adjust(sg_h, n, worst_lag, 0.03)
        c5v = compute_c5(sg_h)
        if c5v < sg_c5 - 1e-14:
            sg_c5 = c5v
            no_improve = 0
            if c5v < best_c5:
                best_c5 = c5v
                best_h = sg_h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                sg_h = best_h + 0.03 * np.random.randn(n)
                sg_h = make_feasible(sg_h, n)
                sg_c5 = compute_c5(sg_h)
                no_improve = 0

    # Phase 3: Interleaved SLSQP + subgradient (30% budget)
    t3 = t0 + budget_s * 0.80
    while time.time() < t3:
        # SLSQP polish
        T_use = 5000 + int(15000 * np.random.rand())
        try:
            h_try = do_slsqp(best_h.copy(), T_use, n, 600, 1e-14)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try
        except Exception:
            pass
        # Subgradient burst
        for _ in range(5):
            if time.time() > t3:
                break
            h_try = subgrad_step(best_h, n, 4, 0.01)
            corr = np.correlate(h_try, 1.0 - h_try, mode='full')
            wl = int(np.argmax(corr) - (n - 1))
            h_try = pair_adjust(h_try, n, wl, 0.02)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try

    # Phase 4: Basin hopping with SLSQP (10% budget)
    t4 = t0 + budget_s * 0.90
    bh_h = best_h.copy()
    bh_c5 = best_c5
    while time.time() < t4:
        scale = 0.04 + 0.08 * np.random.rand()
        h_pert = bh_h + scale * np.random.randn(n)
        h_pert = make_feasible(h_pert, n)
        T_use = 2000 + int(20000 * np.random.rand())
        try:
            h_try = do_slsqp(h_pert, T_use, n, 300, 1e-12)
        except Exception:
            h_try = h_pert
        c5v = compute_c5(h_try)
        temp = 0.002 * (1.0 - (time.time() - t0) / max(budget_s, 1e-6)) + 0.0001
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-8)):
            bh_c5 = c5v
            bh_h = h_try
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try

    # Phase 5: Final high-T polish (10% budget)
    t5 = t0 + budget_s * 0.995
    for T in [50000, 200000, 500000]:
        if time.time() > t5:
            break
        try:
            h_try = do_slsqp(best_h.copy(), T, n, 2000, 1e-16)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try
        except Exception:
            pass
    # One more subgradient burst
    for _ in range(20):
        if time.time() > t5:
            break
        h_try = subgrad_step(best_h, n, 5, 0.005)
        c5v = compute_c5(h_try)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_try

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
