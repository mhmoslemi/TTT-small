import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    return np.max(vals), vals

def exact_project(y, S):
    """Euclidean projection onto {h: 0<=h<=1, sum(h)=S}."""
    n = len(y)
    lo, hi = -max(y) - 1, max(y) + 1  # tighter bounds
    # Actually lambda range: h_i = clip(y_i - lambda, 0, 1)
    # sum = S. As lambda goes from -inf to +inf, sum goes from n to 0.
    lo, hi = -max(0.0, np.max(y)) - 1.0, max(0.0, np.min(y)) + 1.0
    # More carefully: we need lambda such that sum(clip(y-lambda, 0, 1)) = S
    # lambda in [min(y)-1, max(y)+1]
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    # Fix tiny rounding
    diff = S - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            # All at boundaries, just scale non-boundary... shouldn't happen
            h = np.clip(h + diff / n, 0, 1)
    return h

def compute_g_grad(h, d):
    n = len(h)
    if d == 0:
        g = np.sum(h - h*h)
        grad = 1.0 - 2.0*h
    elif d > 0:
        g = 0.0
        grad = np.zeros(n)
        for i in range(d, n):
            g += h[i] * (1.0 - h[i-d])
            grad[i] += (1.0 - h[i-d])
            grad[i-d] += (-h[i])
    else:
        L = -d
        g = 0.0
        grad = np.zeros(n)
        for i in range(0, n-L):
            g += h[i] * (1.0 - h[i+L])
            grad[i] += (1.0 - h[i+L])
            grad[i+L] += (-h[i])
    return g, grad

def bundle_method(h0, n, t_end, max_iter=500):
    h = h0.copy()
    best_c5, best_h = compute_c5_full(h)[0], h.copy()
    dx = 2.0 / n
    S = n / 2.0
    active_ks = {}  # dict: d -> (g, grad)
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Compute all correlations
        c5, corr_vals = compute_c5_full(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find active set (top correlations)
        # corr_vals has length 2n-1, index k corresponds to d = k-(n-1)
        max_c5 = np.max(corr_vals)
        eps = 1e-8
        active_d_list = [k - (n-1) for k in np.where(corr_vals >= max_c5 - eps)[0]]
        
        # Limit active set size
        if len(active_d_list) > 40:
            corr_at_active = np.array([corr_vals[d + n - 1] for d in active_d_list])
            order = np.argsort(-corr_at_active)[:40]
            active_d_list = [active_d_list[i] for i in order]
        
        # Compute gradients for active set
        A_rows = []
        b_vals = []
        for d in active_d_list:
            g, grad = compute_g_grad(h, d)
            rhs = g - np.dot(grad, h)  # = g - grad·h (the linearization offset)
            # LP constraint: grad · h' - t <= rhs
            row = np.append(grad, -1.0)
            A_rows.append(row)
            b_vals.append(rhs)
        
        if len(A_rows) == 0:
            break
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        bounds = [(0,1)]*n + [(None, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        
        if not res.success:
            break
        
        h_prop = res.x[:n]
        
        # Check if it's actually different
        if np.max(np.abs(h_prop - h)) < 1e-10:
            # Stuck, add more to active set
            ...
            break
        
        # Evaluate and accept or line search
        c5_prop = compute_c5_full(h_prop)[0]
        if c5_prop < c5 - 1e-12:
            h = exact_project(h_prop, S)
        else:
            # Line search
            best_alpha = 0.0
            best_val = c5
            for alpha in np.linspace(0.1, 1.0, 10):
                h_try = exact_project(h + alpha * (h_prop - h), S)
                c5_try = compute_c5_full(h_try)[0]
                if c5_try < best_val:
                    best_val = c5_try
                    best_alpha = alpha
            if best_alpha > 0:
                h = exact_project(h + best_alpha * (h_prop - h), S)
            else:
                break  # No improvement possible
    
    return best_h, best_c5

            g += h[i] * (1.0 - h[i + L])
            grad[i] += (1.0 - h[i + L])
            grad[i + L] += (-h[i])
    return g, grad

def bundle_step(h, n, S, max_active=30):
    c5, corr_vals = compute_c5_full(h)
    max_cv = np.max(corr_vals)
    idx_mask = corr_vals >= max_cv - 1e-9
    if np.sum(idx_mask) > max_active:
        order = np.argsort(-corr_vals)[:max_active]
        idx_mask = np.zeros_like(corr_vals, dtype=bool)
        idx_mask[order] = True
    active_d = [k - (n - 1) for k in np.where(idx_mask)[0]]
    if not active_d:
        return None, c5
    A_rows, b_vals = [], []
    for d in active_d:
        g, grad = compute_g_grad(h, d)
        rhs = g - np.dot(grad, h)
        A_rows.append(np.append(grad, -1.0))
        b_vals.append(rhs)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, 1)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'maxiter': 10000})
    if not res.success:
        return None, c5
    return res.x[:n], c5

def run_bundle(h0, n, t_end, max_iter=800, stab=0.005):
    h = h0.copy()
    S = n / 2.0
    best_c5, best_h = compute_c5_full(h)[0], h.copy()
    stall = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        h_prop, c5_now = bundle_step(h, n, S)
        if h_prop is None:
            break
        if np.max(np.abs(h_prop - h)) < 1e-11:
            stall += 1
            if stall > 30:
                break
            continue
        c5_prop, _ = compute_c5_full(h_prop)
        if c5_prop < c5_now - 1e-12:
            alpha = 1.0
        else:
            best_a, best_v = 0.0, c5_now
            for a in [0.5, 0.3, 0.15, 0.05]:
                h_t = exact_project(h + a * (h_prop - h), S)
                v_t, _ = compute_c5_full(h_t)
                if v_t < best_v - 1e-14:
                    best_v, best_a = v_t, a
            if best_a == 0.0:
                stall += 1
                if stall > 40:
                    break
                h += stab * (np.random.rand(n) - 0.5)
                h = exact_project(h, S)
                continue
            alpha = best_a
        h = exact_project(h + alpha * (h_prop - h), S)
        c5_new, _ = compute_c5_full(h)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            h = exact_project(best_h + 0.02 * np.random.randn(n), S)
            stall = 0
    return best_h, best_c5

def coord_opt(h, n, i, S):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        c = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            if j2 != j1:
                c += (-h[j2])
            else:
                c += (-h[j2])
        coef[k + n - 1] = c

    def f1d(hi):
        return np.max(corr_base + hi * coef) * dx

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    S = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5_full(best_h)[0]
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i, S)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = exact_project(h_try, S)
            c5_try, _ = compute_c5_full(h_try)
            if c5_try < best_c5 - 5e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_smooth(h, T, n, S, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def con_fun(x):
        return x.sum() - S
    res = minimize_scalar
    from scipy.optimize import minimize
    r = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                 constraints=[{'type':'eq','fun':con_fun}],
                 options={'maxiter':maxiter, 'ftol':1e-15})
    return exact_project(r.x, S)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    S = n / 2.0
    best_h = exact_project(best_h, S)
    best_c5 = compute_c5_full(best_h)[0]

    def elapsed():
        return time.time() - t0

    # Phase 1: Bundle method (main optimizer)
    t1 = t0 + budget_s * 0.35
    bh, bc = run_bundle(best_h, n, t1, max_iter=2000)
    if bc < best_c5:
        best_c5, best_h = bc, bh

    # Phase 2: Coordinate descent polish
    t2 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=120)

    # Phase 3: Bundle method round 2
    t3 = t0 + budget_s * 0.72
    bh, bc = run_bundle(best_h, n, t3, max_iter=1000)
    if bc < best_c5:
        best_c5, best_h = bc, bh

    # Phase 4: More coordinate descent
    t4 = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=80)

    # Phase 5: Try perturbed restarts
    for restart in range(4):
        if elapsed() > budget_s * 0.93:
            break
        pert = best_h + 0.015 * np.random.randn(n)
        pert = exact_project(pert, S)
        t_sub = min(t0 + budget_s * 0.96, time.time() + (budget_s - elapsed()) / 5.0)
        bh, bc = run_bundle(pert, n, t_sub, max_iter=200)
        if bc < best_c5:
            best_c5, best_h = bc, bh
            pc, _ = compute_c5_full(exact_project(best_h + 0.005*np.random.randn(n), S))
            if pc < best_c5:
                best_c5, best_h = pc, exact_project(best_h + 0.005*np.random.randn(n), S)

    # Phase 6: Final SLSQP polish with high T
    if elapsed() < budget_s * 0.98:
        for T in [5000, 20000]:
            if elapsed() > budget_s * 0.99:
                break
            try:
                hc = slsqp_smooth(best_h, T, n, S, maxiter=600)
                c5v, _ = compute_c5_full(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    # Final exact coordinate descent
    if elapsed() < budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=30)

    best_h = exact_project(best_h, S)
    best_c5 = compute_c5_full(best_h)[0]
    return best_h, best_c5, len(best_h)
