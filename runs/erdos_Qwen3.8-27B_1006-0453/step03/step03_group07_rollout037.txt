```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def subgrad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    lo1, hi1 = max(0, lag), min(n, n + lag)
    lo2, hi2 = max(0, -lag), min(n, n - lag)
    if hi1 > lo1:
        grad[lo1:hi1] += 1.0 - h[lo1 - lag:hi1 - lag]
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 + lag:hi2 + lag]
    return grad

def subgrad_phase(h, n, max_iter, deadline):
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5
    momentum = np.zeros(n)
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = np.max(corr) * (2.0 / n)
        K = min(6 + it // 30, 20)
        top_idx = np.argsort(corr)[-K:]
        spread = np.max(corr[top_idx]) - np.min(corr[top_idx]) + 1e-10
        T = max(spread * 0.6, 2.0)
        w = np.exp((corr[top_idx] - np.max(corr[top_idx])) / T)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            lag = int(ki) - (n - 1)
            grad += wi * subgrad_lag(h, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        momentum = 0.55 * momentum - 0.25 * grad
        best_alpha = 0.005
        best_c5_ls = c5_now
        for alpha in [0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.01]:
            h_try = make_feasible(h + alpha * momentum, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5_ls:
                best_c5_ls = c5_try
                best_alpha = alpha
            elif c5_try > c5_now + 0.005:
                break
        h = make_feasible(h + best_alpha * momentum, n)
        if best_c5_ls < c5_best:
            c5_best = best_c5_ls
            h_best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 800:
            h = make_feasible(h + 0.04 * np.random.randn(n), n)
            no_improve = 0
    return h_best, c5_best

def periodic_obj(xp, n_total, period, T):
    h = np.tile(xp, n_total // period)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n_total)

def periodic_optimize(n_total, period, deadline, x0):
    def obj(xp, T=150):
        return periodic_obj(xp, n_total, period, T)
    bnds = [(0.0, 1.0)] * period
    res = minimize(obj, x0, method='Nelder-Mead',
                   options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
    h = np.tile(np.clip(res.x, 0, 1), n_total // period)
    h = make_feasible(h, n_total)
    return h, compute_c5(h)

def slsqp_polish(h, n, T, maxiter, deadline):
    def sm_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)
    def sc(x):
        return x.sum() - n / 2.0
    res = minimize(sm_obj, h.copy(), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': sc}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return make_feasible(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float).copy()
    else:
        best_h = np.zeros(100)
        best_h[0::2] = 1.0
        best_h = make_feasible(best_h, 100)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    # Phase 1: Subgradient (45% budget)
    dl1 = t0 + budget_s * 0.45
    h1, c5_1 = subgrad_phase(best_h, n, 200000, dl1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1
    # Phase 2: Periodic structure search
    dl2 = t0 + budget_s * 0.62
    for period in [4, 5, 6, 8, 10, 12, 15, 20]:
        if time.time() > dl2:
            break
        if n % period != 0:
            continue
        for trial in range(2):
            if time.time() > dl2:
                break
            if trial == 0:
                x0 = np.resize(best_h[:period * 2], period).astype(float)
                x0 = np.clip(x0, 0.1, 0.9)
            else:
                x0 = np.random.uniform(0.15, 0.85, period)
            try:
                h_p, c5_p = periodic_optimize(n, period, dl2, x0)
                if c5_p < best_c5:
                    best_c5, best_h = c5_p, h_p
            except Exception:
                pass
    # Phase 3: Subgradient from best (12%)
    dl3 = t0 + budget_s * 0.74
    h3, c5_3 = subgrad_phase(best_h, n, 200000, dl3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3
    # Phase 4: SLSQP polish with multiple T values
    dl4 = t0 + budget_s * 0.92
    for T_val in [300, 1000, 5000]:
        if time.time() > dl4:
            break
        try:
            h4 = slsqp_polish(best_h, n, T_val, 3000, dl4)
            c5_4 = compute_c5(h4)
            if c5_4 < best_c5:
                best_c5, best_h = c5_4, h4
        except Exception:
            pass
    # Phase 5: Multi-n exploration
    dl5 = t0 + budget_s * 0.97
    for n_try in [80, 120, 160]:
        if time.time() > dl5:
            break
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=n)
        h_interp = make_feasible(h_interp, n_try)
        h_e, c5_e = subgrad_phase(h_interp, n_try, 50000, dl5)
        if c5_e < best_c5:
            best_c5, best_h = c5_e, h_e
    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
