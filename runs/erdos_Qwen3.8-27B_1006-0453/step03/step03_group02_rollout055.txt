def compute_c5_fast(h, dx):
    """Compute C5 using only non-negative lags (symmetric)."""
    N = len(h)
    one_minus_h = 1.0 - h
    # Compute correlation for lags 0 to N-1
    # corr_full[N-1+L] = sum_{i=0}^{N-1-L} h[i+L] * (1-h[i])
    # We can compute this efficiently
    max_val = 0.0
    for L in range(N):
        if L == 0:
            val = np.dot(h, one_minus_h)
        else:
            val = np.dot(h[L:], one_minus_h[:N-L])
        if val > max_val:
            max_val = val
    return max_val * dx

def project_feasible(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

        else:

            h_r = project_feasible(h_r, target)
            hr_dead = time.time() + min((deadline - time.time()) * 0.5, 120)
            h4, c4 = optimize(h_r, N, target, hr_dead,
                              30, 500, 8, 1, 0.03, 0.001)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()

    # Final polish on the best solution found
    Nb = len(best_h)
    if time.time() < deadline - 6:
        target_b = Nb / 2.0
        hb, cb = optimize(best_h.copy(), Nb, target_b, deadline - 2,
                          200, 8000, 1, 1, 0.001, 0.00003)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Final micro local search
    if time.time() < deadline - 1:
        hn2, cn2 = local_search(best_h.copy(), len(best_h), len(best_h) // 2,
                                deadline, 300, np.random.RandomState(777))
        if cn2 < best_c5:
            best_c5, best_h = cn2, hn2

    return best_h, best_c5, len(best_h)
