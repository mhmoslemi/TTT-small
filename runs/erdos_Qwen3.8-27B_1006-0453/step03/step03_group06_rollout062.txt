def coord_descent_fast(h, n, n_sweeps=5):
    h = h.copy()
    corr = compute_corr(h)
    center = n - 1
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            v = h[j]
            # Compute ms for all shifts
            ms = np.zeros(2*n-1)
            for s in range(-(n-1), n):
                I1 = 1.0 if (0 <= j-s < n) else 0.0
                I2 = 1.0 if (0 <= j+s < n) else 0.0
                ms[s+center] = (1.0-h[j-s])*I1 - h[j+s]*I2
            cs = corr - v * ms
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi-lo)/3
                m2 = hi - (hi-lo)/3
                if np.max(ms*m1+cs) < np.max(ms*m2+cs):
                    hi = m2
                else:
                    lo = m1
            v_new = (lo+hi)/2
            dv = v_new - v
            if abs(dv) > 1e-12:
                h[j] = v_new
                corr += dv * ms
                improved = True
        h = project_feasible(h, n)
        # After projection, recompute corr to avoid drift
        if improved:
            corr = compute_corr(h)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / n)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t < 1e-15:
            return h
        h += room * (diff / t)
    else:
        room = h.copy()
        t = room.sum()
        if t < 1e-15:
            return h
        h += room * (diff / t)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_val(h, T):
    corr = compute_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / len(h))

def smooth_c5_grad(h, T):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    ex = np.exp(T * (corr - m))
    w = ex / ex.sum()
    center = n - 1
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        ws = w[s + center]
        if ws < 1e-15:
            continue
        if s >= 0:
            grad[s:] += ws * (1.0 - h[:n - s])
            grad[:n - s] -= ws * h[s:]
        else:
            ss = -s
            grad[:n - ss] += ws * (1.0 - h[ss:])
            grad[ss:] -= ws * h[:n - ss]
    return grad * (2.0 / n)

def slsqp_opt(h0, n, T, maxiter=400):
    res = minimize(smooth_c5_val, h0, jac=smooth_c5_grad, args=(T,),
                   method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, n_sweeps=5):
    h = h.copy()
    corr = compute_corr(h).copy()
    center = n - 1
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            v = h[j]
            ms = np.zeros(2 * n - 1)
            for s in range(-(n - 1), n):
                I1 = 1.0 if (0 <= j - s < n) else 0.0
                I2 = 1.0 if (0 <= j + s < n) else 0.0
                ms[s + center] = (1.0 - h[j - s]) * I1 - h[j + s] * I2
            cs = corr - v * ms
            lo, hi = 0.0, 1.0
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(ms * m1 + cs) < np.max(ms * m2 + cs):
                    hi = m2
                else:
                    lo = m1
            v_new = (lo + hi) / 2.0
            dv = v_new - v
            if abs(dv) > 1e-13:
                h[j] = v_new
                corr += dv * ms
                improved = True
        h = project_feasible(h, n)
        if improved:
            corr = compute_corr(h).copy()
        else:
            break
    return h

def pair_local_search(h, n, n_trials=5000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    h = h.copy()
    best = compute_c5(h)
    for _ in range(n_trials):
        i = rng.integers(n)

            h[j] += amt
            c5 = compute_c5(h)
            if c5 < best - 1e-15:
                best = c5
                break
            h[i] += amt
            h[j] -= amt
    return h, best
