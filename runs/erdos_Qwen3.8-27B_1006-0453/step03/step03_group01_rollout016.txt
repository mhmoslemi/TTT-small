def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_grad(h, N, corr, topK, tau):
    center = N - 1
    sc = tau * corr - tau * corr.max()

        else:

            g[M:] -= wi * h[:N - M]
