    # Phase 6: Final CD + SLSQP polish (13%)
    t6 = t0 + budget_s * 0.96
    best_h, best_c5 = cd_loop(best_h, n, t6, max_sweeps=1000)

    if time.time() < t0 + budget_s * 0.99:
        for T in [5000, 20000]:
            if time.time() > t0 + budget_s * 0.99:
                break
            try:
                hn = slsqp_polish(best_h, T, n, maxiter=800)
                cn = compute_c5(hn)
                if cn < best_c5:
                    best_h, best_c5 = hn, cn
            except Exception:
                pass
        best_h, best_c5 = cd_loop(best_h, n, t0 + budget_s * 0.995, max_sweeps=200)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
