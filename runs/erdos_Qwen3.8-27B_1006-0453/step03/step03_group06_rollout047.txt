import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def _sum_con(x):
    return x.sum() - len(x) / 2.0

def _smooth_obj(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k is not None and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        tv = corr[idx]
    else:
        tv = corr
    m = np.max(tv)
    return (m + np.log(np.sum(np.exp(T * (tv - m)))) / T) * dx

def slsqp_opt(h0, T, k, maxiter=500):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con}]
    try:
        res = minimize(_smooth_obj, h0, args=(T, k), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def sg_step(h, lr, k_top=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argpartition(corr, -k_top)[-k_top:]
    grad = np.zeros(n)
    for m in top:
        s = m - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 - s:hi2 - s]
    gmax = np.max(np.abs(grad))
    if gmax < 1e-14:
        return h.copy()
    h_new = h - lr * grad / gmax
    return project_feasible(h_new, n)

def coord_descent(h, n_passes=5, deltas=None):
    n = len(h)
    best_c5 = compute_c5(h)
    if deltas is None:
        deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            for d in deltas:
                h_try = h.copy()
                h_try[j] += d
                h_try = project_feasible(h_try, n)
                c5v = compute_c5(h_try)
                if c5v < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5v
                    improved = True
                    break
            if not improved:
                break
        if not improved:
            break
    return h, best_c5

def multi_slq(h0, n, t_list, k_list, maxiter, time_limit):
    best = h0.copy()
    best_c5 = compute_c5(best)
    for T in t_list:
        for k in k_list:
            if time.time() > time_limit:
                return best, best_c5
            hc = slsqp_opt(best, T, k, maxiter)
            c5v = compute_c5(hc)
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                best = hc
    return best, best_c5
