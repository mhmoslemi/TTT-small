def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_c5_val(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and 0 < topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-12:
            return h
        h = h * (target / tot)
    return np.clip(h, 0.0, 1.0)

    v = smooth_c5_val(x, T, topk)
    s = x.sum()
    return v + pw * (s - float(n) / 2.0) ** 2

        corr = corr[idx]
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-12:
            return h
        h = h * (target / tot)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_c5_val(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and 0 < topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-12:
            return h
        h = h * (target / tot)
    return np.clip(h, 0.0, 1.0)


def obj_lbf(x, T, topk, pw, n):
    v = smooth_c5_val(x, T, topk)
    s = x.sum()
    return v + pw * (s - float(n) / 2.0) ** 2


def lbfq_refine(h0, n, T, topk, maxiter):
    bnds = [(0.0, 1.0)] * n
    best = h0.copy()
    best_c5 = compute_c5(best)
    pw = 50.0
    for outer in range(100):
        try:
            res = minimize(obj_lbf, best, args=(T, topk, pw, n),
                           method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': maxiter, 'ftol': 1e-15,
                                    'gtol': 1e-6, 'maxls': 50})
            hnew = project_feasible(res.x, n)
            c5n = compute_c5(hnew)
            if c5n < best_c5 - 1e-15:
                old = best_c5
                best = hnew
                best_c5 = c5n
                pw = max(pw * 0.15, 0.5)
                if old - c5n < 1e-14:
                    break
            else:
                pw *= 4.0
                if pw > 1e10 or (res.nit <= 3 and pw > 200):
                    break
        except Exception:
            break
    return best, best_c5


def sum_eq(x, n):
    return x.sum() - float(n) / 2.0


def sm_obj(x, T, topk):
    return smooth_c5_val(x, T, topk)


def slsqp_refine(h0, n, T, topk, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    res = minimize(sm_obj, h0, args=(T, topk),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    h = project_feasible(res.x, n)
    return h, compute_c5(h)


def coord_descent(h, n, n_sweeps, d0):
    best_c5 = compute_c5(h)
    d = d0
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dd in (d, -d, d * 0.25, -d * 0.25):
                h2 = h.copy()
                h2[j] += dd
                h2 = project_feasible(h2, n)
                c52 = compute_c5(h2)
                if c52 < best_c5 - 1e-14:
                    h = h2
                    best_c5 = c52
                    improved = True
                    break
        if not improved:
            d *= 0.4
            if d < 1e-8:
                break
    return h, best_c5


def flip_search(h, n):
    best_c5 = compute_c5(h)
    for _ in range(5):
