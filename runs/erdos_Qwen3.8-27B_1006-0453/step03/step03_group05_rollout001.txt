        best_h = make_step_fn(150, 0.5)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Annealed L-BFGS-B on initial (5% budget)
    for T in [100, 300, 1000, 3000, 10000]:
        if time.time() - t0 > budget_s * 0.05:
            break
        hc = fast_local_ld(best_h, T, 250, 10.0)
        cc = compute_c5(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 2: Basin hopping with diverse perturbations (40% budget)
    elapsed_target_2 = budget_s * 0.45
    trial = 0
    while time.time() - t0 < elapsed_target_2:
        r = rng.random()
        if r < 0.25:
            hp = perturb_shift(best_h, best_n, rng)
        elif r < 0.45:
            sigma = 0.10 * (1.0 - min(trial / 60.0, 1.0)) + 0.02
            hp = perturb_noise(best_h, best_n, sigma, rng)
        elif r < 0.60:
            hp = perturb_dipole(best_h, best_n, rng, int(rng.integers(3, 8)))
        elif r < 0.75:
            hp = perturb_flip(best_h, best_n, rng)
        else:
            # Larger jump
            hp = project_feasible(best_h + rng.standard_normal(best_n) * 0.2, best_n)
        T_local = int(500 + 2000 * rng.random())
        hc = fast_local_ld(hp, T_local, 200, 5.0)
        cc = compute_c5(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc
        trial += 1

    # Phase 3: Multi-resolution search (15% budget)
    for nr in [120, 180, 250, 350]:
        if time.time() - t0 > budget_s * 0.72:
            break
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, nr, endpoint=False)
        hu = np.interp(x_new, x_old, best_h, period=2.0)
        hu = project_feasible(hu, nr)
        hu = project_feasible(hu + rng.standard_normal(nr) * 0.04, nr)
        hh = fast_local_ld(hu, 800, 300, 8.0)
        ch = compute_c5(hh)
        # Also try topk objective at this resolution
        hh2 = fast_local_topk(hu, 3, 250, 5.0)
        ch2 = compute_c5(hh2)
        if min(ch, ch2) < best_c5:
            best_c5, best_h, best_n = min(ch, ch2), (hh if ch <= ch2 else hh2), nr

    # Phase 4: Subgradient + local refinement (12% budget)
    if time.time() - t0 < budget_s * 0.84:
        for _ in range(8):
            if time.time() - t0 > budget_s * 0.84:
                break
            hs = best_h.copy()
            for _ in range(50):
                hs = subgrad_step(hs, 4, 0.008)
            cc = compute_c5(hs)
            if cc < best_c5:
                best_c5, best_h = cc, hs
            hp = fast_local_ld(best_h, 5000, 300, 20.0)
            cc = compute_c5(hp)
            if cc < best_c5:
                best_c5, best_h = cc, hp

    # Phase 5: Different starting points to escape (8% budget)
    if time.time() - t0 < budget_s * 0.92:
        for frac in [0.4, 0.55, 0.45]:
            if time.time() - t0 > budget_s * 0.92:
                break
            hstart = make_step_fn(best_n, frac)
            hstart = project_feasible(hstart + rng.standard_normal(best_n) * 0.3, best_n)
            ho = fast_local_ld(hstart, 1000, 300, 5.0)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho

    # Phase 6: Final intensive polish (8% budget)
    if time.time() - t0 < budget_s * 0.97:
        for T in [5000, 20000, 500]:
            if time.time() - t0 > budget_s * 0.97:
                break
            hc = fast_local_ld(best_h, T, 500, 50.0)
            cc = compute_c5(hc)
            if cc < best_c5:
                best_c5, best_h = cc, hc
        hc = fast_local_topk(best_h, 3, 400, 5.0)
        cc = compute_c5(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
