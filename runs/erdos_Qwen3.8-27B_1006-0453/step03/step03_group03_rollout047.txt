            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500, ftol=1e-16):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def active_set_obj(x, top_idx, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    return np.max(c[top_idx])

def active_set_grad(x, top_idx, n, tol=1e-8):
    c = np.correlate(x, 1.0 - x, mode='full')
    vals = c[top_idx]
    m = np.max(vals)
    act = top_idx[vals >= m - tol]
    g = np.zeros(n)
    for ii in act:
        k = ii - (n - 1)
        if k >= 0:
            g[:n-k] += (1.0 - x[k:])
            g[k:] += -x[:n-k]
        else:
            kp = -k
            g[kp:] += (1.0 - x[:n-kp])
            g[:n-kp] += -x[kp:]
    g /= len(act)
    return g

def active_set_solve(h, top_k=12, tol=1e-9, maxiter=600):
    n = len(h)
    _, corr = compute_c5(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(active_set_obj, h, args=(top_idx, n),
                      jac=lambda x: active_set_grad(x, top_idx, n, tol=tol),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def expand_active(h, base_k=8, margin=0.02):
    n = len(h)
    _, corr = compute_c5(h)
    m = np.max(corr)
    thresh = m * (1.0 - margin)
    idx = np.where(corr >= thresh)[0]
    return idx[:base_k + 30]

def level_bisect(h, lo, hi, tol_c5=2e-6, max_iter=15):
    n = len(h)
    best = h.copy()
    for _ in range(max_iter):
        mid = 0.5 * (lo + hi)
        target = mid / (2.0 / n)
        _, corr = compute_c5(h)
        top_idx = np.where(corr >= target * 0.95)[0]
        if len(top_idx) == 0:
            top_idx = np.argsort(corr)[::-1][:15]
        def cons_fun(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return c[top_idx] - target
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq}]
        for i in range(min(len(top_idx), 20)):
            cons.append({'type': 'ineq', 'fun': lambda x, idx=top_idx[i]: target - np.correlate(x, 1.0-x, mode='full')[idx]})
        try:
            res = minimize(lambda x: x.sum()*0 + 0, h, method='SLSQP',
                          bounds=bnds, constraints=cons, options={'maxiter': 200})
            h_try = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            h_try = h
        c5_try, _ = compute_c5(h_try)
        if c5_try <= mid:
            hi = mid
            best = h_try
        else:
            lo = c5_try
    return best

def fourier_start(n, seed_val=0):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2, n)
    h = 0.5 * np.ones(n)
    for m in range(1, 6):
        amp = rng.uniform(0.05, 0.35)
        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.sin(m * np.pi * x + phase)
    return project_feasible(h, n)

def block_start(n, seed_val=0):
    rng = np.random.RandomState(seed_val)
    nblocks = rng.randint(4, 12)
    bs = n // nblocks
    h = np.zeros(n)
    for i in range(nblocks):
        s = i * bs
        e = min(s + bs + (1 if i < n % nblocks else 0), n)
        h[s:e] = rng.choice([0.0, 1.0], p=[0.4, 0.6])
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(80)
        best_h[:40] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Escalating T SLSQP from initial
    for T in [800, 4000, 20000, 80000, 300000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=500)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Active-set refinement (multiple sizes)
    if time.time() - t0 < budget_s * 0.22:
        for tk in [8, 12, 18, 25]:
            if time.time() - t0 > budget_s * 0.22:
                break
            h_try = active_set_solve(best_h, top_k=tk, maxiter=700)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
            h_try = slsqp_smooth(best_h, 100000, maxiter=400)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 3: Multi-resolution multi-start
    n_candidates = sorted(set([best_n, 64, 70, 72, 74, 76, 80, 82, 84, 86, 90, 96, 100, 104, 108]))
    for n_try in n_candidates:
        if time.time() - t0 > budget_s * 0.60:
            break
        if n_try % 2 != 0 or n_try < 40:
            continue
        starts = []
        if n_try == best_n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            starts.append(project_feasible(np.interp(x_new, x_old, best_h), n_try))
        for s in range(4):
            starts.append(fourier_start(n_try, seed + n_try * 10 + s))
        for s in range(3):
            starts.append(block_start(n_try, seed + n_try * 20 + s))
        if n_try != best_n:
            h_rot_src = np.interp(np.linspace(0, 2, n_try), np.linspace(0, 2, best_n), best_h)
            for shift in [n_try // 4, n_try // 3]:
                starts.append(project_feasible(np.roll(h_rot_src, shift), n_try))
        for h_start in starts:
            if time.time() - t0 > budget_s * 0.60:
                break
            h_opt = slsqp_smooth(h_start, 8000, maxiter=350)
            h_opt = active_set_solve(h_opt, top_k=10, maxiter=400)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try

    # Phase 4: Active-set + smooth refinement loop
    if time.time() - t0 < budget_s * 0.80:
        for _ in range(6):
            if time.time() - t0 > budget_s * 0.80:
                break
            idx_exp = expand_active(best_h, base_k=10, margin=0.015)
            h_try = active_set_solve(best_h, top_k=len(idx_exp), tol=1e-10, maxiter=800)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
            h_try = slsqp_smooth(best_h, 200000, maxiter=500)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 5: Small perturbations + refinement
    if time.time() - t0 < budget_s * 0.90:
        for scale in [0.02, 0.01, 0.005, 0.003]:
            if time.time() - t0 > budget_s * 0.90:
                break
            for trial in range(4):
                if time.time() - t0 > budget_s * 0.90:
                    break
                h_p = best_h + np.random.randn(best_n) * scale
                h_p = project_feasible(h_p, best_n)
                h_opt = active_set_solve(h_p, top_k=12, maxiter=500)
                h_opt = slsqp_smooth(h_opt, 400000, maxiter=400)
                c5_opt, _ = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Phase 6: Final ultra-high-T polish
    if time.time() - t0 < budget_s * 0.97:
        for T in [500000, 1000000, 2000000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            h_try = slsqp_smooth(best_h, T, maxiter=400, ftol=1e-18)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        h_try = active_set_solve(best_h, top_k=15, tol=1e-12, maxiter=600)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
