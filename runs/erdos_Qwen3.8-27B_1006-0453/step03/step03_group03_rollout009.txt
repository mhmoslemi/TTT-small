def binary_local_search(h, n, time_limit=None, t0=None):
    # Convert to binary
    order = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[order[n//2:]] = 1.0
    best_c5, _ = compute_c5(h_bin)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    stall = 0
    for it in range(5000):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        best_flip = None
        best_val = best_c5
        # Try flipping one 1 to 0 and one 0 to 1
        # To save time, only try a subset
        for oi in ones:
            for zi in zeros:
                if oi == zi:
                    continue
                h_bin[oi] = 0.0
                h_bin[zi] = 1.0
                c5, _ = compute_c5(h_bin)
                h_bin[oi] = 1.0
                h_bin[zi] = 0.0
                if c5 < best_val - 1e-12:
                    best_val = c5
                    best_flip = (oi, zi)
        if best_flip is not None:
            h_bin[best_flip[0]] = 0.0
            h_bin[best_flip[1]] = 1.0
            best_c5 = best_val
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return h_bin, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            total = room.sum()
            if total < 1e-14:
                break
            h[mask] += room[mask] * (diff / total)
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            room = np.where(mask, h, 0.0)
            total = room.sum()
            if total < 1e-14:
                break
            h[mask] -= room[mask] * (-diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    sw = exps.sum()
    weights = exps / sw
    obj = (m + np.log(sw) / T) * dx
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        k = j - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * (1.0 - h[k:n])
            grad[k:] += w * (-h[:n - k])
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n - kk])
            grad[:n - kk] += w * (-h[kk:])
    grad *= dx
    return obj, grad

def lbfgs_optimize(h, T, penalty, maxiter, time_limit=None, t0=None):
    n = len(h)
    def obj_grad(x):
        xc = np.clip(x, 0, 1)
        f, g = smooth_c5_and_grad(xc, T)
        s = xc.sum() - n / 2.0
        f += penalty * s * s
        g = g + 2.0 * penalty * s
        return f, g
    try:
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': 5000})
        out = project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        out = h
    return out

def binary_local_search(h_init, n, max_iters=300, time_limit=None, t0=None):
    order = np.argsort(h_init)
    hb = np.zeros(n)
    hb[order[n // 2:]] = 1.0
    best_c5, _ = compute_c5(hb)
    ones_list = list(np.where(hb == 1.0)[0])
    zeros_list = list(np.where(hb == 0.0)[0])
    rng = np.random.RandomState(0)
    stall = 0
    for it in range(max_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        improved = False
        # Try a random subset of flips
        n_try = min(len(ones_list) * len(zeros_list), 200)
        pairs = []
        for _ in range(n_try):
            oi = ones_list[rng.randint(len(ones_list))]
            zi = zeros_list[rng.randint(len(zeros_list))]
            if oi != zi:
                pairs.append((oi, zi))
        best_flip = None
        best_val = best_c5
        for oi, zi in pairs:
            hb[oi] = 0.0
            hb[zi] = 1.0
            c5, _ = compute_c5(hb)
            hb[oi] = 1.0
            hb[zi] = 0.0
            if c5 < best_val - 1e-12:
                best_val = c5
                best_flip = (oi, zi)
        if best_flip is not None:
            hb[best_flip[0]] = 0.0
            hb[best_flip[1]] = 1.0
            best_c5 = best_val
            ones_list = list(np.where(hb == 1.0)[0])
            zeros_list = list(np.where(hb == 0.0)[0])
            stall = 0
            improved = True
        else:
            stall += 1
            if stall >= 5:
                break
    return hb, best_c5

    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: L-BFGS-B with T annealing on parent
    T_list = [200, 800, 3000, 12000, 50000, 200000]
    for T in T_list:
        if time.time() - t0 > budget_s * 0.25:
            break
        h_try = lbfgs_optimize(best_h, T, penalty=500.0, maxiter=300)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Binary local search
    if time.time() - t0 < budget_s * 0.45:
        tl = budget_s * 0.45 - (time.time() - t0)
        hb, c5b = binary_local_search(best_h, best_n, max_iters=400,
                                      time_limit=tl, t0=t0)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb
        # Refine binary solution with L-BFGS-B
        if time.time() - t0 < budget_s * 0.55:
            for T in [2000, 10000]:
                h_try = lbfgs_optimize(best_h, T, penalty=500.0, maxiter=400)
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try

    # Phase 3: Explore nearby n values
    n_candidates = sorted(set([best_n - 2, best_n, best_n + 2, best_n + 4,
                                72, 76, 80, 84, 88, 92, 96]))
    for n_try in n_candidates:
        if time.time() - t0 > budget_s * 0.75:
            break
        if n_try < 40 or n_try % 2 != 0:
            continue
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_feasible(h_base, n_try)
        # Binary init for this n
        order = np.argsort(h_base)
        h_bin = np.zeros(n_try)
        h_bin[order[n_try // 2:]] = 1.0
        # Quick binary local search
        if time.time() - t0 < budget_s * 0.70:
            hb, c5b = binary_local_search(h_bin, n_try, max_iters=100,
                                          time_limit=budget_s * 0.70, t0=t0)
            h_base = hb
        # L-BFGS-B refine
        if time.time() - t0 < budget_s * 0.75:
            h_try = lbfgs_optimize(h_base, 5000, penalty=500.0, maxiter=300)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                best_n = n_try

    # Phase 4: Multi-start with perturbations at best_n
    if time.time() - t0 < budget_s * 0.90:
        for trial in range(8):
            if time.time() - t0 > budget_s * 0.90:
                break
            scale = 0.03 + 0.05 * rng.rand()
            h_p = best_h + rng.randn(best_n) * scale
            h_p = project_feasible(h_p, best_n)
            # Binary snap then refine
            if trial % 2 == 0:
                order = np.argsort(h_p)
                h_bin = np.zeros(best_n)
                h_bin[order[best_n // 2:]] = 1.0
                h_p = h_bin
            h_try = lbfgs_optimize(h_p, 5000, penalty=500.0, maxiter=300)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 5: Final high-T polish with very tight tolerance
    if time.time() - t0 < budget_s * 0.97:
        for T in [100000, 500000]:
            if time.time() - t0 > budget_s * 0.97:

            h_try = lbfgs_optimize(best_h, T, penalty=1000.0, maxiter=500)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:

                best_h = h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
