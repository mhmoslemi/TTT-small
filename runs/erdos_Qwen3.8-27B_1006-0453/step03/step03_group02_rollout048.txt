import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    N = len(h)
    shift = diff / N
    hs = h - shift
    if np.all(hs >= -1e-12) and np.all(hs <= 1.0 + 1e-12):
        return np.clip(hs, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        sm = float(np.clip(h - mid, 0.0, 1.0).sum())
        if sm > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_grad(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    center = N - 1
    if K >= len(corr):
        idx = np.arange(len(corr))
        cv = corr
    else:
        idx = np.argpartition(corr, -K)[-K:]
        cv = corr[idx]
    sc = tau * cv
    sc -= sc.max()
    w = np.exp(sc)
    sw = w.sum()
    if sw < 1e-30:
        return np.zeros(N), c5
    w /= sw
    grad = np.zeros(N)
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-10:
            continue
        k = int(idx[ix])
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            nv = N - L
            grad[L:] += wi * (1.0 - h[:nv])
            grad[:nv] -= wi * h[L:]
        else:
            M = -L
            nv = N - M
            grad[:nv] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:nv]
    return grad, c5

def opt_phase(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K0, K1, iters):
    best_h = h.copy()
    best_c = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.95, 1e-8
    it = 0
    while time.time() < deadline:
        fr = min((it + 1) / max(iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        K = max(1, int(round(K0 + (K1 - K0) * fr)))
        g, cv = get_grad(h, N, tau, K)
        if cv < best_c - 1e-14:
            best_c = cv
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c

def pair_polish(h, N, target, dx, deadline, n_trials, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, dx)
    dvals = np.array([0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08])
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.randint(N)
        j = rng.randint(N)
        if j == i:
            j = (j + 3) % N
        d = dvals[rng.randint(len(dvals))]
        sgn = 1.0 if rng.randint(2) == 0 else -1.0
        trial = best_h.copy()
        trial[i] += sgn * d
        trial[j] -= sgn * d
        if trial[i] < -1e-9 or trial[i] > 1.0 + 1e-9 or trial[j] < -1e-9 or trial[j] > 1.0 + 1e-9:
            continue
        c = compute_c5(trial, dx)
        if c < best_c - 5e-15:
            best_c = c
            best_h = trial
    return best_h, best_c

def block_shift_polish(h, N, target, dx, deadline, n_trials, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, dx)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        length = rng.randint(3, 12)
        start = rng.randint(0, N - length)
        amount = rng.uniform(-0.04, 0.04)
        seg = trial[start:start + length].copy()
        trial[start:start + length] = seg + amount
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c - 5e-15:
            best_c = c
            best_h = trial
    return best_h, best_c
