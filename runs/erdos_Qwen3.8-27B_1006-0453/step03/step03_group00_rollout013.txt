            old = best_h[i]

            # The function f(v) = max_k (corr_base[k] + delta_k(v)) * dx
            # is the max of linear functions (for k≠0) and one concave quadratic (k=0)
            # on [0,1], so it's unimodal-ish. Use bounded minimization.
            
            hi = old  # store old
            best_v = old
            best_f = best_c5

            # Actually let's just use the analytic approach for linear parts
            # and check the quadratic for k=0

            lags = np.arange(-(n-1), n)

            idx2 = i + lags

            valid2 = (idx2 >= 0) & (idx2 < n)
            coeff[valid1] += (1.0 - best_h[idx1[valid1]])
            coeff[valid2] += (-best_h[idx2[valid2]])

            # Actually for the linear approximation at lag 0: coeff[0] = (1-h[i]) + (-h[i]) = 1-2*h[i]

                return np.max(c_new)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        tot = room.sum()

            return h
        h = h + room * (diff / tot)

        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.91, max_sweeps=200)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 7: Final SLSQP + coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=400)
    if elapsed() < budget_s * 0.97:
        for T in [8000, 30000]:
            if elapsed() > budget_s * 0.97:
                break
            try:
                hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
