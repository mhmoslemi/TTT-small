import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s < target:
        room = 1.0 - h
        tr = room.sum()
        if tr < 1e-13:
            return h
        h = h + room * ((target - s) / tr)
    else:
        th = h.sum()
        if th < 1e-13:
            return h
        h = h * (1.0 - (s - target) / th)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = T * e / se  # weights, sum to 1
    val = (m + np.log(se) / T) * dx
    # grad
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wi = w[n-1+k]
        if wi < 1e-15:
            continue
        if k >= 0:
            # (1-h[i-k]) for i in [k, n-1]
            grad[k:] += wi * (1.0 - h[:-k] if k > 0 else 1.0 - h)
            # -h[i+k] for i in [0, n-1-k]
            grad[:n-k] -= wi * (h[k:] if k > 0 else h)
        else:
            m2 = -k
            # (1-h[i-k]) = (1-h[i+m2]) for i in [0, n-1-m2]
            grad[:n-m2] += wi * (1.0 - h[m2:])
            # -h[i+k] = -h[i-m2] for i in [m2, n-1]
            grad[m2:] -= wi * h[:n-m2]
    return val, grad * dx

def slsqp_opt(h, T, maxiter=500, ftol=1e-14):
    n = len(h)
    def obj(x):
        return smooth_c5_grad(x, T)[0]
    def grad(x):
        return smooth_c5_grad(x, T)[1]
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1))

def coord_opt(h, i, n):
    """Optimize h[i] exactly given other coords fixed."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        c = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        # When k=0: j1=j2=i, so we get (1-h[i]) + (-h[i]) = 1 - 2*h[i]
        # That's correct since the term is h[i](1-h[i]) and d/dh[i] = 1-2h[i]
        # Wait, corr[k=0] = sum_j h[j](1-h[j]). d/dh[i] = 1-2h[i].
        # My formula: j1=i, c += 1-h[i]; j2=i, c += -h[i]. Total: 1-2h[i]. ✓
        coef[k + n - 1] = c
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coef) * dx

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tr = room.sum()
        if tr < 1e-14:
            return h
        h += room * ((target - s) / tr)
    else:
        th = h.sum()
        if th < 1e-14:
            return np.full(n, 0.5)
        h *= (1.0 - (s - target) / th)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = T * e / se
    val = (m + np.log(se) / T) * dx
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wi = w[n-1+k]
        if wi < 1e-15:
            continue
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        else:
            mk = -k
            grad[:n-mk] += wi * (1.0 - h[mk:])
            grad[mk:] -= wi * h[:n-mk]
    return val, grad * dx

def slsqp_opt(h, T, maxiter=500, ftol=1e-14):
    n = len(h)
    def obj(x):
        return smooth_c5_and_grad(x, T)[0]
    def grad(x):
        return smooth_c5_and_grad(x, T)[1]
    def con_fun(x):
        return x.sum() - n / 2.0
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: ones}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1))

def coord_opt(h, i):
    n = len(h)
    h_base = h.copy()
    hi = h[i]
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        c = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            c += 1.0 - h[j1]
        if 0 <= j2 < n:
            c -= h[j2]
        coef[k + n - 1] = c
    dx = 2.0 / n
    def f1d(t):
        return np.max(corr_base + t * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return float(res.x)

def coord_descent_polish(h, t_end, n_sweeps=500):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    stall = 0
    for sweep in range(n_sweeps):
        if time.time() > t_end:

        imp = False

            t = coord_opt(best, i)
            if abs(t - best[i]) > 1e-12:

                best = project(best)
                imp = True
        if imp:
            c5 = compute_c5(best)
            if c5 < best_c5:
                best_c5 = c5
            stall = 0
        else:
            stall += 1
    return best, best_c5

def perturb_block(h, size, mag):
    n = len(h)
    idx = np.random.randint(0, max(1, n - size))
    h[idx:idx+size] += np.random.randn(size) * mag
    return project(h)

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tr = room.sum()
        if tr < 1e-14:
            return h
        h += room * ((target - s) / tr)
    else:
        th = h.sum()
        if th < 1e-14:
            return np.full(n, 0.5)
        h *= (1.0 - (s - target) / th)
    return np.clip(h, 0.0, 1.0)

def sc5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = T * e / se
    val = (m + np.log(se) / T) * dx
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        if k >= 0:
            g[k:] += wk * (1.0 - h[:n - k])
            g[:n - k] -= wk * h[k:]
        else:
            mk = -k
            g[:n - mk] += wk * (1.0 - h[mk:])
            g[mk:] -= wk * h[:n - mk]
    return val, g * dx

def slsqp(h, T, mi=500):
    n = len(h)
    def f(x): return sc5_grad(x, T)[0]
    def j(x): return sc5_grad(x, T)[1]
    ones = np.ones(n)
    res = minimize(f, h, jac=j, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                 'jac': lambda x: ones}],
                   options={'maxiter': mi, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1))

def copt(h, i):
    n = len(h)
    hb = h.copy()
    hb[i] = 0.0
    cb = np.correlate(hb, 1.0 - hb, mode='full')
    cf = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        c = 0.0
        j1, j2 = i + k, i - k
        if 0 <= j1 < n:
            c += 1.0 - h[j1]
        if 0 <= j2 < n:
            c -= h[j2]
        cf[k + n - 1] = c
    dx = 2.0 / n
    def f1(t):
        return np.max(cb + t * cf) * dx
    return float(minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12}).x)

def cd_solve(h, t_end, ns=500):
    n = len(h)
    best = h.copy()
    bc5 = compute_c5(best)
    for _ in range(ns):
        if time.time() > t_end:
            break
        for i in np.random.permutation(n):
            t = copt(best, i)
            best[i] = t
            best = project(best)
        c5 = compute_c5(best)
        if c5 < bc5:
            bc5 = c5
    return best, bc5

def perturb(h, sz, mg):
    n = len(h)
    st = np.random.randint(0, max(1, n - sz))
    h[st:st + sz] += np.random.randn(sz) * mg
    return project(h)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)

    n = len(best_h)
    best_h = project(best_h)
    best_c5 = compute_c5(best_h)

    def el():
        return time.time() - t0

        if bc < best_c5:

        xo = np.linspace(0, 2, n, endpoint=False)
