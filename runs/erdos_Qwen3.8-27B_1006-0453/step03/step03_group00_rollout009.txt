    """Subgradient step focusing on active constraints."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    active = np.where(corr > m - 1e-4)[0]  # near-active constraints
    # Subgradient of max: average of gradients of active constraints
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + k < n:
                g += 1.0 - h[i+k]
            if 0 <= i - k < n:
                g -= h[i-k]
            grad[i] += g
    grad /= len(active)
    h_new = h - step_size * grad
    return project(h_new, n)

        if time.time() > t0 + budget_s * 0.94:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=2000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 7: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t_end - 3, max_sweeps=200)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
