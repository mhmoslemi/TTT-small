```python
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Proportional projection
    if s > 0:
        scale = target / s
        h_new = h * scale
        h_new = np.clip(h_new, 0.0, 1.0)
        # If clipping caused issues, try additive
        if abs(h_new.sum() - target) > 1e-8:
            diff = target - h_new.sum()
            if diff > 0:
                room = (1.0 - h_new)
                tot = room.sum()
                if tot > 1e-12:
                    h_new += room * (diff / tot)
            else:
                tot = h_new.sum()
                if tot > 1e-12:
                    h_new *= (1.0 + diff / tot)
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    h = np.full(n, 0.5)
    h[:int(target)] = 1.0  # approximate
    return project(h, n)

    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = (1.0 - h).sum()
            if room < 1e-14: break
            h += (1.0 - h) * (diff / room)
        else:
            if s < 1e-14: break
            h *= 1.0 + diff / s
        h = np.clip(h, 0.0, 1.0)
    return h

                   options={'maxiter': maxiter, 'ftol': ftol})

    ells = np.arange(-(n-1), n)

    # Second part: i + ell in [0, n)  
    iq = i + ells

def coord_descent(h, n, t_end, seed=42, max_sweeps=500):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    patience = 0
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        any_change = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coef = make_coefs(best_h, i, n)
            # 1D subproblem: min_{t in [0,1]} max_k (corr[k] + t * coef[k])
            res = minimize_scalar(
                lambda t: np.max(corr + t * coef),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11}
            )
            t_new = res.x
            if abs(t_new - best_h[i]) > 1e-11:
                delta = t_new - best_h[i]
                corr += delta * coef
                best_h[i] = t_new
                any_change = True
        if any_change:
            # Project and recompute
            best_h = project(best_h, n)
            corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
            c5_now = float(np.max(corr) * dx)
            if c5_now < best_c5:
                best_c5 = c5_now
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
        if patience >= 5:
            break
    return best_h, best_c5
```

One more optimization: in `cdesc`, instead of calling `minimize_scalar` (which does ~10+ function evaluations), I can implement a bisection on the subgradient directly. The function f(t) = max_k (corr[k] + t*coef[k]) is convex piecewise linear. Its subgradient at t is the set of slopes of the active (maximizing) pieces at t. The minimum is where 0 is in the subgradient.

The subgradient can be computed as:
- Find all k where corr[k] + t*coef[k] = f(t) (the active set)
- The subgradient is [min(coef[k] for k in active), max(coef[k] for k in active)]
- If 0 is in this range, t is optimal
- If all active coef > 0, decrease t
- If all active coef < 0, increase t

This gives a bisection-like method in O(n) per iteration, converging in O(log(1/ε)) iterations. Total: O(n log(1/ε)) per coordinate, vs O(n * 100) for minimize_scalar. About 10x faster.

```python
        vals = corr + mid * coef
        mx = np.max(vals)
        active = vals >= mx - 1e-15
        slopes = coef[active]
        if np.min(slopes) <= 0 and np.max(slopes) >= 0:
            break
        elif np.min(slopes) > 0:
            hi = mid
```

```python
def f1d_min(corr, coef):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * coef
        mx = float(np.max(vals))
        active = vals >= mx - 1e-14
        smin = float(np.min(coef[active]))
        smax = float(np.max(coef[active]))
        if smin <= 0.0 and smax >= 0.0:
            break
        if smin > 0.0:
            hi = mid
        else:
            lo = mid
    t = 0.5 * (lo + hi)
    f0 = float(np.max(corr))
    f1 = float(np.max(corr + coef))
    ft = float(np.max(corr + t * coef))
    if f0 <= ft and f0 <= f1: return 0.0
    if f1 <= ft and f1 <= f0: return 1.0
    return t
```

This is O(n * 50) per coordinate, vs O(n * 100) for minimize_scalar. About 2x faster. And it avoids the overhead of scipy's minimize_scalar.

```python
def cdesc(h, n, te, sd=0, ms=300):
    rng = np.random.RandomState(sd)
    bh = h.copy()
    co = np.correlate(bh, 1.0 - bh, mode='full').astype(float)
    dx = 2.0/n; bv = float(np.max(co)*dx); pt = 0
    for _ in range(ms):
        if time.time() > te: break
        ch = False
        for i in rng.permutation(n):
            if time.time() > te: return bh, bv
            cf = make_coefs(bh, i, n)
            tn = f1d_min(co, cf)
            if abs(tn - bh[i]) > 1e-11:
                co += (tn - bh[i]) * cf
                bh[i] = tn; ch = True
        if ch:
            bh = project(bh, n)
            co = np.correlate(bh, 1.0-bh, mode='full').astype(float)
            c = float(np.max(co)*dx)
            if c < bv: bv = c; pt = 0
            else: pt += 1
        else: pt += 1
        if pt >= 5: break
    return bh, bv
```

Clean and simple. No global variables needed for coordinate descent.
