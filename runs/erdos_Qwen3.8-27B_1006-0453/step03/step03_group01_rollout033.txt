import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def round_to_binary(h, N):
    target = N // 2
    order = np.argsort(-h)
    h_bin = np.zeros(N)
    h_bin[order[:target]] = 1.0
    return h_bin

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def swap_optimize(h, corr, N, deadline, max_passes=800, seed=42):
    dx = 2.0 / N

    N2 = 2 * N - 1

    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = 0
        zeros = np.where(h == 0.0)[0]
        ones = np.where(h == 1.0)[0]
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            if h[i] != 0.0:
                continue
            for j in ones:
                if h[j] != 1.0:
                    continue
                delta.fill(0.0)
                Lmin = max(0, i - (N - 1))
                Lmax = min(i, N - 1)
                if Lmax >= Lmin:
                    Ls = np.arange(Lmin, Lmax + 1)
                    delta[Ls + N - 1] += 1.0 - h[i - Ls]
                Lmax2 = N - 1 - i
                if Lmax2 >= 0:

                if Lmax3 >= Lmin3:

                Lmax4 = N - 1 - j
                if Lmax4 >= 0:
                    Ls = np.arange(0, Lmax4 + 1)
                    delta[Ls + N - 1] += h[j + Ls]

                if new_max < best_c5 / dx - 1e-12:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta
                    best_c5 = new_max * dx

                    improved += 1
        if improved > 0:
            no_improve = 0
        else:
            no_improve += 1

def gradient_polish(h, N, target, deadline, lr=0.003, mom=0.85, max_iter=2000):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    h = h.astype(float).copy()
    v = np.zeros(N)
    center = N - 1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        tau = 80.0
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        grad = np.zeros(N)
        active = w > 1e-5
        for k in np.where(active)[0]:
            wi = w[k]
            L = k - center
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N - L])
                grad[:N - L] -= wi * h[L:]
            else:
                M = -L
                grad[:N - M] -= wi * h[M:]
                grad[M:] += wi * (1.0 - h[:N - M])
        v = mom * v + grad
        h = h - lr * v
        h = project_sum(h, target)
        if lr > 0.0001:
            lr *= 0.995
    return best_h, best_c5
