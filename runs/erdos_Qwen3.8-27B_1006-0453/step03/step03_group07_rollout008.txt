def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5_corr(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h_raw, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h_raw - mid, 0, 1).sum() > target:

        else:
            hi = mid

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

        total = room.sum()
        if total > 1e-14:

    else:

        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)

                elif trial == 1:
                    xp = 0.5 + 0.4 * np.cos(2 * np.pi * np.arange(p) / p)
                    xp = np.clip(xp, 0, 1)
                else:
                    xp = np.random.uniform(0.05, 0.95, p)
                xp = xp / xp.sum() * (p / 2.0)
                xp = np.clip(xp, 0, 1)
                for T in [500, 3000]:
                    if rem() < budget_s * 0.005:
                        break
                    try:
                        bnds = [(0.0, 1.0)] * p
                        res = minimize(periodic_obj, xp.copy(), args=(p, n_p, T),
                                       method='L-BFGS-B', bounds=bnds, maxiter=200)
                        xp_opt = np.clip(res.x, 0, 1)
                        h_try = project_feasible(np.tile(xp_opt, n_p // p), n_p)
                        c5v = compute_c5(h_try)
                        if c5v < best_c5:
                            best_c5 = c5v
                            best_h = h_try
                    except Exception:
                        pass

    # Phase 3: Targeted refinement + gradient
    n_cur = len(best_h)
    if rem() > budget_s * 0.50:
        h_td, c5_td = targeted_refine(best_h.copy(), n_cur, 300, t0, budget_s, 0.55)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td
        h_g, c5_g = proj_grad_step(best_h.copy(), 5000, n_cur, 0.2, 200,
                                    t0, budget_s, 0.62)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
        try:
            h_new = slsqp_step(best_h.copy(), 5000, n_cur, 800, 1e-15)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    # Phase 4: Basin hopping with diverse moves
    n_cur = len(best_h)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    max_bh = min(150, int(rem() / 3.0))
    max_bh = max(max_bh, 30)
    temp = 0.004
    for it in range(max_bh):
        if rem() < 8:
            break
        rtype = it % 5
        if rtype == 0:
            h_p = bh_h.copy()
            bl = max(2, n_cur // 10)
            s1 = np.random.randint(0, n_cur - bl)
            s2 = np.random.randint(0, n_cur - bl)
            tmp = h_p[s1:s1+bl].copy()
            h_p[s1:s1+bl] = h_p[s2:s2+bl]
            h_p[s2:s2+bl] = tmp
        elif rtype == 1:
            h_p = bh_h.copy()
            nf = max(2, n_cur // 8)
            idx = np.random.choice(n_cur, nf, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        elif rtype == 2:
            h_p = bh_h + 0.1 * np.random.randn(n_cur)
        elif rtype == 3:
            h_p = bh_h.copy()
            bl = np.random.randint(3, max(4, n_cur // 4))
            sh = np.random.randint(1, bl)
            s1 = np.random.randint(0, n_cur - bl)
            h_p[s1:s1+bl] = np.roll(h_p[s1:s1+bl], sh)
        else:
            h_p = bh_h + 0.18 * np.random.randn(n_cur)
        h_p = project_feasible(h_p, n_cur)
        T_use = 500 + 300 * (it // 15)
        try:
            h_new = slsqp_step(h_p, T_use, n_cur, 100, 1e-12)
            c5v = compute_c5(h_new)
        except Exception:
            h_new = h_p
            c5v = compute_c5(h_p)
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        temp *= 0.996

    # Phase 5: Multi-resolution
    n_cur = len(best_h)
    if rem() > budget_s * 0.12:
        for n_try in [100, 140, 160, 200, 240, 280]:
            if rem() < 15:
                break
            if n_try == n_cur:
                continue
            x_old = np.linspace(0, 2, n_cur, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            try:
                h_g, c5_g = proj_grad_step(h_interp, 2000, n_try, 0.2, 150,
                                            t0, budget_s, 0.01)
                h_new = slsqp_step(h_g, 5000, n_try, 800, 1e-14)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
                    n_cur = n_try
            except Exception:
                pass

    # Phase 6: Final targeted + gradient + SLSQP polish
    n_cur = len(best_h)
    if rem() > 10:
        h_td, c5_td = targeted_refine(best_h.copy(), n_cur, 150, t0, budget_s, 0.93)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td
    if rem() > 5:
        h_g, c5_g = proj_grad_step(best_h.copy(), 10000, n_cur, 0.1, 200,
                                    t0, budget_s, 0.96)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
    if rem() > 3:
        try:
            h_new = slsqp_step(best_h.copy(), 20000, n_cur, 2000, 1e-16)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
