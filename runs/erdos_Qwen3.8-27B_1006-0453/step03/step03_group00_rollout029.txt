```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < target:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-15:
            h += room * (target - s) / tr
    elif s > target:
        if s > 1e-15:
            h *= (1.0 - (s - target) / s)
    return np.clip(h, 0.0, 1.0)

def build_PN(h, n):
    """Build P[p,d] and N[p,d] matrices for binary swap delta computation."""
    d = np.arange(-(n-1), n)  # (2n-1,)
    idx = np.arange(n)  # (n,)
    # P[p,d] = 1 if 0<=p+d<n, times (1-h[p+d])
    pd = idx[:, None] + d[None, :]  # (n, 2n-1)
    p_valid = (pd >= 0) & (pd < n)
    pd_clip = np.clip(pd, 0, n-1)
    P = p_valid * (1.0 - h[pd_clip])
    # N[p,d] = 1 if 0<=p-d<n, times h[p-d]
    pm = idx[:, None] - d[None, :]  # (n, 2n-1)
    m_valid = (pm >= 0) & (pm < n)
    pm_clip = np.clip(pm, 0, n-1)
    N = m_valid * h[pm_clip]
    return P, N

def binary_swap_search(h, n, max_passes=50, t_end=None):
    """Best-improvement binary swap local search."""
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c5 = corr.max()
    
    for p in range(max_passes):
        if t_end is not None and time.time() > t_end:
            break
        P, N = build_PN(h, n)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # delta[i,j,:] = (P[j,:]-P[i,:]) + (N[i,:]-N[j,:])
        # Shape: (n_ones, n_zeros, 2n-1)
        # For memory: n_ones * n_zeros * (2n-1) ≈ 80*80*319 ≈ 2M floats ≈ 16MB for n=160
        # That's fine.
        
        po = P[ones]  # (no, 2n-1)
        pj = P[zeros]  # (nz, 2n-1)
        no_ = N[ones]
        nj_ = N[zeros]
        
        # delta[i_idx, j_idx, k] = (P[j]-P[i]) + (N[i]-N[j])
        delta = pj[None, :, :] - po[:, None, :] + no_[:, None, :] - nj_[None, :, :]
        # Shape: (no, nz, 2n-1)
        
        new_corr = corr[None, None, :] + delta
        new_max = new_corr.max(axis=2)  # (no, nz)
        
        best_val = new_max.min()
        if best_val >= c5 - 1e-12:
            break
        bi, bj = np.unravel_index(new_max.argmin(), new_max.shape)
        i_idx = ones[bi]
        j_idx = zeros[bj]
        
        h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
        corr = corr + delta[bi, bj, :]
        c5 = best_val
    
    return h, c5

def coord_descent_1d(h, n, t_end, max_sweeps=30):
    """Exact 1D coordinate descent for fractional h."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    d = np.arange(-(n-1), n)
    center = n - 1  # index for d=0
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Compute corr_base (h with h[i]=0) and coefficients
            hi_saved = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            # For d != 0: corr[k] = corr_base[k] + x * ((1-h[i+d]) - h[i-d])
            # For d == 0: corr[center] = corr_base[center] + x - x^2
            
            ip = i + d
            im = i - d
            ipv = (ip >= 0) & (ip < n)
            imv = (im >= 0) & (im < n)
            ip_vals = np.where(ipv, best_h[np.clip(ip, 0, n-1)], 0.0)
            im_vals = np.where(imv, best_h[np.clip(im, 0, n-1)], 0.0)
            
            coef = (1.0 - ip_vals) - im_vals  # for d≠0 linear coefficient
            # For d=0 (center index): coef should give x*(1-x), not x*coef
            # So we handle center separately
            
            def f1d(x):
                c = corr_base + x * coef
                c[center] = corr_base[center] + x * (1.0 - x)
                return np.max(c) * dx
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 200})
            x_opt = res.x
            best_h[i] = x_opt
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                improved = True
            else:
                best_h[i] = hi_saved
        if not improved:
            break
    return best_h, best_c5

    t0 = time.time()
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < target:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-15:
            h += room * (target - s) / tr
    elif s > target:
        if s > 1e-15:
            h *= (1.0 - (s - target) / s)
    return np.clip(h, 0.0, 1.0)

def build_PN(h, n):
    d = np.arange(-(n-1), n)
    idx = np.arange(n)
    pd = idx[:, None] + d[None, :]
    p_valid = (pd >= 0) & (pd < n)
    pd_clip = np.clip(pd, 0, n-1)
    P = p_valid * (1.0 - h[pd_clip])
    pm = idx[:, None] - d[None, :]
    m_valid = (pm >= 0) & (pm < n)
    pm_clip = np.clip(pm, 0, n-1)
    N = m_valid * h[pm_clip]
    return P, N

def binary_swap_search(h, n, max_passes=200, t_end=None):
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c5 = corr.max()
    for _ in range(max_passes):
        if t_end is not None and time.time() > t_end:
            break
        P, N = build_PN(h, n)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        po, pj = P[ones], P[zeros]
        no_, nj_ = N[ones], N[zeros]
        delta = pj[None, :, :] - po[:, None, :] + no_[:, None, :] - nj_[None, :, :]
        new_max = (corr[None, None, :] + delta).max(axis=2)
        best_val = new_max.min()
        if best_val >= c5 - 1e-12:
            break
        bi, bj = np.unravel_index(new_max.argmin(), new_max.shape)
        i_idx, j_idx = ones[bi], zeros[bj]
        h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
        corr = corr + delta[bi, bj, :]
        c5 = best_val
    return h, c5

def coord_descent_1d(h, n, t_end, max_sweeps=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    d = np.arange(-(n-1), n)
    center = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_saved = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            ip = i + d
            im = i - d
            ipv = (ip >= 0) & (ip < n)
            imv = (im >= 0) & (im < n)
            ip_vals = np.where(ipv, best_h[np.clip(ip, 0, n - 1)], 0.0)
            im_vals = np.where(imv, best_h[np.clip(im, 0, n - 1)], 0.0)
            coef = (1.0 - ip_vals) - im_vals
            a0 = corr_base[center]
            c_base = corr_base + coef
            c_base[center] = a0
            def f1d(x):
                c = c_base + x * coef
                c[center] = a0 + x * (1.0 - x)
                return np.max(c) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 300})
            x_opt = float(res.x)
            best_h[i] = x_opt
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                improved = True
            else:
                best_h[i] = hi_saved
        if not improved:
            break
    return best_h, best_c5

def perturb_binary(h, n, n_flips):
    h2 = h.copy()
    ones = np.where(h2 > 0.5)[0]
    zeros = np.where(h2 < 0.5)[0]
    for _ in range(min(n_flips, len(ones), len(zeros))):
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h2[i], h2[j] = h2[j], h2[i]
    return h2

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Phase 2: Fractional coordinate descent (15% time)
    t2 = t0 + budget_s * 0.30
    best_h, best_c5 = coord_descent_1d(best_h, n, t2, max_sweeps=60)

    # Phase 3: Interleaved binary + fractional with perturbations (35% time)
    t3 = t0 + budget_s * 0.65
    rounds = 8
    for r in range(rounds):
        frac = (t3 - time.time()) / (rounds - r)
        if frac < 5:
            break
        tb = time.time() + frac * 0.35
        tc = time.time() + frac * 0.95
        # Binary search with random perturbation
        n_flips = 2 + r
        h_pert = perturb_binary((best_h > 0.5).astype(float), n, n_flips)
        h_b, c5_b = binary_swap_search(h_pert, n, max_passes=200, t_end=tb)
        if c5_b < best_c5:
            best_h, best_c5 = h_b.copy(), c5_b

        h_f = best_h + 0.05 * np.random.randn(n)
        h_f = project(h_f, n)
        best_h, best_c5 = coord_descent_1d(h_f, n, tc, max_sweeps=30)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
