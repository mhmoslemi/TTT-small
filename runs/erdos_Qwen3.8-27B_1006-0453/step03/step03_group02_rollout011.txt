def grad_at_lag(h, N, L):
    """Gradient of c_{center+L} w.r.t. h."""
    grad = np.zeros(N)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:N-L])

    idx = np.argsort(corr)[-K:]
    c_top = corr[idx]
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    return idx, w

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, N, L):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def optimize(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K0, K1,

        frac = min(it / max(iters, 1), 1.0)
