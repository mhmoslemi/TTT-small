    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n_top=20, T=200.0):
    """Gradient of smooth approximation to max of top correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weighting over top lags
    top_idx = np.argpartition(corr, -n_top)[-n_top:]
    weights = np.zeros(2*n-1)
    vals = corr[top_idx]
    w = np.exp(T * (vals - vals.max()))
    w /= w.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for i, k in enumerate(top_idx):
        d = int(k - center)
        if d == 0:
            grad += w[i] * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += w[i] * (1.0 - h[:-d])
            grad[:-d] -= w[i] * h[d:]
        else:
            e = -d
            grad[:-e] += w[i] * (1.0 - h[e:])
            grad[e:] -= w[i] * h[:-e]
    return grad

def block_optimize(h, n, start, length, maxiter=50):
    """Optimize h[start:start+length] while fixing the rest"""
    if length < 2:
        return h
    fixed = h.copy()
    def obj(x):
        h_try = fixed.copy()
        h_try[start:start+length] = x
        h_try = np.clip(h_try, 0, 1)
        return compute_c5(h_try)
    
    x0 = h[start:start+length].copy()
    try:
        res = minimize(obj, x0, method='Nelder-Mead', 
                      options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        if res.fun < compute_c5(h):
            h_new = fixed.copy()
            h_new[start:start+length] = np.clip(res.x, 0, 1)
            return h_new
    except:
        pass
    return h
