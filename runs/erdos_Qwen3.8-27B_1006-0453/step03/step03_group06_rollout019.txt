corr[k] = sum_i h[i] * (1-h[i-m]) for valid i, i-m

Wait, let me re-derive. corr[k] with shift m = k - (n-1):
corr[k] = sum_{i: 0<=i<n, 0<=i-m<n} h[i] * (1-h[i-m])

Let u = i, v = i-m. Then corr[k] = sum_{valid u,v} h[u] * (1-h[v]) where v = u-m.

d/dh[j]:
- h[j] appears in h[u] when u=j: contribution (1-h[j-m]) if j-m is valid
- h[j] appears in h[v] when v=j, i.e., u=j+m: contribution -h[j+m] if j+m is valid

So d(corr[k])/dh[j] = (1-h[j-m]) * [0<=j-m<n] - h[j+m] * [0<=j+m<n]

This is clean! The subgradient of max_k corr[k] is the average of gradients of the active (top) k's.

def subgrad_step(h, k_set, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k_set)[-k_set:]
    
    grad = np.zeros(n)
    for k in idx:
        m = k - (n - 1)
        j = np.arange(n)
        # First term: (1-h[j-m]) where j-m in [0,n-1]
        jm = j - m
        m1 = (jm >= 0) & (jm < n)
        # Second term: -h[j+m] where j+m in [0,n-1]
        jp = j + m
        m2 = (jp >= 0) & (jp < n)
        
        g = np.zeros(n)
        g[m1] = (1.0 - h[jm[m1]])
        g[m2] -= h[jp[m2]]
        grad += g
    
    # Normalize
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h
    h_new = h - lr * grad / gn
    return project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h = h * (target / tot)
    return np.clip(h, 0.0, 1.0)

def sum_eq_fn(x):
    return x.sum() - len(x) / 2.0

def smooth_max_obj(h, args):
    T, k = args
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    tk = corr[idx]
    m = np.max(tk)
    val = m + np.log(np.sum(np.exp(T * (tk - m)))) / T
    return val * dx

def do_slq(h0, T, k, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fn}]
    try:
        res = minimize(smooth_max_obj, h0.copy(), args=(T, k),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def interp_to(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)

    idx = np.argpartition(corr, -k_set)[-k_set:]

    grad = np.zeros(n)
    for k in idx:
        m = k - (n - 1)
        jm = j - m
        jp = j + m
        mask1 = (jm >= 0) & (jm < n)
        mask2 = (jp >= 0) & (jp < n)
        g = np.zeros(n)
        g[mask1] = (1.0 - h[jm[mask1]])
        g[mask2] -= h[jp[mask2]]
        grad += g
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return grad
    return grad / gn

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
