import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(30):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            ss = (1.0 - h[mask]).sum()
            if ss < 1e-14:
                break
            h[mask] += diff / ss
        else:
            mask = h > 1e-14
            ss = h[mask].sum()
            if ss < 1e-14:
                break
            h[mask] += diff / ss
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def coord_opt(h, n, i):
    h_old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_old
    coef = np.zeros(2 * n - 1)
    for k in range(1, n):
        c_fwd = 0.0
        if i + k < n:
            c_fwd += (1.0 - h[i + k])
        if i - k >= 0:
            c_fwd += (-h[i - k])
        coef[n - 1 + k] = c_fwd
        c_rev = 0.0
        if i - k >= 0:
            c_rev += (1.0 - h[i - k])
        if i + k < n:
            c_rev += (-h[i + k])
        coef[n - 1 - k] = c_rev
    idx0 = n - 1
    cb0 = corr_base[idx0]
    coef[idx0] = 0.0
    dx = 2.0 / n

    def f1d(hi):
        d = hi - h_old
        vals = corr_base + d * coef
        vals[idx0] = cb0 + d * (1.0 - h_old - hi)
        return float(np.max(vals) * dx)

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 120})
    return res.x

def cd_pass(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = np.random.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                break
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 5e-10:
                continue
            best_h[i] = hi_new
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Heavy coordinate descent
    t_end1 = t0 + budget_s * 0.40
    best_h, best_c5 = cd_pass(best_h, n, t_end1, max_sweeps=5000)

    # Phase 2: Restart with perturbation
    t_end2 = t0 + budget_s * 0.65

        step = 0.01 + 0.04 * np.random.rand()
        h_p = best_h + step * (np.random.rand(n) - 0.5)
        h_p = project(h_p, n)
        hp, cp = cd_pass(h_p, n, min(t_end2, time.time() + budget_s * 0.015), max_sweeps=80)
        if cp < best_c5:

            best_h = hp

    # Phase 3: Try other resolutions

        x_old = np.linspace(0, 2, n, endpoint=False)

        h_int = np.interp(x_new, x_old, best_h)
        h_int = project(h_int, n_try)
