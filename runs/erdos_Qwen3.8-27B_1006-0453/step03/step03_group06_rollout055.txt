Let me also add a **coordinate descent** step:
For each j, the contribution to the smoothed max is:
F ≈ sum_s p_s * c_s (approximately, when T is large)

And ∂F/∂h[j] is linear in h[j]:
∂F/∂h[j] = A_j - 2*B_j*h[j] (approximately)

Wait, is it? Let's check:
∂c_s/∂h[j] involves h[j±τ] but NOT h[j] itself (for τ≠0).
For τ=0: ∂c_0/∂h[j] = 1-2h[j] (linear in h[j])

So for the smoothed max gradient:
grad[j] = dx * [sum_{τ≠0} p_{n-1+τ} * (1-h[j-τ]) + ... + p[n-1]*(1-2h[j])]

The term involving h[j] directly is ONLY from τ=0: p[n-1]*(1-2h[j]).
All other terms involve h[j±τ] for τ≥1, which are constants when optimizing h[j].

So: grad[j] = C_j - 2*dx*p[n-1]*h[j]

where C_j = all other terms (independent of h[j]).

This means the coordinate-wise problem is:
minimize C_j*h[j] - dx*p[n-1]*h[j]²  (up to a constant)
subject to 0≤h[j]≤1 and the sum constraint.

Without the sum constraint, the optimum is:
h[j] = C_j / (2*dx*p[n-1]) clipped to [0,1]

With the sum constraint, it's more complex but still tractable.

Actually, for the smoothed max (not just linear combination), the Hessian is:
∂²F/∂h[j]² involves the derivative of p_s which depends on h[j]... so it's not exactly quadratic. But for large T, the top-k p's are concentrated, and the approximation is good.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def grad_smoothed_c5(h, T, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if topk is not None and topk < 2 * n - 1:
        idx = np.argpartition(corr, -topk)[-topk:]
        c_sub = corr[idx]
        m_top = c_sub.max()
        exps = np.exp(T * (c_sub - m_top))
        Z = exps.sum()
        p = np.zeros(2 * n - 1)
        p[idx] = exps / Z
        m = m_top
    else:
        m = corr.max()
        exps = np.exp(T * (corr - m))
        Z = exps.sum()
        p = exps / Z
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        ps = p[s]
        if ps < 1e-14:
            continue
        tau = s - (n - 1)
        if tau == 0:
            grad += ps * (1.0 - 2.0 * h)
        elif tau > 0:
            grad[tau:] += ps * (1.0 - h[:n - tau])
            grad[:n - tau] -= ps * h[tau:]
        else:
            t2 = -tau
            grad[:n - t2] += ps * (1.0 - h[t2:])
            grad[t2:] -= ps * h[:n - t2]
    grad *= dx
    val = (m + np.log(Z) / T) * dx
    return val, grad

def slsqp_opt(h0, T, topk, maxiter=1000, ftol=1e-14):
    n = len(h0)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        if topk is not None and topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:]
            cs = c[idx]
            m = cs.max()
            return (m + np.log(cs.sum() if topk == len(c) else np.exp(T * (cs - m)).sum()) / T) * dx
        m = c.max()
        return (m + np.log(np.exp(T * (c - m)).sum()) / T) * dx
    def obj_grad(x):
        _, g = grad_smoothed_c5(x, T, topk)
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1))

def subgradient_descent(h0, T_schedule, n_steps, momentum=0.9):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h)
    step_size = 0.05
    for it in range(n_steps):
        frac = it / max(n_steps - 1, 1)
        T = T_schedule[0] + (T_schedule[1] - T_schedule[0]) * frac
        val, g = grad_smoothed_c5(h, T, topk=None)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        g_normed = g / g_norm
        v = momentum * v - step_size * g_normed
        h_new = h + v
        h_new = project_feasible(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        step_size *= 0.999
        h = h_new
    return best, best_c5

def coordinate_descent(h0, T, topk, max_rounds=50):
    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            _, g = grad_smoothed_c5(h, T, topk)
            gc = g[j]
            p0 = 0.0
            _, ppr = grad_smoothed_c5(h, T, topk)
            h_trial = h.copy()
            if gc < 0:
                h_trial[j] = 1.0
            else:
                h_trial[j] = 0.0

            if c5t < best_c5 - 1e-12:
                h = h_trial
                best = h.copy()
                best_c5 = c5t
                improved = True
        if not improved:
            break
    return best, best_c5
