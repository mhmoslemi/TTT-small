from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, T, top_k=3):
    """Smooth max of top_k entries in corr using log-sum-exp."""
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    v = corr[sorted_idx]
    m = np.max(v)
    return m + np.log(np.sum(np.exp(T * (v - m)))) / T

def slsqp_smooth(h, T, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, T, top_k) * dx
    def obj_grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        v = corr[sorted_idx]
        m = np.max(v)
        weights = np.exp(T * (v - m)) / np.sum(np.exp(T * (v - m)))
        # gradient of smooth max w.r.t. corr entries, then chain rule
        g_corr = np.zeros(2*n-1)
        for j, idx in enumerate(sorted_idx):
            g_corr[idx] = weights[j]
        # gradient of corr w.r.t. x
        g = np.zeros(n)
        for idx in sorted_idx:
            if g_corr[idx] < 1e-12:
                continue
            k = idx - (n - 1)

                g[:n-k] += g_corr[idx] * (-x[k:])

                best_c5 = c5_opt
                best_h = h_opt

    # Phase 5: Re-identify active set and focused optimization
    if elapsed() < budget_s * 0.82:
        _, corr = compute_c5(best_h)
        top_idx = np.argsort(corr)[::-1][:4]
        # Optimize only the top constraints with very high T
        for T in [1000000, 5000000]:
            if elapsed() > budget_s * 0.82:
                break
            h_try = slsqp_smooth(best_h, T, 3, 600)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.95:
        h_f = slsqp_top_constraints(best_h, top_m=10, maxiter=1000)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_s = slsqp_smooth(best_h, 10000000, 2, 400)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Final micro-polish: tiny random nudges
    if elapsed() < budget_s * 0.98:
        for _ in range(3):
            if elapsed() > budget_s * 0.98:
                break
            h_p = best_h + rng.randn(best_n) * 0.003
            h_p = project_feasible(h_p, best_n)
            h_p = slsqp_smooth(h_p, 5000000, 2, 300)
            c5_p, _ = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
