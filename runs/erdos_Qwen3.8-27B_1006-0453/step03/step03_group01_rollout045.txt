                h[i] = best_h[i]
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    dx_par = 2.0 / n_par
    best_c5 = compute_c5(h_par, dx_par)
    best_h = h_par.copy()
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)

    sizes = [380, 320, 440]
    time_per_size = (budget_s - 120) / len(sizes)

    for N in sizes:
        if time.time() > deadline - 60:
            break
        size_dead = min(time.time() + time_per_size, deadline - 40)
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_base = np.interp(xN, x_par, h_par)
        h_base = project_sum(h_base, target)

        n_att = 3
        per_att = (size_dead - time.time() - 8) / n_att
        if per_att < 30:
            n_att = max(1, int((size_dead - time.time() - 8) / 30))

        for at in range(n_att):
            if time.time() > size_dead - 12:
                break
            trial_dead = min(time.time() + per_att, size_dead - 4)
            if trial_dead - time.time() < 20:
                break

            if at == 0:
                h_cur = h_base.copy()
            elif at == 1:
                h_cur = h_base + rng.randn(N) * 0.06
                h_cur = project_sum(h_cur, target)
            else:
                h_cur = (h_base > 0.5).astype(float)
                for _ in range(15):
                    idx = rng.choice(N, size=8, replace=False)
                    h_cur[idx] = 1.0 - h_cur[idx]
                h_cur = project_sum(h_cur, target)

            t_rem = trial_dead - time.time()
            if t_rem < 20:
                break
            p1 = time.time() + t_rem * 0.38
            h_cur, c1 = grad_descent(h_cur, N, target, p1, 3.0, 60.0, 40, 10)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h_cur.copy()
            if time.time() < trial_dead - 8:
                p2 = time.time() + (trial_dead - time.time()) * 0.40
                h_cur, c2 = grad_descent(h_cur, N, target, p2, 60.0, 400.0, 10, 3)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cur.copy()
            if time.time() < trial_dead - 3:
                h_cur, c3 = grad_descent(h_cur, N, target, trial_dead, 300.0, 1200.0, 4, 2)
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = h_cur.copy()

    if time.time() < deadline - 15:
        Nb = len(best_h)
        h_f, c_f = grad_descent(best_h.copy(), Nb, Nb / 2.0, time.time() + 30,
                                500.0, 2000.0, 3, 2)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    if time.time() < deadline - 5:
        Nb2 = len(best_h)
        best_h, best_c5 = coord_polish(best_h.copy(), Nb2, Nb2 / 2.0,
                                       deadline - 2, rng)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
