def project_sum(h, target):

    s = float(h.sum())

        return h
    h2 = h - diff / len(h)

    Tc -= Tc.max()

    p = exps / exps.sum()

f_hinge(h, t) = sum_k max(0, corr[k] - t) + (1/T) * log(sum exp(T * corr[k]))

Where t is the current max (or a fraction of it). This gives a smooth approximation that heavily weights lags near the max while still being differentiable.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_weights(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * corr
    Tc -= Tc.max()
    exps = np.exp(Tc)
    total = exps.sum()
    p = exps / total
    return corr, p

def grad_correlation(h, p):
    N = len(h)
    nc = 2 * N - 1
    center = N - 1
    grad = np.zeros(N)
    thresh = 1.0 / (nc * 10.0)
    active = np.where(p > thresh)[0]
    for ai in active:
        L = ai - center
        wi = p[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    grad -= np.sum(grad) / N
    return grad

def smooth_objective(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * corr
    Tc -= Tc.max()
    return float(Tc.max() + np.log(np.exp(Tc).sum()) / T)

def lbgfs_refine(h, N, T, maxiter=150):
    target = N / 2.0
    dx = 2.0 / N

    def obj(v):
        hp = project_sum(v, target)
        return smooth_objective(hp, T)

    def jac(v):
        hp = project_sum(v, target)
        _, p = corr_weights(hp, T)
        return T * grad_correlation(hp, p)

    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
    h_out = project_sum(res['x'], target)
    return h_out, compute_c5(h_out, dx)

def gd_loop(h, N, target, deadline, T0, T1, lr0, decay, mom):
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 1500.0, 1.0)
        T = T0 + (T1 - T0) * frac
        _, p = corr_weights(h, T)
        g = T * grad_correlation(h, p)
        v = mom * v + g
        lr = lr0 * (decay ** it)
        h = project_sum(h - lr * v, target)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if lr < 1e-13:
            break
    return best_h, best_c5

def polish_h(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        _, p = corr_weights(best_h, 80.0)
        g = grad_correlation(best_h, p)
        order = np.argsort(-np.abs(g))
        for i in order:
            if time.time() > deadline:
                break
            orig = best_h[i]
            candidates = (0.0, 1.0, orig - 0.01, orig + 0.01, orig - 0.004, orig + 0.004)
            for trial in candidates:
                if trial < -1e-9 or trial > 1 + 1e-9:
                    continue
                if abs(trial - orig) < 1e-9:
                    continue
                th = best_h.copy()
                th[i] = trial
                th = project_sum(th, target)
                c5 = compute_c5(th, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = th
                    improved = True
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)

    # Phase 1: GD with moderate temperature on parent grid
    d1 = t0 + budget_s * 0.30
    h1, c1 = gd_loop(best_h.copy(), n_par, n_par / 2.0, d1,
                     T0=3.0, T1=80.0, lr0=0.04, decay=0.995, mom=0.80)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: L-BFGS-B refinement
    d2 = t0 + budget_s * 0.48
    if time.time() < d2 - 10:
        h2, c2 = lbgfs_refine(best_h.copy(), n_par, T=150.0, maxiter=120)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: High-T GD focusing on max lag
    d3 = t0 + budget_s * 0.62
    if time.time() < d3 - 10:
        h3, c3 = gd_loop(best_h.copy(), n_par, n_par / 2.0, d3,
                         T0=150.0, T1=800.0, lr0=0.008, decay=0.992, mom=0.85)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    # Phase 4: Try nearby N values
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)
    for Nd in [n_par + 16, n_par - 16, n_par + 40]:
        if time.time() > deadline - 80:
            break
        if Nd < 200 or Nd > 800:
            continue
        x_new = np.linspace(0.0, 2.0, Nd, endpoint=False)
        h_try = np.interp(x_new, x_par, best_h)
        h_try = project_sum(h_try, Nd / 2.0)
        t_rem = deadline - time.time() - 40
        if t_rem < 40:
            break
        h_t, c_t = gd_loop(h_try, Nd, Nd / 2.0, time.time() + t_rem * 0.55,
                           T0=30.0, T1=400.0, lr0=0.015, decay=0.993, mom=0.82)
        if time.time() < deadline - 35:
            h_t, c_t = polish_h(h_t, Nd, Nd / 2.0, deadline - 35)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t

    # Phase 5: L-BFGS-B on best found
    if time.time() < deadline - 50:
        Nb = len(best_h)
        h_l, c_l = lbgfs_refine(best_h.copy(), Nb, T=600.0, maxiter=100)
        if c_l < best_c5:
            best_c5 = c_l
            best_h = h_l

    # Phase 6: Multi-restart GD from best with perturbations
    if time.time() < deadline - 40:
        Nb = len(best_h)
        for trial in range(3):
            if time.time() > deadline - 20:
                break
            pert = 0.02 / (trial + 1)
            h_p = best_h + rng.randn(Nb) * pert
            h_p = project_sum(h_p, Nb / 2.0)
            t_rem = deadline - time.time() - 15
            if t_rem < 15:
                break
            h_r, c_r = gd_loop(h_p, Nb, Nb / 2.0, time.time() + t_rem * 0.7,
                               T0=200.0, T1=800.0, lr0=0.004, decay=0.994, mom=0.86)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r

    # Phase 7: Final polish
    if time.time() < deadline - 5:
        Nb = len(best_h)
        best_h, best_c5 = polish_h(best_h, Nb, Nb / 2.0, deadline - 2)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
