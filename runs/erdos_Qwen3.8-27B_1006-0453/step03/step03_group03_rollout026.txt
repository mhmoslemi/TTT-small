    # Phase 5: Momentum subgradient at best n (15% time)
    t5_end = t0 + budget_s * 0.85
    remain = max(60, t5_end - time.time())
    n_iters = int(remain * 5)
    n_iters = min(n_iters, 8000)
    h_sub, c5_sub = momentum_opt(best_h, n_iters=n_iters, lr0=0.04, mom=0.72, topk=10)
    if c5_sub < best_c5:
        best_c5 = c5_sub
        best_h = h_sub
    h_pol = slsqp_pnorm(best_h, 1000, maxiter=500)
    c5_pol, _ = compute_c5(h_pol)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol

    # Phase 6: Final polish - multi-start + high-p SLSQP (10% time)
    t6_end = t0 + budget_s * 0.97
    for scale in [0.02, 0.01, 0.005]:
        if time.time() > t6_end:
            break
        h_p = best_h + np.random.randn(best_n) * scale
        h_p = project_feasible(h_p, best_n)
        h_opt = slsqp_pnorm(h_p, 500, maxiter=400)
        h_opt = slsqp_lse(h_opt, 80000, maxiter=400)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(h_opt)

    # Rotation search
    if time.time() < t0 + budget_s * 0.98:
        for shift in [1, 2, 3, 5]:
            if time.time() > budget_s * 0.98 + t0:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_opt = slsqp_pnorm(h_rot, 800, maxiter=300)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
