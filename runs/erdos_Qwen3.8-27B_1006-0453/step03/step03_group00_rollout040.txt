def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()

            if ts < 1e-14:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n * 0.5
    for _ in range(200):
        d = t - h.sum()
        if abs(d) < 1e-14: return h
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14: break
            h += r * d / s
        else:
            s = h.sum()
            if s < 1e-14: break
            h *= 1.0 + d / s
        np.clip(h, 0, 1, out=h)
    return h

def smax(h, T, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def slsqp(h0, n, T, t_end, mit=1000, ftol=1e-15):
    def f(x): return smax(x, T, n)
    def g(x): return x.sum() - n * 0.5
    r = minimize(f, h0, method='SLSQP',
                 bounds=[(0,1)]*n, constraints=[{'type':'eq','fun':g}],
                 options={'maxiter':mit,'ftol':ftol})
    if time.time() > t_end: return h0
    return proj(np.clip(r.x, 0, 1), n)

def cdesc(h, n, t_end, ns=100):
    bh, bc = h.copy(), compute_c5(h)
    dx = 2.0/n
    for sw in range(ns):
        if time.time() > t_end: break
        imp = False
        for i in range(n):
            if time.time() > t_end: return bh, bc
            ho = bh[i]
            bh[i] = 0.0
            cb = np.correlate(bh, 1.0 - bh, mode='full')
            cf = np.zeros(2*n-1)
            for k in range(1, n):
                q1, p1 = i-k, i+k
                c1 = 0.0
                if 0 <= q1 < n: c1 += 1.0 - bh[q1]
                if 0 <= p1 < n: c1 -= bh[p1]
                cf[n-1+k] = c1
                q2, p2 = i+k, i-k
                c2 = 0.0
                if 0 <= q2 < n: c2 += 1.0 - bh[q2]
                if 0 <= p2 < n: c2 -= bh[p2]
                cf[n-1-k] = c2
            cf[n-1] = 1.0
            bh[i] = ho
            def f1(a, cb=cb, cf=cf, dx=dx):
                v = cb + a*cf
                v[n-1] -= a*a
                return v.max()*dx
            res = minimize_scalar(f1, bounds=(0.0,1.0), method='bounded',
                                  options={'xatol':1e-9})
            bh[i] = res.x
            ct = compute_c5(bh)
            if ct < bc - 1e-13:
                bc, imp = ct, True
