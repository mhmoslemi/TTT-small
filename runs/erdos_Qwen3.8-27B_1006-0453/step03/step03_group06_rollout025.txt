def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-14:
            h += room * (diff / t)
    else:
        t = h.sum()
        if t > 1e-14:
            h *= (1.0 + diff / t)
    return np.clip(h, 0.0, 1.0)

def constraint_sum(x):
    return x.sum() - len(x) / 2.0

def soft_max_obj(x, T, topk):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        c = corr
    m = float(c.max())
    if T < 1:
        return m
    return m + float(np.log(np.sum(np.exp(T * (c - m)))) / T)

    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    res = minimize(soft_max_obj, h0, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                   args=(T, topk), options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

    h0 = h.copy(); h0[j] = 0.0
    h1 = h.copy(); h1[j] = 1.0

    c1 = np.correlate(h1, 1.0 - h1, mode='full') * dx

def coord_sweep(h, n, n_passes=1):
    for p in range(n_passes):

        for j in order:
            t_opt = coord_optimize(h, j, n)
            h[j] = t_opt
            h = project(h, n)
    return h

def subgradient_step(h, n, T, topk, alpha):
    dx = 2.0 / n
    one_m_h = 1.0 - h
    corr = np.correlate(h, one_m_h, mode='full') * dx
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    m = c.max()
    w = np.exp(T * (c - m))
    w /= w.sum()
    grad = np.zeros(n)
    for ii in range(len(idx)):
        k = int(idx[ii])
        wi = w[ii]
        s = k - (n - 1)

def coord_sweep(h, n, fwd=True):
    order = np.arange(n) if fwd else np.arange(n - 1, -1, -1)
    for j in order:
        t_opt = coord_optimize(h, int(j), n)
        h[j] = t_opt
        h = project(h, n)
    return h

def subgrad_step(h, n, T, topk, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:].astype(int)
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    m = c.max()
    w = np.exp(T * (c - m))
    w /= w.sum()
    grad = np.zeros(n)
    for ii in range(len(idx)):
        k = int(idx[ii])
        wi = w[ii]
        s = k - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += wi * dx * (1.0 - h[s:])
                grad[s:] -= wi * dx * h[:ns]
        else:
            ss = -s
            ns = n - ss
            if ns > 0:
                grad[ss:] += wi * dx * (1.0 - h[:ns])
                grad[:ns] -= wi * dx * h[ss:]
    return h - alpha * grad

def interp_to(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(xn, xo, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(100, 0.5)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP cleanup (10% time)
    if elapsed() < budget_s * 0.10:
        for T, tk, mi in [(100, None, 800), (500, 5, 800), (2000, 3, 1000), (8000, 3, 1200)]:
            if elapsed() > budget_s * 0.10:
                break
            try:
                hc = slsqp_opt(best_h, T, tk, mi)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = hc
            except Exception:
                pass

    # Phase 2: Coordinate descent - main refinement (30% time)
    t_cd = elapsed()
    cd_budget = budget_s * 0.30
    while elapsed() - t_cd < cd_budget and elapsed() < budget_s * 0.45:
        best_h = coord_sweep(best_h, n, fwd=True)
        cv = compute_c5(best_h)
        if cv < best_c5:
            best_c5 = cv
        best_h = coord_sweep(best_h, n, fwd=False)
        cv = compute_c5(best_h)
        if cv < best_c5:
            best_c5 = cv

    # Phase 3: Subgradient with periodic coord sweep (15% time)
    t_sg = elapsed()
    sg_budget = budget_s * 0.15
    alpha = 0.03
    T_sg = 300
    while elapsed() - t_sg < sg_budget and elapsed() < budget_s * 0.62:
        h_try = subgrad_step(best_h, n, T_sg, 5, alpha)
        h_try = project(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_h = h_try
            best_c5 = c5_try
            alpha = min(alpha * 1.02, 0.08)
        else:
            alpha *= 0.75
        if int((elapsed() - t_sg) * 5) % 2 == 0:
            best_h = coord_sweep(best_h, n, fwd=np.random.rand() > 0.5)
            cv = compute_c5(best_h)
            if cv < best_c5:
                best_c5 = cv

    # Phase 4: Multi-resolution exploration (20% time)
    t_mr = elapsed()
    for n_try in [120, 200, 280, 80, 320]:
        if elapsed() - t_mr > budget_s * 0.20 or elapsed() > budget_s * 0.82:
            break
        h_try = interp_to(best_h, n_try)
        try:
            h_try = slsqp_opt(h_try, 500, 5, 400)
        except Exception:
            pass
        h_try = coord_sweep(h_try, n_try, fwd=True)
        h_try = coord_sweep(h_try, n_try, fwd=False)
        cv = compute_c5(h_try)
        if cv < best_c5:
            best_c5 = cv
            best_h = h_try
            n = n_try

    # Phase 5: More coordinate descent on best (15% time)
    n = len(best_h)
    t_cd2 = elapsed()
    while elapsed() - t_cd2 < budget_s * 0.15 and elapsed() < budget_s * 0.96:
        best_h = coord_sweep(best_h, n, fwd=True)
        cv = compute_c5(best_h)
        if cv < best_c5:
            best_c5 = cv
        best_h = coord_sweep(best_h, n, fwd=False)
        cv = compute_c5(best_h)
        if cv < best_c5:
            best_c5 = cv

    # Phase 6: Final SLSQP + coord polish (5% time)
    n = len(best_h)
    if elapsed() < budget_s * 0.99:
        try:
            hc = slsqp_opt(best_h, 10000, 3, 800)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc
        except Exception:
            pass
    if elapsed() < budget_s * 0.995:
        best_h = coord_sweep(best_h, len(best_h), fwd=True)
        best_c5 = compute_c5(best_h)
        best_h = coord_sweep(best_h, len(best_h), fwd=False)
        best_c5 = compute_c5(best_h)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
