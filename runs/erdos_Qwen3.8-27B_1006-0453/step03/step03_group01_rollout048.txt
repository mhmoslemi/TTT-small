    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_weighted(h, N, w):
    dx = 2.0 / N
    center = N - 1
    grad = np.zeros(N)
    active = np.where(w > 1e-10)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad * dx

def adam_loop(h, N, target, deadline, rng, cfg):
    tau0, tau1 = cfg['tau0'], cfg['tau1']
    k0, k1 = cfg['k0'], cfg['k1']
    lr = cfg['lr']
    b1, b2, eps = 0.9, 0.999, 1e-8
    restart_every = cfg.get('restart_every', 400)
    restart_mag = cfg.get('restart_mag', 0.03)
    warmup = cfg.get('warmup', 50)
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    no_improve = 0
    est_iters = max(1, int((deadline - time.time()) * 200))
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / max(1, est_iters))
        tau = tau0 + (tau1 - tau0) * frac
        topK = max(2, int(k0 + (k1 - k0) * frac))
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_val = float(corr.max())
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > restart_every and it > warmup:
            h = h + rng.randn(N) * restart_mag
            h = project_sum(h, target)
            m[:] = 0.0
            v[:] = 0.0
            no_improve = 0
            continue
        nc = len(corr)
        idx_top = np.argsort(corr)[-topK:]
        vals = corr[idx_top] - corr[idx_top].max()
        p = np.exp(tau * vals)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(len(idx_top)) / len(idx_top)
        else:
            p /= p_sum
        w = np.zeros(nc)
        w[idx_top] = p
        g = grad_weighted(h, N, w)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h - step, target)
    return best_h, best_c5

def momentum_loop(h, N, target, deadline, rng, cfg):
    tau = cfg['tau']
    topK = cfg['topK']
    lr = cfg['lr']
    mom = cfg['mom']
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr_cur = lr
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_val = float(corr.max())
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        nc = len(corr)
        idx_top = np.argsort(corr)[-max(2, topK):]
        w = np.zeros(nc)
        w[idx_top] = 1.0 / len(idx_top)
        g = grad_weighted(h, N, w)
        v = mom * v + g
        lr_cur *= 0.9995
        h = project_sum(h - lr_cur * v, target)
    return best_h, best_c5

def polish(h, N, target, deadline, rng):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            trials = [0.0, 1.0, max(0.0, h[i] - 0.05), min(1.0, h[i] + 0.05),
                      max(0.0, h[i] - 0.01), min(1.0, h[i] + 0.01)]
            orig = h[i]
            for tr in trials:
                if abs(tr - orig) < 1e-9:
                    continue
                h[i] = tr
                hp = project_sum(h, target)
                c5 = compute_c5(hp, dx)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = hp.copy()
                    improved = True
                    break
            h = best_h.copy()
        # Random block perturbation
        if time.time() < deadline - 2:
            for _ in range(5):
                if time.time() > deadline:
                    break
                b = rng.randint(0, max(1, N - 10))
                w = rng.randint(5, 15)
                w = min(w, N - b)
                pert = best_h.copy()
                pert[b:b + w] += rng.randn(w) * 0.04
                pert = project_sum(pert, target)
                c5 = compute_c5(pert, dx)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = pert.copy()
                    improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)
    # Phase 1: Adam optimization at multiple resolutions
    sizes = [380, 480, 576]
    phase1_dead = deadline - 120
    per_size = (phase1_dead - t0) / len(sizes)
    for si, N in enumerate(sizes):
        if time.time() > phase1_dead - 30:
            break
        size_dead = min(t0 + (si + 1) * per_size, phase1_dead - 15)
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_base = np.interp(xN, x_par, h_par)
        h_base = project_sum(h_base, target)
        n_starts = 3
        per_start = max(10.0, (size_dead - time.time() - 5) / n_starts)
        for st in range(n_starts):
            if time.time() > size_dead - 10:
                break
            trial_dead = time.time() + per_start
            if trial_dead > size_dead - 3:
                trial_dead = size_dead - 3
            if trial_dead - time.time() < 15:
                break
            if st == 0:
                h_cur = h_base.copy()
            elif st == 1:
                h_cur = h_base + rng.randn(N) * 0.05
                h_cur = project_sum(h_cur, target)
            else:
                h_cur = h_base + rng.randn(N) * 0.10
                h_cur = project_sum(h_cur, target)
            cfg1 = {'tau0': 5.0, 'tau1': 80.0, 'k0': 30, 'k1': 4,
                    'lr': 0.008, 'restart_every': 500, 'restart_mag': 0.04, 'warmup': 100}
            t_mid = time.time() + (trial_dead - time.time()) * 0.55
            h_cur, c1 = adam_loop(h_cur, N, target, t_mid, rng, cfg1)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h_cur.copy()
            if time.time() < trial_dead - 5:
                cfg2 = {'tau0': 80.0, 'tau1': 400.0, 'k0': 5, 'k1': 2,
                        'lr': 0.003, 'restart_every': 350, 'restart_mag': 0.02, 'warmup': 50}
                h_cur, c2 = adam_loop(h_cur, N, target, trial_dead, rng, cfg2)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cur.copy()
    # Phase 2: Momentum fine-tuning on best
    if time.time() < deadline - 80:
        Nb = len(best_h)
        target_b = Nb / 2.0
        cfg_m = {'tau': 500.0, 'topK': 3, 'lr': 0.004, 'mom': 0.92}
        h_f, c_f = momentum_loop(best_h.copy(), Nb, target_b,
                                 min(time.time() + 40, deadline - 50), rng, cfg_m)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f
    # Phase 3: Polish
    if time.time() < deadline - 10:
        Nb2 = len(best_h)
        best_h, best_c5 = polish(best_h, Nb2, Nb2 / 2.0, deadline - 2, rng)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
