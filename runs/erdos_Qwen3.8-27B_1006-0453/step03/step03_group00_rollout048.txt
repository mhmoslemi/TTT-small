from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, n):
    """Project to sum = n/2 while maintaining [0,1] bounds."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # Scale down the non-zero entries
        mask = h > 1e-10
        h[mask] *= (target / s)
    else:
        # Scale up the non-one entries
        mask = h < 1.0 - 1e-10
        if mask.sum() == 0:
            return h
        room = 1.0 - h[mask]
        h[mask] = h[mask] + room * ((target - s) / room.sum())
    return np.clip(h, 0.0, 1.0)

def coord_gradient(h, i):
    """Compute coef[k] = d(corr[k])/d(h[i]) for all lags k."""
    n = len(h)
    coef = np.zeros(2 * n - 1)
    for k in range(n):  # k is the relative lag in 0..2n-2

        # Let's use the formula: corr[n-1+k] = sum_j h[j+k]*(1-h[j])
        # d/dh[i] of corr[n-1+lag]:
        # From h[j+lag]=h[i] => j = i-lag: (1-h[i-lag])

def project_sum(h, n):
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return np.clip(h, 0, 1)
    if s > target:
        mask = h > 1e-10
        if mask.sum() == 0:
            return np.clip(h, 0, 1)

    else:

            return np.clip(h, 0, 1)
        h = h.copy()
        room = 1.0 - h[mask]
        h[mask] += room * ((target - s) / room.sum())
    return np.clip(h, 0, 1)

    n = len(h)
    dx = 2.0 / n

                c += -h[j2]
            else:
                c += -h[j2]

        return np.max(corr_base + hi * coef) * dx
