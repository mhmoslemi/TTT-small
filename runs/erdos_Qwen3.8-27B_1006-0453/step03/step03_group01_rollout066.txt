C5_smooth(τ) = (1/τ) * log(Σ exp(τ * corr[k] * dx))

As τ → ∞, this approaches the max. For finite τ, it's smooth and differentiable.

where w[k] = exp(τ*corr[k]*dx) / Σ exp(τ*corr[m]*dx)

This is a weighted average of the subgradients at all lags, weighted by their "importance" (exponential of correlation). At the limit τ→∞, only the maximum lag contributes.

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    """Compute gradient of the smoothed C5 objective."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    sc = tau * corr * dx
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    
    center = N - 1
    nc = len(corr)
    grad = np.zeros(N)
    
    active = np.where(w > 1e-6)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad * dx, float(np.max(corr)) * dx

def gradient_opt(h, N, target, deadline, tau0, tau1, lr0, decay, period, mom):
    """Gradient-based optimization with annealing."""
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/N)
    v = np.zeros(N)
    lr = lr0
    it = 0
    total = 1000
    while time.time() < deadline:
        it += 1
        frac = min(it / total, 1.0)
        tau = tau0 + (tau1 - tau0) * frac
        grad, c5 = softmax_grad(h, N, tau)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        v = mom * v + (1-mom) * grad
        h = h - lr * v
        h = project_sum(h, target)
        if it % period == 0:
            lr *= decay
            if lr < 1e-12:
                break
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    sc = tau * corr * dx
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    center = N - 1
    grad = np.zeros(N)
    active = np.where(w > 1e-6)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad * dx, float(np.max(corr)) * dx

def gradient_opt(h, N, target, deadline, tau0, tau1, lr0, decay, period, mom):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/N)
    v = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 800.0, 1.0)
        tau = tau0 + (tau1 - tau0) * frac
        grad, c5 = softmax_grad(h, N, tau)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        v = mom * v + (1-mom) * grad
        h = h - lr * v
        h = project_sum(h, target)
        if it % period == 0:
            lr *= decay
            if lr < 1e-12:
                break
    return best_h, best_c5

def coord_optimize(h, N, dx, deadline, max_sweeps=200, topK=40):
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    M_arr = np.abs(np.arange(-(N-1), N))
    center = N - 1
    sweep = 0
    stall = 0
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        b0 = float(np.sum(h*(1.0-h)) - h[0]*(1.0-h[0])) if N > 1 else 0.0
        for j in range(N):
            if time.time() > deadline:
                break
            valid1 = M_arr <= j
            valid2 = M_arr < (N - j)
            a = np.zeros(2*N-1)
            a[valid1] = (1.0 - h[j - M_arr[valid1]])
            a[valid2] -= h[j + M_arr[valid2]]
            b = corr_raw - a * h[j]
            # L=0 is special: corr(0) = b0_rest + x - x^2 (concave)
            b0_rest = float(np.sum(h*(1.0-h)) - h[j]*(1.0-h[j]))
            candidates = np.array([0.0, 1.0, h[j], 0.5])
            # Add uniform grid for nonlinear L=0
            candidates = np.concatenate([candidates, np.linspace(0, 1, 16)])
            vals_cur = a * h[j] + b
            top_idx = np.argsort(vals_cur)[-topK:]
            for ii in range(min(topK, len(top_idx))):
                li = top_idx[ii]
                if li == center:
                    continue
                for jj in range(ii+1, min(topK, len(top_idx))):
                    lj = top_idx[jj]
                    if lj == center:
                        continue
                    da = a[li] - a[lj]
                    if abs(da) > 1e-10:
                        x_int = (b[lj] - b[li]) / da
                        if 0.0 <= x_int <= 1.0:
                            candidates = np.append(candidates, x_int)
                    # Also intersect with parabola: a*li*x + b[li] = b0_rest + x - x^2
                    # x^2 + (a[li]-1)*x + (b[li]-b0_rest) = 0
                    A_q = 1.0
                    B_q = a[li] - 1.0
                    C_q = b[li] - b0_rest
                    disc = B_q*B_q - 4*A_q*C_q
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        x1 = (-B_q + sq) / (2*A_q)
                        x2 = (-B_q - sq) / (2*A_q)
                        if 0.0 <= x1 <= 1.0:
                            candidates = np.append(candidates, x1)
                        if 0.0 <= x2 <= 1.0:
                            candidates = np.append(candidates, x2)
            candidates = np.clip(candidates, 0.0, 1.0)
            if len(candidates) > 400:
                candidates = candidates[np.linspace(0, len(candidates)-1, 400).astype(int)]
            candidates = np.unique(np.round(candidates, 10))
            # Evaluate: C5(x) = max(max_{L!=0}(a_L*x+b_L), b0_rest+x-x^2) * dx
            lin_vals = (a[None,:] * candidates[:,None] + b[None,:])
            # Zero out the L=0 column for linear part
            lin_vals[:, center] = -np.inf
            lin_max = lin_vals.max(axis=1)
            par_max = b0_rest + candidates - candidates*candidates
            total_max = np.maximum(lin_max, par_max) * dx
            best_idx = total_max.argmin()
            best_x = candidates[best_idx]
            if abs(best_x - h[j]) > 1e-10:
                old = h[j]
                h[j] = best_x
                corr_raw += (best_x - old) * a
                improved = True
        h = project_sum(h, target)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if not improved:
            break
        if stall >= 5:
            # Perturb to escape local optimum
            h = best_h + np.random.RandomState(sweep*7).randn(N) * 0.01
            h = project_sum(h, target)
            stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
