def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, int(np.argmax(corr))

def compute_c5_full(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_at(h, N, k_star):
    """Exact gradient of c(k_star) w.r.t. h"""
    grad = np.zeros(N)
    center = N - 1
    L = k_star - center
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:N-L])
        grad[:N-L] -= h[L:]
    else:
        M = -L
        grad[:N-M] += (1.0 - h[M:])
        grad[M:] -= h[:N-M]
    return grad

def multi_lag_gradient(h, N, K, tau):
    """Weighted gradient over top-K lags"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-K:]
    c_top = corr[top_idx]
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), float(np.max(corr)), top_idx[0]
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(K):
        wi = w[ix]
        if wi < 1e-8:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, float(np.max(corr)), top_idx[np.argmax(c_top)]

def run_subgrad(h, N, target, dx, deadline, tau_seq, K, lr_init, lr_end):
    """Adam-based multi-lag subgradient descent"""
    best_h = h.copy()

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.time() < deadline:
        it += 1
        tau = tau_seq[min(it // 50, len(tau_seq)-1)]
        frac = min(it / max(500, 1), 1.0)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, c5_raw, _ = multi_lag_gradient(h, N, K, tau)
        c5_val = c5_raw * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad**2
        mc = m / (1 - b1**it)
        vc = v / (1 - b2**it)
        h = h - lr * mc / (np.sqrt(vc) + eps)
        h = project_sum(h, target)
    
    return best_h, best_c5

def coordinate_descent(h, N, target, dx, n_iter, deadline):

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)[0]

                best_c5 = c
                break
    return best_h, best_c5
