    n = len(h)

    return float(np.max(corr)) * dx

def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h *= (total + diff) / total
    return np.clip(h, 0.0, 1.0)

def subgrad_h(h, s, n):
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    g = np.zeros(n)
    idx1 = np.arange(lo, hi + 1)
    g[idx1] += (1.0 - h[idx1 + s])
    idx2 = np.arange(lo, hi + 1) + s
    g[idx2] += -h[idx2 - s]
    return g

def subgradient_descent(h0, n, n_iter, lr, topk, momentum=0.85):

        corr = compute_c5_corr(h)
        idx = np.argpartition(corr, -topk)[-topk:]

            s = int(ki) - (n - 1)

            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    for topo in [3, 5]:
        if elapsed() > budget_s * 0.985:
            break
        try:
            hc = slsqp_smooth(best_h, n_cur, 8000, topo, 800)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
