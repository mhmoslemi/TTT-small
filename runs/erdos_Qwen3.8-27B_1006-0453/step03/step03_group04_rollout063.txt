def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        elif L < 0:
            M = -L
            grad[:N - M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:N - M]
    return grad, peak

    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    eps = 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def binarize(h, N):
    h_b = (h > 0.5).astype(float)
    target = N // 2
    n_ones = int(h_b.sum())
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        h_b[idx[target:]] = 0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:target - n_ones]] = 1
    return h_b

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def transfer(h, new_N):
    old_N = len(h)
    x_old = np.linspace(0.0, 2.0, old_N, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_N, endpoint=False)
    return np.interp(x_new, x_old, h)

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        elif L < 0:
            M = -L
            grad[:N - M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)

    best = h_b.copy()
    best_pk = float(np.max(compute_corr(best)))
    total_time = max(deadline - time.time(), 1.0)
    time_per_run = total_time / n_restarts
    t0, t1 = 15.0, 0.002

    for restart in range(n_restarts):
        if time.time() > deadline - 2:
            break
        run_dl = min(deadline - 2, time.time() + time_per_run)
        cur = h_b.copy()
        for _ in range(30 + 40 * restart):

            zeros = np.where(cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:

        cur_pk = float(np.max(compute_corr(cur)))

                        best = cur.copy()
                else:

    # Greedy polish: try all single-swap improvements repeatedly
    for _ in range(30):
        if time.time() > deadline - 1:
            break
        improved = False
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        for _ in range(len(ones)):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best[i], best[j] = 0.0, 1.0
            new_pk = float(np.max(compute_corr(best)))

                best_pk = new_pk
                improved = True
                ones = np.where(best == 1)[0]
                zeros = np.where(best == 0)[0]
            else:
                best[i], best[j] = 1.0, 0.0
        if not improved:
            break

    return best, best_pk * (2.0 / N)
