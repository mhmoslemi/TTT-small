        hi = min(n, n - shift)
        for i in range(lo, hi):
            grad[i] += (1.0 - h[i + shift])
            grad[i + shift] -= h[i]
    grad /= (len(bad) * dx)
    return grad

def coordinate_descent(h, n, best_c5, max_iters, dx_min=1e-5):
    for it in range(max_iters):
        improved = False
        for j in range(n):
            for d in [1e-4, 5e-5, 1e-5, -1e-4, -5e-5, -1e-5]:
                ht = h.copy()
                ht[j] += d
                ht = project_feasible(ht, n)
                c = compute_c5(ht)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = ht
                    improved = True
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def slsqp_polish(h, n, T, topk, maxiter=2000, ftol=1e-16):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_obj(x, T, topk), h,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def swap_descent(h, n, best_c5, n_trials=500):
    rng = np.random.default_rng(0)
    for _ in range(n_trials):
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i1 == i2:
            continue
        for d in [0.01, 0.005, 0.002, 0.001, -0.01, -0.005]:
            ht = h.copy()
            ht[i1] += d
            ht[i2] -= d
            ht = project_feasible(ht, n)
            c = compute_c5(ht)
            if c < best_c5 - 1e-14:
                best_c5 = c
                h = ht
                break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n = 160
        best_h = np.ones(n) * 0.5
        best_h[:n // 4] = 1.0
        best_h[n * 3 // 4:4 * n] = 0.0
        best_h = project_feasible(best_h, n)

    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: T-sweep SLSQP at native resolution
    t_sched = [5, 20, 80, 300, 1000, 4000, 15000, 50000, 150000]
    k_sched = [None, 5, 3, 2]
    for T in t_sched:
        if time.time() - t0 > budget_s * 0.30:
            break
        for k in k_sched:
            if time.time() - t0 > budget_s * 0.32:
                break
            hc = slsqp_polish(best_h, n, T, k, 1500, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 2: Subgradient method for non-smooth max
    for it in range(300):
        if time.time() - t0 > budget_s * 0.50:
            break
        g = subgrad_step(best_h, n, 8)
        if g is None:
            break
        lr = 0.5
        accepted = False
        for _ in range(30):
            ht = best_h - lr * g
            ht = project_feasible(ht, n)
            c5v = compute_c5(ht)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = ht
                accepted = True
                break
            lr *= 0.5
        if not accepted and it > 200:
            break

    # Phase 3: Basin hopping with shift-aware moves
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = 100
    for it in range(n_bh):
        if time.time() - t0 > budget_s * 0.65:
            break
        step = 0.08 * (1.0 - 2.0 * it / n_bh) + 0.01
        r = rng.random()
        hp = bh_h.copy()
        if r < 0.35:
            hp += step * rng.standard_normal(n)
        elif r < 0.65:
            blen = int(rng.integers(3, n // 3 + 1))
            start = int(rng.integers(0, n - blen))
            hp[start:start + blen] += step * (rng.random(blen) * 2 - 1)
        else:
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            hp[i1] += step * 0.5
            hp[i2] -= step * 0.5
        hp = project_feasible(hp, n)
        T_try = rng.choice([200, 1000, 5000])
        k_try = int(rng.choice([0, 3, 5]))
        hc = slsqp_polish(hp, n, T_try, k_try if k_try else None, 400, 1e-13)
        c5v = compute_c5(hc)
        temp = 0.004 * (1.0 - it / n_bh) + 0.0002
        if c5v < bh_c5 or rng.random() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = hc
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: Multi-resolution exploration
    for n_try in [100, 120, 140, 200, 240]:
        if time.time() - t0 > budget_s * 0.80:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        for trial in range(2):
            if time.time() - t0 > budget_s * 0.82:
                break
            h0 = h_interp + rng.uniform(-0.03, 0.03, n_try)
            h0 = project_feasible(h0, n_try)
            for T in [200, 2000, 20000]:
                if time.time() - t0 > budget_s * 0.84:
                    break
                hc = slsqp_polish(h0, n_try, T, 3, 800, 1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    h_interp = np.interp(x_new, x_old, best_h)

    # Phase 5: Fine-grained coordinate + swap descent
    best_h, best_c5 = coordinate_descent(best_h, len(best_h), best_c5, 200)
    if time.time() - t0 < budget_s * 0.90:
        best_h, best_c5 = swap_descent(best_h, len(best_h), best_c5, 800)

    # Phase 6: Final ultra-high-T polish
    n_cur = len(best_h)
    for T in [50000, 200000, 500000]:
        if time.time() - t0 > budget_s * 0.97:
            break
        hc = slsqp_polish(best_h, n_cur, T, 2, 3000, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    for k in [3, 5]:
        if time.time() - t0 > budget_s * 0.985:
            break
        hc = slsqp_polish(best_h, n_cur, 100000, k, 2000, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final micro coordinate descent
    best_h, best_c5 = coordinate_descent(best_h, len(best_h), best_c5, 100)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
