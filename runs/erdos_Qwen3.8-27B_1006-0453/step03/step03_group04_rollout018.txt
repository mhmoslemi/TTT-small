def compute_corr(h):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]

        h_b[j] = 1
        best_corr_max = best_improve

    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)


def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak


def adam_optimize(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)


def to_binary(h, N):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    half = N // 2
    if n_ones > half:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        h_b[idx[order[half:]]] = 0
    elif n_ones < half:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(h[idx])[::-1]
        h_b[idx[order[:half - n_ones]]] = 1
    return h_b


def hill_climb(h_b, N, deadline, max_rounds=80):
    best_h = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        cur_best = best_pk
        cur_pair = None
        for i in ones:
            h_b[i] = 0
            for j in zeros:
                h_b[j] = 1
                pk = float(np.max(compute_corr(h_b)))
                if pk < cur_best:
                    cur_best = pk
                    cur_pair = (i, j)
                h_b[j] = 0
            h_b[i] = 1
        if cur_pair is None:
            break
        i, j = cur_pair
        h_b[i] = 0
        h_b[j] = 1
        best_pk = cur_best
        best_h = h_b.copy()
    return best_h, best_pk


def ils_search(h_b, N, deadline, rng, n_restarts=30):
    best_h = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    for r in range(n_restarts):
        if time.time() > deadline - 3:
            break
        h_cur = best_h.copy()
        n_pert = rng.randint(2, 7)
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        for _ in range(n_pert):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
        dl = min(deadline - 2, time.time() + 25)
        h_new, pk_new = hill_climb(h_cur, N, dl, max_rounds=30)
        if pk_new < best_pk:
            best_pk = pk_new
            best_h = h_new
    return best_h, best_pk


def structured_perturb(h_b, N, rng, n_moves=3):
    h_cur = h_b.copy()
    corr = compute_corr(h_b)
    peak_idx = np.argmax(corr)
    center = N - 1
    shift = peak_idx - center
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    for _ in range(n_moves):
        if abs(shift) < N:
            if shift > 0:
                cand_i = ones[(ones + shift) % N < N]
                cand_j = zeros[zeros < N - shift]
                if len(cand_i) > 0 and len(cand_j) > 0:
                    i = cand_i[rng.randint(len(cand_i))]
                    j = cand_j[rng.randint(len(cand_j))]
                    h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                    h_cur = h_cur
                    ones = np.where(h_cur == 1)[0]
                    zeros = np.where(h_cur == 0)[0]
                    continue
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
    return h_cur


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_c5 = compute_c5(h0, n0)
    best_h = h0.copy()

    # Phase 1: Continuous Adam refinement
    if time.time() < deadline - 400:
        d1 = time.time() + min(80, (deadline - time.time()) * 0.12)
        h1, c1 = adam_optimize(h0.copy(), n0, n0 / 2.0, d1, 25, 0.05, 0.99)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        # Second Adam pass with different k
        if time.time() < deadline - 350:
            d1b = time.time() + min(40, (deadline - time.time()) * 0.06)
            h1b, c1b = adam_optimize(best_h.copy(), n0, n0 / 2.0, d1b, 60, 0.03, 0.995)
            if c1b < best_c5:
                best_c5, best_h = c1b, h1b.copy()

    # Phase 2: Round to binary and hill climb
    h_b = to_binary(best_h, n0)
    c_b = compute_c5(h_b, n0)
    if c_b < best_c5:
        best_c5, best_h = c_b, h_b.copy()

    if time.time() < deadline - 250:
        hc, pk_hc = hill_climb(h_b.copy(), n0, time.time() + 40, max_rounds=60)
        c_hc = pk_hc * (2.0 / n0)
        if c_hc < best_c5:
            best_c5, best_h = c_hc, hc.copy()

    # Phase 3: ILS with structured perturbations
    if time.time() < deadline - 150:
        il_deadline = time.time() + min(180, (deadline - time.time()) * 0.3)
        best_bin = best_h if len(best_h) == n0 else h_b.copy()
        if not (best_bin[0] in (0.0, 1.0)):
            best_bin = to_binary(best_bin, n0)
        h_il, pk_il = ils_search(best_bin.copy(), n0, il_deadline, rng, n_restarts=25)
        c_il = pk_il * (2.0 / n0)
        if c_il < best_c5:
            best_c5, best_h = c_il, h_il.copy()

    # Phase 4: Try other resolutions
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    other_Ns = [460, 550, 420, 680]
    for N in other_Ns:
        if time.time() > deadline - 80:
            break
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, best_h)
        h_init = project_sum(h_init, target)

        # Adam
        if time.time() < deadline - 60:
            d = time.time() + min(35, (deadline - time.time()) * 0.1)
            h_ad, c_ad = adam_optimize(h_init, N, target, d, 20, 0.04, 0.99)
            h_bin = to_binary(h_ad, N)
            c_bin = compute_c5(h_bin, N)
            if c_bin < c_ad:
                h_ad = h_bin
                c_ad = c_bin
            if c_ad < best_c5:
                best_c5, best_h = c_ad, h_ad.copy()

        # Hill climb
        if time.time() < deadline - 50:
            h_bin = to_binary(h_ad, N)
            hc2, pk2 = hill_climb(h_bin, N, min(deadline - 40, time.time() + 25), max_rounds=35)
            c2 = pk2 * (2.0 / N)
            if c2 < best_c5:
                best_c5, best_h = c2, hc2.copy()

        # ILS
        if time.time() < deadline - 35:
            h_src = hc2 if c2 < best_c5 else h_bin
            il_dl = min(deadline - 25, time.time() + (deadline - time.time()) * 0.12)
            h_il2, pk_il2 = ils_search(h_src, N, il_dl, rng, n_restarts=15)
            c_il2 = pk_il2 * (2.0 / N)
            if c_il2 < best_c5:
                best_c5, best_h = c_il2, h_il2.copy()

    # Phase 5: Final polish on best solution
    Nb = len(best_h)
    is_binary = np.all((best_h == 0.0) | (best_h == 1.0))
    if is_binary and time.time() < deadline - 15:
        hf, pkf = hill_climb(best_h.copy(), Nb, time.time() + 20, max_rounds=40)
        cf = pkf * (2.0 / Nb)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()
        if time.time() < deadline - 8:
            hf2, pkf2 = ils_search(best_h.copy(), Nb, deadline - 3, rng, n_restarts=10)
            cf2 = pkf2 * (2.0 / Nb)
            if cf2 < best_c5:
                best_c5, best_h = cf2, hf2.copy()
    elif not is_binary and time.time() < deadline - 10:
        hF = best_h.copy() + rng.randn(Nb) * 0.02
        hF = project_sum(hF, Nb / 2.0)
        hF, cF = adam_optimize(hF, Nb, Nb / 2.0, deadline - 3, 8, 0.02, 0.995)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
