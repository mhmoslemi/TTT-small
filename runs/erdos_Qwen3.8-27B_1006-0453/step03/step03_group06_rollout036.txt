import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return project_simple(h, target)
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def project_simple(h, target):
    """Simple projection: distribute uniformly among non-saturated coords"""
    h = h.copy()
    s = h.sum()
    diff = target - s
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    if free.sum() > 0:
        h[free] += diff / free.sum()
    else:
        # All saturated, try adding to zeros or removing from ones
        if diff > 0:
            zeros = (h == 0)
            h[zeros] += diff / max(1, zeros.sum())
        else:
            ones = (h == 1)
            h[ones] += diff / max(1, ones.sum())
    return np.clip(h, 0, 1)

def coord_sweep(h, n, lam, n_grid=300):
    """One pass of coordinate descent with Lagrange penalty."""
    h = h.copy()
    for j in range(n):
        old_val = h[j]
        h[j] = 0.0
        corr0 = np.correlate(h, 1.0 - h, mode='full')
        h[j] = 1.0
        corr1 = np.correlate(h, 1.0 - h, mode='full')
        h[j] = old_val
        slopes = corr1 - corr0
        c = slopes + lam
        d = corr0
        X = np.linspace(0.0, 1.0, n_grid)
        G = c[:, None] * X[None, :] + d[:, None]
        g = G.max(axis=0)
        h[j] = X[g.argmin()]
    return h

def lam_sweep(h, n, n_grid=300, n_outer=30):
    """Bisection on lambda, inner coordinate sweeps."""
    target_sum = n / 2.0
    lam_lo, lam_hi = -10.0, 10.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for outer in range(n_outer):
        lam = (lam_lo + lam_hi) / 2.0
        h_new = h
        for inner in range(5):
            h_new = coord_sweep(h_new, n, lam, n_grid)
        s = h_new.sum()
        if s > target_sum:
            lam_lo = lam
        else:
            lam_hi = lam
        if abs(s - target_sum) < 1e-6:
            break
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def rounds_to_binary(h, n, threshold=0.15):
    """Push values toward 0 or 1, maintain sum approximately."""
    h = h.copy()
    changes = 0
    for _ in range(100):
        below = h < threshold
        above = h > 1.0 - threshold
        if not (below.any() or above.any()):
            break
        target = n / 2.0
        new_sum = h.sum()
        if below.any():
            new_sum -= below.sum() * (h[below] + (1.0 - threshold)) / 2.0  # approx
        # Simpler: just clip and re-project
        h = np.where(h < threshold, 0.0, h)
        h = np.where(h > 1.0 - threshold, 1.0, h)
        h = project_feasible(h, n)
        break
    return h

            pass

        val1 = np.zeros(2*n-1)
        lo1 = max(0, (n-1)-j)
        hi1 = min(2*n-1, (n-1)+j+1)
        if lo1 < hi1:

            h_idx1 = j + idx1 - (n-1)
            val1[idx1] = 1.0 - h[h_idx1]

        lo2 = max(0, j)

        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            h_idx2 = j - idx2 + (n-1)
            val2[idx2] = -h[h_idx2]

        # corr0 = corr - old_val * slopes
        corr0 = corr - old_val * slopes

        c = slopes + lam
        X = np.linspace(0.0, 1.0, n_grid)
        G = c[:, None] * X[None, :] + corr0[:, None]
        g = G.max(axis=0)
        new_val = X[g.argmin()]
        
        # Update
        h[j] = new_val
        corr += (new_val - old_val) * slopes
    
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)

                h_try = project_feasible(h_try, n_try)
            else:
                h_try = h_interp + np.random.randn(n_try) * 0.04
                h_try = project_feasible(h_try, n_try)
            h_opt = lam_sweep(h_try, n_try, 400, 45, 5)
            h_opt = project_feasible(h_opt, n_try)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h, best_n = c5_opt, h_opt, n_try

    # Phase 3: Random restarts with varying scales
    for r in range(40):
        if elapsed() > budget_s * 0.78:
            break
        scale = 0.02 + 0.06 * (1.0 - r / 40.0)
        h_pert = best_h + np.random.randn(best_n) * scale
        h_pert = project_feasible(h_pert, best_n)
        h_opt = lam_sweep(h_pert, best_n, 300, 30, 4)
        h_opt = project_feasible(h_opt, best_n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    # Phase 4: Block and swap perturbations
    for r in range(25):
        if elapsed() > budget_s * 0.88:
            break
        h_pert = best_h.copy()
        r_type = np.random.rand()
        if r_type < 0.5:
            blen = np.random.randint(3, max(4, best_n // 6))
            start = np.random.randint(0, best_n - blen)
            h_pert[start:start + blen] += np.random.choice([-1, 1]) * 0.08
        elif r_type < 0.8:
            i1 = np.random.randint(0, best_n)
            i2 = np.random.randint(0, best_n)
            swap_amt = np.random.uniform(0.03, 0.12)
            h_pert[i1] += swap_amt
            h_pert[i2] -= swap_amt
        else:
            h_pert = best_h + np.random.uniform(-0.05, 0.05, best_n)
        h_pert = project_feasible(h_pert, best_n)
        h_opt = lam_sweep(h_pert, best_n, 350, 35, 5)
        h_opt = project_feasible(h_opt, best_n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    # Phase 5: Fine polish with increasing grid resolution
    for ng in [500, 900, 1500, 2500]:
        if elapsed() > budget_s * 0.99:
            break
        h_opt = lam_sweep(best_h, best_n, ng, 50, 8)
        h_opt = project_feasible(h_opt, best_n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
