                ht[i] = tt
                ht = project_sum(ht, target)
                c5t = compute_c5(ht, dx)
                if c5t < best_lc - 1e-13:
                    best_lc = c5t
                    best_local = tt
                    ht_save = ht.copy()
            if best_lc < best_c5 - 1e-13:
                best_h = ht_save
                best_c5 = best_lc
                improved = True
    return best_h, best_c5

def multi_scale_optimize(h_par, n_par, target_scales, deadline, rng):
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)
    for N_try in target_scales:
        if N_try <= 0 or time.time() > deadline - 30:
            break
        sub_dead = time.time() + (deadline - time.time()) / max(1, len(target_scales)) - 5
        xN = np.linspace(0.0, 2.0, N_try, endpoint=False)
        h_init = np.interp(xN, x_par, h_par)
        target = N_try / 2.0
        h_init = project_sum(h_init, target)
        h_init += rng.randn(N_try) * 0.03
        h_init = project_sum(h_init, target)
        h1, c1 = optimize_phase(h_init, N_try, target, time.time() + (sub_dead - time.time()) * 0.5, 1)
        h2, c2 = plateau_optimize(h1, N_try, target, sub_dead)
        hc = h2 if c2 < c1 else h1
        cc = min(c2, c1)
        if N_try == n_par:
            if cc < best_c5:
                best_c5, best_h = cc, hc
        else:
            x_par2 = np.linspace(0.0, 2.0, n_par, endpoint=False)
            h_mapped = np.interp(x_par2, xN, hc)
            h_mapped = project_sum(h_mapped, n_par / 2.0)
            cm = compute_c5(h_mapped, 2.0 / n_par)
            if cm < best_c5:
                best_c5, best_h = cm, h_mapped
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    dx_par = 2.0 / n_par
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, dx_par)

    total = deadline - t0
    t_phase1 = t0 + total * 0.30
    t_phase2 = t0 + total * 0.55
    t_phase3 = t0 + total * 0.85

    # Phase 1: Smooth log-sum-exp optimization
    h1, c1 = optimize_phase(best_h.copy(), n_par, n_par / 2.0, t_phase1, 1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Plateau-focused optimization
    h2, c2 = plateau_optimize(best_h.copy(), n_par, n_par / 2.0, t_phase2)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Multi-scale exploration
    scales = [n_par + 60, n_par - 30]
    h3, c3 = multi_scale_optimize(best_h, n_par, scales, t_phase3, rng)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Final tight optimization on best solution
    if time.time() < deadline - 50:
        Nb = len(best_h)
        h4, c4 = plateau_optimize(best_h.copy(), Nb, Nb / 2.0, time.time() + 40)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Coordinate polish
    if time.time() < deadline - 3:
        Nb2 = len(best_h)
        best_h, best_c5 = coord_polish(best_h, Nb2, Nb2 / 2.0, deadline - 2)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
