import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def lse_grad_vec(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-13:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad[:] += wi * (1.0 - h)
                grad[:] -= wi * h
            else:
                grad[:n - k] += wi * (1.0 - h[k:])
                grad[k:] -= wi * h[:n - k]
        else:
            kp = -k
            grad[kp:] += wi * (1.0 - h[:n - kp])
            grad[:n - kp] -= wi * h[kp:]
    return grad * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def sum_eq_grad(x):
    return np.ones(len(x))

def slsqp_smooth(h, T, maxiter=600):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_grad}]
    def obj(x):
        return lse_value(x, T)
    def grd(x):
        return lse_grad_vec(x, T)
    try:
        res = minimize(obj, h, jac=grd, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_momentum(h, n_iters=2000, top_m=8):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.04
    mom = 0.8
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(h_cur)
        m = np.max(corr)
        thresh = m - 0.02 * m
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        active = active[:top_m]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] += (1.0 - h_cur)
                    grad[:] -= h_cur
                else:
                    grad[:n - k] += (1.0 - h_cur[k:])
                    grad[k:] -= h_cur[:n - k]
            else:
                kp = -k
                grad[kp:] += (1.0 - h_cur[:n - kp])
                grad[:n - kp] -= h_cur[kp:]
        grad /= len(active)
        v = mom * v - lr * grad
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.3)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if stall > 300:
                break
    return best_h, best_c5

def periodic_init(n, period, phase=0):
    x = np.arange(n)
    h = np.sin(2.0 * np.pi * (x + phase) / period)
    h = np.clip(0.5 + 0.5 * h, 0, 1)
    return project_feasible(h, n)

def active_set_polish(h, n_lags=5, maxiter=400):

    _, corr = compute_c5(h)
    m = np.max(corr)
    thresh = m - 0.05 * m
    active = np.where(corr >= thresh)[0][:n_lags]
    if len(active) == 0:

    bnds = [(0.0, 1.0)] * n

        sel = c[active]
        mm = np.max(sel)
        return (mm + np.log(np.sum(np.exp(T * (sel - mm)))) / T) * (2.0 / n)
    def grd(x):

        sel = c[active]

                else:
                    grad[:n - k] += wi * (1.0 - x[k:])
                    grad[k:] -= wi * x[:n - k]

                kp = -k
                grad[kp:] += wi * (1.0 - x[:n - kp])
                grad[:n - kp] -= wi * x[kp:]
        return grad * (2.0 / n)
    try:
        res = minimize(obj, h, jac=grd, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
