import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14: break
            h *= (target / total) if total > 1e-14 else 1.0
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_penalty(x, T, k, n, pw):
    h = np.clip(x, 0, 1)
    pen = pw * (h.sum() - n/2.0)**2
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k > 0 and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        c = corr[idx]
    else:
        c = corr
    m = c.max()
    lse = m + np.log(np.exp(T*(c-m)).sum())/T
    return lse * dx + pen

def lbfqsb(h0, T, k, n, maxit, pw):
    bnds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_max_penalty(x, T, k, n, pw)
    try:
        res = minimize(f, h0, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except:
        return h0

def targeted_descent(h, n, n_steps=500):
    """Coordinate descent targeting worst shifts"""
    best = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0-h, mode='full')
        # Top 3 worst shifts
        top3 = np.argsort(corr)[-3:]
        for lag_idx in top3:

            alpha = 0.01 * (1.0 - step/n_steps) + 0.001
            # For this lag, find pairs contributing most
            if lag >= 0:

                    if contrib > 0.1:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h *= (target / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, m):
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g

def smooth_obj(x, T, k, n, pw):
    h = np.clip(x, 0, 1)
    pen = pw * (h.sum() - n / 2.0) ** 2
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if 0 < k < len(corr):
        idx = np.argpartition(corr, -k)
        c = corr[idx[-k:]]
    else:
        c = corr
    m = c.max()
    return (m + np.log(np.exp(T * (c - m)).sum()) / T) * dx + pen

def lbfgsb_opt(h0, T, k, n, maxit, pw):
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T, k, n, pw), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-9})
    return project_feasible(res.x, n)

def subgrad(h, n, steps, top_k, alpha0):
    best = h.copy()
    best_c5 = compute_c5(h)
    for s in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        wts = corr[idx]
        ws = wts.sum()
        wts = wts / ws if ws > 1e-14 else np.ones(top_k) / top_k
        grad = np.zeros(n)
        for w, ii in zip(wts, idx):
            lag = abs(int(ii - (n - 1)))
            grad += w * grad_lag(h, lag)
        alpha = alpha0 * (0.999 ** s)
        h = project_feasible(h - alpha * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: L-BFGS-B with various objectives
    temps = [50, 200, 1000, 5000]
    kvals = [0, 3, 5, 8, 15]
    for T in temps:

            hh, cc = subgrad(best_h, best_n, 500, tk, 0.02)

            if r < 0.4:

            elif r < 0.7:

The smooth objective (for the full correlation, not just top-k):

∂f/∂h[k] = dx * Σ_j p_j * ∂c_j/∂h[k]

And ∂c_j/∂h[k] is the gradient of the correlation at lag j, which is grad_lag(h, |j-(n-1)|) as computed before.

So: ∂f/∂h = dx * Σ_j p_j * grad_lag(h, |j-(n-1)|)

This is a weighted sum of lag gradients, with weights given by the softmax of the correlation values.

This is O(n^2) to compute (n lags, each gradient is O(n)). But we can optimize:
- Only the top few lags have significant p_j weight
- For very large T, p_j ≈ one-hot at the max

def smooth_obj_grad(x, T, k, n, pw):
    h = np.clip(x, 0, 1)
    # Compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    
    # Select top-k or all
    if 0 < k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        c = np.zeros(len(corr))
        c[idx] = corr[idx]
    else:
        c = corr.copy()
    
    # Softmax
    m = c.max()
    exp_c = np.exp(T * (c - m))
    p = exp_c / exp_c.sum()
    
    # Value
    val = (m + np.log(exp_c.sum()) / T) * dx
    
    # Penalty
    pen_val = pw * (h.sum() - n/2.0)**2
    pen_grad = 2 * pw * (h.sum() - n/2.0)
    
    # Gradient
    grad = np.zeros(n)
    for j in range(len(corr)):
        if p[j] > 1e-10:
            lag = abs(j - (n-1))
            grad += p[j] * grad_lag(h, lag)
    grad *= dx
    grad += pen_grad  # penalty gradient (uniform)
    
    return val + pen_val, grad
