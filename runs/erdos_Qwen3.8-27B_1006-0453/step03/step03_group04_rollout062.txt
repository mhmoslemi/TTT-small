    cur_pk = best_pk
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    total_est = max(int((deadline - time.time()) * 35), 3000)
    no_improve = 0
    patience = 6000
    for it in range(total_est):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / total_est)
        if rng.random() < 0.6:
            corr = compute_corr(h_b)
            peak_idx = int(np.argmax(corr))
            L = peak_idx - (N - 1)
            if L > 0:
                cands = [n for n in range(N - L) if h_b[n + L] == 1 and h_b[n] == 0]
                if cands:
                    i = cands[rng.randint(len(cands))]
                    j = zeros[rng.randint(len(zeros))]
                    h_b[i], h_b[j] = 0, 1
                    new_pk = float(np.max(compute_corr(h_b)))
                    delta = new_pk - cur_pk
                    if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / T)):
                        cur_pk = new_pk
                        ones = np.where(h_b == 1)[0]
                        zeros = np.where(h_b == 0)[0]
                        if new_pk < best_pk:
                            best_pk = new_pk
                            best = h_b.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                        continue
                    else:
                        h_b[i], h_b[j] = 1, 0
                        no_improve += 1
                        continue
            elif L < 0:
                M = -L
                cands = [n for n in range(N - M) if h_b[n] == 1 and h_b[n + M] == 0]
                if cands:
                    j = cands[rng.randint(len(cands))]
                    i = zeros[rng.randint(len(zeros))]
                    h_b[i], h_b[j] = 0, 1
                    new_pk = float(np.max(compute_corr(h_b)))
                    delta = new_pk - cur_pk
                    if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / T)):
                        cur_pk = new_pk
                        ones = np.where(h_b == 1)[0]
                        zeros = np.where(h_b == 0)[0]
                        if new_pk < best_pk:
                            best_pk = new_pk
                            best = h_b.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                        continue
                    else:
                        h_b[i], h_b[j] = 1, 0
                        no_improve += 1
                        continue
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = 0, 1
        new_pk = float(np.max(compute_corr(h_b)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_b[i], h_b[j] = 1, 0
            no_improve += 1
        if no_improve > patience:
            no_improve = 0
            for _ in range(10):
                i2 = ones[rng.randint(len(ones))]
                j2 = zeros[rng.randint(len(zeros))]
                h_b[i2], h_b[j2] = 0, 1
            cur_pk = float(np.max(compute_corr(h_b)))
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
    return best, best_pk * (2.0 / N)

def quantize_binary(h, N):
    order = np.argsort(-h)
    h_b = np.zeros(N)
    h_b[order[:N // 2]] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    target = N / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N)

    # Phase 1: Quick Adam to get a smoother starting point
    d1 = t0 + 40
    h1 = best_h + rng.randn(N) * 0.02
    h1 = project_sum(h1, target)
    h1, c1 = adam_run(h1, N, target, d1, k=60, lr0=0.03, decay=0.995)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Intensive binary annealing (random swaps)
    h_b2 = quantize_binary(best_h, N)
    d2 = t0 + 370
    h_b2, c2 = binary_anneal(h_b2, N, d2, rng, T0=5.0, T1=0.0008)
    if c2 < best_c5:
        best_c5, best_h = c2, h_b2.copy()

    # Phase 3: Directed annealing targeting peak lags
    h_b3 = quantize_binary(best_h, N)
    d3 = t0 + 570
    h_b3, c3 = directed_anneal(h_b3, N, d3, rng, T0=3.0, T1=0.0008)
    if c3 < best_c5:
        best_c5, best_h = c3, h_b3.copy()

    # Phase 4: Continuous refinement around best binary
    h4 = best_h.astype(float) + rng.randn(N) * 0.025
    h4 = project_sum(h4, target)
    d4 = t0 + 720
    h4, c4 = adam_run(h4, N, target, d4, k=80, lr0=0.012, decay=0.997)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Another binary round from refined solution
    h_b5 = quantize_binary(best_h, N)
    d5 = t0 + 900
    h_b5, c5 = binary_anneal(h_b5, N, d5, rng, T0=2.0, T1=0.0005)
    if c5 < best_c5:
        best_c5, best_h = c5, h_b5.copy()

    # Phase 6: Fine continuous polish
    if time.time() < deadline - 10:
        h6 = best_h.astype(float) + rng.randn(N) * 0.01
        h6 = project_sum(h6, target)
        h6, c6 = adam_run(h6, N, target, deadline - 3, k=100, lr0=0.005, decay=0.998)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    # Phase 7: Try N=460 if time permits
    if time.time() < deadline - 80:
        N2 = 460
        target2 = N2 / 2.0
        x0 = np.linspace(0.0, 2.0, N, endpoint=False)
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h_t = np.interp(x2, x0, best_h)
        h_t = project_sum(h_t, target2)
        h_bt = quantize_binary(h_t, N2)
        d7 = time.time() + (deadline - time.time()) * 0.45
        h_bt, c7 = binary_anneal(h_bt, N2, d7, rng, T0=3.0, T1=0.001)
        if c7 < best_c5:
            best_c5, best_h = c7, h_bt.copy()
        if time.time() < deadline - 10:
            h8 = best_h.astype(float) + rng.randn(len(best_h)) * 0.015
            h8 = project_sum(h8, len(best_h) / 2.0)
            h8, c8 = adam_run(h8, len(best_h), len(best_h) / 2.0, deadline - 2, k=60, lr0=0.01, decay=0.997)
            if c8 < best_c5:
                best_c5, best_h = c8, h8

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
