def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0/n
    C = np.correlate(h, 1-h, mode='full')  # length 2n-1
    m = np.max(C)
    w = np.exp(T*(C - m))
    w /= np.sum(w)
    
    # grad[i] = dx/T * Σ_k w[k] * (∂C[k]/∂h[i])
    # ∂C[n-1+k]/∂h[i] = (1-h[i+k])*(i+k<n) - h[i-k]*(i-k>=0)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        if w[n-1+k] < 1e-15:
            continue
        wk = w[n-1+k]
        # First term: (1-h[i+k]) for valid i
        if k >= 0:
            for i in range(n-k):
                grad[i] += wk * (1 - h[i+k])
        else:
            for i in range(-k, n):
                grad[i] += wk * (1 - h[i+k])
        # Second term: -h[i-k] for valid i
        if k >= 0:
            for i in range(k, n):
                grad[i] -= wk * h[i-k]
        else:
            for i in range(0, n+k):
                grad[i] -= wk * h[i-k]
    
    return grad * dx / T, float((m + np.log(np.sum(np.exp(T*(C-m))))/T) * dx)

```python
def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0/n
    C = np.correlate(h, 1-h, mode='full')
    m = np.max(C)
    w = np.exp(T*(C - m))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # For each shift k, the gradient contribution is:
    # (1-h shifted by -k) - (h shifted by k)
    # where "shifted" means with zeros padding
    
    # First term: for shift k, add w[n-1+k] * (1-h[i+k]) to grad[i]
    # This is: grad += sum_k w[n-1+k] * pad(1-h[i+k])
    # Second term: for shift k, subtract w[n-1+k] * h[i-k] from grad[i]
    # This is: grad -= sum_k w[n-1+k] * pad(h[i-k])
    
    # Vectorized:
    for k in range(-(n-1), n):
        if w[n-1+k] < 1e-15:
            continue
        wk = w[n-1+k]
        # First term
        ik = np.arange(0, n-k) if k >= 0 else np.arange(-k, n)
        grad[ik] += wk * (1 - h[ik + k])
        # Second term
        ik2 = np.arange(k, n) if k >= 0 else np.arange(0, n+k)
        grad[ik2] -= wk * h[ik2 - k]
    
    val = (m + np.log(np.sum(np.exp(T*(C-m))))/T) * dx
    return val, grad
```

This is still O(n²) but vectorized inner loop. For n=200 and 399 shifts, it's 399 * 200 ≈ 80K operations. Fine.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-14:
            h += room * (diff / t)
    else:
        t = h.sum()
        if t > 1e-14:
            h += h * (diff / t)
    return np.clip(h, 0.0, 1.0)

def soft_max_val(corr, T):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(T * (corr - m)) + 1e-300)) / T)

def slsqp_opt(h0, T, maxiter=500, ftol=1e-16):
    n = len(h0)
    dx = 2.0 / n
    def obj(x):
        return soft_max_val(np.correlate(x, 1 - x, mode='full'), T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(res.x, n)
    except Exception:
        return h0.copy()

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(C)
    shifted = T * (C - m)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    val = (m + np.log(w_sum) / T) * dx
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[n - 1 + k]
        if wk < 1e-16:
            continue
        if k >= 0:
            idx1 = np.arange(0, n - k)
            grad[idx1] += wk * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= wk * h[idx2 - k]
        else:
            idx1 = np.arange(-k, n)
            grad[idx1] += wk * (1.0 - h[idx1 + k])
            idx2 = np.arange(0, n + k)
            grad[idx2] -= wk * h[idx2 - k]
    return val, grad * dx / T

    h = h0.copy()

    v_v = np.zeros(n)
    prev_val = compute_c5(h)

        it += 1
        ptype = it % 5
        frac = (time.time() - t0 - budget_s * 0.27) / max(1e-6, dl3 - t0 - budget_s * 0.27)
        step = 0.14 * max(0.1, 1.0 - frac) + 0.03
        if ptype == 0:
            h_p = bh_h + step * np.random.randn(n_bh)
        elif ptype == 1:
            h_p = bh_h.copy()
            bl = max(2, n_bh // (10 + it // 40))
            s1 = np.random.randint(0, n_bh - bl)
            s2 = np.random.randint(0, n_bh - bl)
            h_p[s1:s1 + bl], h_p[s2:s2 + bl] = h_p[s2:s2 + bl].copy(), h_p[s1:s1 + bl].copy()
        elif ptype == 2:
            h_p = bh_h.copy()
            nf = max(2, n_bh // (8 + it // 60))
            idx = np.random.choice(n_bh, nf, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        elif ptype == 3:
            h_p = (bh_h > 0.5).astype(float)
            nf = max(2, n_bh // 10)
            idx = np.random.choice(n_bh, nf, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        else:
            h_p = bh_h + step * 1.5 * np.random.randn(n_bh)
            h_p = (h_p > 0.5).astype(float)
        h_p = make_feasible(h_p, n_bh)
        T_use = min(6000, 100 + 300 * (it // 12))
        h_new = adam_opt(h_p, T_use, 0.015, 800)
        c5_new = compute_c5(h_new)
        temp = 0.004 * max(0.05, 1.0 - frac) + 0.0001
        if c5_new < bh_c5 or np.random.rand() < np.exp(-(c5_new - bh_c5) / temp):
            bh_c5 = c5_new
            bh_h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Multi-resolution (15%)
    dl4 = t0 + budget_s * 0.80
    for scale in [1.3, 1.6, 2.0]:
        if time.time() > dl4:
            break
        n_new = int(len(best_h) * scale)
        n_new = (n_new // 2) * 2
        if n_new < 60 or n_new > 600:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_int = make_feasible(np.interp(x_new, x_old, best_h), n_new)
        for T in [100, 500]:
            if time.time() > dl4:
                break
            h_new = adam_opt(h_int, T, 0.02, 1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 5: Final polish (15%)
    dl5 = t0 + budget_s * 0.99
    n_fin = len(best_h)
    for T in [3000, 12000, 50000]:
        if time.time() > dl5:
            break
        h_new = slsqp_opt(best_h.copy(), T, 3000, 1e-18)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    if time.time() < dl5:
        for _ in range(300):
            if time.time() > dl5:
                break
            improved = False
            for _ in range(40):
                j = np.random.randint(n_fin)
                delta = np.random.choice([-0.04, -0.015, 0.015, 0.04, -0.07, 0.07])
                h_try = best_h.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                h_try = make_feasible(h_try, n_fin)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
            if not improved:
                break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
