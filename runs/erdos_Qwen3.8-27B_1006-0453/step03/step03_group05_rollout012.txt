import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_grad(h, top_k=3):
    """Compute weighted gradient for top-k overlaps. O(n) per shift."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    w = corr[idx].astype(float)
    ws = w.sum()
    w = w / ws if ws > 0 else np.ones(top_k) / top_k
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        s = ki - (n - 1)
        if s > 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n-s]
        elif s < 0:
            s2 = -s
            grad[s2:] += wi * (1.0 - h[:n-s2])
            grad[:n-s2] -= wi * h[s2:]
        else:
            # s=0: corr = sum h[j](1-h[j]) = sum h - sum h^2
            # d/dh[i] = 1 - 2*h[i]
            grad += wi * (1.0 - 2.0 * h)
    return grad

def fast_subgrad(h, n_steps, top_k, alpha0, decay):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for it in range(n_steps):
        grad = topk_grad(h, top_k)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
        h = h_new
        alpha = alpha0 * (decay ** it)
    return best, best_c5

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_wrap(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def sa_optimize(h, n_steps, T_start, T_end, top_k=3):
    """Simulated annealing with smart moves."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    for it in range(n_steps):
        T = T_start * (T_end / T_start) ** (it / max(1, n_steps - 1))
        # Propose move
        r = np.random.random()
        if r < 0.3:
            # Small perturbation
            hp = h + np.random.randn(n) * 0.03
        elif r < 0.5:
            # Swap two values
            hp = h.copy()
            i, j = np.random.choice(n, 2, replace=False)
            hp[i], hp[j] = hp[j], hp[i]
        elif r < 0.7:
            # Block shift
            hp = h.copy()
            start = np.random.randint(0, n - 3)
            length = np.random.randint(2, min(15, n // 4))
            end = min(start + length, n)
            shift = np.random.randint(1, min(3, length))
            hp[start:end] = np.roll(hp[start:end], shift)
        else:
            # Targeted: move mass from high to low at worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            s_star = int(np.argmax(corr) - (n - 1))
            hp = h.copy()
            for j in range(n):
                j2 = j + s_star
                if 0 <= j2 < n:
                    if h[j] > 0.5 and h[j2] < 0.5:
                        d = min(h[j] * 0.2, 1.0 - h[j2])
                        hp[j] -= d
                        hp[j2] += d
                    elif h[j] < 0.5 and h[j2] > 0.5:
                        d = min((1.0 - h[j]) * 0.2, h[j2])
                        hp[j] += d
                        hp[j2] -= d
        hp = project_feasible(hp, n)
        c5_new = compute_c5(hp)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h = hp
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best = h.copy()
    return best, best_c5

def make_symmetric_start(n, n_periods):
    """Create a starting point with n_periods identical blocks."""
    block_len = n // n_periods
    # Create one block with a bump-like shape
    x = np.linspace(0, 1, block_len)
    block = 0.5 + 0.4 * np.sin(2 * np.pi * x)
    # Tile
    h = np.tile(block, n_periods)
    if len(h) < n:
        h = np.concatenate([h, h[:n - len(h)]])
    h = h[:n]
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_gradient(h, corr, top_k):
    n = len(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    w = corr[idx].astype(float)
    ws = w.sum()
    w = w / ws if ws > 0 else np.ones(top_k) / top_k
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        s = ki - (n - 1)
        if s > 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n-s]
        elif s < 0:
            s2 = -s
            grad[s2:] += wi * (1.0 - h[:n-s2])
            grad[:n-s2] -= wi * h[s2:]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad

def fast_subgrad(h, n_steps, top_k, alpha0, decay):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    alpha = alpha0
    stuck = 0
    for it in range(n_steps):
        grad = topk_gradient(h, corr, top_k)
        h_new = project_feasible(h - alpha * grad, n)
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                alpha *= 0.7
                stuck = 0
        h = h_new
        alpha = alpha0 * (decay ** it)
    return best, best_c5

def sa_optimize(h, n_steps, T_start, T_end, top_k=3):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    c5_cur = best_c5
    for it in range(n_steps):
        T = T_start * (T_end / T_start) ** (it / max(1, n_steps - 1))
        r = np.random.random()

            sig = 0.015 + 0.02 * (T / T_start)
            hp = h + np.random.randn(n) * sig
        elif r < 0.65:
            corr_cur = np.correlate(h, 1.0 - h, mode='full')
            s_star = int(np.argmax(corr_cur) - (n - 1))
            hp = h.copy()
            for j in range(n):
                j2 = j + s_star
                if 0 <= j2 < n:
                    if h[j] > 0.5 and h[j2] < 0.5:
                        d = min(h[j] * 0.15, 1.0 - h[j2])
                        hp[j] -= d
                        hp[j2] += d
                    elif h[j] < 0.5 and h[j2] > 0.5:
                        d = min((1.0 - h[j]) * 0.15, h[j2])
                        hp[j] += d
                        hp[j2] -= d
        elif r < 0.80:
            hp = h.copy()
            start = np.random.randint(0, n - 3)
            length = np.random.randint(3, min(20, n // 3))
            end = min(start + length, n)
            shift = np.random.choice([-2, -1, 1, 2])
            hp[start:end] = np.roll(hp[start:end], shift)
        else:
            hp = h.copy()
            i, j = np.random.choice(n, 2, replace=False)
            hp[i], hp[j] = hp[j], hp[i]
        hp = project_feasible(hp, n)

            h = hp
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best = h.copy()
                corr = np.correlate(h, 1.0 - h, mode='full')
    return best, best_c5

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_avg_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_wrap(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def make_periodic_start(n, n_periods, phase_shift=0.0):
    block_len = n // n_periods
    x = np.linspace(0, 1, block_len + 1)[:-1]
    block = 0.5 + 0.35 * np.sin(2 * np.pi * (x + phase_shift))
    h = np.tile(block, n_periods)
    if len(h) < n:
        h = np.concatenate([h, block[:n - len(h)]])
    h = h[:n]
    return project_feasible(h, n)
