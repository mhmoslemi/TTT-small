                                 time.time() + (deadline - time.time()) * 0.45,
                                 tau0=20.0, tau1=300.0, K=min(15, 2*N-1),
                                 lr0=0.012, lr1=0.001)
            if cr < best_c5:
                best_c5 = cr; best_h = hr.copy()
            if cr < 0.381:
                all_good.append((hr.copy(), cr, N))

    # Average multiple good solutions at same N for improvement
    if len(all_good) > 1:
        by_n = {}
        for ah, ac, an in all_good:
            if an not in by_n:
                by_n[an] = []
            by_n[an].append(ah)
        for N in by_n:
            if time.time() > deadline - 20:
                break
            hs = by_n[N]
            if len(hs) < 2:
                continue
            dx = 2.0 / N
            target = N / 2.0
            avg = np.mean(hs, axis=0)
            avg = project_sum(avg, target)
            c_avg = compute_c5(avg, dx)
            if c_avg < best_c5:
                best_c5 = c_avg; best_h = avg.copy()
            # Polish the average
            if time.time() < deadline - 15:
                hp, cp = optimize(avg.copy(), N, target, dx,
                                  time.time() + (deadline - time.time()) * 0.5,
                                  n_act=3, lr0=0.0005, lr1=0.00002)
                if cp < best_c5:
                    best_c5 = cp; best_h = hp.copy()

    # Final ultra-fine polish on best
    Nb = len(best_h)
    if time.time() < deadline - 4:
        dx_b = 2.0 / Nb
        hf, cf = optimize(best_h.copy(), Nb, Nb // 2, dx_b,
                          deadline - 1, n_act=1, lr0=0.0001, lr1=0.000005)
        if cf < best_c5:
            best_c5 = cf; best_h = hf

    fc = compute_c5(best_h, 2.0 / len(best_h))
    if fc < best_c5:
        best_c5 = fc
    return best_h, best_c5, len(best_h)
