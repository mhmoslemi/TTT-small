                j = gj[rng.randint(len(gj))] if len(gj) > 0 else zeros[rng.randint(len(zeros))]
            elif L < 0 and -L < N:
                M = -L
                vi = zeros[zeros + M < N]
                ci = vi[cur[vi + M] == 1]
                i = ci[rng.randint(len(ci))] if len(ci) > 0 else zeros[rng.randint(len(zeros))]
                vj = ones[ones + M < N]
                gj = vj[cur[vj] == 0]
                j = gj[rng.randint(len(gj))] if len(gj) > 0 else ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            cur[i], cur[j] = cur[j], cur[i]
            np_ = float(np.max(compute_corr(cur)))
            delta = np_ - cur_peak
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
            cur[i], cur[j] = cur[j], cur[i]
        if best_delta < -1e-10 and best_i >= 0:
            cur[best_i], cur[best_j] = cur[best_j], cur[best_i]
            cur_peak = float(np.max(compute_corr(cur)))
            no_improve = 0
            if cur_peak < best_peak:
                best_peak = cur_peak
                best = cur.copy()
                best_c5 = cur_peak * dx
                if target_peak is not None and best_peak <= target_peak:
                    return best, best_c5
        else:
            no_improve += 1
            if no_improve > 500:
                # Perturb and restart hill climbing
                for _ in range(30):
                    oi = np.where(cur == 1)[0]
                    oz = np.where(cur == 0)[0]
                    a = oi[rng.randint(len(oi))]
                    b = oz[rng.randint(len(oz))]
                    cur[a], cur[b] = cur[b], cur[a]
                cur_peak = float(np.max(compute_corr(cur)))
                no_improve = 0
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)
    # Multi-resolution: focus on N=500 for finer granularity
    N_list = [500]
    for N in N_list:
        if time.time() > deadline - 200:
            break
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, best_h)
        h_init = project_sum(h_init, target)
        # Phase 1: Nesterov smooth (40% of remaining)
        remain = deadline - time.time()
        d1 = time.time() + remain * 0.35
        h1, c1 = nesterov_smooth(h_init.copy(), N, target, d1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        # Phase 2: Adam refinement (20%)
        if time.time() < deadline - 100:
            d2 = time.time() + (deadline - time.time()) * 0.22
            h2, c2 = adam_run(h1.copy(), N, target, d2, 50, 0.03, 0.992)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()
        else:
            h2 = h1
        # Phase 3: Binary local search (30%)
        if time.time() < deadline - 50:
            d3 = time.time() + (deadline - time.time()) * 0.35
            # Target: get peak <= floor(best_c5 * N / 2)
            tp = int(np.floor(best_c5 * N / 2.0)) - 1
            h3, c3 = binary_local_search(h2, N, d3, rng, target_peak=tp)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
        # Phase 4: Final continuous polish
        if time.time() < deadline - 10:
            d4 = deadline - 3
            h4, c4 = adam_run(best_h.copy(), N, target, d4, 20, 0.02, 0.996)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()
    # Also try N=600 if time allows
    N2 = 600
    if time.time() < deadline - 80 and N2 != len(best_h):
        target2 = N2 / 2.0
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        hb = np.interp(x2, np.linspace(0.0, 2.0, len(best_h), endpoint=False), best_h)
        hb = project_sum(hb, target2)
        d = deadline - 10
        hA, cA = nesterov_smooth(hb, N2, target2, d, mu0=10.0, k_cap=100, lr0=0.03)
        if cA < best_c5:
            best_c5, best_h = cA, hA
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
