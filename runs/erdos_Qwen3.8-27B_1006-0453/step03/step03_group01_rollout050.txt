import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def corr_weights(h, N, tau, topK):
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    nc = len(corr)
    sc = tau * (corr - corr.max())
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 12.0) / 50.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    return corr, w, float(np.max(corr) * (2.0 / N))

def grad_weights(h, N, w):
    center = N - 1
    nc = len(w)
    grad = np.zeros(N)
    thresh = 1e-5
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def nesterov_opt(h, N, target, deadline, tau0, tau1, k0, k1, lr0, decay, period, mom0, mom1, accept_worse=False):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    x = h.copy()
    x_prev = h.copy()
    lr = lr0
    t_start = time.time()
    it = 0
    while time.time() < deadline - 0.5:
        it += 1
        frac = min(0.3 + 0.7 * it / max(800, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        topK = int(round(k0 + (k1 - k0) * frac))
        mom = mom0 + (mom1 - mom0) * frac

        y = x + mom * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        corr, w, c5_y = corr_weights(y, N, tau, topK)
        if c5_y < best_c5 - 1e-14:
            best_c5 = c5_y
            best_h = y.copy()
        elif accept_worse and it > 100:
            accept_p = 0.1 * np.exp(-frac * 5.0)
            if np.random.random() < accept_p and c5_y < best_c5 + 0.002:
                pass

        grad = grad_weights(y, N, w)
        x_new = y - lr * grad
        x_new = project_sum(x_new, target)
        x_prev = x
        x = x_new

        if it % period == 0:
            lr *= decay
            if lr < 1e-13:
                break
    c5_final = compute_c5(best_h, dx)
    if c5_final < best_c5:
        best_c5 = c5_final
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    n_passes = 0
    while time.time() < deadline and n_passes < 30:
        n_passes += 1
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = best_h[i]
            trials = (0.0, 1.0)
            for d in (0.01, 0.03, 0.06, 0.10, 0.15):
                trials = trials + (max(0.0, orig - d), min(1.0, orig + d))
            for tr in trials:
                if abs(tr - orig) < 1e-9:
                    continue
                best_h[i] = tr
                tmp = project_sum(best_h, target)
                c5 = compute_c5(tmp, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = tmp.copy()
                    improved = True
                    break
            else:
                best_h[i] = orig
        if not improved:
            break
    return best_h, best_c5

def block_polish(h, N, target, deadline, bsz=2):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    while time.time() < deadline:
        improved = False
        for st in range(0, N - bsz + 1):
            if time.time() > deadline:
                break
            idx = np.arange(st, st + bsz)
            orig = best_h[idx].copy()
            candidates = [np.zeros(bsz), np.ones(bsz)]
            for d in (0.05, 0.1):
                candidates.append(np.clip(orig + d, 0, 1))
                candidates.append(np.clip(orig - d, 0, 1))
            for cand in candidates:
                if np.max(np.abs(cand - orig)) < 1e-9:
                    continue
                best_h[idx] = cand
                tmp = project_sum(best_h, target)
                c5 = compute_c5(tmp, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = tmp.copy()
                    improved = True
                    break
            else:
                best_h[idx] = orig
        if not improved:
            break
    return best_h, best_c5
