import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    total = room.sum()
    if total < 1e-12:
        return h
    h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_val(corr, T):
    m = np.max(corr)
    s = np.sum(np.exp(T * (corr - m)))
    return m + np.log(max(s, 1e-300)) / T

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_val(corr, T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = np.sum(exps)
    w = exps / Z
    val = smooth_val(corr, T) * dx
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        L = j - n + 1
        wj = w[j]
        if wj < 1e-15:
            continue
        # First term: max(0,L) <= t <= min(n-1, n-1+L)
        t_lo1 = max(0, L)
        t_hi1 = min(n - 1, n - 1 + L)
        if t_lo1 <= t_hi1:
            grad[t_lo1:t_hi1+1] += wj * (1.0 - h[t_lo1-L:t_hi1+1-L])
        # Second term: max(0,-L) <= t <= min(n-1, n-1-L)
        t_lo2 = max(0, -L)
        t_hi2 = min(n - 1, n - 1 - L)
        if t_lo2 <= t_hi2:
            grad[t_lo2:t_hi2+1] -= wj * h[t_lo2+L:t_hi2+1+L]
    return val, grad * dx

def coord_descent(h, n, max_rounds=200, tol=1e-14):
    """Exact coordinate descent: for each coordinate, minimize the objective
    by varying only that coordinate while keeping others fixed."""
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    
    for rd in range(max_rounds):
        improved = False
        for i in range(n):
            # Fix all except i, vary h[i]
            # corr[j] as function of h[i]:
            # c[j] = A_j + h[i]*B_j (linear in h[i] for fixed others)
            # Actually c[j] = sum_k h[k](1-h[k-L]) which is not linear in h[i]
            # c[j] includes terms h[i](1-h[i-L]) and h[i+L](1-h[i]) and h[i-L](-h[i])...
            # Let me just do a 1D search
            
            old_val = h[i]
            # The contribution of h[i] to corr:
            # For each lag L, h[i] appears as:
            #   h[i] * (1 - h[i-L]) (first factor)
            #   h[i+L] * (1 - h[i]) (second factor, when the "i" in the second h is our index)
            #   -h[i] * h[i-L] (from expanding)
            #   -h[i+L] * h[i] (similarly)
            
            # Let's just compute corr at a few candidate values
            best_h_i = old_val
            best_c5_i = compute_c5(h)
            
            # Try a few candidate values
            candidates = [0.0, 1.0, old_val]
            # Also try values near old_val
            for dv in [-0.1, -0.05, -0.01, 0.01, 0.05, 0.1]:
                cv = old_val + dv
                if 0 <= cv <= 1:
                    candidates.append(cv)
            
            best_val = best_c5_i
            for cv in candidates:
                h[i] = cv
                # Adjust sum constraint: we need sum(h) = target
                # This is tricky in coordinate descent...
                # Let's just check without projection for now
                c5_try = compute_c5(h)
                s = h.sum()
                if abs(s - target) < 1e-10 and c5_try < best_val:
                    best_val = c5_try
                    best_h_i = cv
            h[i] = best_h_i
            if best_h_i != old_val:
                improved = True
        
        # Project after each round
        h = make_feasible(h, n)
        if not improved:
            break
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    total = room.sum()
    if total < 1e-13:
        return h
    h += room * (diff / total)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_val(corr, T):
    m = np.max(corr)
    s = np.sum(np.exp(T * (corr - m)))
    return m + np.log(max(s, 1e-300)) / T

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_val(corr, T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = np.sum(exps)
    w = exps / Z
    val = smooth_val(corr, T) * dx
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        L = j - n + 1
        wj = w[j]
        if wj < 1e-14:
            continue
        t_lo1 = max(0, L)
        t_hi1 = min(n - 1, n - 1 + L)
        if t_lo1 <= t_hi1:
            idx = np.arange(t_lo1, t_hi1 + 1)
            grad[idx] += wj * (1.0 - h[np.clip(idx - L, 0, n - 1)])
        t_lo2 = max(0, -L)
        t_hi2 = min(n - 1, n - 1 - L)
        if t_lo2 <= t_hi2:
            idx = np.arange(t_lo2, t_hi2 + 1)
            grad[idx] -= wj * h[np.clip(idx + L, 0, n - 1)]
    return val, grad * dx

def lbfgs_optimize(h0, n, T, maxiter, t_limit):
    try:
        res = minimize(lambda h: smooth_c5(h, T), h0.copy(),
                       jac=lambda h: smooth_c5_grad(h, T)[1],
                       method='L-BFGS-B', bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-24, 'gtol': 1e-18})
        h_new = make_feasible(res.x, n)
        if time.time() > t_limit:
            return h0
        c5v = compute_c5(h_new)
        return h_new, c5v
    except Exception:
        return h0

def binary_local_search(h0, n, t_limit, max_iter=50000):
    h = np.where(h0 > 0.5, 1.0, 0.0).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    temp = 0.003
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5_try = compute_c5(h)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h.copy()
            ones_idx, zeros_idx = np.where(h > 0.5), np.where(h < 0.5)
        else:
            if np.random.rand() < np.exp(-(c5_try - best_c5) / temp):
                ones_idx, zeros_idx = np.where(h > 0.5), np.where(h < 0.5)
            else:
                h[i], h[j] = h[j], h[i]
                ones_idx, zeros_idx = np.where(h > 0.5), np.where(h < 0.5)
        temp *= 0.9999
    return best_h, best_c5

def nesterov_momentum(h0, n, t_limit, T=3000, n_steps=3000):
    h = h0.copy()
    v = np.zeros(n)
    lr = 0.005
    beta = 0.9
    for step in range(n_steps):
        if time.time() > t_limit:
            break
        val, grad = smooth_c5_grad(h, T)
        v = beta * v + (1.0 - beta) * grad
        h = h - lr * v
        h = make_feasible(h, n)
        if step % 500 == 499:
            lr *= 0.9
            beta = min(0.99, beta + 0.001)
    return h, compute_c5(h)

def fourier_optimize(n, K, t_limit):
    def params_to_h(params):
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.full(n, 0.5)
        for k in range(1, K + 1):
            h += params[2*(k-1)] * np.cos(np.pi * k * x)
            h += params[2*(k-1)+1] * np.sin(np.pi * k * x)
        h = make_feasible(np.clip(h, 0.0, 1.0), n)
        return h
    
    def obj(params):
        return compute_c5(params_to_h(params))
    
    best_c5 = np.inf
    best_h = None
    for trial in range(5):
        if time.time() > t_limit:
            break
        p0 = np.random.uniform(-0.15, 0.15, 2 * K)
        try:
            res = minimize(obj, p0, method='Nelder-Mead',
                          options={'maxiter': 5000, 'fatol': 1e-14})
            h_try = params_to_h(res.x)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try
        except Exception:
            pass
    return best_h, best_c5
