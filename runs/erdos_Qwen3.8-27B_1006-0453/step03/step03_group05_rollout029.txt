    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14: break
            h += room * (diff / t)
        else:
            t = h.sum()
            if t < 1e-14: break
            h -= h * (-diff / t)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute correction to non-boundary elements
        idx = np.where((h > 1e-6) & (h < 1-1e-6))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h

    """Coordinate descent: optimize one element at a time."""
    h = h.copy()
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            # Compute gradient of each corr with respect to h[j]
            # d corr[n-1+m] / d h[j] = (1-h[j-m])_{valid} - h[j+m]_{valid}

            # h[j] appears in (1-h[i]) when i=j (i.e., shift m, j+m must be valid)
            for m in range(-(n-1), n):
                idx = n - 1 + m
                g = 0.0
                if 0 <= j - m < n:
                    g += (1.0 - h[j - m])
                if 0 <= j + m < n:
                    g -= h[j + m]
                grad[idx] = g

                h[j] = best_h0
                improved = True

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += room * (diff / t)
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h -= h * (-diff / t)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_at(h, j, n):
    grad = np.zeros(2 * n - 1)
    # Part 1: (1-h[j-m]) for m where 0<=j-m<n
    # j-m in [0,n-1] => m in [j-n+1, j], clipped to [-(n-1), n-1]
    m_lo = max(-(n-1), j-(n-1))
    m_hi = min(n-1, j)
    if m_lo <= m_hi:
        ms = np.arange(m_lo, m_hi+1)
        idx = n - 1 + ms
        h_idx = j - ms
        grad[idx] += (1.0 - h[h_idx])
    # Part 2: -h[j+m] for m where 0<=j+m<n
    # j+m in [0,n-1] => m in [-j, n-1-j], clipped to [-(n-1), n-1]
    m_lo2 = max(-(n-1), -j)
    m_hi2 = min(n-1, n-1-j)
    if m_lo2 <= m_hi2:
        ms2 = np.arange(m_lo2, m_hi2+1)
        idx2 = n - 1 + ms2
        h_idx2 = j + ms2
        grad[idx2] -= h[h_idx2]
    return grad

def coord_descent(h, n, n_sweeps, rng, n_top=5):
    h = h.copy()
    for sweep in range(n_sweeps):
        corr = get_corr(h)
        improved = False
        order = rng.permutation(n)
        for j in order:
            grad = compute_grad_at(h, j, n)
            k = min(n_top, 2*n-1)
            top_idx = np.argpartition(corr, -(k))[-(k):]
            c_top = corr[top_idx]
            g_top = grad[top_idx]
            
            candidates = [0.0, 1.0]
            for i in range(min(3, len(c_top))):
                for ii in range(i+1, min(5, len(c_top))):
                    dg = g_top[ii] - g_top[i]
                    if abs(dg) > 1e-12:
                        h_eq = h[j] + (c_top[i] - c_top[ii]) / dg
                        if 0.0 < h_eq < 1.0:
                            candidates.append(h_eq)
            
            best_h0 = h[j]
            best_val = float(corr.max())
            for h0 in candidates:
                dm = (h0 - h[j])
                if abs(dm) < 1e-14:
                    continue
                new_max = float((corr + dm * grad).max())
                if new_max < best_val:
                    best_val = new_max
                    best_h0 = h0
            
            if abs(best_h0 - h[j]) > 1e-13:
                corr += (best_h0 - h[j]) * grad
                h[j] = best_h0
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h)

def round_to_binary(h, n):
    """Round h to binary, preserving sum = n/2."""
    target = n / 2
    h_bin = (h >= 0.5).astype(float)
    # Fix sum
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        # If need more ones: turn some 0s with high h value to 1
        # If need fewer: turn some 1s with low h value to 0
        if diff > 0:
            n_flip = int(round(diff))
            zeros = np.where(h_bin == 0)[0]
            z_scores = h[zeros]
            flip = zeros[np.argsort(z_scores)[-n_flip:]]
            h_bin[flip] = 1
        elif diff < 0:
            n_flip = int(round(-diff))
            ones = np.where(h_bin == 1)[0]
            o_scores = h[ones]
            flip = ones[np.argsort(o_scores)[:n_flip]]
            h_bin[flip] = 0
    return h_bin
