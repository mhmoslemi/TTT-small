import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_onto_constraint(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    # Binary search for threshold
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_gradient(h, N, beta):
    """Compute soft-max weighted gradient of max(corr)"""
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    scaled = beta * corr
    scaled -= np.max(scaled)  # for numerical stability
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Gradient of corr[n] w.r.t. h[j]
    # corr[n] = sum_i h[i] * (1-h[i+n-(N-1)])
    # Let L = n - (N-1) be the lag
    # corr at lag L: for L>=0: sum_{i=0}^{N-1-L} h[i+L]*(1-h[i])
    #               for L<0:  sum_{i=-L}^{N-1} h[i]*(1-h[i+L])
    
    grad = np.zeros(N)
    center = N - 1
    for n in range(2 * N - 1):
        w = weights[n]
        if w < 1e-10:
            continue
        L = n - center
        if L == 0:
            # corr = sum h[i]*(1-h[i]), grad = 1-2h
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            # corr = sum_{i=0}^{N-1-L} h[i+L]*(1-h[i])
            # d/dh[j]: for j as the h[i+L] term: (1-h[j-L]) if j-L >= 0, i.e., j >= L
            #          for j as the (1-h[i]) term: -h[j+L] if j+L < N, i.e., j < N-L
            grad[L:] += w * (1.0 - h[:N-L])  # j >= L: contribution from h[j] as first factor
            grad[:N-L] -= w * h[L:]           # j < N-L: contribution from (1-h[j]) as second factor

            #          for j < M: h[j+M]*(1-h[j]) -> -h[j+M]
            grad[M:] += w * (1.0 - h[:N-M])
            grad[:M] -= w * h[M:]
    
    return grad * dx, float(np.max(corr)) * dx

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_onto_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_grad(h, N, beta):
    """Full soft-max weighted gradient of max correlation."""
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    # Only need lags 0 to N-1 (by symmetry)
    corr_lags = corr[center:center + N]  # lags 0, 1, ..., N-1
    
    # Softmax weights (account for symmetry factor of 2 for L>0)
    scaled = beta * corr_lags * dx
    scaled -= np.max(scaled)
    exp_s = np.exp(scaled)
    exp_s[1:] *= 2.0  # factor of 2 for symmetric lags
    total = exp_s.sum()
    weights = exp_s / total  # weights[0] for lag 0, weights[L] for lag L>0
    
    # Compute gradient
    grad = np.zeros(N)
    # Lag 0
    grad += weights[0] * (1.0 - 2.0 * h)
    # Lags 1 to N-1
    for L in range(1, N):
        w = weights[L]
        if w < 1e-12:
            continue
        grad[L:] += w * (1.0 - h[:N-L])
        grad[:N-L] -= w * h[L:]
    
    return grad * dx, float(np.max(corr)) * dx

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    total_time = max(deadline - time.time(), 1.0)
    it = 0
    
    while time.time() < deadline:
        # Progressive beta

        beta = beta0 + (beta1 - beta0) * progress
        
        grad, val = softmax_grad(h, N, beta)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        lr = lr0 * (decay ** (it // 100))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_onto_constraint(h, target)
    
    return best_h, best_val

import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def softmax_grad(h, N, beta):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    cl = corr[center:center + N]
    sc = beta * cl * dx
    sc -= np.max(sc)
    e = np.exp(sc)
    e[1:] *= 2.0
    w = e / e.sum()
    grad = w[0] * (1.0 - 2.0 * h).copy()
    for L in range(1, N):
        wt = w[L]
        if wt < 1e-12:
            continue
        grad[L:] += wt * (1.0 - h[:N - L])
        grad[:N - L] -= wt * h[L:]
    return grad * dx, float(np.max(corr)) * dx

def adam_refine(h, N, target, deadline, beta0, beta1, lr0, lr_decay):
    best_h = h.copy()
    best_v = compute_c5(h, N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    t_start = time.time()
    it = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / max(1.0, deadline - t_start))
        beta = beta0 + (beta1 - beta0) * frac
        grad, val = softmax_grad(h, N, beta)
        if val < best_v:
            best_v = val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        lr = lr0 * (lr_decay ** (it // 200))
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_v

def swap_delta(h, i, j, N):
    """Compute change in corr_lags after swap h[i]:1->0, h[j]:0->1. Returns new_lags correction array."""
    d = np.zeros(N)
    # h[i] was 1, becomes 0
    d[:i + 1] -= (1.0 - h[i::-1])       # term: -(1-h[i-L]) for L=0..i
    d[:N - i] += h[i:]                    # term: +h[i+L] for L=0..N-1-i
    # h[j] was 0, becomes 1
    d[:j + 1] += (1.0 - h[j::-1])       # term: +(1-h[j-L]) for L=0..j
    d[:N - j] -= h[j:]                    # term: -h[j+L] for L=0..N-1-j
    return d

def binary_search(h, N, deadline, rng):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    n_target = N // 2
    n_ones = int(h_b.sum())
    if n_ones > n_target:
        idx = np.where(h_b == 1)[0]
        h_b[idx[n_target:]] = 0
    elif n_ones < n_target:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:n_target - n_ones]] = 1
    
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    center = N - 1
    lags = corr[center:center + N].copy()
    cur_max = float(np.max(lags))
    best_h = h_b.copy()
    best_max = cur_max
    
    T0, T1 = 3.0, 0.005
    n_total = max(int((deadline - time.time()) * 100), 1000)
    
    for it in range(n_total):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_total)
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        d = swap_delta(h_b, i, j, N)
        new_max = float(np.max(lags + d))
        delta = new_max - cur_max
        
        if delta < -1e-12 or (T > 0.01 and rng.random() < np.exp(-delta / T)):
            h_b[i], h_b[j] = 0.0, 1.0
            lags += d
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h_b.copy()
        
        if it % 500 == 0:
            corr = np.correlate(h_b, 1.0 - h_b, mode='full')
            lags = corr[center:center + N].copy()
            cur_max = float(np.max(lags))
    
    return best_h, best_max * dx
