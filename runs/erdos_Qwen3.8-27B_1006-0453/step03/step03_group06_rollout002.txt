    while len(centers) < K:
        centers.append(1.0 + 0.1 * (len(centers) - K / 2))
        widths.append(0.5 / K)
    total_w = sum(widths)
    if total_w > 0:
        widths = [w / total_w for w in widths]
    return centers, widths

def slsqp_refine(h0, n, T, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    if topk is not None:
        res = minimize(smooth_topk_c5, h0, args=(T, topk),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(smooth_c5, h0, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n = 160
        best_h = np.zeros(n)
        best_h[:n//2] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    print(f"Initial: n={n}, C5={best_c5:.6f}")

    # Phase 1: Interval-based optimization with Nelder-Mead
    for K in [2, 3, 4, 5, 6, 8]:
        if elapsed() > budget_s * 0.22:
            break
        centers, widths = extract_intervals(best_h, n, K)
        params0 = []
        for c, w in zip(centers, widths):
            params0.append(c)
            params0.append(w)
        bounds_nm = None
        for T in [100, 500, 2000]:
            if elapsed() > budget_s * 0.25:
                break
            for trial in range(3):
                if elapsed() > budget_s * 0.26:
                    break
                p = params0.copy()
                for j in range(len(p)):
                    p[j] += np.random.uniform(-0.05, 0.05)
                try:
                    res = minimize(interval_obj, p, args=(n, T, K),
                                 method='Nelder-Mead',
                                 options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
                    c_r = res.x[0:2*K:2]
                    w_r = res.x[1:2*K:2]
                    tw = w_r.sum()
                    if tw > 0:
                        w_r = w_r / tw
                        h_try = intervals_to_array(c_r, w_r, n)
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_h = h_try
                            print(f"Interval K={K},T={T},tr={trial}: C5={best_c5:.6f}")
                except Exception:
                    pass

    # Phase 2: SLSQP multi-T refinement
    t_list = [30, 80, 200, 500, 1200, 3000, 8000]
    k_list = [None, 3, 5, 8]
    for T in t_list:
        if elapsed() > budget_s * 0.45:
            break
        for k in k_list:
            if elapsed() > budget_s * 0.47:
                break
            try:
                hc = slsqp_refine(best_h, n, T, k, 600, 1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 3: Basin hopping with adaptive schedule
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = 200
    for it in range(n_bh):
        if elapsed() > budget_s * 0.70:
            break
        progress = it / max(1, n_bh)
        step = 0.07 * (1.0 - progress) + 0.015
        T_bh = 200.0 * (1.0 + 2.0 * progress)
        r = np.random.rand()
        if r < 0.25:
            h_pert = bh_h + step * np.random.randn(n)
        elif r < 0.45:
            blen = np.random.randint(3, max(4, n // 8))
            start = np.random.randint(0, n - blen)
            h_pert = bh_h.copy()
            h_pert[start:start+blen] += step * np.random.choice([-1, 1]) * np.random.rand(blen)
        elif r < 0.65:
            h_pert = bh_h.copy()
            i1 = np.random.randint(0, n // 2)
            i2 = n - 1 - i1
            swap = step * np.random.choice([-1, 1])
            h_pert[i1] = np.clip(h_pert[i1] + swap, 0, 1)
            h_pert[i2] = np.clip(h_pert[i2] - swap, 0, 1)
        elif r < 0.80:
            h_pert = bh_h.copy()
            nswaps = np.random.randint(2, 6)
            for _ in range(nswaps):
                idx = np.random.randint(0, n)
                h_pert[idx] = np.clip(h_pert[idx] + step * np.random.randn(), 0, 1)
        else:
            # Block flip
            blen = np.random.randint(2, max(3, n // 10))
            start = np.random.randint(0, n - blen)
            h_pert = bh_h.copy()
            h_pert[start:start+blen] = 1.0 - h_pert[start:start+blen]
        h_pert = project_feasible(h_pert, n)
        try:
            k_kick = np.random.choice([None, 3, 5, 8])
            hc = slsqp_refine(h_pert, n, T_bh, k_kick, 150, 1e-12)
            c5v = compute_c5(hc)
            temp = 0.004 * (1.0 - progress) + 0.0001
            if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
                bh_c5 = c5v
                bh_h = hc
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass

    # Phase 4: Targeted coordinate descent on worst overlaps
    if elapsed() < budget_s * 0.90:
        for iteration in range(400):
            if elapsed() > budget_s * 0.91:
                break
            improved = False
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            worst_idx = np.argsort(corr)[-6:]
            for k_sh in worst_idx:
                if improved:
                    break
                shift = k_sh - (n - 1)
                if shift >= 0:
                    x_lo, x_hi = 0, n - 1 - shift
                else:
                    x_lo, x_hi = -shift, n - 1
                if x_hi - x_lo < 1:
                    continue
                if shift >= 0:
                    ov = best_h[x_lo:x_hi+1] * (1.0 - best_h[x_lo+shift:x_hi+shift+1])
                else:
                    ov = best_h[x_lo:x_hi+1] * (1.0 - best_h[x_lo+shift:x_hi+shift+1])
                top_pts = np.argsort(ov)[-4:] + x_lo
                for j in top_pts:
                    for delta in [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.07, -0.07]:
                        h_try = best_h.copy()
                        h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
                            break
                    if improved:
                        break

    # Phase 5: Final high-T polish
    for T in [5000, 20000, 80000]:
        if elapsed() > budget_s * 0.97:
            break
        try:
            hc = slsqp_refine(best_h, n, T, None, 2000, 1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    for k in [2, 3, 5]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_refine(best_h, n, 10000, k, 1500, 1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    print(f"Final: C5={best_c5:.6f}")
    return best_h, best_c5, len(best_h)
