def compute_c5(h, N):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_c5(h, N, tau):
    """Compute soft approximation to max corr * dx using log-sum-exp."""
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    # log-sum-exp for numerical stability
    m = corr.max()
    val = m + np.log(np.sum(np.exp((corr - m) * tau)) / len(corr)) / tau
    return float(val * dx), corr * dx

def project_to_box_sum(h, target_sum):

import numpy as np
import time

def compute_c5(h, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def gradient_obj(h, N, tau, K):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(corr.max()) * dx
    if K >= len(corr):
        w = np.ones(len(corr)) / len(corr)
    else:
        idx = np.argsort(corr)[-K:]
        c_sel = corr[idx]
        sc = tau * (c_sel - c_sel.max())
        w_sel = np.exp(sc)
        w_sel /= w_sel.sum()
        w = np.zeros(len(corr))
        w[idx] = w_sel
    grad = np.zeros(N)
    center = N - 1
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-10:
            continue
        L = k - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] -= wk * h[L:]
            grad[L:] += wk * (1.0 - h[:N - L])
        else:
            M = -L
            grad[M:] += wk * (1.0 - h[:N - M])
            grad[:N - M] -= wk * h[M:]
    return grad, c5

def optimize_phase(h, N, target, deadline, tau_s, tau_e, lr_s, lr_e, K, iters_hint):
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / max(iters_hint, 1), 1.0)
        tau = tau_s * (tau_e / tau_s) ** frac
        lr = lr_s * (lr_e / lr_s) ** frac
        grad, c5v = gradient_obj(h, N, tau, K)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_to_box_sum(h, target)
    return best_h, best_c5

def fourier_smooth(h, N, cutoff_frac):
    H = np.fft.rfft(h)
    n_keep = max(2, int(len(H) * (1 - cutoff_frac)))
    H[n_keep:] = 0
    hs = np.fft.irfft(H, n=N)
    return hs

def local_search(h, N, target, n_trials, deadline, scale=0.015):
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    rng = np.random.RandomState(999)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        n_mod = rng.randint(1, 4)
        idx = rng.choice(N, size=n_mod, replace=False)
        trial[idx] += rng.randn(n_mod) * scale
        trial = project_to_box_sum(trial, target)
        c = compute_c5(trial, N)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = trial
    return best_h, best_c5

def fine_local(h, N, target, n_trials, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    rng = np.random.RandomState(4242)
    for _ in range(n_trials):

        c = compute_c5(trial, N)
        if c < best_c5 - 1e-15:
            best_c5 = c
            best_h = trial
    return best_h, best_c5
