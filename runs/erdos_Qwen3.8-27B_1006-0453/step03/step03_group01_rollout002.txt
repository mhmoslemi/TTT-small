    wts /= wts.sum()
    return lags, wts

def optimize_nesterov(h, N, target, deadline, phase_fracs):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h_cur = h.copy()
    h_prev = h.copy()
    t_start = time.time()
    total_t = max(deadline - t_start, 1.0)
    stall = 0

    for ph in phase_fracs:
        if time.time() >= deadline:
            break
        ph_dead = t_start + ph['cum'] * total_t
        if ph_dead > deadline:
            ph_dead = deadline
        topk = ph['topk']
        spike = ph['spike']
        lr = ph['lr']
        mom = ph['mom']
        h_prev = h_cur.copy()

        while time.time() < ph_dead:
            y = h_cur + mom * (h_cur - h_prev)
            y = np.clip(y, 0.0, 1.0)
            lags, wts = top_lags_weights(np.correlate(y, 1.0 - y, mode='full'), N, topk, spike)
            g, _ = grad_weighted(y, N, lags, wts)
            step = lr * (1.0 + 0.5 * (time.time() - t_start) / total_t)
            h_new = y - step * g
            h_new = project_sum(h_new, target)
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            if stall > 800:
                h_new += np.random.randn(N) * 0.01
                h_new = project_sum(h_new, target)
                stall = 0
            h_prev = h_cur
            h_cur = h_new
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = h[i]
            candidates = [0.0, 1.0, orig - 0.03, orig + 0.03, orig - 0.01, orig + 0.01]
            for trial in candidates:
                if trial < 0 or trial > 1 or abs(trial - orig) < 1e-8:
                    continue
                h[i] = trial
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    N = len(h_par)
    target = N / 2.0
    dx = 2.0 / N
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, dx)

    phases = [
        dict(cum=0.45, topk=12, spike=0.55, lr=0.025, mom=0.82),
        dict(cum=0.75, topk=5, spike=0.45, lr=0.007, mom=0.90),
        dict(cum=1.00, topk=3, spike=0.35, lr=0.002, mom=0.94),
    ]

    n_trajs = 4
    prep_time = 20
    traj_time = (deadline - t0 - prep_time - 50) / n_trajs

    for t in range(n_trajs):
        if time.time() > deadline - 50:
            break
        traj_dead = t0 + prep_time + (t + 1) * traj_time
        if traj_dead > deadline - 45:
            traj_dead = deadline - 45

        if t == 0:
            h_cur = h_par.copy()
        elif t == 1:
            h_cur = h_par + rng.randn(N) * 0.04
            h_cur = project_sum(h_cur, target)
        elif t == 2:
            h_cur = h_par.copy()
            h_cur[rng.choice(N, N // 4, replace=False)] = 1.0
            h_cur = project_sum(h_cur, target)
        else:
            h_cur = h_par + rng.randn(N) * 0.06
            h_cur = project_sum(h_cur, target)

        h_opt, c5_opt = optimize_nesterov(h_cur, N, target, traj_dead, phases)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time.time() < deadline - 20:
        best_h, best_c5 = optimize_nesterov(best_h.copy(), N, target, min(time.time() + 25, deadline - 10),
            [dict(cum=1.0, topk=2, spike=0.3, lr=0.001, mom=0.95)])

    if time.time() < deadline - 5:
        best_h, best_c5 = coord_polish(best_h, N, target, deadline - 2)

    final_c5 = compute_c5(best_h, dx)
    best_c5 = min(best_c5, final_c5)
    return best_h, best_c5, N
