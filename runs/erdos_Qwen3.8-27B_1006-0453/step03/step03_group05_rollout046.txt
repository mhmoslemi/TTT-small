                break
            h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def topk_obj(x, k, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def sum_constraint(x):
    return float(x.sum() - len(x) / 2.0)

def vec_subgrad_step(h, top_k, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    wts = corr[idx].astype(float)
    ws = wts.sum()
    wts = wts / ws if ws > 1e-14 else np.ones(top_k) / top_k
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        s = int(ki - (n - 1))
        if s >= 0:
            if s < n:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
        else:
            if -s < n:
                grad[:n + s] += w * (1.0 - h[-s:])
                grad[-s:] -= w * h[:n + s]
    return project_feasible(h - alpha * grad, n)

def slsqp_opt(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def fourier_perturb(h, n_freq, scale):
    n = len(h)
    nf = np.size(np.fft.rfft(h))
    H = np.fft.rfft(h)
    perturb = np.zeros(nf, dtype=complex)
    nn = min(n_freq, nf)
    perturb[:nn] = (np.random.randn(nn) + 1j * np.random.randn(nn)) * scale
    h_new = np.fft.irfft(H + perturb, n=n)
    return project_feasible(h_new, n)

def conflict_balancing(h, top_k, n_iter=15):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(n)
        for ki in idx:
            s = int(ki - (n - 1))
            if s > 0 and s < n:
                ov = h[s:] * (1.0 - h[:n - s])
                grad[s:] += 1.0 * ov
                grad[:n - s] -= 0.5 * ov
            elif s < 0 and -s < n:
                ov = h[:n + s] * (1.0 - h[-s:])
                grad[:n + s] += 1.0 * ov
                grad[-s:] -= 0.5 * ov
        alpha = 0.03 * (1.0 - 0.8 * it / n_iter)
        h = project_feasible(h - alpha * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Vectorized sub-gradient descent (fast, many iterations)
    for tk in [3, 5, 8]:
        if time.time() - t0 > budget_s * 0.08:
            break
        n_iter = 120
        for it in range(n_iter):
            alpha = 0.012 * (1.0 - 0.7 * it / n_iter)
            h_try = vec_subgrad_step(best_h, tk, alpha)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 2: Conflict balancing
    if time.time() - t0 < budget_s * 0.18:
        for tk in [3, 5, 8]:
            if time.time() - t0 > budget_s * 0.18:
                break
            h_cb, c5_cb = conflict_balancing(best_h, tk, 20)
            if c5_cb < best_c5:
                best_c5 = c5_cb
                best_h = h_cb
                best_n = len(best_h)

    # Phase 3: SLSQP with high-temperature softmax
    if time.time() - t0 < budget_s * 0.30:
        for T in [80, 400, 2000]:
            if time.time() - t0 > budget_s * 0.30:
                break
            h_try = slsqp_opt(best_h, softmax_obj, (T, best_n), 300, 1e-13)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                best_n = len(best_h)

    # Phase 4: Multi-resolution spectral search
    res_list = [100, 150, 200, 250, 300]
    for nr in res_list:
        if time.time() - t0 > budget_s * 0.58:
            break
        if nr == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, nr)
        # Two perturbations: smooth and rough
        for ptype in [0, 1]:
            if time.time() - t0 > budget_s * 0.58:
                break
            if ptype == 0:
                h_pert = fourier_perturb(h_up, min(5, nr // 5), 0.04)
            else:
                h_pert = project_feasible(h_up + np.random.randn(nr) * 0.02, nr)
            h_opt = slsqp_opt(h_pert, softmax_obj, (500, nr), 180, 1e-12)
            h_opt = slsqp_opt(h_opt, topk_obj, (3, nr), 180, 1e-13)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = nr
            else:
                # Downsample and compare
                x_dn = np.linspace(0, 2, best_n)
                h_dn = np.interp(x_dn, np.linspace(0, 2, nr), h_opt)
                h_dn = project_feasible(h_dn, best_n)
                c5_dn = compute_c5(h_dn)
                if c5_dn < best_c5:
                    best_c5 = c5_dn
                    best_h = h_dn

    # Phase 5: Intensive top-k SLSQP
    if time.time() - t0 < budget_s * 0.72:
        for k in [2, 3, 4]:
            if time.time() - t0 > budget_s * 0.72:
                break
            h_try = slsqp_opt(best_h, topk_obj, (k, best_n), 400, 1e-14)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                best_n = len(best_h)

    # Phase 6: Basin hopping with Fourier perturbation
    n_trials = 50
    for trial in range(n_trials):
        if time.time() - t0 > budget_s * 0.87:
            break
        scale = 0.03 + 0.025 * (1.0 - trial / float(n_trials))
        if trial % 3 == 0:
            h_mut = project_feasible(best_h + np.random.randn(best_n) * 0.025, best_n)
        elif trial % 3 == 1:
            h_mut = fourier_perturb(best_h, 4, scale)
        else:
            h_mut = vec_subgrad_step(best_h, 3, 0.02)
        h_ref = slsqp_opt(h_mut, softmax_obj, (800, best_n), 150, 1e-12)
        h_ref = slsqp_opt(h_ref, topk_obj, (3, best_n), 150, 1e-13)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = len(best_h)

    # Phase 7: Final multi-objective polish
    if time.time() - t0 < budget_s * 0.94:
        for k in [2, 3]:
            if time.time() - t0 > budget_s * 0.94:
                break
            h_fin = slsqp_opt(best_h, topk_obj, (k, best_n), 500, 1e-15)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin
        h_fin2 = slsqp_opt(best_h, softmax_obj, (20000, best_n), 400, 1e-15)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2

    # Phase 8: Micro sub-gradient + final SLSQP
    if time.time() - t0 < budget_s * 0.985:
        for it in range(80):
            h_try = vec_subgrad_step(best_h, 3, 0.002 * (1.0 - 0.5 * it / 80.0))
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        h_last = slsqp_opt(best_h, topk_obj, (2, best_n), 300, 1e-15)
        c5_last = compute_c5(h_last)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
