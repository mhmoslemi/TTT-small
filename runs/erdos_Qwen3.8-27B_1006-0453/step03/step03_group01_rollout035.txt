import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_grad(corr, h, tau, topK):
    N = len(h)
    center = N - 1
    nc = len(corr)
    sc = tau * (corr - corr.max())
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    if 0 < topK < nc:
        idx_top = np.argsort(corr)[-topK:]
        w_top = np.zeros(nc)
        w_top[idx_top] = 1.0 / len(idx_top)
        alpha = min(1.0, max(0.0, (tau - 8.0) / 25.0))
        w = (1.0 - alpha) * w_sm + alpha * w_top
    else:
        w = w_sm
    grad = np.zeros(N)
    active = np.where(w > 1e-8)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def gd_phase(h, N, target, deadline, tau0, tau1, tk0, tk1, lr0, dec_per, decay, mom):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 3000.0, 1.0)
        tau = tau0 + (tau1 - tau0) * frac
        tk = max(1, int(tk0 + (tk1 - tk0) * frac))
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(corr.max()) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        grad = get_grad(corr, h, tau, tk)
        v = mom * v + (1.0 - mom) * grad
        h = project_sum(h - lr * v, target)
        if it % dec_per == 0:
            lr *= decay
            if lr < 1e-13:
                break
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                return best_h, best_c5
            hi = best_h[i]
            cands = []
            if hi < 0.15:
                cands.append(0.0)
            elif hi > 0.85:
                cands.append(1.0)
            else:
                cands.extend([max(0.0, hi - 0.08), min(1.0, hi + 0.08), 0.0, 1.0])
            for c in cands:
                if abs(c - hi) < 1e-10:
                    continue
                trial = best_h.copy()
                trial[i] = c
                trial = project_sum(trial, target)
                c5 = compute_c5(trial, dx)
                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = trial
                    improved = True
                    break
    return best_h, best_c5

def pair_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    rng = np.random.RandomState(777)
    below = np.where(best_h < 0.5)[0]
    above = np.where(best_h > 0.5)[0]
    ntrial = 0
    max_trial = 30000
    while ntrial < max_trial and time.time() < deadline:
        if len(below) == 0 or len(above) == 0:
            break
        i = below[rng.randint(len(below))]
        j = above[rng.randint(len(above))]
        delta = 0.03
        trial = best_h.copy()
        trial[i] = min(1.0, trial[i] + delta)
        trial[j] = max(0.0, trial[j] - delta)
        trial = project_sum(trial, target)
        c5 = compute_c5(trial, dx)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = trial
            below = np.where(best_h < 0.5)[0]
            above = np.where(best_h > 0.5)[0]
        ntrial += 1
    return best_h, best_c5
