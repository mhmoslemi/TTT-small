    n = len(h)

    val = m + np.log(np.sum(np.exp(T * (corr - m))) + 1e-300) / T
    return val * dx

    n = len(h)
    dx = 2.0 / n

    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()

        s = 0.0
        for idx in range(2 * n - 1):

            wi = w[idx]
            if 0 <= i + k < n:

            if 0 <= i - k < n:
                s += -wi * h[i - k]

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s

        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        # Scale down: h * (1 + diff/s) might overshoot below 0
        # Better: reduce non-zero entries proportionally
        mask = h > 0
        s_mask = h[mask].sum()
        if s_mask < 1e-14:
            return h
        h[mask] *= (1.0 + diff / s_mask)
    return np.clip(h, 0.0, 1.0)

    # coeff for h[i]

        c = 0.0
        j1 = i + k
        j2 = i - k

            c += (-h[j2])

    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=40)
    for T in [30000, 80000]:
        if elapsed() > budget_s * 0.98:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final micro-polish: nudge non-boundary coords
    if elapsed() < budget_s * 0.99:
        inner = np.where((best_h > 1e-6) & (best_h < 1.0 - 1e-6))[0]
        if len(inner) > 0:
            for _ in range(5):
                h_try = best_h + 1e-5 * np.random.randn(n)
                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
