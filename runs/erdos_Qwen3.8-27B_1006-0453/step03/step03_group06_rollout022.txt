                        best = ht
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP schedule
    t1 = 0.18 * budget_s
    best_h, best_c5 = slsqp_schedule(best_h,
        [10, 30, 100, 300, 1000, 3000, 8000],
        [None, 5, 10, 20, 50], 500, 1e-14, t1, t0)

    # Phase 2: Interval-based diverse starts
    t2 = 0.38 * budget_s
    n = len(best_h)
    for K in [3, 4, 5, 6, 7, 8, 10]:
        if (time.time() - t0) > t2:
            break
        for trial in range(3):
            if (time.time() - t0) > t2:
                break
            cs = np.sort(np.random.uniform(0.15, 1.85, K))
            ws = np.random.uniform(0.08, 0.45, K)
            ws /= ws.sum()
            p0 = np.zeros(2 * K)
            for i in range(K):
                p0[2*i] = cs[i]
                p0[2*i+1] = ws[i]
            try:
                res = nelder_mead(interval_obj, p0, args=(n,),
                                  maxiter=2000, xatol=1e-9, fatol=1e-11, adaptive=True)
                h_int = interval_to_samples(res.x, n)
                h_int = project_feasible(h_int, n)
                h_int, c5_int = slsqp_schedule(h_int, [50, 200, 800, 3000],
                                               [None, 10, 30], 300, 1e-14, t2, t0)
                if c5_int < best_c5:
                    best_c5 = c5_int
                    best_h = h_int
            except Exception:
                pass

    # Phase 3: Aggressive basin hopping
    t3 = 0.68 * budget_s
    bh = best_h.copy()
    bh_c5 = best_c5
    n = len(bh)
    n_iter = 120
    for it in range(n_iter):
        if (time.time() - t0) > t3:
            break
        step = 0.12 * (1.0 - it / n_iter) + 0.02
        r = np.random.rand()
        if r < 0.35:
            hp = bh + step * np.random.randn(n)
        elif r < 0.55:
            blen = np.random.randint(5, n // 3 + 1)
            st = np.random.randint(0, n - blen)
            hp = bh.copy()
            hp[st:st+blen] += step * np.random.uniform(0.3, 1.0, blen)
        elif r < 0.75:
            hp = bh.copy()
            q = n // 4
            hp[:q] += step * np.random.uniform(-1, 1, q)
            hp[-q:] += step * np.random.uniform(-1, 1, q)
        else:
            hp = bh.copy()
            for _ in range(np.random.randint(3, 8)):
                i1, i2 = np.random.randint(0, n, 2)
                sv = step * 0.4 * np.random.choice([-1, 1])
                hp[i1] += sv
                hp[i2] -= sv
        hp = project_feasible(hp, n)
        try:
            T_k = int(np.random.choice([50, 150, 500, 1500]))
            k_k = int(np.random.choice([0, 5, 10, 20]))
            k_use = None if k_k == 0 else k_k
            hc = opt_smooth(hp, T_k, k_use, 250, 1e-12)
            c5v = compute_c5(hc)
            temp = 0.004 * max(0.0, 1.0 - it / 60) + 0.0002
            if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
                bh_c5 = c5v
                bh = hc
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass

    # Phase 4: Multi-resolution
    t4 = 0.87 * budget_s
    n_best = len(best_h)
    for n_try in [200, 240, 280, 180]:
        if (time.time() - t0) > t4:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        hi = np.interp(x_new, x_old, best_h)
        hi = project_feasible(hi, n_try)
        for tr in range(4):
            if (time.time() - t0) > t4:
                break
            h0 = hi + np.random.uniform(-0.025, 0.025, n_try)
            h0 = project_feasible(h0, n_try)
            try:
                hc, c5v = slsqp_schedule(h0, [30, 100, 500, 2000, 6000],
                                         [None, 8, 20, 50], 400, 1e-14, t4, t0)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n_best = n_try
                    x_old = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
            except Exception:
                pass

    # Phase 5: Targeted coordinate descent
    best_h, best_c5 = targeted_coord_descent(best_h, 400, 0.95 * budget_s, t0)

    # Phase 6: Final polish
    n = len(best_h)
    for T in [10000, 30000, 100000]:
        if (time.time() - t0) > budget_s * 0.99:
            break
        try:
            hc = opt_smooth(best_h, T, None, 1500, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    for k in [3, 5, 8]:
        if (time.time() - t0) > budget_s * 0.995:
            break
        try:
            hc = opt_smooth(best_h, 5000, k, 800, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
