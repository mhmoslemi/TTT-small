def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-15:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-15:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_con(x):
    return x.sum() - len(x) / 2.0

def lse_topk(h, T, k):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    if k is not None and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        topk = corr[idx]
    else:
        topk = corr
    m = np.max(topk)
    val = m + np.log(np.sum(np.exp(T * (topk - m)))) / T
    return val * dx

def slsqp_min(h0, T, k, maxiter, ftol):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    args = (T, k)
    res = minimize(lse_topk, h0, args=args, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def coordinate_descent(h, n, passes=3, refine=5):
    """Single-pass coordinate descent with projection."""
    best = h.copy()
    best_c5 = compute_c5(best)

                    best = h_try

        fixed[free] = False

            h_try[free] = x
            # Adjust sum: redistribute to nearest 0 or 1 variable
            diff = n/2.0 - h_try.sum()
            if abs(diff) > 1e-10:

                if diff > 0:

                    if len(zeros) > 0:
                        h_try[zeros[0]] += diff
                else:

                    if len(ones) > 0:
                        h_try[ones[0]] += diff  # diff is negative
            h_try = project_feasible(h_try, n)
            return compute_c5(h_try)
        
        from scipy.optimize import minimize as sc_min
        res = sc_min(obj, x0, method='Nelder-Mead',

        ht = project_feasible(ht, n)
        c5_new = compute_c5(ht)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best = ht
            best[best < thresh * 0.5] = 0.0
            best[best > 1.0 - thresh * 0.5] = 1.0
            best = project_feasible(best, n)
            best_c5 = compute_c5(best)
        else:
            break
    return best, best_c5

def coord_descent(h, n, npasses=5, ntry=9):
    best = h.copy()
    best_c5 = compute_c5(best)
    for p in range(npasses):
        improved = False
        order = np.random.permutation(n)
        for j in order:
            partner = (j + n // 3) % n
            vals = np.linspace(-0.3, 0.3, ntry)
            for d in vals:
                ht = best.copy()
                ht[j] += d
                ht[partner] -= d
                ht = project_feasible(ht, n)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    best = ht
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: quick SLSQP to settle
    if elapsed() < budget_s * 0.08:
        for T in [20, 100]:
            if elapsed() > budget_s * 0.08:
                break
            try:
                hc = slsqp_min(best_h, lambda x: lse_topk(x, T, None), 400, 1e-12)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_h, best_c5 = hc, c5v
            except Exception:
                pass

    # Phase 2: sparse optimization (key differentiator)
    if elapsed() < budget_s * 0.25:
        for r in range(3):
            if elapsed() > budget_s * 0.25:
                break
            h_s, c5_s = sparse_optimize(best_h, n, thresh=0.04, max_rounds=20, nm_maxiter=500)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s

    # Phase 3: SLSQP with progressive smoothing
    if elapsed() < budget_s * 0.45:
        schedule = [(30, None), (100, 5), (300, 10), (1000, 5), (3000, None), (3000, 10)]
        for T, k in schedule:
            if elapsed() > budget_s * 0.45:
                break
            try:
                hc = slsqp_min(best_h, lambda x, _T=T, _k=k: lse_topk(x, _T, _k), 600, 1e-13)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_h, best_c5 = hc, c5v
            except Exception:
                pass

    # Phase 4: basin hopping with sparse local solver
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = 120
    for it in range(n_bh):
        if elapsed() > budget_s * 0.72:
            break
        step = 0.06 * (1.0 - it / n_bh) + 0.015
        r = np.random.rand()
        if r < 0.35:
            pert = step * np.random.randn(n)
        elif r < 0.65:
            pert = np.zeros(n)
            blen = max(4, n // 6)
            start = np.random.randint(0, n - blen)
            pert[start:start+blen] = np.random.choice([-1, 1]) * step * 1.5
        else:
            pert = np.zeros(n)
            npts = np.random.randint(4, 12)
            idxs = np.random.choice(n, npts, replace=False)
            pert[idxs] = step * np.random.choice([-1, 1], npts)
        h_pert = project_feasible(bh_h + pert, n)
        try:
            hc, c5v = sparse_optimize(h_pert, n, thresh=0.03, max_rounds=10, nm_maxiter=300)
            if c5v < bh_c5:
                bh_c5 = c5v
                bh_h = hc
            else:
                temp = 0.004 * (1.0 - it / n_bh) + 0.0002
                if np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
                    bh_h = hc
                    bh_c5 = c5v
            if bh_c5 < best_c5:
                best_h, best_c5 = bh_h.copy(), bh_c5
        except Exception:
            pass

    # Phase 5: more sparse optimization on best
    if elapsed() < budget_s * 0.85:
        for thresh in [0.05, 0.03, 0.02]:
            if elapsed() > budget_s * 0.85:
                break
            h_s, c5_s = sparse_optimize(best_h, n, thresh=thresh, max_rounds=15, nm_maxiter=400)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s

    # Phase 6: coordinate descent polish
    if elapsed() < budget_s * 0.93:
        h_cd, c5_cd = coord_descent(best_h, n, npasses=8, ntry=11)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 7: final high-T SLSQP
    for T, k in [(5000, None), (10000, 5), (20000, None)]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_min(best_h, lambda x, _T=T, _k=k: lse_topk(x, _T, _k), 3000, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 8: try one alternate n
    if elapsed() < budget_s * 0.995 and n != 120:
        try:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, 120, endpoint=False)
            h120 = project_feasible(np.interp(x_new, x_old, best_h), 120)
            c5_120 = compute_c5(h120)
            h_s, c5_s = sparse_optimize(h120, 120, thresh=0.03, max_rounds=10, nm_maxiter=300)
            if c5_s < best_c5:
                best_h, best_c5, n = h_s, c5_s, 120
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
