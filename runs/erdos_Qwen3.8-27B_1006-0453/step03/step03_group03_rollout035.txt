    """Project h onto {0<=h_i<=1, sum(h)=target}."""
    lo, hi = -h.max() - 0.1, 1 - h.min() + 0.1

        s = np.clip(h + mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def binary_simulated_annealing(h, n_steps=5000, time_limit=30.0):

def smooth_max_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    occ = np.exp(T * (corr - np.max(corr)))
    w = occ / occ.sum()
    grad = np.zeros(n)
    for i in range(n):
        si = 0.0
        for L in range(-(n - 1), n):
            c = 0.0
            if 0 <= i - L < n:
                c += (1.0 - h[i - L])
            if 0 <= i + L < n:
                c += (-h[i + L])
            si += w[n - 1 + L] * c
        grad[i] = si
    return grad * dx

def fista_refine(h0, T, n_steps=500, time_limit=20.0):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    step = 1.0 / (T * n)
    best_h = h.copy()

    t0 = time.time()

        if time.time() - t0 > time_limit:
            break
        g = smooth_max_grad(y, T)
        y_new = project_box_sum(y - step * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - h)
        h = y_new
        t_k = t_new
        c5, _ = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_direct(h0, maxiter=500):
    n = len(h0)
    _, corr = compute_c5(h0)
    t0 = float(np.max(corr)) + 0.02
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    def obj(x):
        return x[-1]
    def eq(x):
        return x[:-1].sum() - len(x[:-1]) / 2.0
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full')
        return t - c
    cons = [{'type': 'eq', 'fun': eq}, {'type': 'ineq', 'fun': ineq}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_box_sum(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h0

    best_h = project_box_sum(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
