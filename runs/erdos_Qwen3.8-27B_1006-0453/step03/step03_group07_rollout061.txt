```python
    l1_clipped = np.clip(l1, 0, n-1)
    l2_clipped = np.clip(l2, 0, n-1)
    
    grad = dx * (M1 * (1.0 - h[l1_clipped]) - M2 * h[l2_clipped]).T @ w
    # Wait, M1 * (1-h[l1_clipped]) has shape (n, 2n-1)
    # We want sum_k M1[j,k] * (1-h[l1[j,k]]) * w[k] = (M1 * (1-h[l1_clipped])) @ w
    # But this gives a vector of shape (n,) where entry j is sum_k ...
    
    # Actually: grad[j] = dx * sum_k [M1[j,k]*(1-h[l1[j,k]]) - M2[j,k]*h[l2[j,k]]] * w[k]
    # = dx * [(M1 * (1-h[l1_clipped])) @ w - (M2 * h[l2_clipped])) @ w]
    # = dx * [(M1 * (1-h[l1_clipped]) - M2 * h[l2_clipped])].T @ w ... no
    # = dx * ((M1 * (1-h[l1_clipped]) - M2 * h[l2_clipped]) @ w)
    
    return obj, dx * ((M1 * (1.0 - h[l1_clipped]) - M2 * h[l2_clipped]) @ w)
```

Actually, let me think about this problem from a different angle. The Erdős problem is related to the "minimum overlap" of a set with its translates. For the binary case (h ∈ {0,1}), the problem becomes: find a set A ⊂ {0,...,n-1} with |A| = n/2 that minimizes max_k |A ∩ (A_k)^c| where A_k is A shifted by k.

|A ∩ (A_k)^c| = |A| - |A ∩ A_k| = n/2 - |A ∩ A_k|

So minimizing the max overlap is equivalent to maximizing the min |A ∩ A_k| over all k.

For the binary case with n=2m: we want a set A of size m that maximizes min_k |A ∩ (A+k)|.

This is related to the "autocorrelation" of the indicator function of A. We want the minimum autocorrelation value to be as large as possible.

For a random set A of size m = n/2, the expected |A ∩ (A+k)| ≈ m²/n = n/4 for small k. The variance is O(n). So the minimum over ~n lags would be around n/4 - O(√(n log n)).

The C₅ value for the binary case would be:
C₅ = max_k (n/2 - |A ∩ (A+k)|) * 2/n ≈ max_k (n/2 - (n/4 - Δk)) * 2/n = max_k (n/4 + Δk) * 2/n = 1/2 + 2Δk/n

For a good set, Δk (the maximum deviation) is small, so C₅ ≈ 1/2 - small.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    Z = e.sum()
    return (m + np.log(Z) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    Z = e.sum()
    w = e / Z
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    l1 = j_idx + k_idx - n + 1
    l2 = j_idx - k_idx + n - 1
    M1 = ((l1 >= 0) & (l1 < n)).astype(float)
    M2 = ((l2 >= 0) & (l2 < n)).astype(float)
    l1c = np.clip(l1, 0, n - 1)
    l2c = np.clip(l2, 0, n - 1)
    g = M1 * (1.0 - h[l1c]) - M2 * h[l2c]
    return dx * (g @ w)

def objective_wrapper(x, T):
    return softmax_obj(x, T)

def gradient_wrapper(x, T):
    return softmax_grad(x, T)

def slsqp_refine(h0, T, n, maxiter=2000, ftol=1e-18):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(objective_wrapper, h0.copy(), args=(T,),
                   jac=gradient_wrapper, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def targeted_reduction(h, n, num_lags=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    target = np.max(corr)
    subgrad = np.zeros(n)
    order = np.argsort(corr)[::-1]
    for k in order[:num_lags]:
        if corr[k] < target - 1e-10:
            break
        l = k - n + 1
        for j in range(n):
            if 0 <= j + l < n:
                subgrad[j] += (1.0 - h[j + l])
            if 0 <= j - l < n:
                subgrad[j] -= h[j - l]
    subgrad *= dx
    step = 0.01
    h_new = h - step * subgrad
    h_new = make_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float).copy()
    else:
        n = 200
        best_h = np.zeros(n)
        best_h[0::2] = 1.0
        best_h = make_feasible(best_h, n)

    n_best = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient, increasing T
    for T in [50, 100, 200, 500, 1000, 3000]:
        if time.time() - t0 > budget_s * 0.25:
            break
        try:
            h_new = slsqp_refine(best_h, T, n_best, 1500, 1e-16)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
                n_best = len(best_h)
        except Exception:
            pass

    # Phase 2: Targeted subgradient reduction of top lags
    for it in range(200):
        if time.time() - t0 > budget_s * 0.40:
            break
        h_try = targeted_reduction(best_h, n_best, num_lags=3)
        c5v = compute_c5(h_try)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_try
        else:
            # Try smaller step
            dx = 2.0 / n_best
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            target = np.max(corr)
            subgrad = np.zeros(n_best)
            order = np.argsort(corr)[::-1]
            for k in order[:3]:
                l = k - n_best + 1
                for j in range(n_best):
                    if 0 <= j + l < n_best:
                        subgrad[j] += (1.0 - best_h[j + l])
                    if 0 <= j - l < n_best:
                        subgrad[j] -= best_h[j - l]
            subgrad *= dx
            h_try = best_h - 0.003 * subgrad
            h_try = make_feasible(h_try, n_best)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try

    # Phase 3: Basin hopping with SLSQP local refinement
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    remaining = budget_s * 0.65 - (time.time() - t0)
    max_iters = max(20, int(remaining / 8.0))
    for it in range(max_iters):
        if time.time() - t0 > budget_s * 0.65:
            break
        ptype = it % 3
        step = 0.08 * (1.0 - it / max(1, max_iters)) + 0.01
        if ptype == 0:
            h_pert = bh_h + step * np.random.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(2, n_bh // 15)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1 + blen].copy()
            h_pert[s1:s1 + blen] = h_pert[s2:s2 + blen]
            h_pert[s2:s2 + blen] = tmp
        else:
            h_pert = bh_h.copy()
            nflip = max(1, n_bh // 10)
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        h_pert = make_feasible(h_pert, n_bh)
        T_use = 300 + 200 * (it // 10)
        try:
            h_new = slsqp_refine(h_pert, T_use, n_bh, 500, 1e-14)
            c5v = compute_c5(h_new)
        except Exception:
            h_new = h_pert
            c5v = compute_c5(h_pert)
        temp = 0.004 * (1.0 - it / max(1, max_iters)) + 0.0002
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 4: Try higher resolution
    for n_try in [256, 320]:
        if time.time() - t0 > budget_s * 0.80:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [500, 2000]:
            if time.time() - t0 > budget_s * 0.80:
                break
            try:
                h_new = slsqp_refine(h_interp, T, n_try, 1000, 1e-15)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
                    n_best = n_try
            except Exception:
                pass

    # Phase 5: Final high-T polish
    n_cur = len(best_h)
    for T in [5000, 20000, 100000]:
        if time.time() - t0 > budget_s * 0.96:
            break
        try:
            h_new = slsqp_refine(best_h, T, n_cur, 3000, 1e-17)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    # Phase 6: Final targeted tweak
    if time.time() - t0 < budget_s * 0.98:
        for _ in range(50):
            h_try = targeted_reduction(best_h, len(best_h), num_lags=2)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
