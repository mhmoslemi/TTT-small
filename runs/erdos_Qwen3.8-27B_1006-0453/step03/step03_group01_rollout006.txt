            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def optimize_phase(h, N, target, deadline, mode, tau_start, tau_end, topk_start, topk_end,
                   lr0, lr_decay, mom, bb_enable):
    """Optimize with adaptive parameters. mode: 'smooth' or 'exact'."""
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    n_it = 0
    lr = lr0
    h_prev = h.copy()
    grad_prev = np.zeros(N)
    
    while time.time() < deadline:
        n_it += 1
        frac = min(n_it / 800.0, 1.0)
        
        if mode == 'smooth':
            tau = tau_start * (tau_end / tau_start) ** frac
            corr = np.correlate(h, 1.0 - h, mode='full')
            nc = len(corr)
            center = N - 1
            sc = (corr - corr.max()) / tau
            e = np.exp(sc)
            w = e / e.sum()
            thresh = 1e-5
            active = np.where(w > thresh)[0]
            w_active = w[active]
            grad = grad_from_lags(h, N, active, w_active)
            c5_val = float(np.max(corr)) * dx
        else:
            topk = max(1, int(topk_start + (topk_end - topk_start) * frac))
            corr = np.correlate(h, 1.0 - h, mode='full')
            nc = len(corr)
            center = N - 1
            idx_top = np.argsort(corr)[-topk:]
            w_active = np.ones(topk) / topk
            grad = grad_from_lags(h, N, idx_top, w_active)
            c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        nesterov_v = mom * v + grad
        if bb_enable and n_it > 1:
            ys = np.dot(grad - grad_prev, h - h_prev)
            ss = np.dot(h - h_prev, h - h_prev)
            if ss > 1e-12 and abs(ys) > 1e-12:
                lr_bb = max(1e-6, min(0.2, ss / max(ys, 1e-12)))
                lr = 0.5 * lr + 0.5 * lr_bb
        else:
            if n_it % 40 == 0:
                lr *= lr_decay
        
        h_new = h - lr * nesterov_v
        h_new = project_sum(h_new, target)
        
        grad_prev = grad
        h_prev = h
        v = nesterov_v
        h = h_new
    
    return best_h, best_c5

def polish_coords(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for _ in range(20):
        if time.time() > deadline:
            break
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = best_h[i]
            for trial in (0.0, 1.0, max(0.0, orig - 0.02), min(1.0, orig + 0.02),
                          max(0.0, orig - 0.005), min(1.0, orig + 0.005)):
                if abs(trial - orig) < 1e-9:
                    continue
                best_h[i] = trial
                best_h = project_sum(best_h, target)
                c5 = compute_c5(best_h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def swap_polish(h, N, target, deadline):
    """Try swapping pairs of coordinates to reduce C5."""
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    rng = np.random.RandomState(1234)
    for _ in range(50000):
        if time.time() > deadline:
            break
        i, j = rng.randint(0, N, 2)
        if i == j:
            continue
        save_i, save_j = best_h[i], best_h[j]
        best_h[i], best_h[j] = save_j, save_i
        best_h = project_sum(best_h, target)
        c5 = compute_c5(best_h, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        else:
            best_h[i], best_h[j] = save_i, save_j
            best_h = project_sum(best_h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)
    
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)
    
    # Multi-resolution, multi-attempt optimization
    configs = [(380, 4), (440, 2), (520, 2)]
    total_time = budget_s - 90
    time_alloc = []
    total_w = sum(c[1] for c in configs)
    for N, n_att in configs:
        t = total_time * n_att / total_w
        time_alloc.append((N, n_att, t / n_att))
    
    for N, n_att, per_time in time_alloc:
        if time.time() > deadline - 60:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_base = np.interp(xN, x_par, h_par)
        h_base = project_sum(h_base, target)
        
        for at in range(n_att):
            if time.time() > deadline - 40:
                break
            trial_dead = time.time() + per_time
            if trial_dead > deadline - 25:
                trial_dead = deadline - 25
            t_rem = trial_dead - time.time()
            if t_rem < 20:
                break
            
            if at == 0:
                h_cur = h_base.copy()
            elif at == 1:
                h_cur = project_sum(h_base + rng.randn(N) * 0.05, target)
            elif at == 2:
                h_cur = project_sum(h_base + rng.randn(N) * 0.10, target)
            else:
                # Structured perturbation: flip a random block
                h_cur = h_base.copy()
                blen = N // 10
                bstart = rng.randint(0, N - blen)
                h_cur[bstart:bstart+blen] = 1.0 - h_cur[bstart:bstart+blen]
                h_cur = project_sum(h_cur, target)
            
            # Phase 1: Smooth exploration (45% of trial time)
            p1 = t_rem * 0.45
            h_cur, c1 = optimize_phase(h_cur, N, target, time.time() + p1,
                                       'smooth', 8.0, 0.3, 20, 2,
                                       0.06, 0.95, 0.80, True)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h_cur.copy()
            
            if time.time() > trial_dead - 15:
                break
            
            # Phase 2: Exact top-K refinement (30%)
            p2 = min(trial_dead - time.time() - 8, t_rem * 0.30)
            h_cur, c2 = optimize_phase(h_cur, N, target, time.time() + p2,
                                       'exact', 1, 1, 5, 2,
                                       0.02, 0.97, 0.88, False)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h_cur.copy()
            
            if time.time() > trial_dead - 8:
                break
            
            # Phase 3: Fine-tuning top-1 (remaining)
            p3 = trial_dead - time.time() - 2
            if p3 > 5:
                h_cur, c3 = optimize_phase(h_cur, N, target, time.time() + p3,
                                           'exact', 1, 1, 1, 1,
                                           0.005, 0.985, 0.92, False)
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = h_cur.copy()
    
    # Final refinement on best solution
    Nb = len(best_h)
    if time.time() < deadline - 40:
        h_f, c_f = optimize_phase(best_h.copy(), Nb, Nb/2.0, time.time() + 25,
                                  'exact', 1, 1, 2, 1, 0.003, 0.99, 0.93, False)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f
    
    if time.time() < deadline - 15:
        best_h, best_c5 = swap_polish(best_h, len(best_h), len(best_h)/2.0, 
                                       min(time.time() + 12, deadline - 10))
    
    if time.time() < deadline - 3:
        best_h, best_c5 = polish_coords(best_h, len(best_h), len(best_h)/2.0, deadline - 1)
    
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
