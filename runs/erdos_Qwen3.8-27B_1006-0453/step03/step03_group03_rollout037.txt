def bb_projected_grad(h, T, penalty, max_iter=5000):

    best_h = h_cur.copy()

        T_eff = T
        m = np.max(corr)
        w = np.exp(T_eff * (corr - m))

                g[k:] += -wi * h_cur[:n-k]

                g[:n-kp] += -wi * h_cur[kp:]
        # constraint gradient
        g += penalty * 2 * (h_cur.sum() - n/2.0)
        return g * (2.0/n)  # dx factor

        s_prev = s
        y_prev = y
        
        h_cur = h_new
    return best_h, best_c5

    """Returns (value, gradient) of smoothed C5 + lambda*(sum - n/2)^2"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

            g[kp:] += wi * (1.0 - h[:n-kp])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_val_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    eT = np.exp(T * (corr - cm))
    S = eT.sum()
    val = (cm + np.log(S) / T) * dx
    viol = h.sum() - n / 2.0
    val += lam * viol * viol
    w = eT / S
    g = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[:n - k] += wi * (1.0 - h[k:])
            g[k:] -= wi * h[:n - k]
        else:
            kp = -k
            g[kp:] += wi * (1.0 - h[:n - kp])
            g[:n - kp] -= wi * h[kp:]
    g *= dx
    g += lam * 2.0 * viol
    return val, g

def lbfgsb_smooth(h, T, lam, max_iter=4000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_val_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bnds, args=(T, lam),
                       options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_smooth(h, T, maxiter=800):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(lambda x: smooth_val_grad(x, T, 0.0)[0], h,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter=600):
    n = len(h)
    _, corr = compute_c5(h)
    t0 = float(np.max(corr)) + 0.03
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def in_fun(x):
        hh = x[:-1]
        tt = x[-1]
        c = np.correlate(hh, 1.0 - hh, mode='full')
        return tt - np.max(c)
    cons = [{'type': 'eq', 'fun': eq_fun}, {'type': 'ineq', 'fun': in_fun}]
    try:
        res = minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

            best_h = h_f
        h_s = slsqp_smooth(best_h, 5000000, maxiter=800)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
        for scale in [0.005, 0.002]:
            if elapsed() > budget_s * 0.97:
                break
            h_p = best_h + np.random.randn(best_n) * scale
            h_p = project_feasible(h_p, best_n)
            h_opt = slsqp_smooth(h_p, 2000000, maxiter=600)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
