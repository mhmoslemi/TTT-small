    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 0)
    return float((m + np.log(np.sum(np.exp(sh))) / T) * dx)

def project(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
        h = np.clip(h, 0, 1)
        # re-project if needed
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 0:
            room = 1.0 - h
            h += room * (diff2 / room.sum())
        elif diff2 < 0:
            h *= target / h.sum()
            h = np.clip(h, 0, 1)
    return np.clip(h, 0.0, 1.0)

    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')

    coef = np.zeros(2 * nn - 1)

        j2 = i - k

                c += (-h[j2])

                c += (-h[j2])

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})

    gc = np.zeros((2*n-1, n))

            idx2 = a - shift

                g += (1.0 - h[idx1])
            if 0 <= idx2 < n:
                g += (-h[idx2])
            gc[k, a] = g

    h_base[i] = 0.0

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up polish
    t1 = t0 + budget_s * 0.08
    for T in [20, 80, 400, 1500]:
        if time.time() > t1:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=200, ftol=1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Heavy coordinate descent
    t2 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=400)

    # Phase 3: Iterated Local Search with structured perturbations
    t3 = t0 + budget_s * 0.62
    work_h = best_h.copy()
    work_c5 = best_c5
    n_it = 0
    max_it = 100
    while time.time() < t3:
        n_it += 1
        cool = max(0.0, 1.0 - 0.92 * n_it / max_it)
        if n_it % 20 == 0:
            step = 0.01 + 0.05 * cool
        else:
            step = 0.008 + 0.03 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = project(h_pert, n)
        hc = slsqp_opt(h_pert, 600, n, maxiter=150, ftol=1e-13)
        td = min(t3, time.time() + budget_s * 0.008)
        hc, c5v = coord_descent(hc, n, td, max_sweeps=25)
        temp = 0.001 * cool + 0.00015
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h = hc
            work_c5 = c5v
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Final coordinate descent
    t4 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=500)

    # Phase 5: Multi-resolution exploration
    t5 = t0 + budget_s * 0.93
    for n_try in [120, 200, 240]:
        if time.time() > t5:
            break
        if n_try == n:
            continue
        h_int = interpolate_h(best_h, n_try)
        h_int = project(h_int, n_try)
        hc = slsqp_opt(h_int, 1000, n_try, maxiter=200, ftol=1e-14)
        td = min(t5, time.time() + budget_s * 0.015)
        hc, c5v = coord_descent(hc, n_try, td, max_sweeps=60)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            n = n_try

    # Phase 6: Final high-T SLSQP + coordinate descent
    t6 = t0 + budget_s * 0.99
    for T in [5000, 20000]:
        if time.time() > t6 - 2:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if time.time() < t6:
        best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=200)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
