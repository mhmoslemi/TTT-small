    """Compute h from Fourier coefficients."""
    i = np.arange(N)
    h = np.full(N, 0.5)
    for m in range(1, M+1):
        a = params[2*(m-1)]
        b = params[2*(m-1)+1]
        h += a * np.cos(2*np.pi*m*i/N) + b * np.sin(2*np.pi*m*i/N)
    return h

def fourier_objective(params, N):
