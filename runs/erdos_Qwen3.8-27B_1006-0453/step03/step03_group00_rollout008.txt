import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, n):
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def minimize_pwl_sub(a_vec, b_vec, d_lo, d_hi):
    """Minimize max_k(a_vec[k] + d*b_vec[k]) over d in [d_lo, d_hi]."""
    if d_hi - d_lo < 1e-14:
        return (d_lo + d_hi) / 2
    lo, hi = d_lo, d_hi
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a_vec + mid * b_vec
        m = np.max(vals)
        tol = 1e-10 * (m + 1.0)
        active = vals >= (m - tol)
        if not np.any(active):
            active[np.argmax(vals)] = True
        sg_lo = np.min(b_vec[active])
        sg_hi = np.max(b_vec[active])
        if sg_lo > 1e-12:
            hi = mid
        elif sg_hi < -1e-12:
            lo = mid
        else:
            break
    d = (lo + hi) / 2.0
    f_d = np.max(a_vec + d * b_vec)
    f_lo = np.max(a_vec + d_lo * b_vec)
    f_hi = np.max(a_vec + d_hi * b_vec)
    if f_lo <= f_d and f_lo <= f_hi:
        return d_lo
    elif f_hi <= f_d and f_hi <= f_lo:
        return d_hi
    return d

def coord_descent_fast(h, n, t_end, max_sweeps=200, tol=1e-14):
    """Fast coordinate descent with incremental corr updates."""
    dx = 2.0 / n
    corr = compute_corr(h).astype(float)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    n_corr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        # Active-set ordering: prioritize coords that affect the current max
        # For simplicity, use random order but with a few passes
        order = np.random.permutation(n)
        for i in order:
            hi = h[i]
            # Compute coefficient vector: how corr changes when h[i] changes by 1

                c = 0.0
                j1 = i + lag
                if 0 <= j1 < n:
                    c += 1.0 - h[j1]
                j2 = i - lag
                if 0 <= j2 < n:
                    c -= h[j2]
                coef[idx] = c

            d_hi = 1.0 - hi
            d_opt = minimize_pwl_sub(corr, coef, d_lo, d_hi)

                corr += d_opt * coef
                new_c5 = np.max(corr) * dx

                    improved = True
                # We always accept (since it's minimization of a convex function

                    corr -= d_opt * coef

        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_sum(h, n):
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def minimize_1d(corr, coef, quad_flag, hi, n):
    """Minimize max(corr + d*coef - d^2*quad_flag) over d in [-hi, 1-hi]."""
    dx = 2.0 / n
    d_lo = -hi
    d_hi = 1.0 - hi
    if d_hi - d_lo < 1e-14:
        return (d_lo + d_hi) * 0.5
    # Grid + refine
    grid = np.linspace(d_lo, d_hi, 80)
    f_vals = np.max(corr[None, :] + grid[:, None] * coef[None, :] -
                    (grid[:, None] ** 2) * quad_flag[None, :], axis=1)
    best_i = int(np.argmin(f_vals))
    # Refine around best
    if 0 < best_i < len(grid) - 1:
        d1, d2 = grid[best_i - 1], grid[best_i + 1]
    else:
        d1, d2 = d_lo, d_hi
    grid2 = np.linspace(d1, d2, 60)
    f2 = np.max(corr[None, :] + grid2[:, None] * coef[None, :] -
                (grid2[:, None] ** 2) * quad_flag[None, :], axis=1)
    return float(grid2[int(np.argmin(f2))])

def get_coef(h, i, n, n_corr):
    """Coefficient vector for perturbing h[i] by delta (linear part)."""
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    coef = np.zeros(n_corr)
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coef[v1] += 1.0 - h[j1[v1]]
    coef[v2] -= h[j2[v2]]
    return coef

def coord_descent_fast(h, n, t_end, max_sweeps=400, verbose=False):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    center = n - 1
    quad_flag = np.zeros(n_corr)
    quad_flag[center] = 1.0
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    stalled = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            hi = h[i]
            coef = get_coef(h, i, n, n_corr)
            d_opt = minimize_1d(corr, coef, quad_flag, hi, n)
            if abs(d_opt) > 1e-14:
                new_hi = hi + d_opt
                new_hi = float(np.clip(new_hi, 0.0, 1.0))
                d_actual = new_hi - hi
                if abs(d_actual) > 1e-14:
                    h[i] = new_hi
                    corr += d_actual * coef - (d_actual ** 2) * quad_flag
                    nc5 = float(np.max(corr) * dx)
                    if nc5 < best_c5 - 1e-14:
                        best_c5 = nc5
                        best_h = h.copy()
                        improved = True
        # Project and recompute
        h = project_sum(h, n)
        corr = compute_corr(h)
        c5_now = float(np.max(corr) * dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if not improved:
            stalled += 1
            if stalled > 3:
                break
        else:
            stalled = 0
    return best_h, best_c5

def slsqp_polish(h, n, T, maxiter=600):
    target = n / 2.0
    def obj(x):
        return smooth_c5(x, T)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq',
                                 'fun': lambda x: x.sum() - target}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(np.clip(res.x, 0, 1), n)

def interpolate_resolution(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    # Phase 2: Main coordinate descent (fast, many sweeps)

            except Exception:
