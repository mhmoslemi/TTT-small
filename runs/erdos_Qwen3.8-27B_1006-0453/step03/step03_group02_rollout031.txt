        h1, c1 = optimize(cur.copy(), N, target, p1, 'act', 0.05, 0.01, 400, act_delta=0.1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        cur = h1
        # Phase 2: Active-set, tighter
        if time.time() < deadline - 80:
            p2 = time.time() + (deadline - time.time()) * 0.30
            h2, c2 = optimize(cur.copy(), N, target, p2, 'act', 0.02, 0.004, 400, act_delta=0.03)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()
            cur = h2
        # Phase 3: Top-5 sharpened
        if time.time() < deadline - 50:
            p3 = time.time() + (deadline - time.time()) * 0.30
            h3, c3 = optimize(cur.copy(), N, target, p3, 'k5', 0.008, 0.001, 350)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
            cur = h3
        # Phase 4: K=1 subgradient polish
        if time.time() < deadline - 25:
            p4 = deadline - 15
            h4, c4 = optimize(cur.copy(), N, target, p4, 'k1', 0.003, 0.0002, 300)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()
            cur = h4
        # Local search
        if time.time() < deadline - 8:
            hls, cls = local_search(cur.copy(), N, target, dx, 1500, deadline - 3)
            if cls < best_c5:
                best_c5, best_h = cls, hls.copy()
        # Random restart at first size
        if time.time() < deadline - 150 and N == 150:
            h_r = h_init + rng.randn(N) * 0.2
            h_r = project_sum(h_r, target)
            hr_dead = time.time() + (deadline - time.time()) * 0.45
            hr, cr = optimize(h_r, N, target, hr_dead, 'act', 0.04, 0.005, 500, act_delta=0.08)
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()
            elif cr < best_c5 + 0.003 and time.time() < deadline - 80:
                hp, cp = optimize(hr.copy(), N, target, deadline - 40, 'k1', 0.002, 0.0001, 300)
                if cp < best_c5:
                    best_c5, best_h = cp, hp.copy()
    # Final ultra-polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 5:
        hf, cf = optimize(best_h.copy(), Nb, Nb // 2, deadline - 0.5, 'k1', 0.001, 0.00003, 120)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
