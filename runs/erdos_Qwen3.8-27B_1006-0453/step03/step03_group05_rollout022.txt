def get_corr(h):

def slsqp_topk(h, k, n, maxiter=300):
    """SLSQP with top-k average objective."""
    def obj(x):
        corr = get_corr(x)
        dx = 2.0 / n
        top = np.sort(corr)[::-1][:k]
        return float(np.sum(top) * dx / k)
    
    def con_fun(x):
        return float(x.sum() - n / 2.0)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def softmax_slsqp(h, T, n, maxiter=300):

    def obj(x):

        dx = 2.0 / n
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    except Exception:

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def topk_obj(x, k, n):
    corr = get_corr(x)
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(x, T, n):
    corr = get_corr(x)
    dx = 2.0 / n
    m = float(np.max(corr))
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def hinge_obj(x, n, tau):
    corr = get_corr(x)
    dx = 2.0 / n
    excess = np.maximum(0.0, corr[2:2 * n - 2] - tau)
    return float(np.sum(excess * excess) * dx)

def topk_grad(h, k, n):
    corr = get_corr(h)
    dx = 2.0 / n
    idx = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(n)
    j = np.arange(n)
    for ki in idx:
        s = int(ki - (n - 1))
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            g = np.zeros(n)
            m1 = j >= s
            g[m1] = 1.0 - h[j[m1] - s]
            m2 = j < n - s
            g[m2] -= h[j[m2] + s]
            grad += g
        else:
            ss = -s
            g = np.zeros(n)
            m1 = j < n - ss
            g[m1] = 1.0 - h[j[m1] + ss]
            m2 = j >= ss
            g[m2] -= h[j[m2] - ss]
            grad += g
    return grad * (dx / k)

def adam_opt(h0, max_iter, top_k, lr, n, deadline):
    h = h0.copy()
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    b1, b2, eps = 0.8, 0.999, 1e-8
    for t in range(1, max_iter + 1):
        if t % 200 == 0 and time.time() > deadline:
            break
        grad = topk_grad(h, top_k, n)
        m_s = b1 * m_s + (1.0 - b1) * grad
        v_s = b2 * v_s + (1.0 - b2) * (grad * grad)
        m_hat = m_s / (1.0 - b1 ** t)
        v_hat = v_s / (1.0 - b2 ** t)
        h = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_opt(h, obj_func, args, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def make_binary_init(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h

def make_cosine_init(n):
    x = np.linspace(0, 2, n)
    h = np.clip(0.5 + 0.5 * np.cos(np.pi * x), 0, 1)
    return h

def make_2freq_init(n):
    x = np.linspace(0, 2, n)
    h = np.clip(0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x), 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_binary_init(200)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam with exact gradient (key innovation vs subgradient)
    for tk in [3, 5]:
        h_a, c_a = adam_opt(best_h, 4000, tk, 0.004, best_n, deadline - 600)
        if c_a < best_c5:
            best_c5, best_h = c_a, h_a

    # Phase 2: SLSQP top-k polish
    for k in [3, 5]:
        if time.time() > deadline - 500:
            break
        hs = slsqp_opt(best_h, topk_obj, (k, best_n), best_n, 500)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 3: Hinge-loss SLSQP (new objective)
    if time.time() < deadline - 350:
        corr_now = get_corr(best_h)
        tau = float(np.max(corr_now)) * 0.999
        hs = slsqp_opt(best_h, hinge_obj, (best_n, tau), best_n, 400)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        hs = slsqp_opt(best_h, hinge_obj, (best_n, tau * 0.95), best_n, 400)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 4: Softmax SLSQP
    for T in [1000, 5000]:
        if time.time() > deadline - 300:
            break
        hs = slsqp_opt(best_h, softmax_obj, (T, best_n), best_n, 400)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 5: Multi-start from diverse inits
    inits = [make_binary_init(best_n), make_cosine_init(best_n), make_2freq_init(best_n)]
    inits.append(np.random.random(best_n))
    for h_init in inits:
        if time.time() > deadline - 300:
            break
        h_init = project_feasible(h_init, best_n)
        h_a, c_a = adam_opt(h_init, 1500, 3, 0.008, best_n, deadline - 30)
        h_s = slsqp_opt(h_a, topk_obj, (3, best_n), best_n, 300)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s

    # Phase 6: Multi-resolution (go higher, check if better)
    for nr in [300, 400]:
        if time.time() > deadline - 350:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        h_a, c_a = adam_opt(hu, 2000, 3, 0.003, nr, deadline - 60)
        h_s = slsqp_opt(h_a, topk_obj, (3, nr), nr, 300)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5, best_h, best_n = c_s, h_s, nr
        else:
            x_dn = np.linspace(0, 2, best_n)
            hd = np.interp(x_dn, x_new, h_s)
            hd = project_feasible(hd, best_n)
            c_d = compute_c5(hd)
            if c_d < best_c5:
                best_c5, best_h = c_d, hd

    # Try lower resolution too
    if best_n > 100 and time.time() < deadline - 200:
        x_old = np.linspace(0, 2, best_n)
        x_low = np.linspace(0, 2, 100)
        h_low = np.interp(x_low, x_old, best_h)
        h_low = project_feasible(h_low, 100)
        h_a, c_a = adam_opt(h_low, 3000, 3, 0.006, 100, deadline - 60)
        h_s = slsqp_opt(h_a, topk_obj, (3, 100), 100, 500)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5, best_h, best_n = c_s, h_s, 100

    # Phase 7: Perturbation + quick refine
    n_trials = min(25, int(max(0, deadline - time.time()) // 10))
    for trial in range(n_trials):
        if time.time() > deadline - 80:
            break
        sc = 0.012 * (1.0 - trial / max(1, n_trials)) + 0.004
        hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        hp = slsqp_opt(hp, topk_obj, (3, best_n), best_n, 200)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_c5, best_h = cp, hp

    # Phase 8: Final intensive polish
    if time.time() < deadline - 30:
        hf = slsqp_opt(best_h, topk_obj, (3, best_n), best_n, 600)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hf2 = adam_opt(best_h, 1000, 3, 0.001, best_n, deadline - 10)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2
        hf3 = slsqp_opt(best_h, softmax_obj, (8000, best_n), best_n, 400)
        cf3 = compute_c5(hf3)
        if cf3 < best_c5:
            best_c5, best_h = cf3, hf3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
