def compute_corr(h):
    return np.correlate(h, 1.0h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def grad_sm(h, N, k, temp):
    corr = compute_corr(h)
    c = N-1
    pk = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    sh = (corr[top]-pk)/temp
    w = np.exp(sh)
    w /= w.sum()
    g = np.zeros(N)
    for i, wi in zip(top, w):
        L = i-c
        if L == 0:
            g += wi*(1.0-2.0*h)
        elif L > 0:
            g[L:] += wi*(1.0-h[:N-L])
            g[:N-L] -= wi*h[L:]
        else:
            M = -L
            if M < N:
                g[:N-M] += wi*(1.0-h[M:])
                g[M:] -= wi*h[:N-M]
    return g, pk
