import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h2 = np.clip(h - mid, 0.0, 1.0)
        if float(h2.sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fast_grad(h, N, top_idx, top_w, center):
    grad = np.zeros(N)
    for idx, w in zip(top_idx, top_w):
        if w < 1e-7:
            continue
        L = idx - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        else:
            M = -L
            grad[:N - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:N - M]
    return grad

def optimize_phase(h, N, target, deadline, n_top, tau, lr, mom, lr_decay, decay_period, binary_period):
    dx = 2.0 / N
    center = N - 1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    it = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_top:]
    top_w = np.ones(n_top) / n_top

    while time.time() < deadline:
        it += 1
        if it % 3 == 1:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
            max_c = corr.max()
            w = np.exp((corr - max_c) / max(tau, 1.0))
            w /= w.sum()
            top_idx = np.argsort(corr)[-n_top:]
            top_w = w[top_idx].copy()
            top_w /= top_w.sum()

        grad = fast_grad(h, N, top_idx, top_w, center)
        v = mom * v + grad
        h = project_sum(h - lr * v, target)

        if binary_period > 0 and it % binary_period == 0:
            push = min(0.35, 0.05 + 0.05 * (it / 200.0))
            h_bin = np.where(h > 0.5, 1.0, 0.0)
            h = (1.0 - push) * h + push * h_bin
            h = project_sum(h, target)

        if it % decay_period == 0:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_c5

def block_swap_search(h, N, target, deadline, seed=99):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    h_s = np.convolve(h, np.ones(7) / 7, mode='same')
    trans = np.where(np.diff((h_s > 0.5).astype(int)) != 0)[0]
    if len(trans) == 0:
        return best_h, best_c5
    blocks = []
    start = 0
    for t in trans:
        if t + 1 > start:
            blocks.append((start, t + 1))
        start = t + 1
    if start < N:
        blocks.append((start, N))
    attempts = 0
    while time.time() < deadline and attempts < 300:
        attempts += 1
        if len(blocks) < 2:
            break
        i, j = rng.choice(len(blocks), 2, replace=False)
        si, ei = blocks[i]
        sj, ej = blocks[j]
        if ei - si != ej - sj:
            continue
        tmp = h[si:ei].copy()
        h[si:ei] = h[sj:ej]
        h[sj:ej] = tmp
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[sj:ej] = h[si:ei]
            h[si:ei] = tmp
    return best_h, best_c5

def local_polish(h, N, target, deadline, seed=42):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    stall = 0
    while time.time() < deadline:
        improved = False
        order = rng.permutation(N)
        for i in order:
            if time.time() > deadline:
                return best_h, best_c5
            orig = best_h[i]
            for trial in (0.0, 1.0, orig - 0.05, orig + 0.05):
                if trial < 0.0 or trial > 1.0 or abs(trial - orig) < 1e-7:
                    continue
                best_h[i] = trial
                best_h = project_sum(best_h, target)
                c5 = compute_c5(best_h, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
                    break
                best_h[i] = orig
            if improved:
                break
        for _ in range(80):
            if time.time() > deadline:
                return best_h, best_c5
            i, j = rng.choice(N, 2, replace=False)
            delta = rng.choice([-1.0, 1.0]) * rng.choice([0.05, 0.1, 0.2, 0.3])
            if not (0.0 <= best_h[i] + delta <= 1.0 and 0.0 <= best_h[j] - delta <= 1.0):
                continue
            best_h[i] += delta
            best_h[j] -= delta
            c5 = compute_c5(best_h, dx)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                improved = True
            else:
                best_h[i] -= delta
                best_h[j] += delta
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 4:
                break
    return best_h, best_c5

def multi_start_refine(h, N, target, deadline, n_starts, seed):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    per_start = (deadline - time.time()) / max(n_starts, 1)
    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        h_p = best_h.copy()
        n_pert = max(3, N // 15)
        idx = rng.choice(N, n_pert, replace=False)
        h_p[idx] += rng.randn(n_pert) * (0.12 + 0.04 * s)
        h_p = project_sum(h_p, target)
        t_end = min(deadline - 3, time.time() + per_start)
        h_o, c_o = optimize_phase(h_p, N, target, t_end,
                                  n_top=8, tau=4.0, lr=0.012, mom=0.88,
                                  lr_decay=0.96, decay_period=35, binary_period=200)
        if c_o < best_c5:
            best_c5 = c_o
            best_h = h_o.copy()
    return best_h, best_c5
