def project(h, n):

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def _topk_obj(x, T, k):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    tk = corr[idx]
    m = np.max(tk)
    return (m + np.log(np.sum(np.exp(T * (tk - m)))) / T) * dx

def sum_fn(x):
    return x.sum() - len(x) / 2.0

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14:
            return h
        h += r * (diff / t)
    else:
        r = h.copy()
        t = r.sum()
        if t < 1e-14:
            return h
        h += r * (diff / t)
    return np.clip(h, 0.0, 1.0)

def do_slsqp(h0, T, k, n, mi, ft):
    cons = [{'type': 'eq', 'fun': sum_fn}]
    bnds = [(0.0, 1.0)] * n
    if k is not None:
        res = minimize(_topk_obj, h0, args=(T, k), method='SLSQP',
                       bounds=bnds, constraints=cons, options={'maxiter': mi, 'ftol': ft})
    else:
        res = minimize(_lse_obj, h0, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons, options={'maxiter': mi, 'ftol': ft})
    return project(np.clip(res.x, 0, 1), n)

def pgd_optimize(h, n, T, n_iters, step_init, accept_bad):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        T_cur = T * (1.0 + 4.0 * frac)
        h1m = 1.0 - best
        corr = np.correlate(best, h1m, mode='full')
        mcorr = np.max(corr)
        w = np.exp(T_cur * (corr - mcorr))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(w)
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        grad = np.zeros(n)
        nonz = np.where(w > 1e-12)[0]
        for ki in nonz:
            shift = ki - (n - 1)
            wt = w[ki]
            if shift >= 0:
                grad[:n - shift] += wt * h1m[shift:]
                grad[shift:] -= wt * best[:n - shift]
            else:
                s = -shift
                grad[s:] += wt * h1m[:-s]
                grad[:n - s] -= wt * best[s:]
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = step_init * (1.0 - 0.6 * frac)
        step = min(step, 1.0 / (gnorm + 1e-10))
        h_new = best + step * grad
        h_new = project(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new
            best_c5 = c5_new
        elif accept_bad and np.random.rand() < np.exp(-(c5_new - best_c5) / max(0.0005 * (1.0 - frac), 1e-8)):
            best = h_new
            best_c5 = c5_new
    return best, best_c5

def coord_search(h, n, n_iters):
    best = h.copy()
    best_c5 = compute_c5(best)
    deltas = np.array([0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.04, -0.04])
    for it in range(n_iters):
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        top_idx = np.argsort(corr)[-4:]
        for ki in top_idx:
            shift = ki - (n - 1)
            lo = max(0, - shift if shift < 0 else 0)
            hi = min(n, n - shift if shift >= 0 else n)
            for j in range(lo, hi, 2):
                for d in deltas:
                    v = best[j] + d
                    if v < 0.0 or v > 1.0:
                        continue
                    h_try = best.copy()
                    h_try[j] = v
                    h_try = project(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            for _ in range(3):
                j = np.random.randint(n)
                d = np.random.choice(deltas)
                v = best[j] + d
                if v < 0 or v > 1:
                    continue
                h_try = best.copy()
                h_try[j] = v
                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best = h_try
                    improved = True
                    break
    return best, best_c5

def basin_hop(h0, n, n_iter, t0, t1):
    best = h0.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        temp = t0 + (t1 - t0) * (it / max(n_iter - 1, 1))
        step = 0.06 * (1.0 - 0.7 * it / max(n_iter, 1)) + 0.008
        h_pert = best + step * np.random.randn(n)
        r = np.random.rand()
        if r < 0.3:
            blen = np.random.randint(3, n // 6 + 1)
            st = np.random.randint(0, max(n - blen, 1))
            h_pert = best.copy()
            h_pert[st:st + blen] += step * (2.0 * (np.random.rand(blen) < 0.5) - 1.0)
        h_pert = project(h_pert, n)
        T_opt = 500
        k_opt = None
        if np.random.rand() < 0.3:
            k_opt = int(np.random.choice([3, 5, 8]))
        try:
            hc = do_slsqp(h_pert, T_opt, k_opt, n, 250, 1e-12)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best = hc
                best_c5 = c5v
            elif np.random.rand() < np.exp(-(c5v - best_c5) / max(temp, 1e-9)):
                best = hc
                best_c5 = c5v
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(80)
        best_h[:40] = 1.0
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)

    # Phase 1: SLSQP sweep on initial
    if elapsed() < budget_s * 0.08:
        for T in [50, 200, 1000, 5000]:
            for k in [None, 5, 10]:
                if elapsed() > budget_s * 0.08:
                    break
                try:
                    hc = do_slsqp(best_h, T, k, n0, 400, 1e-14)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    # Phase 2: PGD at initial resolution
    if elapsed() < budget_s * 0.15:
        best_h, best_c5 = pgd_optimize(best_h, n0, T=200, n_iters=800, step_init=0.02, accept_bad=True)

    # Phase 3: SLSQP + PGD at n=200
    if elapsed() < budget_s * 0.55:
        n_new = 200
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project(h_new, n_new)
        for T in [100, 500, 2000]:
            for k in [None, 5]:
                if elapsed() > budget_s * 0.30:
                    break
                try:
                    hc = do_slsqp(h_new, T, k, n_new, 350, 1e-13)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                        h_new = hc.copy()
                except Exception:
                    pass
        if elapsed() < budget_s * 0.45:
            h_new, c5v = pgd_optimize(h_new, n_new, T=300, n_iters=600, step_init=0.015, accept_bad=True)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        # Basin hop at n=200
        if elapsed() < budget_s * 0.55:
            best_h, best_c5 = basin_hop(best_h, len(best_h), 30, 0.004, 0.0002)

    # Phase 4: Try n=240
    if elapsed() < budget_s * 0.70:
        n2 = 240
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h240 = np.interp(x_new, x_old, best_h)
        h240 = project(h240, n2)
        c5_240 = compute_c5(h240)
        if c5_240 <= best_c5 + 0.003:
            for T in [200, 1000, 5000]:
                for k in [None, 5]:
                    if elapsed() > budget_s * 0.62:
                        break
                    try:
                        hc = do_slsqp(h240, T, k, n2, 300, 1e-13)
                        c5v = compute_c5(hc)
                        if c5v < best_c5:
                            best_c5 = c5v
                            best_h = hc
                            h240 = hc.copy()
                    except Exception:
                        pass
            if elapsed() < budget_s * 0.70:
                h240, c5v = pgd_optimize(h240, n2, T=200, n_iters=400, step_init=0.01, accept_bad=True)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h240

    # Phase 5: Coordinate descent
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.85:
        best_h, best_c5 = coord_search(best_h, n_cur, 300)

    # Phase 6: Final PGD + SLSQP polish
    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = pgd_optimize(best_h, n_cur, T=500, n_iters=500, step_init=0.008, accept_bad=False)
    for T in [10000, 50000]:
        if elapsed() > budget_s * 0.97:
            break
        try:
            hc = do_slsqp(best_h, T, None, n_cur, 1500, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    for k in [3, 5]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = do_slsqp(best_h, 3000, k, n_cur, 1000, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final coordinate search
    if elapsed() < budget_s * 0.998:
        best_h, best_c5 = coord_search(best_h, n_cur, 100)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
