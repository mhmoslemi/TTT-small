def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            rm = 1.0 - h; t = rm.sum()
            if t > 1e-14: h += rm * (d / t)
        else:
            rm = h.copy(); t = rm.sum()
            if t > 1e-14: h += rm * (d / t)
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(c):

    return smooth_c5(x, T, topk)
