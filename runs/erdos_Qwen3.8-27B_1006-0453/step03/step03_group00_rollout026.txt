    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:

        if tot < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_coef(h, n, j):
    k_idx = np.arange(2 * n - 1)
    i1 = j + k_idx - n + 1
    i2 = j - k_idx + n - 1
    coef = np.zeros(2 * n - 1)
    v1 = (i1 >= 0) & (i1 < n)
    v2 = (i2 >= 0) & (i2 < n)
    coef[v1] += 1.0 - h[i1[v1]]
    coef[v2] += -h[i2[v2]]
    return coef

def coord_descent(h, n, max_sweeps=200, t_end=None, tol=1e-11):
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = np.max(corr) * (2.0 / n)
    x_grid = np.linspace(0.0, 1.0, 81)
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        for j in range(n):
            if t_end is not None and (sweep % 20 == 19 and time.time() > t_end):
                return h, best_c5
            old = h[j]
            coef = get_coef(h, n, j)
            corr_0 = corr - old * coef
            vals = corr_0[None, :] + x_grid[:, None] * coef[None, :]
            obj = np.max(vals, axis=1)
            idx_min = int(np.argmin(obj))
            x_best = x_grid[idx_min]
            if idx_min > 0 and idx_min < 80:
                for d in (0.006, 0.012, -0.006, -0.012):
                    xt = x_best + d
                    if 0.0 <= xt <= 1.0:
                        vt = np.max(corr_0 + xt * coef)
                        if vt < obj[idx_min] - 1e-14:
                            x_best = xt
                            obj[idx_min] = vt
            if abs(x_best - old) > 1e-10:
                h[j] = x_best
                corr = corr_0 + x_best * coef
                improved = True
        c5_now = np.max(corr) * (2.0 / n)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not improved:
            break
    return h, best_c5

def coord_descent_exact(h, n, max_sweeps=50, t_end=None):
    """Exact vertex-based 1D minimization for final polish."""
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = np.max(corr) * (2.0 / n)
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        for j in range(n):
            if t_end is not None and time.time() > t_end:
                return h, best_c5
            old = h[j]
            coef = get_coef(h, n, j)
            corr_0 = corr - old * coef
            candidates = [0.0, 1.0]
            k_arr = np.arange(2 * n - 1)
            for a in range(2 * n - 2):
                for b in range(a + 1, 2 * n - 1):
                    dc = coef[a] - coef[b]
                    if abs(dc) < 1e-14:
                        continue
                    xi = (corr_0[b] - corr_0[a]) / dc
                    if -1e-9 <= xi <= 1.0 + 1e-9:

            best_x = old
            best_v = np.max(corr_0 + old * coef)

def interp_to(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def smooth_c5(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def slsqp_polish(h, n, T, maxiter=400):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return project(h, n)
