def compute_c5(h):

    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    max_shifted = np.max(shifted)
    val = m + (max_shifted / T + np.log(np.sum(np.exp(shifted - max_shifted))) / T)
