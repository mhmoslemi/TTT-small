    if time.time() < dl - 1:
        hl, cl = local_search(best_h, Nb, tgt_b, dx_b, dl - 0.5, seed=999)
        if cl < best_c5:
            best_c5, best_h = cl, hl
    best_h = project_sum(best_h, float(len(best_h)) / 2.0)
    fc = compute_c5(best_h, 2.0 / len(best_h))
    if fc < best_c5:
        best_c5 = fc
    return best_h, best_c5, len(best_h)
