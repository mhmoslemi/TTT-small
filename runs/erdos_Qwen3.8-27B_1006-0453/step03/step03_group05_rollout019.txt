    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += h * (diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return x.sum() - len(x) / 2.0

def topk_mean_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.mean(top)) * dx

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = float(np.max(corr))
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val) * dx

            hp = block_shift(best_h, best_n)
        elif r < 0.60:
            hp = binary_snap(best_h, best_n)
            hp = project_feasible(hp + np.random.randn(best_n) * 0.02, best_n)
        elif r < 0.75:
            hp = gaussian_perturb(best_h, best_n, 0.01)
            hp = slsqp_opt(hp, topk_mean_obj, (2, best_n), 100, 1e-12)
        else:
            hp = best_h.copy()
            idx = np.random.choice(best_n, best_n // 4, replace=False)
            hp[idx] = np.clip(hp[idx] + np.random.randn(len(idx)) * 0.05, 0, 1)
            hp = project_feasible(hp, best_n)
        # Local refine: SLSQP then topk
        ho = slsqp_opt(hp, topk_mean_obj, (3, best_n), 250, 1e-13)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho
        # Secondary refine with softmax
        if time.time() - t0 < budget_s * 0.65:
            ho2 = slsqp_opt(hp, softmax_obj, (500, best_n), 200, 1e-13)
            co2 = compute_c5(ho2)
            if co2 < best_c5:
                best_c5, best_h = co2, ho2

    # Phase 5: Multi-resolution
    for nr in [90, 120, 150, 250]:
        if time.time() - t0 > budget_s * 0.80:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hh = slsqp_opt(hu, topk_mean_obj, (3, nr), 300, 1e-13)
        c5_hi = compute_c5(hh)
        # Also try with perturbation
        hu2 = project_feasible(hu + np.random.randn(nr) * 0.015, nr)
        hh2 = slsqp_opt(hu2, softmax_obj, (400, nr), 250, 1e-13)
        c5_hi2 = compute_c5(hh2)
        if min(c5_hi, c5_hi2) < best_c5:
            if c5_hi < c5_hi2:
                best_c5, best_h, best_n = c5_hi, hh, nr
            else:
                best_c5, best_h, best_n = c5_hi2, hh2, nr
        elif nr < best_n:
            # Try downsample from current best to nr for comparison
            pass

    # Phase 6: Powell's method (derivative-free, handles non-smooth max)
    if time.time() - t0 < budget_s * 0.90:
        hp = powell_opt(best_h, 200)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_c5, best_h = cp, hp

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.95:
        hf = slsqp_opt(best_h, topk_mean_obj, (2, best_n), 600, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hf2 = slsqp_opt(best_h, topk_mean_obj, (3, best_n), 400, 1e-15)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2

    # Phase 8: Micro subgradient + final check
    if time.time() - t0 < budget_s * 0.985:
        hm, cm = subgrad_momentum(best_h, 200, 2, 0.003, 0.999, 0.92)
        if cm < best_c5:
            best_c5, best_h = cm, hm
        hf3 = slsqp_opt(best_h, topk_mean_obj, (2, best_n), 200, 1e-15)
        cf3 = compute_c5(hf3)
        if cf3 < best_c5:
            best_c5, best_h = cf3, hf3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
