import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_onto_simplex_bounds(h):
    """Project h to [0,1]^n with sum = n/2 (approximately)."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        frac = 1.0 + diff / s
        h = h * frac
    return np.clip(h, 0.0, 1.0)

def slsqp_minimize(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_onto_simplex_bounds(np.clip(res.x, 0, 1))

def subgrad_step(h, alpha, n_topk=5):
    """One subgradient step on the max (using top-k active set)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k indices
    topk_idx = np.argsort(corr)[-n_topk:]
    # Subgradient: average of gradients of active corr_k
    # d(corr[m])/d(h[i]) = (1-h[j1]) - h[j2] where j1=i+m-n+1, j2=i-m+n-1
    # Vectorized:
    i_idx = np.arange(n)[:, None]
    m_idx = topk_idx[None, :]
    j1 = i_idx + m_idx - n + 1
    j2 = i_idx - m_idx + n - 1
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    j1c = np.clip(j1, 0, n-1)
    j2c = np.clip(j2, 0, n-1)
    # Gradient of corr[m] w.r.t. h[i]:
    d_out = (1.0 - h[j1c]) * valid1.astype(float) - h[j2c] * valid2.astype(float)
    # Average over active set
    grad = (d_out.mean(axis=1)) * dx
    # Subgradient step (we want to DECREASE the max, so subtract gradient)
    h_new = h - alpha * grad
    return project_onto_simplex_bounds(h_new)

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_step(h, alpha, n_topk=10):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    i_idx = np.arange(n)[:, None]
    m_idx = topk_idx[None, :]
    j1 = i_idx + m_idx - n + 1
    j2 = i_idx - m_idx + n + 1 - 2  # i - m + n - 1
    j2 = i_idx - m_idx + n - 1
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    j1c = np.clip(j1, 0, n - 1)
    j2c = np.clip(j2, 0, n - 1)
    d_out = (1.0 - h[j1c]) * valid1.astype(np.float64) - h[j2c] * valid2.astype(np.float64)
    grad = d_out.mean(axis=1) * dx
    h_new = h - alpha * grad
    return project_h(h_new)

    j2 = i - k_range

    valid2 = (j2 >= 0) & (j2 < n)

            best_c5 = c5v
            best_h = hc

    # Phase 3: Basin hopping (dual track)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(80):
        if elapsed() > budget_s * 0.52:
            break
        cool = 1.0 - 0.9 * (it / 80.0)
        mod = it % 4
        if mod == 0:
            pert = (0.05 + 0.15 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 9)
            phase = np.random.uniform(0, 2 * np.pi)
            pert = (0.04 + 0.12 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif mod == 2:
            mask = np.random.random(n) < 0.3
            pert = np.zeros(n)
            pert[mask] = (0.1 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            pert = (0.03 + 0.08 * cool) * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T_bh = 500 + 4000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=600, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.002 * cool + 0.0002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Extended coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.75, max_sweeps=200)

    # Phase 5: Basin hopping round 2 (finer)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(40):
        if elapsed() > budget_s * 0.84:
            break
        cool = 1.0 - 0.8 * (it / 40.0)
        step = 0.02 + 0.06 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = project_h(h_pert)
        T_bh = 1000 + 5000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=100)

    # Phase 7: Ultra-high-T SLSQP polish
    for T in [200000, 1000000]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coordinate descent
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=50)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
