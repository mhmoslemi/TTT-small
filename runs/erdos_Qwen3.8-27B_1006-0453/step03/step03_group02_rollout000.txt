                           15.0, 300.0, 40, 5, 0.04, 0.001)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    if time.time() > deadline - 80:
        return best_h, best_c5, N

    # Phase 2: Structured starts tournament (25%)
    p2_dead = t0 + (budget_s - 5) * 0.60
    n_starts = 5
    per_start = max((p2_dead - time.time()) / n_starts - 5, 15)
    starters = []
    for si in range(n_starts):
        sh = structured_start(N, target, rng, style=si % 4)
        # Also try mixing with best
        if si >= 2:
            sh = project_sum(0.5 * best_h + 0.5 * sh, target)
        starters.append(sh)
    for si, sh in enumerate(starters):
        rem = p2_dead - time.time()
        if rem < 15:
            break
        seg_dead = min(time.time() + min(per_start, rem - 5), p2_dead - 3)
        ho, co = adam_optimize(sh, N, target, dx, seg_dead,
                               30.0, 600.0, 25, 4, 0.03, 0.002)
        if co < best_c5:
            best_c5, best_h = co, ho
    if time.time() > deadline - 50:
        return best_h, best_c5, N

    # Phase 3: Deep optimization on current best (20%)
    p3_dead = t0 + (budget_s - 5) * 0.80
    h3, c3 = adam_optimize(best_h.copy(), N, target, dx, p3_dead,
                           100.0, 2000.0, 15, 3, 0.01, 0.0003)
    if c3 < best_c5:
        best_c5, best_h = c3, h3
    if time.time() > deadline - 30:
        return best_h, best_c5, N

    # Phase 4: Local search (15%)
    p4_dead = t0 + (budget_s - 5) * 0.95
    best_h, best_c5 = local_search(best_h, N, target, dx,
                                   n_trials=10000, deadline=p4_dead, rng=rng)

    # Phase 5: Final ultra-polish
    if time.time() < deadline - 3:
        hf, cf = adam_optimize(best_h.copy(), N, target, dx, deadline - 1,
                               2000.0, 8000.0, 4, 2, 0.002, 0.00005)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, N
