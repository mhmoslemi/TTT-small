def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            return np.clip(h, 0.0, 1.0)
        h[mask] += diff / cnt
    else:
        mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            return np.clip(h, 0.0, 1.0)
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def corr_val_grad(h, s, n):
    dx = 2.0 / n
    if s >= 0:
        m = n - s
        val = float(np.dot(h[s:], 1.0 - h[:m]))
        grad = np.zeros(n)
        grad[s:] += (1.0 - h[:m])
        grad[:m] -= h[s:]
    else:
        ma = -s
        m = n - ma
        val = float(np.dot(h[:m], 1.0 - h[ma:]))
        grad = np.zeros(n)
        grad[:m] += (1.0 - h[ma:])
        grad[ma:] -= h[:m]
    return val * dx, grad * dx
