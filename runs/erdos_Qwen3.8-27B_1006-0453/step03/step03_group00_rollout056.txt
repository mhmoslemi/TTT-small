    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m))) + 1e-300) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)]*n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0.0, 1.0), n)

def cd_sweep_fast(h, n, t_end):
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    dx = 2.0 / n
    for i in range(n):
        if time.time() > t_end:
            break
        # Compute coef for coordinate i using current h_cur
        coef = np.zeros(2 * n - 1)
        for k in range(-(n-1), n):
            c = 0.0
            j1 = i + k
            j2 = i - k
            if 0 <= j1 < n:
                c += 1.0 - h_cur[j1]
            if 0 <= j2 < n and j2 != j1:
                c += -h_cur[j2]
            elif k == 0:
                c += -h_cur[j2]
            coef[k + n - 1] = c
        
        # Now corr represents the correlation with h_cur[i] = current value
        # We want to find hi that minimizes max(corr_old + (hi - h_cur[i]) * coef)
        # where corr_old = corr - h_cur[i] * coef (the correlation when h[i]=0)
        # Actually: corr_with_hi = (corr - h_cur[i]*coef) + hi*coef
        base = corr - h_cur[i] * coef  # correlation when h[i] = 0
        old_val = h_cur[i]
        
        def f1d(hi):
            return np.max(base + hi * coef) * dx
        
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
        hi_new = res.x
        
        if abs(hi_new - old_val) > 1e-12:
            corr += (hi_new - old_val) * coef
            h_cur[i] = hi_new
    
    return h_cur, float(np.max(corr) * dx)
