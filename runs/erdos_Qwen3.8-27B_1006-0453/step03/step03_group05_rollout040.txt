The projection is the tricky part. Project onto {h : 0≤h≤1, sum(h)=n/2}.

The projection onto {0≤h≤1} is just clipping. The projection onto sum(h)=n/2 with box constraints is:
- Binary search on a threshold: find λ such that sum(clip(h - λ, 0, 1)) = n/2

This is the standard approach for Euclidean projection onto the intersection of a box and a hyperplane.

```python
        s = x.sum()

            lo = mid
        else:

        x = np.clip(x, 0, 1)
```

```python
    n = len(v)
    if abs(v.sum() - target) < 1e-15:
        return np.clip(v, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target:

            hi = mid
    lam = (lo + hi) / 2

    diff = target - x.sum()

def grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        m = -k
        if n - m > 0:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]

        g += grad_for_lag(h, k)
    g /= len(active_lags)
    g *= dx
    h_new = project_box_sum(h - alpha * g, n / 2.0)
    return h_new

def bb_step_size(h, f_prev, g_prev, h_new, f_new, alpha_prev):
    if g_prev is None:
        return alpha_prev
    s = h_new - h
    y = 2.0 / len(h) * (np.correlate(h_new, 1-h_new, 'full') - np.correlate(h, 1-h, 'full'))
```

Let me also think about the coordinate descent more carefully:

For each coordinate i, fixing all other h[j] (j≠i):
- Each corr_k becomes: A_k + b_k * h[i] (linear in h[i])
  where A_k = corr_k - b_k * h[i]_old and b_k = d(corr_k)/d(h[i])
  
- The objective max_k (dx * (A_k + b_k * h[i])) is a convex piecewise-linear function of h[i]
- We want to minimize it over h[i] ∈ [0,1] (and subject to sum constraint)

The optimal h[i] is where the derivative changes sign, or at the boundary. We can find it by:
- The breakpoints are where A_k + b_k * h[i] = A_l + b_l * h[i] for some k,l
- Or at h[i] = 0 or h[i] = 1

This is a 1D convex minimization, solvable by ternary search or by finding the point where the slope changes from negative to positive.

Actually, for a piecewise linear convex function, the minimum is at a point where the subgradient contains 0. The subgradient of max_k (A_k + b_k * x) at x is:
{max_k b_k * 1_{k active} : ...} 

More precisely, the subdifferential at x is the convex hull of {b_k : c_k(x) = max(c)}.

The minimum is where 0 ∈ conv({b_k : k active}), i.e., min_k b_k ≤ 0 ≤ max_k b_k among active k's. Or at a boundary.

In practice, I'll just do a simple scan or bisection on [0,1] to find the minimizer.

But the sum constraint complicates this. If I change h[i], I need to compensate elsewhere. One approach:
- After updating h[i], rescale all other coordinates to maintain the sum
- Or: incorporate the sum as a penalty

Let me simplify: do coordinate descent WITHOUT the sum constraint explicitly, then project at the end of each full sweep.

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    x = np.clip(v - lam, 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1 - 1e-10)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, 1)
    return x

def grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if n > k:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        m = -k
        if n > m:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    return g

def subgrad_method(h, n_iters, k_active, alpha0, decay):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for _ in range(n_iters):
        corr = np.correlate(h, 1-h, mode='full')
        subj = np.argpartition(corr, -k_active)[-k_active:]
        lags = subj - (n-1)
        g = np.zeros(n)
        for ki in lags:
            g += grad_for_lag(h, ki)
        g *= dx / k_active
        h_new = project_box_sum(h - alpha * g, n/2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best = c5_new, h_new.copy()
        if c5_new <= compute_c5(h):
            h = h_new
        else:
            alpha *= 0.8
        alpha *= decay
    return best, best_c5

def coord_descent(h, n_sweeps=5):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        for i in range(n):
            # Minimize over h[i] with others fixed
            # Each corr_k is linear in h[i]: corr_k = A_k + b_k * h[i]
            # Find optimal h[i] in [0,1]
            bi = np.zeros(2*n-1)  # coefficient of h[i] in each corr
            for k in range(-(n-1), n):
                g = grad_for_lag(h, k)
                bi[k + n - 1] = g[i]
            # corr_k at h[i]=0: corr_k(h) - bi[k] * h[i]
            corr_base = np.correlate(h, 1-h, mode='full') - bi * h[i]
            # Minimize max_k dx*(corr_base[k] + bi[k]*x) over x in [0,1]
            # This is a 1D convex piecewise-linear minimization
            # Try: sweep candidate points
            # The minimum of max of lines is where the upper envelope has slope 0
            # Simple approach: ternary search
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi-lo)/3
                m2 = hi - (hi-lo)/3
                f1 = np.max(corr_base + bi*m1)
                f2 = np.max(corr_base + bi*m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_opt = (lo+hi)/2
            h_new = h.copy()
            h_new[i] = x_opt
            h_new = project_box_sum(h_new, n/2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best = c5_new, h_new.copy()
            h = h_new
    return best, best_c5

def lin_LP_step(h, delta):
    """Solve linearized LP to find improvement direction."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    
    cv = cp.Variable(n)
    t = cp.Variable()
    cons = [cv >= -h + 1e-10, cv <= 1-h - 1e-10, cp.sum(cv) == 0,
            cv >= -delta, cv <= delta]
    
    # Add linearized constraints for top active lags
    top_k = np.argpartition(corr, -10)[-10:]
    for p in top_k:
        k = int(p) - (n-1)
        g = grad_for_lag(h, k) * dx
        cons.append(t >= corr[p]*dx + g @ cv)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS)
        if cv.value is not None:
            h_new = project_box_sum(h + cv.value, n/2.0)
            return h_new
    except:
        pass
    return h

def slsqp_topk(h, k, maxiter=200):
    n = len(h)
    dx = 2.0/n
    def obj(x):
        corr = np.correlate(x, 1-x, mode='full')
        top = np.sort(corr)[::-1][:k]
        return float(np.sum(top)*dx/k)
    def obj_grad(x):
        corr = np.correlate(x, 1-x, mode='full')
        subj = np.argpartition(corr, -k)[-k:]
        g = np.zeros(n)
        for p in subj:
            g += grad_for_lag(x, int(p)-(n-1))
        return g * dx / k
    def sum_con(x):
        return float(np.sum(x) - n/2.0)
    bnds = [(0,1)]*n
    cons = [{'type':'eq','fun':sum_con}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter':maxiter, 'ftol':1e-14})
        return project_box_sum(res.x, n/2.0)
    except:
        return h

    np.random.seed(seed)
    t0 = time.time()

        hsg, c5sg = subgrad_method(best_h, 300, ka, 0.05, 0.9997)

        for delta in [0.15, 0.08, 0.04, 0.02]:
            if time.time()-t0 > budget_s*0.35: break
            for _ in range(5):

                hlp = lin_LP_step(best_h, delta)
                clp = compute_c5(hlp)
                if clp < best_c5: best_c5, best_h = clp, hlp
                else: break
    
    # Phase 4: SLSQP with analytical gradient (top-k)
    if time.time()-t0 < budget_s*0.50:

            if time.time()-t0 > budget_s*0.50: break
            hs = slsqp_topk(best_h, k, 300)
            cs = compute_c5(hs)

            # Refine
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    x = np.clip(v - lam, 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1 - 1e-10)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, 1)
    return x

def grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if n > k:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        m = -k
        if n > m:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    return g

def subgrad_method(h, n_iters, k_active, alpha0, decay):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for _ in range(n_iters):
        corr = np.correlate(h, 1-h, mode='full')
        subj = np.argpartition(corr, -k_active)[-k_active:]
        g = np.zeros(n)
        for p in subj:
            g += grad_for_lag(h, int(p) - (n-1))
        g *= dx / k_active
        h_new = project_box_sum(h - alpha * g, n/2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best = c5_new, h_new.copy()
        if c5_new <= compute_c5(h) + 1e-14:
            h = h_new
        else:
            alpha *= 0.7
        alpha *= decay
    return best, best_c5

def coord_descent(h, n_sweeps=3):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_sweeps):
        for i in range(n):
            bi = np.zeros(2*n-1)
            for k in range(-(n-1), n):
                bi[k + n - 1] = grad_for_lag(h, k)[i]
            corr_base = np.correlate(h, 1-h, mode='full') - bi * h[i]
            lo, hi = 0.0, 1.0
            for _ in range(25):
                m1 = lo + (hi-lo)/3
                m2 = hi - (hi-lo)/3
                if np.max(corr_base + bi*m1) < np.max(corr_base + bi*m2):
                    hi = m2
                else:
                    lo = m1
            h[i] = (lo+hi)/2
            h = project_box_sum(h, n/2.0)
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5, best = c5_new, h.copy()
    return best, best_c5

def lp_step(h, delta, top_k=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    cv = cp.Variable(n)
    t = cp.Variable()
    cons = [cv >= -h + 1e-10, cv <= 1-h-1e-10,
            cp.sum(cv) == 0, cv >= -delta, cv <= delta]
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    for p in top_idx:
        k = int(p) - (n-1)
        g = grad_for_lag(h, k) * dx
        cons.append(t >= corr[p]*dx + g @ cv)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.ECOS, max_time=5)
        if cv.value is not None and t.value < compute_c5(h) + delta*dx:
            return project_box_sum(h + np.array(cv.value), n/2.0)
    except:
        pass
    return h

def slsqp_topk(h, k, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1-x, mode='full')
        top = np.sort(corr)[::-1][:k]
        return float(np.sum(top)*dx/k)
    def obj_jac(x):
        corr = np.correlate(x, 1-x, mode='full')
        subj = np.argpartition(corr, -k)[-k:]
        g = np.zeros(n)
        for p in subj:
            g += grad_for_lag(x, int(p)-(n-1))
        return g * dx / k
    def sum_con(x):
        return float(np.sum(x) - n/2.0)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':sum_con}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter':maxiter, 'ftol':1e-14})
        return project_box_sum(res.x, n/2.0)
    except:
        return h

def softmax_slsqp(h, T, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1-x, mode='full')
        m = np.max(corr)
        return float((m + np.log(np.sum(np.exp(T*(corr-m))))/T)*dx)
    def sum_con(x):
        return float(np.sum(x) - n/2.0)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':sum_con}]
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter':maxiter, 'ftol':1e-13})
        return project_box_sum(res.x, n/2.0)
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = compute_c5(best_h)

                            hp[i] -= d

                else:
```
