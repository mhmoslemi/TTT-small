    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

class _Opt1D:
    cf = None
    coeff = None
    hi = 0.0
    n = 0
    dx = 0.0
    def val(self, v):
        nc = self.cf + (v - self.hi) * self.coeff
        nc[self.n - 1] = nc[self.n - 1] + v * (1.0 - v) - self.hi * (1.0 - self.hi)
        return float(nc.max()) * self.dx

_opt1d = _Opt1D()

def fast_coord_sweep(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    k_vals = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        best_h = project(best_h, n)
        corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                best_h = project(best_h, n)
                best_c5 = compute_c5(best_h)
                return best_h, best_c5
            hi = best_h[i]
            j1 = i + k_vals
            j2 = i - k_vals
            v1 = (j1 >= 0) & (j1 < n) & (k_vals != 0)
            v2 = (j2 >= 0) & (j2 < n) & (k_vals != 0)
            coeff = np.zeros(2 * n - 1)
            if v1.any():
                coeff[v1] += (1.0 - best_h[j1[v1]])
            if v2.any():
                coeff[v2] -= best_h[j2[v2]]
            _opt1d.cf = corr_full
            _opt1d.coeff = coeff
            _opt1d.hi = hi
            _opt1d.n = n
            _opt1d.dx = dx
            res = minimize_scalar(_opt1d.val, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 60})
            v_new = float(res.x)
            dv = v_new - hi
            if abs(dv) > 1e-9:
                best_h[i] = v_new
                corr_full += dv * coeff
                corr_full[n - 1] += v_new * (1.0 - v_new) - hi * (1.0 - hi)
                c_try = float(corr_full.max()) * dx
                if c_try < best_c5 - 1e-14:
                    best_c5 = c_try
                    improved = True
                else:
                    best_h[i] = hi
                    corr_full -= dv * coeff
                    corr_full[n - 1] -= v_new * (1.0 - v_new) - hi * (1.0 - hi)
        if not improved:
            break
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter=400):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish (10%)
    t_p1 = t0 + budget_s * 0.10
    for T in [30, 150, 600, 2000]:
        if time.time() > t_p1:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Heavy coordinate descent (55%)
    t_p2 = t0 + budget_s * 0.65
    best_h, best_c5 = fast_coord_sweep(best_h, n, t_p2, max_sweeps=3000)

    # Phase 3: SLSQP re-polish with high T
    if time.time() < t0 + budget_s * 0.72:
        for T in [1000, 5000, 15000]:
            if time.time() > t0 + budget_s * 0.72:
                break
            try:
                hc = slsqp_polish(best_h, T, n, maxiter=600)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass
            best_h, best_c5 = fast_coord_sweep(best_h, n, t0 + budget_s * 0.72, max_sweeps=300)

    # Phase 4: Basin hopping (18%)
    t_p4 = t0 + budget_s * 0.90
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(40):
        if time.time() > t_p4:
            break
        cool = 1.0 - 0.85 * (it / 40.0)
        step = 0.02 + 0.10 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = project(h_pert, n)
        try:
            hc = slsqp_polish(h_pert, 500, n, maxiter=250)
            c5v = compute_c5(hc)
            temp = 0.004 * cool + 0.0004
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
                work_h = hc
                work_c5 = c5v
        except Exception:
            continue
        if time.time() < t_p4:
            work_h, work_c5 = fast_coord_sweep(work_h, n, min(time.time() + 5.0, t_p4), max_sweeps=80)
        if work_c5 < best_c5:
            best_c5 = work_c5
            best_h = work_h.copy()

    # Phase 5: Multi-resolution (7%)
    t_p5 = t0 + budget_s * 0.97
    for n_try in [120, 144, 180, 200]:
        if time.time() > t_p5:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        h_interp, c5_try = fast_coord_sweep(h_interp, n_try, t_p5, max_sweeps=200)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_interp
            n = n_try

    # Phase 6: Final polish (3%)
    t_final = t0 + budget_s * 0.99
    best_h, best_c5 = fast_coord_sweep(best_h, n, t_final, max_sweeps=500)
    T_final = 30000
    rem = t_final - time.time()
    if rem > 3.0:
        try:
            hc = slsqp_polish(best_h, T_final, n, maxiter=int(rem * 30))
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                best_h, best_c5 = fast_coord_sweep(best_h, n, t0 + budget_s, max_sweeps=200)
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
