def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def snap_binary(h, target):
    n_ones = int(round(target))
    idx = np.argsort(h)[::-1]
    out = np.zeros(len(h))
    out[idx[:n_ones]] = 1.0
    return out

def grad_analytic(h, N, topK, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(corr.max())
    center = N - 1
    nc = len(corr)
    if tau > 0:
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        wsum = w.sum()
        if wsum > 1e-30:
            w /= wsum
        else:
            w = np.zeros(nc)
            w[int(np.argmax(corr))] = 1.0
        k = min(topK, nc)
        idx_top = np.argsort(corr)[-k:]
        w_top = np.zeros(nc)
        w_top[idx_top] = 1.0 / k
        w = 0.5 * w + 0.5 * w_top
    else:
        k = min(topK, nc)
        idx_top = np.argsort(corr)[-k:]
        w = np.zeros(nc)
        w[idx_top] = 1.0 / k
    grad = np.zeros(N)
    thresh = 1e-5
    for ai in range(nc):
        wi = w[ai]
        if wi < thresh:
            continue
        L = ai - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5_val

def phase_gradient(h, N, target, deadline, tau, topK, lr_init, mom):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr_init
    while time.time() < deadline:
        grad, c5_raw = grad_analytic(h, N, topK, tau)
        c5 = c5_raw * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        v = mom * v + grad
        h = h - lr * v
        h = project_sum(h, target)
    return best_h, best_c5

def phase_sa(h, N, target, deadline, seed):
    dx = 2.0 / N
    h = snap_binary(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    rng = np.random.RandomState(seed)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T0 = 0.008
    T1 = 0.00005
    est_total = 300000
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / float(est_total), 1.0)
        T = T0 * (T1 / T0) ** frac
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-9 and rng.rand() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def phase_refine(h, N, target, deadline):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    mom = 0.9
    lr = 0.003
    tau = 200.0
    tk = 2
    while time.time() < deadline:
        grad, c5_raw = grad_analytic(h, N, tk, tau)
        c5 = c5_raw * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        v = mom * v + grad
        h = h - lr * v
        h = project_sum(h, target)
        lr *= 0.997
        if lr < 1e-9:
            break
    return best_h, best_c5

def phase_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                return best_h, best_c5
            orig = best_h[i]
            cands = []
            if orig > 1e-9:
                cands.append(0.0)
            if orig < 1.0 - 1e-9:
                cands.append(1.0)
            if orig > 0.15:
                cands.append(orig - 0.08)
            if orig < 0.85:
                cands.append(orig + 0.08)
            if orig > 0.08:
                cands.append(orig - 0.03)
            if orig < 0.92:
                cands.append(orig + 0.03)
            for tv in cands:
                if tv < -1e-9 or tv > 1.0 + 1e-9:
                    continue
                if abs(tv - orig) < 1e-9:
                    continue
                trial = best_h.copy()
                trial[i] = tv
                trial = project_sum(trial, target)
                c5 = compute_c5(trial, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = trial
                    orig = tv
                    improved = True
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    h = np.array(initial_h_values, dtype=float)
    N = len(h)
    dx = 2.0 / N
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Phase 1a: Coarse gradient (early, wide smoothing)
    d1a = t0 + budget_s * 0.18
    h1, c1 = phase_gradient(h.copy(), N, target, d1a, tau=8.0, topK=20, lr_init=0.035, mom=0.82)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 1b: Mid gradient
    d1b = t0 + budget_s * 0.33
    h1b, c1b = phase_gradient(best_h.copy(), N, target, d1b, tau=40.0, topK=8, lr_init=0.015, mom=0.86)
    if c1b < best_c5:
        best_c5 = c1b
        best_h = h1b.copy()

    # Phase 1c: Fine gradient
    d1c = t0 + budget_s * 0.45
    h1c, c1c = phase_gradient(best_h.copy(), N, target, d1c, tau=150.0, topK=3, lr_init=0.006, mom=0.88)
    if c1c < best_c5:
        best_c5 = c1c
        best_h = h1c.copy()

    # Phase 2: Simulated annealing on binary (35% time)
    d2 = t0 + budget_s * 0.80
    best_h, best_c5 = phase_sa(best_h.copy(), N, target, d2, seed)

    # Phase 3: Post-SA gradient refinement (12% time)
    d3 = t0 + budget_s * 0.92
    best_h, best_c5 = phase_refine(best_h.copy(), N, target, d3)

    # Phase 4: Coordinate polish (remaining)
    best_h, best_c5 = phase_polish(best_h, N, target, deadline)

    # Final: check binary snap
    hb = snap_binary(best_h, target)
    cb = compute_c5(hb, dx)
    if cb < best_c5 - 1e-14:
        best_h = hb
        best_c5 = cb

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
