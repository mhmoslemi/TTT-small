```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    w_sum = np.sum(exp_t)
    val = (m + np.log(w_sum) / T) * dx
    w = exp_t / w_sum
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for l in range(1, n):
        wl = w[n - 1 + l] + w[n - 1 - l]
        if wl < 1e-16:
            continue
        a = np.zeros(n)
        b = np.zeros(n)
        a[l:] = 1.0 - h[:n - l]
        b[:n - l] = h[l:]
        grad += wl * (a - b)
    grad *= dx
    return val, grad

def smooth_obj_f(h, T):
    return smooth_obj_grad(h, T)[0]

def smooth_grad_f(h, T):
    return smooth_obj_grad(h, T)[1]

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=600):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj_f, h, args=(T,), jac=lambda x, T2: smooth_grad_f(x, T2),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def proj_subgrad(h, max_iters=3000, T_init=2000):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.15
    T = T_init
    for it in range(max_iters):
        val, grad = smooth_obj_grad(h_cur, T)
        step = lr
        h_try = h_cur
        c5_try, _ = compute_c5(h_cur)
        for _ in range(15):
            h_try = project_feasible(h_cur - step * grad, n)
            c5_try, _ = compute_c5(h_try)
            if c5_try < c5_try and c5_try < best_c5 - 1e-14:
                break
            if c5_try < val - 0.5 * step * np.dot(grad, h_try - h_cur) / max(step, 1e-15) - 1e-12:
                break
            step *= 0.5
        h_try = project_feasible(h_cur - step * grad, n)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_cur = h_try
            lr = min(lr * 1.01, 0.4)
        else:
            h_cur = h_try
            lr *= 0.92
        if it % 300 == 0:
            T = min(T * 1.8, 50000.0)
    return best_h, best_c5

def binary_local_search(h, max_rounds=80):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(-h)
    h_bin[order[:n // 2]] = 1.0
    best_c5, _ = compute_c5(h_bin)
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    for _ in range(max_rounds):
        best_c5_i = best_c5
        best_pair = None
        for j in zeros:
            h_bin[j] = 1.0
            for i in ones:
                if i == j:
                    continue
                h_bin[i] = 0.0
                c5_i, _ = compute_c5(h_bin)
                if c5_i < best_c5_i:
                    best_c5_i = c5_i
                    best_pair = (i, j)
                h_bin[i] = 1.0
            h_bin[j] = 0.0
        if best_pair is None:
            break
        i_a, j_a = best_pair
        h_bin[j_a] = 1.0
        h_bin[i_a] = 0.0
        best_c5 = best_c5_i
        ones = list(np.where(h_bin == 1.0)[0])
        zeros = list(np.where(h_bin == 0.0)[0])
    return h_bin, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.concatenate([np.ones(39), np.zeros(39)])
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP with analytic gradient, T sweep
    for T in [500, 3000, 15000, 80000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=700)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Projected subgradient with backtracking
    if time.time() - t0 < budget_s * 0.28:
        remain = budget_s * 0.18 - (time.time() - t0)
        nit = max(1500, int(remain * 2.5))
        h_sub, c5_sub = proj_subgrad(best_h, max_iters=nit, T_init=2000)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub
        # Second pass with higher T
        if time.time() - t0 < budget_s * 0.28:
            h_sub2, c5_sub2 = proj_subgrad(best_h, max_iters=2000, T_init=10000)
            if c5_sub2 < best_c5:
                best_c5 = c5_sub2
                best_h = h_sub2

    # Phase 3: SLSQP refine
    if time.time() - t0 < budget_s * 0.35:
        h_try = slsqp_smooth(best_h, 50000, maxiter=800)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: Binary local search + continuous refine
    if time.time() - t0 < budget_s * 0.50:
        h_bin, c5_bin = binary_local_search(best_h, max_rounds=60)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        h_r = slsqp_smooth(h_bin, 30000, maxiter=600)
        c5_r, _ = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 5: Multi-n exploration
    for n_try in [76, 80, 84, 90]:
        if time.time() - t0 > budget_s * 0.62:
            break
        if n_try == best_n:
            continue
        if n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_try = slsqp_smooth(h_interp, 10000, maxiter=500)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_try
            h_try2 = slsqp_smooth(h_try, 60000, maxiter=500)
            c5_try2, _ = compute_c5(h_try2)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2

    # Phase 6: Diverse perturbations + subgradient + SLSQP
    if time.time() - t0 < budget_s * 0.80:
        for trial in range(10):
            if time.time() - t0 > budget_s * 0.80:
                break
            scale = 0.02 + 0.05 * np.random.rand()
            h_pert = best_h + np.random.randn(best_n) * scale
            h_pert = project_feasible(h_pert, best_n)
            h_opt, _ = proj_subgrad(h_pert, max_iters=1200, T_init=3000)
            h_opt = slsqp_smooth(h_opt, 40000, maxiter=500)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 7: Rotation variants
    if time.time() - t0 < budget_s * 0.88:
        for shift in [1, 2, 3, 5, 8]:
            if time.time() - t0 > budget_s * 0.88:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_opt, _ = proj_subgrad(h_rot, max_iters=1500, T_init=5000)
            h_opt = slsqp_smooth(h_opt, 60000, maxiter=500)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Final ultra-high T polish
    if time.time() - t0 < budget_s * 0.97:
        for T in [100000, 200000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            h_f = slsqp_smooth(best_h, T, maxiter=1000)
            c5_f, _ = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
