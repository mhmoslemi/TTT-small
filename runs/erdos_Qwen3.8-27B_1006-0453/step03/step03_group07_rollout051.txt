import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff * (1.0 - h[mask]) / ((1.0 - h[mask])[mask].sum() if diff > 0 else h[mask].sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        r = 1.0 - h
    else:
        r = h.copy()
    tot = r.sum()
    if tot > 1e-14:
        h += r * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    ev = np.exp(T * (corr - m))
    sw = ev.sum()
    w = ev / sw
    obj = (m + np.log(sw) / T) * dx
    grad = np.zeros(n)
    for k in range(0, n):
        idx = n - 1 + k
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        if k == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
    for k in range(1, n):
        idx = n - 1 - k
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        grad[k:] += wk * (1.0 - h[:n-k])
        grad[:n-k] -= wk * h[k:]
    return obj, grad

def fista_optimize(h0, T, n, iters, alpha0):

    h = y.copy()

        h_new = proj_feasible(h_new, n)

    best_val = compute_c5(h)
    deltas = np.array([0.03, -0.03, 0.06, -0.06, 0.01, -0.01, 0.015, -0.015])

            d = rng.choice([0.02, -0.02, 0.04, -0.04, 0.005, -0.005])
            h_try = h.copy()
            h_try[j1] += d
            h_try[j2] -= d
            h_try = make_feasible(h_try, n)
            v = compute_c5(h_try)
            if v < best_val - 1e-15:
                best_val = v
                h = h_try
                improved = True
        if not improved:
            break
    return h, best_val

def slsqq_refine(h0, T, n, maxiter):
    def objfun(x):
        return softmax_obj_and_grad(x, T)[0]
    def summ(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': summ}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(objfun, h0.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[0::2] = 1.0
    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: FISTA at parent resolution, multiple T
    for T in [50, 150, 500, 2000]:
        if time.time() - t0 > budget_s * 0.18:
            break
        h_try = fista_optimize(best_h, T, best_n, 800, 1.0 / max(T, 50))
        h_try = make_feasible(h_try, best_n)
        c5v = compute_c5(h_try)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_try

    # Phase 2: Multi-resolution exploration
    for n_try in [100, 150, 300, 400]:
        if time.time() - t0 > budget_s * 0.42:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [100, 500]:
            if time.time() - t0 > budget_s * 0.42:
                break
            h_try = fista_optimize(h_interp, T, n_try, 500, 1.0 / max(T, 50))
            h_try = make_feasible(h_try, n_try)
            c5v = compute_c5(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try
                best_n = n_try

    # Phase 3: Coordinate descent for local refinement
    if time.time() - t0 < budget_s * 0.50:
        h_cd, c5_cd = coord_descent(best_h, best_n, max_rounds=30)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 4: Basin hopping with FISTA + pair descent
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = best_n
    while time.time() - t0 < budget_s * 0.78:
        step_sz = 0.10 + 0.05 * np.random.rand()
        h_pert = bh_h + step_sz * np.random.randn(n_bh)
        h_pert = make_feasible(h_pert, n_bh)
        T_use = 300
        h_try = fista_optimize(h_pert, T_use, n_bh, 300, 1.0 / T_use)
        h_try = make_feasible(h_try, n_bh)
        h_try, c5_try = pair_descent(h_try, n_bh, 30, rng)
        if c5_try < bh_c5:
            bh_c5 = c5_try
            bh_h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 5: SLSQP polish at high T
    for T in [3000, 10000, 30000]:
        if time.time() - t0 > budget_s * 0.96:
            break
        h_new = slsqq_refine(best_h, T, best_n, 2000)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 6: Final coordinate descent
    if time.time() - t0 < budget_s * 0.99:
        h_cd, c5_cd = coord_descent(best_h, best_n, max_rounds=50)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
