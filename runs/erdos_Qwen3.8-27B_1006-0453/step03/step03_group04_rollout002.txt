    h = np.clip(h, 0.0, 1.0)

def grad_soft_max(h, N, threshold_frac=0.95):
    """Gradient using all peaks above a threshold fraction of the max."""
    corr = compute_corr(h)
    peak = float(np.max(corr))
    threshold = peak * threshold_frac
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    k = len(active)
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k

        L = ai - center

            M = -L
            if M < N:

                grad[M:] -= wk * h[:N - M]

                            h_b[start:start+blen] = 0
                            h_b[zstart:zstart+blen] = 1
                            new_pk = float(np.max(compute_corr(h_b)))
                            delta = new_pk - cur_pk
                            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                                cur_pk = new_pk
                                if new_pk < best_pk:
                                    best_pk = new_pk
                                    best = h_b.copy()
                            else:
                                h_b[start:start+blen] = 1
                                h_b[zstart:zstart+blen] = 0
                            it += 1
                            continue
        # Single swap
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        new_pk = float(np.max(compute_corr(h_b)))
        delta = new_pk - cur_pk
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
        it += 1
    return best, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    target = N / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam with soft-max gradient (12%)
    d1 = t0 + budget_s * 0.12
    h1, c1 = adam_run(h0.copy(), N, target, d1, lr0=0.08, frac=0.90)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Binary anneal with block moves (35%)
    d2 = t0 + budget_s * 0.47
    h2, c2 = binary_anneal(h1, N, d2, rng)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Continuous relaxation + Adam (18%)
    d3 = t0 + budget_s * 0.65
    h3 = best_h.astype(float) + rng.randn(N) * 0.04
    h3 = project_sum(h3, target)
    h3, c3 = adam_run(h3, N, target, d3, lr0=0.04, frac=0.94)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Second binary pass (17%)
    d4 = t0 + budget_s * 0.82
    h4, c4 = binary_anneal(best_h, N, d4, rng)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Multi-restart continuous refinement (10%)
    d5 = deadline - 3
    best_in_phase = best_h.copy()
    best_c5_phase = best_c5
    n_restarts = 3
    seg_time = (d5 - time.time()) / n_restarts
    for ri in range(n_restarts):
        if time.time() > d5:
            break
        seg_end = time.time() + seg_time
        h5 = best_in_phase + rng.randn(N) * 0.02 * (ri + 1)
        h5 = project_sum(h5, target)
        h5, c5 = adam_run(h5, N, target, seg_end, lr0=0.03 / (ri + 1), frac=0.95)
        if c5 < best_c5_phase:
            best_c5_phase, best_in_phase = c5, h5

    if best_c5_phase < best_c5:
        best_c5, best_h = best_c5_phase, best_in_phase

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, N
