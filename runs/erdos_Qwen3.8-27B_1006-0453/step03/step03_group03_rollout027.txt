def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def smooth_obj(h, T, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    idx = np.argsort(corr)[::-1][:top_m]
    c_sel = corr[idx]
    shifted = T * (c_sel - m)
    ms = np.max(shifted)
    es = np.exp(shifted - ms)
    val = (m + (ms + np.log(es.sum())) / T) * dx
    return val

def smooth_grad(h, T, top_m):

    es = np.exp(shifted - ms)
    weights = es / es.sum()
    grad = np.zeros(n)

        w = weights[ii]
        k = idx[ii] - (n - 1)

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
