        if bs >= N:
            bs = 1
        a = rng.randint(0, N - bs + 1)
        found = False
        for _ in range(40):
            b = rng.randint(0, N - bs + 1)
            if a != b and hb[a] != hb[b]:
                found = True
                break
        if not found:
            continue
        sa = hb[a:a + bs].copy()
        sb = hb[b:b + bs].copy()
        hb[a:a + bs], hb[b:b + bs] = sb, sa
        npk = float(np.max(np.correlate(hb, 1.0 - hb, mode='full')))
        d = npk - cur_pk
        if d < 0 or (T > 0.002 and rng.random() < np.exp(-d / T)):
            cur_pk = npk
            if npk < best_pk:
                best_pk = npk
                best = hb.copy()
        else:
            hb[a:a + bs], hb[b:b + bs] = sa, sb
    return best, best_pk * dx

def deterministic_improve(hb, N, max_rounds=50):
    dx = 2.0 / N
    best_pk = float(np.max(np.correlate(hb, 1.0 - hb, mode='full')))
    for _ in range(max_rounds):
        improved = False
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        best_swap = None
        best_val = best_pk
        for i in range(len(ones)):
            for j in range(len(zeros)):
                if i % 50 == 0 and j % 50 == 0:
                    hb[ones[i]], hb[zeros[j]] = hb[zeros[j]], hb[ones[i]]
                    pk = float(np.max(np.correlate(hb, 1.0 - hb, mode='full')))
                    if pk < best_val:
                        best_val = pk
                        best_swap = (i, j)
                    hb[ones[i]], hb[zeros[j]] = hb[zeros[j]], hb[ones[i]]
        if best_swap is not None:
            i, j = best_swap
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
            hb[ones[i]], hb[zeros[j]] = hb[zeros[j]], hb[ones[i]]
            best_pk = best_val
            improved = True
        if not improved:
            break
    return hb, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    sizes = [N for N in [400, 480, 560, 640] if N % 2 == 0]
    nsz = len(sizes)
    total_avail = dl - t0 - 15
    per_size = total_avail / nsz

    for si, N in enumerate(sizes):
        phase_end = t0 + 15 + (si + 1) * per_size
        if phase_end > dl - 15:
            phase_end = dl - 15
        if time.time() > phase_end - 50:
            break
        tgt = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        hi = np.interp(xN, x0, best_h)
        hi = project_sum(hi, tgt)
        ts = time.time()
        ta = phase_end - ts
        if ta < 60:
            break

        d1 = ts + ta * 0.14
        h1, c1 = adam_opt(hi.copy(), N, tgt, d1, 20, 0.04, 0.97, 120)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

        d2 = ts + ta * 0.32
        h2, c2 = adam_opt(h1.copy(), N, tgt, d2, 5, 0.012, 0.99, 100)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

        d3 = ts + ta * 0.60
        h3, c3 = sa_binary(h2, N, d3, rng, blocks=[1, 1, 1, 1, 2, 2])
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

        d4 = ts + ta * 0.82
        h4, c4 = sa_binary(h3, N, d4, rng, blocks=[1, 1, 2, 2, 3, 3, 4, 5])
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

        if time.time() < phase_end - 2:
            h5 = h4.copy()
            nf = max(4, N // 60)
            on = np.where(h5 == 1)[0]
            zo = np.where(h5 == 0)[0]
            for _ in range(nf):
                ii = rng.randint(len(on))
                jj = rng.randint(len(zo))
                h5[on[ii]], h5[zo[jj]] = h5[zo[jj]], h5[on[ii]]
                on = np.where(h5 == 1)[0]
                zo = np.where(h5 == 0)[0]
            d5 = phase_end
            h5, c5 = sa_binary(h5, N, d5, rng, blocks=[1, 1, 1, 2, 2, 3])
            if c5 < best_c5:
                best_c5, best_h = c5, h5.copy()

    Nb = len(best_h)
    if time.time() < dl - 2:
        hF = best_h + rng.randn(Nb) * 0.01
        hF = project_sum(hF, Nb / 2.0)
        hF, cF = adam_opt(hF, Nb, Nb / 2.0, dl - 1, 5, 0.008, 0.995, 80)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, min(best_c5, final_c5), len(best_h)
