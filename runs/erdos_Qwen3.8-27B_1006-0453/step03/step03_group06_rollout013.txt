from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def topk_grad_c5(h, T, k):
    """Compute soft-max over top-k shifts, with analytical gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx]
    m = np.max(c)
    exp_c = np.exp(T * (c - m))
    w = exp_c / exp_c.sum()
    val = (m + np.log(exp_c.sum()) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    for ki in range(k):
        kcorr = idx[ki]
        wi = w[ki]
        if wi < 1e-15:
            continue
        shift = kcorr - (n - 1)
        # df/dh[j] = (1-h[j+shift])*I[0<=j+shift<n] - h[j-shift]*I[0<=j-shift<n]
        # Part 1: (1-h[j+shift]) for j in [max(0,-shift), min(n, n-shift))
        j1_lo = max(0, -shift)
        j1_hi = min(n, n - shift)
        if j1_hi > j1_lo:
            grad[j1_lo:j1_hi] += wi * (1.0 - h[j1_lo + shift:j1_hi + shift])
        # Part 2: -h[j-shift] for j in [max(0,shift), min(n, n+shift))
        j2_lo = max(0, shift)
        j2_hi = min(n, n + shift)
        if j2_hi > j2_lo:
            grad[j2_lo:j2_hi] += -wi * h[j2_lo - shift:j2_hi - shift]
    
    grad *= dx
    return val, grad

def slsqp_with_grad(h0, T, k, maxiter, ftol):
    n = len(h0)
    def obj(x):
        v, _ = topk_grad_c5(x, T, k)
        return v
    def grad(x):
        _, g = topk_grad_c5(x, T, k)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def momentum_gd(h0, T, k, lr, n_iter, mu=0.9):
    n = len(h0)
    h = h0.copy()
    vel = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        val, grad = topk_grad_c5(h, T, k)
        vel = mu * vel - lr * grad
        h_new = project_feasible(h + vel, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
        h = h_new
        # Adaptive learning rate
        if it % 50 == 0 and it > 0:
            lr *= 0.85
    return best, best_c5

                    best_c5, best_h = c5_c, h_c
                    break

    # Phase 6: Final aggressive polish
    n = len(best_h)
    for T, k in [(3000, 8), (8000, 6), (20000, 5)]:
        if elapsed() > budget_s * 0.97:
            break
        h_c = slsqp_opt(best_h, T, k, 1200, 1e-16)
        c5_c = compute_c5(h_c)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c

    # Phase 7: Coordinate descent on worst shifts
    if elapsed() < budget_s * 0.99:
        for _ in range(30):
            if elapsed() > budget_s * 0.99:
                break
            improved = False
            corr = np.correlate(best_h, 1.0 - best_h, 'full')
            top5 = np.argsort(corr)[-5:]
            for kc in top5:
                s = kc - (n - 1)
                for j in range(n):
                    for d in (0.03, -0.03):
                        h_try = best_h.copy()
                        h_try[j] = np.clip(h_try[j] + d, 0.0, 1.0)
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-12:
                            best_c5, best_h = c5_try, h_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
