from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    """Project to {0<=h<=1, sum(h)=n/2} using bisection on shift."""
    target = n / 2.0
    vec = np.array(h, dtype=float)
    def sum_at(s):
        return np.clip(vec + s, 0.0, 1.0).sum()
    # Check if already feasible
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vec) - 1.0
    hi = 1.0 - np.min(vec) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if sum_at(mid) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(res.x, n)

def coord_opt_direct(h, n, i):

        h_try = h.copy()
        h_try[i] = hi
        corr = np.correlate(h_try, 1.0 - h_try, mode='full')
        return np.max(corr) * (2.0 / n)

                          options={'xatol': 1e-12})

    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):

            break
        improved = False

        for i in order:

            hi_new = coord_opt_direct(best_h, n, i)

            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

        t2 = np.zeros(n)
        if k >= 0:
            if k < n:
                t1[:n - k] = 1.0 - h[k:]
            if k > 0:
                t2[k:] = -h[:n - k]
        else:
            kk = -k
            t1[kk:] = 1.0 - h[:n - kk]
            t2[:n - kk] = -h[kk:]
        grad += l * (t1 + t2)
    grad *= dx
    h_new = h - alpha * grad
    return project(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP T-schedule (first 12% of budget)
    for T in [10, 50, 200, 1000, 4000, 12000]:
        if elapsed() > budget_s * 0.12:
            break
        try:
            hc = slsqp_opt(best_h.copy(), T, n, maxiter=1000, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Intensive coordinate descent (35% of budget)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.47, max_sweeps=3000)

    # Phase 3: Subgradient + SLSQP diversification (25%)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 40
    for it in range(n_iter):
        if elapsed() > budget_s * 0.72:
            break
        cool = 1.0 - it / float(n_iter)
        # Mix: perturb or subgradient
        if it % 3 == 0:
            step_sz = 0.02 + 0.10 * cool
            h_pert = work_h + step_sz * np.random.randn(n)
        else:
            alpha = 0.01 * cool + 0.002
            T_sg = 500
            h_pert = subgradient_step(work_h, n, T_sg, alpha)
        h_pert = project(h_pert, n)
        # Quick SLSQP
        try:
            T_s = 300 + 2000 * cool
            hc = slsqp_opt(h_pert, T_s, n, maxiter=250, ftol=1e-13)
        except Exception:
            hc = h_pert
        c5v = compute_c5(hc)
        # Short CD polish
        hc, c5v = coord_descent(hc, n, min(t0 + budget_s * 0.72, time.time() + budget_s * 0.015), max_sweeps=60)
        temp = 0.003 * cool + 0.0003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Final intensive CD (15%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.88, max_sweeps=2000)

    # Phase 5: Final SLSQP polish + CD
    for T in [5000, 20000, 50000]:
        if elapsed() > budget_s * 0.95:
            break
        try:
            hc = slsqp_opt(best_h.copy(), T, n, maxiter=1500, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=200)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
