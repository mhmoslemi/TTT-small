    h2[h2 < 0.3] = 0.0
    h2[h2 > 0.7] = 1.0
    h2 = project_sum(h2, target)
    c, _ = compute_c5(h2, dx)
    if c < compute_c5(h, dx)[0] - 1e-6:
        h2, c2 = grad_descent_phase(h2, N, target, dx, deadline,
                                     50, 300, 0.02, 0.003, 20, 5, 0.9)
        return h2, c2
    return h, compute_c5(h, dx)[0]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = 0.5 * np.ones(150)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)[0]
    
    sizes = [100, 120, 150, 200, 250]
    
    for N in sizes:
        if time.time() > deadline - 30:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, best_h)
        h_init = project_sum(h_init, target)
        remain = deadline - time.time()
        
        # Phase 1: Broad exploration
        p1_dl = time.time() + remain * 0.25
        h1, c1 = grad_descent_phase(h_init.copy(), N, target, dx, p1_dl,
                                     tau0=15.0, tau1=80.0, lr0=0.04, lr1=0.01,
                                     K0=50, K1=15, mom=0.85)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
        
        if time.time() > deadline - 30:
            break
        
        # Phase 2: Focused refinement
        remain2 = deadline - time.time()
        p2_dl = time.time() + remain2 * 0.30
        h2, c2 = grad_descent_phase(h1.copy(), N, target, dx, p2_dl,
                                     tau0=80.0, tau1=400.0, lr0=0.01, lr1=0.002,
                                     K0=15, K1=5, mom=0.9)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
        
        if time.time() > deadline - 20:
            break
        
        # Phase 3: Ultra-focused polish
        remain3 = deadline - time.time()
        p3_dl = time.time() + remain3 * 0.20
        h3, c3 = grad_descent_phase(h2.copy(), N, target, dx, p3_dl,
                                     tau0=400.0, tau1=2000.0, lr0=0.004, lr1=0.0004,
                                     K0=8, K1=3, mom=0.92)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
        
        if time.time() > deadline - 15:
            break
        
        # Coordinate descent polish
        rem4 = deadline - time.time()
        cd_dl = time.time() + min(rem4 * 0.3, 60.0)
        h4, c4 = coord_descent(h3.copy(), N, target, dx, cd_dl)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()
        
        # Random restart at this size
        if time.time() < deadline - 40:
            h_r = 0.5 + 0.4 * np.sin(2.0 * np.pi * np.arange(N) * 3.0 / N + rng.uniform(0, 6.28))
            h_r = np.clip(h_r, 0.0, 1.0)
            h_r = project_sum(h_r, target)
            rem5 = deadline - time.time()
            hr_dl = time.time() + rem5 * 0.5
            hr, cr = grad_descent_phase(h_r, N, target, dx, hr_dl,
                                         tau0=20.0, tau1=500.0, lr0=0.03, lr1=0.003,
                                         K0=40, K1=6, mom=0.88)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()
    
    # Final ultra-polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 4:
        dx_b = 2.0 / Nb
        rem_f = deadline - time.time()
        hf, cf = grad_descent_phase(best_h.copy(), Nb, Nb // 2, dx_b,
                                     time.time() + rem_f * 0.7,
                                     tau0=800.0, tau1=5000.0, lr0=0.002, lr1=0.0001,
                                     K0=5, K1=2, mom=0.93)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf
    
    final_c5, _ = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
