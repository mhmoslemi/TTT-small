def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-15:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-15:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_obj_and_grad(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)

    w = exps / s

    # d(corr[k])/dh[j] = (1-h)[j+k-(n-1)] - h[j-k+(n-1)] (with validity checks)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-15:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-15:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    s = np.sum(exps)
    w = exps / s
    val = (m + np.log(s) / T) * dx
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        js = max(0, -off)
        je = min(n, n - off)
        if js < je:
            grad[js:je] += wk * one_minus_h[js + off:je + off]
        js2 = max(0, off)
        je2 = min(n, n + off)
        if js2 < je2:
            grad[js2:je2] -= wk * h[js2 - off:je2 - off]
    return val, grad

def sum_con_fun(x):
    return x.sum() - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x)) / len(x)

def slsqp_refine(h0, T, maxiter=500):
    n = len(h0)
    def obj(x):
        return smooth_obj(x, T)
    def obj_grad(x):
        return smooth_obj_and_grad(x, T)[1]
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return make_feasible(np.clip(res.x, 0, 1), n)

def pgd_step(h, T, n_iters, lr_init=0.08):
    h = h.copy()
    n = len(h)
    lr = lr_init
    prev_val = smooth_obj(h, T)
    for it in range(n_iters):
        val, grad = smooth_obj_and_grad(h, T)
        gc = grad - grad.mean()
        h_new = h - lr * gc
        h_new = make_feasible(h_new, n)
        val_new = smooth_obj(h_new, T)
        if val_new > val + 1e-12:
            for _ in range(8):
                lr *= 0.5
                h_new = h - lr * gc
                h_new = make_feasible(h_new, n)
                val_new = smooth_obj(h_new, T)
                if val_new <= val + 1e-12:
                    break
        else:
            lr = min(lr * 1.05, 0.5)
        h = h_new
        if abs(gc).max() < 1e-11:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        n_init = 200
        best_h = np.zeros(n_init)
        best_h[0:n_init // 2] = 1.0
        best_h = make_feasible(best_h, n_init)

    best_c5 = compute_c5(best_h)

    # Phase 1: PGD with increasing T
    for T in [80, 300, 1500, 8000, 50000]:
        if time.time() - t0 > budget_s * 0.22:
            break
        h_new = pgd_step(best_h.copy(), T, 300, 0.06)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 2: SLSQP with analytical gradient
    for T in [1000, 5000, 30000]:
        if time.time() - t0 > budget_s * 0.38:
            break
        try:
            h_new = slsqp_refine(best_h.copy(), T, 800)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    # Phase 3: Multi-resolution exploration
    for n_try in [100, 150, 200, 240, 300, 400]:
        if time.time() - t0 > budget_s * 0.58:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [300, 5000]:
            if time.time() - t0 > budget_s * 0.58:
                break
            h_new = pgd_step(h_interp.copy(), T, 150, 0.06)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 4: Basin hopping
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    elapsed = time.time() - t0
    max_iter = max(25, int((budget_s * 0.82 - elapsed) / 3.5))
    for it in range(max_iter):
        if time.time() - t0 > budget_s * 0.82:
            break
        step = 0.12 * (1.0 - 0.6 * it / max(1, max_iter)) + 0.01
        ptype = it % 3
        if ptype == 0:
            h_pert = bh_h + step * np.random.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(2, n_bh // 10)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1 + blen].copy()
            h_pert[s1:s1 + blen] = h_pert[s2:s2 + blen]
            h_pert[s2:s2 + blen] = tmp
        else:
            h_pert = bh_h.copy()
            nflip = max(1, n_bh // 6)
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        h_pert = make_feasible(h_pert, n_bh)
        T_use = 500 + 2000 * (it // 12)
        h_new = pgd_step(h_pert, T_use, 60, 0.05)
        c5v = compute_c5(h_new)
        temp = 0.004 * (1.0 - it / max(1, max_iter)) + 0.0002
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 5: Bang-bang snap + re-optimize
    if time.time() - t0 < budget_s * 0.92:
        for thresh_lo, thresh_hi in [(0.1, 0.9), (0.2, 0.8), (0.15, 0.85)]:
            if time.time() - t0 > budget_s * 0.95:
                break
            h_snap = best_h.copy()
            h_snap[h_snap < thresh_lo] = 0.0
            h_snap[h_snap > thresh_hi] = 1.0
            h_snap = make_feasible(h_snap, len(h_snap))
            h_new = pgd_step(h_snap, 5000, 200, 0.05)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 6: Final high-T SLSQP polish
    if time.time() - t0 < budget_s * 0.99:
        for T in [50000, 500000]:
            if time.time() - t0 > budget_s * 0.99:
                break
            try:
                h_new = slsqp_refine(best_h.copy(), T, 2000)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass

    # Final PGD polish
    if time.time() - t0 < budget_s * 0.995:
        h_new = pgd_step(best_h.copy(), 200000, 500, 0.02)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
