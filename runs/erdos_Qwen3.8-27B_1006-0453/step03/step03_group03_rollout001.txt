def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_bisect(h, n):
    h = h.astype(float).copy()
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) * 0.5
    h_p = np.clip(h + alpha, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        free = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if free.sum() > 0:
            h_p[free] += diff / free.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    p = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = p[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            m2 = -k
            grad[m2:] += w * (1.0 - h[:n - m2])
            grad[:n - m2] -= w * h[m2:]
    return grad * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    v = np.zeros(n)

        T = T_lo * (T_hi / T_lo) ** frac

            h_c = h_new

            stall = 0
        else:
            stall += 1
            lr *= 0.93
            if stall > 150:
                break
    return best_h, best_c5
