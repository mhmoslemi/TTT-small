    # For each lag k = -(n-1)..(n-1), compute change

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]

        h[a] = 0
        h[b] = 1
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr) * (2.0/n)

        else:

            else:
                accept = True

            curr_corr = new_corr
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            h[a] = 1
            h[b] = 0
        
        n_steps += 1
        # Cool
        T *= 0.9998
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
