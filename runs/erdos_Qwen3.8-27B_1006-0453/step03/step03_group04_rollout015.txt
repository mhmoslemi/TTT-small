    s = float(h.sum())

        return h

        else:

def adam_refine(h, N, target, deadline, lr0, k=5):
    """Adam optimization with top-k subgradient."""
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_peak = float(np.max(corr))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    center = N - 1
    it = 0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        # Top-k subgradient
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L > 0:
                grad[:N-L] += wk * h[L:]  # d(corr[L])/d(h[i]) for h[i] in first part
                grad[L:] -= wk * h[:N-L]  # for h[i] in second part
            elif L < 0:

                grad[:N-M] -= wk * (1.0 - h[M:])

        h = project_sum(h, target)

    return best_h, best_peak * dx

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-8:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N, rng):
    h_b = (h + 0.5 * rng.rand(N) > 1.0).astype(float)  # slight randomness in threshold
    # Actually let's just threshold at 0.5
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    target = N // 2
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        perm = rng.permutation(len(idx))[:n_ones - target]
        h_b[idx[perm]] = 0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        perm = rng.permutation(len(idx))[:target - n_ones]
        h_b[idx[perm]] = 1
    return h_b

def binary_search(h, N, deadline, rng):
    """Intensive binary local search with random swaps and 2-swap kicks."""
    dx = 2.0 / N
    h_b = h.copy()
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')

    zeros = np.where(h_b == 0)[0].copy()
    
    remaining = deadline - time.time()
    n_iters = max(int(remaining * 50), 2000)
    
    T = 2.0
    T_min = 0.0005
    cool_rate = (T_min / T) ** (1.0 / n_iters)

    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        if it > 0 and it % 200 == 0:
            T = max(T * cool_rate ** 200, T_min)
        
        # Choose move
        if stuck > 500 and rng.random() < 0.3:
            # 2-swap kick: swap two pairs
            i1 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            i2 = ones[rng.randint(len(ones))]
            j2 = zeros[rng.randint(len(zeros))]

                continue
            h_b[i1], h_b[j1] = 0.0, 1.0
            h_b[i2], h_b[j2] = 0.0, 1.0

                cur_pk = new_pk
                ones = np.where(h_b == 1)[0]
                zeros = np.where(h_b == 0)[0]
                if new_pk < best_pk:
                    best_pk, best = new_pk, h_b.copy()
                stuck = 0
            else:
                h_b[i1], h_b[j1] = 1.0, 0.0
                h_b[i2], h_b[j2] = 1.0, 0.0
        else:

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

            new_corr = np.correlate(h_b, 1.0 - h_b, mode='full')
            new_pk = float(np.max(new_corr))
            delta = new_pk - cur_pk
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                cur_pk = new_pk
                # Update ones and zeros
                idx_i = np.where(ones == i)[0][0]
                idx_j = np.where(zeros == j)[0][0]
                ones[idx_i] = j
                zeros[idx_j] = i
                if new_pk < best_pk:
                    best_pk, best = new_pk, h_b.copy()
                    stuck = 0
                else:
                    stuck += 1
            else:
                h_b[i], h_b[j] = 1.0, 0.0
    
    return best, best_pk * dx

def adam_refine(h, N, target, deadline, lr0=2.0, k=5):
    """Adam with top-k subgradient for continuous refinement."""
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    center = N - 1
    it = 0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak, best_h = peak, h.copy()
        
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L > 0:
                grad[L:] += wk * (1.0 - h[:N-L])
                grad[:N-L] -= wk * h[L:]
            elif L < 0:
                M = -L
                grad[:N-M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N-M]
        
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        h = h - lr * (m / (1 - b1**it)) / (np.sqrt(v / (1 - b2**it)) + eps)
        h = project_sum(h, target)
        if it % 300 == 0:
            lr *= 0.97
    
    return best_h, best_peak * dx
