    """Solve: min t s.t. corr_k(h0) + g_k·(h-h0) <= t, 0<=h<=1, sum(h)=n/2"""
    corr0 = compute_corr(h0)
    m = 2*n - 1
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(h) == n/2.0,
        h >= 0,
        h <= 1
    ]
    
    for k_idx in range(m):
        k = k_idx - (n - 1)
        # gradient of corr_k w.r.t. h
        grad = np.zeros(n)

            if 0 <= j-k < n:

    prob = cp.Problem(cp.Minimize(t), constraints)

    from scipy.optimize import linprog
    corr0 = compute_corr(h0)
    m = 2*n - 1
    
    if top_k is not None:
        # Only use top k correlations
        top_idx = np.argsort(corr0)[::-1][:top_k]
    else:
        top_idx = np.arange(m)
    
    A_rows = []
    b_vals = []
    for k_idx in top_idx:

        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j+k < n:
                grad[j] += (1.0 - h0[j+k])
            if 0 <= j-k < n:
                grad[j] -= h0[j-k]

        row = np.append(grad, -1.0)
        A_rows.append(row)
        b_vals.append(-corr0[k_idx] + grad @ h0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [n/2.0]
    
    # Objective: minimize t
    c = np.zeros(n+1)
    c[n] = 1.0
    
    bounds = [(0, 1)]*n + [(None, None)]  # h in [0,1], t free

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_qp_step(h0, n, k_constraint=None):
    """LP-based QP step: linearize correlations, minimize max."""
    corr0 = compute_corr(h0)
    m = 2 * n - 1
    if k_constraint is not None:
        idx = np.argsort(corr0)[::-1][:k_constraint]
    else:
        idx = np.arange(m)

    b_vals = []

        for j in range(n):

            if 0 <= j - k < n:
                grad[j] -= h0[j - k]

        b_vals.append(grad @ h0 - corr0[k_idx])

    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, 1)] * n + [(0, None)]

                  bounds=bounds, method='highs')
    if res.success:
        return np.clip(res.x[:n], 0.0, 1.0)
    return h0.copy()

    """Multiple LP-QP iterations with line search."""
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        h_new = lp_qp_step(h, n, k_constraint)
        # Line search
        c5_new = compute_c5(h_new)
        if c5_new > best_c5:
            # Try half step
            h_half = 0.5 * h + 0.5 * h_new
            h_half = project_feasible(h_half, n)
            c5_half = compute_c5(h_half)
            if c5_half < best_c5:
                h = h_half
                best_c5 = c5_half
                best = h.copy()
            # Try quarter
            h_quarter = 0.75 * h + 0.25 * h_new
            h_quarter = project_feasible(h_quarter, n)
            c5_q = compute_c5(h_quarter)
            if c5_q < best_c5:
                h = h_quarter
                best_c5 = c5_q
                best = h.copy()
            # Reset h to best
            h = best.copy()
        else:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        if best_c5 < tol:
            break
    return best, best_c5

def topk_obj(h, k):
    n = len(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def softmax_obj(h, T):
    n = len(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def lp_qp_step(h0, n, k_constraint=None):
    A_rows, b_vals = build_lp_matrices(h0, n, k_constraint)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_rows, b_ub=b_vals, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:n], 0.0, 1.0)
    except Exception:
        pass
    return h0.copy()

def lp_qp_iterate(h, n, n_iter, k_constraint, tol=1e-13):
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        h_lp = lp_qp_step(h, n, k_constraint)
        c5_lp = compute_c5(h_lp)
        if c5_lp >= best_c5:
            h_mix = 0.5 * h + 0.5 * h_lp
            h_mix = project_feasible(h_mix, n)
            c5_mix = compute_c5(h_mix)
            if c5_mix >= best_c5:
                h_mix = 0.75 * h + 0.25 * h_lp
                h_mix = project_feasible(h_mix, n)
                c5_mix = compute_c5(h_mix)
                if c5_mix < best_c5:
                    best = h_mix.copy()
                    best_c5 = c5_mix
                    h = h_mix.copy()
                return best, best_c5
            best = h_mix.copy()
            best_c5 = c5_mix
            h = h_mix.copy()
        else:
            best = h_lp.copy()
            best_c5 = c5_lp
            h = h_lp.copy()
        if best_c5 < tol:
            break
    return best, best_c5

def topk_avg_obj(h, k):
    n = len(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(h, T):
    n = len(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_optimize(h, obj_func, args, maxiter, ftol, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_step(h, n, top_k, alpha):
    corr = compute_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    wts = corr[idx].astype(float)
    ws = wts.sum()
    if ws > 0:
        wts = wts / ws
    else:
        wts = np.ones(top_k) / top_k
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        k = int(ki) - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:n])
            grad[k:] -= w * h[:n - k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n - kk])
            grad[:n - kk] -= w * h[kk:]
    return project_feasible(h - alpha * grad, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: LP-QP with all constraints (primary driver)
    deadline = t0 + budget_s * 0.35
    stall = 0
    while time.time() < deadline:
        h_new, c5_new = lp_qp_iterate(best_h, best_n, 10, 2 * best_n - 1)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
            if stall > 4:
                break
            best_h = project_feasible(best_h + np.random.randn(best_n) * 0.005, best_n)

    # Phase 2: LP-QP with top-k for finer focus
    deadline = t0 + budget_s * 0.50
    for k_c in [100, 50, 30, 20, 10]:
        if time.time() > deadline:
            break
        for _ in range(5):
            if time.time() > deadline:
                break
            h_new, c5_new = lp_qp_iterate(best_h, best_n, 8, k_c)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
            else:
                break

    # Phase 3: Subgradient refinement
    deadline = t0 + budget_s * 0.60
    if time.time() < deadline:
        alpha = 0.02
        for _ in range(200):
            if time.time() > deadline:
                break
            h = subgrad_step(best_h, best_n, 5, alpha)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
            alpha *= 0.9995

    # Phase 4: SLSQP polish
    deadline = t0 + budget_s * 0.75
    if time.time() < deadline:
        for T in [2000, 8000]:
            if time.time() > deadline:
                break
            h_t = slsqp_optimize(best_h, softmax_obj, (T,), 250, 1e-14, best_n)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_h = h_t
                best_c5 = c5_t
        for k in [3, 5, 8]:
            if time.time() > deadline:
                break
            h_k = slsqp_optimize(best_h, topk_avg_obj, (k,), 250, 1e-14, best_n)
            c5_k = compute_c5(h_k)
            if c5_k < best_c5:
                best_h = h_k
                best_c5 = c5_k

    # Phase 5: Multi-resolution
    deadline = t0 + budget_s * 0.88
    for nr in [300, 400, 150, 100]:
        if time.time() > deadline:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, nr, endpoint=False)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        h_new, c5_new = lp_qp_iterate(hu, nr, 25, min(120, 2 * nr - 1))
        h_p = slsqp_optimize(h_new, topk_avg_obj, (5,), 200, 1e-13, nr)
        c5_p = compute_c5(h_p)
        # Downsample to best_n
        x_dn = np.linspace(0, 2, best_n, endpoint=False)
        hd = np.interp(x_dn, x_new, h_p)
        hd = project_feasible(hd, best_n)
        c5_dn = compute_c5(hd)
        if c5_dn < best_c5:
            best_c5 = c5_dn
            best_h = hd
        elif c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p
            best_n = nr
            x_old = x_new

    # Phase 6: Final LP-QQ + SLSQP
    deadline = t0 + budget_s * 0.96
    stall = 0
    while time.time() < deadline:
        h_new, c5_new = lp_qp_iterate(best_h, best_n, 5, 2 * best_n - 1)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
            if stall > 6:
                break

    if time.time() < t0 + budget_s * 0.985:
        h_f = slsqp_optimize(best_h, topk_avg_obj, (3,), 400, 1e-15, best_n)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_h = h_f
            best_c5 = c5_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
