import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, N, corr, tau, K):
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(c_top)) / len(c_top)
    else:
        w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-6:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[:seg] += wi * (1.0 - h[L:L + seg])
            grad[L:L + seg] -= wi * h[:seg]
        else:
            M = -L
            seg = N - M
            grad[M:M + seg] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[M:M + seg]
    return grad, w, top_idx

def sa_binary_search(h, N, dx, deadline, max_iters, seed_val):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    rng = np.random.RandomState(seed_val)
    T = 0.015
    T_min = 0.0005
    it = 0
    best_it = it
    patience = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        best_h[i] = 1.0
        best_h[j] = 0.0
        new_c5 = compute_c5(best_h, dx)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            patience = 0
            ones = np.concatenate([ones[ones != j], [i]])
            zeros = np.concatenate([zeros[zeros != i], [j]])
        else:
            delta = new_c5 - best_c5
            accept = delta <= 0 or (rng.rand() < np.exp(-delta / max(T, 1e-12)))
            if accept:
                ones = np.concatenate([ones[ones != j], [i]])
                zeros = np.concatenate([zeros[zeros != i], [j]])
                patience += 1
            else:
                best_h[i] = 0.0
                best_h[j] = 1.0
        T = max(T_min, T * 0.9995)

def cont_optimize(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K, seed_val):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    total_iters = max(1, int((deadline - time.time()) / 0.001))
    while time.time() < deadline:
        frac = min(float(it) / max(total_iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')

        it += 1

        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
