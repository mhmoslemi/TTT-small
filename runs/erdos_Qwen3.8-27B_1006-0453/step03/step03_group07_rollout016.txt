        n_cd_iters = max(50, int((budget_s * 0.995 - (time.time() - t0)) / 0.02))
        for _ in range(n_cd_iters):
            if time.time() - t0 > budget_s * 0.995:
                break
            j = rng.randint(n_cd)
            improved = False
            for delta in [-0.04, -0.015, 0.015, 0.04, -0.08, 0.08]:
                h_try = best_h.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                h_try = make_feasible(h_try, n_cd)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
            if not improved and rng.random() < 0.1:
                # Occasionally try a pair update
                j2 = rng.randint(n_cd)
                d1, d2 = rng.choice([-0.03, 0.03], 2)
                h_try = best_h.copy()
                h_try[j] = np.clip(h_try[j] + d1, 0, 1)
                h_try[j2] = np.clip(h_try[j2] + d2, 0, 1)
                h_try = make_feasible(h_try, n_cd)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
