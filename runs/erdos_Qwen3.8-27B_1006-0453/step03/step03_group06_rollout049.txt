def grad_weighted(h, idx, weights):
    """Compute Σ_j weights[j] * ∂corr_{idx[j]}/∂h"""
    n = len(h)
    grad = np.zeros(n)
    dx = 2.0 / n
    for jj in range(len(idx)):
        w = weights[jj]
        if w < 1e-15: continue
        shift = idx[jj] - (n - 1)
        if shift >= 0:
            # ∂/∂h[i] for i in [0, n-1-shift): (1 - h[i+shift])
            # ∂/∂h[i] for i in [shift, n-1]: -h[i-shift]

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:

        h = h + room * (diff / total)

def lgbf_obj_grad(h, T, k, lam):
    """Objective and gradient for L-BFGS-B."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if k is None or k >= len(corr):
        idx = np.arange(len(corr))
        k_use = len(corr)
    else:
        idx = np.argpartition(corr, -k)[-k:]
        k_use = k
    topk = corr[idx]
    m = np.max(topk)
    # Softmax weights
    e = np.exp(T * (topk - m))
    e_max = np.max(e)
    e = e / e_max
    w = e / e.sum()
    val = m + np.log(np.sum(np.exp(T*(topk-m)))) / T
    
    # Gradient of the smooth max (frozen weight)
    grad = np.zeros(n)
    for jj in range(k_use):
        wi = w[jj]
        if wi < 1e-15:
            continue
        shift = idx[jj] - (n - 1)
        if shift >= 0:
            if n - shift > 0:
                grad[:n-shift] += wi * (1.0 - h[shift:])
            if shift > 0:
                grad[shift:] -= wi * h[:n-shift]
        else:
            s = -shift
            if n - s > 0:
                grad[s:] += wi * (1.0 - h[:n-s])
                grad[:n-s] -= wi * h[s:]
    grad *= dx
    
    # Penalty for sum constraint
    s_h = h.sum()
    penalty = lam * (s_h - n/2.0)**2
    pen_grad = 2.0 * lam * (s_h - n/2.0)
    
    return val + penalty, grad + pen_grad

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def _grad_smooth_topk(h, T, k_top):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    total_len = len(corr)
    kk = min(k_top, total_len) if k_top is not None else total_len
    if kk < total_len:
        idx = np.argpartition(corr, -kk)[-kk:]
    else:
        idx = np.arange(total_len)
    topk = corr[idx]
    m = np.max(topk)
    e = np.exp(T * (topk - m))
    e -= e.max()
    w = e / e.sum()
    val = m + np.log(np.sum(np.exp(T * (topk - m)))) / T
    grad = np.zeros(n)
    for jj in range(kk):
        wi = w[jj]
        if wi < 1e-14:
            continue
        shift = idx[jj] - (n - 1)
        if shift >= 0:
            ns = n - shift
            if ns > 0:
                grad[:ns] += wi * (1.0 - h[shift:])
            if shift > 0:
                grad[shift:] -= wi * h[:ns]
        else:
            s = -shift
            ns = n - s
            if ns > 0:
                grad[s:] += wi * (1.0 - h[:ns])
                grad[:ns] -= wi * h[s:]
    grad *= dx
    return val, grad, idx, w

def _lgbf_func(h, T, k_top, lam, n):
    val, grad, _, _ = _grad_smooth_topk(h, T, k_top)
    s_h = h.sum()
    pen = lam * (s_h - n / 2.0) ** 2
    pg = 2.0 * lam * (s_h - n / 2.0)
    return val + pen, grad + pg

def _lgbf_opt(h0, T, k_top, lam, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    res = minimize(_lgbf_func, h0, args=(T, k_top, lam, n),
                   method='L-BFGS-B', bounds=bnds, jac=True,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return project_feasible(res.x, n)

def _coord_descent(h, n, iterations):
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    for _ in range(iterations):
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full') * dx
        m = np.max(corr)
        active = np.where(corr > m - 0.005)[0]
        np.random.shuffle(active)
        for ai in active[:10]:
            shift = ai - (n - 1)
            for _inner in range(3):
                for j in range(n):
                    if j + shift < 0 or j + shift >= n:
                        continue
                    for delta in [0.02, -0.02, 0.01, -0.01]:
                        h_try = best.copy()
                        h_try[j] += delta
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            best = h_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            break
    return best, best_c5

def _structured_init(n, rng):
    """Create diverse structured initial guesses."""
    inits = []
    # Block structure: 1 on first half
    h = np.zeros(n)
    h[:n // 2] = 1.0
    inits.append(project_feasible(h + rng.uniform(-0.05, 0.05, n), n))
    # Two alternating blocks
    h = np.zeros(n)
    block = max(4, n // 8)
    for i in range(0, n, block * 2):
        h[i:i + block] = 1.0
    inits.append(project_feasible(h + rng.uniform(-0.1, 0.1, n), n))
    # Cosine-shaped
    x = np.linspace(0, 2 * np.pi, n)
    h = 0.5 + 0.5 * np.cos(x + rng.uniform(0, 2 * np.pi))
    inits.append(project_feasible(h, n))
    # Smooth binary-like with random breakpoints
    h = np.zeros(n)
    n_blocks = rng.randint(2, 5)
    breaks = np.sort(rng.choice(np.arange(1, n), n_blocks - 1, replace=False))
    breaks = np.concatenate([[0], breaks, [n]])
    for bb in range(n_blocks):
        if rng.rand() > 0.5:
            h[breaks[bb]:breaks[bb + 1]] = 1.0
    h += rng.uniform(-0.03, 0.03, n)
    inits.append(project_feasible(h, n))
    return inits
