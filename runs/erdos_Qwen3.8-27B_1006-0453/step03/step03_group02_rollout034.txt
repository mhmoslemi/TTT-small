import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    # Bisection for the shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    shift = (lo + hi) / 2.0
    return np.clip(h - shift, 0.0, 1.0)

def lse_gradient(h, N, T):
    """Compute gradient of LSE-smoothed max correlation objective.
    Returns (gradient, c5_exact)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_exact = float(np.max(corr)) * (2.0/N)
    
    # Softmax weights
    m = corr.max()
    exp_c = np.exp(T * (corr - m))
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(N)
    center = N - 1
    
    for k in range(len(p)):
        wi = p[k]
        if wi < 1e-8:
            continue
        lag = k - center
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:N-lag] += wi * (1.0 - h[lag:N])
            grad[lag:] -= wi * h[:N-lag]
        else:
            m2 = -lag
            grad[:N-m2] -= wi * h[m2:]
            grad[m2:] += wi * (1.0 - h[:N-m2])
    
    return grad, c5_exact

def lse_optimize(h, N, target, dx, deadline, T_start, T_end, lr_init, iters_hint):

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    m_v = np.zeros(N)

    it = 0

    s = float(h.sum())

    lo, hi = -1.0, 1.0

            lo = mid
        else:
            hi = mid

def lse_grad(h, N, T):

    c5 = float(corr.max()) * (2.0 / N)
    m = corr.max()
    ec = np.exp(T * (corr - m))
    p = ec / ec.sum()
    grad = np.zeros(N)
    ctr = N - 1
    for k in range(len(p)):

        if w < 1e-9:

            grad += w * (1.0 - 2.0 * h)

            grad[:N - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:N - lag]
        else:
            ml = -lag
            grad[:N - ml] -= w * h[ml:]
            grad[ml:] += w * (1.0 - h[:N - ml])
    return grad, c5
