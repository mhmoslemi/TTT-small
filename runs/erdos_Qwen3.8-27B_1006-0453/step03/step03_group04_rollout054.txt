    D = np.zeros((N, 2*N-1))
    idx = np.arange(2*N-1)
    d_lags = idx - (N-1)  # shape (2N-1,), values from -(N-1) to N-1
    
    # For each p, d:
    # A part: 1-h[p+d] if 0≤p+d<N
    # B part: h[p-d] if 0≤p-d<N
    
    p_idx = np.arange(N)[:, None]  # (N, 1)
    d_vals = d_lags[None, :]  # (1, 2N-1)
    
    pd = p_idx + d_vals  # (N, 2N-1)
    p_minus_d = p_idx - d_vals  # (N, 2N-1)
    
    valid_pd = (pd >= 0) & (pd < N)  # (N, 2N-1)
    valid_pmd = (p_minus_d >= 0) & (p_minus_d < N)
    
    # A part: -(1-h[p+d]) * valid_pd
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N-1)], 0)

    cur_pk = best_pk
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / n_iters)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_pk = float(np.max(compute_corr(h_b)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.004 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
    return best, best_pk * (2.0 / N)

def adam_run(h, N, target, deadline, k=40, lr0=0.04, decay=0.992):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wk * (1.0 - h[M:])
                    grad[M:] -= wk * h[:N - M]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
    return best_h, best_peak * dx

def make_init(N, pattern, rng):
    if pattern == 'alt':
        h = np.zeros(N)
        h[::2] = 1.0
    elif pattern == 'blocks':
        h = np.zeros(N)
        h[:N//2] = 1.0
    else:
        h = (rng.random(N) > 0.5).astype(float)
    n_ones = int(h.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Steepest descent on binary version of initial
    N = N0
    h_bin = (h0 > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[:nh - n_ones]] = 1
    d1 = deadline - 300
    h_s = steepest_descent_binary(h_bin, N, max_sweeps=500, deadline=d1)
    c5_s = compute_c5(h_s, N)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Iterated local search (anneal + steepest descent)
    d2 = deadline - 80
    h_cur = best_h.copy()
    for restart in range(20):
        if time.time() > d2:
            break
        remaining = d2 - time.time()
        ann_time = max(10.0, min(25.0, remaining * 0.4))
        n_ann = int(ann_time * 100)
        h_a, c5_a = fast_anneal(h_cur, N, n_ann, rng, T0=3.0, T1=0.003)
        sd_deadline = d2 - 5
        h_s = steepest_descent_binary(h_a, N, max_sweeps=400, deadline=sd_deadline)
        c5_s = compute_c5(h_s, N)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
            h_cur = h_s.copy()

    # Phase 3: Different structured starts
    if time.time() < deadline - 60:
        for pat in ['alt', 'rand']:
            if time.time() > deadline - 60:
                break
            h_init = make_init(N, pat, rng)
            h_s = steepest_descent_binary(h_init, N, max_sweeps=300, deadline=deadline - 55)
            c5_s = compute_c5(h_s, N)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()

    # Phase 4: Continuous Adam polish
    if time.time() < deadline - 15:
        h_pol = best_h.astype(float) + rng.randn(len(best_h)) * 0.08
        h_pol = project_sum(h_pol, len(best_h) / 2.0)
        h_ad, c5_ad = adam_run(h_pol, len(best_h), len(best_h) / 2.0, deadline - 3, k=50, lr0=0.02, decay=0.995)
        if c5_ad < best_c5:
            best_c5, best_h = c5_ad, h_ad.copy()

    # Phase 5: Different N
    for N_try in [N0 + 20, N0 + 40, N0 - 20]:
        if time.time() > deadline - 20 or N_try < 200:
            break
        if N_try % 2 != 0:
            N_try -= 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = (h_interp > 0.5).astype(float)
        n_ones = int(h_interp.sum())
        nh = N_try // 2
        if n_ones > nh:
            idx = np.where(h_interp == 1)[0]
            h_interp[idx[nh:]] = 0
        elif n_ones < nh:
            idx = np.where(h_interp == 0)[0]
            h_interp[idx[:nh - n_ones]] = 1
        h_s = steepest_descent_binary(h_interp, N_try, max_sweeps=200, deadline=deadline - 10)
        c5_s = compute_c5(h_s, N_try)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        if time.time() < deadline - 5:
            h_pol = h_s.astype(float) + rng.randn(N_try) * 0.06
            h_pol = project_sum(h_pol, N_try / 2.0)
            h_ad, c5_ad = adam_run(h_pol, N_try, N_try / 2.0, deadline - 2, k=50, lr0=0.015, decay=0.996)
            if c5_ad < best_c5:
                best_c5, best_h = c5_ad, h_ad.copy()

    # Final: second Adam polish on absolute best
    if time.time() < deadline - 3:
        hF = best_h.astype(float) + rng.randn(len(best_h)) * 0.03
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = adam_run(hF, len(best_h), len(best_h) / 2.0, deadline - 1, k=60, lr0=0.008, decay=0.997)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
