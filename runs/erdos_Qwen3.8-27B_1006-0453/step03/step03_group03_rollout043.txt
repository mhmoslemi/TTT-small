    h = np.clip(h, 0.0, 1.0).copy()

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += r * (d / t)
        else:
            r = h.copy()
            t = r.sum()
            if t < 1e-14:
                break
            h += r * (d / t)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(corr.max())
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / len(x))

def smooth_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(corr.max())
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    g = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[:n-k] -= wi * x[k:]
            g[k:] += wi * (1.0 - x[:n-k])
        else:
            j = -k
            g[:n-j] += wi * (1.0 - x[j:])
            g[j:] -= wi * x[:n-j]
    return g * dx

def _sum_eq(x, n):
    return x.sum() - n / 2.0

                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _dir_ineq(x, dx):

    return x[-1] - np.correlate(h, 1.0 - h, mode='full') * dx

def slsqp_direct(h, maxiter=600):

    dx = 2.0 / n
    _, corr = compute_c5(h)
    t0 = float(corr.max()) * dx + 0.05
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    cons = [
        {'type': 'eq', 'fun': _dir_eq, 'args': (n,)},
        {'type': 'ineq', 'fun': _dir_ineq, 'args': (dx,)}

    try:
        res = minimize(_dir_obj, x0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def grad_descent(h, n_iters=2000, T=10000):

    hc = h.copy()

            hc, lr = hn, min(lr * 1.02, 0.5)
