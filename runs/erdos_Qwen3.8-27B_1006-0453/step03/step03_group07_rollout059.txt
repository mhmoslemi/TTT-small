def compute_c5(h):

    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 0.999999
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.000001
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, T, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        exp_diff = np.exp(T * (corr - m))
        softmax = exp_diff / exp_diff.sum()
        # d(dxcorr_k)/dh[i] = (1-h)[i+off_k] - h[i-off_k]
        offsets = np.arange(len(corr)) - (n - 1)
        g = np.zeros(n)
        for idx in np.where(softmax > 1e-6)[0]:
            off = offsets[idx]
            w = T * dx * softmax[idx]
            # (1-h)[i+off] term: i+off in [0,n-1]
            lo1 = max(0, -off)
            hi1 = min(n, n - off) if off > 0 else n
            if hi1 > lo1:
                g[lo1:hi1] += w * (1.0 - h[lo1+off:hi1+off])
            # -h[i-off] term: i-off in [0,n-1]
            lo2 = max(0, off)
            hi2 = min(n, n + off) if off > 0 else n
            if hi2 > lo2:
                g[lo2:hi2] -= w * h[lo2-off:hi2-off]
        return g
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n/2.0,
             'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

    """Local search on 0-1 valued solutions."""
    rng = np.random.RandomState(seed)
    h = (h0 > 0.5).astype(float)

    zeros = np.where(h == 0)[0]

    while len(ones) > target:

            h_try[src] = 0

        elif op == 1:  # Swap adjacent

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
