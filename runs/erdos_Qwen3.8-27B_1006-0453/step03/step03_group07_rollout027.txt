import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h0

def sweep_optimize(h, n, alpha=0.1, n_sweeps=10):
    """Iterative sweep: for each lag, adjust h to reduce overlap."""
    h = h.copy()
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Process lags in order of worst to best
        order = np.argsort(corr)[::-1]
        for idx in order:
            shift = idx - (n - 1)
            if shift == 0:
                continue
            s = abs(shift)
            if s >= n:
                continue
            if shift > 0:
                for i in range(n - s):
                    d = h[i + s] - h[i]
                    h[i] += alpha * d
                    h[i + s] -= alpha * d
            else:
                for i in range(-s, n):
                    d = h[i + s] - h[i]
                    h[i] += alpha * d
                    h[i + s] -= alpha * d
            h = np.clip(h, 0, 1)
        h = make_feasible(h, n)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h0

def grad_lag(h, idx, n):
    """Gradient of corr[idx] w.r.t. h."""
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
    else:
        s = -k
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    return grad

    else:
        vel = lr * grad
