def get_corr_scaled(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_c5_val(x, T, topk):
    corr = get_corr_scaled(x)
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    m = np.max(corr)
    if T > 100000:
        return m
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return np.clip(h, 0, 1)
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-13:
            return np.clip(h, 0, 1)
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)
