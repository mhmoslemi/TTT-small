            grad += 1.0 - 2.0 * h
        elif k > 0:

            if N - 2*k + 1 > 0:

            grad[0:N-k] += (-h[k:N])

            best_peak = peak
            best_h = h.copy()
        # Find active lags (top-k)
        idx = np.argpartition(corr, -k_active)[-k_active:]
        grad = grad_active(h, N, idx)
        m = b1 * m + (1 - b1) * grad

        h = h - lr_cur * mh / (np.sqrt(vh) + eps)

- Identify the 20 ones that are most "expensive" (contribute most to the max correlation)
- Identify the 20 zeros that would be most "beneficial" to flip
- Try all 400 swaps between them
- Take the best improvement
