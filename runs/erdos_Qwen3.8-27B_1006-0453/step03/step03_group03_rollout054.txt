def linearized_minimax(h, n, M=5):
    """Solve linearized minimax subproblem."""
    _, corr = compute_c5(h)
    dx = 2.0/n
    # Get top M shifts
    top_m = np.argsort(corr)[::-1][:M]
    # Build LP: minimize t s.t. c_k^T h <= t + b_k, 0<=h<=1, sum(h)=n/2
    # where c_k = gradient of corr at h, b_k = corr[k] - c_k^T h
    # Actually: corr_k(h) ≈ corr_k(h*) + sum_j grad_k[j]*(h[j]-h*[j])
    # So: corr_k(h) <= t means sum_j grad_k[j]*h[j] <= t - corr_k(h*) + sum_j grad_k[j]*h*[j]
    
    # Number of variables: n (for h) + 1 (for t)
    # Minimize: t (last variable)
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ [h, t] <= b_ub
    # For each active shift k: sum_j grad_k[j] * h[j] - t <= RHS_k
    # grad_k[j] = (1-h[j+s_k])*[valid] - h[j-s_k]*[valid]
    
    A_rows = []
    b_rows = []
    
    for k in top_m:
        s = k - (n-1)
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j+s <= n-1:
                grad[j] += (1 - h[j+s])
            if 0 <= j-s <= n-1:
                grad[j] -= h[j-s]
        row = np.zeros(n+1)
        row[:n] = grad
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(corr[k] - grad @ h)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds: 0 <= h[j] <= 1, t >= 0
    bounds = [(0,1)]*n + [(0, None)]
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum: add to elements with room
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_box_sum(h, target):
    """Euclidean projection onto {x: 0<=x_i<=1, sum(x)=target}."""
    n = len(h)
    h = np.clip(h, 0, 1).astype(float)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def lse_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def lse_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_c = np.exp(T * (corr - m))
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(2*n-1)  # accumulate per shift
    # For each shift k, compute contribution to each h[j]
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-16:
            continue
        s = k - (n-1)
        # (1-h[j+s])*[0<=j+s<=n-1]
        if s >= 0:
            grad[:n-s] += wk * (1.0 - h[s:])
        else:
            grad[-s:] += wk * (1.0 - h[:-s])
        # -h[j-s]*[0<=j-s<=n-1]
        if s >= 0:
            grad[s:] -= wk * h[:n-s]
        else:
            grad[:n+s] -= wk * h[-s:]
    
    return dx * grad, corr * dx

def fista_lse(h0, T, max_iter=10000, tol=1e-14):
    n = len(h0)
    h = project_box_sum(h0, n/2.0).copy()
    y = h.copy()
    t_k = 1.0
    step = 1.0 / (T + 1)  # initial step size
    best_h = h.copy()
    best_val = lse_objective(h, T)
    
    for it in range(max_iter):
        y = project_box_sum(y, n/2.0)
        g, _ = lse_gradient(y, T)
        # Gradient step
        h_new = project_box_sum(y - step * g, n/2.0)
        val_new = lse_objective(h_new, T)
        
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        
        # Nesterov momentum
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        y = h_new + ((t_k - 1) / t_new) * (h_new - h)
        h = h_new
        t_k = t_new
        
        if it % 500 == 0 and it > 0:
            # Backtrack step size if not decreasing
            if best_val > lse_objective(best_h, T) * (1 + 1e-10) + 1e-15:
                step *= 0.9
        if it % 200 == 0 and it > 0:
            step = min(step * 1.01, 1.0/(T+1))
    
    return best_h, best_val

def linearized_minimax(h, n, M=5, max_rounds=50):
    """Iterative linearized minimax."""
    h_cur = h.copy()
    best_c5, _ = compute_c5(h_cur)
    
    for rnd in range(max_rounds):
        _, corr = compute_c5(h_cur)
        dx = 2.0/n
        top_m = np.argsort(corr)[::-1][:M]
        
        # Build LP
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        
        A_rows = []
        b_rows = []
        
        for k in top_m:
            s = k - (n-1)
            grad_k = np.zeros(n)
            if s >= 0:
                grad_k[:n-s] += (1.0 - h_cur[s:])
                grad_k[s:] -= h_cur[:n-s]
            else:
                grad_k[-s:] += (1.0 - h_cur[:-s])
                grad_k[:n+s] -= h_cur[-s:]
            
            row = np.zeros(n+1)
            row[:n] = grad_k
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(corr[k] - grad_k @ h_cur)
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        bounds = [(0,1)]*n + [(0, None)]
        
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if not res.success:
            break
        
        h_lp = res.x[:n]
        # Step: interpolate between h_cur and h_lp
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = h_cur + alpha * (h_lp - h_cur)
            h_try = project_box_sum(h_try, n/2.0)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                h_cur = h_try
                break
        else:
            if rnd > 10:
                break
    
    return h_cur, best_c5

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    def obj(x):
        return lse_objective(x, T)
    def grad(x):
        g, _ = lse_gradient(x, T)
        return g
    def sum_eq(x):
        return x.sum() - len(x)/2.0
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun': sum_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_box_sum(np.clip(res.x, 0, 1), n/2.0)
    except:
        return h
