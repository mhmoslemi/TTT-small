def smooth_max_and_grad(h, T):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))

    val = (logsum / T) * dx

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h += room * (diff / room.sum())
        else:
            if h.sum() < 1e-14:
                break
            h += h * (diff / h.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def pfg_func(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    logsum = m + np.log(exps.sum())
    val = (logsum / T) * dx
    w = exps / exps.sum()
    s = h.sum() - n / 2.0
    val += lam * s * s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
        else:
            kp = -k
            grad[kp:] += wi * (1.0 - h[:n - kp])
            grad[:n - kp] -= wi * h[kp:]
    grad *= dx
    grad += 2.0 * lam * s
    return (val, grad)

def smv_func(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return ((m + np.log(np.sum(np.exp(T * (corr - m)))))) / T * dx

def smv_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
        else:
            kp = -k
            grad[kp:] += wi * (1.0 - h[:n - kp])
            grad[:n - kp] -= wi * h[kp:]
    return grad * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def lbfgs_opt(h, T, maxiter, lam):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(pfg_func, h, args=(T, lam), jac=True,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7,
                                'maxls': 30, 'maxfun': maxiter * 3})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_opt(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    try:
        res = minimize(smv_func, h, args=(T,),
                       jac=lambda x: smv_grad(x, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_search(h_init, n_iters):
    n = len(h_init)
    order = np.argsort(h_init)
    h_b = np.zeros(n)
    h_b[order[n // 2:]] = 1.0
    best = h_b.copy()
    best_c5, _ = compute_c5(best)
    for _ in range(n_iters):
        mv = np.random.randint(0, 3)
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        t = best.copy()
        if mv == 0:
            idx = np.random.choice(ones) if np.random.rand() < 0.5 else np.random.choice(zeros)
            t[idx] = 1.0 - t[idx]
        elif mv == 1:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            t[i], t[j] = 0.0, 1.0
        else:
            i1, i2 = np.random.choice(ones, 2, replace=False)
            j1, j2 = np.random.choice(zeros, 2, replace=False)
            t[i1], t[i2] = 0.0, 0.0
            t[j1], t[j2] = 1.0, 1.0
        c5t, _ = compute_c5(t)
        if c5t < best_c5 - 1e-15:
            best, best_c5 = t, c5t
    return best, best_c5

def coord_refine(h, n_rounds, T):
    n = len(h)
    best = h.copy()
    best_c5, _ = compute_c5(best)
    for _ in range(n_rounds):
        idx = np.random.choice(n, size=3, replace=False)
        idx_s = np.sort(idx)
        for _ in range(30):
            hp = best.copy()
            hp[idx_s] += np.random.randn(3) * 0.08
            hp = project_feasible(hp, n)
            c5t, _ = compute_c5(hp)
            if c5t < best_c5 - 1e-13:
                best, best_c5 = hp, c5t
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: L-BFGS-B quick refinement
    for T in [300, 2000, 10000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        ht = lbfgs_opt(best_h, T, maxiter=200, lam=30.0)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 2: Binary local search at multiple resolutions
    res_list = [54, 66, 78, best_n, 90, 102, 114]
    for nt in res_list:
        if time.time() - t0 > budget_s * 0.30:
            break
        if nt % 2 != 0 or nt < 20:
            continue
        if nt == best_n:
            hs = best_h.copy()
        else:
            xo = np.linspace(0, 2, best_n)
            xn = np.linspace(0, 2, nt)
            hs = np.interp(xn, xo, best_h)
        hb, c5b = binary_search(hs, n_iters=5000)
        if c5b < best_c5:
            best_c5, best_h, best_n = c5b, hb, nt

    # Phase 3: SLSQP refinement (exact constraint)
    for T in [2000, 10000, 40000]:
        if time.time() - t0 > budget_s * 0.50:
            break
        ht = slsqp_opt(best_h, T, maxiter=400)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 4: L-BFGS-B high T
    for T in [50000, 150000]:
        if time.time() - t0 > budget_s * 0.60:
            break
        ht = lbfgs_opt(best_h, T, maxiter=300, lam=80.0)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 5: Coordinate descent refinement
    if time.time() - t0 < budget_s * 0.70:
        hc, c5c = coord_refine(best_h, n_rounds=200, T=10000)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc
        ht = slsqp_opt(best_h, 30000, maxiter=300)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 6: Perturbation restarts
    for _ in range(18):
        if time.time() - t0 > budget_s * 0.84:
            break
        sc = 0.015 + 0.05 * np.random.rand()
        hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        ht = lbfgs_opt(hp, 5000, maxiter=150, lam=40.0)
        ht = lbfgs_opt(ht, 50000, maxiter=200, lam=100.0)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 7: Rotations
    for sh in [1, 2, 3, 5, 8, 13]:
        if time.time() - t0 > budget_s * 0.92:
            break
        hr = project_feasible(np.roll(best_h, sh), best_n)
        ht = lbfgs_opt(hr, 20000, maxiter=200, lam=60.0)
        ht = lbfgs_opt(ht, 100000, maxiter=200, lam=100.0)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 8: Final SLSQP ultra-high T
    if time.time() - t0 < budget_s * 0.98:
        ht = slsqp_opt(best_h, 100000, maxiter=500)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht
        ht = slsqp_opt(best_h, 300000, maxiter=400)
        c5t, _ = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
