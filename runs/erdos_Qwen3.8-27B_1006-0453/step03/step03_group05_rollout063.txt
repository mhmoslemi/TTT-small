            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_corr = np.exp(T * (corr - m))
    exp_sum = np.sum(exp_corr)
    obj = (m + np.log(exp_sum) / T) * dx
    
    w = exp_corr / exp_sum
    
    # Use only top-k weights if specified
    if top_k is not None and top_k < len(w):
        idx = np.argsort(w)[::-1][:top_k]
        w_sparse = np.zeros_like(w)
        w_sparse[idx] = w[idx]
        w = w_sparse
    
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue

        wk = w[k]

        m_lo = max(0, s)
        m_hi = min(n, s + n)
        if m_lo < m_hi:
            grad[m_lo:m_hi] += wk * (1.0 - h[m_lo - s:m_hi - s])
        # -h[m+s] for m+s in [0,n): m in [-s, n-s) intersect [0,n)
        m_lo2 = max(0, -s)
        m_hi2 = min(n, n - s)
        if m_lo2 < m_hi2:
            grad[m_lo2:m_hi2] -= wk * h[m_lo2 + s:m_hi2 + s]
    
    grad *= dx
    return obj, grad

def projected_gradient_descent(h, T, n_steps, lr, decay, top_k=None):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr_cur = lr
    for i in range(n_steps):
        obj, grad = smooth_obj_grad(h, T, top_k)
        h_new = project_feasible(h - lr_cur * grad, len(h))
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr_cur *= decay
    return best_h, best_c5

def slsqp_topk(h, k, n, maxiter, ftol):

        corr = np.correlate(x, 1.0 - x, mode='full')
        dx = 2.0 / n
        top = np.sort(corr)[::-1][:k]
        return float(np.sum(top) * dx / k)

        dx = 2.0 / n

        g = np.zeros(n)

            if lo < hi:
                g[lo:hi] += (1.0 - x[lo-s:hi-s]) / k

            if lo2 < hi2:
                g[lo2:hi2] -= x[lo2+s:hi2+s] / k
        g *= dx
        return obj(x), g
    bnds = [(0.0, 1.0)] * n

        hi = min(n, s + n)
        if lo < hi:
            grad[lo:hi] += wk * (1.0 - h[lo - s:hi - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 + s:hi2 + s]
    grad *= dx
    return obj, grad

def proj_gd(h, T, n_steps, lr0, decay, top_k=None):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for _ in range(n_steps):
        _, grad = smooth_obj_grad(h, T, top_k)
        h = project_feasible(h - lr * grad, len(h))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= decay
    return best_h, best_c5

def _topk_fun(x, k, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def _topk_jac(x, k, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    top_idx = np.argsort(corr)[::-1][:k]
    g = np.zeros(n)
    nn1 = n - 1
    for ki in top_idx:
        s = ki - nn1
        lo = max(0, s)
        hi = min(n, s + n)
        if lo < hi:
            g[lo:hi] += (1.0 - x[lo - s:hi - s]) / k
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            g[lo2:hi2] -= x[lo2 + s:hi2 + s] / k
    g *= dx
    return g

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_topk(h, k, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_topk_fun, h, args=(k, n), jac=_topk_jac,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _softmax_fun(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def _softmax_jac(x, T, n):
    _, g = smooth_obj_grad(x, T, top_k=None)
    return g

def slsqp_softmax(h, T, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_softmax_fun, h, args=(T, n), jac=_softmax_jac,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient descent, increasing temperature
    for T in [30, 100, 300, 1000, 3000, 10000, 30000]:
        if elapsed() > budget_s * 0.12:
            break
        hng, c5ng = proj_gd(best_h, T, 400, 0.004, 0.999, top_k=25)
        if c5ng < best_c5:
            best_c5, best_h = c5ng, hng

    # Phase 2: SLSQP with analytical top-k gradients
    for k in [3, 5, 8]:
        if elapsed() > budget_s * 0.28:
            break
        hn = slsqp_topk(best_h, k, best_n, 400, 1e-14)
        cn = compute_c5(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 3: SLSQP with softmax objective
    for T in [500, 2000]:
        if elapsed() > budget_s * 0.38:
            break
        hn = slsqp_softmax(best_h, T, best_n, 300, 1e-14)
        cn = compute_c5(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 4: Basin hopping with diverse perturbation + fast local search
    remaining = budget_s * 0.68 - elapsed()
    n_trials = max(int(remaining / 4.0), 15) if remaining > 0 else 0
    for trial in range(n_trials):
        if elapsed() > budget_s * 0.68:
            break
        r = np.random.random()
        if r < 0.25:
            sc = 0.02 + 0.05 * np.random.random()
            hc = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.45:
            hc = best_h.copy()
            blen = np.random.randint(4, best_n // 3 + 1)
            bs = np.random.randint(0, best_n - blen)
            hc[bs:bs + blen] = 1.0 - hc[bs:bs + blen]
            hc = project_feasible(hc, best_n)
        elif r < 0.60:
            hc = best_h.copy()
            ns = np.random.randint(3, 12)
            for _ in range(ns):
                i, j = np.random.choice(best_n, 2, replace=False)
                tmp = hc[i]
                hc[i] = hc[j]
                hc[j] = tmp
            hc = project_feasible(hc, best_n)
        elif r < 0.78:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            s_star = int(np.argmax(corr) - (best_n - 1))
            hc = best_h.copy()
            for _ in range(np.random.randint(5, 15)):
                j = np.random.randint(0, best_n)
                if 0 <= j + s_star < best_n:
                    hc[j] += 0.08 * np.random.randn()
                    hc[j + s_star] -= 0.08 * np.random.randn()
            hc = project_feasible(hc, best_n)
        else:
            hc = project_feasible(best_h + 0.06 * np.random.randn(best_n), best_n)

        # Fast local refine
        hr, cr = proj_gd(hc, 300, 250, 0.003, 0.999, top_k=12)
        hr2 = slsqp_topk(hr, 5, best_n, 200, 1e-13)
        cr2 = compute_c5(hr2)
        if cr2 < cr:
            hr, cr = hr2, cr2
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Phase 5: Multi-resolution (try lower and higher n)
    for nr in [120, 150, 250]:
        if elapsed() > budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu = project_feasible(hu + np.random.randn(nr) * 0.015, nr)
        hr, cr = proj_gd(hu, 500, 350, 0.003, 0.999, top_k=15)
        hr = slsqp_topk(hr, 5, nr, 250, 1e-13)
        cr = compute_c5(hr)
        # Downsample to compare
        hd = np.interp(x_old, x_new, hr)
        hd = project_feasible(hd, best_n)
        cd = compute_c5(hd)
        if cd < best_c5:
            best_c5, best_h = cd, hd
        elif cr < best_c5:
            best_c5, best_h, best_n = cr, hr, nr

    # Phase 6: Intensive final polish
    if elapsed() < budget_s * 0.93:
        for k in [3, 5]:
            hn = slsqp_topk(best_h, k, best_n, 500, 1e-15)
            cn = compute_c5(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        hn, cn = proj_gd(best_h, 5000, 400, 0.001, 0.999, top_k=5)
        if cn < best_c5:
            best_c5, best_h = cn, hn
        hn = slsqp_topk(best_h, 3, best_n, 400, 1e-15)
        cn = compute_c5(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 7: Last micro-adjustments
    if elapsed() < budget_s * 0.98:
        for _ in range(3):
            hn, cn = proj_gd(best_h, 8000, 150, 0.0005, 0.999, top_k=3)
            if cn < best_c5:
                best_c5, best_h = cn, hn
            hn = slsqp_topk(best_h, 3, best_n, 200, 1e-15)
            cn = compute_c5(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
