corr[t] = sum_j h[j] * (1-h[j + t - (n-1)]) for valid j and j+t-(n-1)

Let lag = t - (n-1). Then:
corr[t] = sum_j h[j] * (1 - h[j + lag]) for valid j, j+lag

For lag > 0: j ranges from 0 to n-1-lag
  corr = sum_{j=0}^{n-1-lag} h[j] * (1 - h[j+lag])

For lag < 0: let l = -lag > 0, j ranges from l to n-1
  corr = sum_{j=l}^{n-1} h[j] * (1 - h[j-l])
  = sum_{j=0}^{n-1-l} h[j+l] * (1 - h[j])

Gradient wrt h[m]:
For lag > 0:
  - If m ≤ n-1-lag: h[m] appears as first factor → contribution: (1-h[m+lag])
  - If m ≥ lag: h[m] appears as second factor (1-h[m]) → contribution: -h[m-lag]

For lag < 0 (l = -lag > 0):
  - If m ≥ l: h[m] appears as first factor h[j+l] with j=m-l → contribution: (1-h[m-l])
  - If m ≤ n-1-l: h[m] appears as second factor (1-h[m]) → contribution: -h[m+l]

Note: for lag > 0, the conditions m ≤ n-1-lag and m ≥ lag can both be true (when lag ≤ m ≤ n-1-lag).

def weighted_grad(h, w, n):
    """Compute sum_k w_k * grad f_k for all h."""
    grad = np.zeros(n)
    dx = 2.0/n
    n_shifts = 2*n - 1
    for t in range(n_shifts):
        wt = w[t]
        if wt < 1e-10:
            continue
        lag = t - (n-1)
        if lag == 0:
            # f_0 = sum h[j]*(1-h[j]) = sum h[j] - sum h[j]^2
            # grad = 1 - 2*h
            grad += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            # f = sum_{j=0}^{n-1-lag} h[j]*(1-h[j+lag])
            # grad[m] += (1-h[m+lag]) for m in [0, n-1-lag]
            # grad[m] += -h[m-lag] for m in [lag, n-1]
            grad[:n-lag] += wt * (1.0 - h[lag:])
            grad[lag:] += wt * (-h[:n-lag])
        else:
            l = -lag
            # f = sum_{j=0}^{n-1-l} h[j+l]*(1-h[j])
            # grad[m] += (1-h[m-l]) for m in [l, n-1]
            # grad[m] += -h[m+l] for m in [0, n-1-l]
            grad[l:] += wt * (1.0 - h[:n-l])
            grad[:n-l] += wt * (-h[l:])
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_topk(corr, T, k):
    idx = np.argpartition(corr, -k)[-k:]
    vals = corr[idx]
    m = vals.max()
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    total = room.sum()
    if total < 1e-13:
        return h
    h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def constraint_fn(x):
    return x.sum() - len(x) / 2.0

def weighted_grad(h, w, n):
    grad = np.zeros(n)
    n_shifts = 2 * n - 1
    for t in range(n_shifts):
        wt = w[t]
        if wt < 1e-10:
            continue
        lag = t - (n - 1)
        if lag == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wt * (1.0 - h[lag:])
            grad[lag:] += wt * (-h[:n - lag])
        else:
            l = -lag
            grad[l:] += wt * (1.0 - h[:n - l])
            grad[:n - l] += wt * (-h[l:])
    return grad * (2.0 / n)

def slsqp_opt(h0, T, n, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fn}]
    dx = 2.0 / n
    if topk is not None:
        def _obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return smooth_topk(c, T, topk) * dx
    else:
        def _obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    res = minimize(_obj, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def binary_local_search(h_init, n, max_iters, seed):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    for it in range(max_iters):
        moved = False
        if len(ones_idx) > 0 and len(zeros_idx) > 0:
            i1 = ones_idx[rng.randint(len(ones_idx))]
            i2 = zeros_idx[rng.randint(len(zeros_idx))]
            h[i1], h[i2] = h[i2], h[i1]
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h.copy()
                moved = True
            else:
                h[i1], h[i2] = h[i2], h[i1]
        if not moved and it % 300 == 299:
            for _ in range(4):
                if h.sum() > n / 2 and len(ones_idx) > 0:
                    h[ones_idx[rng.randint(len(ones_idx))]] = 0.0
                elif h.sum() < n / 2 and len(zeros_idx) > 0:
                    h[zeros_idx[rng.randint(len(zeros_idx))]] = 1.0
                else:
                    if len(ones_idx) > 0 and len(zeros_idx) > 0:
                        h[ones_idx[rng.randint(len(ones_idx))]], h[zeros_idx[rng.randint(len(zeros_idx))]] = \
                            h[zeros_idx[rng.randint(len(zeros_idx))]], h[ones_idx[rng.randint(len(ones_idx))]]
            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]
            best_c5 = compute_c5(h)
            best_h = h.copy()
    return best_h, best_c5

def subgradient_polish(h, n, n_iter, eta_init, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_shifts = 2 * n - 1
    w = np.ones(n_shifts) / n_shifts
    momentum = np.zeros(n)
    mu = 0.80
    for it in range(n_iter):
        t = it + 1
        step = eta_init / (np.sqrt(t) + 0.5)
        f = compute_corr(h)
        f_shifted = f - f.max()
        w = w * np.exp(4.0 * f_shifted)
        w /= w.sum()
        grad = weighted_grad(h, w, n)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            step = min(step, 0.1 / gnorm)
        momentum = mu * momentum - step * grad
        h_new = project_feasible(h + momentum, n)
        h = h_new
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n, max_rounds, seed):
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    deltas = [0.01, -0.01, 0.03, -0.03, 0.06, -0.06, 0.1, -0.1]
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for j in order:
            for delta in deltas:
                h_try = best_h.copy()
                h_try[j] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
        if not improved and rnd > 20:
            break
    return best_h, best_c5
