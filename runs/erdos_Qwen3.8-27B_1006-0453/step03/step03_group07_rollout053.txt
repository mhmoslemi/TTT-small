    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for j in range(2*n - 1):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(h + np.random.randn(n) * 1e-9)[::-1][:target]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def subgrad_step(h, T, step):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-13:
            continue
        s = j - (n - 1)
        lo = max(0, s)
        hi = min(n, n + s)
        ir = np.arange(lo, hi)
        g = 1.0 - h[ir - s]
        vp = ir + s
        msk = vp < n
        g[msk] -= h[vp[msk]]
        grad[ir] += wj * g
    return h - step * grad

def subgrad_descend(h, T, n_iter, base_step, bin_every=0):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        step = base_step / np.sqrt(it + 1)
        h_new = subgrad_step(h, T, step)
        h_new = make_feasible(h_new, n)
        if bin_every > 0 and (it + 1) % bin_every == 0:
            h_new = binary_project(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
        h = h_new
    return best_h, best_c5

def pair_swap_local(h, n_iters, T0):
    n = len(h)
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    for it in range(n_iters):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        d = np.random.uniform(-0.15, 0.15)
        h_try = h_cur.copy()
        h_try[i] += d
        h_try[j] -= d
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        temp = T0 * (1.0 - it / max(1, n_iters)) + 1e-4
        if c5_try < best_c5 or np.random.rand() < np.exp(-max(0.0, c5_try - best_c5) / temp):
            h_cur = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
    return h_cur, best_c5

def binary_local(h, n_iters, T0):
    n = len(h)
    h = binary_project(h, n)
    best_c5 = compute_c5(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for it in range(n_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5_try = compute_c5(h)
        temp = T0 * (1.0 - it / max(1, n_iters)) + 1e-5
        if c5_try < best_c5 or np.random.rand() < np.exp(-max(0.0, c5_try - best_c5) / temp):
            best_c5 = c5_try
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i], h[j] = h[j], h[i]
    return h, best_c5

def slsqp_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def slsqp_cons(x, n):
    return x.sum() - n / 2.0

def generate_init(n, typ, seed_val):
    rng = np.random.RandomState(seed_val)
    if typ == 0:
        h = np.zeros(n)
        h[0::2] = 1.0
    elif typ == 1:
        p = max(2, n // 7)
        blk = np.zeros(p)
        blk[:max(1, p // 2)] = 1.0
        h = np.tile(blk, n // p + 1)[:n]
        h = rng.permutation(h)
    elif typ == 2:
        h = (rng.rand(n) < 0.5).astype(float)
    elif typ == 3:
        h = rng.rand(n)
    elif typ == 4:
        p = max(3, n // 5)
        blk = np.zeros(p)
        np.fill blk[:p * 3 // 5], 1.0
        h = np.tile(blk, n // p + 1)[:n]
    else:
        h = rng.rand(n)
    return make_feasible(h, n)
