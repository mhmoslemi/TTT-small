```python
def cheb_values(x, K):
    """Evaluate Chebyshev basis at x ∈ [0,2] mapped to [-1,1]."""
    t = x * 2.0 - 1.0  # Map [0,2] to [-1,1]
    vals = np.zeros((len(x), K+1))
    vals[:, 0] = 1.0
    if K >= 1:
        vals[:, 1] = t
    for k in range(2, K+1):
        vals[:, k] = 2.0 * t * vals[:, k-1] - vals[:, k-2]
    return vals  # shape (n, K+1)
```

Then h = cheb_values(x, K) @ c, and we optimize c.

The integral constraint: ∫₀² h(x) dx = Σ c_k ∫₀² T_k(x) dx = 1

For T_k on [-1,1]: ∫₋₁¹ T_k(t) dt = 0 for k odd, = 2/(1-k²) for k even (k>0), = 2 for k=0.

On [0,2], t = 2x-1, dt = 2dx:
∫₀² T_k(2x-1) dx = (1/2) ∫₋₁¹ T_k(t) dt

So ∫₀² T_0 dx = 1
∫₀² T_k dx = (1/2) * 2/(1-k²) = 1/(1-k²) for even k>0
∫₀² T_k dx = 0 for odd k

The constraint is: c_0 + Σ_{k even, k>0} c_k/(1-k²) = 1

This is a linear constraint! Good.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    sorted_c = np.sort(corr)[::-1]
    return float(np.sum(sorted_c[:k]) * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def shift_grad(h, shifts, n):
    grad = np.zeros(n)
    w = 1.0 / len(shifts)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            ss = -s
            grad[ss:] += w * (1.0 - h[:n - ss])
            grad[:n - ss] -= w * h[ss:]
    return grad

def slsqp_opt(h, obj_f, obj_args, maxiter, ftol, gfun=None):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        if gfun is not None:
            res = minimize(obj_f, h, args=obj_args, method='SLSQP',
                          bounds=bnds, constraints=cons, jac=gfun,
                          options={'maxiter': maxiter, 'ftol': ftol})
        else:
            res = minimize(obj_f, h, args=obj_args, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def cheb_basis_matrix(x, K):
    t = x * 2.0 - 1.0
    M = np.zeros((len(x), K + 1))
    M[:, 0] = 1.0
    if K >= 1:
        M[:, 1] = t
    for k in range(2, K + 1):
        M[:, k] = 2.0 * t * M[:, k - 1] - M[:, k - 2]
    return M

def cheb_integral_weights(K):
    w = np.zeros(K + 1)
    w[0] = 1.0
    for k in range(2, K + 1, 2):
        w[k] = 1.0 / (1.0 - k * k)
    return w

def cheb_optimize(h_init, K, maxiter):
    n = len(h_init)
    x = np.linspace(0, 2, n)
    M = cheb_basis_matrix(x, K)
    c0 = np.linalg.lstsq(M, h_init, rcond=None)[0]
    w_int = cheb_integral_weights(K)
    def obj(c):
        hv = M @ c
        hv = np.clip(hv, 0, 1)
        return topk_obj(hv, 3, n)
    def jac(c):
        hv = M @ c
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        k = 3
        idx = np.argpartition(corr, -k)[-k:]
        shifts = (idx - (n - 1)).astype(int)
        g = shift_grad(hv, shifts, n)
        return M.T @ g
    res = minimize(obj, c0, jac=jac, method='SLSQP',
                  constraints=[{'type': 'eq', 'fun': lambda c: float(w_int @ c - 1.0)}],
                  options={'maxiter': maxiter, 'ftol': 1e-12})
    hv = np.clip(M @ res.x, 0, 1)
    return project_feasible(hv, n)

def multi_subgrad(h, n_steps, top_k, alpha0, decay):
    n = len(h)
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        shifts = (idx - (n - 1)).astype(int)
        grad = shift_grad(h, shifts, n)
        h = project_feasible(h - alpha * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        alpha *= decay
    return best, best_c5

    for i in range(n // 3):

        c5d = compute_c5(hd)
        if c5d < best_c5:
            best_c5, best_h = c5d, hd
        elif c5h < best_c5 * 0.999:
            best_c5, best_h, best_n = c5h, hh, nr

    # Phase 6: Intensive local search on worst shifts
    if time.time() - t0 < budget_s * 0.92:
        for _ in range(8):
            if time.time() - t0 > budget_s * 0.92:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top_idx = np.argpartition(corr, -5)[-5:]
            shifts = (top_idx - (best_n - 1)).astype(int)
            grad = shift_grad(best_h, shifts, best_n)
            for alpha in [0.005, 0.002, 0.001]:
                hc = project_feasible(best_h - alpha * grad, best_n)
                cc = compute_c5(hc)
                if cc < best_c5:
                    best_c5, best_h = cc, hc
                    break

    # Phase 7: Final polish
    if time.time() - t0 < budget_s * 0.97:
        hf = slsqp_opt(best_h, topk_obj, (2, best_n), 500, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hf2 = slsqp_opt(best_h, softmax_obj, (5000, best_n), 400, 1e-15)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2
        # Micro subgrad
        hs, cs = multi_subgrad(best_h, 80, 2, 0.002, 0.998)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        hf3 = slsqp_opt(best_h, topk_obj, (2, best_n), 300, 1e-15)
        cf3 = compute_c5(hf3)
        if cf3 < best_c5:
            best_c5, best_h = cf3, hf3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
