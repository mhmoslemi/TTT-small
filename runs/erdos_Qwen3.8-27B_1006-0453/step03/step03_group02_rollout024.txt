```python
from scipy.optimize import linprog

def solve_lp(h_fixed_g, N, target):
    """Solve: min t s.t. corr_k <= t for all k, sum(h)=target, 0<=h<=1
    where g = 1-h is fixed, so corr_k = sum_n h[n+k]*g[n]"""
    # Variables: [h0, h1, ..., h_{N-1}, t]
    # Objective: minimize t -> c = [0,...,0,1]
    c = np.zeros(N + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_n h[n+k]*g[n] - t <= 0
    center = N - 1
    n_constraints = 2 * N - 1
    A_ub = np.zeros((n_constraints, N + 1))
    b_ub = np.zeros(n_constraints)
    
    g = 1.0 - h_fixed_g  # This is the fixed (1-h)
    
    for k in range(2 * N - 1):
        L = k - center
        if L >= 0:
            # corr[k] = sum_{n=0}^{N-1-L} h[n+L]*g[n]
            for n in range(N - L):
                A_ub[k, n+L] = g[n]
            A_ub[k, N] = -1.0  # -t
        else:
            M = -L
            # corr[k] = sum_{n=M}^{N-1} h[n-M]*g[n]
            for n in range(M, N):
                A_ub[k, n-M] = g[n]
            A_ub[k, N] = -1.0
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * N + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:N], res.x[N]
    return None, None
```

```python
    # Phase 4: Gradient polish at best resolution
    Nb = len(best_h)
    if time.time() < deadline - 50:
        dxb = 2.0 / Nb
        targetb = Nb / 2.0
        d4 = deadline - 15
        h4, c4 = gradient_polish(best_h.copy(), Nb, targetb, dxb, d4, n_iters=500)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()

    # Phase 5: Final LP polish from gradient result
    if time.time() < deadline - 10:
        Nb = len(best_h)
        dxb = 2.0 / Nb
        targetb = Nb / 2.0
        h5, c5 = alternating_lp(best_h.copy(), Nb, targetb, dxb, deadline - 3, max_iter=500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5.copy()

    # Verify and return
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
