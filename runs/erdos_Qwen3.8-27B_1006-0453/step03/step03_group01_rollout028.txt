    h1, c1 = optimize_momentum(best_h.copy(), N, target, p1_dead,
                               2.0, 120.0, 35, 3, 0.035, 0.85, 0.92)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    # Phase 2: Sharpen with high tau (20% time)
    p2_dead = t0 + budget_s * 0.60
    h2, c2 = optimize_momentum(best_h.copy(), N, target, p2_dead,
                               120.0, 800.0, 4, 2, 0.012, 0.88, 0.94)
    if c2 < best_c5:
        best_c5, best_h = c2, h2
    # Phase 3: Multi-scale refinement (15% time)
    if time.time() < deadline - 80:
        N2 = N + 190
        target2 = N2 / 2.0
        x1 = np.linspace(0, 2, N, endpoint=False)
        x2 = np.linspace(0, 2, N2, endpoint=False)
        h_r = np.interp(x2, x1, best_h)
        h_r = project_sum(h_r, target2)
        p3_dead = t0 + budget_s * 0.75
        hr, cr = optimize_momentum(h_r, N2, target2, p3_dead,
                                   40.0, 400.0, 12, 3, 0.02, 0.87, 0.93)
        if cr < best_c5:
            best_c5 = cr
            best_h = np.interp(x1, x2, hr)
            best_h = project_sum(best_h, target)
            best_c5 = compute_c5(best_h, dx)
    # Phase 4: Final sharpen (5% time)
    p4_dead = deadline - 30
    if time.time() < p4_dead:
        Nf = len(best_h)
        tf = Nf / 2.0
        hf, cf = optimize_momentum(best_h.copy(), Nf, tf, p4_dead,
                                   300.0, 1500.0, 3, 2, 0.004, 0.92, 0.96)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    # Phase 5: Snap + gradient local search (remaining time)
    if time.time() < deadline - 5:
        Nb = len(best_h)
        tb = Nb / 2.0
        best_h, best_c5 = snap_and_refine(best_h, Nb, tb, deadline)
        best_h, best_c5 = gradient_local_search(best_h, Nb, tb, deadline - 1)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
