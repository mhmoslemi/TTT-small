    # Phase 3: Multi-restart at multiple resolutions (35%)
    tl = t0 + budget_s * 0.57
    n_resolutions = sorted(set([best_n, 76, 80, 82, 74, 84, 86, 72]))
    for n_try in n_resolutions:
        if time.time() > tl:
            break
        if n_try < 40 or n_try % 2 != 0:
            continue
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_feasible(h_base, n_try)
        n_trials = 10 if n_try == best_n else 6
        for trial in range(n_trials):
            if time.time() > tl:
                break
            if trial < 3:
                scale = 0.02 + 0.04 * trial
            else:
                scale = 0.05 + 0.08 * np.random.rand()
            h_pert = h_base + np.random.randn(n_try) * scale
            h_pert = project_feasible(h_pert, n_try)
            h_opt, c5_opt = coord_sweep(h_pert, n_sweeps=8, n_evals=80)
            if c5_opt < best_c5:
                best_c5, best_h, best_n = c5_opt, h_opt, n_try
                h_base = h_opt.copy()

    # Phase 4: Simulated annealing with coord descent (23%)
    tl = t0 + budget_s * 0.80
    T_sa = 0.0005
    T_min = 1e-7
    cooling = 0.997
    h_sa = best_h.copy()
    c5_sa = best_c5
    n_max_iter = max(2000, int((tl - time.time()) / 0.05))
    for it in range(n_max_iter):
        if time.time() > tl:
            break
        n_move = int(np.random.randint(2, 6))
        idx = np.random.choice(best_n, n_move, replace=False)
        h_trial = h_sa.copy()
        h_trial[idx] = np.random.rand(n_move)
        h_trial = project_feasible(h_trial, best_n)
        c5_trial, _ = compute_c5(h_trial)
        delta = c5_trial - c5_sa
        if delta < 0 or (T_sa > T_min and np.random.rand() < np.exp(-delta / T_sa)):
            h_sa = h_trial
            c5_sa = c5_trial
            if it % 50 == 0:
                h_sa, c5_sa = coord_sweep(h_sa, n_sweeps=3, n_evals=60)
            if c5_sa < best_c5:
                best_c5, best_h = c5_sa, h_sa.copy()
        T_sa *= cooling

    # Phase 5: SLSQP polish (8%)
    tl = t0 + budget_s * 0.88
    if time.time() < tl:
        h_sl = slsqp_polish(best_h, T=80000, maxiter=800)
        c5_sl, _ = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5, best_h = c5_sl, h_sl
        h_cd2, c5_cd2 = coord_sweep(best_h, n_sweeps=20, n_evals=120)
        if c5_cd2 < best_c5:
            best_c5, best_h = c5_cd2, h_cd2

    # Phase 6: Final polish (12%)
    tl = t0 + budget_s * 0.97
    for _ in range(5):
        if time.time() > tl:
            break
        h_cd3, c5_cd3 = coord_sweep(best_h, n_sweeps=15, n_evals=150)
        if c5_cd3 < best_c5:
            best_c5, best_h = c5_cd3, h_cd3
        else:
            break
    # One more SLSQP at very high T
    if time.time() < tl:
        h_sl2 = slsqp_polish(best_h, T=200000, maxiter=400)
        c5_sl2, _ = compute_c5(h_sl2)
        if c5_sl2 < best_c5:
            best_c5, best_h = c5_sl2, h_sl2
            h_cd4, c5_cd4 = coord_sweep(best_h, n_sweeps=10, n_evals=100)
            if c5_cd4 < best_c5:
                best_c5, best_h = c5_cd4, h_cd4

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
