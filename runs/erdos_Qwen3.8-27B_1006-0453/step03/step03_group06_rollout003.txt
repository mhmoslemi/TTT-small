    spread = T * (m - np.min(vals))
    if spread > 50:
        thresh = m - 50.0 / T
        vals = vals[vals >= thresh]
        if len(vals) == 0:
            vals = np.array([m])
    m2 = np.max(vals)
    val = m2 + np.log(np.sum(np.exp(T * (vals - m2)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            h[mask] += room[mask] * (diff / room[mask].sum())
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] -= h[mask] * (-diff / h[mask].sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, T, k, maxiter=1000, ftol=1e-15):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_topk_obj, h0, args=(T, k),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def t_sweep(h0, n, configs):
    best = h0.copy()
    best_c5 = compute_c5(best)
    for T, k, mi in configs:
        hc = slsqp_opt(best, n, T, k, mi, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best = hc
    return best, best_c5

def coord_search(h, n, iterations=400, t_end=0.99):
    t0_global = time.time()
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(iterations):
        if (time.time() - t0_global) > 120:
            break
        improved = False
        for _ in range(8):
            j = np.random.randint(0, n)
            for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.04, -0.04]:
                if best[j] + delta < -0.01 or best[j] + delta > 1.01:
                    continue
                h_try = best.copy()
                h_try[j] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved and it > 250:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: T-sweep at current resolution
    configs1 = [(50, 8, 500), (200, 5, 700), (800, 4, 800),
                (2000, 3, 1000), (8000, 3, 1200), (30000, 2, 1500)]
    if elapsed() < budget_s * 0.2:
        best_h, best_c5 = t_sweep(best_h, n, configs1)

    # Phase 2: Diverse restarts at same resolution
    for trial in range(25):
        if elapsed() > budget_s * 0.45:
            break
        step_mag = 0.06 * (1.0 - trial / 25.0) + 0.01
        r = np.random.rand()
        if r < 0.5:
            h_pert = best_h + step_mag * np.random.randn(n)
        elif r < 0.8:
            h_pert = best_h.copy()
            blen = np.random.randint(5, n // 3 + 1)
            start = np.random.randint(0, n - blen)
            h_pert[start:start+blen] += step_mag * 2 * (np.random.rand(blen) < 0.5) - step_mag
        else:
            h_pert = best_h.copy()
            i1 = np.random.randint(0, n)
            i2 = np.random.randint(0, n)
            d = step_mag * 0.7
            h_pert[i1] += d
            h_pert[i2] -= d
        h_pert = project_feasible(h_pert, n)
        for T, k, mi in [(300, 4, 600), (1500, 3, 800), (5000, 2, 1000)]:
            hc = slsqp_opt(h_pert, n, T, k, mi, 1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Interpolate to higher resolutions
    for n_new in [200, 240]:
        if elapsed() > budget_s * 0.72:
            break
        x_old = np.linspace(0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2.0, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        best_hn = h_interp.copy()
        best_c5n = compute_c5(best_hn)
        for trial in range(4):
            if elapsed() > budget_s * 0.78:
                break
            h0 = best_hn + np.random.uniform(-0.015, 0.015, n_new)
            h0 = project_feasible(h0, n_new)
            configs3 = [(100, 6, 500), (500, 4, 700), (2000, 3, 900), (10000, 2, 1200)]
            hc, c5v = t_sweep(h0, n_new, configs3)
            if c5v < best_c5n:
                best_c5n = c5v
                best_hn = hc
        if best_c5n < best_c5:
            best_c5 = best_c5n
            best_h = best_hn

    # Phase 4: Coordinate search for fine-tuning
    if elapsed() < budget_s * 0.90:
        best_h, best_c5 = coord_search(best_h, len(best_h), 600)

    # Phase 5: Final ultra-high-T polish
    n_cur = len(best_h)
    for T in [50000, 200000]:
        if elapsed() > budget_s * 0.97:
            break
        try:
            hc = slsqp_opt(best_h, n_cur, T, 2, 2000, 1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    # Top-k polish
    for k in [3, 5]:
        if elapsed() > budget_s * 0.985:
            break
        try:
            hc = slsqp_opt(best_h, n_cur, 5000, k, 1500, 1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
