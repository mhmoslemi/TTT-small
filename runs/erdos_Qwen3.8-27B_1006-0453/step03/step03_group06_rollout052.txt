    try:
        res = minimize(topk_smoothed, h0, args=((T, k),),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def coord_descent_polish(h, n, n_steps=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    for step in range(n_steps):
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        top_idx = np.argsort(corr)[-5:]
        for j in range(n):
            for delta in [0.08, -0.08, 0.04, -0.04, 0.02, -0.02]:
                h_try = best.copy()
                h_try[j] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        ih = globals().get('initial_h_values', None)
        if ih is not None:
            best_h = np.array(ih, dtype=float).copy()
        else:
            best_h = np.zeros(160)
            best_h[:80] = 1.0
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish with moderate T
    for T, k in [(100, None), (300, 5), (500, 3)]:
        if elapsed() > budget_s * 0.08:
            break
        hc = slsqp_pass(best_h, T, k, 250, 1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Adaptive T ascent with top-k focus
    t_schedule = [(500, 5), (1000, 3), (2000, 3), (5000, 2), (10000, 2), (20000, 1), (50000, 1), (100000, 1)]
    for T, k in t_schedule:
        if elapsed() > budget_s * 0.35:
            break
        hc = slsqp_pass(best_h, T, k, 400, 1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # Also try from a slightly perturbed version
        if elapsed() > budget_s * 0.33:
            break
        h_pert = best_h + np.random.uniform(-0.02, 0.02, n)
        h_pert = project_feasible(h_pert, n)
        hc2 = slsqp_pass(h_pert, T, k, 300, 1e-13)
        c5v2 = compute_c5(hc2)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = hc2

    # Phase 3: Basin hopping with SLSQP polish
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = 50
    for it in range(n_bh):
        if elapsed() > budget_s * 0.60:
            break
        step = 0.06 * (1.0 - 2.0 * it / n_bh) + 0.01
        r = np.random.rand()
        if r < 0.35:
            h_pert = bh_h + step * np.random.randn(n)
        elif r < 0.65:
            blen = np.random.randint(5, max(6, n // 3))
            start = np.random.randint(0, max(1, n - blen))
            h_pert = bh_h.copy()
            h_pert[start:start+blen] += step * 1.5 * (2.0*np.random.rand(blen) - 1.0)
        else:
            h_pert = bh_h.copy()
            i1 = np.random.randint(0, n)
            i2 = np.random.randint(0, n)
            sw = step * 0.7
            h_pert[i1] += sw
            h_pert[i2] -= sw
            h_pert[(i1+7)%n] += sw * 0.5
            h_pert[(i2+7)%n] -= sw * 0.5
        h_pert = project_feasible(h_pert, n)
        k_kick = [None, 3, 5, 2][np.random.randint(4)]
        hc = slsqp_pass(h_pert, 3000, k_kick, 200, 1e-12)
        c5v = compute_c5(hc)
        temp = 0.004 * (1.0 - it / n_bh) + 0.0002
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-6)):
            bh_c5 = c5v
            bh_h = hc
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: Multi-resolution exploration
    for n_try in [120, 140, 180, 200, 240]:
        if elapsed() > budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_base = np.interp(x_new, x_old, best_h)
        for trial in range(2):
            if elapsed() > budget_s * 0.84:
                break
            h0 = h_base + np.random.uniform(-0.03, 0.03, n_try)
            h0 = project_feasible(h0, n_try)
            for T, k in [(500, 3), (5000, 1)]:
                if elapsed() > budget_s * 0.85:
                    break
                hc = slsqp_pass(h0, T, k, 300, 1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    # Interpolate back to original n
                    x_back = np.linspace(0, 2, n, endpoint=False)
                    best_h = project_feasible(np.interp(x_back, x_new, hc), n)
                    best_c5 = compute_c5(best_h)

    # Phase 5: Coordinate descent polish
    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = coord_descent_polish(best_h, n, n_steps=80)
        n = len(best_h)

    # Phase 6: Final high-T SLSQP polish
    for T, k in [(20000, 1), (50000, 1), (100000, None)]:
        if elapsed() > budget_s * 0.99:
            break
        hc = slsqp_pass(best_h, T, k, 600, 1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coordinate descent
    if elapsed() < budget_s * 0.995:
        best_h, best_c5 = coord_descent_polish(best_h, n, n_steps=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    # Verify
    assert best_h.min() >= -1e-9 and best_h.max() <= 1.0 + 1e-9
    assert abs(best_h.sum() - len(best_h)/2.0) < 0.1
    return best_h, best_c5, len(best_h)
