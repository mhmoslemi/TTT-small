import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 1e-15
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_topk(x, T, K):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    if K < len(corr):
        # Focus on top K
        kth_val = np.partition(corr, len(corr) - K)[len(corr) - K]
        top = corr[corr >= kth_val]
    else:
        top = corr
    m = np.max(top)
    shifted = top - m
    # Numerically stable log-sum-exp
    max_s = np.max(shifted)
    val = m + max_s + np.log(np.sum(np.exp(shifted - max_s))) / T
    return val

def slsqp_opt(h0, T, K, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(lambda x: obj_topk(x, T, K), h0,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return make_feasible(h0, n)

    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    kth_val = np.partition(corr, len(corr) - K)[len(corr) - K]
    worst_mask = corr >= kth_val
    worst_indices = np.where(worst_mask)[0]
    
    grad = np.zeros(n)
    for k_idx in worst_indices:
        shift = k_idx - (n - 1)
        if shift >= 0:
            idx1 = np.arange(0, n - shift)
            grad[idx1] += (1.0 - h[idx1 + shift])
            idx2 = np.arange(shift, n)
            grad[idx2] -= h[idx2 - shift]

            s = -shift
            idx1 = np.arange(s, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    else:
        mask = h > 1e-15
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_topk(x, T, K):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    if K < len(corr):
        thresh = np.partition(corr, len(corr) - K)[len(corr) - K]
        top = corr[corr >= thresh]
    else:
        top = corr
    m = np.max(top)
    if m < 1e-15:
        return 0.0
    shifted = top - m
    ms = np.max(shifted)
    return m + ms + np.log(np.sum(np.exp(shifted - ms))) / T

def slsqp_opt(h0, T, K, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(lambda x: obj_topk(x, T, K), h0,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return make_feasible(h0, n)

def gradient_step(h, K, step_size):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    thresh = np.partition(corr, len(corr) - K)[len(corr) - K]
    worst = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for k_idx in worst:
        s = k_idx - (n - 1)
        if s >= 0:
            i1 = np.arange(0, n - s)
            grad[i1] += (1.0 - h[i1 + s])
            i2 = np.arange(s, n)
            grad[i2] -= h[i2 - s]
        else:
            ss = -s
            i1 = np.arange(ss, n)
            grad[i1] += (1.0 - h[i1 - ss])
            i2 = np.arange(0, n - ss)
            grad[i2] -= h[i2 + ss]
    grad /= max(1, len(worst))
    h_new = h - step_size * grad
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        best_h = np.zeros(100)
        best_h[0::2] = 1.0
        best_h = make_feasible(best_h, 100)

    best_c5 = compute_c5(best_h)

    # Phase 1: Downsample to multiple n, quick SLSQP
    for n_try in [80, 100, 120]:
        if time.time() - t0 > budget_s * 0.10:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = make_feasible(h_init, n_try)
        for T, K in [(100, 10), (400, 15)]:
            if time.time() - t0 > budget_s * 0.10:
                break
            h_new = slsqp_opt(h_init, T, K, 600)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    n_cur = len(best_h)

    # Phase 2: Thorough SLSQP sweep
    for T in [40, 80, 150, 400, 1200]:
        if time.time() - t0 > budget_s * 0.28:
            break
        for K in [3, 5, 10, 20]:
            if time.time() - t0 > budget_s * 0.28:
                break
            h_new = slsqp_opt(best_h.copy(), T, K, 1200)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 3: Gradient descent on worst shifts
    for step_size in [0.08, 0.04, 0.02, 0.01, 0.005]:
        if time.time() - t0 > budget_s * 0.42:
            break
        for _ in range(400):
            if time.time() - t0 > budget_s * 0.42:
                break
            h_new = gradient_step(best_h, 8, step_size)
            c5v = compute_c5(h_new)
            if c5v < best_c5 - 1e-14:
                best_c5 = c5v
                best_h = h_new
            else:
                h_new2 = gradient_step(best_h, 8, step_size * 0.3)
                c5v2 = compute_c5(h_new2)
                if c5v2 < best_c5 - 1e-14:
                    best_c5 = c5v2
                    best_h = h_new2
                else:
                    break

    # Phase 4: Basin hopping with structured perturbations
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    time_left = budget_s * 0.72 - (time.time() - t0)
    max_iters = max(25, int(time_left / 4.5))

    for it in range(max_iters):
        if time.time() - t0 > budget_s * 0.72:
            break
        ptype = it % 5
        if ptype == 0:
            sigma = 0.06 + 0.04 * (1.0 - it / max(1, max_iters))
            h_pert = bh_h + sigma * np.random.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(3, n_bh // 12)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1 + blen].copy()
            h_pert[s1:s1 + blen] = h_pert[s2:s2 + blen]
            h_pert[s2:s2 + blen] = tmp
        elif ptype == 2:
            h_pert = bh_h.copy()
            corr_v = np.correlate(h_pert, 1 - h_pert, mode='full')
            wk = np.argmax(corr_v)
            shift = wk - (n_bh - 1)
            width = max(6, n_bh // 5)
            lo = max(0, shift - width)
            hi = min(n_bh, shift + width)
            if hi > lo:
                h_pert[lo:hi] += 0.1 * (np.random.rand(hi - lo) - 0.4)
        elif ptype == 3:
            h_pert = bh_h.copy()
            nflip = max(2, n_bh // 8)
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        else:
            h_pert = bh_h + 0.15 * np.random.randn(n_bh)

        h_pert = make_feasible(h_pert, n_bh)
        h_new = slsqp_opt(h_pert, 150, 8, 250)
        c5v = compute_c5(h_new)

        temp = 0.003 * (1.0 - it / max(1, max_iters)) + 0.0003
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-8)):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 5: Multi-resolution exploration
    for n_try in [72, 96, 128, 160]:
        if time.time() - t0 > budget_s * 0.86:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T, K in [(80, 5), (500, 10)]:
            if time.time() - t0 > budget_s * 0.86:
                break
            h_new = slsqp_opt(h_interp, T, K, 900)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 6: Final high-precision polish
    n_final = len(best_h)
    for T in [3000, 15000, 80000]:
        if time.time() - t0 > budget_s * 0.985:
            break
        for K in [3, 5]:
            if time.time() - t0 > budget_s * 0.985:
                break
            h_new = slsqp_opt(best_h.copy(), T, K, 4000)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 7: Final gradient micro-steps
    for step_size in [0.005, 0.002, 0.001, 0.0005]:
        if time.time() - t0 > budget_s * 0.995:
            break
        for _ in range(600):
            if time.time() - t0 > budget_s * 0.995:
                break
            h_new = gradient_step(best_h, 5, step_size)
            c5v = compute_c5(h_new)
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                best_h = h_new
            else:
                break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
