```python
def _smooth_jac(x, T, idx, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr[idx]
    m = np.max(c)
    exp_c = np.clip(T * (c - m), -500, 500)
    e = np.exp(exp_c)
    w = e / e.sum()
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-14:
            continue
        s = idx[ii] - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo >= i_hi:
            continue
        i_arr = np.arange(i_lo, i_hi)
        j_arr = i_arr + s
        grad[i_arr] += wk * (1.0 - x[j_arr])
        grad[j_arr] += wk * (-x[i_arr])
    return grad * dx
```
