def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_hc(x, target):
    """Project x onto {h in [0,1]^n: sum(h) = target}"""
    n = len(x)
    lo, hi = -2.0, 3.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-18:
            break
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            room = 1.0 - h
            for i in np.argsort(-room):
                a = min(d, room[i])
                h[i] += a
                d -= a
                if d < 1e-15:
                    break
        else:
            d = -d
            for i in np.argsort(h):
                a = min(d, h[i])
                h[i] -= a
                d -= a
                if d < 1e-15:
                    break
    return h

def sobj(h, T):
    """Smooth max of correlation, returns (value, gradient)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = T * (corr - mx)
    sm = s.max()
    e = np.exp(s - sm)
    Z = e.sum()
    val = (mx + (sm + np.log(Z)) / T) * dx
    sw = e / Z
    g = np.zeros(n)
    for mi in range(len(corr)):
        w = sw[mi]
        if w < 1e-15:
            continue
        k = mi - (n - 1)
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n-k])
            g[:n-k] += -w * h[k:]
        else:
            kp = -k
            g[:n-kp] += w * (1.0 - h[kp:])
            g[kp:] += -w * h[:n-kp]
    g *= T * dx
    return val, g

def lbfgs_optimize(h0, T, maxit, lam):
    n = len(h0)
    tgt = n * 0.5

    def f(h):
        v, _ = sobj(h, T)
        return v + lam * (h.sum() - tgt) ** 2

        _, gr = sobj(h, T)
        gr = gr + 2.0 * lam * (h.sum() - tgt)
        return gr

    res = minimize(f, h0, jac=g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxit, 'ftol': 1e-25, 'gtol': 1e-14,
                            'maxls': 50})
    return project_hc(res.x, tgt)

What about using the duality of the max? The problem is:
min_h max_k f_k(h) subject to constraints

This is equivalent to:
min_{h, t} t subject to f_k(h) ≤ t ∀k

The Lagrangian is: L(h, t, λ) = t + Σ_k λ_k (f_k(h) - t) + μ(∑h - n/2)
= (1-Σλ)t + Σλ_k f_k(h) + μ(∑h - n/2)

For the dual to be bounded below, we need Σλ = 1 (since t appears with coefficient 1-Σλ).

So the dual problem is:
max_{λ≥0, Σλ=1, μ} min_h [Σλ_k f_k(h) + μ(∑h - n/2)]
= max_{λ≥0, Σλ=1, μ} min_h [Σλ_k corr_k(h) + μ(∑h - n/2)] (times dx)

The inner minimization is:
min_h Σ_k λ_k corr_k(h) + μ(∑h - n/2)
= min_h h^T (Σλ_k A_k) h + (b(λ) + μ)·h - μ·n/2

where A_k is a bilinear form. This is a quadratic minimization (but non-convex in general since the h_i * h_{i+k} terms can have positive coefficients).
