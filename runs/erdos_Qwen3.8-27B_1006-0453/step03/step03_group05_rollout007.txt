Yes! So both the positive and negative shift overlaps use the same autocorr value.

overlap(-s) = (n/2 - ones_prefix[s]) - autocorr[n-1+s]  (for shift -s)

And overlap(0) = n/2 - n/2 = 0 for binary.

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def all_overlaps_binary(h):
    """Compute all 2n-1 overlap values for a binary vector h with sum h = n/2"""
    n = len(h)
    autocorr = np.correlate(h, h, mode='full')
    ones_prefix = np.zeros(n + 1)
    ones_prefix[1:] = np.cumsum(h)
    overlaps = np.zeros(2 * n - 1)
    for s in range(1, n):
        overlaps[n - 1 - s] = ones_prefix[n - s] - autocorr[n - 1 + s]
        overlaps[n - 1 + s] = (n / 2.0 - ones_prefix[s]) - autocorr[n - 1 + s]
    overlaps[n - 1] = 0.0
    return overlaps

def binary_hill_climb(h, n, max_sweeps=50, rng=None):
    """Hill climbing: try all swaps, accept the best improvement each sweep"""
    if rng is None:
        rng = np.random.default_rng(42)
    best = h.copy()
    best_ov = np.max(all_overlaps_binary(best))
    for sweep in range(max_sweeps):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        best_improvement = 0
        best_swap = None
        # For efficiency, compute baseline overlaps once
        base_ov = all_overlaps_binary(best)
        for i in ones:
            for j in zeros:
                test = best.copy()
                test[i] = 0
                test[j] = 1
                test_ov = np.max(all_overlaps_binary(test))
                improvement = best_ov - test_ov
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        best[i] = 0
        best[j] = 1
        best_ov = best_ov - best_improvement
    return best, best_ov * (2.0 / n)
