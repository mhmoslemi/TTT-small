```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_val(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = T * (corr - m)
    mx = np.max(sh)
    return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / T) * dx

def lag_g(h, k):
    n = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        g = np.zeros(n)
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
        return g
    kp = -k
    g = np.zeros(n)
    g[:n - kp] += 1.0 - h[kp:]
    g[kp:] -= h[:n - kp]
    return g

def softmax_grad(h, T, top_m=12):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = T * (corr - m)
    mx = np.max(sh)
    w = np.exp(sh - mx)
    w /= w.sum()
    idx = np.argsort(w)[-top_m:]
    wi = w[idx]
    wi /= wi.sum()
    g = np.zeros(n)
    for j, qi in enumerate(idx):
        g += wi[j] * lag_g(h, qi - (n - 1))
    return g * dx

def slsqp_full(h0, T, mi=800):
    n = len(h0)
    b = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        r = minimize(smooth_val, h0, args=(T,), method='SLSQP', bounds=b,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-15})
        return project_feasible(np.clip(r.x, 0, 1), n)
    except Exception:
        return h0

def slsqp_targeted(h0, top_idx, mi=800):
    n = len(h0)
    dx = 2.0 / n
    b = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    T_local = 50000
    def obj_t(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr[top_idx]
        m = np.max(vals)
        sh = T_local * (vals - m)
        mx = np.max(sh)
        return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / T_local) * dx
    try:
        r = minimize(obj_t, h0, method='SLSQP', bounds=b, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return project_feasible(np.clip(r.x, 0, 1), n)
    except Exception:
        return h0

def subgrad_opt(h0, ni=2000, lr=0.03, mom=0.8, tm=10, T=25000):
    n = len(h0)
    hc = h0.copy()
    bh = h0.copy()
    bc, _ = compute_c5(h0)
    v = np.zeros(n)
    st = 0
    for it in range(ni):
        g = softmax_grad(hc, T, tm)
        v = mom * v - lr * g
        hn = project_feasible(hc + v, n)
        c, _ = compute_c5(hn)
        if c < bc - 1e-13:
            bc, bh = c, hn.copy()
            lr = min(lr * 1.01, 0.2)
            hc = hn
            st = 0
        else:
            st += 1
            lr *= 0.96
            if st > 100:
                break
    return bh, bc

def get_active_lags(h, frac=0.998):
    n = len(h)
    _, corr = compute_c5(h)
    m = np.max(corr)
    thresh = m * frac
    active = np.where(corr >= thresh)[0]
    return active[:20]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Targeted SLSQP on active lags (high T, essentially exact)
    active = get_active_lags(best_h, 0.995)
    ht = slsqp_targeted(best_h, active, mi=1000)
    ct, _ = compute_c5(ht)
    if ct < best_c5:
        best_c5, best_h = ct, ht

    # Full SLSQP with high T
    if time.time() - t0 < budget_s * 0.25:
        ht = slsqp_full(best_h, 200000, mi=800)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 2: Subgradient descent
    if time.time() - t0 < budget_s * 0.45:
        hs, cs = subgrad_opt(best_h, 3000, lr=0.025, mom=0.82, tm=12, T=30000)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 3: Re-target and SLSQP
    if time.time() - t0 < budget_s * 0.58:
        active2 = get_active_lags(best_h, 0.99)
        ht = slsqp_targeted(best_h, active2, mi=1000)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
        ht = slsqp_full(best_h, 500000, mi=800)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 4: Try 1-h symmetry
    if time.time() - t0 < budget_s * 0.66:
        hf = project_feasible(1.0 - best_h, best_n)
        ht = slsqp_full(hf, 100000, mi=600)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
        hs, cs = subgrad_opt(ht, 1200, lr=0.02, tm=10, T=30000)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        ht = slsqp_full(best_h, 300000, mi=600)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 5: Resolution search
    for dn in [2, -2, 4, -4, 6, -6, 8]:
        if time.time() - t0 > budget_s * 0.82:
            break
        nt = best_n + dn
        if nt < 50 or nt % 2 != 0:
            continue
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, nt, endpoint=False)
        hi = project_feasible(np.interp(xn, xo, best_h), nt)
        ht = slsqp_full(hi, 50000, mi=600)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h, best_n = ct, ht, nt
            hs, cs = subgrad_opt(best_h, 800, lr=0.02, tm=10)
            if cs < best_c5:
                best_c5, best_h = cs, hs
            ht = slsqp_full(best_h, 200000, mi=500)
            ct, _ = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 6: Structured perturbation + re-optimize
    if time.time() - t0 < budget_s * 0.92:
        g_avg = np.zeros(best_n)
        cnt = 0
        for idx in get_active_lags(best_h, 0.99)[:12]:
            k = idx - (best_n - 1)
            g_avg += lag_g(best_h, k)
            cnt += 1
        if cnt > 0:
            g_avg /= cnt
        for scale in [0.08, 0.15, 0.25]:
            hp = project_feasible(best_h - scale * g_avg, best_n)
            ho, co = subgrad_opt(hp, 800, lr=0.03, tm=10, T=25000)
            ho = slsqp_targeted(ho, get_active_lags(ho, 0.99), mi=600)
            co, _ = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho
                break

    # Phase 7: Final ultra-fine polish
    if time.time() - t0 < budget_s * 0.97:
        ht = slsqp_targeted(best_h, get_active_lags(best_h, 0.995), mi=1000)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
        hs, cs = subgrad_opt(best_h, 400, lr=0.01, mom=0.85, tm=6, T=50000)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        ht = slsqp_full(best_h, 800000, mi=600)
        ct, _ = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
