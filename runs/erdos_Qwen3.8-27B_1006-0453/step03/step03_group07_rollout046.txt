def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h

        total = room.sum()
        if total > 1e-15:

    else:
        total = h.sum()
        if total > 1e-15:
            h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, T, active_frac=1.0):
    """Smooth approximation to max of correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    # Optional: restrict to active set
    if active_frac < 1.0:

        c_sub = corr[idx]

        c_sub = corr
    c_max = np.max(c_sub)
    return c_max + np.log(np.sum(np.exp(T * (c_sub - c_max))) / len(c_sub)) / T

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        total = h.sum()
        if total > 1e-14:
            h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(T * (c - m))) / len(c)) / T

def slsqp_optimize(h0, T, n, maxiter, ftol):
    def obj(x):
        return smooth_max_obj(np.clip(x, 0, 1), T)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

    """L-BFGS-B with penalty for sum constraint."""
    penalty = 1e6

        h = np.clip(x, 0, 1)
        return smooth_max_obj(h, T) + penalty * (np.sum(h) - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0, 1)] * n, maxiter=maxiter)
    return make_feasible(np.clip(res.x, 0, 1), n)
