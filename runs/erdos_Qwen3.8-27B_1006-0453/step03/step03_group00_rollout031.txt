import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def get_corr_and_coefs(h, i):
    """Compute base correlation (with h[i]=0) and coefficient for h[i]."""
    n = len(h)
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        c = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            if j2 != j1:
                c += -h[j2]
            else:  # k=0
                c += -h[j2]
        coef[k + n - 1] = c
    return corr_base, coef

def coord_optimize(h, n, i):
    """Find optimal h[i] given others fixed."""
    corr_base, coef = get_corr_and_coefs(h, i)
    def f1d(t):
        return np.max(corr_base + t * coef)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def solve_1d_max_linear(base, coef, lo=0.0, hi=1.0):
    """minimize max_k (base[k] + t * coef[k]) for t in [lo, hi]"""
    # The function f(t) = max_k (base[k] + t*coef[k]) is convex in t.
    # Minimum is where subgradient contains 0, or at boundary.
    
    # Find active set at t=lo and t=hi
    f_lo = base + lo * coef
    f_hi = base + hi * coef
    k_lo = np.argmax(f_lo)
    k_hi = np.argmax(f_hi)
    
    if k_lo == k_hi:
        # Same active constraint at both ends
        # Minimum is at the end with smaller value, or where this line is minimized
        c = coef[k_lo]
        if c < 0:
            return hi  # decreasing, min at hi
        elif c > 0:
            return lo  # increasing, min at lo
        else:
            return lo  # constant
    else:
        # Different active constraints. Find crossing.
        # The minimum is where the two constraints cross (if in [lo, hi])
        # base[k_lo] + t*coef[k_lo] = base[k_hi] + t*coef[k_hi]
        # t = (base[k_hi] - base[k_lo]) / (coef[k_lo] - coef[k_hi])
        denom = coef[k_lo] - coef[k_hi]
        if abs(denom) < 1e-15:
            return lo if f_lo[k_lo] <= f_hi[k_hi] else hi
        t_cross = (base[k_hi] - base[k_lo]) / denom
        if lo <= t_cross <= hi:
            # Check this is indeed the minimum
            f_cross = np.max(base + t_cross * coef)
            f_lo_val = np.max(base + lo * coef)
            f_hi_val = np.max(base + hi * coef)
            if f_cross <= f_lo_val and f_cross <= f_hi_val:
                return t_cross
            elif f_lo_val <= f_hi_val:
                return lo
            else:
                return hi
        elif t_cross < lo:
            return lo
        else:
            return hi

        vals = base + mid * coef
        mx = np.max(vals)
        active = np.where(vals >= mx - 1e-14)[0]
        c_min = np.min(coef[active])
        c_max = np.max(coef[active])
        if c_min <= 0 and c_max >= 0:
            return mid  # 0 is in subgradient
        elif c_max < 0:
            hi = mid  # all active coefs negative, f decreasing, go right
        else:  # c_min > 0
            lo = mid  # all active coefs positive, f increasing, go left
    return 0.5 * (lo + hi)

When h[i] changes by δ:
- Terms where n=i: h[i] changes by δ, contribution δ*(1-h[i+lag])
- Terms where n+lag=i, i.e., n=i-lag: (1-h[i]) changes by -δ, contribution h[i-lag]*(-δ) = -δ*h[i-lag]

So Δcorr[k] = δ * [(1-h[i+lag_k]) - h[i-lag_k]] (with boundary conditions)

def update_correlation(corr, h, i, new_val):
    """Update correlation after changing h[i] to new_val."""
    n = len(h)
    old_val = h[i]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return corr
    corr_new = corr.copy()
    for k in range(-(n-1), n):
        c = 0.0
        j1 = i + k  # i+lag

            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += -h[j2]
        corr_new[k + n - 1] += delta * c
    h[i] = new_val
    return corr_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d_convex(base, coef, lo=0.0, hi=1.0):
    """Minimize f(t) = max_k(base[k] + t*coef[k]) for t in [lo, hi]."""
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = base + mid * coef
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        c_act = coef[active]
        if np.min(c_act) <= 0.0 and np.max(c_act) >= 0.0:
            return mid
        elif np.max(c_act) < 0.0:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

                j1 = i + k

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d(base, coef):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = base + mid * coef
        mx = np.max(vals)
        c_act = coef[vals >= mx - 1e-14]
        if np.min(c_act) <= 0.0 and np.max(c_act) >= 0.0:
            return mid
        elif np.max(c_act) < 0.0:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def vec_coef(h, i):
    n = len(h)
    k_range = np.arange(-(n-1), n)
    j1 = i + k_range
    j2 = i - k_range
    coef = np.zeros(2 * n - 1)
    m1 = (j1 >= 0) & (j1 < n)
    m2 = (j2 >= 0) & (j2 < n)
    coef[m1] += 1.0 - h[j1[m1]]
    coef[m2] += -h[j2[m2]]
    return coef

def coord_descent(h, n, t_end, max_sweeps=200, reproject_every=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5 = np.max(corr) * dx

            break
        improved = False
        for i in range(n):
            coef = vec_coef(h, i)
            base = corr - h[i] * coef
            t_opt = solve_1d(base, coef)
            if abs(t_opt - h[i]) > 1e-12:

            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        c5 = np.max(corr) * dx
        if c5 < best_c5:

            best_h = h.copy()
        if not improved:

                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(res.x, n)
