```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def _corr_grad_at(h, idx, n):
    grad = np.zeros(n)
    if idx >= n - 1:
        d = idx - (n - 1)
        if d == 0:
            return 1.0 - 2.0 * h
        grad[d:] += (1.0 - h[:-d])
        grad[:n - d] += (-h[d:])
    else:
        s = (n - 1) - idx
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def smooth_max_obj(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        vals = corr
    m = vals.max()
    val = m + np.log(np.sum(np.exp(T * (vals - m)))) / T
    return val * dx, corr, (idx if topk is not None else None)

def smooth_max_grad(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        idx = np.arange(len(corr))
        vals = corr
    m = vals.max()
    exps = T * (vals - m)
    weights = np.exp(exps - exps.max())
    weights /= weights.sum()
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i] * T * dx
        if w < 1e-14:
            continue
        grad += w * _corr_grad_at(h, idx[i], n)
    return grad

def slsqp_analytic(h0, T, n, topk, maxiter, ftol):
    def obj(h):
        v, _, _ = smooth_max_obj(h, T, topk)
        return v
    def obj_grad(h):
        return smooth_max_grad(h, T, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def subgradient_phase(h, n, n_iters, topk, t0, t1, step0, step1):
    best_h = h.copy()
    best_c5 = compute_c5_full(best_h)[0]
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        T = t0 + (t1 - t0) * frac
        step = step0 + (step1 - step0) * frac
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx = np.argpartition(c, -topk)[-topk:]
        vals = c[idx]
        m = vals.max()
        weights = np.exp(T * (vals - m))
        weights /= weights.sum()
        grad = np.zeros(n)
        for i in range(topk):
            if weights[i] < 1e-12:
                continue
            grad += weights[i] * _corr_grad_at(best_h, idx[i], n)
        h_new = best_h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_full(h_new)[0]
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def exact_coord_descent(h, n, max_pass=300):
    best_h = h.copy()
    best_c5 = compute_c5_full(best_h)[0]
    for _ in range(max_pass):
        improved = False
        c5, corr = compute_c5_full(best_h)
        top3 = np.argsort(corr)[-3:]
        for k in top3:
            g = _corr_grad_at(best_h, int(k), n)
            for j in range(n):
                if abs(g[j]) < 1e-10:
                    continue
                direction = -np.sign(g[j])
                for mag in [0.15, 0.05, 0.02, 0.005]:
                    h_try = best_h.copy()
                    h_try[j] += direction * mag
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5_full(h_try)[0]
                    if c5_try < c5 - 1e-14:
                        best_h = h_try
                        c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, compute_c5_full(best_h)[0]

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5, _ = compute_c5_full(best_h)

    # Phase 1: Subgradient warmup
    if time.time() - t0 < budget_s * 0.15:
        best_h, c5 = subgradient_phase(best_h, n, 4000, 5, 50, 500, 0.4, 0.02)
        if c5 < best_c5:
            best_c5, best_h = c5, best_h

    # Phase 2: SLSQP with analytical gradient, multiple T/topk
    t_list = [50, 200, 800, 3000, 10000]
    k_list = [None, 3, 5, 8]
    for T in t_list:
        if time.time() - t0 > budget_s * 0.40:
            break
        for k in k_list:
            if time.time() - t0 > budget_s * 0.40:
                break
            try:
                hc = slsqp_analytic(best_h, T, n, k, 1200, 1e-15)
                c5v = compute_c5_full(hc)[0]
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    # Phase 3: Multi-resolution
    for n_try in [120, 140, 180, 200]:
        if time.time() - t0 > budget_s * 0.72:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [200, 1000, 5000]:
            if time.time() - t0 > budget_s * 0.74:

            try:

                c5v = compute_c5_full(hc)[0]
                if c5v < best_c5:
                    best_c5, best_h, n = c5v, hc, n_try
            except Exception:

        # Also from slight perturbation
        if time.time() - t0 < budget_s * 0.74:

            except Exception:

    if time.time() - t0 < budget_s * 0.88:
```
