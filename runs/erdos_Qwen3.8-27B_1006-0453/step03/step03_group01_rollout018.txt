def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binarize(h, N):
    """Convert to binary, maintaining sum = N/2."""
    idx = np.argsort(h)[::-1]
    b = np.zeros(N)
    b[idx[:N//2]] = 1.0
    return b

def gradient_step(h, N, tau, topK, deadline_check=None):
    """One gradient step using hybrid softmax + topK weighting."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    center = N - 1
    nc = len(corr)
    sc = tau * corr - tau * corr.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = 0.5
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.001
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
