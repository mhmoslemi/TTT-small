def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-11:
            continue
        grad[:n - k] += wk * (1.0 - h[k:])
        if k > 0:
            grad[k:] -= wk * h[:n - k]
    for k in range(1, n):
        wk = w[n - 1 - k]
        if wk < 1e-11:
            continue
        grad[k:] += wk * (1.0 - h[:n - k])
        grad[:n - k] -= wk * h[k:]
    return grad * dx

    def obj(x):
        return smooth_obj(x, T)
    def jac(x):
        return smooth_grad(x, T)

    bnds = [(0.0, 1.0)] * n

    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter):
    n = len(h)
    dx = 2.0 / n

    t0 = float(np.max(corr)) * dx + 0.01

        return x[-1]

        return x[:-1].sum() - n / 2.0
    def ineq_c5(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full')
        return t - np.max(c) * dx
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': eq_sum}, {'type': 'ineq', 'fun': ineq_c5}]
    try:
