import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / len(h), 0.0, 1.0)
    return h

def f1d_eval(hi_new, corr_lin, coef, c0_base, dx):
    vals = corr_lin + hi_new * coef
    k0 = len(coef) // 2
    vals[k0] = c0_base + hi_new - hi_new * hi_new
    return np.max(vals) * dx

def fast_coord_sweep(h, t_end, max_sweeps=5000):
    n = len(h)
    dx = 2.0 / n
    k0 = n - 1
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            hi = h[i]
            coef = np.zeros(2 * n - 1)
            l1s = max(-i, -(n - 1))
            l1e = min(n - 1 - i, n - 1)
            if l1s <= l1e:
                ls = np.arange(l1s, l1e + 1)
                coef[ls + k0] += (1.0 - h[i + ls])
            l2s = max(i - (n - 1), -(n - 1))
            l2e = min(i, n - 1)
            if l2s <= l2e:
                ls = np.arange(l2s, l2e + 1)
                coef[ls + k0] -= h[i - ls]
            c0_base = corr[k0] - hi + hi * hi
            corr_lin = corr - hi * coef
            res = minimize_scalar(f1d_eval, bounds=(0.0, 1.0), method='bounded',
                                  args=(corr_lin, coef, c0_base, dx),
                                  options={'xatol': 1e-12, 'maxiter': 40})
            hi_new = float(res.x)
            if abs(hi_new - hi) > 1e-14:
                h[i] = hi_new
                d = hi_new - hi
                corr = corr + d * coef
                corr[k0] = c0_base + hi_new - hi_new * hi_new
                improved = True
        h = project_sum(h, n / 2.0)
        corr = compute_corr(h)
        c5v = np.max(corr) * dx
        if c5v < best_c5:
            best_c5 = c5v
        elif not improved:
            break
    return h, best_c5

def slsqp_polish(h, T, n, maxiter=600):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def con_f(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_f}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(np.clip(res.x, 0, 1), n / 2.0)

def subgrad_step(h, corr, n, alpha):
    dx = 2.0 / n
    k0 = n - 1
    mval = np.max(corr)
    active = np.where(corr >= mval - 1e-10)[0]
    grad = np.zeros(n)
    for am in active:
        l = am - k0
        for i in range(n):
            c = 0.0
            if 0 <= i + l < n:
                c += (1.0 - h[i + l])
            if 0 <= i - l < n:
                c -= h[i - l]
            if l == 0:
                c = 1.0 - 2.0 * h[i]
            grad[i] += c
    grad /= len(active)
    h = h - alpha * grad
    return project_sum(h, n / 2.0)
