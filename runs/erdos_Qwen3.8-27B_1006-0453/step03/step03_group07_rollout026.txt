import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    else:
        if s > 1e-14:
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def sum_diff(x):
    return float(np.sum(x) - len(x) / 2.0)

def sum_jac(x):
    return np.ones(len(x))

def lse_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def refine_slsqp(h0, T, maxiter=500):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_diff, 'jac': sum_jac}]
    try:
        res = minimize(lse_obj, h0, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def refine_lbfgs(h0, T, maxiter=300):
    """L-BFGS-B by eliminating sum constraint: optimize first n-1 vars, set last."""
    n = len(h0)
    def reduced_obj(u):
        x = np.concatenate([u, np.array([max(0.0, min(1.0, n/2.0 - u.sum()))])])
        x = np.clip(x, 0, 1)
        return lse_obj(x, T)
    x0 = h0[:n-1].copy()
    bnds = [(0.0, 1.0)] * (n-1)
    try:
        res = minimize(reduced_obj, x0, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        x = np.concatenate([res.x, np.array([max(0.0, min(1.0, n/2.0 - res.x.sum()))])])
        return make_feasible(x, n)
    except Exception:
        return h0

def binary_sa(h0, n, iters=400, T_start=0.02, T_end=0.0004):
    target = n // 2
    h = np.round(h0)
    n1 = int(h.sum())
    if n1 != target:
        if n1 < target:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, target - n1, replace=False)] = 1.0
        else:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - target, replace=False)] = 0.0
    c5_cur = compute_c5(h)
    for it in range(iters):
        frac = it / max(1, iters)
        T = T_start * (T_end / T_start) ** frac
        nf = 1 + int(2 * np.random.rand())
        idx = np.random.choice(n, nf, replace=False)
        h_new = h.copy()
        h_new[idx] = 1.0 - h_new[idx]
        diff = target - int(h_new.sum())
        if diff > 0:
            zeros = np.where(h_new == 0)[0]
            if len(zeros) >= diff:
                h_new[np.random.choice(zeros, diff, replace=False)] = 1.0
        elif diff < 0:
            ones = np.where(h_new == 1)[0]
            if len(ones) >= -diff:
                h_new[np.random.choice(ones, -diff, replace=False)] = 0.0
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur or np.random.rand() < np.exp(-(c5_new - c5_cur) / max(T, 1e-12)):
            h = h_new
            c5_cur = c5_new
    return h.copy(), c5_cur

def make_periodic_pattern(n, p, fill_ratio, offset):
    """Create a periodic binary pattern with given period and fill ratio."""
    base = np.zeros(p)
    n_fill = max(1, int(p * fill_ratio))
    base[offset:offset + n_fill] = 1.0
    full = np.tile(base, n // p)
    return full[:n]

def softening(h, n, n_soft=5, strength=0.3):
    """Replace some extreme values with intermediate to reduce peak."""
    h_s = h.copy()
    worst_lag = 0
    corr = np.correlate(h_s, 1.0 - h_s, mode='full')
    # Find positions contributing most to worst lag
    worst_k = int(np.argmax(corr))
    n1 = len(h_s)
    # For lag k, contributions are at positions where both are in range
    k_shift = worst_k - (n1 - 1)
    positions = []
    for j in range(max(0, -k_shift), min(n1, n1 - k_shift)):
        i = j + k_shift
        positions.append((j, i))
    # Soften positions with highest contribution
    contributions = [(abs(h_s[j]) * (1 - h_s[i]) + (1-h_s[j]) * h_s[i], j, i) for j, i in positions]
    contributions.sort(reverse=True)
    for c, j, i in contributions[:n_soft]:
        if h_s[j] > 0.8:
            h_s[j] -= strength * (1.0 - h_s[j])
        elif h_s[j] < 0.2:
            h_s[j] += strength * h_s[j]
        if h_s[i] > 0.8:
            h_s[i] -= strength * (1.0 - h_s[i])
        elif h_s[i] < 0.2:
            h_s[i] += strength * h_s[i]
    return make_feasible(h_s, n)

def ga_crossover(a, b, n):
    mask = np.random.rand(n) < 0.5
    return mask * a + (1.0 - mask) * b

def ga_mutate(h, n, rate=0.1, strength=0.15):
    h_m = h.copy()
    n_mut = max(1, int(n * rate))
    idx = np.random.choice(n, n_mut, replace=False)
    h_m[idx] += strength * np.random.randn(n_mut)
    return make_feasible(h_m, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        n_base = len(ihv)
        best_h = np.array(ihv, dtype=float).copy()
    else:
        n_base = 200
        best_h = make_periodic_pattern(n_base, 4, 0.5, 0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary SA on diverse patterns (12% time)
    t_lim1 = budget_s * 0.12
    candidates = []
    n_targets = []
    for nt in [120, 160, 200, 280, 360]:
        if nt <= 800:
            n_targets.append(nt)
    for nt in n_targets:
        if elapsed() > t_lim1:
            break
        patterns = []
        for p in [4, 5, 6, 7, 8, 10, 12, 14, 15, 20, 24, 30]:
            if nt % p == 0:
                for off in range(min(p, 4)):
                    patterns.append(make_periodic_pattern(nt, p, 0.5, off))
        for _ in range(4):
            pat = np.zeros(nt)
            pat[np.random.choice(nt, nt // 2, replace=False)] = 1.0
            patterns.append(pat)
        # Limit patterns per n
        for pat in patterns[:10]:
            if elapsed() > t_lim1:

            h_b, c5_b = binary_sa(pat, nt, iters=250, T_start=0.015, T_end=0.0003)
            candidates.append((c5_b, h_b, nt))
    candidates.sort(key=lambda x: x[0])
    for c5_c, h_c, n_c in candidates[:3]:
        if c5_c < best_c5:
            if n_c == n_base:
                best_h = h_c.copy()
                best_c5 = c5_c
            else:
                x_old = np.linspace(0, 2, n_c, endpoint=False)
                x_new = np.linspace(0, 2, n_base, endpoint=False)
                best_h = make_feasible(np.interp(x_new, x_old, h_c), n_base)
                best_c5 = compute_c5(best_h)

    # Phase 2: Intensive refinement with increasing T (30% time)
    t_lim2 = budget_s * 0.42
    n_cur = len(best_h)
    h_work = best_h.copy()
    for T in [20, 80, 300, 1000, 4000, 15000, 60000]:

            break
        h_work = refine_slsqp(h_work, T, maxiter=600)
        c5_v = compute_c5(h_work)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h_work.copy()
        # Also try L-BFGS-B variant

        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:

    # Add best binary candidate

        if n_b == n_cur:

    pop_c5 = [compute_c5(p) for p in population]

            child = parent.copy()

    # Phase 4: Final polishing (remaining time)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Coordinate descent micro-adjustments
    if elapsed() < budget_s * 0.97:

            if elapsed() > budget_s * 0.97:

            for j in np.random.permutation(n_final):

    best_c5 = compute_c5(best_h)
