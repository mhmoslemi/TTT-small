import numpy as np
from scipy.optimize import minimize
import time

_opt_state = {'T': 1000.0, 'active_idx': None}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T, active_idx=None):
    n = len(h)
    dx = 2.0 / n
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    if active_idx is not None:
        corr = corr_full[active_idx]
        idxs = active_idx
    else:
        corr = corr_full
        idxs = np.arange(len(corr_full))
    m = np.max(corr)
    shifted = T * (corr - m)
    max_sh = np.max(shifted)
    exps = np.exp(shifted - max_sh)
    log_sum_exp = m + max_sh / T + np.log(np.sum(exps)) / T
    obj = log_sum_exp * dx
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for w, idx in zip(weights, idxs):
        if w < 1e-14:
            continue
        k = int(idx) - (n - 1)
        if k >= 0:
            tmp = np.zeros(n)
            tmp[:n-k] = (1.0 - h[k:])
            tmp[k:] = -h[:n-k]
        else:
            kp = -k
            tmp = np.zeros(n)
            tmp[kp:] = (1.0 - h[:n-kp])
            tmp[:n-kp] = -h[kp:]
        grad += w * tmp
    grad *= dx
    return obj, grad

def _obj_wrap(x):
    h = np.clip(x, 0, 1)
    o, _ = smooth_obj_grad(h, _opt_state['T'], _opt_state['active_idx'])
    return o

def _grad_wrap(x):
    h = np.clip(x, 0, 1)
    _, g = smooth_obj_grad(h, _opt_state['T'], _opt_state['active_idx'])
    return g

def _sum_eq(x):
    return x.sum() - len(x) / 2.0

def _sum_eq_jac(x):
    return np.ones(len(x))

def slsqp_run(h_init, T, maxiter, active_idx=None):
    n = len(h_init)
    _opt_state['T'] = T
    _opt_state['active_idx'] = active_idx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_obj_wrap, h_init, jac=_grad_wrap, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h_init

def subgrad_dir(h, top_m=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    thr = np.max(corr) - 1e-8
    active = np.where(corr >= thr)[0]
    if len(active) > top_m:
        active = active[np.argsort(corr[active])[-top_m:]]
    grad = np.zeros(n)
    for idx in active:
        k = int(idx) - (n - 1)
        if k >= 0:
            tmp = np.zeros(n)
            tmp[:n-k] = (1.0 - h[k:])
            tmp[k:] = -h[:n-k]
        else:
            kp = -k
            tmp = np.zeros(n)
            tmp[kp:] = (1.0 - h[:n-kp])
            tmp[:n-kp] = -h[kp:]
        grad += tmp
    return grad / max(len(active), 1)

def momentum_descent(h_init, n_iters, lr_init=0.04):
    n = len(h_init)
    h_cur = h_init.copy()
    best_local = h_cur.copy()
    best_c5_l, _ = compute_c5(h_cur)
    v = np.zeros(n)
    lr = lr_init
    mom = 0.82
    stall = 0
    for it in range(n_iters):
        g = subgrad_dir(h_cur, top_m=10)
        v = mom * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5_l - 1e-13:
            best_c5_l = c5_new
            best_local = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.025, 0.35)
            stall = 0
        else:
            stall += 1
            lr *= 0.96
            if stall > 250:
                break
    return best_local, best_c5_l

def binary_round(h, n):
    order = np.argsort(h)
    hb = np.zeros(n)
    hb[order[n//2:]] = 1.0
    return hb

    if tl() > 120:

        if c5_s < best_c5:
