def cheb_eval(orders, x):
    """Evaluate Chebyshev polynomial of given orders at x."""
    # T_k(x) recursion
    Tk = np.zeros_like(x)
    for k in range(orders + 1):
        if k == 0:
            Tk = np.ones_like(x)
        elif k == 1:
            Tk = x.copy()
        else:
            Tk = 2 * x * Tk - np.roll(Tk, 1)  # This won't work
    # Better: use recurrence properly
    T = np.zeros((orders + 1, len(x)))
    T[0] = 1.0
    if orders >= 1:
        T[1] = x
    for k in range(2, orders + 1):
        T[k] = 2 * x * T[k-1] - T[k-2]
    return T.T  # shape (len(x), orders+1)

def h_from_cheb(coeffs, n):
    """Evaluate h from Chebyshev coefficients at n equally spaced points."""
    x = np.linspace(-1, 1, n)
    T = cheb_eval(len(coeffs)-1, x)  # (n, K+1)
    return T @ coeffs

def smooth_max(corr, T, n):
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def grad_smooth_max(h, T, n):
    """Gradient of smooth_max w.r.t. h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(T * (corr - m)) / np.sum(np.exp(T * (corr - m)))
    # d(corr[s])/d(h[j]) for shift s:
    # corr[s] = sum_j h[j] * (1 - h[j+s_mod])
    # where s_mod = s - (n-1)
    # d/dh[j] = (1-h[j+s_mod]) + h[j-s_mod] * (-1) if both in range
    grad = np.zeros(n)
    for s in range(2*n - 1):

        if sh == 0:

            j_max = n - sh
            grad[:j_max] += weights[s] * (1 - h[sh:j_max+sh])
        else:
            # sh < 0, let sh2 = -sh
            sh2 = -sh
            j_min = sh2
            grad[j_min:] += weights[s] * (1 - h[:n-sh2])
    return grad * dx

def cheb_obj(coeffs, n, T, x_pts, T_mat):
    """Objective in Chebyshev coefficient space."""
    h = T_mat @ coeffs
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, T, n)

def cheb_grad(coeffs, n, T_temp, x_pts, T_mat):
    """Gradient in Chebyshev coefficient space (approximate)."""
    h = T_mat @ coeffs
    h = np.clip(h, 0, 1)
    # Gradient w.r.t. h
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(T_temp * (corr - m))
    ws = weights.sum()
    weights = weights / ws
    dx = 2.0 / n
    grad_h = np.zeros(n)
    for s in range(2*n - 1):
        sh = s - (n - 1)
        if sh == 0:

        elif sh > 0:

        else:
            sh2 = -sh
            j_min = sh2
            grad_h[j_min:] += weights[s] * (1 - h[:n-sh2])
    grad_h *= dx
    # Chain rule: dc/dcoeffs = T_mat.T @ dc/dh
    return T_mat.T @ grad_h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    m = np.max(corr)
    # Numerical stability
    exps = np.exp(T * (corr - m))
    weights = exps / exps.sum()  # length 2n-1, indexed 0 to 2n-2
    # s = idx - (n-1), so idx = s + (n-1)
    
    grad = np.zeros(n)
    for j in range(n):
        for s in range(-(n-1), n):
            idx = s + (n-1)
            # (1-h[j-s]) * [0 ≤ j-s < n]
            js = j - s
            if 0 <= js < n:
                grad[j] += weights[idx] * (1.0 - h[js])
            # -h[j+s] * [0 ≤ j+s < n]
            js2 = j + s
            if 0 <= js2 < n:
                grad[j] -= weights[idx] * h[js2]
    return grad * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T, n):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    dx = 2.0 / n
    val = (m + np.log(exps.sum()) / T) * dx
    s_vals = np.arange(-(n - 1), n)
    j_mat = np.arange(n)
    js1 = j_mat[:, None] - s_vals[None, :]
    v1 = (js1 >= 0) & (js1 < n)
    A = np.where(v1, 1.0 - h[np.clip(js1, 0, n - 1)], 0.0) @ w
    js2 = j_mat[:, None] + s_vals[None, :]
    v2 = (js2 >= 0) & (js2 < n)
    B = np.where(v2, h[np.clip(js2, 0, n - 1)], 0.0) @ w
    grad = (A - B) * dx
    return val, grad

    v, g = smooth_obj_grad(x, T, n)

    return T

    js2 = j + s_vals

    dc2 = np.where((js2 >= 0) & (js2 < n), h[np.clip(js2, 0, n - 1)], 0.0)
    return delta * (dc1 - dc2)

    dx = 2.0 / n
