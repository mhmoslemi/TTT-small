    cfgs = [
        (n_par, 0.0, 3.0, 80.0, 0.035, 0.84, True),
        (n_par, 0.04, 3.0, 80.0, 0.035, 0.84, True),
        (n_par, 0.0, 30.0, 500.0, 0.012, 0.88, True),
        (n_par, 0.015, 10.0, 200.0, 0.02, 0.86, False),
        (400, 0.0, 3.0, 80.0, 0.03, 0.84, True),
        (n_par, 0.0, 200.0, 2000.0, 0.006, 0.92, True),
        (n_par, 0.06, 5.0, 100.0, 0.04, 0.82, True),
        (n_par, 0.0, 50.0, 300.0, 0.01, 0.90, False),
    ]
    for idx, (N, pert, tau_lo, tau_hi, lr0, mom, nest) in enumerate(cfgs):
        talloc = time_budget * alloc[idx]
        if time.time() > deadline - 20:
            break
        if N == n_par and pert == 0.0:
            h_s = best_h.copy()
        else:
            if N != n_par:
                xN = np.linspace(0.0, 2.0, N, endpoint=False)
                h_s = np.interp(xN, x_par, best_h)
            else:
                h_s = best_h.copy()
            if pert > 0:
                h_s = h_s + rng.randn(N) * pert
            h_s = project_sum(h_s, N / 2.0)
        try_at(N, h_s, talloc, tau_lo, tau_hi, lr0, mom, nest)

    if time.time() < deadline - 35:
        Nb = len(best_h)
        for tau_lb in [50, 150, 400]:
            if time.time() > deadline - 15:
                break
            h_lb, c5_lb = lbfgs_opt(best_h, Nb, Nb / 2.0, tau_lb, deadline)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb

    if time.time() < deadline - 5:
        Nb = len(best_h)
        best_h, best_c5 = snap_polish(best_h, Nb, Nb / 2.0, deadline - 1)

    final_c5 = calc_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
