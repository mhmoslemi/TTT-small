    # Phase 6: Final high-T polish
    n_cur = len(best_h)
    for T in [30000, 100000, 500000]:
        if elapsed_frac() > 0.995:
            break
        h_try = lbfgs_optimize(best_h, T, n_cur, 10000, 1e-18)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
