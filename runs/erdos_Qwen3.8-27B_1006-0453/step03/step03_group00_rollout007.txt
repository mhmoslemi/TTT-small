def project_to_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * ((target - s) / tot)
    else:
        if s < 1e-14:
            return np.full(n, 0.5)
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    sh = np.clip(T * (corr - m), -700.0, 700.0)
    return float((m + np.log(np.sum(np.exp(sh))) / T) * dx)

        return smooth_obj(x, T)
    def g_con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': g_con}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': iters, 'ftol': 1e-14})
        return project_to_sum(np.clip(r.x, 0, 1), n)
    except Exception:
        return project_to_sum(h.copy(), n)

def coord_opt(h, n, i):
    save = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full')
    h[i] = save
    co = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        if k == 0:
            continue
        c = 0.0
        if 0 <= i + k < n:
            c += 1.0 - h[i + k]
        if 0 <= i - k < n:
            c -= h[i - k]
        co[k + n - 1] = c
    dx = 2.0 / n
    def f1(hi):
        v = cb + co * hi
        v[n - 1] = cb[n - 1] + hi * (1.0 - hi)
        return float(np.max(v) * dx)
    r = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
    return float(r.x)

def coord_descent(h, n, t_end, sweeps=100):
    bh = h.copy()
    bc = compute_c5(bh)
    stall = 0
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = 0
        # Priority order: coords most affecting current max
        corr = np.correlate(bh, 1 - bh, mode='full')
        mx = np.max(corr)
        thr = mx - 0.05 * mx
        pri = set()
        off = n - 1
        for lag_idx in np.where(corr > thr)[0]:
            k = lag_idx - off
            lo = max(0, -k)
            hi = min(n, n - k)
            pri.update(range(lo, hi))
        order = list(pri)
        np.random.shuffle(order)
        for i in range(n):
            if i not in pri:
                order.append(i)
        for i in order:
            if time.time() > t_end:
                return bh, bc
            nv = coord_opt(bh, n, i)
            if abs(nv - bh[i]) > 1e-9:
                bh[i] = nv
                bh = project_to_sum(bh, n)
                nc = compute_c5(bh)
                if nc < bc - 1e-14:
                    bc = nc
                    imp += 1
        if imp == 0:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return bh, bc

    x_new = np.arange(n_new) / float(n_new) * 2.0
