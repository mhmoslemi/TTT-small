import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, T, topk):
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        c = corr
    m = np.max(c)
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def obj_lbfgs(x, args):
    T, topk, n, pen_w = args
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    sm = smooth_max_val(corr, T, topk)
    penalty = pen_w * (x.sum() - n / 2.0) ** 2
    return sm * dx + penalty

def grad_smooth_max(h, T, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(len(corr))
    c_sub = corr[idx]
    m = np.max(c_sub)
    w = np.exp(T * (c_sub - m))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ii in zip(w, idx):
        s = int(ii - (n - 1))
        ilo = max(0, -s)
        ihi = min(n - 1, n - 1 - s)
        if ilo > ihi:
            continue
        grad[ilo:ihi+1] += wi
        if s >= 0:
            jlo = max(ilo, 0)
            jhi = min(ihi, n - 1 - s)
        else:
            jlo = max(ilo, -s)
            jhi = ihi
        if jlo <= jhi:
            grad[jlo:jhi+1] -= wi * h[jlo+s:jhi+1+s]
        j2lo = max(0, ilo + s, max(0, s))
        j2hi = min(n - 1, ihi + s, n - 1 + s)
        if j2lo <= j2hi:
            grad[j2lo:j2hi+1] -= wi * h[j2lo-s:j2hi+1-s]
    return grad

def subgrad_step(h, T, topk, lr):
    n = len(h)
    dx = 2.0 / n
    grad = grad_smooth_max(h, T, topk)
    h_new = h - lr * grad * dx
    h_new = project_feasible(h_new, n)
    return h_new

def lbfgs_solve(h0, T, topk, n, maxiter, pen_w):
    bnds = [(0.0, 1.0)] * n
    args = (T, topk, n, pen_w)
    res = minimize(obj_lbfgs, h0, args=args, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, n)

def slsqp_solve(h0, T, topk, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    def eq_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_fun}]
    args = (T, topk, n, 0.0)
    res = minimize(obj_lbfgs, h0, args=args, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(np.clip(res.x, 0, 1), n)

def de_obj(x, n, pen_w):
    h = project_feasible(x, n)
    c5 = compute_c5(h)
    pen = pen_w * abs(h.sum() - n / 2.0)
    return c5 + pen

def de_obj_wrapper(x, args):
    n, pen_w = args
    return de_obj(x, n, pen_w)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n_init = len(best_h)
    best_h = project_feasible(best_h, n_init)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B sweep at current resolution
    n = len(best_h)
    T_sweep = [30, 80, 200, 600, 1500, 5000, 15000]
    k_sweep = [None, 8, 15, 25, 40]
    for T in T_sweep:
        if t_elapsed() > budget_s * 0.22:
            break
        for kk in k_sweep:
            if t_elapsed() > budget_s * 0.25:
                break
            try:
                h_new = lbfgs_solve(best_h, T, kk, n, 3000, 50.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: Vectorized subgradient descent
    T_sg = 800
    k_sg = 20
    lr0 = 0.8
    for it in range(600):
        if t_elapsed() > budget_s * 0.38:
            break
        lr = lr0 / (1.0 + 0.02 * it)
        h_new = subgrad_step(best_h, T_sg, k_sg, lr)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: Downsample + DE for global structure, then upsample
    if t_elapsed() < budget_s * 0.55:
        for n_de in [30, 40, 50]:
            if t_elapsed() > budget_s * 0.50:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_de, endpoint=False)
            h_low = np.interp(x_new, x_old, best_h)
            h_low = project_feasible(h_low, n_de)
            try:
                args_de = (n_de, 5.0)
                de_res = differential_evolution(
                    de_obj_wrapper, [(0.0, 1.0)] * n_de, args=args_de,
                    maxiter=100, popsize=20, tol=1e-8,
                    mutation=(0.5, 1.0), recombination=0.85,
                    seed=seed + n_de, polish=False, init='halton'
                )
                h_de = project_feasible(de_res.x, n_de)
                # Upsample and optimize
                x_up = np.linspace(0, 2, n, endpoint=False)
                h_up = np.interp(x_up, x_new, h_de)
                h_up = project_feasible(h_up, n)
                for T in [200, 1000]:
                    for kk in [None, 15]:
                        if t_elapsed() > budget_s * 0.55:
                            break
                        h_try = lbfgs_solve(h_up, T, kk, n, 1500, 50.0)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_h = h_try
            except Exception:
                pass

    # Phase 4: Multi-resolution refinement
    for n_try in [100, 120, 140, 200, 240]:
        if t_elapsed() > budget_s * 0.75:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [100, 500, 2000, 8000]:
            for kk in [None, 12, 25]:
                if t_elapsed() > budget_s * 0.78:
                    break
                try:
                    h_try = lbfgs_solve(h_interp, T, kk, n_try, 2000, 50.0)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try
                        h_interp = h_try.copy()
                except Exception:
                    pass

    # Phase 5: SLSQP for exact constraint handling
    n = len(best_h)
    if t_elapsed() < budget_s * 0.88:
        for T in [500, 2000, 8000, 30000]:
            if t_elapsed() > budget_s * 0.88:
                break
            for kk in [None, 10, 20]:
                try:
                    h_try = slsqp_solve(best_h, T, kk, n, 800)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try
                except Exception:
                    pass

    # Phase 6: Fine coordinate search targeting worst shifts
    n = len(best_h)
    for outer in range(400):
        if t_elapsed() > budget_s * 0.96:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-10:]
        improved = False
        for ksh in top_idx:
            s = ksh - (n - 1)
            ilo = max(0, -s)
            ihi = min(n - 1, n - 1 - s)
            step_range = max(1, min(6, (ihi - ilo) // 2 + 1))
            for j in range(ilo, min(ilo + step_range, n)):
                for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                    h_try = best_h.copy()
                    h_try[j] += delta
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and outer > 150:
            break

    # Phase 7: Final ultra-high-T L-BFGS-B + SLSQP polish
    n = len(best_h)
    for T in [10000, 50000, 200000]:
        if t_elapsed() > budget_s * 0.99:
            break
        try:
            h_try = lbfgs_solve(best_h, T, None, n, 3000, 200.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        except Exception:
            pass
    if t_elapsed() < budget_s * 0.995:
        for kk in [5, 10]:
            try:
                h_try = slsqp_solve(best_h, 5000, kk, n, 1000)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
