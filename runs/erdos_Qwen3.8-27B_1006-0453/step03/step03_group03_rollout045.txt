def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

    grad = np.zeros(n)

            g[kp:] += (1.0 - h[:n-kp])
            g[:n-kp] += -h[kp:]
        grad += weights[sorted_idx.tolist().index(idx)] * g
    return h - lr * grad

    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m) / max(1, T)))) * T / T
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def nesterov_subgrad(h, n_iters=2000, top_k=12):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    h_cur = h.copy()
    y = h.copy()
    v = np.zeros(n)
    lr = 0.04
    mom = 0.82
    stall = 0
    for it in range(n_iters):
        yc = np.clip(y, 0, 1)
        yc = project_feasible(yc, n)
        _, corr_y = compute_c5(yc)
        m = np.max(corr_y)
        sidx = np.argsort(corr_y)[::-1][:top_k]
        w = np.exp(10.0 * (corr_y[sidx] - m))
        w /= w.sum()
        gy = np.zeros(n)
        for j, idx in enumerate(sidx):
            k = idx - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                g[:n-k] += (1.0 - yc[k:])
                g[k:] += -yc[:n-k]
            else:
                kp = -k
                g[kp:] += (1.0 - yc[:n-kp])
                g[:n-kp] += -yc[kp:]
            gy += w[j] * g
        v = mom * v - lr * gy
        h_new = project_feasible(h_cur + v, n)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * mom * mom)) / 2.0
        y = h_new + (mom / beta) * (h_new - h_cur)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-11:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.015, 0.25)
            stall = 0
        else:
            stall += 1
            lr *= 0.98
        h_cur = h_new
        if stall > 400:
            break
    return best_h, best_c5

def spectral_perturb(h, n_freqs=5, amp=0.04):
    n = len(h)
    x = np.linspace(0, 2, n)
    pert = np.zeros(n)
    for f in range(1, n_freqs + 1):
        ph = np.random.uniform(0, 2 * np.pi)
        pert += np.sin(np.pi * f * x + ph)
    pert /= (np.max(np.abs(pert)) + 1e-10)
    return project_feasible(h + amp * pert, n)

def random_binary(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def blended_init(n, rng):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * x) + 0.1 * np.random.randn(n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP warm start with escalating T
    for T in [300, 2000, 10000, 40000]:
        if elapsed() > budget_s * 0.08:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=500)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Nesterov subgradient
    if elapsed() < budget_s * 0.22:
        h_opt, c5_opt = nesterov_subgrad(best_h, n_iters=4000, top_k=12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        h_f = slsqp_smooth(best_h, 50000, maxiter=500)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 3: Multi-n exploration with diverse inits
    n_list = [42, 48, 54, 60, 66, 72, 78, 84, 90]
    for n_try in n_list:
        if elapsed() > budget_s * 0.62:
            break
        if n_try % 2 != 0:
            continue
        trials = 6 if n_try == best_n else 4
        for trial in range(trials):
            if elapsed() > budget_s * 0.62:
                break
            if trial == 0:
                if n_try == best_n:
                    h_base = best_h.copy()
                else:
                    xo = np.linspace(0, 2, best_n)
                    xn = np.linspace(0, 2, n_try)
                    h_base = project_feasible(np.interp(xn, xo, best_h), n_try)
            elif trial == 1:
                h_base = random_binary(n_try, rng)
            elif trial == 2:
                h_base = blended_init(n_try, rng)
            else:
                amp = 0.03 + 0.05 * rng.rand()
                h_base = spectral_perturb(best_h if n_try == best_n else project_feasible(
                    np.interp(np.linspace(0,2,n_try), np.linspace(0,2,best_n), best_h), n_try),
                    n_freqs=4, amp=amp)
            h_opt, _ = nesterov_subgrad(h_base, n_iters=1200, top_k=10)
            h_opt = slsqp_smooth(h_opt, 15000, maxiter=400)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 4: Focused lag-reduction
    if elapsed() < budget_s * 0.82:
        _, corr = compute_c5(best_h)
        n = best_n
        m = np.max(corr)
        active_lags = np.where(corr > m - 0.02 * m)[0]
        for _ in range(15):
            if elapsed() > budget_s * 0.82:
                break
            for _ in range(300):
                h_p = best_h + np.random.randn(n) * 0.02
                h_p = project_feasible(h_p, n)
                h_p = slsqp_smooth(h_p, 80000, maxiter=500)
                c5_p, _ = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p
                    break
            h_opt, _ = nesterov_subgrad(best_h, n_iters=2000, top_k=15)
            h_opt = slsqp_smooth(h_opt, 100000, maxiter=600)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            else:
                break

    # Phase 5: Rotation + perturbation
    if elapsed() < budget_s * 0.90:
        for shift in [2, 4, 7, 11, 17]:
            if elapsed() > budget_s * 0.90:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_opt, _ = nesterov_subgrad(h_rot, n_iters=2500, top_k=12)
            h_opt = slsqp_smooth(h_opt, 80000, maxiter=600)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 6: Final ultra-precise polish
    if elapsed() < budget_s * 0.97:
        for T in [150000, 400000]:
            if elapsed() > budget_s * 0.97:
                break
            h_f = slsqp_smooth(best_h, T, maxiter=800)
            c5_f, _ = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
        for scale in [0.008, 0.004, 0.002]:
            if elapsed() > budget_s * 0.97:
                break
            h_p = best_h + np.random.randn(len(best_h)) * scale
            h_p = project_feasible(h_p, len(best_h))
            h_f = slsqp_smooth(h_p, 300000, maxiter=700)
            c5_f, _ = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
