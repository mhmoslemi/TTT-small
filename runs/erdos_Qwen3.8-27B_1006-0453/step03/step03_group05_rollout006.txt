    for ki in top_idx:
        s = int(ki - (n - 1))
        j1lo = max(0, -s)
        j1hi = min(n, n - s)
        if j1hi > j1lo:
            grad[j1lo:j1hi] += (1.0 - h[j1lo + s:j1hi + s])
        j2lo = max(0, s)
        j2hi = min(n, s + n)
        if j2hi > j2lo:
            grad[j2lo:j2hi] -= h[j2lo - s:j2hi - s]
    return grad * dx

def topk_avg_obj(x, k, n, lam, target):
    corr = corr_vals(x)
    top = np.sort(corr)[::-1][:k]
    base = np.sum(top) * (2.0 / n) / k
    pen = lam * (x.sum() - target) ** 2
    return base + pen

def topk_avg_grad(x, k, n, lam, target):
    corr = corr_vals(x)
    idx = np.argpartition(corr, -k)[-k:]
    g = subgrad_from_topk(x, n, idx)
    g /= k
    g += 2.0 * lam * (x.sum() - target) * 2.0 / n
    return g

def mom_subgrad(h, n, steps, top_k, alpha0, decay, mom):
    best = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    alpha = alpha0
    for _ in range(steps):
        corr = corr_vals(h)
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        g = subgrad_from_topk(h, n, top_idx)
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g /= gn
        vel = mom * vel - alpha * g
        h_new = project_feasible(h + vel, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            h = h_new
        else:
            alpha *= 0.995
            vel *= 0.5
        alpha *= decay
    return best, best_c5

def lbfgs_pen(h, k, n, n_iter=300, lam0=50.0):
    target = float(n) / 2.0
    x = h.copy()
    for stage in range(4):
        lam = lam0 * (10 ** stage)
        def obj(xx):
            return topk_avg_obj(xx, k, n, lam, target)
        def objg(xx):
            return topk_avg_grad(xx, k, n, lam, target)
        try:
            res = minimize(obj, x, jac=objg, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': n_iter // 4 + 50, 'ftol': 1e-14, 'gtol': 1e-8})
            x = np.clip(res.x, 0, 1)
        except Exception:
            pass
        x = project_feasible(x, n)
    return x

def slsqp_refine(h, k, n, maxiter=300):
    target = float(n) / 2.0
    dx_ = 2.0 / n
    def obj(x):
        corr = corr_vals(x)
        top = np.sort(corr)[::-1][:k]
        return np.sum(top) * dx_ / k
    def objg(x):
        corr = corr_vals(x)
        idx = np.argpartition(corr, -k)[-k:]
        g = subgrad_from_topk(x, n, idx)
        return g / k
    def cfun(x):
        return x.sum() - target
    def cfun_j(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': cfun, 'jac': cfun_j}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def diverse_init(n, seed):
    rs = np.random.RandomState(seed)
    r = rs.random()
    if r < 0.25:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif r < 0.5:
        f = rs.randint(2, 8)
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(np.pi * f * x)
    elif r < 0.75:
        h = np.random.RandomState(seed).rand(n)
    else:
        f1, f2 = rs.randint(2, 6), rs.randint(6, 14)
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(np.pi * f1 * x) + 0.15 * np.cos(np.pi * f2 * x + 1.0)
    return project_feasible(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Aggressive momentum sub-gradient from best (fast)
    for top_k, steps, a0, dec in [(2, 4000, 0.04, 0.9999), (3, 4000, 0.03, 0.9999), (5, 3000, 0.025, 0.9999), (1, 3000, 0.02, 0.9998)]:
        if elapsed() > budget_s * 0.12:
            break
        ht, ct = mom_subgrad(best_h, best_n, steps, top_k, a0, dec, 0.85)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 2: L-BFGS-B with penalty (fast quasi-Newton)
    for k in [1, 2, 3, 5]:
        if elapsed() > budget_s * 0.30:
            break
        ht = lbfgs_pen(best_h, k, best_n, 400)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 3: SLSQP refinement (more precise, slower)
    for k in [1, 2, 3]:
        if elapsed() > budget_s * 0.42:
            break
        ht = slsqp_refine(best_h, k, best_n, 350)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 4: Diverse restarts
    n_div = int((budget_s * 0.60 - elapsed()) / 12) if elapsed() < budget_s * 0.60 else 3
    n_div = max(3, min(n_div, 20))
    for trial in range(n_div):
        if elapsed() > budget_s * 0.62:
            break
        # Mix: sometimes perturb best, sometimes fresh init
        if trial % 3 == 0:
            ht = best_h + np.random.randn(best_n) * 0.04
            ht = project_feasible(ht, best_n)
        else:
            ht = diverse_init(best_n, seed + 100 + trial)
        # Quick sub-gradient warmup
        ht, ct = mom_subgrad(ht, best_n, 1500, 3, 0.03, 0.9999, 0.85)
        # L-BFGS polish
        ht2 = lbfgs_pen(ht, 2, best_n, 300)
        ct2 = compute_c5(ht2)
        if ct2 < best_c5:
            best_c5, best_h = ct2, ht2
        elif ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 5: Multi-resolution
    for nr in [100, 150, 300, 400]:
        if elapsed() > budget_s * 0.82:
            break
        if nr == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        # Add small noise at new resolution
        hu = project_feasible(hu + np.random.randn(nr) * 0.015, nr)
        ht, ct = mom_subgrad(hu, nr, 2000, 3, 0.025, 0.9999, 0.85)
        ht = lbfgs_pen(ht, 2, nr, 300)
        ct = compute_c5(ht)
        # Also try downsampled version
        if nr > best_n:
            x_dn = np.linspace(0, 2, best_n)
            hd = np.interp(x_dn, x_new, ht)
            hd = project_feasible(hd, best_n)
            cd = compute_c5(hd)
            if cd < best_c5:
                best_c5, best_h = cd, hd
        elif ct < best_c5:
            best_c5, best_h, best_n = ct, ht, nr
        elif ct < best_c5:
            best_c5, best_h, best_n = ct, ht, nr

    # Phase 6: Intensive final polish
    if elapsed() < budget_s * 0.92:
        for k in [1, 2]:
            ht = slsqp_refine(best_h, k, best_n, 500)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht
        ht = lbfgs_pen(best_h, 1, best_n, 600)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 7: Micro sub-gradient + final SLSQP
    if elapsed() < budget_s * 0.96:
        ht, ct = mom_subgrad(best_h, best_n, 800, 1, 0.005, 0.9999, 0.9)
        if ct < best_c5:
            best_c5, best_h = ct, ht
        ht = slsqp_refine(best_h, 1, best_n, 400)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
