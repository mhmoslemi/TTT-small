from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-14:
            h += room * (diff / t)
    else:
        room = h.copy()
        t = room.sum()
        if t > 1e-14:
            h += room * (diff / t)
    return np.clip(h, 0.0, 1.0)

def sum_con(h):
    return h.sum() - len(h) / 2.0

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp(h0, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    res = minimize(soft_obj, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=[{'type':'eq','fun':sum_con}],
                   options={'maxiter':maxiter, 'ftol':ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def gradient_step(h, n, lr=0.01):
    """Take a step using the subgradient of the max correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_val = np.max(corr)
    # Soft weights on top correlations
    T = 50
    weights = np.exp(T * (corr - top_val))
    weights /= weights.sum()
    
    # Compute gradient numerically for the top-weighted objective
    # d/dh[i] of sum_j w_j * corr[j]
    # corr[j] involves h and (1-h) at various positions
    # Use central differences for a few key components
    
    grad = np.zeros(n)
    eps = 1e-5
    for i in range(n):
        h_plus = h.copy(); h_plus[i] = min(1, h[i] + eps)
        h_minus = h.copy(); h_minus[i] = max(0, h[i] - eps)
        # Compute weighted correlation
        c1 = np.correlate(h_plus, 1-h_plus, mode='full')
        c2 = np.correlate(h_minus, 1-h_minus, mode='full')
        grad[i] = np.sum(weights * (c1 - c2)) / (2 * eps)
    
    # Project gradient step
    h_new = h - lr * grad
    h_new = make_feasible(h_new, n)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-14:
            h += room * (diff / t)
    else:
        room = h.copy()
        t = room.sum()
        if t > 1e-14:
            h += room * (diff / t)
    return np.clip(h, 0.0, 1.0)

def sum_con(h):
    return h.sum() - len(h) / 2.0

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

    bnds = [(0.0, 1.0)] * n

                   bounds=bnds, constraints=[{'type': 'eq', 'fun': sum_con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)
