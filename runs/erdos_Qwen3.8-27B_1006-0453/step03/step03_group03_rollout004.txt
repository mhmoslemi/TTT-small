import numpy as np
from scipy.optimize import minimize
import time

# All helper functions at top level

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_erdos(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    # Numerical stability: subtract max
    max_s = np.max(shifted)
    if max_s > 500:
        shifted = shifted - max_s
        m_adj = m + max_s / T
    else:
        m_adj = m
    val = m_adj + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def smooth_grad_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    max_s = np.max(shifted)
    if max_s > 500:
        shifted = shifted - max_s
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(n - 1, 2 * n - 1):  # Only lags 0 to n-1 matter? No, all lags.
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad[:] += w[idx] * (1.0 - 2.0 * h)
            else:
                grad[:n-k] += w[idx] * (1.0 - h[k:])
                grad[k:] += -w[idx] * h[:n-k]
        else:
            kp = -k
            grad[kp:] += w[idx] * (1.0 - h[:n-kp])
            grad[:n-kp] += -w[idx] * h[kp:]
    return grad * dx

def sum_const(x):
    return x.sum() - len(x) / 2.0

def sum_const_jac(x):
    return np.ones(len(x))

def lbfgs_smooth(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_const, 'jac': sum_const_jac}]
    try:
        res = minimize(smooth_erdos, h.copy(), args=(T,), method='L-BFGS-B',
                      bounds=bnds, constraints=cons, jac=smooth_grad_fn,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def nesterov_subgrad(h, n_iters, lr_init=0.06, decay=0.97):
    n = len(h)
    y = h.copy()
    x = h.copy()
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    lr = lr_init
    mom = 0.85
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(y)
        m = np.max(corr)
        thresh = m - 1e-7
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 8:
            top = np.argsort(corr[active])[::-1][:8]
            active = active[top]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] += (1.0 - 2.0 * h)
                else:
                    grad[:n-k] += (1.0 - h[k:])
                    grad[k:] += -h[:n-k]
            else:
                kp = -k
                grad[kp:] += (1.0 - h[:n-kp])
                grad[:n-kp] += -h[kp:]
        grad /= len(active)
        y_new = project_feasible(y - lr * grad, n)
        x_new = y_new + mom * (y_new - y)
        x_new = project_feasible(x_new, n)
        y = x_new
        c5_new, _ = compute_c5(y)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = y.copy()
            lr = min(lr * 1.015, 0.30)
            mom = min(mom + 0.0005, 0.95)
            stall = 0
        else:
            stall += 1
            lr *= decay
            mom = max(mom - 0.002, 0.5)
            if stall > 250:
                break
    return best_h, best_c5

def slsqp_direct(h, maxiter):
    n = len(h)
    _, corr = compute_c5(h)
    t0 = float(np.max(corr)) + 0.03
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': _obj_eq_list}, {'type': 'ineq', 'fun': _obj_in_list}]
    try:
        res = minimize(_obj_t, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def _obj_t(x):
    return x[-1]

def _obj_eq_list(x):
    return x[:-1].sum() - len(x[:-1]) / 2.0

def _obj_in_list(x):
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def snap_binary(h, threshold=0.02):
    h = h.copy()
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    return project_feasible(h, len(h))

The weights w will be extremely peaked (essentially a one-hot vector on the max correlation lag), and the gradient will just be the subgradient of the active constraint. That's actually fine - it's the correct behavior for large T.

Let me also think about whether I should use a different approach for the very high T regime. When T → ∞, the problem becomes: minimize the maximum of the correlations, which is a min-max problem. The KKT conditions say that at the optimum, all active constraints (correlations at the max) should have zero derivative along feasible directions.

Actually, for the min-max problem, we can use a different formulation: minimize t subject to corr_k ≤ t for all k. This is a linear programming problem in h (since corr_k is bilinear in h, it's not linear). So it's a quadratic program with the extra variable t.

Wait, corr_k = ∑ h[i]*(1-h[i+k]) is quadratic (bilinear) in h. So the min-max is a quadratic optimization. For small n (78), we could potentially use a QP solver. But scipy doesn't have a great QP solver for this formulation.

Let me also think about whether there's a structural property I'm missing. For n=78, the sequence has 78 elements with 39 ones. The correlations at different lags have different "weights" because the number of valid pairs varies.

For lag k (0 ≤ k ≤ n-1), the number of valid pairs is n-k. So:
corr_k = (1/(n-k)) * ∑_{i=0}^{n-1-k} h[i]*(1-h[i+k]) (normalized)

But in our formulation, we're NOT normalizing - we're just summing. The dx factor applies to all. So the max is over the raw sums times dx.

For k=0: n/2 - ∑h[i]² (times dx) → minimized by making h binary
For k=n-1: only 1 term: h[0]*(1-h[n-1]) or h[1]*(1-h[0]) etc. → very small
For intermediate k: the typical term

The max is likely achieved at small positive k values (k=1, 2, 3, ...) where there are many overlapping pairs.

This suggests that the main constraint is on small lags. The subgradient focusing on the top few is appropriate.
