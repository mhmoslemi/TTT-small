    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)
    N = n_par
    target = N / 2.0
    h = h_par.copy()
    # Main optimization at parent size
    phases = [
        (15.0, 80.0, 0.06, 0.96, 50, 0.75),
        (80.0, 300.0, 0.025, 0.975, 70, 0.82),
        (300.0, 1000.0, 0.01, 0.985, 90, 0.87),
        (1000.0, 3000.0, 0.004, 0.99, 110, 0.91),
    ]
    remaining = deadline - t0 - 120
    for pi, (ts, te, lr0, dec, per, mom) in enumerate(phases):
        ph_end = min(t0 + remaining * (pi+1) / len(phases), deadline - 100)
        if time.time() > ph_end - 5:
            break
        h, c5v = optimize_phase(h, N, target, ph_end, ts, te, lr0, dec, per, mom, rng)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    # Try higher resolution
    if time.time() < deadline - 80:
        for N2 in [460, 530]:
            if time.time() > deadline - 50:
                break
            x2 = np.linspace(0, 2, N2, endpoint=False)
            x_p = np.linspace(0, 2, n_par, endpoint=False)
            h2 = np.interp(x2, x_p, h_par)
            h2 = project_sum(h2, N2 / 2.0)
            t_rem2 = deadline - time.time() - 25
            for ts2, te2, lr2, dec2, per2, mom2 in [(30.0, 200.0, 0.04, 0.97, 60, 0.78),
                                                      (200.0, 800.0, 0.015, 0.98, 80, 0.85),
                                                      (800.0, 2500.0, 0.006, 0.99, 100, 0.90)]:
                if time.time() > deadline - 25:
                    break
                h2, c52 = optimize_phase(h2, N2, N2/2.0, min(time.time() + t_rem2/4.0, deadline-25),
                                         ts2, te2, lr2, dec2, per2, mom2, rng)
                if c52 < best_c5:
                    best_c5 = c52
                    best_h = h2.copy()
    # Final short optimization at best size
    if time.time() < deadline - 20:
        Nb = len(best_h)
        h_f = best_h.copy()
        h_f, cf = optimize_phase(h_f, Nb, Nb/2.0, min(time.time() + 25, deadline - 10),
                                  2000.0, 8000.0, 0.002, 0.992, 80, 0.92, rng)
        if cf < best_c5:
            best_c5 = cf
            best_h = h_f
    # Coordinate polish
    if time.time() < deadline - 4:
        Nb2 = len(best_h)
        best_h, best_c5 = polish_coords(best_h, Nb2, Nb2/2.0, deadline - 2, rng)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
