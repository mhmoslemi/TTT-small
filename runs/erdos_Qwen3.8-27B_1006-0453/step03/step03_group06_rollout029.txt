from scipy.optimize import minimize, differential_evolution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

    def obj(h):
        return smooth_c5(h, T, topk) + pen * (h.sum() - n/2.0)**2
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bnds,
                  options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(np.clip(res.x, 0, 1), n)

def slsqp_opt(h0, T, n, topk, maxiter):
    def obj(h):
        return smooth_c5(h, T, topk)
    def cons_fn(h):
        return h.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fn}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(np.clip(res.x, 0, 1), n)

        pos += gaps[i]
        lo = pos
        hi = pos + widths[i]
        h[(x >= lo) & (x < hi)] = 1.0
        pos = hi
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def lbfgs_opt(h0, T, n, topk, maxiter, pen=300.0):
    def obj(h):
        return smooth_c5(h, T, topk) + pen * (h.sum() - n / 2.0) ** 2
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(np.clip(res.x, 0, 1), n)

def slsqp_opt(h0, T, n, topk, maxiter):
    def obj(h):
        return smooth_c5(h, T, topk)
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(np.clip(res.x, 0, 1), n)

def softmax_to_intervals(a, b, K):
    exp_a = np.exp(a - a.max())
    gaps = exp_a / exp_a.sum()
    exp_b = np.exp(b - b.max())
    widths = exp_b / exp_b.sum()
    centers = np.zeros(K)
    pos = 0.0
    for i in range(K):
        pos += gaps[i]
        centers[i] = pos + widths[i] / 2.0
        pos += widths[i]
    return centers, widths

def hard_interval_h(centers, widths, K, n):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for i in range(K):
        lo = centers[i] - widths[i] / 2.0
        hi = centers[i] + widths[i] / 2.0
        h[(x >= lo) & (x < hi)] = 1.0
    return project_feasible(h, n)

def smooth_interval_h(centers, widths, K, n, sigma):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for i in range(K):
        c, w = centers[i], widths[i]
        if w <= 1e-8:
            continue
        lo, hi = c - w / 2.0, c + w / 2.0
        s_lo = 1.0 / (1.0 + np.exp(-(x - lo) / sigma))
        s_hi = 1.0 / (1.0 + np.exp(-(hi - x) / sigma))
        h = np.maximum(h, s_lo * s_hi)
    return project_feasible(h, n)

def interval_search_obj(z, K, n, sigma, T):
    a = z[:K + 1]
    b = z[K + 1:]
    centers, widths = softmax_to_intervals(a, b, K)
    h = smooth_interval_h(centers, widths, K, n, sigma)
    return smooth_c5(h, T, None)

def make_diverse_h(n, rng, mode):
    h = np.zeros(n)
    if mode == 0:
        h[:n // 2] = 1.0
    elif mode == 1:
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 2:
        h[n // 4:n // 2] = 1.0
        h[3 * n // 4:n] = 1.0
    elif mode == 3:
        for i in range(0, n, 4):
            h[i:i + 2] = 1.0
    else:
        k = max(2, n // 8)
        idx = rng.choice(n, size=k, replace=False)
        idx.sort()
        for i in idx:
            h[i] = 1.0
    return project_feasible(h, n)
