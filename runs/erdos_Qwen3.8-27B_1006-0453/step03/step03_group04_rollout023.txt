```python
                best_peak = new_peak
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
    
    return best_h, best_peak * dx

def continuous_optimize(h, N, deadline, lr=0.05, k=10):
    """Min-max optimization using subgradient of top-k peaks."""
    target = N / 2.0
    h = project(h.copy(), N)
    best_h = h.copy()
    best_c = max_corr(h)
    dx = 2.0 / N
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    
    while time.time() < deadline:
        corr = corr_full(h)
        peak = float(np.max(corr))
        if peak < best_c:
            best_c = peak
            best_h = h.copy()
        
        # Top-k peaks
        top = np.argpartition(corr, -k)[-k:]
        center = N - 1
        
        # Compute subgradient (average of top-k peak gradients)
        grad = np.zeros(N)
        for ai in top:
            L = ai - center
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            elif L < 0:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - h[M:])
                    grad[M:] -= h[:N - M]
        grad /= k
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project(h, N)
        
        if it % 200 == 0:
            lr *= 0.99
        
        # Periodically binarize and re-continue
        if it % 2000 == 0 and it > 2000:
            h = make_binary(h, N)
            m = np.zeros(N)
            v = np.zeros(N)
            it = 0
    
    return best_h, best_c * dx
```

```python
import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def max_peak(h):
    return float(np.max(corr_full(h)))

def make_binary(h, N):
    target = N // 2
    idx = np.argsort(-h, kind='stable')[:target]
    h_b = np.zeros(N)
    h_b[idx] = 1.0
    return h_b

def project_con(h, N):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binary_anneal(h, N, deadline, rng, T_factor=0.3):
    h_b = make_binary(h, N)
    peak = max_peak(h_b)
    best_h = h_b.copy()
    best_peak = peak
    dx = 2.0 / N
    
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    
    T = peak * T_factor
    T_min = peak * 0.003
    cool_rate = 0.9997
    n_improve_window = 0
    window = 500
    
    while time.time() < deadline:
        T = max(T * cool_rate, T_min)
        
        i_pos = rng.randint(len(ones))
        j_pos = rng.randint(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]
        
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_peak = max_peak(h_b)
        delta = new_peak - peak
        
        if delta <= 0 or (T > T_min * 1.02 and rng.random() < np.exp(-delta / T)):
            peak = new_peak
            ones[i_pos], zeros[j_pos] = j, i
            if new_peak < best_peak:
                best_peak = new_peak
                best_h = h_b.copy()
                n_improve_window = 0
            else:
                n_improve_window += 1
```
