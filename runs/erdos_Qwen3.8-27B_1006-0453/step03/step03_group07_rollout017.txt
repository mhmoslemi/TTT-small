def make_feasible(h, n):

    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:

    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-15:

        if total > 1e-15:

def project_feasible(g, n):
    """Project g onto {0<=h<=1, sum(h)=n/2} using bisection on lambda"""
    target = n / 2.0
    lo, hi = g.min() - 2.0, g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_try = np.clip(g - mid, 0.0, 1.0)
        s = h_try.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def subgrad_c5(h, T=200, top_m=20):
    """Compute weighted subgradient of soft-max of correlations and the current value."""
    n = len(h)
    dx = 2.0 / n
    omm = 1.0 - h
    corr = np.correlate(h, omm, mode='full')
    mx = corr.max()
    c = corr - mx
    w = np.exp(T * c)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    # Keep only top m for speed

        idx = np.argsort(w)[-top_m:]
        w_sub = w[idx]
        w_sub /= w_sub.sum()
        shifts = idx - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-15:
            h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot > 1e-15:
            h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def project_feasible(g, n):
    target = n / 2.0
    lo, hi = g.min() - 2.0, g.max() + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(g - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_topm(h, T, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    c = corr - mx
    w = np.exp(T * c)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    if top_m < len(corr):
        idx = np.argsort(w)[-top_m:]
        w = w[idx]
        w /= w.sum()
        shifts = idx - (n - 1)
    else:
        shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for sh, wi in zip(shifts, w):
        if wi < 1e-12:
            continue
        lo1 = max(0, -sh)
        hi1 = min(n, n - sh)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * dx * (1.0 - h[lo1 + sh:hi1 + sh])
        lo2 = max(0, sh)
        hi2 = min(n, n + sh)
        if hi2 > lo2:
            grad[lo2:hi2] += wi * dx * (-h[lo2 - sh:hi2 - sh])
    return grad, mx * dx

def pg_line_search(h, grad, lr_init, T, top_m, n, max_back=20):
    omm = 1.0 - h
    corr0 = np.max(np.correlate(h, omm, mode='full')) * (2.0 / n)
    cur = h.copy()
    for bb in range(max_back):
        alpha = lr_init * (0.5 ** bb)
        cand = project_feasible(h - alpha * grad, n)
        c1 = np.max(np.correlate(cand, 1.0 - cand, mode='full')) * (2.0 / n)
        if c1 < corr0 - 1e-14:
            return cand, c1, alpha
        cur = cand
    return cur, compute_c5(cur), lr_init * (0.5 ** (max_back - 1))

def slsqq(h0, T, n, maxiter=500):
    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        if T < 1e6:
            s = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        else:
            s = m
        return s * 2.0 / n
    def cfun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, method='SLSQP',
                   constraints=[{'type': 'eq', 'fun': cfun}],
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return make_feasible(np.clip(res.x, 0, 1), n)
