def binary_search_sa(h_init, n_iters, t0, t_end):
    h = h_init.copy()
    c5 = compute_c5(h)
    n = len(h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        h[i1], h[i0] = h[i0], h[i1]
        c5_new = compute_c5(h)
        delta = c5_new - c5
        temp = 0.003 * max(0.01, 1.0 - it / max(1, n_iters))
        if delta < 0 or (delta < 0.005 and np.random.rand() < np.exp(-delta / max(temp, 1e-8))):
            c5 = c5_new
        else:
            h[i1], h[i0] = h[i0], h[i1]
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = make_feasible(np.array(ihv, dtype=float), len(ihv))
    else:
        best_h = make_feasible(np.tile([1.0, 0.0], 100), 200)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nelder-Mead / Powell on piecewise-constant at multiple scales
    for n_try in [100, 120, 150, 200]:
        if elapsed() > budget_s * 0.35:
            break
        if n_try == len(best_h):
            h_cur = best_h
        else:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_cur = make_feasible(np.interp(x_new, x_old, best_h), n_try)

        for K in [10, 15, 20, 25, 30, 40, 50]:
            if elapsed() > budget_s * 0.35:
                break
            rep = max(1, n_try // K)
            aK = n_try // rep
            if aK < 4:
                continue
            inits = [h_to_pc(h_cur, aK)]
            inits.append(np.where(np.arange(aK) % 2 == 0, 0.75, 0.25).astype(float))
            inits.append(np.random.uniform(0.3, 0.7, aK))

            for x0 in inits:
                if elapsed() > budget_s * 0.35:
                    break
                x0 = np.clip(x0, 0, 1)
                for T in [300, 2000, 10000]:
                    if elapsed() > budget_s * 0.35:
                        break
                    try:
                        def obj_pc(x):
                            h = pc_to_h(np.clip(x, 0, 1), aK, n_try)
                            corr = np.correlate(h, 1.0 - h, mode='full')
                            return smax(corr, T) * (2.0 / n_try)
                        res = minimize(obj_pc, x0.copy(), method='Powell',
                                     options={'maxiter': 300, 'xtol': 1e-11, 'ftol': 1e-13})
                        h_try = pc_to_h(np.clip(res.x, 0, 1), aK, n_try)
                        c5v = compute_c5(h_try)
                        if c5v < best_c5:
                            best_c5 = c5v
                            best_h = h_try
                    except Exception:
                        pass

    # Phase 2: Nelder-Mead with tighter settings on best found
    n_cur = len(best_h)
    for K in [20, 30, 50, n_cur // 4, n_cur // 3]:
        if elapsed() > budget_s * 0.55:
            break
        if K < 4 or K > n_cur:
            continue
        rep = max(1, n_cur // K)
        aK = n_cur // rep
        if aK < 4:
            continue
        x0 = h_to_pc(best_h, aK)
        for T in [5000, 20000, 100000]:
            if elapsed() > budget_s * 0.55:
                break
            try:
                def obj_nm(x):
                    h = pc_to_h(np.clip(x, 0, 1), aK, n_cur)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    return smax(corr, T) * (2.0 / n_cur)
                res = minimize(obj_nm, x0.copy(), method='Nelder-Mead',
                             options={'maxiter': 50000, 'xatol': 1e-14, 'fatol': 1e-15,
                                     'adaptive': True})
                h_try = pc_to_h(np.clip(res.x, 0, 1), aK, n_cur)
                c5v = compute_c5(h_try)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_try
            except Exception:
                pass

    # Phase 3: SLSQP at reduced resolution for speed
    n_s = min(n_cur, 120)
    if n_s < n_cur:
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n_s, endpoint=False)
        h_s = make_feasible(np.interp(x_new, x_old, best_h), n_s)
    else:
        h_s = best_h.copy()
        n_cur = n_s
    for T in [500, 3000, 15000, 80000]:
        if elapsed() > budget_s * 0.72:
            break
        try:
            h_new = slsqp_refine(h_s, T, maxiter=600, ftol=1e-17)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
                n_cur = n_s
                h_s = h_new
        except Exception:
            pass

    # Phase 4: Binary/near-binary sequence search
    if elapsed() < budget_s * 0.82:
        n_b = len(best_h)
        h_bin = (best_h > 0.4).astype(float)
        h_bin = make_feasible(h_bin, n_b)
        t_end_b = t0 + budget_s * 0.82
        h_b, c5_b = binary_search_sa(h_bin, 4000, t0, t_end_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
        # Also try 3-level quantization
        h_3 = np.zeros(n_b)
        h_3[best_h > 0.7] = 1.0
        h_3[(best_h > 0.3) & (best_h <= 0.7)] = 0.5
        h_3 = make_feasible(h_3, n_b)
        c5_3 = compute_c5(h_3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h_3

    # Phase 5: Targeted local search (worst-lag correction)
    if elapsed() < budget_s * 0.92:
        n_cur = len(best_h)
        t_end5 = t0 + budget_s * 0.92
        for it in range(3000):
            if elapsed() > t_end5:
                break
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            k_star = max_idx - (n_cur - 1)

            # Identify contributing positions
            if k_star >= 0:
                bad = [i for i in range(n_cur - k_star)
                       if h[i + k_star] > 0.6 and h[i] < 0.4]
            else:
                kk = -k_star
                bad = [i for i in range(kk, n_cur)
                       if h[i] > 0.6 and h[i + kk] < 0.4]

            if len(bad) > 0:
                n_mod = min(len(bad), np.random.randint(1, 4))
                mod_idx = np.random.choice(bad, n_mod, replace=False)
                for idx in mod_idx:
                    if k_star >= 0:
                        j = idx + k_star
                        if np.random.rand() < 0.5:
                            h[j] *= 0.7
                            comp = np.random.randint(0, n_cur)
                            h[comp] += (h[j] * 0.3) / max(1, 1)
                        else:
                            h[idx] += 0.15 * h[idx] * 0.3
                            comp = np.random.randint(0, n_cur)
                            h[comp] -= 0.15 * h[idx] * 0.3
                    else:
                        kk = -k_star
                        j = idx + kk
                        if np.random.rand() < 0.5:
                            h[j] += 0.15 * (1 - h[j]) * 0.3
                            comp = np.random.randint(0, n_cur)
                            h[comp] -= 0.15 * (1 - h[j]) * 0.3
                        else:
                            h[idx] *= 0.75
                            comp = np.random.randint(0, n_cur)
                            h[comp] += h[idx] * 0.25
            else:
                # No clear target; small random perturbation
                n_p = np.random.randint(2, max(3, n_cur // 30))
                idx_p = np.random.choice(n_cur, n_p, replace=False)
                h[idx_p] += np.random.uniform(-0.08, 0.08, n_p)

            h = make_feasible(h, n_cur)
            c5v = compute_c5(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h

    # Phase 6: Final high-T SLSQP polish
    if elapsed() < budget_s * 0.98:
        n_cur = len(best_h)
        if n_cur <= 130:
            for T in [50000, 200000]:
                if elapsed() > budget_s * 0.98:
                    break
                try:
                    h_new = slsqp_refine(best_h, T, maxiter=1500, ftol=1e-18)
                    c5v = compute_c5(h_new)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = h_new
                except Exception:
                    pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
