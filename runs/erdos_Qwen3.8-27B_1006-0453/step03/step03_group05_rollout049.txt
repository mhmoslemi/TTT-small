import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0, 1)
    return h

def grad_topk(h, top_k):
    n = len(h)
    corr = get_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(n)
    nm1 = n - 1
    for ki in idx:
        s = int(ki - nm1)
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
        else:
            l = -s
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] -= h[:n - l]
    return grad / top_k

def proj_grad_momentum(h, n_steps, top_k, lr0, t_end):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    momentum = np.zeros(n)
    beta = 0.9
    stuck = 0
    for step in range(n_steps):
        if time.time() > t_end:
            break
        g = grad_topk(h, top_k)
        momentum = beta * momentum + (1 - beta) * g
        h_new = project_feasible(h - lr * momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
            stuck = 0
            lr = min(lr * 1.03, lr0 * 5)
        else:
            h = h_new
            stuck += 1
            lr *= 0.75
            if stuck > 80:
                h = project_feasible(h + np.random.randn(n) * 0.008, n)
                momentum = np.zeros(n)
                stuck = 0
                lr = lr0
    return best, best_c5

def topk_obj(h, k, n):
    corr = get_corr(h)
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_opt(h, obj, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def targeted_repair(h, n):
    corr = get_corr(h)
    s_star = int(np.argmax(corr) - (n - 1))
    h_new = h.copy()
    improved = False
    for i in range(n):
        j = i + s_star

                h_new[i] += delta
                h_new[j] -= delta
                improved = True
    if improved:

    return h_new
