import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_project(h, n):
    """Project to binary with exactly n/2 ones."""
    order = np.argsort(-h)  # descending
    h_bin = np.zeros(n)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def swap_cost(h, i, j):
    """Compute C5 after swapping h[i] and h[j] (assuming h[i]=1, h[j]=0)."""
    h2 = h.copy()
    h2[i] = 0.0
    h2[j] = 1.0
    c5, _ = compute_c5(h2)
    return c5

def best_improvement_search(h, max_iters=100000):
    """Local search: find best swap that reduces C5."""
    n = len(h)
    c5, _ = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        best_cost = c5
        best_pair = None
        for i in ones:
            for j in zeros:
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                c52, _ = compute_c5(h2)
                if c52 < best_cost - 1e-15:
                    best_cost = c52
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        c5 = best_cost
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    return h, c5

def simulated_annealing_binary(h, t_start=1.0, t_end=0.001, n_iters=50000):
    n = len(h)
    c5, _ = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    rng = np.random.RandomState(123)
    
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / n_iters)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new, _ = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            c5 = c5_new
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += room * (diff / t)
        else:
            room = h.copy()
            t = room.sum()
            if t < 1e-14:
                break
            h += room * (diff / t)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n // 2]] = 1.0
    return h_bin

def best_improvement(h, max_scans=200):
    n = len(h)
    ml, _ = compute_corr_max(h)
    best_h = h.copy()
    best_ml = ml
    for scan in range(max_scans):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in ones:
            if improved:
                break
            for j in zeros:
                h[i] = 0
                h[j] = 1
                ml_new, _ = compute_corr_max(h)
                if ml_new < ml - 1e-12:
                    ml = ml_new
                    improved = True
                    if ml < best_ml:
                        best_ml = ml
                        best_h = h.copy()
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
    return best_h, best_ml

def sa_binary(h, rng, n_iters=200000, t_start=2.0, t_end=0.01):
    n = len(h)
    cur = h.copy()
    cur_ml, _ = compute_corr_max(cur)
    best = cur.copy()
    best_ml = cur_ml
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / n_iters)
        i = rng.choice(np.where(cur == 1)[0])
        j = rng.choice(np.where(cur == 0)[0])
        cur[i] = 0
        cur[j] = 1
        new_ml, _ = compute_corr_max(cur)
        delta = new_ml - cur_ml
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            cur_ml = new_ml
            if new_ml < best_ml - 1e-12:
                best_ml = new_ml
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_ml

def slsqp_maxlag(h, maxiter=300, ftol=1e-14):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr)
    def eq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < obj(h):
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

def smooth_logsumexp(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    def eq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    x_old = np.linspace(0, 2, n_old)

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def interp_h(h, n_old, n_new):
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)

def perturb_binary(h, rng, n_swaps):
    cur = h.copy()
    for _ in range(n_swaps):
        o = np.where(cur == 1)[0]
        z = np.where(cur == 0)[0]
        cur[o[rng.randint(len(o))]] = 0
        cur[z[rng.randint(len(z))]] = 1
    return cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search
    h_bin = to_binary(best_h, best_n)
    h_ls, ml_ls = local_search(h_bin, max_rounds=3000)
    c5 = ml_ls * 2.0 / best_n
    if c5 < best_c5:
        best_c5, best_h = c5, h_ls.copy()

    # Phase 2: SA with diverse restarts
    for r in range(12):
        if time.time() - t0 > budget_s * 0.22:
            break
        if r == 0:
            h_s = h_ls.copy()
        elif r < 4:
            h_s = perturb_binary(h_ls, rng, 5 + 3 * r)
        else:
            h_s = h_ls.copy()
            nsw = rng.randint(8, 35)
            h_s = perturb_binary(h_s, rng, nsw)
        iters = 200000 if r < 4 else 120000
        h_sa, ml_sa = sa_search(h_s, rng, iters=iters,
                                t_hot=4.0 + 0.5 * r, t_cold=0.001)
        c5_sa = ml_sa * 2.0 / best_n
        if c5_sa < best_c5:
            best_c5, best_h, h_ls = c5_sa, h_sa.copy(), h_sa.copy()

    # Phase 3: SLSQP continuous polish from best binary
    if time.time() - t0 < budget_s * 0.42:
        h_p = local_search(h_ls, max_rounds=2000)[0]
        for T in [500, 5000, 50000, 300000]:
            if time.time() - t0 > budget_s * 0.42:
                break
            h_s = slsqp_smooth(h_p, T, maxiter=500)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s
                h_p = h_s
    if time.time() - t0 < budget_s * 0.48:
        h_d = slsqp_direct(best_h, maxiter=1000)
        c5_d = compute_c5(h_d)
        if c5_d < best_c5:
            best_c5, best_h = c5_d, h_d

    # Phase 4: Multi-n exploration
    for nt in [76, 80, 82, 74, 84, 86]:
        if time.time() - t0 > budget_s * 0.72:
            break
        if nt % 2:
            continue
        h_i = interp_h(best_h, best_n, nt)
        h_b = to_binary(h_i, nt)
        h_l, ml = local_search(h_b, max_rounds=1500)
        h_s2, ml2 = sa_search(h_l, rng, iters=100000, t_hot=4.0, t_cold=0.003)
        if ml2 < ml:
            h_l, ml = h_s2, ml2
        c5_l = ml * 2.0 / nt
        if c5_l < best_c5:
            best_c5, best_n = c5_l, nt
        h_p = slsqp_smooth(h_l, 10000, maxiter=400)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5, best_h, best_n = c5_p, h_p, nt

    # Phase 5: Pure random restarts (different basin)
    for r in range(5):
        if time.time() - t0 > budget_s * 0.87:
            break
        h_r = np.zeros(best_n)
        idx = rng.choice(best_n, best_n // 2, replace=False)
        h_r[idx] = 1.0
        h_l, ml = local_search(h_r, max_rounds=800)
        h_s2, ml2 = sa_search(h_l, rng, iters=120000, t_hot=6.0, t_cold=0.001)
        c5_r = ml2 * 2.0 / best_n
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_s2.copy()

    # Phase 6: Final high-T polish
    if time.time() - t0 < budget_s * 0.97:
        for T in [100000, 500000]:
            h_f = slsqp_smooth(best_h, T, maxiter=800)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5, best_h = c5_f, h_f
        h_f = slsqp_direct(best_h, maxiter=1200)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
