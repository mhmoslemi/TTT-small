    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, T):
    m = np.max(vals)
    if T < 1e-4:
        return m
    shifted = T * (vals - m)
    # Only compute exp for values not too far below max
    mask = shifted > -50
    if not mask.all():
        s = np.sum(np.exp(shifted[mask])) + np.sum(np.exp(shifted[~mask]))  # ~0 for masked
    else:
        s = np.sum(np.exp(shifted))
    return m + np.log(s) / T

    x = np.arange(n) * 2.0 / n
    h = np.full(n, 0.5)
    for m in range(1, M + 1):
        a = scale / m * 0.5
        b = scale / m * 0.3
        h += a * np.cos(m * np.pi * x) + b * np.sin(m * np.pi * x)
    return h

def lspg_optimize(h0, T, n, maxiter=500, ftol=1e-14):
    """L-BFGS-B with penalty for sum constraint."""
    lam = 500.0
    
    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = smooth_max(corr, T) * (2.0 / n)
        val += lam * (h.sum() - n / 2.0) ** 2
        return val
    
    # No analytical gradient - let L-BFGS-B use numerical
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
    h_new = make_feasible(np.clip(res.x, 0, 1), n)
    return h_new

def slsqp_optimize(h0, T, n, maxiter=300, ftol=1e-14):
    """SLSQP with equality constraint."""
    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, T) * (2.0 / n)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

    for sweep in range(max_sweeps):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, T):
    m = np.max(vals)
    if T < 1e-4:
        return m
    shifted = T * (vals - m)
    shifted = np.where(shifted > -80, shifted, -80)
    return m + np.log(np.sum(np.exp(shifted))) / T

        h += amp * np.cos(m * np.pi * x + phase)
    return h

        corr = np.correlate(h, 1.0 - h, mode='full')

    def obj_jac(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        w = np.exp(T * (corr - m))
        w /= np.sum(w)
        dx = 2.0 / n
        grad = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-10:

            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = i - k + n - 1
                if 0 <= j < n:
                    grad[i] += w[k] * (1 - h[j])
                    grad[j] -= w[k] * h[i]
        return dx * grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    res = minimize(obj, h0, jac=obj_jac, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})

                    improved = True
                    break
        if not improved:
            break
