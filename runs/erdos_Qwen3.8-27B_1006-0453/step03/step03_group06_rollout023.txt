        h_interp = project_feasible(h_interp, n_try)
        # Binary at this resolution
        h_ib = binary_from_continuous(h_interp, n_try)
        h_ib, c5_ib = binary_greedy_search(h_ib, n_try, 200, rng)
        if c5_ib < best_c5:
            best_c5 = c5_ib
            best_h = h_ib.copy()
        # SLSQP at this resolution
        if elapsed() < budget_s * 0.90:
            h_interp, c5_int = slsqp_refine(h_interp, n_try, [500, 3000], [None, 10], 500)
            if c5_int < best_c5:
                best_c5 = c5_int
                best_h = h_interp.copy()

    # Phase 6: Final polish (10% budget)
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = slsqp_refine(best_h, n_cur, [10000, 50000], [None, 3, 5], 1500)
    # Final nudge
    if elapsed() < budget_s * 0.99:
        h_ng, c5_ng = nudge_search(best_h, n_cur, 5000, 8, rng)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
