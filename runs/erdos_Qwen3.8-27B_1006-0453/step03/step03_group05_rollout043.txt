    """Build B_s matrices for all aperiodic shifts s = -(n-1), ..., 0, ..., n-1"""
    # For shift s: (B_s)[j, j+s] = 0.5 for valid j, symmetric
    shifts = np.arange(-(n-1), n)
    B_list = []
    for s in shifts:

        if s >= 0:
            for j in range(n - s):
                B[j, j+s] = 0.5
                B[j+s, j] = 0.5
        else:
            for j in range(n + s):
                B[j, j-s] = 0.5
                B[j-s, j] = 0.5
        B_list.append(B)

            if total < 1e-15: break
            h -= h * (-diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def ones_jac(x):
    return np.ones(len(x))

def smax_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float((m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n))

def smax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14: continue
        l = k_idx - (n - 1)
        if l >= 0:
            if l < n:
                grad[l:] += wk * (1.0 - h[:n-l])
                grad[:n-l] -= wk * h[l:]
        else:
            s = -l
            if s < n:
                grad[:n-s] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:n-s]
    return grad * dx

def slsqp_smooth(h, T, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': ones_jac}]
    try:
        res = minimize(smax_obj, h.copy(), jac=smax_grad, args=(T,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h

def linearized_lp(h, n, K, relax=0.5):
    """Linearized alternating: min t s.t. linearized corr constraints"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top K shifts
    top_idx = np.argpartition(corr, -K)[-K:]
    # Build linear constraints: corr_s(h_new) ≈ sum_j c_j^(s) * h_new[j]
    # where c_j^(s) = 1 - h_current[j - l_s]
    A_rows = []
    for ki in top_idx:
        l = int(ki - (n - 1))
        c = np.zeros(n)
        for j in range(n):
            jm = j - l
            # c[j] is the coefficient of h[j] in the linearized corr
            # corr_s ≈ sum_j h[j] * (1-h_current[j-l])
            # But that's just h[j] * c[j] where c[j] = 1-h_current[j-l]
            # Wait, that makes the constraint: sum_j h[j]*(1-h_cur[j-l])*dx <= t
            # which is: c^T h * dx <= t
            # c[j] = 1-h_cur[j-l] if 0<=j-l<n else 0
            if 0 <= jm < n:
                c[j] = 1.0 - h[jm]
            else:
                # For the other part: corr also has -h[j+l] terms
                # Actually no. The correlation is sum_j h[j]*(1-h[j-l]).
                # Linearizing: fix (1-h[j-l]) at current value.
                # This is an upper bound since we're not optimizing h in that factor.
                c[j] = 0.0  # Actually h[j] appears only in the first factor
        # Hmm wait. corr_s = sum_j h[j] * (1-h[j-l])
        # This is bilinear. If I linearize by fixing (1-h[j-l]) → c_j:
        # corr_s ≈ sum_j c_j * h_j
        # Then the constraint is: sum_j c_j * h_j * dx <= t
        A_rows.append(c * dx)
    
    # LP: min t
    # Variables: [h_0, ..., h_{n-1}, t]  (n+1 vars)
    # Objective: minimize t → c_obj = [0,...,0,1]
    # Inequality: A_rows[i] @ h + (-1)*t <= 0 for each i
    # Equality: sum h = n/2
    # Bounds: 0<=h<=1, t>=0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    for i in range(K):
        A_ub[i, :n] = A_rows[i]
        A_ub[i, -1] = -1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n]
