    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(corr, T):
    m = np.max(corr)
    if T > 1e8:
        return m
    shifted = T * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / T

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        mask = h > 1e-10
        if mask.sum() > 0:
            scale = (target - h[~mask].sum()) / h[mask].sum()
            h[mask] = h[mask] * min(scale, 2.0)
    return np.clip(h, 0.0, 1.0)

def obj_pen(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = smooth_max(corr, T)
    s = h.sum() - n / 2.0
    return sm * dx + lam * s * s

def lbfgsb_step(h, T, lam, maxiter=2000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, h, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 30})
    return project_h(np.clip(res.x, 0, 1), n)

def slsqp_step(h, T, n, maxiter=1500, ftol=1e-15):
    def obj(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * (2.0 / n)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(np.clip(res.x, 0, 1), n)

def coord_1d(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    corr_b = np.correlate(h0, 1.0 - h0, mode='full')
    coef = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        i1 = i + m - (n - 1)
        i2 = i - m + (n - 1)
        c = 0.0
        if 0 <= i1 < n:
            c += 1.0 - h[i1]
        if 0 <= i2 < n:
            c += -h[i2]
        coef[m] = c
    dx = 2.0 / n
    def f1(hi):
        return np.max(corr_b + hi * coef) * dx
    res = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 300})
    return res.x

def coord_2d(h, n, i, j):
    h0 = h.copy()
    h0[i] = 0.0
    h0[j] = 0.0
    corr_b = np.correlate(h0, 1.0 - h0, mode='full')
    c_i = np.zeros(2 * n - 1)
    c_j = np.zeros(2 * n - 1)
    c_ij = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        for idx_hi in [i, j]:
            for idx_hj in [i, j]:
                i1 = idx_hi + m - (n - 1)
                i2 = idx_hi - m + (n - 1)
                if 0 <= i1 < n:
                    if idx_hj == i:
                        c_i[m] += -h[i1] if i1 != idx_hj else -(1-h[i1])
                        # derivative: (1-h[i1]) wrt h[idx_hj] if i1==j, or 0
                    pass
        # Simpler: just compute gradient and cross term directly
    # Actually for 2D it's complex; use direct evaluation
    dx = 2.0 / n
    def f2(xy):
        ht = h.copy()
        ht[i] = xy[0]
        ht[j] = xy[1]
        return compute_c5(ht)
    res = minimize(f2, [h[i], h[j]], method='Nelder-Mead',
                   bounds=[(0,1),(0,1)], options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-14})
    return np.clip(res.x, 0, 1)

def coord_descent(h, n, t_end, use_pairs=False, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            hi = coord_1d(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            c5t = compute_c5(best_h)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                improved = True
        if not improved and not use_pairs:
            break
    return best_h, best_c5

def pair_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(n // 2):
            if time.time() > t_end:
                break
            i, j = np.random.choice(n, 2, replace=False)
            res = coord_2d(best_h, n, i, j)
            best_h[i] = res[0]
            best_h[j] = res[1]
            c5t = compute_c5(best_h)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                improved = True
    return best_h, best_c5

def sa_polish(h, n, t_end, step0=0.02, cool=0.0005):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5
    t = time.time()
    steps = 0
    while time.time() < t_end:
        steps += 1
        i = np.random.randint(n)
        delta = np.random.randn() * step0
        h_try = best_h.copy()
        h_try[i] = np.clip(h_try[i] + delta, 0, 1)
        h_try = project_h(h_try, n)
        c5t = compute_c5(h_try)
        if c5t < cur_c5 or np.random.rand() < np.exp(-(c5t - cur_c5) / max(cool, 1e-8)):
            cur_c5 = c5t
            best_h = h_try
            if c5t < best_c5:
                best_c5 = c5t
        step0 *= 0.9995
        cool *= 0.999
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B with increasing T (fast exploration)
    for T in [5, 15, 50, 200, 800, 3000, 10000]:
        if elapsed() > budget_s * 0.18:
            break
        for lam in [1e3, 1e5, 1e6]:
            if elapsed() > budget_s * 0.18:
                break
            h_new = lbfgsb_step(best_h, T, lam, maxiter=1500)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 2: SLSQP with equality constraint for precision
    for T in [500, 2000, 8000, 30000]:
        if elapsed() > budget_s * 0.30:
            break
        h_new = slsqp_step(best_h, T, n, maxiter=2000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 3: Basin hopping with L-BFGS-B
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(200):
        if elapsed() > budget_s * 0.52:
            break
        cool_frac = 1.0 - 0.9 * (it / 200.0)
        step = 0.02 + 0.12 * cool_frac
        h_pert = work_h + step * np.random.randn(n)
        h_pert = project_h(h_pert, n)
        T_bh = 1000
        h_new = lbfgsb_step(h_pert, T_bh, 1e5, maxiter=800)
        c5 = compute_c5(h_new)
        temp = 0.004 * cool_frac + 0.0002
        if c5 < work_c5 or np.random.rand() < np.exp(-max(0, c5 - work_c5) / temp):
            work_h = h_new
            work_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Coordinate descent (exact 1D)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.70, max_sweeps=150)

    # Phase 5: Pair coordinate descent
    best_h, best_c5 = pair_descent(best_h, n, t0 + budget_s * 0.80, max_sweeps=30)

    # Phase 6: Multi-resolution exploration
    for n_try in [120, 140, 180, 200]:
        if elapsed() > budget_s * 0.88:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_h(h_int, n_try)
        for T in [500, 3000]:
            if elapsed() > budget_s * 0.88:
                break
            h_new = lbfgsb_step(h_int, T, 1e5, maxiter=1000)
            h_new = slsqp_step(h_new, 5000, n_try, maxiter=1000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
                n = n_try

    # Phase 7: SA polish + final SLSQP
    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = sa_polish(best_h, n, t0 + budget_s * 0.93)
    for T in [50000, 200000]:
        if elapsed() > budget_s * 0.97:
            break
        h_new = slsqp_step(best_h, T, n, maxiter=2000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Final cleanup
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
