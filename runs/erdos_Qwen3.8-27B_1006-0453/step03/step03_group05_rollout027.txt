    idx = np.argpartition(corr, -top_k)[-top_k:]
    wts = corr[idx].astype(float)
    wts = wts - wts.min() + 1e-8
    wts /= wts.sum()
    
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        s = int(ki - (n - 1))
        if s >= 0:
            if s < n:
                grad[s:] += w * (1.0 - h[:n-s])
            if s < n:
                grad[:n-s] -= w * h[s:]
        else:
            m = -s
            if m < n:
                grad[:n-m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n-m]
    return grad

def projected_grad_descent(h, n, n_steps, top_k, alpha0, decay, cooldown):
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for step in range(n_steps):
        if no_improve > cooldown:
            break
        grad = subgrad_c5(h, n, top_k)
        h_new = project_feasible(h - alpha * grad, n)

            best = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            alpha *= 0.7
        alpha = max(alpha * decay, 0.001)
    return best, best_c5

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def slsqp_optimize(h, obj_func, args, maxiter, ftol, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def block_swap(h, n, rng):
    """Swap two blocks of h."""
    h_new = h.copy()
    block_size = rng.randint(3, max(4, n // 8))
    i = rng.randint(0, n - block_size)
    j = rng.randint(0, n - block_size)
    if abs(i - j) < block_size:
        return h_new
    block_i = h_new[i:i+block_size].copy()
    block_j = h_new[j:j+block_size].copy()
    h_new[i:i+block_size] = block_j
    h_new[j:j+block_size] = block_i
    return project_feasible(h_new, n)

def local_flip(h, n, rng):

    start = rng.randint(0, n - length)
    h_new[start:start+length] = 1.0 - h_new[start:start+length]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:

            total = room.sum()
            if total < 1e-14:

            h = np.clip(h + room * (diff / total), 0, 1)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h = np.clip(h * (1.0 + diff / total), 0, 1)

    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def slsqp_opt(h, obj_func, args, maxiter, ftol, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def structured_perturb(h, n, rng, mode):
    if mode == 0:
        h_new = h.copy()
        bl = rng.randint(3, max(4, n // 8))
        i = rng.randint(0, n - bl)
        j = rng.randint(0, n - bl)
        if abs(i - j) < bl:
            return h_new
        bi = h_new[i:i+bl].copy()
        bj = h_new[j:j+bl].copy()
        h_new[i:i+bl] = bj
        h_new[j:j+bl] = bi
    elif mode == 1:
        h_new = h.copy()
        ln = rng.randint(2, max(3, n // 6))
        st = rng.randint(0, n - ln)
        h_new[st:st+ln] = 1.0 - h_new[st:st+ln]
    elif mode == 2:
        sc = 0.03 + 0.02 * rng.random()
        h_new = h + rng.randn(n) * sc
    else:
        h_new = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst = int(np.argmax(corr) - (n - 1))
        m = abs(worst)
        if m < n and m > 0:
            if worst >= 0:
                for i in range(min(m, n - m)):
                    diff_v = h[i+m] - h[i]
                    if diff_v > 0.05:
                        sw = min(diff_v * 0.4, 0.15)
                        h_new[i+m] -= sw
                        h_new[i] += sw
            else:
                for i in range(n - m):
                    diff_v = h[i] - h[i+m]
                    if diff_v > 0.05:
                        sw = min(diff_v * 0.4, 0.15)
                        h_new[i] -= sw
                        h_new[i+m] += sw
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient with progressive focusing
    configs = [(25, 0.025, 400), (12, 0.02, 350), (8, 0.015, 300),
               (5, 0.012, 300), (3, 0.008, 350), (2, 0.005, 400)]
    for tk, a0, ns in configs:
        if time.time() - t0 > budget_s * 0.18:
            break
        hg, cg = projected_grad_desc(best_h, best_n, ns, tk, a0, 0.9985, 60)
        if cg < best_c5:
            best_c5, best_h = cg, hg

    # Phase 2: SLSQP top-k, decreasing k
    for k in [10, 7, 5, 4, 3, 2]:
        if time.time() - t0 > budget_s * 0.35:
            break
        ho = slsqp_opt(best_h, topk_obj, (k, best_n), 300, 1e-12, best_n)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 3: Softmax SLSQP, increasing T
    for T in [300, 1000, 3000, 10000, 30000]:
        if time.time() - t0 > budget_s * 0.48:
            break
        ho = slsqp_opt(best_h, softmax_obj, (T, best_n), 250, 1e-12, best_n)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 4: Basin hopping
    elapsed = time.time() - t0
    remaining = budget_s * 0.55 - elapsed
    n_trials = int(remaining / 10) if remaining > 0 else 0
    for trial in range(max(0, n_trials)):
        if time.time() - t0 > budget_s * 0.72:
            break
        mode = int(rng.random() * 4)
        sc_factor = 1.0 - 0.5 * trial / max(1, n_trials)
        h_p = structured_perturb(best_h, best_n, rng, mode)
        if mode == 2:
            h_p = project_feasible(best_h + rng.randn(best_n) * 0.03 * sc_factor, best_n)
        h_r = slsqp_opt(h_p, topk_obj, (4, best_n), 150, 1e-11, best_n)
        h_r = slsqp_opt(h_r, topk_obj, (2, best_n), 150, 1e-12, best_n)
        cr = compute_c5(h_r)
        if cr < best_c5:
            best_c5, best_h = cr, h_r

    # Phase 5: Multi-resolution
    for nr in [250, 300, 150]:
        if time.time() - t0 > budget_s * 0.88:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, nr)
        h_hi = slsqp_opt(h_hi, topk_obj, (3, nr), 200, 1e-11, nr)
        h_hi = slsqp_opt(h_hi, softmax_obj, (3000, nr), 150, 1e-11, nr)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, h_hi, nr
        x_dn = np.linspace(0, 2, best_n)
        h_dn = np.interp(x_dn, x_new, h_hi)
        h_dn = project_feasible(h_dn, best_n)
        c5_dn = compute_c5(h_dn)
        if c5_dn < best_c5:
            best_c5, best_h = c5_dn, h_dn

    # Phase 6: Intensive local refinement
    if time.time() - t0 < budget_s * 0.95:
        for _ in range(4):
            if time.time() - t0 > budget_s * 0.95:
                break
            ho = slsqp_opt(best_h, topk_obj, (2, best_n), 400, 1e-14, best_n)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho
                break
            ho = slsqp_opt(best_h, topk_obj, (1, best_n), 300, 1e-14, best_n)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho
                break

    # Phase 7: Final subgradient polish
    if time.time() - t0 < budget_s * 0.99:
        hf, cf = projected_grad_desc(best_h, best_n, 200, 2, 0.003, 0.999, 50)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hf = slsqp_opt(best_h, topk_obj, (2, best_n), 200, 1e-15, best_n)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
