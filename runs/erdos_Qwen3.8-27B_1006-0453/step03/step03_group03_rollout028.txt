def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            room = h.copy()
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    s = np.clip(T * (corr - m), -400, 0)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx

def slsqp_smooth(h, T, maxiter=600):
    n = len(h)
    def cons_eq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=[{'type': 'eq', 'fun': cons_eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
