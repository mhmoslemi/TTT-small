    m_s = np.zeros(N)
    v_s = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Compute softmax-smoothed gradient
        corr = np.correlate(h, 1.0-h, mode='full') * dx
        maxc = corr.max()
        exp_c = np.exp((corr - maxc) * tau)
        w = exp_c / exp_c.sum()
        # Gradient

            wk = w[k]
            if wk < 1e-12:

            lo1 = max(0, -off)

            hi2 = min(N, N + off)

        m_s = b1*m_s + (1-b1)*grad
        v_s = b2*v_s + (1-b2)*grad*grad

        if c5_val < best_c5 - 1e-12:
