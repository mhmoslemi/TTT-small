    dx = 2.0 / n
    # Compute all D(k) values
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    D = corr * dx  # length 2n-1, D[i] corresponds to lag i-(n-1)
    
    # Softmax weights
    D_shifted = D - np.max(D)
    w = np.exp(T * D_shifted)
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(n):
        s = 0.0
        for i in range(2*n-1):
            k = i - (n-1)  # lag
            # (1-h[m-k]) term
            if 0 <= m-k < n:
                s += w[i] * (1.0 - h[m-k])
            # -h[m+k] term
            if 0 <= m+k < n:
                s -= w[i] * h[m+k]
        grad[m] = s * dx
    
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += h * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    D = corr * dx
    m = np.max(D)
    exp_T = np.exp(T * (D - m))
    w = exp_T / exp_T.sum()
    val = m + np.log(exp_T.sum()) / T
    grad = np.zeros(n)
    for mm in range(n):
        acc = 0.0
        for i in range(2 * n - 1):
            k = i - (n - 1)
            ip = mm + k
            if 0 <= ip < n:
                acc += w[i] * (1.0 - h[ip])
            im = mm - k
            if 0 <= im < n:
                acc -= w[i] * h[im]
        grad[mm] = dx * acc
    return val, grad

def slsqp_opt(h0, T, n, maxiter=400, ftol=1e-16):
    def func(x):
        xc = np.clip(x, 0.0, 1.0)
        return smooth_obj_grad(xc, T)
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(func, h0, jac=True, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def top_k_lags(h, K=10):
    corrs = get_corrs(h)
    idx = np.argsort(corrs)[::-1][:K]
    return idx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        best_h = np.full(200, 0.5)
        best_h = make_feasible(best_h, 200)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Analytical-gradient SLSQP at increasing T (12% budget)
    n_cur = len(best_h)
    for T in [100, 400, 1500, 5000, 20000]:
        if elapsed() > budget_s * 0.12:
            break
        h_new = slsqp_opt(best_h.copy(), T, n_cur, 500, 1e-17)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 2: Multi-resolution sweep (18% budget)
    for n_try in [100, 120, 140, 150, 160, 180, 240, 300]:
        if elapsed() > budget_s * 0.30:
            break
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [500, 3000]:
            if elapsed() > budget_s * 0.30:
                break
            h_new = slsqp_opt(h_interp.copy(), T, n_try, 350, 1e-16)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
                n_cur = n_try
                x_old = np.linspace(0, 2, n_cur, endpoint=False)

    # Phase 3: Basin hopping with gradient-guided perturbations (52% budget)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    end_bh = budget_s * 0.82
    it = 0
    while elapsed() < end_bh:
        ptype = it % 5
        cool = 1.0 - min(elapsed() / end_bh, 1.0)
        step = 0.08 * (0.3 + 0.7 * cool) + 0.01
        if ptype == 0:
            h_pert = bh_h + step * np.random.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(3, n_bh // 10)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1+blen].copy()
            h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
            h_pert[s2:s2+blen] = tmp
        elif ptype == 2:
            # Gradient-guided perturbation
            _, grad = smooth_obj_grad(bh_h, 500)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-10:
                h_pert = bh_h - step * grad / gnorm * np.sqrt(n_bh)
            else:
                h_pert = bh_h + step * np.random.randn(n_bh)
        elif ptype == 3:
            # Flip-based
            h_pert = bh_h.copy()
            nflip = max(2, n_bh // 10)
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = np.clip(h_pert[idx] + np.random.uniform(-0.3, 0.3, nflip), 0, 1)
        else:
            # Block shift
            h_pert = bh_h.copy()
            bsize = max(4, n_bh // 8)
            offset = np.random.randint(1, bsize)
            h_pert[:] = np.roll(bh_h, offset)
            h_pert += 0.05 * np.random.randn(n_bh)

        h_pert = make_feasible(h_pert, n_bh)
        T_use = 300 + int(2000 * cool)
        h_new = slsqp_opt(h_pert, T_use, n_bh, 250, 1e-15)
        c5v = compute_c5(h_new)

        temp = 0.004 * (0.2 + 0.8 * cool) + 0.0002
        if c5v < bh_c5 or (c5v > bh_c5 and np.random.rand() < np.exp(-(c5v - bh_c5) / temp)):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        it += 1

    best_h = bh_h if bh_c5 < best_c5 else best_h
    best_c5 = min(bh_c5, best_c5)
    n_cur = len(best_h)

    # Phase 4: Targeted dominant-lag optimization (12% budget)
    for round in range(5):
        if elapsed() > budget_s * 0.94:
            break
        corrs = get_corrs(best_h)
        top_idx = np.argsort(corrs)[::-1][:15]
        T = 5000 + round * 5000
        h_new = slsqp_opt(best_h.copy(), T, n_cur, 600, 1e-18)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 5: Final ultra-high-T polish (6% budget)
    for T in [50000, 200000, 1000000]:
        if elapsed() > budget_s * 0.985:
            break
        h_new = slsqp_opt(best_h.copy(), T, n_cur, 2000, 1e-18)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 6: Coordinate descent on top lags
    if elapsed() < budget_s * 0.97:
        n_cd = len(best_h)
        for _ in range(200):
            if elapsed() > budget_s * 0.97:
                break
            improved = False
            for _ in range(30):
                j = np.random.randint(0, n_cd)
                delta = np.random.choice([-0.02, -0.05, 0.05, 0.02, -0.08, 0.08])
                h_try = best_h.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                h_try = make_feasible(h_try, n_cd)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
