    s = float(h.sum())
    if abs(s - target) < 1e-12:

    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid

def adam_refine(h, N, target, deadline, k=20, lr0=0.05, decay=0.99):
    best_h = h.copy()
    best_pk = float(np.max(np.correlate(h, 1.0-h, mode='full')))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(corr.max())
        if peak < best_pk:
            best_pk = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N-L])
                grad[:N-L] -= wk * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += wk * (1.0 - h[M:])
                    grad[M:] -= wk * h[:N-M]
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    return best_h, best_pk * (2.0 / N)

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def adam_refine(h, N, target, deadline, k=25, lr0=0.06):

    v = np.zeros(N)
