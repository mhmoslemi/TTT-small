import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_grad(h, N, temp, max_active=60):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    c_shifted = corr - peak
    w = np.exp(c_shifted / temp)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= w_sum
        if n_active := int((w > 1e-10).sum()) > max_active:
            thresh = np.sort(w)[::-1][max_active]
            w[w < thresh] = 0.0
            w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 1e-8)[0]
    for ai in active:
        wi = w[ai]
        L = ai - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            nv = N - L
            grad[L:] += wi * (1.0 - h[:nv])
            grad[:nv] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                nv = N - M
                grad[:nv] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:nv]
    return grad, peak

def topk_grad(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k

        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            nv = N - L
            grad[L:] += wk * (1.0 - h[:nv])
            grad[:nv] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                nv = N - M
                grad[:nv] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:nv]
    return grad, peak

def adam_hc(h, N, target, deadline, lr, k0, k1, t0, t1, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(1.0, it / 3000.0)
        k_eff = max(1, int(k0 + (k1 - k0) * frac))
        temp = t0 * (t1 / t0) ** min(1.0, it / 4000.0)
        if k_eff >= 15:
            grad, peak = topk_grad(h, N, k_eff)
        else:
            grad, peak = softmax_grad(h, N, temp)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 400 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)
