import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(corr, x, n):
    """Compute d(corr[k])/dx as (2n-1, n) structure implicitly via loop."""
    return corr

def obj_grad_smooth(x, T, n, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    Tc = T * (c - m)
    Tc_max = np.max(Tc)
    exp_Tc = np.exp(Tc - Tc_max)
    wts = exp_Tc / exp_Tc.sum()
    val = (m + (Tc_max + np.log(exp_Tc.sum())) / T) * dx
    s_sum = x.sum()
    val += 0.5 * mu * (s_sum - n / 2.0) ** 2
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        k = s + n - 1
        w = wts[k]
        if w < 1e-16:
            continue
        if s >= 0:
            if s < n:
                g[s:] += w * (1.0 - x[:n - s])
                g[:n - s] -= w * x[s:]
        else:
            ss = -s
            g[:n - ss] += w * (1.0 - x[ss:])
            g[ss:] -= w * x[:n - ss]
    g *= dx
    g += mu * (s_sum - n / 2.0) * np.ones(n)
    return val, g

def obj_grad_topk(x, k, n, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(c, -k)[-k:]
    topk_val = np.mean(c[idx])
    val = topk_val * dx
    s_sum = x.sum()
    val += 0.5 * mu * (s_sum - n / 2.0) ** 2
    g = np.zeros(n)

            g[ss:] -= x[:n - ss]

def lbfgs_run(x0, obj_grad_func, n, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad_func, x0, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(x0, n)

            hp[s:e] = np.roll(hp[s:e], sh)
            hp = project_feasible(hp, best_n)
        elif r < 0.65:
            hp = binary_proj(best_h + np.random.randn(best_n) * 0.25, best_n)
        elif r < 0.80:
            # Targeted: push down at worst-shift conflicts
            c = np.correlate(best_h, 1.0 - best_h, mode='full')
            sw = int(np.argmax(c) - (best_n - 1))
            hp = best_h.copy()
            if sw >= 0:
                for i in range(best_n - sw):
                    if hp[i] > 0.5 and hp[i + sw] < 0.5:
                        hp[i] -= 0.1
                        hp[i + sw] += 0.1
            else:
                for i in range(-sw, best_n):
                    if hp[i] > 0.5 and hp[i + sw] < 0.5:
                        hp[i] -= 0.1
                        hp[i + sw] += 0.1
            hp = project_feasible(hp, best_n)
        else:
            hp = project_feasible(np.clip(best_h * (0.7 + 0.6 * np.random.random(best_n)), 0, 1), best_n)

        f1 = make_smooth_obj(500, best_n, mu)
        h_new = lbfgs_run(hp, f1, best_n, maxiter=200)
        f2 = make_topk_obj(4, best_n, mu)
        h_new = lbfgs_run(h_new, f2, best_n, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new
        b = binary_proj(h_new, best_n)
        bc5 = compute_c5(b)
        if bc5 < best_c5:
            best_c5, best_h = bc5, b

    # Phase 5: Multi-resolution
    for nr in [100, 150, 300, 400]:
        if time.time() - t0 > budget_s * 0.90:
            break
        x_old = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        x_new = (np.arange(nr) + 0.5) * (2.0 / nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        f = make_smooth_obj(800, nr, mu)
        hh = lbfgs_run(hu, f, nr, maxiter=250)
        f2 = make_topk_obj(5, nr, mu)
        hh = lbfgs_run(hh, f2, nr, maxiter=250)
        c5_hi = compute_c5(hh)
        if c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, hh, nr
        else:
            # Downsample back and compare
            x_dn = (np.arange(best_n) + 0.5) * (2.0 / best_n)
            hd = np.interp(x_dn, x_new, hh)
            hd = project_feasible(hd, best_n)
            c5_dn = compute_c5(hd)
            if c5_dn < best_c5:
                best_c5, best_h = c5_dn, hd

    # Phase 6: Final intensive polish
    if time.time() - t0 < budget_s * 0.97:
        for _ in range(4):
            if time.time() - t0 > budget_s * 0.97:
                break
            f = make_topk_obj(3, best_n, mu)
            h_new = lbfgs_run(best_h, f, best_n, maxiter=600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new
            f2 = make_smooth_obj(50000, best_n, mu)
            h_new2 = lbfgs_run(best_h, f2, best_n, maxiter=600)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < best_c5:
                best_c5, best_h = c5_new2, h_new2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
