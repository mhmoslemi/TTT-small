def grad_softmax(h, tau, K):
    """Gradient of (1/tau)*log(sum(exp(tau*f_k))) w.r.t. h, where f_k=corr[k].

        h2, c2 = optimize(h1, N, target, dx, p2,
                          400.0, 5000.0, 0.006, 0.0006,
                          60, 5, 600)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
        # Phase 3: ultra-sharp focus
        if time.time() > deadline - 20:
            break
        p3 = time.time() + (deadline - time.time()) * 0.25
        h3, c3 = optimize(h2, N, target, dx, p3,
                          5000.0, 80000.0, 0.0015, 0.00008,
                          5, 1, 500)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
        # Phase 4: directed single-k fix
        if time.time() < deadline - 12:
        hd, cd = directed_fix(h3, N, target, dx, time.time() + (deadline - time.time()) * 0.2, 400)
            if cd < best_c5:
                best_c5 = cd
                best_h = hd.copy()
        # Phase 5: local random search
        if time.time() < deadline - 8:
            hs, cs = local_search(h3, N, target, dx, time.time() + (deadline - time.time()) * 0.3,
                                  4000, 0.008)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()
    # Final ultra-polish at best size
    if time.time() < deadline - 2:
        Nb = len(best_h)
        dx_b = 2.0 / Nb
        hf, cf = optimize(best_h.copy(), Nb, Nb // 2, dx_b, deadline - 1,
                          10000.0, 200000.0, 0.0004, 0.00002,
                          3, 1, 300)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
