            lo = mid

def grad_lag(h, N, k):
    g = np.zeros(N)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:N-k])
        g[:N-k] -= h[k:]
    else:
        M = -k
        g[:N-M] += (1.0 - h[M:])
        g[M:] -= h[:N-M]

def softmax_grad(h, N, topk=8, temp=0.3):
    corr = corr_full(h)
    top_idx = np.argpartition(corr, -topk)[-topk:]
    vals = corr[top_idx]
    w = np.exp((vals - vals.max()) / temp)
    w /= w.sum()
    g = np.zeros(N)
    center = N - 1
    for wi, idx in zip(w, top_idx):
        g += wi * grad_lag(h, N, int(idx - center))
    return float(np.max(corr)), g

def adam_opt(h, N, deadline, lr0=0.02, decay=0.995, topk=8, temp=0.3):
    best_h = h.copy()
    best_pk = float(np.max(corr_full(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        _, g = softmax_grad(h, N, topk, temp)
        it += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project(h, N / 2.0)
        if it % 20 == 0:
            pk = float(np.max(corr_full(h)))
            if pk < best_pk:
                best_pk = pk
                best_h = h.copy()
            lr *= decay
    return best_h, best_pk * dx

        T = T0 * (T1 / T0) ** (it / n_total)
        ones = np.where(hb == 1)[0]

        j = zeros[rng.randint(len(zeros))]

        new_pk = float(np.max(corr_full(hb)))

                best_pk = new_pk
                best = hb.copy()
        else:

    return best, best_pk * dx

    if pattern == 'rand':

            h[pos:pos+blen] = 1.0
            pos += blen
            blen2 = rng.randint(2, max(3, N//20))
            pos += blen2
    return h

- More precisely, corr[k] = sum_n h[n+k](1-h[n])
