import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h -= h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in range(n):
            if diff > 0:
                room = 1.0 - h[idx[i]]
                add = min(room, diff)
                h[idx[i]] += add
                diff -= add
            else:
                take = min(h[idx[i]], -diff)
                h[idx[i]] -= take
                diff += take
            if abs(diff) < 1e-14:
                break
    return h

def obj_smooth(h, T, n, pen):
    hc = np.clip(h, 0, 1)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    if T > 1:
        shifted = T * (corr - m)
        shifted = np.clip(shifted, -700, 700)
        val = m + np.log(np.sum(np.exp(shifted))) / T
    else:
        val = m
    val *= dx
    se = hc.sum() - n / 2.0
    val += pen * se * se
    return val

def obj_topk(h, k, n, pen):
    hc = np.clip(h, 0, 1)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    dx = 2.0 / n
    idx = np.argpartition(corr, -k)[-k:]
    val = np.sum(corr[idx]) * dx / k
    se = hc.sum() - n / 2.0
    val += pen * se * se
    return val

def topk_gradient(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(n)
    for ki in idx:
        s = ki - (n - 1)
        if s >= 0:
            valid = range(s, n)
            hs = h[s:n]
            h0 = h[0:n - s]
            grad[s:] += (1.0 - h0)
            h0b = h[:n - s]
            hb = h[s:]
            grad[:n - s] += (-hb)
        else:
            ss = -s
            valid = range(n - ss)
            h0 = h[:n - ss]
            hs = h[ss:]
            grad[:n - ss] += (1.0 - hs)
            hs2 = h[ss:]
            h0_ = h[:n - ss]
            grad[ss:] += (-h0_)
    grad /= k
    return grad

def lbgfs_opt(h, obj, args, bnds, maxit, ftol):
    try:
        res = minimize(obj, h, args=args, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxit, 'ftol': ftol, 'gtol': 1e-9})
        return project_feasible(np.clip(res.x, 0, 1), len(h))
    except Exception:
        return h

def gradient_ascent(h, n, k, n_steps, alpha0, decay):
    best = h.copy()
    best_c = compute_c5(h)
    alpha = alpha0
    for _ in range(n_steps):
        g = topk_gradient(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = project_feasible(h - alpha * g / gn, n)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best = h_new.copy()
        h = h_new
        alpha *= decay
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    bnds = [(0.0, 1.0)] * best_n

    # Phase 1: L-BFGS-B smooth max at various temperatures
    for T in [100, 500, 2000, 8000, 20000]:
        if time.time() - t0 > budget_s * 0.10:
            break

        cc = compute_c5(hh)
