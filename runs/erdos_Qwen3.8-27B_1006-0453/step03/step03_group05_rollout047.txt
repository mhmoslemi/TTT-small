def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:

            room = 1.0 - h

def targeted_shift_repair(h, n, top_k=5, n_iters=50):
    """Iteratively target the worst shifts"""
    h = h.copy()
    for _ in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top k worst shifts
        worst_idx = np.argsort(corr)[::-1][:top_k]
        for wi in worst_idx:
            s = int(wi - (n - 1))
            if s == 0:
                continue
            # Try to reduce overlap at this shift
            # For each pair (i, i+s) where h[i]*h[i+s] is contributing
            improvement = False
            for i in range(min(n, n - abs(s))):
                j = i + abs(s) if s > 0 else i
                ii = i if s > 0 else i + abs(s)
                jj = i + abs(s)
                if 0 <= ii < n and 0 <= jj < n:
                    # Contribution is h[ii]*(1-h[jj]) + (if symmetric) h[jj]*(1-h[ii])

                        h[jj] -= delta

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')

    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def sum_con(x):
