import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_onto_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_gradient(h, N, tau, topK=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    nc = len(corr)
    dx = 2.0 / N
    sc = tau * corr * dx
    sc -= sc.max()
    exp_sc = np.exp(sc)
    w_sum = exp_sc.sum()
    if w_sum < 1e-300:
        w = np.zeros(nc)
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_sc / w_sum
    
    if topK is not None and topK < nc:
        idx = np.argsort(corr)[-topK:]
        mask = np.zeros(nc, dtype=bool)
        mask[idx] = True
        w[mask] = w[mask] / w[mask].sum()
        w[~mask] = 0.0
    
    grad = np.zeros(N)
    active = np.where(w > 1e-10)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        elif 0 < -L < N:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    
    c5 = float(np.max(corr)) * dx
    return grad, c5

    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    while time.time() < deadline:
        frac = min(float(it) / 300.0, 1.0)
        tau_cur = tau_s * (tau_e / tau_s) ** frac
        lr_cur = lr * (0.85 ** (it / 150.0))
        topK_cur = topK_fn(frac)
        
        grad, c5_val = lse_gradient(h, N, tau_cur, topK_cur)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_constraint(h, target)
        it += 1
    c5_final = compute_c5(h, 2.0 / N)
    return h, c5_final

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_onto_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_gradient(h, N, tau, topK):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    nc = len(corr)
    dx = 2.0 / N
    sc = tau * corr * dx
    sc -= sc.max()
    exp_sc = np.exp(sc)
    w_sum = exp_sc.sum()
    if w_sum < 1e-300:
        w = np.zeros(nc)
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_sc / w_sum
    if topK is not None and topK < nc:
        idx = np.argsort(corr)[-topK:]
        mask = np.zeros(nc, dtype=bool)
        mask[idx] = True
        w = w * mask
        ws = w.sum()
        if ws > 0:
            w /= ws
    grad = np.zeros(N)
    active = np.where(w > 1e-10)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        elif 0 < -L < N:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    c5 = float(np.max(corr)) * dx
    return grad, c5

def topK_fn_b(frac):

def topK_fn_c(frac):

def topK_fn_d(frac):
    return max(3, int(6 - 3 * frac))

def adam_phase(h, N, target, tau_s, tau_e, lr0, tk_fn, deadline):
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    it = 0
    best_local = compute_c5(h, 2.0 / N)
    best_h_local = h.copy()
    stall = 0
    while time.time() < deadline - 2:
        frac = min(float(it) / 500.0, 1.0)
        tau_cur = tau_s * (tau_e / tau_s) ** frac
        lr_cur = lr0 * (0.88 ** (it / 120.0))
        topK_cur = tk_fn(frac)
        grad, c5_val = lse_gradient(h, N, tau_cur, topK_cur)
        if c5_val < best_local - 1e-13:
            best_local = c5_val
            best_h_local = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_constraint(h, target)
        it += 1
    return best_h_local, best_local

def polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    while time.time() < deadline - 1:
        improved = False
        for i in range(N):
            if time.time() > deadline - 1:
                break
            orig = h[i]
            for trial in (0.0, 1.0, orig - 0.004, orig + 0.004, orig - 0.015, orig + 0.015):
                if trial < 0.0 or trial > 1.0 or abs(trial - orig) < 1e-9:
                    continue
                h[i] = trial
                h = project_onto_constraint(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
            h = best_h.copy()
        if not improved:
            break
    return best_h, best_c5
