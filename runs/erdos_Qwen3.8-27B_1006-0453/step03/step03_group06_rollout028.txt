import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-12:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-12:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_c5(h, T):
    corr = compute_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / len(h))

def active_smooth_c5(h, pargs):
    T, idx = pargs
    corr = compute_corr(h)
    top = corr[idx]
    m = np.max(top)
    val = m + np.log(np.sum(np.exp(T * (top - m)))) / T
    return val * (2.0 / len(h))

def grad_smooth_max(h, T):
    n = len(h)
    corr = compute_corr(h)
    mx = np.max(corr)
    w = np.exp(T * (corr - mx))
    w /= w.sum()
    g = np.zeros(n)
    idx_all = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        j1_lo = max(0, sh)
        j1_hi = min(n, n + sh)
        if j1_hi > j1_lo:
            i1 = j1_lo - sh
            g[j1_lo:j1_hi] += wk * (1.0 - h[i1:i1 + (j1_hi - j1_lo)])
        j2_lo = max(0, -sh)
        j2_hi = min(n, n - sh)
        if j2_hi > j2_lo:
            i2 = j2_lo + sh
            g[j2_lo:j2_hi] -= wk * h[i2:i2 + (j2_hi - j2_lo)]
    return g

def gradient_descent(h, n, T, lr, n_steps, momentum=0.85):
    h = h.astype(float).copy()
    v = np.zeros(n)
    for s in range(n_steps):
        g = grad_smooth_max(h, T)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        h = h_new
    return h

def slsqp_active(h0, n, T, active_idx, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(active_smooth_c5, h0, args=(T, active_idx),
                  method='SLSQP', bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(np.clip(res.x, 0, 1), n)

def try_discrete_moves(h, n, n_moves=500):
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = compute_corr(best)
    top_shifts = np.argsort(corr)[-8:]

            best = h2
    return best, best_c5
