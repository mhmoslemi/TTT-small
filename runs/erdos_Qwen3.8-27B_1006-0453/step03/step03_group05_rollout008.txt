def project_feasible(h, n):

def grad_for_shifts(h, shifts):

    n = len(h)

            grad[:n-s] -= h[s:]
        else:
            t = -s
            grad[:n-t] += (1.0 - h[t:])
            grad[t:] -= h[:n-t]
    return grad

            else:

            h = h_new

            h = h_new
            # But track best

    return best, best_c5

def coord_descent(h, n_iters, top_k):

    n = len(h)
    best = h.copy()

def slsqp_wrap(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(x.sum() - len(x) / 2.0)}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-13:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_shift(h, s):
    """Gradient of correlation at shift s (s = k - (n-1)) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def subgrad_momentum(h, n_steps, top_k, alpha0, decay, momentum, focus=30):
    n = len(h)
    v = np.zeros(n)
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for it in range(n_steps):
        corr = compute_correlation(h)
        idx = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[idx]
        vmax = vals.max()
        wts = np.exp(focus * (vals - vmax))
        wts /= wts.sum()
        shifts = (idx - (n - 1)).astype(int)
        grad = np.zeros(n)
        for w, s in zip(wts, shifts):
            if w < 1e-6:
                continue
            grad += w * grad_for_shift(h, s)
        v = momentum * v - alpha * grad
        h_new = project_feasible(h + v, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        alpha *= decay
        if stall > 500 and alpha < 1e-5:
            break
    return best, best_c5

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def slsqp_wrap(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    s_target = n / 2.0
    def sum_con(x):
        return float(np.sum(x) - s_target)
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                       bounds=bnds, constraints=[{'type': 'eq', 'fun': sum_con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def make_initial(n, pattern=0):
    h = np.zeros(n)
    if pattern == 0:
        block = n // 4
        for i in range(0, n - 1, block * 2):
            h[i:min(i + block, n)] = 1.0
    elif pattern == 1:
        x = np.linspace(0, 2, n)
        h[:] = np.clip(0.5 + 0.35 * np.sin(4 * np.pi * x), 0, 1)
    elif pattern == 2:
        n6 = n // 6
        for i in range(0, n - 1, n6 * 2):
            h[i:min(i + n6, n)] = 1.0
    return project_feasible(h, n)

def targeted_attack(h, n, strength=0.15):
    corr = compute_correlation(h)
    top5 = np.argpartition(corr, -5)[-5:]
    hp = h.copy()
    for ki in top5:
        s = int(ki - (n - 1))
        if s >= 0:
            for i in range(n - s):
                if hp[i] > 0.6 and hp[i + s] < 0.4:
                    d = strength * min(hp[i] - 0.5, 0.5 - hp[i + s])
                    hp[i] -= d
                    hp[i + s] += d
        else:
            t = -s
            for i in range(t, n):
                if hp[i] > 0.6 and hp[i - t] < 0.4:
                    d = strength * min(hp[i] - 0.5, 0.5 - hp[i - t])
                    hp[i] -= d
                    hp[i - t] += d
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Initialize from parent or construct
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_initial(200, 0)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy subgradient with momentum (30% budget)
    sg_configs = [
        (3000, 5, 0.025, 0.9999, 0.5, 25),
        (3000, 3, 0.012, 0.9999, 0.7, 40),
        (2000, 8, 0.018, 0.9998, 0.4, 15),
        (2000, 2, 0.008, 0.99995, 0.8, 50),
    ]
    t_limit1 = t0 + budget_s * 0.30
    for cfg in sg_configs:
        if time.time() > t_limit1:
            break
        hsg, c5sg = subgrad_momentum(best_h, *cfg)
        if c5sg < best_c5:
            best_c5, best_h = c5sg, hsg

    # Phase 2: SLSQP polish (15%)
    t_limit2 = t0 + budget_s * 0.45
    for T in [300, 800, 2000]:
        if time.time() > t_limit2:
            break
        ht = slsqp_wrap(best_h, softmax_obj, (T, best_n), 400, 1e-12)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
    for k in [2, 4]:
        if time.time() > t_limit2:
            break
        ht = slsqp_wrap(best_h, topk_obj, (k, best_n), 350, 1e-12)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 3: Basin hopping + refine (30%)
    t_limit3 = t0 + budget_s * 0.75
    n_trials = 80
    for trial in range(n_trials):
        if time.time() > t_limit3:
            break
        r = np.random.random()
        if r < 0.25:
            # Block permutation
            hp = best_h.copy()
            start = np.random.randint(0, max(1, best_n - 5))
            length = np.random.randint(3, max(4, best_n // 5))
            end = min(start + length, best_n)
            shift = np.random.randint(1, max(2, length // 2 + 1))
            hp[start:end] = np.roll(hp[start:end], shift)
        elif r < 0.45:
            # Gaussian noise
            sc = 0.02 + 0.03 * np.random.random() * (1.0 - trial / n_trials)
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.60:
            # Targeted attack on worst shifts
            hp = targeted_attack(best_h, best_n, strength=0.1 + 0.1 * np.random.random())
        elif r < 0.75:
            # Swap a few pairs
            hp = best_h.copy()
            for _ in range(np.random.randint(2, 6)):
                i, j = np.random.choice(best_n, 2, replace=False)
                hp[i], hp[j] = hp[j], hp[i]
        elif r < 0.85:
            # Partial flip in a window
            hp = best_h.copy()
            w = np.random.randint(5, max(6, best_n // 4))
            start = np.random.randint(0, max(1, best_n - w))
            hp[start:start + w] = 1.0 - hp[start:start + w]
        else:
            # Hybrid: keep 80% of best_h, randomize 20%
            hp = best_h.copy()
            n_rand = max(3, best_n // 5)
            idx = np.random.choice(best_n, n_rand, replace=False)
            hp[idx] = np.random.uniform(0, 1, n_rand)
        hp = project_feasible(hp, best_n)
        # Two-stage refine
        ho = slsqp_wrap(hp, softmax_obj, (600, best_n), 200, 1e-11)
        co = compute_c5(ho)
        if co < best_c5:
            ho2 = slsqp_wrap(ho, topk_obj, (3, best_n), 250, 1e-12)
            co2 = compute_c5(ho2)
            if co2 < best_c5:
                best_c5, best_h = co2, ho2
            else:
                best_c5, best_h = co, ho

    # Phase 4: Multi-resolution (10%)
    t_limit4 = t0 + budget_s * 0.85
    for nr in [150, 240, 180, 300]:
        if time.time() > t_limit4:
            break
        if nr == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu_p = project_feasible(hu + np.random.randn(nr) * 0.01, nr)
        hh = slsqp_wrap(hu_p, softmax_obj, (400, nr), 150, 1e-11)
        hh = slsqp_wrap(hh, topk_obj, (3, nr), 200, 1e-12)
        c5_hi = compute_c5(hh)
        x_dn = np.linspace(0, 2, best_n)
        hd = np.interp(x_dn, x_new, hh)

    # Phase 5: Final intensive subgradient + SLSQP (10%)
    t_limit5 = t0 + budget_s * 0.95
    if time.time() < t_limit5:
        hs, cs = subgrad_momentum(best_h, 2000, 3, 0.005, 0.99995, 0.7, 60)
        if cs < best_c5:
            best_c5, best_h = cs, hs
    if time.time() < t_limit5:
        hf = slsqp_wrap(best_h, topk_obj, (2, best_n), 500, 1e-14)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    if time.time() < t_limit5:
        hf2 = slsqp_wrap(best_h, softmax_obj, (8000, best_n), 400, 1e-14)
        cf2 = compute_c5(hf2)

        if cs3 < best_c5:
            best_c5, best_h = cs3, hs3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
