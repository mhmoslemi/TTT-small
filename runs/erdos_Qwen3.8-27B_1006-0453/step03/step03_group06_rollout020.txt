                    best_h = hc2
            else:
                best_c5 = c5h
                best_h = hc

    # Phase 5: Targeted coordinate optimization on worst shifts
    if elapsed() < budget_s * 0.92:
        for _ in range(50):
            if elapsed() > budget_s * 0.92:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top5 = np.argsort(corr)[-5:]
            improved = False
            for k_sh in top5:
                shift = k_sh - (n_cur - 1)
                # Adjust coordinates in the overlap region
                lo = max(0, -shift)
                hi = min(n_cur, n_cur - shift) if shift >= 0 else min(n_cur, n_cur + shift)
                hi = min(n_cur, hi)
                for j in range(lo, hi, max(1, (hi-lo)//8)):
                    for dv in [0.03, -0.03, 0.06, -0.06]:
                        h_try = best_h.copy()
                        h_try[j] += dv
                        h_try = project_feasible(h_try, n_cur)
                        c_try = compute_c5(h_try)
                        if c_try < best_c5 - 1e-13:
                            best_c5 = c_try
                            best_h = h_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

    # Phase 6: Final SLSQP polish with high T
    n_cur = len(best_h)
    for T in [3000, 10000, 30000]:
        if elapsed() > budget_s * 0.99:
            break
        hc = slsqp_polish(best_h, T, n_cur, None, 2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    for k in [3, 5, 8]:
        if elapsed() > budget_s * 0.995:
            break
        hc = slsqp_polish(best_h, 5000, n_cur, k, 1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
