import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-9:
        return h.copy()
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, N, weights):
    center = N - 1
    grad = np.zeros(N)
    thresh = 1e-4
    active = np.where(weights > thresh)[0]
    for ai in active:
        L = ai - center
        wi = weights[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad

Let me also think about whether the binary swap approach can be made more efficient. Instead of random swaps, I can try ALL possible single-coordinate flips (1→0 or 0→1) and pick the best one. This is coordinate descent on the binary structure.

For a binary vector with exactly N/2 ones:
- A single coordinate flip (1→0) changes the sum by -1, violating the constraint.
- So we need paired flips: (1→0) at position a AND (0→1) at position b.

The number of possible swaps is (N/2) * (N/2) = N²/4. For N=380, that's 36,100 swaps. Each swap requires computing the new C₅, which is O(N²) with correlate. Total: 36,100 * 144,000 ≈ 5.2 * 10⁹ operations. That's too slow.

But we can be smarter: for a given swap (a, b), the change in f_k is:
Δf_k = [new terms at a and b] - [old terms at a and b]

For shift k > 0:
Old contribution from position a (which was 1): 
- As h[a]: h[a]*(1-h[a+k]) = 1*(1-h[a+k]) = 1-h[a+k] (if a+k < N)
- As 1-h[a] in h[a-k]*(1-h[a]): h[a-k]*0 = 0 (if a-k >= 0)

New contribution from position a (which becomes 0):
- As h[a]: 0*(1-h[a+k]) = 0
- As 1-h[a]: h[a-k]*1 = h[a-k] (if a-k >= 0)

Similarly for position b (0→1):
Old: h[b]*(1-h[b+k]) = 0, and h[b-k]*(1-h[b]) = h[b-k]
New: h[b]*(1-h[b+k]) = 1-h[b+k], and h[b-k]*(1-h[b]) = 0

So Δf_k = [-(1-h[a+k]) + h[a-k]] + [(1-h[b+k]) - h[b-k]]
= h[a-k] - 1 + h[a+k] + 1 - h[b+k] - h[b-k]  (for valid indices)
= h[a-k] + h[a+k] - h[b-k] - h[b+k]

Where undefined terms are 0.

This is O(1) per (k, swap) pair! So for a given swap, computing Δf_k for all k is O(N). And trying all swaps is O(N² * N) = O(N³). For N=380, that's 55M operations. Doable in a few seconds.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-9:
        return h.copy()
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, N, weights):
    center = N - 1
    grad = np.zeros(N)
    thresh = 1e-4
    active = np.where(weights > thresh)[0]
    for ai in active:
        L = ai - center
        wi = weights[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad

def grad_descent(h, N, target, deadline, tau0, tau1, topK0, topK1, lr_init=0.05, mom=0.80):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr_init
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 2500.0, 1.0)
        tau = tau0 + (tau1 - tau0) * frac
        topK = max(2, int(topK0 + (topK1 - topK0) * frac))
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(corr.max()) * dx
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            tau *= 1.15
            lr *= 0.93
        nc = len(corr)
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws > 0:
            w /= ws
        if topK < nc:
            idx_top = np.argsort(corr)[-topK:]
            w_top = np.zeros(nc)
            w_top[idx_top] = 1.0 / topK
            alpha = min(0.55, max(0.0, (tau - 15.0) / 80.0))
            w = (1.0 - alpha) * w + alpha * w_top
        g = grad_c5(h, N, w)
        gn = np.linalg.norm(g)
        if gn > 1e-14:
            g = g / gn
            v = mom * v + g
            vn = np.linalg.norm(v)
            if vn > 1e-14:
                h = h - lr * (v / vn)
        h = project_sum(h, target)
        if it % 60 == 0:
            lr *= 0.975
            if lr < 1e-9:
                break
    return best_h, best_c5
