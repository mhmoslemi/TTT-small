def nesterov_subgrad(h, n_steps, top_k, alpha0, decay, T_warm, mu=0.9):
    n = len(h)
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    y = h.copy()  # momentum variable
    for step in range(n_steps):
        corr = compute_corr(y)
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[top_idx]
        mx = vals.max()
        T = max(T_warm, 50.0 / (step + 1))
        wts = np.exp((vals - mx) / T)
        wts /= wts.sum()

            best_c5 = c5

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(40):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff * h / tot, 0.0, 1.0)
    return h

def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    else:
        sp = -s
        if sp < n:
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g
