import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_and_p(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)  # numerical stability
    exps = np.exp(shifted)
    p = exps / exps.sum()
    fval = (mc + np.log(exps.sum()) / T) * dx
    return fval, p, corr

def compute_grad(h, p, dx):
    N = len(h)
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        lag = i - (N - 1)
        pi = p[i]
        if pi < 1e-15:
            continue
        if lag >= 0:
            if N - 1 - lag > 0:
                grad[lag:] += pi * (1.0 - h[:N - lag])
                grad[:N - lag] -= pi * h[lag:]
            else:
                grad[N-1] += pi * (1.0 - h[0])
        else:
            lp = -lag
            if N - 1 - lp > 0:
                grad[:N - lp] += pi * (1.0 - h[lp:])
                grad[lp:] -= pi * h[:N - lp]
            else:
                grad[0] += pi * (1.0 - h[N-1])
    return dx * grad

def adam_optimize(h0, T, n_iter, lr, dx):
    N = len(h0)
    h = h0.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    best_f = smooth_c5_and_p(h, T)[0]
    best_h = h.copy()
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    for t in range(1, n_iter + 1):
        f, p, corr = smooth_c5_and_p(h, T)
        if f < best_f:
            best_f = f
            best_h = h.copy()
        g = compute_grad(h, p, 2.0 / N)
        # Project gradient to handle sum constraint
        g_avg = g.mean()
        g = g - g_avg  # project out the mean (sum constraint direction)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, N)
    return best_h, best_f

def slsqp_step(h0, T, n, maxiter, ftol):
    def obj(h):
        return smooth_c5_and_p(h, T)[0]
    def obj_grad(h):
        f, p, _ = smooth_c5_and_p(h, T)
        g = compute_grad(h, p, 2.0 / n)
        return g
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n/2.0,
             'jac': lambda h: np.ones(n) / n}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def periodic_obj(xp, p, n, T):
    h = np.tile(xp, n // p)
    return smooth_c5_and_p(h, T)[0]

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_and_p(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    fval = (mc + np.log(exps.sum()) / T) * dx
    return fval, p

def compute_grad(h, p, dx):
    N = len(h)
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        lag = i - (N - 1)
        pi = p[i]
        if pi < 1e-15:
            continue
        if lag >= 0:
            if N - lag > 1:
                grad[lag:] += pi * (1.0 - h[:N - lag])
                grad[:N - lag] -= pi * h[lag:]
            else:
                grad[N - 1] += pi * (1.0 - h[0])
        else:
            lp = -lag
            if N - lp > 1:
                grad[:N - lp] += pi * (1.0 - h[lp:])
                grad[lp:] -= pi * h[:N - lp]
            else:
                grad[0] += pi * (1.0 - h[N - 1])
    return dx * grad

    return h.sum() - len(h) / 2.0

                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

            break

        if f < best_f:

        g = compute_grad(h, p, dx)

        m = beta1 * m + (1 - beta1) * g

        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, N)
    return best_h, best_f
