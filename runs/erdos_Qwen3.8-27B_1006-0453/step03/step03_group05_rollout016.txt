import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def topo_k_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def active_set_gradient(h, top_k, n):
    """Compute exact gradient for the max of top-k shifts."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    wts = corr[idx].astype(float)
    ws = wts.sum()
    if ws > 0:
        wts = wts / ws
    else:
        wts = np.ones(top_k) / top_k
    
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        s = int(ki - (n - 1))
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            grad[j] += w * g
    return grad, idx

def project_gradient_step(h, grad, alpha, n):
    """Projected gradient step with sum constraint."""
    h_new = h - alpha * grad
    return project_feasible(h_new, n)

def line_search(h, grad, alpha0, n, max_iters=30):
    """Backtracking line search."""
    c0 = compute_c5(h)
    alpha = alpha0
    for _ in range(max_iters):
        h_try = project_gradient_step(h, grad, alpha, n)
        c_try = compute_c5(h_try)
        if c_try < c0:
            return h_try, c_try
        alpha *= 0.5
    return h, c0

    best_c5 = compute_c5(best_h)

The key is to arrange the 1s and 0s so that for every shift k, the overlap ∫h(x)(1-h(x+k))dx is minimized. This is like a "Golay" or "Barker sequence" type problem in continuous form.

For the discrete version with n points: we want a binary (or near-binary) sequence of length n with exactly n/2 ones, such that for every cyclic shift k, the number of positions where h[i]=1 and h[(i+k) mod n]=0 (or vice versa, depending on the convention) is minimized.

Wait, actually the overlap is NOT cyclic. It's on [0,2] with h=0 outside. So the correlation is:

- If s >= 0: corr = sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
- If s < 0: corr = sum_{i=-s}^{n-1} h[i] * (1-h[i+s]) = sum_{i=-s}^{n-1} h[i] * (1-h[i-|s|])

This is NOT a circular correlation! It's a linear correlation. The "edge effects" matter.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_wrap(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def active_set_gradient(h, top_k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    wts = corr[idx].astype(float)
    ws = wts.sum()
    if ws > 0:
        wts = wts / ws
    else:
        wts = np.ones(top_k) / top_k
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        s = int(ki - (n - 1))
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            grad[j] += w * g
    return grad

def line_search(h, grad, alpha0, n):
    c0 = compute_c5(h)
    alpha = alpha0
    for _ in range(25):
        h_try = project_feasible(h - alpha * grad, n)
        c_try = compute_c5(h_try)
        if c_try < c0 - 1e-14:
            return h_try, c_try, True
        alpha *= 0.5
    return h, c0, False

def pair_local_search(h, n, top_k, max_passes=4):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_passes):
        corr = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        improved = False
        for ki in idx:
            s = int(ki - (n - 1))
            if s == 0:
                continue
            if s > 0:
                range_i = range(0, n - s)
            else:
                range_i = range(-s, n)
            for i in range_i:
                j = i + s
                if best[i] > best[j] + 0.01:
                    delta = min(best[i] - best[j], 1.0 - best[j], 0.08)
                    if delta > 1e-5:
                        trial = best.copy()
                        trial[i] -= delta
                        trial[j] += delta
                        trial = project_feasible(trial, n)
                        c_try = compute_c5(trial)
                        if c_try < best_c5 - 1e-11:
                            best = trial
                            best_c5 = c_try
                            improved = True
        if not improved:
            break
    return best, best_c5

def smooth_perturb(h, n, scale):
    noise = np.random.randn(n)
    k5 = np.ones(5) / 5.0
    noise = np.convolve(noise, k5, mode='same')
    return project_feasible(h + scale * noise, n)

def bit_flip_try(h, n, n_trials=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_trials):
        i = np.random.randint(n)
        trial = best.copy()
        trial[i] = 0.0 if trial[i] > 0.5 else 1.0
        trial = project_feasible(trial, n)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-12:
            best = trial
            best_c5 = c_try
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Active-set projected gradient (exact gradient, line search)
    for top_k in [3, 5]:
        if time.time() - t0 > budget_s * 0.08:
            break
        h_curr = best_h.copy()
        for step in range(300):
            grad = active_set_gradient(h_curr, top_k, best_n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            alpha = 0.02 * best_n / (gnorm + 1e-15)
            h_new, c_new, ok = line_search(h_curr, grad, alpha, best_n)
            if ok and c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
                h_curr = h_new
            elif not ok:
                break

    # Phase 2: SLSQP with top-k and high-T softmax
    if time.time() - t0 < budget_s * 0.25:
        for k in [3, 5, 8]:
            ht = slsqp_wrap(best_h, topk_obj, (k, best_n), 400, 1e-13)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 3: Pairwise local search + SLSQP follow-up

    # Phase 5: Basin hopping with structured perturbations
    for trial in range(100):
        if time.time() - t0 > budget_s * 0.78:

        if r < 0.20:

            hp = project_feasible(hp, best_n)

    for nr in [150, 180, 250]:

        hh = slsqp_wrap(hu, topk_obj, (3, nr), 300, 1e-12)
        hh = slsqp_wrap(hh, softmax_obj, (5000, nr), 300, 1e-13)

        hu_p = project_feasible(hu + np.random.randn(nr) * 0.012, nr)

    best_c5 = compute_c5(best_h)
