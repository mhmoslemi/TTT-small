        rng = np.random.default_rng()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.integers(n, permutation=True)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_minimize(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                improved = True
            else:
                # revert if no improvement (shouldn't happen for exact CD)
                pass
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-14):
    def obj(x):
        return smooth_obj(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_eq}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0.0, 1.0), n)

def coord_minimize(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        # Correct derivative: (1-h[i-s])*1_{0<=i-s<n} + (-h[i+s])*1_{0<=i+s<n}
        jm = i - s
        c = 0.0
        if 0 <= jm < n:
            c += (1.0 - h[jm])
        jp = i + s
        if 0 <= jp < n:
            c += (-h[jp])
        coef[k] = c
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr0 + hi * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.integers(n, permutation=True)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_minimize(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-13:
                continue
            best_h[i] = hi
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                improved = True
        best_h = project(best_h, n)
        c5_check = compute_c5(best_h)
        if c5_check < best_c5:
            best_c5 = c5_check
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
            no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:

        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP smooth polish (12% budget)
    t_p1 = t0 + budget_s * 0.12
    for T in [50, 200, 1000, 5000]:
        if time.time() > t_p1:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=600, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Aggressive coordinate descent with fixed derivative (30% budget)
    t_p2 = t0 + budget_s * 0.42
    best_h, best_c5 = coord_descent(best_h, n, t_p2, max_sweeps=300, rng=rng)

    # Phase 3: Basin hopping with structured perturbations (28% budget)
    t_p3 = t0 + budget_s * 0.70
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iters = 50
    for it in range(n_iters):
        if time.time() > t_p3:
            break
        cool = max(0.05, 1.0 - 0.9 * (it / float(n_iters)))
        ptype = it % 5
        if ptype == 0:
            h_p = work_h + (0.03 + 0.12 * cool) * rng.standard_normal(n)
        elif ptype == 1:
            h_p = work_h.copy()
            a = int(rng.integers(0, n - 25))
            b = int(a + rng.integers(8, n - a - 8))
            h_p[a:b], h_p[b:] = h_p[b:].copy(), h_p[a:b].copy()
        elif ptype == 2:
            freq = int(rng.integers(1, 6))
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2, n, endpoint=False)
            h_p = work_h + 0.10 * cool * np.sin(np.pi * freq * x + phase)
        elif ptype == 3:
            h_p = work_h.copy()
            a = int(rng.integers(0, n - 18))
            b = int(a + rng.integers(6, n - a - 6))
            h_p[a:b] = 1.0 - h_p[a:b]
        else:
            h_p = work_h.copy()
            idx = rng.integers(0, n, size=max(3, n // 10))
            h_p[idx] = rng.uniform(0, 1, size=len(idx))

        h_p = project(h_p, n)
        T_bh = int(300 + 2000 * cool)
        try:
            hc = slsqp_opt(h_p, T_bh, n, maxiter=300, ftol=1e-13)
            c5v = compute_c5(hc)
            temp = 0.003 * cool + 0.0004
            if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc.copy()
        except Exception:
            pass
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: More coordinate descent (15% budget)
    t_p4 = t0 + budget_s * 0.88
    best_h, best_c5 = coord_descent(best_h, n, t_p4, max_sweeps=250, rng=rng)

    # Phase 5: Multi-resolution check with n=120 (5% budget)
    t_p5 = t0 + budget_s * 0.93
    if time.time() < t_p5:
        n2 = 120
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n2)
        try:
            hc2 = slsqp_opt(h_interp, 800, n2, maxiter=500, ftol=1e-14)
            c5_2 = compute_c5(hc2)
            if c5_2 < best_c5:
                best_h = hc2
                best_c5 = c5_2
                n = n2
        except Exception:
            pass
        if time.time() < t_p5:
            best_h, best_c5 = coord_descent(best_h, n, t_p5, max_sweeps=80, rng=rng)

    # Phase 6: Final SLSQP + CD polish (7% budget)
    t_end = t0 + budget_s * 0.985
    for T in [8000, 30000]:
        if time.time() > t_end:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=1000, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    if time.time() < t_end:
        best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=100, rng=rng)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
