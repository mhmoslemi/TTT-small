import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def wgrad_smooth(h, T):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    ev = np.exp(T * (corr - m))
    wts = ev / ev.sum()
    grad = np.zeros(n)
    for ki in range(len(corr)):
        w = wts[ki]
        if w < 1e-11:
            continue
        s = ki - (n - 1)
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
        else:
            ss = -s
            grad[ss:] += w * (1.0 - h[:n-ss])
            grad[:n-ss] -= w * h[ss:]
    return grad

def wgrad_topk(h, k):
    n = len(h)
    corr = get_corr(h)
    top_idx = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:]) / k
            grad[s:] -= h[:n-s] / k
        else:
            ss = -s
            grad[ss:] += (1.0 - h[:n-ss]) / k
            grad[:n-ss] -= h[ss:] / k
    return grad

def nesterov_subgrad(h, n_steps, grad_mode, T, k, alpha0, decay, momentum):
    n = len(h)
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    y = h.copy()
    for step in range(n_steps):
        if grad_mode == 'smooth':
            g = wgrad_smooth(y, T)
        else:
            g = wgrad_topk(y, k)
        v_new = momentum * v - alpha * g
        y_new = project_feasible(h + v_new, n)
        v = v_new
        c5 = compute_c5(y_new)
        if c5 < best_c5:
            best_c5 = c5
            best = y_new.copy()
        h = y_new
        alpha *= decay
    return best, best_c5

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def topk_obj(h, k, n):
    corr = get_corr(h)
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(h, T, n):
    corr = get_corr(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

                else:
                    best_c5, best_h = c5s, hs2
    # Random balanced starts
    for _ in range(5):
        if elapsed() > budget_s * 0.72:
            break
        hr = random_balanced(best_n, rng)
        hr2, c5r = nesterov_subgrad(hr, 2000, 'topk', 0, 3, 0.01, 0.999, 0.92)
        c5r = compute_c5(hr2)
        if c5r < best_c5:
            hr3 = slsqp_opt(hr2, topk_obj, (3, best_n), 150, 1e-12)
            c5r3 = compute_c5(hr3)
            if c5r3 < best_c5:
                best_c5, best_h = c5r3, hr3
            else:
                best_c5, best_h = c5r, hr2

    # Phase 5: Basin hopping on best
    n_trials = int(budget_s * 0.15 / 10)
    for trial in range(min(n_trials, 40)):
        if elapsed() > budget_s * 0.88:
            break
        sc = 0.025 * np.exp(-0.03 * trial) + 0.005
        hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        ho, co = nesterov_subgrad(hp, 2000, 'topk', 0, 3, 0.005, 0.999, 0.90)
        co = compute_c5(ho)
        if co < best_c5:
            ho2 = slsqp_opt(ho, topk_obj, (3, best_n), 100, 1e-13)
            co2 = compute_c5(ho2)
            if co2 < best_c5:
                best_c5, best_h = co2, ho2
            else:
                best_c5, best_h = co, ho

    # Phase 6: Final intensive polish
    if elapsed() < budget_s * 0.96:
        for T in [3000, 8000, 20000]:
            ht, ct = nesterov_subgrad(best_h, 5000, 'smooth', T, 3, 0.002, 0.9998, 0.95)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht
    if elapsed() < budget_s * 0.985:
        for k in [1, 2, 3]:
            hf = slsqp_opt(best_h, topk_obj, (k, best_n), 300, 1e-15)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
