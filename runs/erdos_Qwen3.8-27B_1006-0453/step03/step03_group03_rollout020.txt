    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.sort(corr)[::-1][:n_top]
    m = top[0]
    val = m + np.log(np.sum(np.exp(T * (top - m)))) / T
    return val * (2.0 / len(h))

def soft_max_grad(corr, h, T, n_top=5):

    m = np.max(corr)
    top_idx = np.argsort(corr)[::-1][:n_top]

            kp = -lag
            grad[kp:] += wi * (1.0 - h[:n-kp])
            grad[:n-kp] += -wi * h[kp:]

    return x.sum() - len(x) / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(corr_vals, h, T, n_top=5):
    n = len(h)
    m = np.max(corr_vals)
    top_idx = np.argsort(corr_vals)[::-1][:n_top]
    w = np.exp(T * (corr_vals[top_idx] - m))
    w /= w.sum()
    grad = np.zeros(n)
    for j, idx in enumerate(top_idx):
        lag = idx - (n - 1)
        wi = w[j]
        if lag >= 0:
            grad[lag:] += wi * (1.0 - h[:n - lag])
            grad[:n - lag] += -wi * h[lag:]
        else:
            kp = -lag
            grad[:n - kp] += wi * (1.0 - h[kp:])
            grad[kp:] += -wi * h[:n - kp]
    return grad * (2.0 / n)

def soft_max_obj(h, T, n_top=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.sort(corr)[::-1][:n_top]
    m = top[0]
    return (m + np.log(np.sum(np.exp(T * (top - m)))) / T) * (2.0 / len(h))

def soft_max_jac(h, T, n_top=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr_grad(corr, h, T, n_top)

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def pg_optimize(h, n_iters=4000, T_init=5000, T_growth=1.003, step_init=0.02):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    T = T_init
    step = step_init
    stall = 0
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr_grad(corr, h, T, n_top=5)
        h_new = project_feasible(h - step * g, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.03, 0.09)
            stall = 0
        else:
            stall += 1
            step *= 0.88
            if step < 1e-7:
                break
            if stall > 500:
                break
        T *= T_growth
    return best_h, best_c5

def slsqp_polish(h, T, maxiter=400, n_top=5):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(soft_max_obj, h, args=(T, n_top), jac=lambda x: soft_max_jac(x, T, n_top),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or res.fun < soft_max_obj(h, T, n_top):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def binary_perturb(h, n, rng, n_flips=3):
    h_new = h.copy()
    for _ in range(n_flips):
        pos = rng.randint(1, n - 2)
        block = rng.randint(1, 4)
        if h_new[pos] > 0.5:
            h_new[pos:pos+block] = 0.0
        else:
            h_new[pos:pos+block] = 1.0
    return project_feasible(h_new, n)

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: PG optimizer (warm start)
    if time.time() < t_deadline - budget_s * 0.15:
        best_h, best_c5 = pg_optimize(best_h, n_iters=3000, T_init=3000, T_growth=1.004, step_init=0.025)

    # Phase 2: SLSQP with escalating T
    for T in [20000, 80000, 300000]:
        if time.time() > t_deadline - budget_s * 0.18:
            break
        h_try = slsqp_polish(best_h, T, maxiter=500, n_top=5)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        best_h, best_c5 = pg_optimize(best_h, n_iters=1500, T_init=100000, T_growth=1.002, step_init=0.015)

    # Phase 3: Multi-n search
    n_candidates = sorted(set([best_n - 2, best_n - 4, best_n, best_n + 2, best_n + 4, best_n + 6, best_n + 8]))
    for n_try in n_candidates:
        if time.time() > t_deadline - budget_s * 0.30:
            break
        if n_try < 40 or n_try % 2 != 0:
            continue
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            h_base = project_feasible(interp_h(best_h, best_n, n_try), n_try)
        h_opt, c5_opt = pg_optimize(h_base, n_iters=2000, T_init=5000, T_growth=1.003, step_init=0.02)
        h_opt = slsqp_polish(h_opt, 100000, maxiter=400, n_top=5)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try

    # Phase 4: Binary perturbation + re-optimize
    n_pert = 12
    for trial in range(n_pert):
        if time.time() > t_deadline - budget_s * 0.25:
            break
        h_pert = binary_perturb(best_h, best_n, rng, n_flips=rng.randint(1, 5))
        h_opt, c5_opt = pg_optimize(h_pert, n_iters=2000, T_init=8000, T_growth=1.003, step_init=0.02)
        h_opt = slsqp_polish(h_opt, 80000, maxiter=400, n_top=5)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Rotation + shift search
    for shift in [1, 2, 3, 4, 5, 7, 13]:
        if time.time() > t_deadline - budget_s * 0.20:
            break
        h_rot = np.roll(best_h, shift)
        h_rot = project_feasible(h_rot, best_n)
        h_opt, c5_opt = pg_optimize(h_rot, n_iters=2500, T_init=10000, T_growth=1.003, step_init=0.02)
        h_opt = slsqp_polish(h_opt, 150000, maxiter=400, n_top=5)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Random restarts at different n
    for n_try in [best_n, best_n + 2, best_n - 2, best_n + 4]:
        if time.time() > t_deadline - budget_s * 0.08:
            break
        if n_try < 40 or n_try % 2 != 0:
            continue
        for trial in range(3):
            if time.time() > t_deadline - budget_s * 0.06:
                break
            h_rand = project_feasible(np.clip(np.interp(np.linspace(0, 2, n_try), np.linspace(0, 2, best_n), best_h) + rng.randn(n_try) * 0.08, 0, 1), n_try)
            h_opt, c5_opt = pg_optimize(h_rand, n_iters=1500, T_init=5000, T_growth=1.004, step_init=0.02)
            h_opt = slsqp_polish(h_opt, 100000, maxiter=300, n_top=5)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 7: Final intensive polish
    if time.time() < t_deadline - budget_s * 0.02:
        best_h, best_c5 = pg_optimize(best_h, n_iters=5000, T_init=200000, T_growth=1.001, step_init=0.01)
        best_h = slsqp_polish(best_h, 500000, maxiter=800, n_top=3)
        c5_check, _ = compute_c5(best_h)
        if c5_check < best_c5:
            best_c5 = c5_check
        best_h, best_c5 = pg_optimize(best_h, n_iters=3000, T_init=500000, T_growth=1.0005, step_init=0.008)

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
