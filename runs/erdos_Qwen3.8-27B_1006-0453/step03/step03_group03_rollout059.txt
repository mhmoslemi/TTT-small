        best_h, best_c5 = subgradient_polish(best_h, n_iters=5000)

    # Phase 3: Multi-n search
    n_list = [76, 78, 80, 82, 84, 86, 88, 90, 96, 100, 108, 120]
    n_list = [nn for nn in n_list if nn % 2 == 0]
    for n_try in n_list:
        if time.time() - t0 > budget_s * 0.65:
            break
        if n_try == best_n:
            h_try = best_h.copy()
        else:
            h_try = interp_h(best_h, best_n, n_try)
        # SLSQP polish at this n
        for T, mi in [(5000, 500), (50000, 600), (300000, 600)]:
            if time.time() - t0 > budget_s * 0.65:
                break
            h_try = slsqp_polish(h_try, T, maxiter=mi)
        # Quick subgradient
        h_try, c5_try = subgradient_polish(h_try, n_iters=1500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_try

    # Phase 4: Subgradient on best
    if time.time() - t0 < budget_s * 0.78:
        best_h, best_c5 = subgradient_polish(best_h, n_iters=4000)

    # Phase 5: High-T SLSQP polish
    if time.time() - t0 < budget_s * 0.90:
        for T, mi in [(500000, 1000), (3000000, 1000), (15000000, 1200)]:
            if time.time() - t0 > budget_s * 0.90:
                break
            h_try = slsqp_polish(best_h, T, maxiter=mi)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 6: Final subgradient + SLSQP
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = subgradient_polish(best_h, n_iters=3000)
        h_try = slsqp_polish(best_h, 5000000, maxiter=1500)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
