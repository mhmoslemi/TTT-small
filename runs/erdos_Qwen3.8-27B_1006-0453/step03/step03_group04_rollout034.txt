def adam_continuous(h, N, target, deadline, k=20, lr0=3.0, decay=0.998):
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wk * (1.0 - h[M:])
                    grad[M:] -= wk * h[:N - M]
        if peak < best_val:
            best_val = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
    return best_h, best_val * dx

def two_swap_sa(h, N, deadline, rng, n_rounds=50):
    dx = 2.0 / N
    h = h.copy()
    corr = compute_corr(h)
    cur_peak = float(corr.max())
    best_h = h.copy()
    best_peak = cur_peak
    T = 1.0

    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = rng.choice(ones, size=2, replace=False)
        j1, j2 = rng.choice(zeros, size=2, replace=False)
        new_corr = corr.copy()
        for ii in [i1, i2]:
            new_corr[N - 1 - ii : 2 * N - 1 - ii] -= (1.0 - h[0:N - ii])
        for jj in [j1, j2]:
            new_corr[N - 1 - jj : 2 * N - 1 - jj] += (1.0 - h[0:N - jj])
        new_peak = float(new_corr.max())
        delta = new_peak - cur_peak
        if delta <= 0 or (T > 0.01 and rng.random() < np.exp(-delta * N / T)):
            corr = new_corr
            h[i1] = 0; h[i2] = 0
            h[j1] = 1; h[j2] = 1
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_h = h.copy()
        T *= 0.98

    return best_h, best_peak * dx

    h_cont = best_h.astype(float) + rng.randn(N) * 0.08
    h_cont = project_sum(h_cont, target)
    h_c1, c_c1 = adam_continuous(h_cont, N, target, d4, k=15, lr0=4.0, decay=0.998)
    if c_c1 < best_c5:
        best_c5, best_h = c_c1, h_c1.copy()

    # Phase 3: Final polish (12%)
    d5 = deadline
    h_fin = best_h.copy() + rng.randn(N) * 0.03
    h_fin = project_sum(h_fin, target)
    h_f1, c_f1 = adam_continuous(h_fin, N, target, d5, k=10, lr0=1.5, decay=0.999)
    if c_f1 < best_c5:
        best_c5, best_h = c_f1, h_f1

    final_c5 = compute_c5(best_h, N)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, N
