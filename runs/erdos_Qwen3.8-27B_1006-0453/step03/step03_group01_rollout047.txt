            best_h = h.copy()
            last_improve = it
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        if it % 40 == 0:
            lr *= lr_dec
        if it - last_improve > stall_patience and it > stall_patience:
            h = best_h.copy()
            noise = np.random.randn(N) * 0.008
            h = project_sum(h + noise, target)
            m[:] = 0.0
            v[:] = 0.0
            last_improve = it
    return best_h, best_c5

def binary_push(h, N, target, deadline, n_rounds=8):
    dx = 2.0 / N
    best_c5, _ = compute_c5(h, dx)
    best_h = h.copy()
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        for i in range(N):
            if time.time() > deadline:
                break
            val = best_h[i]
            if val < 0.12 or val > 0.88:
                continue
            for trial in (0.0, 1.0):
                h = best_h.copy()
                h[i] = trial
                h = project_sum(h, target)
                c5, _ = compute_c5(h, dx)
                if c5 < best_c5 - 5e-11:
                    best_c5 = c5
                    best_h = h.copy()
                    break
    return best_h, best_c5

def coord_sweep(h, N, target, deadline):
    dx = 2.0 / N
    best_c5, _ = compute_c5(h, dx)
    best_h = h.copy()
    steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    for step in steps:
        if time.time() > deadline:
            break
        for _pass in range(2):
            if time.time() > deadline:
                break
            for i in range(N):
                if time.time() > deadline:
                    break
                for d in (-step, step):
                    trial = best_h[i] + d
                    if trial < 0 or trial > 1:
                        continue
                    h = best_h.copy()
                    h[i] = trial
                    h = project_sum(h, target)
                    c5, _ = compute_c5(h, dx)
                    if c5 < best_c5 - 3e-11:
                        best_c5 = c5
                        best_h = h.copy()
                        break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5, _ = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    # Phase 1: Adam optimization at N=380 with binary attraction
    N1 = n0
    tgt1 = N1 / 2.0
    t1_end = t0 + (budget_s - 5) * 0.42
    h1, c1 = optimize_adam(best_h.copy(), N1, tgt1, min(t1_end, deadline - 30),
                           5.0, 120.0, 25, 4, 0.025, 0.97, 0.02, 300)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: Higher resolution N=572
    N2 = 572
    tgt2 = N2 / 2.0
    t2_end = t0 + (budget_s - 5) * 0.68
    if time.time() < t2_end - 15:
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2 = np.interp(x2, x0, best_h)
        h2 = project_sum(h2, tgt2)
        h2, c2 = optimize_adam(h2, N2, tgt2, min(t2_end, deadline - 20),
                               30.0, 300.0, 12, 3, 0.012, 0.975, 0.015, 250)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2
            x0 = x2
            n0 = N2

    # Phase 3: Focused low-temp refinement
    Nb = len(best_h)
    tgtb = Nb / 2.0
    t3_end = t0 + (budget_s - 5) * 0.82
    if time.time() < t3_end - 5:
        hb, cb = optimize_adam(best_h.copy(), Nb, tgtb, min(t3_end, deadline - 15),
                               150.0, 800.0, 4, 1, 0.004, 0.985, 0.01, 200)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb

    # Phase 4: Binary push
    Np = len(best_h)
    tgtp = Np / 2.0
    t4_end = t0 + (budget_s - 5) * 0.92
    if time.time() < t4_end - 3:
        best_h, best_c5 = binary_push(best_h.copy(), Np, tgtp, min(t4_end, deadline - 5))

    # Phase 5: Coordinate sweep
    Ns = len(best_h)
    tgts = Ns / 2.0
    if time.time() < deadline - 2:
        best_h, best_c5 = coord_sweep(best_h.copy(), Ns, tgts, deadline - 1)

    final_c5, _ = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
