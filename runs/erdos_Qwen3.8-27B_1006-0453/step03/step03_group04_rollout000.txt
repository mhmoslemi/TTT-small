    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_soft(h, N, tau, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    w = np.exp(-(peak - corr) / tau)
    total = w.sum()
    if total < 1e-30:
        return np.zeros(N), peak
    w /= total
    top_idx = np.argpartition(w, -k)[-k:]
    top_idx = top_idx[np.argsort(-w[top_idx], kind='stable')]
    w_sel = w[top_idx].copy()
    w_sel /= w_sel.sum()
    grad = np.zeros(N)
    center = N - 1
    for i in range(k):
        ai = top_idx[i]
        L = ai - center
        wi = w_sel[i]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, tau0, tau_decay, lr_decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    tau = tau0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_soft(h, N, tau, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 80 == 0:
            lr *= lr_decay
            tau *= tau_decay
    return best_h, best_peak * (2.0 / N)

def anneal_binary(h, N, deadline, rng, n_restarts=4, T0=5.0, T1=0.002):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        idx = np.where(h_b == 1)[0]
        h_b[idx[N // 2:]] = 0.0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N // 2 - n_ones]] = 1.0
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            h_b = best.copy()
            n_pert = max(2, N // 25)
            for _ in range(n_pert):
                ones = np.where(h_b == 1)[0]
                zeros = np.where(h_b == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_b[i], h_b[j] = 0.0, 1.0
        cur_pk = float(np.max(compute_corr(h_b)))
        n_iter = max(800, int((deadline - time.time()) * 25))
        for it in range(n_iter):
            if time.time() > deadline:
                break
            T = T0 * (T1 / T0) ** (it / n_iter)
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_b[i] = 0.0
            h_b[j] = 1.0
            new_pk = float(np.max(compute_corr(h_b)))
            delta = new_pk - cur_pk
            if delta < 0 or (T > 0.004 and rng.random() < np.exp(-delta / T)):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = h_b.copy()
            else:
                h_b[i] = 1.0
                h_b[j] = 0.0
    return best, best_pk * dx

def binarize_and_repair(h, N):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        idx = np.where(h_b == 1)[0]
        h_b[idx[N // 2:]] = 0.0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N // 2 - n_ones]] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    # Phase 1: Adam on parent resolution with soft-max gradient
    N1 = n0
    target1 = N1 / 2.0
    d1 = time.time() + (deadline - t0) * 0.10
    h1, c1 = adam_run(h0.copy(), N1, target1, d1, k=50, lr0=0.06, tau0=8.0, tau_decay=0.99, lr_decay=0.992)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary annealing (long, multi-restart)
    d2 = time.time() + (deadline - time.time()) * 0.28
    h2, c2 = anneal_binary(best_h, N1, d2, rng, n_restarts=6, T0=6.0, T1=0.001)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 3: Adam refine on the annealed solution
    h3 = best_h + rng.randn(N1) * 0.04
    h3 = project_sum(h3, target1)
    d3 = time.time() + (deadline - time.time()) * 0.18
    h3, c3 = adam_run(h3, N1, target1, d3, k=25, lr0=0.025, tau0=4.0, tau_decay=0.995, lr_decay=0.996)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: More binary annealing
    if time.time() < deadline - 100:
        d4 = time.time() + (deadline - time.time()) * 0.30
        h4, c4 = anneal_binary(best_h, N1, d4, rng, n_restarts=8, T0=4.0, T1=0.001)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Try higher resolution (550)
    N2 = 550
    if time.time() < deadline - 130:
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2i = np.interp(x2, x0, best_h)
        h2i = project_sum(h2i, N2 / 2.0)
        d5 = time.time() + (deadline - time.time()) * 0.30
        h5, c5v = adam_run(h2i, N2, N2 / 2.0, d5, k=40, lr0=0.04, tau0=6.0, tau_decay=0.99, lr_decay=0.993)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()
        if time.time() < deadline - 50:
            d6 = time.time() + (deadline - time.time()) * 0.40
            h6, c6 = anneal_binary(h5, N2, d6, rng, n_restarts=4, T0=5.0, T1=0.002)
            if c6 < best_c5:
                best_c5, best_h = c6, h6.copy()

    # Phase 6: Final polish on 380 (go back to lower res if higher didn't help)
    Nb = min(N1, len(best_h))
    if len(best_h) > Nb:
        xN = np.linspace(0.0, 2.0, Nb, endpoint=False)
        hN = np.interp(xN, np.linspace(0.0, 2.0, len(best_h), endpoint=False), best_h)
        best_h = project_sum(hN, Nb / 2.0)
        best_c5 = compute_c5(best_h, Nb)
    Nb = len(best_h)
    if time.time() < deadline - 8:
        hF = best_h + rng.randn(Nb) * 0.012
        hF = project_sum(hF, Nb / 2.0)
        hF, cF = adam_run(hF, Nb, Nb / 2.0, deadline - 2, k=15, lr0=0.012, tau0=3.0, tau_decay=0.997, lr_decay=0.998)
        if cF < best_c5:
            best_c5, best_h = cF, hF
        # Quick final binarize + local check
        hB = binarize_and_repair(best_h, Nb)
        cB = compute_c5(hB, Nb)
        if cB < best_c5:
            best_c5, best_h = cB, hB

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
