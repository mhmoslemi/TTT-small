    best_n = len(best_h)

    best_c5, _ = compute_c5(best_h)

    # Phase 1: L-BFGS-B with moderate T (fast global-ish optimization)
    for T, lam, mi in [(100, 200, 600), (400, 500, 600), (1500, 1000, 500)]:
        if elapsed() > budget_s * 0.12:
            break
        h_try, _ = lbfgs_smooth(best_h, T, lam, maxiter=mi)
        h_try = project_feasible(h_try, best_n)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Subgradient with line search (main workhorse)
    if elapsed() < budget_s * 0.55:
        remain = min(budget_s * 0.40, budget_s * 0.55 - elapsed())
        nit = max(2000, int(remain * 12))
        h_sub, c5_sub = subgrad_with_line_search(best_h, n_iters=nit, top_k=5, lr0=0.06)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    # Phase 3: SLSQP smooth with high T (polish the smoothed landscape)
    if elapsed() < budget_s * 0.65:
        for T in [5000, 30000]:
            if elapsed() > budget_s * 0.65:
                break
            h_try = slsqp_smooth(best_h, T, maxiter=600)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 4: Multi-resolution search
    res_list = [best_n, best_n - 2, best_n + 2, best_n - 4, best_n + 4, best_n + 6, best_n - 6]
    res_list = sorted(set(r for r in res_list if r >= 50 and r % 2 == 0))
    for n_try in res_list:
        if elapsed() > budget_s * 0.80:
            break
        h_interp = interp_to_n(best_h, n_try)
        # Quick L-BFGS + subgrad at this resolution
        h_try, _ = lbfgs_smooth(h_interp, 300, 200, maxiter=400)
        h_try = project_feasible(h_try, n_try)
        h_try, c5_try = subgrad_with_line_search(h_try, n_iters=1200, top_k=5, lr0=0.05)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_try

    # Phase 5: Binary structure search on best resolution
    if elapsed() < budget_s * 0.88:
        h_bin = (best_h >= 0.5).astype(float)
        # Ensure exactly n/2 ones
        n_half = best_n // 2
        ones = np.where(h_bin == 1.0)[0]
        if len(ones) > n_half:
            h_bin[ones[n_half:]] = 0.0
        elif len(ones) < n_half:
            zeros = np.where(h_bin == 0.0)[0]
            h_bin[zeros[:n_half - len(ones)]] = 1.0
        h_bin, c5_bin = binary_local_search(h_bin, best_n, n_trials=8000)
        # Refine from binary with subgrad
        h_ref, c5_ref = subgrad_with_line_search(h_bin, n_iters=1500, top_k=5, lr0=0.04)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
        # Also refine the binary with L-BFGS
        if elapsed() < budget_s * 0.92:
            h_l, _ = lbfgs_smooth(h_bin, 500, 500, maxiter=500)
            h_l = project_feasible(h_l, best_n)
            h_l, c5_l = subgrad_with_line_search(h_l, n_iters=800, top_k=5, lr0=0.03)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l

    # Phase 6: Perturbation + re-optimize (escape local optima)
    if elapsed() < budget_s * 0.94:
        for scale in [0.03, 0.015]:
            if elapsed() > budget_s * 0.94:
                break
            h_p = best_h + np.random.randn(best_n) * scale
            h_p = project_feasible(h_p, best_n)
            h_p, _ = lbfgs_smooth(h_p, 1000, 500, maxiter=400)
            h_p = project_feasible(h_p, best_n)
            h_p, c5_p = subgrad_with_line_search(h_p, n_iters=1000, top_k=5, lr0=0.04)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

    # Phase 7: Final SLSQP direct polish
    if elapsed() < budget_s * 0.97:
        h_f = slsqp_direct(best_h, maxiter=800)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
