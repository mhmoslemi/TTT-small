import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def obj_smooth_topk(h, T, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        top = corr[idx]
    else:
        top = corr
    return soft_max(top, T) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def sum_con(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h0, T, n, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    res = minimize(obj_smooth_topk, h0, args=(T, topk),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

                if best[i] > best[j]:
