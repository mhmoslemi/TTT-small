    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    diff = target - s

        return h

        if total < 1e-13:

    diff = n/2.0 - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:

        if t < 1e-13: return h
        h += r * (diff/t)

    """Compute dR[k]/dh[j] for all k."""
    shifts = np.arange(-(n-1), n)
    js = j + shifts
    jn = j - shifts

    v2 = (jn >= 0) & (jn < n)
    g = np.zeros(2*n-1)
    g[v1] += 1.0 - h[js[v1]]
    g[v2] -= h[jn[v2]]

    tau_hi = float(np.max(R) + np.max(np.abs(g)) * (U - L) + 1.0)
    for _ in range(n_iters):
        tau_mid = 0.5 * (tau_lo + tau_hi)
        lo_b = L
        hi_b = U
        pos = g > 1e-12
        neg = g < -1e-12
        if np.any(pos):
            hi_b = min(hi_b, float(np.min((tau_mid - R[pos]) / g[pos])))
        if np.any(neg):
            lo_b = max(lo_b, float(np.max((tau_mid - R[neg]) / g[neg])))
        zero_mask = ~pos & ~neg
        if np.any(zero_mask) and np.max(R[zero_mask]) > tau_mid:
            tau_lo = tau_mid
            continue
        if lo_b <= hi_b + 1e-13:
            tau_hi = tau_mid
        else:
            tau_lo = tau_mid
    lo_b = L
    hi_b = U
    pos = g > 1e-12
    if np.any(pos):
        hi_b = min(hi_b, float(np.min((tau_hi - R[pos]) / g[pos])))
    neg = g < -1e-12
    if np.any(neg):
        lo_b = max(lo_b, float(np.max((tau_hi - R[neg]) / g[neg])))
    if lo_b > hi_b:
        lo_b = hi_b = (L + U) * 0.5
    return float(np.clip(0.5 * (lo_b + hi_b), L, U))

def coord_sweep(h, n, order=None, n_iters=40):
    if order is None:
        order = np.arange(n)
    shifts = np.arange(-(n - 1), n)
    R = np.correlate(h, 1.0 - h, mode='full')
    for idx, j in enumerate(order):
        js = j + shifts
        jn = j - shifts
        v1 = (js >= 0) & (js < n)
        v2 = (jn >= 0) & (jn < n)
        g = np.zeros(2 * n - 1)
        if np.any(v1):
            g[v1] += 1.0 - h[js[v1]]
        if np.any(v2):
            g[v2] -= h[jn[v2]]
        if np.all(np.abs(g) < 1e-12):
            continue
        L = -h[j] + 1e-12
        U = 1.0 - h[j] - 1e-12
        if U <= L:
            continue
        delta = solve_1d(R, g, L, U, n_iters)
        if abs(delta) > 1e-10:
            h[j] += delta
            R += delta * g
            h = project_feasible(h, n)
            if idx % 25 == 24:
                R = np.correlate(h, 1.0 - h, mode='full')
    return h

def slsqp_opt(h0, n, T, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    try:
        res = minimize(smooth_max, h0, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def multi_slq(h0, n, t_list, maxiter):
    best = h0.copy()
    best_c5 = compute_c5(best)
    for T in t_list:
        try:
            hc = slsqp_opt(best, n, T, maxiter)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best = hc
        except Exception:
            pass
    return best, best_c5

def swap_search(h, n, n_trials=200, n_iters=30):
    shifts = np.arange(-(n - 1), n)
    R = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(R)) * (2.0 / n)
    for _ in range(n_trials):
        j1, j2 = np.random.choice(n, 2, replace=False)
        g = np.zeros(2 * n - 1)
        js1, jn1 = j1 + shifts, j1 - shifts
        v1 = (js1 >= 0) & (js1 < n)
        v1m = (jn1 >= 0) & (jn1 < n)
        g[v1] += 1.0 - h[js1[v1]]
        g[v1m] -= h[jn1[v1m]]
        js2, jn2 = j2 + shifts, j2 - shifts
        v2 = (js2 >= 0) & (js2 < n)
        v2m = (jn2 >= 0) & (jn2 < n)
        g[v2] -= 1.0 - h[js2[v2]]
        g[v2m] += h[jn2[v2m]]
        L = max(-h[j1], h[j2] - 1.0) + 1e-12
        U = min(1.0 - h[j1], h[j2]) - 1e-12
        if U <= L:
            continue
        delta = solve_1d(R, g, L, U, n_iters)
        if abs(delta) > 1e-10:
            h[j1] += delta
            h[j2] -= delta
            R = np.correlate(h, 1.0 - h, mode='full')
            c5v = float(np.max(R)) * (2.0 / n)
            if c5v < best_c5:
                best_c5 = c5v
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent (primary optimizer)
    for sv in range(400):
        if time.time() - t0 > budget_s * 0.28:
            break
        if sv % 7 == 3:
            order = np.random.permutation(n)
        elif sv % 7 == 5:
            order = np.arange(n - 1, -1, -1)
        else:
            order = np.arange(n)
        h_new = coord_sweep(best_h, n, order, n_iters=35)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: SLSQP polish with increasing T
    if time.time() - t0 < budget_s * 0.42:
        best_h, best_c5 = multi_slq(best_h, n, [100, 500, 2000, 10000], 400)

    # Phase 3: Swap-based refinement
    if time.time() - t0 < budget_s * 0.52:
        for _ in range(3):
            if time.time() - t0 > budget_s * 0.52:
                break
            best_h, best_c5 = swap_search(best_h, n, 300, 30)
            best_h = coord_sweep(best_h, n, np.arange(n), 30)
            best_c5 = compute_c5(best_h)

    # Phase 4: Basin hopping
    if time.time() - t0 < budget_s * 0.72:
        bh_h = best_h.copy()
        bh_c5 = best_c5
        for it in range(400):
            if time.time() - t0 > budget_s * 0.72:
                break
            step = 0.05 * max(1.0 - it / 400.0, 0.1) + 0.004
            r = np.random.rand()
            if r < 0.5:
                h_p = bh_h + step * np.random.randn(n)
            elif r < 0.8:
                h_p = bh_h.copy()
                bl = int(np.random.randint(4, n // 3 + 1))
                st = int(np.random.randint(0, n - bl))
                h_p[st:st + bl] += step * (np.random.rand(bl) * 2.0 - 1.0)
            else:
                h_p = bh_h.copy()
                i1, i2 = np.random.choice(n, 2, replace=False)
                sv_val = step * 1.5
                h_p[i1] += sv_val
                h_p[i2] -= sv_val
            h_p = project_feasible(h_p, n)
            h_new = coord_sweep(h_p, n, np.random.permutation(n), 25)
            c5_new = compute_c5(h_new)
            temp = 0.003 * max(1.0 - it / 400.0, 0.03) + 0.0001
            if c5_new < bh_c5 or (c5_new > bh_c5 and np.random.rand() < np.exp(-(c5_new - bh_c5) / temp)):
                bh_c5 = c5_new
                bh_h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 5: Multi-resolution
    for n_try in [100, 120, 140, 200]:
        if time.time() - t0 > budget_s * 0.88:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_feasible(h_i, n_try)
        for _ in range(6):
            if time.time() - t0 > budget_s * 0.90:
                break
            h_i = coord_sweep(h_i, n_try, np.random.permutation(n_try), 30)
        h_i, _ = swap_search(h_i, n_try, 200, 25)
        h_i = coord_sweep(h_i, n_try, np.arange(n_try), 35)
        c5n = compute_c5(h_i)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_i

    # Phase 6: Final polish
    n_f = len(best_h)
    if time.time() - t0 < budget_s * 0.96:
        for T in [500, 5000, 50000]:
            if time.time() - t0 > budget_s * 0.96:
                break
            h_new = slsqp_opt(best_h, n_f, T, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        for _ in range(5):
            if time.time() - t0 > budget_s * 0.985:
                break
            best_h = coord_sweep(best_h, n_f, np.arange(n_f), 45)
            best_c5 = compute_c5(best_h)

    # Alternate starting point check
    if time.time() - t0 < budget_s * 0.99:
        h_alt = np.zeros(n)
        quarter = n // 4
        h_alt[:quarter] = 1.0
        h_alt[2 * quarter:3 * quarter] = 1.0
        h_alt = project_feasible(h_alt, n)
        h_alt = coord_sweep(h_alt, n, np.arange(n), 30)
        for _ in range(3):
            h_alt = coord_sweep(h_alt, n, np.random.permutation(n), 30)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
