            for j in range(n - k_neg):
                g[j] -= h[j - k_neg]  # j+k = j-k_neg
        grad += g
    
    # Project to maintain sum constraint
    grad -= np.mean(grad)
    
    # Adaptive step size
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = make_feasible(h_new, n)
    return h_new

    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Log-sum-exp for numerical stability
    val = m + np.log(np.mean(np.exp(T * (corr - m) - np.max(T * (corr - m))))) / T
    return val * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = T * (corr - m)
    val = m + np.log(np.sum(np.exp(scaled - np.max(scaled))) + 1e-300) / T
    return val * dx

def lse_pen_obj(x, T, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    scaled = T * (corr - m)
    val = m + np.log(np.sum(np.exp(scaled - np.max(scaled))) + 1e-300) / T
    penalty = lam * (x.sum() - n / 2.0) ** 2
    return (val + penalty) * dx

def con_fun(x):
    return x.sum() - len(x) / 2.0

def slsqp_opt(h0, T, maxiter=400, ftol=1e-15):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(lse_obj, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def lbfgs_opt(h0, T, lam, maxiter=3000):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    res = minimize(lse_pen_obj, h0, args=(T, lam), method='L-BFGS-B',
                   bounds=bnds, maxiter=maxiter,
                   options={'gtol': 1e-10, 'maxls': 40})
    return make_feasible(np.clip(res.x, 0, 1), n)

def periodic_obj(xp, p, n, T, lam):
    h = np.tile(xp, n // p)
    return lse_pen_obj(h, T, lam)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[0::2] = 1.0

    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial multi-method optimization
    deadline1 = t0 + budget_s * 0.08
    for T in [200, 1000, 5000, 20000]:
        if time.time() > deadline1:
            break
        for method in ['slsqp', 'lbfgs']:
            if time.time() > deadline1:
                break
            try:
                if method == 'slsqp':
                    h_new = slsqp_opt(best_h.copy(), T, 500, 1e-15)
                else:
                    h_new = lbfgs_opt(best_h.copy(), T, 50.0, 2000)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: Basin hopping with diverse perturbations
    deadline2 = t0 + budget_s * 0.55
    bh_h = best_h.copy()
    bh_c5 = best_c5
    stuck = 0
    max_iter = 300

    for it in range(max_iter):
        if time.time() > deadline2:
            break
        scale = 0.08 * (0.3 + 0.7 * (1.0 - it / max(1, max_iter)))
        ptype = it % 5
        if ptype == 0:
            h_pert = bh_h + scale * np.random.randn(n)
        elif ptype == 1:
            h_pert = bh_h.copy()

            s1 = np.random.randint(0, n - blen)
            s2 = np.random.randint(0, n - blen)
            tmp = h_pert[s1:s1+blen].copy()
            h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
            h_pert[s2:s2+blen] = tmp
        elif ptype == 2:
            h_pert = bh_h.copy()
            blen = np.random.randint(4, max(5, n // 10))
            s = np.random.randint(0, n - blen)
            h_pert[s:s+blen] = h_pert[s:s+blen][::-1]
        elif ptype == 3:

            nf = np.random.randint(4, max(5, n // 6))

        temp = 0.0015 * np.exp(-4.0 * it / max(1, max_iter))

            bh_h, bh_c5 = h_new, c5v

                best_c5, best_h = c5v, h_new
                stuck = 0
            else:

        else:
            stuck += 1
        if stuck > 25:
            bh_h = best_h + 0.12 * np.random.randn(n)
            bh_h = make_feasible(bh_h, n)
            bh_c5 = compute_c5(bh_h)
            stuck = 0

    # Phase 3: Multi-resolution search at different n
    deadline3 = t0 + budget_s * 0.72
    for n_try in [150, 180, 220, 250, 300]:
        if time.time() > deadline3:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for trial in range(2):
            if time.time() > deadline3:
                break
            h0 = h_interp + np.random.uniform(-0.03, 0.03, n_try)
            h0 = make_feasible(h0, n_try)
            for T in [1000, 10000]:
                if time.time() > deadline3:
                    break
                try:
                    h_new = slsqp_opt(h0, T, 300, 1e-14)
                    c5v = compute_c5(h_new)
                    if c5v < best_c5:
                        best_c5, best_h = c5v, h_new
                except Exception:
                    pass

    # Phase 4: Coordinate descent on active set
    deadline4 = t0 + budget_s * 0.90
    n_cur = len(best_h)
    active = np.where((best_h > 0.01) & (best_h < 0.99))[0]
    sweeps = 0
    while time.time() < deadline4 and sweeps < 40 and len(active) > 0:
        improved = False
        np.random.shuffle(active)
        for j in active:
            if time.time() > deadline4:
                break
            for delta in [0.01, -0.01, 0.03, -0.03, 0.06, -0.06, 0.1, -0.1]:
                h_try = best_h.copy()
                h_try[j] += delta
                h_try = make_feasible(h_try, n_cur)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5, best_h = c5_try, h_try
                    improved = True
                    break
        sweeps += 1
        if not improved:
            break
        active = np.where((best_h > 0.01) & (best_h < 0.99))[0]

    # Phase 5: Periodic structure exploitation
    deadline5 = t0 + budget_s * 0.96
    n_cur = len(best_h)
    for p in [4, 6, 8, 10, 12, 16, 20]:
        if time.time() > deadline5:
            break
        n_p = (n_cur // p) * p
        if n_p < 20:
            continue
        xp = np.resize(best_h, p).astype(float)
        xp = np.clip(xp, 0, 1)
        xp = make_feasible(xp, p)
        for T in [500, 5000]:
            if time.time() > deadline5:

                bnds_p = [(0.0, 1.0)] * p
                res = minimize(periodic_obj, xp, args=(p, n_p, T, 200.0),

                xp_opt = np.clip(res.x, 0, 1)

                if c5v < best_c5:
                    best_c5, best_h = c5v, h_try
            except Exception:
                pass

    # Phase 6: Final ultra-high-T SLSQP polish
    for T in [50000, 200000, 1000000]:
        if time.time() > t0 + budget_s * 0.995:

        try:
            h_new = slsqp_opt(best_h.copy(), T, 5000, 1e-17)

            if c5v2 < best_c5:

            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
