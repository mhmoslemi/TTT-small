import time

def compute_c5(h, N):
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_refine(h, N, target, deadline, k, lr0, decay, b1=0.85, b2=0.999):
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    dx = 2.0 / N
    lr = lr0
    eps = 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    stall = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
        if stall > 2000:
            # Restart with higher LR
            lr = lr0 * 2
            stall = 0
            m[:] = 0
            v[:] = 0
            it = 0
    return best_h, best_peak * dx

def binary_anneal(h, N, deadline, rng, T0=5.0, T1=0.003, restart_every=15000):

    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        idx = np.where(h_b == 1)[0]
        h_b[idx[N // 2:]] = 0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N // 2 - n_ones]] = 1
    best = h_b.copy()
    best_pk = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')))
    cur_pk = best_pk
    it = 0

    total_est = max(total_est, 5000)

        ones = np.where(h_b == 1)[0]

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        new_pk = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):

            if new_pk < best_pk:

                best = h_b.copy()
        else:

        it += 1
    return best, best_pk * dx

        if wk < 1e-10:
            continue

        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad * dx, smooth_obj(h, N, tau)
