```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_bin, n, max_iters=50000):
    """Local search on binary sequences with swap moves."""
    h = h_bin.copy()
    c5, _ = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        # Try a few random swaps
        improved = False
        for _ in range(10):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            if i == j:
                continue
            h[i], h[j] = 1.0, 0.0
            new_c5, _ = compute_c5(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
                break
            else:
                h[i], h[j] = 0.0, 1.0
        
        if not improved:
            # Try systematic swaps
            found = False
            for i in ones:
                for j in zeros:
                    if abs(i-j) < 2:  # skip nearby swaps (tried already?)
                        continue
                    h[i], h[j] = 1.0, 0.0
                    new_c5, _ = compute_c5(h)
                    if new_c5 < best_c5 - 1e-15:
                        best_c5 = new_c5
                        best_h = h.copy()
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        found = True
                        improved = True
                        break
                    else:

                if found:
                    break
            if not improved:
                break
    
    return best_h, best_c5
```

```python
import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr
```

```python
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h.copy()
        total = room.sum()
        if total < 1e-14:
            break
        h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_from_h(h, n):
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def sa_binary(h_bin, n, t_start, t_end, seed=42):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    c5, _ = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    T = 0.01
    T_end = 0.0001
    total_time = t_end - t_start
    
    while time.time() < t_end:
        progress = (time.time() - t_start) / total_time
        T = 0.01 * (0.0001/0.01) ** progress
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5, _ = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1)[0].tolist()
            zeros = np.where(h == 0)[0].tolist()
        elif rng.rand() < np.exp(-(new_c5 - c5) / T):
            # Accept worse
            ones = np.where(h == 1)[0].tolist()
            zeros = np.where(h == 0)[0].tolist()
            c5 = new_c5
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c5

def coord_descent(h, n, t_end, max_passes=20):
    """Coordinate descent on continuous values."""
    h = h.copy()
    c5, _ = compute_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            old_val = h[i]
            # Try a few values

                                                      old_val + 0.05, old_val - 0.05,
                                                      old_val + 0.02, old_val - 0.02,
                                                      np.random.rand()]), 0, 1))
            for cv in candidates:
                h[i] = cv
                h = project_feasible(h, n)
                new_c5, _ = compute_c5(h)
                if new_c5 < c5:
                    c5 = new_c5
                    improved = True
                    break
            if not improved:
                h[i] = old_val
                h = project_feasible(h, n)
    
    return h, c5
```

- If j = i+k for some valid i (i.e., k ≤ j ≤ n-1 and i = j-k): derivative of h[j]*(1-h[j-k]) w.r.t. h[j] = (1-h[j-k])

So: dc_k/dh[j] = (1-h[j-k]) * I(k≤j≤n-1) - h[j+k] * I(0≤j≤n-1-k)

```python
def smooth_grad(x, T, n, dx):
    c = np.correlate(x, 1.0-x, mode='full')
    m = np.max(c)
    w = np.exp(T*(c-m))
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2*n-1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n-1)
        if k >= 0:
            # dc/dh[j] = (1-h[j-k])*I(j>=k) + (-h[j+k])*I(j<=n-1-k)
            if k == 0:
                # dc/dh[j] = (1-h[j]) - h[j] = 1-2h[j]
                g[:] += w[idx] * (1.0 - 2.0*x)
            else:
                g[k:] += w[idx] * (1.0 - x[:n-k])  # (1-h[j-k]) for j=k..n-1
                g[:n-k] += -w[idx] * x[k:]  # -h[j+k] for j=0..n-1-k
        else:
            m_val = -k
            if m_val == 0:  # can't happen since k<0
                pass
            else:
                g[:n-m_val] += w[idx] * (1.0 - x[m_val:])  # (1-h[j+m]) for j=0..n-1-m
                g[m_val:] += -w[idx] * x[:n-m_val]  # -h[j-m] for j=m..n-1
    return g * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h.copy()
        total = room.sum()
        if total < 1e-14:
            break
        h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:n // 2]] = 1.0
    return hb

def sa_search(h0, n, t_end, seed=42, cool_rate=3.5, T_init=0.008, T_min=0.00005):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    c5, _ = compute_c5(h)
    best_c5, best_h = c5, h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    t_start = time.time()
    total = max(t_end - t_start, 0.1)
    while time.time() < t_end:
        frac = (time.time() - t_start) / total
        T = max(T_init * (T_min / T_init) ** frac, T_min)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc5, _ = compute_c5(h)
        if nc5 < best_c5:
            best_c5, best_h = nc5, h.copy()
            c5 = nc5
            ones = np.where(h == 1)[0].tolist()
            zeros = np.where(h == 0)[0].tolist()
        elif nc5 - best_c5 < T * 3 and rng.rand() < np.exp(-(nc5 - c5) / T):
            c5 = nc5
            ones = np.where(h == 1)[0].tolist()
            zeros = np.where(h == 0)[0].tolist()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def smooth_obj_fun(x, T, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx

def smooth_grad_fun(x, T, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    w = np.exp(T * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        if k == 0:
            g[:] += w[idx] * (1.0 - 2.0 * x)
        elif k > 0:
            g[k:] += w[idx] * (1.0 - x[:n - k])
            g[:n - k] += -w[idx] * x[k:]
        else:
            m = -k
            g[:n - m] += w[idx] * (1.0 - x[m:])
            g[m:] += -w[idx] * x[:n - m]
    return g * dx

def sum_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_smooth(h, T, maxiter, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    def obj(x):
        return smooth_obj_fun(x, T, n, dx)
    def jac(x):
        return smooth_grad_fun(x, T, n, dx)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def top_lag_subgrad(h, n, t_end, n_iters=5000):
    h = h.copy()
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    lr = 0.08
    stall = 0
    for it in range(n_iters):
        if time.time() > t_end:
            break
        _, corr = compute_c5(h)
        m = np.max(corr)
        thresh = m - 0.5
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k == 0:
                g += (1.0 - 2.0 * h)
            elif k > 0:
                g[k:] += (1.0 - h[:n - k])
                g[:n - k] += -h[k:]
            else:
                m2 = -k
                g[:n - m2] += (1.0 - h[m2:])
                g[m2:] += -h[:n - m2]
        g /= len(active)
        h_new = project_feasible(h - lr * g, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5, best_h = c5_new, h_new
            h = h_new
            lr = min(lr * 1.04, 0.5)
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if stall > 300:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 78
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Binary local search (50% budget)
    t1_end = t0 + budget_s * 0.50
    h_bin = to_binary(best_h, best_n)
    h_sa, c5_sa = sa_search(h_bin, best_n, min(t1_end, t0 + budget_s * 0.20), seed=seed)
    if c5_sa < best_c5:
        best_c5, best_h = c5_sa, h_sa
    # Second SA with perturbation
    for attempt in range(3):
        if time.time() > t1_end:
            break
        h_pert = best_h.copy()
        rng = np.random.RandomState(seed + 1000 + attempt)
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        n_flip = 2 + attempt
        for _ in range(n_flip):
            if len(ones) == 0 or len(zeros) == 0:
                break
            h_pert[ones[rng.randint(len(ones))]] = 0
            h_pert[zeros[rng.randint(len(zeros))]] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
        t_sub = min(t1_end, time.time() + budget_s * 0.08)
        h_sa2, c5_sa2 = sa_search(h_pert, best_n, t_sub, seed=seed + 2000 + attempt)
        if c5_sa2 < best_c5:
            best_c5, best_h = c5_sa2, h_sa2

    # Phase 2: Top-lag subgradient on continuous (10%)
    t2_end = t0 + budget_s * 0.60
    h_sub, c5_sub = top_lag_subgrad(best_h, best_n, t2_end, n_iters=8000)
    if c5_sub < best_c5:
        best_c5, best_h = c5_sub, h_sub

    # Phase 3: SLSQP with escalating T (25%)
    t3_end = t0 + budget_s * 0.85
    for T in [200, 2000, 20000, 200000, 2000000]:
        if time.time() > t3_end:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=500)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
            # Quick subgradient after SLSQP improvement
            h_sg, c5_sg = top_lag_subgrad(best_h, best_n, t3_end * 0.3, n_iters=2000)
            if c5_sg < best_c5:
                best_c5, best_h = c5_sg, h_sg

    # Phase 4: Different n values (10%)
    t4_end = t0 + budget_s * 0.95
    for n_try in [best_n - 2, best_n + 2, best_n + 4, best_n - 4]:
        if time.time() > t4_end or n_try < 40 or n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_b = to_binary(h_interp, n_try)
        t_sub = min(t4_end, time.time() + budget_s * 0.04)
        h_sa, c5_sa = sa_search(h_b, n_try, t_sub, seed=seed + n_try)
        if c5_sa > best_c5:
            h_try = slsqp_smooth(h_interp, 50000, maxiter=300)
            c5_try, _ = compute_c5(h_try)
        else:
            h_try = h_sa
            c5_try = c5_sa
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_try, n_try

    # Phase 5: Final ultra-fine polish (5%)

    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
