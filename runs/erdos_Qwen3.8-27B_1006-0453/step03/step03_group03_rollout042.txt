def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_convex(pd, cd):
    """Minimize max_i(pd[i] + cd[i]*x) for x in [0,1]. Convex PWL."""
    lo, hi = 0.0, 1.0
    for _ in range(70):

        m2 = hi - (hi - lo) / 3.0
        if np.max(pd + cd * m1) <= np.max(pd + cd * m2):
            hi = m2
        else:
            lo = m1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_convex(pd, cd):
    lo = 0.0
    hi = 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(pd + cd * m1) <= np.max(pd + cd * m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coord_descent(h, n_passes=500, lam_init=0.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    h_rev = h[::-1].copy()
    lam = lam_init
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    n_full = 2 * n - 1

    pd = np.zeros(n_full)
    cd = np.zeros(n_full)
    for p in range(n_passes):
        max_diff = 0.0
        order = np.random.permutation(n)

            hm = h[m]
            c[:] = 1.0
            c[n - 1 - m : 2 * n - 1 - m] -= h
            c[m : m + n] -= h_rev

                corr += delta * c
                h[m] = x_new
                h_rev[n - 1 - m] = x_new

        s = h.sum() - n / 2.0
        lam += 0.08 * s

            lam *= 0.5
        if max_diff < 1e-14:
            break
    return np.clip(h, 0, 1)

def _slsqp_obj(x, T, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def _slsqp_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_polish(h, T, maxiter=500):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(T, dx), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

def binary_local_search(h, n, n_iter=400):
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    deficit = n / 2 - h_bin.sum()
    idx = np.argsort(h)[::-1]
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if deficit > 0:
        for i in range(int(deficit)):
            h_bin[zeros[i]] = 1.0
    elif deficit < 0:
        for i in range(int(-deficit)):
            h_bin[ones[i]] = 0.0
    best_h = h_bin.copy()
    best_c5, _ = compute_c5(best_h)
    for it in range(n_iter):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        trial = best_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        c5_t, _ = compute_c5(trial)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            best_h = trial
    return best_h, best_c5

def _erdos_direct_obj(x):
    return x[-1]

def _erdos_direct_eq(x, n):
    return x[:-1].sum() - n / 2.0

def _erdos_direct_ineq(x, n):
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return t - corr * dx

def slsqp_direct(h, maxiter=400):
    n = len(h)
    _, corr = compute_c5(h)
    t0 = float(np.max(corr) * (2.0 / n)) + 0.02
    x0 = np.concatenate([h, [t0]])
    bnds = [(0, 1)] * n + [(0, 2)]
    cons = [{'type': 'eq', 'fun': _erdos_direct_eq, 'args': (n,)},
            {'type': 'ineq', 'fun': _erdos_direct_ineq, 'args': (n,)}]
    try:
        res = minimize(_erdos_direct_obj, x0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x[:n], 0, 1)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    elif 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[0::2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Coordinate descent (fast, exact 1D subproblems)
    if time.time() - t0 < budget_s * 0.25:
        h_cd = coord_descent(best_h, n_passes=1500)
        h_cd = project_feasible(h_cd, best_n)
        c5_cd, _ = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 2: SLSQP polish
    if time.time() - t0 < budget_s * 0.40:
        for T in [3000, 20000, 100000]:
            if time.time() - t0 > budget_s * 0.40:
                break
            h_s = slsqp_polish(best_h, T, maxiter=600)
            h_s = project_feasible(h_s, best_n)
            c5_s, _ = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s

    # Phase 3: Binary local search + coordinate refinement
    if time.time() - t0 < budget_s * 0.55:
        h_bin, c5_bin = binary_local_search(best_h, best_n, n_iter=600)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        h_ref = coord_descent(h_bin, n_passes=800)
        h_ref = project_feasible(h_ref, best_n)
        h_ref = slsqp_polish(h_ref, 50000, maxiter=400)
        h_ref = project_feasible(h_ref, best_n)
        c5_ref, _ = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Phase 4: Multi-resolution with coordinate descent
    res_list = [best_n, best_n - 2, best_n + 2, best_n - 4, best_n + 4, 80, 84, 88, 92, 96]
    for n_try in res_list:
        if time.time() - t0 > budget_s * 0.78:
            break
        if n_try < 40 or n_try % 2 != 0:
            continue
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_feasible(h_base, n_try)
        for trial in range(5):
            if time.time() - t0 > budget_s * 0.78:
                break
            if trial == 0:
                h_p = h_base.copy()
            else:
                h_p = h_base + np.random.randn(n_try) * (0.04 + 0.02 * trial)
            h_p = project_feasible(h_p, n_try)
            h_cd = coord_descent(h_p, n_passes=600)
            h_cd = project_feasible(h_cd, n_try)
            h_s = slsqp_polish(h_cd, 50000, maxiter=400)
            h_s = project_feasible(h_s, n_try)
            c5_try, _ = compute_c5(h_s)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_s
                best_n = n_try
                h_base = h_s.copy()

    # Phase 5: Rotation + coordinate descent
    if time.time() - t0 < budget_s * 0.88:
        for shift in [1, 3, 7, 13, 21]:
            if time.time() - t0 > budget_s * 0.88:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_cd = coord_descent(h_rot, n_passes=800)
            h_cd = project_feasible(h_cd, best_n)
            h_s = slsqp_polish(h_cd, 100000, maxiter=500)
            h_s = project_feasible(h_s, best_n)
            c5_try, _ = compute_c5(h_s)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_s

    # Phase 6: Direct SLSQP + final polish
    if time.time() - t0 < budget_s * 0.95:
        h_d = slsqp_direct(best_h, maxiter=600)
        h_d = project_feasible(h_d, best_n)
        c5_d, _ = compute_c5(h_d)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d
        h_f = coord_descent(best_h, n_passes=2000)
        h_f = project_feasible(h_f, best_n)
        h_f = slsqp_polish(h_f, 300000, maxiter=800)
        h_f = project_feasible(h_f, best_n)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
