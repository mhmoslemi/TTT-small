    """Compute softmax weights for top-K entries of corr"""
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        c_top = corr[top_idx]

    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()

def compute_c5(h, dx):

    h = np.clip(h, 0.0, 1.0)

    lo, hi = -1.5, 1.5

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    if K >= len(corr):
        top_idx = np.arange(len(corr))

    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()

    center = N - 1

            grad[L:N] -= wi * h[0:nv]
        else:
            M = -L
            nv = N - M
            grad[0:nv] -= wi * h[M:N]
            grad[M:N] += wi * (1.0 - h[0:nv])
    return grad, c5

def gradient_phase(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K, iters_target):
    best_h, best_c5 = h.copy(), compute_c5(h, dx)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m, v, it = np.zeros(N), np.zeros(N), 0
    while time.time() < deadline:
        fr = min(it / max(iters_target, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        grad, c5v = topk_grad(h, N, tau, K)
        if c5v < best_c5 - 1e-14:
            best_c5, best_h = c5v, h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        h = project_sum(h - lr * (m / (1 - b1**it)) / (np.sqrt(v / (1 - b2**it)) + eps), target)
    return best_h, best_c5

def paired_coord_descent(h, N, target, dx, n_sweeps, deadline):
    best_h, best_c5 = h.copy(), compute_c5(h, dx)
    rng = np.random.RandomState(777)
    for sw in range(n_sweeps):
        if time.time() > deadline:
            break
        order = np.arange(N)
        rng.shuffle(order)
        for jp in order:
            for jq in order:
                if jp == jq:
                    continue
                a = best_h[jp]
                b = best_h[jq]
                lo_d = max(-a, -(1.0 - b))
                hi_d = min(1.0 - a, b)
                if hi_d - lo_d < 1e-12:
                    continue
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                base_max = float(np.max(corr))
                best_delta, best_val = 0.0, base_max
                n_eval = 60
                deltas = np.linspace(lo_d, hi_d, n_eval)
                cached = np.zeros(len(corr))
                for d in deltas:
                    best_h_try = best_h.copy()
                    best_h_try[jp] = a + d
                    best_h_try[jq] = b - d
                    c_try = compute_c5(best_h_try, dx)
                    if c_try < best_val - 1e-14:
                        best_val = c_try
                        best_delta = d
                if best_delta != 0.0:
                    best_h[jp] = a + best_delta
                    best_h[jq] = b - best_delta
                    if best_val < best_c5 - 1e-14:
                        best_c5 = best_val
    return best_h, best_c5

def local_perturb(h, N, target, dx, n_trials, deadline, scale=0.03):
    best_h, best_c5 = h.copy(), compute_c5(h, dx)
    rng = np.random.RandomState(9999)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        n_mod = rng.randint(1, 5)
        idx = rng.choice(N, size=n_mod, replace=False)

        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-14:
            best_c5, best_h = c, trial
    return best_h, best_c5
