        m2 = hi - (hi - lo) / 3

        f2 = np.max(A * m2 + B)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h

            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_min_1d(A, B):
    """Minimize max_k(A[k]*t + B[k]) over t in [0,1] using ternary search."""
    lo, hi = 0.0, 1.0
    for _ in range(50):

        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(A * m1 + B)
        f2 = np.max(A * m2 + B)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    n = len(h)

    for sweep in range(n_sweeps):
        for i in range(n):

            valid_ok = (iok >= 0) & (iok < n)
            Ah = np.zeros(2*n - 1)
            Ah[valid_bk] = 1.0 - h[ibk[valid_bk]]
