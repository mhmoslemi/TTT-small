def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N - M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_opt(h, N, target, deadline, lr0, decay, k, b1=0.8, b2=0.95):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    stag = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak - 1e-11:
            best_peak = peak
            best_h = h.copy()
            stag = 0
        else:
            stag += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h = project_sum(h, target)
        if it % 50 == 0:
            lr *= decay
            if stag > 120:
                lr = max(1e-4, lr * 0.3)
                stag = 0
    return best_h, best_peak * (2.0 / N)

def binary_sa(h, N, deadline, rng):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    target = N // 2
    n_ones = int(h_b.sum())
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        h_b[idx[target:]] = 0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:target - n_ones]] = 1
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    t_start = time.time()
    total = max(deadline - t_start, 1.0)
    # Estimate ~30-50 iters/sec for N=760
    est = max(int(total * 35), 400)
    T0, T1 = 4.0, 0.002
    peak_L = 0
    for it in range(est):
        if time.time() > deadline:
            break
        frac = it / est
        T = T0 * (T1 / T0) ** frac
        # Re-estimate peak lag every 8 iterations
        if it % 8 == 0:
            corr_arr = compute_corr(h_b)
            peak_idx = int(np.argmax(corr_arr))
            peak_L = peak_idx - (N - 1)
        # Targeted move (60%) or random (40%)
        did_target = False
        if 0 < abs(peak_L) < N and rng.random() < 0.6:
            M = abs(peak_L)
            if peak_L > 0:
                cands = np.where((h_b[:N - M] == 1) & (h_b[M:] == 0))[0]
            else:
                cands = np.where((h_b[M:] == 1) & (h_b[:N - M] == 0))[0]
            if len(cands) > 0:
                i = cands[rng.randint(len(cands))]
                j = i + M if peak_L > 0 else i - M
                h_b[i] = 0.0
                h_b[j] = 1.0
                new_pk = float(np.max(compute_corr(h_b)))
                delta = new_pk - cur_pk
                if delta < 0 or (T > 0.004 and rng.random() < np.exp(-delta / T)):
                    cur_pk = new_pk
                    if new_pk < best_pk:
                        best_pk = new_pk
                        best = h_b.copy()
                else:
                    h_b[i] = 1.0
                    h_b[j] = 0.0
                did_target = True
        if not did_target:
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_b[i] = 0.0
            h_b[j] = 1.0
            new_pk = float(np.max(compute_corr(h_b)))
            delta = new_pk - cur_pk
            if delta < 0 or (T > 0.004 and rng.random() < np.exp(-delta / T)):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = h_b.copy()
            else:
                h_b[i] = 1.0
                h_b[j] = 0.0
    return best, best_pk * dx

def greedy_pass(h_b, N, rng, max_rounds=30):
    """Greedily try best swap at peak lag for several rounds."""
    dx = 2.0 / N
    cur_pk = float(np.max(compute_corr(h_b)))
    for _ in range(max_rounds):
        corr = compute_corr(h_b)
        peak_idx = int(np.argmax(corr))
        L = peak_idx - (N - 1)
        improved = False
        if 0 < abs(L) < N:
            M = abs(L)
            if L > 0:
                cands = np.where((h_b[:N - M] == 1) & (h_b[M:] == 0))[0]
            else:
                cands = np.where((h_b[M:] == 1) & (h_b[:N - M] == 0))[0]
            # Try up to 10 candidates
            for idx in range(min(10, len(cands))):
                i = cands[idx]
                j = i + M if L > 0 else i - M
                h_b[i] = 0.0
                h_b[j] = 1.0
                new_pk = float(np.max(compute_corr(h_b)))
                if new_pk < cur_pk:
                    cur_pk = new_pk
                    improved = True
                    break
                else:
                    h_b[i] = 1.0
                    h_b[j] = 0.0
        if not improved:
            break
    return h_b, cur_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x_src = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)
    best_n = n0

    # Main resolution: 760
    N = 760
    target = N / 2.0
    xN = np.linspace(0.0, 2.0, N, endpoint=False)
    h_init = np.interp(xN, x_src, best_h)
    h_init = project_sum(h_init, target)

    remain = deadline - time.time()
    if remain > 200:
        # Phase 1: Adam k=20, moderate lr
        d1 = time.time() + remain * 0.22
        h1, c1 = adam_opt(h_init.copy(), N, target, d1, 0.07, 0.97, k=20)
        if c1 < best_c5:
            best_c5, best_h, best_n = c1, h1.copy(), N
            x_src = xN

        # Phase 2: Adam k=8, lower lr
        if time.time() < deadline - 80:
            d2 = time.time() + (deadline - time.time()) * 0.18
            h2, c2 = adam_opt(h1.copy(), N, target, d2, 0.02, 0.985, k=8)
            if c2 < best_c5:
                best_c5, best_h, best_n = c2, h2.copy(), N
                x_src = xN
        else:
            h2 = h1

        # Phase 3: Greedy pass then binary SA
        if time.time() < deadline - 40:
            h3 = (h2 > 0.5).astype(float)
            h3, _ = greedy_pass(h3.copy(), N, rng, max_rounds=40)
            d3 = time.time() + (deadline - time.time()) * 0.40
            h3, c3 = binary_sa(h3, N, d3, rng)
            if c3 < best_c5:
                best_c5, best_h, best_n = c3, h3, N
                x_src = xN

            # Phase 4: Second binary SA from slightly perturbed start
            if time.time() < deadline - 25:
                h4 = best_h + rng.randn(N) * 0.05
                h4 = project_sum(h4, target)
                h4, c4 = binary_sa(h4, N, time.time() + (deadline - time.time()) * 0.35, rng)
                if c4 < best_c5:
                    best_c5, best_h, best_n = c4, h4, N
                    x_src = xN

    # Try a second resolution: 920 if time permits
    if time.time() < deadline - 100 and best_n <= N:
        N2 = 920
        target2 = N2 / 2.0
        xN2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h_init2 = np.interp(xN2, x_src, best_h)
        h_init2 = project_sum(h_init2, target2)
        remain2 = deadline - time.time()
        if remain2 > 100:
            d_a = time.time() + remain2 * 0.25
            ha, ca = adam_opt(h_init2.copy(), N2, target2, d_a, 0.05, 0.97, k=12)
            if ca < best_c5:
                best_c5, best_h, best_n = ca, ha.copy(), N2
                x_src = xN2
            if time.time() < deadline - 30:
                hb, cb = binary_sa(ha, N2, deadline - 10, rng)
                if cb < best_c5:
                    best_c5, best_h, best_n = cb, hb, N2
                    x_src = xN2

    # Final continuous polish
    Nb = len(best_h)
    if time.time() < deadline - 3:
        hF = best_h + rng.randn(Nb) * 0.01
        hF = project_sum(hF, Nb / 2.0)
        hF, cF = adam_opt(hF, Nb, Nb / 2.0, deadline - 1, 0.012, 0.991, k=6)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
