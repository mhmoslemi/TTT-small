def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_active(h, active_idx, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_idx]
    m = np.max(vals)
    val = m + np.log(np.sum(np.exp(T * (vals - m)))) / T
    return val * dx

    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    return idx

    h_bin[order[n // 2:]] = 1.0
    return h_bin

    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.04

    v = np.zeros(n)

        m = np.max(corr)
        thresh = m - 0.02 * m

            else:

            stall = 0

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_active_fn(h, active_idx, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_idx]
    m = np.max(vals)
    val = m + np.log(np.sum(np.exp(T * (vals - m)))) / T
    return val * dx

def smooth_max_all_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_active(h, active_idx, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_max_active_fn, h, args=(active_idx, T),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_full(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_max_all_fn, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def get_active(corr, frac=0.04, max_count=10):
    m = np.max(corr)
    if m <= 0:
        return np.array([len(corr) // 2])
    thresh = m - frac * m
    idx = np.where(corr >= thresh)[0]
    if len(idx) > max_count:
        order = np.argsort(corr[idx])[-max_count:]
        idx = idx[order]
    return idx

def binary_project(h, n):
    order = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[order[n // 2:]] = 1.0
    return h_bin

def binary_greedy(h, max_rounds=80):
    n = len(h)
    h_cur = binary_project(h, n).copy()
    best_c5, _ = compute_c5(h_cur)
    for _ in range(max_rounds):
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        best_val = best_c5
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                c5_t, _ = compute_c5(h_cur)
                h_cur[i] = 1.0
                h_cur[j] = 0.0
                if c5_t < best_val - 1e-14:
                    best_val = c5_t
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h_cur[best_i] = 0.0
        h_cur[best_j] = 1.0
        best_c5 = best_val
    return h_cur, best_c5

def perturb_binary(h, n, n_flips, rng):
    h_bin = binary_project(h, n).copy()
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for _ in range(n_flips):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        mask_o = ones[ones != i]
        mask_z = zeros[zeros != j]
        if len(mask_o) == 0 or len(mask_z) == 0:
            break
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
    return h_bin

def momentum_subgrad(h, n_iters=2000, rng=None):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.035
    mom = 0.82
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(h_cur)
        m = np.max(corr)
        thresh = m - 0.025 * m
        active = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                grad[:n - k] += (1.0 - h_cur[k:])
                grad[k:] += -h_cur[:n - k]
            else:
                kp = -k
                grad[kp:] += (1.0 - h_cur[:n - kp])
                grad[:n - kp] += -h_cur[kp:]
        grad /= len(active)
        v = mom * v - lr * grad
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.25)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if stall > 250:
                break
    return best_h, best_c5

def optimize_from_binary(h_bin, budget_left_frac, t0, budget_s):
    n = len(h_bin)
    h = h_bin.copy()
    best_c5, _ = compute_c5(h)
    _, corr = compute_c5(h)
    active = get_active(corr, frac=0.05, max_count=8)
    T_vals = [3000, 10000, 40000]
    for T in T_vals:
        elapsed = time.time() - t0
        if elapsed > budget_s * (1.0 - budget_left_frac):
            h = binary_greedy(h, max_rounds=30)[0]
            c5_try, _ = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
            break
        h_try = slsqp_active(h, active, T, maxiter=400)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            h = h_try
        _, corr_t = compute_c5(h)
        active = get_active(corr_t, frac=0.04, max_count=8)
        h_bin_new, c5_bin = binary_greedy(h, max_rounds=20)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            h = h_bin_new
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.concatenate([np.ones(39), np.zeros(39)])
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Active-set SLSQP warm-up
    for rnd in range(6):
        if time.time() - t0 > budget_s * 0.12:
            break
        _, corr = compute_c5(best_h)
        active = get_active(corr, frac=0.04, max_count=8)
        T = 1500 * (rnd + 1)
        h_try = slsqp_active(best_h, active, T, maxiter=350)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Binary greedy local search (core novelty)
    if time.time() - t0 < budget_s * 0.25:
        h_bin, c5_bin = binary_greedy(best_h, max_rounds=100)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        h_reopt, c5_re = optimize_from_binary(best_h, 0.20, t0, budget_s)
        if c5_re < best_c5:
            best_c5 = c5_re
            best_h = h_reopt

    # Phase 3: Momentum subgradient on continuous
    if time.time() - t0 < budget_s * 0.40:
        h_sub, c5_sub = momentum_subgrad(best_h, n_iters=2500, rng=rng)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    # Phase 4: Multiple binary perturbations + greedy + re-optimize
    if time.time() - t0 < budget_s * 0.60:
        for n_flips in [1, 2, 3, 4, 5, 6]:
            if time.time() - t0 > budget_s * 0.60:
                break
            for trial in range(3):
                if time.time() - t0 > budget_s * 0.60:
                    break
                h_pert = perturb_binary(best_h, best_n, n_flips, rng)
                h_g, c5_g = binary_greedy(h_pert, max_rounds=50)
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = h_g
                h_o, c5_o = optimize_from_binary(h_g, 0.05, t0, budget_s)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o

    # Phase 5: Multi-resolution exploration
    if time.time() - t0 < budget_s * 0.75:
        for dn in [-4, -2, 2, 4, 6, 8]:
            if time.time() - t0 > budget_s * 0.75:
                break
            n_try = best_n + dn
            if n_try < 40 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_bin = binary_project(h_interp, n_try)
            h_g, c5_g = binary_greedy(h_bin, max_rounds=40)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g
                best_n = n_try
            h_o, c5_o = optimize_from_binary(h_g, 0.04, t0, budget_s)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o
                best_n = n_try

    # Phase 6: Continuous SLSQP polish with very high T
    if time.time() - t0 < budget_s * 0.85:
        _, corr_p = compute_c5(best_h)
        active_p = get_active(corr_p, frac=0.02, max_count=6)
        h_p = slsqp_active(best_h, active_p, 80000, maxiter=600)
        c5_p, _ = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 7: Another binary greedy round from polished solution
    if time.time() - t0 < budget_s * 0.93:
        h_g2, c5_g2 = binary_greedy(best_h, max_rounds=60)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2
        h_m, c5_m = momentum_subgrad(best_h, n_iters=1000, rng=rng)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m

    # Phase 8: Final SLSQP with full objective
    if time.time() - t0 < budget_s * 0.97:
        h_f = slsqp_full(best_h, 100000, maxiter=500)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_bg, c5_bg = binary_greedy(best_h, max_rounds=40)
        if c5_bg < best_c5:
            best_c5 = c5_bg
            best_h = h_bg

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
