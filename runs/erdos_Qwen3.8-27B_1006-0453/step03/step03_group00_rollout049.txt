    if elapsed() < budget_s * 0.65:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.65, rng,
                                     n_iters=150, base_step=0.04, T_smooth=400)

    # Phase 5: More coordinate descent after basin hopping
    if elapsed() < budget_s * 0.78:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.78, max_sweeps=40, rng=rng)

    # Phase 6: Multi-resolution
    for n_try in [120, 144, 192, 200, 240]:
        if elapsed() > budget_s * 0.90:
            break
        h_interp = interpolate_h(best_h, n_try)
        for T in [200, 1500]:
            if elapsed() > budget_s * 0.92:
                break
            try:
                hc = slsqp_opt(h_interp, T, n_try, maxiter=1000, ftol=1e-14)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5, n = hc, cv, n_try
                    h_interp = interpolate_h(best_h, n_try)
            except Exception:
                pass
        if n == n_try:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.93, max_sweeps=20, rng=rng)

    # Phase 7: Final coordinate descent + SLSQP
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=30, rng=rng)
    for T in [15000, 50000]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-15)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
