import numpy as np
from scipy.optimize import minimize
import time

_NG = 0  # global for constraint

def _con_fun(x):
    return x.sum() - _NG * 0.5

def _con_jac(x):
    return np.ones(_NG)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * 2.0 / n)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    e = np.exp(T * (corr - m))
    e_sum = e.sum()
    w = e / e_sum
    val = (m + np.log(e_sum) / T) * dx
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        if s == 0:
            g = 1.0 - 2.0 * h
        elif s > 0:
            g = np.zeros(n)
            g[s:] = 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
        else:
            ss = -s
            g = np.zeros(n)
            g[:n - ss] = 1.0 - h[ss:]
            g[ss:] -= h[:n - ss]
        grad += wi * g
    return val, grad * dx

def project_bp(h, target):
    def s_lam(lam):
        return np.clip(h - lam, 0.0, 1.0).sum()
    lo, hi = -10.0, 10.0
    if s_lam(lo) < target:
        lo = -100.0
    if s_lam(hi) > target:
        hi = 100.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if s_lam(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter):
    global _NG
    _NG = n
    def _obj(x):
        v, _ = smooth_obj_grad(x, T)
        return v
    def _grad(x):
        _, g = smooth_obj_grad(x, T)
        return g
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return np.clip(res.x, 0.0, 1.0)

def subgrad_loop(h, n, max_iter, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = 0.005
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = compute_corr(best_h)
        m = corr.max()
        active = np.where(corr >= m - 1e-12)[0]
        g = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)
            if s == 0:
                g += 1.0 - 2.0 * best_h
            elif s > 0:
                gs = np.zeros(n)
                gs[s:] = 1.0 - best_h[:n - s]
                gs[:n - s] -= best_h[s:]
                g += gs
            else:
                ss = -s
                gs = np.zeros(n)
                gs[:n - ss] = 1.0 - best_h[ss:]
                gs[ss:] -= best_h[:n - ss]
                g += gs
        g /= len(active)
        h_new = project_bp(best_h - lr * g, n * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.08, 0.05)
        else:
            lr *= 0.8
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=50):
    from scipy.optimize import minimize_scalar
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = compute_corr(best_h)
        dx = 2.0 / n
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = compute_corr(h_base)
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i + k
                j2 = i - k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    if k != 0:
                        c += (-best_h[j2])
                    else:
                        c += (1.0 - best_h[j2])
                elif k == 0:
                    c += (1.0 - best_h[j2])
                coef[k + n - 1] = c
            def f1d(hi):
                return float(np.max(corr_base + hi * coef) * dx)
            try:
                res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-10})
                hi_new = res.x
            except Exception:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_bp(h_try, n * 0.5)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                corr = compute_corr(best_h)
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:

        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_bp(best_h, n * 0.5)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP T-schedule
    t1 = t0 + budget_s * 0.15
    for T in [8, 30, 100, 400, 1500, 5000]:
        if elapsed() > t1:

        try:

            pass

    # Phase 2: Subgradient polish
    t2 = t0 + budget_s * 0.22

    t3 = t0 + budget_s * 0.60
    work_h = best_h.copy()
    work_c5 = best_c5
    it = 0
    while elapsed() < t3:
        it += 1
        cool = max(0.0, 1.0 - it / 120.0)
        step = 0.03 + 0.18 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = project_bp(h_pert, n * 0.5)
        T_bh = 400 if cool > 0.3 else 1500
        try:

            hc = project_bp(hc, n * 0.5)
            c5v = compute_c5(hc)
            temp = 0.006 * cool + 0.0004
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_h = hc.copy()
                    best_c5 = c5v
        except Exception:
            pass
    if work_c5 < best_c5:
        best_h = work_h.copy()

    # Phase 4: Coordinate descent
    t4 = t0 + budget_s * 0.78

        x_new = np.linspace(0, 2, n_try, endpoint=False)

                    best_c5 = c5v
                    n = n_try
            except Exception:
                pass

    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = subgrad_loop(best_h, len(best_h), 3000, t0 + budget_s * 0.99)
        try:

    best_h = project_bp(best_h, len(best_h) * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
