        # SLSQP from binary
        for T in [80, 400, 2000]:
            if elapsed() > budget_s * 0.38:
                break
            try:
                ht = slsqp(h_bin, n0, T, 5, 500, 1e-14)
                cv = compute_c5(ht)
                if cv < best_c5:
                    best_h, best_c5 = ht, cv
            except Exception:
                pass

    # Phase 3: Coordinate descent on worst shifts
    if elapsed() < budget_s * 0.45:
        best_h, best_c5 = coord_worst_shift(best_h, n0, 300)

    # Phase 4: Basin hopping with diverse perturbations
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = 120
    for it in range(n_bh):
        if elapsed() > budget_s * 0.72:
            break
        step = 0.10 * (1.0 - it / n_bh) + 0.015
        r = rng.rand()
        if r < 0.3:
            h_p = bh_h + step * np.random.randn(n0)
        elif r < 0.55:
            h_p = bh_h.copy()
            bl = rng.randint(4, n0 // 3)
            st = rng.randint(0, n0 - bl)
            h_p[st:st + bl] = 1.0 - h_p[st:st + bl]
        elif r < 0.8:
            h_p = bh_h.copy()
            sh = rng.randint(2, n0 // 5)
            h_p[sh:] = 0.4 * bh_h[:n0 - sh] + 0.6 * bh_h[sh:]
            h_p[:sh] = 0.6 * bh_h[:sh] + 0.4 * bh_h[n0 - sh:]
        else:
            h_p = bh_h.copy()
            i1 = rng.randint(n0)
            i2 = rng.randint(n0)
            h_p[i1] += step
            h_p[i2] -= step
        h_p = project(h_p, n0)
        try:
            T_u = rng.choice([80, 400, 1500, 5000])
            k_u = rng.choice([None, 3, 5, 8])
            ht = slsqp(h_p, n0, T_u, k_u, 250, 1e-13)
            cv = compute_c5(ht)
            temp = 0.004 * (1.0 - it / n_bh) + 0.0001
            if cv < bh_c5 or rng.rand() < np.exp(-(cv - bh_c5) / max(temp, 1e-12)):
                bh_h, bh_c5 = ht, cv
                if cv < best_c5:
                    best_h, best_c5 = ht, cv
        except Exception:
            pass

    # Phase 5: Multi-resolution
    for n_try in [100, 120, 200]:
        if elapsed() > budget_s * 0.88:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, n_try, endpoint=False)
        hi = project(np.interp(xn, xo, best_h), n_try)
        for trial in range(2):
            if elapsed() > budget_s * 0.91:
                break
            h0 = project(hi + rng.uniform(-0.04, 0.04, n_try), n_try)
            try:
                for T in [80, 500, 3000]:
                    for k in [None, 5]:
                        ht = slsqp(h0, n_try, T, k, 350, 1e-14)
                        cv = compute_c5(ht)
                        if cv < best_c5:
                            best_h, best_c5 = ht, cv
                            h0 = ht
            except Exception:
                pass

    # Phase 6: Final polish - high T + coord descent
    n_cur = len(best_h)
    for T in [8000, 30000, 100000]:
        if elapsed() > budget_s * 0.97:
            break
        try:
            ht = slsqp(best_h, n_cur, T, 3, 800, 1e-15)
            cv = compute_c5(ht)
            if cv < best_c5:
                best_h, best_c5 = ht, cv
        except Exception:
            pass
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_worst_shift(best_h, n_cur, 200)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
