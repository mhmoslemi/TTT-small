    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    target = n / 2.0

    if abs(s - target) < 1e-15:

    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + diff * room / tot
    else:
        tot = h.sum()
        if tot < 1e-15:
            return h
        h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + diff * room / tot

        if tot < 1e-15:

        s2 = h.sum()
        diff2 = target - s2

            tot2 = room.sum()

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + diff * room / tot
    else:
        tot = h.sum()
        if tot < 1e-15:
            return h
        h = np.clip(h * (target / tot), 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if abs(diff2) > 1e-10:
            room = 1.0 - h
            tot2 = room.sum()
            if tot2 > 1e-15:
                h = h + diff2 * room / tot2
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    hi_old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    pos = shifts + mid
    j1 = i - shifts
    j2 = i + shifts
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coef = np.zeros(n_corr)
    coef[pos[v1]] += (1.0 - h[j1[v1]])
    coef[pos[v2]] -= h[j2[v2]]
    cb = corr_base.copy()

    def f1d(t):
        vals = cb + t * coef
        vals[mid] = cb[mid] + t - t * t
        return np.max(vals) * dx

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    h[i] = hi_old
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500, tol=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_opt(best_h, n, i)
            if abs(t_new - best_h[i]) > 1e-13:
                best_h[i] = t_new
                np.clip(best_h, 0.0, 1.0, out=best_h)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def interp_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def make_random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def binary_local_search(n, t_end, rng, max_flips=50000):
    h = make_random_binary(n, rng)
    c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for it in range(max_flips):
        if time.time() > t_end:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new > c5 + 1e-12:
            h[i] = 1.0
            h[j] = 0.0
        else:
            c5 = c5_new
            mask_o = np.where(h > 0.5)[0]
            mask_z = np.where(h < 0.5)[0]
            ones = mask_o
            zeros = mask_z
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    total_budget = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n_best = len(best_h)
    best_h = project(best_h, n_best)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on parent
    t_limit = t0 + total_budget * 0.18
    best_h, best_c5 = coord_descent(best_h, n_best, t_limit, max_sweeps=800)

    # Phase 2: Multi-resolution exploration
    t_limit = t0 + total_budget * 0.32
    n_candidates = [80, 100, 120, 140, 160, 180, 200, 220, 260, 300]
    for n_try in n_candidates:
        if time.time() > t_limit:

        if n_try == n_best:

        h_start = project(interp_h(best_h, n_best, n_try), n_try)
        h_opt, c5_opt = coord_descent(h_start, n_try, t_limit, max_sweeps=400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_best = n_try

    # Phase 3: Many random restarts with coordinate descent
    t_limit = t0 + total_budget * 0.75
    n_cd = 50
    n_restarts = 0
    while time.time() < t_limit:
        frac = float(n_restarts) / 800.0
        amp = 0.06 + 0.10 * (1.0 - min(frac, 1.0))
        h_pert = best_h + amp * rng.standard_normal(n_best)
        h_pert = project(h_pert, n_best)
        h_opt, c5_opt = coord_descent(h_pert, n_best, t_limit, max_sweeps=n_cd)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        n_restarts += 1

    # Phase 4: Binary-inspired restarts
    t_limit = t0 + total_budget * 0.88
    for n_try in [n_best, max(60, n_best - 20), min(400, n_best + 40)]:
        if time.time() > t_limit:
            break
        for rb in range(30):
            if time.time() > t_limit:
                break
            h_bin, c5_bin = binary_local_search(n_try, t_limit, rng, max_flips=8000)
            h_relax = project(h_bin.astype(float), n_try)
            h_opt, c5_opt = coord_descent(h_relax, n_try, t_limit, max_sweeps=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n_best = n_try

    # Phase 5: Second round of small restarts from best
    t_limit = t0 + total_budget * 0.95
    for r in range(200):
        if time.time() > t_limit:
            break
        amp = 0.04 + 0.04 * (1.0 - r / 200.0)
        h_pert = best_h + amp * rng.standard_normal(n_best)
        h_pert = project(h_pert, n_best)
        h_opt, c5_opt = coord_descent(h_pert, n_best, t_limit, max_sweeps=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_best = n_best

    # Phase 6: Final deep coordinate descent
    best_h, best_c5 = coord_descent(best_h, n_best, t0 + total_budget * 0.99, max_sweeps=2000)

    best_h = project(best_h, n_best)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_best
