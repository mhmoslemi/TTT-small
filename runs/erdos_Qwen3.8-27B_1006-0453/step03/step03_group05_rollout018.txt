    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = (1.0 - h)
            s = room.sum()
            if s < 1e-15: break
            h += room * (diff / s)
        else:
            s = h.sum()
            if s < 1e-15: break
            h += h * (diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / n)

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def slsqp(h, obj, args, maxiter, ftol=1e-14):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        r = minimize(obj, h, args=args, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(r.x, 0, 1), n)
    except:
        return h

    for it in range(n_iters):

        zeros = np.where(b < 0.5)[0]
        # Try targeted swaps

        worst = np.argmax(corr)
        shift = int(worst - (n-1))
        improved = False
        # Targeted: for the worst shift, find i where b[i]=1, b[i+shift]=0
        for trial in range(50):

                if len(valid) > 0:

                    j = int(i + shift)

                        tc = np.correlate(tb, 1-tb, mode='full').max()
                        if tc < best_corr_max:
                            b = tb; best_corr_max = tc; improved = True
                            break

            # Random swap
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]

            if tc < best_corr_max:

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
