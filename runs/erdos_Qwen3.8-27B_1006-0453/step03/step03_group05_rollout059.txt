        if abs(diff) < 1e-14:

            total = room.sum()
            if total < 1e-14:

            h += room * (diff / total)
        else:

            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_avg_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_opt(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

    best_c5 = compute_c5(h)

            break

        top_k_idx = np.argsort(corr)[::-1][:top_k]
        top_k_shifts = top_k_idx.astype(int).tolist()
        ov = corr[top_k_idx].astype(float).copy()
        
        order = np.random.permutation(n)
        for i in order:
            a_s = np.zeros(top_k)
            for idx in range(top_k):
                s_off = top_k_shifts[idx] - (n - 1)
                a = 0.0
                if 0 <= i + s_off < n:
                    a += (1.0 - h[i + s_off])
                if 0 <= i - s_off < n:
                    a -= h[i - s_off]
                a_s[idx] = a
            
            b_s = ov - a_s * h[i]
            
            candidates = [0.0, 1.0]
            for p in range(top_k):
                for q in range(p + 1, top_k):
                    da = a_s[p] - a_s[q]
                    if abs(da) > 1e-15:
                        x = (b_s[q] - b_s[p]) / da
                        if 0.0 < x < 1.0:
                            candidates.append(x)
            
            best_val = np.inf
            best_hi = h[i]
            for x in candidates:
                val = np.max(a_s * x + b_s)
                if val < best_val:
                    best_val = val
                    best_hi = x
            
            if abs(best_hi - h[i]) > 1e-15:
                h[i] = best_hi
                ov[:] = a_s * h[i] + b_s
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5

                hp = best_h.copy()
                i, j = np.random.choice(best_n, 2, replace=False)
                hp[i], hp[j] = 1.0 - hp[j], 1.0 - hp[i]
                hp = project_feasible(hp, best_n)
            else:
                # Gaussian in a window
                hp = best_h.copy()
                ctr = np.random.randint(0, best_n)
                w = np.random.randint(5, max(6, best_n // 4))
                mask = np.zeros(best_n)
                lo, hi = max(0, ctr - w), min(best_n, ctr + w)
                mask[lo:hi] = 1.0
                hp = project_feasible(hp + np.random.randn(best_n) * 0.15 * mask, best_n)
            # Refine
            hr, cr = exact_coord_descent(hp, best_n, 5, 40, t0 + 6.0)
            hr = slsqp_opt(hr, topk_avg_obj, (3, best_n), 150, 1e-13)
            hr, cr2 = exact_coord_descent(hr, best_n, 3, 30, t0 + 4.0)
            cr2 = compute_c5(hr)
            cr = min(cr, cr2)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr

    # Phase 5: Intensive final polish
    if elapsed() < budget_s * 0.92:
        for tk in [3, 5, 8]:
            if elapsed() > budget_s * 0.90:
                break
            h_cd, c5_cd = exact_coord_descent(best_h, best_n, tk, 100,
                                              t0 + budget_s * 0.02)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd
        if elapsed() < budget_s * 0.92:
            hf = slsqp_opt(best_h, softmax_obj, (30000, best_n), 500, 1e-15)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5 = cf
                best_h = hf
            hf2 = slsqp_opt(best_h, topk_avg_obj, (3, best_n), 300, 1e-15)
            cf2 = compute_c5(hf2)
            if cf2 < best_c5:
                best_c5 = cf2
                best_h = hf2

    # Phase 6: Try alternate resolutions
    if elapsed() < budget_s * 0.97:
        for nr in [100, 150]:
            if elapsed() > budget_s * 0.96:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, nr)
            hu = np.interp(x_new, x_old, best_h)
            hu = project_feasible(hu, nr)
            hr, cr = exact_coord_descent(hu, nr, 5, 50, t0 + 5.0)
            hr = slsqp_opt(hr, topk_avg_obj, (3, nr), 200, 1e-13)
            cr = compute_c5(hr)
            hr, cr2 = exact_coord_descent(hr, nr, 3, 40, t0 + 4.0)
            cr2 = compute_c5(hr)
            cr = min(cr, cr2)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr
                best_n = nr

    # Final: one more CD pass + SLSQP
    if elapsed() < budget_s * 0.99:
        h_cd, c5_cd = exact_coord_descent(best_h, best_n, 3, 50, t0 + 3.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        hf = slsqp_opt(best_h, topk_avg_obj, (3, best_n), 200, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
