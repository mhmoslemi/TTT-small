    v = np.zeros(n)
    best = h0.copy()
    best_c5 = compute_c5(h0)
    alpha = alpha0
    for it in range(n_steps):

        y = h + beta * (h - h_prev)
        g = grad_max_approx(y, T)

        h_prev = h
        h = h_new
        alpha *= decay
        c5 = compute_c5(h)
        if c5 < best_c5:

            best = h.copy()
    return best, best_c5

def coord_descent(h, n_iters=50):
    """Coordinate descent: for each i, optimize h[i] in 1D."""
    n = len(h)
    h = h.copy()
    target = n / 2.0
    for it in range(n_iters):
        improved = False
        for i in range(n):
            # Optimize h[i] with others fixed
            # The correlations are affine in h[i]
            # We need sum(h) = target, so if we change h[i], we need to adjust
            # Actually, let's handle the constraint by Lagrange or by fixing sum
            # For now, just optimize h[i] in [0,1] ignoring sum, then project
            # Better: optimize h[i] subject to sum constraint by adjusting one other coordinate
            
            # Simpler: just do a small search
            old_val = h[i]
            # Try values in [0,1]
            # The objective max_k c_k is convex in h[i] (max of affine functions)
            # Find minimum by ternary search or by finding breakpoints
            
            # Compute c_k as function of h[i]:
            # c_k = a_k * h[i] + b_k where a_k, b_k depend on other h values
            # For each shift s:
            #   c_s contributes to a: (1-h[i+s]) if valid, -h[i-s] if valid
            #   c_s contributes to b: rest
            
            # Let's compute a_k and b_k for the top shifts
            # Actually, for the max, we need all shifts
            # But we can approximate with top-K shifts
            
            corr_base = compute_corr(h)  # current correlations
            centers = n - 1
            # Top K shifts
            K = min(10, 2*n-1)
            top_idx = np.argpartition(corr_base, -K)[-K:]
            
            # For each top shift, compute a_k (coefficient of h[i]) and b_k (constant)
            a_vals = np.zeros(K)
            b_vals = np.zeros(K)
            for xi, k in enumerate(top_idx):
                s = int(k) - centers
                a = 0.0
                b = corr_base[k] - a * old_val  # We'll recompute
                # c_s = sum_j h[j]*(1-h[j+s]) for valid j
                # Terms involving h[i]:
                # If 0<=i<n and 0<=i+s<n: term is h[i]*(1-h[i+s]), contributes (1-h[i+s])*h[i] to c_s
                #   So a gets (1-h[i+s]) if i+s valid
                # If 0<=i-s<n and 0<=i<n: term is h[i-s]*(1-h[i]), contributes -h[i-s]*h[i] to c_s
                #   So a gets -h[i-s] if i-s valid
                if 0 <= i+s < n:
                    a += (1.0 - h[i+s])
                if 0 <= i-s < n:
                    a -= h[i-s]
                a_vals[xi] = a
                # b_k = c_k - a_k * h[i]
                b_vals[xi] = corr_base[k] - a_vals[xi] * old_val
            
            # Now minimize max_xi (a_vals[xi] * v + b_vals[xi]) for v in [max(0, target-sum_others+sum_others...)]
            # Actually, just minimize on [0,1] for now
            # This is a 1D convex piecewise-linear minimization
            # The minimum is at a breakpoint or at a boundary
            
            # Find all breakpoints: where a_vals[xi]*v + b_vals[xi] = a_vals[xj]*v + b_vals[xj]
            # v = (b_vals[xj] - b_vals[xi]) / (a_vals[xi] - a_vals[xj])
            
            candidates = [0.0, 1.0]
            for xi in range(K):
                for xj in range(xi+1, K):
                    da = a_vals[xi] - a_vals[xj]
                    if abs(da) > 1e-12:
                        v = (b_vals[xj] - b_vals[xi]) / da
                        if 0.0 <= v <= 1.0:
                            candidates.append(v)
            
            candidates = np.array(candidates)
            best_v = candidates[0]
            best_val = np.max(a_vals * best_v + b_vals)
            for v in candidates[1:]:
                val = np.max(a_vals * v + b_vals)
                if val < best_val:
                    best_val = val
                    best_v = v
            
            if abs(best_v - old_val) > 1e-8:
                h[i] = best_v
                improved = True
        
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def slsqp_polish(h, T, maxiter=500):
    n = len(h)
    def obj(x):
        corr = compute_corr(x)
        return softmax_max(corr, T) * (2.0/n)
    def obj_grad(x):
        return (grad_max_approx(x, T) * (2.0/n))
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = np.clip(1.0 - h, 0, 1)
            tr = room.sum()
            if tr < 1e-15:
                break
            h += room * (diff / tr)
        else:
            val = np.clip(h, 0, 1)
            tv = val.sum()
            if tv < 1e-15:
                break
            h += val * (diff / tv)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in np.argsort(1.0 - h)[::-1]:
                d = min(diff, 1.0 - h[i])
                h[i] += d
                diff -= d
                if diff < 1e-14:
                    break
        else:
            for i in np.argsort(h):
                d = min(-diff, h[i])
                h[i] -= d
                diff += d
                if diff > -1e-14:
                    break
    return h

def softmax_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def softmax_weights(corr, T):
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    return e / e.sum()

def grad_softmax(h, T):
    n = len(h)
    corr = compute_corr(h)
    w = softmax_weights(corr, T)
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-16:
            continue
        l = k - center
        if l >= 0:
            if n - l > 0:
                grad[l:n] += wk * (1.0 - h[:n-l])
                grad[:n-l] += -wk * h[l:n]
        else:
            t = -l
            if n - t > 0:
                grad[:n-t] += wk * (1.0 - h[t:n])
                grad[t:n] += -wk * h[:n-t]
    return grad

def nesterov_descent(h0, T, n_steps, alpha0, beta, decay):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    v = np.zeros(n)
    best = h0.copy()
    best_c5 = compute_c5(h0)
    alpha = alpha0
    for it in range(n_steps):
        y = h + beta * (h - h_prev)
        g = grad_softmax(y, T)
        v = beta * v - alpha * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        h_prev = h
        h = h_new
        alpha *= decay
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def coord_descent(h, n_iters=40, topK=12):
    n = len(h)
    h = h.copy()
    for it in range(n_iters):
        improved = False
        corr = compute_corr(h)
        center = n - 1
        top_idx = np.argpartition(corr, -topK)[-topK:] if topK < 2*n-1 else np.arange(2*n-1)
        for i in range(n):
            a_vals = np.zeros(len(top_idx))
            b_vals = np.zeros(len(top_idx))
            for xi, k in enumerate(top_idx):
                l = int(k) - center
                a = 0.0
                if l >= 0:
                    if l <= i < n:
                        a += (1.0 - h[i-l])
                    if 0 <= i < n - l:
                        a -= h[i+l]
                else:
                    t = -l
                    if 0 <= i < n - t:
                        a += (1.0 - h[i+t])
                    if t <= i < n:
                        a -= h[i-t]
                a_vals[xi] = a
                b_vals[xi] = corr[k] - a_vals[xi] * h[i]
            candidates = [0.0, 1.0]
            kk = len(top_idx)
            for xi in range(kk):
                for xj in range(xi + 1, kk):
                    da = a_vals[xi] - a_vals[xj]
                    if abs(da) > 1e-14:
                        vv = (b_vals[xj] - b_vals[xi]) / da
                        if -1e-9 <= vv <= 1.0 + 1e-9:
                            candidates.append(max(0.0, min(1.0, vv)))
            candidates = np.array(candidates)
            best_v = candidates[0]
            best_val = np.max(a_vals * best_v + b_vals)
            for vv in candidates[1:]:
                val = np.max(a_vals * vv + b_vals)
                if val < best_val:
                    best_val = val
                    best_v = vv
            if abs(best_v - h[i]) > 1e-7:
                h[i] = best_v
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def slsqp_polish(h, T, maxiter=400):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        return softmax_max(compute_corr(x), T) * dx
    def obj_grad(x):
        return grad_softmax(x, T) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov with multiple temperatures
    for T in [30, 80, 150, 400, 1000]:
        if time.time() - t0 > budget_s * 0.22:
            break
        h_new, c5_new = nesterov_descent(best_h, T, 400, 0.012, 0.85, 0.9996)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 2: Coordinate descent (exact 1D subproblems)
    if time.time() - t0 < budget_s * 0.35:
        for _ in range(4):
            if time.time() - t0 > budget_s * 0.35:
                break
            h_cd = coord_descent(best_h, n_iters=40, topK=15)
            c5_cd = compute_c5(h_cd)
            if c5_cd < best_c5:
                best_c5, best_h = c5_cd, h_cd

    # Phase 3: SLSQP with gradient
    if time.time() - t0 < budget_s * 0.50:
        for T in [100, 400, 2000]:
            if time.time() - t0 > budget_s * 0.50:
                break
            h_s = slsqp_polish(best_h, T, 500)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s

    # Phase 4: Basin hopping
    end_hopping = budget_s * 0.72
    trial = 0
    while time.time() - t0 < end_hopping and trial < 60:
        trial += 1
        r = np.random.random()
        if r < 0.25:
            sc = 0.02 + 0.01 * np.random.random()
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.45:
            hp = best_h.copy()
            bs = int(np.random.choice([8, 15, 25, 40]))
            start = np.random.randint(0, max(1, best_n - bs))
            hp[start:start+bs] = np.random.permutation(hp[start:start+bs])
            hp = project_feasible(hp, best_n)
        elif r < 0.65:
            corr = compute_corr(best_h)
            k_star = int(np.argmax(corr))
            s_star = k_star - (best_n - 1)
            hp = best_h.copy()
            for j in range(best_n):
                j2 = j + s_star
                if 0 <= j2 < best_n:
                    hp[j] = 1.0 - hp[j]
                    hp[j2] = 1.0 - hp[j2]
            hp = project_feasible(hp, best_n)
        else:
            corr = compute_corr(best_h)
            top3 = np.argpartition(corr, -3)[-3:]
            hp = best_h.copy()
            for k3 in top3:
                s3 = int(k3) - (best_n - 1)
                for j in range(best_n):
                    j2 = j + s3
                    if 0 <= j2 < best_n:
                        if hp[j] > 0.6 and hp[j2] < 0.4:
                            hp[j] -= 0.08
                            hp[j2] += 0.08
                        elif hp[j] < 0.4 and hp[j2] > 0.6:
                            hp[j] += 0.08
                            hp[j2] -= 0.08
            hp = project_feasible(hp, best_n)
        h_r, c5_r = nesterov_descent(hp, 150, 250, 0.006, 0.88, 0.9997)
        h_r = slsqp_polish(h_r, 300, 300)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r

    # Phase 5: Multi-resolution
    for nr in [150, 180, 220, 250]:
        if time.time() - t0 > budget_s * 0.85:
            break
        if nr == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu = project_feasible(hu + np.random.randn(nr) * 0.008, nr)
        h_r, c5_r = nesterov_descent(hu, 150, 200, 0.006, 0.88, 0.9997)
        h_r = coord_descent(h_r, n_iters=20, topK=10)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, h_r, nr

    # Phase 6: Final refinement
    if time.time() - t0 < budget_s * 0.93:
        h_cd = coord_descent(best_h, n_iters=60, topK=20)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd
        h_s = slsqp_polish(best_h, 5000, 800)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s

    # Phase 7: Ultra-fine Nesterov
    if time.time() - t0 < budget_s * 0.97:
        h_f, c5_f = nesterov_descent(best_h, 3000, 150, 0.001, 0.95, 0.9999)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        h_cd2 = coord_descent(best_h, n_iters=30, topK=20)
        c5_cd2 = compute_c5(h_cd2)
        if c5_cd2 < best_c5:
            best_c5, best_h = c5_cd2, h_cd2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
