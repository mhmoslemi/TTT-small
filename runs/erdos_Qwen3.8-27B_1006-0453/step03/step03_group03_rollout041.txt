```python
def compute_delta_c(a, b, h, n):
    """Compute Δc for swap a(1→0) and b(0→1)"""
    k = np.arange(-(n-1), n)  # all possible shifts
    cc = np.zeros(2*n-1)
    
    # Term 1: -(1-h[a-k]) for a-k in [0,n)
    ak = a - k
    mask1 = (ak >= 0) & (ak < n)
    cc[mask1] -= (1 - h[ak[mask1]])
    
    # Term 2: +h[a+k] for a+k in [0,n)
    ap = a + k
    mask2 = (ap >= 0) & (ap < n)
    cc[mask2] += h[ap[mask2]]
    
    # Term 3: +(1-h[b-k]) for b-k in [0,n)
    bk = b - k
    mask3 = (bk >= 0) & (bk < n)
    cc[mask3] += (1 - h[bk[mask3]])
    
    # Term 4: -h[b+k] for b+k in [0,n)
    bp = b + k
    mask4 = (bp >= 0) & (bp < n)
    cc[mask4] -= h[bp[mask4]]
    
    return cc
```
