```python
def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()

    s = h.sum()
    diff = target - s

        if total > 1e-15:

    else:

        if total > 1e-15:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-15:
            h += room * (diff / total)
    else:
        total = h.sum()
        if total > 1e-15:
            h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def get_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj(h, T):
    C = get_corr_dx(h)
    m = C.max()
    spread = T * (C[-1] - m) if C.size > 1 else 0
    if T * (C - m).max() < -40:
        return m
    return m + np.log(np.sum(np.exp(T * (C - m)))) / T

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    C = get_corr_dx(h)
    m = C.max()
    exps = np.clip(T * (C - m), -500, 500)
    w = np.exp(exps)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[C.argmax()] = 1.0
    else:
        w /= w_sum
    
    h_rev = h[::-1]
    term1 = np.array([np.dot(w[j:j+n], h_rev) for j in range(n)])
    term2 = np.array([np.dot(w[n-1-j:n+j], h) for j in range(n)])
    return dx * (1.0 - term1 - term2)

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, T, n, maxiter=1000, ftol=1e-14):
    def obj(h):
        return smooth_obj(h, T)
    def obj_grad(h):
        return smooth_grad(h, T)
    def constr(h):
        return h.sum() - n / 2.0
    def constr_grad(h):
        return np.ones(n) / n  # normalized? no, just ones
    
    cons = [{'type': 'eq', 'fun': constr, 'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

    t0 = time.time()

    if ihv is not None:

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP with increasing T

        else:

        h_pert = make_feasible(h_pert, n)

        c5v = compute_c5(h_new)
        
        temp = 0.003 * (1.0 - it / max(1, max_iters)) + 0.0001
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-10)):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
    
    # Phase 3: Try different resolutions

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [500, 2000, 5000]:
            if time.time() - t0 > budget_s * 0.92:
                break
            h_new = slsqp_opt(h_interp, T, n_try, 1000, 1e-15)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
```

```python
def fista_opt(h0, T, n, maxiter=2000, lr_decay=0.999):
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    lr = 0.1 / T  # initial learning rate
    
    for k in range(maxiter):
        g = smooth_grad(y, T)
        # Update
        y_new = y - lr * g
        y_new = make_feasible(y_new, n)
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = y_new + (t_k - 1.0) / t_new * (y_new - h)
        y = make_feasible(y, n)
        h = y_new
        t_k = t_new
        lr *= lr_decay
    
    return make_feasible(h, n)
```
