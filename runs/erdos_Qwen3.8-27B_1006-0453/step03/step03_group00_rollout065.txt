```python
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgradient(h, n, over, top_k=5):
    """Compute weighted subgradient of max overlap."""
    m = np.max(over)
    k = min(top_k, len(over))
    idx = np.argsort(over)[-k:]
    vals = over[idx]
    weights = np.exp(300.0 * (vals - m))
    ws = weights.sum()
    if ws < 1e-30:
        return np.zeros(n)
    weights /= ws
    
    grad = np.zeros(n)
    for w, m_idx in zip(weights, idx):
        lag = m_idx - (n - 1)
        if lag == 0:
            g = 1.0 - 2.0 * h
        elif lag > 0:
            g = np.zeros(n)
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] += -h[:n - lag]
        else:
            mm = -lag
            g = np.zeros(n)
            g[mm:] += (1.0 - h[:n - mm])
            g[:n - mm] += -h[mm:]
        grad += w * g
    return grad

def sg_descent(h, n, t_end, top_k=5, iters=20000):
    """Subgradient descent with momentum and adaptive step."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    over = all_overlaps(best_h)
    v = np.zeros(n)
    beta = 0.85
    
    for it in range(iters):
        if time.time() > t_end:
            break
        if it % 200 == 0:
            over = all_overlaps(best_h)
        
        g = subgradient(best_h, n, over, top_k)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Polyak-like step
        current = np.max(over)
        lb = 0.3  # rough lower bound
        step = min(0.5, max(0.01, (current - lb) / (gnorm * gnorm + 1e-10)))
        step = step * 0.1  # scale down
        
        v = beta * v - step * g
        h_new = project(best_h + v, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            # Reset momentum on improvement
            v *= 0.5
    
    return best_h, best_c5
```

```python
    m = np.max(over)
    k = min(top_k, len(over))
    idx = np.argsort(over)[-k:]
    vals = over[idx]

        return np.zeros(n)
    w /= sw
    grad = np.zeros(n)
    for wi, mi in zip(w, idx):
        lag = mi - (n - 1)
        if lag == 0:

        elif lag > 0:
            g = np.zeros(n)
```

```python
def coord_one(h, n, i, dx):
    """Optimize h[i] given all other coordinates fixed. Returns new value."""
    h_save = h[i]
    # Compute correlate with h[i]=0
    h0 = h.copy()
    h0[i] = 0.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    # Compute correlate with h[i]=1
    h1 = h.copy()
    h1[i] = 1.0
    c1 = np.correlate(h1, 1.0 - h1, mode='full')
    
    # For h[i]=v: corr = c0 + v*(c1-c0) except at lag 0 where we subtract v^2
    diff = c1 - c0
    lag0_idx = n - 1  # index of lag 0 in full correlate
    
    def f(v):
        corr = c0 + v * diff
        corr[lag0_idx] -= v * v
        return np.max(corr) * dx
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    h[i] = res.x
    return res.x
```

```python
        corr[lag0_idx] -= v * v
        return np.max(corr) * dx
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    h[i] = res.x
    return res.x
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgradient(h, n, over, top_k=5):
    m = np.max(over)
    k = min(top_k, len(over))
    idx = np.argsort(over)[-k:]
    vals = over[idx]
    w = np.exp(300.0 * (vals - m))
    sw = w.sum()
    if sw < 1e-30:
        return np.zeros(n)
    w /= sw
    grad = np.zeros(n)
    for wi, mi in zip(w, idx):
        lag = mi - (n - 1)
        if lag == 0:
            g = 1.0 - 2.0 * h
        elif lag > 0:
            g = np.zeros(n)
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] += -h[lag:]
        else:
            mm = -lag
            g = np.zeros(n)
            g[:n - mm] += (1.0 - h[mm:])
            g[mm:] += -h[:n - mm]
        grad += wi * g
    return grad

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(res.x, n)

            break

            over = all_overlaps(best_h)
        g = subgradient(best_h, n, over, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        h_new = project(best_h + v, n)

def coord_descent(h, n, t_end, max_sweeps=40):

    dx = 2.0 / n
