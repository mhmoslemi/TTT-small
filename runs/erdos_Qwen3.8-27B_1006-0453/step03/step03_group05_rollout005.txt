    if remaining() > budget_s * 0.12:
        hd, cd = coord_descent(best_h, n_rounds=2, n_vars=best_n)
        if cd < best_c5:
            best_c5, best_h = cd, hd
        hd = slsqp_opt(best_h, topk_obj, (1, best_n), 600, 1e-15)
        cd = compute_c5(hd)
        if cd < best_c5:
            best_c5, best_h = cd, hd

    # Phase 5: Try n=90 (simpler, might find better structure)
    if remaining() > budget_s * 0.25:
        n90 = 90
        x_old = np.linspace(0, 2, best_n)
        x90 = np.linspace(0, 2, n90)
        h90 = np.interp(x90, x_old, best_h)
        h90 = project_feasible(h90, n90)
        h90 = slsqp_opt(h90, topk_obj, (2, n90), 500, 1e-14)
        c90 = compute_c5(h90)
        if c90 < best_c5:
            best_c5, best_h, best_n = c90, h90, n90
            h90b, _, _ = binary_local_search(h90, n90, max_rounds=5, time_limit=20)
            h90r = project_feasible(h90b + np.random.randn(n90) * 0.01, n90)
            h90r = slsqp_opt(h90r, topk_obj, (1, n90), 400, 1e-14)
            c90r = compute_c5(h90r)
            if c90r < best_c5:
                best_c5, best_h, best_n = c90r, h90r, n90

    # Phase 6: Multi-resolution up
    if remaining() > budget_s * 0.20 and best_n < 200:
        nr = best_n + 100
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu = project_feasible(hu + np.random.randn(nr) * 0.01, nr)
        hh = slsqp_opt(hu, topk_obj, (3, nr), 300, 1e-13)
        c5_hi = compute_c5(hh)
        if c5_hi < best_c5:
            # Downsample back
            x_dn = np.linspace(0, 2, best_n)
            hd = np.interp(x_dn, x_new, hh)
            hd = project_feasible(hd, best_n)
            c5_dn = compute_c5(hd)
            if c5_dn < best_c5:
                best_c5, best_h = c5_dn, hd
            else:
                best_c5, best_h, best_n = c5_hi, hh, nr

    # Phase 7: Final intensive polish
    if remaining() > 20:
        for k in [1, 2, 3]:
            if remaining() < 15:
                break
            hf = slsqp_opt(best_h, topk_obj, (k, best_n), 800, 1e-16)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        hf = trust_opt(best_h, topk_obj, (1, best_n), 300)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hf2 = slsqp_opt(best_h, softmax_obj, (5000, best_n), 500, 1e-15)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
