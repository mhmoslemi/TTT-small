def weighted_grad(h, N, corr, center, n_top, tau):
    """Compute weighted gradient using softmax or top-k weighting."""
    nc = len(corr)
    if n_top <= 0:
        sc = tau * (corr - corr.max())
        sc = np.clip(sc, -500, 0)
        w = np.exp(sc)
        w /= w.sum()
    else:
        idx = np.argsort(corr)[-n_top:]
        w = np.zeros(nc)
        w[idx] = 1.0 / n_top
    grad = np.zeros(N)

            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:

            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]

def weighted_grad(h, N, corr, n_top, tau):
    nc = len(corr)
    center = N - 1
    if 0 < n_top < nc:
        idx = np.argsort(corr)[-n_top:]
        w = np.zeros(nc)
        w[idx] = 1.0 / n_top
