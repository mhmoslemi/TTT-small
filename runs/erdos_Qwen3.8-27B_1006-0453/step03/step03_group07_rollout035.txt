    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_c = np.max(corr)
    p = np.exp(T * (corr - max_c))
    p /= p.sum()
    
    # Gradient: for each j, sum over k of p[k] * dcorr[k]/dh[j]
    # dcorr[k]/dh[j] = (1-h[j-k]) if 0<=j-k<n else 0, MINUS h[j+k] if 0<=j+k<n else 0
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        # First sum: sum over k where 0 <= j-k < n, i.e., j-n < k <= j
        # k ranges from max(0, j-n+1) to min(j, 2n-2)
        k_lo = max(0, j - (n-1))  # j-k >= 0 means k <= j

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:

        if total > 1e-14:

def smooth_obj_and_grad(h, T):
    """Compute smoothed max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_vals = np.exp(T * (corr - max_c))
    p = exp_vals / exp_vals.sum()
    obj = (max_c + np.log(exp_vals.sum()) / T) * dx
    
    # Compute gradient
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_k p[k] * (1-h[j-k]) for 0<=j-k<n
        k_lo = max(0, j - (n-1))
        k_hi = min(j, 2*n-2)
        if k_lo <= k_hi:
            idx = np.arange(k_lo, k_hi+1)
            grad[j] += np.dot(p[idx], one_minus_h[j-idx])
        # Second term: sum_k p[k] * h[j+k] for 0<=j+k<n
        k_lo2 = max(0, -j)
        k_hi2 = min(n-1-j, 2*n-2)
        if k_lo2 <= k_hi2:
            idx2 = np.arange(k_lo2, k_hi2+1)
            grad[j] -= np.dot(p[idx2], h[j+idx2])
    grad *= dx
    return obj, grad

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        total = h.sum()
        if total > 1e-14:
            h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    return h.sum() - len(h) / 2.0

    dx = 2.0 / n

def pgd_optimize(h, n, time_limit, T_list=None):
    if T_list is None:
        T_list = [40, 150, 600]
    best_h = h.copy()
    best_obj = compute_c5(h)
    t_start = time.time()
    for T in T_list:
        lr = 0.8 / (T + 5)
        stuck = 0
        while time.time() - t_start < time_limit:
            grad = compute_grad(h, T)
            gc = grad - np.mean(grad)
            h_new = h - lr * gc
            h_new = make_feasible(h_new, n)
            obj = compute_c5(h_new)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_h = h_new.copy()
                h = h_new
                lr *= 0.999
                stuck = 0
            else:
                stuck += 1
                if stuck > 300:
                    lr *= 0.5
                    stuck = 0
                    if lr < 1e-7:
                        break
    return best_h, best_obj

def slsqp_refine(h, n, T, maxiter):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_feasible(np.clip(res.x, 0, 1), n)
        return h_new
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
        best_c5 = compute_c5(best_h)
    else:
        n0 = 120
        best_h = np.zeros(n0)
        best_h[0::2] = 1.0
        best_h = make_feasible(best_h, n0)
        best_c5 = compute_c5(best_h)

    # Phase 1: Block structure search (low-dim optimization)
    n_cur = len(best_h)
    if time.time() - t0 < budget_s * 0.12:
        for B in [4, 6, 8, 10, 12, 16, 20, 28, 36]:
            if time.time() - t0 > budget_s * 0.12:
                break
            if best_h is not None and n_cur % B == 0:
                bv = np.array([best_h[i*B:(i+1)*B].mean() for i in range(B)])
            else:
                bv = np.full(B, 0.5)
            bv = np.clip(bv, 0.05, 0.95)
            bv = bv / bv.sum() * (B / 2.0)
            bv = np.clip(bv, 0, 1)

            def bo(x, B=B, nn=n_cur):
                x = np.clip(x, 0, 1)
                hh = np.repeat(x, nn // B)
                if len(hh) < nn:
                    hh = np.concatenate([hh, np.full(nn - len(hh), 0.5)])
                hh = make_feasible(hh, nn)
                return compute_c5(hh)

            for trial in range(3):
                if time.time() - t0 > budget_s * 0.12:
                    break
                x0 = bv.copy() if trial == 0 else np.random.uniform(0.05, 0.95, B)
                x0 = np.clip(x0, 0, 1)
                try:
                    res = minimize(bo, x0, method='Nelder-Mead',
                                   options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11})
                    x_opt = np.clip(res.x, 0, 1)
                    h_try = np.repeat(x_opt, n_cur // B)
                    if len(h_try) < n_cur:
                        h_try = np.concatenate([h_try, np.full(n_cur - len(h_try), 0.5)])
                    h_try = make_feasible(h_try, n_cur)
                    c5v = compute_c5(h_try)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = h_try.copy()
                except Exception:
                    pass

    # Phase 2: Multi-resolution PGD
    n_targets = sorted(set([40, 60, 80, 100, 120, 160, n_cur]))
    n_targets = [x for x in n_targets if x >= 30 and x <= 240]
    t_phase2_end = t0 + budget_s * 0.55
    time_per = (t_phase2_end - time.time()) / (len(n_targets) * 2 + 1)
    time_per = max(time_per, 8.0)

    for n in n_targets:
        if time.time() > t_phase2_end:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_base = np.interp(x_new, x_old, best_h)
        h_base = make_feasible(h_base, n)

        for trial in range(2):
            if time.time() > t_phase2_end:
                break
            if trial == 0:
                h_try = h_base.copy()
            else:
                h_try = h_base + 0.08 * np.random.randn(n)
                h_try = make_feasible(h_try, n)
            h_opt, c5_opt = pgd_optimize(h_try, n, time_limit=time_per,
                                          T_list=[40, 150, 600, 2000])
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: SLSQP refinement
    t_phase3_end = t0 + budget_s * 0.75
    n_cur = len(best_h)
    for T in [100, 500, 2000, 8000]:
        if time.time() > t_phase3_end:
            break
        h_new = slsqp_refine(best_h.copy(), n_cur, T, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Basin hopping with PGD inner loop
    t_phase4_end = t0 + budget_s * 0.93
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    it = 0
    while time.time() < t_phase4_end:
        it += 1
        frac = min(it / 200.0, 1.0)
        step = 0.12 * (1.0 - frac) + 0.02
        h_pert = bh_h + step * np.random.randn(n_bh)
        if it % 4 == 0:
            blen = max(2, n_bh // 10)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            h_pert = bh_h.copy()
            tmp = h_pert[s1:s1+blen].copy()
            h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
            h_pert[s2:s2+blen] = tmp
        elif it % 4 == 2:
            h_pert = bh_h.copy()
            nidx = max(1, n_bh // 6)
            idx = np.random.choice(n_bh, nidx, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        h_pert = make_feasible(h_pert, n_bh)

        tl = min(3.0, (t_phase4_end - time.time()) * 0.3)
        h_new, c5_new = pgd_optimize(h_pert, n_bh, time_limit=max(tl, 1.0),
                                      T_list=[200, 800])
        temp = 0.004 * (1.0 - frac) + 0.0003
        if c5_new < bh_c5 or np.random.rand() < np.exp(-(c5_new - bh_c5) / temp):
            bh_c5 = c5_new
            bh_h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Final polish
    if time.time() < t0 + budget_s * 0.985:
        n_cur = len(best_h)
        for T in [10000, 50000]:
            if time.time() > t0 + budget_s * 0.985:
                break
            h_new = slsqp_refine(best_h.copy(), n_cur, T, maxiter=3000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
