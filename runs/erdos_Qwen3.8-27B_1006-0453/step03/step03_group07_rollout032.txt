```python
import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def circular_acf(v):
    """Compute circular autocorrelation of binary pattern v."""
    p = len(v)
    ac = np.zeros(p)
    for r in range(p):
        ac[r] = np.dot(v, np.roll(v, -r))
    return ac

def gen_patterns(p, k):
    """Generate all binary patterns of length p with exactly k ones."""
    patterns = []
    positions = list(range(p))
    for combo in combinations(positions, k):
        v = np.zeros(p)
        v[list(combo)] = 1.0
        patterns.append(v)
    return patterns
```

Let me also think about what makes C5 small. We want, for EVERY lag k, the overlap ∫h(x)(1-h(x+k))dx to be small. This means:
- Where h is large, h shifted by k should also be large (so 1-h(x+k) is small)
- Or: h is small where h shifted by k is large

This is a "spread out" condition: the set where h≈1 should be spread out so that no shift aligns it with {h≈0}.

This is related to the concept of a "maximum density" or "minimum discrepancy" set.

For a binary set A of measure 1 in [0,2]:
C5 = max_k |(A \ (A+k))| = max_k |A| - |A ∩ (A+k)| = 1 - min_k |A ∩ (A+k)|

So we want to maximize min_k |A ∩ (A+k)|. This is the minimum autocorrelation of the indicator function.

This is related to the "boomerang" or "autocorrelation" problem in additive combinatorics.

        else:

    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

    w = exps / np.sum(exps)

    grad = np.correlate(w, f, mode='valid')

- We want min_h max_k f_k(h) where f_k(h) = corr[k] * dx
- The Lagrangian dual: max_λ min_h sum_k λ_k * f_k(h) + b (constraint)

The inner problem is still non-convex (products of h), but it's a single quadratic function, which might be easier to optimize.

Actually, sum_k λ_k * corr[k] = sum_k λ_k * sum_j h[j] * (1-h[j+offset_k])
= sum_j h[j] - sum_k λ_k sum_j h[j] * h[j+offset_k]
= n/2 - sum_j sum_k λ_k h[j] h[j+offset_k]

The last term can be written as h^T A h where A is a symmetric matrix:
A[j, j'] = (1/2) sum_k λ_k * I[j' = j + offset_k] + I[j' = j - offset_k]

So the inner problem is: min_h n/2 - h^T A h subject to 0≤h≤1, sum h = n/2
= min_h -h^T A h + constant
= max_h h^T A h - constant

This is maximizing a quadratic form over the hypercube with a sum constraint. If A is positive semidefinite, this is convex (maximizing a convex function is non-convex, but we can use SDP relaxation). If A is not PSD, it's non-convex.

import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def proj_sum(h, n):
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / np.sum(exps)
    f = 1.0 - 2.0 * h
    grad = np.correlate(w, f, mode='valid')
    return grad * dx

def pgd_step(h, T, lr):
    n = len(h)
    g = smooth_grad(h, T)
    return proj_sum(h - lr * g, n)

def slsqp_refine(h0, T, n, maxiter=500, ftol=1e-15):
    def obj_f(h):
        return smooth_obj(h, T)
    def con_f(h):
        return h.sum() - n / 2.0
    def con_j(h):
        return np.ones(n) / n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_f, h0, jac=smooth_grad, args=(T,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < smooth_obj(h0, T):
            return proj_sum(res.x, n)
    except Exception:
        pass
    return h0
