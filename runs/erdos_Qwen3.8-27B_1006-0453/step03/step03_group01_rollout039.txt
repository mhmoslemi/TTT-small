    if abs(diff) < 1e-10:

def grad_compute(h, N, weights):
    """Compute weighted gradient of C5 w.r.t. h."""
    center = N - 1
    grad = np.zeros(N)
    active = np.where(weights > 0.001)[0]
    for ai in active:
        L = ai - center
        wi = weights[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def grad_descent(h, N, target, deadline, soft_tau_range=None, topk_range=None, lr=0.03, decay=0.97, period=60, mom=0.87):
    """General gradient descent. Use soft_tau_range for softmax, topk_range for top-K."""
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 2000.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        nc = len(corr)
        if soft_tau_range is not None:
            tau = soft_tau_range[0] + (soft_tau_range[1] - soft_tau_range[0]) * frac
            sc = tau * corr - tau * corr.max()
            w = np.exp(sc)
            w /= w.sum()
        else:
            K = max(1, int(topk_range[0] + (topk_range[1] - topk_range[0]) * frac))
            idx = np.argsort(corr)[-K:]
            w = np.zeros(nc)
            w[idx] = 1.0 / K
        grad = grad_compute(h, N, w)
        v = mom * v + grad
        h = project_sum(h - lr * v, target)
        if it % 5 == 0:
            c5 = compute_c5(h, dx)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
        if it % period == 0:
            lr *= decay
            if lr < 1e-10:
                break
    return best_h, best_c5

def mass_transfer(h, N, target, deadline, max_ops=50000):

        if c5 < best_c5 - 1e-12:

            best_c5 = c5
    return best_h, best_c5

Actually, let me try a completely different angle: **iteratively reweighted least squares (IRLS) style approach**.

The idea: 
1. Compute all correlations corr(0), corr(1), ..., corr(N-1)
2. Assign weights w_k ∝ exp(β * corr(k)) to each lag (higher weight to worse lags)
3. Minimize the weighted sum: sum_k w_k * corr(k)
4. This is a quadratic in h: h^T (sum w_k A_k) h + (sum w_k b_k)^T h
5. Solve the quadratic subproblem (or take a few gradient steps)
6. Update weights and repeat

The advantage is that in step 3, we're minimizing a smooth function (weighted sum) rather than the non-smooth max. As β increases, the weighted sum approaches the max.

For the quadratic subproblem:
corr(k) for lag L: sum_i h[i](1-h[i+L]) = sum_i h[i] - sum_i h[i]h[i+L]

Since sum_i h[i] = N/2 (fixed), minimizing sum_i h[i](1-h[i+L]) is equivalent to maximizing sum_i h[i]h[i+L].
