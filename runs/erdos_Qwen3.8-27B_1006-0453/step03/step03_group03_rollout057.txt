import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc(h, n_sweeps=200, top_m=12):
    n = len(h)
    h = h.copy()
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    for sweep in range(n_sweeps):
        improved = 0
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            a = c1 - c0
            b = c0.copy()
            candidates = [0.0, 1.0, old]
            order = np.argsort(-b)[:top_m]
            for ii in range(len(order)):
                for jj in range(ii + 1, len(order)):
                    si, sj = order[ii], order[jj]
                    da = a[si] - a[sj]
                    if abs(da) > 1e-12:
                        t_int = (b[sj] - b[si]) / da
                        if 0.0 <= t_int <= 1.0:
                            candidates.append(t_int)
            best_t = old
            best_val = float(np.max(a * best_t + b))
            for tc in candidates:
                tc = float(np.clip(tc, 0.0, 1.0))
                val = float(np.max(a * tc + b))
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = tc
            h[j] = best_t
            h = project_feasible(h, n)
            c5_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved += 1
        if improved == 0:
            break
    return h, best_c5

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    val = m + float(np.log(np.sum(np.exp(T * (corr - m)))) / T)
    return val * dx

def lse_sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lse_sum_eq}]
    try:
        res = minimize(lse_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hx = np.clip(res.x, 0.0, 1.0)
        return project_feasible(hx, n)
    except Exception:
        return h

def topm_obj(h, m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argpartition(corr, -m)[-m:]
    return float(np.sum(corr[idx]))

def topm_slsqp(h, m=5, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lse_sum_eq}]
    try:
        res = minimize(topm_obj, h, args=(m,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h

                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd
                    best_n = n_try

    # Phase 6: Fine CD + SLSQP polish
    if time.time() - t0 < budget_s * 0.92:
        h_cd, c5_cd = coord_desc(best_h, n_sweeps=400, top_m=16)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        h_s = slsqp_smooth(best_h, 50000, maxiter=600)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
        h_t = topm_slsqp(best_h, m=8, maxiter=500)
        c5_t, _ = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 7: Final ultra-fine CD
    if time.time() - t0 < budget_s * 0.98:
        h_cd, c5_cd = coord_desc(best_h, n_sweeps=500, top_m=20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        h_s = slsqp_smooth(best_h, 100000, maxiter=400)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
