    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / topK
    alpha = min(1.0, max(0.0, (tau - 15.0) / 60.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def grad_optimize(h, N, target, deadline, tau_lo, tau_hi, tk_lo, tk_hi, lr0, decay, period, mom):
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 2000.0, 1.0)
        tau = tau_lo + (tau_hi - tau_lo) * frac
        topK = int(max(2, tk_lo + (tk_hi - tk_lo) * frac))
        grad, c5_val = compute_grad(h, N, tau, topK)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        v = mom * v + grad
        h = project_sum(h - lr * v, target)
        if it % period == 0:
            lr *= decay
            if lr < 1e-11:
                break
    return best_h, best_c5

def targeted_swap(h, N, target, deadline, n_trials):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    h_cur = h.copy()
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        center = N - 1
        top_idx = np.argsort(corr)[-8:]
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        # Targeted: find pairs affected by top correlations
        candidates = []
        for ai in top_idx:
            L = ai - center
            if L > 0 and L < N:
                for ii in range(0, N - L, max(1, (N - L) // 15)):
                    if h_cur[ii] > 0.5 and h_cur[ii + L] < 0.5:
                        candidates.append((ii, ii + L))
                    elif h_cur[ii] < 0.5 and h_cur[ii + L] > 0.5:
                        candidates.append((ii + L, ii))
            elif L < 0 and abs(L) < N:
                M = -L
                for ii in range(0, N - M, max(1, (N - M) // 15)):
                    if h_cur[ii + M] > 0.5 and h_cur[ii] < 0.5:
                        candidates.append((ii + M, ii))
                    elif h_cur[ii + M] < 0.5 and h_cur[ii] > 0.5:
                        candidates.append((ii, ii + M))
        # Mix with random swaps
        if candidates and np.random.random() < 0.6:
            i, j = candidates[np.random.randint(len(candidates))]
        else:
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5 = compute_c5(h_cur, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
    return best_h, best_c5

def coord_refine(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for _pass in range(80):
        if time.time() > deadline:
            break
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = best_h[i]
            for trial in (0.0, 1.0):
                if abs(trial - orig) < 1e-6:
                    continue
                h_tmp = best_h.copy()
                h_tmp[i] = trial
                h_tmp = project_sum(h_tmp, target)
                c5 = compute_c5(h_tmp, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h_tmp.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def round_to_binary(h, N, target):
    dx = 2.0 / N
    sorted_idx = np.argsort(h)
    binary = np.zeros(N)
    binary[sorted_idx[N // 2:]] = 1.0
    binary = project_sum(binary, target)
    return binary

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    N = len(h_par)
    target = N / 2.0
    dx = 2.0 / N

    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Gradient optimization with multiple restarts (40% budget)
    p1_dead = t0 + budget_s * 0.38
    n_restarts = 4
    per_r = (p1_dead - t0 - 10) / n_restarts
    for r in range(n_restarts):
        if time.time() > p1_dead - 10:
            break
        r_dead = min(time.time() + per_r, p1_dead - 5)
        if r == 0:
            h_cur = best_h.copy()
        elif r == 1:
            h_cur = best_h + rng.randn(N) * 0.02
            h_cur = project_sum(h_cur, target)
        elif r == 2:
            h_cur = best_h + rng.randn(N) * 0.05
            h_cur = project_sum(h_cur, target)
        else:
            h_cur = rng.rand(N) * 0.3
            h_cur = project_sum(h_cur, target)
        h_cur, c1 = grad_optimize(h_cur, N, target, r_dead,
                                  3.0, 400.0, 25, 3, 0.03, 0.96, 50, 0.84)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h_cur.copy()

    # Phase 2: Binary rounding + targeted swap search (25% budget)
    p2_dead = min(time.time() + budget_s * 0.25, deadline - 120)
    h_bin = round_to_binary(best_h, N, target)
    c_bin = compute_c5(h_bin, dx)
    h_work = best_h.copy()
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin.copy()
        h_work = h_bin.copy()
    h_work, c2 = targeted_swap(h_work, N, target, p2_dead, 50000)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h_work.copy()

    # Phase 3: Coordinate refinement (20% budget)
    p3_dead = min(time.time() + budget_s * 0.20, deadline - 40)
    h_cur, c3 = coord_refine(best_h.copy(), N, target, p3_dead)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h_cur.copy()

    # Phase 4: Another round of gradient + swap on the refined solution
    if time.time() < deadline - 30:
        p4_dead = min(time.time() + 80, deadline - 30)
        h_cur, c4 = grad_optimize(best_h.copy(), N, target, p4_dead,
                                  300.0, 2000.0, 4, 2, 0.003, 0.98, 80, 0.88)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h_cur.copy()

    # Phase 5: Final coordinate refinement
    if time.time() < deadline - 5:
        best_h, best_c5 = coord_refine(best_h.copy(), N, target, deadline - 3)

    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, N
