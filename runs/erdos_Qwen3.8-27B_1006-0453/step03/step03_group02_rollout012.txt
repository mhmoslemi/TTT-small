def grad_at_lag(h, N, L):
    """Gradient of f_L(h) = sum_i h[i](1-h[i+L]) with appropriate bounds."""
    N = len(h)
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:N-L] += (1.0 - h[L:])  # from h[i] in term i
        g[L:] -= h[:N-L]          # from h[i+L] in term i
    else:
        M = -L
        g[M:] += (1.0 - h[:N-M])  # from h[i] in term i+M
        g[:N-M] -= h[M:]          # from h[i+M] in term i
    return g

        y = np.clip(y, 0.0, 1.0)

        else:

        h = project_sum(h + v, target)
        it += 1
