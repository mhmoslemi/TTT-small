import time
import cvxpy as cp

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_all_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_for_lags(h, lags):
    """Compute gradient of c(L) w.r.t. h for given lags (L = k - (N-1))"""
    N = len(h)
    center = N - 1
    grads = []
    for k in lags:
        L = k - center
        g = np.zeros(N)
        if L == 0:
            g = 1.0 - 2.0 * h
        elif L > 0:

            g[L:] += (1.0 - h[:N-L])  # j >= L: contribution (1-h[j-L])
            if L < N:

            lo = mid
        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_lag(h, L):
    N = len(h)
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:N-L])
        g[:N-L] -= h[L:]
    else:
        M = -L
        g[:N-M] += (1.0 - h[M:])
        g[M:] -= h[:N-M]
    return g

def topk_grad(h, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0/len(h))
    N = len(h)
    center = N - 1
    if K >= 2*N-1:
        idx = np.arange(2*N-1)
        ct = corr
    else:
        idx = np.argsort(corr)[-K:]
        ct = corr[idx]
    sc = tau * ct - tau * ct.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    g = np.zeros(N)
    for ix in range(len(w)):
        if w[ix] < 1e-7:
            continue
        L = idx[ix] - center
        g += w[ix] * grad_for_lag(h, L)
    return g, c5

def lp_cutting(h, N, target, deadline, max_iters=50, threshold_frac=0.92):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/N)
    h = h.copy()
    all_grads = []
    
    for it in range(max_iters):

            break

        if c5 < best_c5:

            best_h = h.copy()

        thr = float(np.max(corrs)) * threshold_frac
        active = np.where(corrs >= thr)[0]
        # Build cuts for active lags
        center = N - 1
        new_grads = []
        for k in active:
            L = k - center
            g = grad_for_lag(h, L)
            new_grads.append(g)
        # Solve LP

        t_var = cp.Variable()
        cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        for g in new_grads:
            cons.append(g @ h_var <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:

        except:

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_lag(h, L):
    N = len(h)
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:N-L])
        g[:N-L] -= h[L:]
    else:
        M = -L
        g[:N-M] += (1.0 - h[M:])
        g[M:] -= h[:N-M]
    return g

def topk_grad(h, tau, K):

    c5 = float(np.max(corr)) * (2.0 / N)

        ct = corr
    else:
        idx = np.argsort(corr)[-K:]
        ct = corr[idx]

    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    g = np.zeros(N)
    for ix in range(len(w)):
        if w[ix] < 1e-7:
            continue
        L = idx[ix] - center
        g += w[ix] * grad_for_lag(h, L)
    return g, c5

        center = N - 1
        cmax = float(np.max(corrs))
        if cmax < 1e-15:
            break
        thr = cmax * 0.93
        active = np.where(corrs >= thr)[0]
        for k in active:
            L = k - center
            g = grad_for_lag(h, L)
            offset = float(cmax - g @ h)
            stored_grads.append(g)
            stored_offsets.append(offset)
        if len(stored_grads) > 200:
            keep = np.argsort(np.array(stored_offsets))[-120:]
            stored_grads = [stored_grads[i] for i in keep]
            stored_offsets = [stored_offsets[i] for i in keep]
        h_var = cp.Variable(N)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
