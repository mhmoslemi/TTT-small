```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14: break
            h -= h * (abs(diff) / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, T, n, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(T * (corr - m), -400, 400)
    s = np.exp(shifted).sum()
    val = dx * (m + np.log(s) / T)
    sum_err = h.sum() - n / 2.0
    val += pen * sum_err * sum_err
    return val

def lse_grad(h, T, n, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(T * (corr - m), -400, 400)
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total
    W_pos = weights[n-1:]
    first = np.zeros(n)
    for i in range(n):
        first[i] = np.dot(W_pos[:n-i], 1.0 - h[i:])
    second = np.convolve(W_pos, h, 'full')[:n]
    sum_err = h.sum() - n / 2.0
    return dx * (first - second) + 2.0 * pen * sum_err

def lbfgs_optimize(h, n, T_start=20, T_end=5000, pen_start=1.0, n_iters=2000):
    bnds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T = T_start
    pen = pen_start
    iters_per_phase = 200
    phase = 0
    total_iters = 0
    
    while total_iters < n_iters:
        this_iters = min(iters_per_phase, n_iters - total_iters)
        res = minimize(
            lse_obj, h, args=(T, n, pen), jac=lse_grad,
            method='L-BFGS-B', bounds=bnds,
            options={'maxiter': this_iters, 'ftol': 1e-16, 'gtol': 1e-10}
        )
        h = project_feasible(np.clip(res.x, 0, 1), n)
        total_iters += this_iters
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Increase temperature and penalty
        phase += 1
        T = min(T * 2.5, T_end)
        pen = min(pen * 3.0, 1000.0)
        iters_per_phase = min(iters_per_phase + 100, 800)
        
        if T >= T_end and pen >= 1000:
            # Final phase: just polish
            T = T_end
            pen = 1000
    
    return best_h, best_c5

def slsqp_polish(h, n, T, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_obj(x, T, n, 100.0)
    def grad(x):
        return lse_grad(x, T, n, 100.0)
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h2 = project_feasible(np.clip(res.x, 0, 1), n)
        return h2
    except:
        return h

def coordinate_refine(h, n, n_sweeps=3, n_candidates=7):
    best = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        for i in range(n):
            old = h[i]
            best_t = old
            best_c = compute_c5(h)
            # Try candidate values
            for frac in np.linspace(0, 1, n_candidates):
                t = frac
                h[i] = t
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_t = t
            h[i] = best_t
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

    if time.time() - t0 < budget_s * 0.95:

            h6 = slsqp_polish(best_h, n, T, maxiter=500)
```
