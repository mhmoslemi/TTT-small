import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_gradient(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr) * (2.0 / N))
    n_corr = len(corr)
    if K >= n_corr:
        top_idx = np.arange(n_corr)
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top - tau * c_top.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5_val
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-7:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, c5_val

            tau_amp = 5.0 * (ph + 1)

def pair_swap_search(h, N, target, dx, n_rounds, deadline):

        # Sort by value
        sidx = np.argsort(best_h)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_grad(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr) * (2.0 / N))
    nc = len(corr)
    if K >= nc:
        top_idx = np.arange(nc)
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5_val
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):

            grad += wi * (1.0 - 2.0 * h)

            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, c5_val
