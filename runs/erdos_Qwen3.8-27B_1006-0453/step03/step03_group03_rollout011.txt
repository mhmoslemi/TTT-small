import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def barrier_objective(x, mu, penalty):
    n = x.shape[0] - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, 'full')
    diff = corr - t
    # Smooth max approximation: logsumexp
    m = np.max(diff)
    if m > 0:
        barrier = (m + np.log(np.sum(np.exp(mu * (diff - m)))) / mu)
    else:
        barrier = 0.0 + np.log(sum(np.exp(-mu * abs(diff)))) / mu  # this is wrong
    # Actually, softplus approximation:
    barrier = np.sum(np.logaddexp(0.0, mu * diff)) / mu
    sum_pen = penalty * (h.sum() - n / 2.0) ** 2
    return t + barrier + sum_pen

def barrier_grad(x, mu, penalty):
    n = x.shape[0] - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, 'full')
    diff = corr - t
    # Gradient of sum softplus(mu * diff) / mu
    # d/d(h[j]) of softplus(mu * diff_k) / mu = sigmoid(mu * diff_k) * d(diff_k)/d(h[j])
    # d/d(h[j]) of corr_k = (1-h[j+k]) if j+k valid, or -h[j-k] if j-k valid, etc.
    sigmoid_vals = 1.0 / (1.0 + np.exp(-mu * diff))
    # Gradient w.r.t. h
    gh = np.zeros(n)
    for k in range(2 * n - 1):
        if sigmoid_vals[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # corr[k] = sum_i h[i] * (1 - h[i+lag]) for i=0..n-1-lag
            # d/dh[j] corr[k] = (1-h[j+lag]) for j < n-lag, or -h[j-lag] for j >= lag
            if lag == 0:
                gh[:n] += sigmoid_vals[k] * (-(2.0 * h) + 1.0)  # d/dh[j] of h[j](1-h[j]) = 1-2h[j]
            else:
                idx1 = np.arange(0, n - lag)
                gh[idx1] += sigmoid_vals[k] * (1.0 - h[idx1 + lag])
                idx2 = np.arange(lag, n)
                gh[idx2] += sigmoid_vals[k] * (-h[idx2 - lag])
        else:
            lp = -lag
            idx1 = np.arange(lp, n)
            gh[idx1] += sigmoid_vals[k] * (1.0 - h[idx1 - lp])
            idx2 = np.arange(0, n - lp)
            gh[idx2] += sigmoid_vals[k] * (-h[idx2 + lp])
    # Gradient w.r.t. t
    gt = 1.0 - np.sum(sigmoid_vals)
    # Penalty gradient
    gh += 2.0 * penalty * (h.sum() - n / 2.0)
    return np.append(gh, gt)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def barrier_obj(x, T, pen):
    n = x.shape[0] - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, 'full')
    diff = corr - t
    # Soft max: (1/T) * sum log(exp(T*diff) + 1)
    # Use logaddexp for numerical stability
    barrier = np.sum(np.logaddexp(0.0, T * diff)) / T
    sum_pen = pen * (h.sum() - n * 0.5) ** 2
    return t + barrier + sum_pen

def barrier_grad(x, T, pen):
    n = x.shape[0] - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, 'full')
    diff = corr - t
    # sigmoid(T * diff)
    sig = 1.0 / (1.0 + np.exp(-np.clip(T * diff, -500, 500)))
    gh = np.zeros(n)
    for k in range(2 * n - 1):
        s = sig[k]
        if s < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            gh += s * (1.0 - 2.0 * h)
        elif lag > 0:
            gh[:n-lag] += s * (1.0 - h[lag:])

            gh[:n-lp] += s * (-h[lp:])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, T, pen):
    n = x.shape[0]
    h = x
    corr = np.correlate(h, 1.0 - h, 'full')
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T)
    val *= 2.0 / n
    val += pen * (h.sum() - n * 0.5) ** 2
    return val

def smooth_max_grad(x, T, pen):
    n = x.shape[0]
    h = x
    corr = np.correlate(h, 1.0 - h, 'full')
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    w = exp_vals / exp_vals.sum()
    gh = np.zeros(n)
    for k_out in range(2 * n - 1):
        s = (2.0 / n) * w[k_out]
        if s < 1e-12:
            continue
        k = k_out - (n - 1)
        if k == 0:
            gh += s * (1.0 - 2.0 * h)
        elif k > 0:
            jh = np.arange(k, n)
            gh[jh] += s * (1.0 - h[jh - k])
            jl = np.arange(0, n - k)
            gh[jl] += s * (-h[jl + k])
        else:
            lp = -k
            jl = np.arange(0, n - lp)
            gh[jl] += s * (1.0 - h[jl + lp])
            jh = np.arange(lp, n)
            gh[jh] += s * (-h[jh - lp])
    gh += 2.0 * pen * (h.sum() - n * 0.5)
    return gh

def lbfgs_smooth(h, T, pen, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_max_obj, h, args=(T, pen), jac=smooth_max_grad,
                   method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
    return project_feasible(np.clip(res.x, 0, 1), n)

def barrier_obj(x, T, pen):
    n = x.shape[0] - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, 'full')
    diff = corr - t
    barrier = np.sum(np.logaddexp(0.0, T * diff)) / T
    val = (2.0 / n) * (t + barrier) + pen * (h.sum() - n * 0.5) ** 2
    return val

def barrier_grad(x, T, pen):
    n = x.shape[0] - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, 'full')
    diff = corr - t
    sig = 1.0 / (1.0 + np.exp(-np.clip(T * diff, -500, 500)))
    gh = np.zeros(n)
    for k_out in range(2 * n - 1):
        s = sig[k_out] * (2.0 / n)
        if s < 1e-12:
            continue
        k = k_out - (n - 1)
        if k == 0:
            gh += s * (1.0 - 2.0 * h)
        elif k > 0:
            jh = np.arange(k, n)
            gh[jh] += s * (1.0 - h[jh - k])
            jl = np.arange(0, n - k)
            gh[jl] += s * (-h[jl + k])
        else:
            lp = -k
            jl = np.arange(0, n - lp)
            gh[jl] += s * (1.0 - h[jl + lp])
            jh = np.arange(lp, n)
            gh[jh] += s * (-h[jh - lp])
    gt = (2.0 / n) * (1.0 - np.sum(sig))
    gh += 2.0 * pen * (h.sum() - n * 0.5)
    return np.concatenate([gh, [gt]])

def lbfgs_barrier(h, T, pen, maxiter=500):
    n = len(h)
    t0 = float(np.max(np.correlate(h, 1.0 - h, 'full'))) + 0.05
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    res = minimize(barrier_obj, x0, args=(T, pen), jac=barrier_grad,
                   method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
    return project_feasible(np.clip(res.x[:n], 0, 1), n)

    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)
