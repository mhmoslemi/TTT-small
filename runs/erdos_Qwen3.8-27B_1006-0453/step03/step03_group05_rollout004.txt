        else:
            h2 = _slsqp(best_h, _obj_topk, (3, best_n), 150, 1e-13)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 3: Coordinate descent (fast local search)
    for _ in range(6):
        if time.time() > dl - 120:
            break
        h3, c3 = _coord_desc(best_h, best_n, 800, rng)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
        h3 = _slsqp(best_h, _obj_topk, (3, best_n), 200, 1e-14)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Swap-based local search
    for _ in range(5):
        if time.time() > dl - 100:
            break
        h4, c4 = _swap_search(best_h, best_n, 2000, rng)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Basin hopping with SLSQP refinement
    n_trials = 25
    for tr in range(n_trials):
        if time.time() > dl - 60:
            break
        r = rng.random()
        if r < 0.35:
            sc = 0.02 + 0.01 * (1 - tr / n_trials)
            hp = project_feasible(best_h + rng.standard_normal(best_n) * sc, best_n)
        elif r < 0.6:
            hp = best_h.copy()
            s0 = rng.integers(0, best_n - 4)
            ln = int(rng.integers(3, max(4, best_n // 5)))
            hp[s0:s0+ln] = np.roll(hp[s0:s0+ln], int(rng.integers(1, ln)))
        else:
            hp = best_h.copy()
            for _ in range(int(rng.integers(1, 6))):
                i, j = rng.integers(0, best_n, 2)
                hp[i], hp[j] = hp[j], hp[i]
        hs = _slsqp(hp, _obj_soft, (300, best_n), 180, 1e-12)
        hs = _slsqp(hs, _obj_topk, (4, best_n), 180, 1e-13)
        hs = _attack_top(hs, best_n, 3, 120)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 6: Multi-resolution (try lower n for potentially better discrete C5)
    for nr in [80, 100, 120, 150]:
        if time.time() > dl - 40:
            break
        if nr == best_n:
            continue
        if nr < best_n:
            h_r = np.interp(np.linspace(0, 2, nr), np.linspace(0, 2, best_n), best_h)
        else:
            h_r = np.interp(np.linspace(0, 2, nr), np.linspace(0, 2, best_n), best_h)
        h_r = project_feasible(h_r, nr)
        h_r = project_feasible(h_r + rng.standard_normal(nr) * 0.01, nr)
        hr1 = _slsqp(h_r, _obj_topk, (3, nr), 250, 1e-12)
        hr1 = _slsqp(hr1, _obj_soft, (500, nr), 200, 1e-13)
        hr1 = _attack_top(hr1, nr, 3, 150)
        cr = compute_c5(hr1)
        if cr < best_c5:
            best_c5, best_h, best_n = cr, hr1, nr

    # Phase 7: Final intensive polish
    for fun, args, mi in [(_obj_topk, (2, best_n), 500),
                          (_obj_topk, (3, best_n), 500),
                          (_obj_soft, (3000, best_n), 400)]:
        if time.time() > dl - 10:
            break
        hf = _slsqp(best_h, fun, args, mi, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # Phase 8: Micro coordinate descent
    if time.time() < dl - 3:
        hm, cm = _coord_desc(best_h, best_n, 400, rng)
        if cm < best_c5:
            best_c5, best_h = cm, hm
        hf = _slsqp(best_h, _obj_topk, (2, best_n), 300, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
