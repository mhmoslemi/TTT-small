            new_hi = coord_opt_exact(h_current, i)
            h_try = h_current.copy()
            h_try[i] = new_hi
            if compute_c5(h_try) < compute_c5(h_current):
                h_current = h_try
        except:
            continue

    if T <= 0.1:  # Simulated annealing
        perturb_step = 0.02 * (1 - T/100)**(2/3)
        h_pert = h_current + perturb_step * (np.random.rand(n) - 0.5)
        h_pert = project(h_pert, 2.0/n, n/2)
        return h_pert
    else:  # Stronger exploration
        h_pert = h_current + fanout * (np.random.rand(n) - 0.5)
        h_pert = project(h_pert, 2.0/n, n/2)
        return h_pert
