from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Water-filling projection
    if diff > 0:
        # Need to add, prioritize positions closest to 1
        # Actually, just distribute proportionally to available room
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        # Need to subtract, prioritize positions closest to 0
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    B = 1.0 - h
    c = np.correlate(h, B, mode='full')
    cmax = c.max()
    exp_c = np.exp(T * (c - cmax))
    w = exp_c / exp_c.sum()
    obj = (cmax + np.log(exp_c.sum()) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        # First term: B[j-k] for valid j
        if k >= 0:
            # B valid: j-k in [0, n-1-k], so j in [k, n-1]
            j_lo, j_hi = k, n - 1
        else:
            # B valid: j-k in [-k, n-1], so j in [0, n-1+k]
            j_lo, j_hi = 0, n - 1 + k
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w[m] * B[j_lo-k:j_hi-k+1]
        
        # Second term: -h[j+k] for valid j
        if k >= 0:
            # h valid: j in [0, n-1-k]
            j2_lo, j2_hi = 0, n - 1 - k
        else:
            # h valid: j in [-k, n-1]
            j2_lo, j2_hi = -k, n - 1
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] -= w[m] * h[j2_lo+k:j2_hi+k+1]
    
    return obj, dx * grad

One more thought: the "power mean" approximation to max might be better than softmax for this problem:

∂f_p/∂h[j] = dx * (∑c_k^p)^(1/p-1) * ∑_k c_k^(p-1)*∂c_k/∂h[j]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    B = 1.0 - h
    c = np.correlate(h, B, mode='full')
    cmax = c.max()
    exp_c = np.exp(T * (c - cmax))
    sw = exp_c.sum()
    w = exp_c / sw
    obj = (cmax + np.log(sw) / T) * dx
    grad = np.zeros(n)
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-13:
            continue
        k = m - (n - 1)
        if k >= 0:
            jlo, jhi = k, n - 1
        else:
            jlo, jhi = 0, n - 1 + k
        if jlo <= jhi:
            grad[jlo:jhi+1] += wm * B[jlo-k:jhi-k+1]
        if k >= 0:
            j2lo, j2hi = 0, n - 1 - k
        else:
            j2lo, j2hi = -k, n - 1
        if j2lo <= j2hi:
            grad[j2lo:j2hi+1] -= wm * h[j2lo+k:j2hi+k+1]
    return obj, dx * grad

def power_obj_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    B = 1.0 - h
    c = np.correlate(h, B, mode='full')
    cp = c ** p
    S = cp.sum()
    obj = S ** (1.0 / p) * dx
    w = cp / S * p / S * S ** (1.0 / p)
    w = cp / S
    grad = np.zeros(n)
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-13:
            continue
        k = m - (n - 1)
        if k >= 0:
            jlo, jhi = k, n - 1
        else:
            jlo, jhi = 0, n - 1 + k
        if jlo <= jhi:
            grad[jlo:jhi+1] += wm * B[jlo-k:jhi-k+1]
        if k >= 0:
            j2lo, j2hi = 0, n - 1 - k
        else:
            j2lo, j2hi = -k, n - 1
        if j2lo <= j2hi:
            grad[j2lo:j2hi+1] -= wm * h[j2lo+k:j2hi+k+1]
    scale = S ** (1.0/p - 1.0)
    return obj, dx * scale * grad

def proj_grad_step(h, T, lr, n_iter=200):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    for it in range(n_iter):
        obj, grad = smooth_obj_grad(h, T)
        lr_eff = lr * (1.0 - 0.5 * it / n_iter)
        h_new = h - lr_eff * grad
        h_new = make_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < best:
            h = h_new
            best = c
    return h, best

def basin_hopping(h, n, t_remaining, seed):
    rng = np.random.RandomState(seed)
    bh = h.copy()
    bh_c5 = compute_c5(bh)
    it = 0
    while time.time() < t_remaining:
        it += 1
        ptype = it % 5
        step = 0.08 + 0.04 * rng.rand()
        if ptype == 0:
            h_pert = bh + step * rng.randn(n)
        elif ptype == 1:
            h_pert = bh.copy()
            blen = max(2, n // 10)
            s1 = rng.randint(0, n - blen)
            s2 = rng.randint(0, n - blen)
            if s1 != s2:
                tmp = h_pert[s1:s1+blen].copy()
                h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
                h_pert[s2:s2+blen] = tmp
        elif ptype == 2:
            h_pert = bh.copy()
            nflip = max(1, n // 6)
            idx = rng.choice(n, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif ptype == 3:
            h_pert = bh.copy()
            idx = rng.randint(n)
            delta = step * rng.choice([-1, 1])
            h_pert[idx] += delta
        else:
            freq = rng.randint(1, 4)
            amp = step * 0.5
            x = np.arange(n) * 2.0 / n
            h_pert = bh + amp * np.sin(2 * np.pi * freq * x + rng.rand() * 2 * np.pi)
        h_pert = make_feasible(h_pert, n)
        T_use = 200 + 100 * (it // 20)
        h_new, c_new = proj_grad_step(h_pert, T_use, 0.3, n_iter=80)
        temp = 0.003 * max(0, 1.0 - it / 200) + 0.0001
        if c_new < bh_c5 or rng.rand() < np.exp(-(c_new - bh_c5) / max(temp, 1e-10)):
            bh = h_new
            bh_c5 = c_new
    return bh, bh_c5

def multi_start_optimize(n, t_remaining, best_h_global, best_c5_global, seed):
    rng = np.random.RandomState(seed)
    best_h = best_h_global.copy()
    best_c5 = best_c5_global
    x_old = np.linspace(0, 2, len(best_h), endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h_init = np.interp(x_new, x_old, best_h)
    h_init = make_feasible(h_init, n)
    c_init = compute_c5(h_init)
    if c_init < best_c5:
        best_c5 = c_init
        best_h = h_init
    for trial in range(5):
        if time.time() > t_remaining:
            break
        if trial == 0:
            h0 = h_init.copy()
        else:
            h0 = h_init + 0.05 * rng.randn(n)
        for T in [100, 500, 2000]:
            if time.time() > t_remaining:
                break
            obj, grad = smooth_obj_grad(h0, T)
            h_new, c_new = proj_grad_step(h0, T, 0.2, n_iter=300)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
                h0 = h_new
            h0 = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        n0 = 200
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
        best_h = make_feasible(best_h, n0)
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    n_cur = len(best_h)
    
    # Phase 1: Projected gradient with increasing T
    t_phase1 = t0 + budget_s * 0.15
    for T in [50, 100, 300, 1000, 3000, 10000]:
        if time.time() > t_phase1:
            break
        h_new, c_new = proj_grad_step(best_h, T, 0.3, n_iter=400)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
            n_cur = len(best_h)
    
    # Phase 2: Basin hopping on current size
    t_phase2 = t0 + budget_s * 0.45
    best_h, best_c5 = basin_hopping(best_h, n_cur, t_phase2, seed + 1)
    
    # Phase 3: Multi-resolution exploration
    t_phase3 = t0 + budget_s * 0.75
    for n_try in [80, 120, 150, 160, 200, 240]:
        if time.time() > t_phase3:
            break
        if n_try == n_cur:
            continue
        t_sub = min(t0 + budget_s * 0.15, t_phase3)
        best_h, best_c5 = multi_start_optimize(n_try, t_sub, best_h, best_c5, seed + n_try)
        n_cur = len(best_h)
    
    # Phase 4: Fine basin hopping at best resolution
    t_phase4 = t0 + budget_s * 0.90
    best_h, best_c5 = basin_hopping(best_h, n_cur, t_phase4, seed + 100)
    
    # Phase 5: Final high-T polish
    for T in [5000, 20000, 100000]:
        if time.time() > t0 + budget_s * 0.98:
            break
        h_new, c_new = proj_grad_step(best_h, T, 0.1, n_iter=200)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
    
    # Phase 6: Greedy local search
    if time.time() < t0 + budget_s * 0.995:
        n_cur = len(best_h)
        for it in range(300):
            if time.time() > t0 + budget_s * 0.995:
                break
            improved = False
            for _ in range(40):
                j = np.random.randint(n_cur)
                delta = np.random.choice([-0.02, -0.05, 0.02, 0.05, -0.08, 0.08])
                h_try = best_h.copy()
                h_try[j] += delta
                h_try = make_feasible(h_try, n_cur)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-14:
                    best_c5 = c_try
                    best_h = h_try
                    improved = True
                    break
            if not improved and it > 200:
                break
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
