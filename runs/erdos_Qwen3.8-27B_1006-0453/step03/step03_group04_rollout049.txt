    fb = np.fft.rfft(b[::-1], size)

    return fft_correlate(h, 1.0 - h)

    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5

            lo = mid
        else:
            hi = mid

def grad_topk(h, N, k, temp):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top_idx = np.argpartition(corr, -k)[-k:]

    w = np.exp((top_vals - peak) / temp)
    w /= w.sum()

    center = N - 1

        wi = w[i]
        L = idx - center

            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])

            M = -L
