            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    while time.time() < deadline:
        frac = min(it / max(iters, 1), 1.0)
        beta = beta0 * (beta1 / beta0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5_val = lse_gradient(h, beta, dx)
        if c5_val < best_c5 - 1e-12:

            best_h = h.copy()

        it += 1
