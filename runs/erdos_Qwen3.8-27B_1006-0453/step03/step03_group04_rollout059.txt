def fast_corr(h):

def binary_anneal_fast(h, N, deadline, rng):
    """FFT-accelerated binary annealing."""
    dx = 2.0 / N
    h_b = (h > 0.5).astype(np.float64)
    n_ones = int(h_b.sum())
    half = N // 2
    if n_ones > half:
        idx = np.where(h_b == 1)[0]
        rng.shuffle(idx)
        h_b[idx[:n_ones - half]] = 0
    elif n_ones < half:
        idx = np.where(h_b == 0)[0]
        rng.shuffle(idx)
        h_b[idx[:half - n_ones]] = 1
    
    corr = fast_corr(h_b)
    best = h_b.copy()
    best_pk = float(np.max(corr))
    cur_pk = best_pk
    
    T0, T1 = 5.0, 0.002
    # Estimate iterations
    # Each iteration: ~2 FFTs of size 2N + overhead ≈ 100μs for N=380
    # Allocate ~80% of time to iterations
    avail = max(deadline - time.time(), 1)
    # Rough: 5000 iterations per second (conservative for Python)
    n_iter = int(avail * 3000)
    n_iter = max(n_iter, 1000)
    
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]

            break

                zeros[pos_j[0]] = i

        pj = rng.randint(len(zeros_list))
        i = ones_list[pi]
        j = zeros_list[pj]
        h_b[i] = 0.0
        h_b[j] = 1.0
        corr = fast_corr(h_b)
        new_pk = float(np.max(corr))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.002 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
                stall = 0
            else:
                stall += 1
            ones_list[pi] = j
            zeros_list[pj] = i
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
    return best, best_pk * dx

def targeted_refine(h_b, N, deadline, rng, n_peaks=4):
    dx = 2.0 / N
    best = h_b.copy()
    center = N - 1
    best_pk = float(np.max(fast_corr(h_b)))
    stall = 0
    while time.time() < deadline and stall < 300:
        corr = fast_corr(h_b)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best = h_b.copy()
            stall = 0
        else:
            stall += 1
        top_idx = np.argpartition(corr, -n_peaks)[-n_peaks:]
        improved_any = False
        for m in top_idx:
            lag_val = m - center
            if lag_val == 0 or abs(lag_val) >= N:
                continue
            if lag_val > 0:
                L = lag_val
                mask = (h_b[:N - L] == 1) & (h_b[L:] == 0)
            else:
                L = -lag_val
                mask = (h_b[L:] == 1) & (h_b[:N - L] == 0)
            cands = np.where(mask)[0]
            if len(cands) == 0:
                continue
            for _ in range(min(8, len(cands))):
                idx = cands[rng.randint(len(cands))]
                if lag_val > 0:
                    a, b = idx, idx + L
                else:
                    a, b = idx, idx + L
                h_b[a] = 0.0
                h_b[b] = 1.0
                new_pk = float(np.max(fast_corr(h_b)))
                if new_pk < pk:
                    improved_any = True
                    pk = new_pk
                    if new_pk < best_pk:
                        best_pk = new_pk
                        best = h_b.copy()
                        stall = 0
                    break
                else:
                    h_b[a] = 1.0
                    h_b[b] = 0.0
            if improved_any:
                break
        if not improved_any:
            stall += 20
    return best, best_pk * dx

def adam_continuous(h, N, target, deadline, k=30, lr0=0.05, decay=0.99):
    dx = 2.0 / N
    best = h.copy()
    best_pk = float(np.max(fast_corr(h)))
    b1, b2, eps = 0.85, 0.999, 1e-8
    m_arr = np.zeros(N)
    v_arr = np.zeros(N)
    it = 0
    center = N - 1
    while time.time() < deadline:
        corr = fast_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = int(ai) - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            elif L < 0:
                M = -L
                if M < N:
                    grad[:N - M] += wk * (1.0 - h[M:])
                    grad[M:] -= wk * h[:N - M]
        it += 1
        lr = lr0 * (decay ** (it // 200))
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * grad * grad
        mh = m_arr / (1.0 - b1 ** it)
        vh = v_arr / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best, best_pk * dx

def multi_kick_anneal(h, N, deadline, rng, n_kicks=3):
    dx = 2.0 / N
    best = to_binary(h, N).copy()
    best_pk = float(np.max(fast_corr(best)))
    time_per = max((deadline - time.time()) / n_kicks, 1.0)
    for kick in range(n_kicks):
        d = time.time() + time_per * (0.75 if kick > 0 else 1.0)
        if kick > 0:
            h_cur = best.copy()
            n_swap = 3 + kick * 2
            for _ in range(n_swap):
                ones = np.where(h_cur == 1)[0]
                zeros = np.where(h_cur == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h_cur[a], h_cur[b] = h_cur[b], h_cur[a]
        else:
            h_cur = to_binary(h, N).copy()
        h_new, pk = anneal_swap(h_cur, N, d, rng, T_start=5.0, T_end=0.001)
        if pk < best_pk:
            best_pk = pk
            best = h_new.copy()
    return best, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)
    target0 = n0 // 2
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    # Phase 1: Adam at native resolution
    if time.time() < deadline - 200:
        d1 = time.time() + (deadline - time.time()) * 0.12
        h1, c1 = adam_continuous(h0.copy(), n0, float(target0), d1, k=40, lr0=0.04, decay=0.992)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary annealing at native resolution (multi-kick)
    if time.time() < deadline - 150:
        rem = deadline - time.time()
        d2 = time.time() + rem * 0.35
        h2, c2 = multi_kick_anneal(h1 if 'h1' in dir() else best_h, n0, d2, rng, n_kicks=3)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Targeted refinement at native resolution
    if time.time() < deadline - 100:
        rem = deadline - time.time()
        d3 = time.time() + rem * 0.2
        h3 = to_binary(best_h, n0)
        h3, c3 = targeted_refine(h3, n0, d3, rng, n_peaks=4)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Higher resolution attempts
    sizes = [520, 660]
    for N in sizes:
        if time.time() > deadline - 120:
            break
        target = N // 2
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_interp = np.interp(xN, x0, best_h)
        h_interp = project_sum(h_interp, float(target))
        rem = deadline - time.time()
        # Adam at new resolution
        if rem > 80:
            d4 = time.time() + rem * 0.2
            h4, c4 = adam_continuous(h_interp, N, float(target), d4, k=30, lr0=0.03, decay=0.993)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()
        # Binary annealing at new resolution
        if time.time() < deadline - 60:
            rem2 = deadline - time.time()
            d5 = time.time() + rem2 * 0.55
            h5, c5v = multi_kick_anneal(h4 if 'h4' in dir() else h_interp, N, d5, rng, n_kicks=2)
            if c5v < best_c5:
                best_c5, best_h = c5v, h5.copy()
        # Targeted refinement
        if time.time() < deadline - 40:
            rem3 = deadline - time.time()
            d6 = time.time() + rem3 * 0.5
            h6 = to_binary(h5 if 'h5' in dir() else best_h, N)
            h6, c6 = targeted_refine(h6, N, d6, rng, n_peaks=3)
            if c6 < best_c5:
                best_c5, best_h = c6, h6.copy()

    # Phase 5: Final polish with small perturbation + Adam
    if time.time() < deadline - 5:
        Nb = len(best_h)
        hF = best_h.copy() + rng.randn(Nb) * 0.02
        hF = project_sum(hF, Nb / 2.0)
        dF = deadline - 3
        hF, cF = adam_continuous(hF, Nb, Nb / 2.0, dF, k=20, lr0=0.02, decay=0.995)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    # Final: also try binary polish on best
    if time.time() < deadline - 2:
        Nb = len(best_h)
        hB = to_binary(best_h, Nb)
        dB = deadline - 1
        hB, cB = anneal_swap(hB, Nb, dB, rng, T_start=1.0, T_end=0.0005)
        if cB < best_c5:
            best_c5, best_h = cB, hB.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
