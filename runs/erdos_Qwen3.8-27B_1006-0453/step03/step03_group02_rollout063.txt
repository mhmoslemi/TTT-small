import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, N, top_indices):
    grad = np.zeros(N)
    center = N - 1
    for idx in top_indices:
        j = int(idx) - center
        if j == 0:
            grad += 1.0 - 2.0 * h
        elif j > 0:
            if j < N:
                grad[:N - j] += 1.0 - h[j:N]
                grad[j:] -= h[:N - j]
        else:
            jp = -j
            if jp < N:
                grad[jp:] += 1.0 - h[:N - jp]
                grad[:N - jp] -= h[jp:N]
    return grad / float(len(top_indices))
