    return project_feasible(h_new, n_dst)

    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n0 = len(best_h)
    best_h = project_feasible(best_h, n0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Refine at parent resolution
    if elapsed() < budget_s * 0.35:
        best_h, best_c5 = refine_one(best_h, n0, elapsed, budget_s)

    # Phase 2: Try other resolutions
    for n_try in [120, 140, 180, 200, 100]:
        if elapsed() > budget_s * 0.80:
            break
        h_try = interpolate_h(best_h, n0, n_try)
        # Small perturbation to avoid identical landscape
        h_try = h_try + np.random.uniform(-0.02, 0.02, n_try)
        h_try = project_feasible(h_try, n_try)
        c5_try = compute_c5(h_try)
        # Only refine if promising or close
        if c5_try < best_c5 + 0.001:
            h_r, c5_r = refine_one(h_try, n_try, elapsed, min(budget_s, (elapsed() - t0) + budget_s * 0.15))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
                n0 = n_try

    # Phase 3: Random restarts at best resolution
    n_cur = len(best_h)
    for trial in range(10):
        if elapsed() > budget_s * 0.92:
            break
        h_p = best_h + 0.04 * np.random.randn(n_cur)
        h_p = project_feasible(h_p, n_cur)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 + 0.002:
            h_r, c5_r = refine_one(h_p, n_cur, elapsed, min(budget_s, elapsed() + 40))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 4: Final ultra-fine polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, len(best_h), 60, 4)

    # Final SLSQP at very high T
    if elapsed() < budget_s * 0.99:
        n_fin = len(best_h)
        for T, kk in [(50000, 3), (200000, None)]:
            if elapsed() > budget_s * 0.995:
                break
            try:
                hc = slsqp_step(best_h, T, kk, n_fin, 5000, 1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5 - 1e-14:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return best_h, best_c5, len(best_h)
