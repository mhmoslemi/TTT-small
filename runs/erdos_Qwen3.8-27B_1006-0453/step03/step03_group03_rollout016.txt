def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Use log-sum-exp for numerical stability
    shifted = T * (corr - m)
    logsum = m + np.log(np.sum(np.exp(shifted))) / T
    return logsum * dx

def power_obj(h, p, K):
    # (sum of top-K corr^p)^(1/p) / K * dx -- actually let's use all
    n = len(h)

    # Use top K entries
    topk = np.sort(corr)[-K:]
    return (np.sum(topk**p) / K)**(1.0/p) * dx

    return x.sum() - len(x) / 2.0

            grad[:n-kp] += -h[kp:]
    return grad / len(active)

import numpy as np
from scipy.optimize import minimize, minimize_nelder_mead
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    logsum = m + np.log(np.sum(np.exp(shifted))) / T
    return logsum * dx

def power_obj(h, p, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = np.sort(corr)[-K:]
    return (np.sum(topk ** p) / K) ** (1.0 / p) * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def subgrad_dir(h, corr, top_m=8):
    n = len(h)
    m = np.max(corr)
    thresh = m - 1e-6
    active = np.where(corr >= thresh)[0]
    if len(active) > top_m:
        active = active[:top_m]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] += -h[:n - k]
        else:
            kp = -k
            grad[kp:] += (1.0 - h[:n - kp])
            grad[:n - kp] += -h[kp:]
    return grad / len(active)

def momentum_subgrad(h, n_iters=2000, lr_init=0.05, mom=0.82):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(h_cur)
        g = subgrad_dir(h_cur, corr, top_m=8)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        v = mom * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.015, 0.5)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if stall > 350:
                break
    return best_h, best_c5

def slsqp_smooth(h, T, maxiter=800):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_power(h, p, K, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(power_obj, h, args=(p, K), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def snap_binary(h, n, frac=0.05):
    h_new = h.copy()
    h_new[h_new < frac] = 0.0
    h_new[h_new > 1.0 - frac] = 1.0
    return project_feasible(h_new, n)

def nm_erdos(h, maxiter=1500, tol=1e-13):
    n = len(h)
    def obj(x):
        h_p = project_feasible(np.clip(x, 0, 1), n)
        c5, _ = compute_c5(h_p)
        return c5
    try:
        res = minimize_nelder_mead(obj, h, options={'maxiter': maxiter, 'xatol': tol, 'fatol': tol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
