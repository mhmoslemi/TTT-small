import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_topk(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    n = len(corr)
    if K >= n:
        top_idx = np.arange(n)
        c_top = corr
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(c_top)) / len(c_top)
    else:
        w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-7:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def adam_opt(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K, iters_ref):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / max(iters_ref, 1), 1.0)
        tau = tau0 * (tau1 / max(tau0, 1e-10)) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5_val = gradient_topk(h, N, tau, K)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def local_search(h, N, target, dx, deadline, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    n_it = 0
    while time.time() < deadline:
        n_it += 1
        scale = max(0.003, 0.10 * (0.9997 ** n_it))
        r = rng.random()
        if r < 0.25:
            i = rng.randint(N)
            trial = best_h.copy()
            trial[i] += rng.randn() * scale
            trial = project_sum(trial, target)
        elif r < 0.50:
            i, j = rng.choice(N, size=2, replace=False)
            d = rng.uniform(-scale, scale)
            trial = best_h.copy()
            trial[i] += d
            trial[j] -= d
            trial = project_sum(trial, target)
        elif r < 0.75:
            L = rng.randint(2, 7)
            s = rng.randint(0, N - L)
            trial = best_h.copy()
            trial[s:s+L] += rng.randn(L) * scale * 0.4
            trial = project_sum(trial, target)
        elif r < 0.90:
            d = rng.randn(N)
            d -= d.mean()
            d /= (np.std(d) + 1e-15)
            trial = best_h + scale * d
            trial = project_sum(trial, target)
        else:
            # Gradient-guided step
            g, _ = gradient_topk(best_h, N, 200.0, 5)
            d = g / (np.std(g) + 1e-15)
            d -= d.mean()
            trial = best_h - scale * d
            trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c

    return best_h, best_c5

    if time.time() < deadline - 5:

    return best_h, best_c5
