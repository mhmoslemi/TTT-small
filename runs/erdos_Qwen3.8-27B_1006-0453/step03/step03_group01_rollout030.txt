```python
def compute_gradient(h, w_lags, N):
    """
    h: array of length N
    w_lags: dict or array mapping lag L (-(N-1)..N-1) to weight w[L]
    Returns: gradient array of length N
    """
    grad = np.zeros(N)
    # L=0
    grad += w_lags[0] * (1.0 - 2.0 * h)
    
    j = np.arange(N)
    
    for L in range(1, N):
        w_pos = w_lags[L]
        if abs(w_pos) < 1e-10:
            continue
        # Positive lag L
        # Term A: w_pos * (1-h[j-L]) for j >= L
        mask_A = j >= L
        grad[mask_A] += w_pos * (1.0 - h[j[mask_A] - L])
        # Term B: -w_pos * h[j+L] for j+L < N, i.e., j < N-L
        mask_B = j < N - L
        grad[mask_B] -= w_pos * h[j[mask_B] + L]
        
        w_neg = w_lags[-L] if -L in w_lags else 0
        if abs(w_neg) < 1e-10:
            continue
        # Negative lag -L (M=L)
        # Term C: w_neg * (1-h[j+L]) for j+L < N
        grad[mask_B] += w_neg * (1.0 - h[j[mask_B] + L])
        # Term D: -w_neg * h[j-L] for j >= L
        grad[mask_A] -= w_neg * h[j[mask_A] - L]
    
    return grad
```

    # For j = 0, ..., N-1:

    #  - sum_{L=1}^{min(N-1-j, N-1)} w[N-1+L] * h[j+L]  [positive lag, second term]
    #  + sum_{M=1}^{min(N-1-j, N-1)} w[N-1-M] * (1-h[j+M])  [negative lag, first term]

    # = sum_{L=1}^{j} w[N-1+L] - sum_{L=1}^{j} w[N-1+L]*h[j-L]

    wp = w[N:]  # length N-1, wp[L-1] = w[N-1+L] for L=1..N-1
    prefix_wp = np.cumsum(wp)  # prefix_wp[k] = sum_{L=1}^{k+1} w[N-1+L]

    part1a = np.zeros(N)
    part1a[1:] = prefix_wp[:N-1]
    
    # Second part: convolution of wp with h
    # sum_{L=1}^{j} wp[L-1]*h[j-L] = (convolve(wp, h))[j-1] for j >= 1

    # = conv_wp_h[j-1]
