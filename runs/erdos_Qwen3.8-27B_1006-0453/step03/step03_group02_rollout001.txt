                            K=30, mu_s=0.0, mu_e=0.5, total_it=300)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: Higher tau, stronger binary push
        if time.time() < deadline - 40:
            t2 = time.time() + (deadline - time.time()) * 0.30
            h2, c2 = adam_phase(h1.copy(), N, target, t2,
                                tau_s=80.0, tau_e=400.0, lr_s=0.01, lr_e=0.002,
                                K=15, mu_s=0.5, mu_e=2.0, total_it=400)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

        # Phase 3: Very high tau, tiny lr, polish
        if time.time() < deadline - 20:
            t3 = deadline - (deadline - time.time()) * 0.15
            h3, c3 = adam_phase(h2.copy(), N, target, t3,
                                tau_s=400.0, tau_e=2000.0, lr_s=0.003, lr_e=0.0003,
                                K=8, mu_s=2.0, mu_e=5.0, total_it=300)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()

        # Structured local search
        if time.time() < deadline - 10:
            hl, cl = structured_local(h3.copy(), N, target, deadline - 5,
                                      scale=0.03, n_trials=3000)
            if cl < best_c5:
                best_c5 = cl
                best_h = hl.copy()

        # Coordinate descent polish
        if time.time() < deadline - 6:
            hc, cc = coordinate_search(best_h.copy(), N, N // 2, deadline - 3, steps=4)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

    # Final global polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 4:
        hf, cf = adam_phase(best_h.copy(), Nb, Nb // 2, deadline - 1,
                            tau_s=2000.0, tau_e=8000.0, lr_s=0.001, lr_e=0.00005,
                            K=5, mu_s=3.0, mu_e=6.0, total_it=150)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    final_c5 = compute_c5(best_h, len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
