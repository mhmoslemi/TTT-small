```python
    grad = np.zeros(N)
    center = N - 1
    for k in range(N):  # only process non-negligible weights
        w = weights[center + k]
        if w < 1e-8:
            continue
        # Lag = +k
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:N-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:N-k]
    
    for k in range(1, N):
        w = weights[center - k]
        if w < 1e-8:
            continue
        # Lag = -k
        grad[k:] += w * (1.0 - h[:N-k])
        grad[:N-k] -= w * h[k:]
    return grad
```

```python
def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k

        if L == 0:

        elif L > 0:
            grad[:N-L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:N-L]
        else:
            M = -L
            if M < N:
                grad[:N-M] -= wk * h[M:]
                grad[M:] += wk * (1.0 - h[:N-M])
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad

    return best_h, best_peak * dx
```

```python
    corr = compute_corr(h)
    center = N - 1
    # Get top peaks
    peak_indices = np.argpartition(corr, -top_k)[-top_k:]
    hot = np.zeros(N, dtype=int)
    for pi in peak_indices:

        if L == 0:

                if h[i] == 1 and h[i-M] == 0:
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, N):
    h = (h > 0.5).astype(float)
    target = N // 2
    s = int(h.sum())
    if s > target:
        idx = np.where(h == 1)[0]
        h[idx[target:]] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - s]] = 1
    return h

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N-L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:N-L]
        else:
            M = -L
            if M < N:
                grad[:N-M] -= wk * h[M:]
                grad[M:] += wk * (1.0 - h[:N-M])
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 250 == 0:
            lr *= decay
    return best_h, best_peak * dx

def binary_local_search(h_bin, N, deadline, rng, n_trials=20, restart_every=600):
    dx = 2.0 / N
    best = h_bin.astype(float).copy()
    best_val = float(np.max(compute_corr(best)))
    cur = best.copy()
    cur_val = best_val
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    T = 2.5
    T_min = 0.0005
    stuck = 0
    greedy_end = time.time() + (deadline - time.time()) * 0.55
    in_greedy = True

    while time.time() < deadline:
        if in_greedy and time.time() < greedy_end:
            n_o = min(len(ones), n_trials)
            n_z = min(len(zeros), n_trials)
            if n_o < len(ones):
                io = rng.choice(len(ones), n_o, replace=False)
            else:
                io = np.arange(n_o)
            if n_z < len(zeros):
                iz = rng.choice(len(zeros), n_z, replace=False)
            else:
                iz = np.arange(n_z)
            best_sv = cur_val
            best_i, best_j = -1, -1
            for ii in io:
                i = ones[ii]
                cur[i] = 0
                for jj in iz:
                    j = zeros[jj]
                    cur[j] = 1
                    sv = float(np.max(compute_corr(cur)))
                    if sv < best_sv:
                        best_sv = sv
                        best_i, best_j = int(i), int(j)
                    cur[j] = 0
                cur[i] = 1
            if best_i >= 0:
                cur[best_i] = 0
                cur[best_j] = 1
                cur_val = best_sv
                if cur_val < best_val:
                    best_val = cur_val
                    best = cur.copy()
                    stuck = 0
                else:
                    stuck += 1
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
            else:
                stuck += 1
            if stuck >= restart_every:
                np_pert = rng.randint(3, 7)
                for _ in range(np_pert):
                    pi = ones[rng.randint(len(ones))]
                    pj = zeros[rng.randint(len(zeros))]
                    cur[pi] = 0
                    cur[pj] = 1
                cur_val = float(np.max(compute_corr(cur)))
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                stuck = 0
        else:
            if in_greedy and time.time() >= greedy_end:
                in_greedy = False
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            cur[i] = 0
            cur[j] = 1
            new_val = float(np.max(compute_corr(cur)))
            delta = new_val - cur_val
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best = cur.copy()
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
            else:
                cur[i] = 1
                cur[j] = 0
            if T > T_min:
                T *= 0.9996
            if int((deadline - time.time()) * 10) % 3000 == 0:
                if T < 0.1:
                    T = 0.5
    return best, best_val * dx
```
