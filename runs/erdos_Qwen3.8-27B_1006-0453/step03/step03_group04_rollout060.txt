import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h_b = (h > 0.5).astype(float)
    target = N // 2
    n_ones = int(h_b.sum())
    if n_ones > target:
        ones = np.where(h_b == 1)[0]
        h_b[ones[target:]] = 0
    elif n_ones < target:
        zeros = np.where(h_b == 0)[0]
        h_b[zeros[:target - n_ones]] = 1
    return h_b

def fast_anneal(h, N, deadline, rng, T0=5.0, T1=0.001):

    h_b = make_binary(h, N)
    cur_pk = float(np.max(compute_corr(h_b)))
    best_h = h_b.copy()
    best_pk = cur_pk
    t_start = time.time()
    
    while time.time() < deadline:
        elapsed = time.time() - t_start
        T = T0 * (T1 / T0) ** min(elapsed / max(deadline - t_start, 0.1), 1.0)
        
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_pk = float(np.max(compute_corr(h_b)))
        delta = new_pk - cur_pk
        
        if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
    
    return best_h, best_pk * dx

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(2*N-1)  # not useful directly
    # Compute gradient of the top-k weighted objective w.r.t. h
    grad_h = np.zeros(N)
    center = N - 1
    for ai in top:
        L = ai - center
        if L == 0:
            grad_h += (1.0 - 2.0 * h)
        elif L > 0:
            # corr[center+L] = sum_n h[n+L]*(1-h[n])
            # d/dh[i]: if i >= L: (1-h[i-L]); if i < N-L: -h[i+L]
            if L < N:
                grad_h[:N-L] -= h[L:]  # -h[i+L] for i in [0, N-L)
                grad_h[L:] += (1.0 - h[:N-L])  # (1-h[i-L]) for i in [L, N)
        else:
            M = -L
            if M < N:
                # corr[center-M] = sum_n h[n]*(1-h[n+M])
                # d/dh[i]: if i < N-M: (1-h[i+M]); if i >= M: -h[i-M]
                grad_h[:N-M] += (1.0 - h[M:])
                grad_h[M:] -= h[:N-M]
    grad_h /= k
    return grad_h, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
    return best_h, best_peak * dx

        if d < 0 or (T > 0.003 and rng.random() < np.exp(-d / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
    return best_h, best_pk * dx

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad_h = np.zeros(N)
    c = N - 1
    for ai in top:
        L = ai - c
        if L == 0:
            grad_h += (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad_h[:N-L] -= h[L:]
            grad_h[L:] += (1.0 - h[:N-L])
        elif 0 < -L < N:
            M = -L
            grad_h[:N-M] += (1.0 - h[M:])
            grad_h[M:] -= h[:N-M]
    grad_h /= k
    return grad_h, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        g, pk = grad_topk(h, N, k)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g*g
        it += 1
        h = h - lr * (m/(1-b1**it)) / (np.sqrt(v/(1-b2**it)) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
    return best_h, best_pk * dx

    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
