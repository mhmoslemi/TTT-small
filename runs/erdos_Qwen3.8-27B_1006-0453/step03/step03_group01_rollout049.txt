import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_grad(h, N, topK, thresh_frac):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    c5 = float(np.max(corr)) * (2.0 / N)
    cmax = float(np.max(corr))
    if cmax < 1e-15:
        return np.zeros(N), c5
    thresh = cmax - thresh_frac * cmax
    idx_near = np.where(corr >= thresh)[0]
    idx_top = np.argsort(corr)[-topK:]
    idx_all = np.unique(np.concatenate([idx_near, idx_top]))
    nw = float(len(idx_all))
    grad = np.zeros(N)
    for ai in idx_all:
        L = int(ai - center)
        wi = 1.0 / nw
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def adam_run(h, N, target, deadline, lr0, topK0, topK1, th0, th1, b1, b2):
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    while time.time() < deadline:
        t += 1
        frac = min(t / 2500.0, 1.0)
        topK = max(2, int(topK0 + (topK1 - topK0) * frac))
        th = th0 + (th1 - th0) * frac
        lr = lr0 / (1.0 + 0.004 * t)
        grad, c5_val = make_grad(h, N, topK, th)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mc = m / (1.0 - b1 ** t)
        vc = v / (1.0 - b2 ** t)
        h = project_sum(h - lr * mc / (np.sqrt(vc) + 1e-8), target)
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = best_h[i]
            trials = (0.0, 1.0, max(0.0, orig - 0.015), min(1.0, orig + 0.015))
            for tr in trials:
                if abs(tr - orig) < 1e-7:
                    continue
                best_h[i] = tr
                best_h = project_sum(best_h, target)
                c5 = compute_c5(best_h, dx)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    improved = True
                    break
                else:
                    best_h[i] = orig
            best_h = project_sum(best_h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    target = N / 2.0
    dx = 2.0 / N
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Adam from parent (55% time)
    t1 = t0 + (budget_s - 30) * 0.55
    h_cur = h0.copy()
    h_cur, c1 = adam_run(h_cur, N, target, t1, 0.05, 25, 4, 0.35, 0.08, 0.90, 0.999)
    if c1 < best_c5:
        best_c5, best_h = c1, h_cur.copy()

    # Phase 2: Perturbed restarts (30% time)
    t2 = t0 + (budget_s - 30) * 0.85
    n_restarts = 4
    per_r = max(25.0, (t2 - time.time()) / n_restarts)
    for r in range(n_restarts):
        if time.time() > t2 - 15:
            break
        rd = min(time.time() + per_r, t2 - 5)
        scale = 0.04 + 0.025 * r
        h_cur = best_h + rng.randn(N) * scale
        h_cur = project_sum(h_cur, target)
        h_cur, c2 = adam_run(h_cur, N, target, rd, 0.03, 18, 3, 0.25, 0.05, 0.90, 0.999)
        if c2 < best_c5:
            best_c5, best_h = c2, h_cur.copy()

    # Phase 3: Very focused final optimization (8% time)
    t3 = t0 + (budget_s - 30) * 0.93
    if time.time() < t3:
        h_cur = best_h.copy()
        h_cur, c3 = adam_run(h_cur, N, target, t3, 0.008, 3, 2, 0.06, 0.01, 0.92, 0.999)
        if c3 < best_c5:
            best_c5, best_h = c3, h_cur.copy()

    # Phase 4: Coordinate polish (7% time)
    if time.time() < deadline - 3:
        best_h, best_c5 = coord_polish(best_h, N, target, deadline - 2)

    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, N
