    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project(h, n):

    global _OPT_T, _OPT_N, _OPT_USE_PNORM
    _OPT_T = T
    _OPT_N = n
    _OPT_USE_PNORM = False
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fn}]
    res = minimize(_obj_fn, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def pnorm_slsqp(h, p, n, maxiter=500, ftol=1e-15):
    global _OPT_P, _OPT_N, _OPT_USE_PNORM
    _OPT_P = p
    _OPT_N = n
    _OPT_USE_PNORM = True
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fn}]
    res = minimize(_obj_fn, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    dx = 2.0 / n
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        c = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k + n - 1] = c
    # 1D convex piecewise-linear minimization via ternary + fine grid
    a, b = 0.0, 1.0
    for _ in range(10):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(corr_base + m1 * coef) <= np.max(corr_base + m2 * coef):
            b = m2
        else:
            a = m1
    xs = np.linspace(a, b, 300)
    vals = np.array([np.max(corr_base + x * coef) for x in xs])
    return float(xs[np.argmin(vals)])

def coord_descent(h, n, t_end, max_sweeps=300, tol=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - tol:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def binary_snap(h, n):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for i in range(n):
        if 0.01 < best_h[i] < 0.99:
            continue
        for target in [0.0, 1.0]:
            h_try = best_h.copy()
            h_try[i] = target
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish with increasing T
    p1_end = t0 + budget_s * 0.12
    for T in [30, 150, 600, 2500, 10000]:
        if time.time() > p1_end:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=400, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Long coordinate descent
    p2_end = t0 + budget_s * 0.38
    best_h, best_c5 = coord_descent(best_h, n, p2_end, max_sweeps=600, tol=1e-13)
    best_h, best_c5 = binary_snap(best_h, n)

    # Phase 3: Iterated local search with small perturbations
    p3_end = t0 + budget_s * 0.62
    work_h = best_h.copy()
    work_c5 = best_c5
    for itl in range(60):
        if time.time() > p3_end:
            break
        n_pert = max(2, n // 10)
        idx = np.random.choice(n, n_pert, replace=False)
        h_pert = work_h.copy()
        h_pert[idx] += np.random.randn(n_pert) * 0.02
        h_pert = project(h_pert, n)
        t_sub = min(p3_end, time.time() + budget_s * 0.022)
        h_ref, c5_ref = coord_descent(h_pert, n, t_sub, max_sweeps=50, tol=1e-13)
        h_ref, c5_ref = binary_snap(h_ref, n)
        if c5_ref < best_c5 - 1e-14:
            best_c5 = c5_ref
            best_h = h_ref
            work_h = h_ref.copy()
            work_c5 = c5_ref
        elif c5_ref < work_c5:
            work_h = h_ref
            work_c5 = c5_ref

    # Phase 4: p-norm SLSQP for different smoothing
    p4_end = t0 + budget_s * 0.76
    for p in [20, 80, 300]:
        if time.time() > p4_end:
            break
        try:
            hc = pnorm_slsqp(best_h, p, n, maxiter=400, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: More coordinate descent
    p5_end = t0 + budget_s * 0.87
    best_h, best_c5 = coord_descent(best_h, n, p5_end, max_sweeps=500, tol=1e-14)
    best_h, best_c5 = binary_snap(best_h, n)

    # Phase 6: Try alternative resolutions
    p6_end = t0 + budget_s * 0.93
    for n_try in [144, 200]:
        if time.time() > p6_end:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        h_ref, c5_ref = coord_descent(h_interp, n_try, min(p6_end, time.time() + budget_s * 0.03), max_sweeps=80, tol=1e-13)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            n = n_try

    # Phase 7: Final polish
    p7_end = t_end - 3
    best_h, best_c5 = coord_descent(best_h, n, p7_end, max_sweeps=200, tol=1e-14)
    best_h, best_c5 = binary_snap(best_h, n)
    if time.time() < t_end - 8:
        try:
            hc = slsqp_opt(best_h, 30000, n, maxiter=600, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                best_h, best_c5 = coord_descent(best_h, n, t_end - 2, max_sweeps=60, tol=1e-14)
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
