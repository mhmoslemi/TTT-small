- Increase h where h[n+L] is low and h[n] is high

This is a form of coordinate descent on the worst lags.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    if K < len(corr):
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        c_top = corr
    sc = tau * (c_top - c_top.max())
    sc = np.clip(sc, -50, 0)
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-8:
            continue
        k = int(top_idx[ix])
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def run_phase(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K0, K1, use_nesterov=True):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    it = 0
    while time.time() < deadline:
        frac = min(it / 500.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        K = max(4, int(K0 + (K1 - K0) * frac))
        if use_nesterov:
            y = np.clip(h + b1 * m, 0.0, 1.0)
            grad, c5_val = compute_gradient(y, N, tau, K)
        else:
            grad, c5_val = compute_gradient(h, N, tau, K)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    c5_fin = compute_c5(h, dx)
    if c5_fin < best_c5:
        best_c5, best_h = c5_fin, h.copy()
    return best_h, best_c5

def violation_correction(h, N, target, dx, n_rounds=30, deadline=None):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    center = N - 1
    rng = np.random.RandomState(777)
    for r in range(n_rounds):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top5 = np.argsort(corr)[-5:]
        for k in top5:
            L = k - center
            L = abs(L)
            if L == 0:
                high = np.where(h > 0.5)[0]
                if len(high) > 0:
                    j = high[rng.randint(len(high))]
                    h[j] -= 0.02
                    h2 = project_sum(h, target)
                    c = compute_c5(h2, dx)
                    if c < best_c5 - 1e-12:
                        best_c5, best_h = c, h2.copy()
                        h = h2
                    else:
                        h[j] += 0.02
            elif L < N:
                valid = N - L
                overlap = h[L:]* (1.0 - h[:valid])
                idx = np.argmax(overlap)
                if overlap[idx] > 0.01:
                    step = 0.015
                    h[idx] -= step
                    h[idx + L] += step
                    h2 = project_sum(h, target)
                    c = compute_c5(h2, dx)
                    if c < best_c5 - 1e-12:
                        best_c5, best_h = c, h2.copy()
                        h = h2
                    else:
                        h[idx] += step
                        h[idx + L] -= step
    return best_h, best_c5

def local_search(h, N, target, dx, deadline, scale=0.015, n_trials=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(42)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        nm = rng.randint(1, 5)
        idx = rng.choice(N, size=nm, replace=False)
        trial[idx] += rng.randn(nm) * scale
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-12:
            best_c5, best_h = c, trial
    return best_h, best_c5
