import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_tmp_base = np.zeros(1)
_tmp_coef = np.zeros(1)
_sum_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _f1d(x):
    return np.max(_tmp_base + x * _tmp_coef)

def _sum_con(x):
    return x.sum() - _sum_target

def _sum_con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-14):
    global _sum_target
    _sum_target = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent_fast(h, n, t_end, max_sweeps=50000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    k_range = np.arange(-(n - 1), n)
    global _tmp_base, _tmp_coef

    sweep = 0
    while sweep < max_sweeps and time.time() < t_end:
        sweep += 1
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            old_hi = h[i]
            j1 = i - k_range
            j2 = i + k_range
            mask1 = (j1 >= 0) & (j1 < n)
            mask2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2 * n - 1)
            if mask1.any():
                coef[mask1] += 1.0 - h[j1[mask1]]
            if mask2.any():
                coef[mask2] -= h[j2[mask2]]
            _tmp_base = corr - old_hi * coef
            _tmp_coef = coef
            res = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            new_hi = res.x
            delta = new_hi - old_hi
            if abs(delta) > 1e-13:
                new_corr = corr + delta * coef
                new_c5 = np.max(new_corr) * dx
                if new_c5 < best_c5 - 1e-15:
                    h[i] = new_hi
                    corr = new_corr
                    best_c5 = new_c5
                    improved = True
        if not improved:
            break
    return h, best_c5
