        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
        # Final binary polish
        if time.time() < deadline - 12:
            h5, c5v = anneal_binary(h4, N, deadline - 3, rng)
            if c5v < best_c5:
                best_c5, best_h = c5v, h5

    # Also try a different starting point: sinusoidal perturbation of parent
    if time.time() < deadline - 30:
        Nt = n0
        xtry = np.linspace(0.0, 2.0, Nt, endpoint=False)
        h_try = best_h + 0.06 * np.sin(2.0 * np.pi * 3.0 * xtry / 2.0) + rng.randn(Nt) * 0.03
        h_try = project_sum(h_try, Nt / 2.0)
        d_try = deadline - 5
        h_try, c_try = anneal_binary(h_try, Nt, d_try, rng)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
