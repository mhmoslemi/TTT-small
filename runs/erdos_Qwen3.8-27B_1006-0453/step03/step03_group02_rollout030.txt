def worst_lag_gradient(h, N):
    """Compute gradient of the worst lag's overlap."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    center = N - 1
    L = k_star - center
    grad = np.zeros(N)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:N-L])
        grad[:N-L] -= h[L:]
    else:
        M = -L
        grad[:N-M] += (1.0 - h[M:])
        grad[M:] -= h[:N-M]
    return grad, k_star

def topk_gradient(h, N, tau, K):

    center = N - 1
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top - tau * c_top.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N)
    w /= ws
    grad = np.zeros(N)

        wi = w[ix]
        if wi < 1e-7:

        L = k - center
        if L == 0:

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_gradient(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    n_corr = len(corr)
    if K >= n_corr:
        top_idx = np.arange(n_corr)
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = tau * (c_top - c_top.max())
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N)
    w /= ws
    grad = np.zeros(N)
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-8:
            continue
        k = int(top_idx[ix])
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad

def gd_phase(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K, max_iters):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / float(max(max_iters, 1)), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad = topk_gradient(h, N, tau, K)
        c5_val = compute_c5(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5
