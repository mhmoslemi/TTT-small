    m = np.zeros(N)
    v = np.zeros(N)
    lr = lr0
    it = 0
    best_h = h.copy()
    best_pk = float(np.max(np.correlate(h, 1.0-h, mode='full')))

        peak = float(np.max(corr))

import numpy as np
import time

def fast_corr(h):
    N = len(h)
    size = 1 << (2 * N - 2).bit_length()
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    r = np.fft.irfft(H * np.conj(G), size)
    return np.concatenate((r[size - N + 1:size], r[:N]))

def compute_c5(h, N):
    return float(np.max(fast_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def adam_run(h, N, deadline, lr0, k):
    target = N / 2.0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    lr = lr0
    it = 0
    best_h = h.copy()
    best_pk = float(np.max(fast_corr(h)))
    while time.time() < deadline:
        corr = fast_corr(h)
        peak = float(np.max(corr))
        if peak < best_pk:
            best_pk = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            L = ai - center
            if L == 0:
                grad += (1.0 - 2.0 * h) / k
            elif 0 < L < N:
                grad[L:] += (1.0 - h[:N - L]) / k
                grad[:N - L] -= h[L:] / k
            elif -N < L < 0:
                M = -L
                grad[:N - M] += (1.0 - h[M:]) / k
                grad[M:] -= h[:N - M] / k
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        h = h - lr * (m / (1.0 - b1 ** it)) / (np.sqrt(v / (1.0 - b2 ** it)) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= 0.97
    return best_h, best_pk * (2.0 / N)

def sa_binary(h, N, deadline, rng):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(np.float64)
    target = N // 2
    n1 = int(h_b.sum())
    if n1 > target:
        idx = np.where(h_b == 1)[0]
        sel = rng.choice(n1, n1 - target, replace=False)
        h_b[idx[sel]] = 0.0
    elif n1 < target:
        idx = np.where(h_b == 0)[0]
        sel = rng.choice(N - n1, target - n1, replace=False)
        h_b[idx[sel]] = 1.0

    best = h_b.copy()
    best_pk = float(np.max(fast_corr(h_b)))
    cur_pk = best_pk
    T = 2.5
    cool = 0.99997
    min_T = 0.001
    stagnant = 0
    stuck_limit = 8000

    while time.time() < deadline and T > min_T:
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_pk = float(np.max(fast_corr(h_b)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
            stagnant += 1
        T *= cool
        if stagnant > stuck_limit:
            stagnant = 0
            T = min(T * 3.0, 0.5)
            for _ in range(3):
                oi = np.where(h_b == 1)[0]
                oz = np.where(h_b == 0)[0]
                a = oi[rng.randint(len(oi))]
                b = oz[rng.randint(len(oz))]
                h_b[a] = 0.0
                h_b[b] = 1.0
            cur_pk = float(np.max(fast_corr(h_b)))
    return best, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    best_h = h0.copy()
    best_c5 = compute_c5(h0, n0)

    # Phase A: Optimize at native resolution
    N = n0
    if N % 2 == 1:
        N = N + 1  # make even for binary
    if N != n0:
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        best_h = np.interp(xN, x0, h0)
        best_h = project_sum(best_h, N / 2.0)
        best_c5 = compute_c5(best_h, N)

    d_a1 = time.time() + (deadline - time.time()) * 0.10
    h_a1, c_a1 = adam_run(best_h.copy(), N, d_a1, 0.03, 12)
    if c_a1 < best_c5:
        best_c5, best_h = c_a1, h_a1.copy()

    d_a2 = time.time() + (deadline - time.time()) * 0.12
    h_a2, c_a2 = sa_binary(h_a1, N, d_a2, rng)
    if c_a2 < best_c5:
        best_c5, best_h = c_a2, h_a2.copy()

    # Phase B: Upsample to 1.5x and optimize
    N2 = 2 * N
    if N2 % 2 == 1:
        N2 += 1
    x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
    h2 = np.interp(x2, np.linspace(0.0, 2.0, N, endpoint=False), best_h)
    h2 = project_sum(h2, N2 / 2.0)

    d_b1 = time.time() + (deadline - time.time()) * 0.20
    h_b1, c_b1 = adam_run(h2.copy(), N2, d_b1, 0.025, 15)
    if c_b1 < best_c5:
        best_c5, best_h = c_b1, h_b1.copy()

    d_b2 = time.time() + (deadline - time.time()) * 0.40
    h_b2, c_b2 = sa_binary(h_b1, N2, d_b2, rng)
    if c_b2 < best_c5:
        best_c5, best_h = c_b2, h_b2.copy()

    # Phase C: Fine-tune the best
    Nb = len(best_h)
    h_c = best_h.astype(np.float64) + rng.randn(Nb) * 0.02
    h_c = project_sum(h_c, Nb / 2.0)
    d_c = deadline - 3
    h_c, c_c = adam_run(h_c, Nb, d_c, 0.008, 8)
    if c_c < best_c5:
        best_c5, best_h = c_c, h_c.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
