def compute_c5(h, dx):

    lo, hi = -1.5, 1.5

        else:

        L = max_idx - (N - 1)  # the lag

        if rng.random() < 0.5:
            # Targeted: find a 1 at position i where i+L (or i-L) is 0
            if L >= 0:

                    zeros = np.where(h == 0)[0]
                    zeros = zeros[zeros != i + L]  # avoid swapping with the target
                    if len(zeros) == 0:
                        continue
                    j = zeros[rng.randint(len(zeros))]

                    continue

                    if len(zeros) == 0:

                    j = zeros[rng.randint(len(zeros))]
