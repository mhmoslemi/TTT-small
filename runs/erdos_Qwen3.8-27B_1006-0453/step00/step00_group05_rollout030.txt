import time
from scipy.optimize import minimize

def eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = tau * (corr - m)
    return dx * (m + np.log(np.sum(np.exp(scaled))) / tau)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    global initial_h_values
    
    best_c5 = np.inf
    best_h = None
    best_n = 0
    
    def update_best(h, n):
        nonlocal best_c5, best_h, best_n
        c5 = eval_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    # Initial
    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)
    n = len(h0)
    h = h0.copy()
    h = np.clip(h, 0, 1)
    target = n / 2.0
    h += (target - h.sum()) / n
    h = np.clip(h, 0, 1)
    update_best(h, n)
    
    # Phase 1: scipy optimize with smooth approximation
    if time.time() - t0 < budget_s * 0.3:
        dx = 2.0 / n
        tau = 50.0
        res = minimize(
            smooth_obj, h.copy(), args=(tau, dx),
            method='Powell',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
            options={'maxiter': 200, 'xtol': 1e-8}
        )
        h_opt = np.clip(res.x, 0, 1)
        h_opt += (target - h_opt.sum()) / n
        h_opt = np.clip(h_opt, 0, 1)
        update_best(h_opt, n)
    
    # Phase 2: Local SA (continuous transfers)
    T = 0.0005
    current = h.copy() if eval_c5(h, n) <= best_c5 else best_h.copy()
    c5_cur = eval_c5(current, n)

        j = rng.randint(n)
        if i == j:

        eps = rng.uniform(-0.03, 0.03)
        if current[i] + eps < 0 or current[j] - eps < 0 or current[j] - eps > 1:
            # Wait, we transfer from j to i: current[i] += eps, current[j] -= eps
            # Need: current[i] + eps ∈ [0,1], current[j] - eps ∈ [0,1]
            continue
        current[i] += eps
        current[j] -= eps
        c5_new = eval_c5(current, n)
        delta = c5_new - c5_cur
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            c5_cur = c5_new
            if c5_cur < best_c5:

                best_h = current.copy()

    if time.time() - t0 < budget_s * 0.9:

        # Fix to exactly n/2 ones

                h_bin[idx] = 0
                ones_idx = np.where(h_bin == 1)[0]
        elif n_ones < n // 2:

                h_bin[idx] = 1
                zeros_idx = np.where(h_bin == 0)[0]

import numpy as np
import time
from scipy.optimize import minimize

def eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = tau * (corr - m)
    return dx * (m + np.log(np.sum(np.exp(scaled))) / tau)

def fix_sum(h, n):
    target = n / 2.0
    diff = target - h.sum()
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # fine-tune remaining diff
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[:max(1, int(abs(residual)))]
        else:
            idx = np.argsort(h)[:max(1, int(abs(-residual)))]
        for i in idx:
            add = min(residual, 1.0 - h[i]) if residual > 0 else max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-12:
                break
    return h

def continuous_sa(h, n, rng, t_start, t_limit, T_start=0.0008, n_iter=4000000):
    c5_cur = eval_c5(h, n)
    best_c5 = c5_cur
    best_h = h.copy()
    T = T_start
    for it in range(n_iter):
        if it % 1000 == 0:
            if time.time() > t_limit:
                break
            T = max(T * 0.9995, 1e-8)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        eps = rng.uniform(-0.04, 0.04)
        hi_new = h[i] + eps
        hj_new = h[j] - eps
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        h[i] = hi_new
        h[j] = hj_new
        c5_new = eval_c5(h, n)
        delta = c5_new - c5_cur
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i] = hi_new - eps
            h[j] = hj_new + eps
    return best_h, best_c5

def binary_sa(h, n, rng, t_start, t_limit, n_iter=1500000):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    half = n // 2
    if n_ones > half:
        ones_idx = np.where(h_bin == 1)[0]
        for _ in range(n_ones - half):
            idx = rng.choice(ones_idx)
            h_bin[idx] = 0
            ones_idx = np.where(h_bin == 1)[0]
    elif n_ones < half:
        zeros_idx = np.where(h_bin == 0)[0]
        for _ in range(half - n_ones):
            idx = rng.choice(zeros_idx)
            h_bin[idx] = 1
            zeros_idx = np.where(h_bin == 0)[0]
    c5_cur = eval_c5(h_bin, n)
    best_c5 = c5_cur
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    T = 0.004
    for it in range(n_iter):
        if it % 500 == 0:
            if time.time() > t_limit:
                break
            T = max(T * 0.999, 1e-9)
        pi = rng.choice(ones)
        pj = rng.choice(zeros)
        h_bin[pi] = 0
        h_bin[pj] = 1
        c5_new = eval_c5(h_bin, n)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_bin.copy()
        else:
            h_bin[pi] = 1
            h_bin[pj] = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    global initial_h_values
    best_h = None
    best_c5 = np.inf
    best_n = 0

    # Get initial
    if initial_h_values is not None and len(initial_h_values) > 0:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)
    n0 = len(h0)
    n0 = n0 if n0 % 2 == 0 else n0 + 1
    if n0 != len(h0):
        h0 = np.resize(h0, n0)
    h0 = np.clip(h0, 0, 1)
    h0 = fix_sum(h0, n0)
    c5_0 = eval_c5(h0, n0)
    best_h, best_c5, best_n = h0.copy(), c5_0, n0

    # Phase 1: scipy smooth optimization
    t_check = t0 + 30
    if time.time() < t_check:
        dx = 2.0 / n0
        for tau in [30.0, 100.0, 300.0]:
            if time.time() > t_check:
                break
            try:
                res = minimize(
                    smooth_obj, h0.copy(), args=(tau, dx),
                    method='Powell',
                    bounds=[(0.0, 1.0)] * n0,
                    constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n0 / 2.0, 'jac': lambda x: np.ones(n0)}],
                    options={'maxiter': 150, 'xtol': 1e-10, 'ftol': 1e-10}
                )
                h_opt = np.clip(res.x, 0, 1)
                h_opt = fix_sum(h_opt, n0)
                c5_opt = eval_c5(h_opt, n0)
                if c5_opt < best_c5:
                    best_c5, best_h, best_n = c5_opt, h_opt.copy(), n0
            except Exception:
                pass

    # Phase 2: Continuous SA (main workhorse)
    t_lim2 = t0 + budget_s * 0.55
    h_work = best_h.copy()
    h_work, c5_cont = continuous_sa(h_work, n0, rng, t0, t_lim2, T_start=0.001, n_iter=5000000)
    if c5_cont < best_c5:
        best_c5, best_h, best_n = c5_cont, h_work.copy(), n0

    # Phase 2b: Another round with lower T (polish)
    t_lim2b = t0 + budget_s * 0.7
    h_work, c5_cont2 = continuous_sa(h_work, n0, rng, t0, t_lim2b, T_start=0.0003, n_iter=3000000)
    if c5_cont2 < best_c5:
        best_c5, best_h, best_n = c5_cont2, h_work.copy(), n0

    # Phase 3: Binary SA from continuous best
    t_lim3 = t0 + budget_s * 0.85
    h_bin_best, c5_bin = binary_sa(best_h, n0, rng, t0, t_lim3, n_iter=2000000)
    if c5_bin < best_c5:
        best_c5, best_h, best_n = c5_bin, h_bin_best.copy(), n0

    # Phase 3b: Binary SA from scratch (random init)
    t_lim3b = t0 + budget_s * 0.93
    if n0 % 2 == 0:
        h_rand = np.zeros(n0)
        h_rand[:n0 // 2] = 1.0
        rng.shuffle(h_rand)
        h_rand, c5_rand = binary_sa(h_rand, n0, rng, t0, t_lim3b, n_iter=1500000)
        if c5_rand < best_c5:
            best_c5, best_h, best_n = c5_rand, h_rand.copy(), n0

    # Phase 4: Try upsampled n
    t_lim4 = t0 + budget_s * 0.98
    if time.time() < t_lim4:
        for mult in [2, 3]:
            if time.time() > t_lim4:
                break
            n_new = best_n * mult
            if n_new % 2 != 0:
                n_new += 1
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_new, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0, 1)
            h_up = fix_sum(h_up, n_new)
            c5_up = eval_c5(h_up, n_new)
            if c5_up < best_c5:
                best_c5, best_h, best_n = c5_up, h_up.copy(), n_new
            # Brief local search
            t_sub = min(time.time() + 40, t_lim4)
            h_up, c5_up2 = continuous_sa(h_up, n_new, rng, t0, t_sub, T_start=0.0005, n_iter=800000)
            if c5_up2 < best_c5:
                best_c5, best_h, best_n = c5_up2, h_up.copy(), n_new

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = fix_sum(best_h, best_n)
    best_c5 = eval_c5(best_h, best_n)

    return best_h, best_c5, best_n
