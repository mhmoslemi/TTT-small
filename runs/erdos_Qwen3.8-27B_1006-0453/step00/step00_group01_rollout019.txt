        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def hard_c5_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    d = idx - (n - 1)
    c5 = float(corr[idx] * dx)
    grad = np.zeros(n)
    if d >= 0:
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] -= h[:n - d]
    else:
        dd = -d
        grad[dd:] += (1.0 - h[:n - dd])
        grad[:n - dd] -= h[dd:]
    grad *= dx
    return c5, grad

def smooth_c5_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    ev = np.exp(alpha * (corr - max_c))
    se = ev.sum()
    w = ev / se
    f = max_c + np.log(se) / alpha
    grad = np.zeros(n)
    ny = len(corr)
    for k in range(ny):
        d = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        if d >= 0:
            grad[:n - d] += wk * (1.0 - h[d:])
            grad[d:] -= wk * h[:n - d]
        else:
            dd = -d
            grad[dd:] += wk * (1.0 - h[:n - dd])
            grad[:n - dd] -= wk * h[dd:]
    return f, grad

def pg_optimize(h, n, alpha, n_iter, lr0):
    h = h.copy()
    for it in range(n_iter):
        f, g = smooth_c5_grad(h, n, alpha)
        lr = lr0 / (1.0 + 0.005 * it)
        h_new = h - lr * g
        h = project_feasible(h_new, n)
    return h

def sa_binary(h, n, n_iter, t_high, t_low):
    cur = h.copy()
    cur_c5 = compute_c5(cur, n)
    best = cur.copy()
    best_c5 = cur_c5
    for it in range(n_iter):
        t = t_high * (t_low / t_high) ** (it / max(n_iter - 1, 1))
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[np.random.randint(len(ones))]
        oz = zeros[np.random.randint(len(zeros))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = compute_c5(cur, n)
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        elif np.random.random() < np.exp(-(new_c5 - cur_c5) / max(t, 1e-12)):
            cur_c5 = new_c5
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
    return best, best_c5

def local_search_binary(h, n, max_rounds=50):
    cur = h.copy()
    cur_c5 = compute_c5(cur, n)
    for _ in range(max_rounds):
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        best_delta = 0.0
        best_move = (-1, -1)
        for oi in ones:
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c5 = compute_c5(cur, n)
                delta = cur_c5 - c5
                if delta > best_delta:
                    best_delta = delta
                    best_move = (oi, oz)
                cur[oi] = 1.0
                cur[oz] = 0.0
        if best_move[0] < 0:
            break
        cur[best_move[0]] = 0.0
        cur[best_move[1]] = 1.0
        cur_c5 = compute_c5(cur, n)
    return cur, cur_c5

def random_binary(n):
    h = np.zeros(n)
    h[np.random.permutation(n)[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s * 0.93

    best_h = None
    best_c5 = 1.0
    best_n = 0

    def record(h, n):
        nonlocal best_h, best_c5, best_n
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n
        return c5

    # Phase 1: Work from initial_h_values
    h_work = None
    h_init_n = 0
    try:
        h_work = np.array(initial_h_values, dtype=float)
        h_init_n = len(h_work)
    except Exception:
        h_work = None

    if h_work is not None and h_init_n > 0:
        record(h_work, h_init_n)
        n = h_init_n
        # Smooth projected gradient polish
        for alpha in [30, 80, 200, 500]:
            if time.time() > deadline:
                break
            h_new = pg_optimize(h_work, n, alpha, 300, 0.15)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
                h_work = h_new.copy()
        # Hard gradient polish (fewer steps, active constraint)
        if time.time() < deadline:
            h_c = h_work.copy()
            for it in range(500):
                c5_cur, g = hard_c5_grad(h_c, n)
                lr = 0.05 / (1.0 + 0.02 * it)
                h_c = project_feasible(h_c - lr * g, n)
            record(h_c, n)
            if compute_c5(h_c, n) < best_c5:
                h_work = h_c.copy()

    # Phase 2: Binary local search + SA for even n
    n_list = [44, 46, 48, 50, 52, 54, 56, 58, 60, 64, 68, 72, 80, 96]
    for n in n_list:
        if time.time() > deadline:
            break
        n_restarts = 3 if n <= 60 else 2
        for restart in range(n_restarts):
            if time.time() > deadline:
                break
            h_bin = random_binary(n)
            # Greedy local search
            h_ls, c5_ls = local_search_binary(h_bin, n, max_rounds=20)
            record(h_ls, n)
            # SA
            if time.time() > deadline:
                break
            sa_iters = 80000 if n <= 56 else 40000
            h_sa, c5_sa = sa_binary(h_ls, n, sa_iters, 0.04, 0.0005)
            record(h_sa, n)
            # Final local search polish
            if time.time() > deadline:
                break
            h_fin, c5_fin = local_search_binary(h_sa, n, max_rounds=30)
            record(h_fin, n)

    # Phase 3: Try SLSQP on continuous version near best
    if best_h is not None and time.time() < deadline:
        n = best_n
        # Try a small perturbation + smooth optimization
        for alpha in [100, 300]:
            if time.time() > deadline:
                break
            h_pert = project_feasible(best_h + 0.01 * np.random.randn(n), n)
            h_opt = pg_optimize(h_pert, n, alpha, 400, 0.1)
            record(h_opt, n)

    # Fallback
    if best_h is None:
        n = 46
        best_h = random_binary(n)
        best_h, best_c5 = local_search_binary(best_h, n, 30)
        best_n = n

    return (best_h, best_c5, best_n)
