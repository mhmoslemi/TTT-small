import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_t = 0.01

def _c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def _lse_gh(h, t):
    n = len(h)
    dx = 2.0 / n
    cv = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(cv)
    ev = np.exp((cv - mc) / t)
    se = np.sum(ev)
    val = mc + t * np.log(se)
    w = ev / se
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] += wk * (-h[lag:])
        else:
            m = -lag
            g[:n - m] += wk * (1.0 - h[m:])
            g[m:] += wk * (-h[:n - m])
    return val, g

def _f(x):
    return _lse_gh(x, _g_t)[0]

def _fg(x):
    return _lse_gh(x, _g_t)

def _ce(x):
    return np.sum(x) - _g_n * 0.5

def _cej(x):
    return np.ones(_g_n)

def _proj(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    tg = 0.5 * n
    if abs(np.sum(h) - tg) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > tg:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = tg - np.sum(hp)
    if abs(d) > 1e-10:
        mk = (hp < 1.0) if d > 0 else (hp > 0.0)
        c = np.sum(mk)
        if c > 0:
            hp[mk] += d / c
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _make_starts(n, in_h, in_n, seed):
    starts = []
    x_init = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, in_n), in_h)
    starts.append(_proj(x_init))
    x = np.linspace(0, 2, n)
    tri = _proj(1.0 - 2.0 * np.abs(x - 1.0))
    starts.append(tri)
    two = np.zeros(n)
    two[:n // 2] = 1.0
    starts.append(_proj(two * 0.5 + 0.25))
    rng = np.random.RandomState(seed)
    rnd = _proj(0.2 + 0.6 * rng.random(n))
    starts.append(rnd)
    return starts

def _slsqp_optimize(h0, n, t, maxiter):
    _g_n = n
    _g_t = t
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ce, 'jac': _cej}]
    try:
        r = minimize(_fg, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return _proj(r.x)
    except Exception:
        return _proj(h0)

def _grad_refine(h, n, t, steps, t0, budget):
    _g_n = n
    _g_t = t
    lr = 0.05
    best = _c5(h)
    best_h = h.copy()
    for _ in range(steps):
        if time.time() - t0 > budget:
            break
        _, g = _lse_gh(h, t)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        h_new = _proj(h - lr * g)
        cn = _c5(h_new)
        if cn < best - 1e-14:
            best = cn
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.7
        lr = min(lr, 0.1)
    return best_h, best

    bc = _c5(bh)
