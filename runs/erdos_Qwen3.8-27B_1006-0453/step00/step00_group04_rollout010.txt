        for j in range(target - curr):

    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')

    h = ((idx / phi) % 1.0 < 0.5).astype(np.float64)

        for j in range(curr - target):

    rng = np.random.default_rng(seed)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()

        h[j] = 1.0

import numpy as np
import time


def upsample(parent, n_points):
    n_parent = len(parent)
    h = np.zeros(n_points, dtype=np.float64)
    base = n_points // n_parent
    rem = n_points % n_parent
    pos = 0
    for i in range(n_parent):
        cnt = base + (1 if i < rem else 0)
        h[pos:pos + cnt] = parent[i]
        pos += cnt
    target = n_points // 2
    curr = int(np.sum(h))
    if curr > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        for j in range(curr - target):
            h[idx[j]] = 0.0
    elif curr < target:
        idx = np.where(h < 0.5)[0]
        np.random.shuffle(idx)
        for j in range(target - curr):
            h[idx[j]] = 1.0
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def make_golden(n, seed=0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    idx = np.arange(n)
    h = ((idx.astype(np.float64) / phi) % 1.0 < 0.5).astype(np.float64)
    fix_sum(h, n, seed)
    return h


def fix_sum(h, n, seed=0):
    target = n // 2
    curr = int(np.sum(h))
    if curr > target:
        rng = np.random.default_rng(seed)
        idx = np.where(h > 0.5)[0]
        rng.shuffle(idx)
        for j in range(curr - target):
            h[idx[j]] = 0.0
    elif curr < target:
        rng = np.random.default_rng(seed)
        idx = np.where(h < 0.5)[0]
        rng.shuffle(idx)
        for j in range(target - curr):
            h[idx[j]] = 1.0


def simulated_annealing(h, time_limit, seed=42):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()

    start_time = time.time()
    T = 0.008
    min_T = 0.00005
    cool_rate = 0.9998
    best_streak = 0
    no_improve = 0

    while time.time() - start_time < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)

        if new_c5 <= current_c5:
            current_c5 = new_c5
            no_improve = 0
            T = max(min_T, T * cool_rate)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                best_streak += 1
                T = max(min_T, T * 0.999)
        else:
            delta = new_c5 - current_c5
            accept = False
            if T > min_T * 2:
                ratio = -delta / T
                if ratio > -50.0:
                    accept = rng.random() < np.exp(ratio)
            if accept:
                current_c5 = new_c5
            else:
                h[i] = 1.0
                h[j] = 0.0
            no_improve += 1
            T = max(min_T, T * cool_rate)

        if no_improve > 3000:
            T = min(0.008, T * 3.0)
            no_improve = 0
            best_streak = 0

    return best_h, best_c5
