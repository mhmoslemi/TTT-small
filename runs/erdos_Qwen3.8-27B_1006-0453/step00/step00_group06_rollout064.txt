    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # proportional adjustment
    if s > target:
        # reduce elements proportional to their value
        reduction = (s - target) / s
        h = h * (1.0 - reduction)

        capacity = (1.0 - h).sum()
        needed = target - s
        frac = needed / capacity if capacity > 0 else 0
        h = h + frac * (1.0 - h)
        h = np.clip(h, 0, 1)
    # fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0, 1)

Actually, let me think about what really matters here. The problem is to minimize the maximum of a set of quadratic forms. This is a min-max problem. 

One effective approach: 
- Start from a good initial point
- Do coordinate descent: for each coordinate i, find the best value of h[i] (given all others fixed) that minimizes the max
- Repeat

For coordinate i, given h with h[i] = t, the correlation values are:
For shift k: corr[k] = sum_j h[j](1-h[j-k]) 
The terms involving h[i] are: h[i]*(1-h[i-k]) + (1-h[i])*h[i+k] (approximately)
= (1-h[i-k]) * t + h[i+k] * (1-t)
= h[i+k] + t*(1-h[i-k]-h[i+k])

So each corr[k] is linear in t (for fixed k where i and i±k are in range).

So the max over k of these linear functions in t is a convex piecewise-linear function. We can minimize it by finding the point where the maximum is minimized (like a 1D min-max).

This is actually a great approach! For each coordinate, the optimal value is where the "upper envelope" of the linear functions is minimized.

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def repair(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            m = (h < 1.0).astype(float)
        else:
            m = (h > 0.0).astype(float)
        s = m.sum()
        if s > 0:
            h += (d / s) * m
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def init_spread(n, rng):
    h = np.zeros(n)
    half = n // 2
    if n % 2 == 0:
        step = n / half
        offsets = rng.uniform(0, step * 0.6, half)
        pos = np.round(np.arange(half) * step + offsets).astype(int) % n
        pos = np.unique(np.sort(pos))
        if len(pos) < half:
            extra = np.setdiff1d(np.arange(n), pos)
            pos = np.sort(np.concatenate([pos, rng.choice(extra, half - len(pos), replace=False)]))
        h[pos[:half]] = 1.0
    else:
        hi = (n - 1) // 2
        step = n / (hi + 1)
        offsets = rng.uniform(-step * 0.2, step * 0.2, hi)
        pos = np.round(np.arange(hi) * step + step / 2 + offsets).astype(int) % n
        pos = np.unique(np.sort(pos))
        if len(pos) < hi:
            extra = np.setdiff1d(np.arange(n), pos)
            pos = np.sort(np.concatenate([pos, rng.choice(extra, hi - len(pos), replace=False)]))
        h[pos[:hi]] = 1.0
        rest = np.where(h < 0.5)[0]
        if len(rest) > 0:
            h[rest[0]] = 0.5
    return h

def resample_to_n(h_old, n_new, rng):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_float = np.interp(x_new, x_old, h_old)
    h_bin = (h_float > 0.5).astype(float)
    tgt = n_new / 2.0
    diff = tgt - h_bin.sum()
    if abs(diff) <= 2:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = zeros[np.argsort(-h_float[zeros])]
            h_bin[order[:int(abs(diff))]] = 1.0
        elif diff < 0:
            ones = np.where(h_bin == 1)[0]
            order = ones[np.argsort(h_float[ones])]
            h_bin[order[:int(abs(diff))]] = 0.0
        if h_bin.sum() == tgt:
            return h_bin
    return repair(h_float, n_new)

def sa_search(h, n, n_iters, t_hi, t_lo, rng):
    tgt = 0.5 * n
    is_binary = np.all((h == 0) | (h == 1) | (np.abs(h - 0.5) < 1e-10))
    best_h = h.copy()
    best_c = eval_c5(h)
    cur = h.copy()
    cur_c = best_c
    stuck = 0

    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        temp = t_hi * (t_lo / t_hi) ** frac
        trial = cur.copy()

        if is_binary:
            ones_idx = np.where(cur >= 0.5)[0]
            zeros_idx = np.where(cur < 0.5)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                continue
            if rng.random() < 0.7:
                i = ones_idx[rng.randint(len(ones_idx))]
                j = zeros_idx[rng.randint(len(zeros_idx))]
                trial[i], trial[j] = trial[j], trial[i]
            else:
                # multi-swap: move a block
                blen = rng.randint(1, 4)
                i = ones_idx[rng.randint(len(ones_idx))]
                j = zeros_idx[rng.randint(len(zeros_idx))]
                trial[i], trial[j] = trial[j], trial[i]
        else:
            r = rng.random()
            if r < 0.5:
                i, j = rng.choice(n, 2, replace=False)
                trial[i], trial[j] = trial[j], trial[i]
            else:
                w = rng.randint(2, min(8, n // 4) + 1)
                s = rng.randint(0, n - w)
                trial[s:s + w] += rng.randn(w) * 0.15
                trial = repair(trial, n)

        tc = eval_c5(trial)
        delta = tc - cur_c
        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / temp)):
            cur = trial
            cur_c = tc
            stuck = 0
            if tc < best_c:
                best_c = tc
                best_h = trial.copy()
        else:
            stuck += 1

    return best_h, best_c
