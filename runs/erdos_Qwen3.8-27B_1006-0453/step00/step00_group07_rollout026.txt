import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

    n = _cur_n
    c5 = compute_c5(h, n)
    pen = 100.0 * (np.sum(h) - n * 0.5) ** 2
    return c5 + pen

    _cur_n = n

                    current_c5 = c5
                    improved = True
