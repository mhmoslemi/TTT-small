```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max_obj(h, dx, temp):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(logsumexp(corr / temp)) * temp


def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(mask.sum())
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    n = 80
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except NameError:
        h0 = np.full(n, 0.5)

    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = repair_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def time_left():
        return budget_s - (time.time() - t0)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]

    # Phase 1: SLSQP with decreasing temperature
    for temp in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if time_left() < 100:
            break

        def obj(x):
            return smooth_max_obj(x, dx, temp)

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
        h_new = repair_sum(res.x, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 2: SLSQP from perturbed starts
    for attempt in range(4):
        if time_left() < 90:
            break
        scale = 0.03 * (attempt + 1)
        h_start = repair_sum(best_h + scale * rng.standard_normal(n), target_sum)
        temp = 0.003

        def obj2(x):
            return smooth_max_obj(x, dx, temp)

        res = minimize(obj2, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
        h_new = repair_sum(res.x, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 3: Random pairwise coordinate moves
    if time_left() > 40:
        steps = np.array([0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002])
        improved_recently = 0
        for it in range(50000):
            if time_left() < 20:

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

            vj = best_h[j] - sgn * step
            if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
```
