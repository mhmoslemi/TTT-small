        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        lo = max(current_h[i] - 1.0, -current_h[j])
        hi = min(current_h[i], 1.0 - current_h[j])
        if lo >= hi:
            continue

        scale = min(1.0, max(0.05, T))
        mid = 0.5 * (lo + hi)
        half_range = 0.5 * (hi - lo) * scale
        delta = mid + half_range * (2.0 * rng.random() - 1.0)
        if delta < lo:
            delta = lo
        elif delta > hi:
            delta = hi

        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c5 = evaluate_c5(new_h)
        dc = new_c5 - current_c5

        if dc < 0:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            exponent = -dc / T
            if exponent > -700.0 and rng.random() < math.exp(exponent):
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h.copy()

        T = max(T * cool_rate, T_min)

    return best_h, best_c5


def make_random_start(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return fix_sum(h, n / 2.0, rng)


def make_cosine_start(n, rng, freq, amp, phase):
    xs = np.linspace(0.0, 2.0, n)
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * xs + phase)
    h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, n / 2.0, rng)


def make_block_start(n, rng):
    pattern = np.array([1.0, 1.0, 0.0, 1.0, 0.0, 0.0])
    h = np.tile(pattern, n // len(pattern))
    if len(h) < n:
        h = np.concatenate([h, np.zeros(n - len(h))])
    h = h[:n]
    return fix_sum(h, n / 2.0, rng)


def make_smooth_start(n, rng):
    xs = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.3 * np.cos(3.0 * np.pi * xs) + 0.15 * np.cos(5.0 * np.pi * xs + 0.7)
    h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, n / 2.0, rng)


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(80, 0.5)

    n0 = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n0 / 2.0, rng)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: SA from parent construction
    rem = max(0.0, budget_s - (time.time() - t_start))
    t1 = rem * 0.38
    h1, c5_1 = sa_optimize(h0, t1, rng, T0=1.0, cool_rate=0.999999, T_min=0.001)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Refine with moderate temperature
    rem = max(0.0, budget_s - (time.time() - t_start))
    if rem > 5.0:
        t2 = rem * 0.30
        h2, c5_2 = sa_optimize(best_h, t2, rng, T0=0.2, cool_rate=0.999999, T_min=0.0005)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    # Phase 3: Low-temperature polish
    rem = max(0.0, budget_s - (time.time() - t_start))
    if rem > 5.0:
        t3 = rem * 0.35
        h3, c5_3 = sa_optimize(best_h, t3, rng, T0=0.03, cool_rate=0.9999995, T_min=0.0001)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 4: Alternative starting points
    rem = max(0.0, budget_s - (time.time() - t_start))
    if rem > 20.0:
        alt_starts = [
            make_random_start(80, rng),
            make_cosine_start(80, rng, 1.5, 0.4, 0.4),
            make_block_start(80, rng),
            make_smooth_start(80, rng),
            make_random_start(80, rng),
        ]
        for h_s in alt_starts:
            rem = max(0.0, budget_s - (time.time() - t_start))
            if rem < 12.0:
                break
            t_s = min(rem * 0.28, rem - 3.0)
            if t_s < 2.0:
                break
            h_t, c5_t = sa_optimize(h_s, t_s, rng, T0=0.6, cool_rate=0.999998, T_min=0.001)
            if c5_t < best_c5:
                best_c5, best_h = c5_t, h_t
            # Polish the alternate if it improved
            if c5_t < best_c5 + 0.005:
                rem2 = max(0.0, budget_s - (time.time() - t_start))
                if rem2 > 10.0:
                    t_p = rem2 * 0.3
                    h_p, c5_p = sa_optimize(best_h, t_p, rng, T0=0.05, cool_rate=0.999999, T_min=0.0002)
                    if c5_p < best_c5:
                        best_c5, best_h = c5_p, h_p

    # Final validation
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
