            if time.time() > t0 + budget_s * 0.95:
                break
            h_new2, c5_new2 = slsqp_opt(h2, n2, target2, beta, 200)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2
                n = n2
                target = target2
            h2 = h_new2

        # Coordinate polish on higher-res
        if time.time() < t0 + budget_s * 0.97:
            best_h, best_c5 = coordinate_descent_polish(best_h, target, len(best_h), 20)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5_final = compute_c5(best_h)

    return best_h, best_c5_final, len(best_h)
