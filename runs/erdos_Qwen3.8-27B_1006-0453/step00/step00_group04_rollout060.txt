import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def project_to_constraint(h, n):
    """Project h onto {0<=h<=1, sum=h_target} where h_target = n/2"""
    target = n / 2.0
    current = h.copy()
    current = np.clip(current, 0, 1)
    diff = target - np.sum(current)
    # Add/subtract uniformly to free (non-bound) entries
    for _ in range(50):
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum; add to entries < 1
            mask = current < 1.0
            if not np.any(mask):
                break
            add = diff / np.sum(mask)
            current[mask] = np.minimum(1.0, current[mask] + add)
        else:
            # Need to decrease sum; subtract from entries > 0
            mask = current > 0.0
            if not np.any(mask):
                break
            sub = -diff / np.sum(mask)
            current[mask] = np.maximum(0.0, current[mask] - sub)
        diff = target - np.sum(current)
    return current

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    """Smooth approximation of max"""
    shifted = T * vals  # vals already scaled by dx
    m = np.max(shifted)
    return (m + np.log(np.sum(np.exp(shifted - m)))) / T

    return np.max(corr) * dx

def project_constraint(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:

            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:

def compute_grad(h, n, T):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx  # shape (2n-1,)
    # Softmax weights
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    p = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros(n)
    # For each lag k, τ = k - (n-1)
    taus = np.arange(2*n-1) - (n-1)  # from -(n-1) to n-1
    
    for k in range(2*n-1):
        tau = taus[k]
        pk = p[k]
        # ∂c[k]/∂h[m] = (1-h[m-tau])*[0<=m-tau<n] - h[m+tau]*[0<=m+tau<n]
        # = g[m-tau]*[0<=m-tau<n] - h[m+tau]*[0<=m+tau<n]

        lo2 = max(0, -int(tau))
        hi2 = min(n, n - int(tau))
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2+tau:hi2+tau]
    
    grad *= dx
    return grad

    g = 1.0 - h

    for k in range(2*n-1):
        tau = int(taus[k])

        lo2 = max(0, -tau)
        hi2 = min(n, n - tau)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2+tau:hi2+tau]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_constraint(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h[mask] = np.clip(h[mask], 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    ev = np.exp(T * (corr - m))
    p = ev / np.sum(ev)
    g = 1.0 - h

    for k in range(2 * n - 1):
        tau = k - (n - 1)
        pk = p[k]
        lo1 = max(0, tau)

        lo2 = max(0, -tau)
