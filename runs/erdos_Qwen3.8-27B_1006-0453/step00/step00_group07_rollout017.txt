            for _ in range(n_dir // len(step_sizes)):
                if time_left() < 5:
                    break
                # Random direction with zero sum
                d = rng.randn(n)
                d -= np.mean(d)
                d /= np.linalg.norm(d)
                
                # Find max step that keeps in [0,1]
                pos_mask = d > 1e-12
                neg_mask = d < -1e-12
                if np.any(pos_mask):
                    max_pos = np.min((1 - best_h[pos_mask]) / d[pos_mask])
                else:
                    max_pos = np.inf
                if np.any(neg_mask):
                    max_neg = np.min(-best_h[neg_mask] / d[neg_mask])
                else:
                    max_neg = np.inf
                max_step = min(step, max_pos, max_neg)
                
                if max_step < 1e-10:
                    continue
                
                h_try = best_h + max_step * d
                h_try = np.clip(h_try, 0, 1)
                h_try = enhance_sum(h_try, target_sum)
                
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
    
    # Phase 3: Coordinate descent (focused)
    if time_left() > 20:
        imp_count = 0
        while time_left() > 10:
            improved = False
            for i in range(n):
                if time_left() < 5:
                    break
                for j in range(n):
                    if i == j or time_left() < 5:
                        continue
                    for delta in [0.01, -0.01, 0.02, -0.02]:
                        h_try = best_h.copy()
                        h_try[i] += delta
                        h_try[j] -= delta
                        if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                            c5_try = compute_c5(h_try)
                            if c5_try < best_c5 - 1e-12:
                                best_c5 = c5_try
                                best_h = h_try.copy()
                                improved = True
                                break
            if not improved:
                break
    
    # Final
    best_h = np.clip(best_h, 0, 1)
    best_h = enhance_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    if tau < 1.0:
        return m
    exps = np.exp(np.clip(tau * (vals - m), -500, 500))
    return float(m + np.log(np.sum(exps)) / tau)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-9) & (h < 1.0 - 1e-9)
    n_adj = int(np.sum(mask))
    if n_adj > 0:
        h[mask] += diff / n_adj
    else:
        h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        h += diff2 / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, dx, target_sum, tau, max_iter, cons):
    bounds = [(0, 1)] * n
    result = minimize(
        lambda x: smooth_max(x, dx, tau),
        h0, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    if result.x is not None:
        return fix_sum(np.clip(result.x, 0, 1), target_sum)
    return h0.copy()

def random_direction_search(h_best, c5_best, n, target_sum, rng, n_iters, time_fn):
    for _ in range(n_iters):
        if time_fn() < 8:
            break
        d = rng.randn(n)
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            continue
        d /= norm
        for step in [0.3, 0.1, 0.05, 0.02]:
            pos = d > 1e-12
            neg = d < -1e-12
            max_s = np.inf
            if np.any(pos):
                max_s = min(max_s, np.min((1.0 - h_best[pos]) / d[pos]))
            if np.any(neg):
                max_s = min(max_s, np.min(-h_best[neg] / d[neg]))
            alpha = min(step, max_s)
            if alpha < 1e-10:
                continue
            h_try = fix_sum(h_best + alpha * d, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-12:
                return h_try, c5_try
    return h_best, c5_best

def coordinate_descent(h_best, c5_best, n, target_sum, rng, max_rounds, time_fn):
    for _ in range(max_rounds):
        if time_fn() < 8:
            break
        improved = False
        for i in range(n):
            if time_fn() < 5:
                break
            candidates = rng.choice([j for j in range(n) if j != i], size=min(8, n-1), replace=False)
            for j in candidates:
                for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                    h_try = h_best.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                        c5_try = compute_c5(h_try)
                        if c5_try < c5_best - 1e-12:
                            h_best = h_try
                            c5_best = c5_try
                            improved = True
                            break
        if not improved:
            break
    return h_best, c5_best

def bangbang_push(h, target_sum, n):
    h_work = h.copy()
    count = 0
    for _ in range(200):
        mid = np.where((h_work > 0.15) & (h_work < 0.85))[0]
        if len(mid) == 0:
            break
        idx = mid[count % len(mid)]
        cur = h_work[idx]
        if cur < 0.5:
            new_val = min(1.0, (target_sum - np.sum(h_work) + cur) / 1.0)
            if new_val < 0.05:
                new_val = 0.0
            else:
                new_val = max(0.0, new_val)
        else:
            new_val = max(0.0, target_sum - np.sum(h_work) + cur - 1.0)
            if new_val > 0.95:
                new_val = 1.0
            else:
                new_val = min(1.0, new_val)
        h_work[idx] = new_val
        count += 1
    return fix_sum(h_work, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir():
        try:
            h0 = np.array(initial_h_values, dtype=float)
        except (NameError, Exception):
            h0 = np.full(80, 0.5)
    else:
        h0 = np.full(80, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP with annealed tau
    tau_schedule = [20, 50, 100, 200, 500, 1000, 2000, 4000]
    cur_h = h0.copy()
    for tau in tau_schedule:
        if time_left() < 50:
            break
        cur_h = slsqp_optimize(cur_h, n, dx, target_sum, tau, 300, cons)
        c5 = compute_c5(cur_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cur_h.copy()

    # Phase 2: Multiple SLSQP restarts with perturbation
    if time_left() > 100:
        for restart in range(5):
            if time_left() < 50:
                break
            h_pert = best_h + rng.randn(n) * 0.05
            h_pert = fix_sum(h_pert, target_sum)
            for tau in [100, 500]:
                if time_left() < 30:
                    break
                h_new = slsqp_optimize(h_pert, n, dx, target_sum, tau, 200, cons)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_pert = h_new.copy()

    if time_left() > 30:
        n_search = min(3000, int(time_left() * 40))
        best_h, best_c5 = random_direction_search(
            best_h, best_c5, n, target_sum, rng, n_search, time_left)

    # Phase 4: Coordinate descent
    if time_left() > 20:
        max_rounds = min(50, int(time_left() * 0.5))
        best_h, best_c5 = coordinate_descent(
            best_h, best_c5, n, target_sum, rng, max_rounds, time_left)

    # Phase 5: Bang-bang pushing + re-optimize
    if time_left() > 40:
        h_bb = bangbang_push(best_h, target_sum, n)
        c5_bb = compute_c5(h_bb)
        if c5_bb < best_c5:
            best_c5 = c5_bb
            best_h = h_bb.copy()
        h_refined = slsqp_optimize(h_bb, n, dx, target_sum, 1000, 200, cons)
        c5_ref = compute_c5(h_refined)
        if c5_ref < best_c5:

            best_h = h_refined.copy()

        c5_polish = compute_c5(h_polish)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()
        best_h, best_c5 = random_direction_search(
            best_h, best_c5, n, target_sum, rng, 500, time_left)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()

    return best_h, best_c5, n
