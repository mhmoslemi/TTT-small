    return np.max(vals)

def smooth_obj(h, dx, beta):
    vals = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def smooth_grad(h, dx, beta):
    n = len(h)
    vals = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(vals)
    ev = np.exp(beta * (vals - m))
    w = ev / ev.sum()
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for i in range(n):
        v1 = (i - lags >= 0) & (i - lags < n)
        v2 = (i + lags >= 0) & (i + lags < n)
        t1 = np.dot(w[v1], 1.0 - h[i - lags[v1]])
        t2 = np.dot(w[v2], h[i + lags[v2]])
        grad[i] = dx * (t1 - t2)
