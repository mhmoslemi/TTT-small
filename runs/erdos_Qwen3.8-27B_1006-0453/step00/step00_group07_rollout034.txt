```python
def gradient_step(h, n, dx, T, rng):
    """Take a gradient-guided pair swap step."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    worst = int(np.argmax(corr))
    
    # Compute gradient of the worst term
    grad = np.zeros(n)
    k = abs(worst - (n - 1))
    
    if k == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif worst > n - 1:
        # positive shift k: f = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])*dx
        for i in range(n - k):
            grad[i] -= h[i+k] * dx
            grad[i+k] += (1.0 - h[i]) * dx
    else:
        # negative shift k: f = sum_{i=k}^{n-1} h[i]*(1-h[i-k])*dx
        for i in range(k, n):
            grad[i] += (1.0 - h[i-k]) * dx
            grad[i-k] -= h[i] * dx
    
    # Find best transfer: from high grad to low grad
    # We want to decrease h where grad > 0, increase where grad < 0
    # Transfer from j (grad > 0) to i (grad < 0)
    
    # Sort by gradient
    order = np.argsort(grad)
    # Best donor: smallest grad (most negative) -> we want to increase here
    # Best receiver: largest grad (most positive) -> we want to decrease here
    
    # Actually, we want to DECREASE f:
    # df ≈ grad . dh
    # If we increase h[i] by d and decrease h[j] by d:
    # df ≈ grad[i]*d - grad[j]*d = d*(grad[i] - grad[j])
    # We want this negative, so grad[i] < grad[j]
    # Best: i = argmin(grad), j = argmax(grad)
    
    i = int(order[0])  # min grad -> increase
    j = int(order[-1])  # max grad -> decrease
    if i == j:
        return None
    
    # Transfer amount
    d_max = min(h[j], 1.0 - h[i])
    d_min = max(-h[i], h[j] - 1.0)
    if d_max <= d_min + 1e-12:
        return None
    
    # Scale by learning rate
    lr = rng.uniform(0.01, 0.3)
    d = lr * d_max
    d = np.clip(d, d_min, d_max)
    
    return i, j, d
```

```python
import time

def run(seed=42, budget_s=1100, **kwargs):

        else:
```

```python
    best = arr.copy()
    best_c5 = float(np.max(np.correlate(best, 1.0 - best, mode='full') * dx))

    def try_swap(h, i, j, dval, T):
        """Apply swap, return (new_c5, accepted)."""
        h[i] += dval
        h[j] -= dval
        c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
        if c5 <= 0:
            h[i] -= dval
            h[j] += dval
            return c5, False

        h[i] -= dval
        h[j] += dval
        return c5, False

                j = int(rng.integers(0, n))
```
