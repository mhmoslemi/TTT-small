import numpy as np
from scipy.optimize import minimize
import time

def make_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-10:
        h = h * target_sum / s
        h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 0.99999)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-5)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, args):
    tau, dx = args
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    if tau < 1e-8:
        return cmax
    sh = (corr - cmax) / tau
    return cmax + tau * np.log(np.sum(np.exp(sh)))

def eq_fun(h, args):
    return np.sum(h) - args[0]

def eq_jac(h, args):
    return np.ones(len(h))

def try_round(h, target, levels):
    h_r = np.round(h / levels) * levels
    h_r = np.clip(h_r, 0.0, 1.0)
    if abs(np.sum(h_r) - target) > 1e-8:
        h_r = make_feasible(h_r, target)
        if abs(np.sum(h_r) - target) > 1e-6:
            return None
    return h_r
