    P[0] = h0.copy()
    Pb[0] = h0.copy()
    PbV[0] = eval_c5(P[0])

    for i in range(1, pop_size):
        P[i] = repair_feasibility(h0 + 0.25 * rng.standard_normal(n), target, rng)
        PbV[i] = eval_c5(P[i])
        Pb[i] = P[i].copy()

    gi = int(np.argmin(PbV))
    gb = Pb[gi].copy()
    gbV = PbV[gi]

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        w = w_init - (w_init - w_end) * it / max(max_iters, 1)
        for i in range(pop_size):
            r1 = rng.random(n)
            r2 = rng.random(n)
            V[i] = w * V[i] + c1 * r1 * (Pb[i] - P[i]) + c2 * r2 * (gb - P[i])
            P[i] = repair_feasibility(P[i] + V[i], target, rng)
            val = eval_c5(P[i])
            if val < PbV[i]:
                PbV[i] = val
                Pb[i] = P[i].copy()
                if val < gbV:
                    gbV = val
                    gb = P[i].copy()

    return Pb[gi].copy(), float(PbV[gi]), gb.copy(), float(gbV)


def local_search(h, target, n, rng, deadline):
    c5 = eval_c5(h)
    step = 0.12
    stagnant = 0
    max_stagnant = 3000

    while time.time() < deadline:
        j = rng.integers(n)
        k = rng.integers(n)
        if j == k:
            continue
        d = rng.uniform(-step, step)
        nj = h[j] + d
        nk = h[k] - d
        if 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[j] = nj
            h[k] = nk
            val = eval_c5(h)
            if val < c5:
                c5 = val
                step = min(0.4, step * 1.03)
                stagnant = 0
            else:
                h[j] = nj - d
                h[k] = nk + d
                step = max(0.008, step * 0.97)
                stagnant += 1
        else:
            step = max(0.008, step * 0.95)
            stagnant += 1

        if stagnant > max_stagnant:
            h = repair_feasibility(h + 0.15 * rng.standard_normal(n), target, rng)
            c5 = eval_c5(h)
            step = 0.12
            stagnant = 0

    return h, c5


def fine_tune(h, target, n, rng, deadline):
    c5 = eval_c5(h)
    while time.time() < deadline:
        j = rng.integers(n)
        k = rng.integers(n)
        if j == k:
            continue
        for _ in range(3):
            d = rng.uniform(-0.03, 0.03)
            nj = h[j] + d
            nk = h[k] - d
            if 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
                h[j] = nj
                h[k] = nk
                val = eval_c5(h)
                if val < c5:
                    c5 = val
                    break
                else:
                    h[j] = nj - d
                    h[k] = nk + d
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except (NameError, UnboundLocalError):
        n = 80
        h0 = np.ones(n) * 0.5

    target = n / 2.0

    best_h = repair_feasibility(h0.copy(), target, rng)
    best_c5 = eval_c5(best_h)

    # Phase 1: PSO exploration
    pso_deadline = t0 + min(180.0, budget_s * 0.3)
    p_h, p_c5, gb, gb_c5 = pso_phase(
        best_h, target, n, rng, max_iters=200,
        time_limit=pso_deadline, t0=t0
    )
    if p_c5 < best_c5:
        best_c5 = p_c5
        best_h = p_h.copy()
    if gb_c5 < best_c5:
        best_c5 = gb_c5
        best_h = gb.copy()

    # Phase 2: Local search from best
    ls_deadline = t0 + budget_s * 0.65
    best_h, best_c5 = local_search(best_h.copy(), target, n, rng, ls_deadline)

    # Phase 3: Try local search from a few other individuals
    if time.time() < t0 + budget_s * 0.8:
        for trial in range(3):
            cand = repair_feasibility(
                best_h + 0.1 * rng.standard_normal(n), target, rng
            )
            sub_deadline = min(t0 + budget_s * 0.8, time.time() + 60.0)
            if time.time() >= sub_deadline:
                break
            cand, cand_c5 = local_search(cand, target, n, rng, sub_deadline)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

    # Phase 4: Fine tuning
    ft_deadline = t0 + budget_s * 0.97
    if time.time() < ft_deadline:
        best_h, best_c5 = fine_tune(best_h.copy(), target, n, rng, ft_deadline)

    # Final repair and validation
    best_h = repair_feasibility(best_h.copy(), target, rng)
    final_c5 = eval_c5(best_h)
    if not np.isfinite(final_c5):
        best_h = repair_feasibility(h0.copy(), target, rng)
        final_c5 = eval_c5(best_h)

    return (best_h, float(final_c5), n)
