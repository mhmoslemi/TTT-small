    # Project h onto {0<=h<=1, sum(h)=target_sum}
    # h'[i] = clip(h[i] - lambda, 0, 1), find lambda s.t. sum = target_sum
    lo, hi = -1.0, 1.0  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Small correction for floating point
    diff = target_sum - np.sum(h_proj)
    # Distribute to free variables
    free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
    if np.any(free):
        h_proj[free] += diff / np.sum(free)
        h_proj = np.clip(h_proj, 0, 1)
    return h_proj

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    c5_val = np.max(corr) * dx
    return c5_val, corr * dx

def compute_grad_smooth(h, alpha=200.0):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full') * dx
    
    # Softmax weights
    shifted = alpha * corr - alpha * np.max(corr)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    # Exclude negligible weights
    weights[weights < 1e-8] = 0.0
    
    # Gradient: grad[j] = dx * sum_k weights[k] * dc[k]/dh[j]
    # dc[k]/dh[j] = (1-h[j+l_k])*[0<=j+l_k<n] - h[j-l_k]*[0<=j-l_k<n]
    # where l_k = k - (n-1)
    
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        l = k - (n - 1)
        w = weights[k]
        # (1-h[j+l]) for valid j+l
        lo1 = max(0, -l)
        hi1 = min(n, n - l)  # j+l < n means j < n-l
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+l:hi1+l])
        # -h[j-l] for valid j-l
        lo2 = max(0, l)
        hi2 = min(n, n + l) if l > 0 else n  # j-l >= 0 means j >= l
        if l >= 0:
            lo2 = l
            hi2 = n
        else:
            lo2 = 0
            hi2 = n + l
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-l:hi2-l]
    
    grad *= dx
    return grad

        # Term 1: (1-h[j-k]) for 0<=j-k<n → j in [k, k+n-1] ∩ [0, n-1]

def project_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_p)
    if abs(diff) > 1e-13:
        free = (h_p > 1e-14) & (h_p < 1.0 - 1e-14)
        if np.any(free):
            h_p[free] += diff / np.count_nonzero(free)
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def optimize_smooth_pg(h0, max_iters=8000, alpha_start=80.0, time_limit=60.0):
    start = time.time()
    n = len(h0)
    target = n / 2.0
    h = project_constraint(h0.astype(float).copy(), target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha_start
    lr = 0.05
    no_improve = 0
    patience = 300
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        c5 = compute_c5(h)
        grad = compute_smooth_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_constraint(h - step * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-12:
                h = h_new
                c5 = c5_new
                improved = True
                break
            step *= 0.5
        if not improved:
            h = project_constraint(h - step * grad * 0.5, target)
            c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve == patience:
            alpha = max(10.0, alpha * 0.6)
            lr = max(0.005, lr * 1.3)
            no_improve = 0
        elif no_improve > patience * 3:
            alpha = min(500.0, alpha * 1.25)
            lr = max(0.003, lr * 0.7)
            no_improve = patience
        if it > 0 and it % 200 == 0:
            if improved:
                lr = min(0.1, lr * 1.05)
            else:
                lr = max(0.005, lr * 0.9)
    return best_h, best_c5

def local_search(h0, rng, n_perturb=500, time_limit=15.0):
    start = time.time()
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    scale = 0.1
    for _ in range(n_perturb):
        if time.time() - start > time_limit:
            break
        mask = rng.random(n) < 0.3
        h_new = best_h + scale * (rng.rand(n) - 0.5) * mask
        h_new = project_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            scale = min(0.3, scale * 1.1)
        else:
            scale = max(0.02, scale * 0.95)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed())

    # Gather candidates
    candidates = []
    try:
        h_init = np.array(initial_h_values, dtype=float)
        candidates.append(h_init)
    except Exception:
        h_init = None

    n_vals = [len(h_init)] if h_init is not None else []
    for n in [33, 40, 47, 50, 64, 80]:
        if n not in n_vals:
            n_vals.append(n)

    for n in n_vals:
        if elapsed() > budget_s * 0.15:
            break
        h_blk = np.zeros(n)
        h_blk[:n // 2] = 1.0
        if n % 2 == 1:
            h_blk[n // 2] = 0.5
        candidates.append(h_blk.copy())
        h_r = rng.rand(n)
        h_r = project_constraint(h_r, n / 2.0)
        candidates.append(h_r)
        # Periodic pattern
        h_pat = np.zeros(n)
        p = max(3, n // 6)
        for i in range(n):
            phase = (i % p) / p
            h_pat[i] = 1.0 if phase < 0.5 else 0.0
        h_pat = project_constraint(h_pat, n / 2.0)
        candidates.append(h_pat)

    # Optimize each candidate
    for idx, h0 in enumerate(candidates):
        if remaining() < 20:
            break
        t_alloc = min(remaining() * 0.15, 90.0)
        h_opt, c5_opt = optimize_smooth_pg(h0, max_iters=5000, time_limit=t_alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        # Quick local search on result
        if remaining() > 15:
            h_ls, c5_ls = local_search(h_opt, rng, n_perturb=200, time_limit=min(8.0, remaining() * 0.05))
            if c5_ls < best_c5:
                best_c5 = c5_LS if False else c5_ls
                best_c5 = c5_ls
                best_h = h_ls

    # Multi-round refinement on best
    if best_h is not None:
        for rnd in range(3):
            if remaining() < 30:
                break
            t_alloc = min(remaining() * 0.25, 120.0)
            alpha_use = 150.0 + rnd * 100.0
            h_ref, c5_ref = optimize_smooth_pg(best_h, max_iters=8000,
                                               alpha_start=alpha_use, time_limit=t_alloc)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
            if remaining() > 10:
                h_ls, c5_ls = local_search(best_h, rng, n_perturb=400,
                                           time_limit=min(12.0, remaining() * 0.1))
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls

    # Final local search
    if best_h is not None and remaining() > 10:
        h_fin, c5_fin = local_search(best_h, rng, n_perturb=1000,
                                     time_limit=min(20.0, remaining() * 0.8))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    if best_h is None:
        best_h = np.array(initial_h_values, dtype=float)
        best_c5 = compute_c5(best_h)

    # Final verification
    best_h = project_constraint(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
