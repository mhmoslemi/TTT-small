    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full") * dx
    max_val = np.max(c)
    mask = c >= (max_val - 1e-12)
    count = mask.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for idx in range(2*n-1):
        if not mask[idx]:
            continue
        m = idx - (n - 1)  # lag

        # p-m in [0, n-1] => p in [m, m+n-1] intersect [0, n-1]
        p_start = max(0, m)

        src_start = p_start - m
        src_end = src_start + (p_end - p_start)

            h[mask] = np.minimum(h[mask] + adj, 1.0)

            adj = diff / mask.sum()
            h[mask] = np.maximum(h[mask] + adj, 0.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def obj_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = float(np.max(c))
    mask = c >= (max_val - 1e-13)
    count = float(mask.sum())
    grad = np.zeros(n)
    omh = 1.0 - h
    for idx in np.where(mask)[0]:
        m = int(idx) - (n - 1)
        ps = max(0, m)
        pe = min(n, m + n)
        ss = ps - m
        grad[ps:pe] += omh[ss:ss + (pe - ps)]
        ps2 = max(0, -m)
        pe2 = min(n, n - m)
        ss2 = ps2 + m
        grad[ps2:pe2] -= h[ss2:ss2 + (pe2 - ps2)]
    grad *= dx / count
    return max_val, grad

        strategy = r % 5
        if strategy == 0:
            h_try = best_h + 0.08 * rng.randn(n0)
        elif strategy == 1:
            freq = rng.randint(1, 4)
            phase = rng.uniform(0, 2 * np.pi)
            h_try = best_h + 0.12 * np.sin(2 * np.pi * freq * np.arange(n0) / n0 + phase)
        elif strategy == 2:
            h_try = best_h * rng.uniform(0.7, 1.3)
        elif strategy == 3:
            h_try = np.where(rng.rand(n0) < 0.3, rng.rand(n0) * 0.3, best_h)
        else:
            h_try = np.minimum(1.0, np.maximum(0.0, best_h + 0.15 * rng.randn(n0) * rng.rand(n0)))
        h_try = project_feasible(h_try, n0)

        # SLSQP from perturbed start
        res = run_slq(h_try, n0, 300)
        if np.isfinite(res.fun) and res.fun < best_val:
            best_val = float(res.fun)
            best_h = np.clip(res.x, 0.0, 1.0)

        # Coordinate descent from perturbed start
        if elapsed() < budget_s - 120:
            h_cd = coord_descent_step(h_try, 200)
            h_cd = project_feasible(h_cd, n0)
            v = compute_c5(h_cd)
            if v < best_val:
                best_val = v
                best_h = h_cd.copy()

    # Upscale to 2x and refine
    if elapsed() < budget_s - 100:
        n1 = n0 * 2
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, n1)
        v = compute_c5(h_up)
        if v < best_val:
            best_val = v
            best_h = h_up.copy()

        if elapsed() < budget_s - 80:
            res = run_slq(h_up, n1, 400)
            if np.isfinite(res.fun) and res.fun < best_val:
                best_val = float(res.fun)
                best_h = np.clip(res.x, 0.0, 1.0)

        if elapsed() < budget_s - 80:
            for _ in range(15):
                if elapsed() > budget_s - 80:
                    break
                h_cd = coord_descent_step(best_h, 200)
                h_cd = project_feasible(h_cd, n1)
                v = compute_c5(h_cd)
                if v < best_val:
                    best_val = v
                    best_h = h_cd.copy()

        # A few restarts at 2x
        for r in range(5):
            if elapsed() > budget_s - 80:
                break
            rng = np.random.RandomState(seed + 500 + r)
            h_try = best_h + 0.04 * rng.randn(n1)
            h_try = project_feasible(h_try, n1)
            res = run_slq(h_try, n1, 200)
            if np.isfinite(res.fun) and res.fun < best_val:
                best_val = float(res.fun)
                best_h = np.clip(res.x, 0.0, 1.0)

    # Final polish with coordinate descent
    if elapsed() < budget_s - 20:
        n_final = len(best_h)
        for _ in range(20):
            if elapsed() > budget_s - 20:
                break
            h_cd = coord_descent_step(best_h, 200)
            h_cd = project_feasible(h_cd, n_final)
            v = compute_c5(h_cd)
            if v < best_val:
                best_val = v
                best_h = h_cd.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_val = compute_c5(best_h)
    # Verify constraint
    assert abs(np.sum(best_h) - n_final / 2.0) < 1e-8, "Integral constraint violated"
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15), "Box constraint violated"

    return best_h, best_val, n_final
