    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    global _n_global, _T_global, _penalty_global
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    best_h = np.asarray(initial_h_values, dtype=float).copy()
    best_n = len(best_h)
    best_c5 = compute_c5_internal(best_h)

    resolutions = [best_n, best_n * 2, best_n * 4, best_n * 8]
    resolutions = [r for r in resolutions if r <= 672]

    for target_n in resolutions:
        if elapsed() > budget_s * 0.85:
            break

        if target_n == best_n:
            h_curr = best_h.astype(float).copy()
        else:
            h_curr = interpolate_to(best_h, target_n)

        _n_global = target_n
        bounds = [(0.0, 1.0)] * target_n

        # Multi-restart SLSQP with increasing temperature
        temps = [3.0, 6.0, 12.0, 25.0, 50.0, 100.0, 200.0]
        for T in temps:
            if elapsed() > budget_s * 0.88:
                break
            _T_global = T
            _penalty_global = 1e4
            res = minimize(penalized_obj, h_curr, method='SLSQP',
                           bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-12, 'iprint': 0})
            h_curr = project_feasible(res.x)
            c5 = compute_c5_internal(h_curr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()
                best_n = target_n

            # Random restarts
            if target_n <= 84:
                for restart in range(3):
                    if elapsed() > budget_s * 0.88:
                        break
                    noise = np.random.uniform(-0.3, 0.3, target_n)
                    h_try = project_feasible(h_curr + noise)
                    _T_global = T
                    res2 = minimize(penalized_obj, h_try, method='SLSQP',
                                    bounds=bounds,
                                    options={'maxiter': 500, 'ftol': 1e-12, 'iprint': 0})
                    h_try = project_feasible(res2.x)
                    c5_try = compute_c5_internal(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        best_n = target_n
                        h_curr = h_try.copy()

        # Final refinement at current best with raw max via many small steps
        if elapsed() < budget_s * 0.92:
            h_curr = best_h.astype(float).copy()
            if target_n != best_n:
                h_curr = interpolate_to(best_h, best_n)
            _n_global = best_n
            _penalty_global = 1e6
            for T in [100.0, 300.0, 500.0]:
                if elapsed() > budget_s * 0.94:
                    break
                _T_global = T
                res = minimize(penalized_obj, h_curr, method='SLSQP',
                               bounds=[(0.0, 1.0)] * best_n,
                               options={'maxiter': 1000, 'ftol': 1e-14, 'iprint': 0})
                h_curr = project_feasible(res.x)
                c5 = compute_c5_internal(h_curr)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_curr.copy()

    # Final subgradient / coordinate descent polish
    if elapsed() < budget_s * 0.96:
        _n_global = len(best_h)
        h = best_h.astype(float).copy()
        for it in range(500):
            if elapsed() > budget_s * 0.97:
                break
            improved = False
            order = np.random.permutation(len(h))
            for i in order:
                if elapsed() > budget_s * 0.97:
                    break
                old_val = h[i]
                # Try small perturbations
                for delta in [0.05, -0.05, 0.02, -0.02, 0.1, -0.1, 0.01, -0.01]:
                    h[i] = np.clip(old_val + delta, 0.0, 1.0)
                    h = project_feasible(h)
                    c5 = compute_c5_internal(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] = old_val
            if not improved:
                # Try larger steps
                for _ in range(10):
                    if elapsed() > budget_s * 0.97:
                        break
                    idx = np.random.randint(len(h))
                    h[idx] = np.clip(h[idx] + np.random.uniform(-0.5, 0.5), 0.0, 1.0)
                    h = project_feasible(h)
                    c5 = compute_c5_internal(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h)
    final_c5 = compute_c5_internal(best_h)

    return (best_h, final_c5, len(best_h))
