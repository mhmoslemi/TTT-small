        c5 = (2.0 / n) * corr.max()
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 800:
            h = project_feasible(h + rng.normal(0, 0.15, n), n)
            alpha = 0.08
            stuck = 0
        idx = int(corr.argmax())
        lag = idx - (n - 1)
        g = grad_lag(h, n, lag)
        step = alpha * g
        h_new = project_feasible(h - step, n)
        c5_new = get_c5(h_new)
        if c5_new <= c5 + 1e-12:
            h = h_new
            alpha = min(alpha * 1.02, 0.3)
        else:
            alpha *= 0.9
            if alpha < 0.003:
                alpha = 0.08
    return best_h, best_c5

def make_binary_pattern(n, rng):
    h = np.zeros(n)
    ones_needed = n // 2
    half_needed = 1 if n % 2 == 1 else 0
    perm = rng.permutation(n)
    h[perm[:ones_needed]] = 1.0
    if half_needed:
        h[perm[ones_needed]] = 0.5
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = make_binary_pattern(48, rng)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = get_c5(h_init)

    # Phase 1: Optimize from initial at its n
    h1, c5_1 = local_optimize(h_init.copy(), n, t0 + budget_s * 0.35, rng)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Binary push and re-optimize
    if n % 2 == 0:
        hb = (h1 > 0.5).astype(float)
        hb = project_feasible(hb, n)
        c5_b = get_c5(hb)
        if c5_b < best_c5:
            best_h, best_c5 = hb, c5_b
        h2, c5_2 = local_optimize(hb.copy(), n, t0 + budget_s * 0.55, rng)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: Try other n values (even, allows binary)
    for n_try in [46, 48, 50, 52, 60, 80, 100, 150, 200]:
        if time.time() - t0 > budget_s - 60:
            break
        remaining = budget_s - (time.time() - t0)
        if remaining < 25:
            break
        sub_deadline = t0 + time.time() - t0 + min(remaining * 0.15, 45)
        for trial in range(3):
            if time.time() > sub_deadline:
                break
            h_rand = make_binary_pattern(n_try, rng)
            # Add small noise to break symmetry
            h_rand = project_feasible(h_rand + rng.normal(0, 0.02, n_try), n_try)
            h_opt, c5_opt = local_optimize(h_rand, n_try, sub_deadline, rng)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Final fine-tuning on best solution
    n_best = len(best_h)
    deadline_final = t0 + budget_s - 5
    if time.time() < deadline_final:
        h_f, c5_f = local_optimize(best_h.copy(), n_best, deadline_final, rng)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Verify
    best_h = project_feasible(best_h, n_best)
    best_c5 = get_c5(best_h)

    return (best_h, best_c5, len(best_h))
