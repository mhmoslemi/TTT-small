import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

class ObjSmooth:
    def __init__(self, T):
        self.T = T
    def __call__(self, h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr * dx, self.T)

class ConsSum:
    def __init__(self, n):
        self.target = n / 2.0
        self.n = n
    def __call__(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(self.n)

def make_constraint(n):
    c = ConsSum(n)
    return {'type': 'eq', 'fun': c, 'jac': c.jac}

def try_optimize(h0, n_pts, maxiter, T, state):
    bounds = [(0.0, 1.0)] * n_pts
    cons = make_constraint(n_pts)
    obj = ObjSmooth(T)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = project_feasible(res.x, n_pts)
        c5_new = compute_c5(h_new)
        if c5_new < state[0]:
            state[0] = c5_new
            state[1] = h_new.copy()
    except Exception:
        pass
    return state

def try_nm(h0, n_pts, state, maxiter=200):
    def obj_nm(x):
        return compute_c5(x)
    bounds = [(0.0, 1.0)] * n_pts
    try:
        res = minimize(obj_nm, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
        h_new = project_feasible(res.x, n_pts)
        c5_new = compute_c5(h_new)
        if c5_new < state[0]:
            state[0] = c5_new
            state[1] = h_new.copy()
    except Exception:
        pass
    return state

def interpolate_parent(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
