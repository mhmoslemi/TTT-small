import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def hess_lag(h, m, N):
    """Gradient of the correlation at lag m (m = i - (N-1)) w.r.t. h."""
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += (1.0 - h[:N - m])
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize_h(h, N, target, dx, t0, budget_s, rng, lr_init=0.03):
    """Optimize h using Adam with smooth-max subgradient."""
    best_local = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    tau = 8.0

    for it in range(5000):
        if time.time() - t0 > budget_s - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local:
            best_local = c5_val
            best_h = h.copy()

        # Softmax weights for smooth max
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()

        # Weighted gradient
        thresh = 0.005 / N
        grad = np.zeros(N)
        active = np.where(w > thresh)[0]
        for idx in active:
            lag = int(idx) - (N - 1)
            wi = w[idx]
            g = hess_lag(h, lag, N)
            grad += wi * g

        # Adam update

        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))

            tau *= 1.8
