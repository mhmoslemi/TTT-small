    return res

import numpy as np
import scipy.optimize as opt
import time, math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(vals, alpha):
    m = np.max(vals)
    shifted = alpha * (vals - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def obj_lse(h, n, alpha):
    return smooth_max(c5_values(h), alpha)

def obj_topk(h, n, k):
    vals = c5_values(h)
    idx = np.argpartition(vals, -k)[-k:]
    return np.sum(vals[idx])

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        h = np.clip(h - diff / n, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(free) > 0:
            h[free] -= diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_run(x0, n, obj_fn, max_iter=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = opt.minimize(obj_fn, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        return res

        return None
