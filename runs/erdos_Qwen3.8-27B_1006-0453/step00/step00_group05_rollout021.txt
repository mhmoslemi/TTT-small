    corr = compute_corr(h, n)
    max_c = np.max(corr)
    weights = np.exp(T * (corr - max_c))
    weights /= np.sum(weights)

        w = weights[m]
        if w < 1e-12:
            continue
        s = m - (n-1)  # lag
        # contribution to grad: w * dx * ((1-h[shifted]) - h[other_shift])
        # For valid j: (1-h[j-s]) - h[j+s]
        # j ranges such that 0<=j<n
        # (1-h[j-s]) is valid when 0<=j-s<n, i.e., s<=j<n+s
        # h[j+s] is valid when 0<=j+s<n, i.e., -s<=j<n-s
        
        lo_comp = max(0, s)  # j-s >= 0 => j >= s (for s>0), or j >= 0

        lo_h = max(0, -s)  # j+s >= 0 => j >= -s (for s<0)
        hi_h = min(n, n - s)  # j+s < n => j < n-s (for s>0)

        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if hi1 > lo1:
