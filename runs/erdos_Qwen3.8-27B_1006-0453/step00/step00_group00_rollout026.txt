    return float(np.max(corr)) * dx

def subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    active = np.where(corr >= mv - 1e-8)[0]
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        sp, ep = max(0, -lag), min(n, n - lag)
        if ep > sp:
            g[sp:ep] += 1.0 - h[sp+lag:ep+lag]
        sm, em = max(0, lag), min(n, n + lag)
        if em > sm:
            g[sm:em] -= h[sm-lag:em-lag]
    return g * dx / len(active)

    h = np.clip(h, 0.0, 1.0)

                  make_binary(n, seed + trial * 13 + 37),
                  make_struc(n, seed + trial * 13 + 71)]
        if best_h is not None and len(best_h) != n:
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n)
            starts.append(project(np.interp(xn, xo, best_h), n))

        for s in starts:
            if time.time() - t_start > budget_s - 15:
                break
            h2, c2 = grad_local(s, tlim=min(budget_per * 0.6, 12.0))
            if c2 < best_c5:
                best_h, best_c5 = h2, c2
            h3, c3 = swap_local(h2, tlim=min(budget_per * 0.3, 6.0))
            if c3 < best_c5:
                best_h, best_c5 = h3, c3
        trial += 1

    # Extended search on best n if time remains
    if time.time() - t_start < budget_s - 30 and best_h is not None:
        n_best = len(best_h)
        for t in range(3):
            if time.time() - t_start > budget_s - 10:
                break
            pert = best_h + 0.15 * (np.random.rand(n_best) - 0.5)
            pert = project(pert, n_best)
            h4, c4 = grad_local(pert, tlim=15.0)
            if c4 < best_c5:
                best_h, best_c5 = h4, c4
            h5, c5 = swap_local(h4, tlim=8.0)
            if c5 < best_c5:
                best_h, best_c5 = h5, c5
        # Final polish
        if time.time() - t_start < budget_s - 5:
            h6, c6 = grad_local(best_h, tlim=budget_s - (time.time() - t_start) - 3)
            if c6 < best_c5:
                best_h, best_c5 = h6, c5 if False else c6
                best_h, best_c5 = h6, c6
            h7, c7 = swap_local(best_h, tlim=max(2.0, budget_s - (time.time() - t_start) - 2))
            if c7 < best_c5:
                best_h, best_c5 = h7, c7

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
