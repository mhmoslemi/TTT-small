import numpy as np
import time

def _project_box_sum(h, target):
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return h

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def _sobj_grad(h, n, T):
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode="full") * dx
    TL = T * (lags - lags.max())
    EL = np.exp(TL)
    SE = EL.sum()
    obj = lags.max() + np.log(SE) / T
    w = EL / SE
    t1 = np.zeros(n)
    t2 = np.zeros(n)
    t3 = np.zeros(n)
    for p in range(2 * n - 1):
        k = p - (n - 1)
        wp = w[p]
        if wp < 1e-16:
            continue
        a, b = max(0, -k), min(n, n - k)
        if b > a:
            t2[a:b] += wp * h[a + k:b + k]
            t1[a:b] += wp
        a3, b3 = max(0, k), min(n, n + k)
        if b3 > a3:
            t3[a3:b3] += wp * h[a3 - k:b3 - k]
    return obj, T * dx * (t1 - t2 - t3)
