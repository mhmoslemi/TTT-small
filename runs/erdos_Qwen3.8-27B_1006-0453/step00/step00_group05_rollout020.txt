    t_start = time.time()
    stuck_count = 0
    while time.time() - t_start < max_time:
        improved = False
        order = rng.permutation(len(ones))
        for idx in order:
            if time.time() - t_start > max_time:
                break
            i = ones[idx]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c = int(np.max(np.correlate(h, 1.0 - h, mode="full")))
            if c < best:
                best = c
                ones[idx] = j
                zeros[zeros.index(j)] = i
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            stuck_count += 1
            if stuck_count > 5:
                # Perturb: random swap
                i = int(rng.integers(len(ones)))
                j = int(rng.integers(len(zeros)))
                h[ones[i]], h[zeros[j]] = 0.0, 1.0
                ones[i], zeros[j] = zeros[j], ones[i]
                best = int(np.max(np.correlate(h, 1.0 - h, mode="full")))
                stuck_count = 0
    return h, best * dx

def continuous_refine(h, n, rng, max_time=60.0):
    h = h.astype(np.float64).copy()
    target = n / 2.0
    h = project_sum_box(h, target)
    best = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    alpha = 0.005
    for it in range(20000):
        if time.time() - t_start > max_time:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 0.005)[0]
        grad = np.zeros(n)
        for m in active:
            shift = m - (n - 1)
            j1 = np.arange(n) + shift
            mask1 = (j1 >= 0) & (j1 < n)
            grad[mask1] += (1.0 - h[j1[mask1]])
            j2 = np.arange(n) - shift
            mask2 = (j2 >= 0) & (j2 < n)
            grad[mask2] += -h[j2[mask2]]
        grad /= len(active)
        grad *= dx
        h_new = project_sum_box(h - alpha * grad + 0.001 * rng.standard_normal(n), target)
        c_new = compute_c5(h_new)
        if c_new < best:
            h = h_new
            best = c_new
            best_h = h.copy()
            alpha = min(alpha * 1.1, 0.1)
        else:
            alpha *= 0.8
    return best_h, best

def upsample_binary(h_old, n_new, rng):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_cont = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]))
    h_bin = (h_cont > 0.5).astype(np.float64)
    target = n_new // 2
    s = int(h_bin.sum())
    if s > target:
        idx_1 = list(np.where(h_bin > 0.5)[0])
        for _ in range(s - target):
            k = int(rng.integers(len(idx_1)))
            h_bin[idx_1[k]] = 0.0
            idx_1.pop(k)
    elif s < target:
        idx_0 = list(np.where(h_bin <= 0.5)[0])
        for _ in range(target - s):
            k = int(rng.integers(len(idx_0)))
            h_bin[idx_0[k]] = 1.0
            idx_0.pop(k)
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h0 = None

    best_h = None
    best_c5 = 1.0

    if h0 is not None and len(h0) > 0:
        h0_bin = (h0 > 0.5).astype(np.float64)
        c5 = compute_c5(h0_bin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0_bin.copy()

    # Phase 1: Binary local search at original resolution
    n0 = len(h0) if h0 is not None else 70
    h_start = h0_bin.copy() if h0 is not None else np.zeros(n0)
    if h0 is None:
        h_start[:n0 // 2] = 1.0

    t1 = time.time()
    alloc1 = min(budget_s * 0.25, 300.0)
    h_b, c5_b = binary_local_search(h_start, rng, max_time=alloc1)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Phase 2: Multi-resolution upsampling + search
    resolutions = [140, 200, 280, 400]
    t2 = time.time()
    for n_new in resolutions:
        if time.time() - t0 > budget_s * 0.75:
            break
        remaining = budget_s * 0.75 - (time.time() - t0)
        if remaining < 20:
            break
        h_up = upsample_binary(best_h, n_new, rng)
        t_mid = time.time()
        alloc = min(remaining * 0.3, 180.0)
        h_new, c5_new = binary_local_search(h_up, rng, max_time=alloc)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Continuous refinement on the best binary
        if time.time() - t0 < budget_s * 0.7:
            h_cont, c5_cont = continuous_refine(best_h, len(best_h), rng, max_time=30.0)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                # Keep the continuous version
                best_h = h_cont.copy()
            else:
                best_h = h_new.copy()

    # Phase 3: Multiple random restarts at best resolution
    n_best = len(best_h)
    t3 = time.time()
    restart_time = 20.0
    n_restarts = int(min((budget_s * 0.85 - (time.time() - t0)) / restart_time, 20))
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.85:
            break
        # Generate random binary with correct sum
        h_rand = np.zeros(n_best)
        perm = rng.permutation(n_best)
        h_rand[perm[:n_best // 2]] = 1.0
        h_rand, c5_rand = binary_local_search(h_rand, rng, max_time=restart_time)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()

    # Final: ensure valid
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # If not exactly binary, round but keep sum
    if np.max(np.abs(best_h - np.round(best_h))) < 0.01:
        best_h = np.round(best_h)
    else:
        # Keep as continuous, just project
        best_h = project_sum_box(best_h, n_final / 2.0)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, int(n_final))
