    h_sorted = np.argsort(h)

def anneal(h, n, rng, t_start, t_end, T0, T1):
    """Simulated annealing for binary h."""
    h = h.copy()
    cur_c5 = compute_c5(h, n)
    best_c5 = cur_c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()

        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] -= eps
            h[j] += eps
            h = np.clip(h, 0.0, 1.0)
            deficit = target_sum - h.sum()
            if abs(deficit) > 1e-10:
                h += deficit / n
                h = np.clip(h, 0.0, 1.0)
        step = max(step * 0.9999, 0.001)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5.0

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0, n0)
    best_n = n0

    # Binary at n0
    h_bin = to_binary(h0)
    c5_bin = compute_c5(h_bin, n0)
    if c5_bin < best_c5:
        best_c5, best_h, best_n = c5_bin, h_bin.copy(), n0

    # Phase 1: Binary search on n0 (40% budget)
    t1 = min(t0 + budget_s * 0.4, t_end)
    h1, c5_1 = local_search_binary(h_bin, n0, rng, t1)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1.copy(), n0

    # Phase 2: Continuous refinement on n0 (15% budget)
    if time.time() < t_end:
        t2 = min(time.time() + budget_s * 0.15, t_end)
        h2, c5_2 = continuous_refine(best_h, best_n, rng, t2)
        if c5_2 < best_c5:
            best_c5, best_h, best_n = c5_2, h2.copy(), best_n

    # Phase 3: Try n=140 (25% budget)
    n2 = n0 * 2
    if time.time() < t_end:
        h2_init = np.zeros(n2)
        h2_init[0::2] = best_h
        h2_init[1::2] = best_h
        h2_bin = to_binary(h2_init)
        c5_2b = compute_c5(h2_bin, n2)
        t3 = min(time.time() + budget_s * 0.25, t_end)
        h3, c5_3 = local_search_binary(h2_bin, n2, rng, t3)
        if c5_3 < best_c5:
            best_c5, best_h, best_n = c5_3, h3.copy(), n2

    # Phase 4: Random restarts on n0 if time remains
    t_remaining = t_end - time.time()
    if t_remaining > 30:
        n_restart = best_n
        n_trials = int(min(t_remaining / 15, 20))
        for trial in range(n_trials):
            if time.time() >= t_end - 5:
                break
            if trial == 0:
                hr = to_binary(best_h[:n0]).copy() if best_n == n0 else to_binary(np.interp(np.linspace(0, 2, n0), np.linspace(0, 2, best_n), best_h))
            else:
                hr = np.zeros(n0)
                hr[rng.choice(n0, n0 // 2, replace=False)] = 1.0
            t_sub = min(time.time() + 12, t_end - 10)
            hr2, c5r = local_search_binary(hr, n0, rng, t_sub)
            if c5r < best_c5:
                best_c5, best_h = c5r, hr2.copy()

    # Phase 5: Final continuous tweak if time remains
    if time.time() < t_end - 3 and best_n == n0:
        t5 = min(time.time() + 30, t_end)
        h_fin, c5_fin = continuous_refine(best_h, best_n, rng, t5)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin.copy()

    # Final validation
    n_out = len(best_h)
    dx_out = 2.0 / n_out
    assert abs(best_h.sum() * dx_out - 1.0) < 0.01, f"Sum constraint violated: {best_h.sum()*dx_out}"
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, n_out)

    return list(best_h), final_c5, n_out
