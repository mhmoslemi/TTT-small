    N = len(h)
    dx = 2.0 / N

    return np.max(corr) * dx

def grad_shift(h, shift):
    N = len(h)
    g = np.zeros(N)
    if shift >= 0:
        g[shift:] += 1.0
        g[shift:] -= h[:N - shift]
        g[:N - shift] -= h[shift:]
    else:
        s = -shift
        g[:N - s] += 1.0
        g[:N - s] -= h[s:]
        g[s:] -= h[:N - s]
    return g

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    shift = k_max - (N - 1)
    return grad_shift(h, shift) * dx

def smooth_grad_c5(h, beta=50.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    vmax = vals.max()
    exps = np.exp(beta * (vals - vmax))
    w = exps / exps.sum()

            shift = k - (N - 1)
            g += w[k] * grad_shift(h, shift)

    return g * dx, c5_smooth

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def grad_shift(h, s):
    N = len(h)
    g = np.zeros(N)
    if s >= 0:
        g[s:] += 1.0
        g[s:] -= h[:N - s]
        g[:N - s] -= h[s:]
    else:
        t = -s
        g[:N - t] += 1.0
        g[:N - t] -= h[t:]
        g[t:] -= h[:N - t]
    return g


def smooth_grad(h, beta):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    vmax = vals.max()
    ex = np.exp(beta * (vals - vmax))
    w = ex / ex.sum()
    g = np.zeros(N)
    for k in range(len(corr)):
        if w[k] > 1e-7:
            g += w[k] * grad_shift(h, k - (N - 1))
    cv = float(vmax + np.log(ex.sum()) / beta)
    return g * dx, cv


def project(h, target):
    lo = float(h.min()) - float(target) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def optimize_smooth(h0, time_limit, beta_hi=100.0, beta_lo=8.0, max_iter=50000):
    N = len(h0)
    target = N / 2.0
    h = np.array(h0, dtype=float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    step = 0.3
    mom = 0.0
    vg = np.zeros(N)
    n_iter = 0

    while n_iter < max_iter and (time.time() - t_start) < time_limit:
        prog = min(1.0, n_iter / 3000.0)
        beta = beta_hi * (beta_lo / beta_hi) ** prog
        g, _ = smooth_grad(h, beta)
        vg = 0.7 * vg + 0.3 * g
        h_new = h - step * vg
        h_new = project(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        c5_cur = compute_c5(h)
        if c5_new <= c5_cur + 1e-12:
            h = h_new
            step = min(step * 1.03, 2.0)
        else:
            step *= 0.6
            if step < 1e-7:
                break
        n_iter += 1
    return best_h, best_c5, n_iter


def optimize_exact(h0, time_limit, max_iter=20000):
    N = len(h0)
    target = N / 2.0
    h = np.array(h0, dtype=float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    step = 0.1
    n_iter = 0
    while n_iter < max_iter and (time.time() - t_start) < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        s = k_max - (N - 1)
        g = grad_shift(h, s) * (2.0 / N)
        h_new = h - step * g
        h_new = project(h_new, target)

        n_iter += 1

def interp_up(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def make_initial(n, rng):
    h = np.full(n, 0.5)
    freqs = rng.uniform(1.0, 6.0, size=min(n - 1, 20))
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    amps = rng.uniform(0.05, 0.25, size=len(freqs))
    for i in range(n):
        x = 2.0 * i / n
        v = 0.5
        for j in range(len(freqs)):
            v += amps[j] * np.sin(2 * np.pi * freqs[j] * x / 2.0 + phases[j])
        h[i] = v
    return h
