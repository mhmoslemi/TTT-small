    n = len(x)

import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if T < 1.0:
        return m
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    return m + (1.0 / T) * np.log(np.sum(np.exp(shifted)))

def eq_constraint(x, n):
    return np.sum(x) - n * 0.5

def contract_constraints(x, n):
    """Project x to satisfy 0<=x<=1 and sum x = n/2."""
    x = np.clip(x, 0.0, 1.0)
    diff = n * 0.5 - np.sum(x)
    if diff > 0:
        # Need to increase sum
        order = np.argsort(-x)  # largest first
        for k in range(n):
            add = min(1.0 - x[order[k]], diff)
            x[order[k]] += add
            diff -= add
            if diff <= 0:
                break
    elif diff < 0:
        order = np.argsort(x)  # smallest first
        for k in range(n):
            sub = min(x[order[k]], -diff)
            x[order[k]] -= sub
            diff += sub
            if diff >= 0:
                break
    return x

def greedy_construct(n, rng, n_swap_passes=10):
    """Greedy construction of binary sequence."""
    h = np.random.permutation(np.concatenate([np.ones(n//2), np.zeros(n - n//2)]))
    current_c = compute_c5(h)
    
    for _ in range(n_swap_passes):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                new_c = compute_c5(h)
                if new_c < current_c:
                    current_c = new_c
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, current_c

    best_h = h.copy()
    best_c = current_c

        T = T_init * (T_final / T_init) ** frac

        if not ones_mask.any() or not (~ones_mask).any():

                best_h = h.copy()
        else:

import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if T < 1.0:
        return m
    shifted = np.clip(T * (corr - m), -200, 200)
    return m + (1.0 / T) * np.log(np.sum(np.exp(shifted)))

def eq_constraint(x, n):
    return np.sum(x) - n * 0.5

def contract_constraints(x, n):
    x = np.clip(x, 0.0, 1.0).astype(float)
    diff = n * 0.5 - np.sum(x)
    if diff > 1e-12:
        order = np.argsort(-x)
        for k in range(n):
            add = min(1.0 - x[order[k]], diff)
            x[order[k]] += add
            diff -= add
            if diff <= 0:
                break
    elif diff < -1e-12:
        order = np.argsort(x)
        for k in range(n):
            sub = min(x[order[k]], -diff)
            x[order[k]] -= sub
            diff += sub
            if diff >= 0:
                break
    return x

def sa_binary(h, n, n_iter, rng, T_init, T_final):
    h = h.copy().astype(float)
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_init * (T_final / T_init) ** frac
        ones_mask = h > 0.5
        if not ones_mask.any() or not (~ones_mask).any():
            break
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(~ones_mask)[0]
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        delta = new_c - current_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def hill_climb_binary(h, n, max_sweeps=20):
    h = h.copy().astype(float)
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    for _ in range(max_sweeps):
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        best_delta = 0.0
        best_pair = None
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h)
                h[i], h[j] = h[j], h[i]
                if c < current_c - 1e-15:
                    best_delta = current_c - c
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        current_c = compute_c5(h)
        improved = True
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[:n // 2] = 1.0
    h = h[idx]
    return h
