import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = float(n) / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def obj_func(h, n, target, pen_weight):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * (2.0 / n)
    s = np.sum(h) - target
    return c5 + pen_weight * s * s


def run(seed=42, budget_s=1100, **kwargs):
    import time
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    best_h = None
    best_score = np.inf

    def update(h, sc):
        nonlocal best_h, best_score
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
        parent = make_feasible(parent)
        update(parent, compute_c5(parent))

    n_list = [45, 90, 135, 200, 300]

    for n_pts in n_list:
        if elapsed() > budget_s - 90:
            break
        n = n_pts
        target = float(n) / 2.0
        bounds = [(0.0, 1.0)] * n
        pen = 500.0

        starts = []
        if parent is not None:
            if n_pts == 45:
                starts.append(parent.copy())
            else:
                xp = np.linspace(0.0, 2.0, 45)
                xn = np.linspace(0.0, 2.0, n)
                h_up = interp1d(xp, parent, kind='linear', bounds_error=False,
                                fill_value='extrapolate')(xn)
                starts.append(make_feasible(h_up))

        n_random = 4 if n <= 90 else 2
        for _ in range(n_random):
            h_r = (rng.rand(n) > 0.5).astype(float)
            s_r = float(np.sum(h_r))
            diff = target - s_r
            idx = np.where((diff > 0) == (h_r < 1.0))[0]
            if len(idx) > abs(diff):
                order = idx[:int(abs(diff))]
                h_r[order] += np.sign(diff)
            h_r[idx[int(abs(diff))]] += diff - np.sign(diff) * int(abs(diff))
            starts.append(make_feasible(h_r))

        for h0 in starts:
            if elapsed() > budget_s - 50:
                break
            rem = budget_s - 50 - elapsed()
            maxfev = int(min(2000, rem * 60))
            if maxfev < 40:
                continue
            try:
                res = minimize(obj_func, make_feasible(h0), args=(n, target, pen),
                               method='COBYLA', bounds=bounds,
                               options={'maxfev': maxfev, 'rhobeg': 0.02})
                if res.x is not None and len(res.x) == n:
                    h_c = make_feasible(res.x)
                    sc = compute_c5(h_c)
                    update(h_c, sc)
            except Exception:
                pass

        # Nelder-Mead refinement on current best if same n
        if best_h is not None and len(best_h) == n and elapsed() < budget_s - 30:
            for _ in range(2):
                if elapsed() > budget_s - 25:
                    break
                h0 = make_feasible(best_h + rng.randn(n) * 0.005)
                try:
                    res = minimize(obj_func, h0, args=(n, target, 1e6),
                                   method='Nelder-Mead',
                                   options={'maxiter': 8000, 'xatol': 1e-11, 'fatol': 1e-13})
                    if res.x is not None:
                        h_c = make_feasible(res.x)
                        sc = compute_c5(h_c)
                        update(h_c, sc)
                except Exception:
                    pass

    # Final global refinement on best solution found
    if best_h is not None and elapsed() < budget_s - 15:
        n = len(best_h)
        target = float(n) / 2.0
        for attempt in range(6):
            if elapsed() > budget_s - 10:
                break
            noise = 0.015 * (attempt + 1) ** 0.5
            h0 = make_feasible(best_h + rng.randn(n) * noise)
            pen = 1e5
            try:
                res = minimize(obj_func, h0, args=(n, target, pen),
                               method='Nelder-Mead',
                               options={'maxiter': 15000, 'xatol': 1e-13, 'fatol': 1e-15})
                if res.x is not None:
                    h_c = make_feasible(res.x)
                    sc = compute_c5(h_c)
                    update(h_c, sc)
            except Exception:
                pass

    # Additional polish: small COBYLA on best
    if best_h is not None and elapsed() < budget_s - 5:
        n = len(best_h)
        target = float(n) / 2.0
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_func, make_feasible(best_h), args=(n, target, 1e4),
                           method='COBYLA', bounds=bounds,
                           options={'maxfev': 300, 'rhobeg': 0.005})
            if res.x is not None:
                h_c = make_feasible(res.x)
                sc = compute_c5(h_c)
                update(h_c, sc)
        except Exception:
            pass

    if best_h is not None:
        h_final = make_feasible(best_h)
        c5 = compute_c5(h_final)
        return h_final, c5, len(h_final)

    if parent is not None:
        return parent, compute_c5(parent), len(parent)

    h = np.full(45, 0.5)
    return h, compute_c5(h), 45
