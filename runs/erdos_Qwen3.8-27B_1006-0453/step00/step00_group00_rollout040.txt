def compute_c5_and_grad(h, n):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(g)
    s = max_idx - (n - 1)
    
    # Gradient of the active term
    grad = np.zeros(n)
    imin = max(0, -s)
    imax = min(n-1, n-1-s)

        k = i + s
        grad[k] += (1 - h[i])
        grad[i] += -h[k]

def project_feasible(h, n):

    for _ in range(200):
        h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break

        free = (h > 0) & (h < 1)
        if not np.any(free):
            # Degenerate: all saturated or zero
            # Adjust the closest to boundary
            if diff > 0:

                h[candidates[np.argmax(h[candidates])]

                h[candidates[np.argmin(h[candidates])]

            continue

import numpy as np
import time

def _eval(h, n):
    """Compute C5 value."""
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full')
    return (np.max(g) * dx)

def _project(h, n):
    """Project onto {x: sum(x)=n/2, 0<=x<=1}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # Adjust non-saturated coordinates
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.sum(free) == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def _pair_optimize(h, n, t0, deadline):
    """Do one sweep of pairwise optimization."""
    target = n / 2.0
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                if time.time() > deadline:
                    return h
                # Try transferring mass between i and j
                hi, hj = h[i], h[j]
                if hi < 1e-10 and hj < 1e-10:
                    continue
                if hi > 1-1e-10 and hj > 1-1e-10:
                    continue
                # Try several t values
                best_t = 0.0
                best_c5 = _eval(h, n)
                candidates = []
                if hi > 1e-10:
                    candidates.append(-hi)  # empty i
                    candidates.append(-hi/2)
                if hj > 1e-10:
                    candidates.append(hj)  # empty j
                    candidates.append(hj/2)
                if 1-hi > 1e-10:
                    candidates.append(-(1-hi))  # fill i
                    candidates.append(-(1-hi)/2)
                if 1-hj > 1e-10:
                    candidates.append(1-hj)  # fill j
                    candidates.append((1-hj)/2)
                
                for t in candidates:
                    if t == 0:
                        continue
                    ni, nj = hi + t, hj - t
                    if ni < -1e-10 or ni > 1+1e-10 or nj < -1e-10 or nj > 1+1e-10:
                        continue
                    h[i], h[j] = max(0,ni), min(1,nj)
                    c5 = _eval(h, n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_t = t
                    h[i], h[j] = hi, hj
                
                if best_t != 0:
                    h[i] = np.clip(hi + best_t, 0, 1)
                    h[j] = np.clip(hj - best_t, 0, 1)
                    improved = True
    return h
