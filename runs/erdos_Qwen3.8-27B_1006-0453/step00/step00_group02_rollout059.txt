def smooth_c5_and_grad(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # Scale to objective values
    vals = r * dx
    # Log-sum-exp for numerical stability
    v_max = np.max(vals)
    log_sum = v_max + np.log(np.sum(np.exp(T * (vals - v_max))))
    f = log_sum / T
    
    # Gradient: p_k = exp(T*(v_k - v_max)) / sum_j exp(T*(v_j - v_max))
    p = np.exp(T * (vals - v_max))
    p /= np.sum(p)
    
    # df/dh[j] = sum_k p_k * dx * dr_k/dh[j]
    # dr_k/dh[j] = (1-h[j+m_k])*[valid] - h[j-m_k]*[valid] where m_k = k-(n-1)
    grad = np.zeros(n)
    for k in range(len(r)):
        if p[k] < 1e-10:
            continue
        m = k - (n - 1)
        j1_lo = max(0, -m)
        j1_hi = min(n, n - m)
        if j1_lo < j1_hi:
            idx = np.arange(j1_lo, j1_hi)
            grad[idx] += p[k] * (1.0 - h[idx + m])
        j2_lo = max(0, m)
        j2_hi = min(n, n + m)
        if j2_lo < j2_hi:
            idx = np.arange(j2_lo, j2_hi)
            grad[idx] -= p[k] * h[idx - m]
    
    grad *= dx
    return f, grad

import numpy as np
import time
from scipy.optimize import minimize


def project_constraint(g, target_sum):
    n = len(g)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -np.max(g) - 0.5
    hi = 1.0 - np.min(g) + 0.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(g + (lo + hi) / 2.0, 0.0, 1.0)


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(r) * dx)


def smooth_obj_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    hc = np.clip(h, 0.0, 1.0)
    r = np.correlate(hc, 1.0 - hc, mode='full')
    vals = r * dx
    v_max = np.max(vals)
    shifted = T * (vals - v_max)
    log_sum = v_max + np.log(np.sum(np.exp(shifted))) / T
    f = log_sum
    p = np.exp(shifted)
    p /= np.sum(p)
    grad = np.zeros(n)
    nf = len(r)
    for k in range(nf):
        pk = p[k]
        if pk < 1e-13:
            continue
        m = k - (n - 1)
        j1_lo = max(0, -m)
        j1_hi = min(n, n - m)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += pk * (1.0 - hc[j1_lo + m:j1_hi + m])
        j2_lo = max(0, m)
        j2_hi = min(n, n + m)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= pk * hc[j2_lo - m:j2_hi - m]
    grad *= dx
    sum_dev = np.sum(hc) - target_sum
    f += penalty * sum_dev * sum_dev
    grad += 2.0 * penalty * sum_dev
    return f, grad


def optimize_lBFGSB(h0, max_time, T=40.0):
    n = len(h0)
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n

    def fun(h):
        if time.time() - t_start > max_time:
            raise TimeoutError()
        f, _ = smooth_obj_grad(h, T, 1e4)
        return f

    def jac(h):
        if time.time() - t_start > max_time:
            raise TimeoutError()
        _, g = smooth_obj_grad(h, T, 1e4)
        return g

    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 400, 'maxfun': 800,
                                'ftol': 1e-14, 'gtol': 1e-10})
        h_out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_out = h0
    return project_constraint(h_out, n / 2.0)


def project_subgrad(h0, n_iters, lr0, max_time):
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    t_start = time.time()
    h = project_constraint(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = c5_value(h)
    for t in range(1, n_iters + 1):
        if time.time() - t_start > max_time:
            break
        r = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(r))
        m = k_star - (n - 1)
        grad = np.zeros(n)
        j1lo, j1hi = max(0, -m), min(n, n - m)
        if j1lo < j1hi:
            grad[j1lo:j1hi] += (1.0 - h[j1lo + m:j1hi + m])
        j2lo, j2hi = max(0, m), min(n, n + m)
        if j2lo < j2hi:
            grad[j2lo:j2hi] -= h[j2lo - m:j2hi - m]
        lr = lr0 / (1.0 + 0.03 * np.sqrt(t))
        h_new = project_constraint(h - lr * dx * grad, target_sum)
        c5_new = c5_value(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def upsample_linear(h, new_n):
    n = len(h)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    return np.interp(x_new, x_old, h)


def try_binary_projection(h, target_sum):
    """Try snapping values to 0/1 and see if it improves."""
    n = len(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Adjust count to match target
    nums = np.sum(h_bin)
    if nums > target_sum:
        order = np.argsort(h_bin - h_bin)
        idx = np.argsort(h)[:int(nums - target_sum)]
        h_bin[idx] = 0.0
    elif nums < target_sum:
        idx = np.argsort(h)[::-1][:int(target_sum - nums)]
        h_bin[idx] = 1.0
    h_bin = clip_sum_project(h_bin, target_sum)
    return h_bin

    return project_constraint(h, target_sum)
