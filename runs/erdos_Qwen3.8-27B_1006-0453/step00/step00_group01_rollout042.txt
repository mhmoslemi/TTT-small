import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    """Project h onto {0≤x≤1, sum(x)=n/2}"""
    target = n / 2.0
    # Find lambda such that sum(clamp(h + lambda, 0, 1)) = n/2
    def f(lam):

    lam = optimize.brentq(f, -2, 2)

def _gradient_descent(h, best_c5, best_h, alpha, max_iters, budget_s, t0, min_time=60.0):
    n = len(h)
    f_h = compute_c5(h)
    lr = 2.0 / (alpha + 1.0)
    momentum = np.zeros(n)
    beta_mom = 0.6 if alpha < 50 else 0.8
    for it in range(max_iters):
        if time.time() - t0 > budget_s - min_time:
            break
        grad = smooth_max_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        momentum = beta_mom * momentum - lr * grad
        h_try = project_feasible(h + momentum)
        f_try = compute_c5(h_try)
        accepted = False
        if f_try < f_h:
            h = h_try
            f_h = f_try
            accepted = True
        else:
            # Try without momentum
            h_try2 = project_feasible(h - lr * grad)
            f_try2 = compute_c5(h_try2)
            if f_try2 < f_h:
                h = h_try2
                f_h = f_try2
                momentum = -lr * grad
                accepted = True
            else:
                lr *= 0.5
                if lr < 1e-6:
                    break
        if f_h < best_c5[0]:
            best_c5[0] = f_h
            best_h[:] = h
        if it > 0 and it % 5000 == 0:
            lr *= 1.05
    return best_c5[0], best_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    n = 47
    h0 = None
    try:
        iv = globals().get('initial_h_values', None)
        if iv is not None:
            h0 = np.array(iv, dtype=float).flatten()
            n = len(h0)
    except Exception:
        pass

    if h0 is None:
        h0 = 0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(n) / 7.0)
    h0 = project_feasible(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth gradient descent with increasing alpha
    for alpha in [5.0, 15.0, 50.0, 150.0, 500.0, 2000.0]:
        if time.time() - t0 > budget_s - 80:
            break
        h = best_h.copy()
        bc5 = best_c5
        bh = best_h.copy()
        max_it = 50000 if alpha < 50 else 20000
        best_c5, best_h = _gradient_descent(h, best_c5, best_h, alpha, max_it, budget_s, t0)

    # Phase 2: Small perturbations and short restarts
    for _ in range(10):
        if time.time() - t0 > budget_s - 40:
            break
        h_p = best_h + rng.normal(0.0, 0.03, n)
        h_p = project_feasible(h_p)
        for alpha_r in [200.0, 1000.0]:
            h_p2 = h_p.copy()
            lr = 0.005
            for it in range(5000):
                if time.time() - t0 > budget_s - 40:
                    break
                grad = smooth_max_grad(h_p2, alpha_r)
                h_new = project_feasible(h_p2 - lr * grad)
                f_new = compute_c5(h_new)
                if f_new < compute_c5(h_p2):
                    h_p2 = h_new
                else:
                    lr *= 0.7
                if f_new < best_c5:
                    best_c5 = f_new
                    best_h = h_p2.copy()
            h_p = h_p2

    # Phase 3: SLSQP polish
    if time.time() - t0 < budget_s - 20:
        try:
            res = optimize.minimize(compute_c5, best_h, method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}],
                options={'maxiter': 3000, 'ftol': 1e-14, 'iprint': 0})
            if compute_c5(res.x) < best_c5:
                best_h = project_feasible(res.x)
                best_c5 = compute_c5(best_h)
        except Exception:
            pass

    # Final feasibility check
    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
