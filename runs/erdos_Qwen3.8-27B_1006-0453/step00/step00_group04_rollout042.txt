def compute_gradient(h, weights):
    """
    Compute gradient of sum_k weights[k] * correlate(h, 1-h, 'full')[k] w.r.t. h.
    weights has length 2n-1.
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n - 1):

        w = weights[k]
        if w == 0:

        # d/dc: for each valid i, d/dh[j] contributes
        # h[i+s] term: d/dh[j] = 1 if j = i+s, contributes (1-h[i])
        # (1-h[i]) term: d/dh[j] = -1 if j = i, contributes -h[i+s]

                j_pos = i + s  # where h appears as first factor
                j_neg = i      # where h appears as second factor (1-h)
                grad[j_pos] += w * (1 - h[i])
                grad[j_neg] -= w * h[i + s]

                j_pos = i + s
                j_neg = i
                grad[j_pos] += w * (1 - h[i])
                grad[j_neg] -= w * h[i - s]

import numpy as np
from scipy.optimize import minimize
import time

def c5_from_correlation(c, dx):
    return float(np.max(c) * dx)

def c5_score(h, dx):
    c = np.correlate(h, 1 - h, mode="full")
    return float(np.max(c) * dx)

def fix_sum(h, target_sum):
    """Project h to have sum = target_sum, staying in [0,1]."""
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Proportional adjustment
    if diff > 0:
        # Need to increase sum; spread increase to entries not at 1
        mask = h < 1 - 1e-10
        if mask.sum() == 0:
            return h
        adj = diff / mask.sum()
        h[mask] = np.minimum(1, h[mask] + adj)
        # If still not at target, redistribute
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            mask2 = h < 1 - 1e-10
            if mask2.sum() > 0:
                h[mask2] += diff2 / mask2.sum()
                h = np.clip(h, 0, 1)
    else:
        mask = h > 1e-10
        if mask.sum() == 0:
            return h
        adj = diff / mask.sum()
        h[mask] = np.maximum(0, h[mask] + adj)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            mask2 = h > 1e-10
            if mask2.sum() > 0:
                h[mask2] += diff2 / mask2.sum()
                h = np.clip(h, 0, 1)
    # Final correction
    h = np.clip(h, 0, 1)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-10:
        # Adjust the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + residual, 0, 1)
    return h

def gradient_of_weighted_correlation(h, weights):
    """Gradient of sum_k weights[k] * correlate(h, 1-h, 'full')[k] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            if m <= 0:
                continue
            grad[s:] += w * (1 - h[:m])
            grad[:m] -= w * h[s:]
        else:
            t = -s
            m = n - t
            if m <= 0:
                continue
            grad[:m] += w * (1 - h[t:])
            grad[t:] -= w * h[:m]
    return grad

def smooth_max_optimize(h0, dx, target_sum, max_iter=5000, time_limit=60):
    """Optimize h using smooth max approximation and gradient descent."""
    n = len(h0)
    h = h0.copy()
    start_time = time.time()
    
    best_h = h.copy()
    best_score = c5_score(h, dx)
    
    tau = 50.0  # soft focus parameter
    lr = 0.01
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        c = np.correlate(h, 1 - h, mode="full")
        # Smooth max weights
        c_scaled = c * dx / tau
        c_max = np.max(c_scaled)
        weights = np.exp(c_scaled - c_max)
        weights /= weights.sum()
        
        # Compute gradient
        grad = gradient_of_weighted_correlation(h, weights * dx)
        
        # Gradient descent step
        h = h - lr * grad
        h = fix_sum(h, target_sum)
        
        # Update best
        score = c5_score(h, dx)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        # Adaptive learning rate
        if it % 100 == 99:
            tau *= 1.5
            lr *= 0.9
    
    return best_h, best_score

def binary_local_search(h0, dx, target_sum, max_iter=50000, time_limit=30):
    """Local search on binary solutions."""
    n = len(h0)
    # Round to binary
    h = (h0 >= 0.5).astype(float)
    # Fix sum
    ones_count = int(h.sum())
    if ones_count > target_sum:
        # Remove excess ones (from positions with smallest "importance")
        excess = ones_count - int(target_sum)
        order = np.argsort(h0)[::-1]  # h0 ascending, so we remove the smallest

    c = np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:

    else:

    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-10:

        else:

import numpy as np
import time


def c5_score(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-9
    else:
        mask = h > 1e-9
    if mask.sum() == 0:
        return h
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-8:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-9:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h


def grad_wcorr(h, W):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = W[k]
        if abs(w) < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            if m <= 0:
                continue
            grad[s:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[s:]
        else:
            t = -s
            m = n - t
            if m <= 0:
                continue
            grad[t:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[t:]
    return grad


def opt_continuous(h0, dx, tsum, tl):
    h = h0.copy()
    st = time.time()
    bh = h.copy()
    bs = c5_score(h, dx)
    tau = 40.0
    lr = 0.004
    while time.time() - st < tl:
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        cm = np.max(c)
        w = np.exp((c - cm) / (tau * dx))
        ss = w.sum()
        if ss < 1e-30:
            break
        w /= ss
        g = grad_wcorr(h, w)
        h = h - lr * g
        h = fix_sum(h, tsum)
        s = c5_score(h, dx)
        if s < bs:
            bs = s
            bh = h.copy()
        tau *= 1.15
        lr *= 0.96
        if lr < 1e-7:
            break
    return bh, bs
