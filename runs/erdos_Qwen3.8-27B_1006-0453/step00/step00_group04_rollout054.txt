    return float(np.max(c))

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def interp_parent(parent, n):
    m = len(parent)
    if m == n:
        return parent.copy()
    x0 = np.linspace(0.0, 2.0, m)
    x1 = np.linspace(0.0, 2.0, n)
    f = np.interp(x1, x0, parent)
    f = np.clip(f, 0.0, 1.0)
    s = np.sum(f)
    target = n / 2.0
    if s > 1e-10:
        f *= target / s
        f = np.clip(f, 0.0, 1.0)
    return f

def smooth_max(c, beta):
    cmax = np.max(c)
    shifted = np.clip(beta * (c - cmax), -700.0, 0.0)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

    cvals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    val = smooth_max(cvals, beta)
    pen = mu * (np.sum(h) - n * 0.5) ** 2
    return val + pen

Let me optimize: instead of computing the full correlation for each candidate swap, I can compute it incrementally.

When we flip h[i] from 1 to 0 and h[j] from 0 to 1:
The correlation vector changes by a specific pattern. Let me compute it:

For the full correlation of length 2n-1:
correlate(h, 1-h, 'full')[m] = sum_p h[p] * (1-h[p + m - n + 1])

Let lag = m - (n-1). Then:
C[lag] = sum_p h[p] * (1-h[p+lag]) for valid p.

When h changes by d (where d has d[i]=-1, d[j]=+1, else 0):
new_C[lag] = C[lag] + sum_p (d[p]*(1-h[p+lag]) + h[p]*(-d[p+lag]))
           = C[lag] + (1-h[i+lag])*(-1) + h[i+lag - lag]... 

Hmm, let me be more careful:
new_C[lag] = C[lag] + sum_p d[p]*(1-h[p+lag]) - sum_p h[p]*d[p+lag]
= C[lag] + (-1)*(1-h[i+lag]) + (+1)*(1-h[j+lag]) - h[i]*(-1) + h[j]*(+1)
Wait, d[p+lag]: d is non-zero only at i and j.
- d[i] = -1, d[j] = +1
- sum_p d[p]*(1-h[p+lag]) = -1*(1-h[i+lag]) + 1*(1-h[j+lag])
- sum_p h[p]*d[p+lag]: d[p+lag] is non-zero when p+lag = i (p = i-lag) or p+lag = j (p = j-lag)
  = h[i-lag]*(-1) + h[j-lag]*(+1) = -h[i-lag] + h[j-lag]

So: new_C[lag] = C[lag] + [-(1-h[i+lag]) + (1-h[j+lag])] - [-h[i-lag] + h[j-lag]]
= C[lag] - (1-h[i+lag]) + (1-h[j+lag]) + h[i-lag] - h[j-lag]

This is O(1) per lag! So to compute the max of new_C, I need to evaluate this for all lags (O(n)) and take the max.

But I can be smarter: I only need to check lags where the change is non-zero. The change at lag q is:
delta(q) = -(1-h[i+q]) + (1-h[j+q]) + h[i-q] - h[j-q]

This is non-zero only when i+q, j+q, i-q, or j-q is in [0, n-1]. For most q, at least one of these is valid, so the change is non-zero for most lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def get_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def interp_parent(parent, n):
    m = len(parent)
    if m == n:
        return parent.copy()
    x0 = np.linspace(0.0, 2.0, m)
    x1 = np.linspace(0.0, 2.0, n)
    f = np.interp(x1, x0, parent)
    f = np.clip(f, 0.0, 1.0)
    s = np.sum(f)
    target = n / 2.0
    if s > 1e-10:
        f *= target / s
        f = np.clip(f, 0.0, 1.0)
    return f

def smooth_max(c, beta):
    cmax = float(np.max(c))
    shifted = np.clip(beta * (c - cmax), -700.0, 0.0)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def obj_func(h, n, beta, mu):
    cvals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    val = smooth_max(cvals, beta)
    pen = mu * (float(np.sum(h)) - n * 0.5) ** 2
    return val + pen

def binary_greedy(h, rng, deadline, max_iters=5000):
    n = len(h)
    h = h.copy()
    cvals = get_lag_values(h)
    best_c = float(np.max(cvals))
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    stale = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c = float(np.max(get_lag_values(h)))
        K = min(30, len(ones) * len(zeros))
        improved = False
        for _ in range(K):
            if time.time() > deadline:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c = float(np.max(get_lag_values(h)))
            h[i] = 1.0
            h[j] = 0.0
            if new_c < best_c - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                best_c = new_c
                improved = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                stale = 0
                break
        if not improved:
            stale += 1
            if stale > 50:
                if len(ones) > 1 and len(zeros) > 1:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h <= 0.5)[0]
                    stale = 0
            if stale > 300:
                break
    return h, best_c

def fix_binary_count(h_bin, h_cont, n):
    target = n // 2
    n_ones = int(np.sum(h_bin > 0.5))
    if n_ones > target:
        excess = n_ones - target
        idxs = np.where(h_bin > 0.5)[0]
        order = np.argsort(h_cont[idxs])
        for k in range(min(excess, len(order))):
            h_bin[idxs[order[k]]] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        idxs = np.where(h_bin <= 0.5)[0]
        order = np.argsort(-h_cont[idxs])
        for k in range(min(deficit, len(order))):
            h_bin[idxs[order[k]]] = 1.0
    return h_bin
