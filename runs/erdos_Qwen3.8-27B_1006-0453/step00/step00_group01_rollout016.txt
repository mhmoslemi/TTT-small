    jm = idx - lag
    mask1 = (jm >= 0) & (jm < n)
    grad[mask1] += 1.0 - h[jm[mask1]]
    jp = idx + lag
    mask2 = (jp >= 0) & (jp < n)
    grad[mask2] -= h[jp[mask2]]
    return grad

def optimize_single(h0, end_time, rng, cos_time_frac=0.95):
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.05
    min_lr = 1e-7
    stagnant = 0
    max_stagnant = 300
    deadline = start_time_global + (end_time - start_time_global) * cos_time_frac

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        grad = active_gradient(h, k_star)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = lr / max(gnorm, 0.01)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
            lr = min(lr * 1.03, 0.3)
        else:
            stagnant += 1
            lr *= 0.93
            if stagnant > max_stagnant:
                h = best_h + 0.03 * rng.standard_normal(n)
                h = project_feasible(h, target_sum)
                lr = 0.05
                stagnant = 0
    return best_h, best_c5

def local_search(h0, end_time, rng):
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    deadline = time.time() + 0.3 * (end_time - time.time())
    while time.time() < deadline:
        i = rng.randint(n)
        h_new = h.copy()
        h_new[i] += rng.choice([-1, 1]) * rng.uniform(0.05, 0.3)
        h_new[i] = np.clip(h_new[i], 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = np.clip(h - 0.5 * (h - best_h) * 0.01 * rng.standard_normal(n), 0, 1)
            h = project_feasible(h, target_sum)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global start_time_global
    start_time_global = time.time()
    end_time = start_time_global + budget_s
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(64, 0.5)
    n1 = len(h_init)
    h_init = project_feasible(h_init, n1 / 2.0)

    best_h, best_c5 = h_init.copy(), compute_c5(h_init)

    phase2_deadline = start_time_global + (end_time - start_time_global) * 0.35
    h_opt, c5_opt = optimize_single(h_init, phase2_deadline, rng)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    n_opt = len(h_opt)
    pert_deadline = start_time_global + (end_time - start_time_global) * 0.5
    for trial in range(4):
        if time.time() > pert_deadline:
            break
        frac = (pert_deadline - time.time()) / max(1, 4 - trial)
        h_pert = h_opt + 0.06 * rng.standard_normal(n_opt)
        h_pert = project_feasible(h_pert, n_opt / 2.0)
        dh, dc = optimize_single(h_pert, time.time() + frac * 0.9, rng)
        if dc < best_c5:
            best_h, best_c5 = dh, dc
            h_opt = dh.copy()

    upsample_deadline = start_time_global + (end_time - start_time_global) * 0.85
    for factor in [2, 3, 4, 5]:
        if time.time() > upsample_deadline:
            break
        n2 = n1 * factor
        if n2 > 500:
            continue
        h_up = np.repeat(h_opt, factor)
        h_up = project_feasible(h_up, n2 / 2.0)
        remaining = upsample_deadline - time.time()
        if remaining < 10:
            break
        dh, dc = optimize_single(h_up, time.time() + remaining * 0.9, rng)
        if dc < best_c5:
            best_h, best_c5 = dh, dc

    if time.time() < end_time - 20:
        n_ls = len(best_h)
        h_ls, c5_ls = local_search(best_h, end_time - 10, rng)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
