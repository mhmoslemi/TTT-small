            total_room = room.sum()
            if total_room < 1e-12:

            h += scale * room

            room = h.copy()
            total_room = room.sum()
            if total_room < 1e-12:
                break
            scale = min(1.0, (-diff) / total_room)
            h -= scale * room

    for _ in range(k):
        h[rng.randint(n)] = rng.uniform(0.0, 1.0)
    return h


def _swap_perturb(h, rng, k):
    h = h.copy()
    n = len(h)
    for _ in range(k):
        i, j = rng.choice(n, size=2, replace=False)
        h[i], h[j] = h[j], h[i]
    return h


def _interp_h(h_src, n_dst):
    n = len(h_src)
    xs = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    return np.interp(xd, xs, h_src)


def _random_binary(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h


def _optimize_from(h, n, rng, max_passes, n_grid, t0, budget_frac):
    c5 = _compute_c5(h)
    stuck = 0
    for _ in range(max_passes):
        if time.time() > t0 + budget_frac:
            break
        _coord_pass(h, n, rng, n_grid)
        h = _repair_sum(h, n)
        new_c5 = _compute_c5(h)
        if new_c5 < c5 - 1e-11:
            c5 = new_c5
            stuck = 0
        else:
            stuck += 1
        if stuck > 8:
            h = _perturb(h, rng, rng.randint(1, 4))
            h = _repair_sum(h, n)
            stuck = 0
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = np.clip(initial_h_values.copy().astype(float), 0.0, 1.0)
    n = len(best_h)
    best_h = _repair_sum(best_h, n)
    best_c5 = _compute_c5(best_h)

    # Phase 1: Coordinate descent on parent (35% budget)
    h = best_h.copy()
    h, c5 = _optimize_from(h, n, rng, 500, 40, t0, t0 + budget_s * 0.35)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 2: Multi-restart perturb + CD (30% budget)
    t_phase2_end = t0 + budget_s * 0.65
    while time.time() < t_phase2_end:
        r = rng.random()
        if r < 0.25:
            h = _random_binary(n, rng)
        elif r < 0.5:
            h = _swap_perturb(best_h, rng, rng.randint(2, 6))
            h = _perturb(h, rng, rng.randint(1, 3))
        else:
            h = _perturb(best_h, rng, rng.randint(1, 5))
        h = _repair_sum(np.clip(h, 0.0, 1.0), n)
        h, c5 = _optimize_from(h, n, rng, 30, 30, t0, t_phase2_end)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 3: Alternate resolutions (25% budget)
    t_phase3_end = t0 + budget_s * 0.93
    for n_try in [50, 60, 80, 100, 120, 150]:
        if time.time() >= t_phase3_end:
            break
        for init_mode in range(3):
            if time.time() >= t_phase3_end:
                break
            if init_mode == 0:
                h_new = _interp_h(best_h, n_try).copy()
            elif init_mode == 1:
                h_new = _random_binary(n_try, rng)
            else:
                h_new = np.clip(rng.uniform(0.15, 0.85, size=n_try), 0.0, 1.0)
            h_new = _perturb(h_new, rng, 3)
            h_new = _repair_sum(np.clip(h_new, 0.0, 1.0), n_try)
            h_new, c5 = _optimize_from(h_new, n_try, rng, 25, 25, t0, t_phase3_end)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new

    # Final polish (use remaining time)
    n_final = len(best_h)
    t_end = t0 + budget_s * 0.995
    while time.time() < t_end:
        _coord_pass(best_h, n_final, rng, 50)
        best_h = _repair_sum(best_h, n_final)
        c5 = _compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, len(best_h))
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, len(best_h)
