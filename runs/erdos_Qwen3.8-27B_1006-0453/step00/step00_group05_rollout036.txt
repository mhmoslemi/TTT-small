import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_valid_binary(h, n, rng):
    h = (h >= 0.5).astype(float)
    target = n // 2
    s = int(h.sum())
    while s > target:
        idx = int(rng.integers(n))
        if h[idx] == 1:
            h[idx] = 0; s -= 1
    while s < target:
        idx = int(rng.integers(n))
        if h[idx] == 0:
            h[idx] = 1; s += 1
    return h

def full_pass(h, n, max_passes, deadline):
    best_c5 = compute_c5(h)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_val = best_c5
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                c5 = compute_c5(h)
                h[i] = 1; h[j] = 0
                if c5 < best_val - 1e-14:
                    best_val = c5
                    best_swap = (i, j)
        if best_swap is None:
            break
        h[best_swap[0]] = 0
        h[best_swap[1]] = 1
        best_c5 = best_val
    return h, best_c5

def random_hill(h, n, iters, rng, deadline=None):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(iters):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = 0, 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_c5

    T0 = 0.1
    T1 = 0.001
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    for step in range(iters):

            break

        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
