import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def optimize_at_level(h_in, n, t0, budget_s, max_inner_time):
    bounds = [(0.0, 1.0)] * n
    schedules = [
        (0.08, 200.0, 250),
        (0.03, 1000.0, 200),
        (0.008, 5000.0, 200),
        (0.002, 20000.0, 150),
    ]
    cur = h_in.copy()
    best_local = compute_c5(cur)
    for temp, pen, maxit in schedules:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.82:
            break
        res = minimize(
            smooth_c5, cur, args=(temp, pen),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best_local:
            best_local = c5v
        cur = nh
        if time.time() - t0 > t0 + max_inner_time:
            break
    return cur, best_local

def random_perturb(h, sigma):
    n = len(h)
    noise = np.random.randn(n) * sigma
    h2 = np.clip(h + noise, 0.0, 1.0)
    return repair_h(h2)

def run(seed=42, budget_s=1100, **kwargs):

    cur_n = len(cur_h)

    upscale_factors = [2, 2, 2]
    for fi, uf in enumerate(upscale_factors):
        if time.time() - t0 > budget_s * 0.7:
            break
        cur_h = interp_upscale(cur_h, uf)
        cur_h = repair_h(cur_h)
        cur_n = len(cur_h)

        max_inner = budget_s * 0.18 * (1.0 / (fi + 1))
