import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, tau):
    m = float(np.max(vals))
    if tau * m > 400.0:
        return m
    shifted = tau * (vals - m)
    return m + float(np.log(np.sum(np.exp(shifted)) / len(vals))) / tau

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(corr, tau)

        cnt = float(np.sum(mask))

            h[mask] += diff / cnt

def interpolate_h(h_src, n_src, n_tgt):
