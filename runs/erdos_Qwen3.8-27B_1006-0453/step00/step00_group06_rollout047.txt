import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = np.maximum(1.0 - h, 1e-10)
            rs = room.sum()
            if rs > 1e-12:
                h += diff * room / rs
        else:
            room = np.maximum(h, 1e-10)
            rs = room.sum()
            if rs > 1e-12:
                h += diff * room / rs
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
            h[idx] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h

def make_block_init(n, rng):
    h = np.zeros(n)
    length = n // 2
    start = int(rng.integers(0, n - length + 1))
    h[start:start+length] = 1.0
    if length * 1.0 != n * 0.5:
        h[start + length - 1] = n * 0.5 - (length - 1)
    return repair(h)

def make_random_init(n, rng):
    h = rng.uniform(0, 1, n)
    return repair(h)
