    m = np.max(cor)
    return m + np.log(np.sum(np.exp(beta * (cor - m)))) / beta

    m = np.max(cor)
    weights = np.exp(beta * (cor - m))
    weights /= np.sum(weights)
    return weights

def compute_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # weights[k] corresponds to correlation at "index" k in full mode

                g -= h[j+d]
            grad[j] += w * g

    n_ones = int(round(target_sum))
    frac = target_sum - n_ones
    h = np.zeros(n)
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    if abs(frac) > 1e-10:
        h[idx[0]] = frac if frac > 0 else 1.0 + frac
    return make_feasible(h, target_sum)

def gen_spread(n, seed):
    rng = np.random.RandomState(seed)
    target_sum = n / 2.0
    n_ones = int(round(target_sum))
    h = np.zeros(n)
    if n_ones > 0:
        positions = np.linspace(0, n, n_ones + 1).astype(int)[:n_ones]
        jitter = rng.randint(-2, 3, n_ones)
        positions = np.clip(positions + jitter, 0, n - 1)
        positions = np.unique(positions)
        h[positions[:min(n_ones, len(positions))]] = 1.0
    return make_feasible(h, target_sum)

def try_round(h, target_sum):
    n = len(h)
    h_nr = np.where(h > 0.5, 1.0, 0.0)
    deficit = target_sum - np.sum(h_nr)
    if abs(deficit) > 1e-10:
        fr = np.round(h_nr)
        sorted_idx = np.argsort(np.abs(h - 0.5))[::-1]
        if deficit > 0:
            for i in sorted_idx:
                if fr[i] < 1.0:
                    need = min(deficit, 1.0 - fr[i])
                    fr[i] += need
                    deficit -= need
                    if deficit < 1e-10:
                        break
        else:
            for i in sorted_idx:
                if fr[i] > 0.0:
                    need = min(-deficit, fr[i])
                    fr[i] -= need
                    deficit += need
                    if deficit > -1e-10:
                        break
    return np.clip(fr, 0.0, 1.0)

def optimize(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(h)

    beta = 15.0
    lr = 0.008
    patience = 0
    stall_count = 0

    while time.time() - t0 < time_limit:
        cor = compute_cor(h)
        weights = smooth_max_weights(cor, beta)
        grad = compute_grad(h, weights)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm

        improved = False
        for trial in range(10):
            alpha = lr * (0.55 ** trial)
            h_trial = project_sum(h - alpha * grad, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-10:
                h = h_trial
                best_h = h.copy()
                best_c5 = c5_trial
                improved = True
                break
        if not improved:
            # Try a direct round-and-repair
            h_r = try_round(h, target_sum)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5 - 1e-8:
                h = h_r
                best_h = h.copy()
                best_c5 = c5_r
                improved = True

        if improved:
            patience = 0
            beta = min(beta * 1.03, 150.0)
            stall_count = 0
        else:
            patience += 1
            stall_count += 1
            if stall_count % 40 == 0:
                lr = max(lr * 0.6, 0.0005)
                beta = max(beta * 0.6, 5.0)
            if patience > 400:
                scale = max(lr, 0.003)
                noise = rng.randn(n) * scale
                h = project_sum(h + noise, target_sum)
                patience = 0
                stall_count = 0
                lr = 0.008
                beta = 10.0

    # Final round attempt
    h_r = try_round(best_h, target_sum)
    c5_r = compute_c5(h_r)
    if c5_r < best_c5:
        best_h = h_r
        best_c5 = c5_r

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')

    def elapsed():
        return time.time() - t_start

    # Use initial_h_values if available
    start_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        start_h = np.array(initial_h_values, dtype=float)
        n_init = len(start_h)
        start_h = make_feasible(start_h, n_init / 2.0)
        c5_s = compute_c5(start_h)
        best_h = start_h.copy()
        best_c5 = c5_s

    # Phase 1: Optimize from initial (largest chunk of time)
    if start_h is not None:
        rem = budget_s - elapsed() - 60
        if rem > 30:
            h_opt, c5_opt = optimize(start_h, rem * 0.45, seed)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Phase 2: Try multiple n and starting patterns
    n_list = [45, 50, 60, 75, 55, 65]
    seen_ns = set()
    if start_h is not None:
        seen_ns.add(len(start_h))
    n_list = [x for x in n_list if x not in seen_ns]

    for ni in n_list:
        rem = budget_s - elapsed() - 30
        if rem < 40:
            break
        for gen_fn in [gen_binary, gen_spread]:
            rem = budget_s - elapsed() - 30
            if rem < 20:
                break
            for att in range(3):
                rem = budget_s - elapsed() - 30
                if rem < 15:
                    break
                h0 = gen_fn(ni, seed + att * 13 + ni)
                t_lim = min(rem * 0.2, 80.0)
                if t_lim < 5:
                    break
                h_opt, c5_opt = optimize(h0, t_lim, seed + att + ni * 7)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                if elapsed() > budget_s - 60:
                    break

    # Phase 3: Final polish on best
    rem = budget_s - elapsed() - 5
    if rem > 15 and best_h is not None:
        h_opt, c5_opt = optimize(best_h, rem, seed + 7777)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Verify feasibility
    best_h = make_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
