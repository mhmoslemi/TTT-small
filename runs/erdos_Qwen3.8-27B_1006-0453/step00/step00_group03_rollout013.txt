import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    k = int(np.argmax(c))
    m = k - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        j = i + m
        if 0 <= j < n:
            grad[j] += (1.0 - h[i])
            grad[i] -= h[j]
    # Add subgradient averaging over all k that are near-max
    c_max = c[k]
    mask = c > (c_max - 1e-10)
    if np.sum(mask) > 1:
        grad = np.zeros(n)
        for kk in np.where(mask)[0]:
            mm = kk - (n - 1)
            g2 = np.zeros(n)
            for i in range(n):
                j = i + mm
                if 0 <= j < n:
                    g2[j] += (1.0 - h[i])
                    g2[i] -= h[j]
            grad += g2
        grad /= np.sum(mask)
    return grad * dx, c_max

def fix_h(h, n):
