def project_sum(h, target):
    """Project h onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(h)
    def sum_at_shift(shift):
        return np.sum(np.clip(h + shift, 0, 1))
    s0 = np.sum(h)
    if abs(s0 - target) < 1e-12:
        return h
    # Bisection for shift
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_at_shift(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def smooth_c5_and_grad(h, alpha, penalty=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    e = np.clip(alpha * (corr - cmax), -500, 500)
    e_exp = np.exp(e)
    w_sum = np.sum(e_exp)
    w = e_exp / w_sum
    f_val = cmax + np.log(w_sum) / alpha
    if penalty > 0:
        s = np.sum(h) - n / 2.0
        f_val += penalty * s * s
    grad = np.zeros(n)
    if penalty > 0:
        grad += 2.0 * penalty * s
    for m in range(n):
        k = m + n - 1
        wc = w[k] * dx
        if wc < 1e-14:
            continue
        cls = n - m
        grad[0:cls] += wc * (1.0 - h[m:m+cls])
        grad[m:n] -= wc * h[0:cls]
    for p in range(1, n):
        k = n - 1 - p
        wc = w[k] * dx
        if wc < 1e-14:
            continue
        cls = n - p
        grad[p:n] += wc * (1.0 - h[0:cls])
        grad[0:cls] -= wc * h[p:n]
    return f_val, grad

def project_sum(h, target):
    n = len(h)
    s0 = np.sum(h)
    if abs(s0 - target) < 1e-12:
        return h.copy()
    lo, hi = -max(s0 - target, 0) - 0.1, max(target - s0, 0) + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def make_starts(n, seed):
    rng = np.random.RandomState(seed)
    starts = []
    # Sine-based
    for freq in [2, 3, 4, 5]:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(np.pi * freq * x)
        h = np.clip(h, 0, 1)
        starts.append(project_sum(h, n / 2.0))
    # Random binary
    for _ in range(3):
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        starts.append(h)
    # Periodic blocks
    for block in [4, 5, 6, 7]:
        h = np.zeros(n)
        h[0:n // 2] = 1.0
        # Permute with block structure
        perm = np.tile(np.arange(block), n // block + 1)[:n]
        h = h[0:0] + (np.arange(n) % block < block // 2).astype(float)
        starts.append(project_sum(h, n / 2.0))
    return starts

def interp_parent(parent, n):
    xp = np.linspace(0, 2, len(parent))
    xn = np.linspace(0, 2, n)
    h = np.interp(xn, xp, parent)
    return project_sum(np.clip(h, 0, 1), n / 2.0)

def optimize_start(h0, n, t0, budget, max_time):
    deadline = t0 + max_time
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    alphas = [15, 40, 80, 150, 300, 600, 1200]

            return smooth_c5_and_grad(x, alpha, pen)[0]

            return smooth_c5_and_grad(x, alpha, pen)[1]

                best_h = h_new
            else:

def binary_local_search(h, max_flips=2000):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
