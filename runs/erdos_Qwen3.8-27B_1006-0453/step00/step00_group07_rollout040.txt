def _smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def _obj(x):

    beta = _OPT['beta']
