def repair_sum(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = float(np.sum(h))

def gradient_max_c5(h, top_m=5):
    """Compute subgradient of C5 using top-m lags."""
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

    m = min(top_m, len(c5_vals))

        d = abs(k)
        g = np.zeros(n)
        g[d:] += (1.0 - h[:n-d])
        g[:n-d] -= h[d:]
        total_grad += weights[top_idx.tolist().index(idx)] * g
