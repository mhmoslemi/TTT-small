import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    scaled = T * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    z = np.sum(exp_s)
    weights = exp_s / z
    val = (max_s + np.log(z)) / T
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = weights[idx] * dx
        if wk < 1e-16:
            continue
        k = idx - (n - 1)
        js1 = max(0, k)
        je1 = min(n - 1, k + n - 1)
        if js1 <= je1:
            grad[js1:je1 + 1] += wk * (1.0 - h[js1 - k:je1 - k + 1])
        js2 = max(0, -k)
        je2 = min(n - 1, n - 1 - k)
        if js2 <= je2:
            grad[js2:je2 + 1] += wk * (-h[js2 + k:je2 + k + 1])
    return val, grad

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_resolution(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_constraint(h_new, target)

def lbfgs_step(h, T, target, max_iter, ftol):
    def obj(x):
        v, _ = smooth_obj_grad(x, T)
        pen = 100.0 * (np.sum(x) - target) ** 2
        return v + pen
    def grad(x):
        _, g = smooth_obj_grad(x, T)
        g = g + 200.0 * (np.sum(x) - target) / len(x)
        return g
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * len(h),
                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-7})
    return project_constraint(res.x, target)

def slsqp_step(h, T, target, max_iter):
    def obj(x):
        v, _ = smooth_obj_grad(x, T)
        return v
    def jfun(x):
        _, g = smooth_obj_grad(x, T)
        return g
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(len(h))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h, jac=jfun, method='SLSQP',
                   bounds=[(0.0, 1.0)] * len(h),
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9})
    return project_constraint(res.x, target)

def coord_descent(h, target, max_sweeps=50, deadline=None):
    n = len(h)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            hi = 1.0
            lo = 0.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                h[j] = mid
                h = project_constraint(h, target)
                c = compute_c5(h)
                if c < best_c5:
                    lo = mid
                    best_c5 = c
                else:
                    hi = mid
            h = project_constraint(h, target)
            c = compute_c5(h)
            if c >= best_c5 - 1e-12:
                h[j] = old
                h = project_constraint(h, target)
            else:
                best_c5 = c
                improved = True
        if not improved:
            break
    return h, best_c5
