def project_feasible(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    hp = np.clip(h - lam, 0.0, 1.0)

    if abs(res) > 1e-14:
        mask = (hp > 1e-15) & (hp < 1.0 - 1e-15)
        if mask.any():
            hp[mask] += res / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def sm_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    p = np.exp(alpha * (corr - m))
    p /= p.sum()
    omh = 1.0 - h
    jj = np.arange(n)[:, None]
    mm = np.arange(n)[None, :]
    d = mm - jj
    pp = p[d + n - 1]
    pn = p[n - 1 - d]
    return dx * (omh[None, :] * pp - h[None, :] * pn).sum(axis=1)

def sm_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m)))) + m

    t = np.arange(n) * n0 / n
    i0 = t.astype(int) % n0
    i1 = (i0 + 1) % n0
    f = t - t.astype(int)
    return h_init[i0] * (1 - f) + h_init[i1] * f
