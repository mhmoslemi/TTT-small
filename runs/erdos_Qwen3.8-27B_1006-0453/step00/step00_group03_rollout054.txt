```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        free = (proj > 1e-8) & (proj < 1 - 1e-8)
        nf = np.sum(free)
        if nf > 0:
            proj[free] += diff / nf
            proj = np.clip(proj, 0, 1)
    return proj

def subgrad_c5(h, n_active=1):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    lam_max = np.max(corr)
    active = np.where(corr >= lam_max - 1e-8)[0]
    if len(active) > n_active:
        active = active[np.argsort(corr[active])[-n_active:]]
    grad = np.zeros(n)
    for m in active:
        lag = m - (n - 1)
        # d/dh[j] of corr[m] = (1-h[j+lag])*I(0<=j+lag<n) - h[j-lag]*I(0<=j-lag<n)
        # Vectorized:
        jp = np.arange(n) + lag
        jm = np.arange(n) - lag
        valid_p = (jp >= 0) & (jp < n)
        valid_m = (jm >= 0) & (jm < n)
        gp = np.where(valid_p, 1 - h[jp], 0)
        gm = np.where(valid_m, h[jm], 0)
        grad += gp - gm
    grad *= dx / len(active)
    return grad

def psgd(h0, budget_s=100):
    """Projected subgradient descent"""
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    alpha = 0.5
    rho = 1.0
    for it in range(20000):
        if time.time() - t0 > budget_s:
            break
        g = subgrad_c5(h, n_active=min(5, n))
        alpha = 2.0 / (rho * np.sqrt(it + 1) + 1)
        h_new = h - alpha * g
        h_new = project_feasible(h_new, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        rho = max(rho, np.max(np.abs(g)) + 1e-8)
    return best_h, best_c

def slsqp_smooth(h0, beta=100, budget_s=100):
    """SLSQP with log-sum-exp smooth approximation"""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    t0 = time.time()
    
    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        scaled = beta * corr
        mx = np.max(scaled)
        return (mx + np.log(np.sum(np.exp(scaled - mx)))) / beta * dx
    
    def obj_grad(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        scaled = beta * corr
        mx = np.max(scaled)
        weights = np.exp(scaled - mx)
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for m_idx, w in enumerate(weights):
            lag = m_idx - (n - 1)
            jp = np.arange(n) + lag
            jm = np.arange(n) - lag
            valid_p = (jp >= 0) & (jp < n)
            valid_m = (jm >= 0) & (jm < n)
            gp = np.where(valid_p, 1 - h[jp], 0)
            gm = np.where(valid_m, h[jm], 0)
            grad += w * (gp - gm)
        grad *= beta * dx
        return grad
    
    x0 = project_feasible(h0.copy(), target)
    bounds = [(0, 1)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    
    result = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-10})
    if time.time() - t0 > budget_s:
        return x0, compute_c5(x0)
    
    h = project_feasible(result.x, target)
    return h, compute_c5(h)

def interpolate_to(h, n_new):
    """Interpolate h from its current length to n_new"""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_interp, n_new / 2.0)

def random_feasible(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.random(n)
    return project_feasible(h, n / 2.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time
```


```python
def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0, 1)
        if p.sum() > tgt:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0, 1)
    d = tgt - p.sum()
    if abs(d) > 1e-10:
        fr = (p > 1e-7) & (p < 1 - 1e-7)
        if fr.sum() > 0:
            p[fr] += d / fr.sum()
            p = np.clip(p, 0, 1)
    return p


def _subgrad(h, tk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argsort(corr)[-tk:]
    g = np.zeros(n)
    for m in idx:
        lag = m - (n - 1)
        jp = np.arange(n) + lag
        jm = np.arange(n) - lag
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        g += np.where(vp, 1 - h[jp], 0) - np.where(vm, h[jm], 0)
    return g * dx / len(idx)
```

```python
def _sobj(x, beta, n):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    s = beta * c
    m = s.max()
    return (m + np.log(np.sum(np.exp(s - m)))) / beta * dx


def _sjac(x, beta, n):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    s = beta * c
    m = s.max()
    w = np.exp(s - m)
    w /= w.sum()
    g = np.zeros(n)
    for mi in range(len(w)):
        if w[mi] < 1e-5:
            continue
        lag = mi - (n - 1)
        jp = np.arange(n) + lag
        jm = np.arange(n) - lag
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        g += w[mi] * (np.where(vp, 1 - x[jp], 0) - np.where(vm, x[jm], 0))
    return g * beta * dx


def _eqfun(x, tgt):
    return x.sum() - tgt


def _eqjac(x, tgt):
    return np.ones_like(x)


def _slsqp(h0, beta, tb):
    n = len(h0)
    tgt = n * 0.5
    x0 = _proj(h0.copy(), tgt)
    cons = [{'type': 'eq', 'fun': _eqfun, 'jac': _eqjac, 'args': (tgt,)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(_sobj, x0, jac=_sjac, args=(beta, n), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': 400, 'ftol': 1e-13})
    h = _proj(res.x, tgt)
    return h, _c5(h)


def _interp(h, nn):
    no = len(h)
    if nn == no:
        return h.copy()
    xo = np.arange(no) * 2.0 / no
    xn = np.arange(nn) * 2.0 / nn
    hi = np.interp(xn, xo, h, period=2.0)
    return _proj(hi, nn * 0.5)


def _randf(n, s):
```
