    mx = np.max(sh)
    e = np.exp(sh - mx)
    w = e / np.sum(e)
    j = np.arange(n)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if w[i] < 1e-14:
            continue
        lag = i - (n - 1)
        up = j + lag
        dn = j - lag
        mu = (up >= 0) & (up < n)
        md = (dn >= 0) & (dn < n)
        if np.any(mu):
            grad[mu] += w[i] * (1.0 - h[up[mu]])
        if np.any(md):
            grad[md] -= w[i] * h[dn[md]]
    return grad * dx

def hard_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    j = np.arange(n)
    up = j + lag
    dn = j - lag
    grad = np.zeros(n)
    mu = (up >= 0) & (up < n)
    md = (dn >= 0) & (dn < n)
    if np.any(mu):
        grad[mu] = (1.0 - h[up[mu]])
    if np.any(md):
        grad[md] -= h[dn[md]]
    return grad * dx, float(corr[k_max])

def project_feasible(v, n):
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
    return np.clip(h, 0.0, 1.0)

def optimize_pgd(h_init, n, tau, max_iter, t_limit, use_smooth=True):
    h = project_feasible(h_init, n)
    best = compute_c5(h)
    lr = 0.5
    t_start = time.time()
    stall = 0
    for it in range(max_iter):
        if time.time() - t_start > t_limit:
            break
        if use_smooth and tau > 0:
            g = smooth_grad(h, tau)
        else:
            g, _ = hard_subgrad(h)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        accepted = False
        for bs in range(50):
            h_new = project_feasible(h - lr * g * gn, n)
            v = compute_c5(h_new)
            if v < best - 1e-14:
                h = h_new
                best = v
                accepted = True
                break
            lr *= 0.5
        if accepted:
            stall = 0
        else:
            stall += 1
            lr = min(lr * 1.5, 2.0)
        if stall > 100:
            break
    return h, best

def binary_ls(h, n, max_iters, rng):
    h = (h > 0.5).astype(float)
    h = project_feasible(h, n)
    best = compute_c5(h)
    idx = np.arange(n)
    for _ in range(max_iters):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        v = compute_c5(h)
        if v < best - 1e-15:
            best = v
        else:
            h[i], h[j] = h[j], h[i]
    return h, best

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    return f(x_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Stage 1: Optimize at parent resolution with smooth gradient
    for tau in [30.0, 80.0, 150.0]:
        if time.time() - t0 > budget_s * 0.20:
            break
        tlim = min(30, budget_s * 0.06)
        h_opt, c5_opt = optimize_pgd(best_h, n_init, tau, 800, tlim, use_smooth=True)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Stage 2: Hard subgradient refinement
    if time.time() - t0 < budget_s * 0.35:
        h_opt, c5_opt = optimize_pgd(best_h, n_init, 0, 500, budget_s * 0.08, use_smooth=False)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Stage 3: Binary local search
    if time.time() - t0 < budget_s * 0.50:
        h_bin, c5_bin = binary_ls(best_h, n_init, 50000, rng)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
        # Re-optimize continuously from improved binary
        h_opt, c5_opt = optimize_pgd(best_h, n_init, 100.0, 400, budget_s * 0.05, use_smooth=True)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        h_bin, c5_bin = binary_ls(best_h, n_init, 50000, rng)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Stage 4: Try slightly larger n for finer resolution
    for n_try in [56, 84, 112]:
        if time.time() - t0 > budget_s * 0.72:
            break
        h_up = upsample(best_h, n_try)
        h_up = project_feasible(h_up, n_try)
        h_opt, c5_opt = optimize_pgd(h_up, n_try, 80.0, 400, budget_s * 0.06, use_smooth=True)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            h_bin, c5_bin = binary_ls(best_h, n_try, 20000, rng)
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin, c5_bin
            h_opt, c5_opt = optimize_pgd(best_h, n_try, 150.0, 300, budget_s * 0.04, use_smooth=True)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Stage 5: Random restarts at best n
    n_best = len(best_h)
    attempt = 0
    while time.time() - t0 < budget_s * 0.90 and attempt < 80:
        noise_scale = 0.05 + 0.05 * rng.random()
        pert = rng.normal(0, noise_scale, n_best)
        h_try = project_feasible(best_h + pert, n_best)
        tlim = min(8, (budget_s * 0.90 - (time.time() - t0)))
        if tlim < 2:
            break
        h_opt, c5_opt = optimize_pgd(h_try, n_best, 100.0, 200, tlim, use_smooth=True)
        if c5_opt < best_c5 - 1e-8:
            best_h, best_c5 = h_opt, c5_opt
            h_bin, c5_bin = binary_ls(best_h, n_best, 20000, rng)
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin, c5_bin
        attempt += 1

    # Final binary polish
    if time.time() - t0 < budget_s * 0.96:
        h_bin, c5_bin = binary_ls(best_h, n_best, 30000, rng)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
        h_opt, c5_opt = optimize_pgd(best_h, n_best, 200.0, 300, max(1, budget_s * 0.02), use_smooth=True)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final projection and verification
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
