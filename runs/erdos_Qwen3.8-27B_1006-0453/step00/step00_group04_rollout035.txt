import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def single_subgrad(h, lag):
    n = len(h)
    sg = np.zeros(n)
    if lag == 0:
        sg[:] = 1.0 - 2.0 * h
    elif lag > 0:
        sg[lag:] += (1.0 - h[:n - lag])
        sg[:n - lag] -= h[lag:]
    else:
        s = -lag
        sg[:n - s] += (1.0 - h[s:])
        sg[s:] -= h[:n - s]
    return sg


def multi_subgrad(h, top_k=5, sharpness=15.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    vals = np.array([corr[i] for i in top_idx])
    wts = np.exp((vals - vals[0]) * sharpness)
    wts /= wts.sum()
    sg = np.zeros(n)
    for w, idx in zip(wts, top_idx):
        lag = int(idx) - (n - 1)
        sg += w * single_subgrad(h, lag)
    return sg


def optimize_continuous(h, n, rng, time_limit):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.02
    stall = 0
    while time.time() - start < time_limit:
        for _ in range(500):
            if time.time() - start > time_limit:
                break
            sg = multi_subgrad(h, top_k=5, sharpness=15.0)
            h = project_feasible(h - alpha * sg, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
            alpha = min(alpha * 1.03, 0.12)
        else:
            stall += 1
            alpha *= 0.97
            if stall >= 8:
                h = project_feasible(h + rng.normal(0, 0.06, n), n)
                alpha = 0.02
                stall = 0
    return best_h, best_c5


def make_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_b))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_b == 1)[0]
        flip = rng.choice(ones, n_ones - target, replace=False)

    elif n_ones < target:
        zeros = np.where(h_b == 0)[0]
        flip = rng.choice(zeros, target - n_ones, replace=False)
        h_b[flip] = 1
    return h_b

def interpolate(h_src, n_dst):
    n_src = len(h_src)
    out = np.zeros(n_dst)
    for i in range(n_dst):
        pos = i * n_src / n_dst
        lo = int(pos)
        if lo >= n_src:
            lo = n_src - 1
        frac = pos - lo
        out[i] = (1.0 - frac) * h_src[lo] + frac * h_src[(lo + 1) % n_src]
    return out
