from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def binarize(h):
    n = len(h)
    target = n // 2
    if np.allclose(h, (h > 0.5).astype(float)):
        return (h > 0.5).astype(float)
    idx = np.argsort(-h)

    h_bin[idx[:target]] = 1.0
    return h_bin

def hill_climb(h, time_limit=30):
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    start = time.time()
    stalls = 0
    max_stalls = 30

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
