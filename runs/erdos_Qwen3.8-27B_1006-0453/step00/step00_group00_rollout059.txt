    for lam in (50.0, 500.0, 5000.0):
        if time.time() - t0 > budget_s * 0.15:
            break
        try:
            res = minimize(obj_pen, best_h, method='L-BFGS-B',
                          bounds=bounds, args=(dx, ts, lam),
                          options={'maxiter': 3000, 'ftol': 1e-15})
            hc = repair_sum(np.clip(res.x, 0, 1), ts)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c
        except Exception:
            pass

    # Phase 2: SLSQP with equality constraint
    if time.time() - t0 < budget_s * 0.3:
        try:
            cons = [{'type': 'eq', 'fun': cons_eq,
                     'jac': cons_eq_jac, 'args': (ts,)}]
            res = minimize(obj_direct_x, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          args=(dx,), options={'maxiter': 5000, 'ftol': 1e-15})
            hc = repair_sum(np.clip(res.x, 0, 1), ts)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c
        except Exception:
            pass

    # Phase 3: Many restarts with L-BFGS-B penalty
    n_rest = 300
    for r in range(n_rest):
        if time.time() - t0 > budget_s * 0.78:
            break
        scale = 0.01 + 0.12 * np.sqrt((r + 1) / float(n_rest))
        h_start = np.clip(best_h + rng.normal(0, scale, n), 0, 1)
        h_start = repair_sum(h_start, ts)
        try:
            res = minimize(obj_pen, h_start, method='L-BFGS-B',
                          bounds=bounds, args=(dx, ts, 5000.0),
                          options={'maxiter': 1500, 'ftol': 1e-15})
            hc = repair_sum(np.clip(res.x, 0, 1), ts)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c
        except Exception:
            pass

    # Phase 4: SLSQP from best found so far (polishing)
    if time.time() - t0 < budget_s * 0.85:
        try:
            cons = [{'type': 'eq', 'fun': cons_eq,
                     'jac': cons_eq_jac, 'args': (ts,)}]
            res = minimize(obj_direct_x, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          args=(dx,), options={'maxiter': 8000, 'ftol': 1e-16})
            hc = repair_sum(np.clip(res.x, 0, 1), ts)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c
        except Exception:
            pass

    # Phase 5: Swap-based local search (preserves sum exactly)
    if time.time() - t0 < budget_s * 0.88:
        improved = True
        while improved and time.time() - t0 < budget_s * 0.96:
            improved = False
            for _ in range(5000):
                if time.time() - t0 > budget_s * 0.96:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for d in (0.01, 0.005, 0.002, 0.001, 0.02, 0.05,
                          -0.01, -0.005, -0.002, -0.001, -0.02, -0.05):
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if hi > 1.0 or hj < 0.0:
                        continue
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    c5_try = compute_c5(h_try, dx)
                    if c5_try < best_c5 - 1e-16:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break

    # Phase 6: Very fine swap search
    if time.time() - t0 < budget_s * 0.93:
        improved = True
        while improved and time.time() - t0 < budget_s * 0.97:
            improved = False
            for _ in range(5000):
                if time.time() - t0 > budget_s * 0.97:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for d in (0.0005, 0.0001, 5e-5, -0.0005, -0.0001, -5e-5):
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if hi > 1.0 or hj < 0.0:
                        continue
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    c5_try = compute_c5(h_try, dx)
                    if c5_try < best_c5 - 1e-16:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break

    best_h = repair_sum(best_h, ts)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
