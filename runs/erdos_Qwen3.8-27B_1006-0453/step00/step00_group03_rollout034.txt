    return np.max(c)

def smooth_max_val(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    if sigma < 1e-6:
        return m
    return m + sigma * np.log(np.sum(np.exp((c - m) / sigma)))

def smooth_max_grad(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    if sigma < 1e-6:
        # Just return gradient at argmax
        k_star = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[k_star] = 1.0
    else:
        w = np.exp((c - m) / sigma)
        w /= np.sum(w)

        wk = dx * w[k]
        if abs(wk) < 1e-12:
            continue

        i_hi = min(n, s + n)
        if i_lo < i_hi:
