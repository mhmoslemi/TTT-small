import time
import math

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):

    # Simulated Annealing parameters
    T = 0.025
    T_min = 1e-10
    cool_rate = 0.99999
    stuck = 0
    max_stuck = 80000
    it = 0

    while True:
        it += 1
        if it % 4000 == 0:
            if time.time() - start_time > budget_s - 4:
                break

        # Pick two indices for transfer
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (i + 1) % n

        # Generate perturbation
        delta = rng.randn() * T * 2.5
        # Feasibility bounds: h[i]+d in [0,1] and h[j]-d in [0,1]
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi:
            continue
        if delta < d_lo:
            delta = d_lo
        elif delta > d_hi:
            delta = d_hi

        # Apply move
        h[i] += delta
        h[j] -= delta
        c5_new = compute_c5(h)

        # Acceptance criterion
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            diff = c5_new - best_c5
            if diff < T:
                if rng.random_sample() < math.exp(-diff / T):
                    pass  # accept worse solution
                else:
                    h[j] += delta
                    h[i] -= delta
            else:
                h[j] += delta
                h[i] -= delta

        # Cooling schedule
        T = max(T * cool_rate, T_min)

        # Restart if stuck in local optimum
        if stuck > max_stuck:
            h = best_h.copy()
            T = 0.012 + 0.008 * rng.random_sample()
            stuck = 0
            max_stuck = max(max_stuck // 2, 20000)

    # Final evaluation
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
