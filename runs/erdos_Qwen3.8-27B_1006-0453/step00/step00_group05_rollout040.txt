        w = p[m]
        if w < 1e-15:

        valid_hi = (j_from + lag >= 0) & (j_from + lag < n)
        grad[valid_hi] -= w * h[np.minimum(np.maximum(j_from[valid_hi] + lag, 0), n-1)]

        # c_m = sum_i h[i+lag] * (1-h[i]) for valid i
        # ∂c_m/∂h[j]:
        #   If j = i+lag for some i (i.e., i = j-lag, valid): contributes (1-h[j-lag])
        #   If j = i for some i (i.e., i = j, valid and j+lag valid): contributes -h[j+lag]

        #   j ranges from max(0, lag) to min(n-1, n-1) (always n-1 for the upper bound)
        #   Actually j ranges over [0, n) and we need j-lag ∈ [0, n), so j ∈ [max(0,lag), min(n-1, n-1+lag))
        #   For lag >= 0: j ∈ [lag, n-1], contribution: 1-h[j-lag]
        #   For lag < 0: j ∈ [0, n-1+lag], contribution: 1-h[j-lag]
        
        # For the second term: j ∈ [0, n) and j+lag ∈ [0, n)
        #   For lag >= 0: j ∈ [0, n-1-lag], contribution: -h[j+lag]
        #   For lag < 0: j ∈ [-lag, n-1], contribution: -h[j+lag]
        
        if lag >= 0:
            # First term: j from lag to n-1
            js = np.arange(lag, n)
            grad[js] += w * (1 - h[js - lag])
            # Second term: j from 0 to n-1-lag
            js2 = np.arange(0, n - lag)
            grad[js2] -= w * h[js2 + lag]
        else:
            # lag < 0
            neg_lag = -lag
            # First term: j from 0 to n-1-neg_lag
            js = np.arange(0, n - neg_lag)
            grad[js] += w * (1 - h[js + neg_lag])  # j - lag = j + neg_lag
            # Second term: j from neg_lag to n-1
            js2 = np.arange(neg_lag, n)
            grad[js2] -= w * h[js2 - neg_lag]  # j + lag = j - neg_lag
    
    return grad

import time

def make_binary(h, n_target_ones):
    """Convert h to binary with exactly n_target_ones ones."""
    n = len(h)
    h_bin = (h >= 0.5).astype(float)

        # Sort by h value (lowest first)

    n = len(h)

    dx = 2.0 / n
    current_c5 = eval_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    T = t0

        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]

        h[a] = 0; h[b] = 1
        new_c5 = eval_c5(h)

import time

def make_binary(h, n_ones):

    cnt = int(h_bin.sum())
    if cnt > n_ones:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])
        for i in range(cnt - n_ones):
            h_bin[idx[order[i]]] = 0
    elif cnt < n_ones:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(n_ones - cnt):
            h_bin[idx[order[i]]] = 1
    return h_bin

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def random_binary(n, n_ones, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)[:n_ones]
    h[idx] = 1.0
    return h

def sa_swap(h, rng, max_iters=50000):
    n = len(h)
    h = h.copy()
    current_c5 = eval_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    T = max(current_c5 * 0.1, 0.05)
    T_end = 0.005
    alpha = (T_end / T) ** (1.0 / max(1, max_iters))
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(max_iters):
        T = max(T * alpha, T_end)
        a_idx = rng.integers(len(ones))
        b_idx = rng.integers(len(zeros))
        a = ones[a_idx]
        b = zeros[b_idx]
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = eval_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_c5 = new_c5
            ones[a_idx] = b
            zeros[b_idx] = a
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    return best_h, best_c5

def slsqp_optimize(h0, eps=8.0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(eps * corr)
        return float(np.log(np.sum(np.exp(eps * corr - m))) + m) / eps
    
    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        sh = eps * corr - np.max(eps * corr)
        p = np.exp(sh)
        p /= p.sum()
        g = np.zeros(n)
        for m_idx in range(2 * n - 1):
            lag = m_idx - (n - 1)
            w = p[m_idx] * dx
            if w < 1e-14:
                continue
            if lag >= 0:
                js = np.arange(lag, n)
                g[js] += w * (1.0 - h[js - lag])
                js2 = np.arange(0, n - lag)
                g[js2] -= w * h[js2 + lag]
            else:
                nl = -lag
                js = np.arange(0, n - nl)
                g[js] += w * (1.0 - h[js + nl])
                js2 = np.arange(nl, n)
                g[js2] -= w * h[js2 - nl]
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=grad, bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})

import numpy as np
from scipy.optimize import minimize
import time

def make_binary(h, n_ones):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    cnt = int(h_bin.sum())
    if cnt > n_ones:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])
        for i in range(cnt - n_ones):
            h_bin[idx[order[i]]] = 0.0
    elif cnt < n_ones:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(n_ones - cnt):
            h_bin[idx[order[i]]] = 1.0
    return h_bin

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def random_binary(n, n_ones, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)[:n_ones]
    h[idx] = 1.0
    return h

def sa_swap(h, rng, n_iters, T_start, T_end):
    n = len(h)
    h = h.copy()
    cur_c5 = eval_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T_start
    alpha = (T_end / T_start) ** (1.0 / max(1, n_iters))
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for it in range(n_iters):
        T *= alpha
        a_i = int(rng.integers(len(ones)))
        b_i = int(rng.integers(len(zeros)))
        a, b = ones[a_i], zeros[b_i]
        h[a], h[b] = 0.0, 1.0
        new_c5 = eval_c5(h)
        d = new_c5 - cur_c5
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            cur_c5 = new_c5
            ones[a_i], zeros[b_i] = b, a
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a], h[b] = 1.0, 0.0
    return best_h, best_c5

def slsqp_opt(h0, eps, maxiter):
    n = len(h0)
    dx = 2.0 / n
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(eps * c)
        return float(np.log(np.sum(np.exp(eps * c - m))) + m) / eps
    def grad(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        sh = eps * c - np.max(eps * c)
        p = np.exp(sh)
        p /= p.sum()
        g = np.zeros(n)
        for m_i in range(2 * n - 1):
            lag = m_i - (n - 1)
            w = p[m_i] * dx
            if w < 1e-14:
                continue
            if lag >= 0:
                g[lag:] += w * (1.0 - h[:n - lag])
                g[:n - lag] -= w * h[lag:]
            else:
                nl = -lag
                g[:n - nl] += w * (1.0 - h[nl:])
                g[nl:] -= w * h[:n - nl]
        return g
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0}]
    bnds = tuple((0.0, 1.0) for _ in range(n))
    res = minimize(obj, h0, method='SLSQP', jac=grad,
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x, eval_c5(res.x)

    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0}]
    bnds = tuple((0.0, 1.0) for _ in range(n))
    res = minimize(obj, h0, method='SLSQP', jac=grad,
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x, eval_c5(res.x)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')

    def time_left():
        return budget_s - (time.time() - t0)

    def try_solution(h):
        nonlocal best_h, best_c5
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    h_init_bin = None
    try:
        hv = np.array(initial_h_values, dtype=float)
        n0 = len(hv)
        h_init_bin = make_binary(hv, n0 // 2)
        try_solution(h_init_bin)
    except Exception:
        pass

    n_list = [70, 100, 140, 200, 300, 500]
    if h_init_bin is not None and len(h_init_bin) not in n_list:
        n_list.insert(0, len(h_init_bin))

    for n in n_list:
        if time_left() < 90:
            break
        n_ones = n // 2
        n_restarts = 8 if n <= 100 else (5 if n <= 200 else 3)

        for r in range(n_restarts):
            if time_left() < 30:
                break

            if h_init_bin is not None and r == 0:
                if n == len(h_init_bin):
                    h = h_init_bin.copy()
                else:
                    x_old = np.linspace(0.0, 2.0, len(h_init_bin) + 1)
                    x_new = np.linspace(0.0, 2.0, n + 1)
                    h_interp = np.interp(x_new[:-1], x_old[:-1], h_init_bin)
                    h = make_binary(h_interp, n_ones)
            elif r == 1 and n > 1:
                if best_h is not None:
                    x_old = np.linspace(0.0, 2.0, len(best_h))
                    x_new = np.linspace(0.0, 2.0, n)
                    h_interp = np.interp(x_new, x_old, best_h)
                    h = make_binary(h_interp, n_ones)
                else:
                    h = random_binary(n, n_ones, rng)
            else:
                h = random_binary(n, n_ones, rng)

            iters = min(300000, max(20000, int(time_left() * 25)))
            h_sa, c5_sa = sa_swap(h, rng, iters, 0.08, 0.002)
            try_solution(h_sa)

            if time_left() > 25:
                for eps_v in [5.0, 10.0, 20.0]:
                    if time_left() < 10:
                        break
                    h_sl, c5_sl = slsqp_opt(h_sa, eps_v, 300)
                    try_solution(h_sl)
                    h_b = make_binary(h_sl, n_ones)
                    try_solution(h_b)
                    if time_left() > 20:
                        h_sa2, c5_sa2 = sa_swap(h_b, rng, 80000, 0.04, 0.002)
                        try_solution(h_sa2)

    if best_h is None:
        n = 70
        best_h = random_binary(n, n // 2, rng)
        best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
