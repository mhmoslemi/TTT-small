def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(h, rng):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rem = rng.choice(ones, n_ones - target, replace=False)
        h[rem] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, target - n_ones, replace=False)
        h[add] = 1.0
    return h

def greedy_hill_climb(h, time_limit):
    n = len(h)
    h = h.astype(float).copy()
    target = n // 2
    h = make_binary(h, np.random.RandomState(0))
    best_c5 = evaluate_c5(h)

def _slsqp_obj(x):
    n = _opt_n
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    beta = 8.0
    m = np.max(corr)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - m))))

def _slsqp_eq(x):
    n = _opt_n
    return x.sum() - n / 2.0

def slsqp_optimize(h0, time_limit=120):

                      bounds=[(0.0, 1.0)] * n,
