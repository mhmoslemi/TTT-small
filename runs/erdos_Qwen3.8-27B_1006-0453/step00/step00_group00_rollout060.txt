import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(v, S):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_all(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_c5_for_lag(k, h, n, dx):
    """Gradient of dx * correlate(h, 1-h, 'full')[k] w.r.t. h"""
    grad = np.zeros(n)
    if k < n:
        # corr[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        for i in range(n - k):
            j = i + k  # index in h for the "a" part
            grad[j] += dx * (1.0 - h[i])
            if j < n:
                grad[i] += dx * (-h[j])
    else:
        # Use symmetry: correlate(h,1-h,'full')[k] = correlate(1-h,h,'full')[2n-2-k]
        # So we can compute as if it were correlate(1-h, h) at lag k'
        k2 = 2 * n - 2 - k
        # corr = sum_{i=0}^{n-1-k2} (1-h[i+k2])*h[i]  ... wait

        # So with a=1-h, v=h: sum_{i=0}^{n-1-k2} (1-h[i+k2])*h[i]
        for i in range(n - k2):
            j = i + k2
            # d/dh[j] of (1-h[j])*h[i] = -h[i]
            grad[j] += dx * (-h[i])
            # d/dh[i] of (1-h[j])*h[i] = (1-h[j])
            grad[i] += dx * (1.0 - h[j])
    return grad

def subgrad_descent(h, dx, S, lr=0.01, max_iter=5000, tol=1e-10):

    max_c5 = c5_vals.max()

def local_moves(h, dx, S, n, time_fn):
    c5 = c5_value(h, dx)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for s in steps:
        for _pass in range(50):
            if time_fn() < 20:
                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    for d in (s, -s):
                        hi, hj = h[i], h[j]
                        if hi + d < 0 or hi + d > 1 or hj - d < 0 or hj - d > 1:
                            continue
                        h[i] += d
                        h[j] -= d
                        c5_new = c5_value(h, dx)
                        if c5_new < c5 - 1e-12:
                            c5 = c5_new
                            improved = True
                        else:
                            h[i] = hi
                            h[j] = hj
            if not improved:
                break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n = len(h_init)
    dx = 2.0 / n
    S = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = h_init.copy()
    best_c5 = c5_value(best_h, dx)

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with penalty, various noise scales
    scales = [0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7]
    for scale in scales:
        if tleft() < 30:
            break
        for att in range(3):
            if tleft() < 20:
                break
            if scale == 0.0:
                h0 = best_h.copy()
            else:
                h0 = best_h + scale * rng.standard_normal(n)
                h0 = project_feasible(np.clip(h0, 0.0, 1.0), S)
            h_trial = do_lbfgs(h0, dx, S, n, bounds, pen=50.0, maxiter=2000)
            c5_trial = c5_value(h_trial, dx)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial

    # Phase 2: SLSQP with exact constraint
    if tleft() > 60:
        for att in range(10):
            if tleft() < 50:
                break
            noise = 0.02 * (att + 1)
            if att == 0:
                h0 = best_h.copy()
            else:
                h0 = best_h + noise * rng.standard_normal(n)
                h0 = project_feasible(np.clip(h0, 0.0, 1.0), S)
            h_trial = do_slsqp(h0, dx, S, n, bounds)
            c5_trial = c5_value(h_trial, dx)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial

    # Phase 3: Local coordinate moves
    if tleft() > 40:
        h_tmp = best_h.copy()
        best_h, best_c5 = local_moves(h_tmp, dx, S, n, tleft)

    # Phase 4: Try different n_points via interpolation
    if tleft() > 90:
        for n_try in [n - 1, n + 1, n - 2, n + 2, n + 5, n + 10, 2 * n]:
            if tleft() < 60 or n_try < 4:
                break
            dx_t = 2.0 / n_try
            S_t = n_try / 2.0
            bt_t = [(0.0, 1.0)] * n_try
            x_o = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_n = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_interp = np.interp(x_n, x_o, best_h, period=2.0)
            h_interp = project_feasible(h_interp, S_t)
            for att in range(3):
                if tleft() < 30:
                    break
                if att == 0:
                    h0 = h_interp.copy()
                else:
                    h0 = h_interp + 0.04 * (att) * rng.standard_normal(n_try)
                    h0 = project_feasible(np.clip(h0, 0.0, 1.0), S_t)
                h_trial = do_lbfgs(h0, dx_t, S_t, n_try, bt_t, pen=50.0, maxiter=1500)
                c5_trial = c5_value(h_trial, dx_t)
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                    n = n_try
                    dx = dx_t
                    S = S_t
                    bounds = bt_t
            # Local moves at new n if improved
            if tleft() > 30 and len(best_h) == n_try:
                h_tmp = best_h.copy()
                best_h, best_c5 = local_moves(h_tmp, dx, S, n_try, tleft)

    # Phase 5: Final polish
    if tleft() > 30:
        h_trial = do_lbfgs(best_h, dx, S, n, bounds, pen=200.0, maxiter=3000)
        c5_trial = c5_value(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
        if tleft() > 20:
            h_tmp = best_h.copy()
            best_h, best_c5 = local_moves(h_tmp, dx, S, n, tleft)

    # Final clip for safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = c5_value(best_h, 2.0 / n)

    return (best_h, best_c5, n)
