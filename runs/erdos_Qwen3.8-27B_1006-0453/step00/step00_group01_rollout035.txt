import time

def compute_c5(h, dx):
    return float(dx * np.max(np.correlate(h, 1.0 - h, mode='full')))

    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h
