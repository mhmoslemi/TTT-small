import numpy as np
import time
from scipy.optimize import minimize

def _eval_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_obj(h, n, st, pw):
    dx = 2.0 / n
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    ex = np.exp(corr - cm)
    sz = np.sum(ex)
    w = ex / sz
    f = (cm + np.log(sz)) * dx / st + pw * (np.sum(h) - n * 0.5) ** 2
    return f

def _smooth_grad(h, n, st, pw):
    dx = 2.0 / n
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    ex = np.exp(corr - cm)
    sz = np.sum(ex)
    w = ex / sz
    s = np.sum(h)
    tgt = n * 0.5
    shifts = np.arange(len(corr)) - (n - 1)
    g = np.empty(n)
    for j in range(n):
        va = j - shifts
        oka = (va >= 0) & (va < n)
        A = np.dot(w[oka], h[va[oka]]) if np.any(oka) else 0.0
        vb = j + shifts
        okb = (vb >= 0) & (vb < n)
        B = np.dot(w[okb], h[vb[okb]]) if np.any(okb) else 0.0
        g[j] = dx / st * (1.0 - A - B) + 2.0 * pw * (s - tgt)
    return g

def _project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(100):
        s = np.sum(h)
        diff = tgt - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if not np.any(mask):
            mask = np.ones(n, dtype=bool)
        cnt = np.count_nonzero(mask)
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def _eval_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_weights(h, n, st):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    ex = np.exp(corr - cm)
    sz = ex.sum()
    w = ex / sz
    return corr, cm, sz, w

def _smooth_obj(x, n, st, pw):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr, cm, sz, w = _corr_weights(h, n, st)
    f = (cm + np.log(sz)) * dx / st
    s = h.sum()
    f += pw * (s - 0.5 * n) ** 2
    return f

def _smooth_grad(x, n, st, pw):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr, cm, sz, w = _corr_weights(h, n, st)
    s = h.sum()
    tgt = 0.5 * n
    shifts = np.arange(2 * n - 1) - (n - 1)
    g = np.empty(n)
    for j in range(n):
        va = j - shifts
        oka = (va >= 0) & (va < n)
        A = float(np.dot(w[oka], h[va[oka]])) if oka.any() else 0.0
        vb = j + shifts
        okb = (vb >= 0) & (vb < n)
        B = float(np.dot(w[okb], h[vb[okb]])) if okb.any() else 0.0
        g[j] = dx / st * (1.0 - A - B) + 2.0 * pw * (s - tgt)
    return g
