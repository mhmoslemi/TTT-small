import numpy as np
import time

def _compute_comp(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(_compute_comp(h)) * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        tau = (lo + hi) / 2.0
        s = np.sum(np.clip(h + tau, 0, 1))
        if s < target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h + tau, 0, 1)

def _grad_c5_soft(h, n, t):
    comp = _compute_comp(h)  # length 2n-1
    dx = 2.0 / n
    vals = comp * dx
    # Soft max weights
    vmax = np.max(vals)
    exps = np.exp(t * (vals - vmax))
    w = exps / np.sum(exps)
    # Gradient
    grad = np.zeros(n)
    for ind in range(2 * n - 1):
        wl = w[ind]
        if wl < 1e-15:
            continue
        l = ind - (n - 1)  # lag
        if l >= 0:
            # (1-h[m+l]) for m=0..n-1-l
            if n - 1 - l >= 0:
                grad[:n-l] += wl * (1.0 - h[l:n])
            # -h[m-l] for m=l..n-1
            if l < n:
                grad[l:] -= wl * h[:n-l]
        else:
            ll = -l
            # (1-h[m-ll]) for m=ll..n-1
            grad[ll:] += wl * (1.0 - h[:n-ll])
            # -h[m+ll] for m=0..n-1-ll
            grad[:n-ll] -= wl * h[ll:n]
    return grad * dx

def _optimize_softmax(h, n, time_budget, target_c5):
    best_h = h.copy()
    best_c5 = _c5(h)
    t = 10.0  # softness parameter
    lr = 0.1
    start = time.time()
    
    for step in range(5000):
        if time.time() - start > time_budget:
            break
        if _c5(best_h) <= target_c5:
            break
        
        grad = _grad_c5_soft(h, n, t)
        # Gradient descent step
        h = h - lr * grad
        h = _project(h, n)
        
        c5 = _c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive learning rate
        if step % 100 == 0:
            if step > 0 and best_c5 < _c5(best_h):
                break
            lr = max(0.001, lr * 0.9)
            t = min(50.0, t * 1.05)
    
    return best_h, best_c5

            j_range = slice(0, n - jm)

import numpy as np
import time

def _c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(comp)) * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + tau, 0.0, 1.0))
        if s < target:
            lo = tau
        else:
            hi = tau
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _grad_soft(h, n, t):
    comp = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    vals = comp * dx
    vm = np.max(vals)
    ex = np.exp(t * (vals - vm))
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for ind in range(2 * n - 1):
        wl = w[ind]
        if wl < 1e-16:
            continue
        m = ind - (n - 1)
        if m >= 0:
            if n - m > 0:
                grad[:n - m] += wl * (1.0 - h[m:])
                grad[m:] -= wl * h[:n - m]
        else:
            jm = -m
            if n - jm > 0:
                grad[jm:] += wl * (1.0 - h[:n - jm])
                grad[:n - jm] -= wl * h[jm:]
    return grad * dx

        g = _grad_soft(h, n, t)

            t = min(t_max, t * 1.12)

def _bin_search(h, n, max_time, rng):

    h = np.zeros(n)

        improved = False
