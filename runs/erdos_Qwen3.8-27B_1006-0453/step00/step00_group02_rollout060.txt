import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - result.sum()
    free = (result > 1e-10) & (result < 1.0 - 1e-10)
    if np.any(free) and abs(diff) > 1e-13:
        result[free] += diff / np.sum(free)
        result = np.clip(result, 0.0, 1.0)
    return result

def grad_for_lag(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            g[k:n] += (1.0 - h[0:nk])
            g[0:nk] -= h[k:n]
    else:
        s = -k
        ns = n - s
        if ns > 0:
            g[0:ns] += (1.0 - h[s:n])
            g[s:n] -= h[0:ns]
    return g

def smooth_gradient(h, temp, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr - corr.max()
    ec = np.exp(c / temp)
    w = ec / ec.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-8:
            continue
        k = idx - (n - 1)
        grad += wi * grad_for_lag(h, k, n)
    grad *= dx
    return grad

def optimize_at_res(h0, n, time_budget, seed=42):
    rng = np.random.default_rng(seed)
    target_sum = n / 2.0

    best_c5 = compute_c5(h)

    t0 = time.time()
    temp = 0.03
    step = 0.08
    no_improve = 0
    min_no_improve = 200

    for it in range(50000):
        if time.time() - t0 > time_budget:
            break
        grad = smooth_gradient(h, temp, n)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-14:
            break
        grad_norm = grad / gnorm
        h_new = h - step * grad_norm
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if it % 25 == 24:
            temp = max(0.0005, temp * 0.65)
            step = max(0.002, step * 0.88)
            if no_improve > min_no_improve:
                h = best_h + rng.normal(0, 0.03, n)
                h = project_box_sum(h, target_sum)
                no_improve = 0

def upscale(h, n_old, n_new):

    h_new = np.interp(x_new, x_old, h)
