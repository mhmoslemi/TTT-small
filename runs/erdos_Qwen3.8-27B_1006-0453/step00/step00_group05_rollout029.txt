import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def sa_refine(h, time_limit, seed=42, T0=0.01, Tend=0.00003):
    rng = np.random.RandomState(seed)
    h = (np.array(h, dtype=np.float64) > 0.5).astype(np.float64).copy()
    n = len(h)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t0 = time.time()
    no_improve = 0
    max_no_improve = 80000 if n <= 140 else 200000

    while time.time() - t0 < time_limit and no_improve < max_no_improve:
        frac = min((time.time() - t0) / time_limit, 1.0)
        temp = T0 * (Tend / T0) ** frac

        ones_idx = np.flatnonzero(h == 1.0)
        zeros_idx = np.flatnonzero(h == 0.0)
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]

            if new_c5 < best_c5 - 1e-12:

                no_improve = 0
