import numpy as np
import time
import scipy.optimize as so

_dx_g = 0.0
_S_g = 0.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, S):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = S - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = int(np.sum(m))
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * _dx_g)

def _con(x):
    return float(np.sum(x) - _S_g)

def pair_refine(h, dx, n, c5, tl):

        (30, [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.15, -0.15]),

    ]
    for nsw, deltas in phases:
        for _ in range(nsw):

                return h, c5

                for j in range(i + 1, n):
