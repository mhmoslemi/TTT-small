        cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
        res = minimize(obj_gp, best_h, jac=grad_gp, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-12})
        h_fin = np.clip(res.x, 0, 1)
        h_fin = project_sum(h_fin, n_best / 2.0)
        c5_fin = true_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Ensure exact constraint
    n_final = len(best_h)
    best_h = project_sum(best_h, n_final / 2.0)
    best_c5 = true_c5(best_h)
    return (best_h, best_c5, n_final)
