                             h_r, method='SLSQP', bounds=bnds,
                             constraints=cons,
                             options={'maxiter': 150, 'ftol': 1e-13})
                if r.x is not None and np.all(np.isfinite(r.x)):
                    h_new = repair_sum(np.clip(r.x, 0, 1), target)
                    c5 = compute_c5(h_new, n0)
                    if c5 < best_c5:
                        best_c5, best_h, best_n = c5, h_new.copy(), n0
                    h_r = h_new
            except Exception:
                pass
        rem = budget_s - (time.time() - t_start)
        if rem > 20:
            h_r, c5_r = local_search(h_r, n0, min(rem - 10, 30), seed + 2000)
            if c5_r < best_c5:
                best_c5, best_h, best_n = c5_r, h_r.copy(), n0

    # Final verification
    best_h = repair_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
