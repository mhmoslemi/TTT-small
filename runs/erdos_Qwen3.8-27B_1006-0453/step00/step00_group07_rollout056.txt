import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < 1e-15:
        return m
    s = np.clip((corr - m) / max(T, 1e-12), -60, 60)
    return m + T * np.log(np.sum(np.exp(s)))

def enforce_constraints(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    adj = (h > 1e-9) & (h < 1.0 - 1e-9)
    if np.any(adj):
        h[adj] += diff / np.count_nonzero(adj)
        h = np.clip(h, 0.0, 1.0)
    else:
        h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def slsqp_step(h, n, target_sum, T):
    def obj(x):
        return smooth_max_obj(x, T)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(x.shape)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return enforce_constraints(res.x, target_sum)
    return h

def local_search(h, n, target_sum, rng, t_end, max_iter=50000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T_sa = 0.0008

    for _ in range(max_iter):
        if time.time() > t_end:
            break
        i = int(rng.integers(n))
        j = (i + int(rng.integers(1, n))) % n
        d = float(rng.uniform(-0.06, 0.06))
        h_try = h.copy()
        new_i = h[i] + d
        new_j = h[j] - d
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            d *= 0.5
            new_i = h[i] + d
            new_j = h[j] - d
            if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
                no_improve += 1
                continue
        h_try[i] = new_i
        h_try[j] = new_j
        new_c5 = compute_c5(h_try)
        if new_c5 <= cur_c5:
            h = h_try
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_c5 - cur_c5) / max(T_sa, 1e-12)):
                h = h_try
                cur_c5 = new_c5
            no_improve += 1
        T_sa = max(T_sa * 0.9995, 1e-10)
        if no_improve > len(h) * 200:
            break
    return best_h, best_c5
