    return project_feasible(h_new, len(h_new))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.astype(float).copy()
    n = len(h_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: PTD at n=70 with multiple restarts
    n_start = 8
    for attempt in range(n_start):
        if time_left() < 120:
            break
        budget_this = min(55, time_left() - 110)
        if budget_this < 5:
            break

        if attempt == 0:
            h0 = h_init.copy()
        elif attempt <= 2:
            h0 = project_feasible(h_init + 0.04 * (attempt + 1) * np.random.randn(n), n)
        elif attempt <= 4:
            h0 = project_feasible(h_init + 0.1 * np.random.randn(n), n)
        else:
            h0 = project_feasible(np.random.random(n), n)

        h_opt, c_opt = pgd(h0, budget_this, lr0=0.5, seed=seed + attempt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Also try SLSQP on best found
    if time_left() > 150:
        h_opt, c_opt = slsqp_optimize(best_h, n, maxiter=200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 2: Upscale to n=140
    if time_left() > 150:
        for trial in range(3):
            if time_left() < 130:
                break
            budget_this = min(120, time_left() - 120)
            if budget_this < 10:
                break
            if trial == 0:
                h0 = upscale(best_h, 2)
            elif trial == 1:
                h0 = project_feasible(upscale(best_h, 2) + 0.04 * np.random.randn(len(best_h) * 2), len(best_h) * 2)
            else:
                h0 = project_feasible(upscale(best_h, 2) + 0.1 * np.random.randn(len(best_h) * 2), len(best_h) * 2)
            h_opt, c_opt = pgd(h0, budget_this, lr0=0.3, mom=0.7, seed=seed + 100 + trial)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

        # SLSQP at n=140
        if time_left() > 100:
            h_opt, c_opt = slsqp_optimize(best_h, len(best_h), maxiter=150)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    # Phase 3: Upscale to n=280
    if time_left() > 100:
        n_cur = len(best_h)
        budget_this = min(150, time_left() - 50)
        if budget_this > 20:
            h0 = upscale(best_h, 2)
            h_opt, c_opt = pgd(h0, budget_this, lr0=0.2, mom=0.7, seed=seed + 200)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    # Final polish
    if time_left() > 15:
        budget_this = min(30, time_left() - 5)
        if budget_this > 3:
            h_opt, c_opt = pgd(best_h, budget_this, lr0=0.05, mom=0.5, seed=seed + 300)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    return best_h, best_c, len(best_h)
