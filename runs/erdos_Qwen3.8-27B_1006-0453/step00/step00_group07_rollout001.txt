import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            idx = np.argsort(h)
            for k in idx:
                if diff2 <= 0:
                    break
                add = min(1.0 - h[k], diff2)
                h[k] += add
                diff2 -= add
            idx_r = idx[::-1]
            for k in idx_r:
                if diff2 >= 0:
                    break
                sub = min(h[k], -diff2)
                h[k] -= sub
                diff2 += sub
    return h

def sa_optimize(h, n, dx, target_sum, rng, t_limit, T_init, T_min, cooling, batch):
    deadline = time.time() + t_limit
    h = h.copy()
    current_c = compute_c5(h, dx)
    best_h = h.copy()
    best_c = current_c
    T = T_init
    stagnation = 0

    while T > T_min and time.time() < deadline:
        sigma = 0.05 + 0.8 * (T / T_init)
        for _ in range(batch):
            if time.time() > deadline:
                return best_h, best_c
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

            h[j] -= delta

                h[i] -= delta
                h[j] += delta
                continue
            new_c = compute_c5(h, dx)
            dc = new_c - current_c
            if dc < 0.0 or rng.random() < np.exp(-dc / T):
                current_c = new_c
                if new_c < best_c:

                else:

def polish(h, n, dx, target_sum, rng, t_limit):
    deadline = time.time() + t_limit
    h = h.copy()
    current_c = compute_c5(h, dx)
    best_h = h.copy()
    best_c = current_c
    for step in range(200):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for sign in (1, -1):
                for mag in (0.01, 0.005, 0.001):
                    delta = sign * mag
                    h[i] += delta
                    if h[i] < 0.0 or h[i] > 1.0:
                        h[i] -= delta
                        continue
                    # Distribute -delta across other indices
                    remaining = -delta
                    others = [j for j in range(n) if j != i]
                    rng.shuffle(others)
                    for j in others:
                        if abs(remaining) < 1e-12:
                            break
                        if remaining > 0:
                            add = min(1.0 - h[j], remaining)
                            h[j] += add
                            remaining -= add
                        else:
                            sub = min(h[j], -remaining)
                            h[j] -= sub
                            remaining += sub

                    if new_c < current_c - 1e-12:

                        improved = True
                    else:

                else:
