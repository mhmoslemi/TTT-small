    s = e.sum()

    val = m + (1.0 / alpha) * np.log(s)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-16:
            continue
        sft = k - (n - 1)
        if sft >= 0:
            m2 = n - sft
            g[sft:sft + m2] += wi * dx * (1.0 - h[:m2])
            g[:m2] -= wi * dx * h[sft:sft + m2]
        else:
            t = -sft
            m2 = n - t
            g[:m2] += wi * dx * (1.0 - h[t:t + m2])
            g[t:t + m2] -= wi * dx * h[:m2]
    return val, g


def project_simplex(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    p = np.clip(h - t, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-14:
        free = (p > 1e-10) & (p < 1.0 - 1e-10)
        if free.sum() > 0:
            p[free] += diff / free.sum()
        else:
            p += diff / n
        p = np.clip(p, 0.0, 1.0)
    return p


def _slsqp_solve(h_init, n, alpha, maxiter):
    global _target_sum, _n_pts
    _target_sum = n / 2.0
    _n_pts = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    def f(x):
        v, _ = smooth_obj_grad(x, alpha)
        return v

    def df(x):
        _, g = smooth_obj_grad(x, alpha)
        return g

    res = minimize(f, h_init, jac=df, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return res


def _subgradient_polish(h, n, steps, lr0):
    global _target_sum, _n_pts
    _target_sum = n / 2.0
    _n_pts = n
    best = h.copy()
    best_c = compute_c5(best)
    lr = lr0
    for _ in range(steps):
        _, g = smooth_obj_grad(best, 2000)
        h_try = project_simplex(best - lr * g, n)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best, best_c = h_try, c_try
            lr = min(lr * 1.2, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-7:
                break
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    global _target_sum, _n_pts
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(47, 0.5)

    n = len(h0)
    _target_sum = n / 2.0
    _n_pts = n
    best_h = project_simplex(h0, n)
    best_c = compute_c5(best_h)

    # Main SLSQP from initial
    res = _slsqp_solve(best_h, n, 50, 1000)
    if time_left() < 10:
        return best_h, best_c, n
    h_new = project_simplex(res.x, n)
    c_new = compute_c5(h_new)
    if c_new < best_c:
        best_h, best_c = h_new, c_new

    # Polish
    best_h, best_c = _subgradient_polish(best_h, n, 80, 0.01)

    # SLSQP with tighter alpha
    res2 = _slsqp_solve(best_h, n, 100, 500)
    if time_left() < 30:
        return best_h, best_c, n
    h_new = project_simplex(res2.x, n)
    c_new = compute_c5(h_new)
    if c_new < best_c:
        best_h, best_c = h_new, c_new
    best_h, best_c = _subgradient_polish(best_h, n, 50, 0.005)

    # Random restarts near best
    for trial in range(40):
        if time_left() < 60:
            break
        scale = 0.02 + 0.03 * (trial % 4)
        h_try = project_simplex(best_h + rng.randn(n) * scale, n)
        res3 = _slsqp_solve(h_try, n, 50, 400)
        if time_left() < 30:
            break
        h_new = project_simplex(res3.x, n)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
            best_h, best_c = _subgradient_polish(best_h, n, 40, 0.005)

    # Try larger n (double)
    if time_left() > 180:
        n2 = n * 2
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_simplex(h2, n2)
        c2 = compute_c5(h2)
        res4 = _slsqp_solve(h2, n2, 50, 600)
        if time_left() > 30:
            h_new = project_simplex(res4.x, n2)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
                best_h, best_c = _subgradient_polish(best_h, n2, 50, 0.005)

    # Try n=50 as alternative
    if time_left() > 120:
        n_alt = 50
        h_alt = np.interp(np.linspace(0, 2, n_alt, endpoint=False),
                          np.linspace(0, 2, n, endpoint=False), best_h)
        h_alt = project_simplex(h_alt, n_alt)
        res5 = _slsqp_solve(h_alt, n_alt, 50, 600)
        if time_left() > 20:
            h_new = project_simplex(res5.x, n_alt)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Final verification
    best_c = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
