def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_objective(h, dx, eps):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    if cmax < 1e-12:
        return 0.0
    shifted = (c - cmax) / eps
    shifted -= np.max(shifted)  # additional stability
    return dx * (cmax + eps * np.log(np.sum(np.exp(shifted))))

    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:
        return h
    h = h.copy()
    h[mask] += diff / count
    return np.clip(h, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj(h, dx, eps):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    cs = (c - cmax) / eps
    m = cs.max()
    return float(dx * (cmax + eps * np.log(np.sum(np.exp(cs - m)))))

def con_sum(h, target):
    return np.sum(h) - target

def adjust_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def upsample_to(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = np.arange(n) * (float(n_new) / n)
    x_new = np.arange(n_new, dtype=float)
    return np.interp(x_new, x_old, h)

def make_candidate(h_src, n, rng, noise_scale):
    h = upsample_to(h_src, n)
    if noise_scale > 0:
        h = h + noise_scale * rng.randn(n)
    return adjust_sum(h, n / 2.0)
