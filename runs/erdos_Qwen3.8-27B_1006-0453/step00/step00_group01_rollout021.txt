import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def fix_sum(h, target):
    h = h.copy()
    for _ in range(2):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = int(m.sum())
        if c > 0:
            h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h


def interp_h(hs, ns, nd):
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return fix_sum(np.clip(np.interp(xd, xs, hs), 0, 1), nd / 2.0)
