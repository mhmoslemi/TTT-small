def compute_score(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')

def gradient_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        a = j - shift
        if 0 <= a < n:
            grad[j] += 1.0 - h[a]
        b = j + shift
        if 0 <= b < n:
            grad[j] -= h[b]
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0

        if s > target:

        else:
