                best = h.copy()

            h[b] = 0

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def fix_binary_sum(h, target):
    h = np.clip(np.round(h), 0, 1)
    cur = int(np.sum(h))
    if cur < target:
        idx = np.where(h == 0)[0]
        for i in range(target - cur):
            h[idx[i]] = 1
    elif cur > target:
        idx = np.where(h == 1)[0]
        for i in range(cur - target):
            h[idx[i]] = 0
    return h

def greedy_ls(h, deadline):
    h = h.copy()
    best_mc = compute_max_corr(h)
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_pair = None
        best_new = best_mc
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                mc = compute_max_corr(h)
                h[i] = 1
                h[j] = 0

            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
            best_mc = best_new
        else:
            break
