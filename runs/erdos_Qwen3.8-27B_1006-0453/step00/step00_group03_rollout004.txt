import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, alpha):
    xm = np.max(x)
    return float(xm + np.log(np.sum(np.exp(alpha * (x - xm)))) / alpha)

def make_obj(n, alpha):
    dx = 2.0 / n
    def obj(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr * dx, alpha)
    return obj

def make_cons(n):
    def eq(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    return {'type': 'eq', 'fun': eq, 'jac': eq_jac}

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            room = np.sum(1.0 - h[mask])
        else:
            mask = h > 0.0
            room = np.sum(h[mask])
        if room < 1e-14:
            break
        step = min(1.0, 2.0 * abs(diff) / room)
        h[mask] += step * diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.arange(n_src) + 0.5
    x_dst = np.arange(n_dst) + 0.5
    f = interp1d(x_src, h_src, kind='cubic', fill_value='extrapolate')
    h_new = f(x_dst)
    return project_feasible(h_new, n_dst)

def coordinate_descent(h, n, max_sweeps=10, step=0.05):
    best_c5 = compute_c5_value(h)
    vals = np.arange(0.0, 1.01, step)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            orig = h[i]
            for v in vals:
                h[i] = v
                h[f] = np.clip(h, 0, 1)
                s = np.sum(h)
                if abs(s - n/2.0) > 1e-6:
                    h[i] = orig
                    continue
                c5v = compute_c5_value(h)
                if c5v < best_c5 - 1e-10:
                    best_c5 = c5v
                    improved = True
            if not improved:
                break
    h[f""] = None
    return h, best_c5

    rng = np.random.RandomState(seed)

    h_parent = project_feasible(h_parent, n_parent)
    best_c5 = compute_c5_value(h_parent)
    best_h = h_parent.copy()

    resolutions = sorted(set([n_parent, min(n_parent * 2, 200), min(n_parent * 3, 350)]))
    resolutions = [r for r in resolutions if r <= 500]

    for n in resolutions:
        t_rem = budget_s - (time.time() - t0)
        if t_rem < 25:
            break

        guesses = []
        guesses.append(interpolate_to(h_parent, n))
        if len(best_h) != n:
            guesses.append(interpolate_to(best_h, n))
        elif len(best_h) == n:
            guesses.append(best_h.copy())

        for _ in range(2):
            h_r = 0.5 + 0.35 * rng.randn(n)
            guesses.append(project_feasible(h_r, n))

        x = np.linspace(0, 4 * np.pi, n)

            best_local = compute_c5_value(h_cd)

                v = float(np.clip(v, 0, 1))
                h_cd[i] = v
                s = np.sum(h_cd)
                if abs(s - n_final / 2.0) > 1e-8:
                    continue
                c5v = compute_c5_value(h_cd)

                    best_local = c5v
            if best_local < best_c5 - 1e-10:

                best_c5 = compute_c5_value(h_cd)
                best_h = h_cd.copy()
            else:
                h_cd[i] = orig

    # Final projection
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5_value(best_h)
    return (best_h, final_c5, len(best_h))
