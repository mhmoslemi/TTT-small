    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    grad = np.zeros(n)

            grad[j] += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            grad[j] += -h[j + lag]

    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)

    if diff > 0:
        room = 1.0 - h
    else:
        room = h
    total = room.sum()
    if total > 1e-14:
        h += np.sign(diff) * room * (abs(diff) / total)
    h = np.clip(h, 0.0, 1.0)
    # Exact fix

        order = np.argsort(room)[::-1] if diff > 0 else np.argsort(h)[::-1]
        for i in order:

                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:

                h[i] -= sub
                diff += sub
