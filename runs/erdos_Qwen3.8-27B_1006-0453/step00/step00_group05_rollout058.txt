import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def obj_func(h, n):
    return evaluate_c5(h, n)

def constraint_sum(h, n):
    return float(np.sum(h) - n / 2.0)

def adjust_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 0.999
    else:
        mask = h > 0.001
    count = np.sum(mask)
    if count == 0:
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        h[mask] += diff / count
        h = np.clip(h, 0, 1)
    return h

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    best_c5 = np.inf
    best_h = None
    best_n = 70
    
    # Get initial values

    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        best_n = len(h0)
        best_h = adjust_sum(h0, best_n / 2.0)
        best_c5 = evaluate_c5(best_h, best_n)
    else:
        best_n = 70
        best_h = np.full(best_n, 0.5)
        best_c5 = evaluate_c5(best_h, best_n)
    
    def try_optimize(h_init, n, maxiter):
        nonlocal best_c5, best_h, best_n
        if time.time() - start_time > budget_s * 0.92:
            return
        target = n / 2.0
        h_init = adjust_sum(h_init.copy(), target)
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
        try:

                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
            if res.x is not None:

                c5 = evaluate_c5(h_opt, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n
        except Exception:
            pass

    try_optimize(best_h, best_n, 300)

        if r == 0:

        elif r == 1:

            idx = rng.choice(best_n, size=best_n//4, replace=False)
            h_try[idx] = rng.uniform(0, 1, size=len(idx))
        elif r == 2:

        else:
            h_try = rng.uniform(0, 1, size=best_n)

    best_h = adjust_sum(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h, best_n)

But the max in C5 is non-smooth, which L-BFGS-B doesn't handle well. SLSQP uses numerical gradients and should be more robust.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def obj_func(h, n):
    return evaluate_c5(h, n)

def constraint_sum(h, n):
    return float(np.sum(h) - n / 2.0)

def adjust_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 0.999
    else:
        mask = h > 0.001
    count = int(np.sum(mask))
    if count == 0:
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        h = h.copy()
        h[mask] += diff / count
        h = np.clip(h, 0, 1)
    return h

def upsample_linear(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def optimize_point(h_init, n, maxiter):
    target = n / 2.0
    h_init = adjust_sum(h_init.copy(), target)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
    try:
        res = minimize(obj_func, h_init, method='SLSQP', args=(n,),
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_opt = adjust_sum(res.x, target)
            c5 = evaluate_c5(h_opt, n)
            return h_opt, c5
    except Exception:
        pass
    return None, np.inf

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_c5 = np.inf
    best_h = None
    best_n = 70

    global initial_h_values
    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        best_n = len(h0)
        best_h = adjust_sum(h0, best_n / 2.0)
        best_c5 = evaluate_c5(best_h, best_n)
    else:
        best_n = 70
        best_h = np.full(best_n, 0.5)
        best_c5 = evaluate_c5(best_h, best_n)

    # Phase 1: Optimize from the initial construction
    h_opt, c5 = optimize_point(best_h, best_n, 400)
    if h_opt is not None and c5 < best_c5:
        best_c5, best_h = c5, h_opt

    # Phase 2: Try higher resolutions by upsampling
    for factor in [2, 3, 4, 5, 6, 8]:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.55:
            break
        new_n = best_n * factor
        if new_n > 1200 or new_n == best_n:
            continue
        h_up = upsample_linear(best_h, best_n, new_n)
        h_up = adjust_sum(h_up, new_n / 2.0)
        h_opt, c5 = optimize_point(h_up, new_n, 200)
        if h_opt is not None and c5 < best_c5:
            best_c5, best_h, best_n = c5, h_opt, new_n

    # Phase 3: Randomized restarts at best_n
    elapsed = time.time() - start_time
    remaining = budget_s * 0.92 - elapsed
    n_restarts = max(1, int(remaining / 8.0))
    n_restarts = min(n_restarts, 120)

    for trial in range(n_restarts):
        if time.time() - start_time > budget_s * 0.92:
            break
        r = trial % 6
        if r == 0:
            h_try = best_h + rng.normal(0, 0.08, size=best_n)
        elif r == 1:
            h_try = best_h.copy()
            k = max(2, best_n // 6)
            idx = rng.choice(best_n, size=k, replace=False)
            h_try[idx] = rng.uniform(0, 1, size=k)
        elif r == 2:
            phase = rng.uniform(0, 2 * np.pi)
            h_try = 0.5 + 0.25 * np.sin(np.linspace(0, 4 * np.pi, best_n) + phase)
        elif r == 3:
            phase = rng.uniform(0, 2 * np.pi)
            h_try = 0.5 + 0.2 * np.cos(np.linspace(0, 7 * np.pi, best_n) + phase)
        elif r == 4:
            blend = rng.uniform(0.1, 0.4)
            h_try = blend * best_h + (1.0 - blend) * rng.uniform(0, 1, size=best_n)
        else:
            h_try = rng.uniform(0.1, 0.9, size=best_n)

        h_try = adjust_sum(h_try, best_n / 2.0)
        h_opt, c5 = optimize_point(h_try, best_n, 200)
        if h_opt is not None and c5 < best_c5:
            best_c5, best_h = c5, h_opt

    # Phase 4: One more upsample attempt if time remains
    if time.time() - start_time < budget_s * 0.9:
        alt_n = best_n * 2
        if alt_n <= 1200:
            h_alt = upsample_linear(best_h, best_n, alt_n)
            h_alt = adjust_sum(h_alt, alt_n / 2.0)
            h_opt, c5 = optimize_point(h_alt, alt_n, 150)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, alt_n

    # Final validation
    best_h = adjust_sum(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
