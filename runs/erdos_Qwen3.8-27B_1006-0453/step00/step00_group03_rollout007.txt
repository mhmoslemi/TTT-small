import numpy as np
import time
from scipy.optimize import minimize

_N = 70

def _obj(x):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    beta = 40.0
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def _con(x):
    return np.sum(x) - _N / 2.0

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(h, n):
    h = h.astype(float).copy()
    h += (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def _make_starts(h_init, n):

    if h_init is not None:
        xo = np.linspace(0, 2, len(h_init), endpoint=False)
        xn = np.linspace(0, 2, n, endpoint=False)

        starts.append(_project(hr, n))
