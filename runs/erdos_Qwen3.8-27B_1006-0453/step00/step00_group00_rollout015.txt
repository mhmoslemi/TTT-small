import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def repair_h(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
            continue
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def penalty_obj(x, n, lam):
    x = np.clip(x, 0.0, 1.0)
    c5 = compute_c5(x, n)
    sp = lam * (np.sum(x) - n / 2.0) ** 2
    return c5 + sp

def smooth_obj(x, n, beta):
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    shifted = corr - np.max(corr)
    return np.max(corr) + (1.0 / beta) * np.log(np.sum(np.exp(beta * shifted)))

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and initial_h_values is not None:

        n = 45
        best_h = np.full(n, 0.5)

    best_c5 = compute_c5(best_h, n)

    # Phase 1: Nelder-Mead with penalty

        if time.time() - t0 > budget_s * 0.45:

            hc = repair_h(res.x, n)

            pass
