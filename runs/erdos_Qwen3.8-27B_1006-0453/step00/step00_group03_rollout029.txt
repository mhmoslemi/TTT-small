import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_h(x, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _con_fun(x):
    n = len(x)
    return x.sum() - float(n) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def do_slsqp(h_init, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_h, h_init, args=(n,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_cand = project_feasible(res.x, n)
        return h_cand, compute_c5(h_cand, 2.0 / n)
    except Exception:
        return None, float('inf')

def do_coord_descent(h_init, n, t0, budget_s, reserve, rng_seed):
    np.random.seed(rng_seed)
    dx = 2.0 / n
    h = h_init.copy()
    c5_curr = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_curr
    temp = 0.0005
    min_temp = 1e-8
    no_improve = 0

    while time.time() < budget_s - reserve:
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        delta = np.random.uniform(-0.05, 0.05)
        hi = np.clip(h[i] + delta, 0.0, 1.0)
        hj = np.clip(h[j] - (hi - h[i]), 0.0, 1.0)
        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj
        if abs(h_new.sum() - n / 2.0) > 1e-10:
            continue
        c5_new = compute_c5(h_new, dx)
        diff = c5_new - c5_curr
        if diff < 0 or (diff < 0.005 and np.random.rand() < np.exp(-diff / temp)):
            h = h_new
            c5_curr = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
                temp = max(temp * 0.999, min_temp)
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 200000:
            h = best_h.copy()
            c5_curr = best_c5
            no_improve = 0
            temp = 0.0005
    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    global initial_h_values
    n0 = len(initial_h_values)
    h_best = project_feasible(np.array(initial_h_values, dtype=float), n0)
    c5_best = compute_c5(h_best, 2.0 / n0)
    best_n = n0

    # Phase 1: SLSQP polish on initial
    if time.time() - t0 < budget_s - 600:
        h_c, c5_c = do_slsqp(h_best, n0, 8000, 1e-14)
        if c5_c < c5_best:
            c5_best, h_best = c5_c, h_c

    # Phase 2: Try other resolutions
    for n_try in [100, 140, 168, 175, 200]:
        if time.time() - t0 > budget_s - 400:
            break
        h_init = interpolate_h(h_best, best_n, n_try)
        h_c, c5_c = do_slsqp(h_init, n_try, 2500, 1e-12)
        if c5_c < c5_best:
            c5_best = c5_c
            h_best = h_c
            best_n = n_try

    # Phase 3: Coordinate descent with simulated annealing
    h_cd, c5_cd = do_coord_descent(h_best, best_n, t0, budget_s, 100, seed + 7)
    if c5_cd < c5_best:
        c5_best, h_best = c5_cd, h_cd

    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s - 25:
        h_c, c5_c = do_slsqp(h_best, best_n, 5000, 1e-14)
        if c5_c < c5_best:
            c5_best, h_best = c5_c, h_c

    # Phase 5: Short coordinate descent after final SLSQP
    if time.time() - t0 < budget_s - 10:
        h_cd2, c5_cd2 = do_coord_descent(h_best, best_n, t0, budget_s, 5, seed + 13)
        if c5_cd2 < c5_best:
            c5_best, h_best = c5_cd2, h_cd2

    # Final projection and evaluation
    h_best = project_feasible(h_best, best_n)
    c5_final = compute_c5(h_best, 2.0 / best_n)
    return h_best, c5_final, best_n
