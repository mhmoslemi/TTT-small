    # Continuous refinement on best
    if best_h is not None and remaining() > 40:
        h = best_h.copy()
        n = len(h)
        c5 = best_c5
        for _ in range(200000):
            if remaining() < 5:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = float(rng.uniform(-0.04, 0.04))
            old_i, old_j = h[i], h[j]
            new_i = min(1.0, max(0.0, h[i] + delta))
            new_j = h[j] - (new_i - old_i)
            if new_j < -1e-9 or new_j > 1.0 + 1e-9:
                continue
            h[i] = new_i
            h[j] = max(0.0, min(1.0, new_j))
            nc = compute_c5(h)
            if nc < c5 - 1e-14:
                c5 = nc
            else:
                h[i], h[j] = old_i, old_j
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Second round: more aggressive exploration
    if remaining() > 100:
        for _ in range(5):
            if remaining() < 80:
                break
            if best_h is not None:
                n = len(best_h)
                mut = best_h.copy()
                for k in range(int(rng.integers(3, 8))):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j:
                        mut[i], mut[j] = mut[j], mut[i]
            else:
                n = 128
                mut = balance_binary(make_thue_morse(n), n, rng)
            h_opt, c5_opt = local_search(mut, t0, budget_s, rng, time_pad=10)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    if best_h is None:
        n = 96
        best_h = np.zeros(n)
        best_h[0::2] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
