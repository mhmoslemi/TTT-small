import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target_sum, n_iter=50):
    for _ in range(n_iter):
        h += (target_sum - h.sum()) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h
