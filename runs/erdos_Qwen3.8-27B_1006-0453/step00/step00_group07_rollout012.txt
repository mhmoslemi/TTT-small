import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def subgradient_c5(h, corr):
    """Compute subgradient of C5 = max_k corr[k]"""
    n = len(h)
    dx = 2.0 / n
    max_val = np.max(corr)
    # Active set: indices where corr is close to max
    tol = 1e-8
    active = corr >= max_val - tol
    if not np.any(active):
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
    
    # For each active k (0-indexed in the full correlation array of length 2n-1),
    # the shift is k - (n-1)
    # dc[k]/dh[j] = dx * [(1-h[j-k']) if 0<=j-k'<n else 0] + dx * [-h[j+k'] if 0<=j+k'<n else 0]
    # where k' = k - (n-1) is the actual shift
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if not active[k]:
            continue
        shift = k - (n - 1)  # actual lshift
        # First term: (1-h[j-shift]) for j-shift in [0,n) -> j in [shift, n+shift)
        # Second term: -h[j+shift] for j+shift in [0,n) -> j in [-shift, n-shift)

        j_start1 = max(0, shift)
        j_end1 = min(n, n + shift)
        if j_start1 < j_end1:
            idx = np.arange(j_start1, j_end1)
            idx_shift = idx - shift  # these are valid n indices
            grad[idx] += dx * (1.0 - h[idx_shift])
        
        # Second term: j ranges from max(0, -shift) to min(n-1, n-1-shift)
        j_start2 = max(0, -shift)
        j_end2 = min(n, n - shift)
        if j_start2 < j_end2:
            idx = np.arange(j_start2, j_end2)
            idx_shift = idx + shift  # these are valid n+k indices
            grad[idx] += dx * (-h[idx_shift])
    
    return grad / np.sum(active)  # Average over active set (subgradient of max)

            idx = np.arange(j_lo2, j_hi2 + 1)
            grad[idx] += -h[idx + s]
