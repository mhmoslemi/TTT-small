def targeted_local_search(h, n_points, max_iter=50000):
    dx = 2.0 / n_points
    n = n_points
    best_c5 = compute_c5(h, n)
    
    for iteration in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Convert max_idx to actual shift
        # full correlation: index k corresponds to shift (k - (n-1))
        shift = max_idx - (n - 1)
        
        # Find positions where h[i]=1 and h[i+shift]=0 (these contribute to corr at this shift)
        if shift >= 0:
            valid_i = range(n - shift)
        else:
            valid_i = range(-shift, n)
        
        contributors = []
        for i in valid_i:
            j = i + shift
            if 0 <= j < n:
                if h[i] == 1 and h[j] == 0:
                    contributors.append((i, j))

            h[j] = 1

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def thue_morse_binary(n, seed=0):
    """Generate Thue-Morse based binary sequence of length n."""
    # Generate full Thue-Morse up to n
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # Ensure exactly n/2 ones (for even n)
    if n % 2 == 0:
        # Thue-Morse of length 2^k has exactly 2^(k-1) ones
        # For arbitrary n, we might need to adjust
        ones = np.sum(t)
        target = n // 2
        if ones > target:
            # Flip some 1s to 0s
            idx = np.where(t == 1)[0]
            rng = np.random.RandomState(seed)
            to_flip = rng.choice(idx, size=ones - target, replace=False)
            t[to_flip] = 0
        elif ones < target:
            idx = np.where(t == 0)[0]
            rng = np.random.RandomState(seed)
            to_flip = rng.choice(idx, size=target - ones, replace=False)
            t[to_flip] = 1
    return t.astype(float)
