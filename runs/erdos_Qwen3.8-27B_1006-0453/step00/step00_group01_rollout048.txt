        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.ones(47) * 0.5
    n = len(h)
    target_sum = n / 2.0
    h = project_h(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Softmax subgradient (30% time)
    h1, c5_1 = subgrad_refine(h, target_sum, n_iter=2000, use_soft=True, T=100.0)
    if elapsed() > budget_s * 0.15:
        h1 = h
        c5_1 = best_c5
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Hard subgradient (20% time)
    if elapsed() < budget_s * 0.30:
        h2, c5_2 = subgrad_refine(best_h, target_sum, n_iter=2000, use_soft=False)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: Coordinate descent polish (20% time)
    if elapsed() < budget_s * 0.55:
        h3 = best_h.copy()
        prev = best_c5 + 1
        while elapsed() < budget_s * 0.55 and prev - best_c5 > 1e-10:
            prev = best_c5
            h3 = coord_descent_pass(h3, target_sum, passes=3)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_h, best_c5 = h3, c5_3

    # Phase 4: Random restarts (30% time)
    for restart in range(500):
        if elapsed() > budget_s * 0.88:
            break
        scale = 0.08 + 0.04 * restart / 500
        h_r = best_h + np.random.randn(n) * scale
        h_r = project_h(h_r, target_sum)
        # Quick refine
        h_r, c5_r = subgrad_refine(h_r, target_sum, n_iter=200, use_soft=True, T=100.0)
        h_r = coord_descent_pass(h_r, target_sum, passes=2)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_h, best_c5 = h_r.copy(), c5_r

    # Phase 5: Final polish
    if elapsed() < budget_s - 5:
        h_f = best_h.copy()
        for _ in range(10):
            if elapsed() > budget_s - 5:
                break
            h_f = coord_descent_pass(h_f, target_sum, passes=2)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
        # Tiny subgradient nudge
        h_f = project_h(best_h - 0.002 * subgradient(best_h), target_sum)
        h_f = coord_descent_pass(h_f, target_sum, passes=2)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
