def project_box_sum(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    n = len(h)
    for _ in range(100):  # iterate
        # Uniform shift
        shift = (target_sum - np.sum(h)) / n
        h_new = h + shift
        h_new = np.clip(h_new, 0, 1)
        if abs(np.sum(h_new) - target_sum) < 1e-8:
            break
        # For free variables, adjust
        free = (h_new > 0) & (h_new < 1)
        if np.any(free):
            adj = (target_sum - np.sum(h_new)) / np.sum(free)
            h_new[free] += adj
            h_new = np.clip(h_new, 0, 1)
        h = h_new
    return h_new

        np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free[0]] += diff
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_grad(h, n, T):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    max_c = corr.max()
    scaled = np.clip(T * (corr - max_c) * dx, -300.0, 0.0)
    exp_s = np.exp(scaled)
    se = exp_s.sum()
    w = exp_s / se
    obj = (max_c + np.log(se) / T) * dx
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        wk = w[s + n - 1]
        if wk < 1e-13:
            continue
        jlo = max(0, s)
        jhi = min(n - 1, s + n - 1)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += wk * omh[jlo - s:jhi - s + 1]
        jlo2 = max(0, -s)
        jhi2 = min(n - 1, n - 1 - s)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2 + 1] -= wk * h[jlo2 + s:jhi2 + s + 1]
    grad *= dx
    return obj, grad

def optimize_phase(h, n, T_start, T_end, max_iters, deadline):
    target_sum = n / 2.0
    h = project_box_sum(h, target_sum)
    best_c5 = compute_c5(h)
    lr = 0.08
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * min(1.0, it / max(1, max_iters))
        obj, grad = obj_grad(h, n, T)
        if not np.isfinite(obj) or not np.isfinite(grad).all():
            T -= 5
            if T < 10:
                continue
            continue
        accepted = False
        best_trial_c5 = best_c5
        best_trial_h = h
        for ls in range(18):
            step = lr * (0.5 ** ls)
            h_trial = h - step * grad
            h_trial = project_box_sum(h_trial, target_sum)
            c5_t = compute_c5(h_trial)
            if c5_t < best_trial_c5:
                best_trial_c5 = c5_t
                best_trial_h = h_trial
            if c5_t < best_c5 + 1e-10:
                accepted = True
                break
        if not accepted:
            stagnant += 1
            lr *= 0.6
            if stagnant > 40:
                lr = 0.08
                stagnant = 0
                if it > 50:
                    break
            continue
        h = best_trial_h
        best_c5 = best_trial_c5
        stagnant = 0
        lr = min(lr * 1.05, 0.3)
    return h, best_c5

def upsample(h, target_n):
    old_n = len(h)
    if target_n == old_n:
        return h.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.arange(target_n, dtype=float)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_box_sum(h_new, target_n / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(45, 0.5)
        h_start[::2] = 1.0
        h_start = project_box_sum(h_start, len(h_start) / 2.0)

    n_start = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    starts = [h_start.copy()]
    starts.append(project_box_sum(h_start + 0.08 * rng.randn(n_start), n_start / 2.0))
    starts.append(project_box_sum(
        np.where((np.arange(n_start) * 7 + 3) % 5 < 2, 1.0, 0.0), n_start / 2.0))

    resolutions = [n_start, 90, 180, 360, 500]

    for h_s in starts:
        if time.time() > deadline - 45:
            break
        h_cur = h_s.copy()
        n_cur = len(h_cur)
        h_cur, c5 = optimize_phase(h_cur, n_cur, 25, 80, 400, deadline - 30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
        for target_n in resolutions:
            if target_n <= n_cur:
                continue
            if time.time() > deadline - 30:
                break
            h_up = upsample(h_cur, target_n)
            c5_up = compute_c5(h_up)
            if c5_up > c5 + 0.02:
                continue
            h_cur, c5 = optimize_phase(h_up, target_n, 25, 120, 600, deadline - 30)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
            n_cur = target_n

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
