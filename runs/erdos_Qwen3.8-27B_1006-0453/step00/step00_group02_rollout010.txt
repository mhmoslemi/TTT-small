import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smoothed_max(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / alpha

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_polish(h, n, max_sweeps=50):
    best = compute_c5(h, n)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # For fixed others, c_k is linear in h_j
            # Find optimal h_j in [0,1] minimizing max_k c_k
            # Use ternary search on [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[0 if len(h)==1 else j] = m1
                f1 = compute_c5(h, n)
                h[0 if len(h)==1 else j] = m2
                f2 = compute_c5(h, n)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            h[j] = (lo + hi) / 2.0
            # Fix sum by adjusting a non-boundary element
            diff = n / 2.0 - np.sum(h)
            if abs(diff) > 1e-10:
                # Distribute to free elements
                free = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
                if len(free) > 0:
                    h[free] += diff / len(free)
                    h = np.clip(h, 0.0, 1.0)
                    # Final fix
                    h = fix_sum(h, n)
            c = compute_c5(h, n)
            if c < best - 1e-10:
                best = c
                improved = True
        if not improved:
            break
    return h, best
