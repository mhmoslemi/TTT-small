import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_max(c, T):
    m = np.max(c)
    shifted = T * (c - m)
    shifted = np.clip(shifted, -50, 50)
    return m + (1.0 / T) * np.log(np.mean(np.exp(shifted)))

def obj_func(h, n, T):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return smooth_max(c, T)

def obj_penalty(h, n, T, lam):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    val = smooth_max(c, T)
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def interp_step(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    fracs = np.arange(n_new) * n_old / n_new
    idx = np.minimum(fracs.astype(int), n_old - 1)
    return h_old[idx].copy()

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

                    h_try[i] = hi + d

                    c = compute_c5(h_try)
                    if c < best_c5 - 1e-12:
