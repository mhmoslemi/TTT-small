    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    vals = alpha * (corr - m)
    vmax = np.max(vals)
    s = np.sum(np.exp(vals - vmax))
    return float(m + (vmax + np.log(s)) / alpha)

def constrain_sum(h):

def project_feasible(h, n):
