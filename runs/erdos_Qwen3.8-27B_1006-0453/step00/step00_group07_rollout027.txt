        if cc < bc:
            bc = cc
            bh = hc.copy()
            ni = 0
            ds = max(0.02, ds * 0.98)
        else:
            ni += 1
            if ni > 25:
                hc = make_feasible(bh + rng.normal(0, 0.06, n), target)
                cc = eval_raw(hc)
                bc = cc
                bh = hc.copy()
                T = max(T, 0.002)
                ds = 0.3
                ni = 0
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    B = max(1, int(budget_s))
    dl = t0 + B - 5
    n = 80
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    else:
        h = np.full(n, 0.5)
    dx = 2.0 / n
    tgt = n / 2.0
    h = make_feasible(h, tgt)

    bH = h.copy()
    bC = eval_raw(h) * dx
    print(f"Initial: {bC:.6f}")

    h1, c1 = greedy(h.copy(), 30)
    if c1 * dx < bC:
        bC = c1 * dx
        bH = h1.copy()
    print(f"Greedy: {bC:.6f}")

    sa_dl = t0 + B * 0.82
    starts = [h1.copy()]
    alt = np.tile([1.0, 0.0], n // 2)
    blk = np.concatenate([np.ones(n // 2), np.zeros(n // 2)])
    for s in (alt, blk):
        s = make_feasible(s, tgt)
        sg, _ = greedy(s, 15)
        starts.append(sg)

    sa_count = 0
    for r, st in enumerate(starts):
        for k in range(3):
            if time.time() > sa_dl:
                break
            T0 = 0.004 * (0.7 ** (r * 3 + k))
            sh, sc = sa(st.copy(), tgt, T0, 1e-7, 0.999, sa_dl, seed + 13 * (sa_count + 1))
            C = sc * dx
            if C < bC:
                bC = C
                bH = sh.copy()
                print(f"SA r{r}k{k}: {bC:.6f}")
            sa_count += 1

    if time.time() < dl:
        h2, c2 = greedy(bH.copy(), 60)
        if c2 * dx < bC:
            bC = c2 * dx
            bH = h2.copy()
    print(f"Final: {bC:.6f}")
    return (bH, bC, n)
