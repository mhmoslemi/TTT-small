from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(scores), scores

def smooth_max(scores, tau):
    m = np.max(scores)
    if m == 0:
        max_val = 0.0
    else:
        max_val = tau * np.log(np.sum(np.exp((scores - m) / tau))) + m
    return max_val

def project_feasible(v, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

def score_gradient(h, n, tau=0.005):
    """Compute gradient of smooth max of scores."""
    dx = 2.0 / n
    _, scores = compute_c5(h)
    m = np.max(scores)
    exps = np.exp((scores - m) / tau)
    p = exps / np.sum(exps)
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lag = k - n + 1
        for j in range(n):
            g = 0.0
            if 0 <= j + lag < n:
                g += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] += p[k] * dx * g
    return grad

    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    tau = tau_init
    lr = lr_init
    
    for it in range(max_iter):
        grad = score_gradient(h, n, tau)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Reduce tau gradually
            tau = max(tau * 0.99, 0.001)
        else:
            h = h_new  # still take the step for exploration
            lr *= 0.95
        
    return best_h, best_c5

d(c[idx])/dh[j] = Σ_m [δ(m,j) * (1-h[m-k_shift]) + h[m] * (-δ(m-k_shift, j))]

First term: m=j, need 0≤j<n (always true) and 0≤j-k_shift<n:

Second term: m-k_shift=j, so m=j+k_shift, need 0≤j+k_shift<n and 0≤j<n:
= -h[j+k_shift] * 1{0≤j+k_shift<n}

So: d(c[idx])/dh[j] = (1-h[j-k_shift])*1{0≤j-k_shift<n} - h[j+k_shift]*1{0≤j+k_shift<n}
