    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
    res = minimize(objective_func, x0, args=(n,), method='trust-constr',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'gtol': 1e-8, 'xtol': 1e-10})
    return proj_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except Exception:
        h_parent = make_spectral_init(70, 0)
    n_parent = len(h_parent)
    h_parent = proj_feasible(h_parent, n_parent)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    # Resolutions to explore
    resolutions = sorted(set([n_parent, 100, 140, 200, 280, 350, 420]))

    for idx, n_pts in enumerate(resolutions):
        elapsed = time.time() - t0
        if elapsed > budget_s - 90:
            break

        # Interpolate best to current resolution
        if n_pts == best_n:
            h0 = best_h.copy()
        else:
            h0 = interpolate_h(best_h, best_n, n_pts)
            h0 = proj_feasible(h0, n_pts)

        # Try SLSQP with multiple starts
        n_starts = min(6, max(2, int((budget_s - 90 - elapsed) / max(1.0, n_pts * 0.05))))

        for s in range(n_starts):
            if time.time() - t0 > budget_s - 60:
                break

            if s == 0:
                x0 = h0.copy()
            elif s == 1:
                x0 = h0 + 0.02 * np.random.randn(n_pts)
                x0 = proj_feasible(x0, n_pts)
            elif s == 2:
                x0 = make_spectral_init(n_pts, seed + idx * 100 + s)
            else:
                x0 = h0 + 0.05 * s * np.random.randn(n_pts)
                x0 = proj_feasible(x0, n_pts)

            # SLSQP
            try:
                h_cand = slsqp_optimize(x0, n_pts, maxiter=250)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand
                    best_n = n_pts
                    h0 = h_cand.copy()
            except Exception:
                pass

            # Trust-constr (sometimes finds different basins)
            if time.time() - t0 > budget_s - 45:
                break
            try:
                h_cand2 = trustconstr_optimize(h0, n_pts, maxiter=150)
                c5_cand2 = compute_c5(h_cand2)
                if c5_cand2 < best_c5:
                    best_c5 = c5_cand2
                    best_h = h_cand2
                    best_n = n_pts
                    h0 = h_cand2.copy()
            except Exception:
                pass

        # Update best_n tracking for chain interpolation
        best_n = n_pts

    # Final: also try a few more starts at the best resolution
    elapsed = time.time() - t0
    if elapsed < budget_s - 30:
        n_use = best_n
        for s in range(10):
            if time.time() - t0 > budget_s - 10:
                break
            x0 = best_h + 0.03 * np.random.randn(n_use)
            x0 = proj_feasible(x0, n_use)
            try:
                h_cand = slsqp_optimize(x0, n_use, maxiter=200)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand
            except Exception:
                pass

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = proj_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
