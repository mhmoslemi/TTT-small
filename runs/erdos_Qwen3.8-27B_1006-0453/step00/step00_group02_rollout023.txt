    h = np.clip(h, 0, 1)

    # If sum is still off, force fix
    target_sum = final_n / 2.0
    if abs(np.sum(best_h) - target_sum) > 1e-10:
        best_h = best_h * (target_sum / np.sum(best_h))
        best_h = np.clip(best_h, 0.0, 1.0)
        # Distribute remaining
        for _ in range(1000):
            diff = target_sum - np.sum(best_h)
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                m = best_h < 1.0
            else:
                m = best_h > 0.0
            if not np.any(m):
                break
            best_h[m] += diff / np.sum(m)
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, final_n
