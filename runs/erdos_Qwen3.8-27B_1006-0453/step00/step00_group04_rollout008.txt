import numpy as np
from scipy.optimize import minimize
import time

_opt_params = {'n': 42, 'temp': 0.03}

def compute_c5(h, dx):
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if len(free) > 0:
                h[free] += diff / len(free)
        else:
            free = np.where(h > 1e-12)[0]
            if len(free) > 0:
                h[free] += diff / len(free)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        rem = diff
        for j in order:
            if rem > 0:
                adj = min(rem, 1.0 - h[j])
                h[j] += adj
                rem -= adj
            else:
                adj = max(rem, -h[j])
                h[j] += adj
                rem -= adj
            if abs(rem) < 1e-15:
                break
    return h

def smooth_max_val(h_flat, n, temp):
    h = h_flat.reshape(-1)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(temp * (c - m)))) / temp

def _objective(x):
    return smooth_max_val(x, _opt_params['n'], _opt_params['temp'])

def coordinate_descent(h, n, iterations=50, step=0.01):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    for _ in range(iterations):
        improved = False
        for j in range(n):
            for delta in [step, -step, step*0.5, -step*0.5]:
                candidate = best.copy()
                candidate[j] += delta
                candidate = project_feasible(candidate, n)
                c5 = compute_c5(candidate, dx)
                if c5 < best_c5 - 1e-12:
                    best = candidate
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def interpolate_to(h_src, n_src, n_dst):
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.arange(n_dst) * (2.0 / n_dst)
    h_lin = np.interp(x_dst, x_src, h_src)
    h_per = np.interp(x_dst, x_src, h_src, period=2.0)
    return h_lin, h_per

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.ones(42) * 0.5

    n0 = len(h_init)
    best_h = project_feasible(h_init, n0)
    best_c5 = compute_c5(best_h, 2.0 / n0)

    def elapsed():
        return time.time() - t0

    def try_optimize(h_start, n, dx):
        nonlocal best_h, best_c5
        # Powell with multiple temperatures
        for temp in [0.05, 0.02, 0.01]:
            if elapsed() > budget_s * 0.82:
                return
            _opt_params['n'] = n
            _opt_params['temp'] = temp
            try:
                res = minimize(_objective, h_start.copy(), method='powell',
                               options={'maxiter': 400, 'xtol': 1e-8})
                h_opt = project_feasible(res.x, n)
                c5_opt = compute_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
            except Exception:
                pass

    def try_restarts(h_base, n, dx, n_restarts):
        nonlocal best_h, best_c5
        for r in range(n_restarts):
            if elapsed() > budget_s * 0.85:
                return
            if r == 0:
                h_cur = h_base.copy()
            else:
                scale = 0.08 / (r + 1)
                h_cur = h_base + scale * np.random.randn(n)
                h_cur = project_feasible(h_cur, n)
            try_optimize(h_cur, n, dx)

    # Phase 1: Optimize at initial resolution
    try_restarts(best_h, n0, 2.0 / n0, 3)
    # Coordinate descent refinement
    best_h, best_c5 = coordinate_descent(best_h, n0, iterations=80, step=0.005)

    # Phase 2: Try higher resolutions
    try_ns = [50, 60, 80, 100, 120, 150]
    for n_try in try_ns:
        if elapsed() > budget_s * 0.75:
            break
        dx_try = 2.0 / n_try
        h_lin, h_per = interpolate_to(best_h, len(best_h), n_try)
        for h_cand in [h_lin, h_per]:
            h_cand = project_feasible(h_cand, n_try)
            c5_cand = compute_c5(h_cand, dx_try)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
        try_restarts(best_h, n_try, dx_try, 2)
        best_h, best_c5 = coordinate_descent(best_h, n_try, iterations=40, step=0.003)

    # Phase 3: If still above target, try random structured starts
    if best_c5 > 0.3808 and elapsed() < budget_s * 0.6:
        n_struct = 60
        dx_s = 2.0 / n_struct
        for trial in range(5):
            if elapsed() > budget_s * 0.8:
                break
            # Half-wave construction
            phase = np.random.uniform(0, 2.0)
            h_s = 0.5 + 0.3 * np.sin(np.pi * (np.arange(n_struct) * 2.0 / n_struct + phase))
            h_s = project_feasible(h_s, n_struct)
            # Optimize
            try_optimize(h_s, n_struct, dx_s)

    # Final refinement at best resolution
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h, best_c5 = coordinate_descent(best_h, n_final, iterations=100, step=0.002)
    _opt_params['n'] = n_final
    _opt_params['temp'] = 0.005
    try:
        res = minimize(_objective, best_h.copy(), method='powell',
                       options={'maxiter': 200, 'xtol': 1e-10})
        h_opt = project_feasible(res.x, n_final)
        c5_opt = compute_c5(h_opt, dx_final)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    except Exception:
        pass

    best_h, best_c5 = coordinate_descent(best_h, n_final, iterations=60, step=0.001)

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h, 2.0 / n_final)

    # Use evaluate_erdos_solution if available
    if 'evaluate_erdos_solution' in globals():
        try:
            result = evaluate_erdos_solution(best_h)
            if result is not None:
                if isinstance(result, tuple):
                    best_c5 = float(result[0])
                elif isinstance(result, (int, float, np.floating)):
                    best_c5 = float(result)
        except Exception:
            pass

    return best_h, float(best_c5), int(len(best_h))
