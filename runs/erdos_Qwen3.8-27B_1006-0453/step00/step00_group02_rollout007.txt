import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def obj_function(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx


def con_function(x):
    return np.sum(x) - len(x) / 2.0


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(40):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def binary_search(h0, rng, max_iters):
    n = len(h0)
    h = h0.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    stalls = 0
    it = 0
    best_c5 = current
    while it < max_iters and stalls < 400:
        it += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(25):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            if new_c5 < current:
                current = new_c5
                if current < best_c5:
                    best_c5 = current
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                improved = True
                break
            else:
                h[i], h[j] = h[j], h[i]
        if improved:
            stalls = 0
        else:
            stalls += 1
    return h, best_c5


def make_binary(n, ref_h, rng):
    target = n // 2
    if ref_h is not None and len(ref_h) != n:
        h_int = np.interp(np.linspace(0, len(ref_h) - 0.01, n),
                         np.arange(len(ref_h)), ref_h)
        h_bin = (h_int > 0.5).astype(float)
    elif ref_h is not None:
        h_bin = (ref_h > 0.5).astype(float)
    else:
        h_bin = np.zeros(n)
        h_bin[:target] = 1.0
    n_ones = int(np.sum(h_bin > 0.5))
    if n_ones > target:
        idx = np.where(h_bin > 0.5)[0]
        rng.shuffle(idx)
        h_bin[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h_bin <= 0.5)[0]
        rng.shuffle(idx)
        h_bin[idx[:target - n_ones]] = 1.0
    return h_bin


def cobyla_optimize(h0, n, maxfev, rhobeg):
    res = minimize(obj_function, h0, method='COBYLA',
                   bounds=[(0.0, 1.0)] * n,
                   constraints={'type': 'eq', 'fun': con_function},
                   options={'maxfev': maxfev, 'rhobeg': rhobeg})
    return res


def run(seed=42, budget_s=1100, **kwargs):
