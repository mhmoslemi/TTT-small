import numpy as np
import time


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _sa_optimize(h_init, n_iter, t0, tmin, rng, max_step, time_limit):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c = _compute_c5(h)
    c = best_c
    cool = (tmin / t0) ** (1.0 / max(n_iter, 1))
    t = t0
    start = time.time()
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        dmax = min(h[i], 1.0 - h[j])
        if dmax > 1e-9:
            delta = float(rng.uniform(0.01, min(max_step, dmax)))
            h[i] += delta
            h[j] -= delta
            fwd = 1
        else:
            dmax = min(h[j], 1.0 - h[i])
            if dmax <= 1e-9:
                continue
            delta = float(rng.uniform(0.01, min(max_step, dmax)))
            h[i] -= delta
            h[j] += delta
            fwd = -1
        new_c = _compute_c5(h)
        accept = new_c <= c
        if not accept and t > 1e-15:
            accept = float(rng.random()) < np.exp(-(new_c - c) / t)
        if accept:
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            if fwd == 1:
                h[i] -= delta
                h[j] += delta
            else:
                h[i] += delta
                h[j] -= delta
        t = max(t * cool, tmin)
    return best_h, best_c


def _binary_refine(h, rng, n_passes=50, time_limit=30):
    """Refine by trying swaps in the binary projection."""
    n = len(h)
    target = n // 2
    # Start from thresholding
    sorted_idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[sorted_idx[:target]] = 1.0
    best_c = _compute_c5(binary)
    best_b = binary.copy()
    start = time.time()
    for _ in range(n_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        ones = np.where(binary > 0.5)[0]
        zeros = np.where(binary <= 0.5)[0]
        for _oi in range(len(ones)):
            for _zi in range(len(zeros)):
                oi = int(ones[_oi])
                zi = int(zeros[_zi])
                binary[oi] = 0.0
                binary[zi] = 1.0
                c = _compute_c5(binary)
                if c < best_c - 1e-12:
                    best_c = c
                    best_b = binary.copy()
                    improved = True
                    ones = np.where(binary > 0.5)[0]
                    zeros = np.where(binary <= 0.5)[0]
                    break
                else:
                    binary[oi] = 1.0
                    binary[zi] = 0.0
            if improved:
                break
        if not improved:
            # Try random swaps to escape local opt
            if time.time() - start > time_limit:
                break
            for _ in range(20):
                oi = int(rng.integers(0, len(np.where(best_b > 0.5)[0])))
                ones_list = np.where(best_b > 0.5)[0]
                zeros_list = np.where(best_b <= 0.5)[0]
                oi = int(ones_list[rng.integers(0, len(ones_list))])
                zi = int(zeros_list[rng.integers(0, len(zeros_list))])
                best_b[oi] = 0.0
                best_b[zi] = 1.0
                c = _compute_c5(best_b)
                if c < best_c:
                    best_c = c
                else:
                    best_b[oi] = 1.0
                    best_b[zi] = 0.0
    return best_b, best_c


def _normalize_sum(h, n):
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-10:
        h = np.clip(h * (target / s), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        for _ in range(n * 20):
            idx = int(np.argmin(np.abs(h - 0.5))) if diff > 0 else int(np.argmax(h))
            if diff > 0:
                room = 1.0 - h[idx]
                a = min(room, diff)
            else:
                a = min(h[idx], -diff)
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h
