import numpy as np
from scipy.optimize import minimize
import time


def obj_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def cons_sum(h):
    return h.sum() - len(h) / 2.0


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        h = h + (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, n_new)


def slsqp_opt(h0, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_sum}]
    res = minimize(obj_c5, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    return fix_sum(np.clip(res.x, 0.0, 1.0), n)

    rng = np.random.RandomState(seed_val)
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

    kernel /= kernel.sum()
    h = np.convolve(h, kernel, mode='same')
    h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, n)
