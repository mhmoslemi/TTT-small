import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search_binary(h, dx, max_sweeps=200, sa_temp=0.0, sa_cool=0.95):
    """Local search for binary sequence via pair swaps"""
    h = np.array(h, dtype=float)
    n = len(h)
    best_c5 = compute_c5(h, dx)
    rng = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Shuffle for diversity
        ones_idx = rng.permutation(len(ones))
        zeros_idx = rng.permutation(len(zeros))
        
        found = False
        for ii in ones_idx:
            for jj in zeros_idx:
                i, j = ones[ii], zeros[jj]
                old_i, old_j = h[i], h[j]
                h[i], h[j] = old_j, old_i
                c5 = compute_c5(h, dx)
                delta = c5 - best_c5
                if delta < -1e-12 or (sa_temp > 0 and rng.random() < np.exp(-delta / sa_temp)):
                    best_c5 = c5
                    found = True
                else:
                    h[i], h[j] = old_i, old_j
                if found:
                    break
            if found:
                break
        
        if not found:
            if sa_temp > 0:
                sa_temp *= sa_cool
            else:
                break
    
    return h, best_c5

def make_binary_candidate(n, rng):
    """Create a random binary sequence with exactly n/2 ones"""
    h = np.zeros(n)
    m = n // 2
    idx = rng.permutation(n)
    h[idx[:m]] = 1.0
    return h

def make_fractional_candidate(n, rng):
    """Create a sequence with sum = n/2, mostly binary"""
    h = np.zeros(n)
    m = n // 2
    remainder = n / 2.0 - m  # 0 or 0.5
    idx = rng.permutation(n)
    h[idx[:m]] = 1.0
    if remainder > 0:
        h[idx[m]] = remainder
    return h

        dx = 2.0 / n
        target_sum = n / 2.0

        for s in range(n_starts):

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search(h, dx, max_sweeps, rng, sa_temp=0.0, sa_cool=0.95):
    h = np.array(h, dtype=float).copy()
    n = len(h)
    best_c5 = compute_c5(h, dx)
    cur_temp = sa_temp

    for sweep in range(max_sweeps):
        if cur_temp > 0:
            cur_temp *= sa_cool
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        perm_o = rng.permutation(len(ones))
        perm_z = rng.permutation(len(zeros))
        improved = False
        for pi in range(len(perm_o)):
            i = ones[perm_o[pi]]
            for pj in range(len(perm_z)):
                j = zeros[perm_z[pj]]
                h[i] = 0.0
                h[j] = 1.0

                if accept:
                    best_c5 = c5
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        if not improved and cur_temp <= 1e-12:
            break
    return h, best_c5

    h = np.zeros(n)

    h = np.zeros(n)

            remaining -= 1
    if n % 2 == 1:
        h[np.where(h == 0.0)[0][0]] = 0.5
        h[np.where(h == 1.0)[0][-1]] = 0.0
    return h
