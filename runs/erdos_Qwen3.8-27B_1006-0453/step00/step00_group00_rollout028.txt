                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5


def _random_start(n, best_h, r):
    if r == 0:
        return best_h + np.random.randn(n) * 0.08
    elif r % 5 == 1:
        return best_h + np.random.randn(n) * 0.2
    elif r % 5 == 2:
        return np.random.beta(3, 3, n)
    elif r % 5 == 3:
        v = np.random.rand(n)
        return np.where(v < 0.5, 0.0, 1.0)
    else:
        return best_h + np.random.randn(n) * 0.12


def _gradient_step(h, n, step_size):
    """Take a step in the direction that reduces the active (max) correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    max_val = np.max(corr)
    # Find active lags (within 2% of max)
    active = np.where(corr > max_val * 0.98)[0]
    # Build gradient direction
    grad = np.zeros(n)
    n_half = n - 1
    for k in active:
        s = k - n_half  # shift
        if s == 0:
            grad -= (1.0 - 2.0 * h) * 0.3
        elif s > 0:
            for i in range(n - s):
                grad[i] += (1.0 - h[i + s])
                grad[i + s] -= h[i]
        else:
            s2 = -s
            for i in range(s2, n):
                grad[i] += (1.0 - h[i - s2])
                grad[i - s2] -= h[i]
    # Normalize
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-12:
        grad = grad / gnorm
    h_new = _repair(h - step_size * grad, n)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    n = 45
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h0 = _repair(h0, n)
    best_h = h0.copy()
    best_c5 = _c5_val(h0)

    def elapsed():
        return time.time() - t0

    # Phase 1: NM from initial
    h_try, c5_try = _nm_run(h0, 15000, 1e-8, 1e-10)
    if c5_try < best_c5:
        best_c5, best_h = c5_try, h_try

    # Phase 2: Swap search
    h_try, c5_try = _swap_search(best_h, n, 300, t0 + budget_s * 0.35, t0)
    if c5_try < best_c5:
        best_c5, best_h = c5_try, h_try

    # Phase 3: Gradient descent on active constraints
    if elapsed() < budget_s * 0.45:
        h_cur = best_h.copy()
        step = 0.05
        for it in range(500):
            if elapsed() > budget_s * 0.45:
                break
            h_new = _gradient_step(h_cur, n, step)
            c5_new = _c5_val(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5, best_h = c5_new, h_new
                h_cur = h_new
                if step < 0.2:
                    step *= 1.02
            else:
                step *= 0.85
                if step < 0.001:
                    break
        h_try, c5_try = _swap_search(best_h, n, 100, t0 + budget_s * 0.50, t0)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 4: Multiple restarts with NM
    t_restart_end = t0 + budget_s * 0.88
    n_restarts = 25
    for r in range(n_restarts):
        if time.time() > t_restart_end:
            break
        h_start = _random_start(n, best_h, r)
        h_start = _repair(h_start, n)
        c5_start = _c5_val(h_start)
        if c5_start < best_c5:
            best_c5, best_h = c5_start, h_start.copy()
        rem = t_restart_end - time.time()
        if rem > 3:
            h_try, c5_try = _nm_run(h_start, 8000, 1e-9, 1e-10)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try
            # Quick swap
            if time.time() < t_restart_end - 2:
                h_try, c5_try = _swap_search(best_h, n, 50, t_restart_end - 1, t0)
                if c5_try < best_c5:
                    best_c5, best_h = c5_try, h_try

    # Phase 5: Final polish
    if elapsed() < budget_s * 0.97:
        h_try, c5_try = _nm_run(best_h, 15000, 1e-9, 1e-11)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        h_try, c5_try = _swap_search(best_h, n, 200, t0 + budget_s * 0.97, t0)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h)
    return (best_h, best_c5, n)
