    if diff > 1e-12:
        order = np.where(h < 1.0)[0]

            if diff <= 1e-12:

import numpy as np
import time


def _project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if d > 1e-10:
        for _ in range(len(h)):
            if d <= 1e-10:
                break
            rooms = 1.0 - h
            idx = int(np.argmax(rooms))
            a = min(d, rooms[idx])
            h[idx] += a
            d -= a
    elif d < -1e-10:
        for _ in range(len(h)):
            if d >= -1e-10:
                break
            idx = int(np.argmin(h))
            a = max(d, -h[idx])
            h[idx] += a
            d -= a
    return h


def _c5(h, dx):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _sa_phase(curr, curr_c5, best, best_c5, n, dx, target,
              T_init, cool_rate, step_size, deadline, rng):

    T_floor = 1e-7
    T = T_init
    iters_since_best = 0
    total = 0

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if j == i:
            j = (j + 1) % n

        h[j] -= delta

                best_c5 = c
