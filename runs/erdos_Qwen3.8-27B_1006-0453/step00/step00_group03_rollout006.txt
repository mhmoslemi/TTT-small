import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    shift = k_max - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        j1 = np.arange(shift, n)
        grad[j1] += (1.0 - h[j1 - shift])
        j2 = np.arange(0, n - shift)
        grad[j2] += (-h[j2 + shift])
    else:
        s = -shift
        j1 = np.arange(0, n - s)
        grad[j1] += (1.0 - h[j1 + s])
        j2 = np.arange(s, n)
        grad[j2] += (-h[j2 - s])
    return grad * dx

def soft_c5(h, n, temp=0.008):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def project_c(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        adj = target - s
        pos = h < 1.0
        neg = h > 0.0
        if adj > 0 and np.any(pos):
            h[pos] += adj / np.sum(pos)
        elif adj < 0 and np.any(neg):
            h[neg] += adj / np.sum(neg)
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = project_c(np.clip(h0, 0, 1), n0)
        best_h = h0.copy()
        best_c5 = compute_c5(best_h)
    else:
        n0 = 70
        best_h = np.full(n0, 0.5)
        best_c5 = compute_c5(best_h)

    n_list = [70, 98, 140, 196]
    max_iters = [300, 250, 180, 120]

    for idx_n, n in enumerate(n_list):
        if time.time() - t0 > budget_s * 0.82:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]

        inits = []
        h_int = project_c(interp_n(best_h, n), n)
        inits.append(h_int)
        for _ in range(2):
            hr = rng.rand(n) * 0.5 + 0.25
            inits.append(project_c(hr, n))
        for fc in [0.35, 0.5, 0.65]:
            hb = np.zeros(n)
            hb[:int(n * fc)] = 1.0
            inits.append(project_c(hb, n))
        for start_pos in [0, n // 4, n // 2]:
            hb2 = np.zeros(n)
            blk = int(n * 0.5)
            end_pos = min(start_pos + blk, n)
            hb2[start_pos:end_pos] = 1.0
            inits.append(project_c(hb2, n))

                c_try = compute_c5(ht)
                if c_try < best_c5 - 1e-11:
                    best_h = ht.copy()
                    best_c5 = c_try
                    hc = ht.copy()
                    break

    best_h = project_c(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
