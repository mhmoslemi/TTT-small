import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def correlate_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def grad_correlation(h, k):
    """Gradient of corr[k] w.r.t. h, where k is index in full correlation output."""
    n = len(h)
    dx = 2.0 / n
    s = k - (n - 1)
    grad = np.zeros(n)
    # Term 1: h[j] appears as first factor when j and j+s are both valid
    # i.e., 0 <= j < n and 0 <= j+s < n
    lo1, hi1 = max(0, -s), min(n, n - s)
    if lo1 < hi1:
        grad[lo1:hi1] += (1 - h[lo1+s:hi1+s])
    # Term 2: h[j] appears as second factor (in 1-h[j]) when j-s and j are both valid
    # i.e., 0 <= j-s < n and 0 <= j < n
    lo2, hi2 = max(0, s), min(n, n + s)
    if lo2 < hi2:
        grad[lo2:hi2] += (-h[lo2-s:hi2-s])
    return grad * dx

def projected_gradient_step(h, grad, lr, n):
    """Take a projected gradient step maintaining sum(h)=n/2."""
    h_new = h - lr * grad
    # Maintain sum constraint
    h_new = h_new - (np.sum(h_new) - n/2) / n
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Re-project sum
    for _ in range(10):
        diff = n/2 - np.sum(h_new)
        if abs(diff) < 1e-10:
            break
        # Adjust only where not at bounds
        if diff > 0:
            mask = h_new < 1
        else:
            mask = h_new > 0
        if mask.any():
            h_new[mask] += diff / mask.sum()
        h_new = np.clip(h_new, 0, 1)
    return h_new

def optimize_pg(h0, n, max_iter=5000, lr_init=0.01, time_limit=100):
    """Projected gradient optimization."""
    t_start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = correlate_vec(h)
        k_star = np.argmax(corr)
        c_val = corr[k_star]
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        grad = grad_correlation(h, k_star)
        h = projected_gradient_step(h, grad, lr, n)

        if corr_new[k_star] > c_val + 1e-6:

        elif corr_new[k_star] < c_val - 1e-6:

    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    lo1, hi1 = max(0, -s), min(n, n - s)
    if lo1 < hi1:

    lo2, hi2 = max(0, s), min(n, n + s)

def upsample(h_coarse, n_new):

    h_fine = np.zeros(n_new)

        idx = int(pos)
        frac = pos - idx
        if idx + 1 < n_old:
            h_fine[i] = h_coarse[idx] * (1 - frac) + h_coarse[idx + 1] * frac
        else:
            h_fine[i] = h_coarse[min(idx, n_old - 1)]
    return project_box_sum(h_fine, n_new / 2.0)

def slsqp_polish(h0, time_limit=60):
    n = len(h0)
    dx = 2.0 / n
    t_start = time.time()
    t0 = compute_c5(h0)
    x0 = np.append(h0.copy(), t0)

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x[:n]) - n / 2.0,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    for k in range(2 * n - 1):
        def _mk(kk=k):
            def cf(x):
                h = x[:n]
                return x[n] - np.correlate(h, 1 - h, mode='full')[kk] * dx

                h = x[:n]
                g = np.zeros(n + 1)
                g[n] = 1.0
                g[:n] = -grad_corr_at(h, kk, dx)
                return g
            return cf, cg
        cf, cg = _mk()
        cons.append({'type': 'ineq', 'fun': cf, 'jac': cg})

    return h0, t0
