For n points, dx = 2/n. The correlation at lag k is:
corr[k] = sum_{i} h[i] * (1-h[i+k])  (for the valid range of i)

For binary h, this counts the number of 1s in h that, when shifted by k, land on 0s in h.

The total over all lags k is: sum_k corr[k] = sum_i h[i] * (number of 0s in h) = (n/2) * (n/2) = n²/4 (approximately, for even n).

There are 2n-1 lags, so the average correlation is n²/4 / (2n-1) ≈ n/8.
Average C₅ contribution: (n/8) * (2/n) = 1/4.

But we're looking at the maximum, not the average. For a random binary sequence, the maximum would be above average, maybe around n/8 * 1.5 = 3n/16, giving C₅ ≈ 3/8 = 0.375. That's in the right ballpark!

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def repair_sum(h, n):
    h = h.astype(float).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(-h)
    else:
        order = np.argsort(h)
    remaining = diff
    for i in order:
        if abs(remaining) < 1e-12:
            break
        if remaining > 0:
            step = min(1.0 - h[i], remaining)
            h[i] += step
            remaining -= step
        else:
            step = min(h[i], -remaining)
            h[i] -= step
            remaining += step
    return h

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    if n % 2 == 0:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    else:
        idx = rng.choice(n - 1, size=(n - 1) // 2, replace=False)
        h[idx] = 1.0
        h[-1] = 0.5
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    h = repair_sum(h, n)
    return h

def make_blocks(n, k, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pattern = []
    for _ in range(k):
        pattern.extend([1] * rng.randint(1, max(2, n // (2 * k))))
        pattern.extend([0] * rng.randint(1, max(2, n // (2 * k))))
    h[:n] = pattern[:n]
    h = np.round(h)
    target = n / 2.0
    diff = int(round(target - np.sum(h)))
    if diff > 0:
        idx = np.where(h == 0)[0]
        if diff <= len(idx):
            h[rng.choice(idx, size=diff, replace=False)] = 1.0
    elif diff < 0:
        idx = np.where(h == 1)[0]
        if -diff <= len(idx):
            h[rng.choice(idx, size=-diff, replace=False)] = 0.0
    if abs(np.sum(h) - target) < 1e-10:
        return h
    return repair_sum(h, n)

def local_search(h, n, time_limit, rng):
    t_start = time.time()
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    while time.time() - t_start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, best_c5

def anneal(h, n, time_limit, rng, T0=0.02):
    t_start = time.time()
    h = h.astype(float).copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T0
    while T > 1e-5 and time.time() - t_start < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-6 and np.log(rng.random() + 1e-15) > -delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= 0.997
    return best_h, best_c5

    if n_src == n_dst:
        return h_src.astype(float).copy()
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False)
    x_src = np.linspace(0, 2.0, n_src, endpoint=False)

        idx = np.where(h_dst == 0)[0]
        if diff <= len(idx):

    elif diff < 0:
        idx = np.where(h_dst == 1)[0]
        if -diff <= len(idx):
            h_dst[np.random.choice(idx, size=-diff, replace=False)] = 0.0
    if abs(np.sum(h_dst) - target) > 1e-10:
        h_dst = repair_sum(h_dst, n_dst)
    return h_dst
