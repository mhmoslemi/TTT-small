```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    # Numerical stability: shift so max is 0
    scaled = scaled - m
    soft = (np.log(np.sum(np.exp(scaled))) / temp) + (m / temp)
    return soft * dx

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-8:
        return h
    # Linear adjustment on non-saturated values
    free = (h > 1e-6) & (h < 1.0 - 1e-6)
    if not np.any(free):
        h = h + (target - s) / n
        h = np.clip(h, 0.0, 1.0)
        return h
    diff = target - s
    # Distribute proportionally
    per = diff / np.sum(free)
    h[free] = np.clip(h[free] + per, 0.0, 1.0)
    # If still off, iterate
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if not np.any(free):
            break
        diff = target - s
        per = diff / np.sum(free)
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    h0 = repair_h(h0, n)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    
    def elapsed():
        return time.time() - t_start
    
    # Phase 1: SLSQP with decreasing temperature (annealing)
    temperatures = [10, 20, 40, 80, 150, 300]
    
    for temp in temperatures:
        if elapsed() > budget_s - 60:
            break
        
        maxiter = max(50, int((budget_s - 60 - elapsed()) * 5))
        maxiter = min(maxiter, 500)
        
        result = minimize(
            lambda x: smooth_obj(x, n, temp),
            best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False}
        )
        
        h_trial = repair_h(result.x, n)
        c5_trial = compute_c5(h_trial, n)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    
    # Phase 2: Multi-start with perturbations
    n_starts = 0
    while elapsed() < budget_s - 60:
        n_starts += 1
        scale = 0.05 * np.exp(-n_starts / 30)
        candidate = best_h + np.random.uniform(-scale, scale, size=n)
        candidate = repair_h(candidate, n)
        
        remaining = budget_s - 60 - elapsed()
        if remaining < 5:
            break
        
        maxiter = min(300, max(50, int(remaining * 8)))
        
        for temp in [30, 100]:
            if elapsed() > budget_s - 30:
                break
            result = minimize(
                lambda x: smooth_obj(x, n, temp),
                candidate,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}],
                options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False}
            )
            h_trial = repair_h(result.x, n)
            c5_trial = compute_c5(h_trial, n)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                candidate = best_h.copy()
    
    # Phase 3: Bang-bang refinement
    if elapsed() < budget_s - 20:
        # Try rounding to {0, 1} with some freedom
        for threshold in [0.3, 0.4, 0.5, 0.6, 0.7]:
            if elapsed() > budget_s - 20:
                break
            bb = (best_h > threshold).astype(float)
            # Adjust to meet sum constraint

            target = n / 2.0
            if s > target:
                # Need to turn some 1s to 0s

                np.random.shuffle(ones)

                    if s == target:
                        break
                    bb[i] = 0.0
                    s -= 1
            elif s < target:
                zeros = np.where(bb == 0.0)[0]
                np.random.shuffle(zeros)
                for i in zeros:
                    if s == target:
                        break
                    bb[i] = 1.0

                lambda x: smooth_obj(x, n, 50),
                bb,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,

                best_h = h_trial.copy()
    
    # Final answer
    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj_func(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    scaled = scaled - m
    return (np.log(np.sum(np.exp(scaled))) / temp + m / temp) * dx


def sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = np.sum(free)
        if nfree == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h


def slsqp_solve(x0, n, bounds, temp, maxiter):
    cons = {'type': 'eq', 'fun': sum_constraint, 'args': (n,)}
    res = minimize(
        smooth_obj_func, x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[cons],
        args=(n, temp),
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return res.x


def try_improve(cand_h, best_h, best_c5, n, bounds, t_start, budget_s, time_margin=30):

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
            cand_h = h_t.copy()
    return best_h, best_c5
```
