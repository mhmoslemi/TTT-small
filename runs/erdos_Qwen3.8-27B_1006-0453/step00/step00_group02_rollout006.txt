        maxfev = max(600, int(2500 * (1.0 - 0.5 * frac)))
        res = minimize(obj_func, x0, method='COBYLA', bounds=bounds,
                       constraints=constraints,
                       options={'maxfev': maxfev, 'rhobeg': 0.02})
        h_new = repair(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 4: Binary swap local search
    if budget_s - 25 > (time.time() - t0):
        h_bin = make_binary(n, rng)
        c5_bin = compute_c5(h_bin)
        for it in range(500):
            if budget_s - 15 < (time.time() - t0):
                break
            i_sw = int(rng.integers(0, n))
            j_sw = int(rng.integers(0, n))
            if i_sw == j_sw:
                continue
            h_bin[i_sw], h_bin[j_sw] = h_bin[j_sw], h_bin[i_sw]
            c5_try = compute_c5(h_bin)
            if c5_try > c5_bin + 1e-14:
                h_bin[i_sw], h_bin[j_sw] = h_bin[j_sw], h_bin[i_sw]
            elif c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_bin.copy()
                c5_bin = c5_try
            else:
                c5_bin = min(c5_bin, c5_try)
            if it % 100 == 99 and rng.random() < 0.4:
                h_bin = make_binary(n, rng)
                c5_bin = compute_c5(h_bin)

    # Phase 5: Fine coordinate descent
    if budget_s - 10 > (time.time() - t0):
        stagnant = 0
        for _ in range(3000):
            if budget_s - 5 < (time.time() - t0):
                break
            idx = int(rng.integers(0, n))
            found = False
            for scale in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.015, -0.015]:
                h_trial = best_h.copy()
                h_trial[idx] = np.clip(h_trial[idx] + scale, 0.0, 1.0)
                diff = n * 0.5 - np.sum(h_trial)
                if abs(diff) > 1e-10:
                    mask = np.ones(n, dtype=bool)
                    mask[idx] = False
                    if diff > 0:
                        mask &= (h_trial < 1.0 - 1e-9)
                    else:
                        mask &= (h_trial > 1e-9)
                    others = np.where(mask)[0]
                    if len(others) == 0:
                        continue
                    other = int(rng.integers(0, len(others)))
                    h_trial[others[other]] = np.clip(h_trial[others[other]] + diff, 0.0, 1.0)
                    if abs(np.sum(h_trial) - n * 0.5) > 0.05:
                        continue
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-14:
                    best_c5 = c5_trial
                    best_h = h_trial
                    found = True
                    stagnant = 0
                    break
            if not found:
                stagnant += 1
                if stagnant > 500:
                    break

    return best_h, best_c5, n
