import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _swap_perturb(h, n, rng, step):
    i, j = rng.randint(0, n, 2)
    if i == j:
        return None
    d = rng.uniform(-step, step)
    hi, hj = h[i] + d, h[j] - d
    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
        return None
    h[i] = hi
    h[j] = hj
    return h

def _block_shift(h, n, rng):
    blen = rng.randint(2, min(14, n - 2))
    start = rng.randint(0, n - blen)
    sh = rng.choice(np.array([-2, -1, 1, 2]))
    ns = start + sh
    if ns < 0 or ns + blen > n:
        return None
    tmp = h[start:start + blen].copy()
    h[start:start + blen] = h[ns:ns + blen]
    h[ns:ns + blen] = tmp
    return h

def _multi_perturb(h, n, rng, step):
    npts = rng.randint(2, 5)

        if 0 <= h[idx[ii]] + d <= 1 and 0 <= h[comp] - d <= 1:
            h[idx[ii]] += d
            h[comp] -= d
            ok = True
    return h if ok else None
