    hb = (h > 0.5).astype(float)

                constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 400, 'ftol': 1e-12}
            )
            x_new = project_feasible(res.x, n)
            c5_new = compute_c5(x_new, dx)
            if c5_new < best_c5:
                best_h = x_new
                best_c5 = c5_new
            hb3 = round_binary(x_new, n)
            c5b3 = compute_c5(hb3, dx)
            if c5b3 < best_c5:
                best_h = hb3.copy()
                best_c5 = c5b3
        except Exception:
            pass

    # Nelder-Mead local refinement on raw max
    if time.time() - t0 < budget_s - 30:
        def nm_obj(x, n_v, dx_v):
            xc = project_feasible(x, n_v)
            return compute_c5(xc, dx_v)
        try:
            res = minimize(nm_obj, best_h, args=(n, dx),
                           method='Nelder-Mead',
                           options={'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-12})
            x_new = project_feasible(res.x, n)
            c5_new = compute_c5(x_new, dx)
            if c5_new < best_c5:
                best_h = x_new
                best_c5 = c5_new
        except Exception:
            pass

    # Try higher resolutions
    for n_try in [84, 168, 336]:
        if time.time() - t0 > budget_s - 60:
            break
        if n_try >= 1000:
            break
        x_lim = np.linspace(0, 2, n, endpoint=False)
        y_lim = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(y_lim, x_lim, best_h)
        h_interp = project_feasible(h_interp, n_try)
        dx_try = 2.0 / n_try
        c5_try = compute_c5(h_interp, dx_try)
        if c5_try <= best_c5:
            try:
                res = minimize(
                    smooth_obj, h_interp, args=(n_try, 0.01, dx_try),
                    jac=smooth_grad,
                    method='SLSQP',
                    constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n_try,)}],
                    bounds=[(0.0, 1.0)] * n_try,
                    options={'maxiter': 300, 'ftol': 1e-12}
                )
                h_new = project_feasible(res.x, n_try)
                c5_new = compute_c5(h_new, dx_try)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    n = n_try
                    dx = dx_try
            except Exception:
                pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    final_dx = 2.0 / len(best_h)
    best_c5 = compute_c5(best_h, final_dx)

    # Verify with pre-imported evaluator if available
    try:
        verified = evaluate_erdos_solution(best_h)
        if verified < best_c5:
            best_c5 = verified
    except Exception:
        pass

    return best_h, best_c5, len(best_h)
