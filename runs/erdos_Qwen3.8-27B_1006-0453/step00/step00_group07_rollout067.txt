def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def fix_sum(h, n):
    """Adjust h to have sum exactly n/2, keeping in [0,1]."""
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not mask.any():
        return h
    h = h.copy()
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0, 1)
    # Fine adjustment
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            m2 = h < 1.0
        else:
            m2 = h > 0
        if m2.any():
            h[m2] += diff2 / m2.sum()
            h = np.clip(h, 0, 1)
    return h

        return h.copy()
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h.copy()
    h = h.copy()
    h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        m2 = (h < 1.0) if diff2 > 0 else (h > 0.0)
        if m2.any():
            h[m2] += diff2 / m2.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h


def sa_optimize(h0, n, dx, T0, cool, T_end, t_deadline, rng):
    h = h0.copy()
    C = compute_c5(h, dx)
    T = T0
    step_scale = 0.15
    while T > T_end:
        if time.time() >= t_deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            T *= cool
            continue
        mp = min(1.0 - h[i], h[j])
        mn = min(h[i], 1.0 - h[j])
        if mp + mn < 1e-12:
            T *= cool
            continue
        scale = step_scale * (1.0 + 0.5 * (T / T0))
        lo = max(-mn, -scale)
        hi = min(mp, scale)
        if hi <= lo + 1e-14:
            T *= cool
            continue
        d = rng.uniform(lo, hi)
        if abs(d) < 1e-14:
            T *= cool
            continue
        h[i] += d
        h[j] -= d
        Cn = compute_c5(h, dx)
        if Cn <= C or (T > 1e-14 and rng.random() < np.exp(-(Cn - C) / T)):
            C = Cn
        else:
            h[i] -= d
            h[j] += d
        T *= cool
    return h, C


def cd_optimize(h0, n, dx, t_deadline, rng, step):
    h = h0.copy()
    C = compute_c5(h, dx)
    cur_step = step
    while time.time() < t_deadline and cur_step >= 0.0005:
        improved = False
        for i in range(n):
            if time.time() >= t_deadline:
                return h, C
            best_d = 0.0
            best_C = C
            best_j = -1
            for j in range(i + 1, n):
                for sgn in (1, -1):
                    d = sgn * cur_step
                    if h[i] + d < -1e-12 or h[i] + d > 1.0 + 1e-12:
                        continue
                    if h[j] - d < -1e-12 or h[j] - d > 1.0 + 1e-12:
                        continue
                    h[i] += d
                    h[j] -= d
                    Cn = compute_c5(h, dx)
                    h[i] -= d
                    h[j] += d
                    if Cn < best_C - 1e-14:
                        best_C = Cn
                        best_d = d
                        best_j = j
            if best_d != 0.0 and best_j >= 0:
                h[i] += best_d
                h[best_j] -= best_d
                C = best_C
                improved = True
        if not improved:
            cur_step *= 0.6
    return h, C


def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    except (NameError, TypeError):
        h0 = np.zeros(80)
        h0[:40] = 1.0

    n0 = len(h0)
    dx0 = 2.0 / n0
    h_best = fix_sum(h0, n0)
    c5_best = compute_c5(h_best, dx0)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SA from parent (multiple passes)
    for pass_i in range(3):
        rem = remaining()
        if rem < 15:
            break
        alloc = rem * (0.12 if pass_i == 0 else 0.08)
        T0 = 3.0 / (pass_i + 1)
        cool = 0.9996 if pass_i == 0 else 0.9998
        h1, C1 = sa_optimize(h_best, n0, dx0, T0, cool, 0.005, time.time() + alloc, rng)
        if C1 < c5_best:
            h_best, c5_best = h1, C1

    # Phase 2: CD polish at parent n
    rem = remaining()
    if rem > 15:
        h1, C1 = cd_optimize(h_best, n0, dx0, time.time() + rem * 0.12, rng, 0.02)
        if C1 < c5_best:
            h_best, c5_best = h1, C1

    # Phase 3: Explore different n values
    n_options = [40, 64, 96, 120, 160, 200, 280, 320]
    for n_try in n_options:
        rem = remaining()
        if rem < 35:
            break
        if n_try == len(h_best):
            continue
        dx_try = 2.0 / n_try
        h_try = np.clip(interpolate_h(h_best, n_try), 0.0, 1.0)
        h_try = fix_sum(h_try, n_try)

        alloc = rem * 0.07
        h1, C1 = sa_optimize(h_try, n_try, dx_try, 2.0, 0.9996, 0.01, time.time() + alloc, rng)

        rem = remaining()
        if rem > 12:
            h1, C1 = cd_optimize(h1, n_try, dx_try, time.time() + min(rem * 0.04, 18), rng, 0.01)

        if C1 < c5_best:
            h_best, c5_best = h1, C1

    # Phase 4: Random restarts at best n
    n_cur = len(h_best)
    dx_cur = 2.0 / n_cur
    for _ in range(5):
        rem = remaining()
        if rem < 25:
            break
        h_rand = rng.uniform(0.0, 1.0, n_cur)
        h_rand = fix_sum(h_rand, n_cur)
        alloc = rem * 0.07
        h1, C1 = sa_optimize(h_rand, n_cur, dx_cur, 4.0, 0.9994, 0.01, time.time() + alloc, rng)
        if C1 < c5_best:
            h_best, c5_best = h1, C1

    # Phase 5: Try structured starts (sinusoidal, block patterns)
    for pattern in range(3):
        rem = remaining()
        if rem < 20:
            break
        x = np.linspace(0.0, 2.0, n_cur, endpoint=False)
        if pattern == 0:
            h_s = 0.5 + 0.4 * np.sin(np.pi * x + 0.3)
        elif pattern == 1:
            h_s = 0.5 + 0.4 * np.sin(2.0 * np.pi * x)
        else:
            h_s = 0.5 + 0.35 * np.sin(3.0 * np.pi * x + 0.5)
        h_s = np.clip(h_s, 0.0, 1.0)
        h_s = fix_sum(h_s, n_cur)
        alloc = rem * 0.05
        h1, C1 = sa_optimize(h_s, n_cur, dx_cur, 1.5, 0.9997, 0.01, time.time() + alloc, rng)
        if C1 < c5_best:
            h_best, c5_best = h1, C1

    # Phase 6: Final refinement SA + CD
    n_cur = len(h_best)
    dx_cur = 2.0 / n_cur
    rem = remaining()
    if rem > 15:
        h1, C1 = sa_optimize(h_best, n_cur, dx_cur, 0.4, 0.9999, 0.001, time.time() + rem * 0.3, rng)
        if C1 < c5_best:
            h_best, c5_best = h1, C1
    rem = remaining()
    if rem > 8:
        h1, C1 = cd_optimize(h_best, n_cur, dx_cur, time.time() + rem * 0.5, rng, 0.005)
        if C1 < c5_best:
            h_best, c5_best = h1, C1

    c5_final = compute_c5(h_best, 2.0 / len(h_best))
    return h_best, c5_final, len(h_best)
