import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            mask = h < 1.0 - 1e-10
            count = np.sum(mask)
            if count > 0:
                h[mask] += diff / count
                h = np.clip(h, 0, 1)
        else:
            mask = h > 1e-10
            count = np.sum(mask)
            if count > 0:
                h[mask] += diff / count
                h = np.clip(h, 0, 1)
    # Fix any remaining small error
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0, 1)
    return h

import numpy as np
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h[n // 2] = np.clip(h[n // 2] + diff, 0.0, 1.0)
    return h

def grad_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = corr.max()
    tol = max(1e-8, mv * 1e-6)
    idxs = np.where(corr >= mv - tol)[0]
    g = np.zeros(n)
    for li in idxs:
        lag = li - (n - 1)
        if lag >= 0:
            a = h[lag:]
            b = h[:n - lag]
            g[lag:] += 1.0 - b
            g[:n - lag] -= a
        else:
            s = -lag
            a = h[:n - s]
            b = h[s:]
            g[:n - s] += 1.0 - b
            g[s:] -= a
    return g, corr, mv

def targeted_swap(h, corr, n, rng):
    """Find worst lag and do targeted improvement."""
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    c5_val = corr.max() * (2.0 / n)
    
    if lag >= 0:
        # c[lag] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        # To reduce: transfer from h[i+lag] to h[i] where h[i+lag]*(1-h[i]) is large
        a = h[lag:]
        b = h[:n - lag]
        contrib = a * (1.0 - b)
    else:
        s = -lag
        a = h[:n - s]
        b = h[s:]
        contrib = a * (1.0 - b)
    
    # Find the top contributor
    top_idx = np.argmax(contrib)
    amt = min(contrib[top_idx], 0.5) * rng.uniform(0.3, 0.8)
    
    if lag >= 0:
        i_from = top_idx + lag
        i_to = top_idx
    else:
        s = -lag
        i_from = top_idx
        i_to = top_idx + s
    
    h[i_from] -= amt
    h[i_to] += amt
