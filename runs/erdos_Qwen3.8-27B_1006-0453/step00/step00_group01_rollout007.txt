import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(s)) / alpha

def project_sum(h, target, n_iters=20):
    h = h.copy()
    for _ in range(n_iters):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask_pos = h < 1.0
        mask_neg = h > 0.0
        if diff > 0:
            cnt = np.sum(mask_pos)
            if cnt == 0:
                break
            h[mask_pos] += diff / cnt
        else:
            cnt = np.sum(mask_neg)
            if cnt == 0:
                break
            h[mask_neg] += diff / cnt
        h = np.clip(h, 0, 1)
    return h

def optimize_lbfgsb(h, n, time_limit):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alphas = [15, 40, 100, 250, 600]
    t0 = time.time()
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        def obj(params):
            h_local = np.zeros(n)
            h_local[:-1] = params
            h_local[-1] = target - np.sum(params)
            if h_local[-1] < -0.05 or h_local[-1] > 1.05:
                penalty = abs(max(h_local[-1], 0) - 1 - min(h_local[-1], 1))
                return 10.0 + penalty
            h_local = np.clip(h_local, 0, 1)
            h_local = project_sum(h_local, target)
            dx = 2.0 / n
            corr = np.correlate(h_local, 1 - h_local, mode="full") * dx
            return smooth_max(corr, alpha)
        bounds = [(0, 1)] * (n - 1)
        x0 = best_h[:-1].copy()
        try:
            res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10})
            h_new = np.zeros(n)
            h_new[:-1] = res.x
            h_new[-1] = target - np.sum(res.x)
            h_new = np.clip(h_new, 0, 1)
            h_new = project_sum(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def pair_swap_cd(h, n, max_sweeps=60, tol=1e-10):
    target = n / 2.0
    h = h.copy().astype(float)
    dx = 2.0 / n
    m_corr = 2 * n - 1
    corr0 = np.correlate(h, 1 - h, mode="full").copy()
    n_grid = 21
    for sweep in range(max_sweeps):
        improved = 0
        for i in range(n):
            for j in range(i + 1, n):
                t_lo = max(-h[i], h[j] - 1.0)
                t_hi = min(1.0 - h[i], h[j])
                if t_lo >= t_hi - 1e-12:
                    continue
                A = np.zeros(m_corr)
                for k in range(-(n - 1), n):
                    idx = k + (n - 1)
                    if 0 <= i - k < n:
                        A[idx] += (1.0 - h[i - k])
                    if 0 <= i + k < n:
                        A[idx] -= h[i + k]
                    if 0 <= j - k < n:
                        A[idx] -= (1.0 - h[j - k])
                    if 0 <= j + k < n:
                        A[idx] += h[j + k]
                if np.all(np.abs(A) < 1e-14):
                    continue
                cur_max = np.max(corr0)
                trial_ts = np.linspace(t_lo, t_hi, n_grid)
                best_t = 0.0
                best_val = cur_max
                for t in trial_ts:
                    v = np.max(corr0 + t * A)
                    if v < best_val:
                        best_val = v
                        best_t = t
                if best_val < cur_max - tol:
                    h[i] += best_t
                    h[j] -= best_t
                    corr0 += best_t * A
                    improved += 1
        if improved == 0:
            break
    h = project_sum(h, target)
    return h, compute_c5(h)

def interpolate_h(h_in, n_in, n_out):

        return h_in.copy()
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    h_out = np.interp(x_out, x_in, h_in)
    return h_out
