    t_end = t0 + budget_s
    
    best_global_c5 = float('inf')
    best_global_h = None
    best_global_n = 95
    
    # Try initial_h_values
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except:
        pass
    
    n_candidates = []
    if h_init is not None and len(h_init) > 0:
        n_candidates.append(len(h_init))
    for nc in [95, 94, 96, 97, 98]:
        if nc not in n_candidates:
            n_candidates.append(nc)
    
    time_per_n = max(30, (budget_s - 20) / len(n_candidates))
    
    for idx, n in enumerate(n_candidates):
        t_phase_end = min(t_end - 10, t0 + (idx + 1) * time_per_n)
        if time.time() > t_phase_end - 5:
            continue
        target = n / 2.0
        
        if n == len(h_init) if h_init is not None else False:
            h = snap_binary(h_init, target)
        else:
            h = snap_binary(rng.rand(n), target)
        
        cur_c5 = compute_c5(h)
        best_h = h.copy()
        best_c5 = cur_c5
        
        # Phase 1: Local search
        h, best_c5 = local_search(h, target, best_c5, rng, min(t_phase_end - 10, t0 + time_per_n * 0.4))
        if best_c5 < best_global_c5:
            best_global_c5 = best_c5
            best_global_h = h.copy()
            best_global_n = n
        
        # Phase 2: Simulated annealing
        h = best_h.copy()
        best_c5, best_h = sa_search(h, target, best_c5, best_h, rng, min(t_phase_end - 5, t0 + time_per_n * 0.7))
        if best_c5 < best_global_c5:
            best_global_c5 = best_c5
            best_global_h = best_h.copy()
            best_global_n = n
        
        # Phase 3: Fine tuning
        h = best_h.copy()
        best_c5, best_h = fine_tune(h, target, best_c5, best_h, rng, t_phase_end - 3)
        if best_c5 < best_global_c5:
            best_global_c5 = best_c5
            best_global_h = best_h.copy()
            best_global_n = n
    
    # Final round: more aggressive optimization on best
    if best_global_h is not None and time.time() < t_end - 15:
        h = best_global_h.copy()
        n = best_global_n
        target = n / 2.0
        # Mixed flips and continuous moves
        T = 0.002
        for _ in range(1000000):
            if time.time() > t_end - 5:
                break
            mode = rng.random()
            if mode < 0.6:
                new_h = try_flip(h, rng)
                if new_h is None:
                    continue
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                while j == i:
                    j = rng.randint(n)
                di = rng.uniform(-0.03, 0.03)
                new_h = h.copy()
                new_h[i] = np.clip(new_h[i] + di, 0.0, 1.0)
                new_h[j] = np.clip(new_h[j] - di, 0.0, 1.0)
                diff = target - new_h.sum()
                if abs(diff) > 1e-12:
                    new_h[i] = np.clip(new_h[i] + diff, 0.0, 1.0)
            
            new_c5 = compute_c5(new_h)
            cur_c5 = compute_c5(h)
            if new_c5 < best_global_c5:
                best_global_c5 = new_c5
                best_global_h = new_h.copy()
            delta = new_c5 - cur_c5
            if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
                h = new_h
            T *= 0.99995
            if T < 1e-7:
                T = 0.001
    
    if best_global_h is None:
        n = 95
        best_global_h = snap_binary(rng.rand(n), n / 2.0)
        best_global_n = n
        best_global_c5 = compute_c5(best_global_h)
    
    # Final fix
    n = best_global_n
    target = n / 2.0
    best_global_h = np.clip(best_global_h, 0.0, 1.0)
    best_global_h = fix_sum(best_global_h, target)
    best_global_c5 = compute_c5(best_global_h)
    
    return best_global_h, best_global_c5, best_global_n
