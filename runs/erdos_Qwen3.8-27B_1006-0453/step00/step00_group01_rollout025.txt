def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

    n = len(h)

    f = compute_f(h)
    max_f = f.max()

    w = exp_f / exp_f.sum()
