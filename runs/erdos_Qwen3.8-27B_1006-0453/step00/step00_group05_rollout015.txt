import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def local_search(h, n, max_rounds=500, time_limit=50):
    """Best-improvement local search with single swaps."""
    h = h.copy()
    start = time.time()
    c5 = compute_c5(h, n)
    n_pos = np.where(h > 0.5)[0]  # positions where h=1
    n_neg = np.where(h < 0.5)[0]  # positions where h=0
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        best_delta = 0
        best_pair = None
        best_new_c5 = c5
        
        for i in n_pos:
            for j in n_neg:
                if i == j:
                    continue
                # Try swap
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h, n)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            n_pos = np.where(h > 0.5)[0]
            n_neg = np.where(h < 0.5)[0]
            c5 = best_new_c5
        else:
            break

import numpy as np
from scipy.optimize import minimize
import time

_n_global = [70]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def to_binary(h, n):
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    cur = int(np.sum(h))
    if cur > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:cur - target]] = 0.0
    elif cur < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - cur]] = 1.0
    return h

def constraint_fun(x):
    return np.sum(x) - _n_global[0] / 2.0

def local_search_binary(h, n, time_limit=50):
    h = to_binary(h, n).copy()
    t0 = time.time()
    c5 = compute_c5(h, n)
    stall = 0
    while time.time() - t0 < time_limit:
        pos = np.where(h == 1)[0]
        neg = np.where(h == 0)[0]
        improved = False
        for _ in range(80):

            if nc < c5 - 1e-14:
                c5 = nc
                improved = True
                break
            h[i], h[j] = 1.0, 0.0
        if not improved:
            best_nc, best_pair = c5, None
            for i in pos:
                for j in neg:
                    h[i], h[j] = 0.0, 1.0
                    nc = compute_c5(h, n)
                    if nc < best_nc - 1e-14:
                        best_nc, best_pair = nc, (i, j)
                    h[i], h[j] = 1.0, 0.0
            if best_pair:
                h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
                c5 = best_nc
                improved = True
        if not improved:
            stall += 1
            if stall > 6:
                break
            pos = np.where(h == 1)[0]
            neg = np.where(h == 0)[0]
            for _ in range(np.random.randint(2, 5)):
                i = pos[np.random.randint(len(pos))]
                j = neg[np.random.randint(len(neg))]
                h[i], h[j] = 0.0, 1.0
            stall = 0
    return h, c5

def sa_binary(h, n, time_limit=40):
    h = to_binary(h, n).copy()
    t0 = time.time()
    c5 = compute_c5(h, n)
    best_c5, best_h = c5, h.copy()
    T = 0.04
    while time.time() - t0 < time_limit:
        T = max(T * 0.9993, 0.0005)
        pos = np.where(h == 1)[0]
        neg = np.where(h == 0)[0]
        i = pos[np.random.randint(len(pos))]
        j = neg[np.random.randint(len(neg))]
        h[i], h[j] = 0.0, 1.0
        nc = compute_c5(h, n)
        d = nc - c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-12)):
            c5 = nc
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def slsqp_opt(h0, n, maxiter=200):
    _n_global[0] = n
    h0 = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    try:
        res = minimize(compute_c5, h0, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-8})
        hx = np.clip(res.x, 0, 1)
        return hx, compute_c5(hx, n)
    except Exception:
        return h0, compute_c5(h0, n)

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old).clip(0, 1)
    s = h_new.sum()
    if s > 0:
        h_new *= (n_new / 2.0) / s
        h_new = h_new.clip(0, 1)
    return h_new

def random_binary(n):

    idx = np.random.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
