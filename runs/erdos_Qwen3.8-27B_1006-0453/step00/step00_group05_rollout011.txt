import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def make_valid(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Iterative projection
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Add/subtract proportionally to headroom
        room_up = 1.0 - h
        room_down = h
        total_room = room_up.sum() + room_down.sum()
        if diff > 0:
            h += diff * room_up / room_up.sum() if room_up.sum() > 0 else diff / n
        else:
            h += diff * room_down / room_down.sum() if room_down.sum() > 0 else diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def local_optimize(h, n, rng, budget_s):
    h = h.astype(float).copy()
    t0 = time.time()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    T = 0.005
