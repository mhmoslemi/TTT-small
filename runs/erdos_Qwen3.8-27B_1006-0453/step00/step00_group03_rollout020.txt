def softmax_objective(params, n, beta, dx):
    h = project_onto_manifold(params, n / 2.0)
    corr = compute_corr(h, dx)
    # Soft max
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    return max_c + np.log(np.sum(exps)) / beta

def softmax_gradient(params, n, beta, dx):
    h = project_onto_manifold(params, n / 2.0)
    corr = compute_corr(h, dx)
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(len(corr)):
        s = k - (n - 1)
        grad += w[k] * compute_grad_single(h, s, n)
    return grad

import numpy as np

def project_onto_manifold(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} via bisection on water level."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_at_lag(h, s, n):
    """d/dh of correlation at shift s."""
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[s:] += 1.0 - h[:m]
            g[:m] -= h[s:]
    else:
