import numpy as np
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = float(x.min()) - 1.0, float(x.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(y)
    if abs(diff) > 1e-10:
        free = (y > 1e-8) & (y < 1.0 - 1e-8)
        nf = int(np.sum(free))
        if nf > 0:
            y[free] += diff / nf
    return np.clip(y, 0.0, 1.0)

def grad_top_lags(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = min(top_k, len(corr))
    top_idx = np.argsort(corr)[-top_k:][::-1]
    grad = np.zeros(n)
    for k in top_idx:
        d = int(k) - (n - 1)
        istart = max(0, -d)
        iend = min(n, n - d)
        if iend > istart:
            idx1 = np.arange(istart, iend)
            idx2 = idx1 + d
            grad[idx1] += (1.0 - h[idx2])
            grad[idx2] += (-h[idx1])
    return grad

def optimize_h(h_init, max_time_s, max_iter):
    n = len(h_init)
    h = project_feasible(h_init.astype(float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.05
    stuck = 0
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > max_time_s:
            break
        tk = min(9, 3 + it // 200)
        grad = grad_top_lags(h, tk)
        h_trial = project_feasible(h - lr * grad, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-12:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
            stuck = 0
            lr = min(lr * 1.03, 0.2)
        else:
            stuck += 1
            lr *= 0.82
            if lr < 5e-5:
                h = project_feasible(h + np.random.randn(n) * 0.015, n)
                lr = 0.03
                stuck = 0
        if stuck > 800:
            break
    return best_h, best_c5

def binarize(h):
    n = len(h)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin
