def grid_adaptation(h_ref):
    if len(h_ref) <= n//2:  # Coarser to finer
        # Downsample nearest neighbor but ensure integral preservation
        h_coarse = np.zeros(n//2)
        for i in range(n//2):
            idx1 = 2*i
            idx2 = (2*i+1) % n if n//2 <= n else 2*i - n
            h_coarse[i] = (h_ref[idx1] + h_ref[idx2]) / 2
        return h_coarse * (n/(n//2))
