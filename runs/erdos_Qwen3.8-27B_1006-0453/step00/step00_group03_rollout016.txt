Wait, I realize I need to be more careful about the correlation. Let me double-check what np.correlate(h, 1-h, mode='full') computes.

For arrays a (length N) and v (length M):
np.correlate(a, v, 'full')[k] = sum_n a[n+k] * v[n]  (for valid k)

More precisely, the full correlation has length N+M-1. The index k in the result corresponds to:
result[k] = sum_n a[n + (k - (M-1))] * v[n]  for k = 0, ..., N+M-2

When N = M = n (our case):
result[k] = sum_{i=0}^{n-1} a[i + k - (n-1)] * v[i]  (with a[j]=0 for j outside [0,n-1])

So for k = n-1 (the middle), we get sum_i a[i] * v[i] (zero lag)
For k = n-1 + d, we get sum_i a[i+d] * v[i] (positive lag d)
For k = n-1 - d, we get sum_i a[i-d] * v[i] (negative lag d)

In our case, a=h, v=1-h:
result[n-1+d] = sum_i h[i+d] * (1-h[i])

The overlap at shift k is: sum_i h[i] * (1-h[i+k]) = sum_i h[i-k] * (1-h[i]) = result[n-1-k]

So the overlap at shift k corresponds to index n-1-k in the full correlation.

Since we take max over ALL k (including negative), we take max over all indices of the full correlation. So max(np.correlate(h, 1-h, mode='full')) is correct (times dx gives the integral approximation).

    if time.time() - t0 < budget_s * 0.3:
        h_ls, c5_ls = local_search(best_h, best_n, best_n * 60, seed + 1,
                                   budget_s * 0.35, t0)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls

    # Phase 3: Binary hill climbing
    if time.time() - t0 < budget_s * 0.45:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin > 0.5))
        target_ones = n // 2
        if n_ones > target_ones:
            idx = np.where(h_bin > 0.5)[0]
            for k in range(n_ones - target_ones):
                h_bin[idx[k]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin <= 0.5)[0]
            for k in range(target_ones - n_ones):
                h_bin[idx[k]] = 1.0
        h_b, c5_b = binary_hill_climb(h_bin, best_n, best_n * 30, seed + 2,
                                      budget_s * 0.5, t0)
        if c5_b < best_c5:
            best_c5, best_h, best_n = c5_b, h_b, best_n

    # Phase 4: Multi-start Powell
    if time.time() - t0 < budget_s * 0.6:
        target = best_n / 2.0
        for trial in range(4):
            if time.time() - t0 > budget_s * 0.7:
                break
            h_rand = best_h + 0.15 * rng.randn(best_n)
            h_rand = fix_sum(h_rand, best_n)
            try:
                res = minimize(obj_with_penalty, h_rand, args=(best_n, target, 100.0),
                               method='Powell', options={'maxiter': 200, 'xtol': 1e-9})
                h_try = fix_sum(np.clip(res.x, 0, 1), best_n)
                c5_try = corr_max_val(h_try, best_n)
                if c5_try < best_c5:
                    best_c5, best_h, best_n = c5_try, h_try, best_n
            except Exception:
                pass

    # Phase 5: Higher resolution attempts
    for n2 in [100, 140]:
        if time.time() - t0 > budget_s * 0.82:
            break
        h0 = interpolate_to(best_h, best_n, n2)
        h0 = fix_sum(h0, n2)
        target2 = n2 / 2.0
        try:
            res = minimize(obj_with_penalty, h0, args=(n2, target2, 100.0),
                           method='Powell', options={'maxiter': 200, 'xtol': 1e-9})
            h_try = fix_sum(np.clip(res.x, 0, 1), n2)
            c5_try = corr_max_val(h_try, n2)
            if c5_try < best_c5:
                best_c5, best_h, best_n = c5_try, h_try, n2
        except Exception:
            pass

        if time.time() - t0 < budget_s * 0.88:
            h_ls, c5_ls = local_search(best_h, best_n, best_n * 25, seed + 10,
                                       budget_s * 0.92, t0)
            if c5_ls < best_c5:
                best_c5, best_h = c5_ls, h_ls

    # Phase 6: Final local polish
    if time.time() - t0 < budget_s * 0.95:
        h_ls, c5_ls = local_search(best_h, best_n, best_n * 40, seed + 99,
                                   budget_s * 0.98, t0)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls

    # Ensure final feasibility
    best_h = fix_sum(best_h, best_n)
    best_c5 = corr_max_val(best_h, best_n)

    return best_h, best_c5, best_n
