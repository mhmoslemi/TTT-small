def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm == 0:
            if diff > 0:
                idx = np.random.randint(0, n, size=min(int(np.ceil(diff)) + 1, n))
                h[idx] = 1.0
            else:
                idx = np.random.randint(0, n, size=min(int(np.ceil(-diff)) + 1, n))
                h[idx] = 0.0
            continue
        h[mask] += diff / nm
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_linear(h_parent, n_new):
    n_parent = len(h_parent)
    x_old = (np.arange(n_parent) + 0.5) * (2.0 / n_parent)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h_parent, kind='linear',
                 fill_value=(h_parent[0], h_parent[-1]), bounds_error=False)
    h_new = f(x_new)
    return fix_sum(h_new, n_new)

            h_cd, c5_cd = coord_descent(hs, n, max_passes=150)
            _try(h_cd, n)
            # SLSQP if budget allows
            if time.time() < _g_deadline and n <= 300:
                h_sl, c5_sl = slsqp_opt(h_cd, n, maxiter=150)
                _try(h_sl, n)

    # Final polish on best
    if best_h is not None and time.time() < _g_deadline:
        h_pol, c5_pol = coord_descent(best_h, best_n, max_passes=200, tol=1e-12)
        _try(h_pol, best_n)
        if time.time() < _g_deadline and best_n <= 250:
            h_sl2, c5_sl2 = slsqp_opt(h_pol, best_n, maxiter=100)
            _try(h_sl2, best_n)
            h_pol2, c5_pol2 = coord_descent(h_sl2, best_n, max_passes=200, tol=1e-12)
            _try(h_pol2, best_n)

    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)
        best_h = fix_sum(best_h, best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
