def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c


def smooth_obj(h, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(c)
    return mc + np.log(np.sum(np.exp(t * (c - mc)))) / t
