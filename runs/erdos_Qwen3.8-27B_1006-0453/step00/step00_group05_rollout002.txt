import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx, c

def upsample(parent, new_n):
    """Upsample parent array to new_n points while preserving integral"""
    n_old = len(parent)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / new_n
    
    # Linear interpolation
    old_x = np.linspace(0, 2, n_old, endpoint=False) + dx_old/2
    new_x = np.linspace(0, 2, new_n, endpoint=False) + dx_new/2
    
    # Use step function interpolation (nearest)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        # Find which old cell this new cell belongs to
        pos = new_x[i]
        old_idx = int(pos / dx_old)
        old_idx = min(old_idx, n_old - 1)
        h_new[i] = parent[old_idx]
    
    # Adjust to get correct sum
    target_sum = new_n / 2
    current_sum = h_new.sum()
    if current_sum != target_sum:
        # Adjust: add or remove from the "middle" values
        diff = target_sum - current_sum
        # Sort by value, adjust from the extreme
        if diff > 0:
            # Need to increase: clip and redistribute
            h_new = np.clip(h_new + diff / new_n, 0, 1)
        else:
            h_new = np.clip(h_new + diff / new_n, 0, 1)
    
    # More careful adjustment
    h_new = np.clip(h_new, 0, 1)
    # Iteratively adjust to hit the target
    for _ in range(100):
        cur = h_new.sum()
        if abs(cur - target_sum) < 1e-9:
            break
        diff = target_sum - cur
        if diff > 0:
            mask = h_new < 1 - 1e-6
            if not mask.any():
                break
            h_new[mask] += diff / mask.sum()
        else:
            mask = h_new > 1e-6
            if not mask.any():
                break
            h_new[mask] += diff / mask.sum()
        h_new = np.clip(h_new, 0, 1)
    
    return h_new

def bisect_placement(n, m):
    h = np.zeros(n)
    positions = np.floor(np.arange(m) * n / m).astype(int)
    seen = set()
    for p in positions:
        p = min(p, n - 1)
        if p not in seen:
            h[p] = 1.0
