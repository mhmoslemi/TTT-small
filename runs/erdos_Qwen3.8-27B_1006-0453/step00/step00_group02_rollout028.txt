_WS = {'n': 0, 'pen': 50.0}


def _ws_fun(x):
    n = _WS['n']

    dx = 2.0 / n

    pen = _WS['pen']

    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)
    grad = np.zeros(n)

    v1 = (js >= 0) & (js < n)
    grad[v1] += (1.0 - x[js[v1]])

def _upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.clip(np.interp(x_new, x_old, h), 0.0, 1.0)


def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def _make_binary(n, rng):
    h = rng.randint(0, 2, n).astype(float)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
