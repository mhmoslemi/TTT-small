    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def scale_and_round(h_src, n_new, rng):
    n_src = len(h_src)
    x_src = np.linspace(0, 1, n_src, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_cont = np.interp(x_new, x_src, h_src)
    # Round to 0-1 maintaining sum
    target = n_new // 2
    h = (h_cont >= 0.5).astype(float)
    s = int(h.sum())
    if s < target:
        # Need more ones; pick from highest h_cont among zeros
        frac = h_cont.copy()
        frac[h == 1] = -1
        order = np.argsort(-frac)
        for idx in order:
            if s >= target:
                break
            h[idx] = 1
            s += 1
    elif s > target:

        frac[h == 0] = 10
        order = np.argsort(frac)

            s -= 1

import numpy as np
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def scale_and_round(h_src, n_new, rng):
    n_src = len(h_src)
    x_src = np.linspace(0, 1, n_src, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_cont = np.interp(x_new, x_src, h_src)
    target = n_new // 2
    h = (h_cont >= 0.5).astype(float)
    s = int(h.sum())
    if s < target:
        frac = h_cont.copy()
        frac[h == 1] = -1.0
        order = np.argsort(-frac)
        for idx in order:
            if s >= target:
                break
            h[idx] = 1.0
            s += 1
    elif s > target:
        frac = h_cont.copy()
        frac[h == 0] = 10.0
        order = np.argsort(frac)
        for idx in order:
            if s <= target:
                break
            h[idx] = 0.0
            s -= 1
    return h

def random_01(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def local_search(h, max_iters=20000, rng=None, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy().astype(float)
    best_c5 = c5_value(h)
    no_improve = 0
    max_no_improve = 300

    for it in range(max_iters):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = min(300, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            c5_new = c5_value(h)
            h[i], h[j] = 1.0, 0.0
            if c5_new < best_c5 - 1e-12:
                h[i], h[j] = 0.0, 1.0
                best_c5 = c5_new
                improved = True
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return h, best_c5

def simulated_annealing(h, max_iters=40000, rng=None, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy().astype(float)
    best_c5 = c5_value(h)
    best_h = h.copy()
    cur_c5 = best_c5
    T0, T1 = 0.004, 0.00005

    for it in range(max_iters):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        new_c5 = c5_value(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5
