def project_box_sum(x, target_sum):
    n = len(x)
    lo = -10.0
    hi = 10.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    h = np.clip(x - t, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False) + 0.5/n_old
    x_new = np.linspace(0, 1, n_new, endpoint=False) + 0.5/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new * 0.5)

def slsqp_optimize(h0, n, time_limit=60):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    def obj_grad(h):
        corr = np.correlate(h, 1-h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n-1)
        grad = np.zeros(n)
        lo1, hi1 = max(0, k), min(n, k+n)
        if lo1 < hi1:
            grad[lo1:hi1] += (1 - h[lo1-k:hi1-k])
        lo2, hi2 = max(0, -k), min(n, n-k)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2+k:hi2+k]
        return obj(h), dx * grad
    try:
        res = minimize(obj_grad, h0, method='SLSQP',
                       jac=True,
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq', 'fun': lambda h: h.sum()-n/2,
                                     'jac': lambda h: np.ones(n)}],
                       options={'maxiter': 200, 'ftol': 1e-8})
        h_opt = np.clip(res.x, 0, 1)
        h_opt = project_box_sum(h_opt, n/2)
        return h_opt
    except:
        return h0

def compute_c5_and_grad(h, tau=10.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    
    # Log-sum-exp
    cmax = corr.max()
    exp_c = np.exp(tau * (corr - cmax))
    w = exp_c / exp_c.sum()  # softmax weights, length 2n-1
    
    value = (cmax + np.log(exp_c.sum())) / tau * dx  # ≈ max(corr)*dx
    
    # Gradient
    A = 1 - h  # length n
    B = h  # length n
    w_rev = w[::-1]  # length 2n-1
    
    # first_term[j] = sum_p A[p] * w[j+n-1-p] = conv(A, w, 'full')[j+n-1]
    conv1 = np.convolve(A, w, mode='full')  # length n + (2n-1) - 1 = 3n-2
    first_term = conv1[n-1 : 2*n-1]  # length n
    
    # second_term[j] = sum_p B[p] * w[p-j+n-1] = conv(B, w_rev, 'full')[n-1+j]
    conv2 = np.convolve(B, w_rev, mode='full')  # length 3n-2
    second_term = conv2[n-1 : 2*n-1]  # length n
    
    grad = dx * (first_term - second_term)
    return value, grad
