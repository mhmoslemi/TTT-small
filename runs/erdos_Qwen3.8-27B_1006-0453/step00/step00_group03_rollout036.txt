import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for j in order:
                adj = min(1.0 - h[j], diff)
                h[j] += adj
                diff -= adj
                if abs(diff) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for j in order:
                adj = min(h[j], -diff)
                h[j] -= adj
                diff += adj
                if abs(diff) < 1e-15:
                    break
    return h

def soft_max(vals, beta):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def smooth_obj(h, beta):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return soft_max(corr, beta)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += wk * (1.0 - h[i + lag])
                grad[i + lag] += wk * (-h[i])
        else:
            la = -lag
            for i in range(la, n):
                grad[i] += wk * (1.0 - h[i - la])
                grad[i - la] += wk * (-h[i])
    return grad

def pg_optimize(h, n, beta, lrs, max_iter, deadline_fn):
    h = h.copy()
    best_local = h.copy()
    best_c = compute_c5(h)
    for lr in lrs:
        for it in range(max_iter):
            if deadline_fn():
                return best_local, best_c
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_local = h.copy()
            g = smooth_grad(h, beta)
            hn = project_feasible(h - lr * g, n)
            cn = compute_c5(hn)
            if cn < c - 1e-15:
                h = hn
            else:
                break
    return best_local, best_c
