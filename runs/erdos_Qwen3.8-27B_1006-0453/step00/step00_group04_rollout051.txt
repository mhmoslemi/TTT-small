def compute_overlaps(h):

def compute_gradient_smooth_max(h, beta):
    """Gradient of (1/beta)*log(sum_k exp(beta * c[k] * dx))"""
    n = len(h)
    dx = 2.0 / n
    c = compute_overlaps(h)  # length 2n-1
    d = c * dx  # actual overlap integrals
    
    # Softmax weights
    d_max = np.max(d)
    exp_d = np.exp(beta * (d - d_max))
    p = exp_d / np.sum(exp_d)  # shape (2n-1,)
    
    # Gradient: dx * sum_k p[k] * dc[k]/dh[j]
    # dc[k]/dh[j] = (1-h[j-s]) * 1[0<=j-s<n] - h[j+s] * 1[0<=j+s<n]
    # where s = k - n + 1, ranges from -(n-1) to (n-1)
    
    g = np.zeros(n)
    # For each shift s = k - n + 1:
    # Contribution to g[j] from shift s: p[k] * dx * [(1-h[j-s])*1[0<=j-s<n] - h[j+s]*1[0<=j+s<n]]
    
    for k in range(2*n - 1):
        s = k - n + 1
        w = p[k] * dx
        # (1-h[j-s]) for j such that 0<=j-s<n, i.e., s<=j<s+n
        j_start = max(0, s)
        j_end = min(n, s + n)
        if j_start < j_end:

        # -h[j+s] for j such that 0<=j+s<n, i.e., -s<=j<n-s
