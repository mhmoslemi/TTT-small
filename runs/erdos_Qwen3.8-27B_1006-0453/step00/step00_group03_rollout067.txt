import numpy as np
from scipy.optimize import minimize
import time

def _cost(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _grad_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()
    
    grad = np.zeros(n)
    # All lags from -(n-1) to n-1
    for k in range(-(n-1), n):
        idx = n - 1 + k
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:n] += wk * (1.0 - h[0:n-k])
            grad[0:n-k] -= wk * h[k:n]
        else:
            ak = -k
            grad[0:n-ak] += wk * (1.0 - h[ak:n])
            grad[ak:n] -= wk * h[0:n-ak]
    return grad

def _project_feasible(h, n):
    # Project to sum(h) = n/2, 0 <= h <= 1
    # Simple: scale and clip, then adjust
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        # Distribute the difference to elements not at bounds
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if not mask.any():
            # All at bounds, can't adjust
            break
        h[mask] += diff / mask.sum()
    return h

def _random_feasible(n, rng):
    h = rng.random(n)
    return _project_feasible(h, n)

    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    best_h = None
    best_c = float('inf')
    
    # Try multiple resolutions
    resolutions = [70, 140, 280]
    
    for n in resolutions:
        if time_left() < 100:
            break
        
        # Get starting points
        starts = []
        if n == 70 and initial_h_values is not None and len(initial_h_values) == 70:
            starts.append(np.array(initial_h_values, dtype=float))
        
        # Add random starts
        n_starts = min(20, max(5, int(time_left() / 30)))
        if n == 280:
            n_starts = 5
        for i in range(n_starts):
            starts.append(_random_feasible(n, rng))
        
        best_n = None
        best_cn = float('inf')

            h0 = _project_feasible(h0, n)
            
            c0 = _cost(h0)
            if c0 < best_cn:

                best_n = h0.copy()

            alpha = 50.0
            obj = lambda x: _lse_obj(x, alpha)
            grad = lambda x: _grad_lse(x, alpha)

                obj, h0, jac=grad,

                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 500, 'ftol': 1e-10}

            c_opt = _cost(h_opt)

                best_n = h_opt.copy()

                h = _project(h, n)
                improved = True
        if not improved:
