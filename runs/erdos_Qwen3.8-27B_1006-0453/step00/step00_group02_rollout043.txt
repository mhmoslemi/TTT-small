def fixsum(h, target):
    """Adjust h to have the exact target sum while keeping in [0,1]."""
    n = len(h)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the correction
    if diff > 0:
        # Need to increase sum; add to entries that are < 1
        mask = h < 1.0
        if not np.any(mask):
            return h

        add = diff / np.sum(mask)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(h, dx, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    smooth_max = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    penalty = lam * (np.sum(h) - len(h) / 2.0) ** 2
    return smooth_max + penalty

def smooth_grad(h, dx, beta, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for l in range(n):
        idx_pos = (n - 1) + l  # lag +l
        w = weights[idx_pos]
        if w > 1e-12:
            grad[l:n] += w * (1.0 - h[0:n-l])
            grad[0:n-l] -= w * h[l:n]
        if l > 0:

    """Adjust h to sum to target, keeping in [0,1]."""

    dx = 2.0 / n

    """Compute smoothed C5 objective and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    logsumexp = m + np.log(np.sum(np.exp(beta * (corr - m))))
    obj = logsumexp / beta
    # Penalty for sum constraint
    diff = np.sum(h) - n / 2.0
    obj += 0.5 * lam * diff * diff
    
    # Gradient of smooth max

    wsum = np.sum(weights)
    weights /= wsum  # softmax probabilities
    
    grad = np.zeros(n)
    for l in range(n):
        # Positive lag l
        idx_p = (n - 1) + l
        wp = weights[idx_p]
        if wp > 1e-14:
            grad[l:n] += wp * dx * (1.0 - h[0:n-l])
            grad[0:n-l] -= wp * dx * h[l:n]

np.correlate(h, 1-h, mode='full') gives output of length 2n-1.
The output at index j corresponds to:
Σ_i h[i + j - (n-1)] * (1-h[i])

where the sum is over valid i (both indices in [0, n-1]).

Let l = j - (n-1), the lag.

For l ≥ 0: valid i range: max(0, -l) ≤ i ≤ min(n-1, n-1-l) → 0 ≤ i ≤ n-1-l
For l < 0: valid i range: max(0, -l) ≤ i ≤ min(n-1, n-1-l) → -l ≤ i ≤ n-1

The gradient of the j-th output w.r.t. h[m]:
∂/∂h[m] Σ_i h[i+l]*(1-h[i])

Term 1: h[i+l] = h[m] when i = m-l. Valid if i is in valid range.
Contribution: (1-h[m-l]) * [m-l in valid range for i]

Term 2: h[i] = h[m] when i = m. Valid if m is in valid range.
Contribution: -h[m+l] * [m in valid range for i, and m+l in [0,n-1]]

For l ≥ 0:
Term 1 valid: 0 ≤ m-l ≤ n-1-l → l ≤ m ≤ n-1
Term 2 valid: 0 ≤ m ≤ n-1-l AND 0 ≤ m+l ≤ n-1 (auto for l≥0, m≤n-1-l means m+l ≤ n-1)
→ 0 ≤ m ≤ n-1-l

For l < 0 (let l' = -l):
output = Σ_{i=l'}^{n-1} h[i-l']*(1-h[i])
Term 1: h[i-l']=h[m] when i=m+l'. Valid: l' ≤ m+l' ≤ n-1 → 0 ≤ m ≤ n-1-l'. But also need i=m+l' ≤ n-1, which is m ≤ n-1-l'. And i ≥ l' means m ≥ 0. So 0 ≤ m ≤ n-1-l'.
Contribution: (1-h[m+l'])
Term 2: h[i]=h[m] when i=m. Valid: l' ≤ m ≤ n-1. Also need i-l' = m-l' ≥ 0 (auto if m ≥ l').
Contribution: -h[m-l']

So for l' > 0:
∂/∂h[m] = [0 ≤ m ≤ n-1-l']*(1-h[m+l']) - [l' ≤ m ≤ n-1]*h[m-l']

import numpy as np
from scipy.optimize import minimize
import time


def fix_sum_simple(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
    else:
        mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj_grad(h, dx, beta, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    logse = m + np.log(np.sum(np.exp(beta * (corr - m))))
    obj = logse / beta
    diff = np.sum(h) - n / 2.0
    obj += 0.5 * lam * diff * diff
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for l in range(n):
        wp = weights[(n - 1) + l]
        if wp > 1e-14:
            grad[l:n] += wp * dx * (1.0 - h[0:n - l])
            grad[0:n - l] -= wp * dx * h[l:n]
        if l > 0:
            wn = weights[(n - 1) - l]
            if wn > 1e-14:
                grad[0:n - l] += wn * dx * (1.0 - h[l:n])
                grad[l:n] -= wn * dx * h[0:n - l]
    grad += lam * diff
    return obj, grad


def obj_fun(x, dx, beta, lam):
    return smooth_obj_grad(x, dx, beta, lam)[0]


def grad_fun(x, dx, beta, lam):
    return smooth_obj_grad(x, dx, beta, lam)[1]


def resample_h(h, new_n):
    old_n = len(h)
    x_old = (np.arange(old_n) + 0.5) / old_n

    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum_simple(h_new, new_n / 2.0)
