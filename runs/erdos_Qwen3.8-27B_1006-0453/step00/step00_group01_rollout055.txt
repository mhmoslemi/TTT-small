np.correlate(a, b, mode='full') where a has length n1 and b has length n2:

So for a=h, b=1-h, both length n:

import numpy as np

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_simplex(h, ts):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > ts:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = ts - r.sum()
    if abs(d) > 1e-13:
        idx = np.where((r > 1e-12) & (r < 1.0 - 1e-12))[0]
        if len(idx):
            r[idx[0]] = np.clip(r[idx[0]] + d, 0.0, 1.0)
    return r

def cd_sweep(h, n, grid_pts=801):
    improved = False
    for j in range(n):
        old = h[j]
        h[j] = 0.0
        cb = np.correlate(h, 1.0 - h, mode='full')
        ss = np.arange(2 * n - 1) - (n - 1)
        js_raw = j + ss
        jm_raw = j - ss
        js = np.clip(js_raw, 0, n - 1)
        jm = np.clip(jm_raw, 0, n - 1)
        vjs = (js_raw >= 0) & (js_raw < n)
        vjm = (jm_raw >= 0) & (jm_raw < n)
        coeff = np.where(vjs, 1.0 - h[js], 0.0) - np.where(vjm, h[jm], 0.0)
        const = np.where(vjm, h[jm], 0.0)
        coeff[n - 1] = 1.0
        const[n - 1] = 0.0
        vs = np.linspace(0.0, 1.0, grid_pts)
        C = np.add.outer(vs, coeff) + const + cb[np.newaxis, :]
        C[:, n - 1] = cb[n - 1] + vs * (1.0 - vs)
        best_v = vs[np.argmin(C.max(axis=1))]
        h[j] = best_v
        if best_v != old:
            improved = True
    return improved
