import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, beta):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def obj_func(h, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_max(corr, beta)

def sum_constr(h, n):
    return np.sum(h) - n / 2.0

def double_res(h):
    return np.repeat(h, 2)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def penalty_obj(h, beta, n, lam):
    base = smooth_max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n), beta)
    return base + lam * (np.sum(h) - n / 2.0) ** 2

def optimize_slsqp(h, n, t0, budget_s, frac):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constr, 'args': (n,)}]
    for beta in [30.0, 60.0, 120.0, 250.0]:
        if time.time() - t0 > budget_s * frac:
            break
        try:
            res = minimize(obj_func, best_h, args=(beta, n), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                c5c = compute_c5(res.x)
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = res.x.copy()
        except Exception:
            pass
    return best_h, best_c5

def optimize_lbfgs(h, n, t0, budget_s, frac):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    lams = [0.1, 1.0, 10.0, 100.0]
    betas = [50.0, 100.0, 300.0]
    for lam in lams:
        for beta in betas:
            if time.time() - t0 > budget_s * frac:
                break
            try:
                res = minimize(penalty_obj, best_h, args=(beta, n, lam),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-14})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    hx = fix_sum(res.x, n)
                    c5c = compute_c5(hx)
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = hx.copy()
            except Exception:
                pass
    return best_h, best_c5

def perturb(h, sigma, rng):
    hp = h + sigma * rng.standard_normal(len(h))
    return fix_sum(hp, len(h))
