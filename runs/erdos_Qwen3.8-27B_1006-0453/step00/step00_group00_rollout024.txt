import time

def compute_c5(h, dx):
    """Compute C5 for a given h vector."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_to_sum(h, target_sum):
    """Project h to have sum = target_sum while maintaining [0,1] bounds."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    n = len(h)
    per_element = diff / n
    h_shifted = h + per_element
    if np.all(h_shifted >= 0) and np.all(h_shifted <= 1):
        return h_shifted
    # More careful: adjust only those not at bounds
    if diff > 0:
        # Need to increase sum: add to those below 1
        mask = h < 1.0
        room = 1.0 - h[mask]
        total_room = np.sum(room)
        if total_room < diff:
            return np.clip(h_shifted, 0, 1)  # Can't fully satisfy
        scale = diff / total_room
        # Distribute proportionally to room
        h_new = h.copy()
        h_new[mask] = h[mask] + room * scale
        return h_new
    else:
        # Need to decrease sum: subtract from those above 0
        mask = h > 0.0
        room = h[mask]
        total_room = np.sum(room)
        if total_room < -diff:
            return np.clip(h_shifted, 0, 1)
        scale = (-diff) / total_room
        h_new = h.copy()
        h_new[mask] = h[mask] - room * scale
        return h_new

                    improved = True
