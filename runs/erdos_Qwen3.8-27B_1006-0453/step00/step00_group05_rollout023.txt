import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def fix_sum(h, n):
    """Project h to sum = n/2, clip to [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Iterate to fix
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def binary_local_search(h, n, dx, rng, n_iter=20000, t_limit=8.0):
    """Best-improvement local search on binary h with sum=n/2."""
    start = time.time()
    h = h.copy()
    # Make binary
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    current = compute_c5(h_b, dx)
    
    for it in range(n_iter):
        if time.time() - start > t_limit:
            break
        improved = False
        # Try all swaps
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Random subset of swaps for speed
        n_try = min(len(ones) * len(zeros), 500)
        for _ in range(n_try):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_b[i] = 0
            h_b[j] = 1
            c = compute_c5(h_b, dx)
            if c < current:
                current = c
                improved = True
            else:
                h_b[i] = 1
                h_b[j] = 0
        if not improved:
            # Try a larger random batch
            pass
    return h_b, current

def simulated_annealing(h, n, dx, rng, t_limit=10.0):
    """Simulated annealing on continuous h."""
    start = time.time()
    h = h.copy()
    current = compute_c5(h, dx)
    T = 0.01
    T_min = 0.0001
    
    while time.time() - start < t_limit and T > T_min:
        # Random move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.normal(0, 0.05)
        old_i, old_j = h[i], h[j]
        h[i] = np.clip(old_i + delta, 0, 1)
        h[j] = np.clip(old_j - (h[i] - old_i), 0, 1)  # preserve sum approximately
        h = fix_sum(h, n)
        
        new_c = compute_c5(h, dx)
        diff = new_c - current
        if diff < 0 or rng.random() < np.exp(-diff / T):
            current = new_c
        else:
            h[i], h[j] = old_i, old_j
            h = fix_sum(h, n)
        
        T *= 0.999
    
    return h, current

        # Compute correlation and find active lag
