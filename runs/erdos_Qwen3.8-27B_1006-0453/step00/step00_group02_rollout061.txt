```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    n = len(h)
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def coord_update(h, i, n):
    """Optimize coordinate i, return updated h."""
    old = h[i]
    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old  # restore
    
    # Compute a_k, b_k, q_k for each k
    m = 2 * n - 1
    a = np.zeros(m)
    b = base.copy()
    q = np.zeros(m)
    
    for k in range(m):
        s_k = k - (n - 1)
        if s_k == 0:
            q[k] = -1.0
            a[k] = 1.0
            # b[k] = base[k] (already set)
        else:
            # cond1: h[i] as a[j], j=i, need 0<=i<n and 0<=i+s_k<n
            if 0 <= i + s_k < n:
                a[k] += (1.0 - h[i + s_k])
            # cond2: h[i] as v[j+s_k], j=i-s_k, need 0<=j<n and 0<=j+s_k<n
            j2 = i - s_k
            if 0 <= j2 < n:
                a[k] += -h[j2]
                b[k] += h[j2]
    
    # Minimize max_k (b[k] + a[k]*x + q[k]*x^2) for x in [0,1]
    grid = np.linspace(0.0, 1.0, 201)
    best_x = 0.5
    best_val = np.inf
    for x in grid:
        vals = b + a * x + q * x * x
        v = vals.max()
        if v < best_val:
            best_val = v
            best_x = x
    # Refine
    eps = 0.005
    lo_r = max(0.0, best_x - eps)
    hi_r = min(1.0, best_x + eps)
    if hi_r > lo_r:
        grid2 = np.linspace(lo_r, hi_r, 101)
        for x in grid2:
            vals = b + a * x + q * x * x
            v = vals.max()
            if v < best_val:
                best_val = v
                best_x = x
    h[i] = best_x
    return h

def coordinate_descent(h, n_sweeps, time_limit, rng):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() >= time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            h = coord_update(h, i, n)
        # Project to constraint
        h = project_constraint(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # 10s buffer
    rng = np.random.default_rng(seed)
    
    # Get initial values

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    h0 = project_constraint(h0)

    best_c5 = compute_c5(h0)

    starts = [h0.copy()]
    # Add random starts
    for _ in range(5):
        h_rand = rng.uniform(0, 1, n)
        h_rand = project_constraint(h_rand)
        starts.append(h_rand)
    # Add structured starts
    # Alternating pattern
    h_alt = np.tile(np.array([1.0, 0.0, 0.0, 1.0]), n // 4 + 1)[:n]
    h_alt = project_constraint(h_alt)
    starts.append(h_alt)
    # Block pattern
    h_block = np.zeros(n)
    h_block[::2] = 1.0
    h_block = project_constraint(h_block)
    starts.append(h_block)
    
    # Optimize each start
    for idx, h_start in enumerate(starts):
        remaining = time_limit - time.time()
        if remaining < 30:

        alloc = remaining / len(starts[idx:])  # equal split
        n_sweeps = max(10, int(alloc / 0.5))  # estimate 0.5s per sweep
        h_opt, c5 = coordinate_descent(h_start, n_sweeps, time_limit, rng)
        if c5 < best_c5:

            best_h = h_opt

        for i in range(n):

            h2[2*i+1] = best_h[i]
        h2 = project_constraint(h2)
        
        # Refine at higher resolution
        n_sweeps = int(remaining / 1.0)  # estimate 1s per sweep at higher res

        if c5_2 < best_c5:

            best_h = h2_opt
    
    # Final evaluation

    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
```

```python
def coord_update_fast(h, i, n):
    """Optimize h[i] using 3-point interpolation."""
    n_corr = 2 * n - 1
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 0.5
    corr_half = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    
    # Quadratic fit: f(x) = Q*x^2 + L*x + C
    C = corr0
    Q = 2.0 * corr0 + 2.0 * corr1 - 4.0 * corr_half
    L = -corr0 - corr1 + 4.0 * corr_half
    
    # Minimize max_k (Q[k]*x^2 + L[k]*x + C[k]) for x in [0,1]
    best_x = 0.5
    best_val = np.inf
    # Coarse grid
    for x in np.linspace(0, 1, 101):
        v = (Q * x * x + L * x + C).max()
        if v < best_val:
            best_val = v
            best_x = x
    # Fine grid
    eps = 0.01
    for x in np.linspace(max(0, best_x-eps), min(1, best_x+eps), 101):
        v = (Q * x * x + L * x + C).max()
        if v < best_val:
            best_val = v
            best_x = x
    
    h[i] = best_x
    return h
```
