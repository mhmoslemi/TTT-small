import numpy as np
import time

def compute_corr(h):
    n = len(h)
    one_minus_h = 1.0 - h
    return np.correlate(h, one_minus_h, mode='full')

def compute_D(i0, j0, h, n):
    """Compute D[lag] for the linear coefficient when h[i0]+=delta, h[j0]-=delta"""
    # lag ranges from -(n-1) to (n-1)
    # C[lag] = sum_i h[i] * (1-h[i+lag]) for valid i, i+lag
    # D[lag] = (1-h[i0+lag])_valid - h[i0-lag]_valid - (1-h[j0+lag])_valid + h[j0-lag]_valid
    D = np.zeros(2*n-1)
    for lag in range(-(n-1), n):
        # D[lag] contribution
        val = 0.0
        # i = i0, need i+lag valid
        if 0 <= i0 + lag < n:
            val += (1.0 - h[i0 + lag])
        # i = i0-lag, need i valid
        if 0 <= i0 - lag < n:
            val -= h[i0 - lag]
        # i = j0, need i+lag valid
        if 0 <= j0 + lag < n:
            val -= (1.0 - h[j0 + lag])
        # i = j0-lag, need i valid
        if 0 <= j0 - lag < n:
            val += h[j0 - lag]
        D[lag + n - 1] = val
    return D

import numpy as np
import time
import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fix_sum(h, target, rng=None):
    """Fix sum of h to target while keeping [0,1] bounds."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if rng is not None:
        idx = rng.permutation(len(h))
    else:
        idx = np.argsort(h)
    if diff > 0:
        for k in idx[::-1]:
            add = min(diff, 1.0 - h[k])
            h[k] += add
            diff -= add
            if diff <= 0:
                break
    else:
        for k in idx:
            sub = min(-diff, h[k])
            h[k] -= sub
            diff += sub
            if diff >= 0:
                break
    return np.clip(h, 0.0, 1.0)

def greedy_optimize(h, n, max_sweeps=50, best_improvement=False):
    """Greedy pairwise transfer optimization."""
    target = n / 2.0
    corr = compute_corr(h)
    
    for sweep in range(max_sweeps):
        f_current = np.max(corr)
        best_gain = 0.0
        best_i, best_j, best_d = -1, -1, 0.0
        made_change = False
        
        for i in range(n):
            for j in range(i + 1, n):
                dlo = max(-h[i], h[j] - 1.0)
                dhi = min(1.0 - h[i], h[j])
                if dhi - dlo < 1e-12:
                    continue
                
                lags = np.arange(-(n-1), n)
                ip = i + lags
                im = i - lags
                jp = j + lags
                jm = j - lags
                
                D = np.zeros(2 * n - 1)
                m1 = (ip >= 0) & (ip < n)
                m2 = (im >= 0) & (im < n)
                m3 = (jp >= 0) & (jp < n)
                m4 = (jm >= 0) & (jm < n)
                
                D[m1] += 1.0 - h[ip[m1]]
                D[m2] -= h[im[m2]]
                D[m3] -= 1.0 - h[jp[m3]]
                D[m4] += h[jm[m4]]
                
                lo, hi = dlo, dhi
                for _ in range(25):
                    t1 = 1.0 / 3.0
                    m1v = lo + (hi - lo) * t1
                    m2v = lo + (hi - lo) * (1.0 - t1)
                    if np.max(corr + m1v * D) < np.max(corr + m2v * D):
                        hi = m2v
                    else:
                        lo = m1v
                
                d_opt = 0.5 * (lo + hi)
                f_new = np.max(corr + d_opt * D)
                gain = f_current - f_new
                
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j, best_d = i, j, d_opt
                    if not best_improvement and gain > 1e-10:
                        made_change = True
                        break
            if made_change and not best_improvement:
                break
        
        if best_i >= 0 and best_gain > 1e-14:
            h[best_i] += best_d
            h[best_j] -= best_d
            h = np.clip(h, 0.0, 1.0)
            h = fix_sum(h, target)
            corr = compute_corr(h)
        else:
            if best_improvement:
                # Try first-improvement mode next
                pass
            else:
                break
    
    return h, corr

def random_perturb(h, n, rng, strength=0.15):
    h_new = h.copy()
    n_moves = int(rng.integers(1, 5))
    for _ in range(n_moves):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = rng.uniform(-strength, strength)
        if 0 <= h_new[i] + delta <= 1 and 0 <= h_new[j] - delta <= 1:
            h_new[i] += delta
            h_new[j] -= delta
    return fix_sum(h_new, n / 2.0, rng)

def block_swap(h, n, rng, block_size=4):
    h_new = h.copy()
    for _ in range(int(rng.integers(1, 4))):
        b1 = int(rng.integers(0, n - block_size + 1))
        b2 = int(rng.integers(0, n - block_size + 1))
        if b1 == b2:
            continue
        seg1 = h_new[b1:b1 + block_size].copy()
        seg2 = h_new[b2:b2 + block_size].copy()
        h_new[b1:b1 + block_size] = np.clip(seg2, 0, 1)
        h_new[b2:b2 + block_size] = np.clip(seg1, 0, 1)
    return fix_sum(h_new, n / 2.0, rng)

def make_sparse_init(n, rng, density=0.5):
    h = np.zeros(n)
    k = int(n * density)
    idx = rng.choice(n, size=k, replace=False)
    h[idx] = 1.0
    return fix_sum(h, n / 2.0, rng)

def make_periodic_init(n, rng):
    h = np.zeros(n)
    period = int(n // 3)
    for i in range(n):
        if (i % period) < period // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return fix_sum(h, n / 2.0, rng)
