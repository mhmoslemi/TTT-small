import numpy as np
import time

def compute_score(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h, target_sum):
    n = len(h)
    # Binary search for water level
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix floating point
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.sum(free) > 0:
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_grad_logsumexp(h, corr, t):
    n = len(h)
    max_c = np.max(corr)
    exp_vals = np.exp(t * (corr - max_c))
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = p[m]
        # ∂c_k/∂h[i] = (1-h[i-k])*I(0≤i-k<n) - h[i+k]*I(0≤i+k<n)
        # Vectorized:
        i_start1 = max(0, k)
        i_end1 = min(n, n + k)
        if i_end1 > i_start1:
            idx = np.arange(i_start1, i_end1)
            grad[idx] += wm * (1.0 - h[idx - k])
        
        i_start2 = max(0, -k)
        i_end2 = min(n, n - k)
        if i_end2 > i_start2:
            idx2 = np.arange(i_start2, i_end2)
            grad[idx2] -= wm * h[idx2 + k]
    
    return grad / t

    best_h_local = h.copy()
    lr = lr_init
    t = t0
    no_improve = 0
    
    for it in range(max_iters):

        corr = np.correlate(h, 1.0 - h, mode='full')
        grad = compute_grad_logsumexp(h, corr, t)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_constraint(h_new, target_sum)

            best_h_local = h_new.copy()
            no_improve = 0

            if score < best_local:
                best_local = score
                best_h_local = h_new.copy()
                no_improve = 0
        
        h = h_new if score < compute_score(h) else h

        h = h_new
        
        lr *= 0.998
        if it % 300 == 299:

            no_improve = 0
    
    return best_h_local, best_local

        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        frac = it / max(1, max_iters)
        t = np.exp(np.log(t_start) + frac * (np.log(t_end) - np.log(t_start)))
        grad = compute_grad(h, corr, t)
        h_new = project_constraint(h - lr * grad, target_sum)
        score = compute_score(h_new)
        if score < best_local:
            best_local = score
            best_h_local = h_new.copy()
        h = h_new
        if score < best_local + 0.01:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                lr *= 0.8
            if stall > 400:
                h = project_constraint(h + 0.03 * rng.randn(n), target_sum)
                stall = 0
                lr = lr_start
        lr = max(0.003, lr_start * (1.0 / (1.0 + 0.005 * it)))
    return best_h_local, best_local

def optimize_binary(h_binary, max_iters, deadline, rng):
    n = len(h_binary)
    h = h_binary.copy()
    score = compute_score(h)
    best = h.copy()
    best_score = score
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    temp = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        if i1 == i0:
            continue
        h[i1] = 0.0
        h[i0] = 1.0
        new_score = compute_score(h)
        delta = new_score - score
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-8)):
            score = new_score
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_score < best_score:
                best_score = new_score
                best = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
        temp *= 0.999
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 15

    try:
        h_start = np.array(initial_h_values, dtype=float).copy()
    except (NameError, AttributeError, TypeError):
        h_start = np.full(47, 0.5)

    n0 = len(h_start)
    best_h = project_constraint(h_start, n0 / 2.0)
    best_c5 = compute_score(best_h)

    # Phase 1: Gradient optimization at original resolution
    h_opt, c5_opt = optimize_gradient(
        h_start, n0 / 2.0, 10000, 2.0, 80.0, 0.04, deadline, rng)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 2: Binary local search
    h_bin = (best_h > 0.45).astype(float)
    # Ensure exactly n0//2 ones
    n_ones = int(np.sum(h_bin))
    target_ones = n0 // 2
    if n_ones > target_ones:
        idx = np.where(h_bin == 1.0)[0]
        rng.shuffle(idx)
        for i in range(n_ones - target_ones):
            h_bin[idx[i]] = 0.0
    elif n_ones < target_ones:
        idx = np.where(h_bin == 0.0)[0]
        rng.shuffle(idx)
        for i in range(target_ones - n_ones):
            h_bin[idx[i]] = 1.0
    h_b, s_b = optimize_binary(h_bin, 30000, deadline, rng)
    if s_b < best_c5:
        best_c5 = s_b
        best_h = h_b

    # Phase 3: Upscale to 2x and optimize
    if time.time() < deadline:
        n2 = n0 * 2
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h_up = np.interp(x_new, x_old, best_h)
        h_opt, c5_opt = optimize_gradient(
            h_up, n2 / 2.0, 6000, 3.0, 100.0, 0.02, deadline, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: More random restarts at n0
    while time.time() < deadline - 10:
        h_rand = rng.rand(n0)
        h_rand = project_constraint(h_rand, n0 / 2.0)
        h_opt, c5_opt = optimize_gradient(
            h_rand, n0 / 2.0, 3000, 3.0, 50.0, 0.05, deadline, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        # Binary round of the random
        h_br = (h_rand > 0.5).astype(float)
        n_ones_r = int(np.sum(h_br))
        if n_ones_r != n0 // 2:
            if n_ones_r > n0 // 2:
                idx = np.where(h_br == 1)[0]
                rng.shuffle(idx)
                for i in range(n_ones_r - n0 // 2):
                    h_br[idx[i]] = 0.0
            else:
                idx = np.where(h_br == 0)[0]
                rng.shuffle(idx)
                for i in range(n0 // 2 - n_ones_r):
                    h_br[idx[i]] = 1.0
        h_b2, s_b2 = optimize_binary(h_br, 15000, deadline, rng)
        if s_b2 < best_c5:
            best_c5 = s_b2
            best_h = h_b2

    # Final: also try gradient refinement from best binary
    if time.time() < deadline - 5:
        h_ref = best_h.astype(float).copy()
        h_ref = project_constraint(h_ref, len(h_ref) / 2.0)
        h_opt, c5_opt = optimize_gradient(
            h_ref, len(h_ref) / 2.0, 3000, 5.0, 60.0, 0.02, deadline, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    best_h = project_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_score(best_h)
    return best_h, final_c5, len(best_h)
