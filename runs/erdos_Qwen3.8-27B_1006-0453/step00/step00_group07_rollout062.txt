        # Search from perturbed state
        h_try, c5_try, _ = local_search(h_try, rng2, c5_try, n, t_phase2, 800000, T_init=0.003)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try.copy()
    
    # Phase 3: Final refinement (greedy, no SA)
    t_end = t0 + budget_s - 3
    rng3 = np.random.default_rng(seed + 999)
    h_best, c5_best, _ = local_search(h_best, rng3, c5_best, n, t_end, 3000000, T_init=1e-8)
    
    # Verify constraints
    h_best = np.clip(h_best, 0, 1)
    assert abs(h_best.sum() - target) < 1e-10, f"Sum constraint violated: {h_best.sum()} vs {target}"
    c5_final = compute_c5(h_best)
    
    return h_best, c5_final, n
