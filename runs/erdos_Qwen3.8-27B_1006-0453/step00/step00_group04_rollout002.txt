import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def enforce_sum(h, n, rng):
    target = n // 2
    current = int(np.sum(h >= 0.5))
    if current > target:
        ones = np.where(h >= 0.5)[0]
        for idx in rng.choice(ones, current - target, replace=False):
            h[idx] = 0.0
    elif current < target:
        zeros = np.where(h < 0.5)[0]
        for idx in rng.choice(zeros, target - current, replace=False):
            h[idx] = 1.0
    return h

def upsample(h_init, n):
    n_init = len(h_init)
    if n == n_init:
        return h_init.copy()
    x_old = np.arange(n_init, dtype=np.float64)
    x_new = np.arange(n, dtype=np.float64) * n_init / n
    return np.interp(x_new, x_old, h_init)

def optimize_sa(h_in, time_limit, seed, temp0=0.04, cool=0.9997):
    n = len(h_in)
    rng = np.random.RandomState(seed)
    h = (h_in >= 0.5).astype(np.float64)
    enforce_sum(h, n, rng)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    temp = temp0
    t_start = time.time()

    while time.time() - t_start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h)
        if new_c5 <= cur_c5 or (temp > 1e-12 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        temp *= cool
    return best_h, best_c5

def optimize_pair_swap(h_in, time_limit, seed):
    """Try swapping pairs for more global moves."""
    n = len(h_in)
    rng = np.random.RandomState(seed)
    h = (h_in >= 0.5).astype(np.float64)
    enforce_sum(h, n, rng)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    t_start = time.time()
    temp = 0.03

    while time.time() - t_start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # random pair swap
        i1, i2 = rng.choice(ones, 2, replace=False)
        j1, j2 = rng.choice(zeros, 2, replace=False)
        h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
        new_c5 = evaluate_c5(h)
        if new_c5 <= cur_c5 or (temp > 1e-12 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp)):
            cur_c5 = new_c5

            h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
