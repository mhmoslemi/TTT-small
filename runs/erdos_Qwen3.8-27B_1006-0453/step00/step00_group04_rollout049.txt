import numpy as np
from scipy.optimize import minimize
import time

def project_h(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - n / 2.0
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] -= diff / np.sum(mask)
        else:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += (-diff) / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def soft_max_obj_grad(h, n, alpha):
    """Compute soft-max objective value and gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    f = c * dx
    m = np.max(f)
    exp_f = np.exp(alpha * (f - m))
    weights = exp_f / np.sum(exp_f)
    obj = m + np.log(np.sum(exp_f)) / alpha
    # Gradient: dF/dh[j] = dx * sum_m weights[m] * ((1-h[j+k]) - h[j-k])
    # = dx * (1 - A[j] - B[j])
    h_ext = np.zeros(3 * n - 3)
    h_ext[n-1:2*n-2] = h
    A = np.correlate(h_ext, weights, 'valid')
    w_rev = weights[::-1]
    B = np.correlate(h_ext, w_rev, 'valid')
    grad = dx * (1.0 - A - B)
    return obj, grad

def projected_gradient_step(h, n, lr, alpha):
    obj, grad = soft_max_obj_grad(h, n, alpha)
    h_new = h - lr * grad
    h_new = project_h(h_new, n)
    return h_new, obj

def local_search_binary(h, n, max_iters=2000, rng=None):
    """Local search: flip individual bits to improve C5."""
    if rng is None:
        rng = np.random.default_rng()

    h_bin = project_h(h_bin, n)
    cur_c5 = compute_c5(h_bin, n)
    for it in range(max_iters):
        improved = False
        for j in range(n):
            h_trial = h_bin.copy()
            # Flip: if 1->0, we remove from sum; need to add elsewhere
            # Instead: try swapping h[j] with h[k]
            if h_bin[j] > 0.5:
                candidates = np.where(h_bin < 0.5)[0]
            else:
                candidates = np.where(h_bin > 0.5)[0]
            if len(candidates) == 0:
                continue
            # Try a random candidate for speed
            k = candidates[rng.integers(len(candidates))]
            h_trial[j], h_trial[k] = h_trial[k], h_trial[j]
            c5_trial = compute_c5(h_trial, n)
            if c5_trial < cur_c5 - 1e-12:
                h_bin = h_trial
                cur_c5 = c5_trial
                improved = True
                break
        if not improved:
            break
    return h_bin, cur_c5
