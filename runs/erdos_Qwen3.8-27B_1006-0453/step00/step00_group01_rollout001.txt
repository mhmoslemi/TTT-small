def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return np.max(compute_corr(h)) * dx

def compute_gradient_c5(h, dx, beta=100.0):
    n = len(h)
    corr = compute_corr(h)
    # Soft max gradient
    max_corr = np.max(corr)
    shifted = corr - max_corr
    weights = np.exp(beta * shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - (n - 1)
        
        # First term: (1-h[j-k]) for valid j
        j_lo = max(k, 0)
        j_hi = min(n + k, n)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo - k:j_hi - k])
        
        # Second term: -h[j+k] for valid j
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def compute_gradient_soft(h, dx, beta):
    n = len(h)
    corr = compute_corr(h)
    max_corr = np.max(corr)
    shifted = corr - max_corr
    weights = np.exp(beta * shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - (n - 1)
        j_lo = max(k, 0)
        j_hi = min(n + k, n)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo - k:j_hi - k])
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] += w * (-h[j_lo2 + k:j_hi2 + k])
    return grad * dx

def objective_soft(h, dx, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    return float(dx * (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta))

def objective_and_jac(h, dx, beta):
    h = np.asarray(h, dtype=float).reshape(-1)
    val = objective_soft(h, dx, beta)
    grad = compute_gradient_soft(h, dx, beta)
    return val, grad

def random_feasible(n, rng):
    h = rng.random(n)
    return project_feasible(h, n)

def structured_feasible(n, rng):
    # Create a block-like structure
    nblocks = rng.integers(3, 9)
    block_len = n // nblocks
    h = np.zeros(n)
    for b in range(nblocks):
        start = b * block_len
        end = min(start + block_len, n)
        if rng.random() > 0.3:
            h[start:end] = 1.0 + rng.normal(0, 0.1, end - start)
    return project_feasible(h, n)

def projected_gradient(h, dx, n, budget_end, lr0=0.05, max_iter=5000):
    best_local = compute_c5(h, dx)
    lr = lr0
    patience = 0
    for it in range(max_iter):
        if time.time() > budget_end:
            break
        grad = compute_gradient_soft(h, dx, 30.0)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_local:
            best_local = c5_new
            h = h_new
            patience = 0
            lr = min(lr * 1.03, 0.1)
        else:
            patience += 1
            lr *= 0.97
            if patience > 800:
                break
        if lr < 1e-7:
            break
    return h, best_local

def slsqp_optimize(h0, dx, n, beta, maxiter=500):
    constraints = [{'type': 'eq',
                    'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: objective_soft(x, dx, beta),
                   h0.reshape(-1),
                   jac=lambda x: compute_gradient_soft(x, dx, beta),
                   method='SLSQP',
                   constraints=constraints,
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'eps': 1e-8})
    return res.x, res

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 20

    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.ones(47) * 0.5
    n = len(best_h)
    dx = 2.0 / n
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: SLSQP with soft-max on initial
    try:
        x, _ = slsqp_optimize(best_h, dx, n, beta=40, maxiter=300)
        x = project_feasible(np.clip(x, 0, 1), n)
        c5 = compute_c5(x, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, x
    except Exception:
        pass

    # Phase 2: Projected gradient from best
    if time.time() < deadline:
        h_pg, c5_pg = projected_gradient(best_h.copy(), dx, n, deadline, lr0=0.04, max_iter=8000)
        if c5_pg < best_c5:
            best_c5, best_h = c5_pg, h_pg

    # Phase 3: SLSQP polish
    if time.time() < deadline:
        try:
            x, _ = slsqp_optimize(best_h, dx, n, beta=60, maxiter=400)
            x = project_feasible(np.clip(x, 0, 1), n)
            c5 = compute_c5(x, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, x
        except Exception:
            pass

    # Phase 4: Random restarts
    n_restarts = 6
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if r == 0:
            h_try = random_feasible(n, rng)
        elif r == 1:
            h_try = structured_feasible(n, rng)
        elif r == 2:
            # Perturbed best
            h_try = best_h + rng.normal(0, 0.05, n)
            h_try = project_feasible(h_try, n)
        else:
            h_try = random_feasible(n, rng)

        # Quick SLSQP
        try:
            x, _ = slsqp_optimize(h_try, dx, n, beta=40, maxiter=200)
            x = project_feasible(np.clip(x, 0, 1), n)
            c5 = compute_c5(x, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, x
        except Exception:
            pass

        # Projected gradient polish
        if time.time() < deadline and compute_c5(h_try, dx) < best_c5 + 0.05:
            h_pg, c5_pg = projected_gradient(h_try, dx, n, deadline, lr0=0.03, max_iter=3000)
            if c5_pg < best_c5:
                best_c5, best_h = c5_pg, h_pg

    # Phase 5: Final fine SLSQP
    if time.time() < deadline - 5:
        try:
            x, _ = slsqp_optimize(best_h, dx, n, beta=80, maxiter=500)
            x = project_feasible(np.clip(x, 0, 1), n)
            c5 = compute_c5(x, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, x
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n)
