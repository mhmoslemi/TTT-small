import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            space = 1.0 - h
            total = space.sum()
            if total > 0:
                h = np.minimum(h + space * (diff/total), 1.0)
        else:
            h = np.maximum(h + diff / n, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Ensure exact sum
    remaining = target - h.sum()
    if abs(remaining) > 1e-14:
        h = h.astype(float)
        h[np.argmax(h if remaining < 0 else 1-h)] += remaining
        h = np.clip(h, 0, 1)
    return h

def swap_local_search(h, n, max_iters, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    accept_count = 0
    for it in range(max_iters):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        best_h[i], best_h[j] = best_h[j], best_h[i]
        c5 = compute_c5(best_h)
        if c5 > best_c5 + 1e-14:
            best_h[i], best_h[j] = best_h[j], best_h[i]
        else:
            best_c5 = c5
            accept_count += 1
    return best_h, best_c5

def two_swap_local_search(h, n, max_iters, rng):
    """Try swapping pairs of pairs (4 elements) for more flexibility"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iters):
        idx = rng.choice(n, size=4, replace=False)
        i, j, k, l = idx
        best_h[i], best_h[j] = best_h[j], best_h[i]
        best_h[k], best_h[l] = best_h[l], best_h[k]
        c5 = compute_c5(best_h)
        if c5 > best_c5 + 1e-14:
            best_h[i], best_h[j] = best_h[j], best_h[i]
            best_h[k], best_h[l] = best_h[l], best_h[k]
        else:
            best_c5 = c5
    return best_h, best_c5

def continuous_refine(h, n, max_iters, rng, step=0.05):
    """Small continuous perturbations"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    for it in range(max_iters):
        i = rng.randint(n)
        delta = rng.uniform(-step, step)
        orig = best_h[i]
        best_h[i] = np.clip(orig + delta, 0, 1)
        # Fix sum
        diff = target - best_h.sum()
        if abs(diff) > best_h[i] - orig + 1e-12:
            # Need to adjust elsewhere
            best_h = make_feasible(best_h, n)
        c5 = compute_c5(best_h)
        if c5 > best_c5 + 1e-14:
            best_h[i] = orig
            best_h = make_feasible(best_h, n)
        else:
            best_c5 = c5
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            space = 1.0 - h
            total = space.sum()
            if total > 0:
                h += space * (diff / total)
            else:
                h = np.minimum(h + diff / n, 1.0)
        else:
            h = np.maximum(h + diff / n, 0.0)
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def hill_climb(h, n, max_iters, rng, use_double_swap=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_iters):
        if use_double_swap:
            idx = rng.choice(n, 4, replace=False)
            i, j, k, l = idx
            best_h[i], best_h[j] = best_h[j], best_h[i]
            best_h[k], best_h[l] = best_h[l], best_h[k]
            c5 = compute_c5(best_h)
            if c5 > best_c5 + 1e-14:
                best_h[i], best_h[j] = best_h[j], best_h[i]
                best_h[k], best_h[l] = best_h[l], best_h[k]
            else:
                best_c5 = c5
        else:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            best_h[i], best_h[j] = best_h[j], best_h[i]
            c5 = compute_c5(best_h)
            if c5 > best_c5 + 1e-14:
                best_h[i], best_h[j] = best_h[j], best_h[i]
            else:
                best_c5 = c5
    return best_h, best_c5

def continuous_refine(h, n, max_iters, rng, step=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0

            # Use a pair adjustment

Actually, the continuous refine is tricky because of the sum constraint. Let me think of a better way.

Instead of changing one element, I can change two elements (one up, one down) to maintain the sum exactly:

```python
def continuous_refine(h, n, max_iters, rng, step=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_iters):
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.uniform(-step, step)
        orig_i, orig_j = best_h[i], best_h[j]
        best_h[i] = np.clip(orig_i + delta, 0, 1)
        best_h[j] = np.clip(orig_j - (best_h[i] - orig_i), 0, 1)
        c5 = compute_c5(best_h)
        if c5 > best_c5 + 1e-14:
            best_h[i], best_h[j] = orig_i, orig_j
        else:
            best_c5 = c5
    return best_h, best_c5
```
