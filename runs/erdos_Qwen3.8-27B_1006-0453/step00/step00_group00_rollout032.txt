```python
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= h[idx2 + lag]
    grad /= len(act)
    return grad
```

```python
import numpy as np
import time

def compute_c5_g(h):
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    return g

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_vec(h, g, n):
    mx = g.max()
    tol = 1e-9
    act = np.where(g >= mx - tol)[0]
    grad = np.zeros(n)
    for ki in act:
        lag = ki - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += (1.0 - h[idx - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= h[idx + lag]
    grad /= max(len(act), 1)
    return grad

def local_optimize(h_init, n, rng, end_time):
    target = n / 2.0
    h = project_feasible(h_init.astype(float).copy(), target)
    g = compute_c5_g(h)
    c5 = g.max()
    best_h = h.copy()
    best_c5 = c5

    for epoch in range(2000):
        if time.time() > end_time - 2:
            break
        lr = 2.0 / (1.0 + 0.1 * epoch)
        for _ in range(15):
            if time.time() > end_time - 2:
                break
            grad = subgradient_vec(h, g, n)
            gn = np.linalg.norm(grad)
            if gn < 1e-10:
                break
            h_new = project_feasible(h - lr * grad / gn, target)
            g_new = compute_c5_g(h_new)
            c5_new = g_new.max()
            if c5_new < c5 - 1e-11:
                h, g, c5 = h_new, g_new, c5_new
                if c5 < best_c5:
                    best_h, best_c5 = h.copy(), c5
            else:
                lr *= 0.6
            if lr < 1e-8:
                break
        # SA jump
        scale = max(0.01, 0.4 * np.exp(-epoch / 80.0))
        k = max(2, n // 4)
        idx = rng.integers(0, n, k)
        h[idx] += rng.normal(0, scale, k)
        if rng.random() < 0.3:
            idx2 = rng.integers(0, n, min(3, n))
            h[idx2] = np.clip(h[idx2] + rng.uniform(-0.5, 0.5, len(idx2)), 0, 1)
        h = project_feasible(h, target)
        g = compute_c5_g(h)
        c5 = g.max()
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        elif c5 - best_c5 < 0.05 * scale:
            pass
        else:
            # Revert if too bad
            if c5 > best_c5 + 0.1:
                h = best_h.copy()
                g = compute_c5_g(h)
                c5 = best_c5
    return best_h, best_c5
```
