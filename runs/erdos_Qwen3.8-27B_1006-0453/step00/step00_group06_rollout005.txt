def repair(h, target_sum):

    n = len(h)
    diff = target_sum - h.sum()
    # Distribute the difference evenly, then clip and re-adjust
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # If still off, adjust individual elements
    diff = target_sum - h.sum()
