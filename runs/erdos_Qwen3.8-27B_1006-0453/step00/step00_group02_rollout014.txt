import numpy as np
import scipy.optimize as optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return dx * mc + (dx / T) * np.log(np.sum(np.exp(shifted)))

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    ev = np.exp(shifted)
    w_sum = np.sum(ev)
    weights = ev / w_sum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                idx1 = np.arange(0, n - lag)
                grad[idx1] += w * (1.0 - h[idx1 + lag])
                idx2 = np.arange(lag, n)
                grad[idx2] -= w * h[idx2 - lag]
        else:
            lp = -lag
            idx1 = np.arange(lp, n)
            grad[idx1] += w * (1.0 - h[idx1 - lp])
            idx2 = np.arange(0, n - lp)
            grad[idx2] -= w * h[idx2 + lp]
    return grad * dx

def penalized_obj(h, T, n, lam=1e5):
    d = np.sum(h) - n / 2.0
    return smooth_obj(h, T) + lam * d * d

def penalized_grad(h, T, n, lam=1e5):
    d = np.sum(h) - n / 2.0
    return smooth_grad(h, T) + 2.0 * lam * d

def constrain_sum(h, target):
    h = h.copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            h += diff / len(h)
            h = np.clip(h, 0, 1)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def upscale(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    if n_new > n:
        x_old = np.arange(n, dtype=float)
        x_new = np.linspace(0, n - 1, n_new)
        return np.interp(x_new, x_old, h)
    else:
        x_old = np.arange(n, dtype=float)
        x_new = np.linspace(0, n - 1, n_new)
        return np.interp(x_new, x_old, h)

    bounds = [(0.0, 1.0)] * n_res

    best_c5 = compute_c5(h)
    start = time.time()

    for T in T_schedule:
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 2:
            break

            return penalized_obj(x, T, n_res, lam)

            return penalized_grad(x, T, n_res, lam)

                fun, h, method='L-BFGS-B', jac=gfun,

            )
            h_candidate = np.clip(res.x, 0.0, 1.0)

            h = h_candidate
