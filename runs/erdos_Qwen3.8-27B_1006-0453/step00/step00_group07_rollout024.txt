import numpy as np
from scipy import optimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def make_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    # Sort by value, threshold
    thresh = np.sort(h)[n // 2]
    binary = (h >= thresh).astype(float)
    # Adjust to get exactly n//2 ones
    ones = np.sum(binary)
    if ones > n // 2:
        # Flip some 1s to 0s (the smallest ones)
        idx = np.where(binary == 1)[0]
        binary[idx[:int(ones - n // 2)]] = 0
    elif ones < n // 2:
        idx = np.where(binary == 0)[0]
        binary[idx[:int(n // 2 - ones)]] = 1
    return binary

def swap_neighbors(indices, n, rng):
    """Generate a swap move."""
    i = rng.integers(n)
    j = rng.integers(n)
    return i, j

def sa_optimize_binary(h, n, n_iters, rng):
    """Simulated annealing for binary sequence."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    curr_h = h.copy()
    curr_c5 = best_c5
    T = 0.1
    T_min = 0.001
    
    for it in range(n_iters):
        # Pick a swap
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or curr_h[i] == curr_h[j]:
            continue
        
        # Apply swap
        curr_h[i], curr_h[j] = curr_h[j], curr_h[i]
        new_c5 = evaluate_c5(curr_h, dx)
        
        # Accept?
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        else:
            # Reject
            curr_h[i], curr_h[j] = curr_h[j], curr_h[i]
        
        T = max(T_min, T * 0.9999)
    
    return best_h, best_c5

import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_bin_flex(vals, n):
    """Create binary with exactly n//2 ones from continuous vals."""
    t = n // 2
    idx = np.argsort(vals)[::-1][:t]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def greedy_improve(h, n, max_rounds=100):
    """Greedy: repeatedly find the best swap."""
    dx = 2.0 / n
    h = h.copy()
    curr = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
    for _ in range(max_rounds):
        best_delta = -1e-9
        best_a, best_b = -1, -1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for a in ones:
            for b in zeros:
                h[a], h[b] = 0, 1
                c = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
                if c - curr < best_delta:
                    best_delta = c - curr
                    best_a, best_b = b, a  # swapped back
                h[a], h[b] = 1, 0  # undo
        if best_a >= 0:
            h[best_a], h[best_b] = 1 - h[best_a], 1 - h[best_b]
            curr += best_delta
        else:
            break
    return h, curr
