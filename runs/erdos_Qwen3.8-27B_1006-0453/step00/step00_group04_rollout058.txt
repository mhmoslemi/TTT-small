    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    sh = (corr - mx) * temp
    es = np.exp(sh)
    se = es.sum()

    obj = mx + np.log(se) / temp

        else:
            p = -mk
            g[p:] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:]
    return obj, g

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def smax_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    sh = (corr - mx) * temp
    es = np.exp(sh)
    se = es.sum()
    sw = es / se
    obj = mx + np.log(se) / temp
    g = np.zeros(n)
    for k in range(2 * n - 1):
        mk = k - (n - 1)
        wk = sw[k] * dx
        if mk >= 0:
            g[:n - mk] += wk * (1.0 - h[mk:])
            g[mk:] -= wk * h[:n - mk]
        else:
            p = -mk
            g[p:] += wk * (1.0 - h[:n - p])
            g[:n - p] -= wk * h[p:]
    return obj, g

def sobj_fn(x, dx, temp):
    return smax_grad(x, dx, temp)[0]

def sjac_fn(x, dx, temp):
    return smax_grad(x, dx, temp)[1]

def cfun_fn(x, n):
    return np.sum(x) - n / 2.0

def cjac_fn(x, n):
    return np.ones(n)

def fix(h, n):

    h += (n / 2.0 - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = n / 2.0 - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            for i in np.argsort(-h):
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
                if abs(d) < 1e-12:
                    break
        else:
            for i in np.argsort(h):
                a = min(-d, h[i])
                h[i] -= a
                d += a
                if abs(d) < 1e-12:
                    break
    return h

def interp_n(h, nn):
    no = len(h)
    xo = (np.arange(no) + 0.5) * 2.0 / no

                 options={'maxiter': iters, 'ftol': 1e-12})
    return fix(r.x, n)

def coord_search(h, n, steps=1000, step=0.05):
    dx = 2.0 / n
    best = c5v(h)
    h_cur = h.copy()
    np.random.seed(12345)
    for _ in range(steps):
        j = np.random.randint(n)
        k = np.random.randint(n)
        if j == k:
            k = (j + 1) % n
        delta = np.random.choice([-step, step])
        if h_cur[j] + delta < 0 or h_cur[j] + delta > 1:
            delta = -delta
            if h_cur[j] + delta < 0 or h_cur[j] + delta > 1:
                continue
        h_new = h_cur.copy()

        h_new[k] -= delta
        h_new = np.clip(h_new, 0, 1)

            continue
        c = c5v(h_new)
        if c < best:
            best = c
            h_cur = h_new
    return h_cur, best
