    T = 0.004
    T_floor = 1e-8
    dvals = np.array([0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5])

    iters = 0
    stall_count = 0
    last_improve_iter = 0

    while time.time() - t0 < budget_s - 3:
        i = int(rng.randint(0, n_work))
        j = int(rng.randint(0, n_work))
        if i == j:
            continue
        d = float(dvals[rng.randint(0, len(dvals))])
        if rng.random() < 0.5:
            d = -d

        hi = h_work[i] + d
        hj = h_work[j] - d
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            hi = h_work[i] - d
            hj = h_work[j] + d
            if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                continue

        h_work[i] = hi
        h_work[j] = hj
        new_c5 = compute_c5(h_work)
        dc = new_c5 - cur_c5

        if dc < 0 or rng.random() < np.exp(-dc / T):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h_work.copy()
                last_improve_iter = iters
            elif dc >= 0:
                stall_count += 1
            else:
                stall_count = 0
        else:
            h_work[i] -= d
            h_work[j] += d

        iters += 1
        if iters % 500 == 0:
            T = max(T * 0.93, T_floor)
            if T < T_floor * 10:
                T = 0.004
                stall_count = 0
        # Periodic sum re-fix to combat floating point drift
        if iters % 5000 == 0:
            h_work = fix_sum(h_work, target_work)
            cur_c5 = compute_c5(h_work)
