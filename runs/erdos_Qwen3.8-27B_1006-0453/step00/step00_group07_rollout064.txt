```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    c5 = corr[idx] * dx
    k = idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + k < n:
            g += (1.0 - h[j + k])
        if 0 <= j - k < n:
            g -= h[j - k]
        grad[j] = g * dx
    return c5, grad

def project(h, target_sum):
    # Project onto {0 <= h <= 1, sum(h) = target_sum}
    # Simple: clip then adjust
    n = len(h)
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Redistribute
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if mask.sum() == 0:
        return h
    adj = diff / mask.sum()
    h[mask] = np.clip(h[mask] + adj, 0, 1)
    return h

def subgradient_descent(h, n_iters=5000, lr0=0.01, lr_decay=0.9999):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iters):
        c5, grad = c5_and_grad(h)
        h_new = h - lr * grad
        h_new = project(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= lr_decay
        if lr < 1e-8:
            break
    return best_h, best_c5

def sa_optimize(h, rng, n_iters=50000, T0=0.005, T_min=1e-6, cool=0.9999):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = T0
    for it in range(n_iters):
        # Perturb: pick pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        delta = rng.uniform(-0.1, 0.1)
        new_h = h.copy()
        new_h[i] = np.clip(h[i] + delta, 0, 1)
        actual = new_h[i] - h[i]
        new_h[j] = np.clip(h[j] - actual, 0, 1)
        # Fix sum
        s = new_h.sum() - target_sum
        if abs(s) > 1e-10:
            # Redistribute to all
            new_h -= s / n
            new_h = np.clip(new_h, 0, 1)
            # Still might not be exact, but close enough
        
        new_c5 = compute_c5(new_h)
        if new_c5 <= c5 or rng.random() < np.exp(-(new_c5 - c5) / max(T, 1e-10)):
            h = new_h
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        T *= cool
        T = max(T, T_min)
    return best_h, best_c5

def block_perturb(h, rng):
    n = len(h)
    start = rng.integers(0, n)
    length = rng.integers(2, min(8, n // 3))
    end = min(start + length, n)
    # Generate perturbation that preserves sum
    perm = rng.permutation(end - start)
    shifts = rng.normal(0, 0.05, end - start)
    shifts -= shifts.mean()
    new_h = h.copy()
    for idx in range(end - start):
        pos = start + idx
        new_h[pos] = np.clip(h[pos] + shifts[perm[idx]], 0, 1)
    # Fix sum
    s = new_h.sum() - n / 2.0
    if abs(s) > 1e-10:
        new_h -= s / n
        new_h = np.clip(new_h, 0, 1)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    
    # Initial solution

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init
    
    # Phase 1: Subgradient descent from initial
    h1, c5_1 = subgradient_descent(h_init, n_iters=3000, lr0=0.05, lr_decay=0.99995)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
        best_n = n_init
    if time.time() - t0 > budget_s - 60:
        return best_h, best_c5, best_n
    
    # Phase 2: SA on the best
    h2, c5_2 = sa_optimize(best_h, rng, n_iters=100000, T0=0.003, T_min=1e-6, cool=0.99999)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2
        best_n = n_init
    if time.time() - t0 > budget_s - 60:
        return best_h, best_c5, best_n
    
    # Phase 3: Subgradient again
    h3, c5_3 = subgradient_descent(best_h, n_iters=5000, lr0=0.03, lr_decay=0.9999)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3
        best_n = n_init
    if time.time() - t0 > budget_s - 60:
        return best_h, best_c5, best_n
    
    # Phase 4: Try different n values
    for n_try in [40, 60, 100, 120, 160, 200]:
        if time.time() - t0 > budget_s - 60:

        # Interpolate from best to new n
        x_old = np.linspace(0, 2, n_init + 1)[:-1]
        x_new = np.linspace(0, 2, n_try + 1)[:-1]
        h_interp = np.interp(x_new, x_old, best_h)
        # Adjust sum
        h_interp = np.clip(h_interp, 0, 1)
        target = n_try / 2.0
        h_interp = h_interp * (target / h_interp.sum())

        h_try, c5_try = subgradient_descent(h_interp, n_iters=2000, lr0=0.05)
```
