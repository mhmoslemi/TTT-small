import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def interpolate_h(h, new_n):
    """Interpolate h from len(h) points to new_n points on [0,2]."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def project_to_constraint(h, n):
    """Project h to satisfy sum(h) = n/2 and 0<=h<=1."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    # Distribute uniformly where possible
    if diff > 0:
        # Need to add mass
        mask = h < 1.0
        if mask.any():
            add = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + add)
    elif diff < 0:
        mask = h > 0.0
        if mask.any():
            sub = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - sub)
    return h

def greedy_local_search(h, budget=5.0):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    while time.time() - t0 < budget:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                # Try both directions
                for src, dst in [(j, i), (i, j)]:
                    available = min(h[src], 1.0 - h[dst])
                    for frac in [0.1, 0.3, 0.5, 0.7, 1.0]:
                        delta = available * frac
                        if delta < 1e-10:

                        h[dst] += delta
                        h[src] -= delta

                        if c5_new < best_c5 - 1e-12:

                            improved = True
                        else:

                            # Restore if we didn't accept
                            if c5_new >= best_c5:
                                pass  # already restored
        if not improved:
            break
    return h, best_c5

def compute_c5(h):

    """Ensure 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
