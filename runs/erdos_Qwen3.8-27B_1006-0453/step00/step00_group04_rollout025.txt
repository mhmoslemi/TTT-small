import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def fix_sum(h, target_sum):
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:

        if cnt == 0:
            break

def evaluate_obj(params, beta, lam, n):
    corr = np.correlate(params, 1.0 - params, mode="full")
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    logsum = cmax + np.log(np.sum(np.exp(shifted))) / beta
    dx = 2.0 / n
    val = logsum * dx + lam * (np.sum(params) - n / 2.0) ** 2
