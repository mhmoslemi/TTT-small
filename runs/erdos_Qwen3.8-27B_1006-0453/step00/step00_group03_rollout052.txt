    return project_feasible(h2, n2 / 2.0)

def random_feasible(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n / 2.0)

def perturb_h(h, rng, scale=0.05):
    n = len(h)
    h2 = h + rng.randn(n) * scale
    return project_feasible(h2, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')

    n = 70
    h0 = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    if h0 is None:
        h0 = np.ones(n) * 0.5
    h0 = project_feasible(h0, n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(h0)

    # Phase 1: SLSQP on initial point
    elapsed = time.time() - t0
    if budget_s - elapsed > 60:
        hx, cx = solve_slsqp(h0, maxiter=400)
        if cx < best_c:
            best_h, best_c = hx.copy(), cx

    # Phase 2: Perturbation restarts + SLSQP
    elapsed = time.time() - t0
    n_restarts = 20
    for r in range(n_restarts):
        if budget_s - (time.time() - t0) < 20:
            break
        if r < 8:
            h_start = perturb_h(best_h, rng, scale=0.02 + 0.015 * r)
        elif r < 14:
            h_start = perturb_h(best_h, rng, scale=0.08 + 0.02 * r)
        else:
            h_start = random_feasible(n, rng)
        hx, cx = solve_slsqp(h_start, maxiter=200)
        if cx < best_c:
            best_h, best_c = hx.copy(), cx

    # Phase 3: Projected gradient refinement then SLSQP
    elapsed = time.time() - t0
    if budget_s - elapsed > 80:
        hg, cg = projected_gradient(best_h, n_iter=3000, lr0=0.5)
        if cg < best_c:
            best_h, best_c = hg.copy(), cg
        hx, cx = solve_slsqp(best_h, maxiter=300)
        if cx < best_c:
            best_h, best_c = hx.copy(), cx

    # Phase 4: More diverse restarts
    elapsed = time.time() - t0
    for r in range(10):
        if budget_s - (time.time() - t0) < 20:
            break
        h_start = random_feasible(n, rng)
        hx, cx = solve_slsqp(h_start, maxiter=150)
        if cx < best_c:
            best_h, best_c = hx.copy(), cx

    # Phase 5: Upscale to 2n and optimize
    elapsed = time.time() - t0
    if budget_s - elapsed > 150:
        n2 = n * 2
        h_up = upscale_h(best_h, 2)
        hx2, cx2 = solve_slsqp(h_up, maxiter=250)
        if cx2 < best_c:
            best_h, best_c = hx2.copy(), cx2
        if budget_s - (time.time() - t0) > 80:
            hg2, cg2 = projected_gradient(h_up, n_iter=1500, lr0=0.3)
            if cg2 < best_c:
                best_h, best_c = hg2.copy(), cg2
            hx2, cx2 = solve_slsqp(best_h, maxiter=250)
            if cx2 < best_c:
                best_h, best_c = hx2.copy(), cx2
        # A couple more restarts at higher res
        for r in range(3):
            if budget_s - (time.time() - t0) < 20:
                break
            h_start = perturb_h(best_h, rng, scale=0.02)
            hx2, cx2 = solve_slsqp(h_start, maxiter=150)
            if cx2 < best_c:
                best_h, best_c = hx2.copy(), cx2

    # Phase 6: Final SLSQP polish
    if budget_s - (time.time() - t0) > 10:
        hx, cx = solve_slsqp(best_h, maxiter=200)
        if cx < best_c:
            best_h, best_c = hx.copy(), cx

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c = compute_c5(best_h)

    return best_h, best_c, n_final
