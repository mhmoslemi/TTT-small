import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    return c5, corr

def gradient_corr(h, k_star, n):
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        for j in range(n - lag):
            grad[j] += (1.0 - h[j + lag])
        for j in range(lag, n):
            grad[j] -= h[j - lag]
    else:
        nl = -lag
        for j in range(nl, n):
            grad[j] += (1.0 - h[j - nl])
        for j in range(n - nl):
            grad[j] -= h[j + nl]
    return grad

def project_constrained(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    h -= (np.sum(h) - target) / n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if len(free) == 0:
            break
        h[free] += diff / len(free)
        h = np.clip(h, 0.0, 1.0)
    return h
