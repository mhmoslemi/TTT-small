    target = n / 2.0
    for _ in range(100):

        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            proj = np.clip(h - mid, 0, 1)
            s = np.sum(proj)
            if s > target:
                lo = mid
            else:
                hi = mid

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h = np.clip(h - tau, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
