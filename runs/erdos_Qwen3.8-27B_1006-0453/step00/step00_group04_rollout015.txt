        if time.time() > deadline - 15:
            break

        h0 = make_initial(parent, n_pts)
        h_res, c5_res = projected_gradient(h0, n_pts, 1200, 50.0, 0.02, deadline - 5)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

        # Random restart
        if time.time() < deadline - 35:
            h_rand = project_feasible(rng.rand(n_pts), n_pts)
            h_r2, c5_r2 = projected_gradient(h_rand, n_pts, 600, 50.0, 0.02, deadline - 5)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()

        # Coarse-to-fine: take best at this n, interpolate up
        if time.time() < deadline - 40 and n_pts < 500:
            h_next, c5_next = refine_coarse_to_fine(best_h, min(n_pts * 2, 800), 600, deadline - 5)
            if c5_next < best_c5:
                best_c5 = c5_next
                best_h = h_next.copy()

    # Final polish: binary refinement (push toward 0/1)
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        h_bin = best_h.copy()
        for _ in range(50):
            if time.time() > deadline - 5:
                break
            frac = (h_bin > 0.05) & (h_bin < 0.95)
            if not frac.any():
                break
            h_bin[frac] = np.where(h_bin[frac] < 0.5, 0.0, 1.0)
            h_bin = project_feasible(h_bin, n_best)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            else:
                h_bin = best_h.copy()

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
