import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(x, alpha):
    m = np.max(x)
    if m < 1e-30:
        return 0.0
    scaled = np.clip(alpha * (x - m), -100, 100)
    return m + np.log(np.sum(np.exp(scaled))) / alpha


def obj_fn(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    return smooth_max(corr, alpha)


def ceq_fn(x, n):
    return n / 2.0 - np.sum(x)


def make_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        rem = np.setdiff1d(np.arange(n), idx)
        h[rem[0]] = 0.5
    return h


def binsearch(h, rng, max_iter=50000, tmax=20.0):
    n = len(h)
    h = h.copy()
    bc5 = compute_c5(h)
    t0 = time.time()

            break

        if h[i] > 0.5 > h[j]:
            h[i], h[j] = 0.0, 1.0
            c = compute_c5(h)

                bc5 = c
                stall = 0
            else:
                h[i], h[j] = 1.0, 0.0
                stall += 1
    return h, bc5
