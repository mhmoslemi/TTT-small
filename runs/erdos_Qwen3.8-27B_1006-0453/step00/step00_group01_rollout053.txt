        elif j > 0:
            grad[:n-j] += wk * (1.0 - h[j:])
            grad[j:] += -wk * h[:n-j]
        else:
            jj = -j
            grad[jj:] += wk * (1.0 - h[:n-jj])
            grad[:n-jj] += -wk * h[jj:]
    return obj, grad

def project(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def optimize(h_init, n, tau, t_dead):
    dx = 2.0 / n
    target = n / 2.0
    h = project(h_init.copy(), target)
    obj, _ = smooth_obj_grad(h, dx, tau)
    best_h, best_obj = h.copy(), obj
    for it in range(30000):
        if time.time() > t_dead:
            break
        _, grad = smooth_obj_grad(h, dx, tau)
        gmax = np.abs(grad).max()
        if gmax < 1e-13:
            break
        alpha = min(0.1, 0.01 / gmax)
        improved = False
        for _ in range(50):
            h_new = project(h - alpha * grad, target)
            obj_new, _ = smooth_obj_grad(h_new, dx, tau)
            if obj_new < obj - 1e-15:
                h, obj = h_new, obj_new
                if obj < best_obj:
                    best_obj, best_h = obj, h.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return best_h

def polish_single_shift(h, n, t_dead):
    """Reduce the worst shift via direct subgradient steps."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_s = np.max(compute_scores(h, dx))
    best_h = h.copy()
    lr = 0.01
    for it in range(50000):
        if time.time() > t_dead:
            break
        scores = compute_scores(h, dx)
        k_worst = int(np.argmax(scores))
        j = k_worst - (n - 1)
        g = np.zeros(n)
        if j == 0:
            g = (1.0 - 2.0 * h) * dx
        elif j > 0:
            g[:n-j] = (1.0 - h[j:]) * dx
            g[j:] = -h[:n-j] * dx
        else:
            jj = -j
            g[jj:] = (1.0 - h[:n-jj]) * dx
            g[:n-jj] = -h[jj:] * dx
        h_new = project(h - lr * g, target)
        s_new = np.max(compute_scores(h_new, dx))
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
        if s_new >= best_s * (1 - 1e-10):
            lr *= 0.9
        if lr < 1e-8:
            break
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    T = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    n0f = float(n0)
    best_h = project(h0, n0f)
    best_s = float(np.max(compute_scores(best_h, 2.0/n0)))
    best_n = n0

    def eval_score(h, n):
        return float(np.max(compute_scores(h, 2.0/n)))

    # Main optimization at n0 with tau annealing
    for tau in [10.0, 30.0, 100.0, 300.0, 1000.0]:
        rem = T - time.time() - 40
        if rem < 10:
            break
        h_new = optimize(best_h, best_n, tau, time.time() + min(rem, 180))
        s = eval_score(h_new, best_n)
        if s < best_s:
            best_s, best_h = s, h_new

    # Polish the worst shift at current resolution
    rem = T - time.time() - 30
    if rem > 10:
        h_pol = polish_single_shift(best_h, best_n, time.time() + min(rem, 60))
        s = eval_score(h_pol, best_n)
        if s < best_s:
            best_s, best_h = s, h_pol

    # Random restarts at n0
    for restart in range(3):
        rem = T - time.time() - 30
        if rem < 15:
            break
        noise = np.random.randn(best_n) * 0.05
        h_r = project(best_h + noise, float(best_n))
        for tau in [50.0, 200.0]:
            rem = T - time.time() - 20
            if rem < 10:
                break
            h_new = optimize(h_r, best_n, tau, time.time() + min(rem, 120))
            s = eval_score(h_new, best_n)
            if s < best_s:
                best_s, best_h = s, h_new
                h_r = h_new

    # Try higher resolution (2x)
    rem = T - time.time() - 30
    if rem > 40:
        n1 = 2 * best_n
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n1, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, float(n1))
        for tau in [30.0, 150.0]:
            rem = T - time.time() - 20
            if rem < 10:
                break
            h_new = optimize(h_interp, n1, tau, time.time() + min(rem, 200))
            s = eval_score(h_new, n1)
            if s < best_s:
                best_s, best_h, best_n = s, h_new, n1
                h_interp = h_new

    # Try 4x resolution if time permits
    rem = T - time.time() - 30
    if rem > 40 and best_n < 200:
        n2 = 2 * best_n
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, float(n2))
        for tau in [100.0]:
            rem = T - time.time() - 15
            if rem < 10:
                break
            h_new = optimize(h_interp, n2, tau, time.time() + min(rem, 150))
            s = eval_score(h_new, n2)
            if s < best_s:
                best_s, best_h, best_n = s, h_new, n2

    # Final polish
    rem = T - time.time() - 5
    if rem > 5:
        h_pol = polish_single_shift(best_h, len(best_h), time.time() + rem)
        s = eval_score(h_pol, len(best_h))
        if s < best_s:
            best_s, best_h = s, h_pol

    return best_h, best_s, len(best_h)
