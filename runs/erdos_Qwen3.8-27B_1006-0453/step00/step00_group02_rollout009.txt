import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def fix_sum_constraint(h, target_sum):
    """Adjust h so that sum(h) == target_sum, keeping values in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Adjust free variables first
    free = (h > 0.005) & (h < 0.995)
    nfree = np.sum(free)
    if nfree > 0:
        adj = diff / nfree
        if -0.004 <= adj <= 0.004:
            h[free] += adj
            h = np.clip(h, 0.0, 1.0)
            diff2 = target_sum - np.sum(h)
            if abs(diff2) < 1e-8:
                return h
    # Fallback: spread over all
    h += (target_sum - np.sum(h)) / len(h)
    return np.clip(h, 0.0, 1.0)


def smooth_obj(x, n, tau, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return tau * logsumexp(corr / tau) * dx


def smooth_grad(x, n, tau, dx):
    eps = 1e-5
    g = np.zeros(n)
    base = smooth_obj(x, n, tau, dx)
    for j in range(n):
        xp = x.copy()
        xp[j] += eps
        g[j] = (smooth_obj(xp, n, tau, dx) - base) / eps
    return g
