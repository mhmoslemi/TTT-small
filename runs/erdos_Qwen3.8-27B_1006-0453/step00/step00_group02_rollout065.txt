    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_max_corr(h, n):
    """Gradient of max_k correlate(h, 1-h, 'full')[k] w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr))
    p = np.arange(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
