import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase; add to entries not at 1
        mask = h < 1.0
        if not np.any(mask):
            return h
        per_val = diff / np.sum(mask)
        h[mask] = np.minimum(h[mask] + per_val, 1.0)
        # If still not enough, redistribute
        diff2 = target_sum - np.sum(h)
        if abs(diff2) > 1e-10:
            mask2 = h < 1.0
            if np.any(mask2):
                h[mask2] += diff2 / np.sum(mask2)
                h = np.clip(h, 0, 1)
    else:
        # Need to decrease; reduce from entries not at 0
        mask = h > 0.0
        if not np.any(mask):
            return h
        per_val = diff / np.sum(mask)
        h[mask] = np.maximum(h[mask] + per_val, 0.0)
        diff2 = target_sum - np.sum(h)
        if abs(diff2) > 1e-10:
            mask2 = h > 0.0
            if np.any(mask2):
                h[mask2] += diff2 / np.sum(mask2)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def nm_optimize(h_init, target, max_iter, fade):
    def obj(x):
        return compute_c5(project_constraint(x, target))
    try:
        res = minimize(obj, h_init, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-13, 'fatol': 1e-15,
                                'init_eps': 0.02 * fade if fade else 0.5})
        h_out = project_constraint(res.x, target)
        return h_out, compute_c5(h_out)
    except Exception:
        return h_init.copy(), compute_c5(h_init)

def coord_descent(h, target, rng, max_sweeps=200):
    n = len(h)
    dx = 2.0 / n
    h = project_constraint(h, target)
    best = compute_c5(h)
    k_mid = n - 1
    for _ in range(max_sweeps):
        improved = False
        for p in rng.permutation(n):
            hp = h.copy()
            hp[p] = 0.0
            hp1 = h.copy()
            hp1[p] = 1.0
            c0 = np.correlate(hp, 1.0 - hp, mode='full')
            c1 = np.correlate(hp1, 1.0 - hp1, mode='full')
            A = c1 - c0
            B = c0
            # Handle zero-shift: corr at k_mid is quadratic: x - x^2 + C
            C_zero = float(c0[k_mid])  # value at x=0
            # At x=1: corr = 1 - 1 + C_zero = C_zero... but c1[k_mid] should equal C_zero
            # Actually at x=0: h[p]=0, term contributes 0*(1-0)=0, so C_zero already excludes p's contribution
            # At x=1: h[p]=1, term contributes 1*(1-1)=0, so c1[k_mid] also excludes it
            # The actual zero-shift: x(1-x) + C_zero
            # For the max, use: max(max_{k≠mid}(A_k x + B_k), x - x^2 + C_zero)
            def g(x):
                vals = A * x + B
                vals[k_mid] = x - x * x + C_zero
                return float(np.max(vals) * dx)
            try:
                res = minimize_scalar(g, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-12})
                new_val = float(res.x)
            except Exception:
                new_val = h[p]
            if abs(new_val - h[p]) > 1e-12:
                h[p] = new_val
                h = project_constraint(h, target)
                c5_new = compute_c5(h)
                if c5_new < best - 1e-14:
                    best = c5_new
                    improved = True
        if not improved:
            break
    return h, compute_c5(h)
