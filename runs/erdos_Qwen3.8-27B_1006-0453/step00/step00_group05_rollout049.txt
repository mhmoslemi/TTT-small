import numpy as np
from scipy.optimize import minimize
import time


class P:
    n = 70
    alpha = 40.0


def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h):
    n = P.n
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(P.alpha * (c - m)))) / P.alpha


def sum_con(h):
    return h.sum() - P.n / 2.0


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def local_refine(h, n, max_iter=200):
    """Simple projection-based local search on the non-smooth objective."""
    dx = 2.0 / n
    h = fix_sum(h, n)
    best_c5 = compute_c5(h)
    step = 0.02
    for it in range(max_iter):
        improved = False
        perturbed = h.copy()
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        active = np.where(c > c.max() - 1e-4)[0]
        for lag in active[:5]:
            # Try to reduce overlap at this lag
            shift = lag - (n - 1)
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    for delta in [step, -step]:
                        h_try = h.copy()
                        h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                        h_try[j] = np.clip(h_try[j] - delta, 0, 1)
                        h_try = fix_sum(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-9:
                            h = h_try
                            best_c5 = c5_try
                            improved = True
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global P
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values
    if 'initial_h_values' in globals():
        h0 = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h0 = np.full(70, 0.5)

    n_init = len(h0)
    h0 = fix_sum(h0, n_init)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Resolutions to try
    resolutions = sorted(set([n_init, 80, 100, 140, 200, 300]))
    # Filter to reasonable sizes
    resolutions = [n for n in resolutions if n <= 500]

    for n in resolutions:
        if time.time() - start_time > budget_s - 120:
            break

        P.n = n

        # Starting points
        starts = []
        starts.append(fix_sum(resample(best_h, n), n))
        starts.append(np.full(n, 0.5))
        for _ in range(2):
            h_r = rng.uniform(0.1, 0.9, n)
            starts.append(fix_sum(h_r, n))

        for h_start in starts:
            if time.time() - start_time > budget_s - 60:
                break

            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con}]

            # Phase 1: SLSQP with smooth objective
            P.alpha = 30.0
            try:
                res = minimize(
                    smooth_obj, h_start, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
                )
            except Exception:
                res = None

            if res is not None and hasattr(res, 'x'):
                h_new = fix_sum(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_start = h_new.copy()

            # Phase 2: Tighter smooth max
            P.alpha = 100.0
            try:
                res2 = minimize(
                    smooth_obj, h_start, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': 300, 'ftol': 1e-12, 'disp': False}
                )
            except Exception:
                res2 = None

            if res2 is not None and hasattr(res2, 'x'):
                h_new = fix_sum(res2.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_start = h_new.copy()

            # Phase 3: Local refinement
            if time.time() - start_time < budget_s - 40:
                h_ref, c5_ref = local_refine(h_start, n, max_iter=100)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()

    # Final verification
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
