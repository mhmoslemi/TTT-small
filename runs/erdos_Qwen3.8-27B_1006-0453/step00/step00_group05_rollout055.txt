import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n_half):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > n_half:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, k_lag):
    n = len(h)
    grad = np.zeros(n)
    if k_lag >= 0:
        idx1 = np.arange(0, n - k_lag)
        grad[idx1] += (1.0 - h[idx1 + k_lag])
        idx2 = np.arange(k_lag, n)
        grad[idx2] -= h[idx2 - k_lag]
    else:
        kp = -k_lag
        idx1 = np.arange(kp, n)
        grad[idx1] += (1.0 - h[idx1 - kp])
        idx2 = np.arange(0, n - kp)
        grad[idx2] -= h[idx2 + kp]
    return grad

def resample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n / 2.0)

def local_search(h, max_iters=2000, time_limit=30):
    n = len(h)
    n_half = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    alpha = 0.1
    start_time = time.time()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_max = np.argmax(corr)
        k_lag = k_max - (n - 1)
        
        grad = compute_grad(h, k_lag)

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_half)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha *= 0.8
            if alpha < 0.001:

    return best_h, best_c5

                break
    return best_h, best_c5

def binary_swap_search(h, time_limit=15.0):
    n = len(h)
    n_half = n / 2.0
    h = np.round(h).clip(0, 1)
    h = project_feasible(h, n_half)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    t0 = time.time()
    while time.time() - t0 < time_limit:
        best_c5_now = evaluate_c5(h)
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        improved = False
        for i in ones:
            for j in zeros:
                if time.time() - t0 > time_limit:
                    break
                h[i], h[j] = 0.0, 1.0
                c5 = evaluate_c5(h)
                if c5 < best_c5_now - 1e-12:
                    best_c5_now = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
        best_c5 = best_c5_now
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_global = time.time()
    deadline = t_global + budget_s - 10.0
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(70)
        h0[:35] = 1.0

    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    resolutions = [n0]
    for mult in [2, 4, 8]:
        if n0 * mult <= 560:
            resolutions.append(n0 * mult)

    for res in resolutions:
        if time.time() > deadline - 60:
            break
        if res == n0:
            h_cur = h0.copy()
        else:
            h_cur = resample_h(best_h, res)

        time_left = max(10.0, deadline - time.time())
        frac = 0.7 if res <= 140 else 0.8
        t_alloc = time_left * frac

        # Continuous optimization
        h_new, c5_new = optimize_cont(h_cur, time_limit=t_alloc * 0.6)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        # Try binary round
        if time.time() < deadline - 30:
            h_bin, c5_bin = binary_swap_search(h_new, time_limit=min(15.0, (deadline - time.time()) * 0.15))
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin

        # Local re-optimization from best
        if time.time() < deadline - 20:
            h_reopt = resample_h(best_h, res)
            h_new2, c5_new2 = optimize_cont(h_reopt, time_limit=min(20.0, (deadline - time.time()) * 0.3))
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2

    # Random restarts at best resolution
    if time.time() < deadline - 30:
        n_best = len(best_h)
        n_restarts = 3
        t_per = (deadline - time.time()) / max(n_restarts, 1)
        for r in range(n_restarts):
            if time.time() > deadline - 10:
                break
            h_r = best_h + rng.randn(n_best) * 0.1
            h_r = project_feasible(h_r, n_best / 2.0)
            h_opt, c5_opt = optimize_cont(h_r, time_limit=t_per * 0.7)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, len(best_h))
