```python
import numpy as np
import time

def c5_score(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fix_sum(h, ts, rng):
    h = np.clip(h, 0.0, 1.0).copy()
    d = ts - h.sum()
    if abs(d) < 1e-12:
        return h
    n = len(h)
    r = (1.0 - h) if d > 0 else h.copy()
    tr = r.sum()
    if tr < 1e-12:
        h[:] = ts / n
        return h
    h += d * r / tr
    h = np.clip(h, 0.0, 1.0)
    d2 = ts - h.sum()
    if abs(d2) > 1e-14:
        k = int(rng.integers(n))
        h[k] = float(np.clip(h[k] + d2, 0.0, 1.0))
    return h

def gen_binary(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    else:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        z = np.where(h == 0)[0]
        h[int(rng.choice(z))] = 0.5
    return h

def gen_thue_morse(n, rng):
    h = np.zeros(n)
    counts = np.zeros(n, dtype=int)
    for i in range(1, n):
        counts[i] = counts[i >> 1] + (i & 1)
    h[counts % 2 == 0] = 1.0
    return fix_sum(h, n / 2.0, rng)

def sa_loop(h, dx, rng, t_limit, T_init=0.004, restart_iter=40000):
    n = len(h)
    cur = c5_score(h, dx)
    best = h.copy()
    bc = cur
    T = T_init
    T_min = 1e-6
    stall = 0
    while time.time() < t_limit:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d1 = min(h[i], 1.0 - h[j])
        d2 = min(h[j], 1.0 - h[i])
        r = rng.random()
        did_move = False
        if r < 0.8 and d1 > 1e-4:
            d = float(rng.uniform(0.005, d1))
            h[i] -= d
            h[j] += d
            ri, rj, rd = i, j, d
            did_move = True
        elif d2 > 1e-4:
            d = float(rng.uniform(0.005, d2))
            h[j] -= d
            h[i] += d
            ri, rj, rd = j, i, d
            did_move = True
        if not did_move:
            continue
        nc = c5_score(h, dx)
        if nc <= cur or rng.random() < np.exp(-(nc - cur) / T):
            cur = nc
            stall = 0 if nc < bc else stall + 1
            if nc < bc:
                bc = nc
                best = h.copy()
        else:
            h[ri] += rd
            h[rj] -= rd
            stall += 1
        T = max(T_min, T * 0.9999)
        if stall > restart_iter:
            h[:] = best
            cur = bc
            T = max(T_init * 0.6, 0.003)
            stall = 0
    return best, bc
```
