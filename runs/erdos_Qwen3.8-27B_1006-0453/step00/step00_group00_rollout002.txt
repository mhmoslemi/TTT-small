    return m + t * np.log(np.mean(np.exp((corr - m) / t)))


def constrain_sum(h):
    return np.sum(h) - _target_sum


def fix_sum(h, target):
    h = h.copy()
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = np.sum(room)
            if tr < 1e-15:
                break
            h += diff * room / tr
        else:
            th = np.sum(h)
            if th < 1e-15:
                break
            h += diff * h / th
        np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values, _dx_val, _temp_val, _target_sum
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    best_h = None
    best_c5 = 1.0
    best_n = 45

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float).flatten()
        init = np.clip(init, 0.0, 1.0)
    else:
        init = np.ones(45) * 0.5
    n_init = len(init)

    n_list = sorted(set([n_init, 35, 45, 55]))
    bounds_n = {}

    for n in n_list:
        if time.time() > deadline:
            break
        _dx_val = 2.0 / n
        _target_sum = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons_eq = [{'type': 'eq', 'fun': constrain_sum}]

        starts = []
        if n == n_init:
            starts.append(init.copy())
        else:
            x_new = np.linspace(0, n_init - 1, n)
            starts.append(np.clip(np.interp(x_new, np.arange(n_init), init), 0, 1))

        for _ in range(12):
            starts.append(np.random.rand(n))
        for nb in [3, 4, 5, 7]:
            h_b = np.zeros(n)
            blk = n // nb
            for b in range(nb):
                if np.random.rand() > 0.35:
                    h_b[b * blk:(b + 1) * blk] = np.random.uniform(0.2, 1.0)
            starts.append(h_b)
        for _ in range(6):
            x = np.linspace(0, 2, n)
            freq = np.random.uniform(2.0, 10.0)
            phase = np.random.uniform(0, 2 * np.pi)
            h_s = 0.5 + 0.5 * np.sin(x * freq + phase)
            starts.append(np.clip(h_s, 0, 1))

        for h0 in starts:
            if time.time() > deadline:
                break
            h0 = fix_sum(h0, n / 2.0)
            h_cur = h0.copy()

            for temp in [0.1, 0.015, 0.003]:
                if time.time() > deadline:
                    break
                _temp_val = temp
                res = minimize(
                    fun=smooth_max_obj,
                    x0=h_cur,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_eq,
                    options={'maxiter': 400, 'ftol': 1e-13}
                )
                if res.x is not None:
                    h_cur = fix_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
                    c5 = compute_c5(h_cur)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_cur.copy()
                        best_n = n

        # Additional local search on best found for this n
        if time.time() > deadline:
            break
        if best_n == n and best_h is not None:
            for _ in range(8):
                if time.time() > deadline - 2:
                    break
                h_pert = best_h + np.random.normal(0, 0.03, n)
                h_pert = fix_sum(np.clip(h_pert, 0.0, 1.0), n / 2.0)
                _temp_val = 0.002
                res_ls = minimize(
                    fun=smooth_max_obj,
                    x0=h_pert,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_eq,
                    options={'maxiter': 300, 'ftol': 1e-14}
                )
                if res_ls.x is not None:
                    h_f = fix_sum(np.clip(res_ls.x, 0.0, 1.0), n / 2.0)
                    c5_f = compute_c5(h_f)
                    if c5_f < best_c5:
                        best_c5 = c5_f
                        best_h = h_f.copy()
                        best_n = n

    # Final COBYLA polish on overall best
    if best_h is not None and time.time() < deadline - 3:
        n_final = best_n
        _dx_val = 2.0 / n_final
        _target_sum = n_final / 2.0
        bounds_f = [(0.0, 1.0)] * n_final
        cons_f = [{'type': 'eq', 'fun': constrain_sum}]
        res_cob = minimize(
            fun=compute_c5,
            x0=best_h,
            method='COBYLA',
            constraints=cons_f,
            bounds=bounds_f,
            options={'maxiter': 300, 'rhobeg': 0.005}
        )
        if res_cob.x is not None:
            h_pol = fix_sum(np.clip(res_cob.x, 0.0, 1.0), n_final / 2.0)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_final

    # Validate with pre-imported evaluator if available
    if best_h is not None and 'evaluate_erdos_solution' in globals():
        try:
            ev = evaluate_erdos_solution(best_h)
            if ev is not None and isinstance(ev, (float, int, np.floating)):
                if ev < best_c5:
                    best_c5 = float(ev)
        except Exception:
            pass

    if best_h is None:
        best_h = init.copy()
        best_n = n_init
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
