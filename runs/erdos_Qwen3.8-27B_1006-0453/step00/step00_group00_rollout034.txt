    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def project_to_simplex(h, S, lo=0.0, hi=1.0):
    """Project h onto {lo <= x <= hi, sum(x) = S}"""
    h = np.clip(h, lo, hi)
    # Bisection on offset
    for _ in range(50):
        s = np.sum(h)
        diff = S - s
        if abs(diff) < 1e-12:
            break
        h = h + diff / len(h)
        h = np.clip(h, lo, hi)
    return h

    return float(np.max(corr) * dx)

def project_h(x, S):
    """Project x to {0<=h<=1, sum(h)=S} via bisection on offset."""
    n = len(x)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo, hi = -x.max() - 1, -x.min() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x + mid, 0, 1).sum()
        if s < S:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2, 0, 1)

    """Return (smooth_max_value, softmax_weights, corr_values)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    val = m + np.log(e.sum()) / beta
    return val, w, corr

    n = len(h)

        m2 = (v2 >= 0) & (v2 < n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(x, S):
    n = len(x)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo = -x.max() - 1.0
    hi = -x.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < S:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_weights(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    val = m + np.log(e.sum()) / beta
    return val, w

def smooth_grad(h, beta, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        shift = k - (n - 1)
        fac = wk * dx
        v1 = idx + shift
        m1 = (v1 >= 0) & (v1 < n)
        if m1.any():
            grad[m1] += fac * (1.0 - h[v1[m1]])
        v2 = idx - shift
        m2 = (v2 >= 0) & (v2 < n)
        if m2.any():
            grad[m2] -= fac * h[v2[m2]]
    return grad

def pgd_solve(h0, S, time_start, time_limit, beta_start=40, max_iter=15000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.4
    beta = beta_start
    no_improve = 0

    for it in range(max_iter):
        if time.time() - time_start > time_limit:
            break
        if it % 300 == 0 and it > 0:
            beta = min(800.0, beta * 1.15)

        val, w = smooth_obj_weights(h, beta)
        g = smooth_grad(h, beta, w)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        no_improve = 0

        alpha = lr
        h_new = h
        f_new = val
        for _ in range(30):
            h_new = project_h(h - alpha * g, S)
            f_new, _ = smooth_obj_weights(h_new, beta)
            if f_new < val - 1e-14:
                break
            alpha *= 0.5
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            lr = min(1.5, lr * 1.02)
        else:
            lr = max(0.05, lr * 0.97)
    return best_h, best_c5

def make_structured(n, rng):
    S = n / 2.0
    period = max(2, int(n * rng.uniform(0.15, 0.4)))
    fill = rng.uniform(0.3, 0.7)
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < period * fill:
            h[i] = 1.0
    h += rng.randn(n) * 0.05
    return project_h(h, S)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_structured(45, rng)
    n_init = len(h_init)
    S_init = n_init / 2.0
    h_init = project_h(h_init, S_init)

    # Phase 1: optimize from initial
    h1, c5_1 = pgd_solve(h_init, S_init, t0, t0 + 0.40 * budget_s, beta_start=50, max_iter=20000)
    best_c5 = c5_1
    best_h = h1.copy()

    # Phase 2: restarts with varied n and starts
    n_options = [40, 44, 45, 46, 48, 50, 52, 55, 60, 70, 80, 100, 120]
    phase2_end = t0 + 0.90 * budget_s

    for trial in range(200):
        if time.time() > phase2_end:
            break
        n = n_options[trial % len(n_options)]
        S = n / 2.0
        strat = trial % 5
        if strat == 0:
            h0 = project_h(rng.rand(n), S)
        elif strat == 1:
            h0 = make_structured(n, rng)
        elif strat == 2:
            h0 = np.interp(np.linspace(0, n - 1, n),
                           np.linspace(0, best_h.shape[0] - 1, best_h.shape[0]), best_h)
            h0 = project_h(h0, S)
        elif strat == 3:
            h0 = best_h + rng.randn(len(best_h)) * 0.12
            h0 = project_h(h0, len(h0) / 2.0)
            if n != len(h0):
                h0 = np.interp(np.linspace(0, n - 1, n),
                               np.linspace(0, len(h0) - 1, len(h0)), h0)
                h0 = project_h(h0, S)
        else:
            # Block pattern with random block sizes
            h0 = np.zeros(n)
            pos = 0
            while pos < n:
                blen = rng.randint(1, max(2, n // 6))
                h0[pos:pos + blen] = 1.0
                pos += blen + rng.randint(0, max(1, n // 8))
            h0 = project_h(h0, S)

        remain = max(1.0, phase2_end - time.time() - 2.0)
        h_local, c5_local = pgd_solve(h0, S, t0, t0 + phase2_end, beta_start=35,
                                       max_iter=int(remain * 2000))
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()

    # Phase 3: final polish with high beta
    if time.time() < t0 + budget_s - 3:
        n_b = len(best_h)
        h_fin, c5_fin = pgd_solve(best_h, n_b / 2.0, t0, t0 + budget_s - 2,
                                   beta_start=300, max_iter=10000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
