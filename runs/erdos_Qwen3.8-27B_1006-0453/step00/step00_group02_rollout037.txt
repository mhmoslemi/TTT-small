import numpy as np
from scipy.optimize import minimize
import time

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = np.sum(h)
    if s < 1e-12:
        h[:] = 0.5
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_of(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    else:
        n = 45
        h_best = np.zeros(n)
        h_best[::2] = 1.0
    h_best = repair_h(h_best, n)
    c_best = c5_of(h_best)

    def rem():
        return budget_s - (time.time() - t0)

    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B from multiple starts
    for r in range(8):
        if rem() < 50:
            break
        x0 = h_best.copy() if r == 0 else repair_h(h_best + rng.randn(n) * 0.07, n)
        res = minimize(lambda x: c5_of(x), x0, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 500, 'ftol': 1e-14})
        xo = repair_h(res.x, n)
        cv = c5_of(xo)
        if cv < c_best:
            c_best, h_best = cv, xo

    # Phase 2: SLSQP with equality constraint
    if rem() > 120:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        for r in range(3):
            if rem() < 60:
                break
            x0 = h_best.copy() if r == 0 else repair_h(h_best + rng.randn(n) * 0.04, n)
            res = minimize(lambda x: c5_of(x), x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10})
            xo = repair_h(res.x, n)
            cv = c5_of(xo)
            if cv < c_best:
                c_best, h_best = cv, xo

    # Phase 3: Log-sum-exp smooth objective
    for beta in [50, 150, 400]:
        if rem() < 80:
            break
        def sm(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            e = np.exp(np.clip((c - m) * beta, -700, 0))
            return (m + np.log(e.sum() / len(c)) / beta) * (2.0 / n)
        res = minimize(sm, h_best.copy(), method='L-BFGS-B',

        xo = repair_h(res.x, n)
        cv = c5_of(xo)
        if cv < c_best:
            c_best, h_best = cv, xo

    # Phase 4: Coordinate descent (bang-bang exploration)
    if rem() > 200:
        for _ in range(300):

                break
            imp = False
            for j in np.random.permutation(n):
                for v in (0.0, 1.0):
                    h_t = h_best.copy()
                    h_t[j] = v
                    h_t = repair_h(h_t, n)
                    ct = c5_of(h_t)
                    if ct < c_best - 1e-12:
                        c_best, h_best = ct, h_t
                        imp = True
                        break
            if not imp:
                h_best = repair_h(h_best + rng.randn(n) * 0.03, n)
                c_best = c5_of(h_best)

    # Phase 5: Higher resolution
    for mult in (2, 3):
        nn = n * mult
        if nn > 400 or rem() < 200:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, nn)
        h_i = repair_h(np.interp(x_new, x_old, h_best), nn)
        bn = [(0.0, 1.0)] * nn
        ci = c5_of(h_i)
        for r in range(4):
            if rem() < 60:
                break
            x0 = h_i.copy() if r == 0 else repair_h(h_i + rng.randn(nn) * 0.05, nn)
            res = minimize(lambda x: c5_of(x), x0, method='L-BFGS-B',
                           bounds=bn, options={'maxiter': 300, 'ftol': 1e-14})
            xo = repair_h(res.x, nn)
            cv = c5_of(xo)
            if cv < ci:
                ci, h_i = cv, xo
        if ci < c_best:
            c_best, h_best, n = ci, h_i.copy(), nn

    h_best = repair_h(h_best, n)
    c_best = c5_of(h_best)
    return h_best, c_best, n
