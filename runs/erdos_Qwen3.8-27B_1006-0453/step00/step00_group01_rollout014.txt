                h_opt = satisfy_sum(res.x, target)
            except Exception:
                h_opt = satisfy_sum(h0, target)

            try:
                res2 = minimize(c5_l2, h_opt, args=(n,),
                                method='L-BFGS-B', bounds=bounds,
                                options={'maxiter': 300, 'ftol': 1e-12})
                h_opt = satisfy_sum(res2.x, target)
            except Exception:
                pass

            try:
                res3 = minimize(c5_obj, h_opt, args=(n,), jac=True,
                                method='SLSQP', bounds=bounds,
                                constraints=cons,
                                options={'maxiter': 200, 'ftol': 1e-11})
                h_opt = satisfy_sum(res3.x, target)
            except Exception:
                pass

            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n

    if time.time() - t0 > budget_s * 0.92:
        if best_h is not None:
            return (best_h, best_c5, best_n)
        h_fb = np.full(47, 0.5)
        return (h_fb, compute_c5(h_fb), 47)

    if best_h is None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
        except Exception:
            best_h = np.full(47, 0.5)
            best_n = 47
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
