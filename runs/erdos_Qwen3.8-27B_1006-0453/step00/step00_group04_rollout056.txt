import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_to_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} via bisection on offset."""
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda: sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def overlap_gradient(h, k_shift):
    """Gradient of np.correlate(h, 1-h, 'full')[k_shift] w.r.t. h."""
    n = len(h)
    lag = (n - 1) - k_shift
    g = np.zeros(n)
    valid_plus = (lag >= 0)
    # Vectorized:
    # For index m, contribution is:
    # (1 - h[m+lag]) if 0 <= m+lag < n
    # - h[m-lag] if 0 <= m-lag < n
    if lag >= 0:
        # m+lag < n means m < n-lag
        m_range = np.arange(n - lag)
        g[m_range] += (1.0 - h[m_range + lag])
        # m-lag >= 0 means m >= lag
        m_range2 = np.arange(lag, n)
        g[m_range2] -= h[m_range2 - lag]

        m_range = np.arange(lag2, n)

        m_range2 = np.arange(n - lag2)
        g[m_range2] -= h[m_range2 - lag]  # h[m+lag2]
    return g

    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    # Soft max
    max_ov = np.max(overlaps)
    shifted = overlaps - max_ov
    weights = np.exp(beta * shifted)
    weights /= np.sum(weights)
    return weights

def soft_max_gradient(h, beta=50.0):

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    # Bisection on lambda: sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Final correction for numerical precision
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute evenly among non-saturated entries
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(free):
            result[free] += diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def all_overlap_gradients(h):
    """Return array of shape (2n-1, n) with gradient of each overlap."""
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = (n - 1) - k
        if lag >= 0:

            m1 = np.arange(n - lag)
            grads[k, m1] += (1.0 - h[m1 + lag])
            m2 = np.arange(lag, n)
            grads[k, m2] -= h[m2 - lag]
        else:
            L = -lag
            m1 = np.arange(L, n)
            grads[k, m1] += (1.0 - h[m1 - L])
            m2 = np.arange(n - L)
            grads[k, m2] -= h[m2 + L]
    return grads

def optimize_softgrad(h, time_limit=30, beta=30.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(100000):
        if time.time() - t0 > time_limit:
            break
        
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
        max_ov = np.max(overlaps)
        shifted = overlaps - max_ov
        weights = np.exp(beta * shifted)
        wsum = np.sum(weights)
        if wsum == 0:
            weights = np.zeros(len(overlaps))
            weights[np.argmax(overlaps)] = 1.0
        else:
            weights /= wsum

        g = np.zeros(n)
        for k in range(len(overlaps)):

                continue

            if lag >= 0:
                m1 = np.arange(n - lag)
                g[m1] += weights[k] * (1.0 - h[m1 + lag])
                m2 = np.arange(lag, n)
                g[m2] -= weights[k] * h[m2 - lag]
            else:
                L = -lag
                m1 = np.arange(L, n)
                g[m1] += weights[k] * (1.0 - h[m1 - L])
                m2 = np.arange(n - L)
                g[m2] -= weights[k] * h[m2 + L]
        g *= dx
        
        # Nesterov momentum step
        v = momentum * v + g
        h = h - lr * v
        h = project_feasible(h, target_sum)

        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
            lr = min(lr * 1.05, 0.1)
        elif c5_now > best_c5 * 1.02:
            lr *= 0.8
            h = best_h.copy()
            v *= 0.5
        
        if it % 100 == 0:
            beta = min(beta * 1.02, 200.0)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(free):
            result[free] += diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result


def weighted_gradient(h, weights, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = (n - 1) - k
        if lag >= 0:
            i1 = np.arange(lag, n)
            g[i1] += w * (1.0 - h[i1 - lag])
            i2 = np.arange(n - lag)
            g[i2] -= w * h[i2 + lag]
        else:
            L = -lag
            i1 = np.arange(n - L)
            g[i1] += w * (1.0 - h[i1 + L])
            i2 = np.arange(L, n)
            g[i2] -= w * h[i2 - L]
    return g * dx


def optimize_pgd(h, time_limit=60.0, seed=42):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = 3.0
    beta = 20.0
    mu = 0.7
    v = np.zeros(n)
    no_improve = 0

    for it in range(1000000):
        if time.time() - t0 > time_limit:
            break
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
        mx = np.max(overlaps)
        exps = np.exp(beta * (overlaps - mx))
        ws = np.sum(exps)
        if ws > 1e-300:
            weights = exps / ws
        else:
            weights = np.zeros(len(overlaps))
            weights[int(np.argmax(overlaps))] = 1.0
        g = weighted_gradient(h, weights, dx)
        v = mu * v + g
        h_new = project_feasible(h - lr * v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.03, 15.0)
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.85
            if no_improve > 3000:
                h = best_h + 0.03 * dx * rng.randn(n)
                h = project_feasible(h, target_sum)
                v = np.zeros(n)
                lr = 3.0
                no_improve = 0
        h = h_new
        if it % 1000 == 0:
            beta = min(beta * 1.02, 100.0)
    return best_h, best_c5


def interpolate_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
