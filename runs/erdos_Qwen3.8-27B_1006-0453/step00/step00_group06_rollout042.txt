            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n

    h = h0.copy()
    repair(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_dx(h, dx)

    # Phase 1: intensive local search at original n (55% budget)
    t_end1 = t0 + budget_s * 0.55
    remaining = t_end1 - time.time()
    best_h, best_c5 = local_search_pair(h, n, target_sum, dx, remaining, seed)

    # Phase 2: binary projection + polish (15% budget)
    t_end2 = t_end1 + budget_s * 0.15
    sorted_idx = np.argsort(best_h)[::-1]
    h_b = np.zeros(n)
    n_ones = int(target_sum)
    h_b[sorted_idx[:n_ones]] = 1.0
    h_b[sorted_idx[n_ones]] = target_sum - n_ones
    repair(h_b, target_sum)
    c5_b = compute_c5_dx(h_b, dx)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    remaining = t_end2 - time.time()
    if remaining > 5:
        best_h, best_c5 = local_search_pair(best_h, n, target_sum, dx, remaining, seed + 100)

    # Phase 3: multi-resolution exploration (25% budget)
    t_end3 = t_end2 + budget_s * 0.25
    x_old = np.linspace(dx, 2.0 - dx, n)
    ns_try = [n + 2, n + 4, n + 6, n + 10, n + 14, n + 18, n + 24, n + 30, n + 36, n + 42]
    for n_try in ns_try:
        if time.time() - t0 > t_end3 - 5:
            break
        if n_try < 10 or n_try % 2 == 0 or n_try >= 500:
            continue
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        x_new = np.linspace(dx_try, 2.0 - dx_try, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        repair(h_interp, target_try)
        c5_try = compute_c5_dx(h_interp, dx_try)
        per_n_time = min(15.0, (t_end3 - time.time()) / 2.0)
        if c5_try < best_c5 + 0.01:
            h_interp, c5_try = local_search_pair(h_interp, n_try, target_try, dx_try, per_n_time, seed + n_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_interp.copy()
                n = n_try
                dx = dx_try
                target_sum = target_try
                x_old = np.linspace(dx, 2.0 - dx, n)

    # Phase 4: final polish (5% budget)
    remaining = budget_s - 5 - (time.time() - t0)
    if remaining > 3:
        best_h, best_c5 = local_search_pair(best_h, n, target_sum, dx, remaining, seed + 999)

    repair(best_h, target_sum)
    best_c5 = compute_c5_dx(best_h, dx)
    return (best_h, best_c5, n)
