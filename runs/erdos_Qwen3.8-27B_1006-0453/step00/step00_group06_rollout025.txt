import numpy as np

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def repair_sum(h, target):
    h = h.copy()
    h[h < 0.0] = 0.0
    h[h > 1.0] = 1.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s < diff:
            h[room > 0] = 1.0
        else:
            h += diff * room / s
    else:
        s = h.sum()
        if s < -diff:
            h[h > 0] = 0.0
        else:
            h += diff * h / s
    h[h < 0.0] = 0.0
    h[h > 1.0] = 1.0
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmax(h))
        h[i] += diff2
        h[i] = max(0.0, min(1.0, h[i]))
    return h

    if np.any(vp):
        g[vp] += 1.0 - h[ar[vp] + s]
    if np.any(vm):
        g[vm] -= h[ar[vm] - s]
    return g
