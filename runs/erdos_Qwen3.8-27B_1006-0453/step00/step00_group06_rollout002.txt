    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = repair(h, n)
    best_h, best_c5 = h.copy(), eval_c5(h)

    # Phase 1: Aggressive hill climbing on initial (50% time)
    d1 = t0 + (budget_s - 3) * 0.5
    h, c5 = full_hillclimb(h, n, rng, 5000000, d1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    else:
        h = best_h.copy()
        c5 = best_c5

    # Phase 2: Guided refinement (25% time)
    d2 = t0 + (budget_s - 3) * 0.75
    h, c5 = guided_refine(h, n, rng, 2000000, d2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    else:
        h = best_h.copy()
        c5 = best_c5

    # Phase 3: Try a few other n values (10% time)
    d3 = t0 + (budget_s - 3) * 0.9
    for nn in [94, 96, 98, 100, 128]:
        if time.time() >= d3:
            break
        hh = make_binary_start(nn, rng)
        hh, cc = full_hillclimb(hh, nn, rng, 200000, d3)
        if cc < best_c5:
            best_c5 = cc
            best_h = hh.copy()
            h = hh.copy()

    # Phase 4: Final fine-tuning (10% time)
    n = len(best_h)
    h, c5 = full_hillclimb(best_h.copy(), n, rng, 5000000, deadline)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Verify
    best_h = repair(best_h, len(best_h))
    final_c5 = eval_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
