def smooth_gradient_fast(h, n, beta):

    max_c = np.max(corr)

    p = exp_vals / np.sum(exp_vals)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for ki in range(len(corr)):
        k = ki - (n - 1)
        wk = p[ki] * dx
        if abs(wk) < 1e-15:

    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _corr_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h, n):
    return np.max(_corr_vec(h, n))

def _smooth_grad(h, n, beta):
    corr = _corr_vec(h, n)
    mc = np.max(corr)
    ev = np.exp(beta * (corr - mc))
    p = ev / np.sum(ev)
    dx = 2.0 / n
    grad = np.zeros(n)
    for ki in range(len(corr)):
        k = ki - (n - 1)
        wk = p[ki] * dx
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n-kk])
            grad[:n-kk] -= wk * h[kk:]
    return grad

def _pgd_optimize(h, n, max_iter, time_limit_start, budget_s):
    alpha = 0.02
    beta = 40.0
    for it in range(max_iter):
        if time.time() - time_limit_start > budget_s - 10:
            break
        c5_cur = _c5(h, n)
        grad = _smooth_grad(h, n, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        h_trial = _project_feasible(h - alpha * grad / gn, n)
        c5_trial = _c5(h_trial, n)
        if c5_trial < c5_cur:
            h = h_trial
            alpha = min(alpha * 1.05, 0.1)
        else:
            alpha *= 0.7
            if alpha < 1e-6:
                break
    return h

def _slsqp_optimize(h, n, time_limit_start, budget_s):
    def obj_t(x):
        if time.time() - time_limit_start > budget_s - 5:
            return 999.0
        h_v = x[:n]
        t = x[n]
        corr = _corr_vec(h_v, n)
        return t + 0.01 * np.sum(np.maximum(corr - t, 0)**2)
    def jac_t(x):
        h_v = x[:n]
        t = x[n]
        corr = _corr_vec(h_v, n)
        g = _smooth_grad(h_v, n, 200.0)
        penalty = 0.02 * np.maximum(corr - t, 0)
        for ki in range(len(corr)):
            k = ki - (n - 1)
            pk = penalty[ki]
            if pk < 1e-14:
                continue
            if k == 0:
                g += pk * (1.0 - 2.0 * h_v)
            elif k > 0:
                g[:n-k] += pk * (1.0 - h_v[k:])
                g[k:] -= pk * h_v[:n-k]
            else:
                kk = -k
                g[kk:] += pk * (1.0 - h_v[:n-kk])
                g[:n-kk] -= pk * h_v[kk:]
        jac = np.zeros(n + 1)
        jac[:n] = g
        jac[n] = 1.0
        return jac
    x0 = np.concatenate([h, [_c5(h, n) + 0.01]])
    bounds = [(0, 1)] * n + [(-0.01, 1.5)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n/2.0,

    try:

        return h

    if n == n0:
        return h0.copy()
    x_old = np.linspace(0, 2, n0, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    hh = np.interp(x_new, x_old, h0)
    return _project_feasible(np.clip(hh, 0, 1), n)

    return _project_feasible(np.clip(h, 0, 1), n)
