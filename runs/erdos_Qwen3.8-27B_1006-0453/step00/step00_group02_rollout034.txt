import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def gradient_at_lag(h, lag):
    """Gradient of corr[lag + n - 1] with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        # Term from h[j] * (1-h[j+lag])
        val = 0.0
        j_plus = j + lag
        if 0 <= j_plus < n:
            val += (1.0 - h[j_plus])
        # Term from -h[j] * h[j-lag] (from the (1-h[i]) when i = j-lag)
        j_minus = j - lag
        if 0 <= j_minus < n:
            val -= h[j_minus]
        grad[j] = val
    return grad

def soft_gradient(h, beta=10.0):
    """Compute smooth gradient of max correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Softmax weights
    max_corr = np.max(corr)
    weights = np.exp(beta * (corr - max_corr))
    weights /= np.sum(weights)
    
    # Weighted gradient
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w * gradient_at_lag(h, lag)
    
    return grad * dx  # Scale by dx since objective includes dx

def project_to_manifold(h, target_sum):
    """Project h to {0≤h≤1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    if diff > 0:
        # Need to increase sum; add to elements not at 1
        mask = h < 1.0 - 1e-10
        if np.sum(mask) == 0:
            return h
        per_elem = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + per_elem, 0, 1)
    else:
        # Need to decrease sum; remove from elements not at 0
        mask = h > 1e-10
        if np.sum(mask) == 0:
            return h
        per_elem = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + per_elem, 0, 1)
    
    # Second pass for better convergence
    for _ in range(10):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if np.sum(mask) == 0:

            per_elem = diff / np.sum(mask)
            h[mask] = np.clip(h[mask] + per_elem, 0, 1)
        else:
            mask = h > 1e-12
            if np.sum(mask) == 0:
                break
            per_elem = diff / np.sum(mask)

    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_manifold(h, target_sum):
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(corr)
    ev = np.exp(beta * (corr - cm))
    w = ev / ev.sum()
    o = 1.0 - h
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = k - n1
        if L >= 0:
            m = n - L
            grad[L:n] += wk * o[:m]
            grad[:m] -= wk * h[L:n]
        else:
            Lp = -L
            m = n - Lp
            grad[:m] += wk * o[Lp:n]
            grad[Lp:n] -= wk * h[:m]
    return grad * dx

def optimize_adam(h, n, time_limit, lr=0.02, beta_s=2.0, beta_e=40.0):
    t0 = time.time()
    ts = 0.5 * n
    h = project_manifold(h.copy(), ts)
    best = h.copy()
    best_v = compute_c5(h)
    m = np.zeros(n)
    vv = np.zeros(n)
    b1, b2 = 0.9, 0.999
    ti = 1
    stuck = 0
    while time.time() - t0 < time_limit:
        frac = min(1.0, (time.time() - t0) / max(1.0, time_limit))
        beta = beta_s + (beta_e - beta_s) * frac
        g = smooth_grad(h, beta)
        m = b1 * m + (1.0 - b1) * g
        vv = b2 * vv + (1.0 - b2) * g * g
        mhat = m / (1.0 - b1 ** ti)
        vhat = vv / (1.0 - b2 ** ti)
        step = lr * mhat / (np.sqrt(vhat) + 1e-8)
        h_new = project_manifold(h - step, ts)
        val = compute_c5(h_new)
        if val < best_v - 1e-12:
            best = h_new.copy()
            best_v = val
            stuck = 0
        else:

        h = h_new
        ti += 1
        if stuck > 300:

            h[idx] += 0.15 * np.random.randn(len(idx))

            stuck = 0

    n = len(h)
    if nt == n:
        return h.copy()
    if nt % n == 0:
        h_new = np.repeat(h, nt // n)
    else:
        xo = (np.arange(n) + 0.5) * 2.0 / n
        xn = (np.arange(nt) + 0.5) * 2.0 / nt
        h_new = np.interp(xn, xo, h)
