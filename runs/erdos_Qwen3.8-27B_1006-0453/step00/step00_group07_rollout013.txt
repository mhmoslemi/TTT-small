import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def fix_sum(h, n, rng):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        candidates = np.where(h < 1.0)[0]
    else:
        candidates = np.where(h > 0.0)[0]
    if len(candidates) == 0:
        return h
    per = diff / len(candidates)
    h[candidates] = np.clip(h[candidates] + per, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h


def optimize_sa(h_init, n, time_limit, T0, T_end, rng):
    h = h_init.copy()
    c = compute_c5(h)
    best_local = h.copy()

    no_improve = 0

        elapsed = time.time() - start_t
        frac = min(elapsed / total, 1.0)
        T = T0 * (T_end / T0) ** frac
        # Reheat if stuck

            T = max(T, T0 * 0.3)
            no_improve = 0

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))

        h[j] -= delta

            h[i] = 0.0
