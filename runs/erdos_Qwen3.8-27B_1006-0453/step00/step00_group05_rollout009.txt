from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(x, tau=10):
    m = np.max(x)
    if m < -1000:
        return m
    x_shifted = x - m
    return m + (1/tau) * np.log(np.sum(np.exp(tau * x_shifted)))

def objective_softmax(x, target_sum, tau=50):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    c5_vals = corr * dx
    # soft max
    obj = smooth_max(c5_vals, tau)
    # penalty for constraint violation
    penalty = 100 * (x.sum() - target_sum) ** 2
    # penalty for box constraints
    penalty += 100 * (np.sum(np.maximum(0, -x)**2) + np.sum(np.maximum(0, x-1)**2))

            # Need to decrease; reduce non-zero entries

                if diff <= 0:
                    break

                h[i] -= amt
                diff -= amt

            for i in idx:
                if diff >= 0:
                    break
                amt = min(1 - h[i], -diff)
                h[i] += amt
                diff += amt
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_c5(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if tau > 1e8:
        return m
    shifted = tau * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / tau

def objective_fn(h, target_sum, tau):
    n = len(h)
    val = soft_c5(h, tau)
    s = h.sum() - target_sum
    val += 800.0 * s * s
    val += 800.0 * np.sum(np.maximum(0.0, -h) ** 2)
    val += 800.0 * np.sum(np.maximum(0.0, h - 1.0) ** 2)
    return val

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(h)[::-1]
        rem = diff
        for i in idx:
            t = min(h[i], rem)
            h[i] -= t
            rem -= t
            if rem <= 0:
                break
    else:
        idx = np.argsort(h)
        rem = -diff
        for i in idx:
            t = min(1.0 - h[i], rem)
            h[i] += t
            rem -= t
            if rem <= 0:
                break
    return h

def discrete_local_search(h, n, rng, max_iters=300):
    best_c5 = compute_c5(h)
    h = h.copy()
    for _ in range(max_iters):
        improved = False
        for i in range(n):
            old = h[i]
            for c in [0.0, 1.0]:
                if c == old:
                    continue
                h[i] = c
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    break
                h[i] = old
                h = project_feasible(h, n)
            if not improved and c5 < best_c5 - 1e-12:
                best_c5 = c5
                improved = True
        if not improved:
            break
    return h, best_c5

def continuous_refine(h, n, rng, budget_left, max_powell_iters=800):
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    base_tau = 30.0
    for it in range(15):
        if budget_left() < 15:
            break
        if it == 0:
            x0 = best_h.copy()
        else:
            x0 = best_h.copy()
            k = rng.randint(1, max(2, n // 6))
            idx = rng.choice(n, size=k, replace=False)
            x0[idx] += rng.uniform(-0.4, 0.4, size=k)
            x0 = project_feasible(x0, n)
        tau = base_tau + 5.0 * it
        res = minimize(
            objective_fn, x0, args=(target_sum, tau),
            method='Powell',
            options={'maxiter': min(max_powell_iters, max(50, int(budget_left() / 2.0))),
                     'xatol': 1e-8, 'fatol': 1e-10, 'cratol': 1e-10}
        )
        h_cand = project_feasible(res.x, n)
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
    return best_h, best_c5

def greedy_01_init(n, rng):
    h = np.zeros(n)

    return h
