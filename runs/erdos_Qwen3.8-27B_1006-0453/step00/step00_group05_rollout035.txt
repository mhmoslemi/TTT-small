        h_up3 = _fix_sum(h_up3, n3)
        t6 = min(remaining * 0.5, 120)
        h_b3, c5_b3 = _binary_sa(h_up3, dx3, t6, rng)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3

    # Validate and return
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _c5_of(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
