import numpy as np
from scipy.optimize import minimize
import time

_SCALE = 0.002

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_obj(h, scale):
    c = corr_dx(h)
    m = np.max(c)
    return m + scale * np.log(np.sum(np.exp((c - m) / scale)))

def soft_max_grad(h, scale):
    n = len(h)
    dx = 2.0 / n
    c = corr_dx(h)
    m = np.max(c)
    w = np.exp((c - m) / scale)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        wk = w[n - 1 + k] * dx
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            jm = max(0, k)
            jM = min(n, n + k)
            if jm < jM:
                grad[jm:jM] += wk * (1.0 - h[jm - k:jM - k])
            jm2 = max(0, -k)
            jM2 = n - k
            if jm2 < jM2:
                grad[jm2:jM2] += wk * (-h[jm2 + k:jM2 + k])
    return grad

def obj_func(h):
    return soft_max_obj(h, _SCALE)

def obj_grad_func(h):
    return (soft_max_obj(h, _SCALE), soft_max_grad(h, _SCALE))

    return np.ones(len(h))

    if abs(diff) > 1e-12:

            mask = h > 0.0

            h[mask] += diff / cnt

    return h

    if s > 1e-15:
