    corr = np.correlate(hc, 1.0 - hc, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    grad = np.zeros(n)
    lo = max(0, s)
    hi = min(n, s + n)
    if lo < hi:
        grad[lo:hi] += (1.0 - hc[lo - s:hi - s])
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if lo2 < hi2:
        grad[lo2:hi2] -= hc[lo2 + s:hi2 + s]
    grad *= dx
    return float(corr[k_star] * dx), grad

def constraint_sum(x):
    return float(np.sum(x) - len(x) / 2.0)

def constraint_sum_grad(x):
    return np.ones(len(x))

def do_slsqp(h_init, n, maxiter, ftol):
    h_init = adjust_sum(np.clip(h_init, 0.0, 1.0), n / 2.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_sum_grad}]
    try:
        def _wrapper(x):
            return obj_and_grad(x, n)[0]
        def _grad_wrapper(x):
            return obj_and_grad(x, n)[1]
        res = minimize(_wrapper, h_init, method='SLSQP', jac=_grad_wrapper,
                       bounds=bounds, constraints=cons,
                       options={'disp': False, 'maxiter': maxiter, 'ftol': ftol})
        h_new = adjust_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
        c5 = compute_c5(h_new, n)
        return h_new, c5
    except Exception:
        return None, None

def coord_descent(h, n, n_pairs, rng, deadline):
    h = h.copy()
    target = n / 2.0
    current = compute_c5(h, n)
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        c = h[i] + h[j]
        a_min = max(0.0, c - 1.0)
        a_max = min(1.0, c)
        if a_min >= a_max:
            continue
        n_samp = 15
        for a in np.linspace(a_min, a_max, n_samp):
            h[i] = a
            h[j] = c - a
            f = compute_c5(h, n)
            if f < current:
                current = f
        h[i] = min(max(h[i], a_min), a_max)
        h[j] = c - h[i]
    return h, current

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_fuzzy(n, rng, parent):
    h = random_binary(n, rng)
    h = 0.3 * h + 0.7 * parent
    h = np.clip(h, 0.0, 1.0)
    h = adjust_sum(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except (NameError, AttributeError, TypeError, ValueError):
        h_parent = None
    if h_parent is None or len(h_parent) == 0:
        h_parent = np.zeros(70)
        h_parent[:35] = 1.0

    best_c5 = compute_c5(h_parent, len(h_parent))
    best_h = h_parent.copy()
    best_n = len(h_parent)
    n0 = len(h_parent)
    deadline = start + budget_s - 10

    # Phase 1: SLSQP at parent resolution
    if time.time() < deadline - 30:
        h_new, c5_new = do_slsqp(h_parent, n0, 500, 1e-9)
        if h_new is not None and c5_new < best_c5:
            best_c5, best_h, best_n = c5_new, h_new.copy(), n0

    # Phase 2: Upsample and optimize at higher resolutions
    for factor in [2, 3, 4, 5, 7, 10]:
        if time.time() > deadline - 50:
            break
        n = n0 * factor
        if n > 800:
            break
        h_up = upsample_repeat(best_h, factor)
        h_up = adjust_sum(h_up, n / 2.0)
        maxiter = max(30, 300 * n0 // n)
        h_new, c5_new = do_slsqp(h_up, n, maxiter, 1e-8)
        if h_new is not None and c5_new < best_c5:
            best_c5, best_h, best_n = c5_new, h_new.copy(), n
        if time.time() > deadline - 50:
            break
        # Coordinate descent fine-tune
        deadline_cd = min(deadline - 30, time.time() + 60)
        h_cd, c5_cd = coord_descent(h_new, n, min(2000, 400000 // n), rng, deadline_cd)
        if h_cd is not None and c5_cd < best_c5:
            best_c5, best_h, best_n = c5_cd, h_cd.copy(), n

    # Phase 3: Random restarts at n0
    n_restarts = 25
    for _ in range(n_restarts):
        if time.time() > deadline - 20:
            break
        h_rand = random_fuzzy(n0, rng, best_h)
        h_new, c5_new = do_slsqp(h_rand, n0, 200, 1e-8)
        if h_new is not None and c5_new < best_c5:
            best_c5, best_h, best_n = c5_new, h_new.copy(), n0

    # Phase 4: Coordinate descent on best solution
    if time.time() < deadline - 10:
        deadline2 = min(deadline, time.time() + 300)
        h_cd, c5_cd = coord_descent(best_h, best_n, 5000, rng, deadline2)
        h_cd = adjust_sum(h_cd, best_n / 2.0)
        c5_cd = compute_c5(h_cd, best_n)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    # Final: SLSQP polish
    if time.time() < deadline - 20:
        h_new, c5_new = do_slsqp(best_h, best_n, 300, 1e-10)
        if h_new is not None and c5_new < best_c5:
            best_c5, best_h, best_n = c5_new, h_new.copy(), best_n

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)

    return (best_h, best_c5, best_n)
