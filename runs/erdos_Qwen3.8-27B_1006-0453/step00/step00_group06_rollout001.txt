import numpy as np
import time


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h_init = None

    def c5_of(h):
        n = len(h)
        dx = 2.0 / n
        return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

    if h_init is not None:
        best_h = h_init.copy()
        best_c5 = c5_of(best_h)
    else:
        best_h = np.zeros(96)
        best_h[:48] = 1.0
        rng.shuffle(best_h)
        best_c5 = c5_of(best_h)

    best_n = len(best_h)

    def local_search(h, time_limit, rng_ls):
        n = len(h)
        h = h.copy()
        curr = c5_of(h)
        t_s = time.time()
        stall = 0
        while time.time() - t_s < time_limit:
            i = int(rng_ls.integers(0, n))
            j = int(rng_ls.integers(0, n))
            if i == j:
                continue
            oi, oj = h[i], h[j]
            h[i], h[j] = oj, oi
            new = c5_of(h)
            if new <= curr + 1e-15:
                curr = new
                stall = 0
            else:
                h[i], h[j] = oi, oj
                stall += 1
            if stall > 30000:
                kicks = int(rng_ls.integers(3, 12))
                for _ in range(kicks):
                    a = int(rng_ls.integers(0, n))
                    b = int(rng_ls.integers(0, n))
                    if a != b:
                        h[a], h[b] = h[b], h[a]
                curr = c5_of(h)
                stall = 0
        return h, curr

    def make_random(n, rng_ls):
        h = np.zeros(n)
        half = n // 2
        h[:half] = 1.0
        if n % 2 == 1:
            h[half] = 0.5
        rng_ls.shuffle(h)
        target = n / 2.0
        s = h.sum()
        if abs(s - target) > 1e-12:
            diff = target - s
            h[int(np.argmax(h < 1.0)) if diff > 0 else int(np.argmax(h > 0.0))] += diff
            h = np.clip(h, 0.0, 1.0)
        return h

    iter_count = 0
    while time.time() - t0 < budget_s - 5:
        time_left = budget_s - 5 - (time.time() - t0)
        if time_left < 3:
            break

        mode = iter_count % 5
        if mode == 0:
            chunk = min(20.0, time_left)
            h_w, c5_w = local_search(best_h, chunk, rng)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()
                best_n = len(best_h)
        elif mode == 1:
            h_w = best_h.copy()
            n = len(h_w)
            npert = int(rng.integers(3, min(15, n // 4)))
            for _ in range(npert):
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a != b:
                    h_w[a], h_w[b] = h_w[b], h_w[a]
            chunk = min(20.0, time_left)
            h_w, c5_w = local_search(h_w, chunk, rng)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()
                best_n = len(best_h)
        elif mode == 2:
            n_try = int(rng.integers(92, 116))
            h_r = make_random(n_try, rng)
            chunk = min(20.0, time_left)
            h_r, c5_r = local_search(h_r, chunk, rng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
        elif mode == 3:
            # Focused: reduce worst lag
            n = len(best_h)
            h_w = best_h.copy()
            dx = 2.0 / n
            corr = np.correlate(h_w, 1.0 - h_w, mode="full")
            worst_idx = int(np.argmax(corr))
            worst_lag = worst_idx - n + 1
            chunk = min(15.0, time_left)
            t_f = time.time()
            while time.time() - t_f < chunk:
                contrib_i = []
                for i in range(n):
                    j2 = i + worst_lag
                    if 0 <= j2 < n and h_w[i] > 0.01 and h_w[j2] < 0.99:
                        contrib_i.append(i)
                if not contrib_i:
                    break
                i = int(contrib_i[int(rng.integers(0, len(contrib_i)))])
                j2 = int(rng.integers(0, n))
                if i == j2:
                    continue
                oi, oj = h_w[i], h_w[j2]
                h_w[i], h_w[j2] = oj, oi
                new_c5 = c5_of(h_w)
                if new_c5 < c5_of(best_h) + 1e-15:
                    pass
                else:
                    h_w[i], h_w[j2] = oi, oj
            h_w, c5_w = local_search(h_w, min(5.0, time_left - (time.time() - t0 - (budget_s - 5))), rng)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()
                best_n = len(best_h)
        else:
            # Larger perturbation restart
            h_w = best_h.copy()
            n = len(h_w)
            npert = int(rng.integers(10, min(30, n // 3)))
            for _ in range(npert):
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a != b:
                    h_w[a], h_w[b] = h_w[b], h_w[a]
            chunk = min(20.0, time_left)
            h_w, c5_w = local_search(h_w, chunk, rng)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()
                best_n = len(best_h)

        iter_count += 1

    # Final polish
    if time.time() - t0 < budget_s - 3:
        h_final, c5_final = local_search(best_h, max(3.0, budget_s - 3 - (time.time() - t0)), rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    return best_h, best_c5, len(best_h)
