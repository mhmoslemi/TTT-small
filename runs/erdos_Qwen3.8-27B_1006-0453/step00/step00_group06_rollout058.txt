```python
import numpy as np
from scipy.optimize import minimize
import time

_N = 0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h = h + diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h = h + diff * h / s
    return np.clip(h, 0.0, 1.0)

def _obj(x):
    return compute_c5(x, _N)

def _eqc(x):
    return x.sum() - _N * 0.5

def slsqp(h, n, maxiter=300, ftol=1e-12):
    global _N
    _N = n
    res = minimize(_obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _eqc}],
                   options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
    return res.x
