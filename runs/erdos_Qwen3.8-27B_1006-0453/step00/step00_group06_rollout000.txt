import numpy as np

def eval_c5(h, dx):

    return np.max(c) * dx


def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(np.abs(h - 0.5))
    rem = diff
    for idx in order:
        if rem > 0:
            a = min(rem, 1.0 - h[idx])
            h[idx] += a
            rem -= a
        else:
            a = min(-rem, h[idx])
            h[idx] -= a
            rem += a
        if abs(rem) < 1e-14:
            break
    return h
