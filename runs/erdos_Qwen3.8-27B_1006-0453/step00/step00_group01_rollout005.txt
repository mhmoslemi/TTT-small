```python
    corr = compute_correlations(h)
    # Log-sum-exp weights
    m = np.max(corr)
    scaled = beta * (corr - m)
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)  # shape (2n-1,)
    
    # Compute gradient analytically
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wi = w[idx]
        if abs(wi) < 1e-15:
            continue
        if k >= 0:
            # dc_k/dh[j] = (1-h[j-k]) for j in [k, n-1] - h[j+k] for j in [0, n-1-k]
            for j in range(k, n):
                grad[j] += wi * (1.0 - h[j-k])
            for j in range(0, n - k):
                grad[j] += -wi * h[j+k]
        else:
            kp = -k
            for j in range(0, n - kp):
                grad[j] += wi * (1.0 - h[j+kp])
            for j in range(kp, n):
                grad[j] += -wi * h[j-kp]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    return np.max(compute_correlations(h))

def project(h, n):
    target = n / 2.0
    lo = -1.5
    hi = 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h, beta):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    scaled = beta * (corr - m)
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(n):
        wp = w[k + n - 1]
        wn = w[n - 1 - k]
        if k == 0:
            grad += wp * (1.0 - 2.0 * h)
        else:
            if wp > 1e-12:
                grad[k:] += wp * (1.0 - h[:n - k])
                grad[:n - k] -= wp * h[k:]
            if wn > 1e-12:
                grad[:n - k] += wn * (1.0 - h[k:])
                grad[k:] -= wn * h[:n - k]
    return grad

def optimize_projected(h0, budget_start, max_s=1040, max_iter=20000):

            break

def smooth_objective(h, beta=200.0):
    corr = compute_correlations(h)
    m = np.max(corr)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - m))))

def optimize_lbfgs(h0, budget_start, max_s=1040):
    n = len(h0)
    h_init = project(h0.copy(), n)
    h0_flat = h_init.ravel()
    def obj(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        penalty = 0.001 * (s - n / 2.0) ** 2
        return smooth_objective(h, 200.0) + penalty
    res = minimize(obj, h0_flat, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 3000, 'maxfun': 50000})
    if res.success or res.fun < compute_c5(h_init):
        h_res = project(np.clip(res.x, 0, 1), n)
        c5 = compute_c5(h_res)
        return h_res, c5
    return h_init, compute_c5(h_init)

def make_structured(n, pattern='block'):
    h = np.zeros(n)
    half = n // 2
    if pattern == 'block':
        h[:half] = 1.0
    elif pattern == 'sawtooth':
        for i in range(n):
            h[i] = 1.0 if (i % (2 * half) < half) else 0.0
    elif pattern == 'golden':
        phi = 0.618033988749895
        pos = (np.arange(half) * phi * n) % n
        h[np.sort(pos.astype(int))] = 1.0
    elif pattern == 'random':
        idx = np.random.choice(n, half, replace=False)
        h[idx] = 1.0
    return h

def interpolate_h(h_src, n_src, n_dst):

    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)
    h_dst = np.clip(h_dst, 0, 1)
    return h_dst
```
