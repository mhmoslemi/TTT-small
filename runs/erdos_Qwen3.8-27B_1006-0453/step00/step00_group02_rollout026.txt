import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return np.clip(h, 0.0, 1.0)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def c5_value(h, n):
    dx = 2.0 / n
    h_p = project_feasible(h, n)
    corr = np.correlate(h_p, 1.0 - h_p, mode="full")
    return float(np.max(corr) * dx)


def c5_objective(x, n):
    return c5_value(x, n)


def upsample_linear(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)

    best_h = h0.copy()

    # Resolution schedule: start from n0, double up to 360
    res_list = [n0]

        n = min(n * 2, 360)
        if n not in res_list:

    current_h = h0.copy()

        elapsed = time.time() - t0

        # Config per resolution

        elif n_pts <= 180:
