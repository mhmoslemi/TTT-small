import numpy as np
from scipy.optimize import minimize
import time
import math

def corr_dx(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def c5_val(h):
    n = len(h)
    return float(np.max(corr_dx(h, n)))

def fixed_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, n, eps):
    c = corr_dx(h, n)
    m = np.max(c)
    if m < 1e-15:
        return 0.0
    shifted = (c - m) / eps
    max_s = np.max(shifted)
    return float(m + eps * (max_s + np.log(np.sum(np.exp(shifted - max_s)))))

def obj_grad(h, n, eps):
    c = corr_dx(h, n)
    m = np.max(c)
    if m < 1e-15:
        return np.zeros(n)
    shifted = (c - m) / eps
    w = np.exp(shifted - np.max(shifted))
    w /= np.sum(w)
    dx = 2.0 / n
    grad = np.zeros(n)
    nk1 = 2 * n - 1
    for k in range(nk1):

            if 0 <= js < n:
                val += (1.0 - h[js])
            jm = m_idx - s
            if 0 <= jm < n:
                val -= h[jm]
            if val != 0.0:
                grad[m_idx] += w[k] * dx * val
    return grad
