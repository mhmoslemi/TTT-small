    if time.time() - t_start < budget_s * 0.50:
        h_cont = slsqp_optimize(best_h.copy(), max_iter=400)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_h = h_cont.copy()
            best_c5 = c5_cont
        hb2 = make_binary(h_cont, seed + 13)
        h_cont2 = slsqp_optimize(hb2, max_iter=200)
        c5_c2 = compute_c5(h_cont2)
        if c5_c2 < best_c5:
            best_h = h_cont2.copy()
            best_c5 = c5_c2

    # Phase 4: Upsample and refine
    for target_n in [140, 200, 280, 350, 500, 700]:
        if time.time() - t_start > budget_s * 0.88:
            break
        if target_n <= len(best_h):
            continue
        h_up = upsample(best_h, target_n)
        hb_up = make_binary(h_up, seed + target_n)
        iters = 100000 if target_n <= 300 else 60000
        h_sa, c5_sa = binary_sa(hb_up, rng, iters=iters, t0=0.05, t1=0.0005)
        if target_n <= 280:
            h_fs, c5_fs = full_binary_search(h_sa.copy(), max_sweeps=25)
            if c5_fs < c5_sa:
                h_sa, c5_sa = h_fs, c5_fs
        # Also try continuous at this resolution
        if time.time() - t_start < budget_s * 0.80:
            h_c = slsqp_optimize(h_sa, max_iter=200)
            c5_c = compute_c5(h_c)
            if c5_c < c5_sa:
                h_sa, c5_sa = h_c, c5_c
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa

    # Final: fix sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    n = len(best_h)
    diff = n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(best_h)
            for idx in order:
                if diff <= 0:
                    break
                a = min(1.0 - best_h[idx], diff)
                best_h[idx] += a
                diff -= a
        else:
            order = np.argsort(-best_h)
            for idx in order:
                if diff >= 0:
                    break
                a = min(best_h[idx], -diff)
                best_h[idx] -= a
                diff += a

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
