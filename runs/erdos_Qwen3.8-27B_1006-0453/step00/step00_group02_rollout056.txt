def project_onto_simplex_box(h, target_sum):
    """Project h onto {x ∈ [0,1]^n : sum(x) = target_sum}"""
    # Use bisection on the Lagrange multiplier
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune: adjust the non-boundary elements
    # If still slightly off, distribute the error
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        # Adjust interior points
        interior = (proj > 1e-8) & (proj < 1 - 1e-8)
        if np.any(interior):
            adj = diff / np.sum(interior)
            proj[interior] += adj
            proj = np.clip(proj, 0, 1)

def subgradient(h, active_indices):
    """Compute subgradient of max over active correlation indices"""
    n = len(h)
    grad = np.zeros(n)
    for j in active_indices:
        lag = j - (n - 1)
        # ∂corr_j/∂h[m] for each m
        # Term 1: (1-h[m-lag]) for valid m, m-lag
        # Term 2: -h[m+lag] for valid m, m+lag

            if 0 <= ml < n:

            mp = m + lag
            if 0 <= mp < n:
                g -= h[mp]
            grad[m] += g
    grad /= len(active_indices)

import numpy as np
import time

def project_feasible(h, target_sum):
    """Project h onto {x in [0,1]^n : sum(x) = target_sum} via bisection."""
    n = len(h)
    # The set is non-empty if 0 <= target_sum <= n
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    # Fine-tune for exact sum
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        interior = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if interior.sum() > 0:
            proj[interior] += diff / interior.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_corr(h):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_for_lag(h, lag):
    """Compute gradient of corr at a specific lag with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # corr = sum_i h[i] * (1-h[i-lag]) for valid i
    # valid: 0 <= i < n and 0 <= i-lag < n
    i_lo = max(0, lag)
    i_hi = min(n, n + lag)  # exclusive
    if i_hi <= i_lo:
        return grad
    # For each m, d(corr)/dh[m]:
    # From first factor: if m in [i_lo, i_hi): (1 - h[m-lag])
    # From second factor: if m+lag in [i_lo, i_hi) and 0 <= m < n: -h[m+lag]
    valid_m1 = np.arange(i_lo, i_hi)
    grad[valid_m1] += (1.0 - h[valid_m1 - lag])
    
    # Second part: i-lag = m => i = m+lag, need 0<=m<n and i_lo <= m+lag < i_hi
    m_lo = max(0, i_lo - lag)
    m_hi = min(n, i_hi - lag)
    if m_hi > m_lo:
        valid_m2 = np.arange(m_lo, m_hi)
        grad[valid_m2] -= h[valid_m2 + lag]
    
    return grad

def full_subgradient(h, active_fracs):
    """Average gradient over active correlation indices."""
    n = len(h)
    total = np.zeros(n)
    total_fracs = 0.0
    for j, frac in active_fracs:
        lag = j - (n - 1)
        g = gradient_for_lag(h, lag)
        total += frac * g
        total_fracs += frac
    if total_fracs > 0:
        total /= total_fracs
    return total

def objective_value(h):
    n = len(h)
    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        m = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if m.sum() > 0:
            proj[m] += diff / m.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    lo = max(0, lag)
    hi = min(n, n + lag)
    if hi <= lo:
        return g
    g[lo:hi] += (1.0 - h[lo:hi - lag])
    mlo = max(0, lo - lag)
    mhi = min(n, hi - lag)
    if mhi > mlo:
        g[mlo:mhi] -= h[mlo + lag:mhi + lag]
    return g

def _smooth_obj(h, tau):
    n = len(h)
    c = _corr(h)
    cm = c.max()
    e = np.exp((c - cm) / max(tau, 1e-8))
    s = e.sum()
    val = (cm + tau * np.log(s)) * (2.0 / n)
    return val

def _smooth_grad(h, tau):
    n = len(h)
    c = _corr(h)
    cm = c.max()
    e = np.exp((c - cm) / max(tau, 1e-8))
    w = e / e.sum()
    g = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] > 1e-12:
            g += w[j] * _grad_lag(h, j - (n - 1))
    return g * (2.0 / n)

def _optimize_n(h0, n, time_budget, tau_start=2.0, tau_end=0.02):
    target = n / 2.0
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_val = _smooth_obj(h, tau_start)
    
    tau = tau_start
    t0 = time.time()
    n_anneal = 5  # number of annealing phases
    per_phase = max(5, int(time_budget / n_anneal))
    
    for phase in range(n_anneal):
        tau_cur = tau_start * (tau_end / tau_start) ** (phase / max(n_anneal - 1, 1))
        phase_end = t0 + (phase + 1) * per_phase
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        
        try:
            res = minimize(_smooth_obj, h, args=(tau_cur,),
                          jac=_smooth_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-9})
            if res.fun < best_val:
                # Check actual objective (not smooth)
                actual = np.max(_corr(res.x)) * (2.0 / n)
                if actual < _smooth_obj(best_h, tau_cur) * 1.1:
                    best_h = res.x.copy()
                    best_val = actual
                h = res.x.copy()
        except Exception:
            pass
        
        h = project_feasible(h, target)
        if time.time() > phase_end:
            continue
    
    # Final: compute true objective
    actual_val = np.max(_corr(best_h)) * (2.0 / n)
    return best_h, actual_val

def _upsample(h, n_new):
    n_old = len(h)
    # Treat h as piecewise constant on [0, 2/n_old] each
    x_old = np.arange(n_old) / n_old * 2.0 + 1.0 / n_old
    x_new = np.arange(n_new) / n_new * 2.0 + 1.0 / n_new
    return np.interp(x_new, x_old, h)

def _random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

    t_start = time.time()

    best_h, best_val = _optimize_n(h_init, n_init, t1)

    time_left = budget_s - (time.time() - t_start)

        h_new, v_new = _optimize_n(h_up, n_new, t3)
        if v_new < best_val:
            best_h, best_val = h_new, v_new
    
    # Final: try a few more refinements
    time_left = budget_s
