import numpy as np
from scipy import optimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_to_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        cnt = float(np.sum(mask))
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_run(h_start, n, dx, target_sum, bounds, cons, maxiter, ftol):
    def obj(x):
        return compute_c5(x, dx)
    res = optimize.minimize(obj, h_start, method='SLSQP', bounds=bounds,
                            constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_to_sum(res.x, target_sum)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_to_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def tl():
        return budget_s - (time.time() - start_time)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP directly from initial
    try:
        h_new = _slsqp_run(best_h, n, dx, target_sum, bounds, cons, 1000, 1e-14)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    except Exception:
        pass

    # Phase 2: Random restarts with SLSQP
    scale = 0.15
    stall_count = 0
    while tl() > 60:
        h_start = best_h + rng.normal(0, scale, n)
        h_start = project_to_sum(h_start, target_sum)
        try:
            h_new = _slsqp_run(h_start, n, dx, target_sum, bounds, cons, 400, 1e-13)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5 = h_new, c5_new
                stall_count = 0
                scale = max(0.03, scale * 0.85)
            else:
                stall_count += 1
                if stall_count > 4:
                    scale = min(0.35, scale * 1.2)
                    stall_count = 0
        except Exception:
            pass

    # Phase 3: Coordinate descent polish
    if tl() > 20:
        deltas = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005]
