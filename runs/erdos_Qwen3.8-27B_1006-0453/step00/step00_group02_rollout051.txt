def project_onto_box_sum(h, target_sum):

        else:

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_box_sum(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = tau * corr * dx
    x_max = np.max(x)
    return (x_max + np.log(np.sum(np.exp(x - x_max)))) / tau

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = tau * corr * dx
    x_max = np.max(x)
    p = np.exp(x - x_max) / np.sum(np.exp(x - x_max))
    # grad[m] = dx * (1 - sum_k p[k]*h[m+s_k]_valid - sum_k p[k]*h[m-s_k]_valid)
    grad = np.full(n, dx)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        v = pk * dx
        if s >= 0:
            if n - s > 0:
                grad[:n - s] -= v * h[s:]
                grad[s:] -= v * h[:n - s]
        else:
            t = -s
            if n - t > 0:
                grad[t:] -= v * h[:n - t]
                grad[:n - t] -= v * h[t:]
    return grad

def optimize_tau(h, n, tau, budget_s):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    t0 = time.time()
    
    # Use scipy minimize with SLSQP
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    x0 = h.copy()
    for _ in range(3):  # A few SLSQP calls with increasing tau
        if time.time() - t0 > budget_s:
            break
        res = sp_min(smooth_obj, x0, jac=lambda x: smooth_grad(np.clip(x, 0, 1), tau),
                     bounds=bounds, constraints=cons, method='SLSQP',
                     options={'maxiter': 200, 'ftol': 1e-14})
        x0 = np.clip(res.x, 0, 1)
        x0 = project_onto_box_sum(x0, target)
        c5 = evaluate_c5(x0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x0.copy()
    
    return best_h, best_c5
