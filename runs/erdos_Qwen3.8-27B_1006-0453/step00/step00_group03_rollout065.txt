    f = pos - i0
    return (1.0 - f) * h[i0] + f * h[i0 + 1]

def pgrad(h, t0, budget, max_it=40000, a0=0.01):
    n = len(h)
    bh, bc = h.copy(), compute_c5(h)
    hc, a, st = h.copy(), a0, 0
    for _ in range(max_it):
        if budget - (time.time() - t0) < 10:
            break
        g = compute_c5_grad(hc)
        if np.linalg.norm(g) < 1e-15:
            break
        stp = adjust_sum(hc - a * g)
        cn = compute_c5(stp)
        if cn < bc - 1e-14:
            bc, bh, hc = cn, stp.copy(), stp
            a, st = min(a * 1.3, a0 * 5), 0
        else:
            a, st = a * 0.6, st + 1
            if st > 200:
                hc = adjust_sum(bh + 0.02 * np.random.randn(n))
                a, st = a0, 0
                if np.random.rand() < 0.15:
                    break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    best_c5 = float('inf')
    try:
        h0 = adjust_sum(initial_h_values.copy().astype(float))
    except Exception:
        h0 = adjust_sum(np.random.uniform(0.3, 0.7, 70))
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    n0 = len(h0)
    bounds = [(0.0, 1.0)] * n0
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    # SLSQP at native resolution
    if budget_s - (time.time() - t0) > 60:
        try:
            r = minimize(compute_c5, best_h, method='SLSQP',
                         jac=compute_c5_grad, bounds=bounds, constraints=cons,
                         options={'maxiter': 5000, 'ftol': 1e-14})
            if r.fun < best_c5:
                best_h = adjust_sum(r.x)
                best_c5 = compute_c5(best_h)
        except Exception:
            pass

    # L-BFGS-B with penalty
    if budget_s - (time.time() - t0) > 60:
        try:
            r = minimize(pen_obj, best_h, method='L-BFGS-B', jac=pen_jac,
                         bounds=bounds, args=(1e5,),
                         options={'maxiter': 5000, 'gtol': 1e-10})
            xc = adjust_sum(r.x)
            if compute_c5(xc) < best_c5:
                best_h, best_c5 = xc, compute_c5(xc)
        except Exception:
            pass

    # Projected gradient descent
    if budget_s - (time.time() - t0) > 30:
        hg, cg = pgrad(best_h, t0, budget_s, 40000, 0.01)
        if cg < best_c5:
            best_h, best_c5 = hg, cg

    # Nelder-Mead (derivative-free polish)
    if budget_s - (time.time() - t0) > 50:
        try:
            r = minimize(compute_c5, best_h, method='Nelder-Mead',
                         options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-10})
            xc = adjust_sum(r.x)
            if compute_c5(xc) < best_c5:
                best_h, best_c5 = xc, compute_c5(xc)
        except Exception:
            pass

    # Upsample to 2x and refine
    if budget_s - (time.time() - t0) > 120:
        n2 = n0 * 2
        h2 = adjust_sum(upsample(best_h, 2))
        b2 = [(0.0, 1.0)] * n2
        c2 = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        if budget_s - (time.time() - t0) > 80:
            try:
                r = minimize(compute_c5, h2, method='SLSQP',
                             jac=compute_c5_grad, bounds=b2, constraints=c2,
                             options={'maxiter': 3000, 'ftol': 1e-13})
                if r.fun < best_c5:
                    best_h = adjust_sum(r.x)
                    best_c5 = compute_c5(best_h)
            except Exception:
                pass
        if budget_s - (time.time() - t0) > 60:
            try:
                r = minimize(pen_obj, best_h, method='L-BFGS-B', jac=pen_jac,
                             bounds=b2, args=(1e5,),
                             options={'maxiter': 3000, 'gtol': 1e-10})
                xc = adjust_sum(r.x)
                if compute_c5(xc) < best_c5:
                    best_h, best_c5 = xc, compute_c5(xc)
            except Exception:
                pass
        if budget_s - (time.time() - t0) > 30:
            hg2, cg2 = pgrad(best_h, t0, budget_s, 20000, 0.005)
            if cg2 < best_c5:
                best_h, best_c5 = hg2, cg2

    # Alternative sinusoidal start
    if budget_s - (time.time() - t0) > 120:
        x = np.linspace(0, 2, n0, endpoint=False)
        hs = adjust_sum(0.5 + 0.35 * np.sin(2.0 * np.pi * x))
        try:
            r = minimize(compute_c5, hs, method='SLSQP',
                         jac=compute_c5_grad, bounds=bounds, constraints=cons,
                         options={'maxiter': 3000, 'ftol': 1e-13})
            if r.fun < best_c5:
                best_h = adjust_sum(r.x)
                best_c5 = compute_c5(best_h)
        except Exception:
            pass
        if budget_s - (time.time() - t0) > 30:
            hg3, cg3 = pgrad(best_h, t0, budget_s, 10000, 0.01)
            if cg3 > best_c5:
                pass
            else:
                best_h, best_c5 = hg3, cg3

    # Final polish with small projected gradient
    if budget_s - (time.time() - t0) > 15:
        nf = len(best_h)
        hg4, cg4 = pgrad(best_h, t0, budget_s, 10000, 0.001)
        if cg4 < best_c5:
            best_h, best_c5 = hg4, cg4

    fin = adjust_sum(best_h)
    return fin, compute_c5(fin), len(fin)
