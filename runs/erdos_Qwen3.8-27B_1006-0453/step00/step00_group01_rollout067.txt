    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_c5(h, dx):
    """Compute gradient of C5 = dx * max_s g_s(h).
    Returns gradient w.r.t. h[j] for the argmax shift."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    p = np.argmax(corr)
    s = p - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + s < n:
            g += (1.0 - h[j + s])
        if 0 <= j - s < n:
            g -= h[j - s]
        grad[j] = g
    return grad * dx, s

                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            grad[j] += w[p] * g
    return grad * dx

            break
        progress = it / max(n_iter, 1)

            h = h_new

            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial
            else:

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    
    def obj(x):
        return x[n]
    
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    
    try:

                      options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_softmax(h, dx, beta=30.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        s_lo1 = max(-(n-1), -j)
        s_hi1 = min(n-1, n-1-j)
        if s_lo1 <= s_hi1:
            s1 = np.arange(s_lo1, s_hi1+1)
            grad[j] += np.dot(w[s1+(n-1)], 1.0-h[j+s1])
        s_lo2 = max(-(n-1), j-n+1)
        s_hi2 = min(n-1, j)
        if s_lo2 <= s_hi2:
            s2 = np.arange(s_lo2, s_hi2+1)
            grad[j] -= np.dot(w[s2+(n-1)], h[j-s2])
    return grad * dx

def projected_gradient(h0, dx, target_sum, n_iter=3000, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iter):
        if t_deadline is not None and time.time() > t_deadline:
            break
        beta = 15.0 + 60.0 * (it / max(n_iter, 1))
        lr = 0.04 / (1.0 + 0.005 * it)
        g = grad_softmax(h, dx, beta)
        for shrink in [1.0, 0.5, 0.2, 0.05]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
                break
    return best_h, best_c5

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5
    
    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_sr(h_raw, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    
    def time_ok(margin=8):
        return (time.time() - t0) < budget_s - margin
    
    # Phase 1: SLSQP from initial
    if time_ok():
        h1 = slsqp_solve(h0, dx, target_sum, 300)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    
    # Phase 2: Projected gradient refinement
    if time_ok():
        deadline = t0 + budget_s - 120
        h2, c2 = projected_gradient(best_h, dx, target_sum, 4000, deadline)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
    
    # Phase 3: SLSQP polish
    if time_ok():
        h3 = slsqp_solve(best_h, dx, target_sum, 200)
        c3 = compute_c5(h3, dx)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
    
    # Phase 4: Multi-start with random perturbations
    for r in range(15):
        if not time_ok(30):
            break
        h_r = best_h + 0.08 * np.random.randn(n)
        h_r = project_sr(h_r, target_sum)
        h_r = slsqp_solve(h_r, dx, target_sum, 150)
        c5_r = compute_c5(h_r, dx)
        if c5_r < best_c5 and time_ok(30):
            deadline = t0 + budget_s - 60
            h_r2, c5_r2 = projected_gradient(h_r, dx, target_sum, 2000, deadline)
            if c5_r2 < c5_r:
                c5_r, h_r = c5_r2, h_r2
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
    
    # Phase 5: Random binary starts
    for r in range(8):
        if not time_ok(20):
            break
        h_b = np.zeros(n)
        pos = np.random.permutation(n)
        h_b[pos[:int(target_sum)]] = 1.0
        frac = target_sum - int(target_sum)
        if frac > 0.01:
            h_b[pos[int(target_sum)]] = frac
        h_b = project_sr(h_b, target_sum)
        h_b = slsqp_solve(h_b, dx, target_sum, 100)
        c5_b = compute_c5(h_b, dx)
        if c5_b < best_c5 and time_ok(20):
            deadline = t0 + budget_s - 40
            h_b2, c5_b2 = projected_gradient(h_b, dx, target_sum, 1500, deadline)
            if c5_b2 < c5_b:
                c5_b, h_b = c5_b2, h_b2
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b
    
    # Final polish
    if time_ok(5):
        h_f = slsqp_solve(best_h, dx, target_sum, 100)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f
    
    best_h = project_sr(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
