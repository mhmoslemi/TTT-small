import numpy as np
from scipy.optimize import minimize
import time

def evaluate(h, n_points):
    dx = 2.0 / n_points
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    return float(np.max(corr) * dx)

def logsumexp(y, beta):
    m = np.max(y)
    return m + np.log(np.sum(np.exp(beta * (y - m)))) / beta

def corr_values(h, n_points):
    dx = 2.0 / n_points
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    return corr * dx

def objective(x, n, beta):
    y = corr_values(x, n)
    return logsumexp(y, beta)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        xp = np.clip(x - mid, 0.0, 1.0)
        if np.sum(xp) > target:
            lo = mid
        else:
            hi = mid
    xp = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune to get exact sum
    diff = target - np.sum(xp)
    if abs(diff) > 1e-10:
        # Add/subtract evenly from elements not at boundary
        mask = (xp > 1e-6) & (xp < 1.0 - 1e-6)
        if np.sum(mask) > 0:
            xp[mask] += diff / np.sum(mask)
            xp = np.clip(xp, 0.0, 1.0)
    return xp

def interpolate_to(h_old, n_old, n_new):
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c_best = evaluate(h_best, n_best)
    
    resolutions = [70, 100, 140, 200, 350]
    
    for n in resolutions:
        if time.time() - t_start > budget_s * 0.85:
            break
        
        # Get starting point
        if n == len(h_best):
            x0 = h_best.copy()
        else:
            x0 = interpolate_to(h_best, n_best, n)
            x0 = project_feasible(x0, n)
        
        # Run optimization with decreasing beta
        for beta in [50, 100, 200, 500]:
            if time.time() - t_start > budget_s * 0.9:
                break
            
            bounds = [(0.0, 1.0)] * n
            constraints = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n/2.0,

            res = minimize(objective, x0, args=(n, beta), method='SLSQP',

                c = evaluate(x0, n)
                if c < c_best:

                    h_best = x0.copy()
                    n_best = n

            # Random restart near the current solution

                if c < c_best:
                    c_best = c
                    h_best = x0.copy()
                    n_best = n
    
    return h_best, c_best, n_best
