import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

_g_dx = 0.025
_g_alpha = 100.0
_g_target = 40.0

def obj_smooth(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    v = corr * _g_dx
    m = np.max(v)
    return float(m + (1.0 / _g_alpha) * np.log(np.sum(np.exp(_g_alpha * (v - m)))))

def con_fun(x):
    return float(np.sum(x) - _g_target)

def con_jac(x):
    return np.ones(len(x))

def enforce(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] += d / c
        else:
            m = h > 0.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] += d / c
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14 and h.size > 0:
        i = int(np.argmin(np.abs(h - 0.5)))
        h[i] = float(np.clip(h[i] + d, 0.0, 1.0))
    return h
