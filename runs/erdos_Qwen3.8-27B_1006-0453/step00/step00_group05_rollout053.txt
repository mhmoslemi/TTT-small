    # Phase 1: Optimize parent at its own resolution
    n = best_n
    h_opt, c5_opt = _slsqp(best_h, n, maxiter=1500)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Multiple diverse restarts at parent resolution
    max_trials = 40
    for trial in range(max_trials):
        if tl() < 150:
            break
        tmod = trial % 5
        if tmod == 0:
            h_try = _perturb(best_h, 0.06, rng)
        elif tmod == 1:
            h_try = _random_binary(n, rng)
        elif tmod == 2:
            h_try = _golden_ratio_init(n, rng)
        elif tmod == 3:
            h_try = _structured_init(n, rng)
        else:
            h_try = _perturb(best_h, 0.15, rng)
        mi = 400 if trial < 10 else 250
        h_opt, c5_opt = _slsqp(h_try, n, maxiter=mi)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Higher resolutions
    for target_n in [140, 280, 560]:
        if tl() < 80:
            break
        h_up = _upsample(best_h, target_n)
        mi_main = 400 if target_n <= 140 else 250 if target_n <= 280 else 120
        h_opt, c5_opt = _slsqp(h_up, target_n, maxiter=mi_main)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = target_n

        # Restart at this resolution
        n_restarts = 6 if target_n <= 140 else 4 if target_n <= 280 else 2
        for trial in range(n_restarts):
            if tl() < 80:
                break
            tmod = trial % 3
            if tmod == 0:
                h_try = _perturb(h_up, 0.04, rng)
            elif tmod == 1:
                h_try = _structured_init(target_n, rng)
            else:
                h_try = _random_binary(target_n, rng)
            mi_r = mi_main // 2 + 50
            h_opt, c5_opt = _slsqp(h_try, target_n, maxiter=mi_r)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = target_n

    # Phase 4: Final polish with best solution
    if tl() > 60:
        h_opt, c5_opt = _slsqp(best_h, best_n, maxiter=2000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Ensure validity
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_n / 2.0
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-8:
        best_h += diff / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = _eval_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
