            try:
                h_o, c5_o = pgd_optimize(best_h, len(best_h), steps=5000, time_limit=8.0)
                if c5_o < best_c5:
                    best_c5, best_h = c5_o, h_o
            except Exception:
                pass

    return (best_h, best_c5, len(best_h))
