import numpy as np
import time

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h = h.copy()
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def transfer_move(h, i, j, delta):
    h_new = h.copy()
    if h_new[i] + (-delta) < 0 or h_new[i] + (-delta) > 1:
        return None
    if h_new[j] + delta < 0 or h_new[j] + delta > 1:
        return None
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def local_search(h, target_sum, max_iters=100000, budget_s=200, seed=42):
    rng = np.random.default_rng(seed)
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_val = eval_c5(h)
    current_val = best_val
    start = time.time()
    T_init, T_end = 0.004, 0.00005
    step = 0.04

    for it in range(max_iters):
        if time.time() - start > budget_s:
            break
        progress = min(it / max_iters, 1.0)
        T = T_init * (T_end / T_init) ** progress

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(-step, step)
        h_new = transfer_move(h, i, j, delta)
        if h_new is None:
            continue
        val = eval_c5(h_new)
        df = val - current_val
        if df < 0 or rng.random() < np.exp(-df / T):
            h = h_new
            current_val = val
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val

def pso_optimize(h0, target_sum, n_particles=25, n_iter=400, budget_s=200, seed=42):
    rng = np.random.default_rng(seed)
    n = len(h0)
    particles = np.zeros((n_particles, n))
    velocities = np.zeros((n_particles, n))
    pbest = np.zeros((n_particles, n))
    pbest_val = np.full(n_particles, np.inf)

    for i in range(n_particles):
        if i == 0:
            particles[i] = h0.copy()
        else:
            particles[i] = h0 + rng.normal(0, 0.12, n)
        particles[i] = repair(particles[i], target_sum)
        pbest[i] = particles[i].copy()
        pbest_val[i] = eval_c5(particles[i])

    gbest_idx = int(np.argmin(pbest_val))
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_val[gbest_idx]

    start = time.time()
    w_init, w_end = 0.85, 0.35
    c1, c2 = 1.49, 1.49

    for it in range(n_iter):
        if time.time() - start > budget_s:
            break
        w = w_init + (w_end - w_init) * it / max(n_iter, 1)
        for i in range(n_particles):
            r1 = rng.random(n)
            r2 = rng.random(n)
            velocities[i] = (w * velocities[i]
                             + c1 * r1 * (pbest[i] - particles[i])
                             + c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -0.4, 0.4)
            particles[i] = particles[i] + velocities[i]
            particles[i] = repair(particles[i], target_sum)
            val = eval_c5(particles[i])
            if val < pbest_val[i]:
                pbest_val[i] = val
                pbest[i] = particles[i].copy()
            if val < gbest_val:
                gbest_val = val
                gbest = particles[i].copy()
    return gbest, gbest_val
