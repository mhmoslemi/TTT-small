import numpy as np
import scipy.optimize as opt
import time

_g_n = 70
_g_pen = 1000.0


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-16) if diff > 0 else (h > 1e-16)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _solslsqp_obj(x):
    return compute_c5(x, _g_n)


def _solslsqp_con(x):
    return np.sum(x) - float(_g_n) / 2.0


def _pen_obj(x):
    return compute_c5(x, _g_n) + _g_pen * (np.sum(x) - float(_g_n) / 2.0) ** 2


def _interp(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

    global _g_n
    _g_n = n

    res = opt.minimize(_solslsqp_obj, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    return res
