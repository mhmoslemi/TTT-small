```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _softmax(vals, alpha):
    m = vals.max()
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def _fix(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
            c = mask.sum()
        else:
            mask = h > 0.0
            c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    sm = _softmax(corr, alpha)
    pen = 1e3 * (h.sum() - n / 2.0) ** 2
    return sm + pen

def _scipy(h_start, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    def f(h):
        return _obj(h, n, alpha)
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return _fix(res.x, n)

def _local_search(h_best, c5_best, n, rng, deadline, min_step=0.001):
    for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > deadline:
            break
        if step < min_step:
            break
        stall = 0
        max_stall = 2000
        while time.time() < deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = (rng.random() - 0.5) * 2.0 * step
            ht = h_best.copy()
            ht[i] += delta
            ht[j] -= delta
            if ht[i] < 0 or ht[i] > 1 or ht[j] < 0 or ht[j] > 1:
                stall += 1
            else:
                c = _c5(ht)
                if c < c5_best:
                    h_best, c5_best = ht, c
                    stall = 0
                else:
                    stall += 1
            if stall > max_stall:
                break
    return h_best, c5_best

def _subgradient(h_best, c5_best, n, rng, deadline):
    step = 0.02
    for it in range(50000):
        if time.time() > deadline:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full') * (2.0 / n)
        idx = np.argmax(corr)
        s = idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            g1 = 0.0
            g2 = 0.0
            if 0 <= i + s <= n - 1:
                g1 = (1.0 - h_best[i + s])
            if 0 <= i - s <= n - 1:
                g2 = h_best[i - s]
            grad[i] = (2.0 / n) * (g1 - g2)
        # Project gradient onto sum=0 subspace and feasible directions
        grad -= grad.mean()
        proj = -step * grad / (np.linalg.norm(grad) + 1e-12)
        ht = h_best + proj
        if ht.min() < 0 or ht.max() > 1:
            proj *= 0.5
            ht = h_best + proj
            if ht.min() < 0 or ht.max() > 1:
                continue
        c = _c5(ht)
        if c < c5_best:
            h_best, c5_best = ht.copy(), c
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.95
    return h_best, c5_best
```
