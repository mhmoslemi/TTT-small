    h = x[:n]

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return t - corr

    return x[-1]

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        amount = diff / np.sum(mask)
        amount = min(amount, 1.0 - h[mask]).max() if diff > 0 else max(amount, -(h[mask])).min()
        # Simpler: just distribute
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_descent(h, n, target_sum, n_iters, rng):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    eps_list = [0.01, 0.005, 0.001, 0.0005]
    
    for it in range(n_iters):

        for eps in eps_list:

            j = rng.randint(n)

                    improved = True
                else:
                    h[i] += eps

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def eq_constraint(x, n, target_sum):
    return np.sum(x[:n]) - target_sum

def ineq_constraint(x, n, dx):
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return t - corr

def objective_fn(x):
    return x[-1]

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        nm = int(np.sum(mask))
        if nm == 0:
            break
        h[mask] += diff / nm
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_solve(x0, n, dx, target_sum, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_constraint, 'args': (n, target_sum)},
        {'type': 'ineq', 'fun': ineq_constraint, 'args': (n, dx)}
    ]
    try:
        res = minimize(objective_fn, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair_sum(res.x[:n], target_sum)
    except Exception:
        pass
    return None

def coord_descent(h, n, target_sum, rng, n_rounds):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    eps_list = [0.02, 0.01, 0.005, 0.002, 0.001]
    for rnd in range(n_rounds):
        improved = False
        for _ in range(20):
            eps = eps_list[rng.randint(len(eps_list))]
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for di, dj in [(1, -1), (-1, 1)]:
                hi2, hj2 = h[i] + di * eps, h[j] + dj * eps
                if hi2 < -1e-10 or hi2 > 1.0 + 1e-10 or hj2 < -1e-10 or hj2 > 1.0 + 1e-10:
                    continue
                h[i] = hi2
                h[j] = hj2
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    improved = True
                else:
                    h[i] -= di * eps
                    h[j] -= dj * eps
        if not improved and (rnd + 1) % 50 == 0:
            break
    return h, best_c5
