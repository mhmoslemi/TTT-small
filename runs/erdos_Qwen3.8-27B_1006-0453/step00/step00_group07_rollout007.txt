        if c5 < best_c5:
            best_h = cand
            best_c5 = c5

    # Phase 4: More local search
    dl2 = min(t0 + budget_s * 0.95, time.time() + 60)
    best_h, best_c5 = pair_local_search(best_h, dx, target_sum, dl2, n_trials=6000)

    # Phase 5: Try perturbed restarts
    if time.time() - t0 < budget_s * 0.92:
        best_h, best_c5 = pair_local_search(best_h, dx, target_sum,
                                            min(t0 + budget_s * 0.97, time.time() + 30),
                                            n_trials=3000)

    # Final validation
    best_h = enforce_sum(best_h, target_sum)
    best_c5, _ = compute_c5(best_h, dx)

    return best_h, best_c5, n
