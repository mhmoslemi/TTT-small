import numpy as np
from scipy import optimize
import time

_g_dx = 1.0
_g_ts = 0.0

def _pobj(h):
    c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * _g_dx
    return c5 + 1e8 * (np.sum(h) - _g_ts) ** 2

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _fixsum(h, tgt):
    h = h.astype(np.float64).copy()
    for _ in range(3000):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        i = int(np.argmin(np.minimum(h, 1.0 - h)))
        h[i] = float(np.clip(h[i] + d, 0.0, 1.0))
    return h

def _mkbin(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=np.float64)
    hb[idx[:n // 2]] = 1.0
    return hb

def _slsqp(h, dx, tgt, mi=400):
    global _g_dx, _g_ts
    _g_dx, _g_ts = dx, tgt
    bd = [(0.0, 1.0)] * len(h)
    try:
        r = optimize.minimize(_pobj, h, method='SLSQP', bounds=bd,
                              options={'maxiter': mi, 'ftol': 1e-14})
        out = _fixsum(r.x, tgt)
        return out
    except Exception:
        return _fixsum(h.copy(), tgt)
