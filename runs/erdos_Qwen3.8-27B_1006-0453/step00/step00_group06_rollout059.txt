    jdx = idx + L
    grad[jdx] += h[idx]

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = proj < 1.0 - 1e-12
        else:
            mask = proj > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
    return np.clip(proj, 0.0, 1.0)

def eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_corr_k(h, n, k):
    grad = np.zeros(n)
    i_min = max(0, -k)
    i_max = min(n - 1, n - 1 - k)
    if i_min > i_max:
        return grad
    idx = np.arange(i_min, i_max + 1)
    j_idx = idx + k
    grad[j_idx] += (1.0 - h[idx])
    grad[idx] -= h[idx + k]
    return grad

def subgrad_step(h, n, step, smooth_beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_top = min(5, 2 * n - 1)
    top_idx = np.argsort(corr)[-n_top:]
    shift = corr[top_idx] - corr.max() + 1e-12
    w = np.exp(smooth_beta * shift)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(top_idx)):
        g += w[ii] * grad_corr_k(h, n, int(top_idx[ii]))
    return h - step * g, corr.max() * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    global initial_h_values
    h_init = np.array(initial_h_values, dtype=float)
    n = int(len(h_init))

    best_h = project_feasible(h_init.copy(), n)
    best_c5 = eval_c5(best_h, n)

    h = best_h.copy()
    iter_count = 0
    patience = 0
    max_patience = 400
    beta = 3.0

    while time.time() < deadline - 30:
        iter_count += 1
        step = 0.025 / (1.0 + iter_count / 600.0)
        step = max(step, 0.003)

        h_new, _ = subgrad_step(h, n, step, beta)
        h_new = project_feasible(h_new, n)
        c5_new = eval_c5(h_new, n)

        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new
            patience += 1

        if patience > max_patience:
            h = best_h.copy() + 0.025 * np.random.randn(n)
            h = project_feasible(h, n)
            patience = 0
            iter_count = 0
            beta = np.random.uniform(1.0, 6.0)

        if patience > 0 and patience % 200 == 0:
            h = best_h.copy() + 0.015 * np.random.randn(n)
            h = project_feasible(h, n)
            patience = 0

    # Local coordinate search
    while time.time() < deadline:
        h_try = best_h.copy()
        nc = np.random.randint(1, 5)
        ci = np.random.choice(n, nc, replace=False)
        h_try[ci] += np.random.randn(nc) * 0.008
        h_try = project_feasible(h_try, n)
        c5_try = eval_c5(h_try, n)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    final_c5 = eval_c5(best_h, n)
    return best_h, final_c5, n
