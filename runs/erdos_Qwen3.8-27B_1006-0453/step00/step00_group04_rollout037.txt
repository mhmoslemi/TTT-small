import numpy as np
import time

def evaluate_c5(h):
    """Compute C5 value for binary h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    return np.max(corr) * dx

def scale_to_length(h_orig, n_new, rng):
    """Scale h from len(h_orig) to n_new maintaining sum = n_new//2."""
    n_orig = len(h_orig)
    # Use nearest-neighbor scaling
    h_new = np.zeros(n_new)
    for i in range(n_new):
        idx = int(i * n_orig / n_new)
        idx = min(idx, n_orig - 1)
        h_new[i] = h_orig[idx]
    # Fix sum
    current_sum = int(np.sum(h_new))
    target_sum = n_new // 2
    if current_sum > target_sum:
        # Flip some 1s to 0s
        one_indices = np.where(h_new == 1)[0]
        to_flip = rng.choice(one_indices, current_sum - target_sum, replace=False)
        h_new[to_flip] = 0
    elif current_sum < target_sum:
        # Flip some 0s to 1s
        zero_indices = np.where(h_new == 0)[0]
        to_flip = rng.choice(zero_indices, target_sum - current_sum, replace=False)
        h_new[to_flip] = 1
    return h_new

def thue_morse(n):
    """Generate Thue-Morse-like balanced sequence of length n."""
    # Generate basic Thue-Morse
    tm = [0]
    for _ in range(int(np.log2(n)) + 2):
        tm = tm + [1 - x for x in tm]
    tm = np.array(tm[:n])
    # Ensure exactly n//2 ones
    ones_needed = n // 2 - int(np.sum(tm))
    if ones_needed > 0:
        zeros = np.where(tm == 0)[0]
        flips = np.random.default_rng(0).choice(zeros, ones_needed, replace=False)
        tm[flips] = 1
    elif ones_needed < 0:
        ones = np.where(tm == 1)[0]
        flips = np.random.default_rng(0).choice(ones, -ones_needed, replace=False)
        tm[flips] = 0
    return tm.astype(float)

    t_end = start_time + time_budget
    temperature = 0.01

import numpy as np
import time


def evaluate_c5(h):
    h = np.asarray(h, dtype=np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def scale_to_length(h_orig, n_new, rng):
    n_orig = len(h_orig)
    if n_orig == n_new:
        return h_orig.copy()
    h_new = np.zeros(n_new, dtype=np.float64)
    for i in range(n_new):
        src = min(int(round(i * n_orig / n_new)), n_orig - 1)
        h_new[i] = h_orig[src]
    h_new = (h_new > 0.5).astype(np.float64)
    cur = int(h_new.sum())
    tgt = n_new // 2
    if cur > tgt:
        idx = np.where(h_new == 1)[0]
        sel = rng.choice(idx, cur - tgt, replace=False)
        h_new[sel] = 0
    elif cur < tgt:
        idx = np.where(h_new == 0)[0]
        sel = rng.choice(idx, tgt - cur, replace=False)
        h_new[sel] = 1
    return h_new


def thue_morse_balance(n):

    tgt = n // 2
    if cur < tgt:
        zeros = np.where(tm == 0)[0]
        tm[zeros[:tgt - cur]] = 1
    elif cur > tgt:
        ones = np.where(tm == 1)[0]
        tm[ones[:cur - tgt]] = 0
    return tm.astype(np.float64)


def random_balanced(n, rng):
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h


def block_pattern(n, block_size, rng):
    h = np.zeros(n, dtype=np.float64)
    pattern = np.zeros(block_size, dtype=np.float64)
    ones_per_block = block_size // 2
    pattern[:ones_per_block] = 1
    n_full = n // block_size
    h[:n_full * block_size] = np.tile(pattern, n_full)
    if n % block_size > 0:
        rem = n % block_size
        h[n_full * block_size:] = pattern[:rem]
    h = (h > 0.5).astype(np.float64)
    cur = int(h.sum())
    tgt = n // 2
    if cur > tgt:
        idx = np.where(h == 1)[0]
        sel = rng.choice(idx, cur - tgt, replace=False)
        h[sel] = 0
    elif cur < tgt:
        idx = np.where(h == 0)[0]
        sel = rng.choice(idx, tgt - cur, replace=False)
        h[sel] = 1
    return h


def optimize_h(h_init, n, rng, time_budget):
    h = h_init.copy()
    c5_cur = evaluate_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    t_end = time.time() + time_budget
    temp = 0.006
    min_temp = 0.0002
    no_improve = 0
    patience = 4000
    iter_count = 0
    while time.time() < t_end:
        iter_count += 1
        no_improve += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = evaluate_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
                no_improve = 0
        elif temp > min_temp:
            delta = c5_new - c5_cur
            if delta < 0.08:
                if rng.random() < np.exp(-delta / temp):
                    c5_cur = c5_new
                    if c5_new < c5_best:
                        c5_best = c5_new
                        h_best = h.copy()
                        no_improve = 0
                else:
                    h[i], h[j] = h[j], h[i]
            else:
                h[i], h[j] = h[j], h[i]
        else:
            h[i], h[j] = h[j], h[i]
        if iter_count % 2000 == 0:
            temp = max(min_temp, temp * 0.995)
        if no_improve > patience:
            temp = max(temp, 0.004)
            no_improve = 0
    return h_best, c5_best


def greedy_refine(h, n, rng, time_budget):
    h = h.copy()
    c5_cur = evaluate_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    t_end = time.time() + time_budget
    no_improve = 0
    while time.time() < t_end:
        no_improve += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = evaluate_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
            no_improve = 0
        else:
            h[i], h[j] = h[j], h[i]
        if no_improve > 3000:
            break
    return h_best, c5_best


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    h_parent = None
    try:
        h_parent = np.asarray(initial_h_values, dtype=np.float64)
        h_parent = (h_parent > 0.5).astype(np.float64)
    except Exception:
        h_parent = None
    best_c5 = 0.5
    best_h = None
    best_n = 0
    if h_parent is not None:
        pc5 = evaluate_c5(h_parent)
        best_c5 = pc5
        best_h = h_parent.copy()
        best_n = len(h_parent)
    target_ns = [len(h_parent)] if h_parent is not None else []
    for n in [42, 60, 84, 100, 120, 144, 168, 200, 240, 280, 320, 360, 400, 480, 560]:
        if n not in target_ns:
            target_ns.append(n)
    target_ns = sorted(set(n for n in target_ns if n % 2 == 0))
    reserved = 40.0
    usable = max(60.0, budget_s - reserved)
    n_configs = 0
    for n in target_ns:
        cnt = 0
        if h_parent is not None:
            cnt += 1
        cnt += 1
        if n <= 300:
            cnt += 1
        n_configs += cnt
    per_config = usable / max(1, n_configs)
    for n in target_ns:
        remaining = budget_s - (time.time() - t0) - 12
        if remaining < 5:
            break
        starts = []
        if h_parent is not None:
            starts.append(scale_to_length(h_parent, n, rng))
        starts.append(thue_morse_balance(n))
        if n <= 300:
            starts.append(block_pattern(n, max(4, n // 48), rng))
        tb_per = max(2.0, min(per_config, remaining / max(1, len(starts))))
        for h_init in starts:
            remaining = budget_s - (time.time() - t0) - 10
            if remaining < 3:
                break
            tb = max(2.0, min(tb_per, remaining))
            h_opt, c5_opt = optimize_h(h_init, n, rng, tb)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
    remaining = budget_s - (time.time() - t0) - 2
    if remaining > 10 and best_h is not None:
        h_ref, c5_ref = greedy_refine(best_h, best_n, rng, remaining)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
    if best_h is None:
        n = 42
        best_h = random_balanced(n, rng)
        best_n = n
        best_c5 = evaluate_c5(best_h)
    assert abs(best_h.sum() * (2.0 / best_n) - 1.0) < 1e-6
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    return (best_h, best_c5, best_n)
