                    best_c5 = c_new2
                else:
                    best_h[i] = vi
                    best_h[j] = vj
            else:
                best_h[i] = vi
                best_h[j] = vj

    # Phase 4: Try a few different n values if time allows
    elapsed = time.time() - t0
    if elapsed < budget_s - 120:
        for n_try in [44, 46, 48, 50, 40, 42, 52]:
            if time.time() - t0 > budget_s - 60:
                break
            dx_try = 2.0 / n_try
            tgt = n_try / 2.0
            bnds = [(0.0, 1.0)] * n_try
            # Interpolate from best_h to n_try points
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h, left=0.0, right=0.0)
            h_interp = repair_sum(h_interp, n_try)

            def obj_try(x):
                return compute_c5(x, dx_try)

            cons_try = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt}]
            try:
                res = minimize(obj_try, h_interp, method='SLSQP', bounds=bnds,
                               constraints=cons_try, options={'maxiter': 800, 'ftol': 1e-14})
                if res.x is not None:
                    cand = repair_sum(res.x, n_try)
                    c = compute_c5(cand, dx_try)
                    if c < best_c5:
                        best_h, best_c5, n = cand, c, n_try
                        dx = dx_try
                        target_sum = tgt
                        bounds = bnds
            except Exception:
                pass
            # CD polish on new n
            if n != n_try:
                continue
            for _ in range(30000):
                if time.time() - t0 > budget_s - 30:
                    break
                ii = rng.randint(n)
                jj = rng.randint(n)
                if ii == jj:
                    continue
                dd = float(deltas[rng.randint(len(deltas))])
                vi2, vj2 = best_h[ii], best_h[jj]
                if vi2+dd < 0 or vi2+dd > 1 or vj2-dd < 0 or vj2-dd > 1:
                    continue
                best_h[ii] = vi2 + dd
                best_h[jj] = vj2 - dd
                c_new = compute_c5(best_h, dx)
                if c_new < best_c5 - 1e-15:
                    best_c5 = c_new
                else:
                    best_h[ii] = vi2
                    best_h[jj] = vj2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    n_final = len(best_h)

    return (best_h, best_c5, n_final)
