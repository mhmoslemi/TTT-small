def project_feasible(h, target_sum):
    """Project h onto {0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(h)
    # Bisection on threshold
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0, 1)

def soft_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    f = c * dx
    f_max = np.max(f)
    scaled = beta * (f - f_max)
    exp_f = np.exp(scaled)
    sum_exp = np.sum(exp_f)
    p = exp_f / sum_exp
    obj = f_max + np.log(sum_exp) / beta

        else:
            nn = -j
            grad[nn:] += coeff * (1 - h[:nn])
            grad[:n-nn] -= coeff * h[nn:]
    return obj, grad

            result = np.clip(result, 0.0, 1.0)
    return result

def soft_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    f = c * dx
    f_max = np.max(f)
    scaled = beta * (f - f_max)
    exp_f = np.exp(scaled)
    sum_exp = np.sum(exp_f)
    p = exp_f / sum_exp
    obj = f_max + np.log(sum_exp) / beta
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pk = p[m]
        if pk < 1e-14:
            continue
        j = m - (n - 1)
        coeff = pk * dx
        if j >= 0:
            grad[:n - j] += coeff * (1.0 - h[j:])
            grad[j:] -= coeff * h[:n - j]
        else:
            nn = -j
            grad[nn:] += coeff * (1.0 - h[:nn])
            grad[:n - nn] -= coeff * h[nn:]
    return obj, grad

def optimize_projection(h0, n, beta, max_iter, time_limit):
    h = h0.copy()
    h = project_feasible(h, n / 2.0)
    best_h = h.copy()
    best_obj = soft_max_obj_grad(h, beta)[0]
    lr = 0.02
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        obj, grad = soft_max_obj_grad(h, beta)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        # Scale gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad = grad / gnorm
        # Adaptive step
        for _ in range(20):
            h_new = h - lr * grad * gnorm
            h_new = project_feasible(h_new, n / 2.0)
            obj_new, _ = soft_max_obj_grad(h_new, beta)
            if obj_new < obj:
                h = h_new
                break
            lr *= 0.5
        else:
            lr = max(lr, 0.001)
            break
        lr = min(lr * 1.1, 0.5)
    return best_h, best_obj

def binarize(h, n):
    """Convert to binary with exactly n/2 ones"""
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def local_search_binary(h_bin, n, max_flips, time_limit):
    """Try local flips on binary solution"""
    h = h_bin.copy()
    best = evaluate_erdos(h)
    start = time.time()
    improved = True
    while improved and time.time() - start < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            for j in range(n):
                if j == i:
                    continue
                # Flip i from 1 to 0 and j from 0 to 1 (or vice versa)
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
                    c5 = evaluate_erdos(h)
                    if c5 < best - 1e-8:
                        best = c5
                        improved = True
                    else:
                        h[i], h[j] = h[j], h[i]
    return h, best
