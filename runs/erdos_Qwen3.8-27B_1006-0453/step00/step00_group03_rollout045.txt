import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_max(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(x, n, alpha=8.0):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def make_feasible(x, n, target):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(200):
        adj = (target - np.sum(x)) / n
        x = x + adj
        x = np.clip(x, 0.0, 1.0)
        if abs(np.sum(x) - target) < 1e-14:
            break
    return x

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    target = n / 2.0

    h0 = make_feasible(h0, n, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]

    # Stage 1: SLSQP from parent with direct max
    try:
        res = minimize(obj_max, h0, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-15})
        if res.x is not None:
            c = compute_c5(res.x)
            if c < best_c5:
                best_h, best_c5 = res.x.copy(), c
    except Exception:
        pass

    # Stage 2: SLSQP with smooth objective (power mean approx of max)
    if time.time() - t0 < budget_s - 20:
        try:
            res = minimize(obj_smooth, best_h, args=(n, 10.0), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            if res.x is not None:
                c = compute_c5(res.x)
                if c < best_c5:
                    best_h, best_c5 = res.x.copy(), c
        except Exception:
            pass

    # Stage 3: Multiple perturbation + SLSQP restarts
    n_perts = 40
    for i in range(n_perts):
        if time.time() - t0 > budget_s - 10:
            break
        scale = 0.015 * (1.0 + 0.5 * np.sin(i * 0.3))
        pert = best_h + scale * (np.random.rand(n) - 0.5)
        pert = make_feasible(pert, n, target)
        try:
            res = minimize(obj_max, pert, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-13})
            if res.x is not None:
                c = compute_c5(res.x)
                if c < best_c5:
                    best_h, best_c5 = res.x.copy(), c
        except Exception:
            pass

    # Stage 4: trust-constr refinement
    if time.time() - t0 < budget_s - 15:
        try:
            from scipy.optimize import LinearConstraint
            A = np.ones((1, n))
            lc = LinearConstraint(A, target, target)

                           bounds=bounds, constraints=[lc],

    return best_h, best_c5, n
