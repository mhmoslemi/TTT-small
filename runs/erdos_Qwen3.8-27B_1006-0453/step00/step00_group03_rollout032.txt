    h = np.clip(h.ravel().copy(), 0.0, 1.0)
    tgt = 0.5 * n
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        m = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        if m.any():
            h[m] += d / m.sum()
            h = np.clip(h, 0, 1)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        i = int(np.argmin(np.abs(h - 0.5)))
        h[i] = min(1.0, max(0.0, h[i] + d))
    return h

def gradient_worst(h, n, dx):
    shifts = get_shifts(h, dx)
    worst_k = int(np.argmax(shifts))
    grad = np.zeros(n)
    if worst_k < n:
        k = worst_k
        for m in range(k, n):
            grad[m] += dx * (1.0 - h[m - k])
        for m in range(0, n - k):
            grad[m] -= dx * h[m + k]
    else:
        j = worst_k - n + 1
        for m in range(j, n):
            grad[m] += dx * (1.0 - h[m - j])
        for m in range(0, n - j):
            grad[m] -= dx * h[m + j]
    active_mask = shifts >= shifts.max() - dx * 0.01
    return grad, worst_k, active_mask

            t6 = min(120, rem2 * 0.4)
            h6, c6 = pair_search(best_h, n1, dx1, t6, seed + 200)
            if c6 < best_c:
                best_h, best_c = h6, c6

        # Try scipy at n1
        rem3 = budget_s - (time.time() - t_start) - 30
        if rem3 > 50:
            h7, c7 = smooth_scipy(best_h, n1, dx1, min(80, rem3 * 0.4), alpha=50)
            if c7 < best_c:
                best_h, best_c = h7, c7

    # Phase 4: Final refinement at best resolution
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    rem = budget_s - (time.time() - t_start) - 5
    if rem > 20:
        hf, cf = guided_search(best_h, n_best, dx_best, min(80, rem), seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf
        rem2 = budget_s - (time.time() - t_start) - 3
        if rem2 > 15:
            hp, cp = pair_search(best_h, n_best, dx_best, min(40, rem2), seed + 1234)
            if cp < best_c:
                best_h, best_c = hp, cp

    # Final projection
    nf = len(best_h)
    best_h = project(best_h, nf)
    final_c = compute_c5(best_h, 2.0 / nf)

    return best_h, final_c, nf
