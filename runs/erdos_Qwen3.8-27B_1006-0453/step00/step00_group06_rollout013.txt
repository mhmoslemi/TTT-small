            step_size = max(0.05, step_size * 0.85)

        # Periodic binary local search
        if no_improve % 5 == 0 and time.time() < deadline - 30:
            h_bin = np.zeros(n)
            h_bin[np.argsort(h_try2)[::-1][: n // 2]] = 1.0
            h_bin2, c5_bin = _local_search(h_bin.copy(), n, rng, max_rounds=30)
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin2.copy(), c5_bin
                no_improve = 0

    return (best_h, best_c5, n)
