import numpy as np


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _fix_sum(h, target):
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = np.maximum(1.0 - h, 0.0)
        tr = np.sum(room)
        if tr < 1e-14:
            return h
        h = h + (diff / tr) * room
    else:
        tot = np.sum(np.maximum(h, 0.0))
        if tot < 1e-14:
            return h
        h = h * max(1.0 + diff / tot, 0.0)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = max(0.0, min(1.0, h[idx] + diff2))
    return h


def _sa_perturb(h, n, rng, scale):
    r = rng.random()
    h_new = h.copy()
    if r < 0.5:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            j = (j + 1) % n
        d = rng.uniform(-scale, scale)
        h_new[i] += d
        h_new[j] -= d
    elif r < 0.8:
        k = int(rng.integers(2, min(7, n)))
        idx = rng.choice(n, size=k, replace=False)
        deltas = rng.normal(0.0, scale * 0.5, size=k)
        deltas -= np.mean(deltas)
        h_new[idx] += deltas
    else:
        blen = int(rng.integers(2, min(6, n)))
        bstart = int(rng.integers(max(1, n - blen + 1)))
        block = rng.normal(0.0, scale * 0.35, size=blen)
        block -= np.mean(block)
        h_new[bstart:bstart + blen] += block
    np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new
