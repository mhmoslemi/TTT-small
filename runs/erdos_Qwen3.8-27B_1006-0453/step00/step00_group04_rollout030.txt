```python
import numpy as np
import time

def scale_to_size(h_init, target_n):
    """Scale h_init to target_n using nearest-neighbor interpolation."""
    n_init = len(h_init)
    if target_n == n_init:
        return h_init.copy()
    # Create indices for interpolation
    indices = np.linspace(0, n_init - 1, target_n, endpoint=False)
    h_scaled = h_init[np.floor(indices).astype(int)]
    # Fix count to be exactly target_n/2
    target_sum = target_n // 2
    current_sum = np.sum(h_scaled)
    if current_sum > target_sum:
        # Flip some 1s to 0s
        ones = np.where(h_scaled == 1)[0]
        n_flip = int(current_sum - target_sum)
        flip_idx = np.random.choice(ones, n_flip, replace=False)
        h_scaled[flip_idx] = 0
    elif current_sum < target_sum:
        zeros = np.where(h_scaled == 0)[0]
        n_flip = int(target_sum - current_sum)
        flip_idx = np.random.choice(zeros, n_flip, replace=False)
        h_scaled[flip_idx] = 1
    return h_scaled

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return np.max(corr) * dx

def local_search(h, n, rng, max_iter=10000):
    """Local search by swapping pairs of bits."""
    current = h.copy().astype(float)
    current_c5 = eval_c5(current)
    
    ones = np.where(current == 1)[0]
    zeros = np.where(current == 0)[0]
    
    no_improve = 0
    for it in range(max_iter):
        # Random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        if i == j:
            continue
        
        current[i] = 0
        current[j] = 1
        
        new_c5 = eval_c5(current)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            no_improve = 0
            # Update tracking
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
        else:
            current[i] = 1
            current[j] = 0
            no_improve += 1
    
    return current, current_c5
```

Let me write the efficient delta computation:

When swapping h[i] from 1 to 0 and h[j] from 0 to 1:
Δh = -1 at i, +1 at j
Δ(1-h) = +1 at i, -1 at j

c[k] = sum_n h[n+k] * (1-h[n])

Δc[k] = sum_n [(-1)·δ(n+k=i)·(1-h[n]) + (1)·δ(n+k=j)·(1-h[n]) + h[n+k]·(1)·δ(n=i) + h[n+k]·(-1)·δ(n=j)]

= -(1-h[i-k]) + (1-h[j-k]) + h[k-i] - h[k-j]

(where we interpret h[m] = 0 if m is out of [0, n-1], and 1-h[m] = 0 for out-of-range... actually no, the sum is only over valid n, so out-of-range terms are just 0)

More carefully:
- Term 1: -1·(1-h[n]) where n+k=i, so n=i-k. Valid if 0≤i-k≤n-1. Value: -(1-h[i-k]) = -(1-h[i-k])
- Term 2: +1·(1-h[n]) where n+k=j, so n=j-k. Valid if 0≤j-k≤n-1. Value: (1-h[j-k])
- Term 3: +h[n+k]·1 where n=i, so n+k=i+k. Valid if 0≤i+k≤n-1. Value: h[i+k]
- Term 4: -h[n+k]·1 where n=j, so n+k=j+k. Valid if 0≤j+k≤n-1. Value: -h[j+k]

So Δc[k] = -(1-h[i-k])·[0≤i-k≤n-1] + (1-h[j-k])·[0≤j-k≤n-1] + h[i+k]·[0≤i+k≤n-1] - h[j+k]·[0≤j+k≤n-1]

For binary h:
- (1-h[i-k]) is 1 iff h[i-k]=0, i.e., i-k is a zero position
- h[i+k] is 1 iff i+k is a one position

This is O(1) per k, and there are 2n-1 k values. So computing all Δc[k] is O(n). Then checking if max(c+Δc) < c_max is O(n).

Total per swap: O(n). Number of candidate swaps to try: we can try all (up to (n/2)² ≈ n²/4) or a random subset.

```python
import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), (1.0 - h).astype(np.float64), mode='full')
    return np.max(corr) * dx

def scale_to_size(h_src, n_target, rng):
    n_src = len(h_src)
    if n_target == n_src:
        return h_src.copy()
    # Nearest neighbor scaling
    indices = np.arange(n_target) * n_src / n_target
    h = h_src[np.floor(indices).astype(int)]
    # Fix to have exactly n_target/2 ones
    target = n_target // 2
    current = int(np.sum(h))
    if current > target:
        ones = np.where(h == 1)[0]
        to_flip = rng.choice(ones, current - target, replace=False)
        h[to_flip] = 0
    elif current < target:
        zeros = np.where(h == 0)[0]
        to_flip = rng.choice(zeros, target - current, replace=False)
        h[to_flip] = 1
    return h

def local_search(h, n, rng, max_iter=100000):
    h = h.copy().astype(np.float64)
    c5 = eval_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    best_local = c5
    no_improve = 0
    max_no_improve = 5000
    
    for it in range(max_iter):
        if no_improve > max_no_improve:
            # Try a bigger jump: flip 5-10 pairs
            n_flips = rng.integers(3, 8)
            while n_flips > 0 and len(ones) > 0 and len(zeros) > 0:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                if i != j:
                    h[i], h[j] = h[j], h[i]
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    n_flips -= 1
            c5 = eval_c5(h)
            no_improve = 0
            continue
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_c5 = eval_c5(h)
        
        if new_c5 <= c5:
            c5 = new_c5
            no_improve = 0
            if c5 < best_local:
                best_local = c5
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[i], h[j] = h[j], h[i]  # revert
            no_improve += 1
    
    return h, c5

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    tm = np.zeros(n, dtype=np.float64)
    val = 0
    for i in range(n):
        # Count bits set in i
        tm[i] = bin(i).count('1') % 2
    return tm
