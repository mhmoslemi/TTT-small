    return compute_c5(x)

def slsqp_optimize(x0, n, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objective_wrap, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

def greedy_refine(h, n, n_iter, rng):
    h = h.copy()
    target = n / 2.0
    best = compute_c5(h)
    for _ in range(n_iter):
        improved = False
        order = rng.permutation(n)
        for i in order:
            orig = h[i]
            best_move = None
            best_val = best
            for dv in (0.05, -0.05, 0.1, -0.1, 0.02, -0.02, 0.15, -0.15):
                nv = orig + dv
                if nv < 0.0 or nv > 1.0:
                    continue
                h[i] = nv
                d = target - h.sum()
                if abs(d) < 1e-12:
                    c = compute_c5(h)
                else:
                    # find j to compensate
                    best_j = -1
                    best_cj = np.inf
                    mask = (h > abs(d) + 1e-14) & (h < 1.0 - abs(d) - 1e-14)
                    if not mask.any():
                        mask = (h > 0) & (h < 1)
                    candidates = np.where(mask)[0]
                    candidates = candidates[candidates != i]
                    if len(candidates) == 0:
                        h[i] = orig
                        continue
                    for j in candidates[:20]:
                        h[j] += d
                        c = compute_c5(h)
                        if c < best_cj:
                            best_cj = c
                            best_j = j
                    if best_j < 0:
                        h[i] = orig
                        continue
                    h[best_j] = 1.0  # placeholder to restore
                    # redo: set h[i]=nv, h[best_j] already has +d
                    h[i] = nv
                    c = best_cj
                    # actually h[best_j] already has the adjustment from the loop
                    # but we need to make sure
                    h[i] = orig  # will set properly below
                    h[best_j] -= d  # undo
                if c < best_val - 1e-13:
                    best_val = c
                    best_move = (i, nv, dv)
            if best_move is not None:
                i_m, nv_m, dv_m = best_move
                h[i_m] = nv_m
                d = target - h.sum()
                if abs(d) > 1e-13:
                    mask = (h > abs(d) + 1e-14) & (h < 1.0 - abs(d) - 1e-14)
                    mask = mask & (np.arange(n) != i_m)
                    if mask.any():
                        js = np.where(mask)[0]
                        j_best = js[0]
                        h[j_best] += d
                h = np.clip(h, 0, 1)
                best = compute_c5(h)
                improved = True
                break
            else:
                h[i] = orig
        if not improved:
            break
    return h, best

def nm_polish(h, n, rng, max_iter=300):
    h = h.copy()
    target = n / 2.0
    def fp(x):
        xx = project_feasible(x, n)
        return compute_c5(xx)
    x0 = h.copy()
    x0 = project_feasible(x0, n)
    # Nelder-Mead with penalty
    def penalized(x):
        xx = np.clip(x, 0, 1)
        pen = 100.0 * (xx.sum() - target) ** 2
        return compute_c5(xx) + pen
    from scipy.optimize import minimize as _min
    res = _min(penalized, x0, method='Nelder-Mead',
               options={'xatol': 1e-8, 'fatol': 1e-10, 'maxiter': max_iter,
                        'adaptive': True})
    h_out = project_feasible(res.x, n)
    return h_out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)

    best_h = project_feasible(h_parent, n_parent)
    best_c = compute_c5(best_h)
    n_best = n_parent
    print(f"Parent: n={n_parent}, C5={best_c:.6f}")

    def t_left():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP from parent (generous iterations)
    h_r = slsqp_optimize(best_h.copy(), n_parent, 2000, 1e-14)
    if h_r is not None:
        c_r = compute_c5(h_r)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
            print(f"  SLSQP main: C5={best_c:.6f}, t={time.time()-t0:.0f}s")

    # Phase 2: SLSQP restarts with perturbations
    n_restarts = 0
    while t_left() > 120:
        scale = 0.05 + 0.15 * (n_restarts % 5) / 5.0
        if n_restarts < 15:
            x0 = h_parent + rng.standard_normal(n_parent) * scale
        elif n_restarts < 30:
            x0 = best_h + rng.standard_normal(n_parent) * scale * 0.5
        else:
            x0 = rng.uniform(0.1, 0.9, n_parent)
        x0 = project_feasible(x0, n_parent)
        h_r = slsqp_optimize(x0, n_parent, 500, 1e-12)
        if h_r is not None:
            c_r = compute_c5(h_r)
            if c_r < best_c - 1e-12:
                best_h, best_c = h_r, c_r
                print(f"  R{n_restarts}: C5={best_c:.6f}, t={time.time()-t0:.0f}s")
        n_restarts += 1
        if n_restarts % 5 == 0:
            print(f"  @{n_restarts} restarts, best={best_c:.6f}, t={time.time()-t0:.0f}s")

    # Phase 3: Greedy coordinate refinement
    if t_left() > 60:
        h_g, c_g = greedy_refine(best_h, n_best, 80, rng)
        if c_g < best_c:
            best_h, best_c = h_g, c_g
            print(f"  Greedy: C5={best_c:.6f}, t={time.time()-t0:.0f}s")

    # Phase 4: Nelder-Mead polish
    if t_left() > 90:
        h_nm = nm_polish(best_h, n_best, rng, max_iter=500)
        c_nm = compute_c5(h_nm)
        if c_nm < best_c:
            best_h, best_c = h_nm, c_nm
            print(f"  NM: C5={best_c:.6f}, t={time.time()-t0:.0f}s")

    # Phase 5: Try slightly higher resolution
    if t_left() > 180:
        n2 = n_parent + 30
        xo = np.linspace(0, 2, n_parent)
        xn = np.linspace(0, 2, n2)
        h_interp = np.interp(xn, xo, h_parent)
        h_interp = project_feasible(h_interp, n2)
        c_int = compute_c5(h_interp)
        if c_int < best_c + 0.01:
            h_r = slsqp_optimize(h_interp, n2, 1000, 1e-12)
            if h_r is not None:
                c_r = compute_c5(h_r)
                if c_r < best_c:
                    best_h, best_c, n_best = h_r, c_r, n2
                    print(f"  HigherRes n={n2}: C5={best_c:.6f}")

    # Final cleanup
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h)
    print(f"FINAL: n={n_best}, C5={best_c:.6f}, time={time.time()-t0:.0f}s")
    return best_h, best_c, n_best
