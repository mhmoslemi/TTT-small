import numpy as np
import time
import scipy.optimize as opt

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_max_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr))
    # Find lags at max
    thresh = max_val - 1e-12
    max_indices = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for idx in max_indices:
        L = int(idx - (n - 1))
        g = np.zeros(n)
        if L >= 0:
            lo1, hi1 = L, n - 1
            lo2, hi2 = 0, n - 1 - L
            g[lo1:hi1+1] += (1.0 - h[lo1-L:hi1+1-L])
            g[lo2:hi2+1] -= h[lo2+L:hi2+1+L]
        else:
            k = -L
            lo1, hi1 = 0, n - 1 - k
            lo2, hi2 = k, n - 1
            g[lo1:hi1+1] += (1.0 - h[lo1+k:hi1+1+k])
            g[lo2:hi2+1] -= h[lo2-k:hi2+1-k]
        grad += g
    grad /= len(max_indices)
    return max_val * dx, grad * dx

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h_parent, n_new):
    n_old = len(h_parent)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_parent)
    return fix_sum(h_new, n_new)

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def perturb(h, rng, sigma=0.05):
    h_new = h + rng.randn(len(h)) * sigma
    return fix_sum(h_new, len(h))

def slsqp_optimize(h0, n, maxiter=300):
    def obj(h):
        return compute_max_grad(h)[0]
    def obj_grad(h):
        return compute_max_grad(h)[1]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n) / n}]
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'disp': False, 'maxiter': maxiter, 'ftol': 1e-12})
    h_out = fix_sum(res.x, n)
    return h_out, eval_c5(h_out)

def nelder_mead_optimize(h0, n, time_limit=30):
    t0 = time.time()
    def obj(h):
        return eval_c5(fix_sum(h, n))
    def cons(h):
        return abs(np.sum(h) - n / 2.0)
    # Use penalty method with Nelder-Mead
    best_h = h0.copy()
    best_f = obj(h0)
    step = 0.02
    for _ in range(500):
        if time.time() - t0 > time_limit:
            break
        direction = np.random.randn(n) * step
        h_try = fix_sum(h0 + direction, n)
        f_try = obj(h_try)
        if f_try < best_f:
            best_h = h_try.copy()
            best_f = f_try
            step = min(step * 1.1, 0.1)
        else:
            step = max(step * 0.7, 0.001)
        h0 = best_h.copy()
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 10 - (time.time() - t_start)

    parent = np.asarray(initial_h_values, dtype=float)
    n_p = len(parent)
    best_h = parent.copy()
    best_c5 = eval_c5(best_h)

    # Optimize at parent resolution
    for attempt in range(4):
        if time_left() < 60:
            break
        if attempt == 0:
            h0 = best_h.copy()
        elif attempt == 1:
            h0 = perturb(best_h, rng, 0.03)
        elif attempt == 2:
            h0 = random_binary(n_p, rng)
        else:
            h0 = perturb(best_h, rng, 0.08)
        try:
            h_opt, c5 = slsqp_optimize(h0, n_p, maxiter=400)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
        except Exception:
            pass

    # Nelder-Mead refinement at parent
    if time_left() > 60:
        h0 = perturb(best_h, rng, 0.02)
        try:
            h_opt, c5 = nelder_mead_optimize(h0, n_p, time_limit=30)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
        except Exception:
            pass

    # Try higher resolutions
    for n_new in [140, 280, 420]:
        if time_left() < 120:
            break
        h_up = upsample(best_h, n_new)
        n_iters = 200 if n_new <= 140 else 100
        for attempt in range(3):
            if time_left() < 60:
                break
            if attempt == 0:
                h0 = h_up if attempt == 0 else upsample(best_h, n_new)
            elif attempt == 1:
                h0 = random_binary(n_new, rng)
            else:
                h0 = perturb(h_up, rng, 0.04)
            try:
                h_opt, c5 = slsqp_optimize(h0, n_new, maxiter=n_iters)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
            except Exception:
                pass
        # NM refinement at this resolution
        if time_left() > 60:
            try:
                h0 = perturb(best_h, rng, 0.02)
                h_opt, c5 = nelder_mead_optimize(h0, len(best_h), time_limit=20)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
            except Exception:
                pass

    # Final SLSQP polish
    if time_left() > 45:
        n_f = len(best_h)
        try:
            h0 = perturb(best_h, rng, 0.01)
            h_opt, c5 = slsqp_optimize(h0, n_f, maxiter=300)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
        except Exception:
            pass

    # Ensure valid output
    best_h = fix_sum(best_h, len(best_h))
    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
