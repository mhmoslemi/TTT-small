- This computes the cross-correlation of h with (1-h)
- For arrays a and v of lengths na and nv, the result has length na+nv-1

For k = n-1-m (negative lag m): sum_i h[i] * (1-h[i+m]) (for valid indices)

            res = opt.minimize(obj_raw, best_h, method='SLSQP',
                               bounds=bounds, constraints=constr,
                               options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = fix_sum(res.x, S)
            c5_new = eval_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_c5(best_h, dx)

    return (best_h, best_c5, n)
