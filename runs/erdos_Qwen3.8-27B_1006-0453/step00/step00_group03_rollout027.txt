    if diff > 1e-10:
        idx = np.argsort(-h)
        for i in idx:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    elif diff < -1e-10:
        idx = np.argsort(h)
        for i in idx:
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
            if abs(diff) <= 1e-12:
                break
    return h

def _make_structured(n, k):
    if k == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif k == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif k == 2:
        h = 0.5 + 0.5 * np.sin(2.0 * np.pi * 3.0 * np.arange(n) / n + np.pi / 2)
    elif k == 3:
        h = 0.5 + 0.5 * np.cos(2.0 * np.pi * 5.0 * np.arange(n) / n)
    else:
        h = np.full(n, 0.5)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_end = t0 + budget_s
    rng = np.random.default_rng(seed)

    n = len(initial_h_values)
    best_h = np.array(initial_h_values, dtype=float).copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize parent directly with multiple methods
    result = _try_slsqp(best_h.copy(), n, 3000)
    if result is not None and result[1] < best_c5:
        best_h, best_c5 = result
    result = _try_lbfgs(best_h.copy(), n, 2000)
    if result is not None and result[1] < best_c5:
        best_h, best_c5 = result
    result = _try_slsqp(best_h.copy(), n, 2000)
    if result is not None and result[1] < best_c5:
        best_h, best_c5 = result

    # Phase 2: Multi-start optimization
    n_multi = 0
    while time.time() < budget_end - 80:
        n_multi += 1
        if n_multi <= 5:
            h0 = _make_structured(n, n_multi - 1)
        else:
            sigma = rng.uniform(0.01, 0.15)
            h0 = best_h + rng.normal(0, sigma, n)
        h0 = np.clip(h0, 0.0, 1.0)
        s = h0.sum()
        if s > 1e-8:
            h0 = h0 * (n / 2.0) / s
            h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_sum(h0, n)

        if rng.random() < 0.65:
            result = _try_slsqp(h0, n, 1000)
        else:
            result = _try_lbfgs(h0, n, 800)

        if result is not None and result[1] < best_c5:
            best_h, best_c5 = result
            result2 = _try_slsqp(best_h.copy(), n, 1000)
            if result2 is not None and result2[1] < best_c5:
                best_h, best_c5 = result2

    # Phase 3: Transition to higher resolution if time allows
    if time.time() < budget_end - 120:
        n2 = n * 2
        h2 = np.interp(np.linspace(0, 1, n2, endpoint=False),
                       np.linspace(0, 1, n, endpoint=False),
                       best_h)
        h2 = _project_sum(h2, n2)
        result = _try_slsqp(h2, n2, 1000)
        if result is not None and result[1] < best_c5:
            best_h, best_c5 = result
            n = n2
            result2 = _try_slsqp(best_h.copy(), n, 1500)
            if result2 is not None and result2[1] < best_c5:
                best_h, best_c5 = result2

    # Phase 4: Final refinement
    if time.time() < budget_end - 20:
        result = _try_slsqp(best_h.copy(), n, 5000)
        if result is not None and result[1] < best_c5:
            best_h, best_c5 = result
        result = _try_lbfgs(best_h.copy(), n, 3000)
        if result is not None and result[1] < best_c5:
            best_h, best_c5 = result

    # Final projection and evaluation
    best_h = _project_sum(best_h, n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
