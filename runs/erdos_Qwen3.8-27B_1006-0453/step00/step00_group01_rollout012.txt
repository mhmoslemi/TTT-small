        dx_f = 2.0 / n_f
        tgt_f = n_f / 2.0
        h_f = best_h.copy()
        h_f, c5_f = local_search(h_f, n_f, dx_f, tgt_f, rng, deadline4)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        # Discrete pass
        deadline4b = t0 + budget_s - 1.0
        if time.time() < deadline4b:
            h_d, c5_d = swap_local_search(best_h.copy(), best_n, 2.0 / best_n,
                                           best_n / 2.0, rng, deadline4b)
            if c5_d < best_c5:
                best_c5, best_h = c5_d, h_d

    # Final validation
    best_h = project_h(best_h, best_n, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
