def compute_c5(h):

    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

    """Numerical gradient of corr[k_star] * dx."""
    n = len(h)
    dx = 2.0 / n
    eps = 1e-6
    grad = np.zeros(n)
    for j in range(n):
        h2 = h.copy()
        h2[j] += eps
        cj = np.correlate(h2, 1-h2, mode='full')[k_star] * dx
        # We need central difference
        h3 = h.copy()
        h3[j] -= eps
        cj2 = np.correlate(h3, 1-h3, mode='full')[k_star] * dx
        grad[j] = (cj - cj2) / (2*eps)
    return grad

    n = len(v)
    lo, hi = v.min() - 1, v.max() + 1

        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:

        else:
            hi = mid

    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)

    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    for it in range(max_iters):
        if time.time() > dt_deadline:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1-h, mode='full')
        k_star = np.argmax(corr)
        
        # Gradient

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')

def compute_correlation(h):

def grad_corr(h, m):
    n = len(h)
    dx = 2.0 / n
    tau = m - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    idx1 = j - tau
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = j + tau
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] -= h[idx2[mask2]]
    return grad * dx
