        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def local_optimize(h, n, rng, max_iter, deadline):
    h = h.copy()
    c5 = compute_c5(h, n)
    steps = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.25]
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        improved = False
        for s in steps:
            for d in (s, -s):
                hi, hj = h[i] + d, h[j] - d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    nc = compute_c5(h, n)
                    if nc < c5:
                        c5 = nc
                        improved = True
                        break
                    h[i], h[j] = hi - d, hj + d
            if improved:
                break
    return h, c5

def kick(h, n, rng, strength):
    nk = rng.randint(3, max(4, n // 10))
    idx = rng.choice(n, nk, replace=False)
    h[idx] = np.clip(h[idx] + rng.normal(0.0, strength, nk), 0.0, 1.0)
    fix_sum(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3.0

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        n = 80
        h_init = np.full(n, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Thorough local optimization
    best_h, best_c5 = local_optimize(best_h, n, rng, 15000, deadline - 300.0)

    # Phase 2: Simulated Annealing with kicks
    h = best_h.copy()
    c5 = best_c5
    T = 0.004
    alpha = 0.9997
    T_min = 1e-8
    last_improve = time.time()

    while time.time() < deadline - 130.0:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (i + 1 + rng.randint(n - 1)) % n

        s = min(rng.lognormal(-3.0, 1.0), 0.35)
        d = s if rng.random() < 0.5 else -s
        hi, hj = h[i] + d, h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            T = max(T * alpha, T_min)
            continue

        h[i], h[j] = hi, hj
        nc = compute_c5(h, n)

        if nc < c5:
            c5 = nc
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improve = time.time()
        elif np.log(rng.random() + 1e-300) > -(nc - c5) / T:
            c5 = nc
        else:
            h[i], h[j] = hi - d, hj + d

        T = max(T * alpha, T_min)

        if time.time() - last_improve > 45.0:
            if time.time() >= deadline - 160.0:
                break
            h, c5 = local_optimize(h, n, rng, 400, deadline - 130.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            kick(h, n, rng, 0.08)
            c5 = compute_c5(h, n)
            T = max(T, 0.002)
            last_improve = time.time()

    # Phase 3: Final thorough local polish
    if time.time() < deadline - 10.0:
        best_h, best_c5 = local_optimize(best_h, n, rng, 30000, deadline)

    # Multi-start: small perturbations of best
    if time.time() < deadline - 15.0:
        for _ in range(4):
            if time.time() > deadline - 8.0:
                break
            h2 = best_h.copy()
            nk2 = rng.randint(2, 6)
            idx2 = rng.choice(n, nk2, replace=False)
            h2[idx2] = np.clip(h2[idx2] + rng.normal(0.0, 0.025, nk2), 0.0, 1.0)
            fix_sum(h2, n)
            h2, c5_2 = local_optimize(h2, n, rng, 4000, deadline - 5.0)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Another final polish
    if time.time() < deadline - 3.0:
        best_h, best_c5 = local_optimize(best_h, n, rng, 5000, deadline)

    fix_sum(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
