import math
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n, target, rng):
    h = (h >= 0.5).astype(np.float64)
    s = int(np.sum(h))
    if s > target:
        idx = list(np.where(h == 1)[0])
        rng.shuffle(idx)
        for k in range(s - target):
            h[idx[k]] = 0
    elif s < target:
        idx = list(np.where(h == 0)[0])
        rng.shuffle(idx)
        for k in range(target - s):
            h[idx[k]] = 1
    return h

def anneal(h, n, target, rng, t0, budget_s, frac_budget):
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    T = c5 * 0.003 + 1e-6
    T_min = 1e-8
    cool = 0.99997
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    while time.time() - t0 < budget_s * frac_budget:
        # Choose move type
        r = rng.random()
        if r < 0.8:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            delta = new_c5 - c5
            if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
                c5 = new_c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                # Update lists
                pi = ones.index(i)
                pj = zeros.index(j)
                ones[pi] = j
                zeros[pj] = i
            else:
                h[i], h[j] = h[j], h[i]
        elif r < 0.95:
            # Pair swap: two ones with two zeros
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)

                c, d = zeros[j1], zeros[j2]

                h[b], h[d] = h[d], h[b]
                new_c5 = compute_c5(h)

                if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
                    c5 = new_c5
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    # Update lists
                    for old, new in [(a, c), (b, d)]:
                        pos = ones.index(old)
                        ones[pos] = new

                        zeros[pos] = new

import math
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, target, rng):
    h = np.array(h, dtype=np.float64)
    h = (h >= 0.5).astype(np.float64)
    s = int(np.sum(h))
    if s > target:
        idx = list(np.where(h == 1)[0])
        rng.shuffle(idx)
        for k in range(s - target):
            h[idx[k]] = 0
    elif s < target:
        idx = list(np.where(h == 0)[0])
        rng.shuffle(idx)
        for k in range(target - s):
            h[idx[k]] = 1
    return h

def anneal(h, rng, t_stop):
    n = len(h)
    target = n // 2
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    T = c5 * 0.005 + 1e-6
    T_min = 1e-9

        if r < 0.75:
            # Single swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            nc = compute_c5(h)
            d = nc - c5
            if d < 0 or rng.random() < math.exp(-d / max(T, 1e-12)):
                c5 = nc
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
        else:

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

                    h[i2], h[j2] = h[j2], h[i2]
