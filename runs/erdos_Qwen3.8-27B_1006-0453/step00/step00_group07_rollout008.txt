import numpy as np
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def do_sa(h, n, rng, T, cool, T_min, step_lo, step_hi, t0, t_end):
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    while time.time() < t_end:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        eps = float(rng.uniform(step_lo, step_hi))
        d = 1.0 if rng.random() < 0.5 else -1.0
        v1 = h[i] + d * eps
        v2 = h[j] - d * eps
        if v1 < 0.0 or v1 > 1.0 or v2 < 0.0 or v2 > 1.0:
            if d > 0:
                max_e = min(1.0 - h[i], h[j])
            else:
                max_e = min(1.0 - h[j], h[i])
            if max_e < 1e-9:

            v2 = h[j] - d * eps

        nh[i] = v1
        nh[j] = v2
        nc = compute_c5(nh)
        delta = nc - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h = nh
            cur_c5 = nc
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        T = max(T * cool, T_min)
        if rng.random() < 0.0005 and T < 0.003:
            T = 0.01
    return best_h, best_c5
