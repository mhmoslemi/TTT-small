            ratio = -(c5_new - c5_cur) / T
            if ratio > -700.0 and rng.random() < np.exp(ratio):
                c5_cur = c5_new
            else:
                h[i] = old_i
                h[j] = old_j
        cnt += 1
        if cnt >= cool_every:
            T = max(T * T_ratio, T_min)
            cnt = 0
    return best_h, best_c5

def do_hill_climb(h_start, n, rng, deadline, max_step):
    h = h_start.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    stuck = 0
    while time.time() < deadline:
        if stuck > 20000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-max_step, max_step)
        old_i = h[i]
        old_j = h[j]
        h[i] = old_i + delta
        h[j] = old_j - delta
        if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
            h[i] = old_i
            h[j] = old_j
            stuck += 1
            continue
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            stuck = 0
        else:
            h[i] = old_i
            h[j] = old_j
            stuck += 1
    return best_h, best_c5

def do_multi_step_sa(h_start, n, rng, T_start, cool_every, T_ratio, T_min, deadline):
    h = h_start.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = T_start
    cnt = 0
    while time.time() < deadline:
        # Sometimes do 2-3 pair moves in one step
        n_moves = 1 + (rng.random() < 0.3) + (rng.random() < 0.1)
        old_vals = []
        valid = True
        for _ in range(n_moves):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            step = 0.003 + 0.04 * T
            delta = rng.uniform(-step, step)
            old_i = h[i]
            old_j = h[j]
            h[i] = old_i + delta
            h[j] = old_j - delta
            if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
                valid = False
                break
            old_vals.append((i, old_i, j, old_j))
        if not valid:
            for (i, oi, j, oj) in old_vals:
                h[i] = oi
                h[j] = oj
            continue
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        if c5_new <= c5_cur:
            c5_cur = c5_new
        elif T > T_min:
            ratio = -(c5_new - c5_cur) / T
            if ratio > -700.0 and rng.random() < np.exp(ratio):
                c5_cur = c5_new
            else:
                for (i, oi, j, oj) in reversed(old_vals):
                    h[i] = oi
                    h[j] = oj
        cnt += 1
        if cnt >= cool_every:
            T = max(T * T_ratio, T_min)
            cnt = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_deadline = t0 + budget_s - 3

    # Get initial or create fallback
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(80, 0.5)

    n = len(h0)
    target = n / 2.0
    h0 = repair_h(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SA on initial construction (50% of budget)
    d1 = t0 + 0.48 * (budget_s - 10)
    if time.time() < d1:
        h1, c51 = do_sa(h0, n, rng, 0.02, 15000, 0.93, 1e-10, d1)
        if c51 < best_c5:
            best_c5 = c51
            best_h = h1

    # Phase 2: Multi-step SA from current best (20%)
    d2 = t0 + 0.68 * (budget_s - 10)
    if time.time() < d2:
        h2, c52 = do_multi_step_sa(best_h, n, rng, 0.04, 20000, 0.94, 1e-10, d2)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: Hill climbing (15%)
    d3 = t0 + 0.83 * (budget_s - 10)
    if time.time() < d3:
        h3, c53 = do_hill_climb(best_h, n, rng, d3, 0.003)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3

    # Phase 4: Random restarts with SA (12%)
    d4 = t0 + 0.95 * (budget_s - 10)
    n_restarts = 3
    rem_time = max(0.0, d4 - time.time())
    for r in range(n_restarts):
        if time.time() > hard_deadline or rem_time < 15:
            break
        alloc = rem_time / (n_restarts - r)
        h_rand = rng.uniform(0, 1, n)
        h_rand = repair_h(h_rand, target)
        dr = min(time.time() + alloc, d4)
        h4, c54 = do_sa(h_rand, n, rng, 0.06, 25000, 0.95, 1e-10, dr)
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4

    # Phase 5: Final polish hill climb
    if time.time() < hard_deadline:
        h5, c55 = do_hill_climb(best_h, n, rng, hard_deadline, 0.001)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5

    # Final repair and evaluation
    best_h = repair_h(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
