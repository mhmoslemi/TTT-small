import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def full_sweep(h):
    """Try all single swaps, return best h and c5"""
    n = len(h)
    h = h.copy()
    curr_c5 = compute_c5(h)
    ones = np.flatnonzero(h >= 0.5)
    zeros = np.flatnonzero(h <= 0.5)
    for i in ones:
        for j in zeros:
            h[i] = 0.0
            h[j] = 1.0
            c5 = compute_c5(h)
            if c5 < curr_c5:
                curr_c5 = c5
            else:
                h[i] = 1.0
                h[j] = 0.0
    return h, curr_c5

    rng = np.random.default_rng(seed)
    h = h.astype(float).copy()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def full_sweep_binary(h):
    n = len(h)
    h = h.copy()
    curr_c5 = compute_c5(h)
    for _ in range(3):
        improved = True
        while improved:
            improved = False
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c5 = compute_c5(h)
                    if c5 < curr_c5 - 1e-15:
                        curr_c5 = c5
                        improved = True
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
    return h, curr_c5

def hill_climb(h, time_limit, seed):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = 0.004
    T_min = 1e-8

    max_stalls = 3000

    while time.time() < end_time:
        r = rng.random()
        if r < 0.55:
            ones = np.flatnonzero(h > 0.4)
            zeros = np.flatnonzero(h < 0.6)
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            si, sj = h[i], h[j]
            h[i] = 0.0
            h[j] = 1.0
            c5 = compute_c5(h)
            d = c5 - best_c5
            if d < -1e-15 or (d > 0 and d < T and rng.random() < np.exp(-d / T)):
                if d < 0:
                    best_c5 = c5
                    best_h = h.copy()
                    stalls = 0
                else:
                    stalls += 1
            else:
                h[i], h[j] = si, sj
                stalls += 1
        elif r < 0.80:

            if len({i1, i2, j1, j2}) < 4:

            sv = (h[i1], h[i2], h[j1], h[j2])
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0

def make_binary_targeted(h_template, n, rng):
    if len(h_template) == n:
        h = h_template.astype(float).copy()
    else:
        x_old = np.linspace(0, 2, len(h_template), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, h_template)
    h = np.where(h > 0.5, 1.0, 0.0)
    target_ones = n // 2
    if n % 2 == 1:
        target_ones = (n - 1) // 2
    curr = int(np.sum(h))
    if curr > target_ones:
        idx = np.where(h == 1.0)[0]
        rng.shuffle(idx)
        h[idx[:curr - target_ones]] = 0.0
    elif curr < target_ones:
        idx = np.where(h == 0.0)[0]
        rng.shuffle(idx)
        h[idx[:target_ones - curr]] = 1.0
    if n % 2 == 1:
        zero_idx = np.where(h == 0.0)[0]
        if len(zero_idx) > 0:
            h[zero_idx[0]] = 0.5
    return h

    target = n_f / 2.0

            if abs(diff) < 1e-14:

    final_c5 = compute_c5(best_h)
