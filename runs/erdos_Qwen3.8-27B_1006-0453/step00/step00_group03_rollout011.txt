def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    return float(np.max(vals))

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _normalize(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (target_sum / s)
    return np.clip(h, 0.0, 1.0)

def _sum_eq(x, target):
    return x.sum() - target

def _sum_eq_jac(x, target):
    return np.ones_like(x)

def _run_slsqp(x0, n, tgt, maxiter, deadline):
    if time.time() > deadline:
        return None, None
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (tgt,)}]
    try:
        res = minimize(_compute_c5, x0, method='SLSQP', bounds=bnds,
                      constraints=cons, maxiter=maxiter, ftol=1e-10,
                      options={'maxls': 40})
        hx = _normalize(res.x, tgt)
        c5v = _compute_c5(hx)
        if np.any(np.isnan(hx)):
            return None, None
        return hx, c5v
    except Exception:
        return None, None

def _run_pounders(x0, n, tgt, max_nfev, deadline):
    if time.time() > deadline:
        return None, None
    bnds = [(0.0, 1.0)] * n
    penalty = 50.0
    def obj_pen(x):
        base = _compute_c5(x)
        viol = abs(x.sum() - tgt)
        return base + penalty * viol

        res = minimize(obj_pen, x0, method='L-BFGS-B', bounds=bnds,

                      options={'maxls': 40, 'maxcor': 20})

        c5v = _compute_c5(hx)
        if np.any(np.isnan(hx)):
            return None, None
        return hx, c5v
