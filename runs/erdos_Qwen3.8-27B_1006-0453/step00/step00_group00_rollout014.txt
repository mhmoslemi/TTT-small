        h = np.clip(h, 0, 1)

        h = np.clip(h, 0, 1)

    val = m + float(np.log(np.sum(np.exp(T * (corr - m))))) / T
    return val * dx


def make_feasible(h, target_sum=22.5):

    if s > 1e-12 and abs(s - target_sum) > 1e-12:
