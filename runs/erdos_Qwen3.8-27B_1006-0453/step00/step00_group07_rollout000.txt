import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_valid(h, n, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        idx = int(np.argmax(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def perturb_two(h, n, rng, sigma):
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        j = (j + 1) % n
    delta = rng.normal(0.0, sigma)
    vi = h[i] + delta
    vj = h[j] - delta
    if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
        return i, j, delta
    # Try smaller
    for _ in range(3):
        delta = rng.normal(0.0, sigma * 0.5)
        vi = h[i] + delta
        vj = h[j] - delta
        if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
            return i, j, delta
    return -1, -1, 0.0


def run_sa_segment(h_best, c_best, n, rng, T_init, sigma_init, max_iters, t_deadline):
    cur_h = h_best.copy()
    cur_c = c_best
    T = T_init
    sigma = sigma_init
    target_sum = n / 2.0
    iters = 0
    no_improve = 0

    while iters < max_iters and time.time() < t_deadline:
        i, j, delta = perturb_two(cur_h, n, rng, sigma)
        if i >= 0:
            cur_h[i] += delta
            cur_h[j] -= delta
            nc = evaluate_c5(cur_h)
            accept = nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / max(T, 1e-14))
            if accept:
                cur_c = nc
                if nc < c_best:
                    c_best = nc
                    h_best = cur_h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                cur_h[i] -= delta
                cur_h[j] += delta
                no_improve += 1

        iters += 1
        if iters % 20000 == 0:
            T *= 0.6
            sigma *= 0.85
            if no_improve > 80000:
                cur_h = h_best.copy()
                cur_c = c_best
                T = T_init * 0.5
                sigma = sigma_init * 0.7
                no_improve = 0
            if T < 1e-8:
                T = 1e-8
            if sigma < 0.002:
                sigma = 0.002
    return h_best, c_best
