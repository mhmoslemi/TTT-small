def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max(), corr

    """Project h onto {sum = target_sum, 0 <= h <= 1}"""
    n = len(h)
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:

        else:

            k = m - (n - 1)
            grad += probs[m] * grad_at_k(h, k, n)

        # Remove mean component for constraint
        grad -= grad.mean()
        
        # Nesterov momentum
        momentum = 0.9 * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        
        # Check
        c5_new = np.max(compute_corr(h_new))
        if c5_new <= c5 + 1e-8:
            h = h_new
            lr = min(lr * 1.01, 0.03)
        else:
            lr *= 0.7
            momentum *= 0.5
        
        it += 1
        if it % 500 == 0:
            print(f"  iter {it}, c5={best_c5:.6f}, T={T:.1f}, lr={lr:.5f}")
    
    return best_h, best_c5

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def full_gradient(h, corr, T):
    """Compute gradient of smoothed max."""
    n = len(h)
    M = corr.max()
    s = np.clip(T * (corr - M), -400, 0)
    e = np.exp(s)
    p = e / e.sum()
    
    # Vectorized gradient computation
    # For each shift m (0 to 2n-2), offset k = m - (n-1)
    # grad += p[m] * grad_at_k(h, k, n)
    # 
    # grad_at_k for k>=0: grad[:n-k] += (1-h[k:]), grad[k:] += -h[:n-k]
    # grad_at_k for k<0:  grad[m:] += (1-h[:n-m]), grad[:n-m] += -h[m:]  where m=-k
    
    grad = np.zeros(n)
    # Process in batch for efficiency
    for m in range(2*n-1):
        if p[m] < 1e-8:
            continue
        k = m - (n-1)
        if k >= 0:
            grad[:n-k] += p[m] * (1.0 - h[k:])
            grad[k:] += p[m] * (-h[:n-k])
        else:
            mm = -k
            grad[mm:] += p[m] * (1.0 - h[:n-mm])
            grad[:n-mm] += p[m] * (-h[mm:])
    return grad
