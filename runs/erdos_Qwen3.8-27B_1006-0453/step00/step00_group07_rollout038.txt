def lse_grad(h, dx, beta):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    
    g = np.zeros(N)
    for k in range(2*N - 1):
        if w[k] < 1e-16:
            continue
        s = k - (N - 1)
        wk = w[k] * dx
        # First term vector: (1-h[i-s]) for i in [0,N-1], 0 if i-s out of range
        first = np.zeros(N)
        # i-s in [0,N-1] means i in [s, s+N-1], intersect [0,N-1]
        lo1 = max(0, s)
        hi1 = min(N, s + N)
        if lo1 < hi1:
            first[lo1:hi1] = 1.0 - h[lo1-s:hi1-s]
        # Second term vector: h[i+s] for i in [0,N-1], 0 if i+s out of range
        second = np.zeros(N)
        # i+s in [0,N-1] means i in [-s, N-1-s], intersect [0,N-1]
        lo2 = max(0, -s)
        hi2 = min(N, N - s)
        if lo2 < hi2:
            second[lo2:hi2] = h[lo2+s:hi2+s]
        g += wk * (first - second)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(h, dx):
    return float(np.max(compute_corr(h, dx)))

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 0.9999999
        else:
            mask = h > 1e-9
        if not mask.any():
            h[:] = target / n
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_box_simplex(h, n):
    target = n * 0.5
    lo, hi = -2.0, 3.0
    for _ in range(200):
        t = 0.5 * (lo + hi)
        v = np.clip(h - t, 0.0, 1.0)
        if v.sum() > target:
            lo = t
        else:
            hi = t
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lse_objective(h, dx, beta):
    vals = compute_corr(h, dx)
    m = vals.max()
    return m + np.log(np.exp(beta * (vals - m)).sum()) / beta

def lse_gradient(h, dx, beta):
    N = len(h)
    corr = compute_corr(h, dx)
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w_index = np.argmax(corr)
        w = np.zeros(2 * N - 1)
        w[w_index] = 1.0
        w_sum = 1.0
    w /= w_sum
    g = np.zeros(N)
    for k in range(2 * N - 1):
        if w[k] < 1e-16:
            continue
        s = k - (N - 1)
        wk = w[k] * dx
        first = np.zeros(N)
        lo1 = max(0, s)
        hi1 = min(N, s + N)
        if lo1 < hi1:
            first[lo1:hi1] = 1.0 - h[lo1 - s:hi1 - s]
        second = np.zeros(N)
        lo2 = max(0, -s)
        hi2 = min(N, N - s)
        if lo2 < hi2:
            second[lo2:hi2] = h[lo2 + s:hi2 + s]
        g += wk * (first - second)
    return g

def max_subgrad(h, dx, top_m=1):
    N = len(h)
    corr = compute_corr(h, dx)
    idx = np.argsort(corr)[::-1][:top_m]
    g = np.zeros(N)
    for k in idx:
        s = k - (N - 1)
        first = np.zeros(N)
        lo1 = max(0, s)
        hi1 = min(N, s + N)
        if lo1 < hi1:
            first[lo1:hi1] = 1.0 - h[lo1 - s:hi1 - s]
        second = np.zeros(N)
        lo2 = max(0, -s)
        hi2 = min(N, N - s)
        if lo2 < hi2:
            second[lo2:hi2] = h[lo2 + s:hi2 + s]
        g += dx * (first - second)
    return g / top_m

def slqp_optimize(h_init, dx, beta, maxiter=120):
    n = len(h_init)
    target = n * 0.5
    res = minimize(
        lambda x: lse_objective(x, dx, beta),
        h_init.copy(),
        jac=lambda x: lse_gradient(x, dx, beta),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return res.x

def projected_grad(h, dx, iters, beta_lse=200.0, time_limit=None, t0=None):
    N = len(h)
    h_cur = h.copy()
    lr = 0.02
    for it in range(iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        g = lse_gradient(h_cur, dx, beta_lse)
        h_new = proj_box_simplex(h_cur - lr * g, N)
        c_new = c5_val(h_new, dx)

            h_cur = h_new
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.8
        if lr < 1e-7:
            break
    return h_cur
