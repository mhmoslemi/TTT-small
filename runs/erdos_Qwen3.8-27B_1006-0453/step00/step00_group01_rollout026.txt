import numpy as np
import time

def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_compute_corr(h)) * dx)

def _project(v, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _gradient_softmax(h, w, n):
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wv = w[k_idx]
        if wv < 1e-6:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += wv * (1.0 - h[:n - k])
                grad[:n - k] -= wv * h[k:]
        else:
            kp = -k
            if kp < n:
                grad[:n - kp] += wv * (1.0 - h[kp:])
                grad[kp:] -= wv * h[:n - kp]
    return grad

def _optimize(h_init, rng, deadline):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = _project(np.clip(h_init, 0.0, 1.0), target_sum)
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    momentum = 0.9
    vel = np.zeros(n)
    it = 0
    while time.time() < deadline:
        it += 1
        progress = min(float(it) / 80000.0, 1.0)
        lr = 0.04 * (1.0 - 0.85 * progress) + 0.002
        T = 3.0 + 25.0 * (1.0 - progress)
        corr = _compute_corr(h)
        shifted = T * (corr - np.max(corr))
        w = np.exp(shifted)
        w /= w.sum()
        grad = _gradient_softmax(h, w, n)
        vel = momentum * vel - lr * grad
        h = _project(h + vel, target_sum)
        c5 = _compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 4000 == 0:
            h = _project(h + 0.06 * rng.standard_normal(n), target_sum)
    return best_h, best_c5

def _binary_local_search(h_bin, n, deadline, max_rounds=50):
    dx = 2.0 / n

    if abs(int(h_bin.sum()) - target_ones) > 0:
        return None, None
    h = h_bin.copy()
    best_c5 = _compute_c5(h, dx)
    rng_ls = np.random.default_rng(123)
    for r in range(max_rounds):

        improved = False
        indices = rng_ls.permutation(n)

                best_c5 = c5
                improved = True
            else:
                h[i] = 1.0 - h[i]
        if not improved:
            break
    return h, best_c5
