    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    s = corr * beta
    m = np.max(s)
    return float(m + np.log(np.sum(np.exp(s - m)))) / beta

def project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)

        else:
