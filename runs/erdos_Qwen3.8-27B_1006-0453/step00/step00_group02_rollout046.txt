def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-12:
        # Add to elements that have room
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:

                adj = min(abs(diff), abs(1.0 - h[i]) if diff > 0 else abs(h[i]))
                h[i] += np.sign(diff) * adj
                diff -= np.sign(diff) * adj
                if abs(diff) < 1e-12:
                    break
    return h

    x_old = np.arange(n_from) / n_from
    x_new = np.arange(n_to) / n_to

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-14:
            order = np.argsort(np.abs(h - 0.5))[::-1]
            for idx in order:
                if diff > 0:
                    adj = min(diff, 1.0 - h[idx])
                    h[idx] += adj
                    diff -= adj
                else:
                    adj = min(-diff, h[idx])
                    h[idx] -= adj
                    diff += adj
                if abs(diff) < 1e-14:
                    break
    return h


def interpolate_h(h_from, n_to):
    n_from = len(h_from)
    if n_from == n_to:
        return h_from.copy()
    x_old = np.arange(n_from, dtype=float) / n_from
    x_new = np.arange(n_to, dtype=float) / n_to
    h_ext = np.concatenate([h_from, [h_from[0]]])
    h_new = np.interp(x_new, x_old, h_ext)
    return fix_sum(h_new, n_to)


def optimize_resolution(h0, n, time_limit):
    best_h = h0.copy()
    best_val = compute_c5_exact(best_h)
    h_curr = h0.copy()

    for stage in range(6):
        if time.time() > time_limit:
            break
        alpha = 20 * (stage + 1)
        lam = 50.0 * (stage + 1)

        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            sm = smooth_max(corr, alpha) * (2.0 / n)
            pen = lam * (h.sum() - n / 2.0) ** 2
            return sm + pen

        def obj_exact(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            return float(np.max(corr) * (2.0 / n))

        maxiter = max(50, 200 - stage * 30)
        res = minimize(
            obj, h_curr, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-7}
        )
        h_new = fix_sum(res.x, n)
        val_new = obj_exact(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        h_curr = h_new

        # Coordinate-descent polishing
        for _ in range(30):
            if time.time() > time_limit:
                break
            improved = False
            for i in range(n):
                if time.time() > time_limit:
                    break
                old = h_curr[i]
                for cand in [old + 0.01, old - 0.01, old + 0.001, old - 0.001, 0.0, 1.0]:
                    if cand < 0 or cand > 1:
                        continue
                    h_test = h_curr.copy()
                    h_test[i] = cand
                    # Adjust one other point to maintain sum
                    diff = h_curr[i] - cand
                    if abs(diff) < 1e-10:
                        continue
                    if diff > 0:
                        j = np.argmax(1.0 - h_test)
                    else:
                        j = np.argmin(h_test)
                    if j == i:
                        continue
                    h_test[j] += diff
                    if h_test[j] < 0 or h_test[j] > 1:
                        continue
                    v = obj_exact(h_test)
                    if v < best_val - 1e-10:
                        h_curr = h_test.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    return best_h, best_val
