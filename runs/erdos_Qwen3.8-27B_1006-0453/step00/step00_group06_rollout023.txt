import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, size=n // 2, replace=False)
    h[pos] = 1.0
    return h

def greedy_construct(n, rng):
    """Greedy: place n/2 ones one at a time to minimize max overlap."""
    m = n // 2
    h = np.zeros(n)
    available = list(range(n))
    rng.shuffle(available)
    
    for _ in range(m):
        best_pos = -1
        best_c5 = float('inf')
        for pos in available:
            h[pos] = 1.0
            c5 = eval_c5(h)
            h[pos] = 0.0
            if c5 < best_c5:
                best_c5 = c5
                best_pos = pos
        h[best_pos] = 1.0
        available.remove(best_pos)
    return h, eval_c5(h)

    h = h.copy()
    cur_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    stall = 0

        j = int(rng.integers(n))

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def make_binary(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, size=n // 2, replace=False)
    h[pos] = 1.0
    return h


def greedy_construct(n, rng):
    m = n // 2
    h = np.zeros(n)
    available = rng.permutation(n).tolist()
    for _ in range(m):
        best_idx = 0
        best_val = float('inf')
        for idx in range(len(available)):
            pos = available[idx]
            h[pos] = 1.0
            val = eval_c5(h)
            h[pos] = 0.0
            if val < best_val:
                best_val = val
                best_idx = idx
        h[available[best_idx]] = 1.0
        available.pop(best_idx)
    return h


def local_search(h, rng, deadline, use_sa, temp_init):

    best_h = h.copy()
