    s = T * c
    m = s.max()
    e = np.exp(s - m)
    p = e / e.sum()
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        if klo <= khi:
            idx = np.arange(klo, khi + 1)
            j = i + idx - (n - 1)
            grad[i] += p[idx] @ (1.0 - h[j])
        klo2 = max(0, i)
        khi2 = min(2 * n - 2, i + n - 1)
        if klo2 <= khi2:
            idx2 = np.arange(klo2, khi2 + 1)
            j2 = i - idx2 + (n - 1)
            grad[i] -= p[idx2] @ h[j2]
    return dx * grad


def objective_and_grad(h, T, dx, target_sum, pw):
    h = np.clip(h, 0.0, 1.0)
    val = c5_smoothed_value(h, T, dx)
    pen = pw * (h.sum() - target_sum) ** 2
    grad = c5_smoothed_grad(h, T, dx) + 2.0 * pw * (h.sum() - target_sum)
    return val + pen, grad


def wrap_obj(h, T, dx, target_sum, pw, _=None):
    return objective_and_grad(h, T, dx, target_sum, pw)


def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_initial_random(n, seed, target_sum):
    rng = np.random.RandomState(seed)
    h = (rng.random(n) < 0.5).astype(float)
    deficit = target_sum - h.sum()
    n_adj = int(round(abs(deficit)))
    if deficit > 0:
        zeros = np.where(h == 0)[0]
        idxs = rng.permutation(zeros)[:min(n_adj, len(zeros))]
        h[idxs] = 1.0
    elif deficit < 0:
        ones = np.where(h == 1)[0]
        idxs = rng.permutation(ones)[:min(n_adj, len(ones))]
        h[idxs] = 0.0
    h = h + (target_sum - h.sum()) / n
    return np.clip(h, 0.0, 1.0)


def make_smooth_random(n, seed, target_sum):
    rng = np.random.RandomState(seed)
    k = rng.randint(2, max(3, n // 4))
    freqs = rng.randint(1, 4, size=k)
    amps = rng.random(k)
    phases = rng.uniform(0, 2 * np.pi, k)
    h = np.zeros(n)
    for i in range(n):
        h[i] = sum(amps[j] * np.sin(freqs[j] * i * 2.0 * np.pi / n + phases[j]) for j in range(k))
    h = h - h.min()
    if h.max() > 0:
        h = h / h.max()
    h = 0.5 + 0.5 * h
    h = project_sum(h, target_sum)
    return h


def optimize_from(h0, n, target_sum, dx, T_schedule, pw, bounds, time_limit):
    h_curr = h0.copy()
    for T in T_schedule:
        if time.time() > time_limit:
            break
        obj = lambda x: objective_and_grad(x, T, dx, target_sum, pw)
        try:
            res = minimize(obj, h_curr, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'gtol': 1e-9, 'ftol': 1e-14, 'maxls': 100})
            h_curr = res.x
        except Exception:
            break
    h_curr = project_sum(h_curr, target_sum)
    return h_curr


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15.0

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_initial_random(47, seed, 23.5)

    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = project_sum(h_init, target_sum)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    # Main optimization: anneal T
    T_sched = [5, 15, 40, 100, 250, 600, 1500, 4000]
    pw_sched = [50, 200, 1000, 5000, 20000, 50000, 100000, 200000]

    for T, pw in zip(T_sched, pw_sched):
        if time.time() > deadline:
            break
        h_curr = best_h.copy()
        for T2 in [T, int(T * 4), int(T * 16)]:
            if time.time() > deadline:
                break
            obj = lambda x, T2=T2: objective_and_grad(x, T2, dx, target_sum, pw)
            try:
                res = minimize(obj, h_curr, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'gtol': 1e-9, 'ftol': 1e-14})
                h_curr = res.x
            except Exception:
                pass
        h_curr = project_sum(h_curr, target_sum)
        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()

    # Random restarts
    rng = np.random.RandomState(seed)
    n_restarts = max(1, int((deadline - time.time()) / 25))
    n_restarts = min(n_restarts, 40)

    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if r == 0:
            h_r = best_h + 0.05 * rng.randn(n)
        elif r % 4 == 1:
            h_r = make_initial_random(n, seed + 1000 + r, target_sum)
        elif r % 4 == 2:
            h_r = make_smooth_random(n, seed + 2000 + r, target_sum)
        elif r % 4 == 3:
            # Block structure
            blen = rng.randint(3, max(4, n // 6))
            n_blocks = (n + blen - 1) // blen
            vals = rng.random(n_blocks)
            h_r = np.repeat(vals, blen)[:n]
            h_r = project_sum(h_r, target_sum)
        else:
            h_r = best_h + 0.2 * rng.randn(n)

        h_r = project_sum(h_r, target_sum)
        T_sub = [20, 100, 500, 2000]
        for T in T_sub:
            if time.time() > deadline:
                break
            obj = lambda x, T=T: objective_and_grad(x, T, dx, target_sum, 50000)
            try:
                res = minimize(obj, h_r, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 150, 'gtol': 1e-8, 'ftol': 1e-13})
                h_r = res.x
            except Exception:
                break
        h_r = project_sum(h_r, target_sum)
        c5 = compute_c5(h_r)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_r.copy()

    # Final polish: direct projected gradient on the hard max
    if time.time() < deadline - 10:
        h_p = best_h.copy()
        lr = 0.01
        for it in range(500):
            if time.time() > deadline:
                break
            c = np.correlate(h_p, 1.0 - h_p, mode='full')
            # Subgradient of max
            max_val = c.max()
            mask = c >= max_val - 1e-10
            p = np.zeros_like(c)
            p[mask] = 1.0 / mask.sum()
            grad = c5_smoothed_grad(h_p, 1.0, dx)  # T=1 gives gradient weighted by p
            # Use the actual subgradient: gradient of c_k* (the max one)
            # Rebuild properly: gradient is dx * (weighted sum)
            grad = np.zeros(n)
            for k in np.where(mask)[0]:
                d = k - (n - 1)
                for i in range(n):
                    g = 0.0
                    if 0 <= i + d < n:
                        g += (1.0 - h_p[i + d])
                    if 0 <= i - d < n:
                        g -= h_p[i - d]
                    grad[i] += p[k] * g
            grad *= dx
            # Projected gradient step
            h_p = np.clip(h_p - lr * grad, 0.0, 1.0)
            h_p = project_sum(h_p, target_sum)
            c5 = compute_c5(h_p)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_p.copy()
            if lr < 0.001:
                lr *= 0.95
            else:
                lr = max(0.0005, lr * 0.99)

    # Final feasibility check
    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
