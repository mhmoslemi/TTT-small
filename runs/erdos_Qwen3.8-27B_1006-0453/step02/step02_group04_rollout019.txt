    return float(np.max(corr) * dx)

def fix_sum(h, n):

    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:

                h[i] += add
            else:
                sub = min(rem, h[i])
                h[i] -= sub

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    tgt = 0.5 * n
    d = tgt - h.sum()
    if abs(d) < 1e-13: return h
    m = (h > 1e-9) & (h < 1.0 - 1e-9)
    c = m.sum()
    if c > 0 and abs(d) < c * 0.35:
        h[m] += d / c
        return np.clip(h, 0, 1)
    idx = np.argsort(h) if d > 0 else np.argsort(-h)
    r = abs(d)
    for i in idx:
        a = min(r, 1-h[i]) if d > 0 else min(r, h[i])
        if d > 0: h[i] += a
        else: h[i] -= a
        r -= a
        if r < 1e-13: break
    return np.clip(h, 0, 1)

def obj_grad(h, T, topk=0):
    n = len(h)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    tgt = 0.5 * n
    d = tgt - h.sum()
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-9) & (h < 1.0 - 1e-9)
    c = m.sum()
    if c > 0 and abs(d) < c * 0.35:
        h[m] += d / c
        return np.clip(h, 0, 1)
    idx = np.argsort(h) if d > 0 else np.argsort(-h)
    r = abs(d)
    for i in idx:
        a = min(r, 1.0 - h[i]) if d > 0 else min(r, h[i])
        if d > 0:
            h[i] += a
        else:
            h[i] -= a
        r -= a
        if r < 1e-13:
            break
    return np.clip(h, 0, 1)

def obj_grad(h, T, topk=0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    L = 2 * n - 1
    if topk > 0 and topk < L:
        ti = np.argsort(c)[-topk:]
        sc = c[ti]
        sm = sc.max()
        sh = np.clip((sc - sm) / T, -80.0, 0.0)
        e = np.exp(sh)
        val = (sm + T * np.log(e.sum())) * dx
        w = np.zeros(L)
        w[ti] = e / e.sum()
    else:
        cm = c.max()
        sh = np.clip((c - cm) / T, -80.0, 0.0)
        e = np.exp(sh)
        val = (cm + T * np.log(e.sum())) * dx
        w = e / e.sum()
    g12h = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for j in range(n):
        m_end = min(n, L - j)
        if m_end > 0:
            grad[j] = dx * np.dot(w[j:j+m_end], g12h[:m_end])
    return val, grad

def proj_grad(h, T, steps, topk=30, lr=0.01):
    n = len(h)
    hc = h.copy()
    for _ in range(steps):
        _, g = obj_grad(hc, T, topk)
        hc = fix_sum(hc - lr * g, n)
    return hc

def slsqp_min(h, T, topk=20, maxiter=2000):
    n = len(h)
    def f(x):
        return obj_grad(fix_sum(x, n), T, topk)[0]
    try:
        r = opt.minimize(f, h.copy(), method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n}],
                         options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(r.x, n) if r.x is not None else h
    except Exception:
        return h

def resample(h, nn):
    no = len(h)
    xo = np.arange(no) * 2.0 / no
    xn = np.arange(nn) * 2.0 / nn
    return fix_sum(np.interp(xn, xo, h, period=2.0), nn)
