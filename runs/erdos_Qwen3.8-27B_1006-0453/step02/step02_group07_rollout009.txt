Given slopes s_k and intercepts b_k for k ≠ n-1, and the quadratic for k=n-1:

f(x) = max( max_{k≠n-1}(s_k*x + b_k), -x² + x + C_{n-1} )

The min of f over [0,1]:
- For the linear part: min of max of lines is at a vertex (crossing of two lines) or at 0 or 1.
- The quadratic part complicates things slightly.

Practical approach: 
1. Find x* that minimizes the linear part (ignoring the quadratic)
2. Check if the quadratic is active at x*, and if so, solve the 1D problem numerically
3. The solution is the better of the two

For step 1: The minimum of max of lines over [0,1] is at the point where the upper envelope has a "flat" part (slope changes from negative to positive) or at 0/1.

To find this: 
- At x=0: active line is argmax(b_k)
- At x=1: active line is argmax(s_k + b_k)  
- The slope of the upper envelope goes from s_{argmax(b_k)} at x=0 to s_{argmax(s_k+b_k)} at x=1.
- If the slope at x=0 ≥ 0: minimum is at x=0
- If the slope at x=1 ≤ 0: minimum is at x=1
- Otherwise: minimum is at a crossing point where slope changes sign.

The crossing point: find the two lines that are active at the minimum. This is the pair (k1, k2) where s_{k1} < 0 < s_{k2} (approximately) and they cross at the point where both are on the upper envelope.

A simple algorithm:
1. Compute f(0) = max(b_k) and f(1) = max(s_k + b_k)
2. If f(0) ≤ f(1) and the slope at 0 ≥ 0: answer is 0
3. If f(1) ≤ f(0) and the slope at 1 ≤ 0: answer is 1
4. Otherwise: the minimum is at a crossing point. Find it by:
   - The slope of f at any x is the slope of the active line
   - We need the x where slope changes from negative to positive
   - This is the crossing of the "lower convex envelope" of the lines...

def compute_slopes_intercepts(h, j, n):
    """For changing h[j] to x, compute slope and intercept of each corr[k]."""
    g = 1.0 - h
    n_corr = 2 * n - 1
    slopes = np.zeros(n_corr)
    intercepts = np.zeros(n_corr)
    
    # For each k, the base correlation (with h[j] removed) plus linear terms
    # corr[k] = sum_i h[i] * g[i-k+n-1]
    # Terms involving h[j]:
    #   i=j: h[j]*g[j-k+n-1] = x * (1-h[j-k+n-1]) if j-k+n-1 valid
    #   i-k+n-1=j (i=j+k-n+1): h[j+k-n+1]*g[j] = h[j+k-n+1]*(1-x) if valid
    #   If both i=j AND i=j+k-n+1 (i.e., k=n-1): x*(1-x)
    
    # Strategy: compute full correlation, then subtract old h[j] contributions, add new
    base = np.correlate(h, g, mode='full')  # current correlation values
    
    for k in range(n_corr):
        # Remove contribution of h[j] from base[k], then add linear in x
        # Contribution of h[j] when i=j (h factor): h[j]*g[j-k+n-1] if valid
        idx_g = j - k + n - 1
        contrib_h = 0.0
        if 0 <= idx_g <= n-1:
            contrib_h = h[j] * g[idx_g]
        # Contribution of h[j] when g factor is at j (g factor): h[j+k-n+1]*g[j] if valid
        idx_h = j + k - n + 1
        contrib_g = 0.0
        if 0 <= idx_h <= n-1:
            contrib_g = h[idx_h] * g[j]
        
        # But if both are the same term (k=n-1, idx_g=j, idx_h=j), we double-counted
        if k == n - 1:
            # The term h[j]*g[j] = h[j]*(1-h[j]) appears once, not twice
            base_no_j = base[k] - h[j] * g[j]
            # New: x*(1-x) + base_no_j = -x^2 + x + base_no_j
            # This is quadratic; handle separately
            slopes[k] = 1.0  # linear part slope (x coefficient)
            intercepts[k] = base_no_j  # will add -x^2 separately
        else:
            base_no_j = base[k] - contrib_h - contrib_g
            # New: x*(1-h[idx_g]) + h[idx_h]*(1-x) + base_no_j (for valid indices)
            a = 0.0  # coefficient of x from h factor
            if 0 <= idx_g <= n-1:
                a = 1.0 - h[idx_g]  # = g[idx_g]
            b = 0.0  # coefficient of (1-x) from g factor = -x coefficient
            if 0 <= idx_h <= n-1:
                b = h[idx_h]
            # Total: base_no_j + a*x + b*(1-x) = (base_no_j + b) + (a-b)*x
            slopes[k] = a - b
            intercepts[k] = base_no_j + b
    
    return slopes, intercepts

def min_coord(h, j, n):
    """Find x in [0,1] minimizing max_k corr_k(x) where only h[j] changes to x."""
    g = 1.0 - h
    corr_orig = np.correlate(h, g, mode='full')
    h0 = h[j]
    n_corr = 2 * n - 1
    
    # For k ≠ n-1: corr[k](x) = corr_orig[k] + (x - h0) * (1 - h[j-k+n-1] - h[j+k-n-1+n-1])
    # Simplify: (1 - h[idx_g]) - h[idx_h] where idx_g = j-k+n-1, idx_h = j+k-n+1
    
    # For k = n-1: corr[n-1](x) = corr_orig[n-1] + x(1-x) - h0(1-h0)
    
    # Compute slopes and base values
    # For k ≠ n-1: corr[k](x) = base_k + slope_k * x
    ks = np.arange(n_corr)
    idx_g = j - ks + n - 1  # index in h for g factor
    idx_h = j + ks - n + 1  # index in h for h factor
    
    valid_g = (idx_g >= 0) & (idx_g < n)
    valid_h = (idx_h >= 0) & (idx_h < n)
    
    # Slope for k ≠ n-1: (1-h[idx_g]) - h[idx_h]
    slopes = np.zeros(n_corr)
    for k in range(n_corr):
        if k == n - 1:
            continue
        sg = 1.0 - h[max(0, min(n-1, idx_g[k]))] if valid_g[k] else 0.0
        sh = h[max(0, min(n-1, idx_h[k]))] if valid_h[k] else 0.0
        slopes[k] = sg - sh
    # At k = n-1, handle as quadratic
    
    # Base: corr_orig[k] - h0 * slope_k (for k ≠ n-1)
    # Actually: corr[k](x) = corr_orig[k] + (x-h0)*slope_k = (corr_orig[k] - h0*slope_k) + slope_k*x
    bases = corr_orig - h0 * slopes  # for k ≠ n-1
    
    # Now find min over x ∈ [0,1] of max( max_{k≠n-1}(bases[k] + slopes[k]*x), corr_orig[n-1] + x(1-x) - h0(1-h0) )
    
    # For the linear part: min of max of lines
    # Evaluate at a set of x values
    # The critical points are where two lines cross: x = (bases[k2]-bases[k1])/(slopes[k1]-slopes[k2])
    # Plus x=0, x=1
    
    linear_part = bases + slopes[:, None] * 0  # Just to have the right shape
    # Actually let me just sample
    xs = np.linspace(0, 1, 200)
    # Evaluate linear part: max over k≠n-1 of (bases[k] + slopes[k]*x)
    lin = bases + slopes[:, None] * xs[None, :]  # (n_corr, 200)
    lin[n-1, :] = -np.inf  # exclude k=n-1 from linear part
    lin_max = np.max(lin, axis=0)
    
    # Quadratic part at k=n-1
    quad = corr_orig[n-1] + xs*(1-xs) - h0*(1-h0)
    
    f = np.maximum(lin_max, quad)
    
    best_i = np.argmin(f)
    best_x = xs[best_i]
    
    # Refine: check neighborhood of best_x more finely
    if 0 < best_i < len(xs)-1:
        xs_fine = np.linspace(best_x - xs[1], best_x + xs[1], 50)
        xs_fine = np.clip(xs_fine, 0, 1)
        lin_f = bases + slopes[:, None] * xs_fine[None, :]
        lin_f[n-1, :] = -np.inf
        lin_max_f = np.max(lin_f, axis=0)
        quad_f = corr_orig[n-1] + xs_fine*(1-xs_fine) - h0*(1-h0)
        f_f = np.maximum(lin_max_f, quad_f)
        best_x = xs_fine[np.argmin(f_f)]
    
    return best_x

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    # Distribute adjustment to interior points
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment
    d2 = target - np.sum(h)
    if abs(d2) > 1e-10:
        idx = np.where(m)[0]
        for i in idx:
            adj = min(abs(d2), abs(h[i] - (0 if d2 > 0 else 1)))
            h[i] += np.sign(d2) * adj
            d2 -= np.sign(d2) * adj
            if abs(d2) < 1e-14:
                break
    return h

def lse_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def slsqp_step(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T, n), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def exact_coord_descent(h, n, n_sweeps=20):
    """Coordinate descent with exact 1D minimization."""
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            best_x = min_coord_val(h, j, n)
            if abs(best_x - h[j]) > 1e-12:
                h[j] = best_x
                improved = True
        h = fix_sum(h, n)
        if not improved:
            break
    return h

def min_coord_val(h, j, n):
    """Find x in [0,1] minimizing max_k corr_k when h[j]=x, others fixed."""
    g = 1.0 - h
    corr_orig = np.correlate(h, g, mode='full')
    h0 = h[j]
    n_corr = 2 * n - 1
    
    ks = np.arange(n_corr, dtype=float)
    idx_g = j - ks + (n - 1)  # float indices for the g factor
    idx_h = j + ks - (n - 1)  # float indices for the h factor
    
    valid_g = (idx_g >= 0) & (idx_g < n)
    valid_h = (idx_h >= 0) & (idx_h < n)
    
    # For k ≠ n-1: slope = (1-h[idx_g]) - h[idx_h]
    slopes = np.zeros(n_corr)
    ig = idx_g.astype(int)
    ih = idx_h.astype(int)
    ig_clipped = np.clip(ig, 0, n-1)
    ih_clipped = np.clip(ih, 0, n-1)
    slopes = np.where(valid_g, 1.0 - h[ig_clipped], 0.0) - np.where(valid_h, h[ih_clipped], 0.0)
    
    # For k = n-1, the relationship is quadratic; set slope to 0 for now
    slopes[n-1] = 0.0
    
    bases = corr_orig - h0 * slopes
    
    # Sample x
    xs = np.linspace(0, 1, 256)
    # Linear part: bases[k] + slopes[k]*x for k ≠ n-1
    lin = bases[None, :] + slopes[None, :] * xs[:, None]  # (256, n_corr)
    lin[:, n-1] = -np.inf  # exclude k=n-1
    
    quad = corr_orig[n-1] + xs * (1 - xs) - h0 * (1 - h0)
    
    f = np.maximum(np.max(lin, axis=1), quad)
    best_i = np.argmin(f)
    best_x = xs[best_i]
    
    # Refine
    if 0 < best_i < len(xs) - 1:
        dx = xs[1] - xs[0]
        xs_fine = np.linspace(best_x - dx, best_x + dx, 64)
        xs_fine = np.clip(xs_fine, 0, 1)
        lin_f = bases[None, :] + slopes[None, :] * xs_fine[:, None]
        lin_f[:, n-1] = -np.inf
        quad_f = corr_orig[n-1] + xs_fine * (1 - xs_fine) - h0 * (1 - h0)
        f_f = np.maximum(np.max(lin_f, axis=1), quad_f)
        best_x = xs_fine[np.argmin(f_f)]
    
    return best_x

def pair_optimize(h, j, l, n):
    """Find (x, y) = (h[j], h[l]) minimizing max_k corr_k, 2D search."""
    g = 1.0 - h
    corr_orig = np.correlate(h, g, mode='full')
    h0_j, h0_l = h[j], h[l]
    n_corr = 2 * n - 1
    
    # This is more complex; use a simple 2D grid
    xs = np.linspace(0, 1, 32)
    ys = np.linspace(0, 1, 32)
    best_val = np.inf
    best_x, best_y = h0_j, h0_l
    
    for x in xs:
        for y in ys:
            h[j], h[l] = x, y
            c = np.max(np.correlate(h, 1.0-h, mode='full'))
            if c < best_val:
                best_val = c
                best_x, best_y = x, y
    h[j], h[l] = best_x, best_y
    return h

def sim_anneal_block(h, n, rng, n_iters, T_start, T_end):
    """Simulated annealing with block moves."""
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(n_iters):
        T = T_start * (T_end/T_start) ** (it / max(1, n_iters-1))
        L = rng.randint(3, min(20, n//4))  # block length
        start = rng.randint(0, n - L + 1)
        
        h_new = h.copy()
        # Random perturbation of block
        h_new[start:start+L] += rng.normal(0, 0.1, L)
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new, n)
        
        c5_new = compute_c5(h_new)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

    rng = np.random.RandomState(seed + 7)

        h = slsqp_step(h, n, T, 500)

    # Phase 5: Final coordinate descent + SLSQP

        h = slsqp_step(h, n, 0.0001, 1000)
        h = exact_coord_descent(h, n, n_sweeps=10)
        c5 = compute_c5(h)
    
    # Phase 6: Try different starting points if time remains
    # ... (skipped for brevity, depends on time)
    
    h = fix_sum(h, n)
    return (h, compute_c5(h), n)
