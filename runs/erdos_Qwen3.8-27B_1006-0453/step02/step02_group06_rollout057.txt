def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, assume_sorted=True)
    return proj_constraint(h_new, new_n)

def block_nelder_mead(h0, n, t_start, time_budget, block_size=6):
    """Optimize blocks of variables using Nelder-Mead."""
    rng = np.random.RandomState(42)
    h = h0.copy()
    c_best = compute_c5(h)
    h_best = h.copy()
    
    def obj_block(i_start, vals):
        h_try = h.copy()
        h_try[i_start:i_start+block_size] = vals
        h_try = proj_constraint(h_try, n)
        return compute_c5(h_try)
    
    n_blocks = n // block_size
    for epoch in range(200):
        if time.time() - t_start > time_budget:
            break
        # Shuffle block order
        order = rng.permutation(n_blocks)
        for bi in order:
            if time.time() - t_start > time_budget:
                break
            i_start = bi * block_size
            x0 = h[i_start:i_start+block_size]
            res = opt.minimize(obj_block, x0, args=(i_start,), method='Nelder-Mead',
                              options={'maxiter': 50, 'xatol': 1e-6, 'fatol': 1e-8, 'adaptive': True})
            if res.fun < c_best:
                h[i_start:i_start+block_size] = res.x
                h = proj_constraint(h, n)
                c_best = res.fun
                h_best = h.copy()
    
    return h_best, c_best

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_sm(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    if T < 1e-15:
        return m * dx
    shifted = c - m
    return (m + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def grad_sm(h, T, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    if T < 1e-15:
        idx = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
    else:
        w = np.exp((c - m) / T)
        w /= w.sum()
    # grad[i] = dx * sum_k w[k] * [(1-h[i+k-n+1])*I1 - h[i-k+n-1]*I2]
    # Vectorized: for each i, sum over m of w[m-i+n-1]*(1-h[m]) - w[i-m+n-1]*h[m]
    one_h = 1.0 - h
    grad = np.zeros(n)
    # Use FFT-based cross-correlation for speed
    # term1[i] = sum_m w[m-i+n-1] * one_h[m]  = cross_corr(w, one_h)[2n-2] at offset...
    # Simpler: use direct vectorized computation
    # For n~336, O(n^2) is ~112K ops, fast enough
    for i in range(n):
        k1 = np.arange(n) - i + (n - 1)  # = m - i + n - 1 for m=0..n-1
        k1 = np.clip(k1, 0, 2 * n - 2)
        k2 = i - np.arange(n) + (n - 1)  # = i - m + n - 1 for m=0..n-1
        k2 = np.clip(k2, 0, 2 * n - 2)
        grad[i] = np.dot(w[k1], one_h) - np.dot(w[k2], h)
    grad *= dx
    return grad

def proj_grad_descent(h0, n, T, max_iters, lr0, t_start, time_limit):
    h = h0.copy()
    mom = np.zeros(n)
    beta = 0.85
    lr = lr0
    best_c = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        g = grad_sm(h, T, dx)
        # Adaptive step: normalize gradient
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-15:
            g = g / gnorm
        mom = beta * mom + g
        h_new = h + lr * mom
        h = proj(h_new, n)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 50 == 0:
            lr *= 0.97
    return best_h, best_c

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    res = opt.minimize(obj_sm, h0, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return proj(res.x, n)

def block_nm(h0, n, t_start, time_limit, bs=4, epochs=100):
    rng = np.random.RandomState(7)
    h = h0.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    n_blk = n // bs
    for ep in range(epochs):
        if time.time() - t_start > time_limit:
            break
        order = rng.permutation(n_blk)
        for bi in order:
            if time.time() - t_start > time_limit:
                break
            i0 = bi * bs
            x0 = h[i0:i0 + bs].copy()
            def obj_b(x):
                ht = h.copy()
                ht[i0:i0 + bs] = x
                ht = proj(ht, n)
                return obj_sm(ht, 0.003)
            r = opt.minimize(obj_b, x0, method='Nelder-Mead',
                            options={'maxiter': 40, 'xatol': 1e-7, 'fatol': 1e-9, 'adaptive': True})
            if r.fun < best_c:
                h[i0:i0 + bs] = r.x
                h = proj(h, n)
                best_c = compute_c5(h)
                best_h = h.copy()
    return best_h, best_c

def upsample(h, new_n):
    old_n = len(h)
    xo = np.linspace(0, 2, old_n, endpoint=False)
    xn = np.linspace(0, 2, new_n, endpoint=False)
    return proj(np.interp(xn, xo, h, left=h[0], right=h[-1]), new_n)

    h = h0.copy()
    c_cur = compute_c5(h)

    best_h = h.copy()

        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T):
            h = h_try
            c_cur = c_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
        T *= 0.996
    return best_h, best_c

            h_best, c5_best = h_opt, c

    # Final gradient descent micro-polish
    if remaining() > 15:
        h_opt, c_opt = proj_grad_descent(h_best.copy(), n_best, 0.0005, 500, 0.002, t0, budget_s - 5)
        if c_opt < c5_best:
            h_best, c5_best = h_opt, c_opt

    h_best = proj(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
