            h[interior] += diff / max(1, int(np.sum(interior)))
            h = np.clip(h, 0, 1)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Aggressive p-norm at parent resolution
    for p in [15, 40, 100, 300, 800]:
        if time.time() - t0 > budget_s - 120:
            break
        h_opt = do_slsqp_lp(h_best, n_best, p, 1200)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 2: LSE gradient descent with decreasing T
    for T in [0.02, 0.008, 0.003, 0.001, 0.0004]:
        if time.time() - t0 > budget_s - 100:
            break
        h_opt = do_slsqp_lse(h_best, n_best, T, 1500)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 3: Try n=420
    if time.time() - t0 < budget_s - 180:
        n2 = 420
        h2 = upsample(h_best, n2)
        for p in [30, 100, 400]:
            if time.time() - t0 > budget_s - 100:
                break
            h_opt = do_slsqp_lp(h2, n2, p, 900)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c
                n_best = n2
                h2 = h_opt.copy()
        for T in [0.005, 0.002, 0.0006]:
            if time.time() - t0 > budget_s - 60:
                break
            h_opt = do_slsqp_lse(h_best, n_best, T, 1200)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c
                n_best = n_best

    # Phase 4: Try n=392
    if time.time() - t0 < budget_s - 120:
        n3 = 392
        if n3 != n_best:
            h3 = upsample(h_best, n3)
            for p in [40, 150]:
                if time.time() - t0 > budget_s - 60:
                    break
                h_opt = do_slsqp_lp(h3, n3, p, 800)
                c = compute_c5(h_opt)
                if c < c5_best:
                    h_best = h_opt
                    c5_best = c
                    n_best = n3
                    h3 = h_opt.copy()
            if time.time() - t0 < budget_s - 40:
                h_opt = do_slsqp_lse(h_best, n_best, 0.002, 1000)
                c = compute_c5(h_opt)
                if c < c5_best:
                    h_best = h_opt
                    c5_best = c

    # Phase 5: Exchange-based local search
    if time.time() - t0 < budget_s - 60:
        h_ex, c_ex = exchange_search(h_best, n_best, t0, budget_s, 2500, seed + 123)
        if c_ex < c5_best:
            h_best = h_ex
            c5_best = c_ex

    # Phase 6: Final ultra-fine LSE polish
    if time.time() - t0 < budget_s - 25:
        for T in [0.0003, 0.0001]:
            if time.time() - t0 > budget_s - 10:
                break
            h_opt = do_slsqp_lse(h_best, n_best, T, 3000)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    h_best = project(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
