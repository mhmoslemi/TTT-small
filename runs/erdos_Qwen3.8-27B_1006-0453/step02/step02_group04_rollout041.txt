```python
def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    # Adjust uniformly on interior points
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)

        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)

                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add

            idx = np.argsort(-h)

                dd -= rem
                if dd <= 1e-12: break
```

```python
def projected_grad_descent(h, n, top_k=10, T_init=0.05, T_final=0.001, max_iter=5000, ...):
    # Subgradient method with adaptive step size
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    c5 = best_c5
    alpha = 0.1  # initial step size
    
    for it in range(max_iter):
        # Compute correlations
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_cur = np.max(corr) * (2.0/n)
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        
        # Identify top-K shifts (in terms of correlation values)
        # Note: corr is indexed 0..2n-2, shift s = idx - (n-1)
        top_idx = np.argsort(corr)[-top_k:]
        
        # Compute soft-max weight
        T = T_init * (T_final/T_init) ** (it / max_iter)
        top_vals = corr[top_idx]
        M = np.max(top_vals)
        shifted = (top_vals - M) / T
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Compute gradient (weighted sum of top-K gradients)
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            s = ki - (n-1)  # shift
            grad += wi * grad_for_shift(h_cur, s)
        
        # Projected gradient step with line search
        for _ in range(5):
            h_trial = np.clip(h_cur - alpha * grad, 0.0, 1.0)
            h_trial = fix_sum(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur:
                break
            alpha *= 0.5
        
        h_cur = h_trial
        c5 = c5_trial        
        
        # Decay and adapt step size
        alpha *= 0.999
    
    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:

            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-13: break
        else:

            idx = np.argsort(-h)
            for i in idx:
                rem = min(dd, h[i])
                h[i] -= rem
                dd -= rem
                if dd <= 1e-13: break

    best_c5 = compute_c5(h_cur)

        top_vals = corr[top_idx]
```

- Analytical gradient (fast)
- Can do many more iterations
- But: may converge slowly near the optimum, may not handle the max function well

Also, I should consider: the SLSQP in the previous code was using numerical gradients which for n=360 means 360+1 function evaluations per gradient step. Each function evaluation calls np.correlate which is O(n^2) for mode='full'. So one SLSQP iteration costs O(n^3) = O(360^3) ≈ 47M operations. In contrast, PGD with top_k=20 costs O(n * top_k) = O(7200) per iteration. So PGD is ~6000x faster per iteration!

This means I can do many more PGD iterations. The question is whether it converges well enough.

Let me also think about the "active set" strategy more carefully. In the optimal solution, typically a small number of shifts achieve the maximum. If I can identify them and focus on just those, the problem becomes a finite-dimensional optimization.

Another idea: **Alternating minimization**. Fix the active set, minimize the sum of those overlaps (which is a quadratic), then update the active set. This is like Benders decomposition.

For a fixed set of shifts S, minimize:
min_h sum_{s in S} corr_s(h) s.t. 0<=h<=1, sum(h)=n/2

The objective sum_s corr_s = sum_s sum_i h[i](1-h[i+s]) is quadratic in h.

The gradient of sum_s corr_s is:
sum_s [(1-h[j+s]) - h[j-s]] (with validity checks)

This is a smooth (quadratic) objective, so I can use proper gradient methods (L-BFGS, conjugate gradient, etc.) much more effectively.

Let me implement this:

```python
def active_set_optimize(h, n, S, max_iter=2000, time_limit=30):
    """Minimize sum of correlations at shifts in S using L-BFGS-B."""
    import scipy.optimize as opt
    
    def obj(x):
        total = 0.0
        for m in S:
            s = m - (n - 1)
            if s >= 0:
                total += np.sum(x[:n-s] * (1.0 - x[s:]))
            else:
                t = -s
                total += np.sum(x[t:] * (1.0 - x[:n-t]))
        return total
    
    def jac(x):
        grad = np.zeros(n)
        for m in S:
            s = m - (n - 1)
            if s >= 0:
                grad[:n-s] += (1.0 - x[s:])
                grad[s:] -= x[:n-s]
            else:
                t = -s
                grad[t:] += (1.0 - x[:n-t])
                grad[:n-t] -= x[t:]
        return grad
    
    res = opt.minimize(obj, h.copy(), jac=jac, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
    return fix_sum(res.x, n)
```

This is much better than SLSQP because:
1. We have analytical gradient (no finite differences)
2. L-BFGS-B handles box constraints natively
3. The objective is smooth (quadratic) for a fixed active set

The algorithm becomes:
1. Compute all correlations
2. Find top-K active shifts
3. Optimize sum of those K correlations using L-BFGS-B
4. Re-evaluate correlations, update active set
5. Repeat

This is the Benders/active-set approach and should be much more effective than gradient descent on the non-smooth max.

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-7) & (h < 1.0 - 1e-7)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.45:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            for i in np.argsort(h):
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            dd = -diff
            for i in np.argsort(-h):
                rem = min(dd, h[i])
                h[i] -= rem
                dd -= rem
                if dd <= 1e-13:
                    break
    return h

def pgd_step(h, corr, top_k, T, alpha, v, momentum, n):
    """One PGD step with momentum."""
    top_idx = np.argsort(corr)[-top_k:]
    top_vals = corr[top_idx]
    M = np.max(top_vals)
    exps = np.exp(np.clip((top_vals - M) / T, -60.0, 0.0))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        s = ki - (n - 1)
        if s >= 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n-s]
        else:
            t = -s
            grad[t:] += wi * (1.0 - h[:n-t])
            grad[:n-t] -= wi * h[t:]
    v = momentum * v - alpha * grad
    h_new = np.clip(h + v, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new, v, alpha
