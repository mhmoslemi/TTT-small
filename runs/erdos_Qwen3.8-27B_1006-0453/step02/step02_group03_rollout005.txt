        per_tl = min(80.0, max(15.0, (dl - 15 - time.time()) / 3.0))
        h, c = opt_alpha(h0, dxt, tst, time.time() + per_tl)
        if c < best_c:
            best_h, best_c = h, c
        if time.time() > dl - 20:
            break

    if time.time() < dl - 35:
        nb = len(best_h)
        dxb, tsb = 2.0 / nb, nb / 2.0
        for amp in (0.12, 0.2, 0.06, 0.15, 0.25):
            if time.time() > dl - 15:
                break
            hp = proj(best_h + amp * rng.randn(nb), tsb)
            h, c = opt_alpha(hp, dxb, tsb, dl - 5)
            if c < best_c:
                best_h, best_c = h, c

    if best_h is not None and time.time() < dl - 10:
        nb = len(best_h)
        dxf, tsf = 2.0 / nb, nb / 2.0
        hf, cf = coord_polish(best_h, dxf, tsf, 15, dl - 3, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = proj(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c, len(best_h)
