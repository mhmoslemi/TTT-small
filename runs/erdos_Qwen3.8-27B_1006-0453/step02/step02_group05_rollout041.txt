def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr - (tau * corr).max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-4
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def adam_step(h, target, grad, lr, b1, b2, eps, m, v, it):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    mh = m / (1 - b1 ** it)
    vh = v / (1 - b2 ** it)
    h_new = h - lr * mh / (np.sqrt(vh) + eps)
    h_new = project_sum(h_new, target)
    return h_new, m, v

def optimize(h, N, target, dx, deadline, tau0, tau1, lr0, decay, decay_every, n_warmup=100):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    max_it = 1000000
    while time.time() < deadline and it < max_it:
        frac = min(it / max(n_warmup * 5, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad, c5_val = softmax_grad(h, N, tau)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        h, m, v = adam_step(h, target, grad, lr, b1, b2, eps, m, v, it + 1)
        it += 1
        if it % decay_every == 0:
            lr *= decay
        if lr < 1e-11:
            break
    return best_h, best_c5

Let me also add a **cycling** strategy: after getting a good solution, identify the top 3 offsets and do alternating minimization on them.

For offset L (the correlation at index center+L):
- If L > 0: corr = Σ_{i=0}^{N-1-L} h[i](1-h[i+L])
- Gradient w.r.t. h[j]: (1-h[j+L]) for j < N-L, and -h[j-L] for j ≥ L
- We want to decrease this, so we move h opposite to the gradient

But we also need to maintain the sum constraint. A simple approach: after each offset-specific step, re-project onto the constraint.

    return np.correlate(h, 1.0 - h, mode='full')

def apply_offset_grad(grad, h, N, L, wi):

    idx = center + L

            idx = np.argpartition(corr, -K)[-K:]
            wk = 1.0 / K
        grad = np.zeros(N)
        for ai in idx:
            apply_offset_grad(grad, h, N, ai - center, wk)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % decay_every == 0:
            lr *= decay
        if lr < 1e-11:
            break
    return best_h, best_c5

def softmax_adam(h, N, target, dx, deadline, tau0, tau1, lr0, decay, decay_every):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    center = N - 1
    while time.time() < deadline:

        c5_val = float(corr.max()) * (2.0 / N)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            grad = np.zeros(N)
        else:

            active = np.where(w > thresh)[0]

                apply_offset_grad(grad, h, N, ai - center, w[ai])
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad

        vh = v / (1 - b2 ** it)

        h = project_sum(h, target)
        if it % decay_every == 0:
            lr *= decay
        if lr < 1e-11:
            break
    return best_h, best_c5

def basin_hopping(h, N, target, dx, deadline, rng, n_hops=5):
    """Perturb and re-optimize to escape local minima."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    per_hop = max(5.0, (deadline - time.time()) / max(n_hops, 1))
    for hop in range(n_hops):
        if time.time() > deadline - 2:
            break
        h_pert = h + rng.randn(N) * 0.03
        h_pert = project_sum(h_pert, target)
        t_end = min(time.time() + per_hop, deadline - 2)
        h_new, c_new = topk_adam(h_pert, N, target, dx, t_end, K=5,
                                  lr0=0.01, decay=0.97, decay_every=30)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
            h = h_new.copy()
    return best_h, best_c5

- h-c is clipped: the result is min(max(h-c, 0), 1)
- sum = count(h > c+1) * 1 + sum(min(max(h-c,0),1) for 0 < h-c < 1) + count(h < c) * 0

def single_offset_ls(h, N, target, dx, deadline, max_iters=10000):
    """Optimize by cycling through top offsets with line search."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    center = N - 1
    it = 0
    alpha = 0.01
    while time.time() < deadline and it < max_iters:
        corr = corr_of(h, N)
        c5_val = float(corr.max()) * (2.0 / N)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        # Get top 3 offsets
        top3 = np.argpartition(corr, -3)[-3:]
        for ti in top3:
            L = ti - center
            g = np.zeros(N)
            apply_offset_grad(g, h, N, L, 1.0)
            # Line search
            best_local_alpha = alpha
            best_local_c5 = c5_val
            for factor in [0.5, 1.0, 2.0]:
                a = alpha * factor
                h_try = h - a * g
                h_try = project_sum(h_try, target)
                c_try = compute_c5(h_try, dx)
                if c_try < best_local_c5:
                    best_local_c5 = c_try
                    best_local_alpha = a
            if best_local_alpha != 0:
                h = project_sum(h - best_local_alpha * g, target)
        # Adaptive alpha
        c5_now = compute_c5(h, dx)
        if c5_now >= c5_val:
            alpha *= 0.7
        elif alpha < 0.05:
            alpha *= 1.1
        it += 1
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_of(h, N):
    return np.correlate(h, 1.0 - h, mode='full')

def apply_offset_grad(grad, h, N, L, wi):
    if L == 0:
        grad += wi * (1.0 - 2.0 * h)
    elif L > 0:
        grad[L:] += wi * (1.0 - h[:N - L])
        grad[:N - L] -= wi * h[L:]
    else:
        M = -L
        grad[:N - M] += wi * (1.0 - h[M:])
        grad[M:] -= wi * h[:N - M]

def topk_adam(h, N, target, dx, deadline, K, lr0, decay, decay_every):

    v = np.zeros(N)

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        kk = min(K, total_offsets)
        if kk >= total_offsets:
            idx = np.arange(total_offsets)
            wk = 1.0 / kk
        else:
            idx = np.argpartition(corr, -kk)[-kk:]
            wk = 1.0 / kk
        grad = np.zeros(N)

        m = b1 * m + (1 - b1) * grad

    """Cyclic line-search on top-K offsets with adaptive step."""

    best_c5 = compute_c5(h, dx)
    center = N - 1
    alpha = 0.02
    stall = 0
    while time.time() < deadline:
        corr = corr_of(h, N)
        c5_val = float(corr.max()) * (2.0 / N)
