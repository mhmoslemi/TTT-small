def proj_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

        cnt = np.sum(mask)
        if cnt > 0:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    val = c_max + T * np.log(np.sum(np.exp((corr - c_max) / T)))
    return val * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
