def lse_obj_and_grad(h, t, pen, topk=None):

    mx = np.max(s)
    exp_s = np.exp(s - mx)
    sum_exp = np.sum(exp_s * L)  # Wait, this doesn't work for topk
    # For topk: only consider the top k values
    # LSE of topk = max(topk) + log(sum(exp(topk - max)))/... 
    # Actually for topk LSE:
    if topk is not None and topk < 2*n-1:
        c_top = corr[np.argsort(corr)[-topk:]]
        s_top = t * c_top
        mx_top = np.max(s_top)
        lse_val = (mx_top + np.log(np.sum(np.exp(s_top - mx_top)))) / t
        # For gradient, need softmax over topk
        P = np.exp(s_top - mx_top) / np.sum(np.exp(s_top - mx_top))
        # Map P back to full corr indices

        P_full[np.argsort(corr)[-topk:]] = P
    else:
        exp_s = np.exp(s - mx)
        P_full = exp_s / np.sum(exp_s)
        lse_val = (mx + np.log(np.sum(exp_s))) / t
    
    val = lse_val * dx + pen * (np.sum(h) - n/2.0)**2
    
    # Gradient
    grad = np.zeros(n)
    # For each lag k (index k_idx in 0..2n-2), k = k_idx - (n-1)
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        Pk = P_full[k_idx]
        if Pk == 0:
            continue
        # Term 1: (1-h[j-k]) for valid j
        # j range: if k>=0: [k, n-1]; if k<0: [0, n-1+k]
        if k >= 0:
            j_lo, j_hi = k, n-1
        else:

        if j_hi >= j_lo:

            grad[js] += Pk * t * (1.0 - h[js - k])
        # Term 2: -h[j+k] for valid j
        # j range: if k>=0: [0, n-1-k]; if k<0: [-k, n-1]
        if k >= 0:

        else:

        if j_hi2 >= j_lo2:
            js2 = np.arange(j_lo2, j_hi2+1)
            grad[js2] += Pk * t * (-h[js2 + k])
    
    grad *= dx
    grad += 2.0 * pen * (np.sum(h) - n/2.0)
    
    return val, grad

Let me also think about the binary local search more carefully. For binary h with exactly n/2 ones:

The key insight is that for a binary solution, the correlation at lag k is:
corr[k] = #{i : h[i+k]=1, h[i]=0} (with appropriate bounds)

This is the number of "1→0" transitions at shift k. We want to minimize the maximum over k.

For a move (1 from i to j):
- Remove 1 from i: affects corr[k] for all k where i is involved
- Add 1 at j: affects corr[k] for all k where j is involved

The net change in corr[k]:
- From removing 1 at i:
  - If i+k was contributing (h[i+k]=1, h[i]=0 was false since h[i]=1): actually wait.
  - Before: h[i]=1. After: h[i]=0.
  - corr[k] = sum h[n+k]*(1-h[n]). The term with n=i: h[i+k]*(1-h[i]). Before: h[i+k]*0=0. After: h[i+k]*1=h[i+k].
    So Δ from n=i: +h[i+k] (if i+k valid)
  - The term with n+k=i, i.e., n=i-k: h[i]*(1-h[i-k]). Before: 1*(1-h[i-k]). After: 0*(1-h[i-k])=0.
    So Δ from n=i-k: -(1-h[i-k]) (if i-k valid)
  
- From adding 1 at j:
  - The term with n=j: h[j+k]*(1-h[j]). Before: h[j+k]*1=h[j+k]. After: h[j+k]*0=0.
    So Δ: -h[j+k] (if j+k valid)
  - The term with n+k=j, n=j-k: h[j]*(1-h[j-k]). Before: 0*(1-h[j-k])=0. After: 1*(1-h[j-k]).
    So Δ: +(1-h[j-k]) (if j-k valid)

Total Δcorr[k] = h[i+k]*I(i+k valid) - (1-h[i-k])*I(i-k valid) - h[j+k]*I(j+k valid) + (1-h[j-k])*I(j-k valid)

For binary h, this is:
= h[i+k] - (1-h[i-k]) - h[j+k] + (1-h[j-k])  (with validity)

This is a simple expression. For the simulated annealing, I need to track the max and update it efficiently.

Actually, for the SA, I'll just recompute the full corr after each move (O(n) per move) since n=280 is small. With 10K-50K iterations, that's 2.8M-14M ops. Fine.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    """Convert to binary with exactly n/2 ones"""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        # Randomly set some 1s to 0s
        ones_idx = np.where(h_bin == 1)[0]
        rng_remove = np.random.permutation(len(ones_idx))[:n_ones - n//2]
        h_bin[ones_idx[rng_remove]] = 0
    elif n_ones < n // 2:
        zeros_idx = np.where(h_bin == 0)[0]
        rng_add = np.random.permutation(len(zeros_idx))[:n//2 - n_ones]
        h_bin[zeros_idx[rng_add]] = 1
    return h_bin

def lse_obj_grad(h, t, pen, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if topk is not None and topk < 2*n-1:
        topk_idx = np.argsort(corr)[-topk:]
        c_top = corr[topk_idx]
        s_top = t * c_top
        mx_top = np.max(s_top)
        exp_top = np.exp(s_top - mx_top)
        sum_exp_top = np.sum(exp_top)
        lse_val = (mx_top + np.log(sum_exp_top)) / t
        P_full = np.zeros(2*n-1)
        P_full[topk_idx] = exp_top / sum_exp_top
    else:
        s = t * corr
        mx = np.max(s)
        exp_s = np.exp(s - mx)
        sum_exp = np.sum(exp_s)
        lse_val = (mx + np.log(sum_exp)) / t
        P_full = exp_s / sum_exp
    
    val = lse_val * dx + pen * (np.sum(h) - n/2.0)**2
    
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        Pk = P_full[k_idx]
        if Pk < 1e-15:
            continue
        k = k_idx - (n-1)
        # Term 1: (1-h[j-k]) for valid j where n=j-k is valid in corr sum
        if k >= 0:
            js = np.arange(k, n)
        else:
            js = np.arange(0, n+k)
        if len(js) > 0:
            grad[js] += Pk * t * (1.0 - h[js - k])
        # Term 2: -h[j+k] for valid j where n=j is valid in corr sum
        if k >= 0:
            js2 = np.arange(0, n-k)
        else:
            js2 = np.arange(-k, n)
        if len(js2) > 0:
            grad[js2] += Pk * t * (-h[js2 + k])
    
    grad *= dx
    grad += 2.0 * pen * (np.sum(h) - n/2.0)
    
    return val, grad

def sa_binary(h_bin, n, rng, n_iter, T_init, T_final):
    """Simulated annealing on binary configs"""
    cur_c5 = compute_c5(h_bin)
    best_c5 = cur_c5
    best_h = h_bin.copy()
    dx = 2.0 / n
    
    for it in range(n_iter):
        T = T_init * (T_final / T_init) ** (it / max(1, n_iter - 1))
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_new = h_bin.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_c5 = compute_c5(h_new)
        
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            h_bin = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        perm = rng.permutation(len(ones_idx))[:n_ones - target]
        h_bin[ones_idx[perm]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        perm = rng.permutation(len(zeros_idx))[:target - n_ones]
        h_bin[zeros_idx[perm]] = 1
    return h_bin

def _lse_fun(x, t, pen, topk):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if topk is not None and topk < 2 * n - 1:
        idx = np.argsort(corr)[-topk:]
        ct = corr[idx]
        st = t * ct
        mx = np.max(st)
        et = np.exp(st - mx)
        lse = (mx + np.log(np.sum(et))) / t
    else:
        st = t * corr
        mx = np.max(st)
        et = np.exp(st - mx)
        lse = (mx + np.log(np.sum(et))) / t
    return lse * dx + pen * (np.sum(x) - n / 2.0) ** 2

def _lse_grad(x, t, pen, topk):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if topk is not None and topk < 2 * n - 1:
        idx = np.argsort(corr)[-topk:]
        ct = corr[idx]
        st = t * ct
        mx = np.max(st)
        et = np.exp(st - mx)
        P_full = np.zeros(2 * n - 1)
        P_full[idx] = et / np.sum(et)
    else:
        st = t * corr
        mx = np.max(st)
        et = np.exp(st - mx)
        P_full = et / np.sum(et)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        Pk = P_full[ki]
        if Pk < 1e-15:
            continue
        k = ki - (n - 1)
        if k >= 0:
            js = np.arange(k, n)
        else:
            js = np.arange(0, n + k)
        if len(js) > 0:
            grad[js] += Pk * t * (1.0 - x[js - k])
        if k >= 0:
            js2 = np.arange(0, n - k)
        else:
            js2 = np.arange(-k, n)
        if len(js2) > 0:
            grad[js2] += Pk * t * (-x[js2 + k])
    grad *= dx
    grad += 2.0 * pen * (np.sum(x) - n / 2.0)
    return grad

def _lse_argfun(x, t, pen, topk):
    args = (t, pen, topk)
    return _lse_fun(x, *args)

def _lse_arggrad(x, t, pen, topk):
    args = (t, pen, topk)
    return _lse_grad(x, *args)

def sa_binary(h_bin, n, rng, n_iter, T_init, T_final):
    cur_c5 = compute_c5(h_bin)
    best_c5 = cur_c5
    best_h = h_bin.copy()
    for it in range(n_iter):
        T = T_init * (T_final / T_init) ** (it / max(1, n_iter - 1))
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    return best_h, best_c5
