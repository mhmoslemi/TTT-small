    max_c = corr.max()
    exp_args = np.minimum(tau * (corr - max_c), 500.0)

def lbfgs_step(h, N, dx, tau, lam, maxiter, time_limit):
    """Run L-BFGS-B on the smooth objective."""
    t_start = time.time()
    
    def func(x):
        if time.time() - t_start > time_limit:
            raise TimeoutError()
        val, _ = lse_obj_and_grad(x, N, dx, tau, lam)
        return val
    
    def grad_func(x):
        if time.time() - t_start > time_limit:
            raise TimeoutError()
        _, g = lse_obj_and_grad(x, N, dx, tau, lam)
        return g
    
    bounds = [(0.0, 1.0)] * N
    try:

                       bounds=bounds, options={'maxiter': maxiter, 'maxfun': maxiter*3,
                                               'ftol': 1e-10, 'gtol': 1e-8})
    except (TimeoutError, Exception):

    new_h = res.x

def compute_c5(h, dx):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, N, dx, tau, lam):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    ea = np.minimum(tau * (corr - max_c), 500.0)
    w = np.exp(ea)
    ws = w.sum()

    w = w / ws
