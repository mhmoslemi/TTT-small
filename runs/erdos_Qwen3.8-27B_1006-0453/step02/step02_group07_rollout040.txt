    dx = 2.0 / len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    Rm = np.max(R)
    if T < 1e-15:

def grad_lse(h, T):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    Rm = np.max(R)
    if T < 1e-15:
        w = np.zeros(2 * n - 1)
        mask = R >= Rm - 1e-10

        n2 = min(560, n * 2)
        h2 = upsample_h(h, n2)
        for T in (0.01, 0.003, 0.001, 0.0003, 0.0001):
            if time.time() > t0 + budget_s - 20:
                break
            h2, c52 = pg_descend(h2, T, 5000, 0.8, 0.02, t0, time.time() + 80)
            h2 = fix_sum(h2, n2)
            c52 = compute_c5(h2)
        if c52 < c5:
            h, n, c5 = h2, n2, c52

    h = fix_sum(h, len(h))
    c5 = compute_c5(h)
    return (h, c5, len(h))
