import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    shifted = (corr - c) / T
    shifted = np.clip(shifted, -60, 60)
    return (c + T * np.log(np.sum(np.exp(shifted)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-9) & (h < 1.0 - 1e-9)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-7 < h[i] < 1.0 - 1e-7:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def slsqp_opt(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-18})
        hn = fix_sum(np.clip(r.x, 0, 1), n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def lgbfgs_opt(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = minimize(lse_obj, h, args=(T,), method='L-BFGS-B',
            bounds=[(0., 1.)] * n,
            options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': iters * 3})
        hn = fix_sum(np.clip(r.x, 0, 1), n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def coord_polish(h, n, n_pass=4):
    for _ in range(n_pass):
        improved = False
        for j in range(n):
            if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
                continue
            c0 = compute_c5(h)
            for delta in (-0.06, -0.02, -0.005, 0.005, 0.02, 0.06):
                if h[j] + delta < 1e-5 or h[j] + delta > 1.0 - 1e-5:
                    continue
                hn = h.copy()
                hn[j] += delta
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                hn[mask] -= delta / (n - 1)
                hn = np.clip(hn, 0, 1)
                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < c0 - 1e-13:
                    c0 = c
                    h = hn
                    improved = True
        if not improved:
            break
    return fix_sum(h, n)

def pair_polish(h, n, rng, n_trials=2000):
    for _ in range(n_trials):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
            continue
        if h[l] <= 1e-5 or h[l] >= 1.0 - 1e-5:
            continue
        c0 = compute_c5(h)
        for d in (-0.05, -0.015, -0.004, 0.004, 0.015, 0.05):
            hj2, hl2 = h[j] + d, h[l] - d
            if hj2 < 1e-5 or hj2 > 1.0 - 1e-5 or hl2 < 1e-5 or hl2 > 1.0 - 1e-5:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj2, hl2
            c = compute_c5(hn)
            if c < c0 - 1e-13:
                h = hn
                break
    return fix_sum(h, n)

def upsample(h, nn):
    n = len(h)
    x0 = np.arange(n, dtype=float) * (2.0 / n)
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def downsample(h, nn):
    n = len(h)
    result = np.zeros(nn)
    for i in range(nn):
        lo = int(i * n / nn)
        hi = int((i + 1) * n / nn)
        if hi > lo:
            result[i] = np.mean(h[lo:hi])
        else:
            result[i] = h[lo % n]
    return fix_sum(result, nn)

def block_perturb(h, n, rng, mode):
    ht = h.copy()
    if mode == 0:
        ht += rng.randn(n) * 0.012
    elif mode == 1:
        k = max(3, n // 60)

    elif mode == 2:
        blen = rng.randint(max(3, n // 12), n // 5)
        bstart = rng.randint(0, n - blen)
        block = ht[bstart:bstart+blen].copy()
        shift = rng.randint(1, blen)
        ht[bstart:bstart+blen] = np.roll(block, shift)
    elif mode == 3:
        blen = rng.randint(3, max(4, n // 10))
        b1 = rng.randint(0, n - blen)

            ht[b1:b1+blen], ht[b2:b2+blen] = ht[b2:b2+blen].copy(), ht[b1:b1+blen].copy()
    else:
        blen = rng.randint(2, max(3, n // 8))
        bstart = rng.randint(0, n - blen)
        ht[bstart:bstart+blen] = 1.0 - ht[bstart:bstart+blen]
    return fix_sum(np.clip(ht, 0, 1), n)
