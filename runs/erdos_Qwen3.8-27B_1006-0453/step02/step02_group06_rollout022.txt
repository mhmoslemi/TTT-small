    n = len(h)
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    k_max = np.argmax(corr)
    # Convert k_max to 0-indexed lag: lag = k_max - (n-1)

    elif lag > 0:

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_max_and_grad(h):
    n = len(h)
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        m = -lag
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return corr, lag, grad

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = (corr - c_max) / T
    shifted -= np.max(shifted)  # numerical stability
    return (c_max + T * np.log(np.sum(np.exp(shifted)))) * dx

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

    corr, lag, grad = corr_max_and_grad(h)
    if momentum_buf is None:
        momentum_buf = np.zeros_like(h)
    momentum_buf = beta * momentum_buf + (1 - beta) * grad
    h_new = h - lr * momentum_buf
    h_new = proj_constraint(h_new, len(h))
    return h_new, momentum_buf

def lbfgs_optimize(h0, T, max_iter, n):
    def obj(x):
        return soft_max_obj(x, T)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n*0.5,
             'jac': lambda x: np.ones(n)}]
    res = opt.minimize(obj, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
    return proj_constraint(res.x, n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] = (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] = (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def weighted_subgrad(h, topk=3):
    corr = get_corr(h)
    n = len(h)
    idx = np.argsort(corr)[::-1][:topk]
    vals = corr[idx]

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    if T < 1e-14:
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)
        g = grad_at_lag(h, lag)
    else:
        sh = (corr - cm) / T
        sh -= sh.max()
        w = np.exp(sh)
        w /= w.sum()
        g = np.zeros(n)
        # Only compute gradient for lags with significant weight
        threshold = 1e-6
        active = w > threshold
        for i in range(len(w)):
            if active[i]:
                lag = i - (n - 1)
                g += w[i] * grad_at_lag(h, lag)
    s = np.sum(h) - 0.5 * n
    g += 2.0 * pen * s * np.ones(n)
    return g * (2.0 / n)  # multiply by dx

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return proj_constraint(np.interp(x_new, x_old, h), new_n)

def downsample(h, new_n):
    old_n = len(h)
    idx = np.minimum((np.arange(new_n) * old_n // new_n), old_n - 1)
    return proj_constraint(h[idx].copy(), new_n)

def subgrad_polish(h0, n_steps, lr0, t_start, time_limit, topk=3):
    h = h0.copy()
    n = len(h)
    mom = np.zeros(n)
    beta = 0.85
    c_best = compute_c5(h)
    h_best = h.copy()
    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        g, _ = weighted_subgrad(h, topk)
        lr = lr0 / (1.0 + 0.01 * step)
        mom = beta * mom + (1.0 - beta) * g
        h = proj_constraint(h - lr * mom, n)
        c = compute_c5(h)
        if c < c_best:
            c_best = c
            h_best = h.copy()
    return h_best, c_best

def do_slsqp(h0, n, T, pen, max_iter):
    res = opt.minimize(soft_max_obj, h0, args=(T, pen),
                       jac=lambda x: soft_max_jac(x, T, pen),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return proj_constraint(res.x, n)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] = 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] = 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def weighted_subgrad(h, topk=3):
    corr = get_corr(h)
    n = len(h)
    idx = np.argsort(corr)[::-1][:topk]
    vals = corr[idx]
    w = np.exp((vals - vals[0]) / 1.0)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(topk):
        lag = int(idx[i]) - (n - 1)
        g += w[i] * grad_at_lag(h, lag)
    return g, float(corr[idx[0]])

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = int(np.sum(interior))
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, T, pen):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(corr)
    if T < 1e-14:
        base = cm
    else:
        sh = (corr - cm) / T
        sh -= sh.max()
        base = cm + T * np.log(np.exp(sh).sum())
    s = np.sum(x) - 0.5 * n
    return base * dx + pen * s * s

def upsample(h, new_n):

    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return proj_constraint(np.interp(x_new, x_old, h), new_n)

def downsample(h, new_n):
    old_n = len(h)
    idx = np.minimum((np.arange(new_n) * old_n) // new_n, old_n - 1)
    return proj_constraint(h[idx].copy(), new_n)

def subgrad_polish(h0, n_steps, lr0, t_start, time_limit, topk=3, beta=0.82):

    n = len(h)
    mom = np.zeros(n)
    c_best = compute_c5(h)
    h_best = h.copy()
    stall = 0
    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        g, _ = weighted_subgrad(h, topk)
        lr = lr0 / (1.0 + 0.005 * step)
        mom = beta * mom + (1.0 - beta) * g
        h = proj_constraint(h - lr * mom, n)
        c = compute_c5(h)
        if c < c_best:
            c_best = c
            h_best = h.copy()
            stall = 0
        else:
