import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, N, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    center = N - 1
    top_idx = np.argsort(corr)[::-1][:K]
    grad = np.zeros(N)
    for ai in top_idx:
        L = ai - center
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] += (1.0 - h[L:N])
            grad[L:] -= h[:N - L]
        else:
            M = -L
            grad[M:] += (1.0 - h[:N - M])
            grad[:N - M] -= h[M:]
    return grad / float(K), true_c5

def fista_opt(h, N, target, deadline, K0, K1, lr0, lr1, max_it=200000):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    x = h.copy()
    z = h.copy()
    t_mom = 1.0
    it = 0
    while time.time() < deadline and it < max_it:
        it += 1
        frac = min(it / max(max_it, 1), 1.0)
        K = max(3, int(K0 + (K1 - K0) * frac))
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5_val = topk_grad(z, N, K)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = z.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = z - lr * grad
        y = y + ((t_mom - 1.0) / t_new) * (z - x)
        x = z
        z = project_sum(y, target)
        t_mom = t_new
        if it % 5000 == 0 and c5_val > best_c5 * 1.001:
            x = z.copy()
            t_mom = 1.0
    return best_h, best_c5

def pair_swap_refine(h, N, target, n_trials, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    dx_val = 2.0 / N
    deltas = [0.01, 0.02, 0.05, 0.1]
    for _ in range(n_trials):
        i, j = int(rng.randint(N)), int(rng.randint(N))
        if i == j:
            continue
        for delta in deltas:
            for sign in [1, -1]:
                d = sign * delta
                if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                    continue
                h2 = h.copy()
                h2[i] += d
                h2[j] -= d
                c5 = compute_c5(h2, dx_val)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h2
                    h = h2
                    break
            else:
                continue
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    best_c5 = compute_c5(best_h, 2.0 / n0)

    sizes = sorted(set([n0, 2 * n0, 460, 520, 572, 640]))
    sizes = [s for s in sizes if s % 2 == 0 and 20 <= s <= 800]

    for N in sizes:
        if time.time() > deadline - 50:

        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, best_h)
        h_init = project_sum(h_init, target)
        h_pert = h_init + rng.randn(N) * 0.04
        h_pert = project_sum(h_pert, target)

        for h_start in [h_init, h_pert]:
            if time.time() > deadline - 30:

            remain = deadline - time.time()
            if remain < 15:

            # Phase 1: coarse FISTA
            d1 = time.time() + remain * 0.40
            h1, c1 = fista_opt(h_start.copy(), N, target, d1,
                               K0=5, K1=40, lr0=0.04, lr1=0.008, max_it=100000)
            # Phase 2: fine FISTA
            if time.time() < deadline - 15:
                d2 = min(time.time() + (deadline - time.time()) * 0.40, deadline - 5)
                h2, c2 = fista_opt(h1.copy(), N, target, d2,
                                   K0=40, K1=120, lr0=0.008, lr1=0.001, max_it=80000)
                if c2 < c1:
                    h1, c1 = h2, c2
            # Pair-swap refinement
            if time.time() < deadline - 3:
                n_sw = min(3000, max(200, int(100.0 / max(deadline - time.time(), 0.01))))
                h3, c3 = pair_swap_refine(h1.copy(), N, target, n_sw, seed=seed + N)
                if c3 < c1:
                    h1, c1 = h3, c3
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()

    # Final polish on best
    Nb = len(best_h)
    if time.time() < deadline - 5:
        target_b = Nb / 2.0
        h_f, c_f = fista_opt(best_h.copy(), Nb, target_b, deadline - 2,
                             K0=60, K1=150, lr0=0.003, lr1=0.0005, max_it=50000)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
