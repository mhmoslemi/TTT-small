from scipy import optimize
import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-9) & (h < 1.0 - 1e-9)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-10:
        m2 = (h > 0) & (h < 1.0)
        cnt = np.sum(m2)
        if cnt > 0:
            h[m2] += d2 / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    return cmax + T * np.log(np.sum(np.exp((c - cmax) / T)))

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    w = np.exp((c - cmax) / T)
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        wk = w[idx]
        if k >= 0:
            if n - k > 0:
                m1 = np.arange(k, n)
                grad[m1] += wk * (1.0 - h[m1 - k])
                m2 = np.arange(0, n - k)
                grad[m2] -= wk * h[m2 + k]
        else:
            ks = -k
            if n - ks > 0:
                m1 = np.arange(0, n - ks)
                grad[m1] += wk * (1.0 - h[m1 + ks])
                m2 = np.arange(ks, n)
                grad[m2] -= wk * h[m2 - ks]
    return grad * dx

def lbfgs_step(h, n, T, maxiter=2000):
    def obj(x):
        return smooth_obj(x, T)
    def objgrad(x):
        return smooth_obj(x, T), smooth_grad(x, T)
    try:
        res = optimize.minimize(objgrad, h, jac=True, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5*n}],

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-9) & (h < 1.0 - 1e-9)
    k = np.sum(m)
    if k > 0:
        step = d / k
        h[m] = np.clip(h[m] + step, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-10:
        m2 = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(m2)
        if cnt > 0:
            h[m2] = np.clip(h[m2] + d2 / cnt, 0.0, 1.0)
    return h

def smooth_obj_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    return cmax + T * np.log(np.sum(np.exp((c - cmax) / T)))

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    logw = (c - cmax) / T
    logw -= np.max(logw)
    w = np.exp(logw)
    w /= np.sum(w)
    grad = np.zeros(n)
    active = w > 1e-8
    for idx in np.where(active)[0]:
        k = idx - (n - 1)
        wk = w[idx]
        if k >= 0:
            if n - k > 0:
                grad[k:] += wk * (1.0 - h[:n-k])
                grad[:n-k] -= wk * h[k:]
        else:
            ks = -k
            if n - ks > 0:
                grad[:n-ks] += wk * (1.0 - h[ks:])
                grad[ks:] -= wk * h[:n-ks]
    return grad * dx

        f = smooth_obj_val(x, T) + pen * (np.sum(x) - 0.5 * n) ** 2
        g = smooth_grad(x, T) + 2.0 * pen * (np.sum(x) - 0.5 * n)
        return f, g
    try:
        res = optimize.minimize(objgrad, h, jac=True, method='L-BFGS-B',
                                bounds=[(0.0, 1.0)] * n,
                                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h

        jp = j + k
        if 0 <= jp < n:

                h = coord_descent(h, n, 15)
                c5 = compute_c5(h)

    # Phase 6: Higher resolution
    if time.time() - t0 < budget_s - 80:
        n2 = 560
        h2 = upsample(h, n2)
        for T in [0.008, 0.002, 0.0005]:
            if time.time() - t0 > budget_s - 30:
                break
            h2 = lbfgs_step(h2, n2, T, 800)
            h2 = fix_sum(h2, n2)
        h2 = coord_descent(h2, n2, 10)
        c5_2 = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2

    h = fix_sum(h, len(h))
    c5 = compute_c5(h)
    return (h, c5, len(h))
