from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 1:
            h[free] += d2 / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def grad_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    lo = max(0, -k)
    hi = min(n, n - k)
    if hi <= lo:
        return grad
    i = np.arange(lo, hi)
    np.add.at(grad, i + k, 1.0 - h[i])
    grad[i] -= h[i + k]
    return grad

def proj_subgrad(h, n, n_iters, alpha0, top_m, T, t0, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        # Softmax weights for top lags
        if T > 1e-9:
            w = np.exp((corr - cmax) / T)
            w /= w.sum()
            lags = np.where(w > 1e-8)[0]
        else:
            lags = np.where(corr >= cmax - 1e-10)[0]
            w = {idx: 1.0 / len(lags) for idx in lags}
        
        grad = np.zeros(n)
        if T > 1e-9:
            for idx in lags:
                if w[idx] < 1e-8:

                grad += w[idx] * grad_lag(h, k)

from scipy import optimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 1:
            h[free] += d2 / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def grad_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    lo = max(0, -k)
    hi = min(n, n - k)
    if hi <= lo:
        return grad
    i = np.arange(lo, hi)
    np.add.at(grad, i + k, 1.0 - h[i])
    grad[i] -= h[i + k]
    return grad

def proj_subgrad(h, n, n_iters, alpha0, top_m, T, deadline, use_momentum=True):
