```python
import numpy as np
import time
from scipy.optimize import minimize

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        cnt = int(m.sum())
        if cnt == 0:
            break
        h[m] += d / cnt
    return h

def c5val(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad(h, w):
    N = len(h)
    dx = 2.0 / N
    o = N - 1
    g = np.zeros(N)
    for j in range(N):
        k1lo = max(-(N-1), j - N + 1)
        k1hi = min(N-1, j)
        if k1lo <= k1hi:
            ks = np.arange(k1lo, k1hi + 1)
            g[j] += np.dot(w[o + ks], 1.0 - h[j - ks])
        k2lo = max(-(N-1), -j)
        k2hi = min(N-1, N - 1 - j)
        if k2lo <= k2hi:
            ks = np.arange(k2lo, k2hi + 1)
            g[j] -= np.dot(w[o + ks], h[j + ks])
    return g * dx

def pgrad(h0, ts, deadline, rng):
    N = len(h0)
    h = h0.copy()
    bc = c5val(h)
    stall = 0
    it = 0
    while time.time() < deadline:
        c = corr(h)
        # Cycle through beta and topk
        phase = it % 8
        if phase < 4:
            beta = [10, 30, 80, 200][phase]
            w = np.exp(beta * (c - c.max()))
            s = w.sum()
            w = w / s if s > 1e-30 else np.ones(2*N-1) / (2*N-1)
        else:
            tk = [3, 5, 10, 20][phase - 4]
            idx = np.argsort(c)[-min(tk, 2*N-1):]
            w = np.zeros(2*N-1)
            w[idx] = 1.0 / len(idx)
        g = grad(h, w)
        lr = 0.02 / (1.0 + 0.0005 * it)
        ok = False
        for sc in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            h2 = proj(h - lr * sc * g, ts)
            c2 = c5val(h2)
            if c2 < bc - 1e-16:
                h, bc = h2, c2
                stall = 0
                ok = True
                break
        if not ok:
            stall += 1
            if stall > 500:
                break
        it += 1
    return h, bc

def bswap(h0, ts, deadline, rng):
    """Binary swap local search"""
    N = len(h0)
    h = h0.copy()
    bc = c5val(h)
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        # Try all pairs
        best_gain = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = c5val(h)
                h[i], h[j] = 1.0, 0.0
                gain = bc - c
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        if best_pair is not None and best_gain > 1e-14:
            i, j = best_pair
            h[i], h[j] = 0.0, 1.0
            bc = c5val(h)
            improved = True
        if not improved:
            break
    return h, bc

def to_binary(h, ts, rng):
    """Round h to binary while maintaining sum"""
    N = len(h)
    target = int(ts)
    frac = ts - target
    # Sort by value, set largest to 1
    order = np.argsort(-h)
    h2 = np.zeros(N)
    h2[order[:target]] = 1.0
    if frac > 1e-6:
        h2[order[target]] = frac
    return proj(h2, ts)

def slsqp_opt(h0, ts, mit=400, deadline=None):
    N = len(h0)
    dx = 2.0 / N
    ns = {'n': N, 'ts': ts, 'dx': dx}
    
    def obj(x):
        return x[N]
    
    def cineq(x):
        return x[N] * ns['dx'] - corr(x[:N])
    
    def ceq(x):
        return float(x[:N].sum() - ns['ts'])
    
    c0 = c5val(h0)
    x0 = np.concatenate([h0, [c0 / dx + 0.02]])
    bnds = [(0, 1)] * N + [(0, float(N))]
    cons = [{'type': 'ineq', 'fun': cineq}, {'type': 'eq', 'fun': ceq}]
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-14})
        if r.x is not None:
            hr = proj(r.x[:N], ts)
            cr = c5val(hr)
            if cr < c0:
                return hr, cr
    except Exception:
        pass
    return h0.copy(), c0

def interp_to_n(h, nt):

    x_new = np.linspace(0, 1, nt, endpoint=False)
    return np.interp(x_new, x_old, h)
```
