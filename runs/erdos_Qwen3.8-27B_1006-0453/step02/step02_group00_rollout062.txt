def compute_c5(h):

    n = len(h)

    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    val = m + float(np.log(np.sum(np.exp(T * (corr - m))))) / T
    return val * dx

    n = len(h)
    s = h.sum() - n / 2.0
    return smooth_c5(h, T) + pen * s * s

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def lbfgs_run(h, T, n, pen=5000.0, maxiter=300):

    res = minimize(obj_pen, h, args=(T, pen), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(np.clip(res.x, 0.0, 1.0), n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, new_n)

def multi_T_opt(h0, n, T_list, t0, budget, frac, use_slsq=True):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    limit = t0 + budget * frac
    for T in T_list:
        if time.time() > limit:
            break
        try:
            hc = lbfgs_run(best_h, T, n, pen=3000.0, maxiter=250)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
        if use_slsq:
            try:
                hc = slsqp_run(best_h, T, n, maxiter=500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass
    return best_h, best_c5

def group_descent(h, n, rng, n_rounds=100):
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        j = rng.randint(n)
        group_size = min(3, n)
        idx = np.random.choice(n, size=group_size, replace=False)
        for coords in range(len(idx) + 1):
            for sub in range(max(1, len(idx) - coords + 1)):
                choice = idx[sub:sub + coords] if coords > 0 else np.array([], dtype=int)
                for deltas in [(0.05,) * coords, (-0.05,) * coords, (0.1,) * coords, (-0.1,) * coords,
                               (0.1, -0.05) if coords == 2 else (), (-0.1, 0.05) if coords == 2 else ()]:
                    if len(deltas) != coords:
                        continue
                    h_try = h.copy()
                    for ii, d in zip(choice, deltas):
                        h_try[ii] = float(np.clip(h_try[ii] + d, 0.0, 1.0))
                    h_try = project_feasible(h_try, n)
                    c = compute_c5(h_try)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        h = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            continue
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)

    # Phase 1: T-sweep L-BFGS+SLSQP at native n
    best_h, best_c5 = multi_T_opt(best_h, n0, [30, 100, 400, 1500], t0, budget_s, 0.22)

    # Phase 2: Multi-resolution with structured starts
    for n_try in (30, 40, 50, 60, 80, 100):
        if time.time() - t0 > budget_s * 0.42:
            break
        h_try = downsample(best_h, n_try)
        h_try = h_try + 0.05 * rng.randn(n_try)
        h_try = project_feasible(h_try, n_try)
        h_opt, c5_opt = multi_T_opt(h_try, n_try, [50, 300, 1200], t0, budget_s, 0.42)
        if c5_opt < best_c5:
            best_h = downsample(h_opt, n0)
            best_c5 = compute_c5(best_h)
        # Also try a binary-block start at this n
        h_blk = np.zeros(n_try)
        q = n_try // 2
        h_blk[:q] = 1.0
        h_blk = project_feasible(h_blk, n_try)
        h_opt2, c5_opt2 = multi_T_opt(h_blk, n_try, [80, 500], t0, budget_s, 0.42)
        if c5_opt2 < best_c5:
            best_h = downsample(h_opt2, n0)
            best_c5 = compute_c5(best_h)

    # Phase 3: DE at low n for global exploration
    if time.time() - t0 < budget_s * 0.52:
        for n_de in (30, 45):
            if time.time() - t0 > budget_s * 0.52:
                break
            global _de_n
            _de_n = n_de
            try:
                bnds_de = [(0.0, 1.0)] * n_de
                res = differential_evolution(
                    de_obj, bnds_de, seed=seed + n_de, maxiter=60,
                    popsize=12, tol=1e-7, mutation=(0.4, 1.2),
                    recombination=0.8, polish=False, init='latinhypercube')
                h_de = project_feasible(res.x, n_de)
                h_up = downsample(h_de, n0)
                h_ref, c5_ref = multi_T_opt(h_up, n0, [200, 800, 3000], t0, budget_s, 0.52)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
            except Exception:
                pass

    # Phase 4: Basin hopping with L-BFGS (fast iterations)
    n_cur = len(best_h)
    for it in range(70):
        if time.time() - t0 > budget_s * 0.76:
            break
        step = 0.14 * max(0.0, 1.0 - it / 70.0) + 0.015
        h_pert = best_h + step * rng.randn(n_cur)
        h_pert = project_feasible(h_pert, n_cur)
        try:
            hc = lbfgs_run(h_pert, 400, n_cur, pen=3000, maxiter=200)
            c5v = compute_c5(hc)
            temp = 0.007 * max(0.0, 1.0 - it / 70.0) + 0.0002
            if c5v < best_c5 or rng.rand() < np.exp(-(c5v - best_c5) / temp):
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
        if it % 5 == 4:
            try:
                hc = slsqp_run(best_h, 600, n_cur, maxiter=300)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 5: High-T SLSQP polish
    n_cur = len(best_h)
    for T in (3000, 10000, 30000, 80000):
        if time.time() - t0 > budget_s * 0.93:
            break
        try:
            hc = slsqp_run(best_h, T, n_cur, maxiter=1200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
        try:
            hc = lbfgs_run(best_h, T, n_cur, pen=50000, maxiter=200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                try:
                    hc2 = slsqp_run(hc, T, n_cur, maxiter=400)
                    c5v2 = compute_c5(hc2)
                    if c5v2 < best_c5:
                        best_c5 = c5v2
                        best_h = hc2
                except Exception:
                    pass
        except Exception:
            pass

    # Phase 6: Group/coordinate descent
    if time.time() - t0 < budget_s * 0.98:
        n_cur = len(best_h)
        best_h, best_c5 = group_descent(best_h, n_cur, rng, 300)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
