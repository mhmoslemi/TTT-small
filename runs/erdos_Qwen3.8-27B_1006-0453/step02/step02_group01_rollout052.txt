    n_half = n // 2
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    
    for round_i in range(max_rounds):
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        best_swap_c5 = best_c5
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h)
                h[i] = 1
                h[j] = 0
                if c5 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            best_c5 = best_swap_c5
            improved = True
        if not improved:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_max(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    n_half = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n_half]] = 1.0
    return b

def greedy_binary(h_bin, n, max_rounds=300):
    h = h_bin.copy()
    best_c5 = float(corr_max(h))
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_swap = best_c5 + 1e-9
        bi, bj = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                v = float(corr_max(h))
                h[i] = 1; h[j] = 0
                if v < best_swap:
                    best_swap = v; bi, bj = i, j
        if bi >= 0:
            h[bi] = 0; h[bj] = 1
            best_c5 = best_swap
            improved = True
        if not improved:
            break
    return h, best_c5

def sa_binary(h0, n, t_start, t_end, n_iter, rng):
    n_half = n // 2
    h = make_binary(h0, n).copy()
    best_h = h.copy()
    best_cm = float(corr_max(h))
    curr_cm = best_cm
    t = t_start
    cool = (t_end / t_start) ** (1.0 / n_iter)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    for it in range(n_iter):
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        i = ones[oi]; j = zeros[zi]
        h[i] = 0; h[j] = 1
        ones[oi] = j; zeros[zi] = i
        new_cm = float(corr_max(h))
        if new_cm < curr_cm or rng.random() < np.exp(-(new_cm - curr_cm) / max(t, 1e-10)):
            curr_cm = new_cm
            if curr_cm < best_cm:
                best_cm = curr_cm
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
            ones[oi] = i; zeros[zi] = j
        t = max(t * cool, t_end)
    return best_h, best_cm

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def _sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    cm = float(corr_max(h))
    t0 = cm + 0.02
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def _obj(x): return x[-1]
    def _eq(x): return x[:n].sum() - n / 2.0
    def _ineq(x):
        hd = x[:n]; tt = x[-1]
        return tt - np.correlate(hd, 1.0 - hd, mode='full')
    cons = [{'type': 'eq', 'fun': _eq}, {'type': 'ineq', 'fun': _ineq}]
    try:
        res = minimize(_obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
