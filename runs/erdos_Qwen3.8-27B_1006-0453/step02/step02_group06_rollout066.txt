```python
def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def lbfgs_optimize(h0, n, T, lam, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        softmax_obj, h0, jac=softmax_grad,
        args=(T, lam),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10, 'iprint': 0}
    )
    return proj_constraint(res.x, n)
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    c_shift = c - c_max
    val = (c_max + T * np.log(np.sum(np.exp(c_shift / T)))) * dx
    s = np.sum(h) - n * 0.5
    val += lam * s * s
    return val

def softmax_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    c_shift = c - c_max
    w = np.exp(c_shift / T)
    w /= w.sum()
    h_rev = h[::-1]
    A = np.correlate(w, h_rev, mode='valid')
    B = np.correlate(w, h, mode='valid')[::-1]
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    C = w_cum[n:2*n] - w_cum[:n]
    grad = dx * (C - A - B)
    grad += 2.0 * lam * (np.sum(h) - 0.5 * n)
    return grad

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def upsample_func(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def downsample_func(h, new_n):
    old_n = len(h)
    idx = np.linspace(0, old_n - 1, new_n).astype(int)
    return proj_constraint(h[idx], new_n)

def lbfgs_opt(h0, n, T, lam, max_iter):
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        softmax_obj, h0, jac=softmax_grad, args=(T, lam),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10}
    )
    return proj_constraint(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Coarse-to-fine L-BFGS-B at current resolution
    for T, lam, mi in [(0.04, 50, 100), (0.015, 200, 200), (0.005, 1000, 400),
                        (0.002, 3000, 600), (0.0008, 5000, 1000), (0.0003, 10000, 2000)]:
        if time.time() - t_start > budget_s - 100:
            break
        h_opt = lbfgs_opt(h_best.copy(), n_best, T, lam, mi)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c
    
    # Phase 2: Try finer resolution
    for n_try in [n_best + 56, n_best + 112]:
        if time.time() - t_start > budget_s - 120:
            break
        h_curr = upsample_func(h_best, n_try)
        for T, lam, mi in [(0.02, 200, 200), (0.005, 1000, 500), (0.001, 5000, 1000)]:
            if time.time() - t_start > budget_s - 60:
                break
            h_opt = lbfgs_opt(h_curr, n_try, T, lam, mi)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best, n_best = h_opt, c, n_try
                h_curr = h_best.copy()
    
    # Phase 3: Multi-start
    for i in range(20):
        if time.time() - t_start > budget_s - 80:
            break
        rng = np.random.RandomState(seed + i * 997 + 13)
        sigma = 0.005 + 0.01 * (i % 4)
        h_pert = np.clip(h_best + rng.randn(n_best) * sigma, 0, 1)
        h_pert = proj_constraint(h_pert, n_best)
        T_p = [0.001, 0.003, 0.0005, 0.002][i % 4]
        h_opt = lbfgs_opt(h_pert, n_best, T_p, 5000, 800)
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    c_shift = c - c_max
    val = (c_max + T * np.log(np.sum(np.exp(c_shift / T)))) * dx
    s = np.sum(h) - n * 0.5
    val += lam * s * s
    return val

def softmax_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    c_shift = c - c_max
    w = np.exp(c_shift / T)
    w /= w.sum()
    h_rev = h[::-1]
    A = np.correlate(w, h_rev, mode='valid')
    B = np.correlate(w, h, mode='valid')[::-1]
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    C = w_cum[n:2*n] - w_cum[:n]
    grad = dx * (C - A - B)
    grad += 2.0 * lam * (np.sum(h) - 0.5 * n)
    return grad

        return h

            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)

    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def downsample_func(h, new_n):
    old_n = len(h)
    idx = np.linspace(0, old_n - 1, new_n).astype(int)
    return proj_constraint(h[idx], new_n)

def lbfgs_opt(h0, n, T, lam, max_iter):
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        softmax_obj, h0, jac=softmax_grad, args=(T, lam),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10}
    )
    return proj_constraint(res.x, n)

def pnorm_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return (np.sum(c**p) / len(c)) ** (1.0 / p) * dx

def pnorm_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    nc = len(c)
    c_abs = np.abs(c) + 1e-30

    B = np.correlate(w, h, mode='valid')[::-1]
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    C = w_cum[n:2*n] - w_cum[:n]
    return dx * (C - A - B)
```
