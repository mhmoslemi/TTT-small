import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = np.sum(interior)
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0, 1)
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    return (c_max + T * np.log(np.sum(np.exp((corr - c_max) / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def ups(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, new_n)

        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
    )
    return project_h(res.x, n)

                if ft < c_best:

        order = np.argsort(fits)
        pop_h = [pop_h[j].copy() for j in order]

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = np.sum(interior)
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0, 1)
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    return (c_max + T * np.log(np.sum(np.exp((corr - c_max) / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def upsample_h(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, new_n)

def downsample_h(h, new_n):
    old_n = len(h)
    idx = np.array([min(int(i * old_n / new_n), old_n - 1) for i in range(new_n)])
    return project_h(h[idx], new_n)

def slsqp_opt(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
    )
    return project_h(res.x, n)

def de_search(h_init, n, pop, gens, t0, budget, seed):
    rng = np.random.RandomState(seed)
    h_pop = [h_init.copy()]
    for i in range(1, pop):
        h_try = h_init + rng.standard_normal(n) * 0.015
        h_pop.append(project_h(h_try, n))
    fits = [compute_c5(p) for p in h_pop]
    c_best = np.min(fits)
    h_best = h_pop[int(np.argmin(fits))].copy()
    stagnant = 0
    for g in range(gens):
        if time.time() - t0 > budget - 30:
            break
        F = 0.35 + 0.35 * rng.rand()
        CR = 0.55 + 0.35 * rng.rand()
        improved_this_gen = False
        for i in range(pop):
            a, b, c = rng.choice(pop, 3, replace=False)
            v = h_pop[a] + F * (h_pop[b] - h_pop[c])
            r = rng.rand(n) < CR
            if not r.any():
                r[rng.randint(n)] = True
            trial = np.where(r, v, h_pop[i])
            trial = project_h(trial, n)
            ft = compute_c5(trial)
            if ft < fits[i] - 1e-15:
                h_pop[i], fits[i] = trial, ft
                improved_this_gen = True
                if ft < c_best:
                    c_best, h_best = ft, trial.copy()
        if improved_this_gen:
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= 20:
            worst = int(np.argmax(fits))
            h_pop[worst] = project_h(h_best + rng.standard_normal(n) * 0.01, n)
            fits[worst] = compute_c5(h_pop[worst])
            stagnant = 0
        order = np.argsort(fits)
        h_pop = [h_pop[j].copy() for j in order]
        fits = [fits[j] for j in order]
    return h_best, c_best

def block_sa(h0, n, t0, budget, seed):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T_sa = 0.0004
    while time.time() - t0 < budget - 20:
        nblk = rng.randint(2, 10)
        start = rng.randint(0, max(1, n - nblk))
        idxs = np.arange(start, start + nblk)
        h_try = h_cur.copy()
        h_try[idxs] = rng.uniform(0, 1, nblk)
        h_try = project_h(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T_sa):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T_sa *= 0.998
    return h_best, c_best
