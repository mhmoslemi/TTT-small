def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(corr - cmax, -80*T, 0)
    logsumexp = np.log(np.sum(np.exp(shifted / T)))
    return (cmax + T * logsumexp) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(corr - cmax, -80*T, 0)
    exp_vals = np.exp(shifted / T)
    p = exp_vals / np.sum(exp_vals)  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    for a in range(-(n-1), n):
        pa = p[a + n - 1]  # weight for lag a
        if pa < 1e-15:
            continue
        # First term: (1-h[j-a]) for j in [max(0,a), min(n-1, n-1+a)]

        j_end1 = min(n-1, n-1+a)
        if j_start1 <= j_end1:

            valid = (idx - a) >= 0  # j-a >= 0
            if np.any(valid):

        # Second term: -h[j+a] for j in [max(0,-a), min(n-1, n-1-a)]

            if np.any(valid):
                grad[idx[valid]] -= pa * h[idx[valid] + a]

        if abs(s - target) < 1e-12:

import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        diff = (target - s) / n
        if abs(diff) < 1e-14:
            break
        h += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(corr - cmax, -80*T, 0)
    logsumexp = np.log(np.sum(np.exp(shifted / T)))
    dx = 2.0 / len(h)
    return (cmax + T * logsumexp) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(corr - cmax, -80*T, 0)
    exp_vals = np.exp(shifted / T)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for a in range(-(n-1), n):
        pa = p[a + n - 1]
        if pa < 1e-12:
            continue
        j1s = max(0, a)
        j1e = min(n-1, n-1+a)
        if j1s <= j1e:
            idx = np.arange(j1s, j1e+1)
            grad[idx] += pa * (1.0 - h[idx - a])
        j2s = max(0, -a)
        j2e = min(n-1, n-1-a)
        if j2s <= j2e:
            idx = np.arange(j2s, j2e+1)
            grad[idx] -= pa * h[idx + a]
    return grad * dx

def do_lbfgs(h, n, T, top_k, max_iter=2000):
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        if top_k < 2*n - 1:
            corr = np.correlate(x, 1.0 - x, mode='full')
            top_idx = np.argsort(corr)[-top_k:]
            vals = corr[top_idx]
            vmax = np.max(vals)
            shifted = vals - vmax
            exps = np.exp(shifted / T)
            p_local = exps / np.sum(exps)
            obj = (vmax + T * np.log(np.sum(exps))) * 2.0/n
            grad = np.zeros(n)
            for ii, tidx in enumerate(top_idx):
                if p_local[ii] < 1e-12:
                    continue
                a = tidx - (n-1)
                j1s = max(0, a)
                j1e = min(n-1, n-1+a)
                if j1s <= j1e:
                    idx = np.arange(j1s, j1e+1)
                    grad[idx] += p_local[ii] * (1.0 - x[idx - a])
                j2s = max(0, -a)
                j2e = min(n-1, n-1-a)
                if j2s <= j2e:
                    idx = np.arange(j2s, j2e+1)
                    grad[idx] -= p_local[ii] * x[idx + a]
            grad *= 2.0/n
            # Add penalty for sum constraint
            s = np.sum(x)
            penalty = 1000.0*(s - n*0.5)
            obj += 0.5*penalty*(s - n*0.5)
            grad += penalty * np.ones(n) / n
        else:
            obj = softmax_obj(x, T)
            grad = softmax_grad(x, T)
            s = np.sum(x)
            penalty = 1000.0*(s - n*0.5)
            obj += 0.5*penalty*(s - n*0.5)
            grad += penalty * np.ones(n) / n
        return obj, grad
    
    res = opt.minimize(obj_grad, h.copy(), method='L-BFGS-B',
                       jac=True, bounds=[(0,1)]*n,
                       options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-16})
    if res.x is not None:
        return fix_sum(res.x, n)
    return h

    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(xn, xo, h, period=2.0)
    return fix_sum(h_new, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    h_best = fix_sum(h_best, n_best)
    c5_best = compute_c5(h_best)

    def try_update(hc, nc):
        nonlocal h_best, n_best, c5_best
        hc = fix_sum(hc, nc)
        c = compute_c5(hc)
        if c < c5_best - 1e-12:
            h_best, n_best, c5_best = hc, nc, c
            return True
        return False

    def check_time(min_remaining):
        return budget_s - elapsed() > min_remaining

    # Phase 1: L-BFGS-B with full soft-max, decreasing T
    T_seqs = [0.1, 0.04, 0.015, 0.005, 0.002, 0.0008, 0.0003]
    for T in T_seqs:
        if not check_time(60):
            break
        hk = len(h_best)
        h_opt = do_lbfgs(h_best, hk, T, 2 * hk - 1, max_iter=4000)
        try_update(h_opt, hk)

    # Phase 2: Focus on top-10 lags
    if check_time(80):
        h_opt = do_lbfgs(h_best, n_best, 0.005, 10, max_iter=5000)
        try_update(h_opt, n_best)
    if check_time(80):
        h_opt = do_lbfgs(h_best, n_best, 0.001, 7, max_iter=5000)
        try_update(h_opt, n_best)
    if check_time(60):
        h_opt = do_lbfgs(h_best, n_best, 0.0003, 5, max_iter=6000)
        try_update(h_opt, n_best)

    # Phase 3: SLSQP polish
    if check_time(60):
        h_opt = do_slsqp(h_best, n_best, 0.001, 5, max_iter=3000)
        try_update(h_opt, n_best)

    # Phase 4: Perturbation + re-optimize (escape local optima)
    for ps in [0.04, 0.08, 0.12]:
        if not check_time(150):
            break
        h_pert = h_best + np.random.randn(n_best) * ps
        h_pert = fix_sum(h_pert, n_best)
        h_opt = do_lbfgs(h_pert, n_best, 0.02, 10, max_iter=3000)
        h_opt = do_lbfgs(h_opt, n_best, 0.002, 5, max_iter=4000)
        try_update(h_opt, n_best)

    # Phase 5: Try alternative resolutions
    for n_try in [480, 600, 320]:
        if not check_time(200):
            break
        h_init = resample(h_best, n_try)
        for T, tk in [(0.03, 12), (0.008, 8), (0.002, 5)]:
            if not check_time(80):
                break
            h_opt = do_lbfgs(h_init, n_try, T, tk, max_iter=4000)
            if try_update(h_opt, n_try):
                h_init = h_best.copy()

    # Phase 6: Return to best resolution and final polish
    if n_best != len(h_best):
        h_best = resample(h_best, n_best)
    if check_time(80):
        h_opt = do_lbfgs(h_best, n_best, 0.0005, 4, max_iter=6000)
        try_update(h_opt, n_best)
    if check_time(50):
        h_opt = do_slsqp(h_best, n_best, 0.0004, 3, max_iter=4000)
        try_update(h_opt, n_best)
    if check_time(40):
        h_opt = do_lbfgs(h_best, n_best, 0.0001, 3, max_iter=5000)
        try_update(h_opt, n_best)

    # Phase 7: Micro-perturbation + polish
    if check_time(50):
        h_micro = h_best + np.random.randn(n_best) * 0.008
        h_micro = fix_sum(h_micro, n_best)
        h_opt = do_lbfgs(h_micro, n_best, 0.001, 5, max_iter=4000)
        h_opt = do_lbfgs(h_opt, n_best, 0.0002, 3, max_iter=4000)
        try_update(h_opt, n_best)

    # Final: one more SLSQP with tight tolerance
    if check_time(30):
        h_opt = do_slsqp(h_best, n_best, 0.0002, 3, max_iter=2000)
        try_update(h_opt, n_best)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
