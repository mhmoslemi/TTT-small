def bin_eval(b, P):
    """Return max autocorr value = n1 - min overlap."""
    n1 = int(b.sum())
    min_ov = P  # initialize high
    for L in range(1, P):
        ov = int(np.dot(b, np.roll(b, L)))

def bin_eval_fast(b, P):
    """Vectorized: compute all overlaps at once."""
    n1 = int(b.sum())
    # overlaps[L] = sum_j b[j]*b[(j+L)%P] for L=0..P-1
    # This is the circular autocorrelation
    # Can compute via FFT: ifft(|fft(b)|^2)
    F = np.fft.fft(b)
    acf = np.real(np.fft.ifft(F * np.conj(F)))
    # acf[L] = sum_j b[j]*b[(j+L)%P] (circular)
    # We want min over L=1..P-1
    return n1 - float(np.min(acf[1:]))

    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while it < n_iter_max and time.time() < deadline:

        tau = tau0 * (tau1 / tau0) ** frac
        grad, c5v = smax_grad(h, N, tau)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad

        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
        if lr < 1e-11:
            break
    return best_h, best_c5

    target0 = N0 / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)

    # Allocate time across strategies
    # Strategy 1: Optimized Adam from parent (40% time)
    # Strategy 2: Nesterov + Adam from perturbed parent (20% time)
    # Strategy 3: Diverse random restarts (20% time)
    # Strategy 4: Final fine-tuning of best (20% time)

    # --- Strategy 1: Improved Adam from parent ---
    t1 = deadline * 0.04 + t0 + (deadline - t0) * 0.40
    t1 = t0 + (deadline - t0) * 0.40
    h1, c1 = multi_optimize(h0.copy(), N0, target0, dx0, t1, rng)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # --- Strategy 2: Nesterov-focused from perturbed parent ---
    t2 = t0 + (deadline - t0) * 0.62
    h2_init = h0 + rng.randn(N0) * 0.03
    h2_init = project_sum(h2_init, target0)
    # Quick Adam warmup then Nesterov
    t2a = t0 + (deadline - t0) * 0.55
    h2a, c2a = adam_phase(h2_init, N0, target0, dx0, t2a,
                          tau0=5.0, tau1=100.0, lr0=0.015, decay=0.99, n_iter_max=6000)
    if c2a < best_c5:
        best_c5 = c2a
        best_h = h2a.copy()
    if time.time() < t2 - 5:
        h2, c2 = nesterov_phase(h2a.copy(), N0, target0, dx0, t2,
                                tau=500.0, lr0=0.003, beta=0.92, n_iter_max=10000)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # --- Strategy 3: Random restarts ---
    t3 = t0 + (deadline - t0) * 0.82
    n_restarts = 4
    rem3 = t3 - time.time()
    if rem3 > 30:
        per_restart = rem3 / n_restarts
        for ri in range(n_restarts):
            if time.time() > t3:
                break
            h_r = 0.5 + rng.randn(N0) * 0.12
            h_r = project_sum(h_r, target0)
            tr = time.time() + per_restart - 2
            h_r, c_r = multi_optimize(h_r, N0, target0, dx0, tr, rng)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()

    # --- Strategy 4: Final fine-tuning of best ---
    rem4 = deadline - time.time()
    if rem4 > 15:
        # Very fine Adam
        hf1, cf1 = adam_phase(best_h.copy(), N0, target0, dx0, time.time() + rem4 * 0.6,
                              tau0=500.0, tau1=3000.0, lr0=0.001, decay=0.995, n_iter_max=30000)
        if cf1 < best_c5:
            best_c5 = cf1
            best_h = hf1.copy()
        # Final Nesterov
        rem4b = deadline - time.time()
        if rem4b > 10:
            hf2, cf2 = nesterov_phase(best_h.copy(), N0, target0, dx0, deadline - 3,
                                      tau=2000.0, lr0=0.0005, beta=0.95, n_iter_max=20000)
            if cf2 < best_c5:
                best_c5 = cf2
                best_h = hf2.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
