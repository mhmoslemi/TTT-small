            idx2 = np.arange(0, n - dd)
            grad[idx2] -= w * h[idx2 + dd]

    n = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(y - lam, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = np.sum(exps)
    val = (m + np.log(Z) / T) * dx
    p = exps / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        w = T * p[k] * dx
        if abs(w) < 1e-15:
            continue
        if d >= 0:
            if d < n:
                grad[d:] += w * (1.0 - h[:n - d])
            if d > 0:
                grad[:n - d] -= w * h[d:]
        else:
            dd = -d
            grad[dd:] += w * (1.0 - h[:n - dd])
            grad[:n - dd] -= w * h[dd:]
    return val, grad

def topk_val_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-K:]
    weight = 1.0 / K
    val = np.mean(corr[top_idx]) * dx
    grad = np.zeros(n)
    for k in top_idx:
        d = k - (n - 1)
        w = weight * dx
        if d >= 0:
            if d < n:
                grad[d:] += w * (1.0 - h[:n - d])
            if d > 0:
                grad[:n - d] -= w * h[d:]
        else:
            dd = -d
            grad[dd:] += w * (1.0 - h[:n - dd])
            grad[:n - dd] -= w * h[dd:]
    return val, grad

def project_box_sum(y, target):
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

def projected_grad(h, T, lr0, n_iter, t0, deadline):
    n = len(h)
    target = n / 2.0
    lr = lr0
    val, grad = smooth_val_grad(h, T)
    for it in range(n_iter):
        if time.time() - t0 > deadline:
            break
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        val_new, grad_new = smooth_val_grad(h_new, T)
        if val_new < val:
            h = h_new
            val = val_new
            lr = min(lr * 1.03, lr0 * 3.0)
        else:
            lr *= 0.65
            if lr < 1e-12:
                break
        grad = grad_new
    return h, val

def pg_topk(h, K, lr0, n_iter, t0, deadline):
    n = len(h)
    target = n / 2.0
    lr = lr0
    val, grad = topk_val_grad(h, K)
    for it in range(n_iter):
        if time.time() - t0 > deadline:
            break
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        val_new, grad_new = topk_val_grad(h_new, K)
        if val_new < val:
            h = h_new
            val = val_new
            lr = min(lr * 1.03, lr0 * 3.0)
        else:
            lr *= 0.65
            if lr < 1e-12:
                break
        grad = grad_new
    return h, val

def slsqp_polish(h, t0, deadline):
    n = len(h)
    def _obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(5000.0 * (corr - m)))) / 5000.0
    def _con(x):
        return x.sum() - len(x) / 2.0
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='SLSQP',
                      bounds=bnds,
                      constraints=[{'type': 'eq', 'fun': _con}],
                      options={'maxiter': 200, 'ftol': 1e-15})
        return project_box_sum(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
    best_n = len(best_h)
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient warmup
    d1 = t0 + budget_s * 0.15
    n_cur = len(best_h)
    for T in [200, 1000, 4000]:
        if time.time() - t0 > d1:
            break
        n_cur = len(best_h)
        lr = 0.8 / (T * 2.0 / n_cur + 1.0)
        h_opt, _ = projected_grad(best_h, T, lr, 3000, t0, d1)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Top-K gradient focusing
    d2 = t0 + budget_s * 0.32
    for K in [1, 3, 5, 8]:
        if time.time() - t0 > d2:
            break
        n_cur = len(best_h)
        lr = 0.08
        h_opt, _ = pg_topk(best_h, K, lr, 2000, t0, d2)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Multi-resolution basin hopping
    d3 = t0 + budget_s * 0.70
    n_cur = len(best_h)
    resolutions = sorted(set([n_cur, 80, 84, 90, 96, 100, 108, 120]))
    for n_try in resolutions:
        if time.time() - t0 > d3:
            break
        if n_try == n_cur:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, n_cur)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_box_sum(h_base, n_try / 2.0)
        n_trials = 7 if n_try <= 100 else 4
        for trial in range(n_trials):
            if time.time() - t0 > d3:
                break
            scale = 0.05 + 0.07 * np.random.rand()
            h_pert = h_base + np.random.randn(n_try) * scale
            h_pert = project_box_sum(h_pert, n_try / 2.0)
            lr1 = 0.4 / (800.0 * 2.0 / n_try + 1.0)
            h_opt, _ = projected_grad(h_pert, 800, lr1, 2000, t0, d3)
            lr2 = 0.05
            h_opt, _ = pg_topk(h_opt, 3, lr2, 1000, t0, d3)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
                n_cur = n_try
                h_base = h_opt.copy()

    # Phase 4: Fine-tuning with high-T smooth + top-K
    d4 = t0 + budget_s * 0.88
    n_cur = len(best_h)
    for T in [5000, 20000, 100000]:
        if time.time() - t0 > d4:
            break
        lr = 0.2 / (T * 2.0 / n_cur + 1.0)
        h_opt, _ = projected_grad(best_h, T, lr, 3000, t0, d4)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    for i in range(25):
        if time.time() - t0 > d4:
            break
        scale = 0.012 * (1.0 - i / 30.0) + 0.003
        h_pert = best_h + np.random.randn(n_cur) * scale
        h_pert = project_box_sum(h_pert, n_cur / 2.0)
        h_opt, _ = pg_topk(h_pert, 5, 0.03, 1500, t0, d4)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: SLSQP polish
    d5 = t0 + budget_s * 0.96
    if time.time() - t0 < d5:
        h_s = slsqp_polish(best_h, t0, d5)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
        h_opt, _ = projected_grad(best_h, 50000, 0.005, 2000, t0, d5)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
