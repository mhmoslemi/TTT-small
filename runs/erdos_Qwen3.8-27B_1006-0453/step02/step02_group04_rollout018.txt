import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.5:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            rem = diff
            for i in idx:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
                if rem <= 1e-12:
                    break
        else:
            rem = -diff
            idx = np.argsort(-h)
            for i in idx:
                sub = min(rem, h[i])
                h[i] -= sub
                rem -= sub
                if rem <= 1e-12:
                    break
    return h

def compute_grad_active(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp((corr - c_max) / T)
    w = w / w.sum()
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for p in range(n):
        s_min = max(0, -p)
        s_max = min(n - 1, n - 1 - p)
        if s_min <= s_max:
            sr = np.arange(s_min, s_max + 1)
            grad[p] += np.dot(w[sr + (n-1)], one_minus_h[p + sr])
        s_min2 = max(-(n-1), p - (n-1))
        s_max2 = min(n - 1, p)
        if s_min2 <= s_max2:
            sr2 = np.arange(s_min2, s_max2 + 1)
            grad[p] -= np.dot(w[sr2 + (n-1)], h[p - sr2])
    return grad * dx, corr

def binary_push(h, alpha):
    h = h.copy()
    mask_low = h < 0.5
    h[mask_low] -= alpha * h[mask_low] * (1.0 - h[mask_low]) * 4.0
    h[~mask_low] += alpha * (1.0 - h[~mask_low]) * h[~mask_low] * 4.0
    return np.clip(h, 0.0, 1.0)

def subgradient_path(h, n, T0, lr0, total_steps, binary_alpha):
    T = T0
    lr = lr0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    decay_T = (0.001 / T0) ** (1.0 / total_steps)
    decay_lr = (0.001 / lr0) ** (1.0 / total_steps)
    for step in range(total_steps):
        grad, corr = compute_grad_active(h, T)
        h_new = h - lr * grad
        h_new = binary_push(h_new, binary_alpha)
        h_new = fix_sum(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        T *= decay_T
        lr *= decay_lr
    return best_h, best_c5

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def do_slsqp(h, n, T, max_iter):
    try:
        res = opt.minimize(soft_max_obj, h.copy(), args=(T,),
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                           options={'maxiter': max_iter, 'ftol': 1e-15})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def njn_optimizer(h, n, max_iter=200):
    """Nelder-Mead with adaptive simplex on the max overlap"""
    def obj(x):
        x_c = fix_sum(x, n)
        return compute_c5(x_c)
    x0 = h.copy()
    try:
        res = opt.minimize(obj, x0, method='Nelder-Mead',
                           options={'maxiter': max_iter, 'fatol': 1e-14,
                                    'xatol': 1e-13, 'adaptive': True})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

    # Phase 1: Coarse-to-fine multi-resolution with subgradient

        if n_work != n_cur:
