import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_smax(h, N, tau):
    """Gradient of soft-max of correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 0.0005)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def nesterov_opt(h, N, target, dx, deadline, tau0, tau1, lr0, decay, momentum=0.9):
    """Nesterov-accelerated gradient descent with line search."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / max(100, (deadline - time.time()) * 10), 1.0)
        # Better: use iteration-based fraction
        frac = min(it / 1500.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        
        # Nesterov lookahead
        y = h + momentum * v
        y = project_sum(y, target)
        grad, c5_val = compute_grad_smax(y, N, tau)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = y.copy()
        
        # Line search
        step = lr
        for _ in range(5):
            new_h = y - step * grad
            new_h = project_sum(new_h, target)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < c5_val * 0.999 or new_c5 < best_c5:
                break
            step *= 0.5
        else:
            step = lr * 0.5
        
        new_h = project_sum(y - step * grad, target)
        v = new_h - h
        h = new_h
        it += 1
        if it % 30 == 0:
            lr *= decay
        if lr < 1e-10:
            break
    return best_h, best_c5

def coord_descent(h, N, target, dx, deadline, n_sweeps=20):
    """Coordinate descent: adjust individual points to reduce max correlation."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(123)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(N)
        for i in order:
            if time.time() > deadline - 2:
                return best_h, best_c5
            # Try a few perturbations
            for delta in [0.02, -0.02, 0.05, -0.05, 0.01, -0.01]:
                h_new = h.copy()
                h_new[i] += delta
                h_new = project_sum(h_new, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    h = h_new
                    h = h.copy()
                    break
                elif c5_new < compute_c5(h, dx):
                    pass  # accept but don't update best
    return best_h, best_c5

Actually, I realize I should also try a different approach entirely: **Projected Subgradient Method with Polyak step size**.

The Polyak step size for minimizing f(h) with known lower bound L:
step = (f(h) - L) / ||g||²

We know L ≥ 0 (correlation is non-negative for h ≥ 0), so we can use L = 0 as a lower bound. This gives an adaptive step size that's large when far from optimal and small when close.

            best_h = h.copy()

        h_c, c4 = optimize(h_c, N, target, dx, p4_dead, 'topk', 1.0, 1.0, 0.002, 0.99, K=5)
        if c4 < best_c5:
            best_c5, best_h = c4, h_c.copy()

    # Phase 5: local refinement
    if time.time() < deadline - 20:
        h_c, c5 = local_refine(h_c, N, target, dx, min(time.time() + 40, deadline - 5), rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h_c.copy()

    # Try N=200 with best_h as init
    if time.time() < deadline - 90:
        N2 = 200
        dx2 = 2.0 / N2
        t2 = N2 / 2.0
        xN2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h_init2 = np.interp(xN2, x0, best_h)
        h_init2 = project_sum(h_init2, t2)
        p5_dead = time.time() + min(45, deadline - time.time() - 25)
        h2, c5v = optimize(h_init2, N2, t2, dx2, p5_dead, 'smax', 10.0, 300.0, 0.02, 0.96)
        if time.time() < deadline - 10:
            h2, c5v2 = local_refine(h2, N2, t2, dx2, deadline - 5, rng)
            if c5v2 < c5v:
                c5v = c5v2
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Final smax polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 4:
        dx_b = 2.0 / Nb
        h_f, cf = optimize(best_h.copy(), Nb, Nb / 2.0, dx_b, deadline - 2,
                           'smax', 200.0, 1000.0, 0.0015, 0.993)
        if cf < best_c5:
            best_c5, best_h = cf, h_f.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
