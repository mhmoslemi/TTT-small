def binary_local_search(h_bin, n, max_sweeps=100):
    """Local search on binary h with exactly n/2 ones."""
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                # Compute delta corr for swap i(1->0), j(0->1)
                # Lag L ranges from -(n-1) to (n-1)
                # corr_full index = L + (n-1)
                L = np.arange(-(n-1), n)  # lags
                # i+L valid: 0 <= i+L < n
                valid_iL = (i + L >= 0) & (i + L < n)
                valid_iLm = (i - L >= 0) & (i - L < n)
                valid_jL = (j + L >= 0) & (j + L < n)
                valid_jLm = (j - L >= 0) & (j - L < n)
                
                h_iL = np.zeros(2*n-1)
                h_iL[valid_iL] = h[i + L[valid_iL]]
                h_jL = np.zeros(2*n-1)
                h_jL[valid_jL] = h[j + L[valid_jL]]
                h_iLm = np.zeros(2*n-1)
                h_iLm[valid_iLm] = h[i - L[valid_iLm]]
                h_jLm = np.zeros(2*n-1)
                h_jLm[valid_jLm] = h[j - L[valid_jLm]]
                
                dcorr = h_iL - h_jL - (1 - h_iLm) * valid_iLm.astype(float) + (1 - h_jLm) * valid_jLm.astype(float)
                
                new_corr = corr + dcorr
                new_max = np.max(new_corr)
                if new_max < np.max(corr) - 1e-10:
                    # Accept swap
                    h[i] = 0; h[j] = 1
                    corr = new_corr
                    best_c5 = new_max * dx
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h, n):
    nb = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:nb]] = 1.0
    return hb

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(50):
        d = tgt - np.sum(h)
        if abs(d) < 1e-13:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = int(np.count_nonzero(m))
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_res(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return np.interp(xd, xs, h)

def blocal(h0, n, max_sweeps, deadline):
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    dx = 2.0 / n
    mx = float(np.max(corr))
    L = np.arange(-(n - 1), n, dtype=np.int32)
    N2 = 2 * n - 1
    for sw in range(max_sweeps):
        if deadline and time.time() >= deadline - 2:
            break
        best_gain = 1e-14
        bi = bj = -1
        bd = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            iL = i + L
            iLm = i - L
            viL = (iL >= 0) & (iL < n)
            viLm = (iLm >= 0) & (iLm < n)
            for j in zeros:
                jL = j + L
                jLm = j - L
                vjL = (jL >= 0) & (jL < n)
                vjLm = (jLm >= 0) & (jLm < n)
                d = np.zeros(N2)
                idx = np.where(viL)[0]
                if len(idx):
                    d[idx] = h[iL[idx]]
                idx = np.where(viLm)[0]
                if len(idx):
                    d[idx] -= 1.0 - h[iLm[idx]]
                idx = np.where(vjL)[0]
                if len(idx):
                    d[idx] -= h[jL[idx]]
                idx = np.where(vjLm)[0]

                    d[idx] += 1.0 - h[jLm[idx]]

                    best_gain = gain
                    bi, bj = int(i), int(j)
                    bd = d
        if bi >= 0:
            h[bi] = 0.0
            h[bj] = 1.0
            corr += bd
            mx = float(np.max(corr))
        else:
            break
    return h, mx * dx

def perturb(h, n, k, rng):
    """Randomly swap k ones with k zeros."""
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    ko = rng.choice(len(ones), min(k, len(ones)), replace=False)
    kz = rng.choice(len(zeros), min(k, len(zeros)), replace=False)
    for a in ko:
        h[ones[a]] = 0.0
    for b in kz:
        h[zeros[b]] = 1.0
    return h

        mx = np.max(c)
        t = 200.0
        s = t * (c - mx)
        s -= np.max(s)
        return (mx + np.sum(np.exp(s)) / t) * (2.0 / n)
