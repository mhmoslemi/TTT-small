```python
def soft_max_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # soft max value
    c_max = np.max(corr)
    shifted = corr - c_max
    shifted = np.clip(shifted, -80*T, 0.0)
    exp_vals = np.exp(shifted / T)
    log_sum = np.log(exp_vals.sum() + 1e-300)
    obj = (c_max + T * log_sum) * dx
    
    # gradient: d/dh_j of soft_max(corr) * dx
    # = dx * (1/T) * sum_m exp(c_m - c_max)/T * dc_m/dh_j
    # = dx * sum_m p_m * dc_m/dh_j where p_m = exp(c_m-c_max)/T / sum exp
    # Actually: d soft_max / d h_j = sum_m (exp(c_m/T) / sum exp(c_m'/T)) * dc_m/dh_j
    # = sum_m p_m * dc_m/dh_j where p_m = exp(c_m/T) / sum exp(c_m/T)
    # Using the shifted version: p_m = exp((c_m - c_max)/T) / sum exp((c_m' - c_max)/T)
    
    p = exp_vals / exp_vals.sum()  # probabilities, length 2n-1
    
    # Now compute grad = sum_m p[m] * dc[m]/dh
    # dc[m]/dh[j] for the m-th entry of full correlate
    # m = 0, ..., 2n-2, lag = m - (n-1)
    # For lag >= 0: c[m] = sum_{n=0}^{n-1-lag} h[n+lag]*(1-h[n])
    #   dc/dh[j]:
    #     j appears in h[n+lag] when n=j-lag (valid: 0<=j-lag<=n-1-lag, i.e., lag<=j<=n-1)
    #       contrib: (1-h[j-lag])
    #     j appears in h[n] when n=j (valid: 0<=j<=n-1-lag)
    #       contrib: -h[j+lag]
    # For lag < 0 (lag = -p): c[m] = sum_{n=p}^{n-1} h[n-p]*(1-h[n])
    #   dc/dh[j]:
    #     j appears in h[n-p] when n=j+p (valid: p<=j+p<=n-1, i.e., 0<=j<=n-1-p)
    #       contrib: (1-h[j+p])
    #     j appears in h[n] when n=j (valid: p<=j<=n-1)
    #       contrib: -h[j-p]
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        if p[m] < 1e-15:
            continue
        lag = m - (n-1)
        pm = p[m]
        if lag >= 0:
            # h[n+lag]*(1-h[n]) for n=0 to n-1-lag
            # dc/dh[j] for j in [lag, n-1]: += pm * (1-h[j-lag])
            # dc/dh[j] for j in [0, n-1-lag]: += pm * (-h[j+lag])
            j1 = max(lag, 0)
            j2 = n-1
            grad[j1:j2+1] += pm * (1.0 - h[j1-lag:j2-lag+1])
            j3 = 0
            j4 = n-1-lag
            grad[j3:j4+1] += pm * (-h[j3+lag:j4+lag+1])
        else:
            pp = -lag  # positive
            # h[n-pp]*(1-h[n]) for n=pp to n-1
            # dc/dh[j]:
            #   j=h[n-pp] when n=j+pp (valid: pp<=j+pp<=n-1, i.e., 0<=j<=n-1-pp)
            #     contrib: (1-h[j+pp])
            #   j=h[n] when n=j (valid: pp<=j<=n-1)
            #     contrib: -h[j-pp]
            j1 = 0
            j2 = n-1-pp
            grad[j1:j2+1] += pm * (1.0 - h[j1+pp:j2+pp+1])
            j3 = pp
            j4 = n-1
            grad[j3:j4+1] += pm * (-h[j3-pp:j4-pp+1])
    
    grad *= dx
    return obj, grad
```

Actually, I can vectorize the gradient computation. Let me think...

For the full correlation c[m] with lag = m-(n-1):
- Positive lags (m >= n-1, lag = m-(n-1) >= 0): c[m] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
  This is like a "sliding" inner product.
  
- Negative lags (m < n-1, lag = m-(n-1) < 0, let p = -(n-1-m) > 0): c[m] = sum_{i=p}^{n-1} h[i-p]*(1-h[i])

The gradient can be computed using the "reverse" of the correlation operation. Specifically:
If c = correlate(h, 1-h), then the Jacobian of c w.r.t. h is a (2n-1) x n matrix.
The i-th column of this Jacobian gives dc/dh[i].

For the positive lag part (lag = 0, 1, ..., n-1):
c[lag + n - 1] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])

dc/dh[j] gets contribution from:
- h[j] as the "first" factor: when j = i+lag, i = j-lag, valid for lag ≤ j (and i ≤ n-1-lag means j ≤ n-1)
  Contribution: (1-h[j-lag]) for lag = 0, ..., j
- h[j] as the "second" factor: when j = i, valid for j ≤ n-1-lag
  Contribution: -h[j+lag] for lag = 0, ..., n-1-j

So for a given j:
dc/dh[j] = sum_{lag=0}^{min(j, n-1)} (1-h[j-lag]) * p[lag+n-1] - sum_{lag=0}^{n-1-j} h[j+lag] * p[lag+n-1]

Where p[m] are the soft-max weights.

This can be computed as:
term1[j] = sum_{lag=0}^{j} (1-h[j-lag]) * p[lag+n-1]  (for j ≤ n-1)
term2[j] = sum_{lag=0}^{n-1-j} h[j+lag] * p[lag+n-1]

These are like convolutions! 
term1[j] = (1-h) convolved with p (specifically, the relevant part of p)
term2[j] = h convolved with a reversed part of p

Let me define:
p_pos[lag] = p[lag + n - 1] for lag = 0, 1, ..., n-1 (positive and zero lags)
p_neg[lag] = p[n - 1 - lag] for lag = 0, 1, ..., n-1 (negative lags, indexed by magnitude)

```python
import numpy as np
import scipy.optimize as opt
import time

def fix_sum(h, n):
    """Project h onto {0<=h<=1, sum=n/2}"""
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    # Simple projection: adjust interior points
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) <= n_int * 0.45:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        # Water-filling
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_from_corr(corr, n):
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    exp_vals = np.exp(shifted / T)
    log_sum = np.log(np.sum(exp_vals) + 1e-300)
    obj = (c_max + T * log_sum) * dx
    
    if T < 1e-15:
        # Just return gradient of max
        k_star = np.argmax(corr)
        grad = dc_dh_single(k_star, h, n)
        return obj, grad * dx
    
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    
    # For each lag m (0 to 2n-2), compute contribution to grad
    # This is O(n * (2n-1)) = O(n²)
    # But we can vectorize some of it
    
    # For positive lags (m >= n-1, lag = m-(n-1) >= 0):
    #   c[m] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
    #   dc/dh[j] += p[m] * (1-h[j-lag]) [if j >= lag and j <= n-1]
    #   dc/dh[j] += p[m] * (-h[j+lag]) [if j <= n-1-lag]
    
    # Vectorized: for each lag, add to grad
    for lag in range(0, n):
        pm = p[lag + n - 1]
        if pm < 1e-16:
            continue
        # First term: (1-h[j-lag]) for j = lag, ..., n-1
        grad[lag:] += pm * (1.0 - h[:n-lag])
        # Second term: -h[j+lag] for j = 0, ..., n-1-lag
        grad[:n-lag] -= pm * h[lag:]
    
    # For negative lags (m < n-1, let p_neg = n-1-m >= 1):
    #   c[m] = sum_{i=p_neg}^{n-1} h[i-p_neg]*(1-h[i])
    #   dc/dh[j] += p[m] * (1-h[j+p_neg]) [if j <= n-1-p_neg]
    #   dc/dh[j] += p[m] * (-h[j-p_neg]) [if j >= p_neg]
    for p_neg in range(1, n):
        m = n - 1 - p_neg
        pm = p[m]
        if pm < 1e-16:
            continue
        # First term: (1-h[j+p_neg]) for j = 0, ..., n-1-p_neg
        grad[:n-p_neg] += pm * (1.0 - h[p_neg:])
        # Second term: -h[j-p_neg] for j = p_neg, ..., n-1
        grad[p_neg:] -= pm * h[:n-p_neg]
    
    grad *= dx
    return obj, grad

def dc_dh_single(m, h, n):
    """Compute dc[m]/dh analytically for a single m (for T=0 case)"""
    lag = m - (n-1)
    grad = np.zeros(n)
    if lag >= 0:
        # c[m] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        for j in range(lag, n):
            grad[j] += (1.0 - h[j-lag])
        for j in range(0, n-lag):
            grad[j] -= h[j+lag]
    else:
        p = -lag
        # c[m] = sum_{i=p}^{n-1} h[i-p]*(1-h[i])
        for j in range(0, n-p):
            grad[j] += (1.0 - h[j+p])
        for j in range(p, n):
            grad[j] -= h[j-p]
    return grad

def do_slsqp_analytic(h, n, T, max_iter, ftol=1e-14):
    def func(x):
        return soft_max_obj_grad(x, T)[0]
    def jac(x):
        return soft_max_obj_grad(x, T)[1]
    # Actually SLSQP needs separate func and jac, but computing both together is more efficient
    # Let me use a trick: compute both in one call
    cache = {'val': None, 'grad': None}
    def obj(x):
        v, g = soft_max_obj_grad(x, T)
        cache['val'] = v
        cache['grad'] = g
        return v
    def grad_fn(x):
        if cache['grad'] is None:
            soft_max_obj_grad(x, T)
        return cache['grad']
    
    try:
        res = opt.minimize(
            obj, h.copy(), jac=grad_fn,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n*0.5,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h
```

```python
def projected_gradient_min(h, n, T, max_iter=5000, lr=None):
    """Nesterov-accelerated projected gradient on soft-max objective"""
    if lr is None:
        lr = 0.01
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_val = soft_max_obj_grad(h_curr, T)[0]
    
    for it in range(max_iter):
        v, g = soft_max_obj_grad(h_curr, T)
        if v < best_val - 1e-14:
            best_val = v
            best_h = h_curr.copy()
        
        # Nesterov momentum
        beta = (1 + np.sqrt(1 + 4*(it+1)**2)) / 2  # standard Nesterov
        beta = min(beta, 1.0)  # cap it
        h_mom = h_curr + (h_curr - h_prev) * min(beta/((it+1)+1), 0.1)
        
        # Gradient step
        h_new = h_mom - lr * g
        h_new = fix_sum(h_new, n)
        
        # Simple backtracking line search
        v_new = soft_max_obj_grad(h_new, T)[0]
        if v_new > v + 1e-10:
            # Reduce step
            h_new = h_curr - lr * 0.1 * g
            h_new = fix_sum(h_new, n)
        
        h_prev = h_curr
        h_curr = h_new
        lr *= 0.999  # slow decay
    
    return best_h
```

- Intersection of two lines: a[i]*x+b[i] = a[j]*x+b[j] → x = (b[j]-b[i])/(a[i]-a[j])

```python
    return h

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    shifted = np.clip(corr - cmax, -80.0 * T, 0.0)
    ev = np.exp(shifted / T)
    obj = (cmax + T * np.log(ev.sum() + 1e-300)) * dx
    p = ev / ev.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-16:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += pm * (1.0 - h[:n - lag])
            grad[:n - lag] -= pm * h[lag:]
        else:
            pp = -lag
            grad[:n - pp] += pm * (1.0 - h[pp:])
            grad[pp:] -= pm * h[:n - pp]
    grad *= dx
    return obj, grad

def topk_obj_grad(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(corr)[::-1][:k]
    ct = corr[idx]
    cmax = np.max(ct)
    shifted = np.clip(ct - cmax, -80.0 * T, 0.0)
    ev = np.exp(shifted / T)
    obj = (cmax + T * np.log(ev.sum() + 1e-300)) * dx
    p = ev / ev.sum()
    grad = np.zeros(n)
    for i in range(k):
        pm = p[i]
        if pm < 1e-16:
            continue
        m = idx[i]
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += pm * (1.0 - h[:n - lag])
            grad[:n - lag] -= pm * h[lag:]
        else:
            pp = -lag
            grad[:n - pp] += pm * (1.0 - h[pp:])
            grad[pp:] -= pm * h[:n - pp]
    grad *= dx
    return obj, grad

def do_slsqp(h, n, T, max_iter=2000, use_topk=False, k_top=5):
    if use_topk:
        def f(x):
            return topk_obj_grad(x, T, k_top)[0]
        def g(x):
            return topk_obj_grad(x, T, k_top)[1]
    else:
        def f(x):
            return soft_max_obj_grad(x, T)[0]
        def g(x):
            return soft_max_obj_grad(x, T)[1]
    try:
        res = opt.minimize(f, h.copy(), jac=g, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(xn, np.r_[xo, xo[-1] + 2.0/old_n, xo[0] - 2.0/old_n],
               np.r_[h[-1:]*0, h, h[:1]*0], period=2.0), new_n) if False else \
           fix_sum(np.interp(xn, xo, h), new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = c5_val(h_best)

    def elapsed():
        return time.time() - t0

    def try_update(hc, nc):
        nonlocal h_best, n_best, c5_best
        hc = fix_sum(hc, nc)
        cv = c5_val(hc)
        if cv < c5_best - 1e-13:
            h_best, n_best, c5_best = hc.copy(), nc, cv
            return True
        return False

    # Phase 1: Coordinate descent (fast 1D minimax per coordinate)
    if budget_s - elapsed() > 60:
        h_cd = coordinate_descent(h_best, n_best, max_rounds=60)
        try_update(h_cd, n_best)

    # Phase 2: SLSQP temperature annealing with analytical gradient
    T_seq = [0.08, 0.03, 0.01, 0.003, 0.001, 0.0004, 0.0001]
    for T in T_seq:
        if budget_s - elapsed() < 60:
            break
        h_opt = do_slsqp(h_best, n_best, T, max_iter=3000)
        try_update(h_opt, n_best)

    # Phase 3: Top-k focused SLSQP
    for k_top in [3, 5, 8]:
        if budget_s - elapsed() < 60:
            break
        for T in [0.005, 0.001, 0.0003]:
            if budget_s - elapsed() < 40:
                break
            h_opt = do_slsqp(h_best, n_best, T, max_iter=2000, use_topk=True, k_top=k_top)
            try_update(h_opt, n_best)

    # Phase 4: Another CD round after SLSQP
    if budget_s - elapsed() > 80:
        h_cd2 = coordinate_descent(h_best, n_best, max_rounds=50)
        try_update(h_cd2, n_best)

    # Phase 5: Multi-start perturbations
    pert_scales = [0.02, 0.05, 0.08, 0.12, 0.15]
    for ps in pert_scales:
        if budget_s - elapsed() < 120:
            break
        h_p = fix_sum(np.clip(h_best + np.random.randn(n_best) * ps, 0, 1), n_best)
        h_p = do_slsqp(h_p, n_best, 0.01, max_iter=1500)
        h_p = do_slsqp(h_p, n_best, 0.001, max_iter=1500)
        try_update(h_p, n_best)

    # Phase 6: Try different resolutions
    for n_try in [200, 500, 720]:
        if budget_s - elapsed() < 150:
            break
        if n_try > n_best:
            price = np.interp(np.arange(n_try)*2.0/n_try, np.arange(n_best)*2.0/n_best, h_best)
        else:
            price = np.interp(np.arange(n_try)*2.0/n_try, np.arange(n_best)*2.0/n_best, h_best)
        h_try = fix_sum(price, n_try)
        h_try = coordinate_descent(h_try, n_try, max_rounds=30)
        for T in [0.01, 0.001]:
            if budget_s - elapsed() < 60:
                break
            h_try = do_slsqp(h_try, n_try, T, max_iter=2000)
            if try_update(h_try, n_try):
                break

    # Phase 7: Final CD + SLSQP polish
    if budget_s - elapsed() > 50:
        h_best = coordinate_descent(h_best, n_best, max_rounds=40)
        for T in [0.0003, 0.0001]:
            if budget_s - elapsed() < 40:
                break
            h_best = do_slsqp(h_best, n_best, T, max_iter=3000)

    # Phase 8: Final micro-perturbation + polish
    if budget_s - elapsed() > 30:
        h_micro = fix_sum(np.clip(h_best + np.random.randn(n_best) * 0.005, 0, 1), n_best)
        h_micro = do_slsqp(h_micro, n_best, 0.0005, max_iter=2000)
        try_update(h_micro, n_best)
        h_best = coordinate_descent(h_best, n_best, max_rounds=20)
        c5_best = c5_val(h_best)

    h_best = fix_sum(np.clip(h_best, 0.0, 1.0), len(h_best))
    c5_final = c5_val(h_best)
    return (h_best, c5_final, len(h_best))
```
