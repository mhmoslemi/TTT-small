from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def lfgs_optimize(h, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h, args=(T,),
                  method='L-BFGS-B', bounds=bnds,
                  options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8, 'maxls': 40})
    return project_feasible(np.clip(res.x, 0, 1), n)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + np.clip(room * (diff / total), -1.0, 1.0)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h - np.clip(room * (-diff / total), 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def lfgs_opt(h, T, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, args=(T,), method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def binary_local_search(h, n, n_iters=5000, rng=None):
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    target = n // 2
    while h_bin.sum() > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[rng.choice(idx)] = 0
    while h_bin.sum() < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[rng.choice(idx)] = 1
    
    best = compute_c5(h_bin)
    for it in range(n_iters):
        j1 = np.where(h_bin == 1)[0]
        j2 = np.where(h_bin == 0)[0]
        a = rng.choice(j1)
        b = rng.choice(j2)
        h_try = h_bin.copy()
        h_try[a] = 0
        h_try[b] = 1
        c5_try = compute_c5(h_try)
        if c5_try < best - 1e-15:
            best = c5_try
            h_bin = h_try
    return h_bin

Actually, I just want to double-check one thing about the correlation computation. For `np.correlate(h, 1-h, mode='full')`:

If h has length n, then the full correlation has length 2n-1.
The value at index m (0-indexed) is:
c[m] = Σ_{i} h[i] * (1-h)[m - i]

where the sum is over all i such that 0 ≤ i < n and 0 ≤ m - i < n.

So c[m] = Σ_{i=max(0, m-n+1)}^{min(n-1, m)} h[i] * (1-h[m-i])

This is the cross-correlation of h with (1-h) at "shift" m.

The physical interpretation: c[m] measures how much h at position i overlaps with (1-h) at position m-i. For the Erdős problem, we want to minimize the maximum of c[m] * dx over all m.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
