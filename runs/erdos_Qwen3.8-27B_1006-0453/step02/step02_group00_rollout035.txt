def build_influence(h, n):
    """Build influence matrix: infl[j,k] = effect on corr[k] per unit change in h[j]"""
    # infl[j,k] = (1-h[j-k])*I(0<=k<=j) - h[j+k]*I(0<=k<=n-1-j)
    infl = np.zeros((n, 2*n-1))

        # Term 1: (1-h[j-k]) for k in [0, j]

        # Term 2: -h[j+k] for k in [0, n-1-j]

    return infl

    return h, compute_c5(h)

def slsqp_optimize(h, T, n, maxiter=600):
    cons = [{'type': 'eq', 'fun': sum_constraint_fun, 'jac': sum_constraint_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP warm start
    for T in [30, 100, 300]:
        if elapsed() > budget_s * 0.08:
            break
        hc = slsqp_optimize(best_h.copy(), T, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Pair-exchange descent (main new technique)
    if elapsed() < budget_s * 0.40:
        h_pe, c5_pe = pair_exchange_descent(best_h.copy(), n, max_passes=200, n_pairs_per_pass=8)
        if c5_pe < best_c5:
            best_c5 = c5_pe
            best_h = h_pe

    # Phase 3: SLSQP refinement after pair exchange
    for T in [500, 1500, 5000]:
        if elapsed() > budget_s * 0.55:
            break
        hc = slsqp_optimize(best_h.copy(), T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Wider pair-exchange (more pairs, different structure)
    if elapsed() < budget_s * 0.70:
        h_pe2, c5_pe2 = pair_exchange_descent(best_h.copy(), n, max_passes=100, n_pairs_per_pass=12)
        if c5_pe2 < best_c5:
            best_c5 = c5_pe2
            best_h = h_pe2

    # Phase 5: Basin hopping with SLSQP polish
    n_iters_bh = 40
    for it in range(n_iters_bh):
        if elapsed() > budget_s * 0.85:
            break
        step = 0.10 * (1.0 - 2.0 * it / n_iters_bh) + 0.02
        h_pert = best_h + step * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        hc = slsqp_optimize(h_pert, 200, n, maxiter=200)
        c5v = compute_c5(hc)
        temp = 0.004 * (1.0 - it / float(n_iters_bh)) + 0.0002
        if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / temp):
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 6: Try different resolutions
    for n_try in [100, 140, 160]:
        if elapsed() > budget_s * 0.92:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h_interp, n_try)
        for T in [200, 800]:
            if elapsed() > budget_s * 0.92:
                break
            hc = slsqp_optimize(h0.copy(), T, n_try, maxiter=500)
            c5v = compute_c5(hc)
            h_pe3, c5_pe3 = pair_exchange_descent(hc.copy(), n_try, max_passes=50, n_pairs_per_pass=8)
            if c5_pe3 < c5v:
                hc = h_pe3
                c5v = c5_pe3
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 7: Final polish
    n_cur = len(best_h)
    for T in [3000, 10000, 30000]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp_optimize(best_h.copy(), T, n_cur, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    if elapsed() < budget_s * 0.995:
        h_final, c5_final = pair_exchange_descent(best_h.copy(), n_cur, max_passes=80, n_pairs_per_pass=10)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
