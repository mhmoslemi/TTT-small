import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_avg_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_wrap(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_descent(h, n_steps, top_k, alpha0, decay):
    n = len(h)
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        wts = corr[idx].astype(float)
        ws = wts.sum()
        if ws <= 0:
            wts = np.ones(top_k) / top_k
        else:
            wts = wts / ws
        
        grad = np.zeros(n)
        for w, ki in zip(wts, idx):
            s = int(ki - (n - 1))
            for j in range(n):
                g = 0.0
                jp = j + s
                jm = j - s
                if 0 <= jp < n:
                    g += (1.0 - h[jp])
                if 0 <= jm < n:
                    g -= h[jm]
                grad[j] += w * g
        
        h = project_feasible(h - alpha * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        alpha *= decay
    return best, best_c5

def block_move(h, n):

    start = np.random.randint(0, max(1, n - 2))
    length = np.random.randint(2, max(3, n // 3))

    shift = np.random.randint(1, max(2, length))
    h_new[start:end] = np.roll(h_new[start:end], shift)
    return h_new

        hsg, c5sg = subgrad_descent(best_h, 250, tk, 0.015, 0.9995)
        if c5sg < best_c5:
            best_c5, best_h = c5sg, hsg

    # Phase 2: Top-k composite SLSQP (different objective than smooth max)
    if time.time() - t0 < budget_s * 0.18:
        for k in [3, 5, 8, 12, 16]:
            ht = slsqp_wrap(best_h, topk_avg_obj, (k, best_n), 350, 1e-12)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 3: Smooth max SLSQP with varied temperatures
    if time.time() - t0 < budget_s * 0.28:
        for T in [100, 500, 2000, 8000]:
            ht = slsqp_wrap(best_h, softmax_obj, (T, best_n), 350, 1e-13)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 4: Structured basin hopping (block moves, swaps, Gaussian)
    n_trials = int(budget_s * 0.30 / 5)
    for trial in range(min(n_trials, 120)):
        if time.time() - t0 > budget_s * 0.60:
            break
        r = np.random.random()
        if r < 0.30:
            hp = block_move(best_h, best_n)
        elif r < 0.55:
            hp = swap_pair(best_h, best_n)
        elif r < 0.70:
            hp = targeted_repair(best_h, best_n)
        else:
            sc = 0.05 * (1.0 - 0.006 * trial) + 0.008
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        # Two-stage refine
        ho = slsqp_wrap(hp, softmax_obj, (400, best_n), 180, 1e-12)
        ho = slsqp_wrap(ho, topk_avg_obj, (5, best_n), 180, 1e-13)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 5: Multi-resolution search
    for nr in [120, 150, 180, 200]:
        if time.time() - t0 > budget_s * 0.80:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        # Small perturbation at higher res
        hu = project_feasible(hu + np.random.randn(nr) * 0.02, nr)
        hh = slsqp_wrap(hu, softmax_obj, (300, nr), 250, 1e-12)
        hh = slsqp_wrap(hh, topk_avg_obj, (5, nr), 250, 1e-13)
        c5_hi = compute_c5(hh)
        # Also try without perturbation
        hu2 = np.interp(x_new, x_old, best_h)
        hu2 = project_feasible(hu2, nr)
        hh2 = slsqp_wrap(hu2, topk_avg_obj, (3, nr), 300, 1e-13)
        c5_hi2 = compute_c5(hh2)
        c5_hi = min(c5_hi, c5_hi2)
        hh = hh if c5_hi2 > c5_hi else hh2
        # Compare: keep at higher res or downsample
        x_dn = np.linspace(0, 2, best_n)
        hd = np.interp(x_dn, x_new, hh)
        hd = project_feasible(hd, best_n)
        c5_dn = compute_c5(hd)
        if c5_dn < best_c5:
            best_c5, best_h = c5_dn, hd
        elif c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, hh, nr

    # Phase 6: Targeted repair + intensive local search
    if time.time() - t0 < budget_s * 0.90:
        for _ in range(5):
            if time.time() - t0 > budget_s * 0.90:
                break
            hr = targeted_repair(best_h, best_n)
            hr = slsqp_wrap(hr, topk_avg_obj, (3, best_n), 300, 1e-14)
            cr = compute_c5(hr)
            if cr < best_c5:
                best_c5, best_h = cr, hr

    # Phase 7: Final polish
    if time.time() - t0 < budget_s * 0.95:
        hf = slsqp_wrap(best_h, topk_avg_obj, (3, best_n), 600, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hf2 = slsqp_wrap(best_h, softmax_obj, (10000, best_n), 600, 1e-15)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2

    # Phase 8: Micro sub-gradient
    if time.time() - t0 < budget_s * 0.985:
        hs, cs = subgrad_descent(best_h, 100, 3, 0.003, 0.999)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        hf3 = slsqp_wrap(best_h, topk_avg_obj, (3, best_n), 200, 1e-15)
        cf3 = compute_c5(hf3)
        if cf3 < best_c5:
            best_c5, best_h = cf3, hf3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
