def binary_local_search(h, n, rng, max_iters=20000):
    """Local search on binary patterns via single swaps."""
    target = n / 2
    # Make binary
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly target ones
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Remove excess ones
        ones_idx = np.where(h_bin == 1)[0]
        remove = rng.choice(ones_idx, n_ones - target, replace=False)
        h_bin[remove] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h_bin[add] = 1
    
    best_c5 = compute_c5(h_bin)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    for it in range(max_iters):
        if not ones or not zeros:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # Swap
        h_bin[i] = 0
        h_bin[j] = 1
        ones.discard(i)
        ones.add(j)
        zeros.discard(j)
        zeros.add(i)
        c5 = compute_c5(h_bin)
        if c5 < best_c5:
            best_c5 = c5
        else:
            # Revert
            h_bin[i] = 1
            h_bin[j] = 0
            ones.discard(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
    return h_bin, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_exact(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_lse(h, t, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nk = len(corr)
    if topk and topk < nk:
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    mx = np.max(corr)
    s = t * (corr - mx)
    sm = np.max(s)
    return (mx + (sm + np.log(np.exp(s - sm).sum())) / t) * dx

def single_k_obj(x, k_star):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return corr[k_star] * (2.0 / len(x))

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceq_jac(x):
    return np.ones((1, len(x)))

def reshape_interp(h, nd):
    n = len(h)
    if n == nd:
        return proj_exact(h, nd / 2.0)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj_exact(np.interp(xd, xs, h), nd / 2.0)

def hill_climb(h, n, rng, max_iters, s0, s1):
    tgt = n / 2.0
    bh = proj_exact(h.copy(), tgt)
    bc = compute_c5(bh)
    for it in range(max_iters):
        sigma = s0 * (s1 / s0) ** (it / max(max_iters - 1, 1))
        p = proj_exact(bh + rng.randn(n) * sigma, tgt)
        c = compute_c5(p)
        if c < bc:
            bc, bh = c, p
    return bh, bc

    bnds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': ceq, 'jac': ceq_jac},)
    r = minimize(obj_lse, h, args=(t, topk), method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-14})
    if r.x is None or not np.all(np.isfinite(r.x)):
        return h
    return proj_exact(np.clip(r.x, 0.0, 1.0), n / 2.0)

def do_single_k(h, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': ceq, 'jac': ceq_jac},)
    k1 = int(np.argmax(np.correlate(h, 1.0 - h, mode='full')))
    r = minimize(single_k_obj, h, args=(k1,), method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-14})
    if r.x is None or not np.all(np.isfinite(r.x)):
        return h

        if r2.x is not None and np.all(np.isfinite(r2.x)):
