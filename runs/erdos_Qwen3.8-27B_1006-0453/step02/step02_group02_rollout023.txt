def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_lse(h, t, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        c = corr[idx]
    else:
        c = corr
    mx = np.max(c)
    s = t * (c - mx)
    se = np.sum(np.exp(s))
    return (mx + np.log(se) / t) * dx

def obj_pen(h, t, k, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

        idx = np.argpartition(corr, -k)[-k:]
        c = corr[idx]
    else:
        c = corr
    mx = np.max(c)
    s = t * (c - mx)

def lbfgs_opt(h0, t, k, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    r = minimize(obj_pen, h0, args=(t, k, 500.0), method='L-BFGS-B', bounds=bnds,
                 options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxiter * n})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def binary_sa(h, n, n_iter, T0, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T0
    cool = max(0.005, (0.001 / T0)) ** (1.0 / n_iter)
    for it in range(n_iter):
        T *= cool
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        new_c5 = compute_c5(h)
        acc = new_c5 <= cur_c5 or rng.rand() < np.exp((cur_c5 - new_c5) / max(T, 1e-12))
        if acc:
            cur_c5 = new_c5
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 280

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = proj(np.array(initial_h_values, dtype=float), n)
    except Exception:
        bh = proj(np.full(n, 0.5) + rng.randn(n) * 0.05, n)
    bc = compute_c5(bh)

    # Alternate structured starts for diversity
    alt_starts = []
    block = np.zeros(n)
    for i in range(0, n, 7):
        block[i:i+4] = 1.0
    alt_starts.append(proj(block, n))
    block2 = np.zeros(n)
    for i in range(0, n, 10):
        block2[i:i+5] = 1.0
    alt_starts.append(proj(block2, n))

    # Phase 1: SLSQP coarse-to-fine t, large topk to small
    phase1_configs = [
        (20, 400, 600, 1e-11),
        (50, 200, 800, 1e-12),
        (100, 100, 1000, 1e-12),
        (250, 60, 1200, 1e-13),
        (500, 40, 1400, 1e-13),
        (1000, 25, 1500, 1e-14),
    ]
    for t_val, k_val, mx_it, ftol in phase1_configs:
        if tleft() < 50:
            break
        for st in [bh] + alt_starts[:1]:
            if tleft() < 30:
                break
            try:
                hc = slsqp_opt(st, t_val, k_val, n, mx_it, ftol)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc
            except Exception:
                pass

    # Phase 2: L-BFGS-B with penalty (different landscape)
    if tleft() > 250:
        for t_val, k_val in [(300, 50), (1000, 30), (3000, 15), (8000, 10)]:
            if tleft() < 30:
                break
            try:
                hc = lbfgs_opt(bh, t_val, k_val, n, 2000, 1e-14)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc
            except Exception:
                pass

    # Phase 3: High-t SLSQP with shrinking topk
    phase3_configs = [
        (2000, 20, 2000, 1e-14),
        (5000, 12, 2500, 1e-14),
        (10000, 8, 3000, 1e-15),
        (20000, 6, 3500, 1e-15),
        (50000, 4, 4000, 1e-15),
        (100000, 3, 5000, 1e-16),
        (200000, 2, 6000, 1e-16),
    ]
    for t_val, k_val, mx_it, ftol in phase3_configs:
        if tleft() < 40:
            break
        # Multiple starts: current + small perturbation
        starts = [bh]
        if tleft() > 80:
            starts.append(proj(bh + rng.randn(n) * 0.005, n))
        for st in starts:
            if tleft() < 25:
                break
            try:
                hc = slsqp_opt(st, t_val, k_val, n, mx_it, ftol)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc
            except Exception:
                pass

    # Phase 4: Binary simulated annealing for diversification
    if tleft() > 200:
        h_bin = proj((bh > 0.5).astype(float), n)
        for T0 in [0.03, 0.015, 0.008]:
            if tleft() < 120:
                break
            n_sa = min(400000, int((tleft() - 120) * 2500))
            if n_sa < 20000:
                n_sa = 20000
            try:
                hb, cb = binary_sa(h_bin, n, n_sa, T0, seed + T0)
                if cb < bc:
                    bc = cb
                    bh = proj(hb, n)
                h_bin = proj(hb, n)
            except Exception:
                pass

    # Phase 5: Re-optimize from binary best (if improved)
    if tleft() > 100:
        h_bin = proj((bh > 0.5).astype(float), n)
        hb_c5 = compute_c5(h_bin)
        for st in [bh, h_bin]:
            if tleft() < 30:
                break
            try:
                hc = slsqp_opt(st, 50000, 4, n, 5000, 1e-16)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc
            except Exception:
                pass

    # Phase 6: Ultra-fine polish
    if tleft() > 20:
        for t_val, k_val in [(300000, 2), (500000, 1)]:
            if tleft() < 12:
                break
            try:
                hc = slsqp_opt(bh, t_val, k_val, n, 8000, 1e-17)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc
            except Exception:
                pass

    # Final binary check
    h_fb = proj((bh > 0.5).astype(float), n)
    if compute_c5(h_fb) < bc:
        bh = h_fb
        bc = compute_c5(bh)

    return bh, compute_c5(bh), n
