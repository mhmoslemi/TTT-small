import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, T, n, maxiter=500, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5, h, args=(T,),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def pair_line_search(h, n, deadline, n_pairs_per_sweep=5000, n_gs_iters=14):
    best = h.copy()
    best_c5 = compute_c5(best)
    gr = (np.sqrt(5) - 1) / 2
    total_sweeps = 0
    no_improve_sweeps = 0
    while time.time() < deadline:
        total_sweeps += 1
        improved = False
        for _ in range(n_pairs_per_sweep):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if i > j:
                i, j = j, i
            rest_sum = best.sum() - best[i] - best[j]
            s_val = n / 2.0 - rest_sum
            lo = max(0.0, s_val - 1.0)
            hi = min(1.0, s_val)
            if lo >= hi - 1e-12:
                continue
            a, b = lo, hi
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            h_c = best.copy()
            h_c[i] = c_
            h_c[j] = s_val - c_
            h_d = best.copy()
            h_d[i] = d_
            h_d[j] = s_val - d_
            f_c = compute_c5(h_c)
            f_d = compute_c5(h_d)
            for _ in range(n_gs_iters):
                if f_c < f_d:
                    b = d_
                    d_ = c_
                    f_d = f_c
                    c_ = b - gr * (b - a)
                    h_c = best.copy()
                    h_c[i] = c_
                    h_c[j] = s_val - c_
                    f_c = compute_c5(h_c)
                else:
                    a = c_
                    c_ = d_
                    f_c = f_d
                    d_ = a + gr * (b - a)
                    h_d = best.copy()
                    h_d[i] = d_

            h_new[i] = best_x

def directed_gradient(h, n, iters=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(iters):
