    return np.max(corr) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        lag = m_idx - (n - 1)
        # First term: v[i+lag] for i in [max(0,-lag), min(n, n-lag))
        i1s = max(0, -lag)
        i1e = min(n, n - lag)
        if i1e > i1s:
            grad[i1s:i1e] += wm * v[i1s + lag:i1e + lag]
        # Second term: -h[i-lag] for i in [max(0,lag), min(n, n+lag))
        i2s = max(0, lag)
        i2e = min(n, n + lag)
        if i2e > i2s:
            grad[i2s:i2e] -= wm * h[i2s - lag:i2e - lag]
    
    return grad * dx

    target = n / 2.0
    s = h.sum()
    diff = target - s

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0, 1)

    grad = smooth_c5_grad(h, T)
    h_new = h - lr * grad
    return project_sum(h_new, n / 2.0)

    h_cur = h.copy()
    f_cur = smooth_c5_val(h_cur, T)

            break
        
        h_cur = h_try
        f_cur = f_try

                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Different resolutions
    for n_try in [80, 100, 140, 160, 200]:
        if elapsed() > budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)
        for T in [20, 100, 400]:
            if elapsed() > budget_s * 0.82:
                break
            h_res = pgd_optimize(h_interp, T, n_try, max_iter=2500, lr_init=0.003)
            c5v = compute_c5(h_res)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_res
            h_interp = h_res

    # Phase 6: Coordinate search on worst shifts
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.93:
        for it_cd in range(400):
            if elapsed() > budget_s * 0.93:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top_k = np.argsort(corr)[-4:]
            improved = False
            for m_idx in top_k:
                for _ in range(8):
                    i = np.random.randint(n_cur)
                    for delta in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                        h_try = best_h.copy()
                        h_try[i] += delta
                        h_try = project_sum(h_try, n_cur / 2.0)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-13:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
                            break
                    if improved:
                        break
            if not improved and it_cd > 80:
                break

    # Phase 7: Final high-T SLSQP
    n_cur = len(best_h)
    for T in [5000, 15000]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_with_grad(best_h, T, n_cur, maxiter=1200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
