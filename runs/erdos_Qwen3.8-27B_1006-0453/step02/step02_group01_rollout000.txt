            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Simulated annealing on exact C5
    if _elapsed() < _budget * 0.88:
        h_sa, c5_sa = sim_anneal(best_h, n_iters=2000, seed=seed + 1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 7: Swap-based local search
    if _elapsed() < _budget * 0.92:
        best_h, best_c5 = swap_perturb(best_h, n_tries=200)

    # Phase 8: Final coordinate descent + SLSQP polish
    if _elapsed() < _budget * 0.95:
        best_h, best_c5 = coord_descent(best_h, rounds=100)

    if _elapsed() < _budget * 0.98:
        h_final = slsqp_topm(best_h, 2, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        h_sm = slsqp_smooth(best_h, 10000, maxiter=400)
        c5_sm = compute_c5(h_sm)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
