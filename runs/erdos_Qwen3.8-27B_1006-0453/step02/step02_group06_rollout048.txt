import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def do_trust(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='trust-constr',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'gtol': 1e-10, 'verbose': 0}
    )
    return proj_constraint(res.x, n)

def subgrad_step(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    offset = k_star - (n - 1)
    grad = np.ones(n)
    if offset > 0:
        grad[:n - offset] -= h[offset:]
        grad[offset:] -= h[:n - offset]
    elif offset < 0:
        grad[-offset:] -= h[:n + offset]
        grad[:n + offset] -= h[-offset:]
    else:
        grad -= 2.0 * h
    return grad

def subgradient_refine(h0, n, t_budget, max_steps, lr_init):
    t_start = time.time()
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    lr = lr_init
    for step in range(max_steps):
        if time.time() - t_start > t_budget:
            break
        grad = subgrad_step(h_cur, n)
        h_new = proj_constraint(h_cur - lr * grad, n)
        c_new = compute_c5(h_new)
        if c_new < c_best:
            c_best = c_new
            h_best = h_new.copy()
        if c_new <= c_cur:
            lr *= 1.03
        else:
            lr *= 0.6
            h_cur = proj_constraint(h_cur + 0.5 * lr * grad * -1, n)
            c_cur = compute_c5(h_cur)
        else:
            h_cur = h_new
            c_cur = c_new
        h_cur = h_new
        c_cur = c_new
    return h_best, c_best

def weighted_lag_obj(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    return np.mean(corr[idx]) * dx

    np.random.seed(seed)

    h_best = np.array(initial_h_values, dtype=float).copy()

            c5_best = c

    # Phase 6: Weighted-lag objective optimization

                    c5_best = c
            except Exception:
                pass

    # Phase 7: Simulated annealing fine-tuning
    elapsed = time.time() - t_start
    if budget_s - elapsed > 40:
        rng = np.random.RandomState(seed + 4242)
        h_cur = h_best.copy()
        c_cur = c5_best
        T_sa = 0.0002
        for step in range(300):
            if time.time() - t_start > budget_s - 10:
                break

            idxs = rng.choice(n_best, size=n_blk, replace=False)
            h_try = h_cur.copy()
            h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_blk) * 0.015, 0, 1)
            h_try = proj_constraint(h_try, n_best)

    if budget_s - elapsed > 20:

            if c < c5_best:

                c5_best = c

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
