            if hpb[1] < bc:
                bh, bc = hpb

    hper = make_periodic(n, ts, rng)
    if time.time() < dl - 80:
        hp, cp = gd_run(hper, dx, ts, dl - 50, rng, 1200)
        if cp < bc:
            bh, bc = hp, cp

    # Phase 8: Try nearby n values
    for dn in (1, -1, 2, -2, 3):
        nt = n + dn
        if nt < 20 or nt > 90 or time.time() > dl - 60:
            break
        dxt = 2.0 / nt
        tst = nt / 2.0
        xq_new = np.linspace(0, 1, nt, endpoint=False)
        xq_old = np.linspace(0, 1, len(bh), endpoint=False)
        h_interp = np.interp(xq_new, xq_old, bh)
        h6 = proj(h_interp, tst)
        h7, c7 = gd_run(h6, dxt, tst, dl - 40, rng, 1500)
        if c7 < bc:
            bh, bc = h7, c7
        if c7 < bc:
            h8, c8 = shift_refine(h7, dxt, tst, dl - 20, 500)
            if c8 < bc:
                bh, bc = h8, c8

    # Final polish
    n = len(bh)
    dx = 2.0 / n
    ts = n / 2.0
    if time.time() < dl - 15:
        hf = slsqp_opt(bh, dx, ts, 100)
        cf = compute_c5(hf, dx)
        if cf < bc:
            bh, bc = hf, cf
    if time.time() < dl - 5:
        hd, cd = coord_descent(bh, dx, ts, dl - 2, rng, 30)
        if cd < bc:
            bh, bc = hd, cd

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
