import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:

            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:

            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=200):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter=200):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.02
    
    def obj(x):
        return x[-1]
    def eq(x):
        return x[:n].sum() - n / 2.0
    def ineq(x):
        h_v = x[:n]
        t = x[-1]
        c = np.correlate(h_v, 1.0 - h_v, mode='full')
        return t - c  # t - c[k] >= 0 for all k
    
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': eq}, {'type': 'ineq', 'fun': ineq}]

                      bounds=bnds, constraints=cons,

        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

    """Genetic algorithm on binary vectors with exactly n/2 ones."""
    t_start = time.time()
    
    # Initialize population
    pop = []
    if start_h is not None:
        pop.append(start_h.copy())
    while len(pop) < pop_size:
        pop.append(random_binary(n, rng))
    
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)
    for h in pop:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    for gen in range(n_gen):
        if time.time() - t_start > time_limit:
            break
        # Evaluate all
        scores = [compute_c5(h) for h in pop]
        # Update best
        for i, c5 in enumerate(scores):
            if c5 < best_c5:
                best_c5 = c5
                best_h = pop[i].copy()
        
        # Selection: tournament
        new_pop = [best_h.copy()]  # elitism
        for _ in range(pop_size - 1):
            # Pick 3, take best
            idx = rng.integers(0, pop_size, 3)
            winner = idx[int(np.argmin([scores[i] for i in idx]))]

- The term h[i-k]*(1-h[i]) goes from h[i-k]*0 to h[i-k]*1: change = h[i-k]
Net change in full_corr[k] from flipping i: h[i+k] - 1 + h[i-k]

When h[j] goes from 0 to 1:
- The term h[j]*(1-h[j+k]) goes from 0 to (1-h[j+k]): change = 1-h[j+k]
- The term h[j-k]*(1-h[j]) goes from h[j-k]*1 to h[j-k]*0: change = -h[j-k]
Net change in full_corr[k] from flipping j: 1-h[j+k] - h[j-k]

Total change in full_corr[k]: (h[i+k] - 1 + h[i-k]) + (1-h[j+k] - h[j-k])
= h[i+k] + h[i-k] - h[j+k] - h[j-k]

So the change in full_corr[k] depends only on h at positions i±k and j±k!

This means I can compute the effect of a swap (i→0, j→1) on all correlation values efficiently:
delta[k] = h[i+k] + h[i-k] - h[j+k] - h[j-k] (with appropriate boundary handling)

Then new_full_corr[k] = full_corr[k] + delta[k], and new C5 = max(new_full_corr) * dx.

This is O(n) per swap evaluation instead of O(n²) for the full correlation. For n=90, that's a 2x speedup. But more importantly, I can precompute the correlation once and then evaluate all swaps quickly.
