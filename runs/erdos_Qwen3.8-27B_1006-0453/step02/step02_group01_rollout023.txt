        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = binary_pattern(42, 90)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with continuation on beta
    for beta in [20, 60, 200, 600]:
        if time.time() - t0 > budget_s * 0.10:
            break
        h_try = slsqp_lse(best_h, beta, 500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Long subgradient with annealing
    h_sg, c5_sg = subgrad_optimize(best_h, best_n, 30000, 30, 3000, 0.10, 0.99985, t0, budget_s * 0.40)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg

    # Phase 3: SLSQP after subgradient
    if time.time() - t0 < budget_s * 0.50:
        h_try = slsqp_lse(best_h, 400, 500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: Tighter subgradient (high beta, small lr)
    h_sg2, c5_sg2 = subgrad_optimize(best_h, best_n, 40000, 500, 15000, 0.04, 0.99992, t0, budget_s * 0.62)
    if c5_sg2 < best_c5:
        best_c5 = c5_sg2
        best_h = h_sg2

    # Phase 5: Try alternative starting points
    if time.time() - t0 < budget_s * 0.70:
        alts = [binary_pattern(42, 90), mseq_pattern(90)]
        for s in range(4):
            if time.time() - t0 > budget_s * 0.72:
                break
            alts.append(project_feasible(np.random.rand(90), 90))
        for h_alt in alts:
            if time.time() - t0 > budget_s * 0.72:
                break
            h_t = slsqp_lse(h_alt, 100, 200)
            h_t, c5_t = subgrad_optimize(h_t, 90, 5000, 50, 2000, 0.06, 0.99988, t0, budget_s * 0.72)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    # Phase 6: Resolution increase (interpolate to 2n, re-optimize)
    if time.time() - t0 < budget_s * 0.78:
        n2 = best_n * 2
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n2)
        h_hires = np.interp(x_new, x_old, best_h)
        h_hires = project_feasible(h_hires, n2)
        h_hires, c5_hires = subgrad_optimize(h_hires, n2, 8000, 100, 3000, 0.03, 0.99993, t0, budget_s * 0.82)
        if c5_hires < best_c5:
            best_c5 = c5_hires
            best_h = h_hires
            best_n = n2

    # Phase 7: Final SLSQP polish at multiple betas
    for beta in [1000, 3000, 10000]:
        if time.time() - t0 > budget_s * 0.95:
            break
        h_try = slsqp_lse(best_h, beta, 500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 8: Final micro-subgradient
    h_fin, c5_fin = subgrad_optimize(best_h, best_n, 15000, 5000, 50000, 0.015, 0.99995, t0, budget_s * 0.97)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
