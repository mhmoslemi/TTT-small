# Top-level helpers (no lambdas, no closures)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return float(dx * (m + np.log(np.exp(T * (corr - m)).sum()) / T))

def smooth_grad_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    ex = np.exp(T * (corr - m))
    s = ex.sum()
    p = ex / s
    val = dx * (m + np.log(s) / T)
    dvals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        d = int(dvals[k])
        lo1, hi1 = max(0, d), min(n, n + d)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1 - d:hi1 - d])
        lo2, hi2 = max(0, -d), min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2 + d:hi2 + d]
    grad *= dx
    return val, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-13:
                h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot > 1e-13:
                h *= (1.0 + diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x, n):
    return x.sum() - n / 2.0

def grad_desc(h, T, n, n_steps, lr0):
    h = h.copy()
    lr = lr0
    for _ in range(n_steps):

        h_try = project_feasible(h - lr * g, n)
        # Backtracking line search

        v1 = smooth_obj(h_try, T)
        if v1 < v0:
            h = h_try
            lr = min(lr * 1.08, 0.2)
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    return h

def slsqp_run(h, T, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
