import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_obj(h, T, dx):
    corr = compute_corr(h)
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * math.log(max(s, 1e-300))) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int == 0:
            break
        delta = diff / n_int
        if abs(delta) > 0.49:
            if delta > 0:
                h[h < 0.5] += 0.49
            else:
                h[h > 0.5] -= 0.49
        else:
            h[interior] += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)

def exact_subgradient(h, n, max_iters=5000, init_step=0.05):
    dx = 2.0 / n
    h_cur = h.copy()
    best = compute_c5(h_cur)
    step = init_step
    for it in range(max_iters):
        corr = compute_corr(h_cur)
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        grad = np.zeros(n)
        v1 = np.arange(n) - k
        m1 = (v1 >= 0) & (v1 < n)
        grad[m1] += (1.0 - h_cur[v1[m1]])
        v2 = np.arange(n) + k
        m2 = (v2 >= 0) & (v2 < n)
        grad[m2] -= h_cur[v2[m2]]
