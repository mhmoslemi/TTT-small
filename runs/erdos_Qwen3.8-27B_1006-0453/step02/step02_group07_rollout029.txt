def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    k = np.sum(interior)

        if len(interior2) > 0:

            h = np.clip(h, 0.0, 1.0)

def circ_smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    C = np.zeros(n)
    Fh = np.fft.fft(h)
    Fomh = np.fft.fft(1.0 - h)
    C = np.real(np.fft.ifft(Fomh * np.conj(Fh)))
    cmax = np.max(C)
    expv = np.exp((C - cmax) / T)
    S = np.sum(expv)
    p = expv / S
    obj = (cmax + T * np.log(S)) * dx
    
    p_flip = np.zeros(n)
    p_flip[0] = p[0]
    if n > 1:
        p_flip[1:] = p[:0:-1]
    
    Fp = np.fft.fft(p)
    Fpf = np.fft.fft(p_flip)
    A = np.real(np.fft.ifft(Fpf * np.conj(Fh)))  # A[j] = Σ_k p[k]h[j+k]
    B = np.real(np.fft.ifft(Fp * np.conj(Fh)))   # B[j] = Σ_k p[k]h[j-k]
    
    grad = dx * (1.0 - A - B)
    return obj, grad
