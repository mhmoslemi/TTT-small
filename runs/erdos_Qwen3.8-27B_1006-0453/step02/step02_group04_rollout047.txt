    x_old = np.arange(old_n) * (2.0 / old_n)

    h_ext = np.concatenate([h, h, h])
    x_ext = np.concatenate([x_old, x_old + 2.0, x_old + 4.0])
    x_clipped = np.clip(x_new, 0, 2.0)
    h_new = np.interp(x_clipped, x_ext, h_ext)
    return fix_sum(h_new, new_n)

def structured_perturb(h, n, rng):
    h_new = h.copy()
    action = rng.randint(4)
    if action == 0:
        shift = rng.randint(1, max(2, n // 5))
        h_new = np.roll(h_new, shift)
    elif action == 1:
        start = rng.randint(0, max(1, n - 10))
        length = rng.randint(5, max(6, n // 4))
        end = min(start + length, n)
        h_new[start:end] = 1.0 - h_new[start:end]
    elif action == 2:
        freqs = rng.choice([2, 3, 4, 5, 7], size=2, replace=False)
        phases = rng.uniform(0, 2 * np.pi, size=2)
        x = np.arange(n)
        noise = np.zeros(n)
        for f, p in zip(freqs, phases):
            noise += 0.08 * np.sin(2 * np.pi * f * x / n + p)
        h_new = h + noise
    else:
        h_new = h + rng.randn(n) * 0.04
    return fix_sum(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    h_best = fix_sum(h_best, n_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t_start

    def try_update(h_cand, n_cand):
        nonlocal h_best, n_best, c5_best
        h_cand = fix_sum(h_cand, n_cand)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best - 1e-15:
            h_best = h_cand.copy()
            n_best = n_cand
            c5_best = c5_cand
            return True
        return False

    # Phase 1: L-BFGS-B temperature sweep (fast)
    T_seq = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.00015, 0.00008]
    for T in T_seq:
        if budget_s - elapsed() < 60:
            break
        h_opt = do_lbfgs(h_best, n_best, T, max_iter=5000)
        try_update(h_opt, n_best)

    # Phase 2: SLSQP fine polish
    for T in [0.0004, 0.00015, 0.00008]:
        if budget_s - elapsed() < 50:
            break
        h_opt = do_slsqp(h_best, n_best, T, max_iter=3000)
        try_update(h_opt, n_best)

    # Phase 3: Active-set minimax (equalize top shifts)
    for top_k in [3, 5, 7, 10]:
        if budget_s - elapsed() < 80:
            break
        for T in [0.001, 0.0003]:
            if budget_s - elapsed() < 50:
                break
            h_ref = active_set_minimax(h_best, n_best, T, top_k=top_k, max_iter=3000)
            try_update(h_ref, n_best)

    # Phase 4: Multi-start with structured perturbations
    for restart in range(40):
        if budget_s - elapsed() < 180:
            break
        if restart % 3 == 0:
            h_pert = h_best + rng.randn(n_best) * 0.04
        elif restart % 3 == 1:
            h_pert = structured_perturb(h_best, n_best, rng)
        else:
            h_pert = h_best + rng.randn(n_best) * 0.12
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = fix_sum(h_pert, n_best)
        h_opt = do_lbfgs(h_pert, n_best, 0.01, max_iter=2000)
        h_opt = do_lbfgs(h_opt, n_best, 0.002, max_iter=2000)
        improved = try_update(h_opt, n_best)
        if improved and budget_s - elapsed() > 80:
            h_ref = active_set_minimax(h_best, n_best, 0.0005, top_k=5, max_iter=2000)
            try_update(h_ref, n_best)

    # Phase 5: Alternative resolutions
    for n_try in [240, 320, 480, 600]:
        if budget_s - elapsed() < 120:
            break
        if n_try == n_best:
            continue
        h_init = periodic_interp(h_best, n_try)
        h_opt = do_lbfgs(h_init, n_try, 0.01, max_iter=3000)
        h_opt = do_lbfgs(h_opt, n_try, 0.002, max_iter=3000)
        if try_update(h_opt, n_try):
            h_opt2 = active_set_minimax(h_best, n_best, 0.0005, top_k=5, max_iter=2000)
            try_update(h_opt2, n_best)

    # Phase 6: Binary snap + refine
    if budget_s - elapsed() > 60:
        h_bin = (h_best > 0.5).astype(float)
        h_bin = fix_sum(h_bin, n_best)
        h_opt = do_lbfgs(h_bin, n_best, 0.005, max_iter=2000)
        h_opt = do_slsqp(h_opt, n_best, 0.0002, max_iter=2000)
        try_update(h_opt, n_best)

    # Phase 7: Boundary snap (nudge values near 0/1)
    if budget_s - elapsed() > 50:
        h_fine = h_best.copy()
        h_fine[h_fine < 0.03] = 0.0
        h_fine[h_fine > 0.97] = 1.0
        h_fine = fix_sum(h_fine, n_best)
        h_opt = do_lbfgs(h_fine, n_best, 0.002, max_iter=2000)
        h_opt = do_slsqp(h_opt, n_best, 0.0001, max_iter=2000)
        try_update(h_opt, n_best)

    # Phase 8: Final SLSQP ultra-fine polish
    for T in [0.0001, 0.00005, 0.00002]:
        if budget_s - elapsed() < 35:
            break
        h_opt = do_slsqp(h_best, n_best, T, max_iter=5000)
        try_update(h_opt, n_best)

    # Phase 9: One last micro-perturbation + polish
    if budget_s - elapsed() > 50:
        h_micro = h_best + rng.randn(n_best) * 0.01
        h_micro = np.clip(h_micro, 0, 1)
        h_micro = fix_sum(h_micro, n_best)
        h_opt = do_lbfgs(h_micro, n_best, 0.0005, max_iter=3000)
        h_opt = do_slsqp(h_opt, n_best, 0.00005, max_iter=3000)
        try_update(h_opt, n_best)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
