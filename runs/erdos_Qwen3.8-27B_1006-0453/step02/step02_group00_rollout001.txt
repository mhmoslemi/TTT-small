    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()

        if tot < 1e-15:
            return h
        h += h * (diff / tot)

        return smooth_c5(x, T)

        return x.sum() - n / 2.0

                   constraints=[{'type': 'eq', 'fun': eqc}],

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        shift = k - (n - 1)
        wi = w[k]
        for i in range(n):
            j1 = i - shift
            j2 = i + shift
            if 0 <= j1 < n:
                grad[i] += wi * (1.0 - h[j1])
            if 0 <= j2 < n:
                grad[i] -= wi * h[j2]
