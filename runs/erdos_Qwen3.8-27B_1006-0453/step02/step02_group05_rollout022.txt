def lag_gradient(h, L):
    """Gradient of corr_L w.r.t. h. L is signed lag."""
    N = len(h)
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        if L >= N:
            return g
        g[:N-L] += 1.0 - h[L:]
        g[L:] -= h[:N-L]
    else:
        M = -L
        if M >= N:
            return g
        g[M:] += 1.0 - h[:N-M]
        g[:N-M] -= h[M:]
    return g

def optimize(h, N, target, deadline, k=3, lr0=0.03, momentum=0.9, decay=0.99):
    """Momentum-based gradient descent with top-k objective."""
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/N)
    dx = 2.0/N
    lr = lr0
    vel = np.zeros(N)
    it = 0
    stagnation = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0-h, mode='full')
        c5_val = np.max(corr) * dx
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Top-k gradient
        center = N - 1
        top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top_idx:
            grad += lag_gradient(h, ai - center)
        grad /= k

        vel = momentum * vel - lr * grad
        h = project_sum(h + vel, target)
        
        it += 1
        if it % 200 == 0:
            lr *= decay
        if lr < 1e-10 or stagnation > 500:
            break
    return best_h, best_c5

        h_new[i] += delta
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5
