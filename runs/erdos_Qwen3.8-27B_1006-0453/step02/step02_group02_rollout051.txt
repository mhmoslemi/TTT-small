```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    return project(h_bin, n)

def binary_local_search(h, n, max_epochs=500, time_limit=None, t0=None):
    """1-swap local search on binary h."""
    h = to_binary(h, n)
    best_c5 = compute_c5(h)
    
    for epoch in range(max_epochs):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = np.max(corr)
        
        best_improve = 0
        best_swap = None
        
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                new_max = np.max(corr_new)
                improve = current_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
                h[j] = 0.0
            h[i] = 1.0
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return h, best_c5
```

    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        c = corr
    mx = np.max(c)
    s = t * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def slsqp(h, n, t_val, topk, maxiter, ftol):
    bnd = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    r = minimize(smooth_obj, h, args=(t_val, topk), method='SLSQP',
                 bounds=bnd, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(r.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    rng = np.random.RandomState(seed)
    n = 280

    try:
        h0 = project(np.array(initial_h_values, dtype=float), n)
    except Exception:
        h0 = project(np.full(n, 0.5) + rng.randn(n) * 0.05, n)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Binary local search from parent (25%)
    d1 = t0 + budget_s * 0.25
    bh, bc = binary_local_search(h0, n, d1)
    if bc < best_c5:
        best_c5, best_h = bc, bh.copy()

    # Phase 2: SLSQP continuous refinement (20%)
    d2 = t0 + budget_s * 0.45
    hc = best_h.copy()
    for topk in [8, 4, 20, 5]:
        if time.time() > d2:
            break
        for t_val in [100, 500, 3000]:
            if time.time() > d2:
                break
            try:
                hn = slsqp(hc, n, t_val, topk, 1200, 1e-15)
                c5n = compute_c5(hn)
                if c5n < best_c5:
                    best_c5, best_h = c5n, hn.copy()
                hc = hn
            except Exception:
                pass

    # Phase 3: Binary SA to escape local optima (15%)
    d3 = t0 + budget_s * 0.60
    bs, bcs = binary_sa(best_h, n, d3, rng, t0_fa=0.005)
    if bcs < best_c5:
        best_c5, best_h = bcs, bs.copy()
    # Local search after SA
    bh3, bc3 = binary_local_search(best_h, n, d3 + (deadline - d3) * 0.3)
    if bc3 < best_c5:
        best_c5, best_h = bc3, bh3.copy()

    # Phase 4: Multi-resolution explore (15%)
    d4 = t0 + budget_s * 0.75
    for n_med in [140, 200]:
        if time.time() > d4:
            break
        hm = interp_down(best_h, n_med)
        hbm = to_binary(hm, n_med)
        hbm, bcm = binary_local_search(hbm, n_med, min(d4, time.time() + 80))
        if bcm < best_c5:
            best_c5 = bcm
            best_h = hbm.copy()
        # Upscale and local search at n=280
        hup = interp_up(hbm, n)
        hbup = to_binary(hup, n)
        hbup, bcup = binary_local_search(hbup, n, min(d4, time.time() + 40))
        if bcup < best_c5:
            best_c5, best_h = bcup, hbup.copy()

    # Phase 5: Fine SLSQP polish (15%)
    d5 = t0 + budget_s * 0.90
    hf = best_h.copy()
    for topk in [3, 5, 8]:
        if time.time() > d5:
            break
        for t_val in [500, 3000, 15000]:
            if time.time() > d5:
                break
            try:
                hn = slsqp(hf, n, t_val, topk, 2500, 1e-16)
                c5n = compute_c5(hn)
                if c5n < best_c5:
                    best_c5, best_h = c5n, hn.copy()
                hf = hn
            except Exception:
                pass

    # Phase 6: Final binary polish
    if time.time() < deadline:
        hb_fin = to_binary(best_h, n)
        hb_fin, bc_fin = binary_local_search(hb_fin, n, deadline)
        if bc_fin < best_c5:
            best_c5, best_h = bc_fin, hb_fin.copy()

    return best_h, best_c5, n
