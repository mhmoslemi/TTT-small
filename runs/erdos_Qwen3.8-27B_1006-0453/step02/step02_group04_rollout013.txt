def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-13:

        else:

            idx = np.argsort(-h)

                h[i] -= rem
