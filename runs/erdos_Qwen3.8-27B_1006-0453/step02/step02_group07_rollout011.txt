import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        mask = corr >= c_max - 1e-10
        w = np.zeros(2 * n - 1)
        w[mask] = 1.0
        w /= w.sum()
    else:
        w = np.exp((corr - c_max) / T)
        w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        j0, j1 = max(0, -s), min(n, n - s)
        if j1 > j0:
            grad[j0:j1] += wk * (1.0 - h[j0 + s:j1 + s])
        j0b, j1b = max(0, s), min(n, n + s)
        if j1b > j0b:
            grad[j0b:j1b] -= wk * h[j0b - s:j1b - s]
    return dx * grad

        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def coord_brent(h, n, rng, n_rounds=2):
    for _ in range(n_rounds):
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            lo, hi = -h[j] + 1e-6, 1.0 - h[j] - 1e-6
            if hi - lo < 1e-4:
                continue
            obj_1d = lambda d: compute_c5(fix_sum(apply_delta(h, n, j, d), n))
            try:
                res = optimize.minimize_scalar(obj_1d, bounds=(lo, hi),
                    method='brent', options={'xatol': 1e-6})
                if res.success and res.x != 0:
                    h = fix_sum(apply_delta(h, n, j, res.x), n)
            except Exception:
                pass
    return fix_sum(h, n)

def apply_delta(h, n, j, d):
    hn = h.copy()
    hn[j] = min(max(hn[j] + d, 0.0), 1.0)
    return hn

def pair_brent(h, n, rng, n_pairs=300, n_rounds=1):

        for _ in range(n_pairs):
            j, l = rng.choice(n, 2, replace=False)
