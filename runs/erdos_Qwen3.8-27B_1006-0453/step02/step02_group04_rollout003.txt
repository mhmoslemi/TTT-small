def compute_c5(h):

    dx = 2.0 / n
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.real(np.fft.ifft(np.conj(H) * G))
    return float(np.max(corr) * dx)

def all_shifts(h):
    n = len(h)
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.real(np.fft.ifft(np.conj(H) * G))
    return corr * (2.0 / n)

def project_feasible(h, n):
    h = h.astype(float)
    s = np.sum(h)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.real(np.fft.ifft(np.conj(H) * G))
    return float(np.max(corr) * dx)

def all_shifts(h):
    n = len(h)
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.real(np.fft.ifft(np.conj(H) * G))
    return corr * (2.0 / n)

def project_feasible(h, n):
    h = h.astype(float)
    target = n * 0.5
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, n, top_m):
    shifts = all_shifts(h)
    top_idx = np.argsort(shifts)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k) % n
        grad += (1.0 - np.roll(h, -k)) - np.roll(h, k)
    grad *= (2.0 / n) / top_m
    return grad, shifts, top_idx

def pairwise_refine(h, n, max_rounds=30):
    h = h.copy()
    c5 = compute_c5(h)

            j = np.random.randint(n)
            if i == j:
                continue
            delta = (h[j] - h[i]) * 0.5 * np.random.uniform(0.3, 1.0)
            if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                continue
            h_trial = h.copy()
            h_trial[i] += delta
            h_trial[j] -= delta
            c5_t = compute_c5(h_trial)

                h = h_trial

def do_slsqp_polish(h, n, T, max_iter=5000):
    dx = 2.0 / n
    def obj(x):
        H = np.fft.fft(x)
        G = np.fft.fft(1.0 - x)
        corr = np.real(np.fft.ifft(np.conj(H) * G)) * dx
        c_max = np.max(corr)
        shifted = np.clip(corr - c_max, -60.0 * T, 0.0)
        s = np.sum(np.exp(shifted / T))
        return (c_max + T * np.log(max(s, 1e-300)))
    def con_fun(x):
        return np.sum(x) - n * 0.5
    try:
        res = opt.minimize(obj, h.copy(), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': con_fun}],
                           options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
