import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return np.sum(x[:_sp['n']]) - _sp['ts']

def _softobj(x):
    beta = _sp['beta']
    dx = _sp['dx']
    cr = np.correlate(x, 1.0 - x, mode='full')
    m = cr.max()
    return (m + np.log(np.sum(np.exp(beta * (cr - m)))) / beta) * dx

def _softcon(x):
    return np.sum(x) - _sp['ts']

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_sm(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    m = cr.max()
    w = np.exp(beta * (cr - m))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        kl1, kh1 = max(-(n-1), j-(n-1)), min(n-1, j)
        if kl1 <= kh1:
            k = np.arange(kl1, kh1+1)
            g[j] += np.dot(w[o+k], 1.0 - h[j-k])
        kl2, kh2 = max(-(n-1), -j), min(n-1, n-1-j)
        if kl2 <= kh2:
            k = np.arange(kl2, kh2+1)
            g[j] -= np.dot(w[o+k], h[j+k])
    return g * dx

def slsqp_aux(h0, dx, ts, mit=200):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-12})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= compute_c5(h0, dx) + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def slsqp_soft(h0, dx, ts, beta, mit=200):
    n = len(h0)
    _sp['n'], _sp['ts'], _sp['beta'], _sp['dx'] = n, ts, beta, dx
    cons = [{'type': 'eq', 'fun': _softcon}]
    bnds = [(0, 1)] * n
    try:
        r = minimize(_softobj, h0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-12})
        if r.x is not None:
            hr = proj(r.x, ts)
            if compute_c5(hr, dx) <= compute_c5(h0, dx) + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

    st = 0
    for i in range(it):
        if dl is not None and time.time() > dl:

        b = 20.0 + 80.0 * max(0.0, 1.0 - i / max(1, 0.3*it))

                break
