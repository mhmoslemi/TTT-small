        starts.append(project_feasible(hq, n))
        # Sawtooth
        hs = np.linspace(0.3, 0.7, n)
        starts.append(project_feasible(hs, n))

        for h0 in starts:
            if time.time() - t0 > budget_s * 0.70:
                break
            for T in [80, 400, 1200]:
                if time.time() - t0 > budget_s * 0.70:
                    break
                hc = lbfgs_optimize(h0, T, n, 1e5, maxiter=350)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc

    # Phase 6: Multi-resolution exploration
    for n_try in [80, 100, 160, 200]:
        if time.time() - t0 > budget_s * 0.83:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [60, 300, 1200]:
            if time.time() - t0 > budget_s * 0.83:
                break
            hc = lbfgs_optimize(h_interp, T, n_try, 1e5, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        # SLSQP at new n
        if time.time() - t0 < budget_s * 0.83:
            hc = slsqp_optimize(project_feasible(best_h if len(best_h) != n_try else best_h, n_try), 800, n_try, maxiter=800)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 7: Coordinate descent polish
    n_cur = len(best_h)
    if time.time() - t0 < budget_s * 0.93:
        no_improve = 0
        for iteration in range(600):
            if time.time() - t0 > budget_s * 0.93:
                break
            improved = False
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            k_max = int(np.argmax(corr))
            # Bias towards indices that contribute to worst correlation
            weights = np.ones(n_cur)
            weights[k_max % n_cur: k_max % n_cur + n_cur // 2] *= 2.0
            probs = weights / weights.sum()
            indices = np.random.choice(n_cur, size=min(20, n_cur), replace=False, p=probs)
            for j in indices:
                for delta in [0.015, 0.03, 0.05, -0.015, -0.03, -0.05]:
                    h_try = best_h.copy()
                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, n_cur)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 60:
                break

    # Phase 8: Final SLSQP high-T polish
    for T in [5000, 15000]:
        if time.time() - t0 > budget_s * 0.985:
            break
        hc = slsqp_optimize(best_h, T, n_cur, maxiter=1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
