    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_c5_obj(h, N, tau, penalty):
    """Returns (smooth_c5 * dx + penalty*(sum-target)^2, gradient)."""
    dx = 2.0 / N
    target = N / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr / tau
    scaled -= scaled.max()
    exps = np.exp(scaled)
    esum = exps.sum()
    w = exps / esum
    
    val = (tau * np.log(esum) + corr.max()) * dx + penalty * (h.sum() - target)**2
    
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 1e-8)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N-L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:N-L]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    grad *= dx
    grad += 2.0 * penalty * (h.sum() - target)
    
    return val, grad

    def fun(h_flat):
        return smooth_c5_obj(h_flat, N, tau, penalty)
    res = minimize(fun, h.copy(), jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)]*N,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': maxiter*3})
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_sum(h_opt, N/2.0)
    return h_opt, compute_c5(h_opt, 2.0/N)

        grad = np.zeros(N)

            else:

                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
        m = b1*m + (1-b1)*grad

        it += 1
