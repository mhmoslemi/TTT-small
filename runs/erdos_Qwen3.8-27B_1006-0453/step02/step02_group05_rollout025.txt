import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soffmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    wsum = w.sum()
    if wsum < 1e-30:
        return 0.0, true_c5, np.zeros(N)
    w /= wsum
    obj = float(np.log(wsum) / tau)
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-8
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return obj, true_c5, grad

def nesterov_opt(h, N, target, deadline, tau0, tau1, max_iter):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    x = h.copy()
    xprev = h.copy()
    beta = 0.85
    alpha = 0.1
    it = 0

        frac = min(it / max(max_iter, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        y = x + beta * (x - xprev)
        y = project_sum(y, target)
        obj_y, c5_y, grad_y = soffmax_grad(y, N, tau)
        if c5_y < best_c5:
            best_c5 = c5_y
            best_h = y.copy()
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-10:
            break
        g_dir = grad_y / gnorm
        found = False
        a_trial = alpha
        for _ in range(25):
            x_new = y - a_trial * g_dir
            x_new = project_sum(x_new, target)
            obj_new, _, _ = soffmax_grad(x_new, N, tau)
            if obj_new < obj_y:
                found = True
                break
            a_trial *= 0.5
        if found:
            alpha = min(a_trial * 1.5, 0.5)
        else:
            alpha *= 0.5
            x_new = x
        xprev = x
        x = x_new
        it += 1
    # Final evaluation

    c5_x = compute_c5(x, 2.0 / N)
    if c5_x < best_c5:
        best_c5 = c5_x
        best_h = x
    return best_h, best_c5

    max_L2 = N - 1 - i
    if max_L2 >= 0:
        Ls = np.arange(max_L2 + 1)
        deriv[center + Ls] -= h[i + Ls]
    
    # L < 0: M = -L, M ranges from 1 to N-1, k = center - M ranges from center-1 down to 0
    # For M=1..N-1:
    #   If i <= N-1-M, i.e., M <= N-1-i: deriv[center-M] += (1-h[i+M])
    #   If M <= i, i.e., M in [1, i]: deriv[center-M] -= h[i-M]
    
    # Condition M <= N-1-i: M in [1, min(N-1, N-1-i)]
    max_M1 = min(N - 1, N - 1 - i)
    if max_M1 >= 1:
        Ms = np.arange(1, max_M1 + 1)
        deriv[center - Ms] += (1.0 - h[i + Ms])
    
    # Condition M <= i: M in [1, min(N-1, i)]
    max_M2 = min(N - 1, i)
    if max_M2 >= 1:
        Ms = np.arange(1, max_M2 + 1)
        deriv[center - Ms] -= h[i - Ms]
    
    return deriv

import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soffmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    wsum = w.sum()
    if wsum < 1e-30:
        return 0.0, true_c5, np.zeros(N)
    w /= wsum
    obj = float(np.log(wsum) / tau)
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-8
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return obj, true_c5, grad

def nesterov_opt(h, N, target, deadline, tau0, tau1, max_iter):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    x = h.copy()
    xprev = h.copy()
    beta = 0.82
    alpha = 0.08
    it = 0
    while time.time() < deadline and it < max_iter:
        frac = min(it / max(max_iter, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        y = x + beta * (x - xprev)
        y = project_sum(y, target)
        obj_y, c5_y, grad_y = soffmax_grad(y, N, tau)
        if c5_y < best_c5:
            best_c5 = c5_y
            best_h = y.copy()
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-12:
            break
        g_dir = grad_y / gnorm
        found = False
        a_trial = alpha
        for _ in range(20):
            x_new = y - a_trial * g_dir
            x_new = project_sum(x_new, target)
            obj_new, _, _ = soffmax_grad(x_new, N, tau)
            if obj_new < obj_y:
                found = True
                break
            a_trial *= 0.5
        if found:
            alpha = min(a_trial * 1.8, 0.3)
            x = x_new
        else:
            alpha *= 0.5
        xprev = x
        it += 1
    x = project_sum(x, target)
    c5_x = compute_c5(x, 2.0 / N)
    if c5_x < best_c5:
        best_c5 = c5_x
        best_h = x
    return best_h, best_c5

def deriv_for_coord(h, N, i):
    M2 = 2 * N - 1
    deriv = np.zeros(M2)
    c = N - 1
    max_L1 = min(i, N - 1)
    if max_L1 >= 0:
        Ls = np.arange(max_L1 + 1)
        deriv[c + Ls] += (1.0 - h[i - Ls])
    max_L2 = N - 1 - i
    if max_L2 >= 0:
        Ls = np.arange(max_L2 + 1)
        deriv[c + Ls] -= h[i + Ls]
    max_M1 = min(N - 1, N - 1 - i)
    if max_M1 >= 1:
        Ms = np.arange(1, max_M1 + 1)
        deriv[c - Ms] += (1.0 - h[i + Ms])
    max_M2 = min(N - 1, i)
    if max_M2 >= 1:
        Ms = np.arange(1, max_M2 + 1)
        deriv[c - Ms] -= h[i - Ms]
    return deriv

def f1d_eval(corr, deriv, t):
    return float(np.max(corr + t * deriv))

def coord_descent(h, N, max_sweeps, time_limit):
    dx = 2.0 / N
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, target)
    best_c5 = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in range(N):
            deriv = deriv_for_coord(h, N, i)
            old = h[i]
            tl, th = -old, 1.0 - old
            if abs(th - tl) < 1e-14:
                continue
            res = minimize_scalar(f1d_eval, args=(corr, deriv), bounds=(tl, th),
                                  method='bounded', options={'xatol': 1e-8})
            new_t = res.x
            c_new = corr + new_t * deriv
            c5_new = float(np.max(c_new)) * dx
            if c5_new <= best_c5 + 1e-12:
                corr = c_new
                h[i] = old + new_t
                if c5_new < best_c5:
                    best_c5 = c5_new
        h = project_sum(h, target)
        c5_check = compute_c5(h, dx)
        if c5_check < best_c5:
            best_c5 = c5_check
    return h, best_c5

    t0 = time.time()

    sizes = sorted(set([n0] + [s for s in [200, 300, 400, 500] if s % 2 == 0]))

    for N in sizes:
        if time.time() > deadline - 40:

        dx = 2.0 / N
        target = N / 2.0
        if N == n0:

        else:

        h_init = project_sum(h_init, target)
        remain = deadline - time.time()
        if remain < 30:

        p1_dead = time.time() + remain * 0.35

            best_c5 = c1
            best_h = h1.copy()
        p2_dead = time.time() + (deadline - time.time()) * 0.45
        if time.time() < deadline - 30:
            h2, c2 = coord_descent(h1.copy(), N, max_sweeps=60, time_limit=p2_dead)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
        if time.time() < deadline - 25:
            h_p = best_h[:N].copy() if N == len(best_h) else np.interp(
                np.linspace(0, 2, N, endpoint=False), x0, h0)
            h_p = h_p + rng.randn(N) * 0.05
            h_p = project_sum(h_p, target)
            h3, c3 = coord_descent(h_p, N, max_sweeps=30, time_limit=deadline - 10)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()
    Nb = len(best_h)

    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
