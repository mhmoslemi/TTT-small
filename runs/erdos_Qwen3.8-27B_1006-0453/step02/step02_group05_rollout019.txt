import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(h, N, tau, top_k=50):
    """Compute gradient using top-k shifts for efficiency."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    
    # Soft-max over all lags, but only use top-k for gradient
    sc = corr / tau
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-6
    active = np.where(w > thresh)[0]
    
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    
    return grad, true_c5

def adam_opt(h, N, target, dx, deadline, tau_start, tau_end, lr0, phases=None):
    """Adam with cosine tau and lr schedule."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    max_iter = 3000
    
    while time.time() < deadline and it < max_iter:
        frac = min(it / max_iter, 1.0)
        # Cosine annealing for tau
        tau = tau_end + (tau_start - tau_end) * 0.5 * (1 + np.cos(np.pi * frac))
        # Cosine annealing for lr
        lr = lr0 * 0.5 * (1 + np.cos(np.pi * frac))
        lr = max(lr, lr0 * 0.01)
        
        grad, c5_val = smax_grad(h, N, tau)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

        v = b2 * v + (1 - b2) * grad * grad
        it += 1

        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = corr / tau
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-5
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def adam_opt(h, N, target, dx, deadline, tau_hi, tau_lo, lr0, max_iter=4000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    total_time = max(1.0, deadline - time.time())
    iter_time = total_time / max_iter
    while time.time() < deadline and it < max_iter:
        frac = float(it) / max_iter
        tau = tau_lo + (tau_hi - tau_lo) * 0.5 * (1.0 + np.cos(np.pi * frac))
        tau = max(tau, 0.5)
        lr = lr0 * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, lr0 * 0.02)
        grad, c5_val = smax_grad(h, N, tau)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def local_polish(h, N, target, dx, deadline):

    h = h.copy()

                    old = trial
                    break
