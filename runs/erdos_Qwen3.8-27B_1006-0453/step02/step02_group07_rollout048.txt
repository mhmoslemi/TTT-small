import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-13:
        return h
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    k = np.sum(interior)
    if k > 0:
        h[interior] += d / k
        h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-10:
        idx = np.where((h > 1e-5) & (h < 1.0 - 1e-5))[0]
        if len(idx) > 0:
            h[idx[0]] = min(max(h[idx[0]] + d2, 0.0), 1.0)
    return h

def sum_eq(h, n):
    return np.sum(h) - 0.5 * n

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    if T < 1e-14:
        return cmax * dx
    s = np.clip((corr - cmax) / T, -400, 0)
    return (cmax + T * np.log(np.exp(s).sum())) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    nc = len(corr)
    off = n - 1
    if T < 1e-14:
        w = (corr >= cmax - 1e-10).astype(float)
        w /= max(w.sum(), 1.0)
    else:
        s = np.clip((corr - cmax) / T, -400, 0)
        e = np.exp(s)
        w = e / e.sum()
    grad = np.zeros(n)
    for i in range(n):
        lo1 = max(0, off - i)
        hi1 = min(nc - 1, off + n - 1 - i)
        if lo1 <= hi1:
            hidx = i + np.arange(lo1, hi1 + 1) - off
            grad[i] += np.dot(w[lo1:hi1+1], 1.0 - h[hidx])
        lo2 = max(0, off + i - n + 1)
        hi2 = min(nc - 1, off + i)
        if lo2 <= hi2:
            hidx = i - (np.arange(lo2, hi2 + 1) - off)
            grad[i] -= np.dot(w[lo2:hi2+1], h[hidx])
    grad *= dx
    return grad

def slsqp_opt(h, n, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,), 'jac': lambda x, n: np.ones(n)}]
    try:
        res = optimize.minimize(
            smooth_obj, h, args=(T,), method='SLSQP',
            jac=lambda x, T: smooth_grad(x, T),
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-18})
        hn = fix_sum(res.x, n)
        if compute_c5(hn) < compute_c5(h):
            return hn
    except Exception:
        pass
    return h

def lfbgbs_opt(h, n, T, maxiter):
    """L-BFGS-B with projection onto box, then fix sum."""
    def obj(x):
        xc = fix_sum(x, n)
        return smooth_obj(xc, T)
    def grd(x):
        xc = fix_sum(x, n)
        g = smooth_grad(xc, T)
        return g
    best_h = h.copy()
    best_c = compute_c5(h)
    try:
        res = optimize.minimize(obj, h, jac=grd, method='L-BFGS-B',
                                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        hn = fix_sum(res.x, n)
        if compute_c5(hn) < best_c:
            best_h = hn
            best_c = compute_c5(hn)
    except Exception:
        pass
    return best_h

def coord_descend(h, n, n_pass=5, n_evals_per_coord=8):
    """Coordinate descent with geometric step sizes."""
    for p in range(n_pass):
        improved = False
        for i in range(n):
            if h[i] <= 1e-5 or h[i] >= 1.0 - 1e-5:
                continue
            c0 = compute_c5(h)
            best_c = c0
            best_v = h[i]
            for step in np.concatenate([np.linspace(0.01, 0.25, 6), -np.linspace(0.01, 0.25, 6)]):
                hv = h[i] + step
                if hv < 1e-5 or hv > 1.0 - 1e-5:
                    continue
                hn = h.copy()
                hn[i] = hv
                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = hv
            if best_v != h[i]:
                h[i] = best_v
                improved = True
        h = fix_sum(h, n)
        if not improved:
            break
    return h

def pair_descend(h, n, rng, n_pairs=800):
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
            continue
        if h[l] <= 1e-5 or h[l] >= 1.0 - 1e-5:
            continue
        c0 = compute_c5(h)
        best_c = c0
        best_d = 0.0
        for d in np.concatenate([np.linspace(0.005, 0.1, 5), -np.linspace(0.005, 0.1, 5)]):
            hj, hl = h[j] + d, h[l] - d
            if hj < 1e-5 or hj > 1.0 - 1e-5 or hl < 1e-5 or hl > 1.0 - 1e-5:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj, hl
            c = compute_c5(hn)
            if c < best_c - 1e-14:
                best_c = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)
