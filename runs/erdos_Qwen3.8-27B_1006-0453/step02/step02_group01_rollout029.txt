def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute equally to non-saturated entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    S = exp_t.sum()
    w = exp_t / S
    obj = (m + np.log(S) / T) * dx
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        wi = w[s + n - 1]
        if wi < 1e-13:
            continue
        lo = max(0, s)
        hi = min(n - 1, n - 1 + s)
        ia = np.arange(lo, hi + 1)
        is_ = ia - s
        grad[ia] += wi * (1.0 - h[is_])
        grad[is_] += -wi * h[ia]
    grad *= dx
    return obj, grad

def lbfgsb_solve(h, T, maxiter, pen):
    n = len(h)
    target = n * 0.5
    def f(hv):
        o, _ = smooth_obj_grad(hv, T)
        return o + pen * (hv.sum() - target) ** 2
    def g(hv):
        _, gg = smooth_obj_grad(hv, T)
        return gg + 2.0 * pen * (hv.sum() - target)
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_top(h, top_k=5, temp=5.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    tc = corr[top_idx]
    w = np.exp(temp * (tc - tc.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        s = idx - (n - 1)
        lo = max(0, s)
        hi = min(n - 1, n - 1 + s)
        ia = np.arange(lo, hi + 1)
        is_ = ia - s
        grad[ia] += wi * (1.0 - h[is_])
        grad[is_] += -wi * h[ia]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B polish at increasing T
    for T in [200, 2000, 20000]:
        if elapsed() > budget_s * 0.06:
            break
        ht = lbfgsb_solve(best_h, T, 300, 500.0)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_c5, best_h = c5t, ht

    # Phase 2: Adam subgradient with periodic L-BFGS polish
    n_cur = best_n
    m_s = np.zeros(n_cur)
    v_s = np.zeros(n_cur)
    t_ad = [0]
    no_improve = 0
    max_adam_steps = 5000
    for i in range(max_adam_steps):
        if elapsed() > budget_s * 0.30:
            break
        if i > 0 and i % 200 == 0:
