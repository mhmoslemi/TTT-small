    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)

        if abs(d) < 1e-14:

    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full') * dx
    sm = cr.max()
    w = np.exp(beta * (cr - sm))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:

    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-15) if d > 0 else (h > 1e-15)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def sgrad(h, beta):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full') * dx
    sm = cr.max()
    w = np.exp(beta * (cr - sm))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    g = np.zeros(n)
    for sv in range(-(n - 1), n):
        k = sv + n - 1
        wk = w[k] * dx
        a, b = max(0, -sv), min(n, n - sv)
        if a < b:
            g[a:b] += wk * (1.0 - h[a + sv:b + sv])
        c, d = max(0, sv), min(n, n + sv)
        if c < d:
            g[c:d] -= wk * h[c - sv:d - sv]
    return g

def nm_opt(h0, ts, dl):
    def obj(x):
        return compute_c5(proj(x, ts))
    try:
        r = minimize(obj, h0, method='Nelder-Mead',
                     options={'maxiter': 300, 'xatol': 1e-10, 'fatol': 1e-12,
                              'adaptive': True, 'maxfev': 1500})
        h = proj(r.x, ts)
        return h, compute_c5(h)
    except Exception:
        return h0.copy(), compute_c5(h0)

def cdesc(h0, ts, nrnd, dl, rng):
    n = len(h0)
    h = h0.copy()
