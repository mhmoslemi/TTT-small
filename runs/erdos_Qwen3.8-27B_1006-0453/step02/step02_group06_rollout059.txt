        sigma = 0.008 + 0.01 * (i % 3)
        h_pert = np.clip(h_best + rng.randn(n_best) * sigma, 0, 1)
        h_pert = proj(h_pert, n_best)
        T_p = [0.003, 0.001, 0.0005, 0.002][i % 4]
        h_new = slsqp_smax(h_pert, T_p, 500)
        c = compute_c5(h_new)
        if c < c5_best:
            h_best, c5_best = h_new, c
        h_new2 = slsqp_topk(h_pert, 5, 400)
        c2 = compute_c5(h_new2)
        if c2 < c5_best:
            h_best, c5_best = h_new2, c2

    # Phase 6: Resolution changes
    for n_try in [280, 420, 504]:
        if time.time() - t0 > budget_s - 120:
            break
        if n_try == n_best:
            continue
        h_up = upsample(h_best, n_try)
        h_new = slsqp_smax(h_up, 0.003, 600)
        c = compute_c5(h_new)
        if c < c5_best:
            h_best, c5_best, n_best = h_new, c, n_try
        h_new2 = slsqp_topk(h_up, 5, 500)
        c2 = compute_c5(h_new2)
        if c2 < c5_best:
            h_best, c5_best, n_best = h_new2, c2, n_try

    # Phase 7: SA binary refinement
    if time.time() - t0 < budget_s - 80:
        h_new, c_new = sa_binary(h_best.copy(), len(h_best), t0, budget_s, seed + 55555, 400)
        if c_new < c5_best:
            h_best, c5_best = h_new, c_new

    # Phase 8: Final ultra-fine polish
    if time.time() - t0 < budget_s - 25:
        h_new = slsqp_smax(h_best.copy(), 0.0001, 2000)
        c = compute_c5(h_new)
        if c < c5_best:
            h_best, c5_best = h_new, c
        if time.time() - t0 < budget_s - 15:
            h_new = slsqp_topk(h_best.copy(), 3, 1000)
            c = compute_c5(h_new)
            if c < c5_best:
                h_best, c5_best = h_new, c

    # Phase 9: One more binary block pass
    if time.time() - t0 < budget_s - 10:
        h_new, c_new = binary_block_search(h_best.copy(), len(h_best), 2, t0, budget_s)
        if c_new < c5_best:
            h_best, c5_best = h_new, c_new

    h_best = proj(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    if c5_final > c5_best:
        c5_final = c5_best
    return (h_best, c5_final, len(h_best))
