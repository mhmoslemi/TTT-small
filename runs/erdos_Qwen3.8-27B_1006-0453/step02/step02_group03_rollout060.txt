        m = (h < 1 - 1e-12) if d > 0 else (h > 1e-12)
        c = m.sum()
        if c == 0:
            return h
        h[m] += d / c
    return h

def grad_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (cr - cr.max()))
    s = w.sum()
    w = w / s if s > 1e-30 else np.ones(len(cr)) / len(cr)
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        lo1 = max(-(n - 1), j - n + 1)
        hi1 = min(n - 1, j)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1 + 1)
            g[j] += np.dot(w[o + ks], 1.0 - h[j - ks])
        lo2 = max(-(n - 1), -j)
        hi2 = min(n - 1, n - 1 - j)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2 + 1)
            g[j] -= np.dot(w[o + ks], h[j + ks])
    return g * dx

def gd(h0, ts, iters, b0, b1, dl=None):
    n = len(h0)
    h = h0.copy()
    vel = np.zeros(n)
    bh, bc = h0.copy(), compute_c5(h0)
    stag = 0
    lr_s = 0.012
    for i in range(iters):
        if dl is not None and time.time() > dl:
            break
        frac = min(1.0, i / max(1, 0.7 * iters))
        beta = b0 + (b1 - b0) * frac
        lr = lr_s / (1.0 + 4e-4 * i)
        g = grad_h(h, beta)
        vel = 0.9 * vel - lr * g
        imp = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            h2 = proj(h + sh * vel, ts)
            c2 = compute_c5(h2)
            if c2 < bc - 1e-16:
                bh, bc, h = h2.copy(), c2, h2
                imp = True
                lr_s = min(lr_s * 1.08, 0.06)
                stag = 0
                break
        if not imp:
            stag += 1
            lr_s = max(lr_s * 0.65, 0.0005)
            vel *= 0.3
            if stag > 120:
                break
    return bh, bc

def _sobj(x):
    return x[_G['n']]

def _sineq(x):
    n = _G['n']
    return x[n] * (n / 2.0) - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return x[:_G['n']].sum() - _G['ts']

def slsqp(h0, ts, mit=400):
    n = len(h0)
    _G['n'] = n
    _G['ts'] = ts
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, 1.0)]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-15})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            cr = compute_c5(hr)
            if cr <= t0 + 0.01:
                return hr, cr
    except Exception:
        pass
    return h0.copy(), compute_c5(h0)

def coord_scan(h, ts, nrounds=4, nvals=25):
    n = len(h)
    bh, bc = h.copy(), compute_c5(h)
    for rd in range(nrounds):
        imp = False
        for j in range(n):
            old = bh[j]
            lo = max(0.0, old - 0.4)
            hi = min(1.0, old + 0.4)
            for t in np.linspace(lo, hi, nvals):
                if abs(t - old) < 1e-9:
                    continue
                ht = bh.copy()
                ht[j] = t
                ht = proj(ht, ts)
                ct = compute_c5(ht)
                if ct < bc - 1e-16:
                    bh, bc = ht.copy(), ct
                    imp = True
        if not imp:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl = t0 + budget_s - 5

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = np.full(50, 0.5)

    n0 = len(hr)
    ts0 = n0 / 2.0
    bh = proj(hr, ts0)
    bc = compute_c5(bh)

    # Main optimization phases: GD with increasing beta, sandwiched with SLSQP
    gd_specs = [
        (2000, 5, 60),
        (2500, 30, 150),
        (3000, 80, 400),
        (2500, 150, 600),
        (2000, 300, 800),
    ]
    for idx, (it, b0, b1) in enumerate(gd_specs):
        if time.time() > dl - 40:
            break
        h, c = gd(bh, ts0, it, b0, b1, dl - 15)
        if c < bc:
            bh, bc = h, c
        # SLSQP polish after each GD phase
        if time.time() > dl - 30:
            break
        h, c = slsqp(bh, ts0, 350)
        if c < bc:
            bh, bc = h, c

    # Coordinate scan for fine adjustment
    if time.time() < dl - 20:
        bh, bc = coord_scan(bh, ts0, 5, 30)

    # SLSQP after coordinate scan
    if time.time() < dl - 15:
        bh, bc2 = slsqp(bh, ts0, 300)
        if bc2 < bc:
            bc = bc2

    # Perturbation-based restarts from best
    for amp in (0.04, 0.02, 0.01, 0.005, 0.002):
        if time.time() > dl - 50:
            break
        hp = proj(bh + amp * rng.randn(n0), ts0)
        hp, cp = gd(hp, ts0, 1500, 10, 120, dl - 20)
        if cp < bc:
            bh, bc = hp, cp
        if time.time() > dl - 30:
            break
        hp, cp = slsqp(hp, ts0, 250)
        if cp < bc:
            bh, bc = hp, cp

    # Try neighboring resolutions
    for dn in (-2, -1, 1, 2, 3, -3, 4, -4, 5):
        nt = n0 + dn
        if nt < 20 or nt > 90 or time.time() > dl - 70:
            break
        tst = nt / 2.0
        hi = np.interp(np.linspace(0, 1, nt, endpoint=False),
                       np.linspace(0, 1, n0, endpoint=False), bh)
        hi = proj(hi, tst)
        hi, ci = gd(hi, tst, 1200, 10, 100, dl - 25)
        hi, ci = slsqp(hi, tst, 200)
        if ci < bc:
            bh, bc = hi, ci

    # Final polish cycle
    if time.time() < dl - 10:
        nf = len(bh)
        tsf = nf / 2.0
        bh, c = slsqp(bh, tsf, 250)
        if c < bc:
            bc = c
        bh, c = coord_scan(bh, tsf, 3, 20)
        if c < bc:
            bc = c
        if time.time() < dl - 5:
            h, c = gd(bh, tsf, 1000, 50, 300, dl - 3)
            if c < bc:
                bh, bc = h, c
        bh, c = slsqp(bh, tsf, 150)
        if c < bc:
            bc = c

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh)
    return bh, bc, len(bh)
