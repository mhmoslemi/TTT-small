import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = np.sum(interior)
    if ni > 0:
        h[interior] += diff / ni
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, n, T, max_iter):
    def obj(h):
        return soft_max_obj(h, T)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    res = opt.minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
    return proj_constraint(res.x, n)

def powell_opt(h0, n, T, max_iter):
    def obj(h):
        return soft_max_obj(h, T)
    res = opt.minimize(obj, h0, method='Powell',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
    return proj_constraint(res.x, n)

def de_search(h0, n, pop_size, n_gen, t_start, budget_end, seed):
    rng = np.random.RandomState(seed)
    target = n * 0.5
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        pop[i] = np.clip(h0 + rng.randn(n) * 0.01, 0, 1)
        pop[i] = proj_constraint(pop[i], n)
    fit = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fit)
    h_best = pop[best_idx].copy()
    c_best = fit[best_idx]
    F = 0.8
    CR = 0.5
    for gen in range(n_gen):
        if time.time() > budget_end:
            break
        for i in range(pop_size):
            a, b, c = rng.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            cross_mask = rng.rand(n) < CR
            cross_mask[rng.randint(n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = proj_constraint(trial, n)
            ft = compute_c5(trial)
            if ft <= fit[i]:
                pop[i] = trial
                fit[i] = ft
                if ft < c_best:
                    c_best = ft
                    h_best = trial.copy()
    return h_best, c_best

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)
