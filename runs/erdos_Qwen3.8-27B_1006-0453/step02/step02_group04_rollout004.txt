    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    d = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    ni = interior.sum()
    if ni > 0 and abs(d) < ni * 0.3:
        h[interior] += d / ni
        return np.clip(h, 0.0, 1.0)
    # Fallback: iterative adjustment
    idx = np.argsort(h) if d > 0 else np.argsort(-h)
    for i in idx:
        room = (1.0 - h[i]) if d > 0 else h[i]
        amt = min(abs(d), room)
        h[i] += np.sign(d) * amt
        d -= np.sign(d) * amt
        if abs(d) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        corr_t = corr[idx]
        c_max = corr_t.max()
        if T < 1e-14:
            mask = corr_t == c_max
            vals = corr_t[mask]
            obj = vals.max() * dx
            # Gradient: average of gradients at active points
            grad = np.zeros(n)
            for ki in idx[mask]:

                        v += 1.0 - h[j + s]
                    if 0 <= j - s < n:
                        v -= h[j - s]
                    grad[j] += v

        w_t = np.exp(shifted / T) / np.sum(np.exp(shifted / T))
        obj = (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

        w = np.exp(shifted / T) / np.sum(np.exp(shifted / T))
        obj = (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx
    
    # Gradient computation O(n^2)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    term3 = np.zeros(n)
    for j in range(n):
        klo = max(0, n - 1 - j)
        khi = min(2*n - 2, 2*n - 2 - j)
        if klo <= khi:
            term1[j] = w[klo:khi+1].sum()
            s_r = np.arange(klo, khi + 1) - (n - 1)
            term2[j] = (w[klo:khi+1] * h[j + s_r]).sum()

    return obj, grad

import scipy.optimize as opt
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    d = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    ni = interior.sum()
    if ni > 0 and abs(d) < ni * 0.3:
        h[interior] += d / ni
        return np.clip(h, 0.0, 1.0)
    idx = np.argsort(h) if d > 0 else np.argsort(-h)
    for i in idx:
        room = (1.0 - h[i]) if d > 0 else h[i]
        amt = min(abs(d), room)
        h[i] += np.sign(d) * amt
        d -= np.sign(d) * amt
        if abs(d) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def obj_grad_full(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    if T < 1e-14:
        return c_max * dx, np.zeros(n)
    shifted = np.clip(corr - c_max, -50.0 * T, 0.0)
    exp_v = np.exp(shifted / T)
    w = exp_v / exp_v.sum()
    obj = (c_max + T * np.log(exp_v.sum())) * dx
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    term3 = np.zeros(n)
    for j in range(n):
        klo = max(0, n - 1 - j)
        khi = min(2 * n - 2, 2 * n - 2 - j)
        if klo <= khi:
            term1[j] = w[klo:khi + 1].sum()
            sr = np.arange(klo, khi + 1) - (n - 1)
            term2[j] = (w[klo:khi + 1] * h[j + sr]).sum()
        klo2 = max(0, j - (n - 1))
        khi2 = min(2 * n - 2, j + (n - 1))
        if klo2 <= khi2:
            sr2 = np.arange(klo2, khi2 + 1) - (n - 1)
            term3[j] = (w[klo2:khi2 + 1] * h[j - sr2]).sum()
    grad = dx * (term1 - term2 - term3)
    return obj, grad

def lbfags_optimize(h, n, T, pen, max_iter=300):
    def fun(x):
        o, g = obj_grad_full(x, T)
        s = x.sum() - n * 0.5
        o += pen * s * s
        g = g + 2.0 * pen * s
        return o, g
    try:
        res = opt.minimize(fun, h, jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10,
                                    'maxls': 20, 'maxfun': max_iter * 3})
        return fix_sum(res.x, n)
    except Exception:
        return h

def slsqp_optimize(h, n, T, max_iter=2000):
    def fun(x):
        return obj_grad_full(x, T)[0] + 50.0 * (x.sum() - n * 0.5) ** 2
    def gfun(x):
        _, g = obj_grad_full(x, T)
        return g + 100.0 * (x.sum() - n * 0.5)
    try:
        res = opt.minimize(fun, h, jac=gfun, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-15})
        return fix_sum(res.x, n)
    except Exception:
        return h

    corr = np.correlate(h, 1.0 - h, mode='full')

    corr = np.correlate(h, 1.0 - h, mode='full')

                if 0 <= j - s < n:
                    v -= h[j - s]
                gm[j] += v

    shifted = np.clip(ct - cm, -50.0 * T, 0.0)
    ev = np.exp(shifted / T)
    wt = ev / ev.sum()
    obj = (cm + T * np.log(ev.sum())) * dx
    w = np.zeros(2 * n - 1)
    w[top_idx] = wt
    term1 = np.zeros(n)
    term2 = np.zeros(n)

        if klo <= khi:

            sr = np.arange(klo, khi + 1) - (n - 1)
            term2[j] = (w[klo:khi + 1] * h[j + sr]).sum()
        klo2 = max(0, j - (n - 1))

            sr2 = np.arange(klo2, khi2 + 1) - (n - 1)
            term3[j] = (w[klo2:khi2 + 1] * h[j - sr2]).sum()
