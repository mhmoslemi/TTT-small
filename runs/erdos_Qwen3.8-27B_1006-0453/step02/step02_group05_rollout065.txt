def project_sum(h, target):

def c5_grad_fft(h, N, tau, M):
    dx = 2.0 / N
    fh = np.fft.rfft(h, M)
    f1 = np.fft.rfft(1.0 - h, M)
    corr = np.real(np.fft.irfft(fh * np.conj(f1), M))[:2*N-1]

    B = conv_w_hf[N-1:2*N-1][::-1]

    best_h = h_init.copy()
    best_c5 = compute_c5(h, 2.0/N)

    v = np.zeros(N)

        tau = tau0 * (tau1 / tau0) ** frac

        if it % 100 == 0:

        if lr < 1e-10:
