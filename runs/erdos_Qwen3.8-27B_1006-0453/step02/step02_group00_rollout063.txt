    dx = 2.0 / n

def project_feasible(h, n):

    target = n / 2.0

        if total < 1e-13:

    c = [{'type': 'eq', 'fun': sum_con}]
    b = [(0.0, 1.0)] * n
    r = minimize(smooth_c5, h0, args=(T,), method='SLSQP',
                 bounds=b, constraints=c,
                 options={'maxiter': mi, 'ftol': ft})
    return project_feasible(np.clip(r.x, 0, 1), n)

def rescale(h, nf, nt):
    if nf == nt:
        return h.copy()
    xo = np.linspace(0, 2, nf, endpoint=False)
    xn = np.linspace(0, 2, nt, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), nt)

def generate_start(n, rng):
    kind = rng.randint(4)
    if kind == 0:
        a = rng.uniform(0.25, 0.65)
        b = rng.uniform(0.5, 1.5)
        h = np.zeros(n)
        h[:int(a * n)] = 1.0
        s2, e2 = int(b * n), min(n, int((b + 1.0 - a) * n))
        if s2 < e2:
            h[s2:e2] = 1.0
    elif kind == 1:
        per = int(rng.uniform(0.15, 0.35) * n)
        duty = int(per * rng.uniform(0.35, 0.65))
        h = np.zeros(n)
        for i in range(0, n, per):
            h[i:min(i + duty, n)] = 1.0
    elif kind == 2:
        x = np.linspace(0, 2, n)
        c = rng.uniform(0.3, 1.7)
        w = rng.uniform(0.15, 0.4)
        h = np.exp(-((x - c) ** 2) / (2 * w ** 2))
    else:
        h = (rng.rand(n) < 0.5).astype(float)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float)
    else:
        bh = np.zeros(60)
        bh[:30] = 1.0
        bh = project_feasible(bh, 60)
    bc = compute_c5(bh)
    nb = len(bh)

    # Phase 1: Polish initial at its resolution
    for T in [40, 150, 600]:
        if budget_s - (time.time() - t0) < 40:
            break
        try:
            h2 = do_slsqp(bh.copy(), T, nb, 400, 1e-13)
            c2 = compute_c5(h2)
            if c2 < bc:
                bc, bh = c2, h2
        except Exception:
            pass

    # Phase 2: Aggressive optimization at n=60
    nw = 60
    hw = rescale(bh, nb, nw)
    for T in [15, 60, 250, 1000, 4000]:
        if budget_s - (time.time() - t0) < 250:
            break
        try:
            h2 = do_slsqp(hw.copy(), T, nw, 700, 1e-14)
            c2 = compute_c5(h2)
            if c2 < bc:
                bc = c2
                bh = rescale(h2, nw, nb)
                bc = compute_c5(bh)
                hw = h2
        except Exception:
            pass

    # Phase 3: Basin hopping at n=60
    bhh, bhc = hw.copy(), compute_c5(hw)
    n_iter = 130
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 280:
            break
        fr = it / float(n_iter)
        stp = 0.13 * (1.0 - fr) + 0.02
        hp = bhh + stp * rng.randn(nw)
        if it % 3 == 0:
            k = (it // 3) % 4
            if k == 0:
                bs = int(rng.randint(0, max(1, nw - 12)))
                bw = int(rng.randint(5, 18))
                hp = bhh.copy()
                hp[bs:bs + bw] = 1.0 - hp[bs:bs + bw]
            elif k == 1:
                hp = generate_start(nw, rng)
            elif k == 2:
                hp = bhh.copy()
                bs = int(rng.randint(0, nw - 6))
                hw2 = int(rng.randint(4, 12))
                hp[bs:bs + hw2] += 0.4 * (rng.rand(hw2) - 0.5)
            else:
                x = np.linspace(0, 2, nw)
                c = rng.uniform(0.2, 1.8)
                w = rng.uniform(0.1, 0.35)
                hp = np.exp(-((x - c) ** 2) / (2 * w ** 2)) + 0.2
        hp = project_feasible(hp, nw)
        T_sh = 300 + 200 * (1.0 - fr)
        try:
            h2 = do_slsqp(hp, T_sh, nw, 280, 1e-12)
            c2 = compute_c5(h2)
            tp = 0.006 * (1.0 - fr) + 0.0002
            if c2 < bhc or (np.random.rand() < np.exp(-(c2 - bhc) / tp)):
                bhc, bhh = c2, h2
                if c2 < bc:
                    bc = c2
                    bh = rescale(h2, nw, nb)
                    bc = compute_c5(bh)
        except Exception:
            pass

    # Phase 4: Structured two-block sweep
    if budget_s - (time.time() - t0) > 180:
        for a in np.linspace(0.28, 0.62, 5):
            for b2 in np.linspace(0.45, 1.45, 6):
                if budget_s - (time.time() - t0) < 130:
                    break
                sp = np.zeros(nw)
                sp[:int(a * nw)] = 1.0
                s2e = int(b2 * nw)
                e2e = min(nw, int((b2 + 1.0 - a) * nw))
                if s2e < e2e:
                    sp[s2e:e2e] = 1.0
                sp = project_feasible(sp, nw)
                for T in [100, 600]:
                    if budget_s - (time.time() - t0) < 130:
                        break
                    try:
                        h2 = do_slsqp(sp, T, nw, 500, 1e-13)
                        c2 = compute_c5(h2)
                        if c2 < bc:
                            bc = c2
                            bh = rescale(h2, nw, nb)
                            bc = compute_c5(bh)
                            hw = h2
                    except Exception:
                        pass

    # Phase 5: Periodic pattern sweep
    if budget_s - (time.time() - t0) > 80:
        for per_f in [0.16, 0.2, 0.25, 0.33]:
            for duty_f in [0.35, 0.5, 0.6]:
                if budget_s - (time.time() - t0) < 50:
                    break
                per = int(per_f * nw)
                duty = int(duty_f * per)
                sp = np.zeros(nw)
                for i in range(0, nw, per):
                    sp[i:min(i + duty, nw)] = 1.0
                sp = project_feasible(sp, nw)
                try:
                    h2 = do_slsqp(sp, 400, nw, 400, 1e-13)
                    c2 = compute_c5(h2)
                    if c2 < bc:
                        bc = c2
                        bh = rescale(h2, nw, nb)
                        bc = compute_c5(bh)
                except Exception:
                    pass

    # Phase 6: Upscale and polish at higher n
    for npol in [80, 100, 120]:
        if budget_s - (time.time() - t0) < 50:
            break
        hp = rescale(bh, nb, npol)
        for T in [200, 900, 3000]:
            if budget_s - (time.time() - t0) < 35:
                break
            try:
                h2 = do_slsqp(hp, T, npol, 400, 1e-14)
                c2 = compute_c5(h2)
                if c2 < bc:
                    bc, bh, nb = c2, h2, npol
                    hp = h2
            except Exception:
                pass

    # Phase 7: Coordinate descent polish
    if budget_s - (time.time() - t0) > 20:
        nc = len(bh)
        for _ in range(200):
            if budget_s - (time.time() - t0) < 12:
                break
            j = int(rng.randint(nc))
            for d in [0.04, -0.04, 0.08, -0.08, 0.02, -0.02, 0.12, -0.12]:
                ht = bh.copy()
                ht[j] = np.clip(ht[j] + d, 0, 1)
                ht = project_feasible(ht, nc)
                ct = compute_c5(ht)
                if ct < bc - 1e-14:
                    bc, bh = ct, ht
                    break

    # Phase 8: Final high-T SLSQP
    if budget_s - (time.time() - t0) > 15:
        try:
            h2 = do_slsqp(bh.copy(), 8000, len(bh), 500, 1e-15)
            c2 = compute_c5(h2)
            if c2 < bc:
                bc, bh = c2, h2
        except Exception:
            pass

    bh = project_feasible(bh, len(bh))
    bc = compute_c5(bh)
    return bh, bc, len(bh)
