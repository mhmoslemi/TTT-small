def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-6) & (h < 1.0 - 1e-6)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        idx = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(idx) > 0:
            h[idx[0]] = min(max(h[idx[0]] + d2, 0.0), 1.0)
    return h

def lbfgs_opt(h, n, T, maxiter, ftol):
    c0 = compute_c5(h)
    penalty = 100.0
    
    def obj(h_in):
        h_in = np.clip(h_in, 0, 1)
        val, _ = lse_obj_and_grad(h_in, T)
        pen = penalty * (np.sum(h_in) - 0.5*n)**2
        return val + pen
    
    def grad(h_in):
        h_in = np.clip(h_in, 0, 1)
        _, g = lse_obj_and_grad(h_in, T)
        g += 2*penalty*(np.sum(h_in)-0.5*n)
        return g
    
    try:
        r = optimize.minimize(obj, h, jac=grad, method='L-BFGS-B',
            bounds=[(0.,1.)]*n,
            options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def swap_polish(h, n, rng, n_swaps=2000):
    """Try swapping values at two positions"""
    c5 = compute_c5(h)
    for _ in range(n_swaps):
        j, l = rng.choice(n, 2, replace=False)
        # Try partial swap
        for t in (0.3, 0.5, 0.7):
            dj = t*(h[j]-h[l])

            hn = fix_sum(hn, n)
            c = compute_c5(hn)
            if c < c5 - 1e-14:

    return h, c5

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    shifted = np.clip((corr - c) / T, -80, 80)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    val = T * (c / T + np.log(s)) * dx + pen * (np.sum(h) - 0.5 * n) ** 2
    w = exp_s / s
    grad = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = lags[k]
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if hi1 >= lo1:
            grad[lo1:hi1 + 1] += wk * (1.0 - h[lo1 - lag:hi1 - lag + 1])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if hi2 >= lo2:
            grad[lo2:hi2 + 1] -= wk * h[lo2 + lag:hi2 + lag + 1]
    grad *= dx
    grad += 2.0 * pen * (np.sum(h) - 0.5 * n)
    return val, grad

        return v

        _, g = lse_obj_grad(np.clip(x, 0, 1), T, pen)
        return g
