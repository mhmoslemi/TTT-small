    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-12:
        return h
    free = (h > 1e-5) & (h < 1.0 - 1e-5)
    k = np.sum(free)
    if k > 0:
        h[free] += d / k
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, n)

def coord_polish(h, n, n_pass=2):
    for _ in range(n_pass):
        for j in range(n):
            if h[j] <= 1e-3 or h[j] >= 1.0 - 1e-3:
                continue

            best_h = h.copy()

                c = compute_c5(hn)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = hn
            h = best_h
    return fix_sum(h, n)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(g, n):
    target = 0.5 * n
    lo, hi = g.min() - 2.0, g.max() + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(g - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    z = (corr - cmax) / T
    ez = np.exp(z)
    p = ez / np.sum(ez)
    grad = np.zeros(n)
    thresh = 1e-12
    for idx in range(len(corr)):
        if p[idx] < thresh:
            continue
        s = idx - (n - 1)
        lo1, hi1 = max(0, -s), min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += p[idx] * (1.0 - h[lo1 + s:hi1 + s])
        lo2, hi2 = max(0, s), min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= p[idx] * h[lo2 - s:hi2 - s]
    return grad * dx

def proj_grad_run(h, T, lr, n_steps, mu=0.82):
    n = len(h)
    v = np.zeros(n)
    obj0 = compute_c5(h)
    for _ in range(n_steps):
        g = smooth_grad(h, T)
        v = mu * v + g
        h = project_box_sum(h - lr * v, n)
    return h, obj0

                cv = compute_c5(hv)
                if cv < bc - 1e-13:
                    bc, bh = cv, hv
            h = bh
    return project_box_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return project_box_sum(np.interp(x1, x0, h, period=2.0), nn)

def mutate(h, n, rng, mode):
    ht = h.copy()
    if mode == 0:
        ht += rng.randn(n) * 0.012
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 20), replace=False)
        ht[idx] += rng.randn(len(idx)) * 0.08
    elif mode == 2:
        seg = max(1, n // 4)
        ht[seg:2 * seg] = ht[seg:2 * seg][::-1].copy()
    elif mode == 3:
        seg = max(1, n // 6)
        sh = rng.randint(1, max(2, n // 20))
        if seg + sh < n:
            tmp = ht[seg:seg + sh].copy()
            ht[seg:seg + sh] = ht[seg + sh:2 * seg].copy()
            ht[seg + sh:2 * seg] = tmp
    else:
        ht += 0.04 * np.sign(ht - 0.5) * (ht - 0.5)
    return project_box_sum(ht, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 7)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = project_box_sum(h, n)
    c5 = compute_c5(h)

    # Phase 1: Projected gradient with momentum, cooling schedule
    T_sched = [(0.08, 0.012, 180), (0.04, 0.010, 200), (0.02, 0.008, 220),
               (0.01, 0.006, 250), (0.005, 0.004, 280), (0.002, 0.003, 300),
               (0.001, 0.002, 350), (0.0005, 0.0015, 400)]
    for T, lr, ns in T_sched:
        if time.time() - t0 > budget_s - 350:
            break
        h, _ = proj_grad_run(h, T, lr, ns, mu=0.82)
        h = project_box_sum(h, n)
        cn = compute_c5(h)
        if cn > c5 + 0.002:
            break
        c5 = cn

    # Phase 2: SLSQP with analytic gradient
    for T in [0.005, 0.001, 0.0003]:
        if time.time() - t0 > budget_s - 220:
            break
        h = sbar(h, n, T)
        cn = compute_c5(h)
        if cn < c5:
            c5 = cn

    # Phase 3: Coordinate polish
    if time.time() - t0 < budget_s - 160:
        h = clipol(h, n, 2)
        c5 = compute_c5(h)

    # Phase 4: Diverse restarts with gradient refinement
    for i in range(35):
        if time.time() - t0 > budget_s - 120:
            break
        h
        _t = mutate(h, n, rng, i % 5)
        for T2 in [0.008, 0.003, 0.001]:
            _t, _ = proj_grad_run(_t, T2, 0.004, 120, mu=0.8)
            _t = project_box_sum(_t, n)
        ct = compute_c5(_t)
        if ct < c5:
            h, c5 = _t, ct
            if time.time() - t0 < budget_s - 80:
                h = sbar(h, n, 0.001)
                h = clipol(h, n, 1)
                c5 = compute_c5(h)

    # Phase 5: Multi-resolution search
    for n2 in [200, 320, 400]:
        if time.time() - t0 > budget_s - 60:
            break
        h2 = upsample(h, n2)
        c5_2 = compute_c5(h2)
        for T in [0.02, 0.005, 0.001]:
            if time.time() - t0 > budget_s - 20:
                break
            h2, _ = proj_grad_run(h2, T, 0.006, 200, mu=0.8)
            h2 = project_box_sum(h2, n2)
            c5_2 = compute_c5(h2)
        if time.time() - t0 < budget_s - 15:
            h2 = sbar(h2, n2, 0.001)
            c5_2 = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2

    # Phase 6: Final SLSQP squeeze
    if time.time() - t0 < budget_s - 10:
        h = sbar(h, n, 0.0002)
        h = project_box_sum(h, n)
        c5 = compute_c5(h)

    # Final coordinate polish if time permits
    if time.time() - t0 < budget_s - 5:
        h = clipol(h, n, 1)
        c5 = compute_c5(h)

    h = project_box_sum(h, n)
    c5 = compute_c5(h)
    return (h, c5, n)
