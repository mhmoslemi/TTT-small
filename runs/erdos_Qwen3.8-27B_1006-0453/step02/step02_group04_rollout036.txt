    dx = 2.0 / n

    omh = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-12:
            continue
        lag = top_idx[i] - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wi * omh[lag:]
            grad[lag:] -= wi * h[:n - lag]
        else:
            a = -lag
            grad[a:] += wi * omh[:n - a]
            grad[:n - a] -= wi * h[a:]
    return grad * dx

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    ni = np.sum(interior)
    if ni > 0 and abs(diff) < ni * 0.3:
        h[interior] += diff / ni
    else:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def pg_optimize(h0, n, T, max_iter=80000, top_k=0, time_limit=60.0, momentum_max=0.3):
    h = h0.copy()
    f_best = compute_c5(h)
    h_best = h.copy()
    lr = 0.008
    momentum = 0.0
    vel = np.zeros(n)
    prev_h = h.copy()
    prev_grad = None
    t0 = time.time()
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        f = soft_max_val(h, T)
        if f < f_best - 1e-14:
            f_best = f
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
        grad = grad_soft_max(h, T, top_k)
        if prev_grad is not None:
            sv = h - prev_h
            yv = grad - prev_grad
            sy = np.dot(sv, yv)
            ss = np.dot(sv, sv)
            if sy > 1e-14:
                bb_lr = ss / sy
                lr = 0.6 * lr + 0.4 * min(max(bb_lr, 1e-6), 0.03)
        vel = momentum * vel - lr * grad
        h_new = h + vel
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        f_new = soft_max_val(h_new, T)
        if f_new > f + 1e-7:
            lr *= 0.5
            vel = np.zeros(n)
            momentum = 0.0
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
        prev_h = h.copy()
        prev_grad = grad.copy()
        h = h_new
        if it > 200:
            momentum = min(momentum_max, momentum + 0.0005)
        if stall > 8000:
            break
    return h_best

def lbfgsb_optimize(h0, n, T, max_iter=500, time_limit=40.0):
    def obj(x):
        xc = np.clip(x, 0.0, 1.0)
        f = soft_max_val(xc, T)
        pen = 50.0 * (np.sum(xc) - n * 0.5) ** 2
        return f + pen
    t0 = time.time()
    try:
        res = opt.minimize(obj, h0.copy(), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-18,
                                    'maxfun': 20000, 'maxls': 50})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    if time.time() - t0 > time_limit:
        return h0
    return h0

def interp_h(h_src, n_src, n_dst):
    x_old = np.arange(n_src) * (2.0 / n_src)
    x_new = np.arange(n_dst) * (2.0 / n_dst)
    h_new = np.interp(x_new, x_old, h_src, left=h_src[0], right=h_src[-1])
    return fix_sum(h_new, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t_start

    def try_update(h_c, n_c):
        nonlocal h_best, n_best, c5_best
        h_c = fix_sum(h_c, n_c)
        c5_c = compute_c5(h_c)
        if c5_c < c5_best - 1e-14:
            h_best = h_c.copy()
            n_best = n_c
            c5_best = c5_c
            return True
        return False

    # Phase 1: L-BFGS-B with annealing (good global structure)
    for T in [0.04, 0.015, 0.006]:
        if elapsed() > budget_s - 200:
            break
        tl = min(50.0, budget_s - elapsed() - 50)
        if tl < 10:
            break
        h_new = lbfgsb_optimize(h_best, n_best, T, max_iter=600, time_limit=tl)
        try_update(h_new, n_best)

    # Phase 2: Projected gradient annealing (fast, many iterations)
    temps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for T in temps:
        if elapsed() > budget_s - 200:
            break
        tl = min(70.0, budget_s - elapsed() - 40)
        if tl < 10:
            break
        h_new = pg_optimize(h_best, n_best, T, max_iter=100000, top_k=0,
                            time_limit=tl, momentum_max=0.25)
        try_update(h_new, n_best)

    # Phase 3: Top-k focused (sharpen active shifts)
    for tk in [30, 15, 8]:
        if elapsed() > budget_s - 120:
            break
        for T in [0.003, 0.001, 0.0004]:
            if elapsed() > budget_s - 100:
                break
            tl = min(35.0, budget_s - elapsed() - 30)
            if tl < 8:
                break
            h_new = pg_optimize(h_best, n_best, T, max_iter=60000,
                                top_k=tk, time_limit=tl, momentum_max=0.2)
            try_update(h_new, n_best)

    # Phase 4: Multi-start perturbations
    scales = [0.015, 0.03, 0.05, 0.08, 0.12]
    for sc in scales:
        if elapsed() > budget_s - 150:
            break
        h_pert = h_best + np.random.randn(n_best) * sc
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = fix_sum(h_pert, n_best)
        tl = min(30.0, budget_s - elapsed() - 40)
        if tl < 8:
            break
        h_new = pg_optimize(h_pert, n_best, 0.008, max_iter=40000,
                            top_k=0, time_limit=tl * 0.6, momentum_max=0.3)
        h_new2 = pg_optimize(h_new, n_best, 0.0008, max_iter=40000,
                             top_k=10, time_limit=tl * 0.4, momentum_max=0.15)
        try_update(h_new2, n_best)

    # Phase 5: Resolution changes
    for n_try in [288, 500, 720]:
        if elapsed() > budget_s - 120:
            break
        h_init = interp_h(h_best, n_best, n_try)
        for T in [0.02, 0.006, 0.0015]:
            if elapsed() > budget_s - 80:
                break
            tl = min(35.0, budget_s - elapsed() - 30)
            if tl < 8:
                break
            h_new = pg_optimize(h_init, n_try, T, max_iter=60000,
                                top_k=0, time_limit=tl * 0.7, momentum_max=0.25)
            if try_update(h_new, n_try):
                h_init = interp_h(h_best, n_best, n_try)
            else:
                h_init = h_new
        # Try top-k at this resolution
        if elapsed() > budget_s - 60:
            break
        tl = min(20.0, budget_s - elapsed() - 20)
        if tl > 8:
            h_new = pg_optimize(h_init, n_try, 0.0005, max_iter=40000,
                                top_k=10, time_limit=tl, momentum_max=0.1)
            try_update(h_new, n_try)

    # Phase 6: Final polish at best resolution
    if elapsed() < budget_s - 15:
        tl = budget_s - elapsed() - 5
        h_new = pg_optimize(h_best, n_best, 0.00008, max_iter=200000,
                            top_k=5, time_limit=tl, momentum_max=0.1)
        try_update(h_new, n_best)
    if elapsed() < budget_s - 10:
        tl = budget_s - elapsed() - 4
        h_new = pg_optimize(h_best, n_best, 0.00003, max_iter=100000,
                            top_k=3, time_limit=tl, momentum_max=0.05)
        try_update(h_new, n_best)

    # Final: one L-BFGS-B pass at very low T
    if elapsed() < budget_s - 8:
        tl = budget_s - elapsed() - 3
        h_new = lbfgsb_optimize(h_best, n_best, 0.0001, max_iter=400, time_limit=tl)
        try_update(h_new, n_best)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
