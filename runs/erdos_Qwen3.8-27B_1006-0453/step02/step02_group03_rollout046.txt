import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    n = _sp['n']
    return float(np.sum(x[:n]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-15) if d > 0 else (h > 1e-15)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def soft_grad(h, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    crs = cr - cr.max()
    w = np.exp(beta * crs)
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        klo = max(-(n - 1), -j)
        khi = min(n - 1, n - 1 - j)
        if klo <= khi:
            ks = np.arange(klo, khi + 1)
            g[j] += np.dot(w[o + ks], 1.0 - h[j + ks])
        k2lo = max(-(n - 1), j - (n - 1))
        k2hi = min(n - 1, j)
        if k2lo <= k2hi:
            ks2 = np.arange(k2lo, k2hi + 1)
            g[j] -= np.dot(w[o + ks2], h[j - ks2])
    return g

def slsqp_opt(h0, dx, ts, maxit=200):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-14,
                     'iprint': 0})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 * dx + 0.02:
                return hr
    except Exception:
        pass
    return h0.copy()

def grad_fd(h0, dx, ts, niter, dl, beta_grid=None):
    n = len(h0)
    if beta_grid is None:
        beta_grid = np.array([30.0, 60.0, 120.0, 250.0, 500.0])
    best_h, best_c = h0.copy(), compute_c5(h0, dx)
    for bi in range(len(beta_grid)):
        beta = beta_grid[bi]
        h = best_h.copy()
        bc = best_c
        stall = 0
        n_this = max(100, niter // len(beta_grid))
        for it in range(n_this):
            if time.time() > dl:
                break
            g = soft_grad(h, beta) * dx
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g = g / gn
            lr = 0.05
            ok = False
            for _ in range(30):
                h2 = proj(h - lr * g, ts)
                c2 = compute_c5(h2, dx)
                if c2 < bc - 1e-15:
                    h, bc, best_h, best_c = h2, c2, h2.copy(), c2
                    ok = True
                    lr = min(0.2, lr * 1.3)
                    break
                lr *= 0.5
            if not ok:
                stall += 1
                if stall > 80:
                    break
    return best_h, best_c

def coord_desc(h0, dx, ts, nrounds, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for r in range(nrounds):
        if time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            if time.time() > dl:
                break
            sv = h[j]
            for d in (0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.08, -0.08):
                h[j] = sv + d
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc:
                    bc = c
                    imp = True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def block_desc(h0, dx, ts, ntrials, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for _ in range(ntrials):
        if time.time() > dl:
            break
        ln = rng.randint(1, max(2, n // 6))
        st = rng.randint(0, n - ln + 1)
        idx = np.arange(st, min(st + ln, n))
        sv = h[idx].copy()
        h[idx] = sv + rng.uniform(-0.08, 0.08, len(idx))
        h = proj(h, ts)
        c = compute_c5(h, dx)
        if c < bc:
            bc = c
        else:
            h[idx] = sv
            h = proj(h, ts)
    return h, bc

def make_starts(n, ts, rng, ref):
    out = []
    if ref is not None:
        if len(ref) == n:
            out.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.insert(0, hi)
    out.append(np.full(n, ts / n))
    for _ in range(2):
        h = np.zeros(n)
        p = rng.permutation(n)
        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[p[int(ts)]] = fr
        out.append(h.copy())
    x = np.arange(n) / float(n)
    for m in (1, 2, 3):
        out.append(0.5 + 0.3 * np.sin(2.0 * np.pi * m * x + rng.uniform(0, 6.28)))
    return [proj(s, ts) for s in out]

def opt_pipeline(h0, dx, ts, dl, rng, tag=""):
    h = slsqp_opt(h0, dx, ts, 150)
    c = compute_c5(h, dx)
    if time.time() < dl - 20:
        h2, c2 = grad_fd(h, dx, ts, 3000, dl - 10,
                         beta_grid=np.array([40.0, 100.0, 300.0]))
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 10:
        h3, c3 = coord_desc(h, dx, ts, 60, dl - 5, rng)
        if c3 < c:
            h, c = h3, c3
    if time.time() < dl - 6:
        h4, c4 = block_desc(h, dx, ts, 200, dl - 3, rng)
        if c4 < c:
            h, c = h4, c4
    if time.time() < dl - 3:
        h5 = slsqp_opt(h, dx, ts, 100)
        c5 = compute_c5(h5, dx)
        if c5 < c:
            h, c = h5, c5
    return h, c
