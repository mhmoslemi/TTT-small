    if T < 1e-15:

    """Compute w[m] = d(corr[m])/d(delta) for pair (j,l)."""
    m_idx = np.arange(2*n-1)
    # For index j: contribution to corr[m]
    # term1: 1 - h[j + m - (n-1)] if 0 <= j+m-(n-1) <= n-1
    idx1_j = j + m_idx - (n-1)
    mask1_j = (idx1_j >= 0) & (idx1_j < n)
    t1_j = np.where(mask1_j, 1.0 - h[idx1_j], 0.0)
    # term2: -h[j - m + (n-1)] if 0 <= j-m+(n-1) <= n-1
    idx2_j = j - m_idx + (n-1)
    mask2_j = (idx2_j >= 0) & (idx2_j < n)
    t2_j = np.where(mask2_j, -h[idx2_j], 0.0)
    # Same for l
    idx1_l = l + m_idx - (n-1)
    mask1_l = (idx1_l >= 0) & (idx1_l < n)
    t1_l = np.where(mask1_l, 1.0 - h[idx1_l], 0.0)
    idx2_l = l - m_idx + (n-1)
    mask2_l = (idx2_l >= 0) & (idx2_l < n)
    t2_l = np.where(mask2_l, -h[idx2_l], 0.0)
    
    return (t1_j + t2_j) - (t1_l + t2_l)
