import numpy as np
import scipy.optimize as opt
import time

_PARAMS = {'T': 0.01, 'mu': 1e4, 'n': 336}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def sm_value(h, T, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    T_s = max(T, 1e-10)
    obj = (c_max + T_s * np.log(np.sum(np.exp((corr - c_max) / T_s)))) * dx
    obj += mu * (np.sum(h) - n * 0.5) ** 2
    return obj

def sm_grad(h, T, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    T_s = max(T, 1e-10)
    w = np.exp((corr - c_max) / T_s) * dx / T_s
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w_m = w[m]
        if w_m < 1e-15:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            js = np.arange(lag, n)
            g[js] += w_m * (1.0 - h[js - lag])
            js = np.arange(n - lag)
            g[js] -= w_m * h[js + lag]
        else:
            ln = -lag
            js = np.arange(n - ln)
            g[js] += w_m * (1.0 - h[js + ln])
            js = np.arange(ln, n)
            g[js] -= w_m * h[js - ln]
    g += 2.0 * mu * (np.sum(h) - n * 0.5)
    return g

def lbgfsb_grad(h):
    return sm_grad(h, _PARAMS['T'], _PARAMS['mu'], _PARAMS['n'])

def do_lbgfsb(h0, n, T, mu, max_iter):

    _PARAMS['mu'] = mu

    return np.ones(len(h))

def slsqp_grad(h):

    obj = (np.sum(corr ** p) / len(corr)) ** (1.0 / p) * dx

    return obj
