import numpy as np
import time
from scipy.optimize import minimize as sp_min

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, N, tau, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    tmax = corr.max()
    if tau < 1e-6:
        tmax = 1e-6
    sh = corr * tau - tmax * tau
    w = np.exp(sh)
    lse = np.log(np.sum(w)) + tmax * tau
    obj = lse / tau + penalty * (h.sum() - 0.5 * N) ** 2
    w = w / w.sum()
    grad = np.zeros(N)
    ctr = N - 1
    active = np.where(w > 1e-5)[0]
    for ai in active:
        L = int(ai - ctr)
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    grad = grad / tau + 2.0 * penalty * (h.sum() - 0.5 * N)
    c5v = float(corr.max()) * (2.0 / N)
    return obj, grad, c5v

def lbfgs_run(h_init, N, tau, penalty, deadline, maxiter=200):
    target = 0.5 * N
    h = project_sum(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / N)

    def f_and_g(x):
        if time.time() > deadline:
            return 999.0, np.zeros(N)
        o, g, _ = lse_obj_grad(np.clip(x, 0, 1), N, tau, penalty)
        return o, g

    res = sp_min(f_and_g, h, jac=True, method='L-BFGS-B',
                 bounds=[(0, 1)] * N,
                 options={'maxiter': maxiter, 'gtol': 1e-6, 'ftol': 1e-10,
                          'maxfun': maxiter * 3})
    h_new = project_sum(np.clip(res.x, 0, 1), target)
    c5_new = compute_c5(h_new, 2.0 / N)
    if c5_new < best_c5:
        best_h = h_new
        best_c5 = c5_new
    return best_h, best_c5

def subgrad_polish(h_init, N, deadline, iters=5000, lr0=0.05, decay=0.999):
    target = 0.5 * N
    dx = 2.0 / N
    h = project_sum(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    lr = lr0
    for it in range(iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(corr.max()) * dx
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        kmax = int(np.argmax(corr))
        L = kmax - (N - 1)
        grad = np.zeros(N)
        if L == 0:
            grad[:] = (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            grad[:N - M] += (1.0 - h[M:])
            grad[M:] -= h[:N - M]
        h = h - lr * grad
        h = project_sum(h, target)
        lr *= decay
        if lr < 1e-9:
            break
    return best_h, best_c5

def structured_init(N, kind, rng):
    h = np.zeros(N)

                    h_up = project_sum(h_up, 0.5 * N0)
                    c_up = compute_c5(h_up, 2.0 / N0)
                    if c_up < best_c5:
                        best_c5 = c_up
                        best_h = h_up.copy()
                    else:
                        # Keep small grid solution if better
                        pass

    # Phase F: Final fine polish
    if time.time() < deadline - 5:
        tau_f = 2000.0
        pen_f = 200.0
        h_f, c_f = lbfgs_run(best_h, N0, tau_f, pen_f, deadline - 2, maxiter=100)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()
        h_f2, c_f2 = subgrad_polish(best_h, N0, deadline - 1, iters=3000, lr0=0.01, decay=0.9999)
        if c_f2 < best_c5:
            best_c5 = c_f2
            best_h = h_f2.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
