            if vb == i:

            v = pop[vb] + F * (pop[r1] - pop[r2])
            v = v + rng.randn(n) * 0.003
            u = rng.randint(n)
            mask = rng.rand(n) < CR
            mask[u] = True
            trial = np.where(mask, v, pop[i])
            trial = proj_constraint(trial, n)
            ft = compute_c5(trial)
            if ft < fitness[i]:
                pop[i] = trial
                fitness[i] = ft
        bi = int(np.argmin(fitness))
        if fitness[bi] < c_cur_best:
            c_cur_best = float(fitness[bi])
            h_cur_best = pop[bi].copy()
    return h_cur_best, c_cur_best

def subgrad_refine(h, n, t_start, deadline, top_m=5, n_iter=500, seed=0):
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    step = 0.002
    for it in range(n_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            s = int(k) - (n - 1)
            s = s % n
            grad += (1.0 - np.roll(h_cur, -s)) - np.roll(h_cur, s)
        grad /= top_m
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_try = h_cur - step * grad / gnorm * 0.5
        h_try = proj_constraint(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur:
            h_cur = h_try
            c_cur = c_try
            step = min(step * 1.05, 0.008)
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        else:
            step *= 0.75
    return h_best, c_best

def coord_descent(h, n, t_start, deadline, n_rounds=50):
    rng = np.random.RandomState(12345)
    h_cur = h.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    dx = 2.0 / n
    order = rng.permutation(n)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            old_val = h_cur[i]
            for trial_v in [old_val + 0.01, old_val - 0.01, old_val + 0.003, old_val - 0.003]:
                if trial_v < 0 or trial_v > 1:
                    continue
                h_try = h_cur.copy()
                h_try[i] = trial_v
                h_try = proj_constraint(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < c_cur - 1e-15:
                    h_cur = h_try
                    c_cur = c_try
                    improved = True
                    break
            if not improved and rnd > 10:
                pass
        if c_cur < c_best:
            c_best = c_cur
            h_best = h_cur.copy()
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP at current resolution with multiple T
    deadline = t_start + budget_s * 0.15
    for T in [0.01, 0.004, 0.001]:
        if time.time() > deadline:
            break
        h_opt = do_slsqp(h_best.copy(), n_best, T, 800)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 2: Differential Evolution - broad exploration
    deadline = t_start + budget_s * 0.40
    h_de, c_de = de_phase(h_best, n_best, 20, 120, t_start, deadline, seed)
    if c_de < c5_best:
        h_best = h_de
        c5_best = c_de

    # Phase 3: Subgradient refinement on top shifts
    deadline = t_start + budget_s * 0.55
    h_sg, c_sg = subgrad_refine(h_best, n_best, t_start, deadline, top_m=5, n_iter=800, seed=seed+1)
    if c_sg < c5_best:
        h_best = h_sg
        c5_best = c_sg

    # Phase 4: DE again - exploitation with smaller population, lower F
    deadline = t_start + budget_s * 0.70
    h_de2, c_de2 = de_phase(h_best, n_best, 16, 100, t_start, deadline, seed + 1000)
    if c_de2 < c5_best:
        h_best = h_de2
        c5_best = c_de2

    # Phase 5: SLSQP polish with decreasing T
    deadline = t_start + budget_s * 0.85
    for T in [0.003, 0.001, 0.0003, 0.0001]:
        if time.time() > deadline:
            break
        h_opt = do_slsqp(h_best.copy(), n_best, T, 1200)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 6: Coordinate descent micro-optimization
    deadline = t_start + budget_s * 0.93
    h_cd, c_cd = coord_descent(h_best, n_best, t_start, deadline, n_rounds=30)
    if c_cd < c5_best:
        h_best = h_cd
        c5_best = c_cd

    # Phase 7: Final ultra-fine SLSQP
    deadline = t_start + budget_s * 0.99
    for T in [0.00005, 0.00002]:
        if time.time() > deadline:
            break
        h_opt = do_slsqp(h_best.copy(), n_best, T, 2000)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Final projection and evaluation
    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final
    return (h_best, c5_best, len(h_best))
