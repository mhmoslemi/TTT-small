    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter=400):
    n = len(h)
    def obj(x):
        return x[-1]
    def eq_c(x):
        return x[:n].sum() - n / 2.0
    def ineq_c(x):
        return x[-1] - np.max(np.correlate(x[:n], 1.0 - x[:n], mode='full'))
    corr = np.correlate(h, 1.0 - h, mode='full')
    x0 = np.concatenate([h, [float(np.max(corr)) + 0.03]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': eq_c}, {'type': 'ineq', 'fun': ineq_c}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def sa_optimize(h, time_budget, rng, T0=0.004, T_end=1e-5, alpha=0.9992):
    n = len(h)
    current = h.copy()
    cur_c5 = compute_c5(current)
    best = current.copy()
    best_c5 = cur_c5
    T = T0
    iters = 0
    stuck = 0
    t_start = time.time()
    while time.time() - t_start < time_budget and T > T_end:
        move = rng.randint(5)
        trial = current.copy()
        if move == 0:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(0.005, 0.12) * rng.choice([-1.0, 1.0])
            if 0 <= trial[i] - d <= 1 and 0 <= trial[j] + d <= 1:
                trial[i] -= d
                trial[j] += d
        elif move == 1:
            i, j = rng.choice(n, 2, replace=False)
            trial[i], trial[j] = trial[j], trial[i]
        elif move == 2:
            if n >= 6:
                blen = rng.randint(3, min(12, n - 2))
                start = rng.randint(0, n - blen)
                sh = rng.choice([-2, -1, 1, 2])
                trial[start:start+blen] = np.roll(current[start:start+blen], sh)
        elif move == 3:
            freq = rng.uniform(1, 4) * n
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.008, 0.04)
            trial += amp * np.sin(np.linspace(0, freq, n) + phase)
        else:
            k = rng.randint(2, min(8, n))
            idx = rng.choice(n, k, replace=False)
            trial[idx] += rng.normal(0, 0.025, k)
        trial = project_feasible(trial, n)
        new_c5 = compute_c5(trial)
        delta = new_c5 - cur_c5
        accept = delta < 0 or (T > T_end and rng.rand() < np.exp(-delta / (T * 0.004)))
        if accept:
            current = trial
            cur_c5 = new_c5
            stuck = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = current.copy()
        else:
            stuck += 1
        if stuck > 30000:
            T = max(T * 1.5, 0.002)
            stuck = 0
        T *= alpha
        iters += 1
        if iters % 40000 == 0 and time.time() - t_start < time_budget - 25:
            pol = slsqp_smooth(current, 800, maxiter=120)
            pol_c5 = compute_c5(pol)
            if pol_c5 < best_c5:
                best_c5 = pol_c5
                best = pol.copy()
                current = pol.copy()
                cur_c5 = pol_c5
    return best, best_c5

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 90
        best_h = np.zeros(n0)
        for i in range(n0):
            best_h[i] = 1.0 if (i // 15) % 2 == 0 else 0.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP warmup increasing T
    for T in [20, 100, 500, 2000, 8000, 20000]:
        if rem() < 25:
            break
        ht = slsqp_smooth(best_h, T, maxiter=350)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 2: Direct SLSQP (two attempts)
    for _ in range(2):
        if rem() < 40:
            break
        ht = slsqp_direct(best_h, maxiter=400)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
            best_n = len(best_h)

    # Phase 3: SA at native resolution
    if rem() > 120:
        alloc = min(rem() * 0.30, 320)
        best_h, best_c5 = sa_optimize(best_h, alloc, rng, T0=0.004, T_end=1e-5)
        best_n = len(best_h)

    # Phase 4: SLSQP after SA
    if rem() > 30:
        for T in [2000, 8000, 30000]:
            ht = slsqp_smooth(best_h, T, maxiter=400)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 5: Multi-resolution exploration
    for n_try in [60, 72, 80, 96, 108, 120]:
        if rem() < 70:
            break
        if n_try == best_n:
            continue
        h_interp = interp_to(best_h, n_try)
        ht = slsqp_smooth(h_interp, 500, maxiter=200)
        ct = compute_c5(ht)
        alloc = min(rem() * 0.08, 50)
        if alloc > 15:
            ht, ct = sa_optimize(ht, alloc, rng, T0=0.003, T_end=2e-5)
        if rem() > 20:
            ht2 = slsqp_smooth(ht, 5000, maxiter=300)
            ct2 = compute_c5(ht2)
            if ct2 < ct:
                ht, ct = ht2, ct2
        if ct < best_c5:
            best_c5, best_h, best_n = ct, ht, n_try

    # Phase 6: Final intensive SA
    if rem() > 60:
        alloc = min(rem() * 0.35, 200)
        best_h, best_c5 = sa_optimize(best_h, alloc, rng, T0=0.002, T_end=5e-6)
        best_n = len(best_h)

    # Phase 7: Final polish
    if rem() > 20:
        for T in [5000, 20000, 50000]:
            ht = slsqp_smooth(best_h, T, maxiter=500)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht
    if rem() > 15:
        ht = slsqp_direct(best_h, maxiter=500)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
