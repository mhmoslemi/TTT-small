from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def slsqp_optimize(h, T, n, maxiter=500, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5, h, args=(T,),
                  method='SLSQP', bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

    """One subgradient step targeting the worst lags."""
    corr = get_corr(h)
    # Find top-m lags
    idx = np.argsort(corr)[-top_m:]
    # Compute subgradient contribution from each top lag
    # corr[k] = sum_i h[i + k - (n-1)] * (1-h[i])
    # d corr[k] / d h[j] involves terms where j = i + k - (n-1) or j = i
    # This is complex, so use a simpler approach:
    # For each top lag k, the overlap pattern is h shifted by delta = k - (n-1)
    # We want to reduce h[i] where (1-h[i+delta]) is high, and increase h[i+delta] where h[i] is high

            hi = h[valid]

            grad[valid - d] -= hi

        h_pert = project_feasible(h_pert, n_bh)
        h_ref = subgrad_refine(h_pert, n_bh, n_iter=35, top_k=6, lr=0.25)
        # occasional SLSQP polish
        if it % 20 == 19 and elapsed() < budget_s * 0.40:
            h_ref = slsqp(h_ref, 500, n_bh, maxiter=200, ftol=1e-13)
        c5v = compute_c5(h_ref)
        temp = 0.004 * (1.0 - frac) + 0.0003
        if c5v < bh_c5 or rng.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-10)):
            bh_c5 = c5v
            bh_h = h_ref
            if c5v < best_c5 - 1e-12:
                best_c5 = c5v
                best_h = bh_h.copy()

    # Phase 3: Targeted swap search
    if elapsed() < budget_s * 0.55:
        n_cur = len(best_h)
        for it in range(500):
            if elapsed() > budget_s * 0.55:
                break
            h_try = targeted_swap(best_h, n_cur, rng)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try

    # Phase 4: Multi-n exploration
    for n_try in [60, 80, 100, 140]:
        if elapsed() > budget_s * 0.75:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h_interp, n_try)
        # also try binary snap
        h_bin = (h_interp > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n_try)
        for h_start in [h0, h_bin]:
            if elapsed() > budget_s * 0.75:
                break
            h_ref = subgrad_refine(h_start, n_try, n_iter=60, top_k=6, lr=0.3)
            for T in [400, 1500]:
                if elapsed() > budget_s * 0.75:
                    break
                hc = slsqp(h_ref, T, n_try, maxiter=400, ftol=1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5 - 1e-12:
                    best_c5 = c5v
                    best_h = hc

    # Phase 5: Aggressive SLSQP polish with high T
    n_cur = len(best_h)
    for T in [3000, 10000, 30000, 80000]:
        if elapsed() > budget_s * 0.93:
            break
        hc = slsqp(best_h, T, n_cur, maxiter=600, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Final micro basin hopping
    if elapsed() < budget_s * 0.98:
        n_cur = len(best_h)
        for it in range(200):
            if elapsed() > budget_s * 0.98:
                break
            h_pert = best_h + 0.01 * rng.randn(n_cur)
            h_pert = project_feasible(h_pert, n_cur)
            h_ref = subgrad_refine(h_pert, n_cur, n_iter=20, top_k=8, lr=0.15)
            c5v = compute_c5(h_ref)
            if c5v < best_c5 - 1e-13:
                best_c5 = c5v
                best_h = h_ref

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
