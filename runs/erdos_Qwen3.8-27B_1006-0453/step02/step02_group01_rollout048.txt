    n = len(h)

    h = np.asarray(h, dtype=float)

        x_old = np.linspace(0, 2, len(h))

        h = np.interp(x_new, x_old, h)
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

    h = h_init.copy()

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_old, h)
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def sa_optimize(h_init, n, seed, max_time, T0=0.02, alpha=0.99997, T_min=1e-10):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    t_end = time.time() + max_time
    T = T0
    n_stall = 0
    stall_limit = 80000
    while time.time() < t_end:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                n_stall = 0
            else:
                n_stall += 1
        else:
            h[i], h[j] = h[j], h[i]
        T *= alpha
        if T < T_min:
            T = T_min
        if n_stall > stall_limit and T > 0.001:
            T = max(T * 1.5, 0.005)
            n_stall = 0
    return best_h, best_c5

def local_search(h, n, max_iter=500, time_limit=30):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_end = time.time() + time_limit
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                best_h[i], best_h[j] = best_h[j], best_h[i]
                c5 = compute_c5(best_h)
                delta = c5 - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                best_h[i], best_h[j] = best_h[j], best_h[i]
        if best_i < 0:
            break
        best_h[best_i], best_h[best_j] = best_h[best_j], best_h[best_i]
        best_c5 += best_delta
    return best_h, best_c5

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h + room * (diff / total), 0, 1)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h + room * (diff / total), 0, 1)
    return h

def smooth_c5_cont(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(x):
    return x.sum() - len(x) / 2.0

def cont_slsqp(h, n, maxiter=300, T=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(smooth_c5_cont, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    best_n = n_parent

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Phase 1: SA on parent resolution (binary)
    n1 = n_parent if n_parent % 2 == 0 else n_parent + 1

    t_alloc1 = min(remaining() * 0.35, 350)

    h_ls, c5_ls = local_search(h_sa, n1, max_iter=300, time_limit=25)
    if c5_ls < best_c5:
        best_c5, best_h, best_n = c5_ls, h_ls, n1

    # Phase 2: Multiple SA restarts at best_n
    if remaining() > 80:

        for k in range(4):

                break
            h_init = best_h.copy()
            for _ in range(rng.randint(3, 8)):

                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_init[i], h_init[j] = h_init[j], h_init[i]
            h_sa2, c5_sa2 = sa_optimize(h_init, best_n, seed + 100 + k, t_each)
            h_ls2, c5_ls2 = local_search(h_sa2, best_n, max_iter=200, time_limit=15)
            if c5_ls2 < best_c5:

    # Phase 3: Try other even resolutions
    if remaining() > 60:

        for n_try in other_ns:

            h_interp = to_binary(best_h, n_try)

        h_cont = project_feasible(best_h.astype(float), n_cur)
        for T in [200, 1000]:
            if remaining() < 10:

            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:

                h_cont = h_opt.copy()

    # Phase 6: Additional SA rounds if time remains
    if remaining() > 60:

        h_bin2 = to_binary(best_h, n_cur if n_cur % 2 == 0 else n_cur + 1)

    # Final: project and verify
    best_h = project_feasible(best_h, len(best_h))
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
