import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    s = h.sum() - n / 2.0
    val += lam * s * s
    return val

def smooth_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    ws = w.sum()
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if L >= 0:
            if n - 1 - L >= 0:
                grad[:n - L] += wk * (1.0 - h[L:])
            if L <= n - 1:
                grad[L:] -= wk * h[:n - L]
        else:
            Lp = -L
            if n - 1 - Lp >= 0:
                grad[Lp:] += wk * (1.0 - h[:n - Lp])
            if n - 1 - Lp >= 0:
                grad[:n - Lp] -= wk * h[Lp:]
    grad *= dx
    s = h.sum() - n / 2.0
    grad += 2.0 * lam * s
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-12:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-12:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def lfbg_optimize(h0, T, n, lam=100.0, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T, lam),
                   jac=smooth_grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return project_feasible(np.clip(res.x, 0, 1), n)

def subgradient_refine(h, n, iters=5000, alpha0=0.01):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(iters):
        corr = compute_corr(h)
        k_max = np.argmax(corr)
        L = k_max - (n - 1)
        # subgradient of corr[k_max] w.r.t. h
        sg = np.zeros(n)
        if L >= 0:
            if n - 1 - L >= 0:
                sg[:n - L] += (1.0 - h[L:])
            if L <= n - 1:
                sg[L:] -= h[:n - L]
        else:
            Lp = -L
            if n - 1 - Lp >= 0:
                sg[Lp:] += (1.0 - h[:n - Lp])
            if n - 1 - Lp >= 0:
                sg[:n - Lp] -= h[Lp:]
        alpha = alpha0 / (1.0 + 0.01 * it)
        h = h - alpha * dx * sg
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    s = h.sum() - n / 2.0
    return val + lam * s * s

def smooth_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = k - (n - 1)
        if L >= 0:
            if n - 1 - L >= 0:
                grad[:n - L] += wk * (1.0 - h[L:])
            if L <= n - 1:
                grad[L:] -= wk * h[:n - L]
        else:
            Lp = -L
            if n - 1 - Lp >= 0:
                grad[Lp:] += wk * (1.0 - h[:n - Lp])
                grad[:n - Lp] -= wk * h[Lp:]
    grad *= dx
    s = h.sum() - n / 2.0
    grad += 2.0 * lam * s
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-12:
            return h
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def lfbg_opt(h0, T, n, lam=100.0, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h0, args=(T, lam), jac=smooth_grad,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(np.clip(h0, 0, 1), n)

def multi_subgrad_refine(h, n, iters=5000, alpha0=0.008, T_soft=40.0):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(iters):
        corr = compute_corr(h)
        m = np.max(corr)
        w = np.exp(T_soft * (corr - m))
        w /= w.sum()
        sg = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-10:
                continue
            L = k - (n - 1)
            if L >= 0:
                if n - 1 - L >= 0:
                    sg[:n - L] += wk * (1.0 - h[L:])
                if L <= n - 1:
                    sg[L:] -= wk * h[:n - L]
            else:
                Lp = -L
                if n - 1 - Lp >= 0:
                    sg[Lp:] += wk * (1.0 - h[:n - Lp])
                    sg[:n - Lp] -= wk * h[Lp:]
        alpha = alpha0 / (1.0 + 0.002 * it)
        h = h - alpha * dx * sg
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
