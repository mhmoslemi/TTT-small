                g[j] -= np.dot(w[o + keys2], h[j + keys2])
    return g * dx

def grad_desc(h0, dx, ts, deadline, rng, max_it=2000):
    n = len(h0)
    h, bh = h0.copy(), h0.copy()
    bc = compute_c5(h0, dx)
    stall = 0
    for i in range(max_it):
        if time.time() > deadline:
            break
        b = 20.0 + 80.0 * min(1.0, i / (0.4 * max_it))
        lr = 0.008 / (1.0 + 0.001 * i)
        g = softmax_grad(h, dx, b)
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                bh, bc, h, stall = h2.copy(), c2, h2, 0
                ok = True
                break
        if not ok:
            stall += 1
            if stall > 200:
                break
    return bh, bc

def coord_search(h0, dx, ts, deadline, rng, max_rounds=500):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    cmax = bc / dx
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        step = max(0.001, 0.008 * 0.96 ** (rnd // 8))
        deltas = np.array([step, -step, step * 0.4, -step * 0.4, step * 2, -step * 2])
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            sv = h[j]
            for d in deltas:
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                h = proj(h, ts)
                nm = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if nm < cmax - 1e-14:
                    cmax, bc, improved = nm, nm * dx, True
                    break
                h[j] = sv + d
            if h[j] != sv:
                if h[j] < 0:
                    h[j] = 0
                elif h[j] > 1:
                    h[j] = 1
            else:
                h[j] = sv
        if not improved:
            found = False
            for _ in range(40):
                if time.time() > deadline:
                    break
                j = int(rng.randint(n))
                d = rng.choice([-1, 1]) * rng.uniform(0.01, 0.05)
                h[j] += d
                if h[j] < -0.01 or h[j] > 1.01:
                    h[j] -= d
                    continue
                h = proj(h, ts)
                nm = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if nm < cmax - 1e-14:
                    cmax, bc, found = nm, nm * dx, True
                    break
                h[j] -= d
            if not found:
                break
    return h, bc

def pair_opt(h0, dx, ts, deadline, rng, n_iter=3000):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        nc = compute_c5(h, dx)
        if nc < bc:
            bc = nc
        else:
            h[i], h[j] = h[j], h[i]
    return h, bc

def interp_h(h_src, n_dst):
    x_src = np.linspace(0, 1, len(h_src), endpoint=False)
    x_dst = np.linspace(0, 1, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl = t0 + budget_s - 5

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = np.full(50, 0.5)

    n0 = len(hr)
    ts0 = n0 / 2.0
    dx0 = 2.0 / n0
    h_init = proj(hr, ts0)
    bh, bc = h_init.copy(), compute_c5(h_init, dx0)

    # Phase 1: SLSQP exact (main workhorse)
    if time.time() < dl - 200:
        h1, c1 = slsqp_exact(h_init, dx0, ts0, dl - 180, 600)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 2: Smooth SLSQP (different landscape)
    if time.time() < dl - 180:
        h2, c2 = slsqp_smooth(bh, dx0, ts0, dl - 160, 300, beta=40)
        if c2 < bc:
            bh, bc = h2, c2

    # Phase 3: Gradient descent
    if time.time() < dl - 150:
        h3, c3 = grad_desc(bh, dx0, ts0, dl - 130, rng, 3000)
        if c3 < bc:
            bh, bc = h3, c3

    # Phase 4: SLSQP again after gradient
    if time.time() < dl - 130:
        h4, c4 = slsqp_exact(bh, dx0, ts0, dl - 110, 400)
        if c4 < bc:
            bh, bc = h4, c4

    # Phase 5: Coordinate search (fine)
    if time.time() < dl - 100:
        h5, c5v = coord_search(bh, dx0, ts0, dl - 80, rng, 600)
        if c5v < bc:
            bh, bc = h5, c5v

    # Phase 6: Pair swaps
    if time.time() < dl - 70:
        h6, c6 = pair_opt(bh, dx0, ts0, dl - 50, rng, 4000)
        if c6 < bc:
            bh, bc = h6, c6

    # Phase 7: Try nearby resolutions
    for dn in (1, -1, 2, -2, 3, 4, -3, 5, -4, 6):
        nt = n0 + dn
        if nt < 30 or nt > 90 or time.time() > dl - 50:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        h_new = proj(interp_h(bh, nt), tst)
        h7, c7 = slsqp_exact(h_new, dxt, tst, dl - 30, 250)
        if c7 < bc:
            if time.time() < dl - 20:
                h8, c8 = coord_search(h7, dxt, tst, dl - 10, rng, 150)
                if c8 < c7:
                    c7, h7 = c8, h8
            bh, bc = h7, c7
            break

    # Phase 8: Perturbation restarts
    if time.time() < dl - 60:
        nb = len(bh)
        dxb, tsb = 2.0 / nb, nb / 2.0
        for amp in (0.04, 0.02, 0.01):
            if time.time() > dl - 20:
                break
            hp = proj(bh + amp * rng.randn(nb), tsb)
            h9, c9 = slsqp_exact(hp, dxb, tsb, dl - 10, 300)
            if c9 < bc:
                if time.time() < dl - 8:
                    h10, c10 = coord_search(h9, dxb, tsb, dl - 4, rng, 100)
                    if c10 < c9:
                        c9, h9 = c10, h9
                bh, bc = h9, c9

    # Phase 9: Final polish
    if time.time() < dl - 10:
        nb = len(bh)
        dxb, tsb = 2.0 / nb, nb / 2.0
        hf, cf = slsqp_exact(bh, dxb, tsb, dl - 3, 200)
        if cf < bc:
            bh, bc = hf, cf
        hd, cd = coord_search(bh, dxb, tsb, dl - 1, rng, 80)
        if cd < bc:
            bh, bc = hd, cd

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
