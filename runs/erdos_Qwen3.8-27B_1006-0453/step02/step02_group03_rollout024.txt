import numpy as np
from scipy.optimize import minimize
import time

_g = {}

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(400):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
    return h

def _sobj(x):
    b = _g['b']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    e = np.exp(b * (corr - m))
    return m + np.log(e.sum()) / b

def _sgrad(x):
    n, b = _g['n'], _g['b']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    w = np.exp(b * (corr - m))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(n):
        s_lo = max(-(n-1), -i)
        s_hi = min(n-1, n-1-i)
        if s_lo <= s_hi:
            s = np.arange(s_lo, s_hi+1)
            g[i] += np.dot(w[n-1+s], 1.0 - x[i+s])
        s_lo2 = max(-(n-1), i-(n-1))
        s_hi2 = min(n-1, i)
        if s_lo2 <= s_hi2:
            s = np.arange(s_lo2, s_hi2+1)
            g[i] -= np.dot(w[n-1+s], x[i-s])
    return g

def _seq(x):
    return x.sum() - _g['ts']

def _seqjac(x):
    return np.ones(_g['n'])

    n = len(h0)

                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.x is not None:
            hp = proj(r.x, ts)
            if compute_c5(hp, 2.0/n) <= compute_c5(h0, 2.0/n) + 1e-12:
                best = hp
    except Exception:
        pass
    return best
