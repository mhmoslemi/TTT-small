import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-12) & (h < 1.0 - 1e-12)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            space = 1.0 - h[i]
            add = min(diff, space)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff <= 1e-12:
                break
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    c = compute_corr(h)
    # f = -T * log(sum exp(-c/T)) * dx
    nc = -c / T
    nc_max = np.max(nc)
    nc_shifted = nc - nc_max
    exp_nc = np.exp(nc_shifted)
    log_sum = nc_max + np.log(np.sum(exp_nc))
    f = -T * log_sum * dx
    q = exp_nc / np.sum(exp_nc)  # softmax weights, length 2n-1
    
    # Gradient: grad[j] = -dx * (A[j] - B[j])
    # A[j] = sum_i (1-h[i]) * q[n-1+j-i]
    # B[j] = sum_i h[i] * q[n-1+i-j]
    one_minus_h = 1.0 - h
    A = np.zeros(n)
    B = np.zeros(n)
    for j in range(n):

        q_idx_b = n - 1 + np.arange(n) - j  # n-1+i-j for i=0..n-1
        A[j] = np.dot(one_minus_h, q[q_idx_a])
        B[j] = np.dot(h, q[q_idx_b])
    grad = -dx * (A - B)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-12) & (h < 1.0 - 1e-12)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-13:
                break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff <= 1e-13:
                break
    return h

def soft_obj_and_grad(h, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    nc = -c / T
    nc -= np.max(nc)
    ec = np.exp(nc)
    logsum = np.log(np.sum(ec))
    f = -T * logsum * dx
    q = ec / np.sum(ec)
    oome = 1.0 - h
    A = np.zeros(n)
    B = np.zeros(n)
    ar = np.arange(n)
    for j in range(n):
        qa = n - 1 + j - ar
        qb = n - 1 + ar - j
        A[j] = np.dot(oome, q[qa])
        B[j] = np.dot(h, q[qb])
    grad = -dx * (A - B)
    return f, grad

def penalized_og(x, n, T, pen):
    x = np.clip(x, 0.0, 1.0)
    f, g = soft_obj_and_grad(x, n, T)
    se = np.sum(x) - n * 0.5
    f += pen * se * se
    g = g + 2.0 * pen * se
    return f, g

def lbfgs_opt(h, n, T, max_iter=3000, pen=500.0):
    res = opt.minimize(lambda x: penalized_og(x, n, T, pen), h.copy(),
                       jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 30})
    if hasattr(res, 'x') and res.x is not None:
        return fix_sum(res.x, n)
    return h

def slsqp_opt(h, n, T, max_iter=2000):
    res = opt.minimize(lambda x: soft_obj_and_grad(x, n, T), h.copy(),
                       jac=True, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5}],
                       options={'maxiter': max_iter, 'ftol': 1e-18})
    if hasattr(res, 'x') and res.x is not None:
        return fix_sum(res.x, n)
    return h

def subgrad_descent(h, n, steps=2000, lr0=0.05):
    h = h.copy()
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(c)
        s = k_max - (n - 1)
        grad = np.zeros(n)
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            valid_hi = np.clip(np.arange(n) - s, 0, n - 1)
            mask_hi = (np.arange(n) - s >= 0) & (np.arange(n) - s < n)
            grad[mask_hi] += (1.0 - h[valid_hi[mask_hi]])
            valid_lo = np.clip(np.arange(n) + s, 0, n - 1)
            mask_lo = (np.arange(n) + s >= 0) & (np.arange(n) + s < n)
            grad[mask_lo] -= h[valid_lo[mask_lo]]
        dx = 2.0 / n
        lr = lr0 / (1.0 + it * 0.005)
        h = h - lr * dx * grad
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum(h, n)
    return h

def coordinate_refine(h, n, passes=3):
    h = h.copy()
    dx = 2.0 / n
    for _ in range(passes):
        improved = False
        for j in range(n):
            best_val = compute_c5(h)
            for trial in [h[j] + 0.1, h[j] - 0.1, h[j] + 0.03, h[j] - 0.03, h[j] + 0.01, h[j] - 0.01]:
                if trial < 0 or trial > 1:
                    continue
                h[j] = trial
                h = fix_sum(h, n)
                v = compute_c5(h)
                if v < best_val:
                    best_val = v
                    improved = True
                else:
                    h[j] = trial  # keep trying
            # restore to best (re-optimize j)
            h[j] = h[j]  # already at best from loop
            h = fix_sum(h, n)
    return h

def resample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_ext = np.concatenate([h, h])
    h_new = np.interp(x_new, np.concatenate([x_old, x_old + 2.0]), h_ext,
                      periodic=False)
    return fix_sum(h_new, new_n)

    h_best = np.array(initial_h_values, dtype=float).copy()

    # Phase 1: L-BFGS-B annealing at n_best

    # SLSQP polish

        if try_update(h_sg, n_best):
