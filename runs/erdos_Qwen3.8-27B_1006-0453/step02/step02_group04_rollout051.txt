    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1

    c_max = np.max(corr)
    shifted = corr - c_max
    shifted = np.clip(shifted, -80*T, 0.0)
    exp_vals = np.exp(shifted / T)
    p = exp_vals / np.sum(exp_vals)  # softmax weights

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5

    s = np.sum(h)
