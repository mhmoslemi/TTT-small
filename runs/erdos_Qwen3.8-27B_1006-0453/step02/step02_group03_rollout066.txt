```python
import time

    h = x[:n]
    ac = np.correlate(h, h, mode='full')
    a = -p * ac
    a -= a.max()
    return np.log(np.sum(np.exp(a))) / p

    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:

        h[m] += d / c

    ts = n / 2.0

                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-15})
        if r.x is not None:

    except Exception:
        pass
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _obj(x):
    n, p = _sp['n'], _sp['p']
    h = x[:n]
    B = np.correlate(h, 1.0 - h, mode='full')
    a = p * B
    a -= a.max()
    return np.log(np.sum(np.exp(a))) / p

def _con(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def proj(h, ts):
    h = np.array(h, float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1) if d > 0 else (h > 0)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def slsqp(h0, p, mit):
    n = len(h0)
    _sp['n'] = n
    _sp['ts'] = n / 2.0
    _sp['p'] = p
    try:
        r = minimize(_obj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con}],
                     options={'maxiter': mit, 'ftol': 1e-16})
        if r.x is not None:
            return proj(r.x, n / 2.0)
    except Exception:
        pass
    return h0.copy()

def peak_grad(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s > 0:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
        else:
            g[:] += 1.0 - 2.0 * h
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def atk(h0, dl, rng):
    n = len(h0)
    ts = n / 2.0
    h, bc = h0.copy(), c5(h0)
    bh = h.copy()
    stale = 0
    for _ in range(30000):
        if time.time() > dl or stale > 500:
            break
        B = np.correlate(h, 1.0 - h, mode='full')
        kw = int(np.argmax(B))
        s = kw - (n - 1)
        g = peak_grad(h, s)
        found = False
        for lr in (0.04, 0.02, 0.01, 0.005, 0.002, 0.001):
            h2 = proj(h - lr * g, ts)
            c2 = c5(h2)
            if c2 < bc - 1e-15:
                h, bc = h2, c2
                bh = h.copy()
                found = True
                stale = 0
                break
        if not found:
            stale += 1
    return bh, bc

def atk_multi(h0, dl, rng):
    n = len(h0)
    ts = n / 2.0
    h, bc = h0.copy(), c5(h0)
    bh = h.copy()
    stale = 0
    for _ in range(20000):
        if time.time() > dl or stale > 400:
            break
        B = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(B)[-3:]
        g = np.zeros(n)
        for kw in top_k:
            s = kw - (n - 1)
            g += peak_grad(h, s)
        g /= 3.0
        found = False
        for lr in (0.03, 0.015, 0.008, 0.004, 0.002):
            h2 = proj(h - lr * g, ts)
            c2 = c5(h2)
            if c2 < bc - 1e-15:
                h, bc = h2, c2
                bh = h.copy()
                found = True
                stale = 0
                break
        if not found:
            stale += 1
    return bh, bc

def cds(h0, dl, rng):
    n = len(h0)
    ts = n / 2.0
    h, bc = h0.copy(), c5(h0)
    bh = h.copy()
    stuck = 0
    for outer in range(800):
        if time.time() > dl or stuck > 40:
            break
        imp = False
        perm = rng.permutation(n)
        for j in perm:
            if time.time() > dl:
                break
            sv = h[j]
            best_c = bc
            best_h = None
            for d in (0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005, 0.03, -0.03):
                v = sv + d
                if v < -1e-12 or v > 1 + 1e-12:
                    continue
                h[j] = v
                np.clip(h, 0, 1, out=h)
                h = proj(h, ts)
                c = c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            if best_h is not None:
                h = best_h
                bc = best_c
                imp = True
                bh = h.copy()
            else:
                h[j] = sv
        if not imp:
            stuck += 1
    return bh, bc

def full_opt(h0, dl, rng):
    n = len(h0)
    h, c = h0.copy(), c5(h0)
    for p in (3, 8, 15, 30, 60, 120):
        if time.time() > dl - 6:
            break
        h2 = slsqp(h, p, 250)
        c2 = c5(h2)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 40:
        h2, c2 = atk(h, dl - 25, rng)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 25:
        h2, c2 = atk_multi(h, dl - 15, rng)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 10:
        h2, c2 = cds(h, dl - 5, rng)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 3:
        h2 = slsqp(h, 150, 200)
        c2 = c5(h2)
        if c2 < c:
            h, c = h2, c2
    return h, c

def starts(n, ts, rng, ref):
    out = []
    if ref is not None:
        if len(ref) == n:
            out.append(ref.copy())
        else:
            out.append(proj(np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(ref)), ref), ts))
    out.append(np.full(n, 0.5))
    x = np.arange(n) / n
    for m, a in ((1, 0.25), (2, 0.25), (3, 0.2), (5, 0.15), (7, 0.1)):
        out.append(proj(0.5 + a * np.sin(2 * np.pi * m * x + 0.3 * m), ts))
    for _ in range(2):
        h = np.zeros(n)
        idx = rng.choice(n, int(ts), replace=False)
        h[idx] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[rng.randint(n)] = fr
        out.append(proj(h, ts))
    out.append(proj(rng.uniform(0, 1, n), ts))
    return out

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl = t0 + budget_s - 3
    bh, bc = None, 1e9

    try:

            if c2 < bc:

                bc = c2

    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = c5(bh)
```
