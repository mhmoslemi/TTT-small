    n = len(h)
    g = 1.0 - h
    # Full correlation: c[lag] for lag = -(n-1), ..., n-1
    # Output index k corresponds to lag = k - (n-1)
    # c[k] = sum_i h[i] * g[i + k - (n-1)] for valid i

    c = np.zeros(2*n - 1)
    for lag in range(-(n-1), n):
        idx = lag + (n-1)
        # valid i: 0 <= i < n and 0 <= i+lag < n
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        c[idx] = np.dot(h[i_start:i_end], g[i_start+lag:i_end+lag])

```python
def project_onto_constraint(h_raw, n):
    """Project h_raw onto {0<=h<=1, sum(h)=n/2}"""
    target = n * 0.5
    # Bisection on lambda
    # h_proj[i] = clip(h_raw[i] - lam, 0, 1)
    # Need sum(h_proj) = target
    
    def sum_at_lam(lam):
        return np.sum(np.clip(h_raw - lam, 0.0, 1.0))
    
    lo, hi = h_raw.min() - 1.0, h_raw.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = sum_at_lam(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h_raw - lam, 0.0, 1.0)
```

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Water-filling projection
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h_out = project(h, n)
    d2 = target - np.sum(h_out)
    if abs(d2) > 1e-13:
        int_mask = (h_out > 1e-10) & (h_out < 1.0 - 1e-10)
        if np.sum(int_mask) > 0:
            h_out[int_mask] += d2 / np.sum(int_mask)
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def project(h_raw, n):
    target = n * 0.5
    lo = h_raw.min() - 2.0
    hi = h_raw.max() + 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h_raw - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h_raw - (lo + hi) * 0.5, 0.0, 1.0)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_topk(h, c, k, T):
    n = len(h)
    idx = np.argsort(c)[-k:]
    ct = c[idx]
    w = np.exp((ct - ct.max()) / T)
    w /= w.sum()
    g = np.zeros(n)
    ctr = n - 1
    for ii in range(k):
        lag = int(idx[ii]) - ctr
        wi = w[ii]
        if lag == 0:
            g += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n - lag] += wi * (1.0 - h[lag:])
            g[lag:] -= wi * h[:n - lag]
        else:
            ell = -lag
            g[ell:] += wi * (1.0 - h[:n - ell])
            g[:n - ell] -= wi * h[ell:]
    return g

def momentum_run(h0, n, t0, deadline, T_start=0.01, T_end=0.0002,
                 lr0=0.025, rho=0.88, k_init=25, k_end=8):
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    T = T_start
    stall = 0
    total_iters = 0
    while time.time() < deadline:
        total_iters += 1
        c = corr_array(h)
        frac = min(1.0, total_iters / 8000.0)
        T_cur = T_start * (T_end / T_start) ** frac
        k_cur = max(k_end, int(k_init - (k_init - k_end) * frac))
        g = grad_topk(h, c, k_cur, T_cur)
        gnorm = np.linalg.norm(g)
        if gnorm > 2.0:
            g *= 2.0 / gnorm
        v = rho * v - lr * g
        h_new = project(h + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best = h_new.copy()
            best_c5 = c5_new
            stall = 0
        elif c5_new < best_c5 + 0.0005:
            stall += 1
        else:
            stall += 3
            v *= 0.3
            lr *= 0.7
        h = h_new
        if stall > 80:
            lr = max(lr * 0.8, 0.0005)
            if stall > 200:
                lr *= 0.5
                v *= 0.1
                stall = 50
    return best, best_c5

def slsqp_opt(h, n, T, maxiter=800):
    k_use = 12
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        top = np.sort(c)[-k_use:]
        m = top.max()
        s = np.sum(np.exp((top - m) / T))
        return float((m + T * np.log(max(s, 1e-300)) / k_use) * (2.0 / n))
    def sc(x):
        return np.sum(x) - n * 0.5
    try:
        res = opt.minimize(obj, h.copy(), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sc}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    hn = np.interp(xn, xo, h, period=2.0)
    return fix_sum(hn, new_n)

    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)

    def rem():

    # Phase 1: Long momentum descent (T 0.02->0.0003, k 30->8)
    if rem() > 350:

        if c1 < c5_best:
            h_best, c5_best = h1, c1

    # Phase 2: Second momentum pass, sharper T focus
    if rem() > 250:
        d2 = t0 + budget_s * 0.62

        if c2 < c5_best:
            h_best, c5_best = h2, c2

    # Phase 3: SLSQP polish cascade
    for T_sl in [0.003, 0.0008, 0.0002, 0.00005]:

            break

        c3 = compute_c5(h3)
        if c3 < c5_best:
            h_best, c5_best = h3, c3

    rng = np.random.default_rng(seed + 7)

        h_up = upsample(h_best, n_up)
        du = t0 + budget_s * 0.95

    return (h_best, c5_final, n)
