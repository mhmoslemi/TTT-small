    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:

    lo, hi = -1.0, 1.0

            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    diff = target - np.sum(h)
    m = (h > 1e-9) & (h < 1 - 1e-9)
    if np.sum(m) > 0 and abs(diff) > 1e-10:
        h[m] += diff / np.sum(m)
        h = np.clip(h, 0, 1)

def obj_smax(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    if T < 1e-14:

    def g(x): return np.sum(x) - n * 0.5
    def gj(x): return np.ones(n)
    r = opt.minimize(f, h0, method='SLSQP',
                     bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':g,'jac':gj}],
                     options={'maxiter':mi,'ftol':1e-14,'iprint':0})
    return proj(r.x, n)

def grad_worst_lag(h, n, top_m=3):
    """Compute gradient w.r.t. h for the top-m worst lags, sum them."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Find top m indices
    top_idx = np.argsort(c)[-top_m:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n - 1)  # actual shift
        if k >= 0:

            for i in range(n - k):

                grad[i+k2] -= h[i]  # d/dh[i+k2] of h[i]*(1-h[i+k2]) = -h[i]
    return grad

def grad_step(h, n, lr, top_m=3):
    """One gradient descent step on the max correlation, projected."""
    g = grad_worst_lag(h, n, top_m)
    h_new = h - lr * g
    return proj(h_new, n)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5 * (lo + hi), 0, 1)
    diff = target - np.sum(h)
    m = (h > 1e-9) & (h < 1.0 - 1e-9)
    if np.sum(m) > 0 and abs(diff) > 1e-10:
        h[m] += diff / np.sum(m)
        h = np.clip(h, 0, 1)
    return h

def obj_smax(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    if T < 1e-14:
        return cm * dx
    return (cm + T * np.log(np.sum(np.exp((c - cm) / T)))) * dx

def slsqp(h0, n, T, mi):
    def f(x):
        return obj_smax(x, T)
    def g(x):
        return np.sum(x) - n * 0.5
    def gj(x):
        return np.ones(n)
    r = opt.minimize(f, h0, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': g, 'jac': gj}],
                     options={'maxiter': mi, 'ftol': 1e-14, 'iprint': 0})
    return proj(r.x, n)

def grad_top_lags(h, n, top_m=3):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-top_m:]
    grad = np.zeros(n)
    for j in top_idx:
        k = j - (n - 1)
        if k >= 0:
            L = n - k
            grad[:L] -= h[k:]
            grad[k:] += (1.0 - h[:L])
        else:
            k2 = -k
            L = n - k2
            grad[:L] += (1.0 - h[k2:])
            grad[k2:] -= h[:L]
    return grad

def gradient_descent(h0, n, n_steps, lr_init, lr_decay, top_m=3):
    h = h0.copy()
    lr = lr_init
    for _ in range(n_steps):
        g = grad_top_lags(h, n, top_m)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        h = proj(h - lr * g / g_norm * 0.01, n)
        lr *= lr_decay
    return h

def block_mutate(h, n, rng, max_block, sigma):
    h_new = h.copy()
    bs = rng.randint(2, max_block + 1)
    start = rng.randint(0, max(1, n - bs))
    h_new[start:start+bs] += rng.randn(bs) * sigma
    return proj(h_new, n)

def de_mutate(pop, n, F, rng):
    idxs = rng.choice(len(pop), size=3, replace=False)
    v = pop[idxs[0]].copy() + F * (pop[idxs[1]] - pop[idxs[2]])
    return proj(v, n)
