def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)

    lo, hi = -1.5, 1.5

        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:

    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

    """Convert to binary with exactly n/2 ones, closest to h."""
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max

            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 6: Final binary polish + SLSQP
    elapsed = time.time() - t_start
    if budget_s - elapsed > 50:
        h_final_bin = to_binary(h_best, n)
        rng_f = np.random.RandomState(seed + 99999)
        t_final = min(30, int((budget_s - elapsed) * 0.5))
        h_f, c_f = batch_sa(h_final_bin, n, t_start, t_start + t_final, rng_f, 0.005, 0.0002, 8)
        if c_f < c5_best:
            h_best = h_f.copy()
            c5_best = c_f
        # One last SLSQP
        elapsed = time.time() - t_start
        if budget_s - elapsed > 15:
            h_opt = do_slsqp(h_best.copy(), n, 0.0001, 500)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    h_best = proj_constraint(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
