    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0

        else:
            hi = mid

def exact_topk_grad(h, k, dx):
    """Compute gradient from top-k maximizing shifts. Returns (grad, c5, top_shifts)."""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * dx
    # Get top-k indices in the full correlation
    topk_idx = np.argsort(corr)[-k:]
    # Convert to shifts: shift = idx - (N-1)
    shifts = topk_idx - (N - 1)

        s = int(s)
        if s >= 0:
            valid = N - s
            if valid > 0:
                grad[s:] += (1.0 - h[:valid])
                grad[:valid] -= h[s:]
        else:
            t = -s
            valid = N - t
            if valid > 0:
                grad[:valid] += (1.0 - h[t:])
                grad[t:] -= h[:valid]
    grad *= dx
    return grad, c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_grad(h, k, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(corr.max()) * dx
    if k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -k)[-k:]
    shifts = idx - (N - 1)
    grad = np.zeros(N)
    for s in shifts:
        s = int(s)
        if s >= 0:
            v = N - s
            if v > 0:
                grad[s:] += (1.0 - h[:v])
                grad[:v] -= h[s:]
        else:
            t = -s
            v = N - t
            if v > 0:
                grad[:v] += (1.0 - h[t:])
                grad[t:] -= h[:v]
    grad *= dx
    return grad, c5
