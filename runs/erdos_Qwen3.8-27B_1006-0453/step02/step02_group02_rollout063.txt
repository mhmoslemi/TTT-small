So the change in f_k is at most ±4 (from the four terms involving a or b).

    for l in range(-(n-1), n):

        if wl < 1e-13:

            grad += dx * wl * (1.0 - 2.0 * h)

        h_bin[b] = 1.0

            h_bin[b] = 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.sum(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    mx = np.max(f)
    s = np.clip(t * (f - mx), -30, 30)
    exps = np.exp(s)
    tot = np.sum(exps)
    obj = mx + np.log(tot) / t + lam * (np.sum(h) - n * 0.5) ** 2
    w = exps / tot
    grad = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-12:
            continue
        if l == 0:
            grad += dx * wl * (1.0 - 2.0 * h)
        elif l > 0:
            grad[:n - l] += dx * wl * (1.0 - h[l:])
            grad[l:] += -dx * wl * h[:n - l]
        else:
            m = -l
            grad[m:] += dx * wl * (1.0 - h[:n - m])
            grad[:n - m] += -dx * wl * h[m:]
    grad += 2.0 * lam * (np.sum(h) - n * 0.5)
    return obj, grad

        else:

        T *= 0.9997
    return best, best_c5
