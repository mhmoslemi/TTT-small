        else:

        if c == 0:
            break

    return x[_SP['n']]

def _sineq(x):
    n = _SP['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _seq(x):
    return float(np.sum(x[:_SP['n']]) - _SP['ts'])

def slsqp_opt(h0, ts, mit=300):
    n = len(h0)
    _SP['n'], _SP['ts'] = n, ts
    t0 = float(np.max(np.correlate(h0, 1.0 - h0, mode='full')))
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0, 1)] * n + [(0.0, float(n) * 2)]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _seq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-14})
        if r.x is not None and r.x[-1] < t0 + 0.5:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr) <= compute_c5(h0) + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def lse_grad(h, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) where c = correlate(h,1-h,'full')"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    p = np.exp(beta * (c - cm))
    p /= p.sum()
    # Weighted combination of individual gradients
    g = np.zeros(n)
    for j in range(n):
        for k in range(2 * n - 1):
            d = k - n + 1  # shift
            # contribution from (1-h[j-d]) * I[0<=j-d<n]
            if 0 <= j - d < n:

            # contribution from -h[j+d] * I[0<=j+d<n]
            if 0 <= j + d < n:
                g[j] -= p[k] * h[j + d]
    return g

    bc = compute_c5(h)
    bh = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

_SP = {'n': 0, 'ts': 0}

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def _sobj(x):
    return x[_SP['n']]

def _sineq(x):
    n = _SP['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _seq(x):
    return float(np.sum(x[:_SP['n']]) - _SP['ts'])

def slsqp_opt(h0, ts, mit=300):
    n = len(h0)
    _SP['n'], _SP['ts'] = n, ts
    t0 = float(np.max(np.correlate(h0, 1.0 - h0, mode='full')))
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n) * 2.0)]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _seq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-15})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr) <= compute_c5(h0) + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def lse_grad(h, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    p = np.exp(beta * (c - cm))
    s = p.sum()
    if s < 1e-30:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p /= s
    g = np.zeros(n)
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(n - 1, j)
        m = np.arange(lo, hi + 1)
        d = (j - m)
        g[j] += np.dot(p[d + n - 1], 1.0 - h[m])
        lo2 = max(0, -j)
        hi2 = min(n - 1, n - 1 - j)
        m2 = np.arange(lo2, hi2 + 1)
        d2 = (m2 - j)
        g[j] -= np.dot(p[d2 + n - 1], h[m2])
    return g

def lse_descend(h0, ts, iters, deadline):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h)
    bh = h.copy()
    st = 0
    for i in range(iters):
        if time.time() > deadline:
            break
        beta = 15.0 + 100.0 * min(1.0, i / max(1, 0.4 * iters))
        lr = 0.015 / (1.0 + 0.0008 * i)
        g = lse_grad(h, beta)
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2)
            if c2 < bc - 1e-16:
                bh, bc, h = h2.copy(), c2, h2
                st = 0
                ok = True
                break
        if not ok:
            st += 1
            if st > 300:
                break
    return bh, bc

def coord_descent(h0, ts, rounds, deadline, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h)
    for r in range(rounds):
        if time.time() > deadline:
            break
        imp = False
        order = rng.permutation(n)
        for j in order:
            sv = h[j]
            for d in (0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.1, -0.1, 0.15, -0.15):
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                h2 = proj(h, ts)
                c = compute_c5(h2)
                if c < bc:
                    h = h2
                    bc = c
                    imp = True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def resample(ref, n_new):
    n_old = len(ref)
    if n_old == n_new:
        return ref.copy()
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(x_new, x_old, ref)

def opt_pipeline(h0, ts, deadline, rng):
    n = len(h0)
    h, c = h0.copy(), compute_c5(h0)
    if time.time() < deadline - 20:
        h1 = slsqp_opt(h, ts, 400)
        c1 = compute_c5(h1)
        if c1 < c:
            h, c = h1, c1
    if time.time() < deadline - 40:
        h2, c2 = lse_descend(h, ts, 5000, deadline - 20)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 20:
        h3 = slsqp_opt(h, ts, 200)
        c3 = compute_c5(h3)
        if c3 < c:
            h, c = h3, c3
    if time.time() < deadline - 10:
        h4, c4 = coord_descent(h, ts, 40, deadline - 5, rng)
        if c4 < c:
            h, c = h4, c4
    if time.time() < deadline - 5:
        h5 = slsqp_opt(h, ts, 100)
        c5 = compute_c5(h5)
        if c5 < c:
            h, c = h5, c5
    return h, c
