import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp(T * (corr - c_max))
    w_sum = w.sum()
    val = (c_max + np.log(w_sum) / T) * dx
    w /= w_sum
    
    k = np.arange(2 * n - 1) - (n - 1)
    j0 = np.arange(n)
    
    diff1 = j0[:, None] - k[None, :]
    valid1 = (diff1 >= 0) & (diff1 < n)
    h_idx1 = np.where(valid1, diff1, 0)
    term1 = np.where(valid1, 1.0 - h[h_idx1], 0.0)
    
    diff2 = j0[:, None] + k[None, :]
    valid2 = (diff2 >= 0) & (diff2 < n)
    h_idx2 = np.where(valid2, diff2, 0)
    term2 = np.where(valid2, -h[h_idx2], 0.0)
    
    grad = (term1 + term2) @ w
    grad *= T * dx
    
    return val, grad

def projected_gradient_step(h, grad, alpha, n):
    h_new = h - alpha * grad
    return project_feasible(h_new, n)

def pg_optimize(h0, T, n, max_iter=5000, alpha0=0.005, alpha_min=1e-8):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    alpha = alpha0
    for it in range(max_iter):
        val, grad = smooth_c5_and_grad(h, T)
        
        # Backtracking line search
        for attempt in range(20):
            h_new = projected_gradient_step(h, grad, alpha, n)
            val_new = compute_c5(h_new)
            if val_new < val or attempt > 15:

            alpha *= 0.5
        
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        
        h = h_new
        alpha = max(alpha * 0.98, alpha_min)
        
        if it > 100 and it % 200 == 0:
            # Occasionally increase T? Or just continue
            pass
    
    return best_h, best_val

def slsqp_optimize(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        val, _ = smooth_c5_and_grad(x, T)
        return val
    
    def grad(x):
        _, g = smooth_c5_and_grad(x, T)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

    corr = np.correlate(h, 1.0 - h, mode='full')

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_vals(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    e = T * (corr - c_max)
    e -= e.max()
    w = np.exp(e)
    w_sum = w.sum()
    val = (c_max + np.log(w_sum) / T) * dx
    w /= w_sum
    return val, w, corr

def compute_grad(h, w):
    n = len(h)
    k = np.arange(2 * n - 1) - (n - 1)
    j0 = np.arange(n)
    d1 = j0[:, None] - k[None, :]
    v1 = (d1 >= 0) & (d1 < n)

    d2 = j0[:, None] + k[None, :]
    v2 = (d2 >= 0) & (d2 < n)
    idx2 = np.where(v2, d2, 0)
    t2 = np.where(v2, -h[idx2], 0.0)
    return (t1 + t2) @ w

    val, _, _ = smooth_vals(x, T)
    return val

    _, w, _ = smooth_vals(x, T)
    n = len(x)
    g = compute_grad(x, w)
    return g * T * (2.0 / n)

    return x.sum() - n / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_vals(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    e = T * (corr - c_max)
    e -= e.max()
    w = np.exp(e)
    w_sum = w.sum()
    val = (c_max + np.log(w_sum) / T) * dx
    w /= w_sum
    return val, w, corr

def compute_grad(h, w):
    n = len(h)
    k = np.arange(2 * n - 1) - (n - 1)
    j0 = np.arange(n)
    d1 = j0[:, None] - k[None, :]
    v1 = (d1 >= 0) & (d1 < n)
    idx1 = np.where(v1, d1, 0)
    t1 = np.where(v1, 1.0 - h[idx1], 0.0)
    d2 = j0[:, None] + k[None, :]
    v2 = (d2 >= 0) & (d2 < n)
    idx2 = np.where(v2, d2, 0)
    t2 = np.where(v2, -h[idx2], 0.0)
    return (t1 + t2) @ w

def obj_fn(x, T):
    val, _, _ = smooth_vals(x, T)
    return val

def grad_fn(x, T):
    _, w, _ = smooth_vals(x, T)
    return compute_grad(x, w) * (2.0 / len(x))

def cons_fn(x, n):
    return x.sum() - n * 0.5

    return np.ones(n)

            best_h = h_new.copy()
        h = h_new
        alpha *= 0.995
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_vals(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    e = T * (corr - c_max)
    e -= e.max()
    w = np.exp(e)
    w_sum = w.sum()
    val = (c_max + np.log(w_sum) / T) * dx
    w /= w_sum
    return val, w

def compute_grad(h, w):
    n = len(h)
    k = np.arange(2 * n - 1) - (n - 1)
    j0 = np.arange(n)
    d1 = j0[:, None] - k[None, :]
    v1 = (d1 >= 0) & (d1 < n)
    idx1 = np.where(v1, d1, 0)
    t1 = np.where(v1, 1.0 - h[idx1], 0.0)
    d2 = j0[:, None] + k[None, :]
    v2 = (d2 >= 0) & (d2 < n)
    idx2 = np.where(v2, d2, 0)
    t2 = np.where(v2, -h[idx2], 0.0)
    return (t1 + t2) @ w

def obj_fn(x, T):
    val, _ = smooth_vals(x, T)
    return val

def grad_fn(x, T):
    _, w = smooth_vals(x, T)
    return compute_grad(x, w) * (2.0 / len(x))

def cons_fn(x, n):
    return x.sum() - n * 0.5

def pg_optimize(h0, T, n, max_iter, alpha0, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, w = smooth_vals(h, T)
        grad = compute_grad(h, w) * (2.0 / n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        for _ in range(30):
            h_new = project_feasible(h - alpha * grad, n)
            val_new, _ = smooth_vals(h_new, T)
            if val_new < val - 1e-16:
                break
            alpha *= 0.5
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        alpha *= 0.996
        if alpha < 1e-8:
            alpha = 1e-8
    return best_h, best_c5

def coordinate_search(h, n, deadline, max_rounds=100):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        idx = np.random.permutation(n)
        for j in idx[:20]:
            for delta in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                h_try = best_h.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved and rnd > 30:
            break
    return best_h, best_c5

    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

        best_h[:60] = 1.0

    best_h = project_feasible(best_h, len(best_h))

    n = len(best_h)

    # Phase 1: PG with T-schedule
    schedule = [(15, 5000, 3.0), (40, 4500, 2.5), (100, 4000, 1.5),
                (300, 3500, 1.0), (800, 3000, 0.7), (2000, 2500, 0.4), (5000, 2000, 0.2)]
    for T, mx, a0 in schedule:
        if time.time() - t0 > budget_s * 0.38:
            break
        deadline = t0 + budget_s * 0.38
        h_t, c5_t = pg_optimize(best_h, T, n, mx, a0, deadline)

    # Phase 2: Diverse starts + PG
    if time.time() - t0 < budget_s * 0.58:

        starts = [best_h.copy(),
                  project_feasible((np.random.rand(n) < 0.5).astype(float), n),
                  project_feasible(0.5 + 0.4 * np.sin(4.0 * np.pi * x), n)]
        hblk = np.zeros(n)
        hblk[::3] = 1.5
        starts.append(project_feasible(hblk, n))
        hq = np.zeros(n)
        q = n // 4
        hq[:q] = 1.0
        hq[2 * q:3 * q] = 1.0
        starts.append(hq)

                if c5_t < best_c5:
                    best_c5 = c5_t

    # Phase 3: Basin hopping with PG refinement
    if time.time() - t0 < budget_s * 0.78:
        for it in range(30):
            if time.time() - t0 > budget_s * 0.78:

            h_t, c5_t = pg_optimize(hp, 400, n, 1500, 0.8, t0 + budget_s * 0.78)

        except Exception:
            pass

    if time.time() - t0 < budget_s * 0.95:

    h_t, c5_t = pg_optimize(best_h, 80000, n, 3000, 0.3, t0 + budget_s * 0.98)

        best_h = h_t

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
