        improved = False
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            best_c5 = compute_c5(h)
            best_h = h.copy()
            lo, hi = -0.1, 0.1
            for _ in range(7):
                mid = (lo + hi) / 2.0
                hn = h.copy()
                hn[:] += -mid / (n - 1)
                hn[j] = min(max(h[j] + mid, 0.0), 1.0)
                hn = np.clip(hn, 0.0, 1.0)
                c = compute_c5(hn)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_h = hn
                    lo = mid
                else:
                    hi = mid
            if best_c5 < compute_c5(h) - 1e-14:
                h = best_h
                improved = True
        h = fix_sum(h, n)
        if not improved:
            break
    return h

def sa_search(h, n, t_start, t_end, n_steps, rng):
    c_cur = compute_c5(h)
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        T_sa = t_start * (t_end / t_start) ** frac
        j = rng.randint(n)
        target = 0.0 if h[j] < 0.5 else 1.0
        hn = h.copy()
        hn[j] = target
        hn = fix_sum(hn, n)
        c_new = compute_c5(hn)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(T_sa, 1e-12)):
            h = hn
            c_cur = c_new
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 7)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)

    # Phase 1: L-BFGS-B with graduated temperature and top-k
    grad_sched = [(0.06, 50, 500, 200), (0.025, 30, 800, 400),
                  (0.01, 20, 1000, 500), (0.004, 15, 1200, 600),
                  (0.0015, 10, 1500, 800), (0.0005, 8, 2000, 1000),
                  (0.0002, 5, 2500, 1200)]
    for T, k, it, pen in grad_sched:
        if elapsed() > budget_s - 450:
            break
        h = lfbgbs_opt(h, n, T, k, it, pen)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 2: SLSQP top-k refinement
    slsqp_sched = [(0.01, 25, 400), (0.004, 15, 600), (0.001, 8, 800),
                   (0.0003, 5, 1000), (0.0001, 3, 1200)]
    for T, k, it in slsqp_sched:
        if elapsed() > budget_s - 350:
            break
        h = slsqp_opt(h, n, T, k, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 3: Coordinate line search
    if elapsed() < budget_s - 200:
        h = coord_ls(h, n, 3)
        c5 = compute_c5(h)

    # Phase 4: Simulated annealing
    if elapsed() < budget_s - 180:
        h = sa_search(h, n, 0.002, 0.0001, 8000, rng)
        c5 = compute_c5(h)
        h = fix_sum(h, n)
        if elapsed() < budget_s - 140:
            h = coord_ls(h, n, 2)
            c5 = compute_c5(h)

    # Phase 5: Diverse restarts
    best_h = h.copy()
    best_c5 = c5
    for i in range(18):
        if elapsed() > budget_s - 120:
            break
        ht = h.copy()
        mode = i % 5
        if mode == 0:
            ht += np.random.randn(n) * 0.012
        elif mode == 1:
            idx = np.random.choice(n, 12, replace=False)
            ht[idx] += np.random.randn(12) * 0.1
        elif mode == 2:
            idx = np.random.choice(n, 20, replace=False)
            for a in range(0, 18, 2):
                ht[idx[a]], ht[idx[a+1]] = ht[idx[a+1]], ht[idx[a]]
        elif mode == 3:
            ht += np.random.randn(n) * 0.004
        else:
            a = rng.randint(0, n - 35)
            ht[a:a+35] = np.clip(rng.rand(35), 0, 1)
        ht = fix_sum(ht, n)
        ht = lfbgbs_opt(ht, n, 0.004, 15, 500, 500)
        ht = lfbgbs_opt(ht, n, 0.001, 8, 800, 800)
        ht = slsqp_opt(ht, n, 0.001, 5, 400)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_h = ht
            best_c5 = ct
            if elapsed() < budget_s - 60:
                best_h = coord_ls(best_h, n, 1)
                best_c5 = compute_c5(best_h)
    h, c5 = best_h, best_c5

    # Phase 6: Final SLSQP polish
    for T, k, it in [(0.0001, 3, 2000), (0.00003, 2, 2500)]:
        if elapsed() > budget_s - 40:
            break
        h = slsqp_opt(h, n, T, k, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 7: Higher resolution attempt
    if elapsed() < budget_s - 80:
        n2 = min(560, n * 2)
        h2 = upsample(h, n2)
        for T, k, it, pen in [(0.01, 20, 300, 200), (0.003, 10, 500, 400),
                              (0.001, 5, 600, 600), (0.0003, 3, 800, 800)]:
            if elapsed() > budget_s - 25:
                break
            h2 = lfbgbs_opt(h2, n2, T, k, it, pen)
            h2 = fix_sum(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
