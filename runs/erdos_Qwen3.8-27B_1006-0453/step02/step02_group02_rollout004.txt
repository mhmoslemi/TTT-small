    return float(np.sum(c ** p) ** (1.0 / p) * dx)

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    if n % nd == 0:
        return proj(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def gen_binary_start(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = proj(np.array(initial_h_values, dtype=float), 280)
    except Exception:
        bh = proj(np.full(280, 0.5) + rng.randn(280) * 0.05, 280)
    bc = compute_c5(bh)
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}

    # Phase 1: Diverse exploration at n=70
    if tleft() > 450:
        h70_starts = [interp_down(bh, 70)]
        for _ in range(4):
            h70_starts.append(gen_binary_start(70, rng))
        for _ in range(2):
            h70_starts.append(proj(h70_starts[0] + rng.randn(70) * 0.15, 70))
        best70 = h70_starts[0].copy()
        best70_c5 = compute_c5(best70)
        for st in h70_starts:
            if tleft() < 25:
                break
            for tv in [40, 150, 500]:
                if tleft() < 12:
                    break
                try:
                    r = minimize(slsqp_obj_full, st.copy(), args=(tv,), method='SLSQP',
                                 bounds=[(0, 1)] * 70, constraints=cons,
                                 options={'maxiter': 600, 'ftol': 1e-14})
                    hc = proj(np.clip(r.x, 0, 1), 70)
                    c5 = compute_c5(hc)
                    if c5 < best70_c5:
                        best70_c5 = c5
                        best70 = hc.copy()
                except Exception:
                    pass
        up70 = interp_up(best70, 280)
        c5u = compute_c5(up70)
        if c5u < bc:
            bc = c5u
            bh = up70

    # Phase 2: Active-lag optimization at n=140 (key innovation)
    if tleft() > 380:
        h140 = interp_up(bh, 140)
        # Also try from 70
        h140b = interp_up(best70, 140)
        best140 = h140.copy()
        best140_c5 = compute_c5(h140)
        if compute_c5(h140b) < best140_c5:
            best140 = h140b.copy()
            best140_c5 = compute_c5(h140b)
        # Add diverse starts
        starts140 = [best140, proj(best140 + rng.randn(140) * 0.08, 140),
                      proj((best140 > 0.5).astype(float), 140)]
        for st in starts140:
            if tleft() < 30:
                break
            # Iterative active-lag refinement
            h_cur = st.copy()
            for it in range(4):
                if tleft() < 15:
                    break
                corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                topk = max(5, len(corr) // 8)
                active = np.argpartition(corr, -topk)[-topk:]
                for tv in [60, 300]:
                    if tleft() < 12:
                        break
                    try:
                        r = minimize(slsqp_obj_active, h_cur.copy(), args=(tv, active),
                                     method='SLSQP', bounds=[(0, 1)] * 140,
                                     constraints=cons,
                                     options={'maxiter': 1200, 'ftol': 1e-14})
                        hc = proj(np.clip(r.x, 0, 1), 140)
                        c5 = compute_c5(hc)
                        if c5 < compute_c5(h_cur):
                            h_cur = hc
                    except Exception:
                        pass
            c5_140 = compute_c5(h_cur)
            if c5_140 < best140_c5:
                best140_c5 = c5_140
                best140 = h_cur.copy()
        # Upgrade to 280
        up140 = interp_up(best140, 280)
        c5u = compute_c5(up140)
        if c5u < bc:
            bc = c5u
            bh = up140

    # Phase 3: Fine active-lag optimization at n=280
    h280 = bh.copy()
    for outer in range(8):
        if tleft() < 25:
            break
        corr = np.correlate(h280, 1.0 - h280, mode='full')
        topk = max(4, len(corr) // 10)
        active = np.argpartition(corr, -topk)[-topk:]
        improved = False
        for tv in [80, 400, 1500]:
            if tleft() < 15:
                break
            try:
                r = minimize(slsqp_obj_active, h280.copy(), args=(tv, active),
                             method='SLSQP', bounds=[(0, 1)] * 280,
                             constraints=cons,
                             options={'maxiter': 2500, 'ftol': 1e-15})
                hc = proj(np.clip(r.x, 0, 1), 280)
                c5 = compute_c5(hc)
                if c5 < compute_c5(h280) - 1e-7:
                    h280 = hc
                    improved = True
                    if c5 < bc:
                        bc = c5
            except Exception:
                pass
        if not improved:
            break

    # Phase 4: L_p norm refinement at n=280
    if tleft() > 60:
        for p in [8, 16, 32]:
            if tleft() < 20:
                break
            corr = np.correlate(h280, 1.0 - h280, mode='full')
            topk = max(4, len(corr) // 8)
            active = np.argpartition(corr, -topk)[-topk:]
            try:
                r = minimize(lambda x: obj_lp(x, p, active), h280.copy(),
                             method='SLSQP', bounds=[(0, 1)] * 280,
                             constraints=cons,
                             options={'maxiter': 1500, 'ftol': 1e-15})
                hc = proj(np.clip(r.x, 0, 1), 280)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    h280 = hc
            except Exception:
                pass

    # Phase 5: Perturbation + quick optimization
    if tleft() > 50:
        for attempt in range(15):
            if tleft() < 12:
                break
            h_p = proj(h280 + rng.randn(280) * 0.015, 280)
            try:
                corr = np.correlate(h_p, 1.0 - h_p, mode='full')
                topk = max(4, len(corr) // 10)
                active = np.argpartition(corr, -topk)[-topk:]
                r = minimize(slsqp_obj_active, h_p, args=(400, active),
                             method='SLSQP', bounds=[(0, 1)] * 280,
                             constraints=cons,
                             options={'maxiter': 800, 'ftol': 1e-14})
                hc = proj(np.clip(r.x, 0, 1), 280)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    h280 = hc
            except Exception:
                pass

    # Phase 6: Full softmax polish
    if tleft() > 15:
        for tv in [3000, 10000]:
            if tleft() < 8:
                break
            try:
                r = minimize(slsqp_obj_full, h280.copy(), args=(tv,), method='SLSQP',
                             bounds=[(0, 1)] * 280, constraints=cons,
                             options={'maxiter': 4000, 'ftol': 1e-16})
                hc = proj(np.clip(r.x, 0, 1), 280)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    h280 = hc
            except Exception:
                pass

    # Check if n=140 version is better
    h140_final = interp_down(h280, 140)
    c5_140 = compute_c5(h140_final)
    c5_280 = compute_c5(h280)
    if c5_140 < c5_280:
        return h140_final, c5_140, 140
    return h280, c5_280, 280
