def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.ravel(), 0.0, 1.0).copy()
    tgt = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = tgt - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = np.clip(t * (corr - mx), -200, 200)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def softmax_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = np.clip(t * (corr - mx), -200, 200)
    exp_s = np.exp(s)
    se = np.sum(exp_s)
    w = exp_s / se
    obj = (mx + np.log(se) / t) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-14:
            continue
        sg = k - (n - 1)
        if sg >= 0:
            grad[:n - sg] += wi * (1.0 - h[sg:])
            grad[sg:] -= wi * h[:n - sg]
        else:
            m = -sg
            grad[m:] += wi * (1.0 - h[:n - m])
            grad[:n - m] -= wi * h[m:]
    return obj, grad * dx

    return np.array([np.sum(x) - len(x) * 0.5])

    return np.ones((1, len(x)))

                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return proj(r.x, n)

    def objective(x):
        return softmax_obj(x, t)
    
    def gradient(x):
        _, g = softmax_grad(x, t)
        return g
    
    # L-BFGS-B doesn't handle equality constraints well, use augmented Lagrangian
    # or just project periodically
    # Actually, let's use a penalty method
    lam = 100.0
    def obj_pen(x):
        s = np.sum(x) - n * 0.5
        return softmax_obj(x, t) + 0.5 * lam * s * s
    
    def grad_pen(x):
        _, g = softmax_grad(x, t)
        s = np.sum(x) - n * 0.5
        return g + lam * s * np.ones(n)
    
    r = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                 bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(r.x, n)

def subgrad_descent(h0, n, t, eta0, niter, beta=0.9):
    h = h0.copy()
    vel = np.zeros(n)
    eta = eta0
    best = h.copy()

        _, g = softmax_grad(h, t)

        h_new = h + vel
        h_new = proj(h_new, n)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best = h_new.copy()
        h = h_new
        if it % 100 == 99:
            eta *= 0.9
    return best, best_c

    c = compute_c5(h)

                h[i], h[j] = 0.0, 1.0

                    c = c_try
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, c

def lbfgs_opt(h0, n, t, maxiter, penalty):

    tgt = n * 0.5

    def obj_pen(x):
        s = np.sum(x) - tgt
        return softmax_obj(x, t) + 0.5 * penalty * s * s

    def grad_pen(x):
        _, g = softmax_grad(x, t)
        s = np.sum(x) - tgt
        return g + penalty * s * np.ones(n)

    r = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                 bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def subgrad_descent(h0, n, t, eta0, niter, beta, t0_ref, budget_fn):
    h = h0.copy()
    vel = np.zeros(n)
    eta = eta0
    best_c = compute_c5(h)
    best = h.copy()
    for it in range(niter):
        if budget_fn() < 25:
            break
        _, g = softmax_grad(h, t)
        vel = beta * vel - eta * g
        h_new = proj(h + vel, n)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best = h_new.copy()
        h = h_new
        if it % 150 == 149:
            eta *= 0.88
    return best, best_c

def binary_local_search(h0, n, max_passes, max_tries, t0_ref, budget_fn):
    h = proj((h0 > 0.5).astype(float), n)
    c = compute_c5(h)
    rng_local = np.random.RandomState(12345)
    for p in range(max_passes):
        if budget_fn() < 15:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        perm_i = rng_local.permutation(len(ones))[:max_tries]
        perm_j = rng_local.permutation(len(zeros))[:max_tries]
        for ii in perm_i:
            for jj in perm_j:
                if budget_fn() < 12:
                    break
                i, j = ones[ii], zeros[jj]
                h[i], h[j] = 0.0, 1.0
                c_try = compute_c5(h)
                if c_try < c - 1e-13:
                    c = c_try
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 280

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = proj(np.array(initial_h_values, dtype=float), n)
    except Exception:
        bh = proj(0.5 + rng.randn(n) * 0.08, n)
    bc = compute_c5(bh)
    best = bh.copy()

    # Phase 1: SLSQP with multiple t values from parent
    for t_val in [40, 150, 500, 1500]:
        if tleft() < 50:
            break
        try:
            hc = slsqp_opt(best, n, t_val, 2500, 1e-15)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                best = hc
        except Exception:
            pass

    # Phase 2: L-BFGS-B gradient-based (faster per iteration than SLSQP)
    for t_val in [500, 2000, 8000]:
        if tleft() < 40:
            break
        try:
            hc = lbfgs_opt(best, n, t_val, 3000, 50.0)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                best = hc
        except Exception:
            pass

    # Phase 3: Subgradient with Nesterov momentum
    if tleft() > 120:
        h_sg, c_sg = subgrad_descent(best, n, t=400, eta0=0.025,
                                     niter=2500, beta=0.9, t0_ref=t0,
                                     budget_fn=tleft)
        if c_sg < bc:
            bc = c_sg
            best = h_sg

    # Phase 4: Binary local search (pairwise swaps)
    if tleft() > 100:
        h_bin, c_bin = binary_local_search(best, n, max_passes=40,
                                           max_tries=120, t0_ref=t0,
                                           budget_fn=tleft)
        if c_bin < bc:
            bc = c_bin
            best = h_bin.copy()

    # Phase 5: SLSQP from binary + noise
    if tleft() > 50:
        h_noisy = proj(h_bin + rng.randn(n) * 0.04, n)
        for t_val in [300, 1000]:
            if tleft() < 30:
                break
            try:
                hc = slsqp_opt(h_noisy, n, t_val, 2000, 1e-15)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    best = hc
            except Exception:
                pass

    # Phase 6: More subgradient at higher t (sharper max)
    if tleft() > 80:
        for t_val in [2000, 6000]:
            if tleft() < 35:
                break
            h_sg, c_sg = subgrad_descent(best, n, t=t_val, eta0=0.012,
                                         niter=1000, beta=0.85, t0_ref=t0,
                                         budget_fn=tleft)
            if c_sg < bc:
                bc = c_sg
                best = h_sg

    # Phase 7: Another binary refinement
    if tleft() > 70:
        h_bin2, c_bin2 = binary_local_search(best, n, max_passes=30,
                                             max_tries=80, t0_ref=t0,
                                             budget_fn=tleft)
        if c_bin2 < bc:
            bc = c_bin2
            best = h_bin2.copy()
        # Quick SLSQP polish on binary
        if tleft() > 30:
            try:
                hc = slsqp_opt(best, n, 2000, 2000, 1e-15)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    best = hc
            except Exception:
                pass

    # Phase 8: L-BFGS-B final
    if tleft() > 35:
        try:
            hc = lbfgs_opt(best, n, 3000, 3000, 200.0)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                best = hc
        except Exception:
            pass

    # Phase 9: Ultra-fine SLSQP
    if tleft() > 20:
        for t_val in [20000, 80000]:
            if tleft() < 12:
                break
            try:
                hc = slsqp_opt(best, n, t_val, 4000, 1e-18)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    best = hc
            except Exception:
                pass

    # Final verification
    best = proj(np.clip(best, 0.0, 1.0), n)
    final_c = compute_c5(best)
    return best, final_c, n
