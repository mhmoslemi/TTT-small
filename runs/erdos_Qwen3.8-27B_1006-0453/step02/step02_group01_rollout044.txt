def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * min(diff / total, 1.0)
        else:
            total = h.sum()
            if total < 1e-14:

            h += h * min(-diff / total, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

        h_opt_f, c5_opt_f = swap_local_search(h_bin_f, n_cur, max_iters=4000, t0=t0, deadline=deadline5, seed=seed + 200)
        if c5_opt_f < best_c5:
            best_c5 = c5_opt_f
            best_h = h_opt_f.copy()
    if elapsed() < budget_s * 0.93:
        h_p = slsqp_direct(best_h, maxiter=500)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
    if elapsed() < deadline5:
        h_p = slsqp_smooth(best_h, T=5000, maxiter=400)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
