def autocorr_circular(h):
    n = len(h)
    H = fft(h, n=2*n)
    return ifft(H * conj(H)).real[:n]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    c = autocorr_circular(h)
    return float(np.max(dx * (np.sum(h) - c)))

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        diff = target - np.sum(h)
    return h

def grad_ascent(h, n, T_init, T_end, lr, max_iter):
    """Gradient ascent for max soft-min autocorr"""
    h = h.copy()
    for it in range(max_iter):
        c = autocorr_circular(h)
        m = np.min(c)
        T = T_init * (T_end / T_init) ** (it / max(max_iter-1, 1))
        w = np.exp((m - c) / T)
        w /= np.sum(w)
        grad = 2.0 * ifft(fft(w) * fft(h)).real
        h = h + lr * grad / np.maximum(np.linalg.norm(grad), 1e-10)
        h = fix_sum(h, n)
    return h

            c5_cur = compute_c5_fft(h)
            improved = True
        if not improved:

def do_slsqp(h, n, T, max_iter):
    def obj(x):
        c = autocorr_circular(x)
        m = np.min(c)
        w = np.exp((m - c) / T)
        w /= np.sum(w)
        return -np.dot(w, c)  # minimize negative soft-min = maximize soft-min
    
    def grad(x):

        m = np.min(c)
        w = np.exp((m - c) / T)
        w /= np.sum(w)
        g = -2.0 * ifft(fft(w) * fft(x)).real
        # Subtract mean for constraint (sum = n/2)
        g -= np.mean(g)
        return g
    
    def sum_con(x, n):
        return np.sum(x) - n * 0.5
    
    try:

                          method='SLSQP',

                          constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],

        if hasattr(res, 'x') and res.x is not None:

    except Exception:

    return h

import numpy as np
import scipy.optimize as opt
import time
from scipy.fft import fft, ifft

def autocorr_circ(h):
    n = len(h)
    H = fft(h, n=2*n)
    return ifft(H * np.conj(H)).real[:n]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    c = autocorr_circ(h)
    return float(np.max(dx * (np.sum(h) - c)))

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-13:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
        diff = target - np.sum(h)
    return h

def grad_ascent_fft(h, n, T_init, T_end, lr, max_iter):
    h = h.copy()
    for it in range(max_iter):
        c = autocorr_circ(h)
        m = np.min(c)
        frac = it / max(max_iter - 1, 1)
        T = T_init * (T_end / T_init) ** frac
        w = np.exp((m - c) / T)
        w /= np.sum(w)
        grad = 2.0 * ifft(fft(w) * fft(h)).real
        ng = np.linalg.norm(grad)
        if ng > 1e-15:
            h = h + (lr / ng) * grad
        h = fix_sum(h, n)
    return h

def binary_local_search(h, n, max_passes=300):
    h = h.copy()
    h = (h > 0.5).astype(float)
    c5_cur = compute_c5_fft(h)
    for _ in range(max_passes):
        c = autocorr_circ(h)
        s = np.sum(h)
        dx = 2.0 / n
        best_gain = 1e-10
        best_i = -1
        best_v = 0
        for i in range(n):
            d = 1.0 - 2.0 * h[i]
            if abs(d) < 1e-10:
                continue
            eff = d * (np.roll(h, -i) + np.roll(h, i))
            c_new = c + eff
            c5_new = np.max(dx * (s + d - c_new))
            gain = c5_cur - c5_new
            if gain > best_gain:
                best_gain = gain
                best_i = i
                best_v = 1.0 - h[i]
        if best_i >= 0:
            h[best_i] = best_v
            c5_cur = compute_c5_fft(h)
        else:
            break
    h = fix_sum(h, n)
    return h

def sa_binary(h, n, T_init=1.0, T_end=0.005, n_iter=5000):
    h = (h > 0.5).astype(float)
    h = fix_sum(h, n)
    c5_cur = compute_c5_fft(h)
    best = h.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_init * (T_end / T_init) ** frac
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5_fft(h)
        if c5_new < c5_cur or np.random.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_c5

def do_slsqp(h, n, T, max_iter):
    def obj(x):
        c = autocorr_circ(x)
        m = np.min(c)
        w = np.exp((m - c) / T)
        w /= np.sum(w)
        return -float(np.dot(w, c))
    def grad(x):
        c = autocorr_circ(x)
        m = np.min(c)
        w = np.exp((m - c) / T)
        w /= np.sum(w)
        g = -2.0 * ifft(fft(w) * fft(x)).real
        g -= np.mean(g)
        return g
    def sum_con(x, nn):
        return np.sum(x) - nn * 0.5
    try:
        res = opt.minimize(obj, h.copy(), jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                          options={'maxiter': max_iter, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

    np.random.seed(seed)
    t0 = time.time()

    h_best = fix_sum(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    def try_update(hc, nc):
        nonlocal h_best, n_best, c5_best
        hc = fix_sum(hc, nc)
        cv = compute_c5(hc)
        if cv < c5_best - 1e-12:
            h_best, n_best, c5_best = hc, nc, cv
            return True
        return False

    # Phase 1: FFT gradient ascent
    h1 = grad_ascent_fft(h_best, n_best, 0.1, 0.001, 0.4, 8000)
    try_update(h1, n_best)

    # Phase 2: Binary snap + local search
    if budget_s - elapsed() > 60:
        hb = (h_best > 0.5).astype(float)
        hb = fix_sum(hb, n_best)
        hbl = binary_local_search(hb, n_best, 300)
        try_update(hbl, n_best)
        h1b = grad_ascent_fft(hbl, n_best, 0.05, 0.001, 0.3, 5000)
        try_update(h1b, n_best)

    # Phase 3: SLSQP polish
    if budget_s - elapsed() > 120:
        try_update(do_slsqp(h_best, n_best, 0.01, 2000), n_best)
    if budget_s - elapsed() > 100:
        try_update(do_slsqp(h_best, n_best, 0.003, 3000), n_best)

    # Phase 4: Simulated annealing on binary
    if budget_s - elapsed() > 80:
        ha, ca = sa_binary(h_best, n_best, 1.0, 0.005, 8000)
        try_update(fix_sum(ha, n_best), n_best)
        ha2, _ = sa_binary(h_best, n_best, 0.5, 0.002, 5000)
        hb2 = binary_local_search(ha2, n_best, 200)
        try_update(hb2, n_best)

    # Phase 5: Multi-start perturbations
    for _ in range(4):
        if budget_s - elapsed() < 120:
            break
        hp = h_best + np.random.randn(n_best) * 0.08
        hp = fix_sum(hp, n_best)
        hg = grad_ascent_fft(hp, n_best, 0.08, 0.001, 0.35, 4000)
        try_update(hg, n_best)
        hb3 = (hg > 0.5).astype(float)
        hb3 = fix_sum(hb3, n_best)
        hbl3 = binary_local_search(hb3, n_best, 200)
        try_update(hbl3, n_best)

    # Phase 6: Alternative resolutions
    for n_try in [240, 480]:
        if budget_s - elapsed() < 100:
            break
        xo = np.arange(n_best) * (2.0 / n_best)
        xn = np.arange(n_try) * (2.0 / n_try)
        ht = np.interp(xn, xo, h_best, period=2.0)
        ht = fix_sum(ht, n_try)
        hg2 = grad_ascent_fft(ht, n_try, 0.05, 0.001, 0.3, 3000)
        try_update(hg2, n_try)
        hb4 = (hg2 > 0.5).astype(float)
        hb4 = fix_sum(hb4, n_try)
        hbl4 = binary_local_search(hb4, n_try, 150)
        try_update(hbl4, n_try)

    # Phase 7: Final SLSQP + binary polish
    if budget_s - elapsed() > 60:
        try_update(do_slsqp(h_best, n_best, 0.001, 3000), n_best)
    if budget_s - elapsed() > 30:
        hb5 = (h_best > 0.5).astype(float)
        hb5 = fix_sum(hb5, n_best)
        hbl5 = binary_local_search(hb5, n_best, 300)
        try_update(hbl5, n_best)

    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
