    """Max circular correlation for period vector a of length p"""
    max_c = 0.0
    for k in range(p):
        shifted = np.roll(a, -k)
        c = np.dot(shifted, 1.0 - a)
        if c > max_c:
            max_c = c
    return max_c

def periodic_objective(params, p):
    a = np.clip(params, 0, 1)
    return circ_max_corr(a, p) * 2.0 / p

def sum_eq_p(x, p):
    return np.sum(x) - p * 0.5

def solve_periodic(p, n_total, seed=42):
    """Solve periodic subproblem, return full h vector"""
    a0 = np.full(p, 0.5)
    rng = np.random.RandomState(seed)
    best_a = a0.copy()
    best_c = periodic_objective(a0, p)
    
    # Multiple starts
    for start_i in range(20):
        if start_i == 0:
            a_try = a0.copy()
        else:
            a_try = np.clip(a0 + rng.randn(p) * 0.15, 0, 1)
            a_try = np.clip(a_try - np.sum(a_try) + p/2, 0, 1)  # rough centering
            # Proper projection
            a_try = (a_try - np.mean(a_try)) / max(np.std(a_try), 1e-10) * 0.1 + 0.5
            a_try = np.clip(a_try, 0, 1)
        
        cons = [{'type': 'eq', 'fun': lambda x, p=p: np.sum(x) - p*0.5,
                 'jac': lambda x, p=p: np.ones(p)}]
        res = opt.minimize(periodic_objective, a_try, args=(p,),
                          method='SLSQP',
                          bounds=[(0,1)]*p,
                          constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-16})
        a_res = np.clip(res.x, 0, 1)
        # Project to sum constraint
        a_res = a_res + (p*0.5 - np.sum(a_res)) / p
        a_res = np.clip(a_res, 0, 1)
        c = periodic_objective(a_res, p)
        if c < best_c:
            best_c = c
            best_a = a_res.copy()
    
    # Expand to full length
    m = n_total // p
    h_full = np.tile(best_a, m)
    if len(h_full) < n_total:
        h_full = np.concatenate([h_full, best_a[:n_total - len(h_full)]])
    else:
        h_full = h_full[:n_total]
    h_full = proj_constraint(h_full, n_total)
    return h_full, compute_c5(h_full)

def do_slsqp_softmax(h0, n, T, max_iter):
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        if T < 1e-15:
            return c_max * 2.0/n
        shifted = corr - c_max
        return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * 2.0/n
    
    def cons_fun(h):
        return np.sum(h) - n*0.5
    
    def cons_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = opt.minimize(obj, h0, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return proj_constraint(res.x, n)

Actually, for k≠0: the correlation at lag k is:

If i is in the range [0, n-1-k): h[i] appears as h[i](1-h[i-k]) → linear in h[i] with coefficient -(1-h[i-k])
If i is in the range [k, n-1]: h[i] appears as h[i-k](1-h[i]) → linear in h[i] with coefficient -h[i-k]

So for each k, the contribution is linear in h[i]. The max of linear functions is piecewise-linear and convex. We can minimize it by ternary search or by finding where the two active constraints cross.
