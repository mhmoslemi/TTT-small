import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def sm_obj(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    S = np.sum(np.exp((corr - m) / T))
    return (m + T * np.log(S)) * dx + mu * (np.sum(h) - n * 0.5) ** 2

def sm_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp((corr - m) / T)
    S = np.sum(exps)
    w = exps / S
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    for kk in range(1, n):
        wp = w[n - 1 + kk]
        wn = w[n - 1 - kk]
        if wp > 1e-12:
            grad[kk:] += wp * (1.0 - h[:n - kk])
            grad[:n - kk] -= wp * h[kk:]
        if wn > 1e-12:
            grad[:n - kk] += wn * (1.0 - h[kk:])
            grad[kk:] -= wn * h[:n - kk]
    grad *= dx
    grad += 2.0 * mu * (np.sum(h) - n * 0.5)
    return grad

def lbgfb_opt(h0, T, mu, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = opt.fmin_l_bfgs_b(sm_obj, h0.astype(float), fprime=sm_grad, args=(T, mu),
                            bounds=bounds, maxiter=maxiter, pgtol=1e-14, iprint=-1)
    return proj_constraint(res, n)
