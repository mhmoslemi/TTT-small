            total = room.sum()
            if total < 1e-14:

            total = room.sum()
            if total < 1e-14:
                break
            frac = min(-diff / total, 1.0)
            h = np.clip(h - room * frac, 0, 1)

def soft_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = T * (corr - m)
    ex -= np.max(ex)
    sw = np.sum(np.exp(ex))
    val = (m + np.log(sw) / T) * dx
    weights = np.exp(ex) / sw
    
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for k in range(len(shifts)):
        w = weights[k]
        if w < 1e-14:
            continue
        s = shifts[k]
        # First term: (1-h[j-s]) for valid j-s
        js = max(0, s)
        je = min(n, s + n)
        if js < je:
            grad[js:je] += w * (1.0 - h[js - s:je - s])
        # Second term: -h[j+s] for valid j+s
        js2 = max(0, -s)
        je2 = min(n, n - s)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 + s:je2 + s]
    grad *= dx
    return val, grad

def adam_optimize(h, T_schedule, n_steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    best_val = compute_c5(h)
    best_h = h.copy()
    
    t_total = len(T_schedule)
    for step in range(n_steps):
        # Interpolate T
        frac = step / max(1, n_steps - 1)
        idx = frac * (len(T_schedule) - 1)
        i0 = int(idx)
        i1 = min(i0 + 1, len(T_schedule) - 1)
        alpha = idx - i0
        T = T_schedule[i0] * (1 - alpha) + T_schedule[i1] * alpha
        
        val, grad = soft_c5_and_grad(h, T)
        # Normalize gradient
        g = grad / (np.linalg.norm(grad) + 1e-12)
        
        m_adam = beta1 * m_adam + (1 - beta1) * g
        v_adam = beta2 * v_adam + (1 - beta2) * (g * g)
        m_hat = m_adam / (1 - beta1 ** (step + 1))
        v_hat = v_adam / (1 - beta2 ** (step + 1))

        h_new = project_feasible(h_new, n)
        
        # Check actual objective

        if actual < best_val:

            best_h = h_new.copy()

def slsqp_final(h, maxiter=300):

        return np.max(corr)

        js2 = max(0, -s)
        je2 = min(n, n - s)
        g[js2:je2] -= x[js2 + s:je2 + s]
        return g * dx
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 
             'jac': lambda x: np.ones(n)}]
    try:

        return project_feasible(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h + room * min(diff / total, 1.0), 0, 1)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h - room * min(-diff / total, 1.0), 0, 1)
    return h

def soft_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = T * (corr - m)
    ex -= np.max(ex)
    sw = np.sum(np.exp(ex))
    val = (m + np.log(sw + 1e-300) / T) * dx
    weights = np.exp(ex) / sw
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for k in range(len(shifts)):
        w = weights[k]
        if w < 5e-15:
            continue
        s = shifts[k]
        js1 = max(0, s)
        je1 = min(n, s + n)
        if js1 < je1:
            grad[js1:je1] += w * (1.0 - h[js1 - s:je1 - s])
        js2 = max(0, -s)
        je2 = min(n, n - s)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 + s:je2 + s]
    grad *= dx
    return val, grad

def adam_run(h, T_sched, steps, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    best_val = compute_c5(h)
    best_h = h.copy()
    for step in range(steps):
        frac = step / max(1, steps - 1)
        idx = frac * (len(T_sched) - 1)
        i0 = int(np.clip(idx, 0, len(T_sched) - 1))
        i1 = min(i0 + 1, len(T_sched) - 1)
        a = idx - i0
        T = T_sched[i0] * (1 - a) + T_sched[i1] * a
        _, grad = soft_c5_grad(h, T)
        m_a = beta1 * m_a + (1 - beta1) * grad
        v_a = beta2 * v_a + (1 - beta2) * (grad * grad)
        m_hat = m_a / (1 - beta1 ** (step + 1))
        v_hat = v_a / (1 - beta2 ** (step + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        actual = compute_c5(h_new)
        if actual < best_val:
            best_val = actual
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

    def obj_grad(x):
        _, g = soft_c5_grad(x, T)
        return g

    bnds = [(0.0, 1.0)] * n
    ones_n = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: ones_n}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
