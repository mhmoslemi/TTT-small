    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = (corr - c_max) / T
    LogSumExp = np.log(np.sum(np.exp(shifted)))
    return (c_max + T * LogSumExp) * dx

def soft_max_obj_2(h, T, alpha):
    """Weighted soft-max that emphasizes top peaks more"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = (corr - c_max) / T
    w = np.exp(alpha * (corr - c_max) / T)
    LogSumExp = np.log(np.sum(w * np.exp(shifted)))
    return (c_max + T * LogSumExp) * dx

def proj_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n, dtype=float)
    x_new = np.arange(new_n, dtype=float) * old_n / new_n
    h_new = np.interp(x_new, x_old, h)
    return proj_feasible(h_new, new_n)

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def do_slsqp(h0, n, T, max_iter, ftol=1e-16):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0}
    )
    return proj_feasible(res.x, n)

def do_lbfgs(h0, n, mu, max_iter=1000):
    def obj_pen(x):
        c = soft_max_obj(x, 0.001)
        pen = mu * (np.sum(x) - n * 0.5) ** 2
        return c + pen
    res = opt.minimize(obj_pen, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-14, 'iprint': 0})
    return proj_feasible(res.x, n)

def coord_descent(h, n, n_sweeps, seed):
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            h_try = h_cur.copy()
            best_val = h_cur[i]
            best_c = c_cur
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                h_try[i] = np.clip(h_cur[i] + delta, 0, 1)
                h_try = proj_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c:
                    best_c = c_try
                    best_val = h_try[i]
                    best_h = h_try.copy()
            if best_c < c_cur:
                h_cur = best_h
                c_cur = best_c
                if c_cur < c_best:
                    c_best = c_cur
                    h_best = h_cur.copy()
    return h_best, c_best

def perturbed_restart(h_best, n, c5_best, rng, T_list, max_iter):
    for i, T in enumerate(T_list):
        sigma = 0.005 + 0.004 * (i % 3)
        h_pert = np.clip(h_best + rng.randn(n) * sigma, 0, 1)
        h_pert = proj_feasible(h_pert, n)
        h_try = do_slsqp(h_pert, n, T, max_iter)
        c_try = compute_c5(h_try)
        if c_try < c5_best:
            h_best = h_try
            c5_best = c_try
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Upsample to n=504 for more flexibility
    n_up = 504
    h_work = upsample_to(h_best, n_up)
    c_work = compute_c5(h_work)

    # SLSQP with careful T annealing at n=504
    T_schedule = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    it_schedule = [300, 400, 500, 600, 700, 800, 1000, 1200]

    for T, mi in zip(T_schedule, it_schedule):
        elapsed = time.time() - t_start
        rem = budget_s - elapsed
        if rem < 150:
            break
        mi_use = min(mi, int(rem / 0.4))
        h_work = do_slsqp(h_work, n_up, T, max(100, mi_use))
        c_work = compute_c5(h_work)
        if c_work < c5_best:
            h_best = h_work.copy()
            c5_best = c_work
            n_best = n_up
            h_work = h_best.copy()

    # Phase 2: L-BFGS-B with penalty at n=504
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        for mu in [5, 20, 100, 500]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 50:
                break
            h_try = do_lbfgs(h_best.copy(), n_best, mu, 800)
            c_try = compute_c5(h_try)
            if c_try < c5_best:
                h_best = h_try
                c5_best = c_try

    # Phase 3: Try at n=448 as alternative
    elapsed = time.time() - t_start
    if budget_s - elapsed > 150:
        n_alt = 448
        h_alt = upsample_to(h_best, n_alt)
        for T in [0.01, 0.004, 0.001]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 60:
                break
            h_alt = do_slsqp(h_alt, n_alt, T, 800)
            c_alt = compute_c5(h_alt)
            if c_alt < c5_best:
                h_best = h_alt
                c5_best = c_alt
                n_best = n_alt

    # Phase 4: Multi-start perturbations
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        rng = np.random.RandomState(seed + 7777)
        h_best, c5_best = perturbed_restart(
            h_best, n_best, c5_best, rng,
            [0.008, 0.004, 0.002], 600
        )

    # Phase 5: Coordinate descent (if time allows)
    elapsed = time.time() - t_start
    if budget_s - elapsed > 100:
        h_cd, c_cd = coord_descent(h_best, n_best, 40, seed + 12345)
        if c_cd < c5_best:
            h_best = h_cd
            c5_best = c_cd

    # Phase 6: Final SLSQP polish with very small T
    elapsed = time.time() - t_start
    if budget_s - elapsed > 50:
        for T in [0.0003, 0.0001, 0.00005]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 25:
                break
            h_try = do_slsqp(h_best.copy(), n_best, T, 2500, 1e-18)
            c_try = compute_c5(h_try)
            if c_try < c5_best:
                h_best = h_try
                c5_best = c_try

    # Phase 7: One more perturbation + optimize cycle
    elapsed = time.time() - t_start
    if budget_s - elapsed > 60:
        rng2 = np.random.RandomState(seed + 31337)
        h_p2 = np.clip(h_best + rng2.randn(n_best) * 0.003, 0, 1)
        h_p2 = proj_feasible(h_p2, n_best)
        h_try = do_slsqp(h_p2, n_best, 0.001, 1500)
        c_try = compute_c5(h_try)
        if c_try < c5_best:
            h_best = h_try
            c5_best = c_try

    # Final projection and evaluation
    h_best = proj_feasible(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
