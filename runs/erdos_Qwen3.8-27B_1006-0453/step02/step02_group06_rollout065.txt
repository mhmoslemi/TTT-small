import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def block_descent(h, n, n_blocks, t_start, budget_s, max_time_per_block=5.0):
    """Block coordinate descent"""
    h = h.copy()
    block_size = n // n_blocks
    c_cur = compute_c5(h)
    
    for iteration in range(200):
        if time.time() - t_start > budget_s - 60:
            break
        improved = False
        for b in range(n_blocks):
            start = b * block_size
            end = (b + 1) * block_size if b < n_blocks - 1 else n
            size = end - start
            
            h_block = h[start:end].copy()
            
            def obj_block(x):
                h_temp = h.copy()
                h_temp[start:end] = x
                return soft_max_obj(h_temp, n, 0.005)
            
            res = opt.minimize(
                obj_block, h_block,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * size,
                options={'maxiter': 200, 'ftol': 1e-14}
            )
            
            h[start:end] = res.x
            h = proj_constraint(h, n)
            
            if time.time() - t_start > budget_s - 60:
                return h, c_cur
        
        c_new = compute_c5(h)
        if c_new < c_cur - 1e-10:
            c_cur = c_new
            improved = True
        elif not improved:
            # Try next block arrangement
            pass
    
    return h, c_cur

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_value_and_weights(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    exp_shifted = np.exp(shifted / T)
    weights = exp_shifted / np.sum(exp_shifted)
    value = (c_max + T * np.log(np.sum(exp_shifted))) * (2.0 / n)
    return value, weights, corr

def compute_gradient(h, weights, n):
    """Compute gradient of soft-max objective w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    nc = 2 * n - 1  # length of correlation
    # weights has length nc, indexed 0..nc-1
    # lag for index c is c - (n-1)
    # For lag k (positive or negative), the gradient contribution to h[i] is:
    #   sign part 1: if |k| <= i <= n-1: (1 - h[i - |k|])
    #   sign part 2: if 0 <= i <= n-1-|k|: -h[i + |k|]
    
    for c in range(nc):
        k = abs(c - (n - 1))
        w = weights[c]
        if w < 1e-15:

        # Part 1: (1 - h[i-k]) for i in [k, n-1]
        if k == 0:

            grad[0:n-k] += w * (-h[k:n])  # -h[i+k] for i in [0, n-1-k] = [0, n-1]
        else:
            grad[k:n] += w * (1.0 - h[0:n-k])
            grad[0:n-k] += w * (-h[k:n])
    return grad * dx

def projected_gradient_hessian_free(h, n, T, t_start, budget_s, max_iters=5000):
    """Projected gradient method with backtracking"""
    h_cur = h.copy()
    val_cur, weights_cur, _ = soft_max_value_and_weights(h_cur, T)
    val_best = val_cur
    h_best = h_cur.copy()
    lr = 0.01
    
    for it in range(max_iters):
        if time.time() - t_start > budget_s - 30:
            break
        grad = compute_gradient(h_cur, weights_cur, n)
        
        # Backtracking line search
        step = lr
        for _ in range(30):
            h_try = proj_constraint(h_cur - step * grad, n)
            val_try, weights_try, _ = soft_max_value_and_weights(h_try, T)
            if val_try <= val_cur - 1e-4 * step * np.dot(grad, h_try - h_cur):
                break
            step *= 0.5
        
        h_cur = h_try
        val_cur = val_try
        weights_cur = weights_try
        
        if val_cur < val_best:
            val_best = val_cur
            h_best = h_cur.copy()
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.95
    
    return h_best, val_best

def slsqp_refine(h0, n, T, max_iter=500):
    """SLSQP refinement"""
    def obj(h):
        n_local = len(h)
        dx = 2.0 / n_local
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = corr - c_max
        return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n*0.5, 'jac': lambda x: np.ones(n)}]
    res = opt.minimize(
        obj, h0, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

    A = np.convolve(W, 1.0 - h, mode='full')[:n]
    B = np.correlate(h, W, mode='full')[n - 1:]
    return dx * (A - B)

def proj_grad(h0, n, T, t_start, t_end):
    h = h0.copy()
    c_best = compute_c5(h)
    h_best = h.copy()
    w, _, _ = get_corr_weights(h, T)
    g = compute_grad(h, w, n)
    gn = np.linalg.norm(g)
    lr = 1.0 / (gn * gn + 1e-6) if gn > 1e-10 else 0.01
    momentum = 0.0
    h_prev = h.copy()
    for it in range(20000):
        if time.time() > t_end:
            break
        h_y = h + momentum * (h - h_prev)
        h_prev = h.copy()
        h_try = proj_constraint(h_y - lr * g, n)
        c_try = compute_c5(h_try)
        if c_try < c_best:
            c_best = c_try
            h_best = h_try.copy()
        h = h_try
        w, _, _ = get_corr_weights(h, T)
        g = compute_grad(h, w, n)
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            lr = min(lr * 1.02, 1.0 / (gn * gn + 1e-6))
        momentum = 0.9 * momentum + 0.05
        if it % 500 == 0:
            lr *= 0.95
    return h_best, c_best

def slsqp_softmax(h0, n, T, max_iter):
    def obj(h):
        nl = len(h)
        dx = 2.0 / nl
        corr = np.correlate(h, 1.0 - h, mode='full')
        cm = np.max(corr)
        sh = corr - cm
        return (cm + T * np.log(np.sum(np.exp(sh / T)))) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5, 'jac': lambda x: np.ones(n)}]
    res = opt.minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    return proj_constraint(res.x, n)

def slsqp_toplevels(h0, n, M, max_iter):
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    top_idx = np.argsort(-corr)[:M]
    def obj(h):
        nl = len(h)
        dx = 2.0 / nl
        c = np.correlate(h, 1.0 - h, mode='full')
        return np.sum(c[top_idx] ** 2) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5, 'jac': lambda x: np.ones(n)}]
    res = opt.minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    return proj_constraint(res.x, n)

def upsample_h(h, new_n):
    old_n = len(h)
    xo = np.linspace(0, 2, old_n, endpoint=False)
    xn = np.linspace(0, 2, new_n, endpoint=False)
    return proj_constraint(np.interp(xn, xo, h), new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: PG with moderate T for smooth convergence
    for T in [0.04, 0.015, 0.006, 0.002]:
        rem = budget_s - (time.time() - t0)
        if rem < 90:
            break
        te = t0 + rem * 0.65
        h_new, cn = proj_grad(h_best, n_best, T, t0, te)
        if cn < c5_best:
            h_best, c5_best = h_new, cn

    # Phase 2: Top-levels focused SLSQP (smooth quadratic)
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        for M in [4, 8, 12]:
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            h_opt = slsqp_toplevels(h_best, n_best, M, 600)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best = h_opt, c

    # Phase 3: SLSQP soft-max refinement
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        for T in [0.003, 0.001, 0.0004]:
            rem = budget_s - (time.time() - t0)
            if rem < 35:
                break
            h_opt = slsqp_softmax(h_best, n_best, T, 800)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best = h_opt, c

    # Phase 4: PG at small T for fine-tuning
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        for T in [0.001, 0.0003]:
            rem = budget_s - (time.time() - t0)
            if rem < 30:
                break
            te = t0 + rem * 0.5
            h_pg, cn = proj_grad(h_best, n_best, T, t0, te)
            if cn < c5_best:
                h_best, c5_best = h_pg, cn

    # Phase 5: Resolution exploration
    rem = budget_s - (time.time() - t0)
    for nn in [n_best - 32, n_best + 32, n_best + 64]:
        if nn < 100 or nn == n_best:
            continue
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        hc = upsample_h(h_best, nn)
        rem2 = budget_s - (time.time() - t0)
        hpg, cn = proj_grad(hc, nn, 0.008, t0, t0 + rem2 * 0.5)
        c = compute_c5(hpg)
        if c < c5_best:
            h_best, c5_best, n_best = hpg, c, nn

    # Phase 6: Block perturbation + local re-opt
    rem = budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed + 31337)
    attempts = 0
    while rem > 25 and attempts < 60:
        attempts += 1
        blk = rng.randint(8, min(40, n_best // 4))
        st = rng.randint(0, max(1, n_best - blk))
        sig = 0.015 + 0.01 * (attempts % 3)
        ht = h_best.copy()
        ht[st:st + blk] += rng.randn(blk) * sig
        ht = proj_constraint(ht, n_best)
        if budget_s - (time.time() - t0) < 15:
            break
        ho = slsqp_softmax(ht, n_best, 0.002, 350)
        c = compute_c5(ho)
        if c < c5_best:
            h_best, c5_best = ho, c

    # Phase 7: Final ultra-fine
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        ho = slsqp_softmax(h_best, n_best, 0.0002, 2500)
        c = compute_c5(ho)
        if c < c5_best:
            h_best, c5_best = ho, c
    if budget_s - (time.time() - t0) > 10:
        hpg, cn = proj_grad(h_best, n_best, 0.0002, t0, t0 + (budget_s - (time.time() - t0)) * 0.7)
        if cn < c5_best:
            h_best, c5_best = hpg, cn

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
