        )
        if c_sg < c5_best:
            h_best = h_sg
            c5_best = c_sg
        # Second pass with fewer top lags
        h_sg2, c_sg2 = subgradient_descent(
            h_best.copy(), n_best, t_start,
            t_start + budget_s * 0.70,
            5000, 0.02, 0.998, 1
        )
        if c_sg2 < c5_best:
            h_best = h_sg2
            c5_best = c_sg2

    # Phase 4: Multi-start SLSQP with top-m objective
    if time.time() < t_start + budget_s * 0.85:
        for i in range(8):
            if time.time() > t_start + budget_s * 0.85:
                break
            rng = np.random.RandomState(seed + i * 104729 + 13)
            sigma = 0.008 + 0.006 * (i % 3)
            h_pert = np.clip(h_best + rng.randn(n_best) * sigma, 0, 1)
            h_pert = proj_constraint(h_pert, n_best)
            T_p = 0.003
            h_opt = do_slsqp(h_pert, n_best, T_p, 600, True, 5)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 5: 1D local search
    if time.time() < t_start + budget_s * 0.92:
        h_ls, c_ls = local_search_1d(
            h_best.copy(), n_best, t_start,
            t_start + budget_s * 0.92, 30
        )
        if c_ls < c5_best:
            h_best = h_ls
            c5_best = c_ls

    # Phase 6: SA with very low temperature
    if time.time() < t_start + budget_s * 0.96:
        h_sa, c_sa = sa_refine(
            h_best, n_best, t_start,
            t_start + budget_s * 0.96,
            2000, seed + 55555, 0.0002, 0.998, 0.015
        )
        if c_sa < c5_best:
            h_best = h_sa
            c5_best = c_sa

    # Phase 7: Final SLSQP polish
    if time.time() < t_start + budget_s - 15:
        for T_final in [0.0003, 0.0001]:
            if time.time() > t_start + budget_s - 10:
                break
            h_opt = do_slsqp(h_best.copy(), n_best, T_final, 2000, False, 1)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Final subgradient micro-tuning
    if time.time() < t_start + budget_s - 5:
        h_sg3, c_sg3 = subgradient_descent(
            h_best.copy(), n_best, t_start,
            t_start + budget_s - 3,
            3000, 0.01, 0.999, 1
        )
        if c_sg3 < c5_best:
            h_best = h_sg3
            c5_best = c_sg3

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
