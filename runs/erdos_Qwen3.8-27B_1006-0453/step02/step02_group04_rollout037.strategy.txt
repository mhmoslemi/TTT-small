### Constraints
1. $0 \leq \phi_i \leq 1 \forall i$
2. $\sum_{i=0}^{n-1} \phi_i = n/2$

### Optimization
Use projected gradient descent with adaptive step sizes. The objective function gradient is computed via finite differences with temperature-smoothing to prevent gradient explosion.

## Implementation Procedure

### 1. Parameter Validation and Initialization

```python
def fix_sum(h, n):
    """Clip h to [0,1] and adjust sum to n/2"""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    target = n * 0.5
    diff = target - total
    if abs(diff) < 1e-8:
        return h
    
    # Space creation
    space = (1 - 1e-4) * (np.logical_and(h < 1e-4, h > 1 - 1e-4))
    n_space = np.sum(space)
    
    if abs(diff) < n_space * 0.4:
        if diff > 0:
            add_per = diff / n_space
            h[space] += add_per
        else:
            subtract_per = diff / n_space
            h[space] -= subtract_per
    else:
        # Binary adjustment
        if diff > 0:
            # Increase bounds
            idx = np.searchsorted(sorted(h), 0.5)
            h[idx] = 0.8
        else:
            # Decrease bounds
            idx = np.searchsorted(sorted(h), 0.5)
            h[idx] = 0.2
    
    return np.clip(h, 0, 1)
```

### 2. Temperature-Annealed Objective Function

```python
def compute_c5(h, T=0.1, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    h_smooth = np.minimum(1, np.maximum(0, h + np.random.normal(0, T, n)))
    corr = np.correlate(h_smooth, 1.0 - h_smooth, mode='full')
    return float(np.max(corr)) * dx
```

### 3. Geometric Optimization with Cross-Validation

```python
def nash_optimize(h, n, T=0.1, max_iter=1000):
    # Convert h to geometric form
    x = np.linspace(0, 2, n)
    freqs = np.fft.fftfreq(n) * 2 * np.pi
    
    # Bases (Fourier at low frequencies)
    basis = np.zeros((n, len(freqs)))
    for i, f in enumerate(freqs):
        basis[:, i] = np.exp(1j * f * x)
    
    # Project h onto basis
    coeffs = np.real(np.fft.fft(h))
    
    # Moreau envelope
    def obj(theta):
        # Reconstruct h from optimized coefficients
        h_proj = np.real(np.fft.ifft(coeffs))
        h_smooth = fix_sum(h_proj, n)
        
        # Edge penalty
        energy = np.sum(np.log(np.abs(coeffs + 1e-4)) * freqs**2)
        
        # Temperature-scheduled overlap
        return compute_c5(h_smooth, T)
    
    # Nonlinear inversion
    from scipy.optimize import minimize
    res = minimize(obj, np.zeros(n-1), method='CG', tol=1e-4)
    
    if res.success:
        coeffs_opt = res.x * 0.9 + 0.1 * coeffs
        h_opt = fix_sum(np.real(np.fft.ifft(coeffs_opt)), n)
        return h_opt
    return h
```

### 4. Cross-Resolution Comparison

```python
def compare_candidates(h, n1, h_ref=None, budget=100):
    """Evaluate candidate across resolutions (n1) and against reference h_ref"""
    # Base evaluation
    h0 = fix_sum(h, n1)
    c5_value = compute_c5(h0)
    
    # Cross-check other resolutions [-40%, +40%]
    n_grid = [int(max(30, n1 * 0.6)), int(n1), int(min(1000, n1 * 1.4))]
    c5s = [c5_value]
    for n_test in n_grid[1:]:
        h_test = fix_sum(h0 if n_test==n1 else downsample(h0, n_test), n_test)
        c5s.append(compute_c5(h_test))
        
    # Against reference if provided
    improvement = None
    if h_ref is not None:
        c5_ref = compute_c5(h_ref, budget=50)
        improvement = c5_value < c5_ref - 1e-8
        
    # Weighted average of similar-population C5 values
    return {
        'c5_aware': np.average(c5s, weights=[1, 0.9, 0.9]),
        'new_best': improvement
    }
```

### 5. Main Optimization Loop

```python
def optimize_phase(h_init, n, budget=300):
    """Primary optimization phase with parallel validation"""
    # Step 1: Temperature annealing
    T_min, T_max = 0.1, 0.5
    T_cur = 0.5
    t0 = time.time()
    best_h = fix_sum(h_init, n)
    best_c5 = compute_c5(best_h)
    
    # Geometric embedding
    h_geom = fix_sum(best_h, n)
    
    for i in range(min(150, int(budget/20))):
        T_cur *= 0.99
            
        # Gradient-based update with restarts
        def temperature_objective(x): return compute_c5(x, T_cur)
        h_temp = fix_sum(np.clip(optimizer(temperature_objective, h_geom), n)
        
        c5_temp = compute_c5(h_temp)
        if c5_temp < best_c5 - 1e-8:
            best_h = h_temp
            best_c5 = c5_temp
            
        # Cross-validation at coarser grid
        if (time.time() - t0) > budget * 0.3:
            cv_result = compare_candidates(best_h, n)
            if cv_result['new_best']:
                return best_h, True
                
    return best_h, False
```

3. **Local Optimization Methods**:
