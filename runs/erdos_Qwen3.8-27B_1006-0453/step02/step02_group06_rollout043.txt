import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += d / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    lse = cmax + T * np.log(np.sum(np.exp((corr - cmax) / T)))
    return lse * dx + mu * (np.sum(x) - 0.5 * n) ** 2

def lse_gradient(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    m = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    w = np.exp((corr - cmax) / T)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            ji = np.arange(lo1, hi1 + 1) + k - m
            grad[lo1:hi1 + 1] -= wk * x[ji]
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k)
        if lo2 <= hi2:
            ji = np.arange(lo2, hi2 + 1) - k + m
            grad[lo2:hi2 + 1] -= wk * x[ji]
    grad *= dx
    if mu > 0:
        grad += 2.0 * mu * (np.sum(x) - 0.5 * n)
    return grad

def upsample(h, new_n):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return proj_constraint(np.interp(x_new, x_old, h), new_n)

def do_lbfgsb(h0, n, T, mu, maxiter, t_start, budget_s):

        return lse_objective(x, T, mu)
    def g(x):
        return lse_gradient(x, T, mu)

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                                'maxfun': maxiter * 2})
    return proj_constraint(res.x, n)

```python
def lse_gradient(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    m = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    w = np.exp((corr - cmax) / T)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        # R1: max(0, n-1-k) to min(n-1, 2n-2-k)
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            j = idx + k - m
            grad[lo1:hi1+1] += wk * (1.0 - x[j])
        # R3: max(0, k-n+1) to min(n-1, k)
        lo3 = max(0, k - n + 1)
        hi3 = min(n - 1, k)
        if lo3 <= hi3:
            idx = np.arange(lo3, hi3 + 1)
            j = idx - k + m
            grad[lo3:hi3+1] -= wk * x[j]
    grad *= dx
    if mu > 0:
        grad += 2.0 * mu * (np.sum(x) - 0.5 * n)
    return grad
```
