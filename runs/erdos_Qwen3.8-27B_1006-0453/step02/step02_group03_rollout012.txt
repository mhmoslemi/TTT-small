import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(120):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def sgrad(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    cm = cr.max()
    w = np.exp(beta * (cr - cm))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        kl1 = max(-(n-1), j-(n-1))
        kh1 = min(n-1, j)
        if kl1 <= kh1:
            k = np.arange(kl1, kh1+1)
            g[j] += np.dot(w[o+k], 1.0 - h[j-k])
        kl2 = max(-(n-1), -j)
        kh2 = min(n-1, n-1-j)
        if kl2 <= kh2:
            k = np.arange(kl2, kh2+1)
            g[j] -= np.dot(w[o+k], h[j+k])
    return g * dx

def slsqp(h0, dx, ts, mit=300):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0, 1)]*n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-14})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def gdes(h0, dx, ts, dl, rng):
    n = len(h0)
    h, bh = h0.copy(), h0.copy()
    bc = compute_c5(h0, dx)
    st = 0
    iters = 0
    while time.time() < dl:
        iters += 1
        b = 8.0 + 250.0 * min(1.0, iters / 3000.0)
        lr = 0.06 / (1.0 + 0.0008 * iters)
        g = sgrad(h, dx, b)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.12, 0.05, 0.02):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                bh, bc, h, st = h2.copy(), c2, h2, 0
                ok = True
                break
        if not ok:
            st += 1
            if st > 300:
                break
    return bh, bc

def sa_opt(h0, dx, ts, dl, rng):
    n = len(h0)
    h = h0.copy()
    bh = h0.copy()
    bc = compute_c5(h0, dx)
    cur_c = bc
    T0, T1 = 0.05, 0.0004
    no_imp = 0
    i = 0
    while time.time() < dl:
        i += 1
        frac = min(1.0, i / 6000.0)
        T = T0 * (T1 / T0) ** frac
        h_old = h.copy()

            j = rng.randint(n)
            h[j] += rng.uniform(-0.04, 0.04)
        elif mt < 0.60:
            i1, i2 = rng.randint(n), rng.randint(n)

                e = rng.uniform(-0.03, 0.03)
                h[i1] += e
                h[i2] -= e
        elif mt < 0.80:

        else:
            h[rng.randint(n)] = rng.uniform(0, 1)
        h = proj(h, ts)
        c = compute_c5(h, dx)
        if c < cur_c:
            cur_c = c
            if c < bc:
                bh = h.copy()
                bc = c
            no_imp = 0
        elif rng.random() < np.exp(-(c - cur_c) / max(T, 1e-12)):
            cur_c = c
            no_imp += 1
        else:
            h = h_old
            no_imp += 1
        if no_imp > 5000 and T < 0.003:
            break
    return bh, bc

def targeted_fix(h, dx, ts, dl, rng):
    """Identify worst shifts and do targeted reductions."""
    n = len(h)
    bh = h.copy()
    bc = compute_c5(h0 := h, dx)
    while time.time() < dl:
        cr = np.correlate(h, 1.0 - h, mode='full')
        kstar = int(np.argmax(cr))
        o = n - 1
        k = kstar - o
        improved = False
        for _ in range(200):
            best_j, best_c = -1, bc
            for j in rng.choice(n, size=min(20, n), replace=False):
                if j + k < 0 or j + k >= n:
                    continue
                ov = h[j] * (1.0 - h[j+k])
                if ov < 1e-6:
                    continue
                for dval in (0.02, 0.04, 0.01):
                    for tgt in (0, 1):
                        h2 = h.copy()
                        if tgt == 0:
                            h2[j] -= dval
                        else:
                            h2[j+k] += dval
                        h2 = proj(h2, ts)
                        c2 = compute_c5(h2, dx)
                        if c2 < best_c - 1e-15:
                            best_c, best_j = c2, (j, tgt, dval)
                            improved = True
            if not best_j or not isinstance(best_j, tuple):
                break
            j, tgt, dval = best_j
            h.copy()
            if tgt == 0:
                h[j] -= dval
            else:
                h[j+k] += dval
            h = proj(h, ts)
            if compute_c5(h, dx) <= bc:
                bc = compute_c5(h, dx)
                bh = h.copy()
                break
        if not improved:
            break
    return bh, bc
