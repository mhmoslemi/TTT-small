                                            tol=1e-5, mutation=(0.5, 1.0),
                                            recombination=0.7,
                                            updating='deferred', polish=False,
                                            init='latinhypercube')
            h_lo = project_feasible(np.clip(de_lo.x, 0, 1), n_lo)
            x_old = np.linspace(0, 2, n_lo, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_up = np.interp(x_new, x_old, h_lo)
            h_up = project_feasible(h_up, n)
            for T, tk in [(200, 6), (1000, 10), (4000, 15)]:
                if time.time() - t0 > budget_s * 0.48:
                    break
                hc = slsqp_opt(h_up, T, tk, maxiter=500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass

    # Phase 3: Basin hopping with decaying perturbation
    if time.time() - t0 < budget_s * 0.72:
        bh_h = best_h.copy()
        bh_c5 = best_c5
        n_iter_bh = 50
        T_bh_list = [300, 800, 2000]
        for it in range(n_iter_bh):
            if time.time() - t0 > budget_s * 0.72:
                break
            T_bh = T_bh_list[it % 3]
            step = 0.06 * (1.0 - 0.6 * it / n_iter_bh) + 0.015
            h_pert = bh_h + step * np.random.randn(n)
            h_pert = project_feasible(h_pert, n)
            try:
                hc = slsqp_opt(h_pert, T_bh, 8, maxiter=350)
                c5v = compute_c5(hc)
                temp = 0.004 * (1.0 - it / n_iter_bh) + 0.0002
                if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-10)):
                    bh_c5 = c5v
                    bh_h = hc
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
            except Exception:
                pass

    # Phase 4: Targeted coordinate descent on worst shifts
    if time.time() - t0 < budget_s * 0.88:
        n_cur = len(best_h)
        stall_count = 0
        for iteration in range(300):
            if time.time() - t0 > budget_s * 0.88:
                break
            improved = False
            # Focus perturbation on indices that affect worst shifts
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top_k_worst = 12
            top_idx = np.argpartition(corr, -top_k_worst)[-top_k_worst:]
            n_try = min(25, n_cur)
            indices = np.random.choice(n_cur, size=n_try, replace=False)
            for j in indices:
                for delta in [0.015, 0.03, 0.05, 0.08, -0.015, -0.03, -0.05, -0.08]:
                    h_try = best_h.copy()
                    h_try[j] = float(np.clip(h_try[j] + delta, 0, 1))
                    h_try = project_feasible(h_try, n_cur)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 60:
                break

    # Phase 5: Final high-T SLSQP polish (very tight)
    n_cur = len(best_h)
    final_sched = [(3000, 10), (10000, 15), (30000, 20), (80000, 25)]
    for T, tk in final_sched:
        if time.time() - t0 > budget_s * 0.985:
            break
        try:
            hc = slsqp_opt(best_h, T, tk, maxiter=2000, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
