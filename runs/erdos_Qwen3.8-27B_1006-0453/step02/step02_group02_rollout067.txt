def get_corr_weights(h, t, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    nf = len(corr)
    if topk < nf:
        idx = np.argpartition(corr, -topk)[-topk:]
        cv = corr[idx]
    else:
        idx = np.arange(nf)
        cv = corr
    mx = np.max(cv)
    s = t * (cv - mx)
    s -= s.max()
    exp_s = np.exp(s)
    w = exp_s / exp_s.sum()
    return idx, w, cv, corr

def grad_smooth_max(h, idx, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.full(n, np.sum(w))
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-14:
            continue
        k = idx[ii]
        m = n - 1 + k
        i_lo = max(0, m - n + 1)
        i_hi = min(n - 1, m)
        if i_lo <= i_hi:
            positions = np.arange(i_lo, i_hi + 1)
            grad[positions] -= 2.0 * wi * h[m - positions]
    return grad * dx

def obj_smooth_max(h, t, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    nf = len(corr)
    if topk < nf:
        idx = np.argpartition(corr, -topk)[-topk:]
        cv = corr[idx]
    else:
        cv = corr
    mx = np.max(cv)
    s = t * (cv - mx)
    s -= s.max()
    return (mx + np.log(np.sum(np.exp(s))) / t) * (2.0 / len(h))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, target=None):
    if target is None:
        target = len(h) / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def obj_sm(h, t, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    nf = len(corr)
    if topk and topk < nf:
        idx = np.argpartition(corr, -topk)[-topk:]
        cv = corr[idx]
    else:
        cv = corr
    mx = np.max(cv)
    s = t * (cv - mx)
    s -= s.max()
    return (mx + np.log(np.sum(np.exp(s))) / t) * (2.0 / n)

def fista_opt(h0, t, topk, lr, n_iter):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    z = h0.copy()
    tk = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        corr = np.correlate(z, 1.0 - z, mode='full')
        nf = len(corr)
        if topk and topk < nf:
            idx = np.argpartition(corr, -topk)[-topk:]
            cv = corr[idx]
        else:
            idx = np.arange(nf)
            cv = corr
        mx = np.max(cv)
        s = t * (cv - mx)
        s -= s.max()
        exp_s = np.exp(s)
        w = exp_s / exp_s.sum()
        dx = 2.0 / n
        grad = np.full(n, np.sum(w))
        for ii in range(len(idx)):
            wi = w[ii]
            if wi < 1e-12:
                continue
            lag = idx[ii] - (n - 1)
            if lag >= 0:
                grad[:n - lag] -= wi * h[lag:]
                grad[lag:] -= wi * h[:n - lag]
            else:
                L = -lag
                grad[L:] -= wi * h[:n - L]
                grad[:n - L] -= wi * h[L:]
        grad *= dx
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) * 0.5
        z = h + ((tk - 1.0) / tk_new) * (h - z)
        h = proj(z - lr * grad, target)
        tk = tk_new
        if (it + 1) % 150 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def interp_to(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = (np.arange(n) + 0.5) / n
    xd = (np.arange(nd) + 0.5) / nd
    return proj(np.interp(xd, xs, h), nd)

def slsqp_refine(h, n, t, topk, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    def cfun(x):
        return np.array([np.sum(x) - n * 0.5])
    def cjac(x):
        return np.array([np.ones((1, n))])
    cons = ({'type': 'eq', 'fun': cfun, 'jac': cjac},)
    r = minimize(obj_sm, h, args=(t, topk), method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(r.x, 0.0, 1.0), n)
