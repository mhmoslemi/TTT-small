        for t_start in [0.008, 0.004, 0.0015, 0.0005]:
            if time.time() > deadline2 - 20:
                break
            dl = min(deadline2 - 15, time.time() + 25)
            hs, cs = binary_sa(bh.copy(), rng, t_start, 30000, dl)
            if cs < bc:
                bc = cs
                bh = hs.copy()
            bh, bc = binary_best_improve(bh, 2000, min(deadline2 - 5, time.time() + 30))

    # Phase 3: SLSQP continuous refinement
    if time.time() < t0 + 600:
        deadline3 = t0 + 600
        h_cont = bh.copy().astype(float) + rng.randn(n) * 0.005
        h_cont = proj(h_cont, n)
        for t_val in [50, 200, 700, 2500]:
            if time.time() > deadline3 - 40:
                break
            try:
                dl = min(deadline3 - 30, time.time() + 55)
                hc = slsqp_run(h_cont, n, t_val, None, 2500, 1e-14, bnds, cons)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = to_binary(hc, n)
                h_cont = hc
            except Exception:
                pass
        corr = np.correlate(h_cont, 1.0 - h_cont, mode='full')
        topk_val = max(5, len(corr) // 7)
        for t_val in [300, 1500, 6000]:
            if time.time() > deadline3 - 30:
                break
            try:
                hc = slsqp_run(h_cont, n, t_val, topk_val, 2000, 1e-15, bnds, cons)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = to_binary(hc, n)
                h_cont = hc
            except Exception:
                pass

    # Phase 4: More binary search + SA
    if time.time() < t0 + 850:
        deadline4 = t0 + 850
        bh, bc2 = binary_best_improve(bh, 5000, min(deadline4 - 40, time.time() + 100))
        if bc2 < bc:
            bc = bc2
        for t_start in [0.003, 0.001, 0.0003]:
            if time.time() > deadline4 - 20:
                break
            hs, cs = binary_sa(bh.copy(), rng, t_start, 25000, min(deadline4 - 15, time.time() + 25))
            if cs < bc:
                bc = cs
                bh = hs.copy()
        bh, bc2 = binary_best_improve(bh, 3000, min(deadline4 - 5, time.time() + 40))
        if bc2 < bc:
            bc = bc2

    # Phase 5: Final SLSQP ultra-fine
    if time.time() < t0 + 1050:
        deadline5 = t0 + 1050
        h_cont = bh.copy().astype(float) + rng.randn(n) * 0.003
        h_cont = proj(h_cont, n)
        for t_val in [5000, 20000, 80000]:
            if time.time() > deadline5 - 15:
                break
            try:
                hc = slsqp_run(h_cont, n, t_val, None, 4000, 1e-16, bnds, cons)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = to_binary(hc, n)
                h_cont = hc
            except Exception:
                pass

    # Phase 6: Final binary polish
    if time.time() < t0 + budget_s - 5:
        bh, bc2 = binary_best_improve(bh, 1000, t0 + budget_s - 5)
        if bc2 < bc:
            bc = bc2
            bh = bh

    final_h = bh.copy()
    final_c5 = compute_c5(final_h)
    if final_c5 <= bc:
        return final_h, final_c5, n
    return final_h, final_c5, n
