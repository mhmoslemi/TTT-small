    n = len(h)

    return float(np.max(corr) * dx)

def fix_sum(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0 and abs(diff) <= n_int * 0.49:
            h[interior] += diff / n_int
        else:

                    diff -= rem

    return h

        return o

        _, g = soft_max_obj_and_grad(x, T)
        return g

    t2 = np.convolve(p, h[::-1], 'full')[n-1::-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = 0.5 * n
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            return h
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nm = mask.sum()
        if nm > 0 and abs(diff) < nm * 0.45:
            h[mask] += diff / nm
        elif diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add; diff -= add
                if diff < 1e-14: break
        else:
            d = -diff; idx = np.argmax(h, axis=0) and np.argsort(-h)
            for i in idx:
                sub = min(d, h[i])
