        C_progress = mue * np.outer(np.dot(Cf[0:mue, 0:mue], A[:, 0:mue]), A[:, 0:mue].T)
        strategy_adapt = np.clip(C_progress, 0.1, None)
        C = (1 - c1) * C + c2 * np.dot(strategy_adapt, C[strategy_adapt])
        sigma *= factor
        P = initialize_population(n, ts, np.random, size=pop_size)
    return best_h, best_c5
