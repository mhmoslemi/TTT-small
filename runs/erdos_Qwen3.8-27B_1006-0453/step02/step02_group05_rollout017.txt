With soft-max at high temperature (τ large), we're approximating the max. The gradient is a weighted average of the gradients of individual lags, weighted by exp(corr[k]/τ). At very high τ, this concentrates on the max lag, giving essentially the subgradient.

With topk, we directly average the gradients of the top k lags. For k=1, this is the subgradient at the argmax. For k>1, it's a smoothed version.

The issue with topk=1: the argmax can jump between lags, making the gradient discontinuous. This can cause oscillation. Topk=3-5 smooths this out.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lag_grad(h, N, L):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:N - L])
        g[:N - L] -= h[L:]
    else:
        M = -L
        g[:N - M] += (1.0 - h[M:])
        g[M:] -= h[:N - M]
    return g

def topk_subgrad(h, N, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr)) * (2.0 / N)
    idx = np.argpartition(corr, -k)[-k:]
    center = N - 1
    grad = np.zeros(N)
    for ai in idx:
        L = int(ai) - center
        grad += lag_grad(h, N, L)
    grad /= k
    return grad, c5_val

def adam_opt(h, N, target, deadline, lr0, decay, topk, b1=0.85, b2=0.999):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    lr = lr0
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    eps = 1e-8
    while time.time() < deadline:
        grad, c5_val = topk_subgrad(h, N, topk)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 150 == 0:
            lr *= decay
        if lr < 1e-12:
            break
    return best_h, best_c5

def binary_round(h, N):
    idx = np.argpartition(h, -N // 2)[-N // 2:]
    b = np.zeros(N)
    b[idx] = 1.0
    return b

def nesterov_opt(h, N, target, deadline, lr0=0.01, topk=3):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    y = h.copy()
    beta = 0.9
    lr = lr0
    it = 0
    while time.time() < deadline:
        grad, c5_val = topk_subgrad(y, N, topk)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = y.copy()
        new_h = project_sum(y - lr * grad, target)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = project_sum(new_h + (beta_new - 1.0) / (beta + 1.0) * (new_h - y), target)
        it += 1
        if it % 200 == 0:

        if lr < 1e-12:

    return best_h, best_c5
