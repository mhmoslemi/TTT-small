```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    log_sum = max_shifted + np.log(np.sum(exps))
    val = (log_sum / T) * dx
    
    # Softmax weights
    w = exps / np.sum(exps)
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - n + 1  # actual shift k in [-(n-1), n-1]
        if k >= 0:
            # valid i: 0 <= i < n-k
            i_end = n - k
            grad[:i_end + k] += wk * (1.0 - h[:i_end])  # first part: (1-h[i+k]) for i=0..i_end-1, added to grad[i+k]
            # Wait, I need to be more careful
            # For shift k >= 0:
            # valid j: 0 <= j < n-k (so that j+k < n)
            # d(corr_k)/dh[j] = (1-h[j+k]) - h[j-k] (but h[j-k] only if j-k >= 0)
            # For k >= 0: j-k >= 0 means j >= k
            # So for j in [0, n-k-1]:
            #   first term: (1-h[j+k]) always contributes
            #   second term: -h[j-k] only for j >= k
            # So grad[j] += wk * (1-h[j+k]) for j in [0, n-k-1]
            #    grad[j] -= wk * h[j-k] for j in [k, n-k-1]
            for j in range(n - k):
                grad[j] += wk * (1.0 - h[j + k])
            for j in range(k, n - k):
                grad[j] -= wk * h[j - k]
        else:
            # k < 0, let k_abs = -k
            k_abs = -k
            # valid j: k_abs <= j < n (so that j+k = j-k_abs >= 0)
            # d(corr_k)/dh[j] = (1-h[j+k]) - h[j-k]
```

```python
def smooth_c5_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    max_s = np.max(shifted)
    exps = np.exp(shifted - max_s)
    sum_exps = np.sum(exps)
    val = (max_s + np.log(sum_exps)) / T * dx
    w = exps / sum_exps
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - n + 1
        if k == 0:
            grad[:] += wk * (1.0 - h)
            grad[:] -= wk * h
        elif k > 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            if n - 2*k >= 1:
                grad[k:n-k] -= wk * h[:n-2*k]
        else:
            ka = -k
            grad[ka:n-ka] += wk * (1.0 - h[:n-2*ka])
            grad[ka:n-ka] -= wk * h[2*ka:]
    grad *= dx
    return val, grad
```

1. **ADMM**: Split the problem into subproblems. The max can be handled with an epigraph variable t:

2. **Iterative linearization**: At each iteration, linearize corr_k around the current h:

minimize t
s.t. sum_i h[i]*(1-h[i+k]) * dx <= t, for all k

At a given h*, linearize:

- The feasible set is a polytope (0<=h<=1, sum(h)=n/2)
- The objective is the max of bilinear functions
- Linearization gives valid upper bound on each bilinear function (if concave) or not (if not concave)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    max_s = np.max(shifted)
    exps = np.exp(shifted - max_s)
    sum_exps = np.sum(exps)
    val = (max_s + np.log(sum_exps)) / T * dx
    w = exps / sum_exps
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - n + 1
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            seg = n - k
            grad[:seg] += wk * (1.0 - h[k:])
            if n - 2 * k >= 1:
                grad[k:seg] -= wk * h[:n - 2 * k]
        else:
            ka = -k
            seg = n - ka
            grad[ka:seg] += wk * (1.0 - h[:n - 2 * ka])
            grad[ka:seg] -= wk * h[2 * ka:]
    grad *= dx
    return val, grad

def slsqp_analytic(h, T, maxiter=2000, ftol=1e-14):
    n = len(h)
    def obj(x):
        v, g = smooth_c5_gradient(x, T)
        return v, g
    def constr(x):
        return x.sum() - n / 2.0
    def constr_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr, 'jac': constr_jac}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def slsqp_analytic_multiT(h, T_list, maxiter=800, ftol=1e-13):
    cur = h.copy()
    for T in T_list:
        try:
            cur = slsqp_analytic(cur, T, maxiter=maxiter, ftol=ftol)
        except Exception:
            pass
    return cur

def basin_hop(h, n, T, n_iter, start_temp, end_temp, maxiter, t0, budget_s, frac):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        if time.time() - t0 > budget_s * frac:
            break
        step = 0.10 * (1.0 - 2.0 * it / n_iter) + 0.02
        h_pert = best_h + step * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        try:
            hc = slsqp_analytic(h_pert, T, maxiter=maxiter, ftol=1e-12)
            c5v = compute_c5(hc)
        except Exception:
            continue
        temp = start_temp * (end_temp / start_temp) ** (it / max(1, n_iter - 1))
        if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / temp):
            best_h = hc
            best_c5 = c5v
    return best_h, best_c5

def diverse_starts(n, seed_rng):
    starts = []
    hb = (seed_rng.random(n) < 0.5).astype(float)
    starts.append(project_feasible(hb, n))
    hblk = np.zeros(n)
    hblk[::2] = 1.0
    starts.append(project_feasible(hblk, n))
    hblk2 = np.zeros(n)
    hblk2[::3] = 1.0
    starts.append(project_feasible(hblk2, n))
    q = n // 4
    hq = np.zeros(n)
    hq[:q] = 1.0
    hq[2*q:3*q] = 1.0
    starts.append(project_feasible(hq, n))
    phases = seed_rng.uniform(0, 2*np.pi, size=3)
    amp = seed_rng.uniform(0.1, 0.3, size=3)
    hs = 0.5 + sum(amp[i] * np.sin(2*np.pi*3*(i+1)*np.linspace(0,2,n)/2 + phases[i]) for i in range(3))
    starts.append(project_feasible(hs, n))
    return starts

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
```
