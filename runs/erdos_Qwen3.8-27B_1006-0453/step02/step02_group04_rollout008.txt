```python
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_corr(h, T):
    corr = compute_corr(h)
    c_max = np.max(corr)
    if T < 1e-15:

    shifted = np.clip(corr - c_max, -80.0*T, 0.0)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
        # Second pass if needed
        s2 = np.sum(h)
        diff2 = target - s2
        interior2 = (h > 1e-10) & (h < 1.0 - 1e-10)
        if abs(diff2) > 1e-12 and np.sum(interior2) > 0:
            h[interior2] += diff2 / np.sum(interior2)
            h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-13: break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13: break
    return h

    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
```

```python
def coord_step(h, r, n, corr, active_lags):
    """Compute optimal h[r] given all other coordinates fixed."""
    base_0 = corr[n-1] - h[r]*(1.0-h[r])
    
    # Compute base and slope for active lags only
    active = active_lags  # indices into corr array
    slopes = np.zeros(len(active))
    bases = np.zeros(len(active))
    for idx, m_idx in enumerate(active):
        k = m_idx - (n-1)
        if k == 0:
            continue
        abs_k = abs(k)
        if k > 0:
            sl = 0.0
            if r >= k: sl += (1.0 - h[r-k])
            if r + k < n: sl += (-h[r+k])
        else:
            kp = -k
            sl = 0.0
            if r + kp < n: sl += (1.0 - h[r+kp])
            if r >= kp: sl += (-h[r-kp])
        slopes[idx] = sl
        bases[idx] = corr[m_idx] - h[r]*sl
    
    # Candidates: 0, 1, 0.5, and quadratic-line intersections
    candidates = [0.0, 1.0, 0.5]
    for idx in range(len(active)):
        m_idx = active[idx]
        if m_idx == n-1: continue
        a = 1.0
        b = slopes[idx] - 1.0
        c = bases[idx] - base_0
        disc = b*b - 4*a*c
        if disc >= 0:
            s = np.sqrt(disc)
            x1 = (-b + s) / (2*a)
            x2 = (-b - s) / (2*a)
            if 0 <= x1 <= 1: candidates.append(x1)
            if 0 <= x2 <= 1: candidates.append(x2)
    
    best_x = h[r]
    best_val = 1e18
    for x in candidates:
        x = max(0.0, min(1.0, x))
        val_quad = base_0 + x - x*x
        val_lines = np.max(bases + x*slopes)
        g = max(val_quad, val_lines)
        if g < best_val:
            best_val = g
            best_x = x
    return best_x
```

This is O(K) per coordinate where K is the number of active lags (~10-20). Very fast!

The coordinate descent should be able to find a better local optimum because:
- Each step is the EXACT 1D optimum (not an approximation)
- It naturally handles the non-convexity (max of quadratics and lines)
- It's not limited by SLSQP's trust region or convergence criteria
- It can do many more iterations

One concern: coordinate descent can converge to a saddle point or get stuck. The multi-start should handle this.

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = 0.5 * n
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
        s2 = np.sum(h)
        diff2 = target - s2
        if abs(diff2) > 1e-12:
            interior2 = (h > 1e-10) & (h < 1.0 - 1e-10)
            n_int2 = int(np.sum(interior2))
            if n_int2 > 0:
                h[interior2] += diff2 / n_int2
                h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-13: break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13: break
    return h

def coord_optimal(h, r, n, corr, active):
    """Find optimal h[r] given all other coordinates, using active lags."""
    base_0 = corr[n-1] - h[r] * (1.0 - h[r])
    m = len(active)
    slopes = np.zeros(m)
    bases = np.zeros(m)
    valid = np.zeros(m, dtype=bool)
    for idx in range(m):
        mi = active[idx]
        k = mi - (n - 1)
        if k == 0:
            valid[idx] = False
            continue
        sl = 0.0
        if k > 0:
            if r >= k: sl += (1.0 - h[r - k])
            if r + k < n: sl += (-h[r + k])
        else:
            kp = -k
            if r + kp < n: sl += (1.0 - h[r + kp])
            if r >= kp: sl += (-h[r - kp])
        slopes[idx] = sl
        bases[idx] = corr[mi] - h[r] * sl
        valid[idx] = True
    
    vm = m  # just use all, invalid ones have slope=0, base=0
    # Candidates
    cands = [0.0, 1.0, 0.5]
    for idx in range(m):
        if not valid[idx]: continue
        a = 1.0
        b = slopes[idx] - 1.0
        c = bases[idx] - base_0
        disc = b * b - 4.0 * a * c
        if disc >= 0:
            sq = np.sqrt(disc)
            x1 = (-b + sq) / 2.0
            x2 = (-b - sq) / 2.0
            if 0.0 <= x1 <= 1.0: cands.append(x1)
            if 0.0 <= x2 <= 1.0: cands.append(x2)
    
    best_x = h[r]
    best_v = 1e18
    for x in cands:
        x = max(0.0, min(1.0, x))
        vq = base_0 + x - x * x
        vl = np.max(bases + x * slopes)
        g = vq if vq > vl else vl
        if g < best_v:
            best_v = g
            best_x = x
    return best_x

def coord_descent(h, n, active_k, n_sweeps, use_fresh=True):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-active_k:]
    active = sorted(top.tolist())
    prev_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        for r in range(n):
            xv = coord_optimal(h, r, n, corr, active)
            h[r] = xv
            if use_fresh and (r + 1) % 20 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full')
                top = np.argsort(corr)[-active_k:]
                active = sorted(top.tolist())
        if use_fresh:
            corr = np.correlate(h, 1.0 - h, mode='full')
            top = np.argsort(corr)[-active_k:]
            active = sorted(top.tolist())
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 >= prev_c5 - 1e-14 and sweep > 5:
            break
        prev_c5 = c5
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    return upsample(h, new_n)

def do_slsqp(h, n, T, max_iter):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cm = np.max(corr)
        sh = np.clip(corr - cm, -80*T, 0)
        return (cm + T*np.log(max(np.sum(np.exp(sh/T)), 1e-300))) * (2.0/n)
    try:
        res = opt.minimize(obj, h.copy(), method='SLSQP',
                           bounds=[(0,1)]*n,
                           constraints=[{'type':'eq','fun':lambda x: np.sum(x)-0.5*n}],
                           options={'maxiter':max_iter,'ftol':1e-16})
        if res.x is not None:
            return fix_sum(res.x, n)
    except: pass
    return h
