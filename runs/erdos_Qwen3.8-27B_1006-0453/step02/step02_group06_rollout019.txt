    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def proj_feasible(h, n):

    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    # Bisection for lambda
    lo, hi = -1.0, 1.0

        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    diff = target - np.sum(h)
    if abs(diff) > 1e-10:

        if np.sum(interior) > 0:

            h = np.clip(h, 0.0, 1.0)

    """Compute weighted subgradient of max correlation."""
    corr = corr_vector(h)
    n_corr = len(corr)
    # Get top k indices
    top_ids = np.argsort(corr)[-top_k:]
    weights = np.exp((corr[top_ids] - corr[top_ids].max()) / 0.001)
    weights /= weights.sum()

    grad = np.zeros(n)

            if 0 <= i + m < n:

        if np.clip(h+mid, 0, 1).sum() > t: hi = mid
        else: lo = mid
    h = np.clip(h + (lo+hi)/2, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-10:
        m = (h > 1e-10) & (h < 1-1e-10)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0, 1)
    return h

def lse_obj(h, T):
    corr = corr_vec(h)
    cmax = corr.max()
    if T < 1e-12: return cmax * (2.0/len(h))
    return (cmax + T * np.log(np.sum(np.exp((corr - cmax)/T)))) * (2.0/len(h))

def lse_grad(h, T):
    n = len(h)
    corr = corr_vec(h)
    cmax = corr.max()
    w = np.exp((corr - cmax) / T)
    w /= w.sum()
    # Gradient of log-sum-exp = weighted avg of individual gradients
    m_offsets = np.arange(len(corr)) - (n-1)
    grad = np.zeros(n)
    for wm, mo in zip(w, m_offsets):
        if wm < 1e-10: continue
        if mo == 0:
            grad += wm * (1 - 2*h)
        elif mo > 0:
            i1 = np.arange(mo, n)
            grad[i1] += wm * (1 - h[i1 - mo])
            i2 = np.arange(0, n - mo)
            grad[i2] -= wm * h[i2 + mo]
        else:
            mm = -mo
            i1 = np.arange(0, n - mm)
            grad[i1] += wm * (1 - h[i1 + mm])
            i2 = np.arange(mm, n)
            grad[i2] -= wm * h[i2 - mm]
    return grad * (2.0/n)

Wait, this gradient computation loops over all n_corr = 2n-1 shifts, which for n=336 is 671 iterations, each doing O(n) work. That's O(n²) total per gradient evaluation, which is the same as computing the correlation. So it's fine.

But I can optimize: only the top K shifts contribute significantly to the log-sum-exp gradient (when T is small). So I can skip shifts with very small weight.

Let me also think about the 0-1 local search more carefully:

For a 0-1 vector with n/2 ones and n/2 zeros:
- A "move" is flipping one 0 to 1 and one 1 to 0 (preserves the sum)
- Or flipping a group: move a block of 1s to a block of 0s
- Evaluate the new max correlation

The evaluation is O(n) (just compute the correlation for the affected shifts).

Actually, for the full correlation, we need to recompute all 2n-1 shifts, which is O(n²). But we can be smarter: if we flip positions i and j (i was 0, j was 1), the change in correlation at shift m is:
- Δc[m] = (1-h[j-m]) - h[i-m] + h[j+m] - (1-h[i+m]) ... this gets complicated.

Let me just recompute the full correlation for each move. For n=336, it's fast enough.

Let me also add a key insight: the problem is invariant under h → 1-h (the correlation of h with 1-h is the same as correlation of 1-h with h, which is the same computation). Wait, no:
correlate(h, 1-h) vs correlate(1-h, h). These are different! correlate(a,b) ≠ correlate(b,a) in general (they're reflections of each other).

Actually: correlate(a, b, 'full')[k] = sum_j a[j] * b[j+k-(n-1)]
correlate(b, a, 'full')[k] = sum_j b[j] * a[j+k-(n-1)]

These are related by: correlate(a,b,'full') = correlate(b,a,'full')[::-1] (reversed).

So max of correlate(h, 1-h) = max of correlate(1-h, h). The problem IS invariant under h → 1-h! Good to know.

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > t:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, args):
    h, T, n = args
    h = proj(x, n)
    corr = corr_vec(h)
    cmax = corr.max()
    if T < 1e-12:
        return cmax * (2.0 / n)
    return (cmax + T * np.log(np.sum(np.exp((corr - cmax) / T)))) * (2.0 / n)

def lse_grad_fn(x, args):
    h, T, n = args
    h = proj(x, n)
    corr = corr_vec(h)
    cmax = corr.max()
    w = np.exp((corr - cmax) / T)
    thr = w.max() * 1e-8
    active = w > thr
    w[~active] = 0.0
    w /= w.sum()
    m_offsets = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for wm, mo in zip(w[active], m_offsets[active]):
        if wm < 1e-10:
            continue
        if mo == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif mo > 0:
            i1 = np.arange(mo, n)
            grad[i1] += wm * (1.0 - h[i1 - mo])
            i2 = np.arange(0, n - mo)
            grad[i2] -= wm * h[i2 + mo]
        else:
            mm = -mo
            i1 = np.arange(0, n - mm)
            grad[i1] += wm * (1.0 - h[i1 + mm])
            i2 = np.arange(mm, n)
            grad[i2] -= wm * h[i2 - mm]
    return grad * (2.0 / n)

def lse_optimize(h0, n, T_list, max_iter_list, t_start, budget_s, reserve=80):
    best_h, best_c = h0.copy(), compute_c5(h0)
    for T, mi in zip(T_list, max_iter_list):
        if time.time() - t_start > budget_s - reserve:
            break
        res = opt.minimize(
            lse_obj, best_h,
            args=(T, n),
            jac=lambda x: lse_grad_fn(x, (T, n)),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0}
        )
        h_new = proj(res.x, n)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    return best_h, best_c

def subgrad_descent(h0, n, t_start, budget_s, n_iters=2000, reserve=80):
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h0)
    lr0 = 0.05
    for it in range(n_iters):
        if time.time() - t_start > budget_s - reserve:
            break
        corr = corr_vec(h)
        cmax = corr.max()
        dx = 2.0 / n
        c = cmax * dx
        # Top-3 weights
        top3 = np.argsort(corr)[-3:]
        w = np.exp((corr[top3] - cmax) / 0.003)
        w /= w.sum()
        m_offsets = top3 - (n - 1)
        grad = np.zeros(n)
        for wm, mo in zip(w, m_offsets):
            if mo == 0:
                grad += wm * (1.0 - 2.0 * h)
            elif mo > 0:
                grad[np.arange(mo, n)] += wm * (1.0 - h[mo:n] if mo < n else 0)
            else:
                mm = -mo
                grad[np.arange(0, n-mm)] += wm * (1.0 - h[mm:])
        # Also subtract second terms
        for wm, mo in zip(w, m_offsets):
            if mo > 0:
                grad[:n-mo] -= wm * h[mo:]
            elif mo < 0:
                mm = -mo
                grad[mm:] -= wm * h[:n-mm]
        lr = lr0 * (1.0 - it / n_iters) ** 1.5
        h = proj(h - lr * grad * dx, n)
        c = compute_c5(h)
        if c < best_c - 1e-9:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def binarize_with_sum(h, n):
    hb = (h >= 0.5).astype(float)
    s = hb.sum()
    target = n * 0.5
    if s == target:
        return hb
    if s > target:
        excess = int(s - target)
        ones = np.where(hb == 1.0)[0]
        pick = np.argsort(h[ones])[:excess]
        hb[ones[pick]] = 0.0
    else:
        deficit = int(target - s)
        zeros = np.where(hb == 0.0)[0]
        pick = np.argsort(-h[zeros])[:deficit]
        hb[zeros[pick]] = 1.0
    return proj(hb, n)

def local_search_01(hb, n, t_start, budget_s, max_iters=50000, reserve=80):
    best_h = hb.copy()
    best_c = compute_c5(hb)
    rng = np.random.RandomState(12345)
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    T_sa = 0.001
    stuck = 0
    for it in range(max_iters):
        if time.time() - t_start > budget_s - reserve:
            break
        if stuck > 5000:
            T_sa = max(T_sa, 0.0005)
            stuck = 0
        # Random swap: pick one 1 and one 0, flip both
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        c_new = compute_c5(new_h)
        if c_new < best_c - 1e-12:
            best_h = new_h
            best_c = c_new
            k = np.where(best_h == 1.0)[0]
            l = np.where(best_h == 0.0)[0]
            ones, zeros = k, l
            stuck = 0
        else:
            stuck += 1
            if rng.rand() < np.exp(-(c_new - best_c) / T_sa):
                best_h = new_h
                ones = np.where(best_h == 1.0)[0]
                zeros = np.where(best_h == 0.0)[0]
        if it % 2000 == 0:
            T_sa *= 0.95
    return best_h, best_c

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj(h_new, new_n)

def block_local(h0, n, t_start, budget_s, block_size=16, reserve=60):
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h0)
    rng = np.random.RandomState(777)
    for _ in range(200):
        if time.time() - t_start > budget_s - reserve:
            break
        start = rng.randint(0, n - block_size)
        idx = np.arange(start, start + block_size)
        h_block = h[idx].copy()
        best_block = h_block.copy()
        # Try small perturbations on the block
        for attempt in range(20):
            trial = h_block + rng.randn(block_size) * 0.02
            trial = np.clip(trial, 0, 1)
            h_trial = h.copy()
            h_trial[idx] = trial
            h_trial = proj(h_trial, n)
            c_trial = compute_c5(h_trial)
            if c_trial < compute_c5(h):
                h = h_trial
                h_block = h[idx].copy()
                break
        c = compute_c5(h)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
