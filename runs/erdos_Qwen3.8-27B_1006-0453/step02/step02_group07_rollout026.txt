from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr.max()
    if T < 1e-15:
        return c * dx
    shifted = (corr - c) / T
    shifted = np.minimum(shifted, 0.0)
    return (c + T * np.log(np.sum(np.exp(shifted)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = m.sum()
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-11:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, n, T, maxiter):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': maxiter, 'ftol': 1e-18})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr.max()
    if T < 1e-15:
        return c * dx
    s = np.minimum((corr - c) / T, 0.0)
    return (c + T * np.log(np.sum(np.exp(s)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = m.sum()
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-11:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return np.clip(h, 0.0, 1.0)

    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': maxiter, 'ftol': 1e-18})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h
