    h = np.clip(h, 0.0, 1.0)

                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < best_c - 1e-11:
                    best_c = c
                    best = hn
            h = best
        h = fix_sum(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 7)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)

    sched = [(0.04, 500), (0.015, 600), (0.006, 800), (0.002, 1000),
             (0.0008, 1200), (0.0003, 1500), (0.0001, 2000)]
    for T, it in sched:
        if time.time() - t0 > budget_s - 500:
            break
        h = slsqp(h, n, T, it)
        c5 = compute_c5(h)

    if time.time() - t0 < budget_s - 350:
        h = peak_shave(h, n, 130)
        c5 = compute_c5(h)

    if time.time() - t0 < budget_s - 300:
        hb, cb = binary_local_search(h, n, rng, 6000)
        h_blend = 0.5 * h + 0.5 * hb
        h_blend = fix_sum(h_blend, n)
        h_blend = slsqp(h_blend, n, 0.002, 600)
        hb_c = compute_c5(h_blend)
        if hb_c < c5:
            h, c5 = h_blend, hb_c

    if time.time() - t0 < budget_s - 250:
        for p in [n // 2, n // 4, n // 5]:
            if n % p == 0 and p >= 14:
                hp = periodic_avg(h, n, p)
                hp = slsqp(hp, n, 0.001, 500)
                hp_c = compute_c5(hp)
                if hp_c < c5:
                    h, c5 = hp, hp_c

    if time.time() - t0 < budget_s - 180:
        h = coord_polish(h, n, 2)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 140:
        h = pair_polish(h, n, rng, 700)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 120:
        h = peak_shave(h, n, 80)
        c5 = compute_c5(h)

    for i in range(18):
        if time.time() - t0 > budget_s - 100:
            break
        mode = i % 3
        if mode == 0:
            ht = fourier_restart(h, n, rng)
        elif mode == 1:
            ht = h.copy()
            ht += np.random.randn(n) * 0.02
            idx = np.random.choice(n, 12, replace=False)
            ht[idx] += np.random.randn(12) * 0.1
        else:
            hb2, _ = binary_local_search(h, n, rng, 2000)
            ht = 0.4 * h + 0.6 * hb2
        ht = fix_sum(ht, n)
        for T in (0.004, 0.001, 0.0003):
            ht = slsqp(ht, n, T, 400)
        ht = peak_shave(ht, n, 40)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() - t0 < budget_s - 50:
                h = pair_polish(h, n, rng, 300)
                h = peak_shave(h, n, 40)
                c5 = compute_c5(h)

    for T, it in ((0.00005, 2500), (0.000015, 3000)):
        if time.time() - t0 > budget_s - 25:
            break
        h = slsqp(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    if time.time() - t0 < budget_s - 10:
        h = peak_shave(h, n, 60)
        h = coord_polish(h, n, 1)
        c5 = compute_c5(h)

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
