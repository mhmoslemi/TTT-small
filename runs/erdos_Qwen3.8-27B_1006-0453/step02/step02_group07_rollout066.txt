The projection onto {h : 0≤h_i≤1, sum(h)=n/2} is a water-filling problem. Given a vector y, find h that minimizes ||h-y||² subject to 0≤h≤1 and sum(h)=n/2.

This can be solved by: h_i = clip(y_i - λ, 0, 1) for the right λ. We can find λ by bisection since sum(clip(y_i - λ, 0, 1)) is monotonically decreasing in λ.

Let me implement the full algorithm:

```python
def project_onto_constraints(y, n):
    """Project y onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection for lambda
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    # Fix any tiny residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def fista(h, T, n_iter, L=None):

    n = len(h)
    if L is None:
        # Estimate Lipschitz constant
        # L ≈ max eigenvalue of the Hessian, roughly bounded by 4*n/dx... 
        # Actually for this problem, L ~ O(n) since each correlation involves O(n) products
        # Let me just use a backtracking approach
        L = 4.0 * n  # rough upper bound
    x = h.copy()
    z = h.copy()
    t = 1.0
    for it in range(n_iter):
        val, grad = smooth_obj_and_grad(z, T)
        # Backtracking line search for L
        while True:
            y = z - grad / L
            h_proj = project_onto_constraints(y, n)
            # Check: f(h_proj) <= f(z) + grad·(h_proj-z) + L/2 * ||h_proj-z||²
            # For simplicity, just use the step
            break  # Skip backtracking for speed, use fixed L
        x_new = h_proj
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        z = x_new + ((t-1)/t_new) * (x_new - x)
        z = project_onto_constraints(z, n)  # Keep z feasible
        x = x_new
        t = t_new
    return x
```

```python
    except Exception:
```
```python
        return h
```

```python
        v, _ = smooth_obj_grad(x, T)
        return v

def coord_descent(h, n, n_pass=3):
```
