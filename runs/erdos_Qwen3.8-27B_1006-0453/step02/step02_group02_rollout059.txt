    corr = np.correlate(h, 1.0 - h, mode='full')

Actually, I realize I should also try a completely different approach: **use the fact that the problem is equivalent to minimizing the maximum cross-correlation between a binary sequence and its complement**.

For a binary sequence s of length n with n/2 ones, define:
- autocorr_s[k] = Σ_i s[i] * s[i+k] (number of 1-1 pairs at lag k)
- The overlap we care about: Σ_i s[i] * (1-s[i+k]) = (number of 1s in positions 0 to n-1-k) - autocorr_s[k]

For the linear case (not circular), this is:
overlap[k] = (n-k) - k*(n/n) - autocorr_s[k] ... no, this isn't right.

Let me just define it directly:
overlap[k] = Σ_{i=0}^{n-1-k} s[i]*(1-s[i+k]) for k > 0

= Σ_{i=0}^{n-1-k} s[i] - Σ_{i=0}^{n-1-k} s[i]*s[i+k]
= (number of 1s in positions 0 to n-1-k) - autocorr[k]

For a "well-distributed" sequence, the number of 1s in any contiguous block of length m is approximately m/2. So:
overlap[k] ≈ (n-k)/2 - autocorr[k]

And autocorr[k] ≈ n/4 for a random sequence.
So overlap[k] ≈ (n-k)/2 - n/4 = (n-2k)/4

For k=0: n/4
For k=n/2: 0
For k close to n: negative (not valid)

To minimize the max, we want overlap[k] to be as uniform as possible. The maximum is at k=0 (or k=1) and equals about n/4.

C₅ ≈ (n/4) * (2/n) = 1/2.

But we're getting 0.38, which is much better than 0.5. So the structure is significantly reducing the peak.

How? By making autocorr[k] higher at small k (so 1s tend to be near other 1s) and lower at other k. This clusters the 1s in a way that reduces the 1→0 transitions at the peak lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = h < 1.0 - 1e-12 if d > 0 else h > 1e-12
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    hb = (h >= 0.5).astype(float)
    oc = int(hb.sum())
    if oc > n // 2:
        idx = np.where(hb == 1)[0]
        order = np.argsort(h[idx])
        for k in range(oc - n // 2):
            hb[idx[order[k]]] = 0
    elif oc < n // 2:
        idx = np.where(hb == 0)[0]
        order = np.argsort(-h[idx])
        for k in range(n // 2 - oc):
            hb[idx[order[k]]] = 1
    return hb

def smax_obj(x, t, topk, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if topk and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        c = corr
    mx = c.max()
    s = t * (c - mx)
    s -= s.max()
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def slsqp_run(h, n, t, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    ceq_f = lambda x: np.array([x.sum() - n / 2.0])
    ceq_j = lambda x: np.ones((1, n))
    cons = [{'type': 'eq', 'fun': ceq_f, 'jac': ceq_j}]
    r = minimize(smax_obj, h, args=(t, topk, n), method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def binary_sa(hb, n, rng, max_iter=80000, T0=0.008, Tmin=0.0005, cool=0.9997):
    c5 = compute_c5(hb)
    T = T0
    for _ in range(max_iter):
        if T < Tmin:
            break
        ones = np.where(hb > 0.5)[0]
        zeros = np.where(hb < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        hb[i], hb[j] = 0.0, 1.0
        new_c5 = compute_c5(hb)
        d = new_c5 - c5
        if d < 0 or rng.random() < np.exp(-d / T):
            c5 = new_c5
        else:
            hb[i], hb[j] = 1.0, 0.0
        T *= cool
    return hb, c5

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    if n % nd == 0:
        return proj(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def greedy_improve(hb, n, rng, rounds=5):
    c5 = compute_c5(hb)
    for _ in range(rounds):
        improved = False
        ones = np.where(hb > 0.5)[0]
        zeros = np.where(hb < 0.5)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros[:20]:
                if i == j:
                    continue
                hb[i], hb[j] = 0.0, 1.0
                nc = compute_c5(hb)
                if nc < c5 - 1e-15:
                    c5 = nc
                    improved = True
                else:
                    hb[i], hb[j] = 1.0, 0.0
        if not improved:
            break
    return hb, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_base = 280
    except Exception:
        n_base = 280
        h_init = make_binary(np.full(n_base, 0.5), n_base)

    best_h = proj(h_init, n_base)
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary simulated annealing at base n
    if tleft() > 200:
        hb = make_binary(best_h, n_base)
        for trial in range(3):
            if tleft() < 60:
                break
            hb, c5 = binary_sa(hb.copy(), n_base, rng,
                               max_iter=40000, T0=0.006 * (trial + 1), Tmin=0.0003)
            hb, c5 = greedy_improve(hb, n_base, rng, rounds=3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hb.copy()

    # Phase 2: SLSQP continuous optimization at base n
    if tleft() > 250:
        h_work = best_h.copy()
        corr0 = np.correlate(h_work, 1.0 - h_work, mode='full')
        n_lags = len(corr0)
        topk1 = max(8, n_lags // 5)
        for t_val in [30, 100, 400, 1500, 5000]:
            if tleft() < 30:
                break
            try:
                hc = slsqp_run(h_work, n_base, t_val, topk1, 1500, 1e-14)
                c5 = compute_c5(hc)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
                    h_work = hc
            except Exception:
                pass

    # Phase 3: Coarse optimization (n=140)
    if tleft() > 200:
        n_c = 140
        hc = interp_down(best_h, n_c)
        hbc = make_binary(hc, n_c)
        hbc, c5c = binary_sa(hbc, n_c, rng, max_iter=30000, T0=0.01, Tmin=0.0004)
        hbc, c5c = greedy_improve(hbc, n_c, rng, rounds=4)
        up140 = interp_up(hbc, n_base)
        c5u = compute_c5(up140)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = up140

        for t_val in [50, 200, 800, 3000]:
            if tleft() < 25:
                break
            try:
                hc2 = slsqp_run(hc, n_c, t_val, 20, 1200, 1e-14)
                c5_2 = compute_c5(hc2)
                up = interp_up(hc2, n_base)
                c5_up = compute_c5(up)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = up
                    hc = hc2
                if c5_up < best_c5:
                    best_c5 = c5_up
                    best_h = up
            except Exception:
                pass

    # Phase 4: Even coarser (n=70)
    if tleft() > 150:
        n_co = 70
        hco = interp_down(best_h, n_co)
        hbio = make_binary(hco, n_co)
        hbio, c5o = binary_sa(hbio, n_co, rng, max_iter=25000, T0=0.012, Tmin=0.0005)
        hbio, c5o = greedy_improve(hbio, n_co, rng, rounds=5)
        up70 = interp_up(hbio, n_base)
        c5_70u = compute_c5(up70)
        if c5_70u < best_c5:
            best_c5 = c5_70u
            best_h = up70

        for t_val in [80, 400, 2000]:
            if tleft() < 20:
                break
            try:
                hc3 = slsqp_run(hco, n_co, t_val, 12, 1000, 1e-14)
                c5_3 = compute_c5(hc3)
                up3 = interp_up(hc3, n_base)
                c5_3u = compute_c5(up3)
                if min(c5_3, c5_3u) < best_c5:
                    best_c5 = min(c5_3, c5_3u)
                    best_h = up3 if c5_3u < c5_3 else interp_up(hc3, n_base)
                    hco = hc3
            except Exception:
                pass

    # Phase 5: Final binary SA + polish at base n
    if tleft() > 60:
        hb2 = make_binary(best_h, n_base)
        for trial in range(2):
            if tleft() < 25:
                break
            hb2, c5_2 = binary_sa(hb2, n_base, rng,
                                  max_iter=30000, T0=0.004, Tmin=0.0002)
            hb2, c5_2 = greedy_improve(hb2, n_base, rng, rounds=4)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = hb2.copy()

    # Phase 6: Final SLSQP polish
    if tleft() > 40:
        for t_val in [3000, 10000, 30000]:
            if tleft() < 15:
                break
            try:
                hc = slsqp_run(best_h, n_base, t_val, 5, 3000, 1e-16)
                c5 = compute_c5(hc)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    best_h = proj(best_h, n_base)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_base
