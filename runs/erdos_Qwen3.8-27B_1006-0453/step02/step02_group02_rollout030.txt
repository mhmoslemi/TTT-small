        j = zeros[rng.randint(n - n_ones)]

        # v[i]: 0->1, v[j]: 1->0

        ones = np.where(hb == 1.0)[0]  # This is O(n) but n=280, fine
        zeros = np.where(hb == 0.0)[0]

            curr_corr = new_corr

                best_hb = hb.copy()
        else:
            # Undo
            hb[i] = 1.0
            hb[j] = 0.0
