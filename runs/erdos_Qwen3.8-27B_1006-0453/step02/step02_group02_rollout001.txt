def proj_sum(h, n):

        h[mask] += step

    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj_sum(np.interp(xd, xs, h), nd)

def swap_local_search(h, n, max_rounds=200, rng=None):
    h = h.copy()
    c5 = compute_c5(h)
    if rng is None:
        rng = np.random.RandomState(0)
    for _ in range(max_rounds):
        improved = False
        for _ in range(30):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] > 0.5 and h[j] < 0.5:
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                if c5_new < c5 - 1e-13:
                    c5 = c5_new
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, c5

def lbfgs_refine(h0, n, t_val, topk, pen, maxiter):
    b = [(0.0, 1.0)] * n
    r = minimize(obj_func, h0, args=(t_val, topk, pen), jac=jac_func,
                 method='L-BFGS-B', bounds=b,
                 options={'maxiter': maxiter, 'ftol': 1e-15,
                          'gtol': 1e-12, 'maxls': 40})
    return proj_sum(np.clip(r.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = proj_sum(np.array(initial_h_values, dtype=float), 280)
    except Exception:
        bh = proj_sum(np.full(280, 0.5) + rng.randn(280) * 0.05, 280)
    bc = compute_c5(bh)

    # Phase 1: Coarse exploration at n=140
    if tleft() > 80:
        h140 = interp_down(bh, 140)
        for t_val, topk in [(30, None), (100, None), (400, 20)]:
            if tleft() < 10:
                break
            try:
                hc = lbfgs_refine(h140, 140, t_val, topk, 1e4, 300)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = interp_up(hc, 280)
                h140 = hc
            except Exception:
                pass

    # Phase 2: Refine at n=280 with L-BFGS-B
    h280 = bh.copy()
    configs280 = [(30, None, 500), (80, None, 400), (200, None, 400),
                  (500, 56, 350), (1000, 28, 300), (3000, 14, 300), (10000, 7, 250)]
    for t_val, topk, miter in configs280:
        if tleft() < 12:
            break
        try:
            hc = lbfgs_refine(h280, 280, t_val, topk, 1e4, miter)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc.copy()
            h280 = hc
        except Exception:
            pass

    # Phase 3: Multi-start at n=280
    if tleft() > 120:
        starts = [
            proj_sum(bh + rng.randn(280) * 0.015, 280),
            proj_sum(bh + rng.randn(280) * 0.04, 280),
            proj_sum((bh > 0.5).astype(float), 280),
            proj_sum(np.sin(np.pi * np.linspace(0, 2, 280)) ** 2, 280),
        ]
        for st in starts:
            if tleft() < 20:
                break
            try:
                hc = lbfgs_refine(st, 280, 100, None, 1e4, 250)
                hc = lbfgs_refine(hc, 280, 800, 42, 1e4, 250)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
                    h280 = hc
            except Exception:
                pass

    # Phase 4: Upsample to n=560 and refine
    if tleft() > 150:
        h560 = interp_up(bh, 560)
        configs560 = [(30, None, 400), (100, None, 350), (300, None, 300),
                      (800, 84, 250), (2000, 42, 250), (5000, 21, 200)]
        for t_val, topk, miter in configs560:
            if tleft() < 12:
                break
            try:
                hc = lbfgs_refine(h560, 560, t_val, topk, 1e4, miter)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
                h560 = hc
            except Exception:
                pass
        c5_560 = compute_c5(h560)
        if c5_560 < bc:
            bc = c5_560
            bh = h560.copy()

    # Phase 5: Binary local search
    if tleft() > 40:
        n_b = len(bh)
        h_bin = (bh > 0.5).astype(float)
        h_bin = proj_sum(h_bin, n_b)
        c5_bin = compute_c5(h_bin)
        if c5_bin < bc:
            bc = c5_bin
            bh = h_bin.copy()
        if tleft() > 30:
            bh, c5_ls = swap_local_search(bh, n_b, 300, rng)
            if c5_ls < bc:
                bc = c5_ls

    # Phase 6: Final high-t polish
    if tleft() > 15:
        n_b = len(bh)
        try:
            hc = lbfgs_refine(bh, n_b, 20000, None, 1e5, 400)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc
        except Exception:
            pass
    if tleft() > 10:
        n_b = len(bh)
        try:
            hc = lbfgs_refine(bh, n_b, 100000, None, 1e5, 300)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc
        except Exception:
            pass

    # Final binary refinement
    if tleft() > 15:
        n_b = len(bh)
        h_bin = (bh > 0.5).astype(float)
        h_bin = proj_sum(h_bin, n_b)
        if compute_c5(h_bin) < bc:
            bh = h_bin
            bc = compute_c5(bh)
        bh, c5_ls = swap_local_search(bh, n_b, 200, rng)
        if c5_ls < bc:
            bc = c5_ls

    final_c5 = compute_c5(bh)
    return bh, final_c5, len(bh)
