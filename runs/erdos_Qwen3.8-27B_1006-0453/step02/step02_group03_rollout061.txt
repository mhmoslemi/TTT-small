def smooth_slsqp(h0, dx, ts, beta_start, beta_end, maxiter):
    n = len(h0)
    bnds = [(0,1)]*n
    cons = [{'type':'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    # Multi-stage: increase beta
    h = h0.copy()
    n_stages = 4
    for s in range(n_stages):
        b = beta_start * (beta_end/beta_start)**(s/n_stages)
        r = minimize(_sfun(b, dx, ts), h, jac=_gfun(b, dx, ts),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter//n_stages, 'ftol':1e-12})
        if r.success or r.fun < compute_c5(h,dx)/dx:

        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def smooth_slsqp(h0, dx, ts, b0, b1, nit):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h = h0.copy()
    stages = 4
    for s in range(stages):
        frac = (s + 1) / stages
        b = b0 * (b1 / b0) ** frac
        _gs['beta'], _gs['dx'], _gs['ts'] = b, dx, ts
        try:
            r = minimize(_obj, h, jac=_objg, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': nit // stages + 1, 'ftol': 1e-14})
            if r.x is not None:
                h2 = proj(r.x, ts)
                if compute_c5(h2, dx) <= compute_c5(h, dx) + 0.005:
                    h = h2
        except Exception:
            pass
    return h

def grad_desc(h0, dx, ts, iters, dl, rng):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h, dx)
    bh, bc = h.copy(), c
    stall = 0
    for i in range(iters):
        if dl is not None and time.time() > dl:
            break
        beta = 20.0 + 100.0 * min(1.0, i / (0.7 * iters))
        lr = 0.04 / (1.0 + 0.001 * i)
        cr = np.correlate(h, 1.0 - h, mode='full')
        m = cr.max()
        e = np.exp(beta * (cr - m))
        s = e.sum()
        w = e / s if s > 1e-30 else np.ones(2 * n - 1) / (2 * n - 1)
        g = np.zeros(n)
        o = n - 1
        for j in range(2 * n - 1):
            wj = w[j]
            if wj < 1e-12:
                continue
            k = j - o
            lo = max(0, -k)
            hi = min(n - 1, n - 1 - k)
            for ii in range(lo, hi + 1):
                ipk = ii + k
                g[ii] += wj * (1.0 - h[ipk])
                g[ipk] -= wj * h[ii]
        g *= dx
        h2 = proj(h - lr * g, ts)
        c2 = compute_c5(h2, dx)
        T = 0.002 * np.exp(-0.008 * i)
        if c2 < c or rng.random() < np.exp(-(c2 - c) / max(T, 1e-12)):
            h, c = h2, c2
            stall = 0
        else:
            stall += 1
        if c2 < bc:
            bh, bc = h2.copy(), c2
        if stall > 200:
            break
    return bh, bc

def coord_desc(h0, dx, ts, rounds, dl, rng):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h, dx)
    for r in range(rounds):
        if dl is not None and time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in (0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.08, -0.08, 0.12, -0.12):
                h[j] = sv + d
                if h[j] < -0.02 or h[j] > 1.02:
                    continue
                h = proj(h, ts)
                c2 = compute_c5(h, dx)
                if c2 < c - 1e-15:
                    c = c2
                    imp = True
                    break
            else:
                h[j] = sv
                h = proj(h, ts)
        if not imp:
            break
    return h, c

def transfer_hc(h0, dx, ts, niter, dl, rng):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h, dx)
    bh, bc = h.copy(), c
    for i in range(niter):
        if dl is not None and time.time() > dl:
            break
        i1, i2 = rng.randint(n), rng.randint(n)
        if i1 == i2:
            continue
        d = rng.uniform(0.03, 0.2) * (1 if rng.random() < 0.5 else -1)
        sv1, sv2 = h[i1], h[i2]
        h[i1] = sv1 - d
        h[i2] = sv2 + d
        h = proj(h, ts)
        c2 = compute_c5(h, dx)
        T = 0.005 * np.exp(-0.003 * i)
        if c2 < c or rng.random() < np.exp(-(c2 - c) / max(T, 1e-12)):
            c = c2
        else:
            h[i1], h[i2] = sv1, sv2
            h = proj(h, ts)
        if c < bc:
            bh, bc = h.copy(), c
    return bh, bc

def make_starts(n, ts, rng, ref):
    out = []
    if ref is not None:
        if len(ref) == n:
            out.append(ref.copy())
        else:
            out.append(np.interp(np.linspace(0, 1, n, endpoint=False),
                                 np.linspace(0, 1, len(ref), endpoint=False), ref))
    out.append(np.full(n, ts / n))
    x = np.arange(n) / n
    for m in (1, 2, 3, 5):
        out.append(0.5 + 0.35 * np.sin(2 * np.pi * m * x + rng.uniform(0, 6.28)))
    p = max(2, n // 4)
    out.append(np.array([1.0 if (i % (2 * p)) < p else 0.0 for i in range(n)]))
    p2 = max(2, n // 3)
    out.append(np.array([1.0 if (i % (2 * p2)) < p2 // 2 else 0.0 for i in range(n)]))
    for _ in range(2):
        h = np.zeros(n)
        perm = rng.permutation(n)
        h[perm[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[perm[int(ts)]] = fr
        out.append(h)
    return [proj(s, ts) for s in out]

def optimize_one(h0, dx, ts, dl, rng):
    h = smooth_slsqp(h0, dx, ts, 10, 250, 200)
    c = compute_c5(h, dx)
    if dl is not None and time.time() < dl - 30:
        h2, c2 = grad_desc(h, dx, ts, 1000, dl - 10, rng)
        if c2 < c:
            h, c = h2, c2
    if dl is not None and time.time() < dl - 15:
        h3, c3 = coord_desc(h, dx, ts, 50, dl - 5, rng)
        if c3 < c:
            h, c = h3, c3
    if dl is not None and time.time() < dl - 10:
        h4, c4 = transfer_hc(h, dx, ts, 2000, dl - 5, rng)
        if c4 < c:
            h, c = h4, c4
    if dl is not None and time.time() < dl - 5:
        h5 = smooth_slsqp(h, dx, ts, 50, 500, 120)
        c5 = compute_c5(h5, dx)
        if c5 < c:
            h, c = h5, c5
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dlim = t0 + budget_s - 5
    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None
    best_h, best_c = None, 1e9
    base_n = len(hr) if hr is not None else 50

    if hr is not None:
        ts0 = base_n / 2.0
        dx0 = 2.0 / base_n
        best_h, best_c = optimize_one(proj(hr, ts0), dx0, ts0, dlim, rng)

    n_list = sorted(set([base_n + d for d in range(-3, 11) if base_n + d >= 35 and base_n + d <= 75]))
    for nt in n_list:
        if time.time() > dlim - 80:
            break
        if nt == base_n and hr is not None:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        starts = make_starts(nt, tst, rng, best_h)
        for hs in starts[:5]:
            if time.time() > dlim - 40:
                break
            h_r, c_r = optimize_one(hs, dxt, tst, dlim - 15, rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    if best_h is not None and time.time() < dlim - 100:
        nb, dxb, tsb = len(best_h), 2.0 / len(best_h), len(best_h) / 2.0
        for amp in (0.12, 0.06, 0.03, 0.015, 0.008):
            if time.time() > dlim - 40:
                break
            hp = proj(best_h + amp * rng.randn(nb), tsb)
            h_r, c_r = optimize_one(hp, dxb, tsb, dlim - 15, rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    if best_h is not None:
        nf, dxf, tsf = len(best_h), 2.0 / len(best_h), len(best_h) / 2.0
        hf = smooth_slsqp(best_h, dxf, tsf, 40, 600, 150)
        cf = compute_c5(hf, dxf)
        if cf < best_c:
            best_h, best_c = hf, cf
        if time.time() < dlim - 3:
            hd, cd = coord_desc(best_h, dxf, tsf, 25, t0 + budget_s - 1, rng)
            if cd < best_c:
                best_h, best_c = hd, cd

    if best_h is None:
        best_h = proj(np.full(50, 0.5), 25.0)
        best_c = compute_c5(best_h, 0.04)

    best_h = proj(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c, len(best_h)
