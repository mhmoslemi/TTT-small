    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def try_update(state, h_cand, n_cand):
    h_cand = fix_sum(h_cand, n_cand)
    c5_cand = compute_c5(h_cand)
    if c5_cand < state['c5'] - 1e-15:
        state['h'] = h_cand
        state['n'] = n_cand
        state['c5'] = c5_cand
        return True
    return False

def make_perturbation(h, n, ptype, ps, rng):
    h_p = h.copy()
    if ptype == 'noise':
        h_p += rng.randn(n) * ps
    elif ptype == 'sine':
        for f in [2, 5, 11, 19, 31]:
            h_p += ps * 0.4 * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.283))
    elif ptype == 'block':
        bw = max(3, n // 8)
        for _ in range(4):
            start = rng.randint(0, max(1, n - bw))
            h_p[start:start + bw] += ps * rng.choice([-1, 1])
    elif ptype == 'mirror':
        shift = rng.randint(n // 6, n // 3)
        h_p = np.roll(h_p, shift)
        h_p += rng.randn(n) * ps * 0.3
    return fix_sum(np.clip(h_p, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    h_best = fix_sum(h_best, n_best)
    state = {'h': h_best, 'n': n_best, 'c5': compute_c5(h_best)}

    def elapsed():
        return time.time() - t_start

    # Phase 1: L-BFGS-B temperature sweep (main optimization)
    T_sweep = [0.10, 0.04, 0.015, 0.006, 0.002, 0.0008, 0.0003]
    for T in T_sweep:
        if budget_s - elapsed() < 150:
            break
        h_opt = do_lbfgsb(state['h'], state['n'], T, max_iter=4000)
        try_update(state, h_opt, state['n'])

    # Phase 2: Projected gradient for different basin
    if budget_s - elapsed() > 120:
        h_pg = projected_gradient(state['h'], state['n'], T=0.003, n_iter=80, lr_init=0.005)
        try_update(state, h_pg, state['n'])

    # Phase 3: Structured multi-start
    starts = [('noise', 0.02), ('noise', 0.06), ('noise', 0.12),
              ('sine', 0.04), ('sine', 0.10),
              ('block', 0.06), ('block', 0.12),
              ('mirror', 0.03), ('noise', 0.20)]
    for ptype, ps in starts:
        if budget_s - elapsed() < 250:
            break
        h_p = make_perturbation(state['h'], state['n'], ptype, ps, rng)
        h_o1 = do_lbfgsb(h_p, state['n'], 0.02, max_iter=2000)
        h_o2 = do_lbfgsb(h_o1, state['n'], 0.002, max_iter=2500)
        try_update(state, h_o2, state['n'])

    # Phase 4: Multi-resolution exploration
    for n_try in [300, 400, 480, 540]:
        if budget_s - elapsed() < 250:
            break
        if n_try > state['n']:
            h_init = resample(state['h'], n_try)
        else:
            h_init = resample(state['h'], n_try)
        h_r1 = do_lbfgsb(h_init, n_try, 0.01, max_iter=2500)
        h_r2 = do_lbfgsb(h_r1, n_try, 0.001, max_iter=2500)
        if try_update(state, h_r2, n_try):
            pass

    # Phase 5: Fine polish at best resolution
    for T in [0.0005, 0.0002, 0.00008, 0.00003]:
        if budget_s - elapsed() < 50:
            break
        h_opt = do_lbfgsb(state['h'], state['n'], T, max_iter=5000)
        try_update(state, h_opt, state['n'])

    # Phase 6: Micro-perturbation + polish
    if budget_s - elapsed() > 60:
        h_micro = state['h'] + rng.randn(state['n']) * 0.008
        h_micro = fix_sum(np.clip(h_micro, 0.0, 1.0), state['n'])
        h_mp = do_lbfgsb(h_micro, state['n'], 0.0005, max_iter=3000)
        try_update(state, h_mp, state['n'])

    # Phase 7: Final projected gradient nudge
    if budget_s - elapsed() > 40:
        h_ng = projected_gradient(state['h'], state['n'], T=0.0002, n_iter=40, lr_init=0.002)
        try_update(state, h_ng, state['n'])

    h_final = fix_sum(state['h'], state['n'])
    c5_final = compute_c5(h_final)
    return (h_final, c5_final, state['n'])
