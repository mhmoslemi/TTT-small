        return budget_s - (time.time() - t0)

    # Initialize from parent or random
    try:
        h_init = np.array(initial_h_values, dtype=float)
        if len(h_init) != n:
            xs = (np.arange(len(h_init)) + 0.5) * 2.0 / len(h_init)
            xd = (np.arange(n) + 0.5) * 2.0 / n
            h_init = np.interp(xd, xs, h_init)
        h_cont = proj(h_init, n)
    except Exception:
        h_cont = proj(np.full(n, 0.5) + rng.randn(n) * 0.1, n)

    best_h = h_cont.copy()
    best_c = compute_c5(best_h)

    # === Phase 1: Binary local search (45% budget) ===
    h_bin = (h_cont > 0.5).astype(float)
    h_bin = proj(h_bin, n)
    if compute_c5(h_bin) < best_c:
        best_c = compute_c5(h_bin)
        best_h = h_bin.copy()

    # Multiple SA restarts
    n_restarts = 4
    for restart in range(n_restarts):
        if tleft() < 100:
            break
        if restart == 0:
            h_sa = h_bin.copy()
        else:
            h_sa = h_bin.copy()
            np.random.seed(seed + restart * 137)
            for _ in range(20):
                i, j = rng.choice(n, 2, replace=False)
                if h_sa[i] != h_sa[j]:
                    h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        frac = 0.35 if restart == 0 else 0.08
        h_sa, sa_c = binary_sa(h_sa, n, rng, t0, tleft(), frac)
        if sa_c < best_c:
            best_c = sa_c
            best_h = h_sa.copy()
            h_bin = h_sa.copy()

    # === Phase 2: Directed binary refinement (10% budget) ===
    if tleft() > 40:
        h = h_bin.copy()
        cur_c = compute_c5(h)
        deadline = t0 + budget_s * 0.84
        while time.time() < deadline:
            if directed_swap(h, n, rng, 1):
                new_c = compute_c5(h)
                if new_c < cur_c:
                    cur_c = new_c
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                        h_bin = h.copy()
                else:
                    # undo - find what we did is hard, just recompute from best
                    pass
            else:
                a = rng.randint(n)
                b = rng.randint(n)
                if h[a] != h[b]:
                    h[a], h[b] = h[b], h[a]
                    new_c = compute_c5(h)
                    if new_c < cur_c:
                        cur_c = new_c
                        if cur_c < best_c:
                            best_c = cur_c
                            best_h = h.copy()
                    else:
                        h[a], h[b] = h[b], h[a]

    # === Phase 3: Continuous SLSQP from binary (30% budget) ===
    if tleft() > 80:
        h = h_bin.copy()
        for topk in [5, 10, 15, 30]:
            if tleft() < 20:
                break
            for t_val in [40, 150, 600, 2500]:
                if tleft() < 12:
                    break
                try:
                    hc = slsqp_call(h, n, obj_topk, (t_val, topk), 400, 1e-15)
                    c5 = compute_c5(hc)
                    if c5 < best_c:
                        best_c = c5
                        best_h = hc
                        h = hc
                except Exception:
                    pass

    # === Phase 4: p-norm refinement (10% budget) ===
    if tleft() > 40:
        h = best_h.copy()
        for p in [8, 30, 100]:
            if tleft() < 12:
                break
            for topk in [8, 20]:
                if tleft() < 8:
                    break
                try:
                    hc = slsqp_call(h, n, obj_pnorm, (p, topk), 300, 1e-15)
                    c5 = compute_c5(hc)
                    if c5 < best_c:
                        best_c = c5
                        best_h = hc
                        h = hc
                except Exception:
                    pass

    # === Phase 5: Fine polish (5% budget) ===
    if tleft() > 15:
        h = best_h.copy()
        for t_val in [5000, 20000, 100000]:
            if tleft() < 8:
                break
            try:
                hc = slsqp_call(h, n, obj_topk, (t_val, 3), 1500, 1e-17)
                c5 = compute_c5(hc)
                if c5 < best_c:
                    best_c = c5
                    best_h = hc
            except Exception:
                pass

    # Final verification
    best_h = proj(best_h, n)
    final_c = compute_c5(best_h)
    return best_h, final_c, n
