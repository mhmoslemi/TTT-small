import numpy as np
import scipy.optimize as opt
import time

def compute_c5_linear(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def circular_corr(h):
    n = len(h)
    H = np.fft.rfft(h)
    G = np.fft.rfft(1.0 - h)
    return np.fft.irfft(H * np.conj(G), n=n)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff <= 1e-12:
                break
    return h

def softmax_obj_fn(x, T, top_k):
    n = len(x)
    dx = 2.0 / n
    R = circular_corr(x)
    idx = np.argpartition(R, -top_k)[-top_k:]
    vals = R[idx]
    c_max = np.max(vals)
    if T < 1e-14:
        return c_max * dx
    shifted = vals - c_max
    shifted = np.clip(shifted, -40.0 * T, 0.0)
    w_sum = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(w_sum, 1e-300))) * dx

def softmax_grad_fn(h, T, top_k):
    n = len(h)
    dx = 2.0 / n
    R = circular_corr(h)
    idx = np.argpartition(R, -top_k)[-top_k:]
    vals = R[idx]
    c_max = np.max(vals)
    js = np.arange(n)
    if T < 1e-14:
        m = int(np.argmax(vals))
        k = int(idx[m])
        return ((1.0 - h[(js + k) % n]) - h[(js - k) % n]) * dx
    shifted = vals - c_max
    shifted = np.clip(shifted, -40.0 * T, 0.0)
    w = np.exp(shifted / T)
    w = w / w.sum()
    grad = np.zeros(n)
    for m in range(top_k):
        k = int(idx[m])
        grad += w[m] * ((1.0 - h[(js + k) % n]) - h[(js - k) % n])
    return grad * dx

def adam_opt(h, n, T, top_k, steps, lr):
    h_c = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    b1, b2 = 0.85, 0.999
    R_init = circular_corr(h_c)
    best_f = float(np.max(R_init) * (2.0 / n))
    best_h = h_c.copy()
    patience = 0
    for t in range(1, steps + 1):
        g = softmax_grad_fn(h_c, T, top_k)
        m_arr = b1 * m_arr + (1.0 - b1) * g
        v_arr = b2 * v_arr + (1.0 - b2) * g * g
        m_hat = m_arr / (1.0 - b1**t)
        v_hat = v_arr / (1.0 - b2**t)
        h_c = h_c - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_c = fix_sum(h_c, n)
        if t % 5 == 0:
            R_cur = circular_corr(h_c)
            f_appr = float(np.max(R_cur) * (2.0 / n))
            if f_appr < best_f - 1e-6:
                best_f = f_appr
                best_h = h_c.copy()
                patience = 0
            else:
                patience += 5
        if patience > 120:
            break
    f_lin_final = compute_c5_linear(h_c)
    f_lin_best = compute_c5_linear(best_h)
    if f_lin_final < f_lin_best:
        best_h = h_c.copy()
    return best_h

def slsqp_obj_wrap(x, T, top_k):
    return softmax_obj_fn(x, T, top_k)

def slsqp_con_wrap(x, n):
    return np.sum(x) - n * 0.5

def slsqp_opt(h, n, T, top_k, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': slsqp_con_wrap, 'args': (n,)}
    res = opt.minimize(
        slsqp_obj_wrap, h.copy(), args=(T, top_k),
        method='SLSQP', bounds=bounds, constraints=[cons],
        options={'maxiter': max_iter, 'ftol': 1e-16}
    )
    if hasattr(res, 'x') and res.x is not None:
        return fix_sum(res.x, n)
    return h

def resample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n, dtype=float)
    x_new = np.arange(new_n, dtype=float)
    h_new = np.interp(x_new, x_old, h, period=old_n)
    return fix_sum(h_new, new_n)

def make_structured_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    n_segs = rng.choice([2, 3, 4])
    total_width = n // 2
    seg_widths = rng.dirichlet(np.ones(n_segs)) * total_width
    seg_widths = seg_widths.astype(int)
    seg_widths[0] += total_width - seg_widths.sum()
    pos = 0
    for sw in seg_widths:
        if pos + sw <= n:
            h[pos:pos+sw] = 1.0
        pos += sw
    h = h + rng.randn(n) * 0.05
    return fix_sum(np.clip(h, 0.0, 1.0), n)
