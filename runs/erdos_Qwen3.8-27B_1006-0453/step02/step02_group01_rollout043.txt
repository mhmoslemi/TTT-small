```python
def subgradient_step(h, step_size):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    d = k_star - (n - 1)  # lag offset
    
    g = np.zeros(n)
    if d == 0:
        g = 1.0 - 2.0 * h
    elif d > 0:
        # c = sum_i h[i+d]*(1-h[i])
        # dg/dh[j] = (1-h[j-d]) if j>=d else 0) - (h[j+d] if j+d<n else 0)
        for j in range(n):
            if j >= d:
                g[j] += (1.0 - h[j - d])
            if j + d < n:
                g[j] -= h[j + d]
    else:
        d_abs = -d
        # c = sum_i h[i-d_abs]*(1-h[i])
        # dg/dh[j] = (1-h[j+d_abs]) if j+d_abs<n else 0) - (h[j-d_abs] if j>=d_abs else 0)
        for j in range(n):
            if j + d_abs < n:
                g[j] += (1.0 - h[j + d_abs])
            if j >= d_abs:
                g[j] -= h[j - d_abs]
    
    h_new = h - step_size * g * dx
    return project_feasible(h_new, n)
```

The overlap integral at lag k is ∫ h(x)(1-h(x+k)) dx. For this to be small for all k, we need h(x) and h(x+k) to be "anti-correlated" for all k. This is like a correlation sequence problem.

For binary h with 45 ones and 45 zeros out of 90:
- At lag 0: ∑ h[i](1-h[i]) = 0 (since h is binary)
- At lag 1: ∑ h[i+1](1-h[i]) = number of (0,1) adjacent pairs
- At lag k: number of positions i where h[i+k]=1 and h[i]=0

So for binary, C₅ = max_k (number of (0→1) transitions at lag k) * dx

This is exactly the "autocorrelation" problem for binary sequences! We want to minimize the maximum cross-correlation between the sequence and its complement at all lags.

This is related to Barker sequences and other low-autocorrelation sequences. The theoretical minimum for a binary sequence of length n with n/2 ones is known in some cases.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += room * (diff / tot)
        else:
            room = h.copy()
            tot = room.sum()
            if tot < 1e-14: break
            h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = corr * dx
    m = cs.max()
    return float(m + np.log(np.sum(np.exp(T * (cs - m)))) / T)

def sum_con(x):
    return x.sum() - len(x) / 2.0

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def erdos_ineq(x):
    n = len(x) - 1
    t = x[-1]
    corr = np.correlate(x[:n], 1.0 - x[:n], mode='full')
    return t - corr

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(corr.max()) + 0.01
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq},
            {'type': 'ineq', 'fun': erdos_ineq}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except:
        return h

def lbfgsb_smooth(h, T, penalty, maxiter=300):
    n = len(h)
    def obj(x):
        base = smooth_c5(x, T)
        sv = (x.sum() - n / 2.0) ** 2
        return base + penalty * sv
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def to_binary(h, n):
    target = int(n / 2)
    idx = np.argsort(-h)[:target]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def binary_local_search(hb, max_passes=500, deadline=None):
    n = len(hb)
    best_c5 = compute_c5(hb)
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        ones = list(np.where(hb == 1.0)[0])
        zeros = list(np.where(hb == 0.0)[0])
        improved = False
        for i in ones:
            if hb[i] != 1.0: continue
            for j in zeros:
                if hb[j] != 0.0: continue
                hb[i], hb[j] = 0.0, 1.0
                c5 = compute_c5(hb)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                else:
                    hb[i], hb[j] = 1.0, 0.0
        if not improved:
            break
    return hb.copy(), best_c5

def subgradient_opt(h, n_iter=500, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.1
    for it in range(n_iter):
        if deadline and time.time() > deadline:
            break
        h_new = subgradient_step(h, step)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
        if it % 100 == 99:
            step *= 0.8
    return best_h, best_c5

def subgradient_step(h, step_size):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ks = int(np.argmax(corr))
    d = ks - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[d:] += 1.0 - h[:n - d]
        g[:n - d] -= h[d:]
    else:
        da = -d
        g[:n - da] += 1.0 - h[da:]
        g[da:] -= h[:n - da]
    h_new = h - step_size * g * dx
    return project_feasible(h_new, n)

def random_binary(n, rng):
    pos = rng.choice(n, size=n // 2, replace=False)
    hb = np.zeros(n)
    hb[pos] = 1.0
    return hb
```

```python
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15: break
            h += r * (d / t)

            t = r.sum()
            if t < 1e-15: break
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h += r * (d / t)
        else:
            r = h.copy()
            t = r.sum()
            if t < 1e-15:
                break
            h += r * (d / t)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = corr * dx
    m = cs.max()
    return float(m + np.log(np.sum(np.exp(T * (cs - m)))) / T)

def lbfgsb_obj(x, n, T, pen):
    return smooth_c5(x, T) + pen * (x.sum() - n * 0.5) ** 2

def eq_con(x):
    n = len(x) - 1
    return x[:n].sum() - n * 0.5

def ineq_con(x):
    n = len(x) - 1
    t = x[-1]
    c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
    return t - c

def obj_t(x):
    return x[-1]

def slsqp_solve(h, maxiter=500):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(c.max()) + 0.02
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': eq_con}, {'type': 'ineq', 'fun': ineq_con}]
    try:
        r = minimize(obj_t, x0, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(r.x[:n], 0, 1), n)
    except Exception:
        return h

def lbfgsb_solve(h, T, pen, maxiter=400):
    n = len(h)
    try:
        r = minimize(lbfgsb_obj, h, args=(n, T, pen), method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def to_binary(h):
    n = len(h)
    idx = np.argsort(-h)[:n // 2]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def bin_ls(hb, rng, max_passes=300, deadline=None):
    n = len(hb)
    best = hb.copy()
    best_c5 = compute_c5(best)
    cur = hb.copy()
    stuck = 0
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        imp = False
        for i in ones:
            if cur[i] != 1.0:
                continue
            for j in zeros:
                if cur[j] != 0.0:
                    continue
                cur[i], cur[j] = 0.0, 1.0
                c5 = compute_c5(cur)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best = cur.copy()
                    imp = True
                else:
                    cur[i], cur[j] = 1.0, 0.0
        if not imp:
            cur = best.copy()
            stuck += 1
            if stuck >= 3:
                if len(ones) > 0 and len(zeros) > 0:
                    si = int(rng.choice(ones))
                    sj = int(rng.choice(zeros))
                    cur[si], cur[sj] = 0.0, 1.0
                stuck = 0
            continue
        stuck = 0
    return best, best_c5

def subgrad_step(h, step, eps_frac):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = corr * dx
    m = cs.max()
    active = np.where(cs >= m * (1.0 - eps_frac))[0]
    g = np.zeros(n)
    for ks in active:
        d = int(ks) - (n - 1)
        gk = np.zeros(n)
        if d == 0:
            gk[:] = 1.0 - 2.0 * h
        elif d > 0:
            gk[d:] += 1.0 - h[:n - d]
            gk[:n - d] -= h[d:]
        else:
            da = -d
            gk[:n - da] += 1.0 - h[da:]
            gk[da:] -= h[:n - da]
        g += gk / len(active)
    return project_feasible(h - step * g * dx, n)

def rand_binary(n, rng):
    pos = rng.choice(n, n // 2, replace=False)
    b = np.zeros(n)
    b[pos] = 1.0
    return b
```
