def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_sweep(h, n, n_iter=60):
    """One full coordinate descent sweep."""
    for i in range(n):
        old = h[i]
        # Set h[i] = 0, compute corr
        h[i] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        # Set h[i] = 1, compute corr
        h[i] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = old
        d = c1 - c0  # slope for each shift
        
        # Minimize max(c0 + t*d) for t in [0,1]
        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(n_iter):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(c0 + m1 * d)
            f2 = np.max(c0 + m2 * d)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        h[i] = (lo + hi) * 0.5
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h = h + room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-13:
                break
            h = h + room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq_fun(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun}]
    try:
        res = minimize(smooth_c5_val, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, n_sweeps=100, n_ternary=50):
    """Coordinate descent with projection."""
    for sweep in range(n_sweeps):
        for i in range(n):
            current = h[i]
            s = np.arange(-(n-1), n)
            idx_p = i + s
            idx_m = i - s
            vp = (idx_p >= 0) & (idx_p < n)
            vm = (idx_m >= 0) & (idx_m < n)
            a = np.zeros(2*n-1)
            a[vp] += 1.0 - h[idx_p[vp]]
            a[vm] -= h[idx_m[vm]]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c0 = corr - current * a
            lo, hi = 0.0, 1.0
            for _ in range(n_ternary):
                m1 = lo + (hi-lo)/3.0
                m2 = hi - (hi-lo)/3.0
                if np.max(c0 + m1*a) < np.max(c0 + m2*a):
                    hi = m2
                else:
                    lo = m1
            h[i] = (lo+hi)*0.5
        h = project_feasible(h, n)
    return h

    t_start = time.time()

        return budget_s - (time.time() - t_start)
    
    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        best_h = np.zeros(90)

        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        else:
            break
    
    # Phase 2: SLSQP polish

        for T in [200, 1000, 5000]:
            if time_left() < budget_s * 0.25:

            h_try = slsqp_smooth(best_h, T, maxiter=400)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        # Additional coord descent after SLSQP
        h_try = coord_descent(best_h.copy(), best_n, n_sweeps=50, n_ternary=50)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    # Phase 3: Multi-resolution (go to higher n)
    if time_left() > budget_s * 0.30:
        for n_new in [120, 150, 200]:
            if time_left() < budget_s * 0.15:

            x_old = np.linspace(0, 2, best_n, endpoint=False) + 1.0/best_n
            x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_new)
            h_try = coord_descent(h_interp, n_new, n_sweeps=30, n_ternary=40)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                best_n = n_new

            best_h = h_try

    return best_h, best_c5, best_n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-13:
                break
            h -= h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, n_sweeps=100, n_ternary=40):
    """Exact coordinate descent exploiting 1D structure of max-correlation."""
    for sweep in range(n_sweeps):
        for i in range(n):
            lo, hi = 0.0, 1.0
            for _ in range(n_ternary):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[i] = m1
                c1 = np.max(np.correlate(h, 1.0 - h, mode='full'))
                h[i] = m2
                c2 = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            h[i] = (lo + hi) * 0.5
        h = project_feasible(h, n)
    return h

    np.random.seed(seed)
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    # Initialize from parent construction

    else:
        n0 = 90

        best_h[2 * n0 // 3:] = 0.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at parent resolution (main novel method)
    while time_left() > budget_s * 0.38:

        else:
