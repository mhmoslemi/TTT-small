def ga_binary(n, pop_size, generations, rng, initial_h=None, t_start=None, t_budget=None):

            ia = rng.randint(pop_size)
            ib = rng.randint(pop_size)
            p1 = pop[ia] if scores[ia] <= scores[ib] else pop[ib]
            ia = rng.randint(pop_size)

            p2 = pop[ia] if scores[ia] <= scores[ib] else pop[ib]

                excess = current_ones - ones_count
                one_pos = np.where(child == 1)[0]
                zero_pos = np.where(child == 0)[0]

    best_h, best_c5_val = local_search_binary(best_h, max_rounds=100)

def slsqp_smooth(h, T, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    
    def eq(x):
        return x.sum() - n / 2.0
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

                op = np.where(h_pert == 1)[0]
                zp = np.where(h_pert == 0)[0]
                if len(op) == 0 or len(zp) == 0:
                    break
                fi = rng.choice(op)
                fj = rng.choice(zp)
                h_pert[fi] = 0.0
                h_pert[fj] = 1.0
            h_ls, c5_ls = local_search_binary(h_pert, max_rounds=60)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 5: Real-valued SLSQP polish on best binary
    if elapsed() < budget_s * 0.88:
        h_real = best_h.astype(float).copy()
        best_n_r = len(h_real)
        for T in [300, 1500, 6000]:
            if elapsed() > budget_s * 0.88:
                break
            h_try = slsqp_smooth(h_real, T, maxiter=350)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                best_n = best_n_r
                h_real = h_try

    # Phase 6: One more local search on rounded best
    h_rounded = to_binary(best_h, best_n, rng)
    h_rounded, c5_rounded = local_search_binary(h_rounded, max_rounds=50)
    if c5_rounded < best_c5:
        best_c5 = c5_rounded
        best_h = h_rounded

    # Final SLSQP if time permits
    if elapsed() < budget_s * 0.96:
        h_final = slsqp_smooth(best_h.astype(float), 3000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = len(h_final)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
