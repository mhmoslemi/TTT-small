    n = len(h)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        s = int(i * old_n / new_n)
        e = max(s + 1, int((i + 1) * old_n / new_n))
        h_new[i] = np.mean(h[s:e])
    return proj_constraint(h_new, new_n)

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = opt.minimize(
            soft_max_obj, h0, args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0}
        )
        return proj_constraint(res.x, n)
    except Exception:
        return h0.copy()

def adaptive_temperature(h):
    corr = get_corr(h)
    srt = np.sort(corr)[::-1]
    gap = max(srt[0] - srt[1], 1e-6)
    T = gap / 6.0
    return float(np.clip(T, 0.0002, 0.02))

def do_adaptive(h0, n, time_limit, max_rounds=12):
    h = h0.copy()
    for _ in range(max_rounds):
        if time.time() >= time_limit:
            break
        T = adaptive_temperature(h)
        mi = 800 if n <= 400 else 500
        h_new = do_slsqp(h, n, T, mi)
        c_new = compute_c5(h_new)
        c_old = compute_c5(h)
        if c_new < c_old - 1e-10:
            h = h_new
        else:
            T2 = max(T * 0.4, 0.0001)
            h2 = do_slsqp(h, n, T2, 400)
            c2 = compute_c5(h2)
            if c2 < c_old:
                h = h2
    return h, compute_c5(h)

def block_swap(h, n, rng):
    h_new = h.copy()
    n_swaps = rng.randint(1, 4)
    for _ in range(n_swaps):
        bl = rng.randint(3, max(4, n // 12))
        i1 = rng.randint(0, max(1, n - bl))
        i2 = rng.randint(0, max(1, n - bl))
        if abs(i1 - i2) < bl:
            continue
        h_new[i1:i1+bl], h_new[i2:i2+bl] = h_new[i2:i2+bl].copy(), h_new[i1:i1+bl].copy()
    return proj_constraint(h_new, n)

def region_invert(h, n, rng):
    h_new = h.copy()
    bl = rng.randint(4, max(5, n // 7))
    st = rng.randint(0, max(1, n - bl))
    h_new[st:st+bl] = 1.0 - h_new[st:st+bl]
    return proj_constraint(h_new, n)

def nudge_step(h, n, n_iter=200, rng=None):
    """Coordinate-wise nudging to reduce max correlation"""
    if rng is None:
        rng = np.random.RandomState(42)
    h_cur = h.copy()
    c_cur = compute_c5(h_cur)
    for _ in range(n_iter):
        idx = rng.randint(n)
        old_val = h_cur[idx]
        # Try a few perturbation sizes
        best_v = old_val
        best_c = c_cur
        for dv in [0.02, -0.02, 0.05, -0.05, 0.01, -0.01]:
            trial = old_val + dv
            if trial < 0 or trial > 1:
                continue
            h_cur[idx] = trial
            h_cur = proj_constraint(h_cur, n)
            c_try = compute_c5(h_cur)
            if c_try < best_c:
                best_c = c_try
                best_v = trial
        h_cur[idx] = best_v
        h_cur = proj_constraint(h_cur, n)
        c_cur = compute_c5(h_cur)
    return h_cur, c_cur
