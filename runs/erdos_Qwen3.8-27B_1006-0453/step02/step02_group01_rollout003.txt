import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / n)

def sum_con(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(smooth_obj, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def rdirect(x):
    return x[-1]

def req(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def rineq(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) * (2.0 / n) + 0.01
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': req}, {'type': 'ineq', 'fun': rineq}]
    try:
        res = minimize(rdirect, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def pair_flip_search(h, max_rounds=5):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        for i in ones:
            for j in zeros:
                h_trial = best.copy()
                h_trial[i] = 0.0
                h_trial[j] = 1.0
                c5 = compute_c5(h_trial)
                if c5 < best_c5 - 1e-13:
                    best = h_trial
                    best_c5 = c5
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    if 'initial_h_values' in dir() and initial_h_values is not None:

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive smooth SLSQP
    for T in [30, 150, 800, 4000]:
        if elapsed() > budget_s * 0.10:
            break

            best_h = h_try

    # Phase 2: Direct SLSQP

    # Phase 3: Pair flip + re-optimize
