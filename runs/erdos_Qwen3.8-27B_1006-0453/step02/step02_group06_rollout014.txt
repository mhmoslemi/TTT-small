            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 3: SLSQP refinement at n=336
    if time_rem() > 60:
        for T in [0.003, 0.001, 0.0005]:
            if time_rem() < 25:
                break
            h_opt = slsqp_step(h_best, n_best, T, 800)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 4: Upsample to n=448 and optimize
    if time_rem() > 120:
        n_up = 448
        h_up = upsample(h_best, n_up)
        for T in [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time_rem() < 30:
                break
            h_opt = lbfgs_step(h_up, n_up, T, 50.0, 1200)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c
                n_best = n_up
                h_up = h_best.copy()
        # SLSQP polish at higher res
        if time_rem() > 40:
            for T in [0.002, 0.0008]:
                if time_rem() < 20:
                    break
                h_opt = slsqp_step(h_best, n_best, T, 600)
                c = compute_c5(h_opt)
                if c < c5_best:
                    h_best = h_opt
                    c5_best = c

    # Phase 5: Try n=504 if time permits
    if time_rem() > 150 and n_best < 504:
        n_up2 = 504
        h_up2 = upsample(h_best, n_up2)
        for T in [0.02, 0.008, 0.003, 0.001, 0.0005]:
            if time_rem() < 30:
                break
            h_opt = lbfgs_step(h_up2, n_up2, T, 50.0, 1000)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c
                n_best = n_up2
                h_up2 = h_best.copy()

    # Phase 6: Multi-start perturbations at best resolution
    if time_rem() > 80:
        rng = np.random.RandomState(seed + 12345)
        for i in range(8):
            if time_rem() < 40:
                break
            sigma = 0.008 + 0.006 * (i % 3)
            h_pert = np.clip(h_best + rng.randn(n_best) * sigma, 0, 1)
            h_pert = proj_constraint(h_pert, n_best)
            h_opt = lbfgs_step(h_pert, n_best, 0.004, 40.0, 600)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 7: Final SLSQP ultra-fine polish
    if time_rem() > 35:
        for T in [0.0008, 0.0003, 0.0001]:
            if time_rem() < 15:
                break
            h_opt = slsqp_step(h_best, n_best, T, 1500)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 8: SA micro-refinement
    if time_rem() > 20:
        rng2 = np.random.RandomState(seed + 777)
        h_sa = h_best.copy()
        c_sa = c5_best
        T_sa = 0.0002
        n_steps = 1500
        for step in range(n_steps):
            if time_rem() < 5:
                break
            n_blk = rng2.randint(1, 4)
            idxs = rng2.choice(n_best, size=n_blk, replace=False)
            h_try = h_sa.copy()
            h_try[idxs] = np.clip(h_try[idxs] + rng2.randn(n_blk) * 0.015, 0, 1)
            h_try = proj_constraint(h_try, n_best)
            c_try = compute_c5(h_try)
            if c_try < c_sa or rng2.rand() < np.exp(-(c_try - c_sa) / T_sa):
                h_sa = h_try
                c_sa = c_try
                if c_try < c5_best:
                    h_best = h_try.copy()
                    c5_best = c_try
            T_sa *= 0.9995

    h_best = proj_constraint(h_best, n_best)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n_best)
