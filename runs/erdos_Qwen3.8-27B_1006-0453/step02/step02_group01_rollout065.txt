from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_direct(h, maxiter=500):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.05
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    t_start = time.time()
    try:
        res = minimize(
            lambda x: x[-1], x0, method='SLSQP', bounds=bnds,
            constraints=[
                {'type': 'eq', 'fun': lambda x: x[:n].sum() - n / 2.0},
                {'type': 'ineq', 'fun': lambda x: x[-1] - np.max(np.correlate(x[:n], 1.0 - x[:n], mode='full'))}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    dx = 2.0 / n
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def to_binary(h, n):
    """Round h to binary with exactly n/2 ones"""
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:n // 2]] = 1.0
    return b

def local_search_binary(h_bin, max_sweeps=200):
    """Local search: try all swaps, pick best improving"""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = compute_c5(cur)
    improved = True
    sweeps = 0
    while improved and sweeps < max_sweeps:
        improved = False
        sweeps += 1
        best_swap = None
        best_val = cur_c5
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        for i in ones:
            for j in zeros:
                cur[i] = 0
                cur[j] = 1
                val = compute_c5(cur)
                cur[i] = 1
                cur[j] = 0
                if val < best_val - 1e-15:
                    best_val = val
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            cur[i] = 0
            cur[j] = 1
            cur_c5 = best_val
            improved = True
    return cur, cur_c5

def sim_anneal_binary(h_bin, t_start, t_end, n_iter, rng):
    """Simulated annealing on binary solutions"""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cur[i] = 0
        cur[j] = 1
        new_c5 = compute_c5(cur)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_c5

        # Polish with SLSQP from binary solution

        h_pol2 = slsqp_smooth(best_h, T=500, maxiter=300)

        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2

        # Polish

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:n // 2]] = 1.0
    return b

def local_search_binary(h_bin, max_sweeps=200):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = compute_c5(cur)
    improved = True
    sweeps = 0
    while improved and sweeps < max_sweeps:
        improved = False
        sweeps += 1
        best_delta = 1e-15
        best_swap = None
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                val = compute_c5(cur)
                cur[i] = 1.0
                cur[j] = 0.0

                    best_swap = (i, j)

            i, j = best_swap
            cur[i] = 0.0
            cur[j] = 1.0
            cur_c5 = cur_c5 - best_delta
            improved = True
    return cur, cur_c5

def slsqp_direct(h, maxiter=400):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.05
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(
            lambda x: x[-1], x0, method='SLSQP', bounds=bnds,
            constraints=[
                {'type': 'eq', 'fun': lambda x: x[:n].sum() - n / 2.0},
                {'type': 'ineq', 'fun': lambda x: x[-1] - np.max(np.correlate(x[:n], 1.0 - x[:n], mode='full'))}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def slsqp_smooth(h, T, maxiter=300):

    dx = 2.0 / n
    def obj(x, T):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

    try:

                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgradient_step(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    # Gradient of f_k*(h) wrt h
    # f_k(h) = sum_i h[i]*(1-h[i+k]) for valid (i, i+k)
    # df/dh[i] = (1-h[i+k]) for valid k, and -h[i-k] for valid -k
    grad = np.zeros(n)
    # For the 'full' correlation, lag k means:
    # corr[k] = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])  for k >= 0
    # corr[2n-1-k] = sum_{i=k}^{n-1} h[i] * (1-h[i-k])  for k >= 0 (by symmetry)
    # Actually, np.correlate(a, b, 'full')[k] = sum_i a[i+k] * b[i] for valid i
    # So corr[k] = sum_i h[i+k] * (1-h[i])
    # d(corr[k])/d(h[j]) = (1-h[j-k]) if j>=k, plus -h[j+(2n-1-k)-j]... 
    # This gets complicated. Let me just use finite differences or the explicit formula.
    # corr[k] = sum_{i=max(0,k-(n-1))}^{min(n-1, n-1+k-n)} h[i+k-(n-1)] * (1-h[i])
    # Actually for np.correlate(h, 1-h, 'full'):
    # result[m] = sum_i h[i] * (1-h)[i + (n-1) - m] for valid indices
    # = sum_i h[i] * (1 - h[i + n - 1 - m])
    
    # Let's use the standard formula:
    # np.correlate(a, v, 'full')[k] = sum_i a[i+k] * v[i] for k = -(len(v)-1) ... len(a)-1
    # With the 'full' mode indexing: index j corresponds to lag j - (len(v)-1)
    # So result[j] = sum_i a[i + j - (len(v)-1)] * v[i]
    
    # For our case: a = h, v = 1-h, both length n
    # result[j] = sum_i h[i + j - (n-1)] * (1 - h[i])
    
    k = k_star - (n - 1)  # actual lag
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        for i in range(n - k):
            grad[i] += -(h[i + k])  # d/dh[i] of h[i+k]*(1-h[i]) = -h[i+k]
            grad[i + k] += (1 - h[i])  # d/dh[i+k]
    else:
        k2 = -k
        # corr = sum_{i=k2}^{n-1} h[i] * (1-h[i-k2])
        for i in range(k2, n):
            grad[i] += (1 - h[i - k2])
            grad[i - k2] += -h[i]
    
    grad = grad * dx
    new_h = h - alpha * grad
    return project_feasible(new_h, n), k_star

def slsqp_direct(h, maxiter=400):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.05
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': erdos_eq, 'args': (n,)},
        {'type': 'ineq', 'fun': erdos_ineq, 'args': (n,)}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgradient_step(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[:m] += (1.0 - h[lag:lag + m])
        grad[lag:lag + m] -= h[:m]
    else:
        l = -lag
        m = n - l
        grad[l:] += (1.0 - h[:m])
        grad[:m] -= h[l:]
    new_h = h - alpha * grad * dx
    return project_feasible(new_h, n)

        h_base = project_feasible(h_base, n_try)
        h_bin = to_binary(h_base, n_try)
        h_ls, c5_ls = local_search_binary(h_bin, max_sweeps=100)
        h_sa, _ = sim_anneal_binary(h_ls, 0.007, 0.0005, 4000, rng)
        h_ls2, c5_ls2 = local_search_binary(h_sa, max_sweeps=50)
        cand = h_ls2 if c5_ls2 < c5_ls else h_ls
        cand_c5 = min(c5_ls2, c5_ls)
        if cand_c5 < best_c5:
            best_c5, best_h, best_n = cand_c5, cand.copy(), n_try
        h_pol, c5_pol = polish(cand, t0 + budget_s * 0.82)
        if c5_pol < best_c5:
            best_c5, best_h, best_n = c5_pol, h_pol, n_try

    # Phase 5: Subgradient + periodic binary check
    if time.time() - t0 < budget_s * 0.88:
        h_sub = best_h.copy()
        alpha = 0.1
        for it in range(300):
            if time.time() - t0 > budget_s * 0.88:
                break
            h_sub = subgradient_step(h_sub, alpha)
            alpha *= 0.995
            if it % 30 == 29:
                c5_ch = compute_c5(h_sub)
                if c5_ch < best_c5:
                    best_c5, best_h = c5_ch, h_sub.copy()
                h_b = to_binary(h_sub, len(h_sub))
                h_l, c5_l = local_search_binary(h_b, max_sweeps=15)
                if c5_l < best_c5:
                    best_c5, best_h = c5_l, h_l.copy()
                    h_sub = h_l.copy()

    # Phase 6: Final intensive polish
    if time.time() - t0 < budget_s * 0.96:
        h_final, c5_final = polish(best_h, t0 + budget_s * 0.98)
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
