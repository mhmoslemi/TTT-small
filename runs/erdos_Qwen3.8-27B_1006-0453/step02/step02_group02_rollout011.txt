    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -topk)[-topk:]
    c = corr[idx]
    mx = np.max(c)
    s = t * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def slsqp_polish(h, n, t_val, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    ncorr = 2 * n - 1
    if topk and topk < ncorr:
        r = minimize(obj_topk, h, args=(t_val, topk), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
    else:
        r = minimize(obj_smooth, h, args=(t_val,), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    N = 280
    dx = 2.0 / N

    def tleft():
        return budget_s - (time.time() - t0)

    # Initialize from parent or random
    try:
        h_init = proj(np.array(initial_h_values, dtype=float), N)
    except Exception:
        h_init = proj(np.full(N, 0.5) + rng.randn(N) * 0.1, N)

    h_bin = to_binary(h_init, N)
    corr0 = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    best_c5 = float(np.max(corr0) * dx)
    best_h = h_bin.copy()

    # --- Binary Simulated Annealing ---
    def sa_run(h_start, T_start, T_end, n_steps, use_gu):
        nonlocal best_h, best_c5
        h = h_start.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = float(np.max(corr) * dx)
        ones = np.where(h > 0.5)[0].tolist()
        zeros = np.where(h < 0.5)[0].tolist()
        T = T_start
        cool_block = 20000
        cool_rate = (T_end / T_start) ** (cool_block / float(n_steps))
        stall = 0

        for step in range(n_steps):
            if step % cool_block == 0:
                if tleft() < 15:
                    break
                T *= cool_rate
                if stall > 80000:
                    # Kick: random restart with perturbation
                    for _ in range(20):
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        h[i] = 0.0; h[j] = = 1.0
                        ones.remove(i); ones.append(j)
                        zeros.remove(j); zeros.append(i)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    c5_cur = float(np.max(corr) * dx)
                    stall = 0

            # Move selection
            if use_gu and rng.random() < 0.35:
                max_idx = int(np.argmax(corr))
                shift = max_idx - (N - 1)
                i = ones[rng.randint(len(ones))]
                j_target = (i - shift) % N
                found = False
                for off in [0, 1, -1, 2, -2, 3, -3, 4, -4]:
                    jj = (j_target + off) % N
                    if h[jj] < 0.5:
                        j = jj
                        found = True
                        break
                if not found:
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

            h[i] = 0.0
            h[j] = 1.0
            corr_new = np.correlate(h, 1.0 - h, mode='full')
            c5_new = float(np.max(corr_new) * dx)

            delta = c5_new - c5_cur
            if delta < 0 or rng.random() < np.exp(delta / (T + 1e-10)):
                c5_cur = c5_new
                corr = corr_new
                ones[ones.index(i)] = j
                zeros[zeros.index(j)] = i
                stall = 0
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
                stall += 1

        return best_h.copy()

    # SA Phase 1: High temperature exploration
    if tleft() > 200:
        sa_run(h_bin, 0.10, 0.003, 400000, True)

    # SA Phase 2: Medium temperature with perturbation from best
    if tleft() > 150:
        h_ph = best_h.copy()
        ones_p = np.where(h_ph > 0.5)[0]
        zeros_p = np.where(h_ph < 0.5)[0]
        for _ in range(15):
            i = ones_p[rng.randint(len(ones_p))]
            j = zeros_p[rng.randint(len(zeros_p))]
            h_ph[i] = 0.0; h_ph[j] = 1.0
        sa_run(h_ph, 0.05, 0.001, 400000, True)

    # SA Phase 3: Low temperature fine-tuning
    if tleft() > 100:
        sa_run(best_h, 0.02, 0.0003, 350000, True)

    # SA Phase 4: Very low temperature
    if tleft() > 60:
        sa_run(best_h, 0.008, 0.00005, 250000, False)

    # --- Continuous SLSQP Polish ---
    h_cont = proj(best_h, N)
    c5_cont = compute_c5(h_cont)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont

    ncorr = 2 * N - 1
    for t_val in [100, 400, 1500, 5000, 15000]:
        if tleft() < 25:
            break
        try:
            hc = slsqp_polish(h_cont, N, t_val, None, 2500, 1e-15)
            c5_hc = compute_c5(hc)
            if c5_hc < best_c5:
                best_c5 = c5_hc
                best_h = hc
            h_cont = hc
        except Exception:
            pass

    # Top-k focused polish
    if tleft() > 40:
        for topk in [max(3, ncorr // 10), max(5, ncorr // 6)]:
            if tleft() < 20:
                break
            try:
                hc = slsqp_polish(h_cont, N, 5000, topk, 3000, 1e-15)
                c5_hc = compute_c5(hc)
                if c5_hc < best_c5:
                    best_c5 = c5_hc
                    best_h = hc
                h_cont = hc
            except Exception:
                pass

    # Final ultra-fine polish
    if tleft() > 15:
        try:
            hc = slsqp_polish(h_cont, N, 50000, None, 5000, 1e-17)
            c5_hc = compute_c5(hc)
            if c5_hc < best_c5:
                best_c5 = c5_hc
                best_h = hc
        except Exception:
            pass

    final_h = best_h
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, N
