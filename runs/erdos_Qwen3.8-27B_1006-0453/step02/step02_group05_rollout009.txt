    for _ in range(50):

        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_and_grad(h, N, tau, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * (corr - corr.max())
    w = np.exp(sc)
    wsum = w.sum()
    w /= wsum
    if top_k > 0 and top_k < len(w):
        idx = np.argsort(w)[-top_k:]
        mask = np.zeros_like(w)
        mask[idx] = w[idx]
        msum = mask.sum()
        if msum > 1e-30:
            w = mask / msum
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-6
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def nesterov_phase(h, N, target, deadline, tau0, tau1, lr, mu, top_k):
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    v = np.zeros(N)
    y = h.copy()
    it = 0
    while time.time() < deadline:
        frac = min(it / 5000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        y_new = y - mu * v
        grad, _ = corr_and_grad(y_new, N, tau, top_k)
        v = v - lr * grad
        y = y_new + v
        h_new = project_sum(y, target)
        c5_new = compute_c5(h_new, N)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h_new
        it += 1
    return best_h, best_c5

def local_refine(h, N, target, deadline, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    while time.time() < deadline:
        n_pert = min(max(N // 20, 5), 15)
        idxs = rng.choice(N, size=n_pert, replace=False)
        improved = False
        for i in idxs:
            for delta in [-0.08, -0.03, 0.03, 0.08, -0.015, 0.015]:
                h_trial = h.copy()
                h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
                h_trial = project_sum(h_trial, target)
                c5_t = compute_c5(h_trial, N)
                if c5_t < best_c5 - 1e-8:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
                    h = h_trial
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    sizes = sorted(set([s for s in [n0, 2 * n0, 92, 190, 380, 572]
                        if s % 2 == 0 and 40 <= s <= 1400]))

    for N in sizes:
        if time.time() > deadline - 60:
            break
        target = N // 2
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, float(target))

        remain = deadline - time.time()
        if remain < 50:
            break

        h_cur = h_init.copy()
        # Phase 1: broad (tau 1->40, no focus)
        if time.time() < deadline - 30:
            d1 = time.time() + remain * 0.22
            h_cur, c1 = nesterov_phase(h_cur, N, target, d1, 1.0, 40.0, 0.014, 0.84, 0)
            if c1 < best_c5:
                best_c5, best_h = c1, h_cur.copy()
        # Phase 2: medium (tau 40->200)
        if time.time() < deadline - 20:
            d2 = time.time() + (deadline - time.time()) * 0.22
            h_cur, c2 = nesterov_phase(h_cur, N, target, d2, 40.0, 200.0, 0.005, 0.90, 0)
            if c2 < best_c5:
                best_c5, best_h = c2, h_cur.copy()
        # Phase 3: tight top-4
        if time.time() < deadline - 15:
            d3 = time.time() + (deadline - time.time()) * 0.20
            h_cur, c3 = nesterov_phase(h_cur, N, target, d3, 200.0, 600.0, 0.0025, 0.92, 4)
            if c3 < best_c5:
                best_c5, best_h = c3, h_cur.copy()
        # Phase 4: very tight top-2
        if time.time() < deadline - 8:
            d4 = time.time() + (deadline - time.time()) * 0.15
            h_cur, c4 = nesterov_phase(h_cur, N, target, d4, 600.0, 2500.0, 0.001, 0.94, 2)
            if c4 < best_c5:
                best_c5, best_h = c4, h_cur.copy()
        # Local refinement
        if time.time() < deadline - 4:
            dl = min(deadline - 2, time.time() + 12)
            h_c, c_l = local_refine(h_cur, N, target, dl, rng)
            if c_l < best_c5:
                best_c5, best_h = c_l, h_c.copy()
        # Random restart
        if time.time() < deadline - 35:
            h_r = h_init + rng.randn(N) * 0.03
            h_r = project_sum(h_r, float(target))
            dr = time.time() + 25
            hr, cr = nesterov_phase(h_r, N, target, dr, 1.0, 80.0, 0.010, 0.86, 0)
            if time.time() < deadline - 5:
                dr2 = time.time() + 15
                hr2, cr2 = nesterov_phase(hr, N, target, dr2, 80.0, 400.0, 0.004, 0.91, 3)
                if cr2 < cr:
                    hr, cr = hr2, cr2
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()

    # Final polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 2:
        h_f, c_f = nesterov_phase(best_h.copy(), Nb, Nb // 2, deadline - 1,
                                  300.0, 1200.0, 0.0015, 0.93, 2)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
