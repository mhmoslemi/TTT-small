    return float(np.max(corr) * dx)

def fix_sum(h, n):

    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)

    G = np.fft.rfft(1.0 - h)

        active = sorted_idx[:n_active]

                grad += (1.0 - 2.0 * h)

        grad /= len(active)

        c5_new = compute_c5(h_new)

            best_h = h_new.copy()
            h = h_new
            no_improve = 0

            h = h_new  # still update h
            no_improve += 1

        lr *= lr_decay

import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

def circ_corr(h):
    n = len(h)
    H = np.fft.rfft(h)
    G = np.fft.rfft(1.0 - h)
    return np.fft.irfft(np.conj(H) * G, n)

def proj_subgrad(h, n, n_iter, lr_init, lr_decay, n_active, time_limit, use_momentum=True):

    v = np.zeros(n) if use_momentum else None

    t0 = time.time()

        if time.time() - t0 > time_limit:
            break

        if use_momentum and v is not None:

            h_eval = fix_sum(h_eval, n)
        cc = circ_corr(h_eval)

        grad = np.zeros(n)
        for k in active:
            if k == 0:
                grad += (1.0 - 2.0 * h_eval)
            else:
                grad += (1.0 - np.roll(h_eval, -k) - np.roll(h_eval, k))
        grad /= len(active)
        if use_momentum:
            v = mu * v - lr * grad
            h = np.clip(h - v, 0.0, 1.0)
            h = fix_sum(h, n)
        else:
            h = np.clip(h - lr * grad, 0.0, 1.0)
            h = fix_sum(h, n)

            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.95
                stall = 0
        lr *= lr_decay
        if use_momentum:
            mu = min(mu + 0.000005, 0.999)
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(xn, xo, h, period=2.0), new_n)

def downsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(xn, xo, h, period=2.0), new_n)

def slsqp_polish(h, n, time_limit=60):
    t0 = time.time()
    T = 0.005
    def obj_fn(x):
        xc = fix_sum(x, n)
        cc = circ_corr(xc)
        top = np.sort(cc)[::-1][:8]
        m = np.max(top)
        s = np.sum(np.exp((top - m) / T))
        return float((m + T * np.log(max(s, 1e-300))) * (2.0 / n))
    def sum_con(x):
        return np.sum(x) - n * 0.5
    try:
        res = opt.minimize(obj_fn, h.copy(), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_con}],
                          options={'maxiter': 300, 'ftol': 1e-14})
        if time.time() - t0 < time_limit and hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h
