def slsqp_run(h, n, t_val, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    if topk is not None and topk < 2 * n - 1:
        r = minimize(obj_topk, h, args=(t_val, topk), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
    else:
        r = minimize(obj_smooth, h, args=(t_val,), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    n = 280
    try:
        h = proj(np.array(initial_h_values, dtype=float), n)
    except Exception:
        h = proj(np.full(n, 0.5) + rng.randn(n) * 0.03, n)

    best_h = h.copy()
    best_c = compute_c5(h)

    # Phase 1: SLSQP from initial with escalating t
    for t_val in [100, 500, 2000, 8000]:
        if tleft() < 25:
            break
        try:
            hc = slsqp_run(h, n, t_val, None, 2000, 1e-15)
            c5 = compute_c5(hc)
            if c5 < best_c:
                best_c = c5
                best_h = hc
            h = hc
        except Exception:
            pass

    # Phase 2: Binary simulated annealing (multi-round)
    if tleft() > 250:
        h_bin = to_binary(best_h, n)
        schedules = [
            (0.8, 0.005, 250000),
            (0.5, 0.003, 300000),
            (0.3, 0.002, 350000),
            (0.15, 0.001, 400000),
        ]
        for (ts, te, ni) in schedules:
            if tleft() < 25:
                break
            cb, c_bin = binary_sa(h_bin, n, ts, te, ni, rng)
            if c_bin < best_c:
                best_c = c_bin
                best_h = cb.astype(float)
            h_bin = to_binary(best_h, n)
            cb2, c_bin2 = binary_sa(h_bin, n, ts, te, ni, rng)
            if c_bin2 < best_c:
                best_c = c_bin2
                best_h = cb2.astype(float)
            h_bin = to_binary(best_h, n)

    # Phase 3: SLSQP polish from best (binary-informed)
    if tleft() > 150:
        h_cont = best_h.astype(float).copy()
        for t_val in [500, 3000, 15000]:
            if tleft() < 20:
                break
            try:
                hc = slsqp_run(h_cont, n, t_val, None, 2500, 1e-16)
                c5 = compute_c5(hc)
                if c5 < best_c:
                    best_c = c5
                    best_h = hc
                h_cont = hc
            except Exception:
                pass

    # Phase 4: Diverse binary restarts
    if tleft() > 120:
        for restart in range(4):
            if tleft() < 25:
                break
            noise_scale = 0.08 + 0.05 * restart
            noisy = best_h + rng.randn(n) * noise_scale
            h_bin = to_binary(proj(noisy, n), n)
            cb, c_bin = binary_sa(h_bin, n, 0.4, 0.004, 250000, rng)
            if c_bin < best_c:
                best_c = c_bin
                best_h = cb.astype(float)
            h_bin = to_binary(best_h, n)
            cb2, c_bin2 = binary_sa(h_bin, n, 0.2, 0.002, 200000, rng)
            if c_bin2 < best_c:
                best_c = c_bin2
                best_h = cb2.astype(float)

    # Phase 5: Top-k SLSQP refinement
    if tleft() > 60:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        for k_frac in [0.15, 0.08]:
            if tleft() < 20:
                break
            k = max(5, int(len(corr) * k_frac))
            for t_val in [1000, 5000]:
                if tleft() < 15:
                    break
                try:
                    hc = slsqp_run(best_h, n, t_val, k, 3000, 1e-16)
                    c5 = compute_c5(hc)
                    if c5 < best_c:
                        best_c = c5
                        best_h = hc
                except Exception:
                    pass

    # Phase 6: Ultra-fine SLSQP
    if tleft() > 25:
        for t_val in [50000, 200000]:
            if tleft() < 12:
                break
            try:
                hc = slsqp_run(best_h, n, t_val, None, 5000, 1e-17)
                c5 = compute_c5(hc)
                if c5 < best_c:
                    best_c = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 7: Final binary micro-search
    if tleft() > 20:
        h_bin = to_binary(best_h, n)
        cb, c_bin = binary_sa(h_bin, n, 0.05, 0.0005, 150000, rng)
        if c_bin < best_c:
            best_c = c_bin
            best_h = cb.astype(float)

    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c, n
