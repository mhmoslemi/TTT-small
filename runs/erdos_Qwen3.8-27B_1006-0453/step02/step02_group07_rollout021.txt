import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-12:
        return h
    m = (h > 1e-6) & (h < 1.0 - 1e-6)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        # find the variable closest to 0.5 and adjust it
        i = np.argmin(np.abs(h - 0.5))
        h[i] = min(max(h[i] + d2, 0.0), 1.0)
    return h

def corr_grad(h, k):
    """Gradient of the k-th element of np.correlate(h, 1-h, 'full') w.r.t. h."""
    n = len(h)
    r = k - (n - 1)
    g = np.zeros(n)
    if r >= 0:
        m = n - r
        if m <= 0:
            return g
        idx1 = np.arange(r, n)
        g[idx1] += (1.0 - h[idx1 - r])
        idx2 = np.arange(0, m)
        g[idx2] -= h[idx2 + r]
    else:
        ll = -r
        if ll >= n:
            return g
        m = n - ll
        idx1 = np.arange(0, m)
        g[idx1] += (1.0 - h[idx1 + ll])
        idx2 = np.arange(ll, n)
        g[idx2] -= h[idx2 - ll]
    return g

def lse_smoothed(h, T):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-18:
        return cmax * dx, None
    e = np.exp((corr - cmax) / T)
    s = e.sum()
    p = e / s
    val = (cmax + T * np.log(s)) * dx
    if T < 1e-10:
        return val, None
    # gradient: sum_k p_k * grad_k
    g = np.zeros(n)
    for k in range(len(corr)):
        if p[k] > 1e-10:
            g += p[k] * corr_grad(h, k)
    g *= dx
    return val, g

def lse_obj_only(h, T):
    """Just the objective for L-BFGS-B (no gradient, use numerical)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-18:
        return cmax * dx
    e = np.exp((corr - cmax) / T)
    return (cmax + T * np.log(e.sum())) * dx

def subgrad_step(h, alpha, max_active=5):
    """One projected subgradient step on the exact max problem."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    # Find active lags (within tolerance of max)
    tol = 1e-8
    active = np.where(corr >= cmax - tol)[0]
    # Combine gradients of active lags
    g = np.zeros(n)
    for k in active[:max_active]:
        g += corr_grad(h, k)
    g /= min(len(active), max_active)
    g *= dx
    # Take step
    h_new = h - alpha * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new

def polyak_subgrad(h, c5_ref, n_iters=200, T_init=0.01):
    """Polyak-type subgradient method with adaptive step size."""
    n = len(h)
    h_best = h.copy()
    c5_best = compute_c5(h)
    alpha = T_init
    for t in range(n_iters):
        h_new = subgrad_step(h, alpha)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            alpha *= 1.05  # slightly increase step
        else:
            alpha *= 0.7  # decrease step
        h = h_new
        # Periodically project and fix
        if t % 50 == 49:
            h = fix_sum(np.clip(h, 0, 1), n)
    return h_best

def lbslb_optimize(h, T, maxiter=200):
    """L-BFGS-B optimization of the smoothed objective."""
    n = len(h)
    def obj(hv):
        return lse_obj_only(hv, T)
    def obj_grad(hv):
        _, g = lse_smoothed(hv, T)
        if g is None:
            # numerical gradient fallback
            eps = 1e-7
            g = np.zeros(n)
            for i in range(n):
                hp = hv.copy(); hp[i] += eps
                hm = hv.copy(); hm[i] -= eps
                g[i] = (obj(hp) - obj(hm)) / (2*eps)
            # add constraint gradient (sum constraint)
            g += 0.01 * np.ones(n) * (np.sum(hv) - 0.5*n)
        return g
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)]*n,
                              options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def coord_pair_polish(h, n, rng, n_pairs=1000):
    """Randomized paired coordinate descent."""
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        hj, hl = h[j], h[l]
        if hj < 1e-5 or hj > 1-1e-5 or hl < 1e-5 or hl > 1-1e-5:
            continue
        c0 = compute_c5(h)
        best_d = 0.0
        best_c = c0
        # Try several step sizes
        for d in (-0.03, -0.01, -0.003, 0.003, 0.01, 0.03):
            h2 = h.copy()
            h2[j] = min(max(hj + d, 0.0), 1.0)
            h2[l] = min(max(hl - d, 0.0), 1.0)
            c = compute_c5(h2)
            if c < best_c:
                best_c = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(hj + best_d, 0.0), 1.0)
            h[l] = min(max(hl - best_d, 0.0), 1.0)
    return fix_sum(h, n)

def generate_binary_init(n, seed):
    """Generate a random binary-like initial solution."""
    rng = np.random.RandomState(seed)
    # Start with a random binary pattern with sum = n/2
    perm = rng.permutation(n)

    # Binarize with some randomness
    h = (h > 0.5).astype(float) * 0.9 + 0.1 * rng.rand(n)
    return fix_sum(h, n)

def upsample(h, nn):

    r = k - (n - 1)
    g = np.zeros(n)

        m = n - r
        if m <= 0:

        g[ll:] -= h[:m]

def _lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(corr))
    if T < 1e-18:
        return cm * dx
    e = np.exp((corr - cm) / T)
    return (cm + T * np.log(np.sum(e))) * dx

def _lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(corr))
    e = np.exp((corr - cm) / T)
    p = e / np.sum(e)
    g = np.zeros(n)
    for k in range(len(corr)):
        if p[k] > 1e-12:
            g += p[k] * _corr_grad(h, k)
    g *= dx
    return g

def slsqp_analytic(h, T, iters):
    n = len(h)
    c0 = compute_c5(h)
    def obj(hv):
        return _lse_obj(hv, T)
    def grad(hv):
        return _lse_grad(hv, T)
    def sumcon(hv):
        return np.sum(hv) - 0.5 * n
    try:
        r = optimize.minimize(obj, h, jac=grad, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sumcon}],
            options={'maxiter': iters, 'ftol': 1e-18})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:

    return h
