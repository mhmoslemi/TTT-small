    if T < 1e-14:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def slsqp(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-18})
        hn = project_simplex(r.x, 0.5 * n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return project_simplex(np.interp(x1, x0, h, period=2.0), 0.5 * nn)

def coord_polish(h, n, n_pass=3):
    target = 0.5 * n
    for _ in range(n_pass):
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            best_c5 = compute_c5(h)
            best_h = h.copy()
            for delta in (-0.04, -0.02, -0.008, -0.002, 0.002, 0.008, 0.02, 0.04):
                hn = h.copy()
                hn[:] += -delta / (n - 1)
                hn[j] = min(max(h[j] + delta, 0.0), 1.0)
                hn = project_simplex(hn, target)
                c = compute_c5(hn)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = hn
            h = best_h
        h = project_simplex(h, target)
    return h

def pair_polish(h, n, rng_obj, n_pairs=500):
    target = 0.5 * n
    for _ in range(n_pairs):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] <= 1e-3 or h[j] >= 1.0 - 1e-3:
            continue
        if h[l] <= 1e-3 or h[l] >= 1.0 - 1e-3:
            continue
        best_c5 = compute_c5(h)
        best_d = 0.0
        for d in (-0.05, -0.02, -0.006, -0.0015, 0.0015, 0.006, 0.02, 0.05):
            if h[j]+d < 1e-3 or h[j]+d > 1-1e-3:
                continue
            if h[l]-d < 1e-3 or h[l]-d > 1-1e-3:
                continue
            hn = h.copy()
            hn[j] += d
            hn[l] -= d
            c = compute_c5(hn)
            if c < best_c5 - 1e-13:
                best_c5 = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return project_simplex(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 7)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    target = 0.5 * n
    h = project_simplex(h, target)
    c5 = compute_c5(h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Quick SLSQP warm-up
    for T, it in [(0.03, 200), (0.01, 300), (0.004, 400)]:
        if remaining() < 200:
            break
        h = slsqp(h, n, T, it)
        h = project_simplex(h, target)
        c5 = compute_c5(h)

    # Phase 2: Nesterov accelerated projected gradient (fast, many iterations)
    for T, iters, lr in [(0.008, 30000, 0.003), (0.003, 30000, 0.002),
                         (0.001, 30000, 0.001), (0.0004, 25000, 0.0005)]:
        if remaining() < 150:
            break
        tl = min(remaining() - 100, 120)
        h = nesterov_proj(h, T, iters, lr, 0.9, tl, t0)
        h = project_simplex(h, target)
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
        h = project_simplex(h, target)

    # Phase 3: SLSQP fine optimization
    for T, it in [(0.002, 600), (0.0005, 800), (0.00015, 1000)]:
        if remaining() < 80:
            break
        h = slsqp(h, n, T, it)
        h = project_simplex(h, target)
        c5 = compute_c5(h)

    # Phase 4: Coordinate and pair polishing
    if remaining() > 200:
        h = coord_polish(h, n, 3)
        h = project_simplex(h, target)
        c5 = compute_c5(h)
    if remaining() > 180:
        h = pair_polish(h, n, rng, 700)
        c5 = compute_c5(h)

    # Phase 5: Diverse restarts with Nesterov + SLSQP
    best_h, best_c5 = h.copy(), c5
    for i in range(8):
        if remaining() < 100:
            break
        ht = h.copy()
        mode = i % 4
        if mode == 0:
            ht += np.random.randn(n) * 0.015
        elif mode == 1:
            idx = np.random.choice(n, 15, replace=False)
            ht[idx] += np.random.randn(15) * 0.1
        elif mode == 2:
            ht += np.random.randn(n) * 0.008
        else:
            ht += np.ones(n) * rng.uniform(-0.02, 0.02)
        ht = project_simplex(ht, target)
        for T, it in [(0.01, 400), (0.002, 500)]:
            ht = slsqp(ht, n, T, it)
            ht = project_simplex(ht, target)
        ht = nesterov_proj(ht, 0.0005, 10000, 0.0004, 0.9, min(remaining()-60, 60), t0)
        ht = project_simplex(ht, target)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
            if remaining() > 120:
                best_h = coord_polish(best_h, n, 2)
                best_h = pair_polish(best_h, n, rng, 300)
                best_h = project_simplex(best_h, target)
                best_c5 = compute_c5(best_h)

    h, c5 = best_h, best_c5

    # Phase 6: Final fine SLSQP
    for T, it in [(0.00005, 2000), (0.00002, 2500)]:
        if remaining() < 30:
            break
        h = slsqp(h, n, T, it)
        h = project_simplex(h, target)
        c5 = compute_c5(h)

    # Phase 7: Higher resolution if time remains
    if remaining() > 150:
        n2 = 400
        h2 = upsample(h, n2)
        target2 = 0.5 * n2
        for T, it in [(0.01, 400), (0.003, 500)]:
            if remaining() < 40:
                break
            h2 = slsqp(h2, n2, T, it)
            h2 = project_simplex(h2, target2)
        h2 = nesterov_proj(h2, 0.0005, 20000, 0.0003, 0.9, min(remaining()-30, 90), t0)
        h2 = project_simplex(h2, target2)
        if compute_c5(h2) < c5:
            h, n, c5 = h2, n2, compute_c5(h2)

    h = project_simplex(h, 0.5 * len(h))
    c5_final = compute_c5(h)
    return (h, c5_final, len(h))
