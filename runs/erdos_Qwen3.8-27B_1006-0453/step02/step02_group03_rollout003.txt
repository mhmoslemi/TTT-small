    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    w = np.exp(beta * (cr - cr.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= s
    
    o = n - 1  # offset: w index o + lag corresponds to lag
    g = np.zeros(n)
    # For each j: g[j] = sum_lag w[o+lag] * (1-h[j-lag]) * I[0<=j-lag<n] - sum_lag w[o+lag] * h[j+lag] * I[0<=j+lag<n]

    # Second sum: lags where 0<=j+lag<n → lag in [-j, n-1-j]

import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    n = _sp['n']
    return float(np.sum(x[:n]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > ts:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h2 = np.clip(h - t, 0, 1)
    d = ts - h2.sum()
    if abs(d) > 1e-12:
        mask = (h2 > 1e-10) & (h2 < 1.0 - 1e-10)
        if mask.sum() > 0:
            h2[mask] += d / mask.sum()
            np.clip(h2, 0, 1, out=h2)
    return h2

def sgrad(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (cr - cr.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        l1 = max(0, j - (n - 1))
        l2 = min(n - 1, j)
        if l1 <= l2:
            idx = np.arange(l1, l2 + 1)
            g[j] += np.dot(w[o + (j - idx)], 1.0 - h[idx])
        l3 = max(0, -j) if j < 0 else 0
        l4 = min(n - 1, n - 1 - j)
        if l3 <= l4:
            idx = np.arange(l3, l4 + 1)
            g[j] -= np.dot(w[o + (idx - j)], h[idx])
    return g * dx

def nesterov(h0, dx, ts, max_it, dl, rng):
    n = len(h0)
    h = h0.copy()
    hp = h.copy()
    mom = 0.85
    bc = compute_c5(h, dx)
    stall = 0
    for i in range(max_it):
        if time.time() > dl:
            break
        beta = 15.0 + 100.0 * min(1.0, i / max(1, 0.35 * max_it))
        lr = 0.04 / (1.0 + 0.0008 * i)
        y = h + mom * (h - hp)
        y = np.clip(y, 0, 1)
        g = sgrad(y, dx, beta)
        ok = False
        for sh in (1.0, 0.6, 0.3, 0.15, 0.07):
            h2 = proj(y - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                hp, h, bc = h, h2, c2
                stall = 0
                ok = True
                break
        if not ok:
            stall += 1
            if stall % 30 == 0:
                mom = max(0.5, mom - 0.05)
                h = proj(h + 0.02 * rng.randn(n), ts)
            if stall > 200:
                break
    return h, bc

def slsqp(h0, dx, ts, mit=200):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-14})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def coord_descent(h0, dx, ts, rnd, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for r in range(rnd):

            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]

                h[j] = sw = sv + d
