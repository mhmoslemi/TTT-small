import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0, 1, out=h)
    for _ in range(300):
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h

def slsqp_mm(h0, dx, ts, maxiter=400):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    cr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = cr0.max() + 0.005
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0, 1)] * n + [(0, float(n) + 1)]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    best_h, best_c = h0.copy(), compute_c5(h0, dx)
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            c = compute_c5(hr, dx)
            if c < best_c:
                best_h, best_c = hr, c
    except Exception:
        pass
    return best_h, best_c

def subgrad(h0, dx, ts, max_iter, deadline, b0=5.0, b1=300.0):

            break
        frac = min(1.0, i / max(1, 0.8 * max_iter))
        beta = b0 + (b1 - b0) * frac
        lr = 0.08 / (1.0 + 0.002 * i)
        cr = np.correlate(h, 1.0 - h, mode='full')
        w = np.exp(beta * (cr - cr.max()))
        sw = w.sum()
        if sw < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= sw
        g = np.zeros(n)
        for j in range(n):
            k1lo = max(-(n-1), j-(n-1))
            k1hi = min(n-1, j)
            if k1lo <= k1hi:
                g[j] += np.dot(w[o + np.arange(k1lo, k1hi+1)], 1.0 - h[j - np.arange(k1lo, k1hi+1)])
            k2lo = max(-(n-1), -j)
            k2hi = min(n-1, n-1-j)
            if k2lo <= k2hi:
                g[j] -= np.dot(w[o + np.arange(k2lo, k2hi+1)], h[j + np.arange(k2lo, k2hi+1)])
        g *= dx
        ok = False
        for f in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            h2 = proj(h - lr * f * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h, bc, st = h2, c2, 0
                ok = True
                break
        if not ok:
            st += 1
            if st > 150:
                break
    return h, bc
