    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fix_sum(h, n):
    target = 0.5 * n
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h += d / n
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    d = target - h.sum()
    if abs(d) > 1e-14:
        # Add to entries with most room
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            room = (1.0 - h[i]) if d > 0 else h[i]
            add = np.sign(d) * min(abs(d), room)
            h[i] += add
            d -= add
            if abs(d) < 1e-14:
                break
    return h

def soft_max_grad(h, T, K):
    n = len(h)
    dx = 2.0 / n
    ov = all_overlaps(h)  # 2n-1 values
    idx = np.argsort(ov)[-K:]
    f = ov[idx]
    cmax = f.max()
    if T < 1e-14:
        return cmax, np.zeros(n)
    sh = np.clip(f - cmax, -50.0*T, 0.0)
    w = np.exp(sh / T)
    sw = w.sum()
    p = w / sw
    val = cmax + T * np.log(sw / len(f))
    
    lags = np.arange(len(ov)) - (n - 1)

def optimize_lbfgs(h, T, K, maxiter, penalty):
    n = len(h)
    target = 0.5 * n
    
    def fgrad(x):
        x = np.clip(x, 0.0, 1.0)
        v, g = soft_max_grad(x, T, K)
        # Penalty for sum constraint
        s = x.sum()
        v += penalty * (s - target)**2
        g += 2.0 * penalty * (s - target)
        return v, g
    
    res = minimize(fgrad, h.copy(), method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': maxiter*20})
    return fix_sum(res.x, n)

def interp_res(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2.0, n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    # Periodic interpolation
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_new = np.interp(x_new, x_ext, h_ext)
    return fix_sum(h_new, new_n)
