    Returns: max_new_corr for each pair, shape (len(ones), len(zeros))
    """

    # Entries 0..n-2: negative shifts (m = n-1, n-2, ..., 1) → corr index 0..n-2

    new_max = np.maximum(neg_part.max(axis=-1), pos_part.max(axis=-1))  # shape (140, 140)
    
    return new_max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    sorted_idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[sorted_idx[:n//2]] = 1.0
    return hb

def obj_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s = np.clip(s, -100, 100)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def interp_to(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def build_D(h, n):
    """Build D1 (pos-shift delta for 1->0) and D2 (neg-shift delta for 1->0)."""
    D1 = np.zeros((n, n - 1))
    D2 = np.zeros((n, n - 1))
    for p in range(n):
        if p > 0:
            idx = np.arange(p - 1, -1, -1)
            D1[p, :p] = -(1.0 - h[idx])
            D2[p, :p] = h[idx]
        if p < n - 1:
            idx = np.arange(p + 1, n)
            D1[p, :len(idx)] += h[idx]
            D2[p, :len(idx)] += -(1.0 - h[idx])
    return D1, D2

def best_swap(corr_full, D1, D2, ones, zeros, n):
    """Find best (a,b) swap to minimize max correlation."""
    neg_ref = corr_full[0:n-1]
    pos_ref = corr_full[n:2n-1]
    o_d1 = D1[ones]
    z_d1 = D1[zeros]
    o_d2 = D2[ones]
    z_d2 = D2[zeros]
    neg_part = neg_ref[None, None, :] + o_d2[:, None, :] - z_d2[None, :, :]
    pos_part = pos_ref[None, None, :] + o_d1[:, None, :] - z_d1[None, :, :]
    new_max = np.maximum(neg_part.max(axis=-1), pos_part.max(axis=-1))
    flat = new_max.ravel()
    bi = np.argmin(flat)
    return bi // len(zeros), bi % len(zeros)

def binary_local_search(h, rng, t0, deadline):
    """Steepest descent 2-swap local search on binary h."""
    n = len(h)
    h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(corr_full.max())
    best_max_c = max_c
    best_h = h.copy()
    stall = 0
    kick_count = 0

    while time.time() < deadline:
        D1, D2 = build_D(h, n)
        ia, ib = best_swap(corr_full, D1, D2, ones, zeros, n)
        a = int(ones[ia])
        b = int(zeros[ib])
        # Compute actual new max
        h[a] = 0.0
        h[b] = 1.0
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(corr_full.max())
        if new_max < max_c - 1e-12:
            max_c = new_max
            oi = np.where(ones == a)[0]
            zi = np.where(zeros == b)[0]
            ones[oi[0]] = b
            zeros[zi[0]] = a
            stall = 0
            if max_c < best_max_c:
                best_max_c = max_c
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
            if stall > 80:
                # Kick: swap 3 random pairs
                for _ in range(3):
                    ai = rng.randint(len(ones))
                    bi2 = rng.randint(len(zeros))
                    a2 = int(ones[ai])
                    b2 = int(zeros[bi2])
                    h[a2] = 0.0
                    h[b2] = 1.0
                    ones[ai] = b2
                    zeros[bi2] = a2
                corr_full = np.correlate(h, 1.0 - h, mode='full')
                max_c = float(corr_full.max())
                stall = 0
                kick_count += 1
                if kick_count > 200:
                    break
    return best_h, best_max_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 280
    dx = 2.0 / n

    # Initialize from parent
    try:
        h0 = np.array(initial_h_values, dtype=float)
        h0 = interp_to(h0, n)
    except Exception:
        h0 = None

    # Build candidate list
    candidates = []
    if h0 is not None:
        candidates.append(to_binary(h0, n))
    # Random binaries
    for _ in range(3):
        idx = rng.choice(n, n // 2, replace=False)
        hr = np.zeros(n)
        hr[idx] = 1.0
        candidates.append(hr)
    # Structured: spread evenly
    hs = np.zeros(n)
    hs[::2] = 1.0
    candidates.append(hs)
    # Structured: quadratic residues mod a prime
    p = 281
    qr = sorted([i * i % p for i in range(p)])
    hs2 = np.zeros(n)
    for q in qr:
        if q < n:
            hs2[q] = 1.0
    hs2 = to_binary(hs2, n)
    candidates.append(hs2)

    # Track global best
    global_best_h = None
    global_best_c5 = float('inf')
    if h0 is not None:
        global_best_c5 = compute_c5(h0)
        global_best_h = h0.copy()

    # Phase 1: Binary local search on each candidate
    deadline_1 = t0 + budget_s * 0.72
    for ci, hc in enumerate(candidates):
        if time.time() > deadline_1:
            break
        per_t = max(30.0, (deadline_1 - time.time()) / max(1, len(candidates) - ci))
        h_opt, max_c = binary_local_search(hc, rng, t0, time.time() + per_t)
        c5 = max_c * dx
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best_h = h_opt.copy()

    # Phase 2: More binary search iterations on the best
    if time.time() < t0 + budget_s * 0.85 and global_best_h is not None:
        deadline_2 = t0 + budget_s * 0.85
        h_opt, max_c = binary_local_search(global_best_h, rng, t0, deadline_2)
        c5 = max_c * dx
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best_h = h_opt.copy()

    # Phase 3: Continuous refinement near best binary
    if time.time() < t0 + budget_s * 0.95 and global_best_h is not None:
        h_cont = global_best_h.copy()
        bnds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
        for t_val in [100, 500, 2000]:
            if time.time() > t0 + budget_s * 0.95:
                break
            try:
                r = minimize(obj_smooth, h_cont, args=(t_val,), method='L-BFGS-B',
                             bounds=bnds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-18})
                hc2 = proj(np.clip(r.x, 0.0, 1.0), n)
                c5 = compute_c5(hc2)
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best_h = hc2.copy()
                h_cont = hc2
            except Exception:
                pass

    # Phase 4: Final binary polish
    if time.time() < t0 + budget_s * 0.98:
        h_bin_final = to_binary(global_best_h, n)
        h_opt, max_c = binary_local_search(h_bin_final, rng, t0, t0 + budget_s * 0.98)
        c5 = max_c * dx
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best_h = h_opt.copy()

    if global_best_h is None:
        global_best_h = proj(np.full(n, 0.5), n)
        global_best_c5 = compute_c5(global_best_h)

    return global_best_h, compute_c5(global_best_h), n
