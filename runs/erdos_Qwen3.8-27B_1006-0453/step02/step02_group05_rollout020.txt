    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    # Only process shifts with significant weight
    active = np.where(w > 1e-5)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            if L < N:
                grad[L:] += wi * (1.0 - h[:N-L])
                grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
    return grad, true_c5

def adam_optimize(h, N, target, dx, deadline, tau0, tau1, lr0, decay):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    max_iters = 5000
    while time.time() < deadline and it < max_iters:
        frac = min(it / float(max_iters), 1.0)
        # Smooth tau ramp
        tau = tau0 * (tau1 / tau0) ** (frac ** 0.7)  # slower ramp at start
        grad, c5_val = softmax_grad(h, N, tau)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        step = lr * mh / (np.sqrt(vh) + eps)
        # Try full step, then half, then quarter if needed
        h_candidate = project_sum(h - step, target)
        c5_new = compute_c5(h_candidate, dx)
        if c5_new > c5_val + 1e-8:
            h_candidate = project_sum(h - 0.5 * step, target)
            c5_new = compute_c5(h_candidate, dx)
            if c5_new > c5_val + 1e-8:
                h_candidate = project_sum(h - 0.25 * step, target)
        h = h_candidate
        if it % 50 == 0:
            lr *= decay
        if lr < 1e-10:
            break
    return best_h, best_c5
