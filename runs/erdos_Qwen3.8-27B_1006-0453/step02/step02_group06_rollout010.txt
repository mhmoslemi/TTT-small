        h_opt = do_slsqp(h_best.copy(), n_best, p_norm_obj, (50, n_best), 600, 1e-16)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c
        h_opt = do_slsqp(h_best.copy(), n_best, p_norm_obj, (100, n_best), 500, 1e-16)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 4: Multi-start with diverse perturbations
    rng_main = np.random.RandomState(seed + 1)
    for i in range(24):
        if time.time() > budget_end - 150:
            break
        sigma = 0.004 + rng_main.rand() * 0.016
        h_pert = np.clip(h_best + rng_main.randn(n_best) * sigma, 0, 1)
        h_pert = proj_constraint(h_pert, n_best)
        if i % 3 == 1 and n_best >= 32:
            bs = int(rng_main.randint(6, 18))
            s1 = int(rng_main.randint(0, n_best - bs))
            s2 = int(rng_main.randint(0, n_best - bs))
            if s2 + bs <= s1 or s1 + bs <= s2:
                h_pert[s1:s1+bs], h_pert[s2:s2+bs] = h_pert[s2:s2+bs].copy(), h_pert[s1:s1+bs].copy()
        T_p = 0.0008 + rng_main.rand() * 0.005
        h_opt = do_slsqp(h_pert, n_best, soft_max_obj, (T_p, n_best), 500, 1e-15)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 5: Upsample to higher resolution
    for n_try in [420, 504]:
        if time.time() > budget_end - 120:
            break
        h_curr = upsample(h_best, n_try)
        for T in [0.004, 0.0015]:
            if time.time() > budget_end - 60:
                break
            h_opt = do_slsqp(h_curr, n_try, soft_max_obj, (T, n_try), 500, 1e-15)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c
                n_best = n_try
                h_curr = h_best.copy()

    # Phase 6: Block coordinate descent
    if time.time() < budget_end - 80:
        h_bcd = block_cd(h_best.copy(), n_best, 0.005, budget_end - 40, 16, seed + 3)
        c = compute_c5(h_bcd)
        if c < c5_best:
            h_best = h_bcd
            c5_best = c

    # Phase 7: SA with block swaps
    if time.time() < budget_end - 40:
        h_sa, c_sa = sa_refine(h_best, n_best, budget_end, 10000, seed + 777)
        if c_sa < c5_best:
            h_best = h_sa
            c5_best = c_sa

    # Phase 8: Final SLSQP polish
    for T in [0.0004, 0.00015]:
        if time.time() > budget_end - 5:
            break
        h_opt = do_slsqp(h_best.copy(), n_best, soft_max_obj, (T, n_best), 2500, 1e-18)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
