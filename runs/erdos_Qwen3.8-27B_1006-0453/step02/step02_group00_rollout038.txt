    else:

        best_h[::2] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Thorough SLSQP T-schedule
    for T in [20, 60, 200, 800, 3000, 10000]:
        if time.time() - t0 > budget_s * 0.12:
            break
        hc = slsqp_step(best_h, T, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: L-BFGS-B refinement
    for T in [500, 4000, 15000]:
        if time.time() - t0 > budget_s * 0.22:
            break
        hc = lbfgsb_step(best_h, T, n, maxiter=8000, ftol=1e-16, mu=5000.0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # Re-polish with SLSQP after L-BFGS-B
        hc2 = slsqp_step(best_h, 3000, n, maxiter=1000, ftol=1e-16)
        c5v2 = compute_c5(hc2)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = hc2

    # Phase 3: Differential Evolution (global exploration)
    if time.time() - t0 < budget_s * 0.40:
        bounds_de = [(0.0, 1.0)] * n
        try:
            de_res = differential_evolution(
                de_obj, bounds_de, seed=seed, popsize=20,
                maxiter=400, tol=1e-10, mutation=(0.4, 1.2),
                recomb=0.8, polish=False, init='sobol',
                updating='immediate', workers=1
            )
            if de_res is not None and de_res.x is not None:
                hc = project_feasible(np.clip(de_res.x, 0, 1), n)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                hc = slsqp_step(hc, 3000, n, maxiter=1000, ftol=1e-16)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass

    # Phase 4: Basin hopping with adaptive step + dual optimizer
    T_bh = 1500
    n_bh_iter = 80
    for it in range(n_bh_iter):
        if time.time() - t0 > budget_s * 0.58:
            break
        step = 0.08 * (1.0 - 2.0 * it / n_bh_iter) + 0.02
        h_pert = best_h + step * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        try:
            hc = slsqp_step(h_pert, T_bh, n, maxiter=500, ftol=1e-14)
            c5v = compute_c5(hc)
            temp = 0.005 * (1.0 - it / n_bh_iter) + 0.0003
            if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / temp):
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                else:
                    best_h = hc
        except Exception:
            pass

    # Phase 5: Multi-start diverse structures
    if time.time() - t0 < budget_s * 0.72:
        starts = []
        for i in range(6):
            hb = (np.random.rand(n) < 0.5).astype(float)
            starts.append(project_feasible(hb, n))
        for period in [3, 5, 7, 13]:
            hblk = np.zeros(n)
            half = period // 2
            for i in range(0, n, period):
                hblk[i:i + half] = 1.0
            starts.append(project_feasible(hblk, n))
        for s in [0.04, 0.1, 0.2]:
            hp = best_h + s * np.random.randn(n)
            starts.append(project_feasible(hp, n))
        hc = 1.0 - best_h
        starts.append(project_feasible(hc, n))

        for h0 in starts:
            if time.time() - t0 > budget_s * 0.72:
                break
            for T in [200, 2500]:
                if time.time() - t0 > budget_s * 0.72:
                    break
                try:
                    hc = slsqp_step(h0, T, n, maxiter=700, ftol=1e-15)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    # Phase 6: Systematic pair-swap local search
    if time.time() - t0 < budget_s * 0.84:
        n_cur = len(best_h)
        stall = 0
        for iteration in range(3000):
            if time.time() - t0 > budget_s * 0.84:
                break
            improved = False
            for _ in range(40):
                i = np.random.randint(n_cur)
                j = np.random.randint(n_cur)
                if i == j:
                    continue
                for delta in [0.015, 0.04, 0.08, 0.05]:
                    if best_h[i] + delta > 1.001 or best_h[j] - delta < -0.001:
                        continue
                    h_try = best_h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    h_try = project_feasible(h_try, n_cur)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                stall = 0
            else:
                stall += 1
            if stall > 150:
                # Retry with SLSQP to escape
                hc = slsqp_step(best_h, 5000, n_cur, maxiter=500, ftol=1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5 - 1e-14:
                    best_c5 = c5v
                    best_h = hc
                    stall = 0
                elif stall > 300:
                    break

    # Phase 7: Try different n values
    for n_try in [80, 96, 144, 160, 192]:
        if time.time() - t0 > budget_s * 0.93:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        for trial in range(3):
            if time.time() - t0 > budget_s * 0.93:
                break
            h0 = h_interp + np.random.uniform(-0.04, 0.04, n_try)
            h0 = project_feasible(h0, n_try)
            for T in [300, 3000]:
                if time.time() - t0 > budget_s * 0.93:
                    break
                try:
                    hc = slsqp_step(h0, T, n_try, maxiter=800, ftol=1e-15)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    # Phase 8: Final ultra-high-T polish
    n_cur = len(best_h)
    for T in [8000, 30000, 80000]:
        if time.time() - t0 > budget_s * 0.99:
            break
        try:
            hc = slsqp_step(best_h, T, n_cur, maxiter=3000, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    # One more L-BFGS-B at very high T
    if time.time() - t0 < budget_s * 0.995:
        hc = lbfgsb_step(best_h, 30000, n_cur, maxiter=10000, ftol=1e-16, mu=10000.0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
