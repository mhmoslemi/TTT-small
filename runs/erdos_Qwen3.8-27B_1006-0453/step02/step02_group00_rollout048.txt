import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def obj_topk(h, T, n, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_use = min(k, len(corr))
    idx = np.argpartition(-corr, k_use - 1)[:k_use]
    topk = corr[idx]
    m = np.max(topk)
    if T < 1e-10:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(T * (topk - m)))) / T) * (2.0 / n)

def obj_full(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1e-10:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def slsqp_topk(h, T, n, k, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(obj_topk, h, args=(T, n, k), method='SLSQP',
                  bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(np.clip(res.x, 0, 1), n)

def slsqp_full(h, T, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(obj_full, h, args=(T, n), method='SLSQP',
                  bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(np.clip(res.x, 0, 1), n)

    """Nelder-Mead on active coordinates only, preserving sum via projection."""
    active = np.where((h > 0.01) & (h < 0.99))[0]
    if len(active) < 3 or len(active) > 40:
        return h
    idx = active

    def nm_obj(u):
        h_try = h.copy()
        h_try[idx] = u
        h_try = project_feasible(h_try, n)
        return compute_c5(h_try)

    res = minimize(nm_obj, h[idx].copy(), method='Nelder-Mead',
                  options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-14})
    h_new = h.copy()
    h_new[idx] = res.x
    return project_feasible(h_new, n)
