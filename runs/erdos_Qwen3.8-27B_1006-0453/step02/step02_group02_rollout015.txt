    c = val[idx]
    mx = np.max(c)
    s = t * (c - mx)
    s -= np.max(s)
    return mx + np.log(np.sum(np.exp(s))) / t

    """Gradient wrt val (full vector, zeros outside idx)."""
    c = val[idx]
    mx = np.max(c)
    s = t * (c - mx)
    s -= np.max(s)
    e = np.exp(s)
    w = e / np.sum(e)
    g = np.zeros_like(val)
    g[idx] = w
    return g

    coeff has length 2n-1 (same as corr in 'full' mode).

    s -= np.max(s)
    val = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    # penalty
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return val + penalty

def obj_grad_topk(h, t, topk, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = min(topk, len(corr))
    idx = np.argpartition(corr, -topk)[-topk:]
    c = corr[idx]
    mx = np.max(c)
    s = t * (c - mx)
    s -= np.max(s)
    e = np.exp(s)
    w = e / np.sum(e)
    coeff = np.zeros(2 * n - 1)
    coeff[idx] = w
    grad = corr_grad(h, coeff) * dx
    # penalty gradient
    sp = np.sum(h) - n / 2.0
    grad += 2.0 * lam * sp * np.ones(n)
    val = (mx + np.log(np.sum(e)) / t) * dx + lam * sp * sp
    return val, grad

    s -= np.max(s)
    val = (mx + np.log(np.sum(np.exp(s))) / t) * dx

def obj_grad_full(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    e = np.exp(s)
    w = e / np.sum(e)
    grad = corr_grad(h, w) * dx
    sp = np.sum(h) - n / 2.0
    grad += 2.0 * lam * sp * np.ones(n)
    val = (mx + np.log(np.sum(e)) / t) * dx + lam * sp * sp
    return val, grad

def lbfgs_opt(h, n, t, topk, lam, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    if topk is not None:
        fun = lambda x: obj_fun_topk(x, t, topk, lam)
        jac = lambda x: obj_grad_topk(x, t, topk, lam)
    else:
        fun = lambda x: obj_fun_full(x, t, lam)
        jac = lambda x: obj_grad_full(x, t, lam)
    try:
        r = minimize(fun, h, jac=jac, method='L-BFGS-B',
                     bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8, 'maxls': 50})
        return proj(np.clip(r.x, 0.0, 1.0), n), r
    except Exception:
        return h, None

        if tleft_check(t0, budget_s) < 25:
            break
        topk = max(5, int(559.0 / (3.0 + tv / 300.0)))
        for tr in range(3):
            if tleft_check(t0, budget_s) < 18:
                break
            st = h.copy()
            if tr == 1:
                st = proj(h + rng.randn(N) * 0.003, N)
            elif tr == 2:
                st = bin_push(h, N)
            hc = _lbfgs(st, N, tv, topk, 5000.0, 2000, 1e-12)
            c5 = compute_c5(hc)
            if c5 < compute_c5(h):
                h = hc
            if c5 < bc:
                bc = c5

    # Phase 4: Peak-focused refinement
    if tleft_check(t0, budget_s) > 70:
        for tv, div in [(3000, 10), (10000, 7), (30000, 5), (80000, 3)]:
            if tleft_check(t0, budget_s) < 12:
                break
            topk = max(3, 559 // div)
            hc = _lbfgs(h, N, tv, topk, 10000.0, 3000, 1e-13)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                h = hc

    # Phase 5: Binary push cycle
    if tleft_check(t0, budget_s) > 50:
        for _ in range(4):
            if tleft_check(t0, budget_s) < 12:
                break
            hb = bin_push(h, N)
            for tv in [500, 3000]:
                if tleft_check(t0, budget_s) < 10:
                    break
                hc = _lbfgs(hb, N, tv, 25, 8000.0, 2000, 1e-12)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    h = hc
                hb = bin_push(hc, N)

    # Phase 6: Try n=400
    if tleft_check(t0, budget_s) > 80:
        h400 = interp_up(h, 400)
        c400 = compute_c5(h400)
        for tv in [200, 1000, 5000]:
            if tleft_check(t0, budget_s) < 25:
                break
            topk = max(5, 799 // 8)
            hc = _lbfgs(h400, 400, tv, topk, 5000.0, 1200, 1e-12)
            c5 = compute_c5(hc)
            if c5 < c400:
                h400 = hc
                c400 = c5
        if c400 < bc:
            bc = c400
            h = h400

    # Phase 7: Final polish
    if tleft_check(t0, budget_s) > 10:
        nf = len(h)
        hc = _lbfgs(h, nf, 50000, 5, 20000.0, 4000, 1e-14)
        c5 = compute_c5(hc)
        if c5 < bc:
            bc = c5
            h = hc

    h = proj(h, len(h))
    fc = compute_c5(h)
    if fc < bc:
        bc = fc
    return h, bc, len(h)
