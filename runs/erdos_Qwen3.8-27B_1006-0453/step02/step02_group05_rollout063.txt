import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    true_c5 = float(corr.max() * dx)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = max(1e-6, 1e-3 / N)
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    grad *= dx
    return grad, true_c5

    v = np.zeros(N)
    it = 0
    t_start = time.time()
    t_total = max(deadline - t_start, 1.0)
    while time.time() < deadline:
        it += 1
        frac = min((time.time() - t_start) / t_total, 1.0)
        if sched == 'cos':
            lr_eff = lr * 0.5 * (1 + np.cos(np.pi * frac))
        else:
            lr_eff = lr
        h_look = h + mu * v
        g, c5v = softmax_grad(h_look, N, tau)
        c5v_real = compute_c5(h, dx)
        if c5v_real < best_c5 - 1e-13:
            best_c5 = c5v_real
            best_h = h.copy()
        v = mu * v - lr_eff * g
        h = h + v
        h = project_sum(h, target)
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    true_c5 = float(corr.max() * dx)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = max(1e-6, 1e-3 / N)
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    grad *= dx
    return grad, true_c5

def nesterov_opt(h, N, target, dx, deadline, tau, lr0, mu):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    t_start = time.time()
    t_dur = max(deadline - t_start, 1.0)
    while time.time() < deadline:
        frac = min((time.time() - t_start) / t_dur, 1.0)
        lr_eff = lr0 * (0.1 + 0.9 * (1.0 + np.cos(np.pi * frac)) * 0.5)
        h_look = h + mu * v
        g, _ = softmax_grad(h_look, N, tau)
        v = mu * v - lr_eff * g
        h = h + v
        h = project_sum(h, target)
        c5v = compute_c5(h, dx)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best_h = h.copy()
    return best_h, best_c5

def coord_polish(h, N, target, dx, deadline, ntern=18):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            old = h[i]
            a = np.zeros(2 * N - 1)
            k1lo = max(0, i - N + 1)
            k1hi = min(i, 2 * N - 2)
            if k1lo <= k1hi:
                idx1 = np.arange(k1lo, k1hi + 1)
                a[idx1] += (1.0 - h[i - idx1])
            k2lo = 0
            k2hi = min(N - 1 - i, 2 * N - 2)
            if k2lo <= k2hi:
                idx2 = np.arange(k2lo, k2hi + 1)
                a[idx2] -= h[i + idx2]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            lo, hi = lo_d, hi_d
            for _ in range(ntern):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = np.max(corr + m1 * a)
                v2 = np.max(corr + m2 * a)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            opt = (lo + hi) / 2.0
            new = max(0.0, min(1.0, old + opt))
            if abs(new - old) > 1e-10:
                h[i] = new
                corr += (new - old) * a
                improved = True
        h = project_sum(h, target)
        c5v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def targeted_subgrad(h, N, target, dx, deadline, step0=0.02):
    """Target the current worst lag repeatedly with decreasing step."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    center = N - 1
    t_start = time.time()
    t_dur = max(deadline - t_start, 1.0)
    while time.time() < deadline:
        frac = (time.time() - t_start) / t_dur
        step = step0 * (1.0 - 0.9 * frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst = np.argmax(corr)
        L = worst - center
        g = np.zeros(N)
        if L == 0:
            g[:] = (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:N - L])
            g[:N - L] -= h[L:]
        else:
            M = -L
            g[:N - M] += (1.0 - h[M:])
            g[M:] -= h[:N - M]
        h = h - step * (2.0 / N) * g
        h = project_sum(h, target)
        c5v = compute_c5(h, dx)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best_h = h.copy()
    return best_h, best_c5
