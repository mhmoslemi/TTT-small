import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(h, c):
    n = len(h)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def pnorm_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.maximum(get_corr(h), 0.0)
    return (np.sum(corr ** p)) ** (1.0 / p) * dx

def logsumexp_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    ws = w.sum()
    w = w / ws
    A = np.zeros(n)
    B = np.zeros(n)
    ks = np.arange(2 * n - 1)
    for j in range(n):
        idx_a = j + ks - (n - 1)
        va = (idx_a >= 0) & (idx_a < n)
        A[j] = np.dot(w[va], h[idx_a[va]])
        idx_b = j - ks + (n - 1)
        vb = (idx_b >= 0) & (idx_b < n)
        B[j] = np.dot(w[vb], h[idx_b[vb]])
    return (1.0 - A - B) * dx

def slsqp_grad_opt(h0, T, n, maxiter=400):
    def obj(h):
        return logsumexp_obj(h, T)
    def gr(h):
        return lse_grad(h, T)
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0, 'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=gr, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_box_sum(np.clip(res.x, 0, 1), n / 2.0)

def pgm_momentum(h, p, n, niter, lr_init, beta, t_start):
    h_cur = h.copy()
    v = np.zeros(n)
    lr = lr_init
    c = n / 2.0
    for it in range(niter):
        if time.time() - t_start > 1100 * 0.95:
            break
        corr = np.maximum(get_corr(h_cur), 0.0)
        sp = np.sum(corr ** p)
        if sp < 1e-30:
            break
        base = sp ** (1.0 / p - 1.0)
        ks = np.arange(2 * n - 1)
        A = np.zeros(n)
        B = np.zeros(n)
        w = corr ** (p - 1)
        ws = w.sum()
        if ws < 1e-30:
            break
        w_n = w / ws
        for j in range(n):
            ia = j + ks - (n - 1)
            va = (ia >= 0) & (ia < n)
            A[j] = np.dot(w_n[va], h_cur[ia[va]])
            ib = j - ks + (n - 1)
            vb = (ib >= 0) & (ib < n)
            B[j] = np.dot(w_n[vb], h_cur[ib[vb]])
        grad = base * (1.0 - A - B) * (2.0 / n)
        v = beta * v - lr * grad
        h_new = project_box_sum(h_cur + v, c)
        f_new = pnorm_obj(h_new, p)
        f_old = pnorm_obj(h_cur, p)
        if f_new > f_old:
            lr *= 0.5
            v = beta * v
            h_new = project_box_sum(h_cur + v, c)
            if f_new < pnorm_obj(h_new, p) + 1e-15:
                pass
            h_new = project_box_sum(h_cur - lr * grad, c)
            if pnorm_obj(h_new, p) < f_old:
                h_cur = h_new
            else:
                h_cur = project_box_sum(h_cur - 0.1 * lr * grad, c)
        else:
            h_cur = h_new
            if it % 20 == 0 and lr < lr_init:
                lr *= 1.1
        if it % 50 == 0:
            lr = max(lr * 0.9, 0.001)
    return h_cur

def active_set_slsqp(h, n, k_top=5, T=200, maxiter=300):
    corr = get_corr(h)
    top_k = np.argsort(corr)[-k_top:]
    def obj(h):
        c = get_corr(h)
        sel = c[top_k]
        m = np.max(sel)
        return (m + np.log(np.sum(np.exp(T * (sel - m)))) / T) * (2.0 / n)
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0, 'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_box_sum(np.clip(res.x, 0, 1), n / 2.0)

def coordinate_descent(h, n, n_pass=20):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for _ in range(n_pass):
        improved = False
        corr = get_corr(h)
        k_worst = np.argmax(corr)
        for j in np.random.permutation(n):
            for d in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                h2 = h.copy()
                h2[j] += d
                h2 = project_box_sum(h2, n / 2.0)
                c5_2 = compute_c5(h2)
                if c5_2 < best_c5 - 1e-13:
                    h = h2
                    best_c5 = c5_2
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[::2] = 1.0
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: SLSQP with gradient, moderate T
    for T in [50, 200, 600]:
        if elapsed() > budget_s * 0.10:
            break
        hc = slsqp_grad_opt(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: P-norm projected gradient with momentum
    p_list = [4, 8, 16, 32, 64]
    for p in p_list:
        if elapsed() > budget_s * 0.35:
            break
        h_pgm = pgm_momentum(best_h, p, n, niter=200, lr_init=0.3, beta=0.5, t_start=t0)
        c5v = compute_c5(h_pgm)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_pgm
        # SLSQP polish after P-norm
        hc = slsqp_grad_opt(best_h, 400, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 3: Active-set focused SLSQP
    if elapsed() < budget_s * 0.50:
        for k_top in [3, 5, 8]:
            hc = active_set_slsqp(best_h, n, k_top=k_top, T=300, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    # Phase 4: Basin hopping with P-norm energy
    T_bh = 300
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = 80
    for it in range(n_bh):
        if elapsed() > budget_s * 0.72:
            break
        step = 0.15 * (1.0 - 2.0 * it / n_bh) + 0.04
        h_pert = bh_h + step * np.random.randn(n)
        h_pert = project_box_sum(h_pert, n / 2.0)
        hc = slsqp_grad_opt(h_pert, T_bh, n, maxiter=250)
        c5v = compute_c5(hc)
        temp = 0.006 * (1.0 - it / n_bh) + 0.0004
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5, bh_h = c5v, hc
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    # Phase 5: Multi-resolution exploration
    for n_try in [80, 96, 100]:
        if elapsed() > budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        for trial in range(3):
            if elapsed() > budget_s * 0.82:
                break
            h0 = project_box_sum(h_int + np.random.uniform(-0.08, 0.08, n_try), n_try / 2.0)
            for T in [100, 500]:
                if elapsed() > budget_s * 0.82:
                    break
                hc = slsqp_grad_opt(h0, T, n_try, maxiter=400)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
                    n = n_try

    # Phase 6: Coordinate descent fine-tuning
    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = coordinate_descent(best_h, len(best_h), n_pass=30)

    # Phase 7: Final high-T SLSQP polish
    n_cur = len(best_h)
    for T in [1000, 5000]:
        if elapsed() > budget_s * 0.98:
            break
        hc = slsqp_grad_opt(best_h, T, n_cur, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
