2. Identify the active set: shifts where C(k) > C_max - ε
3. Formulate: min C_max subject to C(k) ≤ C_max for all active k, 0≤h≤1, sum(h)=n/2
4. This is a nonlinear program (C(k) is quadratic in h)
5. Solve with SLSQP or similar

import numpy as np
import scipy.optimize as opt
from functools import partial
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = 0.5 * n - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    msk = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = int(msk.sum())
    if ni > 0 and abs(diff) < ni * 0.45:
        h[msk] += diff / ni
    if abs(np.sum(h) - 0.5 * n) > 1e-13:
        if diff > 0:
            for i in np.argsort(h):
                a = min(diff, 1.0 - h[i])
                h[i] += a; diff -= a
                if diff <= 1e-13: break
        else:
            diff = -diff
            for i in np.argsort(-h):
                r = min(diff, h[i])
                h[i] -= r; diff -= r
                if diff <= 1e-13: break
    return np.clip(h, 0.0, 1.0)

def scon(x, n):
    return np.sum(x) - 0.5 * n

def scon_j(x, n):
    return np.ones(n)

def lse_obj(h, T):
    n = len(h); dx = 2.0 / n
    c = corr_full(h)
    cm = c.max()
    return float((cm + T * np.log(np.sum(np.exp((c - cm) / T)))) * dx)

def _sg(h, j, dx):
    n = len(h); lag = j - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] += -h[lag:]
    else:
        m = -lag
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] += -h[m:]
    return g * dx

def lse_grad(h, T):
    n = len(h); dx = 2.0 / n
    c = corr_full(h)
    cm = c.max()
    w = np.exp((c - cm) / T)
    w /= w.sum()
    thr = w.max() * 1e-10
    g = np.zeros(n)
    for k in np.where(w > thr)[0]:
        g += w[k] * _sg(h, k, dx)
    return g

def slsq_opt(h, n, T, mit):
    try:
        r = opt.minimize(lse_obj, h.copy(), args=(T,),
                         jac=partial(lse_grad, T=T),
                         method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': scon, 'jac': scon_j, 'args': (n,)}],
                         options={'maxiter': mit, 'ftol': 1e-15})
        if hasattr(r, 'x') and r.x is not None:
            return fix_sum(r.x, n)
    except Exception:
        pass
    return h

def _lbfgs_f(x, T, n, lam):
    x = np.clip(x, 0.0, 1.0)
    return lse_obj(x, T) + lam * (np.sum(x) - 0.5 * n) ** 2

def _lbfgs_g(x, T, n, lam):
    x = np.clip(x, 0.0, 1.0)
    return lse_grad(x, T) + 2.0 * lam * (np.sum(x) - 0.5 * n) * np.ones(n)

def lbfgs_opt(h, n, T, mit, lam):
    try:
        r = opt.minimize(_lbfgs_f, h.copy(), args=(T, n, lam),
                         jac=partial(_lbfgs_g, T=T, n=n, lam=lam),
                         method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': mit, 'gtol': 1e-13, 'ftol': 1e-16})
        if hasattr(r, 'x') and r.x is not None:
            return fix_sum(r.x, n)
    except Exception:
        pass
    return h

def interp_h(h, nn):
    no = len(h)
    xo = np.arange(no) * 2.0 / no
    xn = np.arange(nn) * 2.0 / nn
    return fix_sum(np.interp(xn, xo, h, period=2.0), nn)

def do_update(hb, cb, nb, hc, nc):
    hc = fix_sum(hc, nc)
    c = compute_c5(hc)
    if c < cb - 1e-14:
        return hc, c, nc
    return hb, cb, nb

def perturb_h(h, n, scale, seed_rng):
    hp = h + seed_rng.randn(n) * scale
    return fix_sum(np.clip(hp, 0.0, 1.0), n)

def phase_shift_h(h, n, dpos):
    return fix_sum(np.roll(h, dpos), n)
