def grad_for_lags(h, lag_indices):
    """Compute gradients of c[lag] for specified lags. Returns (n_lags, n) array."""
    n = len(h)
    dx = 2.0 / n
    n_lags = len(lag_indices)
    G = np.zeros((n_lags, n))
    for a in range(n_lags):
        m = lag_indices[a]
        k = m - (n - 1)
        if k >= 0:
            for j in range(n - k):
                G[a, j] += (1.0 - h[j + k])
            for j in range(k, n):
                G[a, j] -= h[j - k]
        else:
            l = -k
            for j in range(l, n):
                G[a, j] += (1.0 - h[j - l])
            for j in range(n - l):
                G[a, j] -= h[j + l]
    G *= dx
    return G

def find_best_swap(h, corr, active_lags, G, c_max, tol=1e-8):
    """Find best coordinate swap. Returns (p, q, delta) or None."""
    n = len(h)
    # E[i,j] = max_a (G[a,i] - G[a,j])
    # Valid if E[i,j] < -tol
    diff = G[:, :, None] - G[:, None, :]  # (n_lags, n, n)
    E = np.max(diff, axis=0)  # (n, n)

    best_h, best_c5 = coord_swap_refine(best_h, t1_end)

    # Phase 2: Subgradient refinement (~200s)
    if elapsed() < budget_s * 0.50:
        t2_end = t_start + budget_s * 0.50
        h2, c2 = subgrad_refine(best_h.copy(), t2_end)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: Coordinate swap again after subgradient (~150s)
    if elapsed() < budget_s * 0.63:
        t3_end = t_start + budget_s * 0.63
        best_h, best_c5 = coord_swap_refine(best_h, t3_end)

    # Phase 4: SLSQP smooth polish at multiple temperatures (~200s)
    if elapsed() < budget_s * 0.80:
        t4_end = t_start + budget_s * 0.80
        for T in [50, 200, 1000, 5000]:
            if elapsed() > t4_end:
                break
            h4 = slsqp_smooth_polish(best_h.copy(), T, maxiter=500)
            c4 = compute_c5(h4)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4

    # Phase 5: Basin hopping with coord swap (~150s)
    if elapsed() < budget_s * 0.93:
        t5_end = t_start + budget_s * 0.93
        n_cur = len(best_h)
        n_trials = int((t5_end - time.time()) / 8)
        for i in range(min(n_trials, 25)):
            if elapsed() > t5_end:
                break
            scale = 0.08 * (1.0 - 0.03 * i)
            h_pert = best_h + np.random.randn(n_cur) * scale
            h_pert = project_feasible(h_pert, n_cur)
            h_opt, c_opt = coord_swap_refine(h_pert, time.time() + 6)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.97:
        h_final = slsqp_smooth_polish(best_h.copy(), 10000, maxiter=600)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final
        best_h, best_c5 = coord_swap_refine(best_h, time.time() + 5)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
