        h_rand = binarize(np.full(n0, 0.5) + rng.randn(n0) * 0.08, n0)
        # Mix with current best
        h_rand = binarize(0.6 * h_rand + 0.4 * h_ls, n0)
        h_ls_r, c5_r = bls(h_rand, n0, tleft, max_iters=80000, seed=seed + 200 + attempt)
        update(h_ls_r, c5_r)

    # Phase 5: Final polish - try tiny continuous adjustments
    if tleft() > 20:
        h_fin = best_h.astype(float)
        dx = 2.0 / best_n
        corr = np.correlate(h_fin, 1.0 - h_fin, mode='full')
        k_worst = int(np.argmax(corr))
        # Nudge values slightly at worst-lag-contributing positions
        for dbg in [0.02, 0.01, 0.005]:
            h_t = h_fin.copy()
            for i in range(best_n):
                j = i + (k_worst - (best_n - 1))
                if 0 <= j < best_n and h_fin[i] > 0.5 and h_fin[j] < 0.5:
                    h_t[i] = max(0.0, h_t[i] - dbg)
                    h_t[j] = min(1.0, h_t[j] + dbg)
            h_t = proj(h_t, best_n)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                update(h_t, c5_t)
                h_fin = h_t

    return best_h, best_c5, best_n
