import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = T * (corr - m)
    sm = np.max(sh)
    return (m + (sm + np.log(np.sum(np.exp(sh - sm)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = T * (corr - m)
    sm = np.max(sh)
    ev = np.exp(sh - sm)
    w = ev / ev.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-14:
            continue
        lag = j - (n - 1)
        if lag == 0:
            grad += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w[j] * (1.0 - h[:-lag])
            grad[:n - lag] += -w[j] * h[lag:]
        else:
            p = -lag
            grad[:n - p] += w[j] * (1.0 - h[p:])
            grad[p:] += -w[j] * h[:-p]
    return grad * dx

def obj_wrapper(h, T):
    return smooth_val(h, T)

def grad_wrapper(h, T):
    return smooth_grad(h, T)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def lbfgsb_opt(h0, T, n, maxiter=800, maxfun=4000):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_wrapper, h0, jac=grad_wrapper, args=(T,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxfun': maxfun})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def swap_perturb(h, n, rng, n_swaps=5):
    hp = h.copy()
    for _ in range(n_swaps):
        i, j = rng.choice(n, 2, replace=False)
        amt = rng.uniform(0.02, 0.12)
        hp[i] -= amt
        hp[j] += amt
    return project_feasible(hp, n)

def block_perturb(h, n, rng):
    hp = h.copy()
    blk = max(3, n // 12)
    for _ in range(3):
        s = rng.randint(0, n - blk)
        shift = rng.uniform(-0.1, 0.1)
        hp[s:s+blk] += shift
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Progressive T annealing with L-BFGS-B
    for T in [5, 15, 50, 150, 400, 1000, 3000, 8000]:
        if rem() < 20:
            break
        hc = lbfgsb_opt(best_h, T, n, maxiter=600, maxfun=3000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Aggressive basin hopping (L-BFGS-B is fast)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    T_bh = 400
    n_iter = 150
    for it in range(n_iter):
        if rem() < 80:
            break
        r = it / n_iter
        # Mix of perturbation types
        ptype = rng.randint(3)
        if ptype == 0:
            step = 0.12 * (1.0 - r) + 0.02
            h_pert = bh_h + step * rng.randn(n)
            h_pert = project_feasible(h_pert, n)
        elif ptype == 1:
            h_pert = swap_perturb(bh_h, n, rng, n_swaps=4)
        else:
            h_pert = block_perturb(bh_h, n, rng)
        hc = lbfgsb_opt(h_pert, T_bh, n, maxiter=150, maxfun=800)
        c5v = compute_c5(hc)
        temp = 0.004 * (1.0 - r) + 0.0002
        if c5v < bh_c5 or rng.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-8)):
            bh_c5 = c5v
            bh_h = hc
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        # Occasionally re-optimize from best with higher T
        if it % 30 == 29 and rem() > 100:
            hc2 = lbfgsb_opt(best_h, 2000, n, maxiter=400, maxfun=2000)
            c5v2 = compute_c5(hc2)
            if c5v2 < best_c5:
                best_c5 = c5v2
                best_h = hc2
                bh_h = best_h.copy()
                bh_c5 = best_c5

    # Phase 3: Diverse structured restarts
    if rem() > 50:
        starts = []
        for _ in range(3):
            hb = (rng.rand(n) < 0.5).astype(float)
            starts.append(project_feasible(hb, n))
        # Quarter-block
        q = n // 4
        hq = np.zeros(n)
        hq[:q] = 1.0
        hq[2*q:3*q] = 1.0
        starts.append(project_feasible(hq, n))
        # Triangle
        ht = np.abs(np.linspace(-1, 1, n))
        ht = project_feasible(ht, n)
        starts.append(ht)

        for h0 in starts:
            if rem() < 30:
                break
            for T in [200, 800]:
                if rem() < 15:
                    break
                hc = lbfgsb_opt(h0, T, n, maxiter=400, maxfun=2000)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc

    # Phase 4: Targeted subgradient on max shift
    if rem() > 30:
        for sweep in range(80):
            if rem() < 10:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            k_max = np.argmax(corr)
            lag = k_max - (n - 1)
            improved = False
            if 0 < abs(lag) < n:
                for _ in range(10):
                    if lag > 0:
                        cands = np.where((best_h > 0.6) & (best_h[lag:] < 0.4))[0]
                        if len(cands) == 0:
                            break
                        j = cands[rng.randint(len(cands))] + lag
                    else:
                        p = -lag
                        cands = np.where((best_h < 0.4) & (best_h[p:] > 0.6))[0]
                        if len(cands) == 0:
                            break
                        j = cands[rng.randint(len(cands))]
                    h_try = best_h.copy()
                    src = j - lag
                    if 0 <= src < n and 0 <= j < n:
                        amt = 0.04
                        if src > j:
                            pass
                        h_try[j] -= amt
                        h_try[src] += amt
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-12:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
                            break
            if not improved and sweep > 40:
                break

    # Phase 5: Final high-T polish
    for T in [5000, 20000, 80000]:
        if rem() < 5:
            break
        hc = lbfgsb_opt(best_h, T, n, maxiter=400, maxfun=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final: verify and return
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
