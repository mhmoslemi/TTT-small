import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    s = (corr - c) / T
    s = np.clip(s, -60, 0)
    return (c + T * np.log(np.sum(np.exp(s)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for i in range(n_ones - target):
            b[ones[order[i]]] = 0
    elif n_ones < target:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(target - n_ones):
            b[zeros[order[i]]] = 1
    return b

def slsqp_step(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def binary_local_search(b, n, max_time, rng):
    c5 = compute_c5(b)
    best = c5
    best_b = b.copy()
    t0 = time.time()
    stale = 0
    while time.time() - t0 < max_time:
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for ii in range(len(zeros)):
            i = zeros[ii]
            for jj in range(len(ones)):
                j = ones[jj]
                if time.time() - t0 > max_time:
                    return best_b, best
                b[i], b[j] = 1.0, 0.0
                c5n = compute_c5(b)
                if c5n < c5 - 1e-12:
                    c5 = c5n
                    if c5 < best:
                        best = c5
                        best_b = b.copy()
                    improved = True
                    break
                b[i], b[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            stale += 1
            if stale >= 10:
                break
        else:
            stale = 0
    return best_b, best
