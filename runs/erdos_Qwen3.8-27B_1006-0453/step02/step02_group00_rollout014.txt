```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    corr = compute_corr(h)
    n = len(h)

    m = np.max(corr)
    if m < 0:
        m = 0
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    if not np.isfinite(val):
        val = m + 1.0/T * np.log(len(corr))
    return val * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    scaled = T * (corr - m)
    scaled = np.clip(scaled, -700, 700)
    val = m + np.log(np.sum(np.exp(scaled)) + 1e-300) / T
    return val * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_topk(h, top_k=3):
    n = len(h)
    corr = compute_corr(h)
    order = np.argsort(corr)[::-1][:top_k]
    vals = corr[order]
    T_w = 80.0
    wts = np.exp(T_w * (vals - np.max(vals)))
    wts /= wts.sum()
    grad = np.zeros(n)
    for w, idx in zip(wts, order):
        k = int(idx) - (n - 1)
        if k >= 0:
            if n - 1 - k >= 0:
                grad[0:n-k] += w * (1.0 - h[k:n])
            if k < n:
                grad[k:n] -= w * h[0:n-k]
        else:
            m = -k
            if n - 1 - m >= 0:
                grad[0:n-m] += w * (1.0 - h[m:n])
            if m < n:
                grad[m:n] -= w * h[0:n-m]
    return grad, np.max(corr)

def subgradient_search(h, n, max_iters=500):
    best = h.copy()
    best_val = compute_c5(best)
    alpha = 0.05
    for it in range(max_iters):
        grad, mc = subgradient_topk(best, top_k=3)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        step = alpha / (1.0 + 0.05 * it)
        h_new = project_feasible(best - step * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_val:
            best = h_new
            best_val = c5_new
        else:
            alpha *= 0.6
            if alpha < 1e-6:
                break
    return best, best_val

def nm_smooth(h, T, n, maxiter=2000):
    def obj(h_in):
        h_c = np.clip(h_in, 0, 1)
        pen = 50.0 * (h_c.sum() - n / 2.0) ** 2
        return smooth_obj(h_c, T) + pen
    res = minimize(obj, h, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-12,
                            'fatol': 1e-14, 'adaptive': True})
    return project_feasible(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, n_iter=300, top_k=5):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        corr = compute_corr(best)
        idxs = np.argsort(corr)[::-1][:top_k]
        improved = False
        for idx in idxs:
            k = int(idx) - (n - 1)
            if k >= 0:
                cands = np.unique(np.concatenate([np.arange(0, n - k), np.arange(k, n)]))
            else:
                cands = np.unique(np.concatenate([np.arange(0, n + k), np.arange(-k, n)]))
            np.random.shuffle(cands)
            cands = cands[:12]
            for j in cands:
                for d in [0.03, -0.03, 0.07, -0.07, 0.12, -0.12]:
                    ht = best.copy()
                    ht[j] += d
                    ht = project_feasible(ht, n)
                    c5t = compute_c5(ht)
                    if c5t < best_c5 - 1e-15:
                        best = ht
                        best_c5 = c5t
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and it > 30:
            break
    return best, best_c5

def downsample(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    n_list = sorted(set([40, 60, 80, 100, len(best_h), 120, 140, 160]))
    
    for n in n_list:
        if elapsed() > budget_s * 0.82:
            break
        if n == len(best_h):
            h_cur = best_h.copy()
        else:
            h_cur = downsample(best_h, n)
        h_cur = project_feasible(h_cur, n)
        c5_cur = compute_c5(h_cur)
        if c5_cur >= best_c5 + 0.005:
            continue

        # Subgradient phase
        h_cur, c5_cur = subgradient_search(h_cur, n, max_iters=400)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

        # Nelder-Mead polish on this n
        for T in [200, 1000, 5000]:
            if elapsed() > budget_s * 0.82:
                break
            try:
                h_nm = nm_smooth(h_cur, T, n, maxiter=1500)
                c5_nm = compute_c5(h_nm)
                if c5_nm < c5_cur:
                    h_cur = h_nm
                    c5_cur = c5_nm
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
            except Exception:
                pass

        # Coordinate descent
        h_cd, c5_cd = coord_descent(h_cur, n, n_iter=200)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_cur = h_cd
            c5_cur = c5_cd

    # Final intensive polish on best
    n_best = len(best_h)
    if elapsed() < budget_s * 0.92:
        h_cur = best_h.copy()
        for T in [3000, 10000, 30000]:
            if elapsed() > budget_s * 0.92:
                break
            try:
                h_nm = nm_smooth(h_cur, T, n_best, maxiter=3000)
                c5_nm = compute_c5(h_nm)
                if c5_nm < best_c5:
                    best_c5 = c5_nm
                    best_h = h_nm
                    h_cur = h_nm
            except Exception:
                pass

    if elapsed() < budget_s * 0.97:
        h_cur = best_h.copy()
        h_cur, c5_sg = subgradient_search(h_cur, n_best, max_iters=600)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_cur

    if elapsed() < budget_s * 0.995:
        h_cd, c5_cd = coord_descent(best_h, n_best, n_iter=400, top_k=7)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
