from scipy.optimize import minimize, minimize_scalar, dual_annealing
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(corr, T):
    m = np.max(corr)
    if m < 1e-15:
        return m
    shifted = corr - m
    vals = np.exp(np.clip(T * shifted, -50, 50))
    return m + np.log(np.sum(vals)) / T

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, T) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, n_sweeps=5):
    """Coordinate descent: for each j, find best h[j] in {0,1,current,±steps}"""
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            best_v = h[j]
            best_c5 = compute_c5(h)
            for v in [0.0, 1.0, 0.5, h[j]+0.15, h[j]-0.15, h[j]+0.05, h[j]-0.05]:
