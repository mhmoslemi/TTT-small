def project_exact(h, c):
    """Exact projection onto {0<=h<=1, sum(h)=c}"""
    n = len(h)
    if c <= 0: return np.zeros(n)
    if c >= n: return np.ones(n)
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

    e = np.exp(s)

            gn = n - d

                g[:gn] -= wk * h[d:]

                 options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-16})

    h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    passes = 0
    while improved and passes < max_passes:

        passes += 1

    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        T = T_start * (T_end / T_start) ** (step / n_steps)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0

        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def _ceq(x):
    return np.array([x.sum() - 140.0])

def _ceqj(x):
    return np.ones((1, 280))

def _obj(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = corr.max()
    s = np.clip(t * (corr - cm), -50, 0)
    val = (cm + np.log(np.sum(np.exp(s))) / t) * dx
    pen = lam * (h.sum() - n * 0.5) ** 2
    return val + pen

def _obj_grad(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = corr.max()
    s = np.clip(t * (corr - cm), -50, 0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        sk = k - (n - 1)
        wk = w[k]
        if sk >= 0:
            gn = n - sk
            if gn > 0:
                g[:gn] += wk * (1.0 - h[sk:])
                g[sk:] -= wk * h[:gn]
        else:
            d = -sk
            gn = n - d
            if gn > 0:
                g[d:] += wk * (1.0 - h[:gn])
                g[:gn] -= wk * h[d:]
    g *= dx
    g += 2.0 * lam * (h.sum() - n * 0.5)
    return g

def lbfgs_opt(h, n, t, lam, maxiter=3000, ftol=1e-16):
    bnds = [(0.0, 1.0)] * n
    r = minimize(_obj, h, jac=_obj_grad, args=(t, lam),
                 method='L-BFGS-B', bounds=bnds,
                 options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-17})
    return project_feasible(r.x, n)

def slsqp_opt(h, n, t, maxiter=2000, ftol=1e-16):
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _ceq, 'jac': _ceqj}
    r = minimize(_obj, h, jac=_obj_grad, args=(t, 0.0),
                 method='SLSQP', bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(r.x, n)

def binary_swap_search(h, n, max_passes=15, time_limit=15.0, t0=None):
    if t0 is None:
        t0 = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passes += 1
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = -1e-14
        best_pair = None
        for idx_i in range(len(ones)):
            i = ones[idx_i]
            for idx_j in range(len(zeros)):
                j = zeros[idx_j]
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = c5 - best_c5
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_pair = (i, j)
            if passes > 3 and (idx_i + 1) % 20 == 0 and time.time() - t0 > time_limit:
                break
        if best_pair is not None and best_delta < -1e-13:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 += best_delta
            improved = True
    return h, best_c5

def sa_binary(h, n, n_steps=3000, T_start=0.05, T_end=0.001, t0=None):
    if t0 is None:
        t0 = time.time()
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() - t0 > 25:
            break
        T = T_start * (T_end / T_start) ** (step / float(n_steps))
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best, best_c5

def interp_up(h, nd):

    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return project_feasible(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    if n % nd == 0:
        return project_feasible(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return project_feasible(np.interp(xd, xs, h), nd)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    N = 280

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = project_feasible(np.array(initial_h_values, dtype=float), N)
    except Exception:
        bh = np.zeros(N)
        bh[np.sort(rng.choice(N, N // 2, replace=False))] = 1.0
    bc5 = compute_c5(bh)

    # Phase 1: L-BFGS-B refinement from initial
    for t_val, lam in [(50, 10.0), (200, 10.0), (500, 5.0), (2000, 2.0)]:
        if tleft() < 30:
            break
        try:
            h_new = lbfgs_opt(bh.copy(), N, t_val, lam, 4000, 1e-16)
            c5 = compute_c5(h_new)
            if c5 < bc5:
                bc5 = c5
                bh = h_new
        except Exception:
            pass

    # Phase 2: SLSQP fine-tune
    if tleft() > 80:
        for t_val in [100, 500, 2000, 8000]:
            if tleft() < 20:
                break
            try:
                h_new = slsqp_opt(bh.copy(), N, t_val, 2500, 1e-16)
                c5 = compute_c5(h_new)
                if c5 < bc5:
                    bc5 = c5
                    bh = h_new
            except Exception:
                pass

    # Phase 3: Binary swap search
    h_bin = (bh > 0.5).astype(float)
    h_bin = project_feasible(h_bin, N)
    c5b = compute_c5(h_bin)
    if c5b < bc5:
        bc5 = c5b
        bh = h_bin
    if tleft() > 60:
        h_bin, c5b = binary_swap_search(h_bin, N, max_passes=12, time_limit=30, t0=time.time())
        if c5b < bc5:
            bc5 = c5b
            bh = h_bin

    # Phase 4: L-BFGS-B on best (including binary)
    if tleft() > 60:
        for t_val, lam in [(80, 5.0), (400, 3.0), (1500, 1.0), (5000, 0.5)]:
            if tleft() < 18:
                break
            try:
                h_new = lbfgs_opt(bh.copy(), N, t_val, lam, 5000, 1e-17)
                c5 = compute_c5(h_new)
                if c5 < bc5 - 1e-10:
                    bc5 = c5
                    bh = h_new
            except Exception:
                pass

    # Phase 5: Simulated annealing binary search
    if tleft() > 50:
        h_sa = (bh > 0.5).astype(float)
        h_sa = project_feasible(h_sa, N)
        h_sa, c5_sa = sa_binary(h_sa, N, n_steps=5000, T_start=0.04, T_end=0.0005, t0=time.time())
        if c5_sa < bc5:
            bc5 = c5_sa
            bh = h_sa
        h_sa, c5_sa = binary_swap_search(h_sa, N, max_passes=8, time_limit=12, t0=time.time())
        if c5_sa < bc5:
            bc5 = c5_sa
            bh = h_sa

    # Phase 6: Random restart with L-BFGS-B
    if tleft() > 100:
        for restart in range(3):
            if tleft() < 40:
                break
            h_rand = np.zeros(N)
            h_rand[np.sort(rng.choice(N, N // 2, replace=False))] = 1.0
            h_rand = project_feasible(h_rand, N)
            # Perturb
            h_rand = project_feasible(h_rand + rng.randn(N) * 0.1, N)
            improved_this = False
            for t_val, lam in [(100, 10.0), (500, 5.0), (2000, 2.0)]:
                if tleft() < 20:
                    break
                try:
                    h_new = lbfgs_opt(h_rand.copy(), N, t_val, lam, 3000, 1e-16)
                    c5 = compute_c5(h_new)
                    if c5 < bc5 - 1e-10:
                        bc5 = c5
                        bh = h_new
                        improved_this = True
                    h_rand = h_new
                except Exception:
                    pass
            if improved_this:
                pass

    # Phase 7: Final SLSQP polish
    if tleft() > 25:
        for t_val in [1000, 5000, 20000]:
            if tleft() < 12:
                break
            try:
                h_new = slsqp_opt(bh.copy(), N, t_val, 3000, 1e-16)
                c5 = compute_c5(h_new)
                if c5 < bc5:
                    bc5 = c5
                    bh = h_new
            except Exception:
                pass

    # Phase 8: Final binary polish
    h_bin = (bh > 0.5).astype(float)
    h_bin = project_feasible(h_bin, N)
    c5b = compute_c5(h_bin)
    if c5b < bc5:
        bc5 = c5b
        bh = h_bin
    if tleft() > 15:
        h_bin, c5b = binary_swap_search(h_bin, N, max_passes=8, time_limit=10, t0=time.time())
        if c5b < bc5:
            bc5 = c5b
            bh = h_bin

    # Phase 9: Final L-BFGS-B ultra-fine
    if tleft() > 15:
        try:
            h_new = lbfgs_opt(bh.copy(), N, 10000, 0.5, 8000, 1e-18)
            c5 = compute_c5(h_new)
            if c5 < bc5:
                bc5 = c5
                bh = h_new
        except Exception:
            pass

    bh = project_feasible(bh, N)
    final_c5 = compute_c5(bh)
    return bh, final_c5, N
