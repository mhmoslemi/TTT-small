```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
        # Re-fix if needed
        s2 = np.sum(h)
        if abs(s2 - target) > 1e-10:
            diff2 = target - s2
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-12:

        else:

            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return h

    return np.correlate(h, 1.0 - h, mode='full')

            g[:m] -= h[s:]                 # -h[j+s]
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return g
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        ss = -s
        m = n - ss
        if m <= 0:
            return g
        g[ss:] += 1.0 - h[:m]
        g[:m] -= h[ss:]
    return g

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    if T < 1e-12:
        return c_max * dx
    shifted = np.clip((corr - c_max) / T, -40.0, 0.0)
    return (c_max + T * float(np.log(np.sum(np.exp(shifted))))) * dx

def sum_con(h, n):
    return float(np.sum(h)) - n * 0.5

def do_slsqp(h, n, T, max_iter):
    try:
        res = opt.minimize(
            soft_max_obj, h.copy(), args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def proj_subgrad(h, n, n_iter, lr_init, top_k, momentum):
    h = h.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for idx in top:
            grad += grad_shift(h, n, int(idx) - (n - 1))
        grad /= top_k
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 200:
            lr *= 0.9
            v *= 0.5
            stall = 0
    return best_h, best_c5

def coord_descent(h, n, n_sweeps, deltas):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
```

```python
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
        np.clip(h, 0.0, 1.0, out=h)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h


def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return g
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        ss = -s
        m = n - ss
        if m <= 0:
            return g
        g[ss:] += 1.0 - h[:m]
        g[:m] -= h[ss:]
    return g


def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    if T < 1e-12:
        return c_max * dx
    sh = np.clip((corr - c_max) / T, -40.0, 0.0)
    return (c_max + T * float(np.log(np.sum(np.exp(sh))))) * dx


def sum_con(h, n):
    return float(np.sum(h)) - n * 0.5

    try:
        res = opt.minimize(
            soft_max_obj, h.copy(), args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h


def proj_subgrad(h, n, n_iter, lr0, tk, mom):
    h = h.copy()
    v = np.zeros(n)
    best_h, best_c5 = h.copy(), compute_c5(h)
    lr, stall = lr0, 0
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(corr)[-tk:]
        grad = np.zeros(n)
        for idx in top:
            grad += grad_shift(h, n, int(idx) - (n - 1))
        grad /= tk
        v = mom * v - lr * grad
        h = fix_sum(h + v, n)
        c5 = compute_c5(h)
        if c5 < best_c5:

        else:
            stall += 1
        if stall > 300:

            v *= 0.3
            stall = 0
    return best_h, best_c5


def coord_desc(h, n, sweeps):
    h = h.copy()
    best_c5 = compute_c5(h)
    dels = [0.01, -0.01, 0.03, -0.03, 0.06, -0.06]
```

```python
import numpy as np
import scipy.optimize as opt
import time
```


```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
```


```python
def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    s = float(np.sum(h))
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = int(np.sum(interior))
    if ni > 0 and abs(diff) <= ni * 0.49:
        h[interior] += diff / ni
        np.clip(h, 0.0, 1.0, out=h)
    elif diff > 0:
        for i in np.argsort(h):
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        da = -diff
        for i in np.argsort(-h):
            rem = min(da, h[i])
            h[i] -= rem
            da -= rem
            if da <= 1e-14:
                break
    s2 = float(np.sum(h))
    if abs(s2 - target) > 1e-12:
        h += (target - s2) / n
        np.clip(h, 0.0, 1.0, out=h)
    return h
```


```python
def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return g
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        ss = -s
        m = n - ss
        if m <= 0:
            return g
        g[ss:] += 1.0 - h[:m]
        g[:m] -= h[ss:]
    return g
```


```python
def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    if T < 1e-12:
        return c_max * dx
    sh = np.clip((corr - c_max) / T, -40.0, 0.0)
    return (c_max + T * float(np.log(np.sum(np.exp(sh))))) * dx
```


```python
def sum_con(h, n):
    return float(np.sum(h)) - n * 0.5
```


```python
def do_slsqp(h, n, T, max_iter):

            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],

        if hasattr(res, 'x') and res.x is not None:

    return h
```
