        scale = 0.05 * np.sqrt(T / T_init)
        h_new = cur.copy()
        h_new[idx] += rng.randn(k) * scale
        h_new = project_feasible(h_new, n)
        f_new = compute_c5(h_new)
        d = f_new - cur_f
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            cur = h_new
            cur_f = f_new
            if f_new < best_f:
                best_f = f_new
                best = cur.copy()
        T = max(T * 0.99996, T_min)
    return best, best_f

def hill_climb_binary(h, n_rounds=50, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    cur = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(cur.sum())
    if n_ones < target:
        zeros = np.where(cur < 0.5)[0]
        cur[zeros[:target - n_ones]] = 1.0
    elif n_ones > target:
        ones = np.where(cur > 0.5)[0]
        cur[ones[:n_ones - target]] = 0.0
    cur = project_feasible(cur, n)
    cur_f = compute_c5(cur)
    for _ in range(n_rounds):
        improved = False
        zeros = np.where(cur < 0.5)[0]
        ones = np.where(cur > 0.5)[0]
        rz = rng.permutation(len(zeros))
        ro = rng.permutation(len(ones))
        for zi in range(len(zeros)):
            for oi in range(len(ones)):
                i = zeros[rz[zi]]
                j = ones[rz[oi]] if False else ones[ro[oi]]
                cur[j] = 0.0
                cur[i] = 1.0
                f = compute_c5(cur)
                if f < cur_f - 1e-12:
                    cur_f = f
                    improved = True
                    zeros = np.where(cur < 0.5)[0]
                    ones = np.where(cur > 0.5)[0]
                else:
                    cur[i] = 0.0
                    cur[j] = 1.0
        if not improved:
            break
    return cur, cur_f

def make_structured_init(n, mode=0):
    h = np.zeros(n)
    if mode == 0:
        h[::2] = 1.0
    elif mode == 1:
        h[:n//2] = 1.0
    elif mode == 2:
        q = 7
        for i in range(n):
            if (i * q) % n < n // 2:
                h[i] = 1.0
    elif mode == 3:
        for i in range(n):
            h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_structured_init(90, mode=0)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary SA from initial (35% of time)
    t1 = t0 + budget_s * 0.35
    h_b1, c5_b1 = binary_sa(best_h, t1, seed=seed, T_init=0.04, cooling=0.99999)
    if c5_b1 < best_c5:
        best_c5 = c5_b1
        best_h = h_b1.copy()

    # Phase 2: Continuous SLSQP from binary result (15% of time)
    t2 = t0 + budget_s * 0.50
    if time.time() < t2:
        h_s = slsqp(best_h, T=200, maxiter=300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Continuous SA (15% of time)
    t3 = t0 + budget_s * 0.65
    if time.time() < t3:
        h_cs, c5_cs = cont_sa(best_h, t3, seed=seed + 777, T_init=0.005)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs.copy()

    # Phase 4: Binary SA restart - structured inits (15% of time)
    t4 = t0 + budget_s * 0.80
    for mode in range(4):
        if time.time() > t4:
            break
        h_init = make_structured_init(best_n, mode=mode)
        h_b, c5_b = binary_sa(h_init, min(time.time() + budget_s * 0.03, t4),
                              seed=seed + 2000 + mode * 100, T_init=0.03, cooling=0.99998)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Quick SLSQP on best binary
        if c5_b < best_c5 + 0.01 and time.time() < t4:
            h_s2 = slsqp(h_b, T=300, maxiter=200)
            c5_s2 = compute_c5(h_s2)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2.copy()

    # Phase 5: Multi-resolution try (5% of time)
    t5 = t0 + budget_s * 0.85
    for n_alt in [84, 96, 80]:
        if time.time() > t5:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_new, x_old, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        h_alt_b, c5_alt_b = binary_sa(h_alt, time.time() + 10, seed=seed + 5000, T_init=0.03, cooling=0.99998)
        c5_alt = compute_c5(h_alt_b)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt_b.copy()
            best_n = n_alt

    # Phase 6: Final SLSQP polish (10% of time)
    t6 = t0 + budget_s * 0.95
    for T in [100, 500, 2000, 8000]:
        if time.time() > t6:
            break
        h_f = slsqp(best_h, T=T, maxiter=400)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 7: One more SA cycle on best (last 3%)
    t7 = t0 + budget_s * 0.985
    if time.time() < t7:
        h_fin, c5_fin = cont_sa(best_h, t7, seed=seed + 9999, T_init=0.003)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
