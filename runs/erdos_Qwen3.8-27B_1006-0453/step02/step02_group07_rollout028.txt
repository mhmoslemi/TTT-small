import time

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return float(c * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, T, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    val = (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx
    if lam > 0:
        val += lam * (np.sum(h) - 0.5 * n) ** 2
    return val

                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < bc - 1e-13:
                    bc = c
                    bd = d
            if bd != 0.0:
                h[:] -= bd / (n - 1)
                h[j] += bd
                h = np.clip(h, 0.0, 1.0)
                h = fix_sum(h, n)
                improved = True
        if not improved:
            break
    return h

def pair_polish(h, n, rng, npairs=400):
    for _ in range(npairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0-1e-4 or h[l] <= 1e-4 or h[l] >= 1.0-1e-4:
            continue
        bc = compute_c5(h)
        bd = 0.0
        for d in (-0.03, -0.01, -0.003, 0.003, 0.01, 0.03):
            if h[j]+d < 1e-4 or h[j]+d > 1-1e-4 or h[l]-d < 1e-4 or h[l]-d > 1-1e-4:
                continue
            hn = h.copy()
            hn[j] += d
            hn[l] -= d
            c = compute_c5(hn)
            if c < bc - 1e-13:
                bc = c
                bd = d
        if bd != 0.0:
            h[j] += bd
            h[l] -= bd
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)
    t0 = time.time()
    rem = lambda: budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    # Phase 1: Fast projected gradient with momentum
    if rem() > 250:
        for T, steps, lr in [(0.03, 3000, 0.015), (0.01, 4000, 0.01),
                             (0.004, 5000, 0.006), (0.0015, 6000, 0.004),
                             (0.0005, 8000, 0.002)]:
            if rem() < 200:
                break
            h = pgd(h, T, steps, lr)
            h = fix_sum(h, n)
            nc = compute_c5(h)
            if nc < c5:
                c5 = nc

    # Phase 2: L-BFGS-B refinement
    if rem() > 150:
        for T in (0.005, 0.001, 0.0003):
            if rem() < 120:
                break
            h = lbfgsb_refine(h, n, T, 300)
            h = fix_sum(h, n)
            nc = compute_c5(h)
            if nc < c5:
                c5 = nc

    # Phase 3: SLSQP precision
    if rem() > 100:
        h = slsqp_refine(h, n, 0.0001, 600)
        h = fix_sum(h, n)
        nc = compute_c5(h)
        if nc < c5:
            c5 = nc

    # Phase 4: Coordinate + pair polish
    if rem() > 60:
        h = coord_polish(h, n, 3)
        c5 = compute_c5(h)
    if rem() > 50:
        h = pair_polish(h, n, rng, 500)
        c5 = compute_c5(h)

    # Phase 5: Diverse restarts
    if rem() > 200:
        for i in range(8):
            if rem() < 150:
                break
            ht = h.copy()
            mode = i % 3
            if mode == 0:
                ht += np.random.randn(n) * 0.015
            elif mode == 1:
                idx = np.random.choice(n, 15, replace=False)
                ht[idx] += np.random.randn(15) * 0.1
            else:
                ht += np.random.randn(n) * 0.008
            ht = fix_sum(ht, n)
            ht = pgd(ht, 0.005, 3000, 0.005)
            ht = pgd(ht, 0.001, 4000, 0.003)
            ht = lbfgsb_refine(ht, n, 0.001, 200)
            ht = fix_sum(ht, n)
            ct = compute_c5(ht)
            if ct < c5:
                h, c5 = ht, ct
                if rem() > 100:
                    h = coord_polish(h, n, 2)
                    c5 = compute_c5(h)

    # Phase 6: Upscale to 2x
    if rem() > 180:
        n2 = min(n * 2, 560)
        h2 = upsample(h, n2)
        for T, steps, lr in [(0.01, 3000, 0.008), (0.003, 4000, 0.005),
                             (0.001, 5000, 0.003)]:
            if rem() < 100:
                break
            h2 = pgd(h2, T, steps, lr)
            h2 = fix_sum(h2, n2)
        if rem() > 80:
            h2 = lbfgsb_refine(h2, n2, 0.001, 250)
            h2 = fix_sum(h2, n2)
        if rem() > 60:
            h2 = slsqp_refine(h2, n2, 0.0001, 400)
            h2 = fix_sum(h2, n2)
        if rem() > 40:
            h2 = coord_polish(h2, n2, 2)
        c5_2 = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2

    # Phase 7: Final SLSQP + polish
    if rem() > 30:
        h = slsqp_refine(h, n, 0.00005, 800)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
    if rem() > 15:
        h = coord_polish(h, n, 2)
        c5 = compute_c5(h)

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
