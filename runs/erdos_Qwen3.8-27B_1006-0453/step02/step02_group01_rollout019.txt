For coordinate descent with the sum constraint:
- We can't change just one coordinate without violating the sum constraint

def coord_descent(h, n_iter=100, penalty=10.0):
    n = len(h)
    h = h.copy()
    for it in range(n_iter):
        improved = False
        for i in range(n):
            # Try to optimize h[i] given all others fixed
            # Objective: max_p corr[p] + penalty * (sum - n/2)^2
            # corr[p] = base[p] + h[i]*(1-2*h[p-i]) for valid p
            # This is linear in h[i], so the max is convex in h[i]
            # => optimal h[i] is at a breakpoint or at the boundary [0,1]
            
            # Compute base correlations (set h[i]=0)
            h_save = h[i]
            h[i] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode='full')
            
            # For each valid p, the slope is (1 - 2*h[p-i])
            # corr[p] = corr_base[p] + h[i] * slope[p]
            # where slope[p] = (1-2*h[p-i]) if 0 <= p-i < n, else 0
            # Wait, we set h[i]=0, so corr_base already has the h[i]=0 contribution
            # The "true" corr with h[i] is:
            # corr_true[p] = corr_base[p] + h[i]*(1-h[p-i]) + h[p-i]*(1-h[i])
            #              = corr_base[p] + h[i] - h[i]*h[p-i] + h[p-i] - h[i]*h[p-i]
            #              = corr_base[p] + h[p-i] + h[i]*(1-2*h[p-i])
            # for p-i in [0,n-1]
            
            slopes = np.zeros(2*n-1)
            lo = max(0, i) # p-i >= 0 → p >= i
            hi = min(2*n-2, i+n-1) # p-i <= n-1 → p <= i+n-1
            if lo <= hi:
                p_range = np.arange(lo, hi+1)
                h_p_i = h[p_range - i]  # h[p-i] for p in range, but h[i] is 0 now!
                # Wait, h[i] = 0 now, so h[p_i] when p_i == i is 0
                # That's fine, the formula still holds
                slopes[lo:hi+1] = 1 - 2*h_p_i
            
            # corr_true[p] = corr_base[p] + h[i]*slopes[p] (approximately, since h[i]=0 in base)
            # Actually more precisely:
            # With h[i] restored to val:
            # The change from setting h[i]=0 to h[i]=val is:
            # For p in [i, i+n-1]: corr changes by val*(1-h[p-i]) + h[p-i]*(1-val) - 0
            # = val - val*h[p-i] + h[p-i] - val*h[p-i]
            # = h[p-i] + val*(1-2*h[p-i])
            # But h[p-i] when p-i==i is h[i]=0 (since we set it to 0)
            # So the formula is: delta[p] = h[p-i] + val*(1-2*h[p-i]) for p in valid range
            # But h[p-i] here is the value with h[i]=0, which is correct
            
            # So corr_with_val[p] = corr_base[p] + h[p-i] + val*(1-2*h[p-i])
            #                       = (corr_base[p] + h[p-i]) + val*(1-2*h[p-i])
            
            # The sum constraint: sum(h) with h[i]=val is (sum_base + val)
            # sum_base = h.sum() with h[i]=0
            # Target: n/2
            # Penalty: (sum_base + val - n/2)^2 * penalty
            
            # Objective as function of val ∈ [0,1]:
            # max_p [(corr_base[p] + h[p-i]) + val*slopes[p]] + penalty*(sum_base + val - n/2)^2
            # = max_p [A[p] + val*slopes[p]] + penalty*(B + val)^2
            # where A[p] = corr_base[p] + h[p-i] (for valid p, else corr_base[p])
            # B = sum_base - n/2
            # slopes[p] = 1-2*h[p-i] for valid p, 0 otherwise
            
            # This is convex in val (max of linear + quadratic)
            # Find minimum by trying critical points
            
            # Binary search or ternary search on [0, 1]
            # Since it's convex, ternary search works
            
            A = corr_base.copy()
            if lo <= hi:
                A[lo:hi+1] += h[p_range - i]
            
            B = h.sum() - n/2.0  # sum_base - n/2
            
            def obj_at_val(val):
                corr_vals = A + val * slopes
                pen = penalty * (B + val)**2
                return np.max(corr_vals) + pen
            
            # Ternary search on [0, 1]
            a, b = 0.0, 1.0
            for _ in range(30):
                m1 = a + (b-a)/3
                m2 = b - (b-a)/3
                if obj_at_val(m1) < obj_at_val(m2):
                    b = m2
                else:
                    a = m1
            
            best_val = (a+b)/2
            if best_val != h_save:
                h[i] = best_val
                improved = True
        
        if not improved:
            break
    
    return project_feasible(h)

            if not time_ok(0.88):
                break
            h_p = best_h + rng.randn(n_cur) * scales[i]
            h_p = project_feasible(h_p)
            h = coord_descent(h_p, n_passes=15)
            h = slsqp_optimize(h, maxiter=250)
            c = compute_c5(h)
            if c < best_c5:
                best_c5, best_h = c, h.copy()

    # Phase 5: Final polish
    if time_ok(0.95):
        h = coord_descent(best_h, n_passes=50, penalty=100.0)
        c = compute_c5(h)
        if c < best_c5:
            best_c5, best_h = c, h
        h = nm_refine(best_h, maxiter=300)
        c = compute_c5(h)
        if c < best_c5:
            best_c5, best_h = c, h
    if time_ok(0.99):
        h = slsqp_optimize(best_h, maxiter=500, ftol=1e-14)
        c = compute_c5(h)
        if c < best_c5:
            best_c5, best_h = c, h

    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
