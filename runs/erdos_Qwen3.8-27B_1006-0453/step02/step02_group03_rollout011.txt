    cr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cr) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    np.clip(h, 0, 1, out=h)
    return h

def grad_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full')
    mx = cr.max()
    w = np.exp(beta * (cr - mx))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1

        k2hi = n - 1 - j
        if k2lo <= k2hi:
            ks = np.arange(k2lo, k2hi + 1)
            idx = j + ks
            g[j] -= np.dot(w[o + ks], h[idx])
    return g * dx

            {'type': 'eq', 'fun': lambda x: eq_sl(x, n)}]

                     bounds=bnds, constraints=cons,

    dx = 2.0 / n
    h = h0.copy()

        t_ad += 1
        m_v = a1 * m_v + (1.0 - a1) * g
        v_v = a2 * v_v + (1.0 - a2) * g * g
        mhat = m_v / (1.0 - a1 ** t_ad)
        vhat = v_v / (1.0 - a2 ** t_ad)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        moved = False
        for s in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            h_new = proj(h - s * step, ts)
            c_new = compute_c5(h_new, dx)
            if c_new <= bc + 1e-14:
                h = h_new
                moved = True
                break
        if not moved:
            continue
        c_now = compute_c5(h, dx)
        if c_now < bc:
            bc = c_now
            bh = h.copy()
    return bh, bc

def sa_opt(h0, n_it, dl, rng):
    n = len(h0)
    ts = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    bc = compute_c5(h, dx)
    bh = h.copy()
    T0 = 0.003
    cool = 0.985
    for i in range(n_it):
        if dl is not None and time.time() > dl:
            break
        T = T0 * (cool ** i)
        idx = rng.randint(n)
        delta = rng.uniform(-0.08, 0.08)
        h_new = h.copy()
        h_new[idx] += delta
        h_new = proj(h_new, ts)
        c_new = compute_c5(h_new, dx)
        dc = c_new - bc
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-12)):
            h = h_new
            if c_new < bc:
                bc = c_new
                bh = h.copy()
    return bh, bc

def slsqp_opt(h0, maxiter=150):
    n = len(h0)
    _sl['n'] = n
    dx = 2.0 / n
    t0 = compute_c5(h0, dx)
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': _l_ineq}, {'type': 'eq', 'fun': _l_eq}]
    try:
        r = minimize(_l_obj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if r.x is not None:
            hr = proj(r.x[:n], n / 2.0)
            if compute_c5(hr, dx) < t0 + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def coord_descent(h0, n_round, dl, rng):
    n = len(h0)
    ts = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    bc = compute_c5(h, dx)
    bh = h.copy()
    for r in range(n_round):
        if dl is not None and time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in (0.04, -0.04, 0.02, -0.02, 0.08, -0.08, 0.01, -0.01):
                h[j] = sv + d
                if h[j] < -1e-9 or h[j] > 1 + 1e-9:
                    continue
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc - 1e-14:
                    bc = c
                    bh = h.copy()
                    imp = True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return bh, bc

def make_starts(n, rng, ref):
    ts = n / 2.0
    out = []
    if ref is not None:
        if len(ref) == n:
            out.append(ref.copy())
        else:
            xn = np.linspace(0, 1, n, endpoint=False)
            xo = np.linspace(0, 1, len(ref), endpoint=False)
            out.append(np.interp(xn, xo, ref))
    out.append(np.full(n, 0.5))
    for p in (max(2, n // 5), max(2, n // 3), max(3, n // 4)):
        half = p // 2
        pat = np.concatenate([np.ones(half), np.zeros(p - half)])
        out.append(np.tile(pat, int(np.ceil(n / float(p))))[:n])
    for _ in range(2):
        h = np.zeros(n)
        perm = rng.permutation(n)
        h[perm[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[perm[int(ts)]] = fr
        out.append(h.copy())
    for amp in (0.15, 0.08):
        out.append(np.full(n, 0.5) + amp * rng.randn(n))
    return [proj(s, ts) for s in out]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 5.0
    bh, bc = None, 1e9

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None

    if hr is not None:
        n0 = len(hr)
        ts0 = n0 / 2.0
        h0 = proj(hr, ts0)
        bh, bc = h0, compute_c5(h0, 2.0 / n0)

        # Phase 1: Adam with moderate beta
        h1, c1 = adam_opt(bh, 600, deadline - 120, lam=0.0)
        if c1 < bc:
            bh, bc = h1, c1
        # Phase 2: Adam with binary push
        h2, c2 = adam_opt(bh, 300, deadline - 100, lam=0.25)
        if c2 < bc:
            bh, bc = h2, c2
        # Phase 3: Simulated annealing
        h3, c3 = sa_opt(bh, 400, deadline - 80, rng)
        if c3 < bc:
            bh, bc = h3, c3
        # Phase 4: SLSQP polish
        h4 = slsqp_opt(bh, 150)
        c4 = compute_c5(h4, 2.0 / n0)
        if c4 < bc:
            bh, bc = h4, c4
        # Phase 5: Coordinate descent
        h5, c5 = coord_descent(bh, 20, deadline - 50, rng)
        if c5 < bc:
            bh, bc = h5, c5
        # Phase 6: Final Adam
        h6, c6 = adam_opt(bh, 500, deadline - 30, lam=0.1)
        if c6 < bc:
            bh, bc = h6, c6
        # Phase 7: Final SLSQP
        h7 = slsqp_opt(bh, 100)
        c7 = compute_c5(h7, 2.0 / n0)
        if c7 < bc:
            bh, bc = h7, c7

    # Try nearby n values
    n_base = len(hr) if hr is not None else 50
    for dn in (1, -1, 2, -2, 3, -3, 4, -4):
        nt = n_base + dn
        if nt < 20 or nt > 95 or time.time() > deadline - 40:
            continue
        dxt = 2.0 / nt
        tst = nt / 2.0
        starts = make_starts(nt, rng, bh)
        for hs in starts[:3]:
            if time.time() > deadline - 15:
                break
            ha, ca = adam_opt(hs, 400, deadline - 10, lam=0.1)
            hsl = slsqp_opt(ha, 80)
            csl = compute_c5(hsl, dxt)
            if csl < ca:
                ha, ca = hsl, csl
            if ca < bc:
                bh, bc = ha, ca

    # Final overall polish
    if bh is not None and time.time() < deadline - 10:
        nf = len(bh)
        hf = slsqp_opt(bh, 100)
        cf = compute_c5(hf, 2.0 / nf)
        if cf < bc:
            bh, bc = hf, cf
        ha, ca = adam_opt(bh, 200, deadline - 3, lam=0.05)
        if ca < bc:
            bh, bc = ha, ca

    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh, 0.04)

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
