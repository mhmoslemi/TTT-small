    return np.max(corr) * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # log-sum-exp for numerical stability
    val = dx * (m + np.log(np.sum(np.exp(T * (corr - m)))) / T)
    # penalty for sum constraint
    s = h.sum() - n / 2.0
    val += penalty * s * s
    return val

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

                break
            noise_scale = 0.25 * (1.0 - trial / 4.0) + 0.05
            h_n = best_h + noise_scale * np.random.randn(n_best)
            h_b, c5_b = binary_local_search(h_n, n_best, max_flips=400)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b
        # Also try direct rounding of current best
        h_b2, c5_b2 = binary_local_search(best_h, n_best, max_flips=200)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2

    # Phase 5: Post-binary L-BFGS-B refinement (relax binary solution)
    if time.time() - t0 < budget_s * 0.95:
        for T in [500, 5000, 20000]:
            if time.time() - t0 > budget_s * 0.95:
                break
            hc = lbfgsb_opt(best_h, T, n_best, 8000, 5000.0)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 6: Final targeted coordinate search
    if time.time() - t0 < budget_s * 0.985:
        for iteration in range(200):
            if time.time() - t0 > budget_s * 0.985:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            k_max = int(np.argmax(corr))
            lag = k_max - (n_best - 1)
            lo = max(0, -lag)
            hi = min(n_best, n_best - lag)
            candidates = np.arange(lo, hi)
            np.random.shuffle(candidates)
            improved = False
            for j in candidates[:20]:
                for delta in [0.02, 0.05, 0.08, -0.02, -0.05, -0.08]:
                    h_try = best_h.copy()
                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, n_best)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved and iteration > 60:
                break

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
