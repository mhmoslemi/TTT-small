def compute_grad(h, t, n, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mu = np.max(corr)
    s = np.clip(t * (corr - mu), -400, 400)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)  # shape (2n-1,)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        # dc[k]/dh[j] = (1-h[j+k-n+1])*1[0<=j+k-n+1<n] - h[j-k+n-1]*1[0<=j-k+n-1<n]
        # First term: j+k-n+1 in [0,n-1] => j in [n-1-k, n-1-k+n-1] intersect [0,n-1]
        #   = [max(0, n-1-k), min(n-1, 2n-2-k)]
        #   Since k in [0,2n-2]: n-1-k in [-(n-1), n-1], so max(0,n-1-k)
        #   2n-2-k in [0, 2n-2], so min(n-1, 2n-2-k)
        j_start1 = max(0, n - 1 - k)
        j_end1 = min(n - 1, 2 * n - 2 - k)
        if j_start1 <= j_end1:
            idx = np.arange(j_start1, j_end1 + 1)
            h_idx = idx + k - n + 1  # range [0, n-1]
            grad[idx] += wk * (1 - h[h_idx])
        
        # Second term: j-k+n-1 in [0,n-1] => j in [k-n+1, k] intersect [0,n-1]
        j_start2 = max(0, k - n + 1)
        j_end2 = min(n - 1, k)
        if j_start2 <= j_end2:
            idx = np.arange(j_start2, j_end2 + 1)
            h_idx = idx - k + n - 1  # range [0, n-1]
            grad[idx] -= wk * h[h_idx]
    
    grad *= dx
    grad += 2 * lam * (np.sum(h) - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_lse_obj(h, t, n, lam, topk=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        c = corr
    mx = np.max(c)
    s = np.clip(t * (c - mx), -400, 400)
    val = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def corr_lse_grad(h, t, n, lam, topk=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
        active = np.zeros(len(corr))
        active[idx] = 1.0
    else:
        c = corr
        active = np.ones(len(corr))
    mx = np.max(c)
    s = np.clip(t * (c - mx), -400, 400)
    exp_s = np.exp(s)
    w_sub = exp_s / np.sum(exp_s)
    w = active * 0  # full weights
    if topk and topk < len(corr):
        w[idx] = w_sub
    else:
        w = w_sub
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        j_s1 = max(0, n - 1 - k)
        j_e1 = min(n - 1, 2 * n - 2 - k)
        if j_s1 <= j_e1:
            idx1 = np.arange(j_s1, j_e1 + 1)
            h_idx = idx1 + k - n + 1
            grad[idx1] += wk * (1.0 - h[h_idx])
        j_s2 = max(0, k - n + 1)
        j_e2 = min(n - 1, k)
        if j_s2 <= j_e2:
            idx2 = np.arange(j_s2, j_e2 + 1)
            h_idx = idx2 - k + n - 1
            grad[idx2] -= wk * h[h_idx]
    
    grad *= dx
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    return grad

def optimize_analytic(h0, n, t, lam, topk=None, maxiter=5000, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    r = minimize(corr_lse_obj, h0, jac=corr_lse_grad,
                 args=(t, n, lam, topk), method='L-BFGS-B',
                 bounds=bnds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12, 'maxfun': 50000})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def binary_push(h, n):
    return proj((h > 0.5).astype(float), n)

def interp(h, n_new):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return proj(np.interp(xd, xs, h), n_new)

    t0 = time.time()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(h, t, n, lam, topk):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk and 0 < topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        c = corr
    mx = np.max(c)
    s = np.clip(t * (c - mx), -400, 400)
    val = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def grad_func(h, t, n, lam, topk):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if topk and 0 < topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
        w = np.zeros(nc)
    else:
        c = corr
        idx = np.arange(nc)
        w = np.zeros(nc)
    mx = np.max(c)
    s = np.clip(t * (c - mx), -400, 400)
    es = np.exp(s)
    w[idx] = es / np.sum(es)
    grad = np.zeros(n)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-16:
            continue
        js1 = max(0, n - 1 - k)
        je1 = min(n - 1, 2 * n - 2 - k)
        if js1 <= je1:
            i1 = np.arange(js1, je1 + 1)
            grad[i1] += wk * (1.0 - h[i1 + k - n + 1])
        js2 = max(0, k - n + 1)
        je2 = min(n - 1, k)
        if js2 <= je2:
            i2 = np.arange(js2, je2 + 1)
            grad[i2] -= wk * h[i2 - k + n - 1]
    grad *= dx
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    return grad

def opt_lbfgs(h0, n, t, lam, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    r = minimize(obj_func, h0, jac=grad_func,
                 args=(t, n, lam, topk), method='L-BFGS-B',
                 bounds=bnds, options={'maxiter': maxiter, 'ftol': ftol,
                                       'gtol': 1e-11, 'maxfun': 60000})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def bin_push(h, n):
    return proj((h > 0.5).astype(float), n)

def interp_h(h, n_new):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return proj(np.interp(xd, xs, h), n_new)
