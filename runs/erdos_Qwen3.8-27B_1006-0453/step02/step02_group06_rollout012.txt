import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max
    shifted = corr - c_max
    return c_max + T * np.log(np.sum(np.exp(shifted / T)))

def topk_obj(h, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_sorted = np.sort(corr)[::-1]
    return np.mean(c_sorted[:k])

def pnorm_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return (np.sum(corr**p) / len(corr))**(1.0/p)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def do_slsqp(h0, n, obj_func, obj_args, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        obj_func, h0, args=obj_args,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def small_perturb(h, n, sigma, seed):
    rng = np.random.RandomState(seed)
    h_p = h.copy()
    n_pert = max(4, n // 25)
    idxs = rng.choice(n, size=n_pert, replace=False)
    h_p[idxs] = np.clip(h_p[idxs] + rng.randn(n_pert) * sigma, 0, 1)
    return proj_constraint(h_p, n)

def block_perturb(h, n, seed):
    rng = np.random.RandomState(seed)
    h_p = h.copy()
    n_blocks = rng.randint(2, 5)
    for _ in range(n_blocks):
        start = rng.randint(0, n - 10)
        length = rng.randint(5, 20)
        end = min(start + length, n)
        shift = rng.uniform(-0.03, 0.03)
        h_p[start:end] = np.clip(h_p[start:end] + shift, 0, 1)
    return proj_constraint(h_p, n)
