def momentum_desc(h, N, target, dx, deadline, K_init, K_final, tau, lr0, decay, momentum):
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr))
    lr = lr0
    vel = np.zeros(N)
    it = 0
    total_iter = 5000
    while time.time() < deadline:
        frac = min(it / float(total_iter), 1.0)
        K = max(3, int(K_init + (K_final - K_init) * frac))
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        grad = topK_gradient(h, corr, K, tau)
        vel = momentum * vel - lr * grad
        h = h + vel
        h = project_sum(h, target)
        it += 1
        if it % 100 == 0:
            lr *= decay
            if lr < 1e-10:
                break
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr))
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h.copy()
    return best_h, best_c5

def coord_refine(h, N, dx, deadline, step0=0.1):
    target = N / 2.0
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    best_h = h.copy()
    step = step0
    rng = np.random.RandomState(123)
    while time.time() < deadline:
        order = np.arange(N)
        rng.shuffle(order)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            hi = h[i]
            candidates = [hi + step, hi - step, hi + step * 0.5, hi - step * 0.5,
                          np.clip(hi + step * 0.25, 0, 1), np.clip(hi - step * 0.25, 0, 1)]
            for dv in candidates:
                if dv < 0 or dv > 1:
                    continue
                if abs(dv - hi) < 1e-10:
                    continue
                h2 = h.copy()
                h2[i] = dv
                h2 = project_sum(h2, target)
                c = float(np.max(np.correlate(h2, 1.0 - h2, mode='full'))) * dx
                if c < best_c5 - 1e-14:
                    h = h2
                    best_c5 = c
                    best_h = h2.copy()
                    improved = True
                    break
        if not improved:
            step *= 0.75
            if step < 1e-5:
                break
    return best_h, best_c5

def perturb_and_optimize(h, N, target, dx, deadline, rng, K=10, step_size=0.05):
    h2 = h.copy()
    mask = rng.rand(N) < 0.1
    h2[mask] += rng.randn(mask.sum()) * step_size
    h2 = project_sum(h2, target)
    h2, c2 = momentum_desc(h2, N, target, dx, deadline, K_init=20, K_final=5,
                           tau=30.0, lr0=0.03, decay=0.995, momentum=0.5)
    return h2, c2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    sizes = [192, 256, 320, 380, 448, 512, 640]
    sizes = [s for s in sizes if s % 2 == 0]

    size_budget = deadline - 120
    # Weight: more time for sizes near 380-512
    weights = np.array([1.0, 1.5, 2.0, 3.0, 2.5, 2.0, 1.2])
    weights /= weights.sum()
    durations = weights * size_budget

    for si, N in enumerate(sizes):
        dur = durations[si]
        if dur < 20:
            dur = 20
        phase_deadline = t0 + 30 + sum(durations[:si+1])
        if phase_deadline > deadline - 60:
            phase_deadline = deadline - 60
        if time.time() > phase_deadline - 10:
            continue

        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)

        # Phase A: coarse momentum descent (60% of phase time)
        pa_dead = time.time() + dur * 0.55
        h_a, c_a = momentum_desc(h_init.copy(), N, target, dx, pa_dead,
                                 K_init=30, K_final=8, tau=20.0,
                                 lr0=0.04, decay=0.994, momentum=0.6)
        if c_a * dx < best_c5:
            best_c5 = c_a * dx
            best_h = h_a.copy()

        # Phase B: fine momentum descent (25% of phase time)
        if time.time() < phase_deadline - 20:
            pb_dead = time.time() + dur * 0.3
            h_b, c_b = momentum_desc(h_a.copy(), N, target, dx, pb_dead,
                                     K_init=8, K_final=4, tau=80.0,
                                     lr0=0.015, decay=0.997, momentum=0.4)
            if c_b * dx < best_c5:
                best_c5 = c_b * dx
                best_h = h_b.copy()

        # Phase C: coordinate refinement (15% of phase time)
        if time.time() < phase_deadline - 5:
            pc_dead = min(time.time() + dur * 0.15, phase_deadline - 2)
            h_c, c_c = coord_refine(h_b, N, dx, pc_dead, step0=0.08)
            if c_c < best_c5:
                best_c5 = c_c
                best_h = h_c.copy()

        # Phase D: perturbation restart if time allows
        if time.time() < phase_deadline - 15:
            pd_dead = time.time() + min(30, phase_deadline - time.time() - 5)
            h_d, c_d = perturb_and_optimize(best_h if len(best_h) == N else h_b,
                                            N, target, dx, pd_dead, rng)
            if c_d * dx < best_c5:
                best_c5 = c_d * dx
                best_h = h_d.copy()

    # Final: interpolate best to its native size and do final coordinate refinement
    Nb = len(best_h)
    if time.time() < deadline - 5:
        dx_b = 2.0 / Nb
        h_f, c_f = coord_refine(best_h.copy(), Nb, dx_b, deadline - 2, step0=0.05)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    # Very final: try a quick momentum pass at the best size
    if time.time() < deadline - 3:
        Nb2 = len(best_h)
        dx2 = 2.0 / Nb2
        h_g, c_g = momentum_desc(best_h.copy(), Nb2, Nb2 // 2, dx2, deadline - 1,
                                 K_init=10, K_final=5, tau=100.0,
                                 lr0=0.008, decay=0.998, momentum=0.3)
        if c_g * dx2 < best_c5:
            best_c5 = c_g * dx2
            best_h = h_g

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
