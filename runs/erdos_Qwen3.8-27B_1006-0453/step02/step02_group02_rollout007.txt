import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = int(np.sum(m))
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, t, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    if topk and topk < m:
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        idx = np.arange(m)
        c = corr
    mx = float(np.max(c))
    s = t * (c - mx)
    s -= np.max(s)
    e = np.exp(s)
    se = float(np.sum(e))
    w = e / se
    obj = (mx + np.log(se) / t) * dx
    w_full = np.zeros(m)
    w_full[idx] = w
    c1 = np.convolve(h, w_full, mode='full')
    c2 = np.correlate(w_full, h, mode='full')
    grad = dx * (1.0 - c1[(n-1):(2*n-1)] - c2[0:n])
    return obj, grad

def pgrad(h, n, t, topk, max_it, lr, elgap):
    h = proj(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = lr
    for _ in range(max_it):
        _, grad = obj_grad(h, t, topk)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        s = step
        accepted = False
        for _bs in range(25):
            h_new = proj(h - s * grad, n)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                accepted = True
                break
            s *= 0.5
        if not accepted:
            h_new = proj(h - 1e-4 * lr * grad / gnorm, n)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                best_h = h.copy()
            else:
                break
        h = h_new
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        step = min(s * 1.05, lr * 2.0)
    return best_h, best_c

def binary_push(h, n):
    return proj((h > 0.5).astype(float), n)

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    if n >= nd * 2 and n % nd == 0:
        return proj(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def slsqp_polish(h, n, t, topk, maxiter, ftol):
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def objf(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = len(corr)
        if topk and topk < m:
            idx = np.argpartition(corr, -topk)[-topk:]
            c = corr[idx]
        else:
            c = corr
        mx = np.max(c)
        s = t * (c - mx)
        s -= np.max(s)
        return (mx + np.log(np.sum(np.exp(s))) / t) * dx
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5,
            'jac': lambda x: np.ones((1, n))}
    r = minimize(objf, h, method='SLSQP', bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)
