```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Use bisection on threshold for exact projection
    # Project onto {0<=h<=1, sum(h)=target}
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s_proj = np.sum(h_proj)
        if s_proj > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine-tune
    residual = target - np.sum(h_proj)
    if abs(residual) > 1e-12:
        interior = (h_proj > 1e-10) & (h_proj < 1.0-1e-10)
        if np.sum(interior) > 0:
            h_proj[interior] += residual / np.sum(interior)
    return np.clip(h_proj, 0.0, 1.0)

def lnorm_obj(h, p):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Use p-norm as smooth approximation to max
    if np.max(corr) <= 0:
        return 0.0
    return float(np.mean(np.abs(corr)**p))**(1.0/p)
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += r / cnt
    return np.clip(h, 0.0, 1.0)

def lnorm_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.maximum(corr, 0.0)
    return float(np.mean(c ** p)) ** (1.0 / p)

def lse_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    sh = np.clip(corr - cm, -50.0 * T, 0.0)
    s = np.sum(np.exp(sh / T))
    return (cm + T * np.log(max(s, 1e-300))) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def do_lbfgsb(h, n, p, max_iter):
    try:
        res = opt.minimize(lnorm_obj, h.copy(), args=(p,),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-12})
        return fix_sum(res.x, n)
    except Exception:
        return h

def do_slsqp(h, n, T, max_iter):
    try:
        res = opt.minimize(lambda x: lse_obj(x, T, n), h.copy(),
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                           options={'maxiter': max_iter, 'ftol': 1e-16})
        return fix_sum(res.x, n)
    except Exception:
        return h

def subgrad_dir(h, n, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        m = k - (n - 1)
        s = abs(m)
        if s == 0:
            grad += 1.0 - 2.0 * h
        else:
            if m > 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:])
            else:
                grad[:n-s] += (1.0 - h[s:])
                grad[s:] += (-h[:n-s])
    grad /= top_k
    return grad * dx, corr * dx

def do_subgrad(h, n, n_iter, step0, top_k=10):
    h_c = h.copy()
    step = step0
    for it in range(n_iter):
        g, _ = subgrad_dir(h_c, n, top_k)
        h_c = fix_sum(h_c - step * g, n)
        step *= 0.998
        if it % 200 == 199:
            step = max(0.0005, step * 2.0)
    return h_c

def interp_h(h, n_old, n_new):
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    return fix_sum(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)
    
    def elapsed():
        return time.time() - t0
    
    def try_update(hc, nc):
        nonlocal h_best, n_best, c5_best
        hc = fix_sum(hc, nc)
        c = compute_c5(hc)
        if c < c5_best - 1e-15:
            h_best, n_best, c5_best = hc, nc, c
            return True
        return False
    
    # Phase 1: L-BFGS-B p-norm annealing
    for p in [8, 16, 32, 64, 128, 256, 512, 1024]:
        if budget_s - elapsed() < 50: break
        h_new = do_lbfgsb(h_best, n_best, p, 800)
        try_update(h_new, n_best)
    
    # Phase 2: SLSQP log-sum-exp
    for T in [0.04, 0.015, 0.006, 0.002, 0.0008, 0.0003]:
        if budget_s - elapsed() < 50: break
        h_new = do_slsqp(h_best, n_best, T, 2000)
        try_update(h_new, n_best)
    
    # Phase 3: Subgradient descent
    if budget_s - elapsed() > 150:
        h_sg = do_subgrad(h_best, n_best, 3000, 0.02, top_k=12)
        try_update(h_sg, n_best)
        if budget_s - elapsed() > 100:
            h_sg2 = do_subgrad(h_best, n_best, 2000, 0.01, top_k=6)
            try_update(h_sg2, n_best)
    
    # Phase 4: Multi-resolution
    for n_try in [240, 300, 420, 480, 560, 640]:
        if budget_s - elapsed() < 180: break
        if n_try == n_best: continue
        h_init = interp_h(h_best, n_best, n_try)
        h_init = do_lbfgsb(h_init, n_try, 64, 500)
        h_init = do_slsqp(h_init, n_try, 0.003, 1500)
        h_init = do_slsqp(h_init, n_try, 0.0005, 1500)
        if try_update(h_init, n_try):
            pass
    
    # Phase 5: Polish with very low T SLSQP
    for T in [0.00015, 0.00005, 0.00002]:
        if budget_s - elapsed() < 35: break
        h_new = do_slsqp(h_best, n_best, T, 4000)
        try_update(h_new, n_best)
    
    # Phase 6: Multiple small perturbation restarts
    for scale in [0.03, 0.015, 0.008, 0.02, 0.01]:
        if budget_s - elapsed() < 70: break
        h_p = h_best + np.random.randn(n_best) * scale
        h_p = fix_sum(np.clip(h_p, 0, 1), n_best)
        h_p = do_slsqp(h_p, n_best, 0.01, 1500)
        h_p = do_slsqp(h_p, n_best, 0.0005, 2000)
        try_update(h_p, n_best)
    
    # Phase 7: Final subgradient + polish cycle
    if budget_s - elapsed() > 120:
        h_sg = do_subgrad(h_best, n_best, 1500, 0.008, top_k=8)
        try_update(h_sg, n_best)
        if budget_s - elapsed() > 50:
            h_new = do_slsqp(h_best, n_best, 0.0001, 3000)
            try_update(h_new, n_best)
    
    h_best = fix_sum(h_best, len(h_best))
    return (h_best, compute_c5(h_best), len(h_best))
```

```python
def active_set_opt(h, n, active_indices, max_iter):
    """Minimize max corr subject to corr[k] = t for k in active_indices"""
    # We want: minimize corr[k_active] (make them equal and as small as possible)
    # Use SLSQP with equality constraints: corr[k_i] - corr[k_0] = 0 for i>0
    # And minimize corr[k_0]
    
    def objective(x):
        corr = np.correlate(x, 1-x, mode='full') * (2.0/n)
        return corr[active_indices[0]]
    
    constraints = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    for i in range(1, len(active_indices)):
        k0, ki = active_indices[0], active_indices[i]
        # corr[k0] - corr[ki] = 0
        def con(x, _k0=k0, _ki=ki):
            c = np.correlate(x, 1-x, mode='full')
            return c[_k0] - c[_ki]
        constraints.append({'type': 'eq', 'fun': con})

                       bounds=[(0.0, 1.0)]*n,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
    return fix_sum(res.x, n)
```

```python
def symproj(h, n):
    """Enforce h[j] + h[n-1-j] = 1"""
    h = h.copy()
    nh = n // 2
    # h[0:nh] are free, h[nh:n] determined (if n even)
    # For n even: h[j] + h[n-1-j] = 1 for j = 0..n/2-1
    # h[n/2 : n] = 1 - h[n-1 : n/2-1 : -1] = 1 - h[n-1], 1-h[n-2], ...
    for j in range(nh):
        partner = n - 1 - j
        h[partner] = 1.0 - h[j]
    # Clip the free half
    h[:nh] = np.clip(h[:nh], 0.0, 1.0)
    h[nh:] = 1.0 - h[nh::-1]  # reverse the free half and subtract from 1
    return h
```

So h[n/2:] = 1 - h[n/2-1::-1] = 1 - reversed(h[:n/2]).

Let me verify sum: sum(h[:n/2]) + sum(h[n/2:]) = sum(h[:n/2]) + sum(1 - h[:n/2][::-1]) = sum(h[:n/2]) + n/2 - sum(h[:n/2]) = n/2. ✓

```python
def obj_sym(free, n):
    """Objective in reduced space. free has length n//2.
    h = [free, 1 - free[::-1]]"""
    h = np.empty(n)
    nh = n // 2
    h[:nh] = free
    h[nh:] = 1.0 - free[::-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
```

One more thought: the subgradient in the symmetric space. If h[j] + h[n-1-j] = 1, then moving h[j] by δ means moving h[n-1-j] by -δ. The effective gradient in the reduced space is:
g_red[j] = g_full[j] - g_full[n-1-j] for j = 0..n/2-1

This is because δh[j] = δ, δh[n-1-j] = -δ, so the directional derivative in the reduced space is g_full[j]*δ + g_full[n-1-j]*(-δ) = (g_full[j] - g_full[n-1-j])*δ.

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += r / cnt
    return np.clip(h, 0.0, 1.0)

def sym_from_free(free):
    n = 2 * len(free)
    h = np.empty(n)
    h[:len(free)] = free
    h[len(free):] = 1.0 - free[::-1]
    return h

def compute_c5_sym(free):
    h = sym_from_free(free)
    return compute_c5(h)

def lnorm_sym(free, p):
    h = sym_from_free(free)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.maximum(corr, 0.0)
    return float(np.mean(c ** p)) ** (1.0 / p)

def lse_sym(free, T):
    h = sym_from_free(free)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    sh = np.clip(corr - cm, -50.0 * T, 0.0)
    s = np.sum(np.exp(sh / T))
    return (cm + T * np.log(max(s, 1e-300))) * dx

def full_from_sym_h(h):
    n = len(h)
    nh = n // 2
    free = h[:nh].copy()
    return sym_from_free(np.clip(free, 0, 1))

def do_lbfgsb_sym(free, p, max_iter):
    nh = len(free)
    try:
        res = opt.minimize(lnorm_sym, free.copy(), args=(p,),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * nh,
                           options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-12})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return free

def do_slsqp_sym(free, T, max_iter):
    nh = len(free)
    try:
        res = opt.minimize(lambda x: lse_sym(x, T), free.copy(),
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * nh,
                           options={'maxiter': max_iter, 'ftol': 1e-16})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return free

def subgrad_dir_sym(free, n, top_k):
    h = sym_from_free(free)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        m = k - (n - 1)
        s = abs(m)
        if s == 0:
            grad += 1.0 - 2.0 * h
        else:
            if m > 0:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] += (-h[s:])
            else:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
    grad /= top_k
    return grad * dx

def do_subgrad_sym(free, n, n_iter, step0, top_k):
    fc = free.copy()
    step = step0
    for it in range(n_iter):
        g = subgrad_dir_sym(fc, n, top_k)
        nh = n // 2
        g_red = g[:nh] - g[:nh][::-1]
        fc -= step * g_red
        fc = np.clip(fc, 0.0, 1.0)
        step *= 0.998
        if it % 200 == 199:
            step = max(0.0005, step * 2.0)
    return fc

def active_set_sym(free, top_k, max_iter):
    h = sym_from_free(free)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    nh = n // 2
    def obj(x):
        h2 = sym_from_free(x)
        c = np.correlate(h2, 1.0 - h2, mode='full')
        return c[idx[0]] * dx
    res = opt.minimize(obj, free.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * nh,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
    return np.clip(res.x, 0.0, 1.0)

def interp_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return fix_sum(np.interp(x_new, x_old, h), n_new)

def get_free(h, n):
    nh = n // 2
    return h[:nh].copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    def try_update(hc, nc):
        nonlocal h_best, n_best, c5_best
        hc = fix_sum(hc, nc)
        c = compute_c5(hc)
        if c < c5_best - 1e-15:
            h_best, n_best, c5_best = hc, nc, c
            return True
        return False

    # Work in symmetric reduced space
    nh = n_best // 2
    free_best = get_free(h_best, n_best)
    free_best = np.clip(free_best, 0.0, 1.0)

    def try_update_sym(free_c):
        nc = 2 * len(free_c)
        hc = sym_from_free(np.clip(free_c, 0, 1))
        return try_update(hc, nc)

    # Phase 1: L-BFGS-B p-norm annealing in symmetric space
    for p in [8, 16, 32, 64, 128, 256, 512]:
        if budget_s - elapsed() < 50:
            break
        free_new = do_lbfgsb_sym(free_best, p, 600)
        if try_update_sym(free_new):
            free_best = free_new

    # Phase 2: SLSQP log-sum-exp in symmetric space
    for T in [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]:
        if budget_s - elapsed() < 50:
            break
        free_new = do_slsqp_sym(free_best, T, 1500)
        if try_update_sym(free_new):
            free_best = free_new

    # Phase 3: Active-set optimization in symmetric space
    for tk in [6, 8, 10, 12]:
        if budget_s - elapsed() < 60:
            break
        free_new = active_set_sym(free_best, tk, 2000)
        if try_update_sym(free_new):
            free_best = free_new

    # Phase 4: Subgradient descent in symmetric space
    if budget_s - elapsed() > 150:
        free_sg = do_subgrad_sym(free_best, n_best, 2000, 0.015, top_k=10)
        if try_update_sym(free_sg):
            free_best = free_sg
        if budget_s - elapsed() > 100:
            free_sg2 = do_subgrad_sym(free_best, n_best, 1500, 0.008, top_k=6)
            if try_update_sym(free_sg2):
                free_best = free_sg2

    # Phase 5: Multi-resolution (symmetric)
    for n_try in [240, 300, 420, 480, 560]:
        if budget_s - elapsed() < 160:
            break
        if n_try == n_best or n_try < 100:
            continue
        if n_try % 2 != 0:
            continue
        h_try = sym_from_free(free_best)
        h_interp = interp_h(h_try, n_best, n_try)
        # Make symmetric
        nh_try = n_try // 2
        h_sym = sym_from_free(h_interp[:nh_try])
        free_try = h_sym[:nh_try].copy()
        free_try = do_lbfgsb_sym(free_try, 64, 400)
        free_try = do_slsqp_sym(free_try, 0.003, 1000)
        free_try = do_slsqp_sym(free_try, 0.0005, 1000)
        if try_update_sym(free_try):
            if n_try == 2 * len(free_try):
                free_best = free_try
                n_best = n_try

    # Phase 6: Very fine polish
    for T in [0.0002, 0.00008, 0.00003]:
        if budget_s - elapsed() < 35:
            break
        free_new = do_slsqp_sym(free_best, T, 2500)
        if try_update_sym(free_new):
            free_best = free_new

    # Phase 7: Small perturbation restarts in symmetric space
    for scale in [0.025, 0.012, 0.006, 0.018, 0.01]:
        if budget_s - elapsed() < 70:
            break
        free_p = free_best + np.random.randn(len(free_best)) * scale
        free_p = np.clip(free_p, 0.0, 1.0)
        free_p = do_slsqp_sym(free_p, 0.01, 1200)
        free_p = do_slsqp_sym(free_p, 0.0005, 1500)
        if try_update_sym(free_p):
            free_best = free_p

    # Phase 8: Final subgradient + polish
    if budget_s - elapsed() > 100:
        free_sg = do_subgrad_sym(free_best, n_best, 1000, 0.006, top_k=8)
        if try_update_sym(free_sg):
            free_best = free_sg
        if budget_s - elapsed() > 40:
            free_new = do_slsqp_sym(free_best, 0.0001, 3000)
            if try_update_sym(free_new):
                free_best = free_new

    # Final: use best (possibly non-symmetric) solution
    if budget_s - elapsed() > 30:
        # Try one more round of non-symmetric refinement on the best
        h_ns = h_best.copy()
        res = opt.minimize(
            lambda x: lse_sym(get_free(sym_from_free(get_free(x.copy())), 2*len(get_free(x))), 0.0001)
            if False else compute_c5(fix_sum(x, n_best)),
            h_ns, method='Nelder-Mead',
            options={'maxiter': 500, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True}
        )
        try_update(res.x, n_best)

    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
```
