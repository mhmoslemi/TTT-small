import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, t, topk=None):
    """Compute soft-max objective and gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c_sub = corr[idx]
    else:
        idx = np.arange(len(corr))
        c_sub = corr
    
    mx = c_sub.max()
    exps = np.exp(t * (c_sub - mx))
    w = exps / exps.sum()
    obj = (mx + np.log(exps.sum()) / t) * dx  # soft max value
    
    # Gradient
    g = np.zeros(n)
    for j, k in enumerate(idx):
        if w[j] < 1e-15:
            continue
        s = k - (n - 1)
        # term1: w_j * (1-h)[i+s]
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        g[lo1:hi1] += w[j] * (1.0 - h[lo1+s:hi1+s])
        # term2: -w_j * h[i-s]
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        g[lo2:hi2] -= w[j] * h[lo2-s:hi2-s]
    
    return obj, g * dx  # gradient of the dx-scaled objective

def pgd_optimize(h, n, budget, t_values, topk_schedule, use_momentum=True):
    """Projected gradient descent with momentum"""
    t0 = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    # Nesterov momentum
    v = np.zeros(n)
    
    for t in t_values:
        if time.time() - t0 > budget:
            break
        for topk in topk_schedule:
            if time.time() - t0 > budget:
                break
            # Run PGD for this (t, topk) pair
            lr = 0.5 / n  # initial learning rate
            mu = 0.9  # momentum
            
            for it in range(5000):
                if time.time() - t0 > budget:
                    break
                # Nesterov step
                y = h + mu * v
                obj, grad = soft_max_obj(y, t, topk)
                
                # Backtracking line search
                step = lr
                f_y = obj
                for _ in range(20):
                    h_new = project(y - step * grad, n)
                    f_new, _ = soft_max_obj(h_new, t, topk)
                    if f_new < f_y - 0.5 * step * np.sum(grad * (h_new - y)):

                    step *= 0.5
                
                v = mu * v + (h - h_new)
                h = h_new
                
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    
    return best_h, best_c

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        # Water filling: if step would push some past bounds, recurse
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, t, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c_sub = corr[idx]
    else:
        idx = np.arange(len(corr))
        c_sub = corr
    
    mx = c_sub.max()
    exps = np.exp(t * (c_sub - mx))
    w = exps / exps.sum()
    obj = (mx + np.log(np.sum(exps)) / t) * dx
    
    # Gradient of soft-max objective
    g = np.zeros(n)
    for j in range(len(idx)):
        if w[j] < 1e-15:
            continue
        k = idx[j]
        s = k - (n - 1)
        # dc_k/dh[i] = (1-h)[i+s] - h[i-s] (with boundary handling)
        lo1, hi1 = max(0, -s), min(n, n - s)
        g[lo1:hi1] += w[j] * (1.0 - h[lo1+s:hi1+s])
        lo2, hi2 = max(0, s), min(n, n + s)
        g[lo2:hi2] -= w[j] * h[lo2-s:hi2-s]
    
    return obj, g * dx

def pgd_nesterov(h0, n, t, topk, max_iter, budget, lr_init=0.01, momentum=0.9):
    t0 = time.time()
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - t0 > budget:
            break
        y = h + momentum * v
        obj, grad = obj_and_grad(y, t, topk)
        
        # Backtracking line search
        step = lr
        obj_new = obj
        for _ in range(30):
            h_new = project(y - step * grad, n)
            obj_try, _ = obj_and_grad(h_new, t, topk)
            if obj_try <= obj - 1e-4 * step * np.dot(grad, h_new - y):
                break
            step *= 0.5
            obj_new = obj_try
            h_new = project(y - step * grad, n)
        
        v = momentum * v + (h_new - h)
        h = h_new
        
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            lr = lr * 1.05  # slight increase
        else:
            lr = lr * 0.95  # slight decrease
    
    return best_h, best_c

def slsqp_step(h, n, t, topk, maxiter=500, ftol=1e-14):
    """One SLSQP optimization step"""
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        c_corr = np.correlate(x, 1.0 - x, mode='full')
        if topk is not None and topk < len(c_corr):
            idx = np.argpartition(c_corr, -topk)[-topk:]
            c_s = c_corr[idx]
        else:
            c_s = c_corr
        mx = c_s.max()
        return (mx + np.log(np.sum(np.exp(t * (c_s - mx)))) / t) * (2.0 / n)
    
    def constr(x):
        return x.sum() - n / 2.0
    
    r = minimize(obj, h, method='SLSQP', bounds=bnds,
                 constraints=[{'type': 'eq', 'fun': constr, 'jac': lambda x: np.ones(n)}],
                 options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(r.x, 0, 1), n)

def binary_push(h, n, rng):
    """Create a binary version with sum constraint"""
    h_bin = (h > 0.5).astype(float)
    # Adjust to meet sum constraint
    target = n // 2
    current = int(h_bin.sum())
    if current > target:
        # Need to flip (current - target) ones to zeros
        ones = np.where(h_bin == 1)[0]
        rng.shuffle(ones)
        for i in range(current - target):
            h_bin[ones[i]] = 0
    elif current < target:
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(zeros)
        for i in range(target - current):
            h_bin[zeros[i]] = 1
    return h_bin

def block_perturb(h, n, rng, n_blocks=3, block_size=10):

        # Flip: h → 1-h in the block
        h_new[start:start+block_size] = 1.0 - h[start:start+block_size]
    return project(h_new, n)

                hk = slsqp(h_blk, n, 200, 30, 700, 1e-13)
                if compute_c5(hk) < bc:
                    bc = compute_c5(hk)
                    bh = hk
            except Exception:
                pass
            # Small random perturbation
            h_pr = project(bh + rng.randn(n) * 0.01, n)
            try:
                hr = slsqp(h_pr, n, 300, 25, 600, 1e-13)
                if compute_c5(hr) < bc:
                    bc = compute_c5(hr)
                    bh = hr
            except Exception:
                pass

    # Phase 6: Ultra-fine top-5 polish
    if tleft() > 30:
        for t in [30000, 150000]:
            if tleft() < 20:
                break
            try:
                hpol = slsqp(bh, n, t, 5, 3000, 1e-16)
                c = compute_c5(hpol)
                if c < bc:
                    bc = c
                    bh = hpol
            except Exception:
                pass

    # Phase 7: Final PGD squeeze
    if tleft() > 15:
        try:
            hp, cp = pgd_momentum(bh, n, 50000, 4, 20000, max(3, tleft() - 5))
            if cp < bc:
                bc = cp
                bh = hp
        except Exception:
            pass

    bh = project(bh, n)
    final_c = compute_c5(bh)
    return bh, final_c, n
