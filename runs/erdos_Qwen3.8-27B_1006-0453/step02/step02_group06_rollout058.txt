import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n, dtype=float) / old_n
    x_new = np.arange(new_n, dtype=float) / new_n
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:

    return np.sum(x) - n * 0.5

    return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = opt.minimize(
            obj_fn, h0, args=obj_args,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
        )
        return proj_constraint(res.x, n)
    except Exception:
        return h0

            if budget_s - (time.time() - t_start) < 25:
                break
            h_opt = do_tksqp(h_best.copy(), n_best, T, 2500)
            c = compute_c5(h_opt)
            if c < c5_best:
                c5_best = c
                h_best = h_opt

    # Phase 6: Try additional resolutions
    if budget_s - (time.time() - t_start) > 90:
        for n_try in [420, 504, 672]:
            if n_try == n_best:
                continue
            if budget_s - (time.time() - t_start) < 45:
                break
            h_curr = upsample(h_best, n_try)
            h_opt = do_tksqp(h_curr, n_try, 0.005, 400)
            h_opt = do_tksqp(h_opt, n_try, 0.001, 800)
            c = compute_c5(h_opt)
            if c < c5_best:
                c5_best = c
                h_best = h_opt
                n_best = n_try

    # Phase 7: Final subgradient + SLSQP squeeze
    if budget_s - (time.time() - t_start) > 50:
        h_sg, c_sg = subgradient_refine(h_best, n_best, 150, 0.008)
        if c_sg < c5_best:
            c5_best = c_sg
            h_best = h_sg
        if budget_s - (time.time() - t_start) > 20:
            h_opt = do_tksqp(h_best.copy(), n_best, 0.0001, 2000)
            c = compute_c5(h_opt)
            if c < c5_best:
                c5_best = c
                h_best = h_opt

    h_best = proj_constraint(h_best, n_best)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n_best)
