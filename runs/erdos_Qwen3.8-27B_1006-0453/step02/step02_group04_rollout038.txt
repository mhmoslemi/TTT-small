    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            m = h < 1.0 - 1e-12
            cnt = np.sum(m)
            if cnt == 0:
                break
            h[m] = np.minimum(h[m] + diff / cnt, 1.0)
        else:
            m = h > 1e-12
            cnt = np.sum(m)
            if cnt == 0:
                break
            h[m] = np.maximum(h[m] - (-diff) / cnt, 0.0)
        diff = target - np.sum(h)
    return h

def solve_1d(C, A, n):
    # Golden section: min_{t in [0,1]} max_k(C[k]+A[k]*t) with quad at k=n-1
    def f(t):
        v = C + A * t
        v[n - 1] = C[n - 1] + t - t * t
        return float(np.max(v))
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(30):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - phi * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + phi * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def cds_sweep(h, n, fwd):
    corr = np.correlate(h, 1.0 - h, mode='full')
    Larr = np.arange(-(n - 1), n)
    A = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    order = range(n) if fwd else range(n - 1, -1, -1)
    for i in order:
        old = h[i]
        idx1 = i + Larr
        idx2 = i - Larr
        A[:] = 0.0
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        A[m1] = 1.0 - h[idx1[m1]]
        A[m2] -= h[idx2[m2]]
        C[:] = corr - old * A
        C[n - 1] = corr[n - 1] - old + old * old
        t_new = solve_1d(C, A, n)
        delta = t_new - old
        if abs(delta) > 1e-15:
            h[i] = t_new
            corr += delta * A
            corr[n - 1] -= delta * delta
    return h, corr

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(300):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0 - 1e-12
            cnt = np.sum(m)
            if cnt == 0:
                break
            h[m] = np.minimum(h[m] + diff / cnt, 1.0)
        else:
            m = h > 1e-12
            cnt = np.sum(m)
            if cnt == 0:
                break
            h[m] = np.maximum(h[m] + diff / cnt, 0.0)
        diff = target - np.sum(h)
    return h

def f_1d(C, A, n, t):
    v = C + A * t
    v[n - 1] = C[n - 1] + t - t * t
    return float(np.max(v))

def solve_1d(C, A, n):
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = f_1d(C, A, n, c)
    fd = f_1d(C, A, n, d)
    for _ in range(30):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = f_1d(C, A, n, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = f_1d(C, A, n, d)
    return (lo + hi) * 0.5

def cds_sweep(h, n, fwd):
    corr = np.correlate(h, 1.0 - h, mode='full')
    Larr = np.arange(-(n - 1), n)
    A = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    order = range(n) if fwd else range(n - 1, -1, -1)
    for i in order:
        old = h[i]
        A[:] = 0.0
        idx1 = i + Larr
        idx2 = i - Larr
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        A[m1] = 1.0 - h[idx1[m1]]
        A[m2] -= h[idx2[m2]]
        C[:] = corr - old * A
        C[n - 1] = corr[n - 1] - old + old * old
        t_new = solve_1d(C, A, n)
        delta = t_new - old
        if abs(delta) > 1e-15:
            h[i] = t_new
            corr += delta * A
            corr[n - 1] -= delta * delta
    return h, corr

def slsqp_obj(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(corr)
    sh = np.clip(corr - cm, -80.0 * T, 0.0)
    s = np.sum(np.exp(sh / T))
    return (cm + T * np.log(max(s, 1e-300))) * (2.0 / n)

def slsqp_con(x, n):
    return np.sum(x) - n * 0.5

def slsqp_polish(h, n, T, max_iter):
    try:
        res = opt.minimize(slsqp_obj, h.copy(), args=(n, T), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}],
                           options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def interp_res(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    hn = np.interp(xn, xo, h, period=2.0)
    return fix_sum(hn, new_n)
