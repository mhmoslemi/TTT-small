    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint_x(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint_x}]
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def fix_binary_count(h, n):
    """Ensure exactly n//2 ones in binary h"""
    target = n // 2
    cnt = int(h.sum())
    if cnt == target:
        return h
    h = h.copy()
    if cnt > target:
        idx = np.where(h == 1)[0]
        h[idx[(cnt - target):]] = 0
    else:
        idx = np.where(h == 0)[0]
        h[idx[:target - cnt]] = 1
    return h

def bin_swap_eval(h, i, j, n):
    """Quick C5 eval after swapping positions i,j in binary h"""
    h[i], h[j] = h[j], h[i]
    c5 = compute_c5(h)
    return c5

def binary_local_search(h_bin, n, max_iter=5000, seed_offset=0):
    rng = np.random.RandomState(seed_offset + 42)
    current = h_bin.copy()
    c5_cur = compute_c5(current)
    ones = np.where(current == 1)[0]
    zeros = np.where(current == 0)[0]
    stall = 0
    for it in range(max_iter):
        improved = False
        for _ in range(80):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            current[i], current[j] = current[j], current[i]
            c5_new = compute_c5(current)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                ones = np.where(current == 1)[0]
                zeros = np.where(current == 0)[0]
                stall = 0
                improved = True
            else:
                current[i], current[j] = current[j], current[i]
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 300:
                break
    return current, c5_cur

def sa_binary(n, n_iter=20000, T_start=0.008, T_end=0.00005, seed_offset=0):
    rng = np.random.RandomState(seed_offset + 7)
    alpha = n // 2
    positions = rng.choice(n, alpha, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    ones_set = set(positions.tolist())
    zeros_list = [i for i in range(n) if i not in ones_set]

    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        T = T_start * (T_end / T_start) ** frac
        i = list(ones_set)[rng.randint(len(ones_set))]
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        ones_set.remove(i)
        ones_set.add(j)
        zi = zeros_list.index(i)
        zj = zeros_list.index(j)
        zeros_list[zi], zeros_list[zj] = j, i

        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            ones_set.remove(j)
            ones_set.add(i)
            zeros_list[zi], zeros_list[zj] = i, j
    return best_h, best_c5

def interp_to_n(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Continuous refinement of initial
    if elapsed() < budget_s * 0.06:
        h_try = slsqp_smooth(best_h, 400, maxiter=250)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
    if elapsed() < budget_s * 0.10:
        h_try = slsqp_erdos(best_h, maxiter=500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 2: Binary optimization at current and nearby resolutions
    res_list = [best_n]
    for dn in [-10, -5, 5, 10, 15, 20, -15]:
        nv = best_n + dn
        if 40 <= nv <= 250 and nv % 2 == 0 and nv not in res_list:
            res_list.append(nv)
    res_list.sort()

    for n_try in res_list:
        if elapsed() > budget_s * 0.55:
            break
        # Initialize binary from best_h
        if n_try == best_n:
            h_interp = best_h
        else:
            h_interp = interp_to_n(best_h, best_n, n_try)
        h_bin = (h_interp > 0.5).astype(float)
        h_bin = fix_binary_count(h_bin, n_try)

        # Multiple SA runs
        for sa_iter in range(3):
            if elapsed() > budget_s * 0.55:
                break
            h_sa, c5_sa = sa_binary(n_try, n_iter=12000, seed_offset=sa_iter * 100 + n_try)
            if c5_sa < best_c5:
                best_c5, best_h, best_n = c5_sa, h_sa.copy(), n_try

        # Local search from best binary found
        if elapsed() < budget_s * 0.55:
            h_ls, c5_ls = binary_local_search(h_sa if 'h_sa' in dir() else h_bin, n_try, max_iter=4000, seed_offset=n_try)
            if c5_ls < best_c5:
                best_c5, best_h, best_n = c5_ls, h_ls.copy(), n_try

        # Continuous refinement of best binary
        if elapsed() < budget_s * 0.55:
            h_ref = slsqp_erdos(best_h if len(best_h) == n_try else h_ls, maxiter=350)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5, best_h, best_n = c5_ref, h_ref.copy(), n_try

    # Phase 3: Basin hopping in continuous space
    if elapsed() < budget_s * 0.80:
        n_cur = len(best_h)
        n_trials = min(25, max(8, int((budget_s * 0.80 - elapsed()) / 20)))
        for i in range(n_trials):
            if elapsed() > budget_s * 0.80:
                break
            scale = 0.04 + 0.04 * np.random.rand()
            h_pert = best_h + np.random.randn(n_cur) * scale
            h_pert = project_feasible(h_pert, n_cur)
            h_opt = slsqp_smooth(h_pert, 600, maxiter=200)
            h_opt = slsqp_erdos(h_opt, maxiter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt

    # Phase 4: Cyclic rotation + refine
    if elapsed() < budget_s * 0.88:
        n_cur = len(best_h)
        for shift in [1, 3, 5, 7, 11, 17]:
            if elapsed() > budget_s * 0.88:
                break
            h_rot = np.roll(best_h, shift)
            h_opt = slsqp_erdos(h_rot, maxiter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt

    # Phase 5: Final intensive polish
    if elapsed() < budget_s * 0.96:
        h_try = slsqp_erdos(best_h, maxiter=1000)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        h_try = slsqp_smooth(best_h, 8000, maxiter=500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        h_try = slsqp_erdos(best_h, maxiter=500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
