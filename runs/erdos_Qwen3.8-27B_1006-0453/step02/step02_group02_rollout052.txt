```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_and_grad(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max
    mx = np.max(corr)
    s = t * (corr - mx)
    logsum = mx + np.log(np.sum(np.exp(s))) / t
    f = logsum * dx
    # Penalty for constraint
    s_h = np.sum(h) - n / 2.0
    f += lam * s_h * s_h
    
    # Gradient of smooth max
    w = np.exp(s) / np.sum(np.exp(s))  # weights for each lag
    # w is indexed by m=0..2n-2, lag k = m-(n-1)
    grad = np.zeros(n)
    for m in range(2*n-1):
        k = m - (n-1)
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        # (1-h[i+k]) for valid i+k
        lo1 = max(0, -k)
        hi1 = min(n, n-k)
        if lo1 < hi1:
            grad[lo1:hi1] += wm * (1.0 - h[lo1+k:hi1+k])
        # -h[i-k] for valid i-k
        lo2 = max(0, k)
        hi2 = min(n, n+k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2-k:hi2-k]
    # Gradient of penalty
    grad += 2.0 * lam * s_h
    return f, grad

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, n, dx, max_iter=50):
    """Greedy local search: try moving mass between pairs of coordinates."""
    for _ in range(max_iter):
        improved = False
        c5_before = compute_c5(h)
        corr_before = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            if abs(h[i] - 0.5) < 1e-6:
                continue
            # Try pushing h[i] toward 0 or 1
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-8:
                continue
            # Find j to compensate
            if delta > 0:
                # Need to reduce h[j] by delta; pick j with smallest h[j] > 0
                candidates = np.where(h > 1e-8)[0]
                if len(candidates) == 0:
                    continue
                j = candidates[np.argmin(h[candidates])]
                amt = min(delta, h[j])
            else:
                # Need to increase h[j] by -delta; pick j with largest h[j] < 1
                candidates = np.where(h < 1 - 1e-8)[0]
                if len(candidates) == 0:
                    continue
                j = candidates[np.argmax(h[candidates])]
                amt = min(-delta, 1.0 - h[j])
            if amt < 1e-8:
                continue
            h[i] += amt * np.sign(delta)
            h[j] -= amt * np.sign(delta)
            h = np.clip(h, 0, 1)
            c5_new = compute_c5(h)
            if c5_new < c5_before:
                c5_before = c5_new
                improved = True
                corr_before = np.correlate(h, 1.0 - h, mode='full')
            else:
                h[i] -= amt * np.sign(delta)
                h[j] += amt * np.sign(delta)
        if not improved:
            break
    return h, compute_c5(h)
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smoothmax_grad(corr, t, dx):
    """Compute smooth max value and gradient weights."""
    mx = np.max(corr)
    s = t * (corr - mx)
    max_s = np.max(s)
    exp_s = np.exp(s - max_s)
    w = exp_s / np.sum(exp_s)
    val = (mx + (max_s + np.log(np.sum(exp_s))) / t) * dx
    return val, w

def obj_grad(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, w = smoothmax_grad(corr, t, dx)
    s_h = np.sum(h) - n / 2.0
    val += lam * s_h * s_h
    
    # Gradient
    grad = np.zeros(n)
    # For each lag index m (0 to 2n-2), corresponding to lag k = m-(n-1)
    # ∂(corr[m])/∂h[i] = (1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]
    # Weighted sum: grad[i] += w[m] * dx * (above)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        k = m - (n - 1)

        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:

        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 - k:hi2 - k]
```

```python
        order = np.argsort(h)
        for idx in order:
            if n_ones <= target:
                break
            if h_bin[idx] == 1.0:
                h_bin[idx] = 0.0
                n_ones -= 1
    elif n_ones < target:
        order = np.argsort(h)[::-1]
        for idx in order:
            if n_ones >= target:
                break
            if h_bin[idx] == 0.0:
                h_bin[idx] = 1.0
                n_ones += 1
    return h_bin

def pair_search(h, n, n_tries, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(n_tries):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, best_c5

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    if n % nd == 0:
        return proj(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    N = 280
    bnds = [(0.0, 1.0)] * N

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = proj(np.array(initial_h_values, dtype=float), N)
    except Exception:
        bh = proj(np.full(N, 0.5) + rng.randn(N) * 0.08, N)
    bc = compute_c5(bh)

    # Phase 1: Coarse optimization at n=70 for global structure
    if tleft() > 300:
        h70 = interp_down(bh, 70)
        b70 = [(0.0, 1.0)] * 70
        for t_val in [30, 100, 400, 1500]:
            if tleft() < 20:
                break
            for rr in range(2):
                if tleft() < 12:
                    break
                hs = h70.copy() if rr == 0 else proj(h70 + rng.randn(70) * 0.04, 70)
                try:
                    r = minimize(obj_grad, hs, args=(t_val, 50.0), method='L-BFGS-B',
                                 jac=True, bounds=b70,
                                 options={'maxiter': 2000, 'ftol': 1e-15, 'maxfun': 30000})
                    h70 = proj(np.clip(r.x, 0, 1), 70)
                except Exception:
                    pass
        bh = interp_up(h70, N)
        bc = compute_c5(bh)

    # Phase 2: L-BFGS-B at n=280 with increasing temperature
    t_schedule = [40, 100, 300, 800, 2000, 6000, 15000, 40000]
    for t_val in t_schedule:
        if tleft() < 25:
            break
        for rr in range(3):
            if tleft() < 12:
                break
            if rr == 0:
                hs = bh.copy()
            elif rr == 1:
                hs = proj(bh + rng.randn(N) * 0.015, N)
            else:
                hs = proj(bh + rng.randn(N) * 0.04, N)
            try:
                r = minimize(obj_grad, hs, args=(t_val, 80.0), method='L-BFGS-B',
                             jac=True, bounds=bnds,
                             options={'maxiter': 4000, 'ftol': 1e-17,
                                      'gtol': 1e-13, 'maxfun': 60000})
                hn = proj(np.clip(r.x, 0.0, 1.0), N)
                c5 = compute_c5(hn)
                if c5 < bc:
                    bc = c5
                    bh = hn
            except Exception:
                pass

    # Phase 3: Binary post-processing with pair search
    if tleft() > 60:
        best_bin = bh.copy()
        best_bin_c5 = bc
        for attempt in range(5):
            if tleft() < 15:
                break
            if attempt < 3:
                h_start = proj(bh + rng.randn(N) * 0.02, N)
            else:
                h_start = proj(bh + rng.randn(N) * 0.06, N)
            h_bin = binary_fix(h_start, N)
            c5b = compute_c5(h_bin)
            h_bin, c5b = pair_search(h_bin, N, 4000, rng)
            if c5b < best_bin_c5:
                best_bin_c5 = c5b
                best_bin = h_bin.copy()
        if best_bin_c5 < bc:
            bc = best_bin_c5
            bh = best_bin

    # Phase 4: Un-binary and re-optimize from best binary
    if tleft() > 40:
        h_start = np.where(bh > 0.5, 0.95, 0.05).astype(float)
        h_start = proj(h_start, N)
        for t_val in [10000, 50000, 200000]:
            if tleft() < 15:
                break
            try:
                r = minimize(obj_grad, h_start, args=(t_val, 200.0), method='L-BFGS-B',
                             jac=True, bounds=bnds,
                             options={'maxiter': 5000, 'ftol': 1e-18,
                                      'gtol': 1e-14, 'maxfun': 80000})
                hn = proj(np.clip(r.x, 0.0, 1.0), N)
                c5 = compute_c5(hn)
                if c5 < bc:
                    bc = c5
                    bh = hn
                    h_start = hn.copy()
            except Exception:
                pass

    # Phase 5: Final binary polish
    if tleft() > 20:
        h_bin = binary_fix(bh, N)
        h_bin, c5b = pair_search(h_bin, N, 5000, rng)
        if c5b < bc:
            bc = c5b
            bh = h_bin

    bh = proj(bh, N)
    final_c5 = compute_c5(bh)
    if final_c5 > bc:
        bh = proj(bh, N)
        final_c5 = compute_c5(bh)
    return bh, final_c5, N
```
