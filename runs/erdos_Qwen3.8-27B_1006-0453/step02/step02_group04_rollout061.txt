```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
        s2 = np.sum(h)
        if abs(s2 - target) > 1e-10:
            h += (target - s2) / n
            h = np.clip(h, 0.0, 1.0)
    else:

            h = np.clip(h, 0.0, 1.0)
            s = np.sum(h)
            if abs(s - target) < 1e-12:
                break

    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top_k active shifts

    # Compute subgradient: average of gradients of active corr components
    # corr[m] = sum_i h[i + (m - (n-1))] * (1 - h[i]) for valid i
    # d(corr[m])/d(h[j]) = (1-h[j-(m-(n-1))]) - h[j+(m-(n-1))] (with bounds)
```

```python
import scipy.optimize as opt
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            h[free] += (target - s) / nf
        else:
            h += (target - s) / n
    return np.clip(h, 0.0, 1.0)

def _grad_corr_at(h, corr_idx, n):
    lg = corr_idx - (n - 1)
    grad = np.zeros(n)
    if lg >= 0:
        if lg < n:
            grad[lg:] += (1.0 - h[:n - lg])
            grad[:n - lg] += (-h[lg:])
    else:
        lga = -lg
        if lga < n:
            grad[:n - lga] += (1.0 - h[lga:])
            grad[lga:] += (-h[:n - lga])
    return grad

def softmax_obj_and_grad(h, T, top_k=0):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k > 0 and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
    else:
        idx = np.arange(len(corr))
    c = corr[idx]
    c_max = np.max(c)
    if T < 1e-14:
        am = np.argmax(c)
        return c_max * dx, _grad_corr_at(h, idx[am], n) * dx
    shifted = np.clip(c - c_max, -80.0 * T, 0.0)
    exps = np.exp(shifted / T)
    s = np.sum(exps)
    val = (c_max + T * np.log(s)) * dx
    weights = exps / s
    grad = np.zeros(n)
    for w, ii in zip(weights, idx):
        if w > 1e-12:
            grad += w * _grad_corr_at(h, ii, n)
    return val, grad * dx

def do_slsqp(h0, n, T, top_k, max_iter):
    def obj(x):
        v, _ = softmax_obj_and_grad(x, T, top_k)
        return v
    def obj_and_grad(x):
        return softmax_obj_and_grad(x, T, top_k)
    try:
        res = opt.minimize(
            obj_and_grad, h0.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': max_iter, 'ftol': 1e-16, 'disp': False}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h0

def do_lbfgs(h0, n, T, top_k, max_iter, pen_weight):
    """L-BFGS-B with penalty for sum constraint"""
    def obj(x):
        v, g = softmax_obj_and_grad(x, T, top_k)
        pen = pen_weight * (np.sum(x) - n * 0.5) ** 2
        return v + pen
    def obj_grad(x):
        v, g = softmax_obj_and_grad(x, T, top_k)
        pen_val = pen_weight * (np.sum(x) - n * 0.5) ** 2
        pen_grad = 2 * pen_weight * (np.sum(x) - n * 0.5)
        return v + pen_val, g + pen_grad
    try:
        res = opt.minimize(obj_grad, h0.copy(), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h0

def subgradient_polish(h0, n, n_iter, step_init, top_k=5):
    h = h0.copy()
    step = step_init
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for ii in idx:
            grad += _grad_corr_at(h, ii, n)
        grad /= top_k
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
        else:
            step *= 0.7
    return h

    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)
```

For corr at lag k ≠ 0: the terms involving h[i] are:

- h[i+k] for k≠0: appears as h[m] where m = i, contributing -h[i+k] to the derivative

So varying h[i] affects multiple corr terms, each linearly in h[i]. The max of these linear functions is a convex piecewise-linear function of h[i] (for fixed other h[j]). So the 1D subproblem is: minimize max_k (a_k * h[i] + b_k) subject to 0 ≤ h[i] ≤ 1.

This is a 1D convex (piecewise linear) minimization that can be solved exactly! The optimum is where the top two active constraints cross, or at a boundary.

This coordinate descent would be a nice polishing step. Let me implement it.

For coord i, the corr values as functions of h[i] = t:
- For each lag lg, corr_lg(t) = A_lg * t + B_lg
  where A_lg and B_lg depend on the other h values.

From the gradient: d(corr_lg)/d(h[i]) = A_lg (constant, since corr_lg is linear in h[i] when others are fixed).

Wait, is corr_lg linear in h[i]? Let's check:
corr_lg = sum_{m} h[m+lg]*(1-h[m])

The terms involving h[i]:
1. When m+lg = i (m = i-lg): h[i]*(1-h[i-lg]) → linear in h[i], coefficient (1-h[i-lg])
2. When m = i: h[i+lg]*(1-h[i]) → linear in h[i], coefficient -h[i+lg]

So corr_lg = (1-h[i-lg])*h[i] - h[i+lg]*h[i] + (terms not involving h[i])
           = [(1-h[i-lg]) - h[i+lg]] * h[i] + (terms not involving h[i])

Yes! It's linear in h[i]. The coefficient is [(1-h[i-lg]) - h[i+lg]] and the constant is the rest.

So for coordinate descent on h[i]:
- Compute A_lg = (1-h[i-lg]) - h[i+lg] for each lag lg (with appropriate bounds)
- Compute B_lg = corr_lg (current value) - A_lg * h[i] (the constant part)
- Minimize max_lg (A_lg * t + B_lg) over t ∈ [0,1]

The 1D problem: minimize max_lg (A_lg * t + B_lg) over t ∈ [0,1].

This is a standard 1D convex piecewise-linear minimization. The optimum is at a breakpoint or boundary.

Breakpoints: where A_p * t + B_p = A_q * t + B_q, i.e., t = (B_q - B_p)/(A_p - A_q) for A_p ≠ A_q.

We can find the optimum by:
1. Evaluating at t=0 and t=1
2. For each pair (p,q) where A_p < A_q (one decreasing, one increasing), compute the crossing point and evaluate there.
3. The optimum is the minimum of all candidate values.

For simplicity, I'll just do: evaluate at t=0, t=1, and the midpoint of the "active" region. Or even simpler: do a ternary search on [0,1] since the function is convex.

Actually, for a convex piecewise-linear function in 1D, I can use a different approach: the subgradient is 0 at the optimum (or contains 0). The subgradient of max_lg (A_lg * t + B_lg) is the set {A_lg : lg is active (achieves the max)}. So the optimum is where 0 is in conv{A_lg for active lg}, or equivalently, where max of increasing A_lg's equals min of decreasing A_lg's (at the crossing).

Simpler approach: just do a golden section search or ternary search on [0,1] for the 1D subproblem. 30 iterations of ternary search gives accuracy of (2/3)^30 ≈ 5e-6, which is more than enough.

Let me implement coordinate descent with ternary search:

```python
def coord_descent_step(h, n, n_sweeps=3):

                m2 = b - (b-a)/3

                v2 = compute_1d_max(h, i, m2, n)
                if v1 < v2:

                else:
                    a = m1
            t_new = (a+b)/2
            v_new = compute_1d_max(h, i, t_new, n)
            h[i] = t_new
    return h

def compute_1d_max(h, i, t, n):
    """Compute max corr when h[i]=t (others fixed)"""
    # Quick computation: start from current corr and adjust
    # For each lag lg, corr_lg changes by A_lg * (t - h[i])
    # where A_lg = (1-h[i-lg]) - h[i+lg] (with bounds)
    # But we need the actual corr_lg values...
    # Simplest: just compute the full correlate with h[i] set to t
    h2 = h.copy()
    h2[i] = t
    corr = np.correlate(h2, 1.0 - h2, mode='full')
    return np.max(corr)
```

Wait, this is O(n^2) per evaluation (for the correlate), and we do 30 evaluations per coordinate, 360 coordinates, 3 sweeps = 30*360*3 = 32400 correlates of size 360. Each correlate is O(n^2) = 130K ops. Total: 4.2 billion ops. Way too slow.

I need a faster way to compute the 1D subproblem. Since corr_lg is linear in h[i], I can precompute the coefficients:

```python
def compute_1d_coeffs(h, i, n):
    """For coordinate i, compute A_lg and B_lg for all lags"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    A = np.zeros(len(corr))
    for corr_idx in range(len(corr)):
        lg = corr_idx - (n - 1)
        a_val = 0.0
        # Contribution from h[i] as h[m+lg]: m = i-lg, coefficient (1-h[m]) = (1-h[i-lg])
        if 0 <= i - lg < n:
            a_val += (1.0 - h[i - lg])
        # Contribution from h[i] as h[m]: m = i, coefficient -h[m+lg] = -h[i+lg]
        if 0 <= i + lg < n:
            a_val += (-h[i + lg])
        A[corr_idx] = a_val
    B = corr - A * h[i]
    return A, B
```

Then the 1D problem is: minimize max_l (A[l]*t + B[l]) for t in [0,1].

And to evaluate this at a given t: max_l(A[l]*t + B[l]) = max(A*t + B) which is O(K) where K=2n-1.

So each ternary search step is O(K) = O(n). Total: 30 * 360 * 3 * 720 ≈ 23M ops. Very fast!

But computing A and B is O(K * n) per coordinate (since for each of K lags, we check 2 conditions). Actually no, for each coordinate i, computing A is O(K) since for each lag we just do 2 lookups. So total: 360 * 720 = 260K for computing A, B. Then 30 * 3 * max(A*t+B) evaluations, each O(720): 30*3*720 = 65K. Very fast.

```python
def coord_descent(h, n, n_sweeps=3, n_ternary=40):
    h = h.copy()
    for sweep in range(n_sweeps):
        order = np.random.permutation(n)
        for i in order:
            A, B = _coord_coeffs(h, i, n)
            # Minimize max(A*t + B) over t in [0,1] via ternary search
            a, b = 0.0, 1.0
            for _ in range(n_ternary):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                v1 = np.max(A * m1 + B)
                v2 = np.max(A * m2 + B)
                if v1 < v2:
                    b = m2
                else:
                    a = m1
            t_new = (a + b) / 2.0
            h[i] = t_new
    return fix_sum(h, n)

def _coord_coeffs(h, i, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = len(corr)
    lags = np.arange(K) - (n - 1)
    A = np.zeros(K)
    for k in range(K):
        lg = lags[k]
        a = 0.0
        if 0 <= i - lg < n:
            a += (1.0 - h[i - lg])
        if 0 <= i + lg < n:
            a += (-h[i + lg])
        A[k] = a
    B = corr - A * h[i]
    return A, B
```

Wait, the loop over K=719 for each of n=360 coordinates is 719*360 = 259K iterations per sweep. In Python, this might be a bit slow (maybe 0.1-0.5 seconds per sweep). With 3 sweeps, that's 0.3-1.5 seconds. Acceptable.

Actually, I can vectorize the A computation:

For coordinate i:
- A[k] = (1-h[i-lags[k]]) * [0<=i-lags[k]<n] + (-h[i+lags[k]]) * [0<=i+lags[k]<n]

The lags are np.arange(K) - (n-1), which goes from -(n-1) to (n-1).

i - lags[k] = i - (k - (n-1)) = i + n - 1 - k
i + lags[k] = i + k - (n-1)

So:
- i - lags[k] is in [0, n-1] iff 0 <= i+n-1-k < n iff i <= k <= i+n-1
  Since k ranges from 0 to 2n-2, and i ranges from 0 to n-1:
  k from i to min(i+n-1, 2n-2) = i to i+n-1 (since i <= n-1, i+n-1 <= 2n-2)
  
- i + lags[k] is in [0, n-1] iff 0 <= i+k-(n-1) < n iff n-1-i <= k <= 2n-2-i
  Since k ranges from 0 to 2n-2:
  k from max(0, n-1-i) to min(2n-2, 2n-2-i) = n-1-i to 2n-2-i (since i >= 0, 2n-2-i <= 2n-2)

```python
def _coord_coeffs_vec(h, i, n):

    A[idx1] += (1.0 - h[n - 1 + i - idx1])  # h[i-lg] = h[i-(k-(n-1))] = h[i+n-1-k]

    # i+lg = i + k - (n-1), valid for k in [n-1-i, 2n-2-i]

    A[idx2] += (-h[idx2 - (n - 1 - i)])  # h[i+lg] = h[i+k-(n-1)]

    return A, B
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            h[free] += (target - s) / nf
        else:
            h += (target - s) / n
    return np.clip(h, 0.0, 1.0)

def _grad_corr_at(h, corr_idx, n):
    lg = corr_idx - (n - 1)
    grad = np.zeros(n)
    if lg >= 0:
        if lg < n:
            grad[lg:] += (1.0 - h[:n - lg])
            grad[:n - lg] += (-h[lg:])
    else:
        lga = -lg
        if lga < n:
            grad[:n - lga] += (1.0 - h[lga:])
            grad[lga:] += (-h[:n - lga])
    return grad

def softmax_obj(x, T, top_k):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_k > 0 and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
    else:
        c = corr
    c_max = np.max(c)
    if T < 1e-14:
        return c_max * dx
    shifted = np.clip(c - c_max, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def softmax_jac(x, T, top_k):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_k > 0 and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
    else:
        idx = np.arange(len(corr))
    c = corr[idx]
    c_max = np.max(c)
    if T < 1e-14:
        am = np.argmax(c)
        return _grad_corr_at(x, idx[am], n) * dx
    shifted = np.clip(c - c_max, -80.0 * T, 0.0)
    exps = np.exp(shifted / T)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for w, ii in zip(weights, idx):
        if w > 1e-12:
            grad += w * _grad_corr_at(x, ii, n)
    return grad * dx

def sum_con_fun(x, n):
    return np.sum(x) - n * 0.5

def sum_con_jac(x, n):
    return np.ones(len(x))

def do_slsqp(h0, n, T, top_k, max_iter):
    try:
        res = opt.minimize(
            softmax_obj, h0.copy(), args=(T, top_k),
            jac=lambda x, a, b: softmax_jac(x, a, b),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-16, 'disp': False}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h0

def lbfgs_pen_obj(x, T, top_k, pw, n):
    v = softmax_obj(x, T, top_k)
    g = softmax_jac(x, T, top_k)
    pen = pw * (np.sum(x) - n * 0.5) ** 2
    pen_g = 2.0 * pw * (np.sum(x) - n * 0.5)
    return v + pen, g + pen_g

def do_lbfgs(h0, n, T, top_k, max_iter, pw):
    try:
        res = opt.minimize(
            lbfgs_pen_obj, h0.copy(), args=(T, top_k, pw, n),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-10, 'maxfun': max_iter * n}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h0

def _coord_coeffs_vec(h, i, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = 2 * n - 1
    A = np.zeros(K)
    lo1 = i
    hi1 = min(i + n, K)
    idx1 = np.arange(lo1, hi1)
    A[idx1] += (1.0 - h[n - 1 + i - idx1])
    lo2 = max(0, n - 1 - i)
    hi2 = min(K, 2 * n - 1 - i)
    idx2 = np.arange(lo2, hi2)
    A[idx2] += (-h[idx2 - (n - 1 - i)])
    B = corr - A * h[i]
    return A, B

def coord_descent(h, n, n_sweeps=3, n_ternary=35):
    h = h.copy()
    for _ in range(n_sweeps):
        for i in np.random.permutation(n):
            A, B = _coord_coeffs_vec(h, i, n)
            a, b = 0.0, 1.0
            for _ in range(n_ternary):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                v1 = np.max(A * m1 + B)
                v2 = np.max(A * m2 + B)
                if v1 < v2:
                    b = m2
                else:
                    a = m1
            h[i] = (a + b) / 2.0
        h = fix_sum(h, n)
    return h

def subgrad_polish(h, n, n_iter, step_init, top_k):
    h = h.copy()
    step = step_init
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for ii in idx:
            grad += _grad_corr_at(h, ii, n)
        grad /= top_k
        h_new = np.clip(h - step * grad, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
        else:
            step *= 0.7
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t_start

    def try_update(h_cand, n_cand):

        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best - 1e-14:
            h_best = h_cand
            n_best = n_cand
            c5_best = c5_cand

        return False

    # Phase 1: SLSQP with analytic gradient, temperature sweep
    for T, tk, mi in [(0.05, 0, 3000), (0.02, 0, 4000), (0.008, 0, 4000),

        if budget_s - elapsed() < 130:

        h_opt = do_slsqp(h_best, n_best, T, tk, mi)

    for T, tk, mi, pw in [(0.01, 0, 4000, 500), (0.003, 5, 4000, 2000),
```
