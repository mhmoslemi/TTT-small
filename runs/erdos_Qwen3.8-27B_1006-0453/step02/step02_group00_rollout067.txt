```python
import cvxpy as cp

def solve_lp_relaxation(h0, n):
    dx = 2.0 / n
    h = cp.Variable(n)
    T = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2.0]
    
    for k in range(2*n - 1):
        m = k - (n - 1)
        # Compute grad at h0
        grad = np.zeros(n)
        if m >= 0:
            grad[:n-m] += (1.0 - h0[m:])  # (1-h0[i+m]) for i=0..n-m-1
            grad[m:] -= h0[:n-m]           # -h0[i-m] for i=m..n-1
        else:
            a = -m
            grad[a:] += (1.0 - h0[:n-a])  # (1-h0[i-m]) for i=a..n-1 (i+m >= 0)
            grad[:n-a] -= h0[a:]          # -h0[i+m] for i=0..n-a-1 (i+m < n)
        
        # corr_k(h0)
        if m >= 0:
            corr0 = np.sum(h0[:n-m] * (1.0 - h0[m:]))
        else:
            a = -m
            corr0 = np.sum(h0[a:] * (1.0 - h0[:n-a]))
        
        # Constraint: grad·h ≤ T - corr0 + grad·h0
        constraints.append(grad @ h <= T - corr0 + grad @ h0)
    
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)  # or CLARABEL if available
    except:
        prob.solve(solver=cp.SCS)
    
    if h.value is not None:
        h_sol = np.clip(h.value, 0, 1)
        h_sol = project_feasible(h_sol, n)
        return h_sol, T.value * dx
    return None, None
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def get_corr_shifts(h, n):
    """Return correlation values for all shifts, and their gradients."""
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    return corr

def slsqp_refine(h, T, n, maxiter=500):
    def obj(x):
        x = project_feasible(x, n)
        return smooth_c5(x, T)
    res = minimize(obj, h.copy(), method='SLSQP',
                  bounds=[(0.0, 1.0)]*n,
                  constraints=[{'type':'eq','fun': lambda x: x.sum() - n/2.0}],
                  options={'maxiter': maxiter, 'ftol': 1e-15})

def lp_iteration(h0, n, trust_radius=0.15):

    dx = 2.0 / n
    corr = get_corr_shifts(h0, n)
    
    h_var = cp.Variable(n)
    T_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
    
    # Trust region
    if trust_radius > 0:
        constraints.append(cp.abs(h_var - h0) <= trust_radius)
    
    for k in range(2*n - 1):
        m = k - (n - 1)
        # Gradient of corr_k at h0
        grad = np.zeros(n)
        if m >= 0:
            if n - m > 0:
                grad[:n-m] += (1.0 - h0[m:])
                grad[m:] -= h0[:n-m]
        else:
            a = -m
            if n - a > 0:
                grad[a:] += (1.0 - h0[:n-a])
                grad[:n-a] -= h0[a:]
        
        # corr_k at h0
        if m >= 0:
            corr0 = float(np.sum(h0[:n-m] * (1.0 - h0[m:]))) if n-m > 0 else 0.0
        else:
            a = -m
            corr0 = float(np.sum(h0[a:] * (1.0 - h0[:n-a]))) if n-a > 0 else 0.0
        
        # Linear constraint: corr0 + grad·(h-h0) <= T/dx
        # => grad·h <= T/dx - corr0 + grad·h0
        rhs = T_var / dx - corr0 + float(np.dot(grad, h0))
        constraints.append(grad @ h_var <= rhs)
    
    prob = cp.Problem(cp.Minimize(T_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None
    
    if h_var.value is None:
        return None
    
    h_sol = np.clip(h_var.value, 0, 1)
    h_sol = project_feasible(h_sol, n)
    return h_sol
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)
```
