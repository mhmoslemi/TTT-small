import time

    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_top_obj(h, T, M):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = min(M, len(corr))
    top = np.argpartition(corr, -M)[-M:]
    c = corr[top]
    cmax = c.max()
    e = (c - cmax) / T
    e -= e.max()
    return (cmax + T * (e.max() + np.log(np.exp(e).sum()))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    if m.sum() > 0:
        h[m] += d / m.sum()
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in np.where((h > 1e-4) & (h < 1.0 - 1e-4))[0]:
            h[i] = np.clip(h[i] + d2, 0, 1)
            break
    return h

    try:
        res = optimize.minimize(lse_top_obj, h, args=(T, M),
            method='SLSQP', bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': maxiter, 'ftol': 1e-15})
        hn = fix_sum(res.x, n)
        cn = compute_c5(hn)
        if cn < c0:
            return hn, cn

        pass
    return h, c0

            continue

        c_top = corr[top_idx]
        cmax = c_top.max()
        best_c5 = (cmax + 0.01 * np.log(np.sum(np.exp((c_top - cmax) / 0.01)))) * (2.0/n)
        best_d = 0.0
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2
        a, b = lo, hi
        x1 = b - gr * (b - a)
        x2 = a + gr * (b - a)
        for _ in range(20):
            h[i] += x1; h[j] -= x1
            c1 = np.max(np.correlate(h, 1.0 - h, mode='full')[top_idx])
            h[i] -= x1; h[j] += x1
            h[i] += x2; h[j] -= x2
            c2 = np.max(np.correlate(h, 1.0 - h, mode='full')[top_idx])
            h[i] -= x2; h[j] += x2
            if c1 < c2:
                b = x2; x2 = x1; x1 = b - gr*(b-a)
            else:
                a = x1; x1 = x2; x2 = a + gr*(b-a)
        best_d = (a + b) / 2
        h[i] = np.clip(h[i] + best_d, 0, 1)
        h[j] = np.clip(h[j] - best_d, 0, 1)
        h = fix_sum(h, n)
    return h

Hmm wait, the golden section search here isn't quite right because the function isn't unimodal in general (it's the max of linear functions of delta, which IS convex in delta... wait, max of linear functions IS convex). So the golden section should work!

Actually, for a pair move h[i]->h[i]+d, h[j]->h[j]-d:
Each corr[k] changes linearly in d (to first order, ignoring the fact that h[i] appears in both factors for k=0).

For the top-M max: max_k (A_k + d * B_k) which is convex in d (max of linear functions). Golden section works for convex functions.

But there's a subtlety: the top-M set changes as d changes. For the golden section, I should use the FIXED top-M from the initial evaluation. That's what I'm doing above (top_idx is fixed). Good.

However, for k=0 (no shift), corr[0] = sum h[i]*(1-h[i]) includes h[i]^2 terms, making it quadratic in d. But this is still convex in d (since it's a sum of quadratics with negative leading coefficients... wait, h[i]*(1-h[i]) = h[i] - h[i]^2, which is concave in h[i]. So corr[0] is concave in d when h[i] changes. This means the max of (some convex + some concave) might not be convex.

In practice, for small d, the linear approximation is good and the golden section should work. Let me just use a grid search with 30-50 points for robustness.

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_top_obj(h, T, M):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = min(M, len(corr))
    top = np.argpartition(corr, -M)[-M:]
    c = corr[top]
    cmax = c.max()
    e = (c - cmax) / T
    e -= e.max()
    return (cmax + T * (e.max() + np.log(np.exp(e).sum()))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e–7)
    if m.sum() > 0:
        h[m] += d / m.sum()
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        idx = np.where((h > 1e-4) & (h < 1.0 - 1e-4))[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + d2, 0, 1)
    return h

def slsqp_step(h, n, T, M, maxiter):
    c0 = compute_c5(h)
    try:
        res = optimize.minimize(lse_top_obj, h, args=(T, M),
            method='SLSQP', bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': maxiter, 'ftol': 1e-15})
        hn = fix_sum(res.x, n)
        cn = compute_c5(hn)
        if cn < c0 - 1e-14:
            return hn, cn
    except Exception:
        pass
    return h, c0

def pair_descent(h, n, rng, n_pairs, top_m=40):
    for _ in range(n_pairs):
        i, j = rng.choice(n, 2, replace=False)
        if h[i] <= 1e-5 or h[i] >= 1.0 - 1e-5:
            continue
        if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
            continue
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        best_c = np.max(corr[top_idx])
        best_d = 0.0
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue
        deltas = np.linspace(lo, hi, 31)
        for d in deltas:
            h[i] += d
            h[j] -= d
            c = np.max(np.correlate(h, 1.0 - h, mode='full')[top_idx])
            if c < best_c - 1e-14:
                best_c = c
                best_d = d
            h[i] -= d
            h[j] += d
        if best_d != 0.0:
            h[i] = np.clip(h[i] + best_d, 0, 1)
            h[j] = np.clip(h[j] - best_d, 0, 1)
        h = fix_sum(h, n)
    return h

def coord_descent(h, n, n_pass=2, top_m=30):
    for _ in range(n_pass):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        for j in range(n):
            if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
                continue
            best_c = np.max(corr[top_idx])
            best_d = 0.0
            lo = max(-h[j], 0.0)
            hi = min(1.0 - h[j], 0.0)
            # Single coordinate: adjust and redistribute to others
            # Just try small shifts
            for d in (-0.08, -0.03, -0.01, 0.01, 0.03, 0.08):
                hn = h.copy()
                hn[j] += d
                if hn[j] < 0 or hn[j] > 1:
                    continue
                hn = fix_sum(hn, n)
                c = np.max(np.correlate(hn, 1.0 - hn, mode='full')[top_idx])
                if c < best_c - 1e-14:
                    best_c = c
                    best_d = d
            if best_d != 0.0:
                h[j] = np.clip(h[j] + best_d, 0, 1)
                h = fix_sum(h, n)
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def downsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)
