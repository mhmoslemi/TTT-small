import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def fix_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = int(np.sum(m))
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def lse_obj_and_grad(h, T, pen=1.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    c_max = np.max(corr)
    exps = np.exp((corr - c_max) / T)
    sum_exp = np.sum(exps)
    log_sum = c_max + T * np.log(sum_exp)
    obj = log_sum * dx + pen * (np.sum(h) - 0.5 * n) ** 2
    
    w = exps / sum_exp
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        w_k = w[m + n - 1]
        if w_k < 1e-14:
            continue
        j_start = max(0, -m)
        j_end = min(n, n - m)
        js = np.arange(j_start, j_end)
        jm = js + m
        dm = (1.0 - h[jm]).copy()
        valid = (js - m) >= 0
        if np.any(valid):
            dm[valid] -= h[js[valid] - m]
        grad[js] += w_k * dm
    grad *= dx
    # Add penalty gradient
    grad += 2.0 * pen * (np.sum(h) - 0.5 * n) * np.ones(n)
    return obj, grad

def lbfgsb_step(h, n, T, pen, maxiter=500):
    T_local = T
    pen_local = pen
    def f(x):
        return lse_obj_and_grad(x, T_local, pen_local)[0]
    def fprime(x):
        return lse_obj_and_grad(x, T_local, pen_local)[1]
    try:
        res = optimize.minimize(f, h, jac=fprime, method='L-BFGS-B',
            bounds=[(0., 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return fix_sum(res.x, n)
    except Exception:
        return h

def subgrad_refine(h, n, n_steps=100, lr=0.01):
    """Take subgradient steps on the max lag."""
    for _ in range(n_steps):
        c5, corr = compute_c5(h)
        k_star = int(np.argmax(corr))
        m = k_star - n + 1  # shift
        # Compute gradient of corr[m] w.r.t. h
        j_start = max(0, -m)
        j_end = min(n, n - m)
        js = np.arange(j_start, j_end)
        jm = js + m
        g = (1.0 - h[jm])
        valid = (js - m) >= 0
        if np.any(valid):
            g = g.copy()
            g[valid] -= h[js[valid] - m]
        # Step: h[j] -= lr * g for active j, compensate sum
        g_full = np.zeros(n)
        g_full[js] = g
        # Adjust to preserve sum: subtract mean
        g_full -= np.mean(g_full[js])
        h_new = np.clip(h - lr * g_full, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new, _ = compute_c5(h_new)

            lr = min(lr * 1.05, 0.1)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def fix_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = int(np.sum(m))
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def lse_obj_and_grad(h, T, pen=1.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    exps = np.exp((corr - c_max) / T)
    sum_exp = np.sum(exps)
    obj = (c_max + T * np.log(sum_exp)) * dx + pen * (np.sum(h) - 0.5 * n) ** 2
    w = exps / sum_exp
    grad = np.zeros(n)
    j_all = np.arange(n)
    for m in range(-(n - 1), n):
        w_k = w[m + n - 1]
        if w_k < 1e-14:
            continue
        jm = j_all + m
        jmm = j_all - m
        g = np.zeros(n)
        v1 = (jm >= 0) & (jm < n)
        v2 = (jmm >= 0) & (jmm < n)
        g[v1] += (1.0 - h[jm[v1]])
        g[v2] -= h[jmm[v2]]
        grad += w_k * g
    grad *= dx
    grad += 2.0 * pen * (np.sum(h) - 0.5 * n)
    return obj, grad

def lbfgsb_step(h, n, T, pen, maxiter=400):
    def f(x):
        return lse_obj_and_grad(x, T, pen)[0]
    def fprime(x):
        return lse_obj_and_grad(x, T, pen)[1]
    try:
        res = optimize.minimize(f, h, jac=fprime, method='L-BFGS-B',
                                bounds=[(0.0, 1.0)] * n,
                                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return fix_sum(res.x, n)
    except Exception:
        return h

def subgrad_refine(h, n, n_steps=150, lr=0.005):
    for _ in range(n_steps):
        c5, corr = compute_c5(h)
        top_k = np.argsort(corr)[-5:]
        m_top = top_k - n + 1
        grad = np.zeros(n)
        j_all = np.arange(n)
        for mm in m_top:
            jm = j_all + mm
            jmm = j_all - mm
            g = np.zeros(n)
            v1 = (jm >= 0) & (jm < n)
            v2 = (jmm >= 0) & (jmm < n)
            g[v1] += (1.0 - h[jm[v1]])
            g[v2] -= h[jmm[v2]]
            grad += g
        grad /= len(m_top)
        grad -= np.mean(grad)
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new, _ = compute_c5(h_new)

            h = h_new
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.65

    return h

def coord_polish(h, n, n_pass=2):
    for _ in range(n_pass):
        for j in range(n):

                    best_h = hn
            h = best_h

        ht = lbfgsb_step(ht, n, 0.0005, pen=20.0, maxiter=300)
        ht = subgrad_refine(ht, n, n_steps=80, lr=0.003)
        ct, _ = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() - t0 < budget_s - 200:
                h = lbfgsb_step(h, n, 0.0001, pen=30.0, maxiter=500)
                h = subgrad_refine(h, n, n_steps=100, lr=0.002)
                c5, _ = compute_c5(h)

    # Phase 4: Upsample to 400 and optimize
    if time.time() - t0 < budget_s - 150:
        n2 = 400
        h2 = upsample(h, n2)
        for T, pen in [(0.03, 5.0), (0.01, 10.0), (0.003, 15.0), (0.001, 25.0), (0.0003, 40.0)]:
            if time.time() - t0 > budget_s - 30:
                break
            h2 = lbfgsb_step(h2, n2, T, pen, maxiter=300)
        h2 = subgrad_refine(h2, n2, n_steps=120, lr=0.003)
        h2 = lbfgsb_step(h2, n2, 0.0001, pen=50.0, maxiter=400)
        c5_2, _ = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2

    # Phase 5: Upsample to 550 if time allows
    if time.time() - t0 < budget_s - 80 and c5 < 0.382:
        n3 = 550
        h3 = upsample(h, n3)
        for T, pen in [(0.02, 5.0), (0.005, 15.0), (0.001, 30.0), (0.0003, 50.0)]:
            if time.time() - t0 > budget_s - 20:
                break
            h3 = lbfgsb_step(h3, n3, T, pen, maxiter=250)
        h3 = subgrad_refine(h3, n3, n_steps=100, lr=0.002)
        c5_3, _ = compute_c5(h3)
        if c5_3 < c5:
            h, n, c5 = h3, n3, c5_3

    # Final polish
    if time.time() - t0 < budget_s - 15:
        h = lbfgsb_step(h, n, 0.00005, pen=50.0, maxiter=500)
        h = subgrad_refine(h, n, n_steps=80, lr=0.001)
        h = coord_polish(h, n, n_pass=1)

    h = fix_sum(h, len(h))
    c5, _ = compute_c5(h)
    return (h, c5, len(h))
