```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0, 1)
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
    )
    return proj(res.x, n)

def lbfgsb_opt(h0, n, T, max_iter, lam=100.0):
    def obj_with_penalty(h):
        base = soft_max_obj(h, T)
        viol = np.sum(h) - n * 0.5
        return base + 0.5 * lam * viol * viol
    res = opt.minimize(
        obj_with_penalty, h0,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'gtol': 1e-12, 'ftol': 1e-16, 'iprint': 0}
    )
    return proj(res.x, n)

def coord_descent(h0, n, max_rounds=50):
    h = h0.copy()
    for r in range(max_rounds):
        improved = False
        for i in range(n):
            best_val = h[i]
            best_c = compute_c5(h)
            # Try a few values around current
            for delta in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                trial = h[i] + delta
                if 0 <= trial <= 1:
                    h[i] = trial
                    h = proj(h, n)
                    c = compute_c5(h)
                    if c < best_c:
                        best_c = c
                        best_val = h[i]
            h[i] = best_val
            h = proj(h, n)
        if not improved:
            break
    # Final proj
    return proj(h, n)
```

Let me just define it as: c[m] = sum over all (i,j) where j-i = m-(n-1) of h[i] * (1-h[j]), with the constraint that 0<=i,j<n.

Equivalently, for lag L = m-(n-1) (ranging from -(n-1) to n-1):
c[m] = sum_{i: 0<=i<n and 0<=i+L<n} h[i] * (1-h[i+L])

For L >= 0: sum_{i=0}^{n-1-L} h[i] * (1-h[i+L])
For L < 0: sum_{i=-L}^{n-1} h[i] * (1-h[i+L]) = sum_{i=0}^{n-1+L} h[i-L] * (1-h[i])

OK so for the correlation, each output element is a sum of products of pairs of h values (and 1-h values).

The gradient of c[m] with respect to h[j]:
For L = m - (n-1):
- If L >= 0: c[m] = sum_{i=0}^{n-1-L} h[i] * (1-h[i+L])
  dc[m]/dh[j] = (1-h[j+L]) * I(j in [0, n-1-L]) + sum_{i: i+L=j} h[i] * (-1)
              = (1-h[j+L]) * I(0<=j<n-L) - h[j-L] * I(0<=j-L<n-1-L+1)
              = (1-h[j+L]) * I(0<=j<n-L) - h[j-L] * I(L<=j<n)

- If L < 0 (let L' = -L > 0): c[m] = sum_{i=0}^{n-1-L'} h[i-L'] * (1-h[i])
  dc[m]/dh[j] = (1-h[j]) * I(j-L' in [0, n-1-L']) + sum_{i: i-L'=j} h[i-L'] * (-1)
              = (1-h[j]) * I(0<=j-L'<n-1-L'+1) - h[j] * I(j in [0, n-1-L'])
              = (1-h[j]) * I(L'<=j<n) - h[j] * I(0<=j<n-L')

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        ni = np.sum(interior)
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def do_slsqp(h0, n, T, max_iter, ftol=1e-16):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0}
    )
    return proj(res.x, n)

def do_lbfgsb(h0, n, T, max_iter, lam=50.0):
    def obj_pen(h):
        base = soft_max_obj(h, T)
        viol = np.sum(h) - n * 0.5
        return base + 0.5 * lam * viol * viol
    res = opt.minimize(
        obj_pen, h0,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-15, 'iprint': 0}
    )
    return proj(res.x, n)

def sa_refine(h0, n, t_start, deadline, max_steps, seed, T_init, T_mult):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T_sa = T_init
    # Momentum
    momentum = np.zeros(n)
    for step in range(max_steps):
        if time.time() > deadline:
            break
        # Adaptive block size
        blk = rng.randint(2, 8 + step % 6)
        start = rng.randint(0, n - blk)
        idxs = np.arange(start, start + blk)
        # Gradient-directed bias
        noise = rng.randn(blk) * 0.03
        # Add small gradient estimate direction
        h_try = h_cur.copy()
        h_try[idxs] = np.clip(h_try[idxs] + noise, 0, 1)
        h_try = proj(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T_sa):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T_sa *= T_mult
    return h_best, c_best

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj(h_new, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with penalty - fast many iterations
    elapsed = time.time() - t_start
    if budget_s - elapsed > 200:
        for T in [0.03, 0.01, 0.004, 0.0015, 0.0006, 0.0002]:
            if time.time() > deadline - 150:
                break
            h_try = do_lbfgsb(h_best, n_best, T, 500, lam=80.0)
            c = compute_c5(h_try)
            if c < c5_best:
                h_best, c5_best = h_try, c

    # Phase 2: SLSQP with soft-max (precise, constraining)
    elapsed = time.time() - t_start
    if budget_s - elapsed > 250:
        for T in [0.04, 0.015, 0.006, 0.002, 0.0008, 0.0003, 0.0001]:
            if time.time() > deadline - 200:
                break
            h_try = do_slsqp(h_best, n_best, T, 800)
            c = compute_c5(h_try)
            if c < c5_best:
                h_best, c5_best = h_try, c

    # Phase 3: Multi-start perturbations with SLSQP
    for i in range(8):
        if time.time() > deadline - 250:
            break
        rng = np.random.RandomState(seed + i * 104729 + 7)
        sigma = 0.008 + 0.006 * (i % 3)
        h_pert = np.clip(h_best + rng.randn(n_best) * sigma, 0, 1)
        h_pert = proj(h_pert, n_best)
        T_p = [0.003, 0.0015, 0.0008, 0.004][i % 4]
        h_try = do_slsqp(h_pert, n_best, T_p, 600)
        c = compute_c5(h_try)
        if c < c5_best:
            h_best, c5_best = h_try, c

    # Phase 4: Simulated annealing with structured moves
    if time.time() > deadline - 180:
        pass
    else:
        sa_deadline = min(deadline - 60, t_start + budget_s * 0.78)
        h_sa, c_sa = sa_refine(h_best, n_best, t_start, sa_deadline, 400, seed + 12345, 0.0008, 0.997)
        if c_sa < c5_best:
            h_best, c5_best = h_sa, c_sa

    # Phase 5: Resolution change attempts
    for n_try in [280, 420, 504]:
        if time.time() > deadline - 80:
            break
        h_up = upsample(h_best, n_try)
        h_try = do_slsqp(h_up, n_try, 0.003, 500)
        c = compute_c5(h_try)
        if c < c5_best:
            h_best, c5_best, n_best = h_try, c, n_try

    # Phase 6: Ultra-fine SLSQP polish
    if time.time() > deadline - 40:
        pass
    else:
        for T in [0.0001, 0.00004, 0.00002]:
            if time.time() > deadline - 15:
                break
            h_try = do_slsqp(h_best, n_best, T, 2000, ftol=1e-18)
            c = compute_c5(h_try)
            if c < c5_best:
                h_best, c5_best = h_try, c

    # Final L-BFGS-B polish
    if time.time() > deadline - 10:
        pass
    else:
        h_try = do_lbfgsb(h_best, n_best, 0.0001, 400, lam=200.0)
        c = compute_c5(h_try)
        if c < c5_best:
            h_best, c5_best = h_try, c

    h_best = proj(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
```
