```python
        x_old = np.linspace(0, 2, best_n)

    best_h = best_b.astype(float)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def to_binary(h, n):
    h = np.array(h, dtype=float).flatten()
    if len(h) != n:
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_old, h)
    b = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(b.sum())
    if n_ones > target:
        idx = np.where(b == 1)[0]
        np.random.shuffle(idx)
        b[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(b == 0)[0]
        np.random.shuffle(idx)
        b[idx[:target - n_ones]] = 1
    return b

def ls_bimprove(b, max_iters=5000, timeout=None, t0=None):
    n = len(b)
    b = b.copy()
    c5 = compute_c5(b)
    for it in range(max_iters):
        if timeout is not None and time.time() - t0 > timeout:
            break
        best_delta = 0.0
        best_move = None
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        for i in ones:
            for j in zeros:
                b[i], b[j] = 0.0, 1.0
                nc = compute_c5(b)
                b[i], b[j] = 1.0, 0.0
                d = nc - c5
                if d < best_delta - 1e-14:
                    best_delta = d
                    best_move = (i, j)
        if best_move is not None:
            b[best_move[0]], b[best_move[1]] = 0.0, 1.0
            c5 += best_delta
        else:
            break
    return b, c5

def ls_2swap(b, max_iters=3000, timeout=None, t0=None):
    """Local search allowing 2-pair swaps"""
    n = len(b)
    b = b.copy()
    c5 = compute_c5(b)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for it in range(max_iters):
        if timeout is not None and time.time() - t0 > timeout:
            break
        best_delta = 0.0
        best_move = None
        # Single swap
        for i in ones:
            for j in zeros:
                b[i], b[j] = 0.0, 1.0
                nc = compute_c5(b)
                b[i], b[j] = 1.0, 0.0
                d = nc - c5
                if d < best_delta - 1e-14:
                    best_delta = d
                    best_move = ('1', i, j)
        # 2-swap (sample to keep tractable)
        if len(ones) >= 2 and len(zeros) >= 2:
            n_samples = min(200, len(ones) * len(zeros))
            for _ in range(n_samples):
                i1 = np.random.choice(ones)
                i2 = np.random.choice(ones[ones != i1])
                j1 = np.random.choice(zeros)
                j2 = np.random.choice(zeros[zeros != j1])
                b[i1], b[i2], b[j1], b[j2] = 0.0, 0.0, 1.0, 1.0
                nc = compute_c5(b)
                b[i1], b[i2], b[j1], b[j2] = 1.0, 1.0, 0.0, 0.0
                d = nc - c5
                if d < best_delta - 1e-14:
                    best_delta = d
                    best_move = ('2', i1, i2, j1, j2)
        if best_move is not None:
            if best_move[0] == '1':
                b[best_move[1]], b[best_move[2]] = 0.0, 1.0
            else:
                b[best_move[1]], b[best_move[2]] = 0.0, 0.0
                b[best_move[3]], b[best_move[4]] = 1.0, 1.0
            c5 += best_delta
        else:
            break
    return b, c5

def sa_binary(b, n_iter=15000, T0=0.04, T1=0.001, timeout=None, t0=None):
    n = len(b)
    b = b.copy()
    c5 = compute_c5(b)
    for it in range(n_iter):
        if timeout is not None and time.time() - t0 > timeout:
            break
        T = max(T0 * (T1 / T0) ** (it / max(n_iter - 1, 1)), 1e-9)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i, j = np.random.choice(ones), np.random.choice(zeros)
        b[i], b[j] = 0.0, 1.0
        nc = compute_c5(b)
        d = nc - c5
        if d < 0 or np.random.random() < np.exp(-d / T):
            c5 = nc
        else:
            b[i], b[j] = 1.0, 0.0
    return b, c5

def ils_search(b, n_restarts=25, timeout=None, t0=None):
    """Iterated local search with perturbations"""
    n = len(b)
    best_b = b.copy()
    best_c5 = compute_c5(b)
    for r in range(n_restarts):
        if timeout is not None and time.time() - t0 > timeout:
            break
        # Perturb: random number of pair flips
        pert_b = best_b.copy()
        n_flips = np.random.randint(2, max(3, n // 15) + 1)
        ones = np.where(pert_b == 1)[0]
        zeros = np.where(pert_b == 0)[0]
        for _ in range(n_flips):
            oi = np.where(pert_b == 1)[0]
            zi = np.where(pert_b == 0)[0]
            pert_b[np.random.choice(oi)], pert_b[np.random.choice(zi)] = 0.0, 1.0
        # Local search
        ls_b, ls_c5 = ls_bimprove(pert_b, max_iters=3000)
        if ls_c5 < best_c5:
            best_b, best_c5 = ls_b, ls_c5
    return best_b, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
    else:
        init = np.zeros(90)
        init[:45] = 1.0

    best_b = to_binary(init, len(init))
    best_c5 = compute_c5(best_b)
    best_n = len(best_b)

    def elapsed():
        return time.time() - t0

    # Phase 1: Convergence local search on parent
    best_b, best_c5 = ls_bimprove(best_b, max_iters=5000, timeout=budget_s * 0.10, t0=t0)

    # Phase 2: 2-swap local search (escape single-swap local optima)
    if elapsed() < budget_s * 0.25:
        b2, c52 = ls_2swap(best_b, max_iters=2000, timeout=budget_s * 0.10, t0=t0)
        if c52 < best_c5:
            best_c5, best_b = c52, b2
            best_b, best_c5 = ls_bimprove(best_b, max_iters=3000, timeout=budget_s * 0.05, t0=t0)

    # Phase 3: Iterated local search (multiple perturb+refine cycles)
    if elapsed() < budget_s * 0.50:
        best_b, best_c5 = ils_search(best_b, n_restarts=20, timeout=budget_s * 0.25, t0=t0)

    # Phase 4: SA exploration from best
    if elapsed() < budget_s * 0.70:
        for sa_round in range(4):
            if elapsed() > budget_s * 0.70:
                break
            b_sa, _ = sa_binary(best_b, n_iter=10000,
                                T0=0.02 + 0.015 * sa_round, T1=0.001,
                                timeout=budget_s * 0.07, t0=t0)
            b_ls, c5_ls = ls_bimprove(b_sa, max_iters=4000, timeout=budget_s * 0.05, t0=t0)
            if c5_ls < best_c5:
                best_c5, best_b = c5_ls, b_ls
                best_n = len(best_b)

    # Phase 5: Multi-resolution search
    for n_try in [80, 84, 96, 100, 110, 120]:
        if elapsed() > budget_s * 0.85:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_b)
        b_init = to_binary(h_interp, n_try)
        b_ls, c5_ls = ls_bimprove(b_init, max_iters=4000, timeout=budget_s * 0.06, t0=t0)
        if c5_ls < best_c5:
            best_c5, best_b, best_n = c5_ls, b_ls, n_try
            # Quick SA polish at new resolution
            if elapsed() < budget_s * 0.85:
                b_sa, _ = sa_binary(best_b, n_iter=5000, T0=0.02, T1=0.001,
                                    timeout=budget_s * 0.04, t0=t0)
                b_ls2, c5_ls2 = ls_bimprove(b_sa, max_iters=3000, timeout=budget_s * 0.03, t0=t0)
                if c5_ls2 < best_c5:
                    best_c5, best_b = c5_ls2, b_ls2

    # Phase 6: Final intensive local search
    if elapsed() < budget_s * 0.96:
        best_b, best_c5 = ls_bimprove(best_b, max_iters=10000, timeout=budget_s * 0.04, t0=t0)
        b2, c52 = ls_2swap(best_b, max_iters=1500, timeout=max(0, budget_s * 0.01 - (elapsed() - t0)), t0=t0)
        if c52 < best_c5:
            best_c5, best_b = c52, b2
            best_b, best_c5 = ls_bimprove(best_b, max_iters=2000, timeout=max(0, budget_s * 0.005 - (elapsed() - t0)), t0=t0)

    best_h = best_b.astype(float)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
