import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_core(x, t, topk, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if topk > 0 and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
        p = np.zeros(len(corr))
        p[idx] = np.exp(t * (c - c.max()))
        p /= p.sum()
    else:
        p = np.exp(t * (corr - corr.max()))
        p /= p.sum()
    mx = c.max() if (topk > 0 and topk < len(corr)) else corr.max()
    s = t * (c - mx)
    val = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    g = 1.0 - x
    conv1 = np.convolve(p, g, mode='full')
    ctrl = np.correlate(p, x, mode='full')
    grad = dx * (conv1[n-1:2*n-1] - ctrl[:n])
    if lam > 0:
        sv = np.sum(x) - n / 2.0
        val += lam * sv * sv
        grad = grad + 2.0 * lam * sv
    return val, grad

def sm_val(x, t, topk, lam):
    return _obj_core(x, t, topk, lam)[0]

def sm_grad(x, t, topk, lam):
    return _obj_core(x, t, topk, lam)[1]

def ceq_fun(x):
    return np.sum(x) - len(x) / 2.0

def ceq_jac(x):
    return np.ones(len(x))

    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return project_feasible(np.interp(xd, xs, h), nd)

def binary_push(h, n):
    hb = (h > 0.5).astype(float)
    return project_feasible(hb, n)
