M2[i,j] = w[i-j+n-1] = w[-(j-i)+n-1]

    for n_try in [54, 66, 78, 96, 108]:
        if elapsed() > budget_s * 0.72:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_try = lbbfgs_opt(h_interp, 500, maxiter=250, penalty=600.0)
        h_try = slsqp_smooth(h_try, 3000, maxiter=200)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            curr_n = len(best_h)

    # Phase 5: High-temperature polish
    curr_n = len(best_h)
    for T in [5000, 15000, 30000]:
        if elapsed() > budget_s * 0.84:
            break
        h_try = lbbfgs_opt(best_h, T, maxiter=400, penalty=1000.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        h_try = slsqp_smooth(best_h, T * 2, maxiter=300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 6: Fine restarts around best
    curr_n = len(best_h)
    n_fine = int(min(25, max(10, budget_s * 0.018)))
    for trial in range(n_fine):
        if elapsed() > budget_s * 0.93:
            break
        scale = 0.015 + 0.01 * np.random.rand()
        h_pert = best_h + np.random.randn(curr_n) * scale
        h_pert = project_feasible(h_pert, curr_n)
        h_try = lbbfgs_opt(h_pert, 10000, maxiter=300, penalty=1200.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 7: Direct SLSQP max-min polish
    if elapsed() < budget_s * 0.97:
        h_try = slsqp_direct(best_h, maxiter=500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        h_try = slsqp_smooth(best_h, 50000, maxiter=400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
