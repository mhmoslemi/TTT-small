def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:

            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

    """Compute full cross-correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_obj(h, T):
    """Smoothed max of correlation values times dx"""
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = (corr - c_max) / T
    shifted = np.clip(shifted, -80.0, 0.0)
    return (c_max + T * np.log(np.sum(np.exp(shifted)))) * dx

def softmax_grad(h, T):
    """Analytical gradient of smoothed max objective.
    Returns gradient of T*log(sum(exp(c_s/T))) * dx w.r.t. h.
    """
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    c_max = np.max(corr)
    if T < 1e-15:
        # Return subgradient of max
        active = np.where(np.abs(corr - c_max) < 1e-10)[0]
        result = np.zeros(n)
        for s_idx in active:
            s = s_idx - (n - 1)  # shift

                jp = (j + s) % n
                result[j] += (1.0 - h[jm]) - h[jp]

    exps = np.exp(shifted)

        jp = (np.arange(n) + s) % n
        result += w * ((1.0 - h[jm]) - h[jp])
    
    return result * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        vals = 1.0 - h if diff > 0 else h.copy()
        idx = np.argsort(vals)
        for i in idx:
            space = (1.0 - h[i]) if diff > 0 else h[i]
            add = min(abs(diff), space)
            h[i] += add if diff > 0 else -add
            diff -= add if diff > 0 else -add
