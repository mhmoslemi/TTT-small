                grad[:rem] -= wm * h[s:]
        else:

            rem = n - r
            if rem > 0:

                grad[r:] -= wm * h[:rem]

    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

    if n % nd == 0:

    xs = (np.arange(n) + 0.5) * 2.0 / n

def binary_ls(h, n, max_its, rng, tb):
    """Binary local search with pair swaps and simulated annealing."""
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    cc, _ = compute_c5(h)
    t_start = time.time()
    T = 0.002  # initial "temperature" for SA
    for it in range(max_its):
        if time.time() - t_start > tb:
            break
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        nc, _ = compute_c5(h)
        if nc < cc:
            cc = nc
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            # Simulated annealing: accept worse move with prob exp(-delta/T)
            if rng.random() < np.exp(-(nc - cc) / T):
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                cc = nc
            else:
                h[i] = 1.0
                h[j] = 0.0
        T *= 0.999
    return h, cc

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        c = np.count_nonzero(m)
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth(h, t, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
        mx = np.max(c)
        w = np.zeros(len(corr))
        w[idx] = np.exp(t * (c - mx))
        sw = np.sum(w)
        w /= sw
    else:
        mx = np.max(corr)
        w = np.exp(t * (corr - mx))
        w /= np.sum(w)
