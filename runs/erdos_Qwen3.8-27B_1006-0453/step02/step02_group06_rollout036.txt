- term h[i]: i=j gives (1-h[j+s])
- term (1-h[i+s]): i+s=j gives -h[j-s]
Total: (1-h[j+s]) - h[j-s]

So the correct formula is: dC[s]/dh[j] = (1-h[j+s]) - h[j-s]

And the soft-max gradient:
grad[j] = dx * sum_s w[s] * ((1-h[j+s]) - h[j-s])
= dx * (1 - sum_s w[s]*h[j+s] - sum_s w[s]*h[j-s])
= dx * (1 - B[j] - A[j])

where A[j] = sum_s w[s]*h[j-s] and B[j] = sum_s w[s]*h[j+s]

This is the same as before! The formula is symmetric in A and B. Good.

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def soft_max_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    weights = np.exp(shifted / T)
    Z = weights.sum()
    obj = (c_max + T * np.log(Z)) * dx
    w_arr = weights / Z
    A = np.convolve(w_arr, h, 'full')[n-1:2*n-1]
    B = np.convolve(w_arr[::-1], h, 'full')[n-1:2*n-1]
    grad = dx * (1.0 - A - B)
    return obj, grad

def sum_con_fn(h, n):
    return np.sum(h) - n * 0.5

    return np.ones(n)

def pnorm_val(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (np.sum(corr**p) / len(corr))**(1.0/p) * dx

    T, lam, n = args
    val, _ = soft_max_val_grad(x, T)
    if lam > 0:
        s = np.sum(x) - n * 0.5
        val += lam * s * s
    return val

def top_grad(x, args):
    T, lam, n = args
    _, g = soft_max_val_grad(x, T)
    if lam > 0:
        s = np.sum(x) - n * 0.5
        g = g + 2.0 * lam * s
    return g

def smax_slq_obj(x, args):
    T = args[0]
    return soft_max_val_grad(x, T)[0]

def do_lbfgsb(h0, n, T, lam, max_iter):
    bounds = [(0.0, 1.0)] * n
    args = (T, lam, n)
    res = opt.minimize(top_obj, h0, jac=top_grad, args=args,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-12})
    return proj_constraint(res.x, n)

def do_slsqp(h0, n, T, max_iter):
    args = (T,)
    cons = [{'type': 'eq', 'fun': sum_con_fn, 'jac': sum_con_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(smax_slq_obj, h0, args=args,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
    return proj_constraint(res.x, n)

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})

    for step in range(max_steps):

        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_blk) * 0.03, 0, 1)

            if c_try < c_best:
