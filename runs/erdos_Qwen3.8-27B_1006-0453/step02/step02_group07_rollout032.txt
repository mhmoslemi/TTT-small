from scipy import optimize
import time

# Helper functions (top-level)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    k = np.sum(free)
    if k > 0:
        h[free] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-3 < h[i] < 1.0 - 1e-3:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def proj_set(h, n):
    target = 0.5 * n
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def slsqp(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(
            lse_obj, h, args=(T,),
            method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def fista_subgrad(h, n, n_iters, T=0.005):
    """FISTA with smoothed subgradient."""
    t = 1.0
    h_old = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    step = 1.0 / (n * dx)  # = 1, roughly
    
    for it in range(n_iters):
        # Anneal T
        frac = it / max(1, n_iters - 1)
        T_cur = T * (0.1) ** frac
        
        # Compute correlation and gradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        exps = np.exp((corr - c_max) / T_cur)
        p = exps / np.sum(exps)
        
        grad = np.zeros(n)
        top_idx = np.argsort(p)[::-1]
        top_idx = top_idx[p[top_idx] > 1e-10]
        for m in top_idx:
            pm = p[m]
            o1 = m - (n - 1)
            ks1, ke1 = max(0, -o1), min(n, n - o1)
            if ks1 < ke1:
                grad[ks1:ke1] += pm * dx * (1.0 - y[ks1+o1:ke1+o1])
            o2 = (n - 1) - m
            ks2, ke2 = max(0, -o2), min(n, n - o2)
            if ks2 < ke2:
                grad[ks2:ke2] -= pm * dx * y[ks2+o2:ke2+o2]
        
        # FISTA step with backtracking
        h_new = proj_set(y - step * grad, n)
        # Check if we made progress
        c5_try = compute_c5(h_new)
        
        # Nesterov momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = max(0, (t - 1.0) / t_next)
        y = h_new + beta * (h_new - h_old)
        h_old = h_new
        h = h_new
        t = t_next
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h.copy()
    
    return best_h, best_c5

def pair_polish(h, n, rng_obj, n_pairs=400):

        j, l = rng_obj.choice(n, 2, replace=False)

Also, the step size matters a lot. Let me think about what step size to use.

The LSE objective f_T(h) = dx * (max(corr) + T * log(Σ exp((corr - max)/T)))

The gradient involves the correlation at each lag, which depends on all h values. The "Lipschitz constant" of the gradient is related to how much the correlation changes when h changes. Since corr[m] = Σ_i h[i](1-h)[i+m-(n-1)], the second derivative is bounded by roughly 2n (each h appears in at most n correlations). So L ≈ 2n.

Step size ≈ 1/L ≈ 1/(2n) ≈ 1/560 for n=280.

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    k = np.sum(free)
    if k > 0:
        h[free] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-3 < h[i] < 1.0 - 1e-3:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def proj_set(h, n):
    target = 0.5 * n
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def fista_opt(h, n, n_iters, T_init=0.005, T_final=0.0005):
    t = 1.0
    h_old = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    step = 1.0 / n
    stall = 0
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        T_cur = T_init * (T_final / T_init) ** frac
        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        exps = np.exp((corr - c_max) / T_cur)
        p = exps / np.sum(exps)
        grad = np.zeros(n)
        top_idx = np.argsort(p)[::-1]
        top_idx = top_idx[p[top_idx] > 1e-10]
        for m in top_idx:
            pm = p[m]
            o1 = m - (n - 1)
            ks1, ke1 = max(0, -o1), min(n, n - o1)
            if ks1 < ke1:
                grad[ks1:ke1] += pm * dx * (1.0 - y[ks1 + o1:ke1 + o1])
            o2 = (n - 1) - m
            ks2, ke2 = max(0, -o2), min(n, n - o2)
            if ks2 < ke2:
                grad[ks2:ke2] -= pm * dx * y[ks2 + o2:ke2 + o2]
        h_new = proj_set(y - step * grad, n)
        c5_try = compute_c5(h_new)
        if c5_try > best_c5 + 0.002:
            h_new = proj_set(y - 0.3 * step * grad, n)
            c5_try = compute_c5(h_new)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        if c5_try < best_c5:
            beta = max(0, (t - 1.0) / t_next)
            stall = 0
            best_c5 = c5_try
            best_h = h_new.copy()
        else:
            beta = 0.0
            stall += 1
            if stall > 400:
                step *= 0.7
                stall = 0
            if stall > 1200:
                y = best_h.copy()
                h_old = best_h.copy()
                h_new = best_h.copy()
                t = 1.0
                stall = 0
                continue
        y = h_new + beta * (h_new - h_old)
        h_old = h_new
        h = h_new
        t = t_next
        step = min(step * 1.05, 5.0 / n)
    return best_h, best_c5

def coord_polish(h, n, n_pass=3):
    for _ in range(n_pass):
        changed = False
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            best_c5 = compute_c5(h)
            best_h = h
            for delta in (-0.04, -0.015, -0.004, 0.004, 0.015, 0.04):
                hn = h.copy()
                hn[:] += -delta / max(1, n - 1)
                hn[j] = min(max(h[j] + delta, 0.0), 1.0)
                hn = np.clip(hn, 0.0, 1.0)
                c = compute_c5(hn)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = hn
                    changed = True
            h = best_h
        h = fix_sum(h, n)
        if not changed:
            break
    return h

def pair_polish(h, n, rng_obj, n_pairs=500):
    for _ in range(n_pairs):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4 or h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue
        best_c5 = compute_c5(h)
        best_d = 0.0
        for d in (-0.04, -0.012, -0.003, 0.003, 0.012, 0.04):
            hj2, hl2 = h[j] + d, h[l] - d
            if hj2 < 1e-4 or hj2 > 1.0 - 1e-4 or hl2 < 1e-4 or hl2 > 1.0 - 1e-4:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj2, hl2
            c = compute_c5(hn)
            if c < best_c5 - 1e-13:
                best_c5 = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)
