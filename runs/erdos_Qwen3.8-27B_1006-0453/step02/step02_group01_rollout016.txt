    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, h, mode='full')[n-1:] * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_optimize(h, n_iter=5000, alpha0=0.5, top_k=5, momentum=0.9, seed_offset=0):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed_offset)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    
    for it in range(n_iter):
        R = compute_R(h)
        c5_cur = 1.0 - R.min()
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        min_idx = np.argsort(R)[:top_k]
        grad = np.zeros(n)
        for k in min_idx:
            if k == 0:
                grad -= 2.0 * h
            else:
                grad[:n-k] -= h[k:]
                grad[k:] -= h[:n-k]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        alpha = alpha0 * alpha0 / (alpha0 + it)
        v = momentum * v - alpha * grad / gnorm
        h_new = h + dx * v
        h_new = project_feasible(h_new, n)

        if c5_new < c5_cur:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    try:

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_R(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, h, mode='full')[n - 1:] * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_optimize(h, n_iter=4000, step0=0.03, top_k=5, momentum=0.85, seed_offset=0):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def erdos_ineq(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def slsqp_erdos(h, maxiter=300):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.01
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq}, {'type': 'ineq', 'fun': erdos_ineq}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)

        return h
