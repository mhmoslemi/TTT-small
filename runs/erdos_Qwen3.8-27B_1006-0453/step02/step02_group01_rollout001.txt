    starts.append((x < 1.0).astype(float))
    for freq in [1, 2, 3, 4]:
        s = 0.5 + 0.5 * np.sin(np.pi * freq * x)
        s = (s > 0.5).astype(float)
        starts.append(s)
    s = np.tile([1.0, 0.0], n // 2 + 1)[:n]
    starts.append(s.astype(float))
    s = np.tile([1.0, 1.0, 0.0, 0.0], n // 4 + 1)[:n]
    starts.append(s.astype(float))
    rng = np.random.RandomState(123)
    for _ in range(4):
        s = (rng.rand(n) > 0.5).astype(float)
        starts.append(s)
    return [project_feasible(s.copy(), n) for s in starts]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 90
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    n = best_n
    # Phase 1: Binary SA from multiple diverse starts
    t_phase = budget_s * 0.25
    starts = diverse_starts(n)
    if not (elapsed() < t_phase - 5):
        starts = starts[:4]
    for idx, s in enumerate(starts):
        if elapsed() > t_phase:
            break
        bh, bc = binary_sa(s, n_iters=20, swaps_per_iter=6000, T0=0.25, cooling=0.98, seed=seed + idx)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh.copy()

    # Phase 2: SLSQP smooth optimization from best binary
    if elapsed() < budget_s * 0.35:
        for T in [50, 200, 1000, 5000]:
            h_try = slsqp_smooth(best_h, T, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 3: Direct Erdos SLSQP
    if elapsed() < budget_s * 0.45:
        h_try = slsqp_erdos(best_h, maxiter=500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: Directed descent (targeting active lags)
    if elapsed() < budget_s * 0.55:
        h_try, c5_try = directed_descent(best_h, n_steps=300, seed=seed)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        h_try = slsqp_smooth(best_h, 500, maxiter=300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: Multi-resolution exploration
    if elapsed() < budget_s * 0.70:
        for n_try in [72, 80, 90, 100, 110, 120]:
            if elapsed() > budget_s * 0.70:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            bh, bc = binary_sa(h_interp, n_iters=10, swaps_per_iter=4000, T0=0.2, cooling=0.97, seed=seed + n_try)
            if bc < best_c5:
                h_smooth = slsqp_smooth(bh, 300, maxiter=200)
                c5_s = compute_c5(h_smooth)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_smooth

    # Phase 6: Fine-grained basin hopping with small perturbations
    if elapsed() < budget_s * 0.85:
        for trial in range(15):
            if elapsed() > budget_s * 0.85:
                break
            scale = 0.03 + 0.02 * np.random.rand()
            h_pert = best_h + np.random.randn(len(best_h)) * scale
            h_pert = project_feasible(h_pert, len(best_h))
            h_opt = slsqp_smooth(h_pert, 500, maxiter=250)
            h_opt = slsqp_erdos(h_opt, maxiter=200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 7: Final intensive polish
    if elapsed() < budget_s * 0.93:
        for T in [2000, 10000]:
            h_try = slsqp_smooth(best_h, T, maxiter=400)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        h_try = slsqp_erdos(best_h, maxiter=600)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 8: One more binary SA round on the rounded current best
    if elapsed() < budget_s * 0.97:
        bh, bc = binary_sa(best_h, n_iters=15, swaps_per_iter=5000, T0=0.1, cooling=0.99, seed=seed + 999)
        bh = slsqp_smooth(bh, 1000, maxiter=300)
        bc = compute_c5(bh)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
