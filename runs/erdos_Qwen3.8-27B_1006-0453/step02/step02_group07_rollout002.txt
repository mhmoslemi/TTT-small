import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_vals(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lse_obj(x, T):
    c = corr_vals(x)
    cmax = np.max(c)
    if T < 1e-12:
        return cmax
    return cmax + T * np.log(np.sum(np.exp((c - cmax) / T)))

def grad_corr_at_k(h, k):
    n = len(h)
    dx = 2.0 / n
    shift = k - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        g[shift:n] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return dx * g

def sum_con(x, n):
    return np.sum(x) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-6) & (h < 1.0 - 1e-6)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-12:
        idx = np.where((h > 1e-4) & (h < 1.0 - 1e-4))[0]
        for i in idx:
            h[i] = np.clip(h[i] + d2, 0.0, 1.0)
            break
    return h

def slsqp_opt(h, n, T, maxiter):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(
            lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def targeted_subgrad(h, n, n_steps, lr0, lr1):
    best = compute_c5(h)
    for it in range(n_steps):
        lr = lr0 * (lr1 / lr0) ** (it / max(n_steps, 1))
        c = corr_vals(h)
        k_star = int(np.argmax(c))
        g = grad_corr_at_k(h, k_star)
        # Also mix in top-3 lags for robustness
        top3 = np.argsort(c)[-3:]
        for kk in top3:
            if kk != k_star:
                g += 0.3 * grad_corr_at_k(h, int(kk))
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best:
            h = h_new
            best = c_new
    return h, best

def fourier_perturb(h, n, rng, max_mode, amp):
    h_new = h.copy()
    x = np.arange(n) * 2.0 / n
    for m in range(1, max_mode + 1):
        a = rng.normal(0, amp / (1.0 + 0.3 * m))
        b = rng.normal(0, amp / (1.0 + 0.3 * m))
        h_new += a * np.sin(np.pi * m * x) + b * np.cos(np.pi * m * x)
    return fix_sum(np.clip(h_new, 0.0, 1.0), n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    h_interp = np.interp(x1, x0, h)
    return fix_sum(h_interp, nn)

def pair_polish(h, n, rng, n_pairs):
    c5 = compute_c5(h)
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] < 1e-3 or h[j] > 1 - 1e-3 or h[l] < 1e-3 or h[l] > 1 - 1e-3:
            continue
        bc = c5
        bd = 0.0
        for d in np.array([-0.025, -0.01, -0.004, 0.004, 0.01, 0.025]):
            hn = h.copy()
            hn[j] = np.clip(h[j] + d, 0, 1)
            hn[l] = np.clip(h[l] - d, 0, 1)
            hn = fix_sum(hn, n)
            cc = compute_c5(hn)
            if cc < bc - 1e-11:
                bc = cc
                bd = d
        if bd != 0.0:
            h[j] = np.clip(h[j] + bd, 0, 1)
            h[l] = np.clip(h[l] - bd, 0, 1)
            h = fix_sum(h, n)
            c5 = bc
    return h, c5
