import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    # Distribute correction to interior points
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    k = np.sum(interior)
    if k > 0:
        h[interior] += d / k
        h = np.clip(h, 0.0, 1.0)
    # If still off, adjust one point
    d2 = target - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = np.clip(h[i] + d2, 0.0, 1.0)
                break
    return h

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def slsqp_step(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0 - 1e-12:
            return hn
    except Exception:
        pass
    return h

import numpy as np
from scipy import optimize

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    k = np.sum(interior)
    if k > 0:
        h[interior] += d / k
        h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = np.clip(h[i] + d2, 0.0, 1.0)
                break
    return h

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def slsqp_step(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0 - 1e-13:
            return hn
    except Exception:
        pass
    return h
