def solve_quadratic(Q, c, N, target):
    # min h^T Q h + c^T h subject to 0 <= h <= 1, sum(h) = target
    # Use scipy.optimize.minimize or cvxpy
    from scipy.optimize import minimize
    h0 = np.ones(N) * target / N
    
    def obj(h):
        return h @ Q @ h + c @ h
    
    def grad(h):
        return 2 * Q @ h + c
    
    constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - target, 'jac': lambda h: np.ones(N)}]
    bounds = [(0, 1)] * N
    
    res = minimize(obj, h0, jac=grad, method='trust-constr', 
                   bounds=bounds, constraints=constraints)
    return res.x

    true_c5 = float(np.max(corr)) * (2.0 / N)
    
    # Soft-max weights
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()

    grad = np.zeros(N)
    center = N - 1  # index of shift 0 in 'full' correlation
    
    for shift in range(-N+1, N):
        wi = w[center + shift]
        if wi < 1e-12:
            continue
        if shift == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif shift > 0:

            # dC/dh[j] = (1-h[j+shift]) for j< N-shift, minus h[j-shift] for j>=shift

            grad[:n_valid] += wi * (1.0 - h[shift:shift+n_valid])
            grad[shift:] -= wi * h[:n_valid]

    return grad, true_c5
