import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descend(h, n, n_sweeps=30, grid_size=200):
    h = h.copy()
    corr_len = 2 * n - 1
    m_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    x_grid = np.linspace(0.0, 1.0, grid_size)
    
    for sweep in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')

            left[valid1] = 1.0 - h[idx1[valid1]]
            
            idx2 = j + m_vals

            b = corr - a * h[j]

    mx = np.max(corr)
    s = t * (corr - mx)
    s -= s.max()  # numerical stability
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

    c = corr[idx]
    mx = np.max(c)
    s = t * (c - mx)
    s -= s.max()
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx
