    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = np.median(corr)
    above = corr[corr >= mid]
    mx = np.max(above)
    s = t * (above - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_up(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h, left=h[0], right=h[-1]), nd)

def interp_down(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    if n % nd == 0:
        return proj(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h, left=h[0], right=h[-1]), nd)

def greedy_binarize(h, n):
    """Greedy binary assignment: place n/2 ones to minimize max correlation."""
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h)
    for i in range(n // 2):
        h_bin[sorted_idx[i]] = 1.0
    return h_bin

def lbfbs_opt(h, n, t, maxiter, use_topk, k):
    bnds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)},)
    if use_topk:
        r = minimize(obj_topk, h, args=(t, k), method='L-BFGS-B',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    else:
        r = minimize(obj_full, h, args=(t,), method='L-BFGS-B',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def coord_swap(h, n, max_passes=10):
    """Iterative coordinate swap to reduce max correlation."""
    h = h.copy()
    dx = 2.0 / n
    for _ in range(max_passes):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        # Find positions to swap: high h with low h
        high = np.where(h > 0.7)[0]
        low = np.where(h < 0.3)[0]
        for i in high:
            if len(high) * len(low) > 50000:
                break
            for j in low:
                if i == j:
                    continue
                delta = min(h[i] - 0.01, 0.01)  # small step
                # Try transfer
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
                if c_new < mx - 1e-10:
                    h = h_new
                    mx = c_new
                    improved = True
                    break
            if not improved:
                continue
        if not improved:
            break

        k_star = np.argmax(corr)

        grad = np.zeros(n)

            # corr[k] = Σ valid m: h[m] * (1-h[m+k-n+1])

        low_idx = np.where(h < 0.5)[0]

            h[j] += delta

Actually, for the 1D problem on [0,2]: if h is the indicator of a set A ⊂ [0,1] (so h(x) = 1 for x ∈ A, h(x) = 0 for x ∈ [1,2]), then ∫h = |A| = 1, so A = [0,1]. That's trivial.

More generally, h is 1 on a set of measure 1 in [0,2]. The "complement" (where h=0) also has measure 1.

For a binary h that is 1 on a set A of measure 1:
overlap at shift s = |A ∩ (A-s)^c| = |A| - |A ∩ (A-s)| = 1 - |A ∩ (A-s)|

So C₅ = max_s (1 - |A ∩ (A-s)|) = 1 - min_s |A ∩ (A-s)|

We want to MINIMIZE C₅, which means MAXIMIZE min_s |A ∩ (A-s)|.

So we want a set A of measure 1 in [0,2] that maximizes the minimum cross-correlation with its shifts. This is related to the "Sidon set" or "B_h sequence" problem!

Actually, |A ∩ (A-s)| is the auto-correlation of the indicator of A. We want to maximize the minimum auto-correlation (over all shifts s). This is the "minimum auto-correlation" problem.

For binary sequences, this is related to the Welch bound and related coding theory results. But for continuous sets in [0,2], it's a measure-theoretic problem.

The key constraint is: A ⊂ [0,2], |A| = 1.
The auto-correlation: r(s) = |(A) ∩ (A-s)| (Lebesgue measure of the intersection).
We want to maximize min_{s∈[-2,2]} r(s).

This is a continuous version of the "minimum overlap" problem. The lower bound on C₅ would be 1 - max(min r(s)).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def obj_topk(h, t, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx].copy()
    mx = np.max(c)
    s = t * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def obj_topk2(h, t, k):
    """Topk with log-sum-exp on the top k, weighted"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx].copy()
    mx = np.max(c)
    # Use a power mean instead of log-sum-exp for sharper focus
    return (mx + np.mean(c - mx) + np.var(c - mx) / (2*t)) * dx
