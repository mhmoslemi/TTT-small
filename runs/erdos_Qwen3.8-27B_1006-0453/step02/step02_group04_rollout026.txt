```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.45:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    shifted = np.clip(corr - c, -100.0, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c + T * np.log(max(s, 1e-300))) * dx

def smooth_max_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    shifted = np.clip(corr - c, -100.0, 0.0)
    weights = np.exp(shifted / T)
    s_total = np.sum(weights)
    if s_total < 1e-300:
        return np.zeros(n)
    p = weights / s_total
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        k = s + n - 1
        pk = p[k]
        if pk < 1e-14:
            continue
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_hi <= i_lo:
            continue
        i_vals = np.arange(i_lo, i_hi)
        j_vals = i_vals + s
        grad[j_vals] += pk * (1.0 - h[i_vals])
        grad[i_vals] += pk * (-h[j_vals])
    return grad * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n + 1) * (2.0 / old_n)  # periodic: add point at 2
    h_old_ext = np.concatenate([h, h[:1]])
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h_old_ext)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n + 1) * (2.0 / old_n)
    h_old_ext = np.concatenate([h, h[:1]])
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h_old_ext)
    return fix_sum(h_new, new_n)

def do_lbfgsb(h, n, T, max_iter):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_obj(x, T, dx)
    def grad(x):
        return smooth_max_grad(x, T, dx)
    try:
        res = opt.minimize(
            obj, h.copy(), jac=grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: sum_con(x, n),
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h
```

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.45:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    shifted = np.clip(corr - c, -120.0, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c + T * np.log(max(s, 1e-300))) * dx

def smooth_max_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    shifted = np.clip(corr - c, -120.0, 0.0)
    weights = np.exp(shifted / T)
    s_total = np.sum(weights)
    if s_total < 1e-300:
        return np.zeros(n)
    p = weights / s_total
    grad = np.zeros(n)
    for s_val in range(-(n - 1), n):
        k = s_val + n - 1
        pk = p[k]
        if pk < 1e-14:
            continue
        i_lo = max(0, -s_val)
        i_hi = min(n, n - s_val)
        if i_hi <= i_lo:
            continue
        i_vals = np.arange(i_lo, i_hi)
        j_vals = i_vals + s_val
        grad[j_vals] += pk * (1.0 - h[i_vals])
        grad[i_vals] += pk * (-h[j_vals])
    return grad * dx

class SoftMaxProb:
    def __init__(self, T, dx):
        self.T = T
        self.dx = dx
    def fun(self, x):
        return smooth_max_obj(x, self.T, self.dx)
    def jac(self, x):
        return smooth_max_grad(x, self.T, self.dx)

class ActiveShiftProb:
    def __init__(self, top_k, n, T, dx):
        self.top_k = top_k
        self.n = n
        self.T = T
        self.dx = dx
    def fun(self, x):
        c = np.correlate(x, 1.0 - x, mode='full')
        tv = c[self.top_k]
        cm = np.max(tv)
        sh = np.clip(tv - cm, -120.0, 0.0)
        s = np.sum(np.exp(sh / self.T))
        return (cm + self.T * np.log(max(s, 1e-300))) * self.dx
    def jac(self, x):
        c = np.correlate(x, 1.0 - x, mode='full')
        tv = c[self.top_k]
        cm = np.max(tv)
        sh = np.clip(tv - cm, -120.0, 0.0)
        weights = np.exp(sh / self.T)
        s = np.sum(weights)
        if s < 1e-300:
            return np.zeros(self.n)
        p = weights / s
        grad = np.zeros(self.n)
        n = self.n
        for idx, ki in enumerate(self.top_k):
            pk = p[idx]
            if pk < 1e-14:
                continue
            s_val = ki - (n - 1)
            i_lo = max(0, -s_val)
            i_hi = min(n, n - s_val)
            if i_hi <= i_lo:
                continue
            i_vals = np.arange(i_lo, i_hi)
            j_vals = i_vals + s_val
            grad[j_vals] += pk * (1.0 - x[i_vals])
            grad[i_vals] += pk * (-x[j_vals])
        return grad * self.dx

def sum_con_fun(x, n):
    return np.sum(x) - n * 0.5

def sum_con_jac(x, n):
    return np.ones(n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n + 1) * (2.0 / old_n)
    h_ext = np.concatenate([h, h[:1]])
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h_ext)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n + 1) * (2.0 / old_n)
    h_ext = np.concatenate([h, h[:1]])
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h_ext)
    return fix_sum(h_new, new_n)

def do_slsqp_analytic(h, n, T, max_iter):
    dx = 2.0 / n
    prob = SoftMaxProb(T, dx)
    try:
        res = opt.minimize(prob.fun, h.copy(), jac=prob.jac,
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sum_con_fun,
                                         'jac': sum_con_jac, 'args': (n,)}],
                           options={'maxiter': max_iter, 'ftol': 1e-17})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_active_refine(h, n, T, max_iter, n_top=4):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-n_top:]
    prob = ActiveShiftProb(top_k, n, T, dx)
    try:
        res = opt.minimize(prob.fun, h.copy(), jac=prob.jac,
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sum_con_fun,
                                         'jac': sum_con_jac, 'args': (n,)}],
                           options={'maxiter': max_iter, 'ftol': 1e-16})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_projected_grad(h, n, T, dx, n_iter=200, lr=0.01):
    """Projected gradient descent with adaptive step"""
    x = h.copy()
    for it in range(n_iter):
        g = smooth_max_grad(x, T, dx)
        x_new = x - lr * g
        x_new = fix_sum(x_new, n)
        obj_new = smooth_max_obj(x_new, T, dx)
        obj_old = smooth_max_obj(x, T, dx)
        if obj_new < obj_old:
            x = x_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return x
