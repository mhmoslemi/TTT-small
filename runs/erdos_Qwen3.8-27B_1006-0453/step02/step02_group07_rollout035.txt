    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    c = np.max(corr)
    if T < 1e-12:
        val = c * dx
        # subgradient: just use argmax
        kmax = np.argmax(corr)
        d = kmax - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            valid_first = np.arange(n) < (n - d)
            grad[valid_first] += (1 - h[d:])[:n-d]  # (1-h[j+d]) for j=0..n-1-d
            valid_second = np.arange(n) >= d
            grad[valid_second] -= h[:n-d]  # -h[j-d] for j=d..n-1
        else:
            dp = -d
            valid_first = np.arange(n) >= dp
            grad[valid_first] += (1 - h[:n-dp])  # (1-h[j-dp]) for j=dp..n-1
            valid_second = np.arange(n) < (n - dp)
            grad[valid_second] -= h[dp:]  # -h[j+dp] for j=0..n-1-dp
        return val, grad * dx
    # Softmax weights
    exps = np.exp((corr - c) / T)
    w = exps / np.sum(exps)

        if wk < 1e-12:
