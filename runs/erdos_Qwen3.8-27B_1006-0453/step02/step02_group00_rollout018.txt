    val = m + (ms + np.log(np.sum(np.exp(shifted - ms)))) / T
    return val * dx

    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

            h_new[j] += delta
            count += 1
