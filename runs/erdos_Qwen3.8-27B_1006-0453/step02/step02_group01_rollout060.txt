import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, n):
    """Ensure h is binary with exactly n//2 ones."""
    target = n // 2
    current = int(np.sum(h >= 0.5))
    if current == target:
        return (h >= 0.5).astype(float)
    b = (h >= 0.5).astype(float)
    if current > target:
        ones_idx = np.where(b == 1)[0]
        n_remove = current - target
        # Remove ones with smallest original h values
        order = np.argsort(h[ones_idx])[:n_remove]
        b[ones_idx[order]] = 0
    else:
        zeros_idx = np.where(b == 0)[0]
        n_add = target - current
        order = np.argsort(h[zeros_idx])[::-1][:n_add]
        b[zeros_idx[order]] = 1
    return b

def sa_binary(h, n_iters, T0, T1, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    cur = h.copy()
    best = h.copy()
    cur_c5 = compute_c5(cur)
    best_c5 = cur_c5
    
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(n_iters - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j] or i == j:
            # retry
            j = rng.randint(n)
            if h[i] == h[j] or i == j:
                continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = compute_c5(cur)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]  # revert
    
    return best, best_c5

def hill_climb_binary(h, max_iters=5000):
    """Local search: try all single swaps, take the best improving move."""
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    improved = True
    while improved and max_iters > 0:
        improved = False
        best_delta = 0
        best_i, best_j = -1, -1
        # Try swaps: for efficiency, sample subset
        n_pairs = min(len(ones) * len(zeros), 500)
        for _ in range(n_pairs):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            cur[i], cur[j] = cur[j], cur[i]
            new_c5 = compute_c5(cur)
            delta = cur_c5 - new_c5
            cur[i], cur[j] = cur[j], cur[i]
            if delta > best_delta:
                best_delta = delta
                best_i, best_j = i, j
        
        if best_i >= 0:
            cur[best_i], cur[best_j] = cur[best_j], cur[best_i]
            cur_c5 -= best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
            improved = True
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            max_iters -= 1
    
    return best, best_c5

    n = len(h)

    target = n // 2
    b = (h >= 0.5).astype(float)

    if cur == target:

    if cur > target:

        b[idx[order]] = 0
    else:

        order = np.argsort(h[idx])[::-1][:target - cur]
        b[idx[order]] = 1
    return b

            cur[jj] = 1
            ones.append(jj)
            zeros.append(ii)
        cur, cur_mc = hill_climb(cur, rng, max_rounds=1500, time_limit=0.4)
        if cur_mc < best_mc:
            best_mc = cur_mc
            best = cur.copy()
    return best, best_mc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return max(budget_s - (time.time() - t0), 0.0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float).copy()
    else:
        init = np.zeros(90)
        init[:45] = 1.0

    best_n = len(init)
    best_h = project_binary(init, best_n)
    best_mc = max_corr(best_h)
    best_c5 = best_mc * 2.0 / best_n

    # Benchmark
    bn = 500
    bs = time.time()
    for _ in range(bn):
        max_corr(best_h)
    bt = time.time() - bs
    ips = bn / max(bt, 1e-9)

    # Phase 1: Long SA (35% budget)
    t1 = tl() * 0.35
    h1, mc1 = sa_run(best_h, int(ips * t1), 2.0, 0.005, rng, t1)
    if mc1 < best_mc:
        best_mc = mc1
        best_h = h1

    # Phase 2: Hill climb
    if tl() > budget_s * 0.08:
        h2, mc2 = hill_climb(best_h, rng, 100000, tl() * 0.05)
        if mc2 < best_mc:
            best_mc = mc2
            best_h = h2

    # Phase 3: ILS (25% budget)
    if tl() > budget_s * 0.15:
        t3 = tl() * 0.25
        h3, mc3 = ils_run(best_h, rng, 100000, t3)
        if mc3 < best_mc:
            best_mc = mc3
            best_h = h3

    # Phase 4: SA + HC polish (15%)
    if tl() > budget_s * 0.12:
        t4 = tl() * 0.15
        h4, mc4 = sa_run(best_h, int(ips * t4 * 0.6), 0.5, 0.001, rng, t4 * 0.6)
        if mc4 < best_mc:
            best_mc = mc4
            best_h = h4
        h5, mc5 = hill_climb(best_h, rng, 50000, t4 * 0.4)
        if mc5 < best_mc:
            best_mc = mc5
            best_h = h5

    # Phase 5: Random restarts (15%)
    if tl() > budget_s * 0.08:
        t5_total = tl() * 0.15
        n_r = max(3, min(20, int(t5_total / 4.0)))
        t_per = t5_total / n_r
        for r in range(n_r):
            if tl() < t_per * 0.5:
                break
            rh = (rng.rand(best_n) < 0.5).astype(float)
            rh = project_binary(rh, best_n)
            hr, mcr = sa_run(rh, int(ips * t_per * 0.5), 1.5, 0.01, rng, t_per * 0.5)
            hr2, mcr2 = hill_climb(hr, rng, 3000, t_per * 0.3)
            if mcr2 < best_mc:
                best_mc = mcr2
                best_h = hr2

    # Phase 6: Other resolutions (5%)
    if tl() > budget_s * 0.03:
        t6 = tl() * 0.05
        for n_try in [80, 100, 120]:
            if tl() < t6 * 0.2:
                break
            xo = np.arange(best_n) * 2.0 / best_n
            xn = np.arange(n_try) * 2.0 / n_try
            hi = np.interp(xn, xo, best_h)
            hb = project_binary(hi, n_try)
            ht, mct = sa_run(hb, int(ips * t6 * 0.2 * 90.0 / n_try), 1.0, 0.01, rng, t6 * 0.2)
            ht2, mct2 = hill_climb(ht, rng, 2000, t6 * 0.08)
            c5_try = mct2 * 2.0 / n_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_mc = mct2
                best_h = ht2
                best_n = n_try

    # Final hill climb
    if tl() > 2.0:
        hf, mcf = hill_climb(best_h, rng, 5000, tl() * 0.7)
        if mcf < best_mc:
            best_mc = mcf
            best_h = hf

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
