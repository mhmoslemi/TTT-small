def top_k_subgradient(h, n, n_iter=10000, top_k=3, base_lr=0.005):
    """Subgradient descent on max of top-k correlations"""
    h = h.copy().astype(float)
    lr = base_lr

    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    vals = corr[idx]
    m = np.max(vals)
    if T < 1e-15:
        return m * dx
    shifted = np.clip(vals - m, -50.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (m + T * np.log(max(s, 1e-300))) * dx

def shift_grad(h, s):
    n = len(h)
    g = 1.0 - h
    grad = np.zeros(n)
    if s >= 0:
        grad[:n-s] += g[s:]
        grad[s:] -= h[:n-s]
    else:
        grad[-s:] += g[:n+s]
        grad[:n+s] -= h[-s:]
    return grad

def top_k_subgrad(h, n, n_iter, top_k, base_lr, momentum):
    h = h.copy().astype(float)
    lr = base_lr
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, 'full')
        cur = float(np.max(corr) * 2.0 / n)
        if cur < best_c5 - 1e-13:
            best_c5 = cur
            best_h = h.copy()
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for m in top_idx:
            s = m - (n - 1)
            grad += shift_grad(h, s)
        grad /= top_k
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c_new = compute_c5(h_new)
        if c_new <= cur + 1e-12:
            h = h_new
            lr = min(lr * 1.003, base_lr * 3.0)
        else:
            lr *= 0.95
            if lr < base_lr * 0.1:
                lr = base_lr
                v *= 0.5
    return best_h

    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    deltas = np.array([-0.05, -0.03, -0.01, 0.01, 0.03, 0.05])
    rng = np.random.RandomState(42)
    for rnd in range(n_rounds):
        idx = rng.choice(n, size=min(n_per_round, n), replace=True)
        for j in idx:
            orig = h[j]
            best_d = 0.0
            best_c = best_c5
            for d in deltas:
                val = orig + d
                if val < 0.0 or val > 1.0:
                    continue
                h[j] = val
                h_f = fix_sum(h, n)
                c = compute_c5(h_f)
                if c < best_c:
                    best_c = c
                    best_d = d
            if best_d != 0.0:
                h[j] = orig + best_d
                h = fix_sum(h, n)
                best_c5 = best_c
            else:
                h[j] = orig
    return h

def do_slsqp(h, n, T, top_k, max_iter):
    try:
        res = opt.minimize(
            soft_max_obj, h.copy(), args=(T, top_k),
            method='SLSQP',

        )
        if hasattr(res, 'x') and res.x is not None:

    except Exception:
        pass
    return h

- For full correlation: corr[m] = sum_i h[i] * g[i + m - (n-1)]
