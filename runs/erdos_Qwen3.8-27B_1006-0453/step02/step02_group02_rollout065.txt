def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_topk(h, k):

    corr = np.correlate(h, 1.0 - h, mode='full')
    if k >= len(corr):
        return np.arange(len(corr))
    return np.argpartition(corr, -k)[-k:].copy()

def corr_grad(lag, h, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        ln = -lag
        g[ln:] += (1.0 - h[:-ln])
        g[:-ln] -= h[ln:]
    return g

def obj_func(h, t, topk_idx, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    fv = corr[topk_idx] * dx
    fm = np.max(fv)
    sc = t * (fv - fm)
    obj = fm + np.log(np.sum(np.exp(sc))) / t
    sv = np.sum(h) - n / 2.0
    obj += lam * sv * sv
    return obj

def grad_func(h, t, topk_idx, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    fv = corr[topk_idx] * dx
    fm = np.max(fv)
    sc = t * (fv - fm)
    esc = np.exp(sc)
    p = esc / np.sum(esc)
    g = np.zeros(n)
    for i in range(len(topk_idx)):
        lag = topk_idx[i] - (n - 1)
        g += p[i] * dx * corr_grad(lag, h, n)
    sv = np.sum(h) - n / 2.0
    g += 2.0 * lam * sv
    return g

def do_opt(h, n, t_val, k, lam, maxiter, bnds):
    topk = get_topk(h, k)
    r = minimize(obj_func, h, args=(t_val, topk, lam, n),
                 jac=grad_func, method='L-BFGS-B',
                 bounds=bnds,
                 options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return proj(np.clip(r.x, 0.0, 1.0), n)
