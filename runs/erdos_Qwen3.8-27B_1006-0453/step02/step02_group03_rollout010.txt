def lse_lbfgs(h0, beta, ts, lam, maxiter=2000):
    n = len(h0)
    def obj(x):
        h = np.clip(x, 0, 1)
        cr = np.correlate(h, 1.0 - h, mode='full')
        bc = beta * cr
        bc_max = bc.max()
        lse = (bc_max + np.log(np.sum(np.exp(bc - bc_max)))) / beta
        penalty = 0.5 * lam * (h.sum() - ts)**2
        return lse + penalty
    
    def grad(x):
        h = np.clip(x, 0, 1)
        cr = np.correlate(h, 1.0 - h, mode='full')
        bc = beta * cr
        bc_max = bc.max()
        w = np.exp(bc - bc_max)
        w /= w.sum()
        o = n - 1
        g = np.zeros(n)
        for j in range(n):
            k_lo = max(j-(n-1), -(n-1))
            k_hi = min(j, n-1)
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi+1)
                g[j] += np.dot(w[o+ks], 1.0 - h[j-ks])
            k_lo2 = max(-j, -(n-1))
            k_hi2 = min(n-1-j, n-1)
            if k_lo2 <= k_hi2:
                ks = np.arange(k_lo2, k_hi2+1)
                g[j] -= np.dot(w[o+ks], h[j+ks])
        g += lam * (h.sum() - ts)
        return g
    
    # Initialize at the projection of h0
    x0 = h0.copy()
    result = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                      bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-14})
    h = np.clip(result.x, 0, 1)
    h = proj(h, ts)
    return h
