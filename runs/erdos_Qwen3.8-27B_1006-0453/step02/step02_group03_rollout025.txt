```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def cr_weighted_grad(h, w):
    N = len(h)
    g = np.zeros(N)
    for m in range(2 * N - 1):
        wm = w[m]
        if abs(wm) < 1e-15:
            continue
        shift = m - (N - 1)
        lo1 = max(0, -shift)
        hi1 = min(N - 1, N - 1 - shift)
        if lo1 <= hi1:
            g[lo1:hi1+1] += wm * (1.0 - h[lo1+shift:hi1+1+shift])
        lo2 = max(0, shift)
        hi2 = min(N - 1, N + shift - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= wm * h[lo2-shift:hi2+1-shift]
    return g

def softmax_obj_and_grad(x, W):
    """Compute smooth max of correlate(x, 1-x) using softmax."""
    h = x  # already projected
    cr = np.correlate(h, 1.0 - h, mode='full')
    beta = W['beta']
    lam = W['lam']
    ts = W['ts']
    
    cr_max = cr.max()
    ex = np.exp(beta * (cr - cr_max))
    s = ex.sum()
    if s < 1e-30:
        w = np.ones(len(cr)) / len(cr)
        obj = cr_max
    else:
        w = ex / s
        obj = cr_max + np.log(s) / beta
    
    # Gradient of smooth max
    g = cr_weighted_grad(h, w)
    
    # Penalty gradient for sum constraint
    penalty = lam * (h.sum() - ts) ** 2
    g_pen = 2 * lam * (h.sum() - ts)
    
    return obj + penalty, g + g_pen

def lbfgsb_phase(h0, ts, betas, iters_per_beta, dl):
    """Run L-BFGS-B with increasing beta."""
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    t_start = time.time()
    
    for beta in betas:
        if time.time() > dl:
            break
        W = {'beta': beta, 'lam': 50.0, 'ts': ts}
        
        def obj(x):
            h_clip = np.clip(x, 0, 1)
            return _soft_obj_val(h_clip, ts, beta, 50.0)
        
        def grad(x):
            h_clip = np.clip(x, 0, 1)
            _, g = _soft_obj_and_grad(h_clip, ts, beta, 50.0)
            # Only return gradient where not at boundary
            g[x <= 0] = max(g[x <= 0].min(), 0) if (x <= 0).any() else 0
            g[x >= 1] = min(g[x >= 1].max(), 0) if (x >= 1).any() else 0
            return g
        
        # Actually, let's use bounds with L-BFGS-B instead of clipping in gradient
        def obj_b(x):
            return _soft_obj_val(x, ts, beta, 50.0)
        
        def grad_b(x):
            return _soft_grad(x, ts, beta, 50.0)
        
        bnds = [(0, 1)] * N
        t0 = time.time()
        try:
            r = minimize(obj_b, h, jac=grad_b, method='L-BFGS-B',
                        bounds=bnds, options={'maxiter': iters_per_beta, 'gtol': 1e-8})
            h = proj(r.x, ts)
        except Exception:
            pass
        
    return h, compute_c5(h, dx)

def _soft_obj_val(h, ts, beta, lam):
    cr = np.correlate(h, 1.0 - h, mode='full')
    cr_max = cr.max()
    s = np.exp(beta * (cr - cr_max)).sum()
    smooth = cr_max + np.log(s) / beta if s > 1e-30 else cr_max
    return smooth + lam * (h.sum() - ts) ** 2

def _soft_grad(h, ts, beta, lam):
    cr = np.correlate(h, 1.0 - h, mode='full')
    cr_max = cr.max()
    ex = np.exp(beta * (cr - cr_max))
    s = ex.sum()
    w = ex / s if s > 1e-30 else np.ones(len(cr)) / len(cr)
    g = cr_weighted_grad(h, w)
    g += 2 * lam * (h.sum() - ts)
    return g

def slsqp_refine(h0, dx, ts, maxiter=200):
    """SLSQP with t-variable for final refinement."""
    N = len(h0)
    
    def s_obj(x):
        return x[N]
    
    def s_ineq(x):
        return x[N] - np.correlate(x[:N], 1.0 - x[:N], mode='full')
    
    def s_eq(x):
        return x[:N].sum() - ts
    
    cr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = cr0.max() + 0.005
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0, 1)] * N + [(0, N)]
    cons = [{'type': 'ineq', 'fun': s_ineq}, {'type': 'eq', 'fun': s_eq}]
    
    try:
        r = minimize(s_obj, x0, method='SLSQP', bounds=bnds,
                    constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if r.success or r.x is not None:
            hr = proj(r.x[:N], ts)
            if compute_c5(hr, dx) <= compute_c5(h0, dx) + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def coord_descent(h0, dx, ts, max_rounds=20, dl=None, rng=None):
    """Coordinate descent with multiple step sizes."""
    N = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    
    for r in range(max_rounds):
        if dl and time.time() > dl:
            break
        improved = False
        for j in rng.permutation(N):
            sv = h[j]
            best_v, best_c = sv, bc
            for d in (0.02, -0.02, 0.04, -0.04, 0.08, -0.08, 0.01, -0.01):
                nv = sv + d
                if nv < -1e-10 or nv > 1 + 1e-10:
                    continue
                h[j] = nv
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < best_c:
                    best_c, best_v = c, nv
            if best_v != sv:
                h[j] = best_v
                h = proj(h, ts)
                bc = best_c
                improved = True
            else:
                h[j] = sv
        if not improved:
            break
    return h, bc

def make_starts(n, ts, rng, ref=None):
    """Generate diverse starting points."""
    out = []
    # Uniform
    out.append(np.full(n, ts / n))
    # Block patterns
    for p in [n//3, n//4, n//5, n//7]:
        p = max(2, p)
        h = np.zeros(n)
        h[::(2*p)][:min(n//(2*p), n)] = 1.0
        # Better: alternating blocks
        h = np.array([1.0 if (i % (2*p)) < p else 0.0 for i in range(n)])
        out.append(proj(h, ts))
    # Sine waves
    x = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7]:
        for amp in [0.3, 0.5]:
            h = 0.5 + amp * np.sin(2*np.pi*freq*x)
            out.append(proj(h, ts))
    # Random
    for _ in range(3):
        out.append(proj(rng.uniform(0, 1, n), ts))
    # Interpolate reference
    if ref is not None:
        if len(ref) == n:
            out.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                          np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.insert(0, proj(hi, ts))
    return out
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(120):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def cr_weighted_grad(h, w):
    N = len(h)
    g = np.zeros(N)
    for m in range(2 * N - 1):
        wm = w[m]
        if abs(wm) < 1e-15:
            continue
        shift = m - (N - 1)
        lo1 = max(0, -shift)
        hi1 = min(N - 1, N - 1 - shift)
        if lo1 <= hi1:
            g[lo1:hi1+1] += wm * (1.0 - h[lo1+shift:hi1+1+shift])
        lo2 = max(0, shift)
        hi2 = min(N - 1, N + shift - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= wm * h[lo2-shift:hi2+1-shift]
    return g

def _soft_obj_val(h, ts, beta, lam):
    cr = np.correlate(h, 1.0 - h, mode='full')
    cr_max = cr.max()
    s = np.exp(beta * (cr - cr_max)).sum()
    smooth = cr_max + np.log(s) / beta if s > 1e-30 else cr_max
    return smooth + lam * (h.sum() - ts) ** 2

def _soft_grad(h, ts, beta, lam):
    cr = np.correlate(h, 1.0 - h, mode='full')
    cr_max = cr.max()
    ex = np.exp(beta * (cr - cr_max))
    s = ex.sum()
    w = ex / s if s > 1e-30 else np.ones(len(cr)) / len(cr)
    g = cr_weighted_grad(h, w)
    g += 2.0 * lam * (h.sum() - ts)
    return g

def _obj(x):
    return _soft_obj_val(x, _P['ts'], _P['beta'], _P['lam'])

def _grad(x):
    return _soft_grad(x, _P['ts'], _P['beta'], _P['lam'])

def _sobj(x):
    return x[_P['n']]

def _sineq(x):
    return x[_P['n']] - np.correlate(x[:_P['n']], 1.0 - x[:_P['n']], mode='full')

def _seq(x):
    return x[:_P['n']].sum() - _P['ts']

def slsqp_refine(h0, dx, ts, maxiter=200):
    N = len(h0)
    _P['n'], _P['ts'] = N, ts
    cr0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = cr0.max() + 0.005
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0, 1)] * N + [(0, float(N))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _seq}]
    c0 = compute_c5(h0, dx)
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                    constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        if r.x is not None:
            hr = proj(r.x[:N], ts)
            if compute_c5(hr, dx) <= c0 + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def lbfgsb_phase(h0, ts, dl, max_iter_total=400):
    N = len(h0)
    h = h0.copy()
    c = compute_c5(h, 2.0 / N)
    betas = [8, 20, 50, 120]
    its = [120, 100, 80, 60]
    bnds = [(0, 1)] * N

        _P['ts'], _P['beta'], _P['lam'] = ts, float(beta), 80.0
        try:

                        bounds=bnds, options={'maxiter': its[i], 'gtol': 1e-10, 'maxfun': its[i]*2})
            h2 = proj(r.x, ts)
            c2 = compute_c5(h2, 2.0 / N)
            if c2 < c - 1e-12:
                h, c = h2, c2
        except Exception:
            pass
    return h, c

def grad_descent(h0, ts, dl, max_iters=3000):
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    c = compute_c5(h, dx)
    stall = 0

            break

        cr = np.correlate(h, 1.0 - h, mode='full')

        s = w.sum()
        w = w / s if s > 1e-30 else np.ones(len(cr)) / len(cr)

        lr = 0.04 / (1.0 + 0.0008 * it)
        improved = False

def coord_descent(h0, ts, max_rounds=20, dl=None, rng=None):
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    bc = compute_c5(h, dx)
```
