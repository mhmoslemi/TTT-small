        k = int(np.sum(m))
        if k > 0:
            h_p[m] += d / k
            np.clip(h_p, 0.0, 1.0, out=h_p)
    return h_p

def pgd(h, T, max_steps, ftol=1e-14):
    """Projected gradient descent with backtracking line search"""
    n = len(h)
    best_val = lse_obj_and_weights(h, T)[0]
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        val, corr, w = lse_obj_and_weights(h, T)
        if val <= best_val * (1 - 1e-15):
            best_val = val
        g = gradient_lse(h, w)
        # Backtracking line search
        alpha = 1.0
        for _ in range(30):
            h_new = project(h - alpha * g, n)
            val_new, _, _ = lse_obj_and_weights(h_new, T)
            if val_new <= val - 1e-4 * alpha * np.sum(g**2):
                break
            alpha *= 0.5
        else:
            break
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if abs(val - best_val) < ftol:
            break
    return h, best_c5

def coord_polish(h, n, n_pass=2):
    for _ in range(n_pass):
        improved = True
        while improved:
            improved = False
            for j in range(n):
                if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                    continue
                c0 = compute_c5(h)

                if best_delta != 0:
                    h[j] = np.clip(h[j] + best_delta, 0, 1)
                    h = fix_sum(h, n)
                    improved = True
    return h

import numpy as np

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-10:
        m = (h > 1e-7) & (h < 1.0 - 1e-7)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d2 / k
            np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_obj_weights(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        idx = np.argmax(corr)
        w = np.zeros_like(corr)
        w[idx] = 1.0
        return c_max * dx, corr, w
    shifted = np.clip((corr - c_max) / T, -500, 0)
    es = np.exp(shifted)
    w = es / es.sum()
    return (c_max + T * np.log(es.sum())) * dx, corr, w

def gradient_lse(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for l in range(2 * n - 1):
        wl = w[l]
        if wl < 1e-12:
            continue
        m = l - (n - 1)
        if m >= 0:
            j1 = np.arange(m, n)
            g[j1] += wl * (1.0 - h[j1 - m])
            j2 = np.arange(0, n - m)
            g[j2] -= wl * h[j2 + m]
        else:
            p = -m
            j1 = np.arange(0, n - p)
            g[j1] += wl * (1.0 - h[j1 + p])
            j2 = np.arange(p, n)
            g[j2] -= wl * h[j2 - p]
    return dx * g
