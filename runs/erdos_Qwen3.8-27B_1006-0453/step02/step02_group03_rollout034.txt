        return x[n]

    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full')
    m = cr.max()
    w = np.exp(beta * (cr - m))

    g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sgrad2(x):
    n = _sp['n']
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    n = _sp['n']
    return float(np.sum(x[:n]) - _sp['ts'])

def compute_c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1.0 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    np.clip(h, 0, 1, out=h)
    return h

def lse_val(h, beta, mu, ts):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full')
    mx = cr.max()
    lse = (mx + np.log(np.sum(np.exp(beta * (cr - mx)))) / beta) * dx
    return lse + mu * (h.sum() - ts) ** 2

def lse_grad(h, beta, mu, ts):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full')
    mx = cr.max()
    e = np.exp(beta * (cr - mx))
    w = e / e.sum()
    g = np.zeros(n)
    og = n - 1
    for j in range(n):
        k1lo = max(-(n-1), j-(n-1))
        k1hi = min(n-1, j)
        if k1lo <= k1hi:
            ks = np.arange(k1lo, k1hi+1)
            g[j] += np.dot(w[og+ks], 1.0 - h[j-ks])
        k2lo = max(-(n-1), -j)
        k2hi = min(n-1, n-1-j)
        if k2lo <= k2hi:
            ks = np.arange(k2lo, k2hi+1)
            g[j] -= np.dot(w[og+ks], h[j+ks])
    g *= dx
    g += 2.0 * mu * (h.sum() - ts)
    return g

def slsqp_max(h0, ts, mit=300):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    dx = 2.0 / n
    t0 = float(np.correlate(h0, 1.0 - h0, mode='full').max()) / dx + 0.005
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0, 1)] * n + [(0, float(n) + 1)]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', jac=_sgrad2, bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-12, 'iprint': 0})
        if r.x is not None:
            return proj(r.x[:n], ts)
    except Exception:
        pass
    return h0.copy()

def lse_lbfgsb(h0, ts, beta, mu, deadline):
    n = len(h0)
    x0 = h0.copy()
    bnds = [(0.0, 1.0)] * n

    def fun(xx):
        return lse_val(xx, beta, mu, ts)

    def fp(xx):
        return lse_grad(xx, beta, mu, ts)

    try:
        res = fmin_l_bfgs_b(fun, x0, fprime=fp, pgtol=1e-10, maxiter=300,
                            maxfun=2000, iprint=0, bounds=bnds)
        return proj(res[0], ts)
    except Exception:
        return h0.copy()

def nesterov_desc(h0, ts, iters, deadline, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = compute_c5(h0)
    v = np.zeros(n)
    mom = 0.8
    stale = 0
    for i in range(iters):
        if time.time() > deadline:
            break
        beta = 15.0 + 70.0 * min(1.0, i / max(1, 0.5 * iters))
        lr = 0.025 / (1.0 + 0.0015 * i)
        g = lse_grad(h, beta, 200.0, ts)
        v = mom * v - lr * g
        h_new = proj(h + v, ts)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-14:
            best_h, best_c, stale = h_new.copy(), c_new, 0
            h = h_new
        else:
            stale += 1
            h = h_new
            v *= 0.4
            if stale > 180:
                h = best_h.copy()
                v = np.zeros(n)
                stale = 0
    return best_h, best_c

def coord_desc(h0, ts, rounds, deadline, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h)
    steps_lo = (0.002, -0.002, 0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.04, -0.04)
    for r in range(rounds):
        if time.time() > deadline:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            ok = False
            for d in steps_lo:
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                h = proj(h, ts)
                c = compute_c5(h)
                if c < bc - 1e-15:
                    bc = c
                    ok = True
                    imp = True
                    break
            if not ok:
                h[j] = sv
        if not imp:
            break
    return h, bc

def full_optimize(h0, deadline, rng):
    n = len(h0)
    ts = n / 2.0
    h = proj(h0, ts)
    c = compute_c5(h)
    td = time.time()

    # Phase 1: L-BFGS-B with continuation
    for beta in (5, 15, 40, 100, 250):
        if time.time() > deadline - 30:
            break
        h2 = lse_lbfgsb(h, ts, beta, 80.0, deadline - 5)
        c2 = compute_c5(h2)
        if c2 < c:
            h, c = h2, c2

    # Phase 2: SLSQP on exact max
    if time.time() < deadline - 40:
        h3 = slsqp_max(h, ts, 350)
        c3 = compute_c5(h3)
        if c3 < c:
            h, c = h3, c3

    # Phase 3: Nesterov gradient descent
    if time.time() < deadline - 25:
        h4, c4 = nesterov_desc(h, ts, 1500, deadline - 8, rng)
        if c4 < c:
            h, c = h4, c4

    # Phase 4: SLSQP again
    if time.time() < deadline - 15:
        h5 = slsqp_max(h, ts, 250)
        c5 = compute_c5(h5)
        if c5 < c:
            h, c = h5, c5

    # Phase 5: Fine coordinate descent
    if time.time() < deadline - 8:
        h6, c6 = coord_desc(h, ts, 60, deadline - 3, rng)
        if c6 < c:
            h, c = h6, c6

    # Phase 6: Final SLSQP
    if time.time() < deadline - 2:
        h7 = slsqp_max(h, ts, 150)
        c7 = compute_c5(h7)
        if c7 < c:
            h, c = h7, c7

    return h, c

def make_starts(n, ts, rng, ref):
    outs = []
    if ref is not None:
        if len(ref) == n:
            outs.append(proj(ref.copy(), ts))
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            outs.append(proj(hi, ts))
    outs.append(proj(np.full(n, 0.5), ts))
    for m in (1, 2, 3, 5, 7):
        x = np.arange(n) / n
        s = 0.5 + 0.2 * np.sin(2 * np.pi * m * x)
        outs.append(proj(s, ts))
    for _ in range(2):
        outs.append(proj(rng.uniform(0, 1, n), ts))
    p = max(2, n // 4)
    outs.append(proj(np.array([1.0 if (i % (2*p)) < p else 0.0 for i in range(n)]), ts))
    return outs
