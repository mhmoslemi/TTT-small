import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    """Compute gradient of soft-max of correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    # Vectorized gradient computation
    for k in range(len(w)):
        wi = w[k]
        if wi < 1e-6:
            continue
        L = k - center  # shift
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            # corr = sum_{i=L}^{N-1} h[i]*(1-h[i-L])
            # d/dh[n] = (1-h[n-L]) for n>=L, minus h[n+L] for n<=N-1-L
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, float(np.max(corr)) * (2.0/N)

def optimize_phase(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, total_iters):
    """Adam optimization with time-based tau and lr schedules."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    # Estimate iterations per second
    while time.time() < deadline:
        frac = min(it / max(total_iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        
        grad, c5_val = softmax_grad(h, N, tau)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def local_search(h, N, target, dx, n_trials=200, deadline=None):
    """Try random coordinate perturbations."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(0)
    for i in range(n_trials):
        if deadline and time.time() > deadline:

        h_trial = best_h.copy()
        # Perturb a few random coordinates

        h_trial = project_sum(h_trial, target)
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < best_c5:
            best_c5 = c5_trial
