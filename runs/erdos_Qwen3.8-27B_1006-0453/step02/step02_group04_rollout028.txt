def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:

            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]

                h[i] += add
                diff -= add

            idx = np.argsort(-h)

                rem = min(diff, h[i])
                h[i] -= rem

    n = len(a)

    prod = fa * np.conj(fb)

    # The full correlation has length n+m-1

    return np.concatenate([corr[size-n+1:], corr[:m+n-1-(size-n+1)]])
