```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n=None):
    if n is None:
        n = len(h)
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
    else:
        if diff > 0:
            idx = np.argsort(h)
            rem = diff
            for i in idx:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
                if rem < 1e-13:
                    break
        else:
            rem = -diff
            idx = np.argsort(-h)
            for i in idx:
                sub = min(rem, h[i])
                h[i] -= sub
                rem -= sub
                if rem < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h, n):
    """Compute subgradient of max_k corr_k * dx at h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cmax = np.max(corr)
    # Active set: lags where corr is within tolerance of max
    tol = 1e-10
    active = np.where(corr >= cmax - tol)[0]
    n_active = len(active)
    
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            else:
                grad[lag:] += (1.0 - h[:n - lag])
                grad[:n - lag] -= h[lag:]
        else:
            l = -lag
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] -= h[:n - l]
    
    grad /= n_active
    grad *= dx
    return grad, cmax * dx, len(active)

def subgradient_descent(h, n, n_iter=10000, alpha0=0.1):
    """Run subgradient descent"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iter):
        grad, c5_val, n_act = compute_subgrad(h, n)
        alpha = alpha0 / (1.0 + 0.01 * it)

        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        if np.max(np.abs(h_new - h)) < 1e-14:

        h = h_new
        if (it + 1) % 100 == 0:

            if c5_check < best_c5:

                best_h = h.copy()
    
    # Final check

    if c5_check < best_c5:

        best_h = h.copy()

def coordinate_descent(h, n, n_passes=5):

            h[i] = 0.0

            # Actually let me compute it directly:
            h[i] = 1.0

            h[i] = 0.0  # reset
            coeff = corr1 - corr0  # this is the coefficient of h[i] in each corr_k
            
            # Now minimize max_j (corr0[j] + coeff[j] * t) for t in [0,1]
            # This is a 1D convex minimization
            # Use the fact that the minimum of max of linear functions is at a breakpoint
            # Simple approach: ternary search (works for convex functions)
            lo, hi = 0.0, 1.0
            for _ in range(50):

                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            
            h[i] = t_opt
        # Fix sum after each pass
        h = fix_sum(h, n)
    
    return h
```

```python
    for p in range(n_passes):
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            coeff = np.zeros(2*n-1)
            first_mask = (lags >= i-n+1) & (lags <= i)
            second_mask = (lags >= -i) & (lags < n-i)
            j1 = i - lags[first_mask]
            j2 = i + lags[second_mask]
            coeff[first_mask] += (1.0 - h[j1])
            coeff[second_mask] -= h[j2]
            base = corr - coeff * h[i]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi-lo)/3.0
                m2 = hi - (hi-lo)/3.0
                if np.max(base + coeff*m1) < np.max(base + coeff*m2):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            corr += (t_opt - h[i]) * coeff
            h[i] = t_opt
        h = fix_sum(h, n)
    return h
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
    else:
        if diff > 0:
            for i in np.argsort(h):
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff < 1e-13: break
        else:
            diff = -diff
            for i in np.argsort(-h):
                sub = min(diff, h[i])
                h[i] -= sub
                diff -= sub
                if diff < 1e-13: break
    return np.clip(h, 0.0, 1.0)

def subgrad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    cmax = np.max(corr)
    active = np.where(corr >= cmax - 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += 1.0 - 2.0 * h
            else:
                grad[lag:] += 1.0 - h[:n-lag]
                grad[:n-lag] -= h[lag:]
        else:
            l = -lag
            grad[:n-l] += 1.0 - h[l:]
            grad[l:] -= h[:n-l]
    grad /= len(active)
