import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def active_gradient(h, top_k=5, T_w=0.02):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    c_max = corr[sorted_idx[0]]
    weights = np.exp((corr[sorted_idx] - c_max) / T_w)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for w, k_idx in zip(weights, sorted_idx):
        k = k_idx - (n - 1)
        g = np.zeros(n)
        if k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        elif k < 0:
            kp = -k
            g[:n-kp] += (1.0 - h[kp:])
            g[kp:] -= h[:n-kp]
        else:
            g[:] = 1.0 - 2.0 * h[:]
        grad += w * g
    return grad

def pg_step(h, n, alpha, top_k=5, T_w=0.02):
    c0 = compute_c5(h)
    grad = active_gradient(h, top_k, T_w)
    for _ in range(30):
        h_try = proj_constraint(h - alpha * grad, n)
        c_try = compute_c5(h_try)
        if c_try < c0:
            return h_try, c_try
        alpha *= 0.5

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def de_evolve(h_best, n, t_start, budget_s, pop_size=15, F=0.8, CR=0.7, seed=0):

    h_best = h_best.copy()

    pop = [h_best.copy()]
    pop_c = [c_best]
    for i in range(pop_size - 1):

        h_i = proj_constraint(h_i, n)
        pop.append(h_i)
        pop_c.append(compute_c5(h_i))
    
    gen = 0
    while time.time() - t_start < budget_s - 120:
        gen += 1
        F_cur = F * (0.95 ** (gen // 10))
        for i in range(pop_size):
            rng2 = np.random.RandomState(rng.randint(2**31))
            indices = [j for j in range(pop_size) if j != i]
            a, b, c = rng2.choice(indices, 3, replace=False)
            v = pop[a] + F_cur * (pop[b] - pop[c])
            # Crossover

            u = np.where(mask, v, pop[i])
            u = proj_constraint(u, n)
            c_u = compute_c5(u)
            if c_u < pop_c[i]:
                pop[i] = u
                pop_c[i] = c_u
            if c_u < c_best:
                c_best = c_u
                h_best = u.copy()
    return h_best, c_best

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_grad(h, top_k, T_w):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    c_max = corr[sorted_idx[0]]
    weights = np.exp((corr[sorted_idx] - c_max) / T_w)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for w, k_idx in zip(weights, sorted_idx):
        k = k_idx - (n - 1)
        g = np.zeros(n)
        if k > 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        elif k < 0:
            kp = -k
            g[:n - kp] += (1.0 - h[kp:])
            g[kp:] -= h[:n - kp]
        else:
            g[:] = 1.0 - 2.0 * h[:]
        grad += w * g
    return grad, c_max

        if c_try < best_c:

        a *= 0.6
        if a < 1e-8:

    return best_h, best_c

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

    return np.sum(h) - n * 0.5

    return np.ones(n)

    res = opt.minimize(soft_max_obj, h0, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
