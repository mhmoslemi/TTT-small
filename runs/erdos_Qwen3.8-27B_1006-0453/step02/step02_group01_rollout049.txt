from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    """Convert to binary with exactly n//2 ones, keeping tie-breaking random."""
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def swap_local_search(h, n, time_limit=None, t_start=None):
    """Exhaustive single-swap greedy local search on binary h."""
    if t_start is None:
        t_start = time.time()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    best_c5 = compute_c5(h)
    
    while True:
        if time_limit and time.time() - t_start > time_limit:
            break
        best_delta = -1e-15
        best_ij = None
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                c5 = compute_c5(h)
                h[i] = 1; h[j] = 0
                d = c5 - best_c5
                if d < best_delta:
                    best_delta = d
                    best_ij = (i, j)
        if best_ij is None:
            break
        i, j = best_ij
        h[i] = 0; h[j] = 1
        ones.remove(i); ones.append(j)
        zeros.remove(j); zeros.append(i)
        best_c5 = best_c5 + best_delta
    return h, best_c5

def annealed_swap(h, n, T0=0.005, T_end=0.0001, max_rounds=2000):
    """Simulated annealing with single swaps."""
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    
    for rnd in range(max_rounds):
        T = T0 * (T_end / T0) ** (rnd / max_rounds)
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0; h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def swap_local_search(h, n, max_rounds=1000, t_start=None, time_limit=None):
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    best_c5 = compute_c5(h)
    if t_start is None:
        t_start = time.time()
    for _ in range(max_rounds):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        best_delta = -1e-15
        best_ij = None
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h)
                h[i] = 1
                h[j] = 0
                d = c5 - best_c5
                if d < best_delta:
                    best_delta = d
                    best_ij = (i, j)
        if best_ij is None:
            break
        i, j = best_ij
        h[i] = 0
        h[j] = 1
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
        best_c5 += best_delta
    return h, best_c5

def anneal_swap(h, n, T0=0.004, T1=0.0002, rounds=8000):
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    for r in range(rounds):
        T = T0 * (T1 / T0) ** (r / max(rounds - 1, 1))
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-9)):
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5 - 1e-13:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5

def slsqp_polish(h, n, maxiter=600):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(compute_c5, h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_slsqp(h, n, T, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
