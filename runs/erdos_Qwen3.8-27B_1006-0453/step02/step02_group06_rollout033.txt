    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)

        c2 = np.sum(m)
        if c2 > 0:
            h[m] += d / c2

    return h

    if idx is not None:
        corr = corr[idx]
    dx = 2.0 / len(h)
    mx = np.max(corr)
    return (mx + T * np.log(np.sum(np.exp((corr - mx) / T)))) * dx

    corr = np.correlate(h, 1.0 - h, mode='full')

    return (np.mean(corr ** p)) ** (1.0 / p) * dx

                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18})
    return proj(r.x, n)

                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18})
