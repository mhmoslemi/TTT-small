def fix_sum(h, n):

    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-12:
        return h
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(mask)
    if k > 0:
        h[mask] += d / k
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-12:

def lse_of_arr(arr, T):
    m = np.max(arr)
    if T < 1e-15:
        return m
    return m + T * np.log(np.sum(np.exp((arr - m) / T)))

def coord_descent(h, n, n_passes=10, T=0.0, n_iter=50):
    """Exact coordinate descent with golden section search."""
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full').astype(float)
    
    for _pass in range(n_passes):
        improved = 0
        for j in range(n):
            old_h = h[j]
            if old_h <= 1e-10 or old_h >= 1.0 - 1e-10:
                continue
            
            # Compute linear coefficients
            coeff = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                s = m - (n - 1)
                if s == 0:
                    continue
                c = 0.0
                if 0 <= j + s < n:
                    c += 1.0 - h[j + s]
                if 0 <= j - s < n:
                    c -= h[j - s]
                coeff[m] = c
            
            # Golden section search
            lo, hi = 0.0, 1.0
            for _it in range(n_iter):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                
                if T < 1e-15:
                    d1 = m1 - old_h
                    nc1 = corr + d1 * coeff
                    nc1[n-1] += m1*(1-m1) - old_h*(1-old_h)
                    c1 = np.max(nc1)
                    
                    d2 = m2 - old_h
                    nc2 = corr + d2 * coeff
                    nc2[n-1] += m2*(1-m2) - old_h*(1-old_h)
                    c2 = np.max(nc2)
                else:
                    d1 = m1 - old_h
                    nc1 = corr + d1 * coeff
                    nc1[n-1] += m1*(1-m1) - old_h*(1-old_h)
                    c1 = lse_of_arr(nc1, T) * dx
                    
                    d2 = m2 - old_h
                    nc2 = corr + d2 * coeff
                    nc2[n-1] += m2*(1-m2) - old_h*(1-old_h)
                    c2 = lse_of_arr(nc2, T) * dx
                
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            
            new_h = (lo + hi) / 2.0
            delta = new_h - old_h
            if abs(delta) > 1e-10:
                corr += delta * coeff
                corr[n-1] += new_h*(1-new_h) - old_h*(1-old_h)
                h[j] = new_h
                improved += 1
        
        if improved == 0:
            break
    
    h = fix_sum(h, n)
    return h

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-13:
        return h
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(mask)
    if k > 0:
        h[mask] += d / k
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-13:
        for i in range(n):
            if 1e-4 < h[i] < 1.0 - 1e-4:

    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def coord_descent(h, n, n_passes=20, n_iter=50, T=0.0):
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    s_vals = np.arange(2 * n - 1) - (n - 1)
    
    for _pass in range(n_passes):
        improved = 0
        for j in range(n):
            old_h = h[j]
            if old_h <= 1e-10 or old_h >= 1.0 - 1e-10:
                continue
            
            idx1 = j + s_vals
            mask1 = (idx1 >= 0) & (idx1 < n) & (s_vals != 0)
            idx2 = j - s_vals
            mask2 = (idx2 >= 0) & (idx2 < n) & (s_vals != 0)
            
            coeff = np.zeros(2 * n - 1)
            coeff[mask1] = 1.0 - h[idx1[mask1]]
            coeff[mask2] += -h[idx2[mask2]]

                    nc2 = corr + (m2 - old_h) * coeff

        coeff_i[mask2_i] += -h[idx2_i[mask2_i]]
        
        idx1_j = j + s_vals

        idx2_j = j - s_vals

            corr = corr_new
            if c_new < best_c5:

The difference only involves k where h_new[k] ≠ h_old[k] (i.e., k=i or k=j) AND k+s where h_new[k+s] ≠ h_old[k+s] (i.e., k+s=i or k+s=j, so k=i-s or k=j-s).

Terms that change:
1. k=i: Old: h_old[i]*(1-h_old[i+s]) = 1*(1-h[i+s]) = 1-h[i+s]
   New: h_new[i]*(1-h_new[i+s]) = 0*(1-h_new[i+s]) = 0
   Contribution to Δ: -(1-h[i+s]) [using old h for the other factor]
   But wait, h_new[i+s] might differ from h_old[i+s] if i+s = j, i.e., s = j-i.
   If i+s = j: New: 0*(1-1) = 0, Old: 1*(1-0) = 1, Δ = -1
   If i+s ≠ j: New: 0*(1-h[i+s]) = 0, Old: 1*(1-h[i+s]) = 1-h[i+s], Δ = -(1-h[i+s])

2. k=j: Old: h_old[j]*(1-h_old[j+s]) = 0
   New: h_new[j]*(1-h_new[j+s]) = 1*(1-h_new[j+s])
   If j+s = i: New: 1*(1-0) = 1, Old: 0, Δ = 1
   If j+s ≠ i: New: 1*(1-h[j+s]) = 1-h[j+s], Old: 0, Δ = 1-h[j+s]

3. k = i-s (where k+s = i): Old: h_old[i-s]*(1-h_old[i]) = h[i-s]*0 = 0
   New: h_new[i-s]*(1-h_new[i]) = h[i-s]*1 = h[i-s]
   (h[i-s] doesn't change since i-s ≠ i,j assuming s≠0 and s≠j-i)
   Δ = h[i-s]

4. k = j-s (where k+s = j): Old: h_old[j-s]*(1-h_old[j]) = h[j-s]*1 = h[j-s]
   New: h_new[j-s]*(1-h_new[j]) = h[j-s]*0 = 0
   Δ = -h[j-s]

5. If s=0: k=i and k+s=i, terms 1 and 3 merge.
   Old: h[i]*(1-h[i]) = 1*0 = 0
   New: h_new[i]*(1-h_new[i]) = 0*1 = 0
   Δ = 0

   Similarly for k=j: Old: 0*(1-0) = 0, New: 1*(1-1) = 0, Δ = 0

So for s≠0:
Δcorr[m] = [-(1-h[i+s]) if i+s≠j else -1] + [(1-h[j+s]) if j+s≠i else 1] + h[i-s] - h[j-s]
All with validity checks (0 ≤ index < n).

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-13:
        return h
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(mask)
    if k > 0:
        h[mask] += d / k
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-13:
        for i in range(n):
            if 1e-4 < h[i] < 1.0 - 1e-4:
                h[i] = np.clip(h[i] + d, 0, 1)
                break
    return h

def upsample_func(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def lse_obj_func(hv, T):
    dx = 2.0 / len(hv)
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    m = np.max(corr)
    if T < 1e-15:
        return m * dx
    return (m + T * np.log(np.sum(np.exp((corr - m) / T)))) * dx

def sum_con_func(hv):
    return np.sum(hv) - 0.5 * len(hv)

def coord_descent(h, n, n_passes=20, n_iter=50, T=0.0):
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    s_vals = np.arange(2 * n - 1) - (n - 1)
    for _pass in range(n_passes):
        improved = 0
        for j in range(n):
            old_h = h[j]
            if old_h <= 1e-10 or old_h >= 1.0 - 1e-10:
                continue
            idx1 = j + s_vals
            mask1 = (idx1 >= 0) & (idx1 < n) & (s_vals != 0)
            idx2 = j - s_vals
            mask2 = (idx2 >= 0) & (idx2 < n) & (s_vals != 0)
            coeff = np.zeros(2 * n - 1)
            coeff[mask1] = 1.0 - h[idx1[mask1]]
            coeff[mask2] += -h[idx2[mask2]]
            self_old = old_h * (1.0 - old_h)
            lo, hi = 0.0, 1.0
            for _it in range(n_iter):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                if T < 1e-15:
                    nc1 = corr + (m1 - old_h) * coeff
                    nc1[n-1] += m1*(1-m1) - self_old
                    c1 = np.max(nc1)
                    nc2 = corr + (m2 - old_h) * coeff
                    nc2[n-1] += m2*(1-m2) - self_old
                    c2 = np.max(nc2)
                else:
                    nc1 = corr + (m1 - old_h) * coeff
                    nc1[n-1] += m1*(1-m1) - self_old
                    mx1 = np.max(nc1)
                    c1 = (mx1 + T*np.log(np.sum(np.exp((nc1-mx1)/T)))) * dx
                    nc2 = corr + (m2 - old_h) * coeff
                    nc2[n-1] += m2*(1-m2) - self_old
                    mx2 = np.max(nc2)
                    c2 = (mx2 + T*np.log(np.sum(np.exp((nc2-mx2)/T)))) * dx
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            new_h = (lo + hi) / 2.0
            if abs(new_h - old_h) > 1e-10:
                delta = new_h - old_h
                corr += delta * coeff
                corr[n-1] += new_h*(1-new_h) - self_old
                h[j] = new_h
                improved += 1
        if improved == 0:
            break
    return fix_sum(h, n)

def binary_sa(h_init, n, n_iter=15000, T0=0.06, T1=0.0005, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = (h_init > 0.5).astype(float)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > n // 2:
        excess = len(ones) - n // 2
        h[rng.choice(ones, excess, replace=False)] = 0.0
    elif len(ones) < n // 2:
        deficit = n // 2 - len(ones)
        h[rng.choice(zeros, deficit, replace=False)] = 1.0
    h = fix_sum(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    t_start = time.time()
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        if time.time() - t_start > 60:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hn = h.copy()
        hn[i], hn[j] = 0.0, 1.0
        c_new = np.max(np.correlate(hn, 1.0 - hn, mode='full')) * dx
        c_old = best_c5
        if c_new < c_old or rng.random() < np.exp(-(c_new - c_old) / max(T, 1e-10)):
            h = hn
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
    return best_h, best_c5
