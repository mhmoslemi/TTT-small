import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def topk_obj(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        vals = corr[idx]
    else:
        vals = corr
    c_max = vals.max()
    shifted = vals - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def constraint_fun(h, n):
    return np.sum(h) - n * 0.5

def constraint_jac(h, n):
    return np.ones(n)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        idx = min(int(i * old_n / new_n), old_n - 1)
        h_new[i] = h[idx]
    return proj(h_new, new_n)

def slsqp_solve(h0, n, T, k, max_iter):
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac, 'args': (n,)}]
    res = opt.minimize(topk_obj, h0, args=(T, k), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    return proj(res.x, n)

def penalty_obj(h, T, k, n, mu):
    base = topk_obj(h, T, k)
    cv = np.sum(h) - n * 0.5
    return base + mu * cv * cv

def lbfbs_solve(h0, n, T, k, max_iter, mu):
    res = opt.minimize(penalty_obj, h0, args=(T, k, n, mu),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return proj(res.x, n)

def sa_refine(h0, n, t_start, budget_s, max_steps, seed):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T_sa = 0.0004
    for step in range(max_steps):
        if time.time() - t_start > budget_s - 30:
            break

        h_try = h_cur.copy()
        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_blk) * 0.018, 0, 1)
        h_try = proj(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T_sa):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T_sa *= 0.9985
    return h_best, c_best

def focus_shifts(h, n, m, T, max_iter):
    """Optimize focusing on the top m shifts specifically."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-m:]
    dx = 2.0 / n

    def focus_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        vals = c[top_idx]
        c_max = vals.max()
        shifted = vals - c_max
        return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac, 'args': (n,)}]
    res = opt.minimize(focus_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    return proj(res.x, n)
