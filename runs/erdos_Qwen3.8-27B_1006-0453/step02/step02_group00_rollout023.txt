def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_c5(h, k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = np.sort(corr)[-k:]
    return np.mean(topk) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

    return h.sum() - len(h) / 2.0

    """One subgradient step on the max correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    # Subgradient of corr[k_max] w.r.t. h
    # corr[k] = sum_i h[i] * (1-h[i-(k-(n-1))])  ...

    # For k_max, the subgradient is:
    # For each i: (1 - h[i - (k_max - (n-1))]) - h[i - (k_max - (n-1)) + 1]... 
    
    # This is getting complex. Let me use numerical subgradient.
    # Actually, let me just do a simple approach: perturb in direction that reduces the max corr.
    
    # The max corr is at k_max. To reduce it, we want to reduce h[j]*(1-h[j-shift])
    # where shift = k_max - (n-1)
    shift = k_max - (n - 1)
    # If shift >= 0: corr = sum_j h[j]*(1-h[j-shift]) for j >= shift
    # To reduce: either decrease h[j] or increase h[j-shift] for the contributing j's

    contrib_mean = contrib.mean()
    contrib_centered = contrib - contrib_mean  # to preserve sum
