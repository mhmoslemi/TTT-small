def fw_step(h, n, topk=1):
    """One Frank-Wolfe step for minimizing max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Find top-k lags
    if topk == 1:
        k_star = np.argmax(corr)
        k_minus_j = k_star - np.arange(n)
        valid = (k_minus_j >= 0) & (k_minus_j < n)
        grad = np.zeros(n)
        grad[valid] = 1.0 - 2.0 * h[k_minus_j[valid]]
    else:
        # Use average of top-k subgradients for smoother direction
        idx = np.argpartition(corr, -topk)[-topk:]
        # Weight by how close to max
        vals = corr[idx]
        mx = vals.max()
        w = np.exp((vals - mx) * 5) / np.sum(np.exp((vals - mx) * 5))
        grad = np.zeros(n)
        for k_idx, wi in zip(idx, w):
            k_minus_j = k_idx - np.arange(n)
            valid = (k_minus_j >= 0) & (k_minus_j < n)
            grad[valid] += wi * (1.0 - 2.0 * h[k_minus_j[valid]])
    
    # Linear oracle: minimize grad·h' s.t. 0<=h'<=1, sum(h')=n/2
    # Set h'=1 where grad is smallest, h'=0 where grad is largest
    order = np.argsort(grad)
    h_new = np.zeros(n)
    h_new[order[:n//2]] = 1.0
    
    return h, h_new, grad

def proj(h, n):

        h_curr = h_new
        c5 = compute_c5(h_curr)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_curr.copy()
    return best_h, best_c5

def swap_local_search(h, n, n_swaps, rng):
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr)
    for _ in range(n_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h_curr[i], h_curr[j] = h_curr[j], h_curr[i]
        c5_new = compute_c5(h_curr)
        if c5_new < c5_curr - 1e-13:
            c5_curr = c5_new
        else:
            h_curr[i], h_curr[j] = h_curr[j], h_curr[i]
    return h_curr, c5_curr

def ceq_fun(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceq_jac(x):
    return np.ones((1, len(x)))

def make_cons():
    return {'type': 'eq', 'fun': ceq_fun, 'jac': ceq_jac}

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = proj(np.array(initial_h_values, dtype=float), 280)
    except Exception:
        bh = proj(np.full(280, 0.5) + rng.randn(280) * 0.03, 280)
    bc5 = compute_c5(bh)

    bnds140 = [(0.0, 1.0)] * 140
    bnds280 = [(0.0, 1.0)] * 280
    cons = make_cons()

    # Phase 1: Frank-Wolfe at n=140 for global structure
    if tleft() > 350:
        h140 = interp_down(bh, 140)
        for topk in [1, 3, 5, 8, 12, 20]:
            if tleft() < 30:
                break
            h140, c140 = fw_iterate(h140, 140, topk, 500, 'accel')
            if c140 < bc5:
                bc5 = c140
                bh = interp_up(h140, 280)
        # Random restarts at coarse level
        for _ in range(4):
            if tleft() < 25:
                break
            h_r = proj(h140 + rng.randn(140) * 0.06, 140)
            h140, c140 = fw_iterate(h_r, 140, 5, 400, 'accel')
            if c140 < bc5:
                bc5 = c140
                bh = interp_up(h140, 280)

    # Phase 2: SLSQP refinement at n=140
    if tleft() > 250:
        h140 = interp_down(bh, 140)
        for t_val in [30, 100, 400, 1500]:
            if tleft() < 20:
                break
            try:
                r = minimize(soft_max_obj, h140, args=(t_val,), method='SLSQP',
                             bounds=bnds140, constraints=cons,
                             options={'maxiter': 1500, 'ftol': 1e-14})
                hc = proj(np.clip(r.x, 0.0, 1.0), 140)
                c5 = compute_c5(hc)
                if c5 < bc5:
                    bc5 = c5
                    bh = interp_up(hc, 280)
                h140 = hc
            except Exception:
                pass

    # Phase 3: Frank-Wolfe at n=280
    if tleft() > 250:
        h280 = bh.copy() if len(bh) == 280 else interp_up(bh, 280)
        for topk in [1, 2, 3, 5, 8, 15]:
            if tleft() < 20:
                break
            h280, cfw = fw_iterate(h280, 280, topk, 400, 'accel')
            if cfw < bc5:
                bc5 = cfw
        # Noise + FW restart
        for _ in range(3):
            if tleft() < 20:
                break
            h_r = proj(h280 + rng.randn(280) * 0.02, 280)
            h280, cfw = fw_iterate(h_r, 280, 5, 300, 'accel')
            if cfw < bc5:
                bc5 = cfw

    # Phase 4: SLSQP with top-k at n=280
    if tleft() > 200:
        corr_tmp = np.correlate(h280, 1.0 - h280, mode='full')
        topk_val = max(4, len(corr_tmp) // 8)
        for t_val in [50, 200, 800, 3000]:
            if tleft() < 20:
                break
            try:
                r = minimize(topk_obj, h280, args=(t_val, topk_val), method='SLSQP',
                             bounds=bnds280, constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-15})
                hc = proj(np.clip(r.x, 0.0, 1.0), 280)
                c5 = compute_c5(hc)
                if c5 < bc5:
                    bc5 = c5
                    h280 = hc
            except Exception:
                pass

    # Phase 5: Binary local search
    if tleft() > 80:
        h_bin = proj((h280 > 0.5).astype(float), 280)
        h_bin, c5_bin = swap_local_search(h_bin, 280, 15000, rng)
        if c5_bin < bc5:
            bc5 = c5_bin
            h280 = h_bin
        # FW from binary
        h280, cfw = fw_iterate(h280, 280, 3, 300, 'accel')
        if cfw < bc5:
            bc5 = cfw

    # Phase 6: More SLSQP with full soft-max
    if tleft() > 80:
        for t_val in [5000, 20000, 100000]:
            if tleft() < 15:
                break
            try:
                r = minimize(soft_max_obj, h280, args=(t_val,), method='SLSQP',
                             bounds=bnds280, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-16})
                hc = proj(np.clip(r.x, 0.0, 1.0), 280)
                c5 = compute_c5(hc)
                if c5 < bc5:
                    bc5 = c5
                    h280 = hc
            except Exception:
                pass

    # Phase 7: Targeted top-k with smaller k
    if tleft() > 40:
        corr_tmp = np.correlate(h280, 1.0 - h280, mode='full')
        topk_small = max(2, len(corr_tmp) // 15)
        for t_val in [2000, 10000]:
            if tleft() < 15:
                break
            try:
                r = minimize(topk_obj, h280, args=(t_val, topk_small), method='SLSQP',
                             bounds=bnds280, constraints=cons,
                             options={'maxiter': 3000, 'ftol': 1e-15})
                hc = proj(np.clip(r.x, 0.0, 1.0), 280)
                c5 = compute_c5(hc)
                if c5 < bc5:
                    bc5 = c5
                    h280 = hc
            except Exception:
                pass

    # Phase 8: Final FW polish + swap
    if tleft() > 20:
        h280, cfw = fw_iterate(h280, 280, 2, 200, 'accel')
        if cfw < bc5:
            bc5 = cfw
        h280, c5s = swap_local_search(h280, 280, 5000, rng)
        if c5s < bc5:
            bc5 = c5s

    h280 = proj(h280, 280)
    final_c5 = compute_c5(h280)
    return h280, final_c5, 280
