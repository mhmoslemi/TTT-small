import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def active_shifts(h, K=4):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1]
    return idx[:K], corr[idx[0]]

def targeted_perturb(h, n, n_active=4, strength=0.03):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[::-1][:n_active]
    # Compute gradient direction for each active shift
    eps = 1e-6
    grad = np.zeros(n)
    for s in top_k:
        g = np.zeros(n)
        for i in range(n):
            h_p = h.copy()
            h_p[i] += eps
            h_p = project_feasible(h_p, n)
            h_m = h.copy()
            h_m[i] -= eps
            h_m = project_feasible(h_m, n)
            cp = np.correlate(h_p, 1.0 - h_p, mode='full')[s]
            cm = np.correlate(h_m, 1.0 - h_m, mode='full')[s]
            g[i] = (cp - cm) / (2.0 * eps)
        grad += g
    # Step opposite to combined gradient, satisfy sum constraint
    grad -= grad.mean()
    h_new = h - strength * grad / (np.max(np.abs(grad)) + 1e-12)
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coarse-to-fine T-schedule
    for T in [50, 150, 400, 1000, 3000, 8000]:
        if elapsed() > budget_s * 0.10:
            break
        hc = slsqp_opt(best_h, T, n, 600, 1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Fine basin hopping (many small steps)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_iter_bh = 150
    for it in range(n_iter_bh):
        if elapsed() > budget_s * 0.52:
            break
        frac = it / n_iter_bh
        step = 0.05 * (1.0 - frac) + 0.012
        # Mix random and targeted perturbation
        if it % 3 == 0:

        elif it % 3 == 1:

        temp = 0.004 * (1.0 - frac) + 0.0002

                           np.random.rand() < np.exp(-(c5v - bh_c5) / temp)):

            h_p = project_feasible(h_p, n_cur_alt)

            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: Coordinate descent / squeeze
    n_cur = len(best_h)
    no_imp = 0
    for outer in range(500):
        if elapsed() > budget_s * 0.94:
            break
        improved = False
        # Single coordinate moves
        for _ in range(20):
            j = np.random.randint(n_cur)
            for delta in [0.008, 0.015, 0.004, -0.008, -0.015, -0.004]:

                    best_h = ht
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for _ in range(12):
                i, j = np.random.choice(n_cur, 2, replace=False)
                for d in [0.006, 0.012, 0.003]:

                        ht = best_h.copy()
                        ht[i] += si * d

                        ht = project_feasible(ht, n_cur)

                            best_h = ht

                    if improved:

        if improved:

        else:

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
