        freqs = np.arange(n)
        ht += 0.04 * np.sin(2 * np.pi * rng.integers(2, 8) * freqs / n) * rng.uniform(0.5, 1.0)
        ht += 0.02 * np.cos(2 * np.pi * rng.integers(3, 12) * freqs / n + rng.uniform(0, 6.28))
    elif mode == 3:
        # Block perturbation
        blk = max(4, n // 40)
        nb = n // blk
        shifts = rng.uniform(-0.08, 0.08, nb)
        ht += np.repeat(shifts, blk)[:n]
    else:
        ht += rng.uniform(-0.015, 0.015, n)
    return fix_sum(ht, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Multi-objective SLSQP cooling (top-k for exploration, LSE for precision)
    sched1 = [(0.05, 400, True), (0.02, 500, True), (0.008, 600, True),
              (0.003, 800, False), (0.001, 1000, False), (0.0004, 1200, False),
              (0.0001, 1500, False)]
    for T, it, use_topk in sched1:
        if remaining() < 250:
            break
        obj_fn = (lambda x, t=T, k=8: topk_obj(x, t, k)) if use_topk else (lambda x, t=T: lse_obj(x, t))
        h = slsqp_opt(h, n, T, it, obj_fn=obj_fn)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: L-BFGS-B refinement
    if remaining() > 200:
        for K in (5, 10, 20):
            if remaining() < 180:
                break
            h = lbfgs_opt(h, n, 0.002, 800, K)
            h = fix_sum(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Phase 3: Coordinate + pair polishing
    if remaining() > 150:
        h = coord_bisect(h, n, n_pass=2, iters_bisect=6)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    if remaining() > 120:
        h = pair_polish(h, n, rng, 500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Restart exploration
    n_restarts = 25
    for i in range(n_restarts):
        if remaining() < 80:
            break
        ht = make_perturb(best_h, n, rng, i % 5)
        # Quick SLSQP on restart
        for T, it in ((0.02, 400), (0.005, 600), (0.001, 800)):
            if remaining() < 60:
                break
            ht = slsqp_opt(ht, n, T, it)
            ht = fix_sum(ht, n)
        # Quick polish
        if remaining() > 70:
            ht = coord_bisect(ht, n, n_pass=1, iters_bisect=5)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht.copy()
            c5 = ct
            h = ht

    # Phase 5: Final fine-tuning on best
    h = best_h.copy()
    for T, it in ((0.0001, 2000), (0.00003, 2500), (0.00001, 3000)):
        if remaining() < 40:
            break
        h = slsqp_opt(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 6: Final coord + pair polish on best
    if remaining() > 30:
        h = best_h.copy()
        h = coord_bisect(h, n, n_pass=1, iters_bisect=5)
        h = pair_polish(h, n, rng, 200)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 7: Try alternate resolution
    if remaining() > 80:
        for n_alt in (200, 350):
            if remaining() < 50:
                break
            h_alt = upsample(best_h, n_alt)
            for T, it in ((0.02, 400), (0.005, 500), (0.001, 700)):
                if remaining() < 40:
                    break
                h_alt = slsqp_opt(h_alt, n_alt, T, it)
                h_alt = fix_sum(h_alt, n_alt)
            h_alt = coord_bisect(h_alt, n_alt, n_pass=1, iters_bisect=5)
            c_alt = compute_c5(h_alt)
            if c_alt < best_c5:
                best_c5, best_h = c_alt, h_alt.copy()

    best_h = fix_sum(best_h, len(best_h))
    return (best_h, compute_c5(best_h), len(best_h))
