    # Phase 1: SLSQP warmup with increasing T
    for T in [20, 80, 400, 1500, 6000]:
        if budget_exhausted(0.06):
            break
        hc = slsqp_opt(best_h, T, n, 400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Differential Evolution (global exploration)
    if not budget_exhausted(0.42):
        T_de = 400
        bounds_de = list(zip(np.zeros(n), np.ones(n)))
        try:
            de_res = differential_evolution(
                de_objective, bounds_de, args=(n, T_de),
                maxiter=100, popsize=14, tol=1e-12,
                mutation=(0.5, 0.9), recombination=0.75,
                seed=seed, polish=False, init='halton-jittered',
                updating='immediate')
            hc = project_feasible(np.clip(de_res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

        # Second DE with different params from best
        if not budget_exhausted(0.42):
            T_de2 = 150
            try:
                h_pert = best_h + np.random.normal(0, 0.08, n)
                h_pert = project_feasible(h_pert, n)
                de_res2 = differential_evolution(
                    de_objective, bounds_de, args=(n, T_de2),
                    maxiter=60, popsize=10, tol=1e-10,
                    mutation=(0.7, 1.0), recombination=0.6,
                    seed=seed + 1, polish=False, init='sobol')
                hc = project_feasible(np.clip(de_res2.x, 0, 1), n)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    # Phase 3: Aggressive SLSQP polish
    for T in [800, 4000, 15000]:
        if budget_exhausted(0.55):
            break
        hc = slsqp_opt(best_h, T, n, 1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 4: Sum-preserving coordinate descent (medium step)
    n_cur = len(best_h)
    best_h, best_c5 = coordinate_refine(
        best_h, n_cur, best_c5, n_trials=50,
        step_lo=0.002, step_hi=0.03, max_iter=300,
        t_limit_fn=lambda: budget_exhausted(0.78))

    # Phase 5: Try alternate resolutions
    for n_try in [100, 150]:
        if budget_exhausted(0.82):
            break
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h_interp + np.random.normal(0, 0.02, n_try), n_try)
        for T in [500, 5000]:
            if budget_exhausted(0.82):
                break
            hc = slsqp_opt(h0, T, n_try, 600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
                n_cur = n_try

    # Phase 6: Micro coordinate search (ultra-fine)
    n_cur = len(best_h)
    best_h, best_c5 = coordinate_refine(
        best_h, n_cur, best_c5, n_trials=80,
        step_lo=0.0001, step_hi=0.004, max_iter=400,
        t_limit_fn=lambda: budget_exhausted(0.95))

    # Phase 7: Final SLSQP at very high T
    n_cur = len(best_h)
    for T in [30000, 100000]:
        if budget_exhausted(0.98):
            break
        hc = slsqp_opt(best_h, T, n_cur, 800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
