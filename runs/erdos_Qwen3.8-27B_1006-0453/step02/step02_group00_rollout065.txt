import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def optimize_slsqp(h, T, n, maxiter=500, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(np.array(h, copy=True), n)

def round_binary(h, n):
    h_r = (h > 0.5).astype(float)
    target = n // 2
    diff = target - int(h_r.sum())
    if diff > 0:
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:diff]:
            h_r[zeros[idx]] = 1.0
    elif diff < 0:
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:(-diff)]:
            h_r[ones[idx]] = 0.0
    return h_r

def gen_starts(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    s = []
    for k in [1, 2, 3, 5]:
        s.append(project_feasible(0.5 + 0.4 * np.cos(np.pi * k * x), n))
    for k in [2, 3, 4]:
        s.append(project_feasible(0.5 + 0.4 * np.sin(2 * np.pi * k * x), n))
    for _ in range(4):
        b = (rng.random(n) < 0.5).astype(float)
        s.append(project_feasible(b, n))
    q = n // 4
    blk = np.zeros(n)
    blk[:q] = 1.0
    blk[2 * q:3 * q] = 1.0
    s.append(project_feasible(blk, n))
    blk2 = np.zeros(n)
    blk2[::3] = 1.5
    s.append(project_feasible(blk2, n))
    half = n // 2
    hp = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x[:half])
    h_ant = np.concatenate([hp, 1.0 - hp])
    s.append(project_feasible(h_ant, n))
    saw = (x * 3) % 2.0
    s.append(project_feasible(0.5 + 0.3 * (1.0 - 2.0 * saw), n))
    return s

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Fine T-schedule from initial (20% budget)
    T_sched = [2, 5, 10, 20, 40, 80, 160, 350, 700, 1500, 4000, 10000]
    for T in T_sched:
        if time.time() - t0 > budget_s * 0.20:
            break
        hc = optimize_slsqp(best_h, T, n, maxiter=1000, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Multi-frequency Fourier basin hopping (25% budget)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_iter_bh = 70
    x_idx = np.arange(n)
    for it in range(n_iter_bh):
        if time.time() - t0 > budget_s * 0.45:
            break
        n_modes = rng.integers(1, 4)
        pert = np.zeros(n)
        amp = 0.10 * (1.0 - 0.5 * it / n_iter_bh) + 0.015
        for _ in range(n_modes):
            freq = rng.integers(1, n // 2 + 1)
            phase = rng.random() * 2.0 * np.pi
            pert += amp * np.cos(2.0 * np.pi * freq * x_idx / n + phase)
        h_pert = project_feasible(bh_h + pert, n)
        T_bh = 400 + int(400 * np.cos(np.pi * it / max(n_iter_bh, 1)))
        hc = optimize_slsqp(h_pert, T_bh, n, maxiter=400, ftol=1e-12)
        c5v = compute_c5(hc)
        temp = 0.004 * (1.0 - it / n_iter_bh) + 0.0001
        if c5v < bh_c5 or rng.random() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = hc
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Diverse multi-start (20% budget)
    starts = gen_starts(n, rng)
    for h0 in starts:
        if time.time() - t0 > budget_s * 0.65:
            break
        for T in [150, 600, 2500]:
            if time.time() - t0 > budget_s * 0.65:
                break
            hc = optimize_slsqp(h0, T, n, maxiter=700, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: Swap + coordinate local search (18% budget)
    n_cur = len(best_h)
    stagnant = 0
    for iteration in range(600):
        if time.time() - t0 > budget_s * 0.83:
            break
        improved = False
        for _ in range(25):
            i = rng.integers(n_cur)
            j = rng.integers(n_cur)
            if i == j:
                continue
            h_try = best_h.copy()
            h_try[i], h_try[j] = h_try[j], h_try[i]
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
                break
        if not improved:
            for _ in range(12):
                j = rng.integers(n_cur)
                for delta in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
                    h_try = best_h.copy()
                    h_try[j] += delta
                    h_try = project_feasible(h_try, n_cur)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
        if improved:
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 120:
            break

    # Phase 4b: Binary rounding trial
    if time.time() - t0 < budget_s * 0.86:
        h_bin = round_binary(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
            hc = optimize_slsqp(h_bin, 2000, n, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 5: High-T SLSQP polish (10% budget)
    n_cur = len(best_h)
    for T in [5000, 15000, 40000, 120000]:
        if time.time() - t0 > budget_s * 0.99:
            break
        hc = optimize_slsqp(best_h, T, n_cur, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final: one more binary round + polish
    if time.time() - t0 < budget_s * 0.995:
        h_bin2 = round_binary(best_h, n_cur)
        c5_b2 = compute_c5(h_bin2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_bin2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
