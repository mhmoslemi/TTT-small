import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def soft_max_uniform_c5(h, T, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    smax = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    top_k = min(20, len(corr))
    sorted_c = np.sort(corr)[-top_k:]
    unif = np.var(sorted_c)
    return (smax + alpha * unif) * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_opt(h, T, maxiter, alpha=0.0):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]

    else:
        fun = lambda x: soft_max_c5(x, T)
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return h

    n = len(h)
    target = int(n / 2)
    h_bin = (h > 0.5).astype(float).copy()

        idx = np.where(h_bin == 1)[0]
        np.random.shuffle(idx)
        for i in range(ones - target):
            h_bin[idx[i]] = 0.0
    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        np.random.shuffle(idx)
        for i in range(target - ones):
            h_bin[idx[i]] = 1.0
    current = h_bin.copy()
    current_cost = compute_c5(current)
    best_h = current.copy()
    best_cost = current_cost
    for it in range(max_iters):
        T = t_start * (t_end / t_start) ** (it / max_iters)
        zeros = np.where(current == 0)[0]
        ones_arr = np.where(current == 1)[0]
        if len(zeros) == 0 or len(ones_arr) == 0:
            break
        i = zeros[np.random.randint(len(zeros))]
        j = ones_arr[np.random.randint(len(ones_arr))]
        trial = current.copy()
        trial[i] = 1.0
        trial[j] = 0.0

            current = trial
            current_cost = trial_cost

                best_h = current.copy()
    return best_h, best_cost
