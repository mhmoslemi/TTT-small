import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h, nd):
    n = len(h)
    if n == nd:
        return proj_constraint(h.copy(), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj_constraint(np.interp(xd, xs, h), nd)

def obj_and_grad(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    csum = float(np.sum(h) - n / 2.0)
    
    # Log-sum-exp of corr
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    e = np.exp(s)
    LSE = mx + np.log(np.sum(e)) / t
    
    obj = LSE * dx + mu * csum * csum
    
    # Gradient
    w = e / (t * np.sum(e))  # shape (2n-1,)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-16:
            continue
        m = k - (n - 1)
        if m >= 0:
            if n - m > 0:
                grad[:n-m] += wk * (1.0 - h[m:])
                grad[m:] -= wk * h[:n-m]
        else:
            j = -m
            if n - j > 0:
                grad[j:] += wk * (1.0 - h[:n-j])
                grad[:n-j] -= wk * h[j:]
    grad *= dx
    grad += 2.0 * mu * csum  # scalar added to each element
    return obj, grad

def topk_obj_and_grad(h, t, mu, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    csum = float(np.sum(h) - n / 2.0)
    
    idx = np.argpartition(corr, -topk)[-topk:]
    cs = corr[idx]
    mx = np.max(cs)
    s = t * (cs - mx)
    s -= np.max(s)
    e = np.exp(s)
    LSE = mx + np.log(np.sum(e)) / t
    
    obj = LSE * dx + mu * csum * csum
    
    w = np.zeros(len(corr))
    w[idx] = e / (t * np.sum(e))
    
    grad = np.zeros(n)
    for k in idx:
        wk = w[k]
        if wk < 1e-16:
            continue
        m = k - (n - 1)
        if m >= 0:
            if n - m > 0:
                grad[:n-m] += wk * (1.0 - h[m:])
                grad[m:] -= wk * h[:n-m]
        else:
            j = -m
            if n - j > 0:
                grad[j:] += wk * (1.0 - h[:n-j])
                grad[:n-j] -= wk * h[j:]
    grad *= dx
    grad += 2.0 * mu * csum
    return obj, grad

def lbfgs_call(h, n, t, mu, topk, maxiter=2000):
    bnds = [(0.0, 1.0)] * n
    if topk is not None:
        r = minimize(topk_obj_and_grad, h, args=(t, mu, topk),
                     method='L-BFGS-B', jac=True, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    else:
        r = minimize(obj_and_grad, h, args=(t, mu),
                     method='L-BFGS-B', jac=True, bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return proj_constraint(np.clip(r.x, 0.0, 1.0), n)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        nn = len(x)
        dx = 2.0 / nn
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr)
        t = 500
        s = t * (corr - mx)
        s -= np.max(s)
        return (mx + np.log(np.sum(np.exp(s))) / t) * dx
    
    r = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                 options={'maxiter': 1000, 'ftol': 1e-15})
    return proj_constraint(np.clip(r.x, 0.0, 1.0), n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h, nd):
    n = len(h)
    if n == nd:
        return proj_constraint(h.copy(), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj_constraint(np.interp(xd, xs, h), nd)

def full_obj_grad(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    csum = float(np.sum(h) - n / 2.0)
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    e = np.exp(s)
    esum = np.sum(e)
    obj = (mx + np.log(esum) / t) * dx + mu * csum * csum
    w = e / (t * esum)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-17:
            continue
        m = k - (n - 1)
        if m >= 0:
            if n - m > 0:
                grad[:n-m] += wk * (1.0 - h[m:])
                grad[m:] -= wk * h[:n-m]

            j = -m

                grad[j:] += wk * (1.0 - h[:n-j])
