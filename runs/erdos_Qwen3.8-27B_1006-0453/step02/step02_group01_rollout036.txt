def compute_subgrad(h, n, tol_frac=0.03):
    corr = compute_corr(h)
    mx = float(corr.max())
    tol = tol_frac * mx
    active = np.where(corr >= mx - tol)[0]
    lags = active - (n - 1)
    g = np.zeros(n)
    m = np.arange(n)
    for lag in lags:
        if lag >= 0:
            I1 = (m >= lag) & (m <= n - 1)
            I2 = (m >= 0) & (m <= n - 1 - lag)
        else:
            I1 = (m >= 0) & (m <= n - 1 + lag)
            I2 = (m >= -lag) & (m <= n - 1)
        c = np.zeros(n)
        c[I1] = 1.0 - h[m[I1] - lag]
        c[I2] += -h[m[I2] + lag]
        g += c
    g /= len(lags)
    return g, mx

    gn = np.linalg.norm(g)

        return h
    h_new = project_feasible(h - step * g / gn, n)
    return h_new

def coord_descent(h, n, sweeps=3, ngrid=30):

            a = c1 - c0
            b = c0

            bi = np.argmin(vals)

                lo, hi = grid[bi-1], grid[bi+1]

                ht = fg[np.argmin(fv)]
            h[mi] = ht
            h = project_feasible(h, n)
        c5_s = compute_c5(h)
        if c5_s < best_c5:
            best_c5 = c5_s
    return h, best_c5

def slsqp_poly(h, maxiter=400):

    except Exception:

            h_r, c5_r = subgrad_run(h_r, best_n, iters=200, s0=0.03, momentum=0.3)
            h_r, c5_r = coord_descent(h_r, best_n, sweeps=2, ngrid=30)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    # Phase 9: Final SLSQP + coord polish
    if rem() > 15:
        hf = slsqp_smooth(best_h, 500, maxiter=400)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    if rem() > 10:
        hf, cf = slsqp_direct(best_h, maxiter=300)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    if rem() > 8:
        hf, cf = coord_descent(best_h, best_n, sweeps=3, ngrid=60)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
