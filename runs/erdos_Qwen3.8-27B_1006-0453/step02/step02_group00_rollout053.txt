from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    if np.max(shifted) > 500:
        shifted = np.clip(shifted, -500, 500)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def sum_func(x):
    return x.sum() - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_func, 'jac': sum_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        xc = project_h(np.clip(res.x, 0, 1), n)
        if compute_c5(xc) > 1.0:
            return project_h(h, n)
        return xc

        return project_h(h, n)
