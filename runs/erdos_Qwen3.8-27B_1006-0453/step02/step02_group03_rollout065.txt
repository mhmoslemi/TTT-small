So d/dh[m] of c_s = (1-h[m-s]) - h[m+s] (for valid indices)

And the full gradient of smooth_max = sum_s w_s * d c_s / d h[m]

g[m] = sum_s w_{n-1+s} * [(1-h[m-s]) - h[m+s]]

where the sum is over valid s.

     = sum_s w_s - (w * h)[m] (convolution) - (w_ref * h)[m] (convolution with reflected w)

    g = np.zeros(n)
    o = n - 1
    for m in range(n):

def compute_grad(h, w, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, ts):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(200):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0 - 1e-13) if d > 0 else (h > 1e-13)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_fin(h0, dx, ts, mit=150):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eq(x):
        return x[:n].sum() - ts
    c0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [c0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mit, 'ftol': 1e-14})
        if r.x is not None:
            hr = project(r.x[:n], ts)
            if compute_c5(hr, dx) <= c0 * dx + 0.005:
                return hr
    except Exception:
        pass
    return h0.copy()

def lse_grad_descent(h0, dx, ts, iters=4000, dl=None):
    n = len(h0)
    h = h0.copy()
    best_h, best_c = h0.copy(), compute_c5(h0, dx)
    o = n - 1
    stall = 0
    for it in range(iters):
        if dl is not None and time.time() > dl:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        frac = it / max(1, iters)
        beta = 8.0 + 100.0 * min(1.0, frac * 1.5)
        top_k = max(2, int(np.ceil(10.0 - 7.0 * frac)))
        top_idx = np.argsort(corr)[::-1][:top_k]
        tc = corr[top_idx]
        w_top = np.exp(0.4 * beta * (tc - tc.max()))
        w_top /= w_top.sum()
        w = np.zeros(2 * n - 1)
        for i in range(top_k):
            w[top_idx[i]] = w_top[i]
        rest = np.setdiff1d(np.arange(2 * n - 1), top_idx)
        if len(rest) > 0:
            w[rest] = 0.02 / len(rest)
        w /= w.sum()
        # Gradient
        g = np.zeros(n)
        for m in range(n):
            s1lo = max(m - (n - 1), -(n - 1))
            s1hi = min(m, n - 1)
            if s1lo <= s1hi:
                s = np.arange(s1lo, s1hi + 1)
                g[m] += np.dot(w[o + s], 1.0 - h[m - s])
            s2lo = max(-m, -(n - 1))
            s2hi = min(n - 1 - m, n - 1)
            if s2lo <= s2hi:
                s = np.arange(s2lo, s2hi + 1)
                g[m] -= np.dot(w[o + s], h[m + s])
        g *= dx
        # Adaptive line search
        base_lr = 0.06 / (1.0 + 0.003 * it)
        found = False
        for sc in (1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.01):
            h_new = project(h - base_lr * sc * g, ts)
            c_new = compute_c5(h_new, dx)
            if c_new < best_c - 1e-15:
                h, best_h, best_c = h_new, h_new.copy(), c_new
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 300:
                break
    return best_h, best_c

def coord_desc(h0, dx, ts, rnds=50, dl=None, rng=None):
    n = len(h0)
    h, best_c = h0.copy(), compute_c5(h0, dx)
    for r in range(rnds):
        if dl is not None and time.time() > dl:
            break
        imp = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for j in order:
            sv = h[j]
            for d in (0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.1, -0.1):
                h[j] = sv + d
                if h[j] < -1e-9 or h[j] > 1 + 1e-9:
                    continue
                np.clip(h, 0, 1, out=h)
                h = project(h, ts)
                c = compute_c5(h, dx)
                if c < best_c - 1e-15:
                    best_c, imp = c, True
                    break
            else:
                h[j] = sv
                h = project(h, ts)
        if not imp:
            break
    return h, best_c

def pair_search(h0, dx, ts, n_tries=500, dl=None, rng=None):
    n = len(h0)
    h, best_c = h0.copy(), compute_c5(h0, dx)
    imp_cnt = 0
    for t in range(n_tries):
        if dl is not None and time.time() > dl:
            break
        if imp_cnt > 100:
            break
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        di, dj = float(rng.uniform(-0.15, 0.15)), float(rng.uniform(-0.15, 0.15))
        si, sj = h[i], h[j]
        h[i], h[j] = si + di, sj - di
        h = project(h, ts)
        c = compute_c5(h, dx)
        if c < best_c - 1e-15:
            best_c, imp_cnt = c, imp_cnt + 1
        else:
            h[i], h[j] = si, sj
            h = project(h, ts)
    return h, best_c

def make_starts(n, ts, rng, ref):
    starts_list = []
    if ref is not None and len(ref) == n:
        starts_list.append(ref.copy())
    elif ref is not None:
        xq = np.linspace(0, 1, n, endpoint=False)
        x0 = np.linspace(0, 1, len(ref), endpoint=False)
        starts_list.append(np.interp(xq, x0, ref))
    starts_list.append(np.full(n, ts / n))
    # Binary block
    h = np.zeros(n)
    p = rng.permutation(n)
    h[p[:int(ts)]] = 1.0
    fr = ts - int(ts)
    if fr > 0.01:
        h[p[int(ts)]] = fr
    starts_list.append(h.copy())
    # Sine patterns
    x = np.arange(n) / n
    for m in (1, 2, 3):
        starts_list.append(0.5 + 0.35 * np.sin(2 * np.pi * m * x + rng.uniform(0, 1)))
    # Random
    starts_list.append(rng.uniform(0, 1, n))
    return [project(s, ts) for s in starts_list]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_end = t_start + budget_s - 8
    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None
    best_h, best_c = None, 1e9

    if hr is not None:

        h0 = project(hr, ts0)
        h0 = slsqp_fin(h0, dx0, ts0, 150)
        best_c = compute_c5(h0, dx0)
        best_h = h0.copy()
        t_mid = t_end - (budget_s * 0.45)
        h1, c1 = lse_grad_descent(h0, dx0, ts0, 3000, dl=min(t_mid, t_end - 200))
        if c1 < best_c:
            best_h, best_c = h1, c1
        h2 = slsqp_fin(best_h, dx0, ts0, 120)
        c2 = compute_c5(h2, dx0)
        if c2 < best_c:

        h3, c3 = lse_grad_descent(best_h, dx0, ts0, 2000, dl=t_end - 150)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Try other n values
    n_base = len(hr) if hr is not None else 50
    n_list = list(range(max(20, n_base - 8), min(70, n_base + 10)))
    rng.shuffle(n_list)
    n_list.sort(key=lambda x: abs(x - n_base))
    
    for nt in n_list:
        if nt == n_base:
            continue
        if time.time() > t_end - 60:
            break
        dxt, tst = 2.0 / nt, nt * 0.5
        starts = make_starts(nt, tst, rng, best_h)
        for hs in starts[:3]:
            if time.time() > t_end - 40:
                break
            t_alloc = min(60, t_end - 40 - time.time())
            if t_alloc < 10:

            h_s = slsqp_fin(hs, dxt, tst, 100)

            h_s = slsqp_fin(h_s, dxt, tst, 80)
            c_s = compute_c5(h_s, dxt)
            if c_s < best_c:
                best_h, best_c = h_s, c_s

    # Final polish
    if best_h is not None:
        nb, dxb, tsb = len(best_h), 2.0 / len(best_h), len(best_h) * 0.5
        h_f = slsqp_fin(best_h, dxb, tsb, 100)
        c_f = compute_c5(h_f, dxb)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        if time.time() < t_end - 30:
            h_f, c_f = lse_grad_descent(best_h, dxb, tsb, 1500, dl=t_end - 15)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
        if time.time() < t_end - 10:

    best_h = project(best_h, len(best_h) * 0.5)
    best_c = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c, len(best_h)
