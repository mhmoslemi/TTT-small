def anneal(h0, dx, ts, t_start, n_iter, deadline, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    best = h.copy()
    bc_best = bc
    for i in range(n_iter):
        if time.time() > deadline:
            break
        T = t_start * (0.999 ** i)
        if T < 1e-8:
            break
        # Random move
        move_type = rng.randint(4)
        if move_type == 0:
            # Single point
            j = rng.randint(n)
            h[j] += rng.uniform(-0.1, 0.1)
        elif move_type == 1:
            # Swap
            i, j = rng.randint(n), rng.randint(n)
            h[i], h[j] = h[j], h[i]
        elif move_type == 2:
            # Block shift
            m = rng.randint(2, max(3, n//5))
            i = rng.randint(n - m)
            block = h[i:i+m].copy()
            h[i:i+m] = rng.permutation(block)
        else:
            # Two-point
            i, j = rng.randint(n), rng.randint(n)
            d = rng.uniform(-0.1, 0.1)
            h[i] += d
            h[j] -= d  # maintain sum
        
        h2 = project(h, ts)
        c2 = compute_c5(h2, dx)
        delta = c2 - bc
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h, bc = h2, c2
            if bc < bc_best:
                best, bc_best = h.copy(), bc
    return best, bc_best

        d = ts - h.sum()
        if abs(d) < 1e-14:

    vals = corr[top_idx]
    w = np.exp(6.0 * (vals - vals.max()))

    _sp_n = n
    _sp_ts = ts

    def _sobj(x):
        return x[_sp_n]

    def _sineq(x):
        return x[_sp_n] - np.correlate(x[:_sp_n], 1.0 - x[:_sp_n], mode='full')

    def _siq(x):
        return float(np.sum(x[:_sp_n]) - _sp_ts)

    t0 = _c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:

            c2 = _c5(h2, dx)
            if c2 < bc - 1e-14:
                h, bc = h2, c2
                best = h.copy()
                stall, ok = 0, True
                break
        if not ok:
            stall += 1
            if stall > 80:
                break
    return best, _c5(best, dx)

def _paircd(h0, dx, ts, nrnd, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), _c5(h0, dx)
    for r in range(nrnd):
        if time.time() > dl:
            break
        imp = False
        for _ in range(6 * n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            si, sj = h[i], h[j]
            done = False
            for di in (0.03, -0.03, 0.06, -0.06, 0.015, -0.015):
                for dj in (0.03, -0.03, 0.015, -0.015):
                    h[i], h[j] = si + di, sj + dj
                    if h[i] < -0.02 or h[i] > 1.02 or h[j] < -0.02 or h[j] > 1.02:
                        continue
                    h2 = _proj(h, ts)
                    c2 = _c5(h2, dx)
                    if c2 < bc - 1e-13:
                        h[:] = h2
                        bc, imp, done = c2, True, True
                        break
                if done:
                    break
            if not done:
                h[i], h[j] = si, sj
        if not imp:
            break
    return h, bc

def _anneal(h0, dx, ts, t0, nit, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), _c5(h0, dx)
    best, bc_best = h.copy(), bc
    for i in range(nit):
        if time.time() > dl:
            break
        T = t0 * (0.9992 ** i)
        if T < 1e-9:
            break
        mt = rng.randint(4)
        if mt == 0:
            j = rng.randint(n)
            h[j] = h[j] + rng.uniform(-0.08, 0.08)
        elif mt == 1:
            i, j = rng.randint(n), rng.randint(n)
            h[i], h[j] = h[j], h[i]
        elif mt == 2:
            m = rng.randint(2, max(3, n // 6))
            i = rng.randint(max(1, n - m))
            blk = h[i:i + m].copy()
            h[i:i + m] = rng.permutation(blk)
        else:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                d = rng.uniform(-0.06, 0.06)
                h[i] += d
                h[j] -= d
        h2 = _proj(h, ts)
        c2 = _c5(h2, dx)
        d = c2 - bc
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            h, bc = h2, c2
            if bc < bc_best:
                best, bc_best = h.copy(), bc
    return best, bc_best

def _starts(n, ts, rng, ref=None):
    out = [np.full(n, ts / n)]
    for _ in range(4):
        h = np.zeros(n)
        p = rng.permutation(n)
        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[p[int(ts)]] = fr
        out.append(h.copy())
    for m in (1, 2, 3, 5, 7):
        x = np.arange(n) / n
        out.append(np.clip(0.5 + 0.3 * np.sin(2 * np.pi * m * x), 0, 1))
    for _ in range(2):
        out.append(rng.uniform(0, 1, n))
    p = max(2, n // 5)
    out.append(np.array([1.0 if (i % (2 * p)) < p else 0.0 for i in range(n)]))
    if ref is not None:
        if len(ref) == n:
            out.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.insert(0, hi)
    return [_proj(s, ts) for s in out]

def _optimize(h0, dx, ts, dl, rng):
    h = _slsqp(h0, dx, ts, 120)
    c = _c5(h, dx)
    if time.time() < dl - 20:
        h2, c2 = _sgdescent(h, dx, ts, 3000, dl - 15)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 15:
        h3, c3 = _paircd(h, dx, ts, 40, dl - 10, rng)
        if c3 < c:
            h, c = h3, c3
    if time.time() < dl - 10:
        h4, c4 = _anneal(h, dx, ts, 0.005, 8000, dl - 5, rng)
        if c4 < c:
            h, c = h4, c4
    if time.time() < dl - 5:
        h5 = _slsqp(h, dx, ts, 80)
        c5 = _c5(h5, dx)
        if c5 < c:
            h, c = h5, c5
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    dl = t_start + budget_s - 5
    bh, bc = None, 1e9

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None

    if hr is not None:
        n0 = len(hr)
        dx0 = 2.0 / n0
        ts0 = n0 / 2.0
        bh, bc = _optimize(_proj(hr, ts0), dx0, ts0, dl - 100, rng)
    else:
        n0 = 50
        bh = np.full(n0, 0.5)
        bc = _c5(bh, 2.0 / n0)

    n_base = len(bh)
    for dn in (-2, 2, -1, 1, -3, 3, -4, 4, 5, -5):
        nt = n_base + dn
        if nt < 25 or nt > 75 or time.time() > dl - 60:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        st = _starts(nt, tst, rng, bh)
        for hs in st[:3]:
            if time.time() > dl - 30:
                break
            hr1, cr1 = _optimize(hs, dxt, tst, dl - 20, rng)
            if cr1 < bc:
                bh, bc = hr1, cr1

    if time.time() < dl - 80:
        nb = len(bh)
        dxb, tsb = 2.0 / nb, nb / 2.0
        for amp in (0.12, 0.06, 0.03):
            if time.time() > dl - 30:
                break
            hp = _proj(bh + amp * rng.randn(nb), tsb)
            hr2, cr2 = _optimize(hp, dxb, tsb, dl - 15, rng)
            if cr2 < bc:
                bh, bc = hr2, cr2

    if bh is not None:
        nf = len(bh)
        dxf, tsf = 2.0 / nf, nf / 2.0
        hf = _slsqp(bh, dxf, tsf, 100)
        cf = _c5(hf, dxf)
        if cf < bc:
            bh, bc = hf, cf
        hd, cd = _paircd(bh, dxf, tsf, 20, t_start + budget_s - 2, rng)
        if cd < bc:
            bh, bc = hd, cd

    if bh is None:
        bh = _proj(np.full(50, 0.5), 25.0)
        bc = _c5(bh, 0.04)

    bh = _proj(bh, len(bh) / 2.0)
    bc = _c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
