    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(corr.max()) * dx
    sc = tau * (corr - corr.max())
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5_val
    w /= ws
    center = N - 1
    grad = np.zeros(N)
    for ai in range(len(corr)):
        wi = w[ai]
        if wi < 1e-4:
            continue
        L = ai - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad * dx, c5_val

        h = project_sum(h, target)
        if it % 50 == 0:

        if lr < 1e-10:
            break
    return best_h, best_c5

                    h = h_new
                    best_c5 = c5_new
                    break
    return h, best_c5

        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 40 == 0:
            lr *= decay
        if lr < 1e-10:
            break
    return best_h, best_c5

def active_set_phase(h, N, target, deadline, n_top=3, lr_init=0.4, use_momentum=True):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr_init
    stall = 0
    beta = 0.3
    mom = np.zeros(N)
    while time.time() < deadline:
        grad, c5_val = true_grad(h, N, n_top)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if use_momentum:
            mom = beta * mom + (1.0 - beta) * grad
        else:
            mom = grad
        found = False
        for bt in range(12):
            step = lr / (2.0 ** bt)
            h_new = np.clip(h - step * mom, 0.0, 1.0)
            h_new = project_sum(h_new, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                found = True
                break
        if not found:
            lr *= 0.4
            if lr < 1e-6:
                break
        else:
            lr = min(lr * 1.15, lr_init)
        if stall > 80:
            break
    return best_h, best_c5

def coord_polish(h, target, dx, rng, n_attempts=2000):
    N = len(h)
    best_c5 = compute_c5(h, dx)
    deltas = [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]
    for _ in range(n_attempts):
        i = rng.randint(0, N)
        j = rng.randint(0, N)
        if i == j:
            continue
        for d in deltas:
            hi_n = h[i] + d
            hj_n = h[j] - d
            if 0.0 <= hi_n <= 1.0 and 0.0 <= hj_n <= 1.0:
                h_orig_i, h_orig_j = h[i], h[j]
                h[i] = hi_n
                h[j] = hj_n
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                else:
                    h[i] = h_orig_i
                    h[j] = h_orig_j
                break
    return h, best_c5

def block_init(N, rng):
    n_blocks = rng.choice([4, 5, 6, 7, 8])
    block_sizes = np.array([N // n_blocks] * n_blocks)
    rem = N - block_sizes.sum()
    block_sizes[:rem] += 1
    n_on = N // 2
    order = rng.permutation(n_blocks)
    h = np.zeros(N)
    filled = 0
    for bi in order:
        need = n_on - filled
        take = min(need, block_sizes[bi])
        start = int(np.sum(block_sizes[:bi]))
        h[start:start + take] = 1.0
        filled += take
        if filled >= n_on:
            break
    h += rng.randn(N) * 0.1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    N = n0
    dx = 2.0 / N
    target = N / 2.0
    best_h = project_sum(h0.copy(), target)
    best_c5 = compute_c5(best_h, dx)
    h_work = best_h.copy()

    # Phase 1: Adam coarse
    rem = deadline - time.time()
    if rem > 150:
        h1, c1 = adam_phase(h_work.copy(), N, target, time.time() + rem * 0.25,
                            tau0=5.0, tau1=60.0, lr0=0.02, decay=0.988, n_steps=3000)
        if c1 < best_c5:
            best_c5, best_h, h_work = c1, h1, h1.copy()

    # Phase 2: Adam fine
    rem = deadline - time.time()
    if rem > 120:
        h2, c2 = adam_phase(h_work.copy(), N, target, time.time() + rem * 0.35,
                            tau0=60.0, tau1=400.0, lr0=0.006, decay=0.993, n_steps=4000)
        if c2 < best_c5:
            best_c5, best_h, h_work = c2, h2, h2.copy()

    # Phase 3: Active-set gradient refinement
    rem = deadline - time.time()
    if rem > 80:
        h3, c3 = active_set_phase(h_work.copy(), N, target, time.time() + rem * 0.45,
                                  n_top=3, lr_init=0.35, use_momentum=True)
        if c3 < best_c5:
            best_c5, best_h, h_work = c3, h3, h3.copy()

    # Phase 4: Coordinate polish
    rem = deadline - time.time()
    if rem > 30:
        n_att = min(4000, int(rem * 40))
        h4, c4 = coord_polish(h_work.copy(), target, dx, rng, n_attempts=n_att)
        if c4 < best_c5:
            best_c5, best_h, h_work = c4, h4, h4.copy()

    # Random restarts with structured initializations
    rem = deadline - time.time()
    n_restarts = min(4, int(rem // 60))
    for ri in range(n_restarts):
        if time.time() > deadline - 70:
            break
        if ri == 0:
            h_r = block_init(N, rng)
        elif ri == 1:
            h_r = (h_work > 0.5).astype(float) + rng.randn(N) * 0.12
        else:
            h_r = h_work + rng.randn(N) * 0.06
        h_r = project_sum(h_r, target)
        t_sub = time.time() + min(35, (deadline - time.time()) * 0.4)
        h_r, c_r = adam_phase(h_r, N, target, t_sub,
                              tau0=10.0, tau1=120.0, lr0=0.014, decay=0.988, n_steps=2000)
        if time.time() < deadline - 25:
            h_r, c_r = active_set_phase(h_r, N, target, time.time() + 15,
                                        n_top=3, lr_init=0.25, use_momentum=False)
        if c_r < best_c5:
            best_c5, best_h, h_work = c_r, h_r, h_r.copy()

    # Final polish
    if time.time() < deadline - 4:
        h_f, c_f = coord_polish(best_h.copy(), target, dx, rng, n_attempts=800)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        if time.time() < deadline - 2:
            h_f2, c_f2 = active_set_phase(best_h.copy(), N, target, deadline - 1,
                                          n_top=2, lr_init=0.1, use_momentum=False)
            if c_f2 < best_c5:
                best_c5, best_h = c_f2, h_f2

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
