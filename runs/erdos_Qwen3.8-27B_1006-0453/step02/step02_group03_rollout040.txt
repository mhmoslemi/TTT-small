        h5, c5v = block_swap(h, dx, ts, 6, dl - 3, rng)
        if c5v < c:
            h, c = h5, c5v
    if dl and time.time() < dl - 3:
        h6 = slsqp_opt(h, ts, 200)
        c6 = compute_c5(h6, dx)
        if c6 < c:
            h, c = h6, c6
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl = t0 + budget_s - 5
    bh, bc = None, 1e9

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None

    if hr is not None:
        n0 = len(hr)
        dx0, ts0 = 2.0 / n0, n0 / 2.0
        bh, bc = optimize_one(proj(hr, ts0), dx0, ts0, dl, rng)
        hc_best = compute_c5(bh, dx0)
        if hc_best < bc:
            bc = hc_best

    n_base = len(hr) if hr is not None else 50
    for dn in (2, -2, 4, -4, 6, -6, 8, -8, 10, 12, -10):
        nt = n_base + dn
        if nt < 30 or nt > 80 or time.time() > dl - 60:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        sts = make_starts(nt, tst, rng, bh)
        for hs in sts[:4]:
            if time.time() > dl - 28:
                break
            hr2, cr2 = optimize_one(hs, dxt, tst, dl - 18, rng)
            if cr2 < bc:
                bh, bc = hr2, cr2

    if bh is not None and time.time() < dl - 55:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        for amp in (0.25, 0.12, 0.06, 0.03, 0.015):
            if time.time() > dl - 22:
                break
            hp = proj(bh + amp * rng.randn(nb), tsb)
            hr2, cr2 = optimize_one(hp, dxb, tsb, dl - 14, rng)
            if cr2 < bc:
                bh, bc = hr2, cr2

    if bh is not None:
        nf, dxf, tsf = len(bh), 2.0 / len(bh), len(bh) / 2.0
        hf = slsqp_opt(bh, tsf, 300)
        cf = compute_c5(hf, dxf)
        if cf < bc:
            bh, bc = hf, cf
        hd, cd = active_grad_descent(bh, dxf, tsf, 500, t0 + budget_s - 6)
        if cd < bc:
            bh, bc = hd, cd
        hp2, cp2 = pair_swap(bh, dxf, tsf, 12, t0 + budget_s - 4, rng)
        if cp2 < bc:
            bh, bc = hp2, cp2
        hf2 = slsqp_opt(bh, tsf, 150)
        cf2 = compute_c5(hf2, dxf)
        if cf2 < bc:
            bh, bc = hf2, cf2

    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh, 0.04)

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
