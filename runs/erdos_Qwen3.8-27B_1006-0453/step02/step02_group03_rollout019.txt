import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def smooth_grad(h, beta, dx):
    n = len(h)
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    m = z.max()
    ez = np.exp(beta * (z - m))
    s = ez.sum()
    w = ez / s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        k_lo = max(0, o - j)
        k_hi = min(2*n-2, o + n - 1 - j)
        if k_lo <= k_hi:
            idx = k_lo + np.arange(k_hi - k_lo + 1)
            h_idx = idx + j - o
            g[j] += np.dot(w[idx], 1.0 - h[h_idx])
        k_lo2 = max(0, j + o - n + 1)
        k_hi2 = min(2*n-2, j + o)
        if k_lo2 <= k_hi2:
            idx = k_lo2 + np.arange(k_hi2 - k_lo2 + 1)
            h_idx = j - idx + o
            g[j] -= np.dot(w[idx], h[h_idx])
    return g * dx

def pgd(h0, dx, ts, dl, rng):
    n = len(h0)
    h = proj(h0.copy(), ts)
    bc = compute_c5(h, dx)
    betas = [3, 5, 10, 20, 40, 80, 160, 320]
    for bi, beta in enumerate(betas):
        iters_per_beta = 200
        for it in range(iters_per_beta):
            if time.time() > dl:
                return h, bc
            g = smooth_grad(h, beta, dx)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            lr = 0.1 / (gn + 1e-10)
            improved = False
            for sh in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_new = proj(h - lr * sh * g, ts)
                c_new = compute_c5(h_new, dx)
                if c_new < bc - 1e-15:
                    h, bc = h_new, c_new
                    improved = True
                    break
            if not improved:
                # Try simulated annealing step
                T = 0.001 * (1.0 - bi / len(betas))
                if np.random.random() < min(1.0, T / (bc + 1e-10)):
                    h_new = proj(h + 0.02 * rng.randn(n), ts)
                    c_new = compute_c5(h_new, dx)
                    if c_new - bc < T:
                        h, bc = h_new, c_new
    return h, bc

def coord_desc(h0, dx, ts, dl, rng, n_rounds=20):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.03, -0.03]
    for rd in range(n_rounds):

            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in deltas:
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:

                h = proj(h, ts)
                c = compute_c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def smooth_grad(h, beta, dx):
    n = len(h)
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    m = z.max()
    ez = np.exp(beta * (z - m))
    s = ez.sum()
    w = ez / s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        k_lo = max(0, o - j)
        k_hi = min(2 * n - 2, o + n - 1 - j)
        if k_lo <= k_hi:
            idx = k_lo + np.arange(k_hi - k_lo + 1)
            h_idx = idx + j - o
            g[j] += np.dot(w[idx], 1.0 - h[h_idx])
        k_lo2 = max(0, j + o - n + 1)
        k_hi2 = min(2 * n - 2, j + o)
        if k_lo2 <= k_hi2:
            idx = k_lo2 + np.arange(k_hi2 - k_lo2 + 1)
            h_idx = j - idx + o
            g[j] -= np.dot(w[idx], h[h_idx])
    return g * dx

def pgd(h0, dx, ts, dl, rng, beta_lo=10, beta_hi=400, nphases=7):
    n = len(h0)
    h = proj(h0.copy(), ts)
    bc = compute_c5(h, dx)
    for pi in range(nphases):
        frac = pi / max(1, nphases - 1)
        beta = beta_lo * (beta_hi / beta_lo) ** frac
        niter = 120 if beta < 50 else 80
        for it in range(niter):
            if time.time() > dl:
                return h, bc
            g = smooth_grad(h, beta, dx)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            lr = 0.08 / (gn + 1e-10)
            improved = False
            for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
                h_new = proj(h - lr * sh * g, ts)
                c_new = compute_c5(h_new, dx)
                if c_new < bc - 1e-15:
                    h, bc = h_new, c_new
                    improved = True
                    break
            if not improved and it > 0 and it % 40 == 0:
                h_new = proj(h + 0.008 * rng.randn(n), ts)
                c_new = compute_c5(h_new, dx)
                if c_new < bc + 0.002 * (1.0 - frac):
                    h, bc = h_new, c_new
    return h, bc

def coord_desc(h0, dx, ts, dl, rng, nrounds=25):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    deltas = (0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.015, -0.015)
    for rd in range(nrounds):
        if time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in deltas:
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                h[j] = min(max(h[j], 0.0), 1.0)
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc - 1e-15:
                    bc = c
                    imp = True
                    break
            if not imp or h[j] != sv:
                pass
            h = proj(h, ts)
            if sv != h[j]:
                h[j] = min(max(sv, 0.0), 1.0)
                h = proj(h, ts)
        if not imp:
            break
    h = proj(h, ts)
    return h, bc

def two_point(h0, dx, ts, dl, rng, nrounds=15):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for rd in range(nrounds):
        if time.time() > dl:
            break
        imp = False
        for _ in range(4 * n):
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            for d in (0.01, -0.01, 0.02, -0.02, 0.005, -0.005):
                hv = h[i] - d

                if -1e-10 > hj or hj > 1 + 1e-10:

                h[i], h[j] = max(hv, 0), min(hj, 1)

        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[p[int(ts)]] = fr
        out.append(h.copy())
    x = np.arange(n) / n
    for m in (1, 2, 3):
        out.append(0.5 + 0.25 * np.sin(2 * np.pi * m * x))
    if ref is not None:
        if len(ref) == n:
            out.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.insert(0, hi)
    return [proj(s, ts) for s in out]

def opt_full(h0, dx, ts, dl, rng):
    h, c = pgd(h0, dx, ts, dl - 20, rng, beta_lo=10, beta_hi=400, nphases=7)
    if time.time() < dl - 15:
        h2, c2 = coord_desc(h, dx, ts, dl - 10, rng, 25)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 8:
        h3, c3 = two_point(h, dx, ts, dl - 5, rng, 10)
        if c3 < c:
            h, c = h3, c3
    if time.time() < dl - 5:
        h4, c4 = pgd(h, dx, ts, dl - 2, rng, beta_lo=100, beta_hi=500, nphases=3)
        if c4 < c:
            h, c = h4, c4
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    df = t0 + budget_s - 12
    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None
    bh, bc = None, 1e9
    if hr is not None:
        n0 = len(hr)
        dx0, ts0 = 2.0 / n0, n0 / 2.0
        h_init = proj(hr, ts0)
        bh, bc = h_init, compute_c5(h_init, dx0)
        h1, c1 = opt_full(bh, dx0, ts0, df - 80, rng)
        if c1 < bc:
            bh, bc = h1, c1
    n_base = len(bh) if bh is not None else 50
    for dn in (1, -1, 2, -2, 3, 4, -3, 5, -4, 6, 7, -5):
        nt = n_base + dn
        if nt < 24 or nt > 75 or time.time() > df - 80:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        st = start_points(nt, tst, rng, bh)
        for hs in st[:3]:
            if time.time() > df - 25:
                break
            hr2, cr2 = opt_full(hs, dxt, tst, df - 15, rng)
            if cr2 < bc:
                bh, bc = hr2, cr2
    if bh is not None:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        h_f, c_f = coord_desc(bh, dxb, tsb, t0 + budget_s - 4, rng, 30)
        if c_f < bc:
            bh, bc = h_f, c_f
        h_g, c_g = two_point(bh, dxb, tsb, t0 + budget_s - 2, rng, 12)
        if c_g < bc:
            bh, bc = h_g, c_g
    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh, 2.0 / 50)
    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
