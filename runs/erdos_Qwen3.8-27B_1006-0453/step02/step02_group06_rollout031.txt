    for n_try in n_tries:
        if time.time() - t_start > budget_s - 120:
            break
        if n_try == n_best:
            h_curr = h_best.copy()
        elif n_try > len(h_best):
            h_curr = upsample(h_best, n_try)
        else:
            h_curr = upsample(h_best, 2048)
            h_curr = h_curr[:n_try] * n_try / 2048.0
            h_curr = np.clip(np.interp(np.linspace(0, 2, n_try, endpoint=False),
                                        np.linspace(0, 2, len(h_best), endpoint=False),
                                        h_best[:n_try * len(h_best) // n_try]) if n_try < len(h_best) else h_best, 0, 1)
            h_curr = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                                np.linspace(0, 2, len(h_best), endpoint=False), h_best)
            h_curr = proj_constraint(h_curr, n_try)
        for T in [0.01, 0.003, 0.0008]:
            if time.time() - t_start > budget_s - 60:
                break
            h_opt = lbfgs_solve(h_curr, n_try, T, 50.0, 250)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c
                n_best = n_try
                h_curr = h_opt.copy()

    # Phase 3: SLSQP with exact gradient (handles equality exactly)
    if time.time() - t_start < budget_s - 80:
        for T in [0.002, 0.0005]:
            if time.time() - t_start > budget_s - 40:
                break
            h_opt = slsqp_solve(h_best.copy(), n_best, T, 400)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 4: Multi-start diverse perturbations
    for i in range(10):
        if time.time() - t_start > budget_s - 90:
            break
        rng = np.random.RandomState(seed + 9973 * (i + 1))
        mode = i % 3
        if mode == 0:
            h_pert = np.clip(h_best + rng.randn(n_best) * 0.015, 0, 1)
        elif mode == 1:
            idx = rng.choice(n_best, size=n_best // 4, replace=False)
            h_pert = h_best.copy()
            h_pert[idx] = 1.0 - h_pert[idx]
        else:
            shift = rng.randint(1, max(2, n_best // 50))
            h_pert = np.roll(h_best, shift)
        h_pert = proj_constraint(h_pert, n_best)
        for T in [0.004, 0.001]:
            if time.time() - t_start > budget_s - 40:
                break
            h_opt = lbfgs_solve(h_pert, n_best, T, 60.0, 300)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 5: Simulated annealing
    time_left = budget_s - (time.time() - t_start) - 50
    if time_left > 60:
        h_sa, c_sa = sa_refine(h_best, n_best, t_start, time_left, 5000, seed + 12345)
        if c_sa < c5_best:
            h_best = h_sa
            c5_best = c_sa

    # Phase 6: Final ultra-fine polish
    if time.time() - t_start < budget_s - 25:
        h_opt = lbfgs_solve(h_best.copy(), n_best, 0.0001, 200.0, 800)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c
    if time.time() - t_start < budget_s - 10:
        h_opt = slsqp_solve(h_best.copy(), n_best, 0.0001, 300)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
