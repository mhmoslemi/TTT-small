import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n*0.5, 'jac': lambda: np.ones(n)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def binary_swap_sa(h_cont, n, t_start, time_budget, seed):
    """SA on binary representations, then project back"""
    rng = np.random.RandomState(seed)
    # Round to binary
    h_bin = (h_cont >= 0.5).astype(float)
    # Fix to exactly n/2 ones
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) > n // 2:
        # Remove some ones
        remove = rng.choice(ones, len(ones) - n//2, replace=False)
        h_bin[remove] = 0
    elif len(ones) < n // 2:
        # Add some ones
        add = rng.choice(zeros, n//2 - len(ones), replace=False)
        h_bin[add] = 1
    
    c_cur = compute_c5(h_bin)
    c_best = c_cur
    h_best = h_bin.copy()
    T = 0.001
    t_end = t_start + time_budget
    
    while time.time() < t_end:
        # Pick a random 1 and a random 0, swap them
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c_try = compute_c5(h_bin)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T):
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]  # undo
        T *= 0.999
    return h_best, c_best

        h_pert = proj_constraint(np.clip(h_best + rng.randn(n) * sigma, 0.0, 1.0), n)
        T_ms = [0.003, 0.0015, 0.004, 0.001, 0.002][i]
        h_opt = do_slsqp(h_pert, n, T_ms, 450)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 6: Try nearby resolutions (80s)
    elapsed = time.time() - t_start
    for n_try in [n - 16, n + 16]:
        if n_try <= 0 or n_try % 2 != 0 or n_try == n:
            continue
        elapsed = time.time() - t_start
        if budget_s - elapsed < 60:
            break
        h_nr = multi_resolution_try(h_best, n, n_try, 0.002, 500)
        c_nr = compute_c5(h_nr)
        if c_nr < c5_best:
            h_best = h_nr
            c5_best = c_nr
            n = n_try

    # Phase 7: Final ultra-fine polish (40s)
    elapsed = time.time() - t_start
    if budget_s - elapsed > 35:
        h_opt = do_slsqp(h_best.copy(), n, 0.0001, 1800)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    h_best = proj_constraint(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
