    s = float(h.sum())
    if abs(s - target) < 1e-12:

        else:
            hi = mid

    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr[N-1:]  # N values: lags 0 to N-1

    sc -= sc.max()

        best_h = h.copy()

    true_c5 = float(np.max(f)) * (2.0 / N)

    sc = tau * f - tau * np.max(f)
    w = np.exp(sc)

    one_minus_h = 1.0 - h

    g2 = np.correlate(h, w, mode='full')[N-1:2*N-1]
    grad = g1 - g2
    return grad, true_c5

    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    stall = 0
    last_best = best_c5
    
    while time.time() < deadline:
        frac = min(it / 5000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        if tau < 0.01:
            tau = 0.01
        
        grad, c5_val = fft_gradient(h, N, tau)
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1

        step = lr * mh / (np.sqrt(vh) + eps)

        h = project_sum(h, target)

        if lr < 1e-12:
            break
        if stall > 500 and it > 200:

    c5_final = compute_c5(best_h, dx)

        return h
    lo, hi = -2.0, 2.0

        else:
            hi = mid

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fft_gradient(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr[N - 1:]
    true_c5 = float(np.max(f)) * (2.0 / N)
    sc = tau * f
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    g1 = np.convolve(w, 1.0 - h, mode='full')[:N]
    g2 = np.correlate(h, w, mode='full')[N - 1:2 * N - 1]
    return g1 - g2, true_c5

def adam_fft(h, N, target, dx, deadline, tau0, tau1, lr0, decay, de):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / 4000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad, cv = fft_gradient(h, N, tau)
        if cv < best_c5:
            best_c5 = cv
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        step = lr * mh / (np.sqrt(vh) + eps)
        gm = np.max(np.abs(step))
        if gm > 0.3:
            step *= 0.3 / gm
        h = project_sum(h - step, target)
        if it % de == 0:
            lr *= decay
        if lr < 1e-12:
            break
    return best_h, best_c5

def fista_opt(h, N, target, dx, deadline, tau, lr):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    y = h.copy()
    x = h.copy()
    t_k = 1.0
    while time.time() < deadline:
        grad, cv = fft_gradient(y, N, tau)
        if cv < best_c5:
            best_c5 = cv
            best_h = y.copy()
        y_new = project_sum(y - lr * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = project_sum(y_new + ((t_k - 1.0) / t_new) * (y_new - x), target)
        t_k = t_new
        x = y_new
    return best_h, best_c5

def binary_proj(h, N):
    idx = np.argsort(h)[::-1][:N // 2]
    b = np.zeros(N)
    b[idx] = 1.0
    return b
