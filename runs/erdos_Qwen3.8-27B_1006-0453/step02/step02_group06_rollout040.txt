    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

def softmax_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr / T
    c_max = np.max(c)
    exps = np.exp(c - c_max)
    sum_exps = np.sum(exps)

    val = T * (c_max + np.log(sum_exps)) * dx

            g = np.zeros(n)
            g[:n-pm] += 1.0 - h[pm:]
            g[pm:] -= h[:n-pm]

    grad *= dx
    return val, grad

    c_best = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    
    for Ti, T in enumerate(T_sched):
        steps = steps_per_T
        lr = lr_init * (0.5 ** Ti)
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        
        for s in range(steps):
            val, grad = softmax_val_grad(h, T)
            # Zero-mean constraint: adjust gradient
            grad -= np.mean(grad)
            
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * grad**2
            m_hat = m_adam / (1 - beta1**(s+1))
            v_hat = v_adam / (1 - beta2**(s+1))
            
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_h(h, n)

        cm = np.max(c)
        if T < 1e-15:
            return float(np.max(c) * (2.0/n))

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n*0.5}]
    res = opt.minimize(obj, h0, method='SLSQP',
        bounds=[(0.0,1.0)]*n, constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return project_h(res.x, n)

    x_new = np.linspace(0, 2, new_n, endpoint=False)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -h.min() - 0.01, 1.0 - h.max() + 0.01
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-9:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += diff2 / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_val_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr / T
    c_shift = c - np.max(c)
    exps = np.exp(c_shift)
    sum_e = np.sum(exps)
    p = exps / sum_e
    val = T * (np.max(c) + np.log(sum_e)) * dx
    grad = np.zeros(n)
    nm1 = n - 1
    for i in range(2 * n - 1):
        w = p[i]
        if w < 1e-14:
            continue

        else:
            m = -k
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
        grad += w * g
    grad *= dx
    return val, grad

    return np.ones(n)

def slsqp_softmax(h, T, n):

        return float((cm + T * np.log(np.sum(np.exp(s / T))))) * dx

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -h.min() - 0.01, 1.0 - h.max() + 0.01
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-9:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += diff2 / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_val_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr / T
    c_max = np.max(c)
    exps = np.exp(c - c_max)
    sum_e = np.sum(exps)
    p = exps / sum_e
    val = T * (c_max + np.log(sum_e)) * dx
    grad = np.zeros(n)
    nm1 = n - 1
    for i in range(2 * n - 1):
        w = p[i]
        if w < 1e-10:
            continue
        k = i - nm1
        g = np.zeros(n)
        if k >= 0:
            if k == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[k:n] += 1.0 - h[0:n-k]
                g[0:n-k] -= h[k:n]
        else:
            mm = -k
            g[0:n-mm] += 1.0 - h[mm:n]
            g[mm:n] -= h[0:n-mm]
        grad += w * g
    grad *= dx
    return val, grad

def sum_eq_func(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac_func(h, n):
    return np.ones(n)

def slsqp_obj(x, T, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(c)
    s = c - cm
    return float((cm + T * np.log(np.sum(np.exp(s / T))))) * dx

def slsqp_refine(h0, T, n, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq_func, 'jac': sum_eq_jac_func, 'args': (n,)}]
    res = opt.minimize(slsqp_obj, h0.copy(), args=(T, n), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return project_h(res.x, n)

def adam_phase(h0, n, T, steps, lr, seed=0):
    h = h0.copy()
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h)
    for s in range(steps):
        val, grad = softmax_val_grad(h, T, n)
        grad -= np.mean(grad)
        m_a = b1 * m_a + (1.0 - b1) * grad
        v_a = b2 * v_a + (1.0 - b2) * grad * grad
        m_hat = m_a / (1.0 - b1 ** (s + 1))
        v_hat = v_a / (1.0 - b2 ** (s + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)

def upsample_h(h, new_n):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return project_h(np.interp(x_new, x_old, h), new_n)

def local_pressure_refine(h0, n, max_rounds, seed):
