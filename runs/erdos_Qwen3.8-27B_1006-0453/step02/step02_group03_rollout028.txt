    for j in range(n):
        kl1, kh1 = max(-(n - 1), j - (n - 1)), min(n - 1, j)
        if kl1 <= kh1:
            k = np.arange(kl1, kh1 + 1)
            g[j] += np.dot(w[o + k], 1.0 - h[j - k])
        kl2, kh2 = max(-(n - 1), -j), min(n - 1, n - 1 - j)
        if kl2 <= kh2:
            k = np.arange(kl2, kh2 + 1)
            g[j] -= np.dot(w[o + k], h[j + k])
    return g * dx

def grad_descent(h0, ts, dl, iters=4000):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    bc = compute_c5(h, dx)
    bh = h.copy()
    stall = 0
    for i in range(iters):
        if time.time() > dl:
            break
        beta = 30.0 + 120.0 * min(1.0, i / (0.5 * iters))
        lr = 0.04 / (1.0 + 0.004 * i)
        g = sgrad(h, dx, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        found = False
        for sh in (lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05, lr*0.01, 0.001, 0.0001):
            h2 = project(h - sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h, bc, bh = h2, c2, h2.copy()
                found = True
                break
        if not found:
            stall += 1
            if stall > 300:
                break
        else:
            stall = 0
    return bh, bc

def coord_descent(h0, ts, dl, rng, rounds=80):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    bc = compute_c5(h, dx)
    bh = h.copy()
    deltas = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.2, -0.2]
    for r in range(rounds):
        if time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            best_v, best_c = sv, bc
            for d in deltas:
                cv = sv + d
                if cv < -0.01 or cv > 1.01:
                    continue
                h[j] = cv
                h = project(h, ts)
                c = compute_c5(h, dx)
                if c < best_c:
                    best_c, best_v = c, cv
            for cv in (0.0, 0.5, 1.0):
                if abs(cv - sv) < 0.005:
                    continue
                h[j] = cv
                h = project(h, ts)
                c = compute_c5(h, dx)
                if c < best_c:
                    best_c, best_v = c, cv
            h[j] = best_v
            h = project(h, ts)
            if best_c < bc - 1e-15:
                bc = best_c
                bh = h.copy()
                imp = True
        if not imp:
            break
    return bh, bc

def make_starts(n, ts, rng, ref=None):
    out = []
    x = np.linspace(0, 2, n, endpoint=False)
    for _ in range(4):
        h = np.zeros(n)
        p = rng.permutation(n)
        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[p[int(ts)]] = fr
        out.append(h.copy())
    for m, amp in [(1, 0.4), (2, 0.35), (3, 0.3), (5, 0.2), (7, 0.15)]:
        out.append(0.5 + amp * np.sin(np.pi * m * x))
    for _ in range(3):
        out.append(rng.uniform(0, 1, n))
    for p in [5, 6, 8]:
        if 2 * p <= n:
            pat = np.array([1.0 if (i % (2*p)) < p//2 else 0.0 for i in range(2*p)])
            out.append(np.tile(pat, n // len(pat) + 1)[:n])
    if ref is not None:
        if len(ref) == n:
            out.insert(0, ref.copy())
        else:
            t = np.linspace(0, 1, n, endpoint=False)
            s = np.linspace(0, 1, len(ref), endpoint=False)
            out.insert(0, np.interp(t, s, ref))
    return [project(s, ts) for s in out]

def opt_pipeline(h0, ts, dl, rng):
    n = len(h0)
    dx = 2.0 / n
    h, c = h0.copy(), compute_c5(h, dx)
    h, c = slsqp_opt(h, ts, 400)
    if time.time() < dl - 20:
        h2, c2 = grad_descent(h, ts, min(dl - 10, time.time() + 150), 5000)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 15:
        h3, c3 = slsqp_opt(h, ts, 350)
        if c3 < c:
            h, c = h3, c3
    if time.time() < dl - 10:
        h4, c4 = coord_descent(h, ts, min(dl - 5, time.time() + 80), rng, 60)
        if c4 < c:
            h, c = h4, c4
    if time.time() < dl - 5:
        h5, c5 = slsqp_opt(h, ts, 250)
        if c5 < c:
            h, c = h5, c5
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl_end = t0 + budget_s - 4

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None

    bh, bc = None, 1e9
    n_base = 50

    if hr is not None:
        n0 = len(hr)
        ts0 = n0 / 2.0
        h0 = project(hr, ts0)
        bh, bc = opt_pipeline(h0, ts0, dl_end - 120, rng)
        n_base = n0
    else:
        bh = project(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh, 0.04)

    # Perturbation restarts at same n
    nb, tsb = len(bh), len(bh) / 2.0
    for amp in (0.1, 0.05, 0.03, 0.015, 0.008):
        if time.time() > dl_end - 150:
            break
        hp = project(bh + amp * rng.randn(nb), tsb)
        hr2, cr2 = opt_pipeline(hp, tsb, dl_end - 30, rng)
        if cr2 < bc:
            bh, bc = hr2, cr2
            nb, tsb = len(bh), len(bh) / 2.0

    # Resolution sweep
    for dn in list(range(1, 12)) + list(range(-1, -12, -1)):
        nt = n_base + dn
        if nt < 30 or nt > 80 or time.time() > dl_end - 80:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        t = np.linspace(0, 1, nt, endpoint=False)
        s = np.linspace(0, 1, n_base, endpoint=False)
        h_init = project(np.interp(t, s, bh), tst)
        best_nt, bc_nt = h_init.copy(), compute_c5(h_init, dxt)

        extra = make_starts(nt, tst, rng, ref=bh)
        for hs in [h_init] + extra[:3]:
            if time.time() > dl_end - 40:
                break
            h_r, c_r = opt_pipeline(hs, tst, dl_end - 20, rng)
            if c_r < bc_nt:
                best_nt, bc_nt = h_r, c_r
        if bc_nt < bc:
            bh, bc, n_base = best_nt, bc_nt, nt

    # Second perturbation round at best n
    nb, tsb = len(bh), len(bh) / 2.0
    for amp in (0.05, 0.03, 0.02, 0.01):
        if time.time() > dl_end - 60:
            break
        hp = project(bh + amp * rng.randn(nb), tsb)
        hr2, cr2 = opt_pipeline(hp, tsb, dl_end - 10, rng)
        if cr2 < bc:
            bh, bc = hr2, cr2

    # Final polish
    nb, tsb = len(bh), len(bh) / 2.0
    if time.time() < dl_end - 8:
        hf1, cf1 = slsqp_opt(bh, tsb, 400)
        if cf1 < bc:
            bh, bc = hf1, cf1
    nb, tsb = len(bh), len(bh) / 2.0
    if time.time() < dl_end - 5:
        hf2, cf2 = coord_descent(bh, tsb, dl_end - 2, rng, 50)
        if cf2 < bc:
            bh, bc = hf2, cf2
    nb, tsb = len(bh), len(bh) / 2.0
    if time.time() < dl_end - 3:
        hf3, cf3 = slsqp_opt(bh, tsb, 200)
        if cf3 < bc:
            bh, bc = hf3, cf3

    bh = project(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
