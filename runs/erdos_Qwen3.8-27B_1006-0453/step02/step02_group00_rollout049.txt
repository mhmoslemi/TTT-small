def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    val = m + np.log(np.sum(np.exp(shifted)) + 1e-30) / T
    return val * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

                   options={'maxiter': maxiter, 'ftol': ftol})

        h_p[s1:e1] = h_p[s2:e2]
        h_p[s2:e2] = tmp

        rng = np.random
    best = h.copy()
    best_c5 = compute_c5(best)

        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:

            best_c5 = c5_try
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -400, 0)

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h

    else:
        tot = h.sum()
        if tot < 1e-14:

        h += h * (diff / tot)

def slsqp(h0, T, n, maxiter=400, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    res = minimize(smooth_c5, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def fourier_perturb(h, n, rng, amplitude):
    h_p = h.copy()
    n_modes = rng.randint(1, 4)
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 3))
        phase = 2.0 * np.pi * rng.random()
        amp = amplitude * rng.uniform(0.4, 1.0)
        h_p += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return project_feasible(h_p, n)

def block_perturb(h, n, rng, n_blocks=3):
    h_p = h.copy()
    for _ in range(n_blocks):
        b1 = rng.randint(0, n)
        b2 = rng.randint(0, n)
        size = rng.randint(1, max(2, n // 6))
        s1, e1 = min(b1, n), min(b1 + size, n)
        s2, e2 = min(b2, n), min(b2 + size, n)
        if e1 - s1 != e2 - s2:
            size2 = e1 - s1
            e2 = min(b2 + size2, n)
            if s1 == e1 or s2 == e2:
                continue
        tmp = h_p[s1:e1].copy()
        h_p[s1:e1] = h_p[s2:e2]
        h_p[s2:e2] = tmp
    return project_feasible(h_p, n)

def de_perturb(pop, rng, F=0.8):
    idx = rng.choice(len(pop), size=3, replace=False)
    a, b, c = pop[idx[0]], pop[idx[1]], pop[idx[2]]
    return project_feasible(a + F * (b - c), len(a))

def swap_search(h, n, n_trials, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_trials):
        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue
        h_try = best.copy()
        h_try[i], h_try[j] = best[j], best[i]
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best = h_try
            best_c5 = c5_try
    return best, best_c5

def point_push(h, n, n_trials, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_trials):
        i = rng.randint(0, n)
        target = 0.0 if best[i] > 0.5 else 1.0
        frac = rng.uniform(0.3, 1.0)
        h_try = best.copy()
        h_try[i] = best[i] + frac * (target - best[i])
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best = h_try
            best_c5 = c5_try
    return best, best_c5

def targeted_adjust(h, n, rng, n_trials=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full')
    k_worst = np.argmax(corr)
    for _ in range(n_trials):
        i = rng.randint(0, n)
