import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_pnorm(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = np.sum(corr**p) ** (1.0/p)
    return val * dx

def constraint_fun(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

        if kind == 0:
            h = (rng.rand(n) < 0.5).astype(float)
        elif kind == 1:
            h = np.zeros(n)
            h[::2] = 1.0
        elif kind == 2:
            q = n // 4
            h = np.zeros(n)
            h[:q] = 1.0
            h[2*q:3*q] = 1.0
        elif kind == 3:
            h = rng.rand(n)
            h = 0.3 + 0.4 * np.abs(np.sin(2*np.pi*np.arange(n)/n * rng.randint(2,7)))
        else:
            h = warm_start.copy()
            h += 0.15 * rng.randn(n)
        pop.append(project_feasible(h, n))
    return pop

def bbo_gen(pop, scores, n, gen, max_gen, rng):
    P = len(pop)
    order = np.argsort(scores)
    ranks = np.empty(P, dtype=float)
    for r, idx in enumerate(order):
        ranks[idx] = r
    if P > 1:
        lam = ranks / (P - 1)
    else:
        lam = np.array([0.5])
    mu = 1.0 - lam
    mu_norm = mu / mu.sum()
    
    mu_rate = 0.12 * (1.0 - gen / max_gen) + 0.01
    mig_str = 0.6 * (1.0 - gen / max_gen) + 0.25
    
    dims = np.arange(n)
    new_pop = []
    for i in range(P):
        h_new = pop[i].copy()
        if mig_str > 0.01:
            mask = rng.rand(n) < mig_str
            if mask.any():
                sources = rng.choice(P, size=n, p=mu_norm)
                for d in dims[mask]:
                    if sources[d] != i:
                        h_new[d] = pop[sources[d]][d]
        n_mut = int(np.ceil(mu_rate * n))
        mut_idx = rng.choice(n, size=n_mut, replace=False)
        h_new[mut_idx] = rng.rand(n_mut)
        new_pop.append(project_feasible(h_new, n))
    return new_pop

def targeted_local_search(h, c5_best, n, max_iters, rng, time_limit, t0):
    best_h = h.copy()
    best_c5 = c5_best
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        worst_idx = np.argmax(corr)
        shift = worst_idx - (n - 1)
        improved = False
        n_tries = 12
        for _ in range(n_tries):
            j = rng.randint(n)
            for delta in [0.02, 0.04, 0.07, 0.1, -0.02, -0.04, -0.07, -0.1]:
                h_try = best_h.copy()
                h_try[j] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            for _ in range(6):
                j1 = rng.randint(n)
                j2 = rng.randint(n)
                if j1 == j2:
                    continue
                for d1 in [0.03, -0.03]:
                    for d2 in [-0.03, 0.03]:
                        if d1 + d2 != 0:
                            continue
                        h_try = best_h.copy()
                        h_try[j1] += d1
                        h_try[j2] += d2
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-13:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved and it > 60:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    n = 120
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP T-schedule polish
    for T in [30, 100, 400, 1500, 5000]:
        if elapsed() > budget_s * 0.10:
            break
        try:
            hc = slsqp_optimize(best_h, T, n, maxiter=500, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    
    # Phase 2: p-norm optimization (different smoothing)
    for p in [8, 16, 32, 64]:
        if elapsed() > budget_s * 0.18:
            break
        try:
            hc = pnorm_optimize(best_h, p, n, maxiter=300, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    
    # Phase 3: BBO population evolution
    pop_size = 16
    max_gen = 70
    pop = bbo_initial(pop_size, n, seed, best_h)
    scores = np.array([compute_c5(h) for h in pop])
    bbo_best_c5 = scores.min()
    bbo_best_h = pop[scores.argmin()].copy()
    if bbo_best_c5 < best_c5:
        best_c5 = bbo_best_c5
        best_h = bbo_best_h.copy()
    
    stagnant = 0
    for gen in range(max_gen):
        if elapsed() > budget_s * 0.52:
            break
        pop = bbo_gen(pop, scores, n, gen, max_gen, rng)
        scores = np.array([compute_c5(h) for h in pop])
        gen_best = scores.min()
        if gen_best < bbo_best_c5 - 1e-11:
            bbo_best_c5 = gen_best
            bbo_best_h = pop[scores.argmin()].copy()
            if bbo_best_c5 < best_c5:
                best_c5 = bbo_best_c5
                best_h = bbo_best_h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 6 == 0:
                worst = scores.argmax()
                pop[worst] = project_feasible(best_h + 0.2 * rng.randn(n), n)
            if stagnant > 35:
                break
    
    # SLSQP polish top-3 BBO solutions
    top3_order = np.argsort(scores)[:3]
    for idx in top3_order:
        if elapsed() > budget_s * 0.62:
            break
        for T in [800, 4000]:
            if elapsed() > budget_s * 0.62:
                break
            try:
                hc = slsqp_optimize(pop[idx], T, n, maxiter=500, ftol=1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    pop[idx] = hc
            except Exception:
                pass
    
    # Phase 4: Targeted local search
    if elapsed() < budget_s * 0.75:
        best_h, best_c5 = targeted_local_search(
            best_h, best_c5, n, 300, rng, budget_s * 0.75, t0)
    
    # Phase 5: Higher resolution attempts
    for n_try in [150, 180, 240]:
        if elapsed() > budget_s * 0.88:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [300, 1500]:
            if elapsed() > budget_s * 0.88:
                break
            try:
                hc = slsqp_optimize(h_interp, T, n_try, maxiter=400, ftol=1e-13)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass
    
    # Phase 6: Final multi-T SLSQP polish
    n_cur = len(best_h)
    for T in [2000, 10000, 40000]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_optimize(best_h, T, n_cur, maxiter=1200, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    
    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
