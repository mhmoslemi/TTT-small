                    best_c5 = c5_try
                    improved = True
                    corr = np.correlate(best, 1.0 - best, mode='full')
                    break
            if improved:
                break
        if not improved:
            break
    return best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Initialize from parent or default
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP warmup on initial
    n_init = len(best_h)
    for T in [100, 500, 2000]:
        if elapsed() > budget_s * 0.06:
            break
        hc = slsqp_optimize(best_h.copy(), T, n_init, maxiter=400)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 2: Differential Evolution (global search) - n=60 for speed
    n_de = 60
    if elapsed() < budget_s * 0.40:
        bnds_de = [(0.0, 1.0)] * n_de
        # Add initial projected to n_de as a seed point via init
        if n_init != n_de:
            x_old = np.linspace(0, 2, n_init, endpoint=False)
            x_new = np.linspace(0, 2, n_de, endpoint=False)
            h_seed = project_feasible(np.interp(x_new, x_old, best_h), n_de)
        else:
            h_seed = best_h.copy()
        try:
            de_res = differential_evolution(
                de_objective, bnds_de, args=(n_de,),
                maxiter=250, popsize=18, tol=1e-10,
                mut=(0.4, 0.9), recomb=0.85, seed=seed,
                init='latinhypercube', polish=False,
                updating='deferred', workers=1
            )
            h_de = project_feasible(np.clip(de_res.x, 0, 1), n_de)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_c5, best_h = c5_de, h_de
        except Exception:
            pass

    # Phase 3: SLSQP multi-temperature refinement
    n_cur = len(best_h)
    T_refine = [40, 150, 500, 1500, 5000, 15000]
    for T in T_refine:
        if elapsed() > budget_s * 0.65:
            break
        hc = slsqp_optimize(best_h.copy(), T, n_cur, maxiter=700, ftol=1e-15)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 4: Basin hopping (compact, SLSQP per step)
    if elapsed() < budget_s * 0.82:
        bh_h = best_h.copy()
        bh_c5 = best_c5
        n_bh = len(bh_h)
        n_iter = 35
        for it in range(n_iter):
            if elapsed() > budget_s * 0.82:
                break
            step = 0.10 * (1.0 - 2.0 * it / n_bh) + 0.02
            h_pert = bh_h + step * np.random.randn(n_bh)
            h_pert = project_feasible(h_pert, n_bh)
            try:
                hc = slsqp_optimize(h_pert, 600, n_bh, maxiter=250, ftol=1e-12)
                c5 = compute_c5(hc)
                temp = 0.006 * (1.0 - it / float(n_iter)) + 0.0004
                if c5 < bh_c5 or np.random.rand() < np.exp(-(c5 - bh_c5) / max(temp, 1e-8)):
                    bh_c5, bh_h = c5, hc
                    if c5 < best_c5:
                        best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 5: Coordinate descent polish
    if elapsed() < budget_s * 0.90:
        n_cur = len(best_h)
        best_h = coordinate_descent(best_h, n_cur, max_rounds=30)
        best_c5 = compute_c5(best_h)

    # Phase 6: Try higher resolution if time allows
    if elapsed() < budget_s * 0.95:
        for n_try in [len(best_h) + 20, len(best_h) + 40]:
            if elapsed() > budget_s * 0.95:
                break
            if n_try > 300:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_try = project_feasible(h_interp + 0.02 * np.random.randn(n_try), n_try)
            hc = slsqp_optimize(h_try, 800, n_try, maxiter=500)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_c5, best_h = c5, hc

    # Phase 7: Final ultra-high-T SLSQP polish
    n_cur = len(best_h)
    for T in [30000, 100000]:
        if elapsed() > budget_s * 0.995:
            break
        hc = slsqp_optimize(best_h.copy(), T, n_cur, maxiter=1000, ftol=1e-16)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
