import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def obj_topk(h, t, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -topk)[-topk:]
    c = corr[idx]
    mx = np.max(c)
    s = t * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    if n % nd == 0:
        return proj(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

    return proj(h, n)

def slsqp_opt(h, n_pts, t_val, topk, maxiter, ftol, bnds):
    corr_len = 2 * n_pts - 1
    if topk is not None and topk < corr_len:
        r = minimize(obj_topk, h, args=(t_val, topk), method='SLSQP',
                     bounds=bnds, constraints={'type': 'eq', 'fun': ceq, 'jac': ceqj},

                     bounds=bnds, constraints={'type': 'eq', 'fun': ceq, 'jac': ceqj},
                     options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n_pts)
