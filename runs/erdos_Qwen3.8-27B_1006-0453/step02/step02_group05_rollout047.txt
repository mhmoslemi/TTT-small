Let me also add: after the main optimization, do a "polish" where I repeatedly:
1. Find k* = argmax(corr)
2. Compute grad of corr[k*]
3. Take a small step: h -= alpha * grad (where alpha is small)
4. Project
5. Repeat until no improvement

This is like steepest descent on the exact (not soft) max, which should give the final polish.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad(h, N, weights, center):
    """Compute gradient given precomputed weights over correlation indices."""
    grad = np.zeros(N)
    active = np.where(weights > 1e-6)[0]
    for ai in active:
        L = ai - center
        wi = weights[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def fista_phase(h, N, target, dx, deadline, lr_init, kind, param, lr_decay):
    """FISTA with Nesterov momentum. kind='soft' (param=tau) or 'topm' (param=m)."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    y = h.copy()
    h_prev = h.copy()
    t_val = 1.0
    lr = lr_init
    center = N - 1
    it = 0
    while time.time() < deadline:
        corr = get_corr(y)
        if kind == 'soft':
            sc = param * corr
            sc -= sc.max()
            w = np.exp(sc)
            w /= w.sum()
        else:
            m = int(param)
            idx = np.argsort(corr)[-m:]
            w = np.zeros(len(corr))
            w[idx] = 1.0 / m
        grad = compute_grad(y, N, w, center)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        if it > 0:
            y_mom = h + (t_val - 1.0) / t_new * (h - h_prev)
        else:
            y_mom = h.copy()
        h_new = y_mom - lr * grad
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        y = h
        t_val = t_new
        it += 1
        if it % 300 == 0:
            lr *= lr_decay
        if lr < 1e-10:
            break
    return best_h, best_c5

def polish_step(h, N, target, dx, alpha):
    """One steepest-descent step on the exact max shift."""
    corr = get_corr(h)
    k_star = np.argmax(corr)
    L = k_star - (N - 1)
    g = np.zeros(N)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] = (1.0 - h[:N - L])
        g[:N - L] -= h[L:]
    else:
        M = -L
        g[:N - M] = (1.0 - h[M:])
        g[M:] -= h[:N - M]
    h_new = h - alpha * g
    h_new = project_sum(h_new, target)
    c5_new = compute_c5(h_new, dx)
    c5_old = compute_c5(h, dx)
    if c5_new < c5_old:
        return h_new, c5_new, True
    return h, c5_old, False

def local_search(h, N, target, dx, deadline, rng):
    """Focused local search: single-worst-shift steepest descent + random pairs."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    # Phase A: steepest descent on worst shift
    alpha = 0.005
    for _ in range(2000):
        if time.time() > deadline:
            break
        h, c5, improved = polish_step(h, N, target, dx, alpha)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            alpha *= 0.5
            if alpha < 1e-5:
                break
    # Phase B: random pair transfers
    alpha_b = 0.03
    for _ in range(3000):
        if time.time() > deadline:
            break
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j:
            continue
        if h[i] < h[j] - 1e-6:
            eps = min(alpha_b, h[j] - h[i]) * 0.5
        elif h[j] < h[i] - 1e-6:
            eps = min(alpha_b, h[i] - h[j]) * 0.5
        else:
            continue
        h2 = h.copy()
        if h[i] < h[j]:
            h2[i] += eps
            h2[j] -= eps
        else:
            h2[j] += eps
            h2[i] -= eps
        c5 = compute_c5(h2, dx)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h2.copy()
            h = h2
    return best_h, best_c5
