```python
def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    c_max = np.max(corr)
    shifted = corr - c_max
    shifted = np.clip(shifted, -100*T, 0)
    exp_vals = np.exp(shifted / T)
    w = exp_vals / np.sum(exp_vals)  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    # For each j, grad[j] = dx * sum_k w[k]*(1-h[j-k])*I[valid] - dx * sum_k w[k]*h[j+k]*I[valid]
    # k ranges from -(n-1) to n-1, indexed as k_idx = k + (n-1)
    
    for j in range(n):
        s = 0.0
        for k_idx in range(2*n-1):
            k = k_idx - (n-1)
            # First term: (1-h[j-k]) if 0<=j-k<=n-1
            ik = j - k
            if 0 <= ik < n:
                s += w[k_idx] * (1.0 - h[ik])
            # Second term: -h[j+k] if 0<=j+k<=n-1
            jk = j + k
            if 0 <= jk < n:
                s -= w[k_idx] * h[jk]
        grad[j] = dx * s
    return grad
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    shifted = np.clip(shifted, -100.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        # Subgradient of max: uniform weight on argmax
        w = np.zeros(2*n-1)
        w[corr == c_max] = 1.0
        w /= w.sum()
    else:
        shifted = np.clip(corr - c_max, -100.0*T, 0.0)
        w = np.exp(shifted / T)
        w /= w.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        # First term: sum_k w[k_idx] * (1-h[j-k]) for valid j-k
        k_min1 = max(-(n-1), 0-j)  # j-k >= 0 => k <= j
        k_max1 = min(n-1, n-1-j)   # j-k <= n-1 => k >= j-(n-1)
        # Wait: 0 <= j-k <= n-1 means j-(n-1) <= k <= j
        k_min1 = max(-(n-1), j-(n-1))
        k_max1 = min(n-1, j)
        s1 = 0.0
        if k_min1 <= k_max1:
            ks = np.arange(k_min1, k_max1+1)
            start = k_min1 + (n-1)
            s1 = np.dot(w[start:start+len(ks)], one_minus_h[j-ks])
        
        # Second term: sum_k w[k_idx] * h[j+k] for valid j+k
        # 0 <= j+k <= n-1 means -j <= k <= n-1-j
        k_min2 = max(-(n-1), -j)
        k_max2 = min(n-1, n-1-j)
        s2 = 0.0
        if k_min2 <= k_max2:
            ks = np.arange(k_min2, k_max2+1)
            start = k_min2 + (n-1)
            s2 = np.dot(w[start:start+len(ks)], h[j+ks])
        
        grad[j] = dx * (s1 - s2)
    return grad

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def sum_con_jac(h, n):
    return np.ones(n)

def fix_sum(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

            soft_max_obj, h.copy(), args=(T,),
            jac=lambda x: soft_max_grad(x, T),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,

        if hasattr(res, 'x') and res.x is not None:

        pass
    return h

def do_lbfgsb(h, n, T, max_iter=500, penalty=50.0):

        return soft_max_obj(x, T) + penalty * (np.sum(x) - n*0.5)**2

        new = cur - lr * g
        new = np.clip(new, 0.0, 1.0)

    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
```

```python
    else:
        shifted = np.clip(corr - c_max, -100.0 * T, 0.0)
        w = np.exp(shifted / T)
        w /= w.sum()
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        k_min1 = max(-(n - 1), j - (n - 1))
        k_max1 = min(n - 1, j)
        s1 = 0.0
        if k_min1 <= k_max1:
            ks = np.arange(k_min1, k_max1 + 1)
            st = k_min1 + (n - 1)
            s1 = np.dot(w[st:st + len(ks)], one_minus_h[j - ks])
        k_min2 = max(-(n - 1), -j)
        k_max2 = min(n - 1, n - 1 - j)
        s2 = 0.0
        if k_min2 <= k_max2:
            ks = np.arange(k_min2, k_max2 + 1)
            st = k_min2 + (n - 1)
            s2 = np.dot(w[st:st + len(ks)], h[j + ks])
        grad[j] = dx * (s1 - s2)
    return grad

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def sum_con_jac(h, n):
    return np.ones(n)

def fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

def do_slsqp_grad(h, n, T, max_iter=2000):
    try:
        res = opt.minimize(
            soft_max_obj, h.copy(), args=(T,),
            jac=lambda x: soft_max_grad(x, T),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,), 'jac': sum_con_jac}],
            options={'maxiter': max_iter, 'ftol': 1e-16}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_lbfgsb(h, n, T, max_iter=500, penalty=200.0):
    def obj_pen(x):
        return soft_max_obj(x, T) + penalty * (np.sum(x) - n * 0.5) ** 2
    def grad_pen(x):
        g = soft_max_grad(x, T)
        g += 2.0 * penalty * (np.sum(x) - n * 0.5)
        return g
    try:
        res = opt.minimize(obj_pen, h.copy(), jac=grad_pen,
                          method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def projected_grad(h, n, T, lr_init, n_steps=300):
    cur = h.copy()
    lr = lr_init
    best_c = compute_c5(cur)
    for _ in range(n_steps):
        g = soft_max_grad(cur, T)
        new = np.clip(cur - lr * g, 0.0, 1.0)
        new = fix_sum(new, n)
        c_new = compute_c5(new)
        if c_new < best_c:
            cur = new
            best_c = c_new
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return cur

def active_shift_opt(h, n, T, top_k, max_iter=1500):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is None:
        top_k = np.argsort(corr)[-10:]
    dx = 2.0 / n
    def as_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        tv = c[top_k]
        tm = np.max(tv)
        sh = np.clip(tv - tm, -100 * T, 0)
        s = np.sum(np.exp(sh / T))
        return (tm + T * np.log(max(s, 1e-300))) * dx
    def as_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        tv = c[top_k]
        tm = np.max(tv)
        sh = np.clip(tv - tm, -100 * T, 0)
        wt = np.exp(sh / T)
        wt /= wt.sum()
        g = np.zeros(n)
        omh = 1.0 - x
        for j in range(n):
            s1 = 0.0
            s2 = 0.0
            for ii, ki in enumerate(top_k):
                k = ki - (n - 1)
                ik = j - k
                if 0 <= ik < n:
                    s1 += wt[ii] * omh[ik]
                jk = j + k
                if 0 <= jk < n:
                    s2 += wt[ii] * x[jk]
            g[j] = dx * (s1 - s2)
        return g
    try:
        res = opt.minimize(as_obj, h.copy(), jac=as_grad,
                          method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,), 'jac': sum_con_jac}],
                          options={'maxiter': max_iter, 'ftol': 1e-15})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def resample_h(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(x_new, x_old, h, period=2.0), new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    def try_update(hc, nc):
        nonlocal h_best, n_best, c5_best
        hc = fix_sum(hc, nc)
        cv = compute_c5(hc)
        if cv < c5_best - 1e-15:
            h_best = hc
            n_best = nc
            c5_best = cv
            return True
        return False

    # Phase 1: SLSQP with analytic gradient, temperature sweep
    T_seq1 = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.00015]
    for T in T_seq1:
        if budget_s - elapsed() < 120:
            break
        max_it = 4000 if T > 0.01 else 3000
        h_opt = do_slsqp_grad(h_best, n_best, T, max_iter=max_it)
        try_update(h_opt, n_best)

    # Phase 2: L-BFGS-B with penalty (different optimizer landscape)
    for T in [0.01, 0.004, 0.001, 0.0003]:
        if budget_s - elapsed() < 100:
            break
        h_opt = do_lbfgsb(h_best, n_best, T, max_iter=600, penalty=300.0)
        try_update(h_opt, n_best)

    # Phase 3: Projected gradient descent for fine-tuning
    if budget_s - elapsed() > 80:
        h_pg = projected_grad(h_best, n_best, 0.001, lr_init=0.5, n_steps=500)
        try_update(h_pg, n_best)
        h_pg2 = projected_grad(h_best, n_best, 0.0003, lr_init=0.2, n_steps=400)
        try_update(h_pg2, n_best)

    # Phase 4: Active shift refinement (top 10, 25, 50)
    for tk_size in [10, 25, 50]:
        if budget_s - elapsed() < 100:
            break
        h_ref = active_shift_opt(h_best, n_best, 0.001, None if tk_size == 10 else np.argsort(
            np.correlate(h_best, 1.0 - h_best, mode='full'))[-tk_size:], max_iter=2000)
        try_update(h_ref, n_best)
        h_ref2 = active_shift_opt(h_best, n_best, 0.0003, None if tk_size == 10 else np.argsort(
            np.correlate(h_best, 1.0 - h_best, mode='full'))[-tk_size:], max_iter=1500)
        try_update(h_ref2, n_best)

    # Phase 5: Multi-start with structured perturbations
    for pert_type in range(8):
        if budget_s - elapsed() < 180:
            break
        scale = 0.03 + 0.02 * pert_type
        if pert_type < 3:
            h_pert = h_best + np.random.randn(n_best) * scale
        elif pert_type < 6:
            freq = 3 + 2 * (pert_type - 3)
            phase = np.random.uniform(0, 2 * np.pi)
            h_pert = h_best + scale * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
        else:
            block = n_best // 8
            h_pert = h_best + scale * (np.random.randn(n_best // block) + 0.5)
            h_pert = np.repeat(np.clip(h_pert, 0, 1), block)[:n_best]
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = fix_sum(h_pert, n_best)
        h_o1 = do_slsqp_grad(h_pert, n_best, 0.01, max_iter=1500)
        h_o2 = do_slsqp_grad(h_o1, n_best, 0.001, max_iter=2000)
        try_update(h_o2, n_best)

    # Phase 6: Alternative resolutions
    for n_alt in [480, 500, 600]:
        if budget_s - elapsed() < 200:
            break
        h_init = resample_h(h_best, n_alt)
        for T in [0.02, 0.005, 0.001, 0.0003]:
            if budget_s - elapsed() < 120:
                break
            h_opt = do_slsqp_grad(h_init, n_alt, T, max_iter=2500)
            if try_update(h_opt, n_alt):
                h_init = h_best.copy() if len(h_best) != n_alt else resample_h(h_best, n_alt)

    # Phase 7: Downsample high-res best to try 360
    if len(h_best) > 360:
        if budget_s - elapsed() > 100:
            h_ds = resample_h(h_best, 360)
            h_opt = do_slsqp_grad(h_ds, 360, 0.001, max_iter=3000)
            try_update(h_opt, 360)

    # Phase 8: Final SLSQP polish at best resolution
    for T in [0.0005, 0.0002, 0.00008]:
        if budget_s - elapsed() < 50:
            break
        h_opt = do_slsqp_grad(h_best, n_best, T, max_iter=5000)
        try_update(h_opt, n_best)

    # Phase 9: Final projected gradient micro-refinement
    if budget_s - elapsed() > 30:
        h_fg = projected_grad(h_best, n_best, 0.0002, lr_init=0.1, n_steps=300)
        try_update(h_fg, n_best)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
```
