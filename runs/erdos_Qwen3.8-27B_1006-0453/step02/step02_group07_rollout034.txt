def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    n = len(h)

    lo = h.min() - 1.0

    cm = c.max()

    g_rev = g[::-1]
    W = np.fft.rfft(w, N)
    G = np.fft.rfft(g_rev, N)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_h(h):
    n = len(h)
    target = 0.5 * n
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_val(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * (2.0 / n)
    cm = c.max()
    return float(T * np.log(np.sum(np.exp((c - cm) / T))))

def grad_fft(h, w, n, N):
    g = 1.0 - 2.0 * h
    g_rev = g[::-1]
    W = np.fft.rfft(w, N)
    G = np.fft.rfft(g_rev, N)
    conv_full = np.fft.irfft(W * G, N)
    return (2.0 / n) * conv_full[n - 1:2 * n - 1]

def pxg_run(h, n, N, T_sched, t0, t_end, base_lr=0.4):
    for T, n_iter, lr_scale in T_sched:
        lr = base_lr * lr_scale
        f_prev = lse_val(h, n, T)
        stall = 0
        for it in range(n_iter):
            if time.time() - t0 > t_end:
                return h
            corr = np.correlate(h, 1.0 - h, mode='full')
            c = corr * (2.0 / n)
            cm = c.max()
            w_exp = np.exp((c - cm) / T)
            ws = w_exp.sum()
            if ws < 1e-30:
                break
            w = w_exp / ws
            g = grad_fft(h, w, n, N)
            h_new = project_h(h - lr * g)
            f_new = lse_val(h_new, n, T)
            if f_new < f_prev - 1e-12:

                f_prev = f_new

                hn = h.copy()
                hn[j] = np.clip(h[j] + d, 0.0, 1.0)
                hn = project_h(hn)
                c = compute_c5(hn)
                if c < bc - 1e-13:
                    bc = c
                    bh = hn
            h = bh
