- Bisection: the derivative of max_k (c_k + δ*d_k) w.r.t. δ is max over active k of d_k
  - If all active d_k > 0: increase δ makes it worse → optimal δ is at lower bound
  - If all active d_k < 0: decrease δ makes it worse → optimal δ is at upper bound
  - Otherwise: the optimal δ is where the max "flips" between different k's

Actually for a simple bisection:

- f'(δ) = max_{k: c_k + δ*d_k = max} d_k (subgradient)
- We want f'(δ) = 0, i.e., the max is achieved by k's with both positive and negative d_k
- Or at a boundary of δ

This is like finding the Chebyshev center in 1D. Simple bisection works:
- If at current δ, all k achieving the max have d_k > 0: we should decrease δ
- If all have d_k < 0: increase δ
- If mixed: we're at (or near) the optimum

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr) * (2.0 / N))
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1

        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

        if improved:

            c = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(c) * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    sizes = sorted(set([n0, 2 * n0, 3 * n0]))
    sizes = [s for s in sizes if s % 2 == 0 and s <= 1200]

    for N in sizes:
        if time.time() > deadline - 120:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)

        # Phase 1: Adam broad
        p1_dead = time.time() + min(180, (deadline - time.time()) * 0.35)
        h1, c1 = adam_opt(h_init.copy(), N, target, dx, p1_dead,
                          tau0=1.0, tau1=50.0, lr0=0.04, decay=0.98)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: Adam fine
        if time.time() < deadline - 80:
            p2_dead = time.time() + min(150, (deadline - time.time()) * 0.3)
            h2, c2 = adam_opt(h1.copy(), N, target, dx, p2_dead,
                              tau0=50.0, tau1=500.0, lr0=0.01, decay=0.985)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

        # Phase 3: Adam very fine
        if time.time() < deadline - 60:
            p3_dead = time.time() + min(100, (deadline - time.time()) * 0.25)
            h3, c3 = adam_opt(h2.copy(), N, target, dx, p3_dead,
                              tau0=500.0, tau1=2000.0, lr0=0.003, decay=0.99)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()

        # Phase 4: CD polish
        if time.time() < deadline - 30:
            cd_dead = time.time() + min(60, deadline - time.time() - 15)
            h4, c4 = cd_polish(h3.copy(), N, target, cd_dead)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

        # Random restart at same size
        if time.time() < deadline - 80:
            h_r = h_init + rng.randn(N) * 0.05
            h_r = project_sum(h_r, target)
            p4_dead = time.time() + min(60, deadline - time.time() - 30)
            h5, c5 = adam_opt(h_r, N, target, dx, p4_dead,
                              tau0=1.0, tau1=300.0, lr0=0.025, decay=0.985)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h5.copy()
            if time.time() < deadline - 20:
                h6, c6 = cd_polish(h5.copy(), N, target, deadline - 15, max_sweeps=200)
                if c6 < best_c5:
                    best_c5 = c6
                    best_h = h6.copy()

    # Final polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 2:
        h_f, c_f = cd_polish(best_h.copy(), Nb, Nb / 2.0, deadline - 1, max_sweeps=300)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
