    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = np.sum(interior)
    if ni > 0 and abs(diff) < ni * 0.3:
        h[interior] += diff / ni
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add; diff -= add
                if diff <= 1e-12: break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem; diff -= rem
                if diff <= 1e-12: break
    return h

def compute_subgrad(h, active_k, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_k:
        m = k - (n - 1)
        if m >= 0:
            nf = n - m
            grad[:nf] += (1.0 - h[m:])
            grad[m:] += (-h[:nf])
        else:
            ma = -m
            nf = n - ma
            grad[ma:] += (1.0 - h[:nf])
            grad[:nf] += (-h[ma:])
    grad /= (len(active_k) * dx)
    return grad

def subgrad_opt(h, n, n_it, lr0, decay, tk):
    h = h.copy()
    for it in range(n_it):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-tk:]
        g = compute_subgrad(h, idx, n)
        lr = lr0 * (decay ** it)
        h = h - lr * g
        h = np.clip(h, 0.0, 1.0)
        h = fix_sum(h, n)
    return h

def slsqp_polish(h, n, T, max_it):
    dx = 2.0 / n
    def fobj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = np.max(c)
        sh = np.clip(c - cm, -80.0 * T, 0.0)
        return (cm + T * np.log(max(np.sum(np.exp(sh / T)), 1e-300))) * dx
    def fcon(x):
        return np.sum(x) - n * 0.5
    try:
        r = opt.minimize(fobj, h.copy(), method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': fcon}],
                         options={'maxiter': max_it, 'ftol': 1e-16})
        if r.x is not None:
            return fix_sum(r.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def el(): return time.time() - t0

    h_best = np.array(initial_h_values, dtype=float).copy()
    n = len(h_best)
    h_best = fix_sum(h_best, n)
    c5_best = compute_c5(h_best)

    def upd(hc, nc):
        nonlocal h_best, c5_best
        hc = fix_sum(hc, nc)
        cc = compute_c5(hc)
        if cc < c5_best - 1e-14:
            h_best = hc; c5_best = cc; return True
        return False

    # Phase 1: Subgradient descent - anneal top_k from wide to narrow
    anneal_cfgs = [(3000, 1.2, 0.9996, 20), (3000, 0.8, 0.9996, 8),
                   (3000, 0.6, 0.9995, 5), (3000, 0.5, 0.9994, 3),
                   (4000, 0.4, 0.9995, 2), (4000, 0.3, 0.9996, 1)]
    for ni, lr, dc, tk in anneal_cfgs:
        if budget_s - el() < 50: break
        h_t = subgrad_opt(h_best, n, ni, lr, dc, tk)
        upd(h_t, n)

    # Phase 2: SLSQP soft-max polish
    for T in [0.008, 0.003, 0.001, 0.0004, 0.00015]:
        if budget_s - el() < 70: break
        h_t = slsqp_polish(h_best, n, T, 3000)
        upd(h_t, n)

    # Phase 3: Alternating subgradient with different top_k
    for tk in [1, 2, 4, 7, 12, 20, 50]:
        if budget_s - el() < 40: break
        h_t = subgrad_opt(h_best, n, 2500, 0.35, 0.9996, tk)
        upd(h_t, n)
        if budget_s - el() < 40: break
        h_t = slsqp_polish(h_best, n, 0.0005, 2000)
        upd(h_t, n)

    # Phase 4: Multi-start perturbations
    scales = [0.01, 0.03, 0.06, 0.10, 0.15, 0.20]
    for sc in scales:
        if budget_s - el() < 100: break
        hp = h_best + np.random.randn(n) * sc
        hp = fix_sum(hp, n)
        h_t = subgrad_opt(hp, n, 2000, 0.6, 0.9995, 5)
        h_t = slsqp_polish(h_t, n, 0.001, 2000)
        upd(h_t, n)

    # Phase 5: Try different resolutions
    for nt in [200, 450, 600]:
        if budget_s - el() < 120: break
        xo = np.arange(n) * (2.0 / n)
        xn = np.arange(nt) * (2.0 / nt)
        hi = np.interp(xn, xo, h_best, period=2.0)
        hi = fix_sum(hi, nt)
        h_t = subgrad_opt(hi, nt, 2000, 0.5, 0.9995, 5)
        h_t = slsqp_polish(h_t, nt, 0.001, 2500)
        if upd(h_t, nt):
            n = nt
            h_best = hi  # keep consistent (actually h_best already updated in upd)

    # Phase 6: Back to best n, fine polish
    if n != len(h_best):
        n = len(h_best)
    if budget_s - el() > 80:
        h_t = subgrad_opt(h_best, n, 5000, 0.15, 0.9998, 1)
        upd(h_t, n)
    if budget_s - el() > 60:
        h_t = slsqp_polish(h_best, n, 0.0002, 4000)
        upd(h_t, n)
    if budget_s - el() > 50:
        h_t = subgrad_opt(h_best, n, 3000, 0.1, 0.9999, 2)
        upd(h_t, n)
    if budget_s - el() > 40:
        h_t = slsqp_polish(h_best, n, 0.0001, 4000)
        upd(h_t, n)

    # Phase 7: Micro-perturbation final polish
    if budget_s - el() > 50:
        for sc in [0.003, 0.008]:
            if budget_s - el() < 30: break
            hm = h_best + np.random.randn(n) * sc
            hm = fix_sum(hm, n)
            h_t = subgrad_opt(hm, n, 2000, 0.2, 0.9997, 1)
            h_t = slsqp_polish(h_t, n, 0.0003, 3000)
            upd(h_t, n)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
