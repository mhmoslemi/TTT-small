                ht = proj(h - lr * f * g, ts)
                ct = compute_c5(ht, dx)
                if ct < bc - 1e-15:
                    bh, bc, h, accepted = ht.copy(), ct, ht, True
                    m = f * 0.85 * m
                    break
        if accepted:
            no_imp = 0
            beta = min(beta * 1.015, 600.0)
        else:
            no_imp += 1
            if no_imp > 50:
                beta = max(beta * 0.9, 5.0)
            if no_imp > 300:
                break
    return bh, bc

def coord_descent(h0, max_rounds=200, dl=None, rng=None):
    n = len(h0)
    ts = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    bc = compute_c5(h, dx)
    steps = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.08, -0.08, 0.005, -0.005]
    for r in range(max_rounds):
        if dl is not None and time.time() > dl:
            break
        imp = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for j in order:
            sv = h[j]
            best_v, best_c = sv, bc
            for d in steps:
                nv = sv + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h[j] = nv
                h_p = proj(h, ts)
                c = compute_c5(h_p, dx)
                if c < best_c:
                    best_v, best_c = nv, c
            if best_v != sv:
                h = proj(np.array([h[j] if k != j else j for k, j in enumerate(h)]), ts)
                h = np.array([best_v if k == j else h[k] for k in range(n)])
                h = proj(h, ts)
                bc = best_c
                imp = True
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_end = t_start + budget_s - 3.0
    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = np.full(50, 0.5)
    n0 = len(hr)
    ts0 = n0 / 2.0
    dx0 = 2.0 / n0
    h0 = proj(hr, ts0)
    bh, bc = h0.copy(), compute_c5(h0, dx0)

    # Phase 1: SLSQP
    if time.time() < t_end - 80:
        h1 = slsqp_min(bh, maxiter=600, ftol=1e-14)
        c1 = compute_c5(h1, dx0)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 2: Nesterov momentum on smooth approx
    if time.time() < t_end - 100:
        h2, c2 = nesterov_gd(bh, max_iter=4000, dl=t_end - 40)
        if c2 < bc:
            bh, bc = h2, c2

    # Phase 3: Coordinate descent polish
    if time.time() < t_end - 30:
        h3, c3 = coord_descent(bh, max_rounds=300, dl=t_end - 10, rng=rng)
        if c3 < bc:
            bh, bc = h3, c3

    # Phase 4: SLSQP polish
    if time.time() < t_end - 15:
        h4 = slsqp_min(bh, maxiter=400, ftol=1e-14)
        c4 = compute_c5(h4, 2.0 / len(b4) if False else 2.0 / len(h4))
        if c4 < bc:
            bh, bc = h4, c4

    # Phase 5: Try nearby resolutions
    nb = len(bh)
    for dn in (-2, -1, 1, 2, -3, 3, -4, 4, -5, 5):
        nt = nb + dn
        if nt < 20 or nt > 130 or time.time() > t_end - 50:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        xi = np.linspace(0, 1, nb, endpoint=False)
        xo = np.linspace(0, 1, nt, endpoint=False)
        hi = proj(np.interp(xo, xi, bh), tst)
        ht = slsqp_min(hi, maxiter=350, ftol=1e-14)
        ct = compute_c5(ht, dxt)
        if ct < bc and time.time() < t_end - 25:
            ht2, ct2 = nesterov_gd(ht, max_iter=2000, dl=t_end - 15)
            if ct2 < ct:
                ht, ct = ht2, ct2
        if ct < bc:
            bh, bc = ht, ct
            nb = len(bh)

    # Phase 6: Perturbation restarts
    if time.time() < t_end - 50:
        nb = len(bh)
        tsb, dxb = nb / 2.0, 2.0 / nb
        for amp in (0.12, 0.06, 0.03, 0.015):
            if time.time() > t_end - 20:
                break
            hp = proj(bh + amp * rng.randn(nb), tsb)
            h_r = slsqp_min(hp, maxiter=300, ftol=1e-13)
            c_r = compute_c5(h_r, dxb)
            if c_r < bc:
                bh, bc = h_r, c_r

    # Final polish
    if bh is not None:
        nb = len(bh)
        dxb = 2.0 / nb
        tsb = nb / 2.0
        hf = slsqp_min(bh, maxiter=300, ftol=1e-14)
        cf = compute_c5(hf, dxb)
        if cf < bc:
            bh, bc = hf, cf
        hd, cd = coord_descent(bh, max_rounds=80, dl=t_end - 1, rng=rng)
        if cd < bc:
            bh, bc = hd, cd

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
