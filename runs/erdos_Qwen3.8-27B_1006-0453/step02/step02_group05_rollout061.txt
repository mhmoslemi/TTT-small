import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, N, weights):
    """Compute weighted gradient. weights has same length as correlate output (2N-1)."""
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-10
    active = np.where(weights > thresh)[0]
    for ai in active:
        L = ai - center
        wi = weights[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            if L < N:
                grad[:N-L] += wi * (1.0 - h[L:])
                grad[L:] -= wi * h[:N-L]
        else:
            M = -L
            if M < N:
                grad[M:] += wi * (1.0 - h[:N-M])
                grad[:N-M] -= wi * h[M:]
    return grad

def get_weights(corr, method, param):
    """Get lag weights. method: 'topK' with param=K, or 'softmax' with param=tau."""
    c5 = float(np.max(corr))
    if method == 'topK':
        K = min(param, len(corr))
        idx = np.argsort(corr)[-K:]
        w = np.zeros(len(corr))
        w[idx] = 1.0 / K
        return w, c5
    else:  # softmax
        tau = param
        sc = tau * corr - tau * corr.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w /= ws
        return w, c5

def gradient_descent(h, N, target, method, param, lr0, lr1, n_iter, deadline, smoothing=0.0):
    """Cosine-annealed gradient descent with Adam."""
    dx = 2.0 / N
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    
    while it < n_iter and time.time() < deadline:
        frac = it / max(n_iter - 1, 1)
        # Cosine annealing
        lr = lr1 + 0.5 * (lr0 - lr1) * (1 + np.cos(np.pi * frac))
        # Ramp parameter
        if method == 'topK':
            K = max(int(param * (1 - 0.7 * frac)), 3)
            p = K
        else:
            p = param * (1 + 5 * frac)  # ramp tau up
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        w, _ = get_weights(corr, method, p)
        g = compute_grad(h, N, w)
        
        # Add smoothing if needed
        if smoothing > 0:
            if N > 2:
                g[1:-1] += smoothing * (2*h[1:-1] - h[:-2] - h[2:])
        
        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    
    return best_h, best_c5

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if s <= 0:
        # distribute equally
        idx = np.random.choice(len(h), int(target), replace=False)
        h = np.zeros(len(h))
        h[idx] = 1.0
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, N, w, center):
    grad = np.zeros(N)
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    for ai in active:
        L = int(ai) - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            if L < N:
                grad[:N-L] += wi * (1.0 - h[L:])
                grad[L:] -= wi * h[:N-L]
        else:
            M = -L
            if M < N:
                grad[M:] += wi * (1.0 - h[:N-M])
                grad[:N-M] -= wi * h[M:]
    return grad

        frac = it / max(n_iter - 1, 1)

            if ws > 1e-30:

                w[np.argmax(corr)] = 1.0

            best_h = h.copy()

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, N, w, center):
    grad = np.zeros(N)
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    for ai in active:
        L = int(ai) - center
        wi = float(w[ai])
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[:N - L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:N - L]
        elif -L < N:
            M = -L
            grad[M:] += wi * (1.0 - h[:N - M])
            grad[:N - M] -= wi * h[M:]
    return grad

def optimize_phase(h, N, target, method, p0, p1, lr0, lr1, n_iter, deadline):
    dx = 2.0 / N
    center = N - 1
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while it < n_iter:
        if time.time() >= deadline:
            break
        frac = it / max(n_iter - 1, 1)
        lr = lr1 + 0.5 * (lr0 - lr1) * (1.0 + np.cos(np.pi * frac))
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if method == 'topK':
            K = max(int(p0 + (p1 - p0) * frac), 3)
            idx = np.argsort(corr)[-K:]
            w = np.zeros(len(corr))
            w[idx] = 1.0 / K
        else:
            tau = p0 + (p1 - p0) * frac
            sc = tau * corr - tau * float(np.max(corr))
            w = np.exp(sc)
            ws = w.sum()
            if ws > 1e-30:
                w /= ws
            else:
                w = np.zeros(len(corr))
                w[np.argmax(corr)] = 1.0
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        g = compute_grad(h, N, w, center)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        step = np.clip(lr * mh / (np.sqrt(vh) + eps), -0.08, 0.08)
        h = project_sum(h - step, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5f = float(np.max(corr)) * dx
    if c5f < best_c5:
        best_c5 = c5f
        best_h = h.copy()
    return best_h, best_c5

def coordinate_polish(h, N, target, max_sweeps=4):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    deltas = (-0.03, -0.015, -0.005, 0.005, 0.015, 0.03)
    for _ in range(max_sweeps):
        improved = False
        for i in range(N):
            saved = h.copy()
            for d in deltas:
                h[i] = float(np.clip(h[i] + d, 0.0, 1.0))
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    break
            else:
                h[:] = saved
        if not improved:
            break
    return h, best_c5
