def sgrad(h, dx, alpha):
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * dx * conv
    zm = z.max()
    w = np.exp(z - zm)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        kl1, kh1 = max(0, -j), min(n - 1, n - 1 - j)
        if kl1 <= kh1:
            ks = np.arange(kl1, kh1 + 1)
            g[j] += np.dot(w[o + ks], 1.0 - h[j + ks])
        kl2, kh2 = max(1 - n, j - (n - 1)), min(n - 1, j)
        if kl2 <= kh2:
            ks = np.arange(kl2, kh2 + 1)
            g[j] -= np.dot(w[o + ks], h[j - ks])
    return g * dx

    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def slsqp(h0, dx, ts, mit=300):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n + 1))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-16})
        if r.x is not None:
            hr = project_feasible(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 * dx + 1e-6:
                return hr
    except Exception:
        pass
    try:
        x0b = np.concatenate([h0, [t0 + 0.005]])
        r = minimize(_sobj, x0b, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-16})
        if r.x is not None:
            hr = project_feasible(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 * dx + 1e-6:
                return hr
    except Exception:
        pass
    return h0.copy()

def sgrad(h, dx, alpha):
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * dx * conv
    zm = z.max()
    w = np.exp(z - zm)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for i in range(n):
        k_lo = max(i - n + 1, -(n - 1))
        k_hi = min(i, n - 1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            g[i] += np.dot(w[o + ks], 1.0 - h[i - ks])
        k_lo2 = max(-i, -(n - 1))
        k_hi2 = min(n - 1 - i, n - 1)
        if k_lo2 <= k_hi2:
            ks = np.arange(k_lo2, k_hi2 + 1)
            g[i] -= np.dot(w[o + ks], h[i + ks])
    return g * dx

def pg_optimize(h0, dx, ts, dl, rng):
    n = len(h0)
    bh, bc = h0.copy(), compute_c5(h0, dx)
    alphas = [15, 30, 60, 100, 180, 300, 600, 1200]
    for alpha in alphas:
        if time.time() > dl - 8:
            break
        h = bh.copy()
        v = np.zeros(n)
        mu = 0.82
        stalled = 0
        for i in range(400):
            if time.time() > dl - 4:
                break
            g = sgrad(h, dx, alpha)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            lr = 0.4 / (1.0 + 0.008 * i) / (gnorm + 1e-10)
            v = mu * v - lr * g
            h2 = project_feasible(h + v, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                bh, bc, h = h2.copy(), c2, h2
                stalled = 0
                mu = 0.82
            else:
                stalled += 1
                mu = max(0.5, mu * 0.92)
                v *= 0.4
            if stalled > 80:
                break
    if time.time() < dl - 6:
        hr = slsqp(bh, dx, ts, 250)
        cr = compute_c5(hr, dx)
        if cr < bc:
            bh, bc = hr, cr
    return bh, bc

def coord_descend(h0, dx, ts, rounds, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for r in range(rounds):
        if time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            improved = False
            for d in (0.015, -0.015, 0.007, -0.007, 0.03, -0.03, 0.003, -0.003):
                nv = sv + d
                if nv < -1e-9 or nv > 1 + 1e-9:
                    continue
                h[j] = nv
                h = project_feasible(h, ts)
                c = compute_c5(h, dx)
                if c < bc - 1e-16:
                    bc, improved, imp = c, True, True
                    break
            if not improved:
                h[j] = sv
        if not imp:
            break
    return h, bc

def pair_swap(h0, dx, ts, iters, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for _ in range(iters):
        if time.time() > dl:
            break
        imp = False
        for _ in range(2 * n):
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, ts)
            c = compute_c5(h, dx)
            if c < bc - 1e-16:
                bc, imp = c, True
            else:
                h[i], h[j] = h[j], h[i]
        if not imp:
            break
    return h, bc

def make_starts(n, ts, rng, ref=None):
    out = []
    if ref is not None:
        if len(ref) == n:
            out.append(project_feasible(ref.copy(), ts))
        else:
            xq = np.linspace(0, 1, n, endpoint=False)
            xref = np.linspace(0, 1, len(ref), endpoint=False)
            out.append(project_feasible(np.interp(xq, xref, ref), ts))
    out.append(project_feasible(np.full(n, ts / n), ts))
    for _ in range(2):
        hb = np.zeros(n)
        p = rng.permutation(n)
        hb[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            hb[p[int(ts)]] = fr
        out.append(project_feasible(hb, ts))
    x = np.arange(n) / n
    for m in (2, 3, 5):
        out.append(project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * m * x), ts))
    out.append(project_feasible(rng.uniform(0, 1, n), ts))
    return out

def full_optimize(h0, dx, ts, dl, rng):
    h, c = pg_optimize(h0, dx, ts, dl, rng)
    if time.time() < dl - 15:
        h2, c2 = coord_descend(h, dx, ts, 15, dl - 8, rng)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 10:
        h3, c3 = pair_swap(h, dx, ts, 8, dl - 5, rng)
        if c3 < c:
            h, c = h3, c3
    if time.time() < dl - 6:
        h4 = slsqp(h, dx, ts, 200)
        c4 = compute_c5(h4, dx)
        if c4 < c:
            h, c = h4, c4
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 6

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None

    bh, bc = None, 1e9

    if hr is not None:
        n0 = len(hr)
        dx0, ts0 = 2.0 / n0, n0 / 2.0
        h0 = project_feasible(hr, ts0)
        bh, bc = full_optimize(h0, dx0, ts0, deadline, rng)

    n_base = len(hr) if hr is not None else 50

    # Try nearby n values
    n_candidates = []
    for dn in (0, 2, -2, 4, -4, 5, 6, -5, 7, 8, -6, 10, -8):
        nt = n_base + dn
        if 30 <= nt <= 90 and nt not in n_candidates:
            n_candidates.append(nt)
    n_candidates.sort(key=lambda x: abs(x - n_base))

    for nt in n_candidates:
        if time.time() > deadline - 80:
            break
        if nt == n_base and bh is not None:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        starts = make_starts(nt, tst, rng, bh)
        for hs in starts[:3]:
            if time.time() > deadline - 30:
                break
            h_r, c_r = full_optimize(hs, dxt, tst, deadline - 10, rng)
            if c_r < bc:
                bh, bc = h_r, c_r

    # Perturbation restarts at best n
    if bh is not None and time.time() < deadline - 60:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        for amp in (0.12, 0.06, 0.03, 0.015):
            if time.time() > deadline - 30:
                break
            hp = project_feasible(bh + amp * rng.randn(nb), tsb)
            h_r, c_r = full_optimize(hp, dxb, tsb, deadline - 10, rng)
            if c_r < bc:
                bh, bc = h_r, c_r

    # Final polish
    if bh is not None and time.time() < deadline - 10:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        hf = slsqp(bh, dxb, tsb, 400)
        cf = compute_c5(hf, dxb)
        if cf < bc:
            bh, bc = hf, cf
        if time.time() < deadline - 3:
            hc, cc = coord_descend(bh, dxb, tsb, 25, t_start + budget_s - 2, rng)
            if cc < bc:
                bh, bc = hc, cc

    if bh is None:
        bh = np.full(50, 0.5)
        bc = compute_c5(bh, 0.04)

    bh = project_feasible(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
