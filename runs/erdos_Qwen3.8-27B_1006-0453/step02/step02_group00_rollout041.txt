For the 1D optimization: minimize f(δ) = max_k (corr[k] + δ*C[k]) over [lo, hi].

This is a convex PL function. The minimum is at a point where the subgradient ∂f contains 0, or at a boundary.

Efficient method:
1. At the current best δ, find all k that are "active" (achieve the max)
2. If any active C[k] < 0 and any other value could be max... 

Actually, the simplest efficient method for convex PL 1D:
- The function f(δ) = max_k (a_k + b_k * δ) where a_k = corr[k], b_k = C[k]
- The minimum over [lo, hi] can be found by:
  - If all b_k ≤ 0: minimum at hi
  - If all b_k ≥ 0: minimum at lo
  - Otherwise: the minimum is at a "crossing point" where two (or more) lines intersect
  
Finding the crossing point: sort the lines by their value at δ=0, then walk through them...

Actually, the simplest approach that's efficient: use binary search on the subgradient.
- At δ, the subgradient is C[k*] where k* is the argmax
- If C[k*] < 0: increase δ (function is decreasing)
- If C[k*] > 0: decrease δ (function is increasing)
- If C[k*] = 0 or multiple k* have mixed signs: we're at the minimum

This converges in O(n) steps (each step jumps to a breakpoint).

def optimize_swap_fast(h, i, j, corr, n, n_grid=60):
    """Fast swap optimization using precomputed correlation."""
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-10:
        return 0.0, corr.copy()
    
    # Compute C[k] for the swap
    # C[k] = h[j-k] - h[i-k] + h[j+k] - h[i+k] (with 0 for out of bounds)
    # corr = np.correlate(h, 1-h, 'full') has length 2n-1
    # corr[k] = sum_m h[m+k] * (1-h[m]) for m=0..n-1 where 0<=m+k<n
    # So k ranges from -(n-1) to n-1, stored as index k+(n-1) in the array
    
    # For the effect of changing h[i] by +1:
    # In corr[k] = sum_m h[m+k]*(1-h[m]):
    #   h[m+k] changes when m+k = i → m = i-k: adds (1-h[i-k])
    #   (1-h[m]) changes when m = i: subtracts h[i+k]
    # So Δ from h[i]: (1-h[i-k]) - h[i+k]
    
    # Let me define helper: safe_h(idx) = h[idx] if 0<=idx<n else 0
    # eff_i[k] = (1 - safe_h(i-k)) - safe_h(i+k)  [for h[i] increase]
    # eff_j[k] = (1 - safe_h(j-k)) - safe_h(j+k)  [for h[j] increase]
    # C[k] = eff_i[k] - eff_j[k] [for h[j] decrease, it's -eff_j]
    # Wait: swap is h[i]+=δ, h[j]-=δ
    # Δcorr[k] = δ * eff_i[k] + (-δ) * eff_j[k] = δ * (eff_i[k] - eff_j[k])
    
    n_corr = 2 * n - 1
    C = np.zeros(n_corr)
    for k_idx in range(n_corr):
        k = k_idx - (n - 1)  # actual shift
        # safe_h(i - k), safe_h(i + k), safe_h(j - k), safe_h(j + k)
        ik1 = i - k
        ik2 = i + k
        jk1 = j - k
        jk2 = j + k
        s_ik1 = h[ik1] if 0 <= ik1 < n else 0.0
        s_ik2 = h[ik2] if 0 <= ik2 < n else 0.0
        s_jk1 = h[jk1] if 0 <= jk1 < n else 0.0
        s_jk2 = h[jk2] if 0 <= jk2 < n else 0.0
        eff_i = (1.0 - s_ik1) - s_ik2
        eff_j = (1.0 - s_jk1) - s_jk2
        C[k_idx] = eff_i - eff_j
    
    # Now minimize max_k (corr[k] + δ * C[k]) over δ ∈ [lo, hi]
    deltas = np.sort(np.unique(np.concatenate([np.linspace(lo, hi, n_grid), [0.0]])))
    best_val = np.max(corr)
    best_delta = 0.0
    for d in deltas:
        v = np.max(corr + d * C)
        if v < best_val - 1e-15:
            best_val = v
            best_delta = d
    
    # Apply the best delta
    h[i] += best_delta
    h[j] -= best_delta
    h = np.clip(h, 0, 1)
    
    # Recompute correlation for next iteration
    new_corr = np.correlate(h, 1.0 - h, mode='full')
    return new_corr

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h = h * (target / total)

def do_cd_swaps_fast(h, n, rng, n_swaps, n_grid=60):
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = 2 * n - 1
    for it in range(n_swaps):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi - lo < 1e-10:
            continue
        # Compute effect vector C
        # For shift k (0 to nc-1, actual shift = k-(n-1)):
        # C[k] = [(1-h[i-shift]) - h[i+shift]] - [(1-h[j-shift]) - h[j+shift]]
        # where h[x] = 0 if x not in [0, n-1]
        shifts = np.arange(nc) - (n - 1)
        # Use padded h
        # h_ext[x] = h[x] if 0<=x<n else 0
        # Need h[i-shift] and h[i+shift] for all shifts
        # i-shift ranges from i-(n-1) to i+(n-1)
        # i+shift ranges from i-(n-1) to i+(n-1)
        
        # Let's use a padded array
        # h_pad of length 3*(n-1)+1, with h at offset (n-1)
        # Actually, simpler: just use np.where
        
        ik_neg = i - shifts  # where to look for h[i-shift]
        ik_pos = i + shifts  # where to look for h[i+shift]
        jk_neg = j - shifts
        jk_pos = j + shifts
        
        # Get values with 0 padding
        def safe_idx(arr, idx):
            mask = (idx >= 0) & (idx < n)
            result = np.zeros(len(idx))
            result[mask] = arr[idx[mask]]
            return result
        
        s_ikn = safe_idx(h, ik_neg)
        s_ikp = safe_idx(h, ik_pos)
        s_jkn = safe_idx(h, jk_neg)
        s_jkp = safe_idx(h, jk_pos)
        
        eff_i = (1.0 - s_ikn) - s_ikp
        eff_j = (1.0 - s_jkn) - s_jkp
        C = eff_i - eff_j
        
        # Minimize max(corr + d*C) over d in [lo, hi]
        grid = np.sort(np.unique(np.concatenate([np.linspace(lo, hi, n_grid), [0.0]])))
        best_d = 0.0
        best_v = np.max(corr)
        for d in grid:
            if abs(d) < 1e-15:
                continue
            v = np.max(corr + d * C)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-15:
            h[i] = np.clip(h[i] + best_d, 0, 1)
            h[j] = np.clip(h[j] - best_d, 0, 1)
            corr = np.correlate(h, 1.0 - h, mode='full')
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h = h * (target / total)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, T, n, maxiter, ftol):
    def obj(x):
        return smooth_c5(x, T)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

    nc = 2 * n - 1
    h_pad = np.zeros(3 * n)
    h_pad[n:2 * n] = h
    corr = np.correlate(h, 1.0 - h, mode='full')

        j = int(rng.integers(n))
        if i == j:

        s_ikp = h_pad[n + i + shifts]
        s_jkn = h_pad[n + j - shifts]
        s_jkp = h_pad[n + j + shifts]
        C = (1.0 - s_ikn) - s_ikp - ((1.0 - s_jkn) - s_jkp)

            v = np.max(corr + d * C)

                best_v = v
                best_d = d

            h[j] = np.clip(h[j] - best_d, 0.0, 1.0)

            c5v = np.max(corr) * (2.0 / n)
            if c5v < best_c5:
                best_c5 = c5v
    
    return h, best_c5
