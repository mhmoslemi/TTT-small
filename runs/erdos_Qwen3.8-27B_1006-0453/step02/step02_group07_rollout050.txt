```python
        j_range = np.arange(n)
        idx1 = j_range - m
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] -= h[idx1[valid1]]
        idx2 = j_range + m
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] -= h[idx2[valid2]]
        return dx * grad
    
    exp_vals = np.exp((corr - c_max) / T)
    p = exp_vals / np.sum(exp_vals)
    
    # For efficiency, only use lags with significant weight
    threshold = 1e-10
    active = p > threshold
    if np.sum(active) == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    j_range = np.arange(n)
    for m in range(-(n-1), n):
        pm = p[m + n - 1]
        if pm < threshold:
            continue
        val = np.ones(n)
        idx1 = j_range - m
        v1 = (idx1 >= 0) & (idx1 < n)
        val[v1] -= h[idx1[v1]]
        idx2 = j_range + m
        v2 = (idx2 >= 0) & (idx2 < n)
        val[v2] -= h[idx2[v2]]
        grad += pm * val
    
    return dx * grad
```

This is O(n * n_active) where n_active is the number of active lags. With LSE smoothing, for moderate T, only lags near the max have significant weight, so n_active might be 50-200. This should be fast enough.
