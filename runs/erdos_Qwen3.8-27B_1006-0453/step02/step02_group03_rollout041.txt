    n = len(h0)
    c = np.correlate(h0, 1.0 - h0, mode='full')
    top_k = np.argsort(c)[-ntop:]  # worst (largest) shifts
    c_top = c[top_k] * dx
    t0 = c_top.max() + 0.01
    
    def obj(x):
        return x[n]
    
    def cons_ineq(x):
        h = x[:n]
        c = np.correlate(h, 1.0 - h, mode='full')
        return x[n] - c[top_k] * dx
    
    def cons_eq(x):
        return np.sum(x[:n]) - ts
    
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': cons_ineq},
            {'type': 'eq', 'fun': cons_eq}]
    
    r = minimize(obj, x0, method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    if r.success or r.x is not None:
        hr = proj(r.x[:n], ts)
        return hr
    return h0.copy()

    bc = compute_c5(h0, dx)
    deltas = (0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.1, -0.1)
    for r in range(maxr):
        if time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in deltas:
                h[j] = min(1.0, max(0.0, sv + d))
                h = proj(h, ts)
                c2 = compute_c5(h, dx)
                if c2 < bc:
                    bc, imp = c2, True
                    sv = h[j]
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def opt_one(h0, n, rng, dl):
    dx, ts = 2.0 / n, n / 2.0
    h = proj(h0, ts)
    c = compute_c5(h, dx)
    if time.time() < dl - 12:
        h2 = slsqp_fast(h, dx, ts, 120)
        c2 = compute_c5(h2, dx)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 6:
        h3, c3 = pgd_run(h, dx, ts, dl - 3, rng)
        if c3 < c:
            h, c = h3, c3
    if time.time() < dl - 3:
        h4, c4 = coord_desc(h, dx, ts, dl - 1, rng)
        if c4 < c:
            h, c = h4, c4
    if time.time() < dl - 1:
        h5 = slsqp_fast(h, dx, ts, 60)
        c5 = compute_c5(h5, dx)
        if c5 < c:
            h, c = h5, c5
    return h, c

def gen_starts(n, ts, rng, ref):
    out = []
    if ref is not None:
        if len(ref) == n:
            out.append(ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.append(hi)
    out.append(np.full(n, ts / n))
    for _ in range(3):
        h = np.zeros(n)
        p = rng.permutation(n)
        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[p[int(ts)]] = fr
        out.append(h)
    x = np.arange(n) / n
    for m in (1, 2, 3, 5):
        out.append(0.5 + 0.25 * np.sin(2.0 * np.pi * m * x))
    out.append(rng.uniform(0, 1, n))
    p = max(2, n // 4)
    out.append(np.array([1.0 if (i % (2 * p)) < p else 0.0 for i in range(n)]))
    return [proj(s, ts) for s in out]
