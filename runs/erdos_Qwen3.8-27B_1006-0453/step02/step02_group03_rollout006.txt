import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _sp_store = {}

def _slsqp_obj(x):
    return x[_sp_store['n']]

def _slsqp_con1(x):
    n = _sp_store['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _slsqp_con2(x):
    return x[:_sp_store['n']].sum() - _sp_store['ts']

def slsqp_mm(h0, dx, ts, mit=250):
    n = len(h0)
    _sp_store['n'], _sp_store['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0.copy(), [t0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _slsqp_con1}, {'type': 'eq', 'fun': _slsqp_con2}]
    try:
        r = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-14})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 + 0.02:
                return hr
    except Exception:
        pass
    return h0.copy()

def lse_grad(h, n, beta, dx):
    cr = np.correlate(h, 1.0 - h, mode='full')
    m = cr.max()
    w = np.exp(beta * (cr - m) * dx)
    w /= w.sum()
    o = n - 1
    g = np.zeros(n)
    for i in range(n):
        k1lo = max(-(n-1), i-(n-1))
        k1hi = min(n-1, i)
        if k1lo <= k1hi:
            kk = np.arange(k1lo, k1hi+1)
            g[i] += np.dot(w[o+kk], 1.0 - h[i-kk])
        k2lo = max(-(n-1), -i)
        k2hi = min(n-1, n-1-i)
        if k2lo <= k2hi:
            kk = np.arange(k2lo, k2hi+1)
            g[i] -= np.dot(w[o+kk], h[i+kk])
    return g * dx

def lse_opt(h0, dx, ts, dl, betas):
    n = len(h0)
    h, hc = h0.copy(), compute_c5(h0, dx)
    for beta in betas:
        if time.time() > dl:
            break
        def obj(x):
            cr = np.correlate(x, 1.0-x, mode='full') * dx
            m = cr.max()
            return (m + np.log(np.sum(np.exp(beta*(cr-m))))) / beta
        def grad(x):
            return lse_grad(x, n, beta, dx)
        try:
            r = minimize(obj, h.copy(), method='L-BFGS-B', jac=grad,
                         bounds=[(0,1)]*n, options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12})
            hr = proj(r.x, ts)
            cr2 = compute_c5(hr, dx)
            if cr2 < hc:
                h, hc = hr, cr2
        except Exception:
            pass
    return h, hc

def coord_desc(h0, dx, ts, rounds, dl, rng):
    n = len(h0)
    h, hc = h0.copy(), compute_c5(h0, dx)
    T = 0.008
    for r in range(rounds):
        if time.time() > dl:
            break
        T *= 0.97
        for j in rng.permutation(n):
            sv = h[j]
            cands = set()
            for d in (0.05,-0.05,0.02,-0.02,0.1,-0.1,0.01,-0.01,0.08,-0.08):
                cv = sv + d
                if 0.0 <= cv <= 1.0:
                    cands.add(round(cv, 10))
            if sv > 0.01:
                cands.add(round(sv*0.5, 10))
                cands.add(0.0)
            if sv < 0.99:
                cands.add(round((sv+1)*0.5, 10))
                cands.add(1.0)
            best_c, best_v = hc, sv
            for cv in cands:
                h[j] = cv
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < best_c - 1e-15:
                    best_c, best_v = c, cv
                elif T > 1e-9 and rng.random() < np.exp(-(c-best_c)/(T*dx)):
                    best_c, best_v = c, cv
            h = np.array(h)
            h[j] = best_v
            h = proj(h, ts)
            if best_c < hc - 1e-15:
                hc = best_c
    return h, hc

def pswap(h0, dx, ts, rnd, dl, rng):
    n = len(h0)
    h, hc = h0.copy(), compute_c5(h0, dx)
    for r in range(rnd):
        if time.time() > dl:
            break
        imp = False
        for _ in range(4*n):
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j or abs(h[i]-h[j]) < 1e-7:
                continue
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h, dx)
            if c < hc - 1e-15:
                hc, imp = c, True
            else:
                h[i], h[j] = h[j], h[i]
        if not imp:
            break
    return h, hc

def pipeline(h0, dx, ts, dl, rng):

    h, c = h0.copy(), compute_c5(h0, dx)
    if time.time() < dl - 30:

            h, c = h2, c2
    if time.time() < dl - 20:

            h, c = h3, c3
    if time.time() < dl - 10:
        h4, c4 = pswap(h, dx, ts, 25, dl - 5, rng)
        if c4 < c:
            h, c = h4, c4
    if time.time() < dl - 5:
        h5 = slsqp_mm(h, dx, ts, 150)
        c5 = compute_c5(h5, dx)
        if c5 < c:
            h, c = h5, c5
    return h, c

def gen_starts(n, ts, rng, ref=None):
    s = [np.full(n, ts/n)]
    for _ in range(4):
        h = np.zeros(n)
        p = rng.permutation(n)
        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if 0.01 < fr < 0.99:
            h[p[int(ts)]] = fr
        s.append(h.copy())
    for m in (1, 2, 3, 5, 7, 11):
        x = np.arange(n)/n
        s.append(0.5 + 0.25*np.sin(2*np.pi*m*x + rng.uniform(0, 6.28)))
    for p in (max(2,n//4), max(2,n//3), max(2,n//2)):
        s.append(np.array([1.0 if (i%(2*p))<p else 0.0 for i in range(n)]))
    for _ in range(3):
        s.append(rng.uniform(0, 1, n))
    if ref is not None:
        if len(ref) == n:
            s.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0,1,n,endpoint=False),
                          np.linspace(0,1,len(ref),endpoint=False), ref)
            s.insert(0, hi)
    return [proj(t, ts) for t in s]
