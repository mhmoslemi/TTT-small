    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Adaptive step size
    alpha = 0.1
    gamma = 0.995  # cooling rate
    
    while time.time() - t0 < budget_s:
        # Compute all autocorrelations
        corr_h = np.correlate(h, h, mode='full')  # length 2n-1
        # Shift so that index n-1 corresponds to lag 0
        # corr_h[n-1+k] = A_k for k = -(n-1), ..., n-1
        A = corr_h  # we'll work with the full correlation

                if 0 <= j - k < n:

        h = project_feasible(h, n)
        
        # Check improvement
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        alpha *= gamma
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_autocorr_min(h):
    """Return min of autocorrelation = n/2 - C5/dx"""
    n = len(h)
    dx = 2.0 / n
    corr_h = np.correlate(h, h, mode='full')
    return np.min(corr_h)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Spread over all non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def grad_autocorr(h, k):
    """Gradient of A_k = sum_j h[j]*h[j+k] w.r.t. h"""
    n = len(h)
    if k < 0:
        k = -k
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 2.0 * h
    else:
        grad[:n-k] += h[k:]
        grad[k:] += h[:n-k]
    return grad

def active_set_ascent(h, n, budget_s, alpha0=0.05, gamma=0.998, eps=0.5):
    """Projected (sub)gradient ascent on min_k A_k"""
    t0 = time.time()
    best_h = h.copy()
    best_minA = compute_autocorr_min(h)
    alpha = alpha0
    
    while time.time() - t0 < budget_s:
        corr_h = np.correlate(h, h, mode='full')
        min_val = np.min(corr_h)
        
        if min_val > best_minA:
            best_minA = min_val
            best_h = h.copy()
        
        # Find near-active lags
        threshold = min_val + eps
        active = np.where(corr_h <= threshold)[0]
        
        # Average gradient over active lags
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += grad_autocorr(h, k)
        grad /= max(1, len(active))
        
        # Step
        h = h + alpha * grad
        h = project_feasible(h, n)
        
        alpha *= gamma
        if alpha < 1e-6:
            alpha = alpha0 * 0.3  # restart with smaller step
    
    return best_h

def pair_local_search(h, n, budget_s, delta=0.05):
    """Local search via pair exchanges targeted at improving worst lag"""
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() - t0 < budget_s:
        corr_h = np.correlate(h, h, mode='full')
        min_val = np.min(corr_h)
        k_min = int(np.argmin(corr_h)) - (n - 1)
        g = grad_autocorr(h, k_min)
        
        improved = False
        # Try all pairs in random order
        idx = np.random.permutation(n)
        for i_idx in range(n):
            if time.time() - t0 > budget_s:
                break
            i = idx[i_idx]
            # Find best j to exchange with
            for j_idx in range(n):
                j = idx[j_idx]
                if i == j:
                    continue
                if g[i] <= g[j]:
                    continue
                d = min(delta, 1.0 - h[i], h[j])
                if d < 1e-10:
                    continue
                h_test = h.copy()
                h_test[i] += d
                h_test[j] -= d
                # Quick improvement check on the active lag
                # A_k changes by d*(g[i]-g[j]) for the active lag
                # But we should check the actual minimum
                new_minA = np.min(np.correlate(h_test, h_test, mode='full'))
                if new_minA > min_val + 1e-12:

        ones.sort(key=lambda i: h[i])

            h_bin[i] = 0
    elif cnt < target:
        zeros = list(np.where(h_bin == 0)[0])
        zeros.sort(key=lambda i: -h[i])
        for i in zeros[:target - cnt]:
            h_bin[i] = 1
    best_c5 = compute_c5(h_bin)
    best = h_bin.copy()
    swaps_done = 0
    for _ in range(200):
        improved = False
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            for j in zeros:
                if swaps_done >= max_swaps:
                    return best, best_c5
                swaps_done += 1
                best[i], best[j] = 0.0, 1.0
                c = compute_c5(best)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    improved = True
                    break
                best[i], best[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def smooth_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(smooth_obj, h, args=(T, n), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0

    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient ascent + SLSQP on parent (25% budget)
    if elapsed() < budget_s * 0.25:
        h_ga = gradient_ascent(best_h, best_n, n_iters=3000, alpha0=0.03, eps=0.8)
        c5 = compute_c5(h_ga)
        if c5 < best_c5:
            best_c5, best_h = c5, h_ga
    if elapsed() < budget_s * 0.30:
        h_s = slsqp_optimize(best_h, maxiter=800)
        c5 = compute_c5(h_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s

    # Phase 2: Multi-resolution gradient + SLSQP (40% budget)
    t_p2_end = budget_s * 0.70
    resolutions = sorted(set([60, 72, 80, 90, 100, best_n]))
    for n_try in resolutions:
        if elapsed() > t_p2_end - 80:
            break
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            h_base = np.interp(np.linspace(0, 2, n_try),
                              np.linspace(0, 2, best_n), best_h)
            h_base = project_feasible(h_base, n_try)
        n_restarts = 5 if n_try <= 90 else 3
        for r in range(n_restarts):
            if elapsed() > t_p2_end - 60:
                break
            scale = 0.06 + 0.06 * np.random.rand()
            h_p = h_base + np.random.randn(n_try) * scale
            h_p = project_feasible(h_p, n_try)
            h_opt = gradient_ascent(h_p, n_try, n_iters=800, alpha0=0.03, eps=0.5)
            h_opt = slsqp_optimize(h_opt, maxiter=500)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt.copy(), n_try
                h_base = h_opt.copy()

    # Phase 3: Binary local search (10% budget)
    if elapsed() < budget_s * 0.78:
        h_bin, c5_bin = binary_local_search(best_h, best_n, max_swaps=8000)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin
        h_s = slsqp_optimize(best_h, maxiter=600)
        c5 = compute_c5(h_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s

    # Phase 4: Fine basin hopping with smooth SLSQP (15% budget)
    t_p4_end = budget_s * 0.93
    n_cur = len(best_h)
    for i in range(15):
        if elapsed() > t_p4_end:
            break
        scale = 0.015 + 0.015 * np.random.rand()
        h_p = best_h + np.random.randn(n_cur) * scale
        h_p = project_feasible(h_p, n_cur)
        h_ga = gradient_ascent(h_p, n_cur, n_iters=1200, alpha0=0.02, eps=0.4)
        h_opt = slsqp_optimize(h_ga, maxiter=700)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt

    # Phase 5: Final polish
    if elapsed() < budget_s * 0.99:
        h_s = slsqp_optimize(best_h, maxiter=1000)
        c5 = compute_c5(h_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s
        h_sm = smooth_grad(best_h, T=2000, maxiter=400)
        c5 = compute_c5(h_sm)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sm
        h_s2 = slsqp_optimize(best_h, maxiter=500)
        c5 = compute_c5(h_s2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
