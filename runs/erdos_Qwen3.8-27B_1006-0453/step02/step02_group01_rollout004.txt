def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)

        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-13:
                break
            h -= room * (-diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def constraint_eq(x):
    n = len(x)
    return x.sum() - n / 2.0

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def full_obj(x):
    return x[-1]

def full_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def full_ineq(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def slsqp_full(h, maxiter=400):
    n = len(h)
    t0 = compute_c5(h) * n / 2.0 + 0.5
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'eq', 'fun': full_eq},
        {'type': 'ineq', 'fun': full_ineq}
    ]
    try:
        res = minimize(full_obj, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def sa_step(h, n, T, rng, mode):
    if mode == 0:
        blen = rng.randint(2, max(3, n // 6))
        start = rng.randint(0, n - blen)
        h_p = h.copy()
        h_p[start:start+blen] += rng.randn(blen) * 0.25
    elif mode == 1:
        freq = rng.uniform(1.0, 10.0)
        amp = rng.uniform(0.04, 0.20)
        phase = rng.uniform(0, 2.0 * np.pi)
        x = np.arange(n)
        h_p = h + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    elif mode == 2:
        h_p = h + rng.randn(n) * 0.06
    else:
        n_flip = rng.randint(3, max(4, n // 8))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_p = h.copy()
        h_p[idx] = rng.uniform(0, 1, n_flip)
    return project_feasible(h_p, n)

def sa_optimize(h0, n, t_start, t_end, n_iter, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    max_no_improve = max(1000, n_iter // 10)

    for i in range(n_iter):
        T = t_start * (t_end / t_start) ** (i / float(n_iter))
        mode = rng.randint(0, 4)
        h_p = sa_step(h, n, T, rng, mode)
        c5_p = compute_c5(h_p)

        if c5_p < cur_c5 or rng.random() < np.exp(-(c5_p - cur_c5) / max(T, 1e-10)):
            h = h_p
            cur_c5 = c5_p
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                h = best_h + rng.randn(n) * 0.15
                h = project_feasible(h, n)
                cur_c5 = compute_c5(h)
                no_improve = 0

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    return best_h, best_c5

def binarize_and_relax(h, n):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    h_relax = h_bin + (h - h_bin) * 0.3
    return project_feasible(h_relax, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Quick SLSQP warmup
    for T in [300, 2000, 10000]:
        if elapsed() > budget_s * 0.04:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=250)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: SA exploration (multiple runs with different params)
    sa_params = [
        (0.6, 0.008, 20000),
        (0.4, 0.004, 18000),
        (0.8, 0.012, 22000),
        (0.3, 0.002, 15000),
    ]
    for (ts, te, ni) in sa_params:
        if elapsed() > budget_s * 0.45:
            break
        h_sa, c5_sa = sa_optimize(best_h, best_n, ts, te, ni, rng)
        h_pol = slsqp_smooth(h_sa, 1500, maxiter=300)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 3: Binarize-relax cycle
    if elapsed() < budget_s * 0.58:
        for _ in range(3):
            if elapsed() > budget_s * 0.58:
                break
            h_br = binarize_and_relax(best_h, best_n)
            h_sa, c5_sa = sa_optimize(h_br, best_n, 0.3, 0.003, 12000, rng)
            h_pol = slsqp_smooth(h_sa, 2000, maxiter=300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 4: Multi-resolution
    for n_try in [100, 80, 120]:
        if elapsed() > budget_s * 0.72:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_sa, c5_sa = sa_optimize(h_interp, n_try, 0.4, 0.005, 12000, rng)
        h_pol = slsqp_smooth(h_sa, 1000, maxiter=200)
        # Interpolate back to best_n
        x_new2 = np.linspace(0, 2, n_try)
        x_old2 = np.linspace(0, 2, best_n)
        h_back = np.interp(x_old2, x_new2, h_pol)
        h_back = project_feasible(h_back, best_n)
        h_back = slsqp_smooth(h_back, 300, maxiter=200)
        c5_back = compute_c5(h_back)
        if c5_back < best_c5:
            best_c5 = c5_back
            best_h = h_back

    # Phase 5: Full-constraint SLSQP (min t formulation)
    if elapsed() < budget_s * 0.82:
        h_full = slsqp_full(best_h, maxiter=500)
        c5_full = compute_c5(h_full)
        if c5_full < best_c5:
            best_c5 = c5_full
            best_h = h_full
        # Second attempt from slightly perturbed
        h_pert = best_h + rng.randn(best_n) * 0.02
        h_pert = project_feasible(h_pert, best_n)
        h_full2 = slsqp_full(h_pert, maxiter=400)
        c5_full2 = compute_c5(h_full2)
        if c5_full2 < best_c5:
            best_c5 = c5_full2
            best_h = h_full2

        h_sa, c5_sa = sa_optimize(best_h, best_n, 0.15, 0.001, 15000, rng)
        c5_pol = compute_c5(h_sa)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_sa
        h_pol = slsqp_smooth(best_h, 5000, maxiter=500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
        h_pol2 = slsqp_smooth(best_h, 50, maxiter=500)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2

    # Phase 7: Final full-constraint + smooth
    if elapsed() < budget_s * 0.98:
        h_fin = slsqp_full(best_h, maxiter=600)
        c5_fin = compute_c5(h_fin)

            best_c5 = c5_fin2
            best_h = h_fin2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
