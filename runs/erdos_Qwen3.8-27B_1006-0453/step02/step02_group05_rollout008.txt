import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(h, N, tau, k_top=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    if k_top > 1:
        k = min(k_top, len(w))
        idx = np.argsort(w)[-k:]
        w[idx] *= 3.0
        w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 1e-5)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def nesterov_opt(h, N, target, dx, deadline, tau0, tau1, lr0, beta,
                 k_top, noise_every, noise_amp, decay, decay_every):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        frac = min(it / 3000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad, c5_val = smax_grad(h, N, tau, k_top)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        v = beta * v - lr * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, target)
        h = h_new
        it += 1
        if noise_every > 0 and it % noise_every == 0:
            h += noise_amp * np.random.randn(N)
            h = project_sum(h, target)
        if it % decay_every == 0:
            lr *= decay
        if lr < 1e-11:
            break
    return best_h, best_c5

def pair_swap_sa(h, N, target, deadline, max_iters=4000):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    temp = 0.005
    h = h.copy()
    cur_c5 = best_c5
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        grad, _ = smax_grad(h, N, 150.0, 3)
        improved = False
        # Gradient-guided pair
        i_inc = int(np.argmin(grad))
        i_dec = int(np.argmax(grad))
        if i_inc != i_dec:
            for delta in [0.02, 0.01, 0.005]:
                h_new = h.copy()
                h_new[i_inc] = min(1.0, h[i_inc] + delta)
                h_new[i_dec] = max(0.0, h[i_dec] - delta)
                h_new = project_sum(h_new, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < cur_c5:
                    h = h_new
                    cur_c5 = c5_new
                    improved = True
                    break
                elif np.random.rand() < np.exp(-(c5_new - cur_c5) / max(temp, 1e-8)):
                    h = h_new
                    cur_c5 = c5_new
                    improved = True
                    break
        if not improved:
            # Random pair
            i1 = np.random.randint(N)
            i2 = np.random.randint(N)
            if i1 != i2 and h[i1] < 0.999 and h[i2] > 0.001:
                delta = 0.01
                h_new = h.copy()
                h_new[i1] = min(1.0, h[i1] + delta)
                h_new[i2] = max(0.0, h[i2] - delta)
                h_new = project_sum(h_new, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < cur_c5:
                    h = h_new
                    cur_c5 = c5_new
                    improved = True
                elif np.random.rand() < np.exp(-(c5_new - cur_c5) / max(temp, 1e-8)):
                    h = h_new
                    cur_c5 = c5_new
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            temp *= 0.85
            if temp < 1e-6:
                break
        it += 1
    return best_h, best_c5

            best_h = h2.copy()

                              k_top=3, noise_every=0, noise_amp=0.0,

                best_h = h5.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
