def smooth_objective(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    # Penalty for sum constraint
    penalty = lam * (h.sum() - n/2.0)**2
    return (val + penalty) * dx

def project_h(h, n):

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    return np.clip(h, 0, 1)

def l-bfgsb_optimize(h0, T, n, lam, maxiter=500):

        i = p + s_vals

        term[valid_j] += (1.0 - h[j[valid_j]])

        grad[p] = dx * T * np.dot(w, term)

    return grad

import time

    corr = np.correlate(h, 1.0 - h, mode='full')

            if total < 1e-14:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    val = (m + np.log(np.sum(exps)) / T) * dx

    w_s = w[::-1]  # w[2n-2], ..., w[0]

        _, g = smooth_obj_and_grad(h, T, lam)

    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return np.clip(res.x, 0, 1)

def coord_descent(h, n, n_sweeps=5, grid_size=40):
    h = h.copy()
    for _ in range(n_sweeps):
        for p in range(n):
            # Compute A_m and B_m for each correlation index m
            # corr[m] = A_m + h[p] * B_m (linear in h[p])
            corr = np.correlate(h, 1.0 - h, mode='full')
            # dc[m]/dh[p]:
            s_vals = np.arange(-(n-1), n)
            d = n - 1  # offset
            # dc[m]/dh[p] = (1-h[p-s])*[0<=p-s<n] - h[p+s]*[0<=p+s<n] where s=n-1-m
            w_s = np.ones(2*n-1)  # we just need the structure, not the weights
            # Actually for the 1D problem, I need B[m] = dc[corr_idx]/dh[p] for each corr index
            # corr_idx m corresponds to s = n-1-m
            # B[m] = (1-h[p-s])*[0<=p-s<n] - h[p+s]*[0<=p+s<n]
            
            s = n - 1 - np.arange(2*n-1)  # s[m] = n-1-m
            j = p - s  # for (1-h[p-s])
            i = p + s  # for h[p+s]
            B = np.zeros(2*n-1)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            B[vj] += (1.0 - h[j[vj]])
            B[vi] -= h[i[vi]]
            
            # A[m] = corr[m] - h[p]*B[m]
            A = corr - h[p] * B
            
            # Find t* = argmin_{t in [0,1]} max_m (A[m] + t*B[m])
            t_grid = np.linspace(0, 1, grid_size)
            vals = np.max(A[None, :] + t_grid[:, None] * B[None, :], axis=1)
            t_star = t_grid[np.argmin(vals)]
            
            # Refine with finer grid around t_star
            if 0 < t_star < 1:
                t_fine = np.linspace(max(0, t_star - 0.025), min(1, t_star + 0.025), 50)
                vals_f = np.max(A[None, :] + t_fine[:, None] * B[None, :], axis=1)
                t_star = t_fine[np.argmin(vals_f)]
            
            h[p] = t_star
        h = project_h(h, n)
    return h
