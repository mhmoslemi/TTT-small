```python
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h += room * (diff / room.sum())
        else:
            if h.sum() < 1e-14:
                break
            h += h * (diff / h.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def gen_initial(n, seed):
    """Generate a structured initial point."""
    rng = np.random.RandomState(seed)
    # Start with alternating pattern
    h = np.zeros(n)
    h[::2] = 1.0
    # Adjust to get sum = n/2
    h = project_feasible(h, n)
    return h

def binary_sa(h_init, n, n_iter=100000, t0=0.3, t1=0.001, seed=0):
    rng = np.random.RandomState(seed)
    h = (h_init > 0.5).astype(np.float64)
    # Fix to exactly n/2 ones
    while h.sum() > n/2:
        h[rng.choice(np.where(h==1)[0])] = 0
    while h.sum() < n/2:
        h[rng.choice(np.where(h==0)[0])] = 1
    
    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    
    for it in range(n_iter):
        T = t0 * (t1/t0) ** (it / max(n_iter-1, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)

            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def lbfgs_smooth(h, n, T, maxiter=500, lam=100.0):
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        sp = lam * (x.sum() - n / 2.0) ** 2
        return val * (2.0 / n) + sp
    
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h

def slsqp_refine(h, n, T=500, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0/n)

                      options={'maxiter': maxiter, 'ftol': 1e-14})

    except:
        return h
```

**Key algorithmic improvement: Use the correlation values to guide the search.**

Instead of random perturbations, I can:
1. Compute all correlations
2. Identify the top-5 worst lags
3. For each worst lag k, identify the positions (i, i+k) that contribute most to corr_k
4. Try to "fix" these by swapping h values

This is a form of coordinate descent / greedy algorithm.

```python
def targeted_fix(h, n, n_steps=1000):
    """Greedy: repeatedly find worst (i,j) pair contributing to worst lag and fix it."""
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_lag_idx = np.argmax(corr)
        worst_lag = worst_lag_idx - (n - 1)  # actual lag
        
        if worst_lag == 0:
            break  # Can't reduce further (shouldn't happen for binary)
        
        # Find the top contributing pairs
        if worst_lag > 0:
            # corr = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
            # Contributing pairs: i where h[i]=1 and h[i+lag]=0
            pairs = [(i, i+worst_lag) for i in range(n-worst_lag) if h[i] > 0.5 and h[i+worst_lag] < 0.5]
        else:
            lag = -worst_lag
            pairs = [(i+lag, i) for i in range(n-lag) if h[i] > 0.5 and h[i+lag] < 0.5]
        
        if not pairs:
            break
        
        # Try flipping the pair that reduces C₅ the most
        # (For binary, flipping h[i]=1->0 and h[j]=0->1 maintains sum)
        best_improvement = 0
        best_pair = None
        for i, j in pairs[:20]:  # Check top 20 pairs
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if best_c5 - new_c5 > best_improvement:
                best_improvement = best_c5 - new_c5
                best_pair = (i, j)
        
        if best_pair is not None and best_improvement > 0:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 -= best_improvement
        else:
            # No improvement, try random swaps
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 > best_c5 + 0.01:  # Revert big worsenings
                h[i] = 1.0
                h[j] = 0.0
            else:
                best_c5 = new_c5
    
    return h, best_c5
```

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h += room * (diff / room.sum())
        else:
            if h.sum() < 1e-14:
                break
            h += h * (diff / h.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def binary_sa(h_init, n, n_iter, t0, t1, seed):
    rng = np.random.RandomState(seed)
    h = (h_init > 0.5).astype(np.float64)
    while h.sum() > n / 2:
        h[rng.choice(np.where(h == 1)[0])] = 0
    while h.sum() < n / 2:
        h[rng.choice(np.where(h == 0)[0])] = 1
    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    for it in range(n_iter):
        T = t0 * (t1 / t0) ** (it / max(n_iter - 1, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0

        if new_c5 <= curr_c5 or rng.random() < np.exp(min(0.0, -(new_c5 - curr_c5) / max(T, 1e-12))):

            if new_c5 < best_c5:

def lbfgs_opt(h, T, maxiter=500, lam=300.0):

        m = np.max(corr)

        return val * (2.0 / n) + lam * (x.sum() - n / 2.0) ** 2
    try:

                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})

    except Exception:
        return h
```
