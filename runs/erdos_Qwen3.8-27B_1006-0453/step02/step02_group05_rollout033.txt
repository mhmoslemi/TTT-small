def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0

        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    """Compute gradient of soft-max and true C5."""
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    f_vals = corr * dx
    true_c5 = float(f_vals.max())
    
    sc = tau * f_vals - tau * f_vals.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    
    if topK > 0:

        w2[idx] = w[idx]
        w2 /= w2.sum()
        w = w2

        else:
            m = -s
            grad[m:] += wi * (1.0 - h[:N-m])
            grad[:N-m] -= wi * h[m:]
    return grad, true_c5

            h = project_sum(h, target)
            v = np.zeros(N)

2. Solve min_h sum_k p_k * f_k(h) (via gradient descent)
3. Update p: p_k ∝ p_k * exp(tau * f_k(h*)) (softmax update)
4. Repeat

This is essentially the **gradient ascent-descent** or **fictitious play** approach for the min-max problem. It's guaranteed to converge to a saddle point under appropriate conditions.

def minmax_opt(h, N, target, deadline, tau_adapt):
    """Alternating min-max: fix p → minimize weighted sum → update p."""
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Initialize p uniformly
    K = 2*N - 1
    p = np.ones(K) / K
    
    while time.time() < deadline:
        # Step 1: Minimize J(h) = sum_k p_k * f_k(h) for fixed p
        # Use a few gradient steps
        for _ in range(50):
            if time.time() >= deadline:
                break
            grad = np.zeros(N)
            for k in range(K):
                if p[k] < 1e-6:
                    continue
                s = k - (N-1)
                pk = p[k]
                if s == 0:
                    grad += pk * (1.0 - 2.0*h)
                elif s > 0:
                    grad[:N-s] += pk * (1.0-h[s:])
                    grad[s:] -= pk * h[:N-s]
                else:
                    m = -s
                    grad[m:] += pk * (1.0-h[:N-m])
                    grad[:N-m] -= pk * h[m:]

        corr = np.correlate(h, 1.0-h, mode='full')

            best_h = h.copy()

        p = np.exp(sc)
        p /= p.sum()
    
    return best_h, best_c5

        lr = lr_start * (lr_end / lr_start) ** frac
        grad, c5_v = grad_smax(h, N, tau, topK)
        if c5_v < best_c5 - 1e-9:
            best_c5 = c5_v
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            # small perturbation to escape plateaus
            h += np.random.randn(N) * 0.005
            h = project_sum(h, target)
    return best_h, best_c5

def tune_single(h_init, N, target, remain_sec, rng):
    """Run multi-phase Adam with varied hyperparams, return best."""
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, 2.0 / N)
    
    configs = [
        # (tau_s, tau_e, lr_s, lr_e, topK, frac_of_time)
        (8.0, 40.0, 0.04, 0.015, -1, 0.25),
        (40.0, 120.0, 0.015, 0.005, -1, 0.25),
        (120.0, 400.0, 0.005, 0.0015, 5, 0.25),
        (400.0, 1200.0, 0.0015, 0.0004, 3, 0.25),
    ]
    
    cur_h = h_init.copy()
    for ts, te, lrs, lre, tk, frac in configs:
        t_end = time.time() + remain_sec * frac
        remain_sec *= (1.0 - frac)
        if time.time() > t_end:
            break
        h_new, c5_new = adam_phase(cur_h, N, target, t_end, ts, te, lrs, lre, tk)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        cur_h = h_new if c5_new <= best_c5 else best_h.copy()
    
    return best_h, best_c5

def random_binary_init(N, rng):
    """Create a random permutation of N/2 ones and N/2 zeros."""
    arr = np.zeros(N)
    arr[:N//2] = 1.0
    rng.shuffle(arr)
    # Smooth slightly
    k = np.ones(5) / 5.0
    arr = np.convolve(arr, k, mode='same')
    return project_sum(arr, N / 2.0)

def block_init(N, rng, nblocks):
    """Create block-structured initial guess."""
    h = np.zeros(N)
    positions = np.sort(rng.choice(range(1, N), size=nblocks, replace=False))
    boundaries = np.concatenate([[0], positions, [N]])
    for i in range(nblocks + 1):
        val = rng.random()
        h[boundaries[i]:boundaries[i+1]] = val
    return project_sum(h, N / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8.0
    rng = np.random.RandomState(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    
    # Sizes to try
    sizes = []
    for s in [n0, n0+40, n0+100, n0+200, 460, 520, 580, 640]:
        s = s if s % 2 == 0 else s + 1
        if s <= 700 and s not in sizes:
            sizes.append(s)
    sizes.sort()
    
    # Time allocation per size
    n_sizes = len(sizes)
    
    for si, N in enumerate(sizes):
        time_left = deadline - time.time()
        if time_left < 60:
            break
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        
        # Budget for this size
        if si == 0:
            alloc = time_left * 0.35
        elif si == 1:
            alloc = time_left * 0.25
        else:
            alloc = time_left * 0.15
        alloc = min(alloc, time_left - 30)
        
        if alloc < 30:
            break
        
        # Starting points
        starts = []
        # Interpolated from parent
        h_interp = np.interp(xN, x0, h0)
        starts.append(project_sum(h_interp, target))
        # Random binary
        if N <= 600:
            starts.append(random_binary_init(N, rng))
            starts.append(block_init(N, rng, 12))
        
        for st_i, h_start in enumerate(starts):
            if time.time() > deadline - 40:
                break
            # Allocate time: more for first start
            if st_i == 0:
                t_alloc = alloc * 0.5
            else:
                t_alloc = alloc * 0.25
            if time.time() + t_alloc > deadline - 20:
                t_alloc = max(30, deadline - 20 - time.time())
            if t_alloc < 30:
                break
            
            h_res, c5_res = tune_single(h_start, N, target, t_alloc, rng)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
        
        # If the best is at this size, do a final polish
        if time.time() < deadline - 50:
            Nb = len(best_h)
            if Nb == N and (deadline - time.time()) > 80:
                h_pol, c5_pol = adam_phase(best_h.copy(), Nb, Nb//2.0,
                                            deadline - 40,
                                            500.0, 2000.0, 0.001, 0.0002, topK=3)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
    
    # Final micro-polish
    if time.time() < deadline - 5:
        Nb = len(best_h)
        h_fin, c5_fin = adam_phase(best_h.copy(), Nb, Nb//2.0,
                                    deadline - 2.0,
                                    1000.0, 3000.0, 0.0005, 0.0001, topK=3)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Verify constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
