                # Upsample to n_init
                h_up = np.interp(np.linspace(0, 2, n_init, endpoint=False),
                                 np.linspace(0, 2, n_low, endpoint=False), h_de_ref)
                h_up = project_feasible(h_up, n_init)
                h_up = slsqp_run(h_up, 500, n_init, maxiter=600, ftol=1e-14)
                c5_up = compute_c5(h_up)
                if c5_up < best_c5:
                    best_c5 = c5_up
                    best_h = h_up
                # Also keep low-res result as candidate
                if c5_de < best_c5 * 1.001:
                    best_h_low = h_de_ref.copy()
                    best_c5_low = c5_de
                    n_keep = n_low

    # Phase 3: DE at current resolution (25% budget)
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.50:
        for T_de in [80, 200]:
            if elapsed() > budget_s * 0.45:
                break
            h_de = de_gsearch(n_cur, T_de, popsize=20, maxiter=45, seed=seed + 100)
            if h_de is None:
                continue
            h_de, c5_de = multi_T_slsqp(h_de, n_cur, [200, 800, 2500], [400, 600, 800])
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de

    # Phase 4: Multi-start SLSQP from diverse initializations (20% budget)
    if elapsed() < budget_s * 0.70:
        n_s = len(best_h)
        starts = []
        # Binary random
        for _ in range(5):
            hb = (np.random.rand(n_s) < 0.5).astype(float)
            starts.append(project_feasible(hb, n_s))
        # Block patterns
        hblk = np.zeros(n_s)
        hblk[::2] = 1.0
        starts.append(project_feasible(hblk, n_s))
        # Quarter blocks
        q = n_s // 4
        hq = np.zeros(n_s)
        hq[:q] = 1.0
        hq[2*q:3*q] = 1.0
        starts.append(project_feasible(hq, n_s))
        # Triangular
        ht = np.sin(np.linspace(0, np.pi, n_s)) ** 2
        ht = project_feasible(ht, n_s)
        starts.append(ht)
        # Sine modulated
        hs = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n_s))
        starts.append(project_feasible(hs, n_s))
        # Best_h with noise
        hn = best_h + 0.1 * np.random.randn(n_s)
        starts.append(project_feasible(hn, n_s))

        for h0 in starts:
            if elapsed() > budget_s * 0.70:
                break
            hc, c5v = multi_T_slsqp(h0, n_s, [30, 200, 800], [300, 500, 700])
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 5: Basin hopping with DE-like kicks (15% budget)
    n_bh = len(best_h)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_iter_bh = 40
    if elapsed() < budget_s * 0.85:
        for it in range(n_iter_bh):
            if elapsed() > budget_s * 0.85:
                break
            # DE-inspired mutation: pick 2 random points, use differential
            a = np.random.randint(n_bh)
            b = np.random.randint(n_bh)
            scale = 0.15 * (1.0 - 2.0 * it / n_bh) + 0.04
            h_pert = bh_h.copy()
            h_pert[a] = np.clip(bh_h[a] + scale * (bh_h[a] - bh_h[b]), 0, 1)
            h_pert = project_feasible(h_pert, n_bh)
            # Also try larger random kick every 5 iterations
            if it % 5 == 0:
                h_pert = bh_h + 0.12 * (1.0 - it / n_bh) * np.random.randn(n_bh)
                h_pert = project_feasible(h_pert, n_bh)
            hc = slsqp_run(h_pert, 500, n_bh, maxiter=400, ftol=1e-13)
            c5v = compute_c5(hc)
            if c5v < bh_c5 or (it > 10 and np.random.rand() < 0.05):
                bh_c5 = c5v
                bh_h = hc
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 6: Coordinate descent targeting worst overlap (8% budget)
    n_cd = len(best_h)
    if elapsed() < budget_s * 0.93:
        for outer in range(80):
            if elapsed() > budget_s * 0.93:
                break
            improved = False
            for _ in range(30):
                if elapsed() > budget_s * 0.93:
                    break
                j = np.random.randint(n_cd)
                best_j_val = best_c5
                best_j_delta = 0
                for d in [0.03, 0.06, 0.1, 0.15, -0.03, -0.06, -0.1, -0.15, 0.01, -0.01]:
                    h_try = best_h.copy()
                    h_try[j] = np.clip(h_try[j] + d, 0, 1)
                    h_try = project_feasible(h_try, n_cd)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_j_val - 1e-13:
                        best_j_val = c5_try
                        best_j_delta = d
                if best_j_delta != 0:
                    best_h[j] = np.clip(best_h[j] + best_j_delta, 0, 1)
                    best_h = project_feasible(best_h, n_cd)
                    best_c5 = best_j_val
                    improved = True
            if not improved and outer > 20:
                break

    # Phase 7: Final high-T SLSQP polish (5% budget)
    n_fin = len(best_h)
    for T in [3000, 10000, 30000]:
        if elapsed() > budget_s * 0.985:
            break
        best_h = slsqp_run(best_h, T, n_fin, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
