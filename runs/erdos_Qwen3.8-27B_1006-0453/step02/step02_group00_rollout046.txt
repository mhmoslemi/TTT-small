Let me just use a simpler formulation. Define:
c[k] = sum_{j} h[j] * (1-h[j+s]) where s = k-(n-1)

For k=0: s = -(n-1), so c[0] = sum_j h[j] * (1-h[j-(n-1)])
Only j=n-1 gives j-(n-1)=0, so c[0] = h[n-1]*(1-h[0])

For k=n-1: s=0, c[n-1] = sum_j h[j]*(1-h[j]) = sum_j h[j] - sum_j h[j]^2

For k=2n-2: s=n-1, c[2n-2] = h[0]*(1-h[n-1])

This makes sense.

Now the gradient:
c[k] = sum_j h[j]*(1-h[j+s]) where s=k-(n-1)

∂c[k]/∂h[m] = (1-h[m+s]) * 1[m+s ∈ [0,n-1]]  [from the first h[j] term, j=m]
            + h[m] * (-1) * 1[m ∈ j range and j+s=m, i.e., j=m-s ∈ [0,n-1]]
            = (1-h[m+s]) * 1[m+s ∈ [0,n-1]] - h[m-s] * 1[m-s ∈ [0,n-1]]

Wait, the second term: in the sum Σ_j h[j]*(1-h[j+s]), the term involving h[m] in the second factor is when j+s=m, i.e., j=m-s. So:
- First factor contributes: (1-h[m+s]) if m is a valid j (always) AND m+s is valid
- Second factor contributes: -h[m-s] if m-s is a valid j AND m is valid (always)

So: ∂c[k]/∂h[m] = (1-h[m+s]) * 1[m+s ∈ [0,n-1]] - h[m-s] * 1[m-s ∈ [0,n-1]]

This looks right. Let me implement this efficiently.

For a given shift s, the gradient contribution to all m is:
grad[m] += w[k] * (1-h[m+s]) for m where m+s ∈ [0,n-1]
grad[m] += w[k] * (-h[m-s]) for m where m-s ∈ [0,n-1]

This can be done with array operations:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softc5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    val = mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
    return float(val) * dx

def softc5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    exps = np.exp(T * (corr - mx))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        m_lo = max(0, -s)
        m_hi = min(n - 1, n - 1 - s)
        if m_lo <= m_hi:
            grad[m_lo:m_hi + 1] += wk * (1.0 - h[m_lo + s:m_hi + 1 + s])
        m_lo2 = max(0, s)
        m_hi2 = min(n - 1, s + n - 1)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] -= wk * h[m_lo2 - s:m_hi2 + 1 - s]
    val = (mx + np.log(exps.sum()) / T) * dx
    return val, grad * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_con_fun(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, T, maxiter=600):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': sum_con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(softc5, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(np.clip(res.x, 0, 1), n)

def lbfgsb_opt(h0, T, maxiter=3000, penalty=5000.0):
    n = len(h0)
    def obj(h):
        v, g = softc5_grad(h, T)
        pen = penalty * (h.sum() - n / 2.0) ** 2
        gpen = 2.0 * penalty * (h.sum() - n / 2.0)
        return v + pen, g + gpen * np.ones(n)
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=True, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return project_feasible(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: SLSQP multi-T polish
    for T in [5, 15, 50, 150, 400, 1000, 2500, 6000]:
        if elapsed() > budget_s * 0.12:
            break
        hc = slsqp_opt(best_h, T, 800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: L-BFGS-B with analytical gradient
    for T in [30, 100, 400, 1500, 5000]:
        if elapsed() > budget_s * 0.22:
            break
        hc = lbfgsb_opt(best_h, T, 5000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 3: Basin hopping (L-BFGS-B based for speed)
    bh_h, bh_c5 = best_h.copy(), best_c5
    n_bh = 100
    for it in range(n_bh):
        if elapsed() > budget_s * 0.58:
            break
        frac = it / n_bh
        step = 0.14 * (1.0 - 0.6 * frac) + 0.025
        rmode = np.random.rand()
        if rmode < 0.35:
            blen = np.random.randint(3, min(n // 4, 20))
            start = np.random.randint(0, n - blen)
            bh_h[start:start + blen] += step * np.random.randn(blen)
        elif rmode < 0.6:
            idx = np.random.choice(n, np.random.randint(4, 15), replace=False)
            bh_h[idx] += step * np.random.randn(len(idx))
        else:
            i1, i2 = np.random.choice(n, 2, replace=False)
            bh_h[i1], bh_h[i2] = bh_h[i2], bh_h[i1]
        h_pert = project_feasible(bh_h, n)
        T_pick = [80, 300, 1000, 3000][np.random.randint(4)]
        if np.random.rand() < 0.4:
            hc = lbfgsb_opt(h_pert, T_pick, 1500)
        else:
            hc = slsqp_opt(h_pert, T_pick, 250)
        c5v = compute_c5(hc)
        temp = 0.005 * (1.0 - 0.85 * frac) + 0.0002
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_h, bh_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    # Phase 4: Multi-resolution
    for n_try in [80, 100, 140]:
        if elapsed() > budget_s * 0.75:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        xn = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        hi = np.interp(xn, xo, best_h)
        for tr in range(3):
            if elapsed() > budget_s * 0.75:
                break
            h0 = project_feasible(hi + np.random.uniform(-0.04, 0.04, n_try), n_try)
            for T in [80, 400, 1500]:
                if elapsed() > budget_s * 0.75:
                    break
                hc = lbfgsb_opt(h0, T, 3000)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc

    # Phase 5: Fine coordinate/pair descent
    n_cur = len(best_h)
    stall = 0
    for it in range(600):
        if elapsed() > budget_s * 0.92:
            break
        improved = False
        if np.random.rand() < 0.5:
            j = np.random.randint(n_cur)
            for d in [0.015, 0.04, 0.08, -0.015, -0.04, -0.08]:
                ht = best_h.copy()
                ht[j] += d
                ht = project_feasible(ht, n_cur)
                cv = compute_c5(ht)
                if cv < best_c5 - 1e-13:
                    best_c5, best_h = cv, ht
                    improved = True
                    break
        else:
            j1, j2 = np.random.choice(n_cur, 2, replace=False)
            for d1 in [0.02, 0.05, 0.1]:
                for d2 in [-0.02, -0.05, -0.1]:
                    ht = best_h.copy()
                    ht[j1] += d1
                    ht[j2] += d2
                    ht = project_feasible(ht, n_cur)
                    cv = compute_c5(ht)
                    if cv < best_c5 - 1e-13:
                        best_c5, best_h = cv, ht
                        improved = True
                        break
                if improved:
                    break
        stall = 0 if improved else stall + 1
        if stall > 150:
            break

    # Phase 6: Final high-T polish
    n_cur = len(best_h)
    for T in [4000, 12000, 40000]:
        if elapsed() > budget_s * 0.995:
            break
        hc = lbfgsb_opt(best_h, T, 5000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
        hc2 = slsqp_opt(best_h, T, 1000)
        c5v2 = compute_c5(hc2)
        if c5v2 < best_c5:
            best_c5, best_h = c5v2, hc2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
