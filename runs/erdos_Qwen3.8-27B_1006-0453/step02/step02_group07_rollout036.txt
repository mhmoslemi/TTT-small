from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fix_sum(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def sum_constraint(h, n):

def smooth_obj(h, T, topk):
    dx = 2.0 / len(h)
    corr = get_corr(h)
    idx = np.argsort(corr)[-topk:]
    cv = corr[idx]
    cmax = np.max(cv)
    if T < 1e-15:
        return cmax * dx
    return (cmax + T * np.log(np.sum(np.exp((cv - cmax) / T)))) * dx

    """Analytical gradient of smoothed max over top-k lags."""
    dx = 2.0 / len(h)
    n = len(h)
    corr = get_corr(h)
    idx = np.argsort(corr)[-topk:]
    cv = corr[idx]
    cmax = np.max(cv)
    # Softmax weights
    w = np.exp((cv - cmax) / T)
    w /= np.sum(w)
    
    # Gradient of each c_k w.r.t. h
    # corr[k] = sum_i h[i] * (1-h[i+m]) where m = k-n+1
    # d(corr[k])/dh[j] = (1-h[j+m])*I[0<=j+m<n] - h[j-m]*I[0<=j-m<n]
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        m = k - n + 1
        # For each j: contribution is w[ii] * [(1-h[j+m])*I - h[j-m]*I]
        if m >= 0:
            # j+m valid: j in [0, n-1-m]
            # j-m valid: j in [m, n-1]
            g[:n-m] += w[ii] * (1.0 - h[m:])
            g[m:] -= w[ii] * h[:n-m]
        else:
            mm = -m
            # j+m valid: j in [mm, n-1]
            # j-m valid: j in [0, n-1-mm]
            g[mm:] += w[ii] * (1.0 - h[:n-mm])  # h[j+m] = h[j-mm]
            g[:n-mm] -= w[ii] * h[mm:]  # h[j-m] = h[j+mm]
    return g * dx

def hess_approx_diag(h, T, topk):
    """Approximate diagonal of Hessian for step sizing."""
    eps = 1e-5
    n = len(h)
    g0 = grad_smooth(h, T, topk)

        diag[j] = (gp[j] - g0[j]) / eps
    # Fill in with median

        diag[diag <= 0] = med
    return diag

    dx = 2.0 / n
    corr = get_corr(h)
    # Top-k lags
    idx = np.argsort(corr)[-topk:]
    cv = corr[idx]

    return h_new

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fix_sum(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def smooth_obj(h, T, topk):
    dx = 2.0 / len(h)
    corr = get_corr(h)
    idx = np.argpartition(corr, -topk)[-topk:]
    cv = corr[idx]
    cmax = np.max(cv)
    if T < 1e-15:
        return cmax * dx
    return (cmax + T * np.log(np.sum(np.exp((cv - cmax) / T)))) * dx

def grad_smooth(h, T, topk):
    dx = 2.0 / len(h)
    n = len(h)
    corr = get_corr(h)
    idx = np.argpartition(corr, -topk)[-topk:]
    cv = corr[idx]
    cmax = np.max(cv)
    w = np.exp((cv - cmax) / T)
    w /= np.sum(w)
    g = np.zeros(n)
    for ii in range(len(idx)):
        k = idx[ii]
        s = k - n + 1
        wv = w[ii]
        # Term 1: (1-h[j-s]) for max(0,s) ≤ j ≤ min(n-1, s+n-1)
        j1lo = max(0, s)
        j1hi = min(n - 1, s + n - 1)
        if j1lo <= j1hi:
            g[j1lo:j1hi+1] += wv * (1.0 - h[j1lo-s:j1hi+1-s])
        # Term 2: -h[j+s] for max(0,-s) ≤ j ≤ min(n-1, n-1-s)
        j2lo = max(0, -s)
        j2hi = min(n - 1, n - 1 - s)
        if j2lo <= j2hi:
            g[j2lo:j2hi+1] -= wv * h[j2lo+s:j2hi+1+s]
    return g * dx

def slsqp_analytic(h, n, T, topk, iters):
    c0 = compute_c5(h)
    try:
        res = optimize.minimize(
            smooth_obj, h, args=(T, topk),
            jac=lambda x: grad_smooth(x, T, topk),
            method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16}
        )
        hn = fix_sum(res.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def lkbfgs(h, n, T, topk, iters):
    c0 = compute_c5(h)
    try:
        res = optimize.minimize(
            smooth_obj, h, args=(T, topk),
            jac=lambda x: grad_smooth(x, T, topk),
            method='L-BFGS-B',
            bounds=[(0., 1.)] * n,
            options={'maxiter': iters, 'gtol': 1e-10, 'maxls': 40}
        )
        hn = fix_sum(res.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def subgrad_mom(h, n, n_iter, alpha0, topk=5):
    v = np.zeros(n)
    beta = 0.9
    alpha = alpha0
    best_h, best_c = h.copy(), compute_c5(h)
    stall = 0
    for it in range(n_iter):
        corr = get_corr(h)
        idx = np.argpartition(corr, -topk)[-topk:]
        cv = corr[idx]
        w = cv - cv.min()
        w /= (w.sum() + 1e-30)
        g = np.zeros(n)
        for ii in range(topk):
            k = idx[ii]
            s = k - n + 1
            wv = w[ii]
            j1lo = max(0, s); j1hi = min(n-1, s+n-1)
            if j1lo <= j1hi:
                g[j1lo:j1hi+1] += wv * (1.0 - h[j1lo-s:j1hi+1-s])
            j2lo = max(0, -s); j2hi = min(n-1, n-1-s)
            if j2lo <= j2hi:
                g[j2lo:j2hi+1] -= wv * h[j2lo+s:j2hi+1+s]
        g *= 2.0 / n
        v = beta * v + (1 - beta) * g
        h2 = np.clip(h - alpha * v, 0, 1)
        h2 = fix_sum(h2, n)
        c2 = compute_c5(h2)
        if c2 < best_c - 1e-11:
            best_h, best_c = h2.copy(), c2
            stall = 0
        else:
            stall += 1
        if stall > 40:
            alpha *= 0.85
            v *= 0.5
        h = h2
    return best_h, best_c

def pair_polish(h, n, rng, n_pairs=600):
    c = compute_c5(h)
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if not (1e-5 < h[j] < 1-1e-5 and 1e-5 < h[l] < 1-1e-5):
            continue
        best_c, best_d = c, 0.0
        for d in (-0.05, -0.02, -0.005, 0.005, 0.02, 0.05):
            if not (0 < h[j]+d < 1 and 0 < h[l]-d < 1):
                continue
            hn = h.copy()
            hn[j], hn[l] = h[j]+d, h[l]-d
            cn = compute_c5(hn)
            if cn < best_c - 1e-13:
                best_c, best_d = cn, d
        if best_d:
            h[j] = np.clip(h[j]+best_d, 0, 1)
            h[l] = np.clip(h[l]-best_d, 0, 1)
            c = best_c
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0/len(h))
    x1 = np.arange(nn, dtype=float) * (2.0/nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)
    t0 = time.time()
    h = np.array(initial_h_values, dtype=float).copy()

    for T, tk, it in sched1:
        if time.time() - t0 > budget_s - 380: break

        c5 = compute_c5(h)

        h_sg, c5_sg = subgrad_mom(h, n, 600, 0.008, topk=5)

        m = i % 4

            ht[:half] += np.random.randn(half) * 0.03
            ht[half:] -= np.random.randn(n-half) * 0.03
        ht = fix_sum(ht, n)
        ht = slsqp_analytic(ht, n, 0.003, 4, 500)

        ht = fix_sum(ht, n)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fix_sum(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def smooth_obj(h, T, topk):
    dx = 2.0 / len(h)
    corr = get_corr(h)
    idx = np.argpartition(corr, -topk)[-topk:]
    cv = corr[idx]
    cmax = np.max(cv)
    if T < 1e-15:
        return cmax * dx
    return (cmax + T * np.log(np.sum(np.exp((cv - cmax) / T)))) * dx

def grad_smooth(h, T, topk):
    dx = 2.0 / len(h)
    n = len(h)
    corr = get_corr(h)
    idx = np.argpartition(corr, -topk)[-topk:]
    cv = corr[idx]
    cmax = np.max(cv)
    w = np.exp((cv - cmax) / T)
    w /= np.sum(w)
    g = np.zeros(n)
    for ii in range(len(idx)):
        s = idx[ii] - n + 1
        wv = w[ii]
        j1lo = max(0, s)
        j1hi = min(n - 1, s + n - 1)
        if j1lo <= j1hi:
            g[j1lo:j1hi + 1] += wv * (1.0 - h[j1lo - s:j1hi + 1 - s])
        j2lo = max(0, -s)
        j2hi = min(n - 1, n - 1 - s)
        if j2lo <= j2hi:
            g[j2lo:j2hi + 1] -= wv * h[j2lo + s:j2hi + 1 + s]
    return g * dx

def slsqp_analytic(h, n, T, topk, iters):
    c0 = compute_c5(h)
    try:
        res = optimize.minimize(
            smooth_obj, h, args=(T, topk),
            jac=grad_smooth,
            method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16}
        )
        hn = fix_sum(res.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def lkbfgs(h, n, T, topk, iters):
    c0 = compute_c5(h)
    try:
        res = optimize.minimize(
            smooth_obj, h, args=(T, topk),
            jac=grad_smooth,
            method='L-BFGS-B',
            bounds=[(0., 1.)] * n,
            options={'maxiter': iters, 'gtol': 1e-10, 'maxls': 40}
        )
        hn = fix_sum(res.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def subgrad_mom(h, n, n_iter, alpha0, topk=5):
    v = np.zeros(n)
    beta = 0.9
    alpha = alpha0
    best_h, best_c = h.copy(), compute_c5(h)
    stall = 0
    for _ in range(n_iter):
        corr = get_corr(h)
        idx = np.argpartition(corr, -topk)[-topk:]
        cv = corr[idx]
        w = cv - cv.min()
        w /= (w.sum() + 1e-30)
        g = np.zeros(n)
        for ii in range(topk):
            s = idx[ii] - n + 1
            wv = w[ii]
            j1lo = max(0, s); j1hi = min(n - 1, s + n - 1)
            if j1lo <= j1hi:
                g[j1lo:j1hi + 1] += wv * (1.0 - h[j1lo - s:j1hi + 1 - s])
            j2lo = max(0, -s); j2hi = min(n - 1, n - 1 - s)
            if j2lo <= j2hi:
                g[j2lo:j2hi + 1] -= wv * h[j2lo + s:j2hi + 1 + s]
        g *= 2.0 / n
        v = beta * v + (1.0 - beta) * g
        h2 = np.clip(h - alpha * v, 0, 1)
        h2 = fix_sum(h2, n)
        c2 = compute_c5(h2)
        if c2 < best_c - 1e-11:
            best_h, best_c = h2.copy(), c2
            stall = 0
        else:
            stall += 1
        if stall > 40:
            alpha *= 0.85
            v *= 0.5
        h = h2
    return best_h, best_c

def pair_polish(h, n, rng, n_pairs=600):
    c = compute_c5(h)
    steps = (-0.05, -0.02, -0.005, 0.005, 0.02, 0.05)
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if not (1e-5 < h[j] < 1 - 1e-5 and 1e-5 < h[l] < 1 - 1e-5):
            continue
        best_c, best_d = c, 0.0
        for d in steps:
            hj2, hl2 = h[j] + d, h[l] - d
            if hj2 < 0 or hj2 > 1 or hl2 < 0 or hl2 > 1:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj2, hl2
            cn = compute_c5(hn)
            if cn < best_c - 1e-13:
                best_c, best_d = cn, d
        if best_d:
            h[j] = np.clip(h[j] + best_d, 0, 1)
            h[l] = np.clip(h[l] - best_d, 0, 1)
            c = best_c
    return fix_sum(h, n)

def coord_sweep(h, n, n_passes=3):
    c = compute_c5(h)
    for _ in range(n_passes):
        for j in range(n):
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            best_c, best_h = c, h.copy()
            for delta in (-0.06, -0.02, 0.02, 0.06):
                nv = h[j] + delta
                if nv < 1e-5 or nv > 1 - 1e-5:
                    continue
                hn = h.copy()
                hn[j] = nv
                dn = h[j] - nv
                lo = (j - 1 + n) % n
                ro = (j + 1) % n
                hn[lo] = np.clip(hn[lo] + 0.5 * dn, 0, 1)
                hn[ro] = np.clip(hn[ro] + 0.5 * dn, 0, 1)
                hn = fix_sum(hn, n)
                cn = compute_c5(hn)
                if cn < best_c - 1e-12:
                    best_c, best_h = cn, hn
            h = best_h
        h = fix_sum(h, n)
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)
