def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)

        h[m] += d / k

def lse_objective_and_grad(h, n, t, lam):
    """Log-sum-exp objective with sum penalty, returns (value, gradient)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # log-sum-exp

    log_sum_exp = cmax + np.log(np.sum(np.exp(corr - cmax)))
    val = (log_sum_exp / t) * dx
    
    # gradient of log-sum-exp part
    w = np.exp(corr - cmax) / np.sum(np.exp(corr - cmax))  # softmax weights
    grad = np.zeros(n)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def lse_grad(h, n, t):
    """Compute (value, gradient) of LSE-smoothed max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    exp_c = np.exp(corr - cmax)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    val = (cmax + np.log(sum_exp)) / t * dx
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-12:
            continue
        s = m - (n - 1)
        if s == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif s > 0:
            nv = n - s
            grad[:nv] += wm * (1.0 - h[s:])
            grad[s:] += -wm * h[:nv]
        else:
            s2 = -s
            nv = n - s2
            grad[s2:] += wm * (1.0 - h[:nv])
            grad[:nv] += -wm * h[s2:]
    grad *= dx / t
    return val, grad

def lbfgs_run(h, n, t, maxiter=300):
    def fun(x):
        return lse_grad(x, n, t)
    try:
        r = optimize.minimize(fun, h, jac=True, method='L-BFGS-B',
            bounds=[(0., 1.)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
        return fix_sum(r.x, n)
    except Exception:
        return h

def topk_grad(h, n, k=30):
    """Gradient of sum of top-k correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k:]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / k
    val = np.sum(corr[idx]) * dx
    grad = np.zeros(n)
    for m in idx:
        s = m - (n - 1)
        if s == 0:
            grad += (1.0 - 2.0 * h) / k
        elif s > 0:
            nv = n - s
            grad[:nv] += (1.0 - h[s:]) / k
            grad[s:] += -h[:nv] / k
        else:
            s2 = -s
            nv = n - s2
            grad[s2:] += (1.0 - h[:nv]) / k
            grad[:nv] += -h[s2:] / k
    grad *= dx
    return val, grad

def proj_grad_descent(h, n, k, n_steps, lr):
    for _ in range(n_steps):
        _, grad = topk_grad(h, n, k)
        h = h - lr * grad
        h = fix_sum(h, n)
    return h

def coord_polish(h, n, n_pass=2):
    for _ in range(n_pass):
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            base_c5 = compute_c5(h)
            best = h.copy()
            best_c = base_c5
            for delta in (-0.04, -0.015, -0.004, 0.004, 0.015, 0.04):
                hn = h.copy()
                hn[:] += -delta / (n - 1)
                hn[j] = min(max(h[j] + delta, 0.0), 1.0)
                hn = np.clip(hn, 0.0, 1.0)
                c = compute_c5(hn)
                if c < best_c - 1e-13:
                    best_c = c
                    best = hn
            h = best
        h = fix_sum(h, n)
    return h

def pair_move(h, n, rng, n_moves=400):
    for _ in range(n_moves):
        j, l = rng.choice(n, 2, replace=False)
        if not (1e-3 < h[j] < 1.0 - 1e-3 and 1e-3 < h[l] < 1.0 - 1e-3):
            continue
        base_c5 = compute_c5(h)
        best_d = 0.0
        for d in (-0.03, -0.01, -0.003, 0.003, 0.01, 0.03):
            if not (1e-4 < h[j]+d < 1.0-1e-4 and 1e-4 < h[l]-d < 1.0-1e-4):
                continue
            hn = h.copy()
            hn[j], hn[l] = h[j]+d, h[l]-d
            if compute_c5(hn) < base_c5 - 1e-13:
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def optimize_at_n(h, n, t_arr, lbfgs_iters, time_limit):
    t0 = time.time()
    for t in t_arr:
        if time.time() - t0 > time_limit:
            break
        h = lbfgs_run(h, n, t, lbfgs_iters)
    # Top-k gradient descent
    if time.time() - t0 < time_limit:
        for k in (20, 50, 100):
            h = proj_grad_descent(h, n, k, 200, 0.01)
    # Coordinate polish
    if time.time() - t0 < time_limit:
        h = coord_polish(h, n, 2)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)
    t0 = time.time()
    
    h = np.asarray(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)
    
    # Phase 1: L-BFGS-B with decreasing temperature at n=280
    temps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for t in temps:
        if time.time() - t0 > budget_s - 400:
            break
        h = lbfgs_run(h, n, t, 400)
        c5 = compute_c5(h)
    
    # Phase 2: Top-k gradient + polish
    if time.time() - t0 < budget_s - 300:
        for k in (20, 40, 80, 200):
            h = proj_grad_descent(h, n, k, 300, 0.008)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 250:
        h = coord_polish(h, n, 3)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 200:
        h = pair_move(h, n, rng, 500)
        c5 = compute_c5(h)
    
    # Phase 3: Restarts
    for i in range(15):
        if time.time() - t0 > budget_s - 150:
            break
        ht = h.copy()
        mode = i % 5
        if mode == 0:
            ht += np.random.randn(n) * 0.015
        elif mode == 1:
            idx = np.random.choice(n, 8, replace=False)
            ht[idx] += np.random.randn(8) * 0.1
        elif mode == 2:
            ht += np.random.randn(n) * 0.008
        elif mode == 3:
            idx = np.random.choice(n, 20, replace=False)
            ht[idx] += np.random.randn(20) * 0.05
        else:
            ht += rng.uniform(-0.02, 0.02, n)
        ht = fix_sum(ht, n)
        for t in (0.005, 0.001, 0.0003):
            ht = lbfgs_run(ht, n, t, 300)
        ht = proj_grad_descent(ht, n, 50, 200, 0.005)
        ht = coord_polish(ht, n, 1)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() - t0 < budget_s - 60:
                h = pair_move(h, n, rng, 300)
                c5 = compute_c5(h)
    
    # Phase 4: Higher resolutions
    best_h, best_c5, best_n = h, c5, n
    for n2 in (320, 400, 500):
        if time.time() - t0 > budget_s - 60:

        h2 = upsample(h, n2)
        for t in (0.01, 0.003, 0.001, 0.0003):
            if time.time() - t0 > budget_s - 30:

            h2 = lbfgs_run(h2, n2, t, 300)
        h2 = proj_grad_descent(h2, n2, 50, 200, 0.005)
        h2 = coord_polish(h2, n2, 1)

    if time.time() - t0 < budget_s - 20:

        h = fix_sum(h, n)
    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
