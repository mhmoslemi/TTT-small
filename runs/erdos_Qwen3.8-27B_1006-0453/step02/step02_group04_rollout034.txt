import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = np.sum(interior)
    if ni > 0 and abs(diff) < ni * 0.4:
        h[interior] += diff / ni
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                if diff <= 1e-12: break
        else:
            d = -diff; idx = np.argsort(-h)
            for i in idx:
                r = min(d, h[i]); h[i] -= r; d -= r
                if d <= 1e-12: break
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, T):
    n = len(h); dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    cm = np.max(corr)
    if T < 1e-14: return cm * dx
    sh = np.clip((corr - cm)/T, -80.0, 0.0)
    e = np.exp(sh); s = np.sum(e)
    return (cm + T * np.log(max(s, 1e-300))) * dx

def lse_grad(h, T):
    n = len(h); dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    cm = np.max(corr)
    if T < 1e-14:
        pr = np.zeros(2*n-1); pr[np.argmax(corr)] = 1.0
    else:
        sh = np.clip((corr-cm)/T, -80.0, 0.0)
        e = np.exp(sh); pr = e / np.sum(e)
    g = np.zeros(n)
    for m in range(2*n-1):
        p = pr[m]
        if p < 1e-15: continue
        k = m - (n-1)
        if k == 0:
            g += p*(1.0 - 2.0*h)
        elif k > 0:
            g[k:] += p*(1.0 - h[:n-k])
            g[:n-k] += -p*h[k:]
        else:
            s = -k
            g[:n-s] += p*(1.0 - h[s:])
            g[s:] += -p*h[:n-s]
    return g * dx

def sum_con(h, n):
    return np.sum(h) - n*0.5

def do_slsqp_grad(h, n, T, max_it=2000):
    try:
        res = opt.minimize(
            lse_obj, h.copy(), args=(T,),
            jac=lambda x, t: lse_grad(x, t),
            method='SLSQP',
            bounds=[(0.0, 1.0)]*n,
            constraints=[{'type':'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_it, 'ftol': 1e-18})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def active_set_slsqp(h, n, T, top_k=8, max_it=2000):
    corr = np.correlate(h, 1.0-h, mode='full')
    top = np.argsort(corr)[-top_k:]
    def obj_top(x, T, top):
        c = np.correlate(x, 1.0-x, mode='full')
        tv = c[top]
        return float(np.max(tv) + T*np.log(np.sum(np.exp((tv-np.max(tv))/T))/len(tv)))
    try:
        res = opt.minimize(
            obj_top, h.copy(), args=(T, top),
            method='SLSQP',
            bounds=[(0.0, 1.0)]*n,
            constraints=[{'type':'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_it, 'ftol': 1e-18})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample(h, new_n):
    on = len(h)
    xo = np.arange(on)*(2.0/on)
    xn = np.arange(new_n)*(2.0/new_n)
    return fix_sum(np.interp(xn, xo, h, period=2.0), new_n)

def make_sinusoidal(n, freq=1.0, phase=0.0, amp=0.4):
    x = np.arange(n)*(2.0/n)
    h = 0.5 + amp * np.sin(2*np.pi*freq*x/2.0 + phase)
    return fix_sum(np.clip(h, 0, 1), n)

def make_block(n, ratio=0.5):
    h = np.zeros(n)
    n_fill = int(n * ratio)
    h[:n_fill] = 1.0
    return fix_sum(h, n)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = np.sum(interior)
    if ni > 0 and abs(diff) < ni * 0.4:
        h[interior] += diff / ni
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                if diff <= 1e-13: break
        else:
            d = -diff; idx = np.argsort(-h)
            for i in idx:
                r = min(d, h[i]); h[i] -= r; d -= r
                if d <= 1e-13: break
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, T):
    n = len(h); dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    if T < 1e-14: return cm * dx
    sh = np.clip((corr - cm) / T, -80.0, 0.0)
    e = np.exp(sh); s = np.sum(e)
    return (cm + T * np.log(max(s, 1e-300))) * dx

def lse_grad(h, T):
    n = len(h); dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    if T < 1e-14:
        pr = np.zeros(2 * n - 1); pr[np.argmax(corr)] = 1.0
    else:
        sh = np.clip((corr - cm) / T, -80.0, 0.0)
        e = np.exp(sh); pr = e / np.sum(e)
    g = np.zeros(n)
    for m in range(2 * n - 1):
        p = pr[m]
        if p < 1e-15: continue
        k = m - (n - 1)
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += p * (1.0 - h[:n - k])
            g[:n - k] += -p * h[k:]
        else:
            s = -k
            g[:n - s] += p * (1.0 - h[s:])
            g[s:] += -p * h[:n - s]
    return g * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def do_slsqp(h, n, T, max_it=2000):
    try:
        res = opt.minimize(
            lse_obj, h.copy(), args=(T,),
            jac=lambda x, t: lse_grad(x, t),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_it, 'ftol': 1e-18})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def active_set_opt(h, n, T, top_k=8, max_it=2000):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-top_k:]
    def obj_top(x, T, top):
        c = np.correlate(x, 1.0 - x, mode='full')
        tv = c[top]
        mx = np.max(tv)
        sh = np.clip((tv - mx) / T, -60.0, 0.0)
        return float((mx + T * np.log(np.sum(np.exp(sh)) / len(tv))) * (2.0 / n))
    try:
        res = opt.minimize(
            obj_top, h.copy(), args=(T, top),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_it, 'ftol': 1e-18})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample(h, new_n):
    on = len(h)
    xo = np.arange(on) * (2.0 / on)
    xn = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(xn, xo, h, period=2.0), new_n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h_best = np.array(initial_h_values, dtype=float).copy()

    h_best = fix_sum(h_best, n_best)
    c5_best = compute_c5(h_best)

    def try_update(hc, nc):

        hc = fix_sum(hc, nc)
        cv = compute_c5(hc)
        if cv < c5_best - 1e-14:
            h_best, n_best, c5_best = hc, nc, cv
            return True
        return False

    # Phase 1: Gradient SLSQP temperature sweep at n_best
    for T in [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]:
        if budget_s - elapsed() < 100: break
        h_opt = do_slsqp(h_best, n_best, T, max_it=3000)

    # Phase 2: Active-set refinement (top-K shifts)

        try_update(h_ref, n_best)

    # Phase 3: Coarse-to-fine from structured starts

        for start_fn in [

    # Phase 6: Try higher resolution

            if budget_s - elapsed() < 100: break

        h_best = do_slsqp(h_best, n_best, 0.0003, max_it=4000)
        try_update(h_best, n_best)
    if budget_s - elapsed() > 50:
        h_best = active_set_opt(h_best, n_best, 0.0001, top_k=4, max_it=3000)
