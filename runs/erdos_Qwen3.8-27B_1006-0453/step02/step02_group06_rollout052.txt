def proj_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):

        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def soft_max_obj(h, T):

                          bounds=[(0.0, 1.0)] * n, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < float('inf'):
            return proj_feasible(res.x, n)
    except:
        pass
    return h0
