import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return np.interp(xd, xs, h)

def binarize(h, n):
    hb = (h > 0.5).astype(float)
    diff = n // 2 - int(hb.sum())
    if diff > 0:
        fringe = np.where(hb == 0)[0]
        order = np.argsort(-h[fringe])[:diff]
        hb[fringe[order]] = 1
    elif diff < 0:
        fringe = np.where(hb == 1)[0]
        order = np.argsort(h[fringe])[-diff:]
        hb[fringe[order]] = 0
    return hb

def effect_vectors(h, pos, n):
    """Compute effect vectors for flipping positions in `pos`.
    For pos where h==0: effect of 0->1
    For pos where h==1: effect of 1->0
    Returns matrix of shape (len(pos), 2n-1)
    """
    m = len(pos)
    if m == 0:
        return np.zeros((0, 2*n-1))
    shifts = np.arange(2*n-1) - (n-1)  # shape (2n-1)
    pos_arr = pos.astype(int)[:, None]  # (m, 1)
    # For 0->1 flip at position p:
    #   +（1-h[p+shift]) where 0<=p+shift<n
    #   - h[p-shift] where 0<=p-shift<n
    idx_plus = pos_arr + shifts[None, :]
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    val_plus = np.where(valid_plus, 1.0 - h[np.clip(idx_plus, 0, n-1)], 0.0)
    idx_minus = pos_arr - shifts[None, :]
    valid_minus = (idx_minus >= 0) & (idx_minus < n)
    val_minus = np.where(valid_minus, h[np.clip(idx_minus, 0, n-1)], 0.0)
    
    # For positions where h==0: effect = val_plus - val_minus (positive contribution)
    # For positions where h==1: effect = -val_plus + val_minus (negative contribution)
    is_one = h[pos] > 0.5
    # For 0->1: +（1-h[p+shift]) - h[p-shift]
    # For 1->0: -（1-h[p+shift]) + h[p-shift]
    # Combined: sign * (val_plus - val_minus) where sign=+1 for 0, -1 for 1
    sign = np.where(h[pos] < 0.5, 1.0, -1.0)[:, None]
    return sign * (val_plus - val_minus)

def binary_local_search(h_bin, n, max_flips=5000, time_limit=300, rng=None):
    """Greedy binary local search."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.copy().astype(float)
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    curr_max = corr.max()
    best_max = curr_max
    best_h = h.copy()
    t0 = time.time()
    
    stalls = 0
    flips = 0
    while flips < max_flips and time.time() - t0 < time_limit:
        if stalls > 300:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        
        # Compute effect vectors
        ep = effect_vectors(h, zeros, n)  # (nz, 2n-1)
        en = effect_vectors(h, ones, n)   # (no, 2n-1)
        
        best_val = np.inf
        best_i = best_j = -1
        for i in range(nz):
            base = corr + ep[i]
            new_maxes = (en + base[None, :]).max(axis=1)
            j_loc = new_maxes.argmin()
            if new_maxes[j_loc] < best_val:
                best_val = new_maxes[j_loc]
                best_i = i
                best_j = j_loc
        
        if best_val < curr_max - 1e-10:
            i_pos = zeros[best_i]
            j_pos = ones[best_j]
            h[i_pos] = 1.0
            h[j_pos] = 0.0
            corr = corr + ep[best_i] + en[best_j]
            curr_max = best_val
            if curr_max < best_max - 1e-12:
                best_max = curr_max
                best_h = h.copy()
            flips += 1
            stalls = 0
        else:
            # Try random perturbations to escape local optimum
            if stalls > 50:
                n_rand = min(nz * no, 200)
                for _ in range(n_rand):
                    i = rng.randint(nz)
                    j = rng.randint(no)
                    i_pos = zeros[i]
                    j_pos = ones[j]
                    new_corr = corr + ep[i] + en[j]
                    new_max = new_corr.max()
                    # Accept with probability based on degradation
                    if new_max < curr_max:
                        h[i_pos] = 1.0
                        h[j_pos] = 0.0
                        corr = new_corr
                        curr_max = new_max
                        if curr_max < best_max:
                            best_max = curr_max
                            best_h = h.copy()
                        break
                    elif rng.random() < np.exp(-(new_max - curr_max) / 0.5):
                        h[i_pos] = 1.0
                        h[j_pos] = 0.0
                        corr = new_corr
                        curr_max = new_max
                        break
            stalls += 1
    
    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - (time.time() - t0)
    
    # Initialize from parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(200, 0.5) + rng.randn(200) * 0.05
        n_parent = 200
    h_parent = proj(h_parent, n_parent)
    
    # Best overall
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary local search at multiple resolutions
    for n_try in [100, 140, 200, 280]:
        if tleft() < 60:
            break
        h_n = interp_to(h_parent, n_try) if n_try != n_parent else h_parent.copy()
        h_bin = binarize(h_n, n_try)
        
        # Multiple restarts with perturbation
        for restart in range(3):
            if tleft() < 30:
                break
            if restart > 0:
                # Random perturbation: flip a few pairs
                h_pert = h_bin.copy()
                zeros = np.where(h_pert == 0)[0]
                ones = np.where(h_pert == 1)[0]
                n_flip = 3 + restart * 2
                for _ in range(n_flip):
                    i = zeros[rng.randint(len(zeros))]
                    j = ones[rng.randint(len(ones))]
                    h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
                h_work = h_pert
            else:
                h_work = h_bin.copy()
            
            time_alloc = min(tleft() * 0.7, 80)
            h_res, c5_res = binary_local_search(h_work, n_try, max_flips=10000,
                                                 time_limit=time_alloc, rng=rng)
            c5_val = compute_c5(h_res)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_res
            h_bin = h_res  # use result as next start
    
    # Phase 2: SLSQP refinement on best binary solution
    if tleft() > 80:
        n_final = len(best_h)
        h_cont = best_h.copy()
        
        def obj_smo(h, t):
            n = len(h)
            dx = 2.0 / n
            corr = np.correlate(h, 1.0-h, mode='full')
            mx = corr.max()
            s = t * (corr - mx)
            s = np.clip(s, -50, 50)
            return (mx + np.log(np.sum(np.exp(s))) / t) * dx
        
        bnds = [(0.0, 1.0)] * n_final
        def ceq(x):
            return np.array([np.sum(x) - n_final/2.0])
        cons = {'type': 'eq', 'fun': ceq, 'jac': lambda x: np.ones((1, n_final))}
        
        for t_val in [100, 500, 2000]:
            if tleft() < 20:
                break
            try:
                r = minimize(obj_smo, h_cont, args=(t_val,), method='SLSQP',
                            bounds=bnds, constraints=cons,
                            options={'maxiter': 1000, 'ftol': 1e-14})
                h_new = proj(np.clip(r.x, 0, 1), n_final)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                h_cont = h_new
            except Exception:
                pass
    
    # Phase 3: Another round of binary search on refined solution
    if tleft() > 40:
        n_fin = len(best_h)
        h_b2 = binarize(best_h, n_fin)
        h_res2, _ = binary_local_search(h_b2, n_fin, max_flips=5000,
                                         time_limit=min(tleft()*0.8, 60), rng=rng)
        c5_c = compute_c5(h_res2)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_res2
        # Also try continuous version
        h_c2 = proj(interp_to(h_res2, n_fin), n_fin)
        c5_c2 = compute_c5(h_c2)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2
    
    return best_h, compute_c5(best_h), len(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return np.interp(xd, xs, h)

def binarize(h, n):
    hb = (h > 0.5).astype(float)
    diff = n // 2 - int(hb.sum())
    if diff > 0:
        idx = np.where(hb == 0)[0]
        order = np.argsort(-h[idx])[:diff]
        hb[idx[order]] = 1.0
    elif diff < 0:
        idx = np.where(hb == 1)[0]
        order = np.argsort(h[idx])[-diff:]
        hb[idx[order]] = 0.0
    return hb

def effect_vectors(h, pos, n):
    m = len(pos)
    if m == 0:
        return np.zeros((0, 2 * n - 1))
    shifts = np.arange(2 * n - 1) - (n - 1)
    pa = pos.astype(np.int64)[:, None]
    ip = pa + shifts[None, :]
    vp = (ip >= 0) & (ip < n)
    ipc = np.clip(ip, 0, n - 1)
    ipc[np.logical_not(vp)] = 0
    val_p = np.where(vp, 1.0 - h[ipc], 0.0)
    im = pa - shifts[None, :]
    vm = (im >= 0) & (im < n)
    imc = np.clip(im, 0, n - 1)
    imc[np.logical_not(vm)] = 0
    val_m = np.where(vm, h[imc], 0.0)
    sign = np.where(h[pos] < 0.5, 1.0, -1.0)[:, None]
    return sign * (val_p - val_m)

def binary_local_search(h_bin, n, max_flips=8000, time_limit=300, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.copy().astype(float)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    curr_max = corr.max()
    best_max = curr_max
    best_h = h.copy()
    t0 = time.time()
    stalls = 0
    flips = 0
    while flips < max_flips and time.time() - t0 < time_limit:
        if stalls > 400:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        ep = effect_vectors(h, zeros, n)
        en = effect_vectors(h, ones, n)
        best_val = np.inf
        best_i = best_j = -1
        for i in range(nz):
            base = corr + ep[i]
            nm = (en + base[None, :]).max(axis=1)
            jloc = nm.argmin()
            if nm[jloc] < best_val:
                best_val = nm[jloc]
                best_i = i
                best_j = jloc
        if best_val < curr_max - 1e-12:
            ip, jp = zeros[best_i], ones[best_j]
            h[ip] = 1.0
            h[jp] = 0.0
            corr = corr + ep[best_i] + en[best_j]
            curr_max = best_val
            if curr_max < best_max - 1e-14:
                best_max = curr_max
                best_h = h.copy()
            flips += 1
            stalls = 0
        else:
            if stalls > 80:
                found = False
                for _ in range(500):
                    ii = rng.randint(nz)
                    jj = rng.randint(no)
                    nc = corr + ep[ii] + en[jj]
                    nm = nc.max()
                    if nm < curr_max or (stalls > 200 and rng.random() < np.exp(-(nm - curr_max) / 2.0)):
                        ip, jp = zeros[ii], ones[jj]
                        h[ip] = 1.0
                        h[jp] = 0.0
                        corr = nc
                        curr_max = nm
                        if curr_max < best_max:
                            best_max = curr_max
                            best_h = h.copy()
                        found = True
                        break
                if not found:
                    stalls += 10
                else:
                    stalls = 0
            else:
                stalls += 1
    return best_h, best_max

    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search at multiple resolutions with restarts
    for n_try in [100, 140, 200, 280]:
        if tleft() < 50:

        h_n = interp_to(h_par, n_try)
        h_bin = binarize(h_n, n_try)
        n_restarts = 4 if n_try <= 140 else 3

            if r == 0:

            else:
                h_work = h_bin.copy()
                zs = np.where(h_work == 0)[0]
                os_ = np.where(h_work == 1)[0]

                                           time_limit=tl, rng=rng)
            c5v = compute_c5(h_res)

                    best_c5 = c5n
                    best_h = h_new
                h_c = h_new

        h_bf = binarize(best_h, n_pf)

        c5cp = compute_c5(h_cp)
        if c5cp < best_c5:
            best_c5 = c5cp
            best_h = h_cp

    return best_h, compute_c5(best_h), len(best_h)
