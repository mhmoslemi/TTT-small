        ts0 = n_base / 2.0
        dx0 = 2.0 / n_base
        for s2 in range(4):
            if time.time() > deadline - 40:
                break
            h_p = proj(bh + 0.06 * rng.randn(n_base), ts0)
            h_c, c_c = cmaes(h_p, n_base, ts0, 40, seed + 100 + s2)
            if c_c < bc:
                bh, bc = h_c, c_c
            h_s = slsqp(bh, dx0, ts0, 150)
            c_s = compute_c5(h_s, dx0)
            if c_s < bc:
                bh, bc = h_s, c_s

    # Try nearby n values with CMA-ES
    for dn in (0, 1, -1, 2, -2, 3, -3, 4, -4):
        if time.time() > deadline - 60:
            break
        nt = n_base + dn
        if nt < 30 or nt > 80:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        # Interpolate best solution to new n
        if bh is not None and len(bh) != nt:
            h_ref = np.interp(np.linspace(0, 1, nt, endpoint=False),
                              np.linspace(0, 1, len(bh), endpoint=False), bh)
            h_ref = proj(h_ref, tst)
        else:
            h_ref = proj(np.full(nt, tst / nt), tst)
        h_c, c_c = cmaes(h_ref, nt, tst, min(35, max(10, budget_s * 0.04)), seed + 50 + dn)
        if c_c < bc:
            bh, bc = h_c, c_c
        h_s = slsqp(bh, dxt, tst, 150)
        c_s = compute_c5(h_s, dxt)
        if c_s < bc:
            bh, bc = h_s, c_s
        # Gradient descent refinement
        if time.time() < deadline - 30:
            h_g, c_g = grad_desc(bh, dxt, tst, 800, deadline - 20, rng)
            if c_g < bc:
                bh, bc = h_g, c_g
            h_s = slsqp(bh, dxt, tst, 100)
            c_s = compute_c5(h_s, dxt)
            if c_s < bc:
                bh, bc = h_s, c_s

    # Final refinement loop
    if bh is not None:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        for amp in (0.03, 0.015, 0.008, 0.004):
            if time.time() > deadline - 15:
                break
            h_p = proj(bh + amp * rng.randn(nb), tsb)
            h_c, c_c = cmaes(h_p, nb, tsb, 20, seed + 200)
            if c_c < bc:
                bh, bc = h_c, c_c
            h_s = slsqp(bh, dxb, tsb, 200)
            c_s = compute_c5(h_s, dxb)
            if c_s < bc:
                bh, bc = h_s, c_s
            h_cd, c_cd = coord_desc(bh, dxb, tsb, 20, deadline - 5, rng)
            if c_cd < bc:
                bh, bc = h_cd, c_cd
            h_s = slsqp(bh, dxb, tsb, 100)
            c_s = compute_c5(h_s, dxb)
            if c_s < bc:
                bh, bc = h_s, c_s

    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh, 0.04)

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
