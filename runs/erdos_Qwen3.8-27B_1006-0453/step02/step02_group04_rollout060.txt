def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def fix_sum(h, n=None):
    if n is None:
        n = len(h)
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.35:
        h[interior] += diff / n_int
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-13:
                break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff <= 1e-13:

def smooth_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n

    c_max = float(np.max(corr))

    return (c_max + T * np.log(max(s, 1e-300))) * dx

    return np.sum(x) - n * 0.5

                           method='SLSQP',

                           constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],

        if hasattr(res, 'x') and res.x is not None:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    if T < 1e-15:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(corr)] = 1.0
    else:
        shifted = np.clip((corr - c_max) / T, -80.0, 0.0)
        exp_vals = np.exp(shifted)
        total = float(np.sum(exp_vals))
        weights = exp_vals / max(total, 1e-300)
    obj_val = (c_max + T * np.log(max(float(np.sum(exp_vals if T >= 1e-15 else 1.0)), 1e-300))) * dx
    s = float(np.sum(h)) - n * 0.5
    obj_val += lam * s * s
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        ell = k_idx - (n - 1)
        lo1 = max(0, ell)
        hi1 = min(n - 1, ell + n - 1)
        if lo1 <= hi1:

        if c5_cand < c5_best - 1e-15:
            h_best = h_opt
            c5_best = c5_cand

    # Micro-perturbation + one last push
    if budget_s - _elapsed() > 50:
        h_micro = h_best + np.random.randn(n_best) * 0.004
        h_micro = np.clip(h_micro, 0.0, 1.0)
        h_micro = fix_sum(h_micro, n_best)
        h_opt = do_lbfgsb(h_micro, n_best, 0.0005, lam, max_iter=3000)
        h_opt = do_slsqp(h_opt, n_best, 0.0002, 2000)
        c5_cand = compute_c5(h_opt)
        if c5_cand < c5_best - 1e-15:
            h_best = h_opt
            c5_best = c5_cand

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    n_final = len(h_best)
    return (h_best, c5_final, n_final)
