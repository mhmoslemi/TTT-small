import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def slsqp(h, T, n, maxiter=600, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def trust_constr_opt(h, T, n, maxiter=600):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5, h, args=(T,), method='trust-constr',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'gtol': 1e-10})
    return project_feasible(np.clip(res.x, 0, 1), n)

def subgrad_step(h, n, frac=0.08):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    thresh = cmax - frac * max(cmax, 1e-10)

    grad = np.zeros(n)
    for m_idx in active:
        lag = m_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] += -h[lag:]
        else:
            k2 = -lag
            grad[:n - k2] += (1.0 - h[k2:])
            grad[k2:] += -h[:n - k2]
    grad /= len(active)
    return grad

def block_parameterize(n, breakpoints, heights):
    h = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False)
    for i, bp in enumerate(breakpoints):
        mask = (x >= bp[0]) & (x < bp[1]) if i < len(breakpoints) - 1 else (x >= bp[0]) & (x <= 2.0)
        h[mask] = heights[i]
    return project_feasible(np.clip(h, 0, 1), n)

def block_obj(params, n):
    K = (len(params) - 1) // 2
    bp_edges = params[:K + 1]
    heights = params[K + 1:]
    bps = [(bp_edges[i], bp_edges[i + 1]) for i in range(K)]
    h = block_parameterize(n, bps, heights)
    return compute_c5(h)
