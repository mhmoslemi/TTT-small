import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    # Use bisection
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(80):
        tau = (lo + hi) / 2
        s = np.sum(np.clip(h - tau, 0, 1))
        if s > target:
            lo = tau  # need to decrease more → increase tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lag_gradient(h, lag, n):
    """Compute d(corr[lag])/dh as a vector of length n"""
    grad = np.zeros(n)
    if lag >= 0:
        valid = n - lag
        if valid > 0:
            grad[lag:] += (1.0 - h[:valid])
            grad[:valid] -= h[lag:]
    else:
        L = -lag
        valid = n - L
        if valid > 0:
            grad[:valid] += (1.0 - h[L:])
            grad[L:] -= h[:valid]
    return grad

def subgradient_step(h, n, topk, temp, alpha):
    """One subgradient step"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k lags
    idx = np.argpartition(corr, -topk)[-topk:]
    c_active = corr[idx]
    # Soft weights
    exps = np.exp(temp * (c_active - c_active.max()))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(len(idx)):
        m = idx[k]
        lag = m - (n - 1)
        grad += weights[k] * lag_gradient(h, lag, n)
    
    h_new = h - alpha * grad
    return project_feasible(h_new, n), corr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - (time.time() - t0)
    
    # Start from initial

        bh = project_feasible(np.array(initial_h_values, dtype=float), n)
    except Exception:
        n = 280
        bh = project_feasible(np.full(n, 0.5) + rng.randn(n) * 0.05, n)
    
    bc = compute_c5(bh)
    
    # Phase 1: Subgradient optimization at n=280
    h = bh.copy()
    for topk in [5, 10, 20, 5]:
        for temp in [20, 50, 150]:
            if tleft() < 30:
                break
            alpha = 0.05
            for it in range(500):
                if tleft() < 5:
                    break
                h, corr = subgradient_step(h, n, topk, temp, alpha)
                c5 = np.max(corr) * (2.0/n)
                if c5 < bc:
                    bc = c5
                    bh = h.copy()
                # Decay alpha
                alpha *= 0.999
            bc = min(bc, compute_c5(h))
    
    # Phase 2: Binary local search
    if tleft() > 100:
        h_bin = project_feasible((h > 0.5).astype(float), n)
        for it in range(20000):
            if tleft() < 10:
                break
            # Try random flips
            i = rng.randint(n)
            j = rng.randint(n)
            if h_bin[i] > 0.5 and h_bin[j] < 0.5:
                # Swap: h[i] 1→0, h[j] 0→1
                h_bin[i] = 0
                h_bin[j] = 1
                c5_new = compute_c5(h_bin)
                if c5_new < bc:
                    bc = c5_new
                    bh = h_bin.copy()
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
        h = h_bin
    
    # Phase 3: Refine with subgradient from binary
    if tleft() > 60:
        h = bh.copy()
        for topk in [3, 7, 15]:
            for temp in [50, 200, 500]:
                if tleft() < 20:
                    break
                alpha = 0.03
                for it in range(500):
                    if tleft() < 5:

                    c5 = np.max(corr) * (2.0/n)
                    if c5 < bc:
                        bc = c5
                        bh = h.copy()
                    alpha *= 0.9995

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Bisection for the right shift
    lo, hi = -max(h.max(), 1.0), -max(1.0 - h.min(), 1.0)
    # Actually, we want: sum(clip(h + c, 0, 1)) = target
    # If diff > 0, we need to increase, so c > 0
    # If diff < 0, we need to decrease, so c < 0

        s = np.sum(np.clip(h + c, 0.0, 1.0))
        if s < target:
            lo = c
        else:
            hi = c
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_hess(h, topk, temp, n):
    """Returns gradient of smooth approx to max_k corr[k]"""
    corr = np.correlate(h, 1.0 - h, mode='full')

    exps = np.exp(temp * (ca - emax))

            grad[:v] -= wts[k] * h[lag:]
        else:
            grad[:v] += wts[k] * (1.0 - h[L:])
            grad[L:] -= wts[k] * h[:v]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(70):
        c = (lo + hi) / 2.0
        ss = np.sum(np.clip(h + c, 0.0, 1.0))
        if ss < target:
            lo = c
        else:
            hi = c
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_compute(h, topk, temp, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -topk)[-topk:]
    ca = corr[idx]
    emax = ca.max()
    exps = np.exp(temp * (ca - emax))
    wsum = exps.sum()
    wts = exps / wsum if wsum > 0 else np.ones(len(idx)) / len(idx)
    grad = np.zeros(n)
    for k in range(len(idx)):
        lag = idx[k] - (n - 1)
        L = abs(lag)
        v = n - L
        if v <= 0:
            continue
        w = wts[k]
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:v])
            grad[:v] -= w * h[lag:]
        else:
            grad[:v] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:v]
    return grad, corr

def heavy_ball_phase(h_start, n, topk, temp, alpha0, decay, n_iter, t0, budget_s):
    h = h_start.copy()
    v = np.zeros(n)
    mu = 0.82
    alpha = alpha0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 3.0:
            break
        grad, corr = subgrad_compute(h, topk, temp, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        v = mu * v - alpha * grad / gn
        h = project_feasible(h + v * 0.5, n)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        alpha *= decay
    return best_h, best_c5

def binary_search_swap(h_start, n, n_trials, rng, t0, budget_s):
    h = h_start.copy()
    h = (h > 0.5).astype(float)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for trial in range(n_trials):
        if budget_s - (time.time() - t0) < 5.0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5
