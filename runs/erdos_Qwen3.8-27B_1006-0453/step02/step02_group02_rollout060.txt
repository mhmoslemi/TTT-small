```python
def compute_gradient_fast(h, u, dx):
    n = len(h)
    k = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    j = np.arange(n)  # shape (n,)
    # First term matrix: (1-h[j-k[m]]) for valid (j,m) pairs
    diff1 = j[:, None] - k[None, :]  # shape (n, 2n-1)
    valid1 = (diff1 >= 0) & (diff1 < n)
    term1 = np.where(valid1, (1-h[diff1]), 0) * u[None, :]
    # Second term matrix: h[j+k[m]] for valid (j,m) pairs
    diff2 = j[:, None] + k[None, :]  # shape (n, 2n-1)
    valid2 = (diff2 >= 0) & (diff2 < n)
    term2 = np.where(valid2, h[diff2], 0) * u[None, :]
    grad = (term1.sum(axis=1) - term2.sum(axis=1)) * dx
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(100):
        d = tgt - np.sum(h)
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    s = t * (corr - M)
    s -= np.max(s)
    e = np.exp(s)
    w = e / np.sum(e)
    obj = (M + np.log(np.sum(e)) / t) * dx
    
    k_lags = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)
    d1 = j[:, None] - k_lags[None, :]
    v1 = (d1 >= 0) & (d1 < n)
    t1 = np.where(v1, 1.0 - h[np.clip(d1, 0, n - 1)], 0.0) * w[None, :]
    d2 = j[:, None] + k_lags[None, :]
    v2 = (d2 >= 0) & (d2 < n)
    t2 = np.where(v2, h[np.clip(d2, 0, n - 1)], 0.0) * w[None, :]
    grad = (t1.sum(axis=1) - t2.sum(axis=1)) * dx
    return obj, grad, w

def lfgb_one(h0, n, t, pen_lam, max_iter):
    k_lags = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)
    dx = 2.0 / n
    tgt = n / 2.0
    
    def f_and_g(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = np.max(corr)
        s = t * (corr - M)
        s -= np.max(s)
        e = np.exp(s)
        w = e / np.sum(e)
        fval = (M + np.log(np.sum(e)) / t) * dx + pen_lam * (np.sum(h) - tgt) ** 2
        d1 = j_idx[:, None] - k_lags[None, :]
        v1 = (d1 >= 0) & (d1 < n)
        t1 = np.where(v1, 1.0 - h[np.clip(d1, 0, n - 1)], 0.0) * w[None, :]
        d2 = j_idx[:, None] + k_lags[None, :]
        v2 = (d2 >= 0) & (d2 < n)
        t2 = np.where(v2, h[np.clip(d2, 0, n - 1)], 0.0) * w[None, :]
        g = (t1.sum(axis=1) - t2.sum(axis=1)) * dx + 2.0 * pen_lam * (np.sum(h) - tgt)
        return fval, g
    
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(f_and_g, h0, jac=True, method='L-BFGS-B',
                     bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16})
        return proj(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        return proj(h0, n)

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

    best_c = compute_c5(hb)

            break

        b = zeros[rng.randint(len(zeros))]

        hb[b] = 1.0
        new_c = compute_c5(hb)
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
        else:
            hb[a] = 1.0
            hb[b] = 0.0
    return hb, best_c
