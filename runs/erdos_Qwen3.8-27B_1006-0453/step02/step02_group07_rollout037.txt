import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    nk = np.sum(m)
    if nk > 0:
        h[m] += d / nk
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def exact_coord_pass(h, n, order=None):
    c = np.correlate(h, 1.0 - h, mode='full')
    n1 = n - 1
    t_grid = np.linspace(0.0, 1.0, 81)
    
    if order is None:
        order = range(n)
    
    for j in order:
        k_lo = max(-n1, j - n1, -j)
        k_hi = min(n1, j, n1 - j)
        if k_lo > k_hi:
            continue
        ks = np.arange(k_lo, k_hi + 1)
        ms = ks + n1
        j_mk = j - ks
        I1 = (j_mk >= 0) & (j_mk <= n1)
        j_pk = j + ks
        I2 = (j_pk >= 0) & (j_pk <= n1)
        A = np.zeros(len(ks))
        if np.any(I1):
            A[I1] += 1.0 - h[j_mk[I1]]
        if np.any(I2):
            A[I2] -= h[j_pk[I2]]
        B = c[ms] - A * h[j]
        # Vectorized grid eval
        f_vals = (B[:, None] + A[:, None] * t_grid[None, :]).max(axis=0)
        t_best = t_grid[f_vals.argmin()]
        # Quick refinement
        fb = f_vals.min()
        d = 0.01
        for _ in range(8):
            tl = max(0.0, t_best - d)
            tr = min(1.0, t_best + d)
            fl = (B + A * tl).max()
            fr = (B + A * tr).max()
            if fl < fb:
                t_best, fb, d = tl, fl, d * 0.5
            elif fr < fb:
                t_best, fb, d = tr, fr, d * 0.5
            else:
                d *= 0.5
        delta = t_best - h[j]
        if abs(delta) > 1e-13:
            h[j] = t_best
            c[ms] = B + A * t_best
    return h

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def slsqp_refine(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h
