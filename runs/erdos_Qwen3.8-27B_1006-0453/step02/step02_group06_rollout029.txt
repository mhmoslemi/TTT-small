    dx = 2.0 / n

    # For each j:
    #   term1 = sum over m where j + (m-(n-1)) is in [0,n): p[m] * (1-h[j+m-(n-1)])
    #   term2 = sum over m where j - (m-(n-1)) is in [0,n): p[m] * h[j-m+(n-1)]
    
    # Vectorized: for each m, the valid j range for term1 is [max(0, n-1-m), min(n-1, 2n-1-m-1+...)]
    # This is getting messy. Let me just loop over m.
    
    for m in range(2 * n - 1):
        if p[m] < 1e-18:
            continue

        # term1: j from max(0, -k) to min(n-1, n-1-k)
        j_lo = max(0, -k)

        j_lo = max(0, k)
        j_hi = min(n - 1, n - 1 + k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] -= p[m] * h[j_lo - k : j_hi + 1 - k]
    
    return grad * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    c_max = np.max(corr)
    exp_c = np.exp((corr - c_max) / T)
    w_sum = np.sum(exp_c)
    w = exp_c / w_sum
    val = (c_max + T * np.log(w_sum)) * dx
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        j1_lo = max(0, -k)
        j1_hi = min(n - 1, n - 1 - k)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi+1] += wm * (1.0 - h[j1_lo+k:j1_hi+1+k])
        j2_lo = max(0, k)
        j2_hi = min(n - 1, n - 1 + k)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] -= wm * h[j2_lo-k:j2_hi+1-k]
    return val, grad * dx

    v, _ = soft_max_val_and_grad(h, T)
