= Σ_k softmax(β·corr_k)[k] · ∂corr_k/∂h[i]

This gives a weighted sum of all the per-lag gradients, with weights proportional to how close each lag is to the max.

```python
def lse_grad(corr, h, N, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*corr_k)))"""
    shifted = beta * corr - beta * corr.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(N)
    center = N - 1
    k_vals = np.arange(2*N-1) - center
    for k_idx in range(2*N-1):
        if weights[k_idx] < 1e-10:
            continue
        k = int(k_vals[k_idx])
        if k == 0:
            grad += weights[k_idx] * (1.0 - 2.0*h)
        elif k > 0:
            m = N - k
            grad[k:] += weights[k_idx] * (1.0 - h[:m])
            grad[:m] -= weights[k_idx] * h[k:]
        else:
            m = N + k
            grad[:m] += weights[k_idx] * (1.0 - h[-k:])
            grad[m:] -= weights[k_idx] * h[:m]
    return grad
```

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.rfft(h.astype(np.float64), L)

    i_pk = i + k_vals
    A = np.zeros(2*N-1)
    B = np.zeros(2*N-1)
    m1 = (i_mk >= 0) & (i_mk < N)
    m2 = (i_pk >= 0) & (i_pk < N)
    A[m1] = 1.0 - h[np.clip(i_mk[m1], 0, N-1)]
    B[m2] = h[np.clip(i_pk[m2], 0, N-1)]
    return A - B

def full_C(h, N):
    return np.array([row_C(h, N, i) for i in range(N)])

def binary_opt(h_in, N, deadline, smp=200, kick=5, thresh=50, max_it=80000):
    h = (h_in > 0.5).astype(np.float64)
    tgt = N // 2
    if h.sum() != tgt:
        idx = np.argsort(-h)
        h = np.zeros(N)
        h[idx[:tgt]] = 1.0
    C = full_C(h, N)
    corr = fft_corr(h)
    P = float(corr.max())
    best_P, best_h = P, h.copy()
    stuck = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_it):
        if time.time() > deadline:
            break
        P = float(corr.max())
        if P < best_P - 1e-12:
            best_P, best_h, stuck = P, h.copy(), 0
        else:
            stuck += 1
        if stuck >= thresh:
            for _ in range(kick):
                p, q = int(np.random.choice(ones)), int(np.random.choice(zeros))
                h[p], h[q] = 0.0, 1.0
                C[p], C[q] = row_C(h, N, p), row_C(h, N, q)
                ones, zeros = np.where(h==1)[0], np.where(h==0)[0]
            stuck = 0
            continue
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns, nz = min(len(ones), smp), min(len(zeros), smp)
        so, sz = np.random.choice(ones, ns, replace=False), np.random.choice(zeros, nz, replace=False)
        newP = (corr[None,None,:] - C[so][:,None,:] + C[sz][None,:,:]).max(axis=2)
        mi, mj = np.unravel_index(np.argmin(newP), newP.shape)
        if newP[mi,mj] < P - 1e-12:
            p, q = int(so[mi]), int(sz[mj])
            h[p], h[q] = 0.0, 1.0
            C[p], C[q] = row_C(h, N, p), row_C(h, N, q)
            corr = fft_corr(h)
            ones, zeros = np.where(h==1)[0], np.where(h==0)[0]
    return best_h, best_P

def lse_opt(h_in, N, deadline, beta0=25.0, lr0=0.004, kt=50):
    h = project(h_in, N*0.5)
    target = N * 0.5
    m = np.zeros(N)
    beta_m = 0.9
    lr = lr0
    beta = beta0
    best_P = float(fft_corr(h).max())
    best_h = h.copy()
    k_vals = np.arange(2*N-1) - (N-1)
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_P - 1e-15:
            best_P, best_h = P, h.copy()
        top = np.argpartition(corr, -kt)[-kt:]
        bc = beta * corr[top]
        bc -= bc.max()
        w = np.exp(bc)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(kt) / kt
        else:
            w /= ws
        grad = np.zeros(N)
        for ii in range(kt):
            wk = w[ii]
            if wk < 1e-10:
                continue
            k = int(k_vals[top[ii]])
            if k == 0:
                grad += wk * (1.0 - 2.0*h)
            elif k > 0:
                m2 = N - k
                grad[k:] += wk * (1.0 - h[:m2])
                grad[:m2] -= wk * h[k:]
            else:
                m2 = N + k
                grad[:m2] += wk * (1.0 - h[-k:])
                grad[-k:] -= wk * h[:m2]
        h_new = project(h - lr * grad, target)
        m = beta_m * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.992
            beta = max(beta0 * 0.3, beta * 0.997)
            beta_m = min(0.97, 0.90 + 0.002 * (it // 400))
    return best_h, best_P

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=np.float64)
    N = len(h0)
    target = N / 2.0
    best_h = project(h0.copy(), target)
    best_c5 = compute_c5(best_h, N)

    # Phase 1: Binary optimization
    d1 = min(deadline - 550.0, t0 + 350.0)
    hb, Pb = binary_opt(best_h, N, d1, smp=200, kick=5, thresh=50)
    c5b = Pb * (2.0/N)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 2: LSE continuous optimization
    if time.time() < deadline - 380.0:
        d2 = min(deadline - 330.0, time.time() + 400.0)
        hc = project(best_h + np.random.randn(N) * 0.03, target)
        hc2, Pc2 = lse_opt(hc, N, d2, beta0=30.0, lr0=0.003, kt=50)
        c5c = Pc2 * (2.0/N)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc2

    # Phase 3: Binary refinement from continuous
    if time.time() < deadline - 200.0:
        d3 = min(deadline - 150.0, time.time() + 180.0)
        hb3, Pb3 = binary_opt(best_h, N, d3, smp=200, kick=6, thresh=40)
        c5b3 = Pb3 * (2.0/N)
        if c5b3 < best_c5:
            best_c5, best_h = c5b3, hb3

    # Phase 4: LSE polish with sharper beta
    if time.time() < deadline - 110.0:
        d4 = min(deadline - 60.0, time.time() + 160.0)
        hc4 = project(best_h + np.random.randn(N) * 0.015, target)
        hc4c, Pc4 = lse_opt(hc4, N, d4, beta0=60.0, lr0=0.002, kt=80)
        c5c4 = Pc4 * (2.0/N)
        if c5c4 < best_c5:
            best_c5, best_h = c5c4, hc4c

    # Phase 5: Try different N (interpolation + optimize)
    if time.time() < deadline - 50.0:
        for Nt in [N+6, N+12, N-6]:
            if Nt < 100 or time.time() > deadline - 20:
                continue
            x_old = np.linspace(0, 2, N, endpoint=False)
            x_new = np.linspace(0, 2, Nt, endpoint=False)
            hi = np.interp(x_new, x_old, best_h)
            hi = project(hi, Nt/2.0)
            d5 = min(deadline - 10.0, time.time() + 30.0)
            hi2, Pi2 = lse_opt(hi, Nt, d5, beta0=40.0, lr0=0.003, kt=40)
            c5i = Pi2 * (2.0/Nt)
            if c5i < best_c5:
                best_c5, best_h = c5i, hi2

    # Final: check binary version too
    Nf = len(best_h)
    hb = (best_h > 0.5).astype(np.float64)
    if hb.sum() != Nf // 2:
        idx = np.argsort(-best_h)
        hb = np.zeros(Nf)
        hb[idx[:Nf//2]] = 1.0
    c5bin = compute_c5(hb, Nf)
    if c5bin < best_c5:
        best_h, best_c5 = hb, c5bin

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
