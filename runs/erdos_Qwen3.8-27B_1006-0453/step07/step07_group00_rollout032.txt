def gradient_descend(h_in, N, deadline, rng, lr=0.01, n_top=40, max_iters=2000):

    vel = np.zeros(N)
    momentum = 0.85
    
    while time.time() < deadline and it < max_iters:
        it += 1
        y = h + momentum * vel
        y = np.clip(y, 0, 1)
        corr = fft_corr(y)

            best_h = h.copy()
        
        top = np.argpartition(corr, -n_top)[-n_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])

            else:

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    return float(fft_corr(h).max()) * 2.0 / len(h)

def make_binary(h, N, rng):
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        h[rng.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        h[rng.choice(idx, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def do_swap(h, corr, s, deltas, p, q, N):
    h[p], h[q] = 1.0 - h[p], 1.0 - h[q]
    corr += deltas[p] - deltas[q]
    s[p], s[q] = -s[p], -s[q]
    deltas[p, :] = 0.0
    deltas[p, p:p + N] = s
    deltas[q, :] = 0.0
    deltas[q, q:q + N] = s

def greedy_sweep(h, corr, s, deltas, N, max_moves=50):
    for _ in range(max_moves):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_val = float(corr.max())
        best_p, best_q = -1, -1
        nz = len(zeros)

            ce = min(ci + 40, nz)
            dc = deltas[zeros[ci:ce]]

                nms = (corr[None, :] + deltas[p][None, :] - dc).max(axis=1)
                mi = int(nms.argmin())

                    best_p, best_q = int(p), int(zeros[ci + mi])
        if best_p < 0:
            break
        do_swap(h, corr, s, deltas, best_p, best_q, N)
    return best_p >= 0

def grad_polish(h_in, N, deadline, rng, lr=0.008, n_top=50):
    h = h_in.astype(float).copy()
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    vel = np.zeros(N)
    beta = 0.85
    it = 0
    while time.time() < deadline and it < 3000:
        it += 1
        y = np.clip(h + beta * vel, 0.0, 1.0)
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -n_top)[-n_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        vel = beta * vel - lr * grad
        h = np.clip(h + vel, 0.0, 1.0)
        h = np.clip(h - (h.sum() - N / 2.0) / N, 0.0, 1.0)
        if it % 400 == 0:
            lr *= 0.97
            beta = min(0.96, 0.85 + it * 0.0001)
    return best_h, best_pk * 2.0 / N
