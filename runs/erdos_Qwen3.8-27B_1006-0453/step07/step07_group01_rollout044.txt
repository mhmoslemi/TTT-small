    return best_h, best_pk * (2.0 / N)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(99)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, max_sweeps=2000, perturb_every=25, seed=42):
    h = ensure_binary(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(seed)

    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
        Dz = D[zeros, :]

        best_new_P = P
        best_pq = None
        for p in ones:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]), 0, 0)

        # 2-swap: try random pairs of swaps when stuck
        n_2swap = 0
        if stall > 15:
            n_2swap = min(30, 5 + stall)
            for _ in range(n_2swap):
                i1 = rng.randint(len(ones))
                i2 = rng.randint(len(ones))
                j1 = rng.randint(len(zeros))
                j2 = rng.randint(len(zeros))
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                cand2 = c + D[p1, :] + D[p2, :] - Dz[j1, :] - Dz[j2, :]
                npk2 = float(cand2.max())
                if npk2 < best_new_P:
                    best_new_P = npk2
                    best_pq = (p1, q1, p2, q2)

        if best_pq is None:
            if stall > perturb_every:
                n_r = min(3 + stall // 10, 8)
                for _ in range(n_r):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                c = fft_corr(h)
                stall = 0
            elif stall > perturb_every // 2:
                # small kick: flip one pair
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                c = fft_corr(h)
            else:
                break
        else:
            p1, q1, p2, q2 = best_pq
            h[p1] = 0.0
            h[q1] = 1.0
            if p2 != 0:
                h[p2] = 0.0
                h[q2] = 1.0
            c = fft_corr(h)

    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Quick continuous optimization (multiple T values)
    for T_val in [60.0, 150.0]:
        if time.time() > deadline - 550:
            break
        hc = h0.copy() + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        d = min(deadline - 500, time.time() + 80)
        h_n, c5_n = nesterov_opt(hc, N0, d, k=30, T=T_val, lr0=0.015, beta=0.90)
        if c5_n < best_c5:
            best_c5, best_h = c5_n, h_n.copy()

    # Phase 2: Main binary search (longest phase)
    if time.time() < deadline - 350:
        h_bin = ensure_binary(best_h, N0)
        d_b = min(deadline - 300, time.time() + 250)
        h_br, c5_br = binary_search(h_bin, N0, d_b, max_sweeps=3000, perturb_every=30, seed=seed)
        if c5_br < best_c5:
            best_c5, best_h = c5_br, h_br.copy()

    # Phase 3: Restart from perturbed binary
    if time.time() < deadline - 250:
        rng = np.random.RandomState(seed + 7)
        h_p = best_h.copy()
        n_flips = max(4, N0 // 80)
        fl = rng.choice(N0, n_flips, replace=False)
        h_p[fl] = 1.0 - h_p[fl]
        h_p = project_sum(h_p, N0 / 2.0)
        h_bin2 = ensure_binary(h_p, N0)
        d3 = min(deadline - 180, time.time() + 130)
        h_br2, c5_br2 = binary_search(h_bin2, N0, d3, max_sweeps=2000, perturb_every=25, seed=seed + 100)
        if c5_br2 < best_c5:
            best_c5, best_h = c5_br2, h_br2.copy()

    # Phase 4: Continuous refinement + re-binarize
    if time.time() < deadline - 130:
        hc2 = best_h.copy() + np.random.randn(N0) * 0.025
        hc2 = project_sum(hc2, N0 / 2.0)
        d4 = min(deadline - 80, time.time() + 60)
        h4, c5_4 = nesterov_opt(hc2, N0, d4, k=50, T=120.0, lr0=0.008, beta=0.92)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
        if time.time() < deadline - 40:
            h_bin3 = ensure_binary(best_h, N0)
            d4b = min(deadline - 5, time.time() + 60)
            h_br3, c5_br3 = binary_search(h_bin3, N0, d4b, max_sweeps=1500, perturb_every=25, seed=seed + 200)
            if c5_br3 < best_c5:
                best_c5, best_h = c5_br3, h_br3.copy()

    # Phase 5: Try higher resolution if time allows
    N1 = 2 * N0
    if N1 <= 2500 and time.time() < deadline - 80:
        h_up = upsample(best_h, N0, N1)
        h_up = project_sum(h_up, N1 / 2.0)
        h_bin_up = ensure_binary(h_up, N1)
        d5 = min(deadline - 30, time.time() + 60)
        h_br_up, c5_up = binary_search(h_bin_up, N1, d5, max_sweeps=500, perturb_every=35, seed=seed + 300)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_br_up.copy()

    # Phase 6: Final polish at best resolution
    N_fin = len(best_h)
    if time.time() < deadline - 15:
        hF = ensure_binary(best_h, N_fin)
        d6 = max(10, deadline - 3 - time.time())
        hF2, c5F = binary_search(hF, N_fin, deadline - 2, max_sweeps=1000, perturb_every=20, seed=seed + 500)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
