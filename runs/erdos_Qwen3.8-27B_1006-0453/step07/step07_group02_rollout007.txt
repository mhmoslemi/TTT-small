        it += 1
        if it % 80 == 0:
            lr *= 0.994
            beta = min(0.97, 0.85 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0_in = np.array(initial_h_values, dtype=float)
    N0 = len(h0_in)
    best_h = h0_in.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Directed binary search with CORRECTED sign (-Co + Cz)
    d1 = min(deadline - 320, t0 + 220)
    h_bin = (h0_in > 0.5).astype(float)
    fix_sum_bin(h_bin, N0 // 2)
    h_s, _ = directed_search(h_bin, N0, d1, sample_sz=200, kick_thresh=45, kick_sz=5)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov continuous
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 90)
        hc = best_h + np.random.randn(N0) * 0.12
        hc = project_01(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=50, lr0=0.012)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Multi-start binary with larger perturbations
    if time.time() < deadline - 220:
        d3 = min(deadline - 200, time.time() + 200)
        per_try = (d3 - time.time()) / 3
        for trial in range(3):
            if time.time() > deadline - 210:
                break
            h3 = (best_h > 0.5).astype(float)
            ones3 = np.where(h3 == 1)[0]
            zeros3 = np.where(h3 == 0)[0]
            np.random.shuffle(ones3)
            np.random.shuffle(zeros3)
            n_flip = min(5 + trial * 4, len(ones3), len(zeros3))
            h3[ones3[:n_flip]] = 0
            h3[zeros3[:n_flip]] = 1
            sub_dl = min(deadline - 210, time.time() + per_try * 0.8)
            h3s, _ = directed_search(h3, N0, sub_dl, sample_sz=200, kick_thresh=40, kick_sz=5)
            c5_3 = compute_c5(h3s, N0)
            if c5_3 < best_c5:
                best_c5, best_h = c5_3, h3s.copy()

    # Phase 4: Second Nesterov with different params
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 80)
        hc4 = best_h + np.random.randn(N0) * 0.08
        hc4 = project_01(hc4, N0 / 2.0)
        hn4, c5n4 = nesterov_opt(hc4, N0, d4, k=80, lr0=0.008)
        if c5n4 < best_c5:
            best_c5, best_h = c5n4, hn4.copy()

    # Phase 5: Multi-resolution
    for N_try in [400, 420, 440]:
        if time.time() > deadline - 80:
            break
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = (h_int > 0.5).astype(float)
        fix_sum_bin(h_bt, N_try // 2)
        sub_dl = min(deadline - 30, time.time() + 50)
        h_rt, _ = directed_search(h_bt, N_try, sub_dl, sample_sz=N_try // 2, kick_thresh=35, kick_sz=4)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final directed polish on best solution
    if time.time() < deadline - 15:
        hF = (best_h > 0.5).astype(float)
        fix_sum_bin(hF, len(best_h) // 2)
        hFs, _ = directed_search(hF, len(best_h), deadline - 2, sample_sz=250, kick_thresh=60, kick_sz=3)
        c5F = compute_c5(hFs, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hFs.copy()

    # Final check: binary vs continuous
    final_c5_b = compute_c5((best_h > 0.5).astype(float), len(best_h))
    if final_c5_b < best_c5:
        best_h = (best_h > 0.5).astype(float)
        best_c5 = final_c5_b

    return best_h, best_c5, len(best_h)
