def nesterov_opt(h, N, deadline, k_active=80, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.8
    it = 0
    patience = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-9:
            best_pk = pk
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        top = np.argpartition(corr, -k_active)[-k_active:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= float(k_active)
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.997
            beta = min(0.96, 0.8 + it * 0.0003)
        if patience > 500:
            break
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fix_sum(h, ntarget):
    n1 = int(np.round(h.sum()))
    diff = n1 - ntarget
    if diff > 0:
        idx = np.where(h == 1)[0]
        if len(idx) >= diff:
            h[np.random.choice(idx, diff, replace=False)] = 0
    elif diff < 0:
        idx = np.where(h == 0)[0]
        if len(idx) >= -diff:
            h[np.random.choice(idx, -diff, replace=False)] = 1
    return h

def c_row(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    in_m = (i_mk >= 0) & (i_mk < N)
    in_p = (i_pk >= 0) & (i_pk < N)
    A = np.where(in_m, 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where(in_p, h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def binary_ls(h, N, deadline, max_iters=50000, ssz=150, kth=60, ks=3):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    Cm = np.stack([c_row(h, N, i) for i in range(N)])
    stuck = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c, best_h, stuck = P, h.copy(), 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), ssz)
        nz = min(len(zeros), ssz)
        oi = np.random.choice(ones, ns, replace=False)
        zi = np.random.choice(zeros, nz, replace=False)
        newP = (corr[None, None, :] - Cm[oi][None, :, :] + Cm[zi][:, None, :]).max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        if float(newP[mi]) < P - 1e-12:
            h[oi[mi[1]]] = 0
            h[zi[mi[0]]] = 1
            Cm[oi[mi[1]]] = c_row(h, N, oi[mi[1]])
            Cm[zi[mi[0]]] = c_row(h, N, zi[mi[0]])
        elif stuck >= kth:
            for _2 in range(ks):
                p, q = int(np.random.choice(ones)), int(np.random.choice(zeros))
                h[p], h[q] = 0, 1
                Cm[p] = c_row(h, N, p)
                Cm[q] = c_row(h, N, q)
                ones, zeros = np.where(h == 1)[0], np.where(h == 0)[0]
            stuck = 0
        else:
            break
    return best_h, best_c * (2.0 / N)

def nesterov(h, N, deadline, ka=80, lr0=0.04):
    h = h.astype(float).copy()
    tgt = N / 2.0
    best_h, best_pk = h.copy(), float(np.max(fft_corr(h)))
    m, lr, beta = np.zeros(N), lr0, 0.75
    pat = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-9:
            best_pk, best_h, pat = pk, h.copy(), 0
        else:
            pat += 1
        top = np.argpartition(corr, -ka)[-ka:]
        g = np.zeros(N)
        ctr = N - 1
        for ai in top:
            L = int(ai - ctr)
            if L == 0:
                g += 1.0 - 2.0 * y
            elif L > 0:
                g[L:] += 1.0 - y[:N - L]
                g[:N - L] -= y[L:]
            else:
                M = -L
                g[:N - M] += 1.0 - y[M:]
                g[M:] -= y[:N - M]
        g /= float(ka)
        h_new = project_01(y - lr * g, tgt)
        m = beta * (h_new - h)
        h = h_new
        if it % 120 == 0:
            lr *= 0.996
            beta = min(0.95, 0.75 + it * 0.0004)
        if pat > 600:
            break
    return best_h, best_pk * (2.0 / N)

def sturmian(N, alpha, beta):
    i = np.arange(N)
    h = (np.mod(i * alpha + beta, 1.0) < 0.5).astype(float)
    fix_sum(h, N // 2)
    return h
