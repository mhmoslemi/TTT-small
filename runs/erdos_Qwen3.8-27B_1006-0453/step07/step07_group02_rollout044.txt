            if L == 0:
                grad += (1.0 - 2.0 * y) * wi
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L]) * wi
                grad[:N - L] -= y[L:] * wi
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:]) * wi
                    grad[M:] -= y[:N - M] * wi
        grad /= len(active)
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.992
            beta = min(0.96, 0.86 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Aggressive binary search
    d1 = min(deadline - 350, t0 + 250)
    h_bin = (h0 > 0.5).astype(float)
    fix_sum(h_bin, N0 // 2)
    h_s, _ = local_search(h_bin, N0, d1, max_iters=20000, sample_sz=250, kick_thresh=60, kick_sz=5)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Weighted subgradient continuous
    if time.time() < deadline - 280:
        d2 = min(deadline - 260, time.time() + 130)
        hc = best_h + np.random.randn(N0) * 0.07
        hc = project_01(hc, N0 / 2.0)
        hn, c5n = weighted_subgrad(hc, N0, d2, k_active=50, lr0=0.012)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search with annealing
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 170)
        h3 = (best_h > 0.5).astype(float)
        fix_sum(h3, N0 // 2)
        h_s3, _ = local_search(h3, N0, d3, max_iters=12000, sample_sz=250, temp=0.25, kick_thresh=50, kick_sz=6)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Perturbed restart
    if time.time() < deadline - 150:
        h4 = (best_h > 0.5).astype(float)
        ones4 = np.where(h4 == 1)[0]
        zeros4 = np.where(h4 == 0)[0]
        for _ in range(20):
            h4[np.random.choice(ones4)] = 0
            h4[np.random.choice(zeros4)] = 1
            ones4 = np.where(h4 == 1)[0]
            zeros4 = np.where(h4 == 0)[0]
        d4 = min(deadline - 130, time.time() + 100)
        h_s4, _ = local_search(h4, N0, d4, max_iters=8000, sample_sz=250, kick_thresh=40, kick_sz=5)
        c5_4 = compute_c5(h_s4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_s4.copy()

    # Phase 5: Multi-resolution exploration
    for N_try in [360, 420, 480]:
        if time.time() > deadline - 90:
            break
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = (h_int > 0.5).astype(float)
        fix_sum(h_bt, N_try // 2)
        sub_dl = min(deadline - 40, time.time() + 55)
        h_rt, _ = local_search(h_bt, N_try, sub_dl, max_iters=5000, sample_sz=min(N_try, 250), kick_thresh=35, kick_sz=4)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Subgradient polish on best
    if time.time() < deadline - 30:
        hF = best_h + np.random.randn(len(best_h)) * 0.03
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = weighted_subgrad(hF, len(best_h), deadline - 12, k_active=70, lr0=0.007)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final binary cleanup
    if time.time() < deadline - 1:
        hB = (best_h > 0.5).astype(float)
        fix_sum(hB, len(best_h) // 2)
        hB, _ = local_search(hB, len(best_h), deadline - 0.3, max_iters=800, sample_sz=250, kick_thresh=25, kick_sz=3)
        c5B = compute_c5(hB, len(best_h))
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    final_c5 = compute_c5(best_h, len(best_h))
    final_c52 = compute_c5((best_h > 0.5).astype(float), len(best_h))
    if final_c52 < final_c5:
        best_h = (best_h > 0.5).astype(float)
        final_c5 = final_c52
    return best_h, final_c5, len(best_h)
