def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:

        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_jac(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    ws = e / e.sum()
    val = mx + np.log(np.e.sum() if False else np.exp(0)) / T + np.log(ws.sum() * np.exp(T * mx) / np.exp(T * mx) + 1e-30) / T
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if ws[i] > 1e-9:
            g += ws[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, N, deadline, Ts=(45.0, 30.0, 20.0, 14.0)):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    for T in Ts:
        if time.time() > deadline:
            break
        for lam in (1.0, 3.0, 10.0):
            if time.time() > deadline:
                break
            res = sp_min(smooth_jac, h, args=(T, target, lam),
                         jac=True, method='L-BFGS-B', bounds=[(0,1)]*N,
                         options={'maxiter':250, 'maxfun':600, 'ftol':1e-15})
            h = proj01(res.x, target)
            c = compute_c5(h, N)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def bin_full_sweep(h, N, deadline, max_sweeps=40, n_tries=0):
    h = (h > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    ones_idx = np.where(h==1)[0]
    zeros_idx = np.where(h==0)[0]
    if n1 > target:
        h[np.random.choice(ones_idx, n1-target, replace=False)] = 0
    elif n1 < target:
        h[np.random.choice(zeros_idx, target-n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(np.max(fft_corr(h)))
        if P < best_c - 1e-15:
            best_c = P
            best_h = h.copy()
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        bd, bi, bj = 0.0, -1, -1
        if n_tries > 0:
            candidates = [(int(ones[np.random.randint(len(ones))]),
                           int(zeros[np.random.randint(len(zeros))])) for _ in range(n_tries)]
        else:
            candidates = [(int(a), int(b)) for a in ones for b in zeros]
        for oi, zi in candidates:
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            h[oi], h[zi] = 1.0, 0.0
            d = P - nP
            if d > bd:
                bd, bi, bj = d, oi, zi
        if bd > 1e-15:
            h[bi], h[bj] = 0.0, 1.0
        else:
            break
    return best_h, best_c * (2.0 / N)

def bin_2swap(h, N, deadline, max_iters=400, n_trials=12):
    h = (h > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 != target:
        ones_i = np.where(h==1)[0]
        zeros_i = np.where(h==0)[0]
        if n1 > target:
            h[np.random.choice(ones_i, n1-target, replace=False)] = 0
        else:
            h[np.random.choice(zeros_i, target-n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        P = float(np.max(fft_corr(h)))
        if P < best_c - 1e-15:
            best_c = P
            best_h = h.copy()
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        bd, bp = 0.0, None
        for _ in range(n_trials):
            i1, i2 = np.random.choice(len(ones), 2, replace=False)
            j1, j2 = np.random.choice(len(zeros), 2, replace=False)
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            h[p1], h[p2], h[q1], h[q2] = 0, 0, 1, 1
            nP = float(np.max(fft_corr(h)))
            h[p1], h[p2], h[q1], h[q2] = 1, 1, 0, 0
            d = P - nP
            if d > bd:
                bd, bp = d, (p1, p2, q1, q2)
        if bp is not None and bd > 1e-15:
            p1, p2, q1, q2 = bp
            h[p1], h[p2], h[q1], h[q2] = 0, 0, 1, 1
        else:
            break
    return best_h, best_c * (2.0 / N)

def subgrad_ne(h, N, deadline, alpha0=0.03, beta=0.92, atol=1e-5):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < 10000:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= max(len(active), 1)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            alpha *= 0.95
            beta = min(0.98, 0.90 + 0.0001 * it)
    return best_h, best_pk * (2.0 / N)
