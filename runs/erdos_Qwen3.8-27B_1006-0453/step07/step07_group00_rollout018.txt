import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(corr.max()) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    sort_idx = np.argsort(-h)
    h[:] = 0.0
    h[sort_idx[:N//2]] = 1.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

                t_start=0.04, max_iters=100000, use_two_swap=False):

    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 800 == 0:
            T = max(T * cool_rate, 0.001)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Peak-focused candidates
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)

                cand_p.extend(bp[sel])
                cand_q.extend(bq[sel])

        cand_p = np.array(sorted(set(cand_p)), dtype=int)

            break
        
        # Single swap evaluation

                else:

        else:

        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):

def nesterov_polish(h_in, N, deadline, lr=0.012, k_top=40):
    h = h_in.astype(float).copy()
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.82
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            lr *= 0.98
            beta = min(0.95, 0.80 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(corr.max()) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    sort_idx = np.argsort(-h, kind='stable')
    h[:] = 0.0
    h[sort_idx[:N // 2]] = 1.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))

    return s, deltas

            d6a = min(deadline - 8, time.time() + 15)
            h6, _ = exhaustive_hill_climb(h6, d6a)
        d6b = max(deadline - 2, time.time() + 3)
        if time.time() < d6b:
            h6, _ = sa_optimize(h6, d6b, n_peaks=12, n_per_peak=35,
                                cool_rate=0.9999, t_start=0.008, use_two_swap=True,
                                two_swap_prob=0.5)
        c6 = compute_c5(h6, N0)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
