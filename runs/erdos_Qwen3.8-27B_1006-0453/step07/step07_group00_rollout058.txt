import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = np.array(h, dtype=float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        vals = np.argsort(h)[::-1]
        for i in range(n1 - target):
            h[vals[i]] = 0
    elif n1 < target:
        vals = np.argsort(h)
        for i in range(target - n1):
            h[vals[i]] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Fix count
    while h.sum() < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx)] = 1
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx)] = 0
    return h

def build_deltas(s):
    N = len(s)
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return deltas

def recompute_delta(s, i, deltas):
    deltas[i, :] = 0.0
    deltas[i, i:i + N] = s

    N = len(h_init)
    h = h_init.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s = 2.0 * h - 1.0
    deltas = build_deltas(s)
    T = t_start
    no_impr = 0
    n_reheat = 0
    it = 0
    
    while time.time() < deadline and no_impr < max_no_impr:
        it += 1

        ones = np.where(h == 1)[0]

        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Sample candidates biased toward peak lags
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                n_pairs = N - k_off
                if n_pairs <= 0: continue
                vi = np.arange(0, n_pairs)

                if mask.any():
                    bp = vi[mask]
                    ns = min(n_per_peak, len(bp))
                    sel = np.random.choice(len(bp), ns, replace=False)
                    cand_p.extend(bp[sel].tolist())
                    cand_q.extend((bp[sel] + k_off).tolist())
            else:
                M = -k_off

            else:

                grad[M:] -= y[:N - M]
        grad /= len(top)
        
        m = beta * m - lr * grad
        h_new = np.clip(h + m, 0.0, 1.0)
        # Project to sum constraint

        h_new = np.clip(h_new - adj, 0.0, 1.0)
        
        if np.sum(np.abs(h_new - h)) < 1e-7:
            lr *= 0.95
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.97
            beta = min(0.95, 0.8 + 0.0003 * it)
    
    return best_h, best_pk * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):

        h[order[:target-n1]] = 1
    h[h > 0.5] = 1.0

    diff = target - int(h.sum())
    if diff > 0:
        idx = np.where(h == 0)[0]

def build_deltas(s):

    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i+N] = s
    return deltas

def continuous_opt(h_init, deadline, lr_init=0.02, k_top=25):
    N = len(h_init)
    h = h_init.astype(float).copy()
    m = np.zeros(N)
    beta = 0.8
    lr = lr_init
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        
        m = beta * m - lr * grad
        step = h + m
        h_new = np.clip(step, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        
        if np.sum(np.abs(h_new - h)) < 1e-8:
            lr *= 0.9
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.98
            beta = min(0.96, 0.78 + 0.0004 * it)
    
    return best_h, best_pk * (2.0 / N)

        h[np.random.choice(idx, diff, replace=False)] = 1
    elif diff < 0:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, -diff, replace=False)] = 0
    return h

def build_deltas(s):
    N = len(s)
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return deltas

def sa_binary(h_init, deadline, swing_mask=None, n_peaks=6, n_per_peak=18,
              t_start=0.045, cool_rate=0.9994, max_no_impr=5000, directed_frac=0.35):
    N = len(h_init)
    h = h_init.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s = 2.0 * h - 1.0
    deltas = build_deltas(s)
    T = t_start
    no_impr = 0
    it = 0
    L_tot = 2 * N - 1
    if swing_mask is None:
        swing_mask = np.ones(N, dtype=bool)
    
    while time.time() < deadline and no_impr < max_no_impr:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.001)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        p = q = -1
        best_new = 1e10
        if np.random.random() < directed_frac:
            worst = int(np.argmax(corr))
            k_off = worst - (N - 1)
            cand_p, cand_q = np.array([], dtype=int), np.array([], dtype=int)
            if k_off > 0 and k_off < N:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                if mask.any():
                    bp = vi[mask]
                    cand_p, cand_q = bp, bp + k_off
            elif k_off < 0 and -k_off < N:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                if mask.any():
                    bp = vi[mask]
                    cand_p, cand_q = bp, bp - M
            if len(cand_p) == 0:
                cand_p = np.random.choice(ones, min(6, len(ones)), replace=False)
                cand_q = np.random.choice(zeros, len(cand_p), replace=False)
            ns = min(n_per_peak, len(cand_p))
            if len(cand_p) > ns:
                sw = swing_mask[cand_p]
                if sw.sum() > 0:
                    sel = np.where(sw)[0]
                else:
                    sel = np.random.choice(len(cand_p), ns, replace=False)
            else:
                sel = np.arange(len(cand_p))
            cand_p = cand_p[sel]
            cand_q = cand_q[sel]
            dp = deltas[cand_p]
            dq = deltas[cand_q]
            nm = (corr[None, :] + dp - dq).max(axis=1)
            idx = int(np.argmin(nm))
            best_new, p, q = float(nm[idx]), int(cand_p[idx]), int(cand_q[idx])
        else:
            peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
            pl_list, ql_list = [], []
            for pk in peak_lags:
                ko = int(pk) - (N - 1)
                if ko > 0 and ko < N:
                    vi = np.arange(0, N - ko)
                    mk = (h[vi] == 1) & (h[vi + ko] == 0)
                    if mk.any():
                        bp = vi[mk]
                        ns2 = min(n_per_peak, len(bp))
                        sel = np.random.choice(len(bp), ns2, replace=False)
                        pl_list.extend(bp[sel].tolist())
                        ql_list.extend((bp[sel] + ko).tolist())
                elif ko < 0 and -ko < N:
                    M2 = -ko
                    vi = np.arange(M2, N)
                    mk = (h[vi] == 1) & (h[vi - M2] == 0)
                    if mk.any():
                        bp = vi[mk]
                        ns2 = min(n_per_peak, len(bp))
                        sel = np.random.choice(len(bp), ns2, replace=False)
                        pl_list.extend(bp[sel].tolist())
                        ql_list.extend((bp[sel] - M2).tolist())
            nr = min(8, len(ones), len(zeros))
            pl_list = pl_list + np.random.choice(ones, nr, replace=False).tolist()
            ql_list = ql_list + np.random.choice(zeros, nr, replace=False).tolist()
            seen = set()
            fp, fq = [], []
            for a, b in zip(pl_list, ql_list):
                if (a, b) not in seen:
                    seen.add((a, b))
                    fp.append(a)
                    fq.append(b)
            cand_pl = np.array(fp, dtype=int)
            cand_ql = np.array(fq, dtype=int)
            if len(cand_pl) == 0:
                cand_pl = np.random.choice(ones, 4, replace=False)
                cand_ql = np.random.choice(zeros, 4, replace=False)
            dp = deltas[cand_pl]
            dq = deltas[cand_ql]
            chunk = 6
            bi, bj = 0, 0
            for ci in range(0, len(cand_pl), chunk):
                ce = min(ci + chunk, len(cand_pl))
                vals = corr[None, None, :] + dp[ci:ce, None, :] - dq[None, :, :]
                nm = vals.max(axis=2)
                mi = np.unravel_index(int(np.argmin(nm)), nm.shape)
                if nm[mi] < best_new:
                    best_new = float(nm[mi])
                    bi, bj = mi[0] + ci, mi[1]
            p, q = int(cand_pl[bi]), int(cand_ql[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-12))):
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p + N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q + N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
    return best_h, best_c

def continuous_opt(h_init, deadline, lr_init=0.018, k_top=30):
    N = len(h_init)
    h = h_init.astype(float).copy()
    m = np.zeros(N)
    beta = 0.82
    lr = lr_init
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        m = beta * m - lr * grad
        h_new = np.clip(h + m, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        if np.sum(np.abs(h_new - h)) < 1e-8:
            lr *= 0.92
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.975
            beta = min(0.95, 0.80 + 0.0003 * it)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: SA warm
    d1 = min(deadline - 380, t0 + 260)
    h1, _ = sa_binary(h_bin, d1, t_start=0.05, cool_rate=0.9993, n_peaks=6, n_per_peak=20)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()
    # Phase 2: Continuous polish to get swing info
    if time.time() < deadline - 220:
        d2 = min(deadline - 170, time.time() + 120)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, c2 = continuous_opt(hc, d2, lr_init=0.015, k_top=35)
        if c2 < best_c5:
            best_c5, best_h = c2, hn.copy()
    # Phase 3: SA guided by swing positions from continuous
    if time.time() < deadline - 130:
        h3 = make_binary(best_h, N0)
        hc_for_swing = best_h if (best_h.min() < 0.99 or best_h.max() > 0.01) else h3.astype(float)
        swing = np.abs(hc_for_swing - 0.5) < 0.2
        swing_count = swing.sum()
        if swing_count < 30:
            top_idx = np.argsort(np.abs(hc_for_swing - 0.5))[:60]
            swing = np.zeros(N0, dtype=bool)
            swing[top_idx] = True
        d3 = min(deadline - 70, time.time() + 220)
        h3, _ = sa_binary(h3, d3, swing_mask=swing, t_start=0.03, cool_rate=0.9996,
                          directed_frac=0.4, n_peaks=7, n_per_peak=22)
        c3 = compute_c5(h3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
    # Phase 4: Multi-start perturbation
    if time.time() < deadline - 100:
        for pert_n in [3, 5]:
            if time.time() > deadline - 60:
                break
            Ny = len(best_h)
            hb = make_binary(best_h, Ny)
            flip_idx = np.random.choice(Ny, pert_n, replace=False)
            for fi in flip_idx:
                hb[fi] = 1.0 - hb[fi]
            # fix count
            diff = Ny // 2 - int(hb.sum())
            if diff > 0:
                z = np.where(hb == 0)[0]
                hb[np.random.choice(z, diff, replace=False)] = 1
            elif diff < 0:
                o = np.where(hb == 1)[0]
                hb[np.random.choice(o, -diff, replace=False)] = 0
            sub_dl = min(deadline - 40, time.time() + 40)
            hr, _ = sa_binary(hb, sub_dl, t_start=0.02, cool_rate=0.9995, directed_frac=0.35)
            cr = compute_c5(hr, Ny)
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()
    # Phase 5: Higher resolution
    for N_try in [440, 520]:
        if time.time() > deadline - 60:
            break
        if N_try % 2 != 0:
            N_try += 1
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, N_try)
        sub_dl = min(deadline - 20, time.time() + 45)
        hr, _ = sa_binary(hb, sub_dl, t_start=0.03, cool_rate=0.9994, directed_frac=0.3)
        cr = compute_c5(hr, N_try)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()
    # Phase 6: Final fine SA
    if time.time() < deadline - 2:
        Ny = len(best_h)
        h5 = make_binary(best_h, Ny)
        h5, _ = sa_binary(h5, deadline - 1, t_start=0.012, cool_rate=0.9998,
                          n_peaks=10, n_per_peak=30, directed_frac=0.5)
        c5 = compute_c5(h5, Ny)
        if c5 < best_c5:
            best_c5, best_h = c5, h5
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
