def subgradient(h, corr, thresh_frac=0.95):
    """Compute subgradient using all lags above threshold fraction of max."""
    N = len(h)

    max_val = np.max(corr)

def de_step(pop, fits, F=0.6, CR=0.8, target_sum=None):

    N_pop = len(pop)
    N = len(pop[0])
    best_idx = np.argmin(fits)
    new_pop = pop.copy()
    new_fits = fits.copy()
    for i in range(N_pop):
        # Select 3 distinct random indices
        idxs = np.random.choice(N_pop, 4, replace=False)
        a, b, c = idxs[0], idxs[1], idxs[2]
        mutant = pop[a] + F * (pop[b] - pop[c])
        rint = np.random.randint(0, N)
        crossover_mask = np.random.random(N) < CR
        crossover_mask[rint] = True
        trial = np.where(crossover_mask, mutant, pop[i])
        trial = project(trial, target_sum)
        ft = compute_c5(trial)
        if ft < fits[i]:
            new_pop[i] = trial
            new_fits[i] = ft
    # Keep best
    if fits[best_idx] < new_fits[best_idx]:
        new_pop[best_idx] = pop[best_idx]
        new_fits[best_idx] = fits[best_idx]
    return new_pop, new_fits

    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def subgradient(h, corr, thresh_frac=0.95):
    N = len(h)
    center = N - 1
    max_val = np.max(corr)
    thresh = thresh_frac * max_val
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([center])
    grad = np.zeros(N)
    for m in active:
        M = int(m) - center
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif abs(M) < N:
            M = abs(M)
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / len(active)

def cont_opt(h, deadline, k_thresh=0.95, lr0=0.02, mom=0.85, max_stale=50):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > max_stale:
            lr0 *= 0.75
            stale = 0
            if lr0 < 1e-5:
                break
        g = subgradient(h, corr, k_thresh)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.7 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_sa(h, deadline, n_crit=30, s_sz=45, T0=0.008, T_cooldown=0.985, max_iters=5000):
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    it = 0
    no_imp = 0
    T = T0
    center = N - 1
    while time.time() < deadline and it < max_iters:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
            T = max(T, T0 * 0.5)
        else:
            no_imp += 1
        T *= T_cooldown
        if no_imp > 1000 and T < T0 * 0.01:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        cur_peak = float(np.max(corr))
        
        # Single swap
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        best_move = None
        best_delta = 0.0
        
        for oi in os_:
            for zi in zs:
                new_peak = 0.0
                for L in top_lags:
                    idx = center + L
                    d = 0.0
                    if 0 <= oi + L < N:
                        d -= h[oi + L]
                    if 0 <= oi - L < N:
                        d -= h[oi - L]
                    if 0 <= zi + L < N:
                        d += (1 - h[zi + L])
                    if 0 <= zi - L < N:
                        d += (1 - h[zi - L])
                    val = corr[idx] + d
                    if val > new_peak:
                        new_peak = val
                delta = new_peak - cur_peak
                if delta < best_delta:
                    best_delta = delta
                    best_move = (oi, zi)
        
        if best_move is not None:
            if best_delta < 0 or (T > 1e-8 and np.random.random() < np.exp(-best_delta / T)):
                h[best_move[0]] = 0
                h[best_move[1]] = 1
        
        # Double swap every few iterations
        elif it % 4 == 0 and len(ones) >= 2 and len(zeros) >= 2:
            o1 = int(np.random.choice(ones))
            o2 = int(np.random.choice(ones))
            if o1 == o2:
                o2 = int(np.random.choice(ones))
            if o1 == o2:
                pass
            else:
                z1 = int(np.random.choice(zeros))
                z2 = int(np.random.choice(zeros))
                if z1 == z2:
                    z2 = int(np.random.choice(zeros))
                if z1 == z2:
                    pass
                else:
                    h[o1] = 0; h[o2] = 0
                    h[z1] = 1; h[z2] = 1
                    corr2 = compute_corr(h)
                    c2 = float(np.max(corr2)) * (2.0 / N)
                    if c2 < c - 1e-12 or (T > 1e-8 and np.random.random() < np.exp(-(c2-c)/T)):
                        if c2 < best_c:
                            best_c = c2
                            best_h = h.copy()
                            no_imp = 0
                    else:
                        h[o1] = 1; h[o2] = 1
                        h[z1] = 0; h[z2] = 0
        
        it += 1
    return best_h, best_c

def de_populate(h_base, N, pop_size, target_sum):
    pop = np.zeros((pop_size, N))
    pop[0] = h_base
    for i in range(1, pop_size):
        noise = np.random.randn(N) * (0.1 + 0.05 * i / pop_size)
        pop[i] = project(h_base + noise, target_sum)
    return pop

def de_opt(h_base, deadline, pop_size=30, max_gens=100):
    N = len(h_base)
    tgt = N / 2.0
    pop = de_populate(h_base, N, pop_size, tgt)
    fits = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fits)
    best_h = pop[best_idx].copy()
    best_c = fits[best_idx]
    gen = 0
    while time.time() < deadline and gen < max_gens:
        F = 0.5 + 0.2 * (gen / max_gens)
        CR = 0.7
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = idxs
            mutant = pop[a] + F * (pop[b] - pop[c])
            mask = np.random.random(N) < CR
            mask[int(np.random.randint(N))] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project(trial, tgt)
            ft = compute_c5(trial)
            if ft < fits[i]:
                pop[i] = trial
                fits[i] = ft
        new_best = np.argmin(fits)
        if fits[new_best] < best_c:
            best_c = fits[new_best]
            best_h = pop[new_best].copy()
        gen += 1
    return best_h, best_c

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def subgradient(h, corr, thresh_frac=0.95):
    N = len(h)
    center = N - 1
    max_val = np.max(corr)
    thresh = thresh_frac * max_val
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([center])
    grad = np.zeros(N)
    for m in active:
        M = int(m) - center
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif abs(M) < N:
            M = abs(M)
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / len(active)

def cont_opt(h, deadline, k_thresh=0.95, lr0=0.02, mom=0.85, max_stale=50):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > max_stale:
            lr0 *= 0.75
            stale = 0
            if lr0 < 1e-5:
                break
        g = subgradient(h, corr, k_thresh)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.7 * v + 0.3 * (h_new - h)

def binary_sa(h, deadline, n_crit=30, s_sz=45, T0=0.008, T_cooldown=0.985, max_iters=5000):

    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    it = 0
    no_imp = 0
    T = T0
    center = N - 1
    while time.time() < deadline and it < max_iters:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
            T = max(T, T0 * 0.5)
        else:
            no_imp += 1
        T *= T_cooldown
        if no_imp > 1200 and T < T0 * 0.005:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        top_idx = np.argsort(corr)[-n_crit:]
        top_lags = (top_idx - center).astype(int)
        top_corr = corr[top_idx]
        cur_peak = float(np.max(corr))
        
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        
        # Vectorized removal/addition effects
        oi_p = os_[:, None] + top_lags[None, :]
        oi_m = os_[:, None] - top_lags[None, :]
        rem = np.zeros((ns, n_crit))
        vm = (oi_p >= 0) & (oi_p < N)
        vm2 = (oi_m >= 0) & (oi_m < N)
        rem[vm] = -h[oi_p[vm]]
        rem[vm2] -= h[oi_m[vm2]]
        
        zi_p = zs[:, None] + top_lags[None, :]
        zi_m = zs[:, None] - top_lags[None, :]
        add = np.zeros((nz, n_crit))
        va = (zi_p >= 0) & (zi_p < N)
        va2 = (zi_m >= 0) & (zi_m < N)
        add[va] = (1.0 - h[zi_p[va]])
        add[va2] += (1.0 - h[zi_m[va2]])
        
        combined = top_corr[None, None, :] + rem[:, None, :] + add[None, :, :]
        new_peaks = combined.max(axis=2)
        
        flat_idx = np.argmin(new_peaks)
        best_i, best_j = np.unravel_index(flat_idx, new_peaks.shape)
        best_delta = new_peaks[best_i, best_j] - cur_peak
        
        if best_delta < 0 or (T > 1e-8 and np.random.random() < np.exp(-best_delta / T)):
            h[os_[best_i]] = 0
            h[zs[best_j]] = 1
        
        if it % 5 == 0 and len(ones) >= 2 and len(zeros) >= 2:
            o1 = int(np.random.choice(ones))
            o2 = int(np.random.choice(ones))
            if o1 != o2:
                z1 = int(np.random.choice(zeros))
                z2 = int(np.random.choice(zeros))
                if z1 != z2:
                    h[o1] = 0; h[o2] = 0
                    h[z1] = 1; h[z2] = 1
                    corr2 = compute_corr(h)
                    c2 = float(np.max(corr2)) * (2.0 / N)
                    if c2 < c - 1e-12 or (T > 1e-8 and np.random.random() < np.exp(-(c2-c)/T)):
                        if c2 < best_c:
                            best_c = c2
                            best_h = h.copy()
                            no_imp = 0
                    else:
                        h[o1] = 1; h[o2] = 1
                        h[z1] = 0; h[z2] = 0
        
        it += 1
    return best_h, best_c

def de_opt(h_base, deadline, pop_size=30, max_gens=80):
    N = len(h_base)
    tgt = N / 2.0
    pop = np.zeros((pop_size, N))
    pop[0] = h_base

                fits[i] = ft
        nb = np.argmin(fits)
        if fits[nb] < best_c:
            best_c = fits[nb]
            best_h = pop[nb].copy()
        gen += 1
    return best_h, best_c

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def project(h, target_sum):

def subgradient(h, corr, thresh_frac=0.95):
    N = len(h)
    center = N - 1
    max_val = np.max(corr)
    thresh = thresh_frac * max_val
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([center])
    grad = np.zeros(N)
    for m in active:
        M = int(m) - center
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif abs(M) < N:
            M = abs(M)
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]

        h_bt, c6 = binary_sa(h_bt, d6, n_crit=25, s_sz=40, T0=0.004, T_cooldown=0.99)
        if c6 < best_c:
            best_c, best_h = c6, h_bt.copy()
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
