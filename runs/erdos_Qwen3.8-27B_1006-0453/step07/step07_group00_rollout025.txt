import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def make_binary(h, N):
    h = np.asarray(h, dtype=float)
    target = N // 2
    idx = np.argsort(-h, kind='stable')
    result = np.zeros(N)
    result[idx[:target]] = 1.0
    return result

def init_state(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    corr = fft_corr(h)
    return s, deltas, corr

def apply_flip(h, s, deltas, corr, p, q):
    N = len(h)
    corr = corr + deltas[p] - deltas[q]
    h[p], h[q] = 0.0, 1.0
    s[p] = -s[p]
    s[q] = -s[q]
    deltas[p, :] = 0.0
    deltas[p, p:p + N] = s
    deltas[q, :] = 0.0
    deltas[q, q:q + N] = s
    return h, s, deltas, corr

def sa_targeted(h_in, deadline, T_start=0.04, cool=0.9992, n_peaks=6, n_per_peak=14, beam=50):
    N = len(h_in)
    h = h_in.copy()
    s, deltas, corr = init_state(h)
    best_c = float(corr.max())
    best_h = h.copy()
    T = T_start
    no_impr = 0
    it = 0
    N1 = N - 1

    while time.time() < deadline and no_impr < 12000:
        it += 1
        if it % 600 == 0:
            T = max(T * cool, 0.002)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        # Get top peak lags, excluding center (lag 0 at index N-1)
        corr_work = corr.copy()
        corr_work[N1] = -1e10
        peak_lags = np.argpartition(corr_work, -n_peaks)[-n_peaks:]
        peak_lags = peak_lags[corr[peak_lags] > 1e-6]

        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - N1
            if 0 < abs(k_off) < N:
                if k_off > 0:
                    vi = np.arange(0, N - k_off)
                    mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                    bp = vi[mask]
                    bq = (vi + k_off)[mask]
                else:
                    M = -k_off
                    vi = np.arange(M, N)
                    mask = (h[vi] == 1) & (h[vi - M] == 0)
                    bp = vi[mask]
                    bq = (vi - M)[mask]
                if len(bp) > 0:
                    ns = min(n_per_peak, len(bp))
                    sel = np.random.choice(len(bp), ns, replace=False)
                    cand_p.extend(bp[sel].tolist())
                    cand_q.extend(bq[sel].tolist())

        nr = min(12, len(ones), len(zeros))
        if nr > 0:
            cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
            cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())

        seen = set()
        moves = []
        for p, q in zip(cand_p, cand_q):
            key = (int(p), int(q))
            if key not in seen:
                seen.add(key)
                moves.append(key)
                if len(moves) >= beam:
                    break

        if not moves:
            no_impr += 1
            continue

        dc_best = 1e10
        best_move = None
        for p, q in moves:
            nm = float((corr + deltas[p] - deltas[q]).max())
            if nm < dc_best:
                dc_best = nm
                best_move = (p, q)

        dc = dc_best - best_c

                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1

        if no_impr == 4000 and T < 0.035:
            T = 0.05

    return best_h, best_c

def greedy_polish(h_in, n_rounds=4):
    h = h_in.copy()
    N = len(h)
    N1 = N - 1
    s, deltas, corr = init_state(h)
    best_c = float(corr.max())

    for rnd in range(n_rounds):
        improved = False
        corr_work = corr.copy()
        corr_work[N1] = -1e10
        peak_lags = np.argpartition(corr_work, -10)[-10:]
        for pk in peak_lags:
            k_off = int(pk) - N1
            if 0 < abs(k_off) < N:
                if k_off > 0:
                    vi = np.arange(0, N - k_off)
                    mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                    bp = vi[mask]
                    bq = (vi + k_off)[mask]
                else:
                    M = -k_off
                    vi = np.arange(M, N)
                    mask = (h[vi] == 1) & (h[vi - M] == 0)
                    bp = vi[mask]
                    bq = (vi - M)[mask]
                for i in range(len(bp)):
                    p, q = int(bp[i]), int(bq[i])
                    nm = float((corr + deltas[p] - deltas[q]).max())
                    if nm < best_c - 1e-12:
                        h, s, deltas, corr = apply_flip(h, s, deltas, corr, p, q)
                        best_c = nm
                        improved = True
        if not improved:
            break
    return h, best_c
