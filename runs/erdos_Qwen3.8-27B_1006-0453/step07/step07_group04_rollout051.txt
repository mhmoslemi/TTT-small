```python
    # For s=0 (m = N-1), both terms merge; set contribution to 0
    is_zero = (sv == 0)
    val = (1.0 - np.where(vm, h[np.clip(im, 0, N-1)], 0.0)) * vm - \
          np.where(vp, h[np.clip(ip, 0, N-1)], 0.0) * vp
    val[is_zero] = 0
    return val
```

```python
def compute_E(h, N):
    """E[i, m] = marginal effect of changing h[i] by +1 on corr[m].
    For binary h, this gives the change when h[i]: 0→1.
    corr[m] = sum_j h[j] * (1-h[j-s]) where s = m-(N-1)
    """
    s = np.arange(2*N-1) - (N-1)  # shape (2N-1,)
    i = np.arange(N)[:, None]      # shape (N, 1)
    sv = s[None, :]                # shape (1, 2N-1)
    
    # Term 1: h[i] appears as h[j] when j=i, s is the lag
    # This contributes (1-h[i-s]) to corr[m] when i-s is valid

    vm = (im >= 0) & (im < N)
    # Only for s != 0 (i.e., im != i)
    term1 = np.where(vm & (sv != 0), 1.0 - h[np.clip(im, 0, N-1)], 0.0)

    return term1 + term2
```
