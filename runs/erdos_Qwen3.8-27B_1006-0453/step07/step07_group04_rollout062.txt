import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_binary(h, N, seed=42):
    target = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 == target:
        return h
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h[idx] = 0
    else:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h[idx] = 1
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def compute_vmat(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    vm = (pm >= 0) & (pm < N)
    pm_c = np.clip(pm, 0, N - 1)
    return (1.0 - h[pm_c]) * vm

def greedy_tabu(h, N, deadline, max_sweeps=5000, sf=1.0, tabu_len=15, kick_every=80, kick_n=4):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    vmat = compute_vmat(h, N)
    stall = 0
    it = 0
    tabu_until = {}
    idx_arr = np.arange(N)
    while it < max_sweeps and time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        if n1 < 2 or n0 < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_n):
                p = ones[np.random.randint(n1)]
                q = zeros[np.random.randint(n0)]
                h[p], h[q] = h[q], h[p]
            vmat = compute_vmat(h, N)
            c = fft_corr(h)
            best_c = min(best_c, float(c.max()))
            if float(c.max()) < best_c + 1e-12:
                best_h = h.copy()
            stall = 0
            it += 1
            continue
        p_list = ones if sf >= 1.0 else np.random.choice(ones, max(20, int(n1 * sf)), replace=False)
        Dz = vmat[zeros, :]
        best_new_P = best_c - 1e-14
        best_pq = None
        for p in p_list:
            key = (int(p),)
            base = c - vmat[p, :]
            all_max = (base[None, :] + Dz).max(axis=1)
            mi = int(np.argmin(all_max))
            if all_max[mi] < best_new_P:
                best_new_P = float(all_max[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            stall += 1
            if stall > 5:
                p = ones[np.random.randint(n1)]
                q = zeros[np.random.randint(n0)]
                h[p], h[q] = h[q], h[p]
                vmat = compute_vmat(h, N)
                c = fft_corr(h)
            it += 1
            continue
        p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        vmat[idx_arr, idx_arr - p + N - 1] += 1.0
        vmat[idx_arr, idx_arr - q + N - 1] -= 1.0
        c = fft_corr(h)
        cur = float(c.max())
        if cur < best_c - 1e-13:
            best_c = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        tabu_until[(p, q)] = it + tabu_len
        it += 1
    return best_h, best_c * (2.0 / N)

def sa_binary(h, N, deadline, max_sweeps=8000, T0=3.0, Tcool=0.993, T_min=0.03, sf=0.8, kick_every=100, kick_n=5):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    vmat = compute_vmat(h, N)
    T = T0
    stall = 0
    it = 0
    idx_arr = np.arange(N)
    while it < max_sweeps and time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        if n1 < 2 or n0 < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_n):
                p = ones[np.random.randint(n1)]
                q = zeros[np.random.randint(n0)]
                h[p], h[q] = h[q], h[p]
            vmat = compute_vmat(h, N)
            c = fft_corr(h)
            cur = float(c.max())
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            stall = 0
            T = T0 * 0.5
            it += 1
            continue
        P = float(c.max())
        p_list = ones if sf >= 1.0 else np.random.choice(ones, max(20, int(n1 * sf)), replace=False)
        Dz = vmat[zeros, :]
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - vmat[p, :]
            all_max = (base[None, :] + Dz).max(axis=1)
            mi = int(np.argmin(all_max))
            if all_max[mi] < best_new_P:
                best_new_P = float(all_max[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:
            p, q = best_pq
            delta = best_new_P - P
            accept = delta < -1e-14
            if not accept and T > T_min:
                accept = np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))
            if accept:
                h[p] = 0.0
                h[q] = 1.0
                vmat[idx_arr, idx_arr - p + N - 1] += 1.0
                vmat[idx_arr, idx_arr - q + N - 1] -= 1.0
                c = fft_corr(h)
                cur = float(c.max())
                if cur < best_c - 1e-13:
                    best_c = cur
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                p = ones[np.random.randint(n1)]
                q = zeros[np.random.randint(n0)]
                h[p], h[q] = h[q], h[p]
                vmat = compute_vmat(h, N)
                c = fft_corr(h)
                stall += 1
        else:
            stall += 1
        T = max(T_min, T * Tcool)
        it += 1
    return best_h, best_c * (2.0 / N)

def subgrad_opt(h_init, N, deadline, k_active=40, T_smooth=50.0, lr0=0.015, shrink=0.9995):
    h = h_init.astype(float).copy()
    h = project_sum(h, N / 2.0)
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    lr = lr0
    momentum = np.zeros(N)
    beta_m = 0.7
    it = 0
    while time.time() < deadline:
        it += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_active)[-k_active:]
        w = np.exp((corr[top] - corr[top].max()) / T_smooth)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for j, ai in enumerate(top):
            L = int(ai - center)
            wi = w[j]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N - L])
                grad[:N - L] -= wi * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - h[M:])
                    grad[M:] -= wi * h[:N - M]
        grad /= np.sum(w)
        momentum = beta_m * momentum - lr * grad
        h_new = project_sum(h + momentum, N / 2.0)
        h = h_new
        if it % 300 == 0:
            lr *= shrink
            beta_m = min(0.95, 0.6 + it * 0.0001)
            T_smooth = max(5.0, T_smooth * 0.95)
    return best_h, best_pk * dx
