def greedy_step(h, N, c, D, sf=1.0):
    """One pass of greedy: find best swap and apply it. Returns (h, new_P, improved)."""
    P = float(c.max())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))

    Dz = D[zeros, :]

        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def nested_ils(h, N, deadline, n_outer=4, perturb_k=6):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        h_p = best_h.copy()
        ones = np.where(h_p == 1)[0]
        zeros = np.where(h_p == 0)[0]
        n_flip = min(perturb_k, len(ones), len(zeros))
        p_idx = np.random.choice(ones, n_flip, replace=False)
        q_idx = np.random.choice(zeros, n_flip, replace=False)
        for i in range(n_flip):
            h_p[p_idx[i]] = 0.0
            h_p[q_idx[i]] = 1.0
        sub_dl = min(deadline - 3, time.time() + 70)
        h_s, c5_s = sa_search(h_p, N, sub_dl, T0=3.5, Tcool=0.994, T_min=0.04, sf=1.0, kick_every=110, kick_size=4)
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_s.copy()
    return best_h, best_c5

def cont_opt(h, N, deadline, k_top=40, lr=0.018, max_it=5000, beta=0.88):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    for it in range(max_it):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        step = lr / max(gnorm, 1e-6)
        h_new = project_sum(y - step * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 600 == 0:
            lr *= 0.92
            beta = min(0.97, 0.82 + it * 0.0001)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 1000, t0 + 30)
    h1, c5_1 = greedy_ls(h_bin, N0, d1, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Main SA at N0
    d2 = min(deadline - 800, t0 + 180)
    h2, c5_2 = sa_search(best_h, N0, d2, T0=5.0, Tcool=0.992, T_min=0.025, sf=1.0, kick_every=90, kick_size=6)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Nested ILS at N0
    d3 = min(deadline - 600, t0 + 380)
    h3, c5_3 = nested_ils(best_h, N0, d3, n_outer=4, perturb_k=5)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Continuous refinement round 1 (larger lr)
    if time.time() < deadline - 450:
        hc = best_h + np.random.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        d4 = min(deadline - 420, t0 + 500)
        h4, c5_4 = cont_opt(hc, N0, d4, k_top=35, lr=0.025, max_it=3000, beta=0.85)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Greedy binarize + continuous round 2
    if time.time() < deadline - 350:
        h5b = ensure_sum(best_h, N0)
        d5 = min(deadline - 330, t0 + 550)
        h5, c5_5 = greedy_ls(h5b, N0, d5, sf=1.0)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
        d5c = min(deadline - 300, t0 + 590)
        hc5 = best_h + np.random.randn(N0) * 0.02
        hc5 = project_sum(hc5, N0 / 2.0)
        h5c, c5_5c = cont_opt(hc5, N0, d5c, k_top=30, lr=0.015, max_it=3000, beta=0.90)
        if c5_5c < best_c5:
            best_c5, best_h = c5_5c, h5c.copy()

    # Phase 6: Second SA with larger kicks
    if time.time() < deadline - 250:
        h6p = best_h.copy()
        ones6 = np.where(h6p == 1)[0]
        zeros6 = np.where(h6p == 0)[0]
        for _ in range(15):
            h6p[ones6[np.random.randint(len(ones6))]] = 0.0
            h6p[zeros6[np.random.randint(len(zeros6))]] = 1.0
        d6 = min(deadline - 200, t0 + 700)
        h6, c5_6 = sa_search(h6p, N0, d6, T0=7.0, Tcool=0.991, T_min=0.02, sf=1.0, kick_every=80, kick_size=8)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Fine continuous polish (small lr, many iterations)
    if time.time() < deadline - 150:
        hc7 = best_h.copy()
        d7 = min(deadline - 120, t0 + 830)
        h7, c5_7 = cont_opt(hc7, N0, d7, k_top=25, lr=0.008, max_it=8000, beta=0.95)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: ILS round 2
    if time.time() < deadline - 100:
        d8 = min(deadline - 70, t0 + 920)
        h8, c5_8 = nested_ils(best_h, N0, d8, n_outer=3, perturb_k=4)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    # Phase 9: Final continuous micro-polish
    if time.time() < deadline - 30:
        hc9 = best_h.copy()
        d9 = min(deadline - 10, t0 + 970)
        h9, c5_9 = cont_opt(hc9, N0, d9, k_top=20, lr=0.004, max_it=10000, beta=0.97)
        if c5_9 < best_c5:
            best_c5, best_h = c5_9, h9.copy()

    # Final greedy polish on binary projection
    Nf = len(best_h)
    if time.time() < deadline - 2:
        hF = ensure_sum(best_h, Nf)
        sf_fin = 1.0 if Nf <= 400 else 0.6
        hF, c5F = greedy_ls(hF, Nf, deadline - 0.5, sf=sf_fin)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, N0 / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
