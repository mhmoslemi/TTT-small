```python
def swap_delta(corr, h, a, b, N):
    """Compute change in corr when h[a]: 1->0 and h[b]: 0->1.
    corr has 2N-1 entries, indexed by lag = idx - (N-1)."""
    ctr = N - 1
    delta = np.zeros(2*N-1)
    for L in range(-N+1, N):
        # h[a+L], h[b+L], h[a-L], h[b-L]
        aL = a + L
        bL = b + L
        aML = a - L
        bML = b - L
        val_aL = h[aL] if 0 <= aL < N else 0.0
        val_bL = h[bL] if 0 <= bL < N else 0.0
        val_aML = h[aML] if 0 <= aML < N else 0.0
        val_bML = h[bML] if 0 <= bML < N else 0.0
        delta[ctr + L] = val_aL - val_bL + val_aML - val_bML
    return delta
```

```python
import time

    L = 2 * N - 1
```

        perm = np.random.permutation(len(idx))[:n1 - nh]
