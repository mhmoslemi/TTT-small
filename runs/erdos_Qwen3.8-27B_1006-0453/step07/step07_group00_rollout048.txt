import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def make_binary(h, N, rng):
    h = h.astype(float).copy()
    target = N // 2
    scores = h.copy()
    n1 = int(np.sum(h >= 0.5))
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    if n1 > target:
        ones = np.where(h == 1)[0]
        # Remove the weakest ones (lowest score)
        order = np.argsort(scores[ones])
        h[ones[order[:n1-target]]] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-scores[zeros])
        h[zeros[order[:target-n1]]] = 1.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i+N] = s
    return s, deltas

    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.8

        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
