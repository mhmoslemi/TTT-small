import numpy as np
import time

def fft_corr(h):
    """Compute linear correlation of h with (1-h), length 2N-1"""
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_to_sum(h, target_sum):
    """Project h to have sum exactly target_sum while staying in [0,1]"""
    h = h.copy()
    diff = target_sum - h.sum()
    # Distribute uniformly
    h += diff / len(h)
    return np.clip(h, 0, 1)

def binary_project(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_binary_sum(h, target):
    h = h.copy()
    n1 = int(np.sum(h == 1))
    if n1 > target:
        ones = np.where(h == 1)[0]
        to_remove = n1 - target
        remove = np.random.choice(ones, to_remove, replace=False)
        h[remove] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        to_add = target - n1
        add = np.random.choice(zeros, to_add, replace=False)
        h[add] = 1
    return h

- If j is in the second factor (i + lag_k = j, i.e., i = j - lag_k): need i valid (0 ≤ j-lag_k < N) and j valid. Contribution: -h[j-lag_k]

So: d(corr[k])/dh[j] = (1-h[j+lag_k]) * 1_{valid} - h[j-lag_k] * 1_{valid}

And the gradient of the max (using softmax weights w[k]):
grad[j] = ∑_k w[k] * ((1-h[j+lag_k]) - h[j-lag_k])

This can be computed as:
grad[j] = ∑_k w[k] - ∑_k w[k]*h[j+lag_k] - ∑_k w[k]*h[j-lag_k]

The first term is 1 (normalized weights).
The second term: ∑_k w[k]*h[j+k-(N-1)] = cross-correlation of w and h.
The third term: ∑_k w[k]*h[j-k+(N-1)] = cross-correlation of w (reversed) and h.

Both can be computed via FFT in O(N log N).

Let me also think about a key insight: for the "full" correlation of length 2N-1, the maximum is typically at lag 0 (k = N-1) or near it. The constraint sum(h) = N/2 means that at lag 0:
corr[N-1] = ∑ h[i]*(1-h[i]) = N/2 - ∑h[i]² 

This is maximized when h[i] = 1/2 for all i (giving N/4), and minimized when h is binary (giving 0). So for binary h, the max is NOT at lag 0.

For continuous h, the trade-off is: making h more binary reduces corr at lag 0 but might increase it at other lags. The optimal balance is what we're searching for.

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def compute_c5(h):
    N = len(h)
    c = fft_corr(h)
    return float(np.max(c)) * (2.0 / N)

def project_sum(h, target):
    h = h.copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not free.any():
            free = (h > 0) | (h < 1)
        if not free.any():
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0, 1)
    return h

def grad_max_corr(h):
    """Compute subgradient of max_k corr[k] using softmax"""
    N = len(h)
    corr = fft_corr(h)
    peak = np.max(corr)
    # Adaptive temperature
    temp = max(0.005, 0.02 / (1 + 0.1 * (peak - corr).std()))
    w = np.exp((corr - peak) / temp)
    w /= w.sum()

        L *= 2

    B = np.real(np.fft.irfft(W * H, L))[N-1:2*N-1]

            corr = fft_corr(h_b)
