        max_k1 = min(p, N - 1)  # for h[p-k] term, need k <= p and p-k <= N-1 (always true)
        # Actually k ranges 0 to N-1, and p-k must be in [0, N-1], so k in [0, p]
        if p >= 0:
            ks = np.arange(0, p + 1)
            passes[p, center + ks] += h[p - ks] - 1.0
        max_k2 = N - 1 - p  # for h[p+k] term, need k <= N-1-p
        if max_k2 >= 0:
            ks = np.arange(0, max_k2 + 1)
            passes[p, center + ks] += h[p + ks]
        # k < 0, M = -k > 0
        # term 1: p <= N-1-M, i.e., M <= N-1-p; h[p+M] - 1, M in [1, N-1-p]
        max_M1 = N - 1 - p
        if max_M1 >= 1:
            Ms = np.arange(1, max_M1 + 1)
            passes[p, center - Ms] += h[p + Ms] - 1.0
        # term 2: M <= p; h[p-M], M in [1, p]
        max_M2 = p
        if max_M2 >= 1:
            Ms = np.arange(1, max_M2 + 1)
            passes[p, center - Ms] += h[p - Ms]
    
    return passes  # delta_remove[p, shift_index]

def sa_optimize(h, N, deadline, max_iters=20000, T0=0.05, T_end=0.001, rng=None):
    """Simulated annealing with 1-swap and 2-swap moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    cur_c = fft_corr(h)
    cur_P = float(cur_c.max())
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = T0
    cooling = (T_end / T0) ** (1.0 / max_iters)
    accept_count = 0
    improve_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cooling
        
        # Recompute delta every few iterations (it changes with h)
        if it % 10 == 0 or it == 0:
            D = compute_delta(h, N)
        
        move_type = rng.random()
        if move_type < 0.6:
            # 1-swap: move 1 from p to q
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            delta = D[p, :] - D[q, :]  # remove p, add q
            new_P = float((cur_c + delta).max())
            d_P = new_P - cur_P
            if d_P < 0 or rng.random() < np.exp(-d_P / max(T, 1e-6)):
                h[p], h[q] = 0.0, 1.0
                cur_c = cur_c + delta
                cur_P = new_P
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if new_P < best_c - 1e-12:
                    best_c = new_P
                    best_h = h.copy()
                    improve_count += 1
                accept_count += 1
        else:
            # 2-swap: move 1 from p1 to q1, 1 from p2 to q2
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            delta = D[p1, :] + D[p2, :] - D[q1, :] - D[q2, :]
            new_P = float((cur_c + delta).max())
            d_P = new_P - cur_P
            if d_P < 0 or rng.random() < np.exp(-d_P / max(T, 1e-6)):
                h[p1], h[p2], h[q1], h[q2] = 0.0, 0.0, 1.0, 1.0
                cur_c = cur_c + delta
                cur_P = new_P
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if new_P < best_c - 1e-12:
                    best_c = new_P
                    best_h = h.copy()
                    improve_count += 1
                accept_count += 1
    
    # Final check
    final_c = float(np.max(fft_corr(h)))
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()
    
    return best_h, best_c * (2.0 / N)

def steepest_descent(h, N, deadline, max_iters=5000, rng=None):
    """Steepest descent with best-improvement 1-swap and guided 2-swap."""
    if rng is None:
        rng = np.random.RandomState(99)
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    cur_c = fft_corr(h)
    cur_P = float(cur_c.max())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        D = compute_delta(h, N)
        
        # Try best 1-swap
        best_1_P = cur_P
        best_1_pq = None
        # For each zero q, the best p to remove:
        # new_P = max(cur_c + D[p,:] - D[q,:])
        # = max over p of (cur_c - D[q,:]) - ... hmm
        # For fixed q: new_P(q, p) = (cur_c - D[q,:])[max] considering p
        # Actually: new_P = max_m (cur_c[m] + D[p,m] - D[q,m])
        # For fixed q: max_p max_m (cur_c[m] - D[q,m] + D[p,m])
        #           = max_m (cur_c[m] - D[q,m] + max_p D[p,m])
        # Wait no, the max over m and max over p don't commute like that.
        # max_p max_m f(p,m) = max_m max_p f(p,m) (this IS true)
        # So for fixed q: best_new_P = max_m (cur_c[m] - D[q,m] + max_p D[p,m])
        # And the best p is argmax_p D[p, argmax_m (...)]
        
        # This is still O(N²) if done naively, but we can do it smarter:
        # D has shape (N, 2N-1), we only need D for ones
        D_ones = D[ones, :]  # shape (N/2, 2N-1)
        
        # For each zero q: best_new_P(q) = max_m (cur_c[m] - D[q,m] + max_p D[p,m])
        max_D = D_ones.max(axis=0)  # shape (2N-1,)
        
        # For each zero:
        for qi in range(len(zeros)):
            q = zeros[qi]
            base = cur_c - D[q, :] + max_D
            new_P = float(base.max())
            if new_P < best_1_P - 1e-14:
                best_1_P = new_P
                m_star = int(np.argmax(base))
                p_star = int(ones[np.argmax(D_ones[:, m_star])])
                best_1_pq = (p_star, int(q))
        
        # Try some 2-swaps
        best_2_P = best_1_P
        best_2_move = None
        n_2_try = min(200, N)
        for _ in range(n_2_try):
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            new_P = float((cur_c + D[p1,:] + D[p2,:] - D[q1,:] - D[q2,:]).max())
            if new_P < best_2_P - 1e-14:
                best_2_P = new_P
                best_2_move = (p1, p2, q1, q2)
        
        if best_2_P < best_1_P - 1e-14:
            p1, p2, q1, q2 = best_2_move
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
            delta = D[p1,:] + D[p2,:] - D[q1,:] - D[q2,:]
        elif best_1_pq is not None:
            p, q = best_1_pq
            h[p] = 0.0
            h[q] = 1.0
            delta = D[p,:] - D[q,:]
        else:
            stall += 1
            if stall > 20:
                return best_h, best_c * (2.0 / N)
            continue
        
        cur_c = cur_c + delta
        cur_P = float(cur_c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if cur_P < best_c - 1e-12:
            best_c = cur_P
            best_h = h.copy()
            stall = 0
    
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
