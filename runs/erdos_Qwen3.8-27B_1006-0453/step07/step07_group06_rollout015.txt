import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)  # Use real FFT for speed
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    N = len(h)

        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)

    return h

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, deadline, T=30.0, max_outer=3):
    N = len(h)
    target = N / 2.0
    h = h.astype(float).copy()
    best_h, best_v = h.copy(), smooth_obj(h, T, target, 0.0)[0]
    lam = 10.0
    for _ in range(max_outer):

        res = sp_min(smooth_obj, h, args=(T, target, lam), jac=True,
                    method='L-BFGS-B', bounds=[(0, 1)] * N,
                    options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-14})

        v = smooth_obj(h, T, target, 0.0)[0]

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1.0
    return h

def exhaustive_2flip(h, max_passes=100, deadline=None):
    N = len(h)
    h = make_binary(h)
    best_c = float(np.max(fft_corr(h)))
    passes = 0
    while passes < max_passes:
        if deadline is not None and time.time() > deadline:
            break
        passes += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_d = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = float(np.max(fft_corr(h)))
                d = best_c - c
                h[i] = 1.0
                h[j] = 0.0
                if d > best_d + 1e-14:
                    best_d = d
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h[best_i], h[best_j] = 0.0, 1.0
        best_c -= best_d
    return h, best_c

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, deadline, T=30.0, max_outer=3):
    N = len(h)
    target = N / 2.0
    h = h.astype(float).copy()
    best_h, best_v = h.copy(), smooth_obj(h, T, target, 0.0)[0]
    lam = 10.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj, h, args=(T, target, lam), jac=True,
                    method='L-BFGS-B', bounds=[(0, 1)] * N,
                    options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj(h, T, target, 0.0)[0]
        if v < best_v:
            best_v, best_h = v, h.copy()
        lam *= 2.0
    return best_h, best_v * (2.0 / N)

def upsample(h, new_N):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, new_N, endpoint=False)
    return np.interp(x_new, x_old, h)

def binary_refine(h, N, deadline, ntri=40, max_iters=400):
    h = make_binary(h)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c, best_h = P, h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        for _ in range(ntri):
            i = np.random.randint(len(ones))
            j = np.random.randint(len(zeros))
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd:
                bd, bi, bj = d, oi, zi
            h[oi], h[zi] = 1.0, 0.0
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            stall = 0
        else:
            stall += 1
    return best_h, best_c

def perturb_4flip(h):
    N = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    ii = np.random.choice(len(ones), 2, replace=False)
    jj = np.random.choice(len(zeros), 2, replace=False)
    h[ii[0]], h[jj[0]] = 0.0, 1.0
    h[ii[1]], h[jj[1]] = 0.0, 1.0
    return h

    t0 = time.time()
    dl = t0 + budget_s - 3.0
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)

    if time.time() < dl - 400:
        d2 = min(dl - 350, time.time() + 180)
        for Nt in [40, 80, 160, 300, 400, 440]:
            if time.time() > d2:
                break

                Nt += 1
            ht = upsample(best_h, Nt)
            ht = make_binary(ht)
            hr, cr = exhaustive_2flip(ht, max_passes=80, deadline=d2)
            c5r = cr * (2.0 / Nt)
            if c5r < best_c5:
                best_c5, best_h = c5r, hr.copy()
                if len(best_h) != N0:
                    best_h = make_binary(upsample(best_h, N0))
                    best_c5 = compute_c5(best_h, N0)

    # Phase 3: More exhaustive 2-flip
    if time.time() < dl - 250:
        d3 = min(dl - 200, time.time() + 160)
        best_h, c3 = exhaustive_2flip(best_h, max_passes=150, deadline=d3)
        if c3 * (2.0 / N0) < best_c5:
            best_c5 = c3 * (2.0 / N0)

            if time.time() > d4:
