import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    idx = np.argsort(h)[::-1]
    h[:] = 0.0
    h[idx[:target]] = 1.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def sa_search(h_bin, deadline, n_peaks=6, n_per_peak=18, cool_rate=0.999, t_start=0.045,
              max_iters=120000, reheat_every=5000, multi_swap_prob=0.15):
    N = len(h_bin)
    h = h_bin.astype(float).copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    T_min = 0.003
    it = 0
    no_impr = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 800 == 0:
            T = max(T * cool_rate, T_min)
        if it % reheat_every == 0 and T < 0.02:
            T = min(T * 1.8, t_start * 0.5)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 3 or len(zeros) < 3:
            break

        # Targeted candidates from top peaks
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                shift = -M
            if k_off >= 0:
                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                bp, bq = vi[mask], (vi + shift)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())

        # Random candidates
        nr = min(10, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())

        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) < 2 or len(cand_q) < 2:
            break

        # Single swap evaluation

            if cand_p[bi2] != cand_p[bi] and cand_q[bj2] != cand_q[bj] and cand_p[bi2] != cand_q[bj2]:
                d2 = deltas[cand_p[bi2]] + deltas[cand_p[bi]] - deltas[cand_q[bj2]] - deltas[cand_q[bj]]
                new_c2 = float((corr + d2).max())

    return best_h, best_c

def cont_optimize(h_in, N, deadline, lr=0.008, k_top=30, beta0=40):
    h = h_in.astype(float).copy()
    mom = np.zeros(N)
    beta_m = 0.82
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    it = 0
    beta = beta0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        c_top = corr[top]
        mx = c_top.max()
        w = np.exp((c_top - mx) * beta)
        w /= w.sum()
        grad = np.zeros(N)
        for wi, t in zip(w, top):
            L = int(t - (N - 1))
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N-L]); grad[:N-L] -= wi * h[L:]
            else:
                M = -L
                grad[:N-M] += wi * (1.0 - h[M:]); grad[M:] -= wi * h[:N-M]
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new -= (h_new.sum() - N * 0.5) / N
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new -= (h_new.sum() - N * 0.5) / N
        h_new = np.clip(h_new, 0.0, 1.0)
        mom = beta_m * mom + (1 - beta_m) * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.975
            beta *= 0.96
            beta = max(beta, 15)
            beta_m = min(0.93, 0.80 + it * 0.00015)
    return best_h, best_pk * (2.0 / N)
