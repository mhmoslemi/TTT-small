import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, n_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h, kind='stable')
    out = np.zeros_like(h)
    out[idx[:n_ones]] = 1.0
    return out

def fix_binary(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def make_sturmian(N, p, q):

    h = np.zeros(N, dtype=int)
    for i in range(N):

    return h.astype(float)

def make_modular(N, a, b, mod):
    i = np.arange(N)
    return ((i * a + b) % mod < mod // 2).astype(float)
