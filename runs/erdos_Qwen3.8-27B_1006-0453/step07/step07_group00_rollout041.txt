import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def make_binary(h, N, rng=None):
    if rng is None:
        rng = np.random
    idx = np.argsort(-np.asarray(h, dtype=float), kind='stable')
    h2 = np.zeros(N)
    h2[idx[:N // 2]] = 1.0
    return h2

def build_d(h, N):
    d = np.zeros((N, 2 * N - 1))
    for i in range(N):
        k = np.arange(2 * N - 1)
        a = i + k - N + 1
        b = i - k + N - 1
        ma = (a >= 0) & (a < N)
        mb = (b >= 0) & (b < N)
        va = np.zeros(2 * N - 1)
        vb = np.zeros(2 * N - 1)
        va[ma] = 1.0 - h[a[ma]]
        vb[mb] = h[b[mb]]
        d[i] = -va + vb
    return d

def update_d(d, N, p, q):
    """Update d after h[p]:1->0, h[q]:0->1"""
    # Vectorized update
    i1 = np.arange(max(0, p - N + 1), min(N, p + 1))
    if len(i1) > 0:
        d[i1, p - i1 + N - 1] -= 1.0
    i2 = np.arange(max(0, p - N + 1), min(N, p + N))
    if len(i2) > 0:

    i3 = np.arange(max(0, q - N + 1), min(N, q + 1))

    i4 = np.arange(max(0, q - N + 1), min(N, q + N))

def sa_optimize(h_bin, deadline, rng, n_peaks=8, n_per_peak=20, t0=0.06, cr=0.9993, stall=12000):

    best_h = h.copy()
    d = build_d(h, N)
    T = t0
    it = 0
    stall_ct = 0

        it += 1

        ones = np.where(h == 1)[0]

        if len(ones) < 2 or len(zeros) < 2:

    return best_h, best_c

def grad_polish(h_in, N, deadline, rng, lr0=0.04, tau=50):
    """Continuous projected gradient with softmax max."""
    h = h_in.astype(float).copy()
    best_c = compute_c5(h, N)
    best_h = h.copy()
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 200 == 0:
            lr *= 0.97
        corr = fft_corr(h)
        # Softmax gradient
        sc = corr * tau
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        # grad of corr[k] w.r.t. h = -d[:, k] (from our derivation)
        # So grad of weighted sum = -d.T @ w
        # But d depends on h! For the gradient, we use current d.
        # Actually, let me just compute the gradient directly.
        # corr[k] = sum_i h[i] * (1-h[i+k-N+1])
        # d(corr[k])/dh[j] = (1-h[j+k-N+1]) - h[j-k+N+1] (with bounds)
        # This equals -d_delta[j][k] where d_delta is our matrix
        # So gradient of sum_k w[k]*corr[k] = sum_k w[k] * (-d_delta[:,k]) = -(d_delta.T @ w)
        # But d_delta depends on h, so this is an approximation (ignoring d²/dh²)
        dmat = build_d(h, N)
        grad = -(dmat.T @ w)

        h_new = np.clip(h_new - adj, 0, 1)

        diff = h_new.sum() - N / 2.0

            else:

            h_new[idx] += diff / len(idx) * -1  # This is rough
            h_new = np.clip(h_new, 0, 1)
        h = h_new
        c = compute_c5(h, N)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

import numpy as np
import time
import math

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def make_binary(h, N, rng=None):
    if rng is None:
        rng = np.random
    idx = np.argsort(-np.asarray(h, dtype=float), kind='stable')
    h2 = np.zeros(N)
    h2[idx[:N // 2]] = 1.0
    return h2

def build_d(h, N):
    d = np.zeros((N, 2 * N - 1))
    k = np.arange(2 * N - 1)
    for i in range(N):
        a = i + k - N + 1
        b = i - k + N - 1
        ma = (a >= 0) & (a < N)
        mb = (b >= 0) & (b < N)
        d[i, ma] = -(1.0 - h[a[ma]])
        d[i, mb] += h[b[mb]]
    return d

def update_d(d, N, p, q):
    i1 = np.arange(0, p + 1)
    d[i1, p - i1 + N - 1] -= 1.0
    i2 = np.arange(N)
    d[i2, i2 - p + N - 1] -= 1.0
    i3 = np.arange(0, q + 1)
    d[i3, q - i3 + N - 1] += 1.0
    i4 = np.arange(N)
    d[i4, i4 - q + N - 1] += 1.0

def grad_polish(h_in, N, deadline, rng, lr0=0.035, tau=80):
    h = h_in.astype(float).copy()
    best_c = compute_c5(h, N)
    best_h = h.copy()
    lr = lr0
    it = 0
    while time.time() < deadline - 2:
        it += 1
        if it % 300 == 0:
            lr *= 0.96
        corr = fft_corr(h)
        sc = corr * tau
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        dmat = build_d(h, N)
        grad = -(dmat.T @ w)
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new -= (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new
        c = float(fft_corr(h).max())
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c * (2.0 / N)
