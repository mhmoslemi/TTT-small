def grad_top(h, corr, k=30):

    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

        else:
            no_imp += 1
        if no_imp > 1200:
            break
        top_idx = np.argsort(corr)[-n_crit:]
        top_L = (top_idx - center).astype(int)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        ns = min(len(ones), nsamp)
        nz = min(len(zeros), nsamp)
        o_s = np.random.choice(ones, ns, replace=False)
        z_s = np.random.choice(zeros, nz, replace=False)
        best_move = None
        best_np = cur_peak + 1.0
        for oi in o_s:
            eff_rm = np.empty(n_crit)
            for j in range(n_crit):
                L = top_L[j]
                d = 0.0
                if 0 <= oi - L < N:
                    d -= (1.0 - h[oi - L])
                if 0 <= oi + L < N:
                    d += h[oi + L]
                eff_rm[j] = d
            for zi in z_s:
                new_pk = -1e9
                for j in range(n_crit):
                    L = top_L[j]
                    da = 0.0
                    if 0 <= zi - L < N:
                        v = 1.0 - h[zi - L]
                        if zi - L == oi:
                            v = 1.0
                        da += v
                    if 0 <= zi + L < N:
                        v = h[zi + L]
                        if zi + L == oi:
                            v = 0.0
                        da -= v
                    nv = corr[top_idx[j]] + eff_rm[j] + da
                    if nv > new_pk:
                        new_pk = nv
                if new_pk < best_np:
                    best_np = new_pk
                    best_move = (int(oi), int(zi))
        if best_move is not None and best_np < cur_peak:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
        elif T > 0.2 and np.random.random() < 0.35:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        T *= T_decay
        it += 1
    return best_h, best_c

def block_swap_opt(h, deadline, n_crit=25, max_iter=20000):
    N = len(h)
    center = N - 1
    h = h.copy()
    best_h = h.copy()
    best_c = float(np.max(compute_corr(h))) * (2.0 / N)
    it = 0
    no_imp = 0
    while time.time() < deadline and it < max_iter:
        corr = compute_corr(h)
        cur_peak = float(np.max(corr))
        cur_c = cur_peak * (2.0 / N)
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 800:
            break
        top_idx = np.argsort(corr)[-n_crit:]
        top_L = (top_idx - center).astype(int)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Try block moves: swap 2 adjacent ones with 2 adjacent zeros
        found = False
        for _ in range(60):
            oi1 = int(np.random.choice(ones))
            if oi1 + 1 < N and h[oi1 + 1] == 1:
                oi2 = oi1 + 1
            elif oi1 - 1 >= 0 and h[oi1 - 1] == 1:
                oi2 = oi1 - 1
            else:
                oi2 = int(np.random.choice(ones))
            if oi2 == oi1:
                continue
            zi1 = int(np.random.choice(zeros))
            if zi1 + 1 < N and h[zi1 + 1] == 0:
                zi2 = zi1 + 1
            elif zi1 - 1 >= 0 and h[zi1 - 1] == 0:
                zi2 = zi1 - 1
            else:
                zi2 = int(np.random.choice(zeros))
            if zi2 == zi1 or zi2 in (oi1, oi2) or zi1 in (oi1, oi2):
                continue
            new_pk = 0.0
            ok = True
            for j in range(n_crit):
                L = top_L[j]
                nv = corr[top_idx[j]]
                for p in (oi1, oi2):
                    if 0 <= p - L < N:
                        nv -= (1.0 - h[p - L])
                    if 0 <= p + L < N:
                        nv += h[p + L]
                for p in (zi1, zi2):
                    if 0 <= p - L < N:
                        v = 1.0 - h[p - L]
                        if p - L in (oi1, oi2):
                            v = 1.0
                        nv += v
                    if 0 <= p + L < N:
                        v = h[p + L]
                        if p + L in (oi1, oi2):
                            v = 0.0
                        nv -= v
                if nv > new_pk:
                    new_pk = nv
            if new_pk < cur_peak:
                h[oi1] = 0
                h[oi2] = 0
                h[zi1] = 1
                h[zi2] = 1
                found = True
                break
        if not found and np.random.random() < 0.3:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Continuous optimization from initial (150s)
    d1 = min(deadline - 800, t0 + 150)
    h1 = project(best_h + np.random.randn(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=35, lr0=0.015, mom=0.90)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Binary optimization at N=380 (250s)
    if time.time() < deadline - 550:
        hb = fix_binary(best_h, N0 // 2)
        d2 = min(deadline - 500, time.time() + 250)
        hb, c2 = binary_opt(hb, d2, n_crit=35, nsamp=45, max_iter=100000)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    # Phase 3: Block swap at N=380 (150s)
    if time.time() < deadline - 350:
        hb3 = fix_binary(best_h, N0 // 2)
        d3 = min(deadline - 300, time.time() + 150)
        hb3, c3 = block_swap_opt(hb3, d3, n_crit=25, max_iter=30000)
        if c3 < best_c:
            best_c, best_h = c3, hb3.copy()

    # Phase 4: Try N=400
    N_alt = 400
    if time.time() < deadline - 250:
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, best_h)
        h_alt = fix_binary(h_alt, N_alt // 2)
        d4 = min(deadline - 230, time.time() + 180)
        h_alt, c4 = binary_opt(h_alt, d4, n_crit=30, nsamp=45, max_iter=60000)
        if c4 < best_c:
            best_c, best_h = c4, h_alt.copy()
        # Continue with block swaps on N=400 result
        if time.time() < deadline - 130:
            hb4 = fix_binary(h_alt, N_alt // 2)
            d4b = min(deadline - 120, time.time() + 100)
            hb4, c4b = block_swap_opt(hb4, d4b, n_crit=25, max_iter=20000)
            if c4b < best_c:
                best_c, best_h = c4b, hb4.copy()

    # Phase 5: Continuous refinement on best (100s)
    if time.time() < deadline - 80:
        Nref = len(best_h)
        h5 = project(best_h + np.random.randn(Nref) * 0.03, Nref / 2.0)
        d5 = min(deadline - 60, time.time() + 100)
        h5, c5 = cont_opt(h5, d5, k=30, lr0=0.01, mom=0.92)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Final binary polish (60s)
    if time.time() < deadline - 10:
        Nf = len(best_h)
        hB = fix_binary(best_h, Nf // 2)
        d6 = min(deadline - 3, time.time() + 60)
        hB, c6 = binary_opt(hB, d6, n_crit=40, nsamp=50, max_iter=50000)
        if c6 < best_c:
            best_c, best_h = c6, hB.copy()

    # Final check: compare binary vs continuous
    Nf = len(best_h)
    hB = fix_binary(best_h, Nf // 2)
    cB = compute_c5(hB)
    cC = compute_c5(best_h)
    if cB < cC:
        best_h, best_c = hB, cB
    else:
        best_c = cC
    return best_h, best_c, len(best_h)
