def binary_optimize(h, N, deadline, rng, max_sweeps=2000, two_swap_prob=0.3, three_swap_prob=0.05):
    h = h.astype(float).copy()
    if int(round(h.sum())) != N // 2:
        h = ensure_sum_binary(h, N, rng)
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_c = P_cur
    best_h = h.copy()
    
    # Precompute mask matrices
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Compute D
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm + hp * vp
        
        # Best single swap
        D_z = D[zeros, :]
        best_new_P = P_cur
        best_pq = None
        for p in ones:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        # 2-swaps
        best_2p = None
        if two_swap_prob > 0 and rng.random() < two_swap_prob:
            n_try = min(300, max(80, N // 2))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = (p1, p2, q1, q2)
        
        # 3-swaps (less frequent)
        best_3p = None
        if three_swap_prob > 0 and rng.random() < three_swap_prob and len(ones) >= 6 and len(zeros) >= 6:
            n_try = min(50, max(20, N // 6))
            for _ in range(n_try):
                i1, i2, i3 = rng.randint(0, len(ones), 3)
                j1, j2, j3 = rng.randint(0, len(zeros), 3)
                if len(set([i1,i2,i3])) < 3 or len(set([j1,j2,j3])) < 3:
                    continue
                p1, p2, p3 = int(ones[i1]), int(ones[i2]), int(ones[i3])
                q1, q2, q3 = int(zeros[j1]), int(zeros[j2]), int(zeros[j3])
                new_P = float((c_cur + D[p1] + D[p2] + D[p3] - D[q1] - D[q2] - D[q3]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = None
                    best_3p = (p1, p2, p3, q1, q2, q3)
        
        if best_3p is not None:
            p1,p2,p3,q1,q2,q3 = best_3p
            h[p1]=h[p2]=h[p3]=0.0
            h[q1]=h[q2]=h[q3]=1.0
        elif best_2p is not None:
            p1,p2,q1,q2 = best_2p
            h[p1]=h[p2]=0.0
            h[q1]=h[q2]=1.0
        elif best_pq is not None:
            h[best_pq[0]]=0.0
            h[best_pq[1]]=1.0
        else:
            stall += 1
            if stall > 40:
                n_rand = min(4, len(ones)//2, len(zeros)//2)
                ones_l = list(ones)
                zeros_l = list(zeros)
                for _ in range(n_rand):
                    pi = rng.randint(len(ones_l))
                    qi = rng.randint(len(zeros_l))
                    p, q = ones_l[pi], zeros_l[qi]
                    ones_l[pi] = zeros_l[qi]
                    zeros_l[qi] = p
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 15:
                break
            else:
                continue
        
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-14:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
    
    return best_h, best_c * (2.0 / N)

def nesterov_continuous(h, N, deadline, k=50, lr0=0.005, smooth_T=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    center = N - 1
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        max_c = corr.max()
        if max_c < best_pk - 1e-13:
            best_pk = max_c
            best_h = h.copy()
        # Softmax-weighted subgradient over top-k
        top = np.argpartition(corr, -k)[-k:]
        wt = np.exp((corr[top] - max_c) / smooth_T)
        wt_sum = wt.sum()
        if wt_sum > 0:
            wt = wt / wt_sum
        grad = np.zeros(N)
        for idx, ai in enumerate(top):
            Ls = int(ai - center)
            g = np.zeros(N)
            if Ls == 0:
                g = (1.0 - 2.0 * y)
            elif Ls > 0:
                g[Ls:] += (1.0 - y[:N - Ls])
                g[:N - Ls] -= y[Ls:]
            else:
                M = -Ls
                if M < N:
                    g[:N - M] += (1.0 - y[M:])
                    g[M:] -= y[:N - M]
            grad += wt[idx] * g
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.992
            beta = min(0.95, 0.85 + it * 0.0001)
            smooth_T = max(0.005, smooth_T * 0.98)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

            for _ in range(n_try3):
                ii = rng.choice(len(ones), 3, replace=False)
                jj = rng.choice(len(zeros), 3, replace=False)
                new_P = float((c_cur + D[ones[ii[0]]] + D[ones[ii[1]]] + D[ones[ii[2]]] - D[zeros[jj[0]]] - D[zeros[jj[1]]] - D[zeros[jj[2]]]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = None
                    best_3p = (int(ones[ii[0]]), int(ones[ii[1]]), int(ones[ii[2]]), int(zeros[jj[0]]), int(zeros[jj[1]]), int(zeros[jj[2]]))
        if best_3p is not None:
            h[best_3p[0]]=h[best_3p[1]]=h[best_3p[2]]=0.0
            h[best_3p[3]]=h[best_3p[4]]=h[best_3p[5]]=1.0
        elif best_2p is not None:
            h[best_2p[0]]=h[best_2p[1]]=0.0
            h[best_2p[2]]=h[best_2p[3]]=1.0
        elif best_pq is not None:
            h[best_pq[0]]=0.0
            h[best_pq[1]]=1.0
        else:
            stall += 1
            if stall > 35:
                nr = min(5, len(ones)//2, len(zeros)//2)
                ol, zl = list(ones), list(zeros)
                for _ in range(nr):
                    pi, qi = rng.randint(len(ol)), rng.randint(len(zl))
                    ol[pi], zl[qi] = zl[qi], ol[pi]
                    h[ol[pi]], h[zl[qi]] = zl[qi], 0.0
                    h[ol[pi]] = 0.0
                    h[zl[qi]] = 1.0
                stall = 0
            elif stall > 12:
                break
            else:
                continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-14:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
    return best_h, best_c * (2.0 / N)

def nesterov_pass(h, N, deadline, k=45, lr0=0.005, Ts=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.87
    center = N - 1
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        max_c = corr.max()
        if max_c < best_pk - 1e-13:
            best_pk = max_c
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        wt = np.exp((corr[top] - max_c) / Ts)
        ws = wt.sum()
        if ws > 0:
            wt = wt / ws
        grad = np.zeros(N)
        for idx in range(k):
            ai = int(top[idx])
            Ls = ai - center
            g = np.zeros(N)
            if Ls == 0:
                g[:] = 1.0 - 2.0 * y
            elif Ls > 0:
                g[Ls:] += (1.0 - y[:N - Ls])
                g[:N - Ls] -= y[Ls:]
            else:
                M = -Ls
                if M < N:
                    g[:N - M] += (1.0 - y[M:])
                    g[M:] -= y[:N - M]
            grad += wt[idx] * g
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.993
            beta = min(0.94, 0.85 + it * 0.00008)
            Ts = max(0.004, Ts * 0.985)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Continuous Nesterov (explore landscape)
    d1 = min(deadline - 550, t0 + budget_s * 0.18)
    hc = best_h + rng.randn(N0) * 0.02
    hc = project_sum(hc, N0 / 2.0)
    h1, c5_1 = nesterov_pass(hc, N0, d1, k=40, lr0=0.004, Ts=0.025)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Thorough binary search
    d2 = min(deadline - 350, time.time() + budget_s * 0.30)
    h_bin = ensure_sum_binary(best_h, N0, rng)
    h2, c5_2 = binary_optimize(h_bin, N0, d2, rng, max_sweeps=2000, two_p=0.45, three_p=0.08)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Short continuous refinement
    d3 = min(deadline - 200, time.time() + budget_s * 0.12)
    hc3 = best_h + rng.randn(N0) * 0.012
    hc3 = project_sum(hc3, N0 / 2.0)
    h3, c5_3 = nesterov_pass(hc3, N0, d3, k=35, lr0=0.003, Ts=0.015)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Perturb + binary (escape local min)
    d4 = min(deadline - 120, time.time() + budget_s * 0.15)
    if time.time() < d4:
        hp = best_h.copy()
        op = np.where(hp > 0.5)[0]
        zp = np.where(hp <= 0.5)[0]
        for _ in range(7):
            hp[op[rng.randint(len(op))]] = 0.0
            hp[zp[rng.randint(len(zp))]] = 1.0
        hp = ensure_sum_binary(hp, N0, rng)
        h4, c5_4 = binary_optimize(hp, N0, d4, rng, max_sweeps=600, two_p=0.4, three_p=0.07)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Second perturb + binary (different seed)
    d5 = min(deadline - 60, time.time() + budget_s * 0.10)
    if time.time() < d5:
        rng2 = np.random.RandomState(seed + 77)
        hp2 = best_h.copy()
        op2 = np.where(hp2 > 0.5)[0]
        zp2 = np.where(hp2 <= 0.5)[0]
        for _ in range(10):
            hp2[op2[rng2.randint(len(op2))]] = 0.0
            hp2[zp2[rng2.randint(len(zp2))]] = 1.0
        hp2 = ensure_sum_binary(hp2, N0, rng2)
        h5, c5_5 = binary_optimize(hp2, N0, d5, rng2, max_sweeps=400, two_p=0.35, three_p=0.06)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final binary polish
    hF = ensure_sum_binary(best_h, N0, rng)
    hF2, c5F = binary_optimize(hF, N0, deadline, rng, max_sweeps=600, two_p=0.35, three_p=0.05)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
