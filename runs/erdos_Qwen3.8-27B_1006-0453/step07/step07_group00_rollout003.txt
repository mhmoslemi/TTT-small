        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * m + (1.0 - beta) * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.988
            beta = min(0.96, 0.82 + it * 0.00012)
    return best_h, best_pk * (2.0 / N)

def structured_perturb(h, n_swaps=12):
    N = len(h)
    h = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = int(np.random.choice(ones))
        q = int(np.random.choice(zeros))
        h[p], h[q] = 0, 1
    return h

def block_reversal_perturb(h):
    N = len(h)
    h = h.copy()
    for _ in range(6):
        start = int(np.random.randint(0, N - 4))
        length = int(np.random.randint(3, min(12, N - start)))
        end = start + length
        ones_in = int(h[start:end].sum())
        zeros_in = length - ones_in
        if ones_in > 0 and zeros_in > 0:
            h[start:end] = h[start:end][::-1]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.asarray(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Strong SA from slightly perturbed start
    d1 = min(deadline - 450, t0 + 280)
    h1 = make_binary(h0 + np.random.randn(N0) * 0.08, N0)
    h1, _ = sa_binary(h1, d1, n_peaks=14, n_per_peak=35, cool_rate=0.9994, t_start=0.04, no_impr_limit=15000)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Nesterov continuous polish
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 160)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.025
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, c2 = nesterov_polish(hc, N0, d2, lr=0.011, k_top=80)
        if c2 < best_c5:
            best_c5, best_h = c2, hn.copy()

    # Phase 3: SA from best with lower temp, more peaks
    if time.time() < deadline - 180:
        h3 = make_binary(best_h, N0)
        d3 = min(deadline - 100, time.time() + 200)
        h3, _ = sa_binary(h3, d3, n_peaks=16, n_per_peak=40, cool_rate=0.9997, t_start=0.02, no_impr_limit=18000)
        c3 = compute_c5(h3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Iterated local search with structured perturbations
    if time.time() < deadline - 80:
        h4 = make_binary(best_h, N0)
        for _ in range(8):
            if time.time() > deadline - 60:
                break
            if np.random.random() < 0.5:
                h4 = structured_perturb(h4, n_swaps=10)
            else:
                h4 = block_reversal_perturb(h4)
            sub_dl = min(deadline - 50, time.time() + 70)
            h4, _ = sa_binary(h4, sub_dl, n_peaks=12, n_per_peak=28, cool_rate=0.9995, t_start=0.025, no_impr_limit=10000)
            c4 = compute_c5(h4, N0)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()

    # Phase 5: Try nearby resolutions
    for N_try in [384, 420, 480, 512]:
        if time.time() > deadline - 90:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == N0:
            continue
        xo = np.linspace(0, 2, N0, endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, N_try)
        sub_dl = min(deadline - 40, time.time() + 60)
        hr, _ = sa_binary(hb, sub_dl, n_peaks=12, n_per_peak=25, cool_rate=0.9994, t_start=0.035, no_impr_limit=10000)
        cr = compute_c5(hr, N_try)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 6: Final targeted SA
    if time.time() < deadline - 4:
        Ny = len(best_h)
        h6 = make_binary(best_h, Ny)
        h6, _ = sa_binary(h6, deadline - 1, n_peaks=20, n_per_peak=45, cool_rate=0.9999, t_start=0.012, no_impr_limit=10000)
        c6 = compute_c5(h6, Ny)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
