```python
    corr = fft_corr(h)
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    w = np.exp(top - top.max())
    w_sum = w.sum()

        return np.zeros(N)
    w /= w_sum
    grad = np.zeros(N)

    for i, ai in enumerate(idx):

        wi = w[i]
        if L == 0:

            grad[L:] += -wi * h[:N - L]
        else:
            M = -L
            if M < N:
                grad[M:] += wi * (1.0 - h[:N - M])
                grad[:N - M] += -wi * h[M:]
    return grad, float(top.max()) * (2.0 / N)

def adam_optimize(h0, N, deadline, k=80, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):

    target = N / 2.0
```

```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(np.flip(1.0 - h), L), L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_softmax(h, N, k):
    corr = fft_corr(h)
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    w = np.exp(top - top.max())
    ws = w.sum() + 1e-30
    w /= ws
    grad = np.zeros(N)
    c = N - 1
    for i in range(k):
        L = int(idx[i] - c)
        wi = w[i]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            if L >= N:
                continue
            grad[:N - L] += wi * (1.0 - h[L:])
            grad[L:] += -wi * h[:N - L]
        else:
            M = -L
            if M >= N:
                continue
            grad[M:] += wi * (1.0 - h[:N - M])
            grad[:N - M] += -wi * h[M:]
    return grad

def adam_opt(h0, N, deadline, k=80, lr=0.004, b1=0.9, b2=0.999):
```

```python
def swap_delta(h, p, q, N):
    """Compute change in correlation when h[p]: 1->0, h[q]: 0->1"""
    # This is O(1) per shift, O(N) total
    delta = np.zeros(2 * N - 1)
    c = N - 1
    # For shift L > 0 (index c+L):
    # remove p: -(1-h[p+L])*[p<=N-1-L] + h[p-L]*[p>=L]
    # add q: +(1-h[q+L])*[q<=N-1-L] - h[q-L]*[q>=L]
    # For all L = 1..N-1:
    Ls = np.arange(1, N)
    idx_pos = c + Ls  # indices in corr for positive shifts
    idx_neg = c - Ls  # indices in corr for negative shifts
    
    # Positive shifts: L = 1..N-1
    # remove p contribution:
    mask1 = Ls <= (N - 1 - p)  # p+L < N
    delta[idx_pos[mask1]] += -(1 - h[p + Ls[mask1]])
    mask2 = Ls <= p  # p-L >= 0
    delta[idx_pos[mask2]] += h[p - Ls[mask2]]
    # add q contribution:
    mask3 = Ls <= (N - 1 - q)
    delta[idx_pos[mask3]] += (1 - h[q + Ls[mask3]])
    mask4 = Ls <= q
    delta[idx_pos[mask4]] += -h[q - Ls[mask4]]
    
    # Negative shifts: -L where L = 1..N-1
    # corr[c-L] = sum_{k=L}^{N-1} h[k]*(1-h[k-L])
    # remove p: -(1-h[p-L])*[p>=L] + h[p+L]*[p+L<=N-1]
    mask5 = Ls <= p
    delta[idx_neg[mask5]] += -(1 - h[p - Ls[mask5]])
    mask6 = Ls <= (N - 1 - p)
    delta[idx_neg[mask6]] += h[p + Ls[mask6]]
    # add q:
    mask7 = Ls <= q
    delta[idx_neg[mask7]] += (1 - h[q - Ls[mask7]])
    mask8 = Ls <= (N - 1 - q)
    delta[idx_neg[mask8]] += -h[q + Ls[mask8]]
    
    return delta
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(np.flip(1.0 - h), L), L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_softmax(h, N, k):
    corr = fft_corr(h)
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    w = np.exp(top - top.max())
    ws = w.sum() + 1e-30
    w /= ws
    grad = np.zeros(N)
    c = N - 1
    for i in range(k):
        L = int(idx[i] - c)
        wi = w[i]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            if L >= N:
                continue
            grad[:N - L] += wi * (1.0 - h[L:])
            grad[L:] += -wi * h[:N - L]
        else:
            M = -L
            if M >= N:
                continue
            grad[M:] += wi * (1.0 - h[:N - M])
            grad[:N - M] += -wi * h[M:]
    return grad

def adam_opt(h0, N, deadline, k=80, lr=0.004, b1=0.9, b2=0.999):
    h = project_sum(h0.astype(float).copy(), N / 2.0)
    m = np.zeros(N)
    v = np.zeros(N)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t = 0
    skip = 1
    while time.time() < deadline:
        t += 1
        g = grad_softmax(h, N, k)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + 1e-8), N / 2.0)
        if t % skip == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
        if t % 3000 == 0:
            lr *= 0.95
            skip = max(1, skip // 2)
    c5 = compute_c5(best_h)
    return best_h, c5

def upsample(h, N_new):
    N_old = len(h)
    if N_old >= N_new:
        return project_sum(h[:N_new].copy(), N_new / 2.0)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    hu = np.interp(x_new, x_old, h)
    return project_sum(hu, N_new / 2.0)

def ensure_binary(h, N, seed=42):
    nh = N // 2
    hb = (h > 0.5).astype(float).copy()
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    rng = np.random.RandomState(seed)
    if n1 > nh:
        ones = np.where(hb == 1)[0]
        hb[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(hb == 0)[0]
        hb[rng.choice(zeros, nh - n1, replace=False)] = 1
    return hb

def binary_opt(h0, N, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = ensure_binary(h0, N, seed)
    P = float(np.max(fft_corr(h)))
    best_c5 = P * (2.0 / N)
    best_h = h.copy()
    temp = 0.012
    stall = 0
    max_sweeps = 8000
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.003, temp * 0.9975)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        improved = False
        n_try = 35
        for _ in range(n_try):
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p] = 0.0
            h[q] = 1.0
            P_new = float(np.max(fft_corr(h)))
            if P_new < P - 1e-13:
                P = P_new
                improved = True
                c5_n = P * (2.0 / N)
                if c5_n < best_c5:
                    best_c5 = c5_n
                    best_h = h.copy()
            else:
                delta = P_new - P
                acc = np.exp(-max(delta, 0.0) / max(temp, 1e-8))
                if rng.random() > acc:
                    h[p] = 1.0
                    h[q] = 0.0
        if len(ones) >= 4 and len(zeros) >= 4 and rng.random() < 0.25:
            for _ in range(4):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                h[p1] = h[p2] = 0.0
                h[q1] = h[q2] = 1.0
                P_new = float(np.max(fft_corr(h)))
                if P_new < P - 1e-13:
                    P = P_new
                    improved = True
                    if P * (2.0 / N) < best_c5:
                        best_c5 = P * (2.0 / N)
                        best_h = h.copy()
                else:
                    h[p1] = h[p2] = 1.0
                    h[q1] = h[q2] = 0.0
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 35:
                n_pert = 3 + int(4 * rng.random())
                ones_r = np.where(h == 1)[0]
                zeros_r = np.where(h == 0)[0]
                for _ in range(n_pert):
                    h[ones_r[rng.randint(len(ones_r))]] = 0.0
                    h[zeros_r[rng.randint(len(zeros_r))]] = 1.0
                P = float(np.max(fft_corr(h)))
                stall = 0
                temp = 0.015
    return best_h, best_c5

    t0 = time.time()
```
