def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = np.asarray(h, dtype=float).copy()
    n1 = int(round(h.sum()))
    target = N // 2
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        perm = np.random.permutation(len(idx))[:n1 - target]
        h[idx[perm]] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        perm = np.random.permutation(len(idx))[:target - n1]
        h[idx[perm]] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h
