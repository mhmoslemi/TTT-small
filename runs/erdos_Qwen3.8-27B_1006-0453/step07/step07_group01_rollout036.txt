    curr_c = best_c
    while time.time() < deadline:

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = ones[random]
        q = zeros[random]
        # Evaluate the swap
        new_c = evaluate_swap(h, p, q)
        delta = new_c - curr_c

            h[p], h[q] = 0, 1
            curr_c = new_c

                best_h = h.copy()
        T *= alpha
