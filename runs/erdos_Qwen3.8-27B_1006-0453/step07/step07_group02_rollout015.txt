import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_single(h, idx):
    N = len(h)
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def fix_sum_binary(h, target_sum):
    h = (h > 0.5).astype(float)
    s = int(h.sum())
    if s > target_sum:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:s - target_sum]] = 0
    elif s < target_sum:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target_sum - s]] = 1
    return h

    stall = 0

        if len(ones) < 1 or len(zeros) < 1:
            break
        cur_peak = float(np.max(corr))
        best_val = cur_peak
        best_i = -1
