import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N, rng):
    h = h.astype(float).copy()
    n1 = int(np.sum(h > 0.5))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[rng.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h <= 0.5)[0]
        h[rng.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))

        deltas[i, i:i + N] = s
