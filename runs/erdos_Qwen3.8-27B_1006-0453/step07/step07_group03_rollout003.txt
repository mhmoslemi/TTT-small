Wait, I need to be careful about numpy's correlate convention.

np.correlate(a, v) gives c[k] = sum_n a[n+k] * v[n] for the "valid" part.
For mode="full", k ranges from -(len(v)-1) to len(a)-1.

So for our case: a = h, v = 1-h, both of length N.
c[k] = sum_n h[n+k] * (1-h[n]) for n such that 0 <= n < N and 0 <= n+k < N.

For k >= 0: n ranges from 0 to N-1-k
For k < 0: n ranges from -k to N-1

∂c[k]/∂h[p]:
- h[p] appears in h[n+k] when n+k = p, i.e., n = p-k. Valid if 0 <= p-k < N.
  Contribution: (1-h[p-k])
- h[p] appears in (1-h[n]) when n = p. Valid if 0 <= p < N (always).
  Contribution: -h[p+k] (if 0 <= p+k < N)

So ∂c[k]/∂h[p] = (1-h[p-k])*I(0<=p-k<N) - h[p+k]*I(0<=p+k<N)

For the weighted sum: grad[p] = sum_k w[k] * ∂c[k]/∂h[p]

Let me just compute this directly with the shifts array:

```python
def compute_grad_direct(h, N, w):
    """w is array of length 2N-1, indexed by shift + N - 1"""
    shifts = np.arange(2*N-1) - (N-1)
    p = np.arange(N)
    grad = np.zeros(N)
    for si, s in enumerate(shifts):
        wi = w[si]
        if wi < 1e-15:
            continue
        # (1-h[p-s]) valid when 0 <= p-s < N
        ps = p - s
        valid1 = (ps >= 0) & (ps < N)
        ps_c = np.clip(ps, 0, N-1)
        contrib1 = np.where(valid1, (1-h[ps_c]) * wi, 0)
        # -h[p+s] valid when 0 <= p+s < N
        pp = p + s
        valid2 = (pp >= 0) & (pp < N)
        pp_c = np.clip(pp, 0, N-1)
        contrib2 = np.where(valid2, -h[pp_c] * wi, 0)
        grad += contrib1 + contrib2
    return grad
```

This is O(N * 2N) = O(N²) per gradient computation. For N=380, that's ~290K ops. Fine.

def compute_grad_vec(h, N, w):

    p = np.arange(N)[:, None]  # N x 1

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_binary(h, N, seed=0):
    h = h.astype(float).copy()
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed + 777)
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        order = idx[np.argsort(h[idx][::-1])]
        h_bin[order[:n1 - target]] = 0
    else:
        idx = np.where(h_bin == 0)[0]
        order = idx[np.argsort(-h[idx])[:target - n1]]
        h_bin[order] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    hp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def full_grad(h, N, w):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p = np.arange(N)[:, None]
    s = shifts[None, :]
    ps = p - s
    pp = p + s
    v1 = (ps >= 0) & (ps < N)
    v2 = (pp >= 0) & (pp < N)
    c1 = np.where(v1, (1.0 - h[np.clip(ps, 0, N - 1)]) * w[None, :], 0.0)
    c2 = np.where(v2, (-h[np.clip(pp, 0, N - 1)]) * w[None, :], 0.0)
    return (c1 + c2).sum(axis=1)

def nesterov_full(h, N, deadline, k=0, lr0=0.008):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        c_max = corr.max()
        if k == 0:
            w = np.exp(40.0 * (corr - c_max))
            wss = w.sum()
            if wss < 1e-30:
                w = np.zeros_like(corr)
                w[corr == c_max] = 1.0 / (corr == c_max).sum()
            else:
                w /= wss
        else:
            top = np.argpartition(corr, -k)[-k:]
            w = np.zeros(len(corr))
            cn = corr[top] - corr[top].max()
            wt = np.exp(20.0 * cn)
            w[top] = wt
            w /= w.sum()
        grad = full_grad(y, N, w)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.99
            beta = min(0.95, 0.82 + it * 0.0002)
    return best_h, best_pk * dx

def binary_anneal(h, N, deadline, max_sweeps=500, two_swap=0.3, rng=None, temp0=0.015):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_binary(h, N)
    if rng is None:
        rng = np.random.RandomState(42)
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    best_c = P_cur
    best_h = h.copy()
    temp = temp0
    stall = 0
    tabu_len = 40
    tabu = {}
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp0 * (0.995 ** sweep))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        # 1-swap: find best (p, q)
        best_dp = None
        best_pq = None
        for p in ones:
            base = c_cur - D[p, :]
            npk = base[None, :] - D_z
            kmax = npk.max(axis=1)
            mi = int(np.argmin(kmax))
            if best_dp is None or kmax[mi] < best_dp:
                best_dp = float(kmax[mi])
                best_pq = (int(p), int(zeros[mi]))
        # 2-swap
        best_2p = None
        if two_swap > 0 and len(ones) >= 4 and len(zeros) >= 4 and rng.random() < two_swap:
            n_try = min(80, max(20, N // 5))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
                if best_2p is None or new_P < best_2p[0]:
                    best_2p = (new_P, p1, p2, q1, q2)
        # Choose best move
        accepted = False
        if best_2p is not None and (best_2p[0] < P_cur or rng.random() < np.exp(-(P_cur - best_2p[0]) / max(temp, 1e-6))):
            _, p1, p2, q1, q2 = best_2p
            key1 = (p1, q1)
            key2 = (p2, q2)
            if key1 not in tabu and key2 not in tabu:
                h[p1] = h[p2] = 0.0
                h[q1] = h[q2] = 1.0
                tabu[key1] = sweep
                tabu[key2] = sweep
                accepted = True
        if not accepted and best_pq is not None:
            p, q = best_pq
            key = (p, q)
            if key not in tabu:
                if best_dp < P_cur or rng.random() < np.exp(-(P_cur - best_dp) / max(temp, 1e-6)):
                    h[p] = 0.0
                    h[q] = 1.0
                    tabu[key] = sweep
                    accepted = True
        if not accepted:
            # random move when stuck
            if stall > 15:
                n_rand = 3
                for _ in range(n_rand):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p] = 0.0
                    h[q] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stall += 1
        if accepted:
            c_cur = fft_corr(h)
            P_cur = float(c_cur.max())
            if P_cur < best_c - 1e-14:
                best_c = P_cur
                best_h = h.copy()
                stall = 0
                tabu = {}
            else:
                stall += 1
        # Clean old tabu entries
        if len(tabu) > tabu_len:
            tabu = {k2: v2 for k2, v2 in tabu.items() if sweep - v2 < tabu_len}
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

        d3 = min(deadline - 200, time.time() + budget_s * 0.12)

    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
