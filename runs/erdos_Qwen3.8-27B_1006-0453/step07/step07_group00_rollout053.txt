import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def grad_corr(h, k_idx, N):
    """Gradient of correlation at lag index k_idx w.r.t. h."""
    g = np.zeros(N)
    s = k_idx - (N - 1)
    if s > 0:
        idx = np.arange(N - s)
        g[idx] += (1.0 - h[idx + s])
        g[idx + s] -= h[idx]
    elif s < 0:
        m = -s
        idx = np.arange(m, N)
        g[idx] += (1.0 - h[idx - m])
        g[idx - m] -= h[idx]
    # s == 0: grad is 1 - 2h
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def subgrad_optimize(h, N, deadline, k_top=60, lr0=0.005, decay=0.995):
    """Subgradient descent with Nesterov momentum on the max correlation."""
    h = h.astype(float).copy()
    v = np.zeros(N)  # momentum
    lr = lr0
    best_h = h.copy()
    corr = fft_corr(h)
    best_val = float(corr.max())
    it = 0
    while time.time() < deadline:
        y = h + v
        corr = fft_corr(y)
        top = np.argpartition(corr, -k_top)[-k_top:]
        # Subgradient: weighted average of top gradients
        weights = np.exp(corr[top] / 0.01)  # softmax-ish for emphasizing max
        weights /= weights.sum()
        g = np.zeros(N)
        for w, t in zip(weights, top):
            g += w * grad_corr(y, t, N)
        # Nesterov update
        v = 0.85 * v - lr * g
        h_new = y + v
        # Project to [0, N/2] simplex
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new - (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize
        if h_new.sum() != N / 2.0:
            # Simple repair
            diff = N / 2.0 - h_new.sum()
            if diff > 0:
                order = np.argsort(h_new)
                for i in range(len(order)):
                    h_new[order[i]] = min(1.0, h_new[order[i]] + diff / (len(order) - i))
                    if h_new.sum() >= N / 2.0 - 1e-9:
                        break
            else:
                order = np.argsort(-h_new)
                for i in range(len(order)):
                    h_new[order[i]] = max(0.0, h_new[order[i]] + diff / (len(order) - i))
                    if h_new.sum() <= N / 2.0 + 1e-9:
                        break
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= decay
            pk = float(fft_corr(h).max())
            if pk < best_val:
                best_val = pk
                best_h = h.copy()
        if it % 1000 == 0:
            pk = float(fft_corr(h).max())
            if pk < best_val:
                best_val = pk
                best_h = h.copy()
    pk = float(fft_corr(h).max())
    if pk < best_val:
        best_val = pk
        best_h = h.copy()
    return best_h, best_val * (2.0 / N)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    # Pad to next power of 2 for speed
    Lp = 1 << (L - 1).bit_length() if L > 1 else 2
    H = np.fft.rfft(h, Lp)
    G = np.fft.rfft(np.flip(1.0 - h), Lp)
    return np.real(np.fft.irfft(H * G, Lp))[:2 * N - 1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > target:
        order = np.argsort(-h)
        for i in range(n1 - target):
            h[order[i]] = 0.0
    elif n1 < target:
        order = np.argsort(h)
        for i in range(target - n1):
            h[order[i]] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def delta_at(h, idx, N):
    """Change in full correlation when h[idx] flips (1→0 or 0→1)."""
    # Returns the vector of changes to corr
    s = 2.0 * h[idx] - 1.0  # +1 if h[idx]=1, -1 if h[idx]=0
    d = np.zeros(2 * N - 1)
    d[idx:idx + N] = s
    return d

def smooth_grad_opt(h_in, N, deadline, k_top=50, lr0=0.008):

        corr = fft_corr(y)

        # Softmax weights emphasizing the max

        w /= w.sum()
        g = np.zeros(N)
        for wi, t in zip(w, top):
            s = int(t) - (N - 1)
            if s > 0:

            elif s < 0:

        h_new = np.clip(h_new, 0.0, 1.0)
        h_new -= (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new, 0.0, 1.0)

        it += 1

            pk = float(fft_corr(h).max())
            if pk < best_v:
                best_v = pk
                best_h = h.copy()
            lr *= 0.993
            beta = min(0.92, 0.80 + it * 0.0001)
    pk = float(fft_corr(h).max())
    if pk < best_v:
        best_v = pk
        best_h = h.copy()
    return best_h, best_v * (2.0 / N)
