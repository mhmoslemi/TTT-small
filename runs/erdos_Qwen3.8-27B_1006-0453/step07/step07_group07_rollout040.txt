import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_corr_top(h, corr, k=40):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def cont_opt(h, deadline, k=40, lr0=0.025, mom=0.85):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 80:
            lr0 *= 0.85
            stale = 0
        g = grad_corr_top(h, corr, k)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

            zeros = np.where(h == 0)[0]
            nr = max(3, N // 25)
            si = np.random.choice(len(ones), size=nr, replace=False)
            for s in si:
                h[ones[s]] = 0
                h[int(np.random.choice(zeros))] = 1
            no_imp = 0
            it += 1
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        na, nz = len(ones), len(zeros)
        if na < 1 or nz < 1:
            break
        top_idx = np.argsort(corr)[-n_crit:]
        lags = (top_idx - center).astype(int)
        h_pad = np.zeros(L)
        h_pad[center:center + N] = h
        pa_idx = center + ones[:, None] + lags[None, :]
        ma_idx = center + ones[:, None] - lags[None, :]
        pb_idx = center + zeros[:, None] + lags[None, :]
        mb_idx = center + zeros[:, None] - lags[None, :]
        pa = np.where((pa_idx >= 0) & (pa_idx < L), h_pad[np.clip(pa_idx, 0, L-1)], 0.0)
        ma = np.where((ma_idx >= 0) & (ma_idx < L), h_pad[np.clip(ma_idx, 0, L-1)], 0.0)
        pb = np.where((pb_idx >= 0) & (pb_idx < L), h_pad[np.clip(pb_idx, 0, L-1)], 0.0)
        mb = np.where((mb_idx >= 0) & (mb_idx < L), h_pad[np.clip(mb_idx, 0, L-1)], 0.0)
        part_a = pa - ma
        part_b = pb - mb
        ct = corr[top_idx]
        new_max = np.max(ct[None, None, :] - part_a[:, None, :] + part_b[None, :, :], axis=2)
        min_val = float(np.min(new_max))
        if min_val < raw_max - 1e-10:
            ai, bi = np.unravel_index(np.argmin(new_max), new_max.shape)
            h[ones[ai]] = 0
            h[zeros[bi]] = 1
        elif no_imp > 1500 and np.random.random() < 0.25:
            h[ones[np.random.randint(na)]] = 0
            h[zeros[np.random.randint(nz)]] = 1
        it += 1
    return best_h, best_c

def interp_to(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2.0, N_old, endpoint=False)
    x_new = np.linspace(0, 2.0, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Continuous optimization
    d1 = min(deadline - 850, t0 + 160)
    h1 = project(best_h + np.random.randn(N0) * 0.05, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=30, lr0=0.016, mom=0.91)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: Binary swap LS - main workhorse
    if time.time() < deadline - 300:
        d2 = min(deadline - 270, time.time() + 480)
        hb = fix_binary(best_h)
        hb, c2 = binary_swap_ls(hb, d2, n_crit=45, max_iters=300000)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    # Phase 3: Continuous refinement
    if time.time() < deadline - 170:
        d3 = min(deadline - 150, time.time() + 70)
        h3 = project(best_h + np.random.randn(N0) * 0.03, N0 / 2.0)
        h3, c3 = cont_opt(h3, d3, k=25, lr0=0.01, mom=0.93)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Binary polish
    if time.time() < deadline - 90:
        d4 = min(deadline - 70, time.time() + 70)
        hb4 = fix_binary(best_h)
        hb4, c4 = binary_swap_ls(hb4, d4, n_crit=40, max_iters=100000)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()

    # Phase 5: Resolution changes
    for N_try in [420, 460, 500]:
        if time.time() > deadline - 80:
            break
        h_int = interp_to(best_h, N_try)
        h_bt = fix_binary(h_int)
        d5 = min(deadline - 60, time.time() + 45)
        h_bt, c5v = binary_swap_ls(h_bt, d5, n_crit=35, max_iters=40000)
        if time.time() < deadline - 40:
            hc = project(h_bt + np.random.randn(N_try) * 0.04, N_try / 2.0)
            d6 = min(deadline - 30, time.time() + 25)
            hc, c6 = cont_opt(hc, d6, k=25, lr0=0.012, mom=0.9)
            if c6 < c5v:
                c5v, h_bt = c6, hc.copy()
        if c5v < best_c:
            best_c, best_h = c5v, h_bt.copy()

    # Phase 6: Final binary polish
    if time.time() < deadline - 12:
        hB = fix_binary(best_h)
        d7 = min(deadline - 2, time.time() + 35)
        hB, c7 = binary_swap_ls(hB, d7, n_crit=50, max_iters=60000)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()

    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
