import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_soft(h, corr, k=50, temp=0.003):
    N = len(h)
    center = N - 1
    c_max = np.max(corr)
    sel = np.argsort(corr)[-k:]
    w = np.exp((corr[sel] - c_max) / temp)
    w /= w.sum()
    grad = np.zeros(N)
    for i, m in enumerate(sel):
        wi = w[i]
        M = abs(int(m) - center)
        if M == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif M < N:
            grad[M:] += wi * (1.0 - h[:-M])
            grad[:N-M] -= wi * h[M:]
    return grad

    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:

        if stale > 45:
            lr0 *= 0.87
            stale = 0
            if lr0 < 1e-7:

        g = grad_soft(h, corr, k, temp)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = mom * v + 0.5 * (h_new - h)
        h = h_new
    return best_h, best_c

    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)

        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

    """Delta in corr[center+L] when h[pos] flips 1->0."""
    d = 0.0
    if 0 <= pos + L < N:
        d -= (1.0 - h[pos + L])
    if 0 <= pos - L < N:
        d += h[pos - L]
    return d

def flip_delta1(h, pos, L, center, N):
    """Delta in corr[center+L] when h[pos] flips 0->1."""

    if 0 <= pos + L < N:

    if 0 <= pos - L < N:
        d -= h[pos - L]
    return d
