    best_c = float(r.max())
    best_h = h.copy()

    momentum = np.zeros(N)
    beta = beta0
    lr = lr0

    for it in range(iters):
        if time.time() > deadline:
            break
        beta = beta0 * (beta_f / beta0) ** (it / max(1, iters * 0.7))
        lr = lr0 * (0.9995 ** it)

        ex = beta * (r - r.max())
        ex = np.clip(ex, -50, 50)
        w = np.exp(ex)
        w /= w.sum()

        grad = np.zeros(N)
        for k in range(L):
            if w[k] < 1e-6:
                continue
            shift = k - (N - 1)
            if shift == 0:
                grad += w[k] * (1.0 - 2.0 * h)
            elif shift > 0:
                grad[shift:] += w[k] * (1.0 - h[:N - shift])
                grad[:N - shift] -= w[k] * h[shift:]
            else:
                M = -shift
                grad[:N - M] += w[k] * (1.0 - h[M:])
                grad[M:] -= w[k] * h[:N - M]

        h_new = project_sum(y - lr * grad, N / 2.0)

            best_h = h.copy()

        if not improved:
            break
    return h, float(np.max(fft_corr(h))) * (2.0 / N)

def upsample_h(h_old, N_old, N_new):
    x_old = np.arange(N_old) / N_old
    x_new = np.arange(N_new) / N_new
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary local search at N0 (main effort)
    h_b = ensure_binary(h0, N0, rng)
    d1 = min(deadline - 350, t0 + budget_s * 0.35)
    h1, c1 = local_search_binary(h_b, N0, d1, rng, max_sweeps=8000, three_rate=0.25)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Continuous relaxation at N0 for new basin
    if time.time() < deadline - 280:
        hc = best_h + rng.randn(N0) * 0.15
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 250, time.time() + budget_s * 0.12)
        h2, c2 = continuous_opt(hc, N0, d2, beta0=40, beta_f=300, lr0=0.012, iters=4000)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        # Binarize and polish
        h2b = ensure_binary(h2, N0, rng)
        d2b = min(deadline - 220, time.time() + budget_s * 0.08)
        h2b, c2b = local_search_binary(h2b, N0, d2b, rng, max_sweeps=3000, three_rate=0.3)
        if c2b < best_c5:
            best_c5, best_h = c2b, h2b.copy()

    # Phase 3: Upscale to 2*N0, optimize, come back
    N1 = 2 * N0
    if time.time() < deadline - 200:
        h_up = upsample_h(best_h, len(best_h), N1)
        h_up_b = ensure_binary(h_up, N1, rng)
        d3 = min(deadline - 160, time.time() + budget_s * 0.18)
        h3, c3 = local_search_binary(h_up_b, N1, d3, rng, max_sweeps=1500, three_rate=0.15)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
        # Downsample back to N0 if 2*N0 is better
        h_ds = upsample_h(h3, N1, N0)
        h_ds = ensure_binary(h_ds, N0, rng)
        d3b = min(deadline - 130, time.time() + budget_s * 0.06)
        h3b, c3b = local_search_binary(h_ds, N0, d3b, rng, max_sweeps=2000, three_rate=0.3)
        if c3b < best_c5:
            best_c5, best_h = c3b, h3b.copy()

    # Phase 4: Another continuous + binary cycle with different perturbation
    if time.time() < deadline - 100:
        hc3 = best_h + rng.randn(N0) * 0.08
        hc3 = project_sum(hc3, N0 / 2.0)
        d4 = min(deadline - 70, time.time() + budget_s * 0.07)
        h4, c4 = continuous_opt(hc3, N0, d4, beta0=80, beta_f=400, lr0=0.006, iters=2000)
        h4b = ensure_binary(h4, N0, rng)
        d4b = min(deadline - 40, time.time() + budget_s * 0.05)
        h4b, c4b = local_search_binary(h4b, N0, d4b, rng, max_sweeps=2000, three_rate=0.35)
        if c4b < best_c5:
            best_c5, best_h = c4b, h4b.copy()

    # Phase 5: Final exhaustive 1-swap polish
    Nf = len(best_h)
    hF, cF = exhaustive_1swap_polish(best_h, Nf, rng, max_passes=30)
    if cF < best_c5:
        best_c5, best_h = cF, hF.copy()

    # Final micro-polish: a few more local search rounds
    if time.time() < deadline - 2:
        hF2, cF2 = local_search_binary(best_h, len(best_h), deadline - 1, rng, max_sweeps=500, three_rate=0.3)
        if cF2 < best_c5:
            best_c5, best_h = cF2, hF2.copy()
        hF3, cF3 = exhaustive_1swap_polish(best_h, len(best_h), rng, max_passes=20)
        if cF3 < best_c5:
            best_c5, best_h = cF3, hF3.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
