import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h[::-1], L)  # flip(1-h)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0].copy()
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0].copy()
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_top(h, corr, k=50):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            grad[M:] += (1.0 - h[:-M])
            grad[:-M] -= h[M:]
    return grad / k

def continuous_opt(h, deadline, lr=0.02, k=40):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stall = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            lr *= 0.9
            stall = 0
            if lr < 1e-5:
                break
        g = grad_top(h, corr, k)
        v = 0.85 * v - lr * g
        h = project_sum(h + v, tgt)
    return best_h, best_c

def binary_sa(h, deadline, t_init=0.004, t_min=0.00005):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    cur_peak = float(np.max(corr))
    cur_c = cur_peak * (2.0 / N)
    best_c = cur_c
    best_h = h.copy()
    temp = t_init
    stall = 0
    dx = 2.0 / N
    while time.time() < deadline:
        stall += 1
        if stall > 80:
            temp = max(t_min, temp * 0.85)
            stall = 0
            if temp < t_min * 1.05:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        r = np.random.random()
        if r < 0.5:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0; h[zi] = 1
            new_corr = compute_corr(h)
            new_peak = float(np.max(new_corr))
            new_c = new_peak * dx
            d = new_c - cur_c
            if d < 0 or np.random.random() < np.exp(-d / temp):
                cur_c = new_c; cur_peak = new_peak; corr = new_corr
                if cur_c < best_c:
                    best_c = cur_c; best_h = h.copy()
            else:
                h[oi] = 1; h[zi] = 0
        elif r < 0.8:
            if len(ones) < 2 or len(zeros) < 2:
                continue
            oi1, oi2 = np.random.choice(ones, 2, replace=False)
            zi1, zi2 = np.random.choice(zeros, 2, replace=False)
            h[oi1] = 0; h[oi2] = 0; h[zi1] = 1; h[zi2] = 1
            new_corr = compute_corr(h)
            new_peak = float(np.max(new_corr))
            new_c = new_peak * dx
            d = new_c - cur_c
            if d < 0 or np.random.random() < np.exp(-d / temp):
                cur_c = new_c; cur_peak = new_peak; corr = new_corr
                if cur_c < best_c:
                    best_c = cur_c; best_h = h.copy()
            else:
                h[oi1] = 1; h[oi2] = 1; h[zi1] = 0; h[zi2] = 0
        else:
            slen = int(np.random.randint(3, max(4, N // 5)))
            a = int(np.random.randint(0, N - slen))
            b = a + slen
            seg = h[a:b].copy()

            new_corr = compute_corr(h)
            new_peak = float(np.max(new_corr))

                    best_c = cur_c; best_h = h.copy()
            else:
                h[a:b] = seg
    return best_h, best_c

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0].copy()
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0].copy()
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_top(h, corr, k=50):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            grad[M:] += (1.0 - h[:-M])
            grad[:-M] -= h[M:]
    return grad / k

def continuous_opt(h, deadline, lr=0.02, k=40):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stall = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            lr *= 0.9
            stall = 0
            if lr < 1e-5:
                break
        g = grad_top(h, corr, k)
        v = 0.85 * v - lr * g
        h = project_sum(h + v, tgt)
    return best_h, best_c

def binary_sa(h, deadline, t_init=0.004, t_min=0.00004, max_stall=120):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    cur_peak = float(np.max(corr))
    cur_c = cur_peak * (2.0 / N)
    best_c = cur_c
    best_h = h.copy()
    temp = t_init
    stall = 0
    dx = 2.0 / N
    while time.time() < deadline:
        stall += 1
        if stall > max_stall:
            temp = max(t_min, temp * 0.82)
            stall = 0
            if temp < t_min * 1.05:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        r = np.random.random()
        if r < 0.5:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0; h[zi] = 1
            nc = compute_corr(h)
            npk = float(np.max(nc))
            nc_c = npk * dx
            d = nc_c - cur_c
            if d < 0 or np.random.random() < np.exp(-d / temp):
                cur_c = nc_c; cur_peak = npk; corr = nc
                if cur_c < best_c:
                    best_c = cur_c; best_h = h.copy()
            else:
                h[oi] = 1; h[zi] = 0
        elif r < 0.82:
            if len(ones) < 2 or len(zeros) < 2:
                continue
            oi1, oi2 = np.random.choice(ones, 2, replace=False)
            zi1, zi2 = np.random.choice(zeros, 2, replace=False)
            h[oi1] = 0; h[oi2] = 0; h[zi1] = 1; h[zi2] = 1
            nc = compute_corr(h)
            npk = float(np.max(nc))
            nc_c = npk * dx
            d = nc_c - cur_c
            if d < 0 or np.random.random() < np.exp(-d / temp):
                cur_c = nc_c; cur_peak = npk; corr = nc
                if cur_c < best_c:
                    best_c = cur_c; best_h = h.copy()
            else:
                h[oi1] = 1; h[oi2] = 1; h[zi1] = 0; h[zi2] = 0
        else:

            a = int(np.random.randint(0, N - slen))
            b = a + slen
            seg = h[a:b].copy()

            nc = compute_corr(h)
            npk = float(np.max(nc))
            nc_c = npk * dx
            d = nc_c - cur_c
            if d < 0 or np.random.random() < np.exp(-d / temp):
                cur_c = nc_c; cur_peak = npk; corr = nc
                if cur_c < best_c:
                    best_c = cur_c; best_h = h.copy()
            else:
                h[a:b] = seg
    return best_h, best_c

def targeted_il(h, deadline, n_tries=300, n_top=12):
    N = len(h)
    h = fix_binary(h.copy())
    dx = 2.0 / N
    corr = compute_corr(h)
    best_c = float(np.max(corr)) * dx
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        max_val = float(np.max(corr))
        c = max_val * dx
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argsort(corr)[-n_top:]
        best_swap = None
        best_nv = max_val
        for _ in range(n_tries):
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0; h[zi] = 1
            nc = compute_corr(h)
            nv = float(np.max(nc))
            if nv < best_nv:
                best_nv = nv
                best_swap = (oi, zi)
            h[oi] = 1; h[zi] = 0
        if best_swap is not None and best_nv < max_val:
            h[best_swap[0]] = 0
            h[best_swap[1]] = 1
    return best_h, best_c
