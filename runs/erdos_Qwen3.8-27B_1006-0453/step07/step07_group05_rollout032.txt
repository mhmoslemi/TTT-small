import numpy as np
import time
from scipy.fft import rfft, irfft

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = rfft(h, L)
    G = rfft((1.0 - h)[::-1], L)
    return np.real(irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def fast_local_search(h, N, deadline, tabu_len=50):
    """Hill climbing with tabu on 2-flip moves."""
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    tabu = {1: set(), 0: set()}  # tabu[i] = iteration when i becomes non-tabu
    it = 0
    no_improve = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline:
        it += 1
        # Try a few random moves, pick the best improving one
        best_delta = 0
        best_oi = -1
        best_zi = -1
        n_trials = 20
        for _ in range(n_trials):
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            if oi == zi:
                continue
            # Tabu check
            if it < tabu[0].get(oi, 0) or it < tabu[1].get(zi, 0):
                continue
            h[oi] = 0
            h[zi] = 1
            new_max = float(np.max(fft_corr(h)))
            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_oi = oi
                best_zi = zi
            h[oi] = 1
            h[zi] = 0
        
        if best_oi == -1 or best_delta >= -1e-14:
            # No improving move found - perturb
            n_swap = max(3, N // 50)
            for _ in range(n_swap):
                oi = int(np.random.choice(np.where(h == 1)[0]))
                zi = int(np.random.choice(np.where(h == 0)[0]))
                h[oi], h[zi] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

            it = 0

            continue

        h[best_zi] = 1
        cur_max += best_delta
        corr = fft_corr(h)
        cur_max = float(np.max(corr))
        
        # Update tabu
        tabu[1][best_oi] = it + tabu_len
        tabu[1][best_zi] = it + tabu_len  # zi becomes 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

            best_c = cur_max * 2.0 / N
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_h, best_c

A swap of h[a]:1→0 and h[b]:0→1 changes auto_k by:

This computes the circular cross-correlation of h with reversed(1-h) at length L=2N-1.

The circular cross-correlation is:
r[m] = ∑ᵢ h[(i) mod L] * rev(1-h)[(m-i) mod L]

where rev(1-h)[j] = (1-h)[N-1-j] for 0 ≤ j < N, and 0 for N ≤ j < L.

So: r[m] = ∑_{i=0}^{N-1} h[i] * (1-h[N-1-((m-i) mod L)]) (for the non-zero part)

Let me verify: if m = 0, r[0] = ∑ᵢ h[i] * (1-h[N-1-(-i) mod L])
For i < N: (N-1-i) mod L = N-1-i (since N-1-i < L for i ≥ 0 and N < L)
So r[0] = ∑ᵢ h[i] * (1-h[N-1-i])

If m = N-1: r[N-1] = ∑ᵢ h[i] * (1-h[N-1-((N-1-i) mod L)])
(N-1-i) mod L = N-1-i for i ≤ N-1
So r[N-1] = ∑ᵢ h[i] * (1-h[i]) = 0 for binary h. ✓

If m = N: r[N] = ∑ᵢ h[i] * (1-h[N-1-((N-i) mod L)])
For i < N: (N-i) mod L = N-i (since 0 < N-i ≤ N < L)
So r[N] = ∑ᵢ h[i] * (1-h[N-1-(N-i)]) = ∑ᵢ h[i] * (1-h[i-1])
(For i=0: h[N-1-(N)] = h[-1 mod L] = h[L-1] = 0 since we zero-padded)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h):
    target = len(h) // 2
    idx = np.argsort(-h)
    out = np.zeros(len(h))
    out[idx[:target]] = 1.0
    return out

def fix_sum(h, target):
    h = h.copy()
    n1 = int(round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def structured_init(N, variant, seed=0):
    rng = np.random.RandomState(seed)
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[rng.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        x = np.arange(N, dtype=float)
        v = np.sin(2*np.pi*x*3.0/N) + 0.9*np.sin(2*np.pi*x*7.0/N+0.3) + 0.7*np.cos(2*np.pi*x*11.0/N+0.7)
        h = (v > np.median(v)).astype(float)
    elif variant == 2:
        h = ((np.arange(N)*7+3) % 13 < 5).astype(float)
    elif variant == 3:
        x = np.arange(N, dtype=float)
        v = np.zeros(N)
        for f in range(1, 7):
            v += 0.5/f * np.sin(2*np.pi*f*x/N + 0.4*f)
        h = (v > np.median(v)).astype(float)
    else:
        h = np.zeros(N)
        h[rng.choice(N, nh, replace=False)] = 1
    return fix_sum(h, nh)

def il_search(h, N, deadline, max_stall=200, n_trials=15, p_range=(3, 6)):

    nh = N // 2
    h = fix_sum(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    stall = 0
    it = 0

    while time.time() < deadline - 0.3:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        bd = 1e-15
        bm = None
        bc = None

        for _ in range(n_trials):
            oi = int(ones[np.random.randint(len(ones))])
            zi = int(zeros[np.random.randint(len(zeros))])
            if oi == zi:
                continue
            h[oi] = 0.0
            h[zi] = 1.0
            nc = fft_corr(h)
            nm = float(np.max(nc))
            d = nm - cur_max
            if d < bd:
                bd = d
                bm = (oi, zi)
                bc = nc
            h[oi] = 1.0
            h[zi] = 0.0

        if bm is not None and bd < -1e-14:
            oi, zi = bm
            h[oi] = 0.0
            h[zi] = 1.0
            cur_max += bd
            corr = bc
            if cur_max * 2.0 / N < best_c:

def il_block_search(h, N, deadline, max_stall=150, n_trials=10):

                h[oi] = 0.0
