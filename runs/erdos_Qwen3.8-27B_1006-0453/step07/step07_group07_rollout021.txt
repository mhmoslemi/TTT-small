def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)

def compute_c5(h):

def project(h, target_sum):

    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5

        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:

def softmax_obj(h, T):
    """Compute softmax-weighted objective (differentiable proxy for max)."""
    corr = compute_corr(h)
    N = len(h)
    # Shift to positive
    c = corr / (2.0 / N)
    m = np.max(c)
    exp_c = np.exp((c - m) / T)
    val = m + T * np.log(np.sum(exp_c))
    return val

def softmax_grad(h, T, n_top=50):
    """Gradient of softmax objective w.r.t. h."""
    N = len(h)
    corr = compute_corr(h)
    dx = 2.0 / N
    c = corr / dx
    m = np.max(c)
    exp_c = np.exp((c - m) / T)
    w = exp_c / np.sum(exp_c)
    
    # d(corr_k)/d(h_i) for each weighted k
    # corr at center+N-1+m = sum over valid i of h[i]*(1-h[i+m])
    # d/dh_i at lag m: (1-h[i+m]) - h[i-m] (with bounds)
    center = N - 1
    grad = np.zeros(N)
    active = np.where(w * N > 0.01)[0]  # Only top weighted lags
    for m_idx in active:
        M = int(m_idx) - center

        if M == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif abs(M) < N:

                lo = np.arange(0, N - M)

                lo = np.arange(0, N - M)
                grad[lo] += wt * (1.0 - h[lo + M])
                grad[hi] += wt * (-h[hi - M])
    return grad * dx

def binary_sa(h, deadline, max_iters=5000, T_start=0.05, T_end=0.005, block_max=4):
    """Binary simulated annealing with block moves."""
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    dx = 2.0 / N
    best_c = float(np.max(corr)) * dx
    best_h = h.copy()
    best_corr = corr.copy()
    
    T = T_start
    cooling = (T_end / T_start) ** (1.0 / max(1, max_iters))
    it = 0
    no_imp = 0
    
    while time.time() < deadline and it < max_iters:
        T *= cooling
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        cur_c = float(np.max(corr))
        
        # Choose move type
        r = np.random.random()
        if r < 0.6:
            # Single swap
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        elif r < 0.85:
            # Block swap (swap two 1s with two 0s)
            b1 = min(block_max, len(ones) - 1)
            b2 = min(block_max, len(zeros) - 1)
            o1 = int(np.random.choice(ones[:max(1,len(ones)-1)]))
            o2 = int(np.random.choice(ones))
            z1 = int(np.random.choice(zeros))
            z2 = int(np.random.choice(zeros))
            # Avoid overlap
            if o1 == o2 or z1 == z2:
                it += 1
                continue
            h[o1] = 0; h[o2] = 0; h[z1] = 1; h[z2] = 1
        else:
            # 1-swap with nearby element (local move)
            oi = int(np.random.choice(ones))
            dist = int(np.random.choice([1,2,3,4]))
            zi = (oi + dist) % N
            if h[zi] == 0:
                h[oi] = 0
                h[zi] = 1
            else:
                zi = (oi - dist) % N
                if h[zi] == 0:
                    h[oi] = 0
                    h[zi] = 1
                else:
                    it += 1
                    continue
        
        new_corr = compute_corr(h)
        new_c = float(np.max(new_corr))
        
        # Accept/reject
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / (T * dx)):
            corr = new_corr
            cur_c = new_c
            no_imp += 1 if cur_c >= best_c else 0
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
                best_corr = corr.copy()
                no_imp = 0
        else:
            # Revert
            if r < 0.6:
                h[zi] = 0; h[oi] = 1
            elif r < 0.85:
                h[o1] = 1; h[o2] = 1; h[z1] = 0; h[z2] = 0
            else:
                h[zi] = 0; h[oi] = 1
        
        it += 1
        if no_imp > 2000:
            break
    return best_h, best_c

def binary_greedy(h, deadline, n_crit=10, max_rounds=200):

    N = len(h)

        best_swap = None

            for zi in zeros:
                if oi == zi:
                    continue
                h[oi] = 0; h[zi] = 1
                new_max = 0.0
                for li in top_idx:
                    if corr[li] > new_max:
                        # Need to recompute this lag
                        pass
                # Quick check: recompute only top lags
                new_peak = 0.0
                for L in top_lags:
                    v = 0.0
                    if L >= 0:
                        for i in range(N - L):
                            v += h[i] * (1 - h[i + L])
                    else:
                        L2 = -L
                        for i in range(L2, N):
                            v += (1 - h[i]) * h[i - L2]
                    if v > new_peak:
                        new_peak = v
                if new_peak > cur_peak:
                    # Also check if any other lag exceeds
                    # (conservative: assume it might)
                    pass
                h[oi] = 1; h[zi] = 0
                improve = cur_peak - new_peak
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (oi, zi)

            h[best_swap[1]] = 1

            c = float(np.max(corr)) * dx
            if c < best_c:

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    F = 1.0 - h
    GH = np.fft.rfft(np.flip(F), L)
    return np.fft.irfft(H * GH, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def compute_topk_sum(h, k=5):

    corr = compute_corr(h)
    topk = np.sort(corr)[-k:]
    return topk.sum() * (2.0 / N)

def project_sum(h, target):

One more thought: the `compute_corr` function using FFT should be verified. Let me trace through:
- h has N elements
- 1-h has N elements
- np.flip(1-h) reverses 1-h
- L = 2N-1
- H = fft(h, L) - zero-padded FFT of h
- G = fft(flip(1-h), L) - zero-padded FFT of reversed (1-h)
- ifft(H*G) gives the circular convolution of h and flip(1-h) in length L

Since L >= N + N - 1, there's no circular wrap-around, so this is the linear convolution:
conv[k] = Σ_{i=0}^{N-1} h[i] * flip(1-h)[k-i]

where flip(1-h)[j] = (1-h)[N-1-j]

So conv[k] = Σ_i h[i] * (1-h)[N-1-(k-i)] = Σ_i h[i] * (1-h)[N-1-k+i]

Let j = N-1-k+i, so i = j - (N-1-k) = j + k - N + 1
conv[k] = Σ_j h[j+k-N+1] * (1-h)[j] (where both indices valid)

This is essentially the cross-correlation, just with a different indexing. The set of values is the same as np.correlate(h, 1-h, 'full'), just in a different order. So max(conv) = max(np.correlate(h, 1-h, 'full')). Good.

        pop[i] = project_sum(best + np.random.randn(N) * 0.08, tgt)
    fits = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    stagnant = 0
    while time.time() < deadline:
        improved = False
        for i in range(pop_size):
            if time.time() > deadline:
                break
            others = [j for j in range(pop_size) if j != i]
            a, b, c = np.random.choice(others, 3, replace=False)
            mutant = np.clip(pop[a] + F * (pop[b] - pop[c]), 0, 1)
            mutant = project_sum(mutant, tgt)
            trial = pop[i].copy()
            mask = np.random.random(N) < CR
            trial[mask] = mutant[mask]
            trial = project_sum(trial, tgt)
            ft = compute_c5(trial)
            if ft < fits[i]:
                pop[i] = trial
                fits[i] = ft
                improved = True
                if ft < best_c:
                    best_c = ft
                    best = trial.copy()
                    stagnant = 0
        if not improved:
            stagnant += 1
            if stagnant > 8:
                break
        # Elitism: copy best into random slot
        bi = np.argmin(fits)
        ri = int(np.random.randint(pop_size))
        pop[ri] = pop[bi]
        fits[ri] = fits[bi]
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Continuous gradient optimization
    d1 = min(deadline - 500, t0 + 100)
    h1 = project_sum(best_h + np.random.randn(N0) * 0.04, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=14, lr0=0.012, mom=0.88)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: DE in continuous space
    if time.time() < deadline - 380:
        d2 = min(deadline - 360, time.time() + 160)
        h2, c2 = de_continuous(best_h, d2, pop_size=30, F=0.5, CR=0.85)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: Binary SA (main workhorse)
    if time.time() < deadline - 260:
        d3 = min(deadline - 240, time.time() + 160)
        hb = fix_binary(best_h)
        hb, c3 = binary_sa(hb, d3, T_start=0.07, T_end=0.002, total_iters=5000)
        if c3 < best_c:
            best_c, best_h = c3, hb.copy()

    # Phase 4: Greedy top-lag refinement
    if time.time() < deadline - 140:
        d4 = min(deadline - 120, time.time() + 80)
        h4, c4 = binary_greedy_top(best_h, d4, n_crit=10, max_rounds=60)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Short continuous polish
    if time.time() < deadline - 100:
        d5 = min(deadline - 80, time.time() + 60)
        h5 = project_sum(best_h + np.random.randn(N0) * 0.015, N0 / 2.0)
        h5, c5 = cont_opt(h5, d5, k=10, lr0=0.006, mom=0.91)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Final binary SA (low temp)
    if time.time() < deadline - 30:
        d6 = min(deadline - 3, time.time() + 90)
        hb6 = fix_binary(best_h)
        hb6, c6 = binary_sa(hb6, d6, T_start=0.025, T_end=0.001, total_iters=3500)
        if c6 < best_c:
            best_c, best_h = c6, hb6.copy()

    # Try resolution 400
    if time.time() < deadline - 25:
        N_try = 400
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d7 = min(deadline - 2, time.time() + 20)
        h_bt, c7 = binary_sa(h_bt, d7, T_start=0.04, T_end=0.003, total_iters=1500)
        if c7 < best_c:
            best_c, best_h = c7, h_bt.copy()

    # Final validation
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
