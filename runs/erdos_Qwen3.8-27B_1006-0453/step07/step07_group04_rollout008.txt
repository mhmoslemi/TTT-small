def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_D(h, N):

                if npk[mi] < best_new_P:
                    best_new_P = float(npk[mi])
                    best_pq = (int(p), int(zeros[mi]))
            if best_pq is None:
                break
            delta = best_new_P - P
            eval_cnt += 1
            if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0, delta) / T)):
                acc_cnt += 1
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
            else:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
        rate = acc_cnt / max(1, eval_cnt)
        if eval_cnt % 100 == 0:
            if rate < 0.15:
                T *= 1.08
            elif rate > 0.5:
                T *= 0.95
            T = max(T_min, min(T, T0))
            acc_cnt = 0
            eval_cnt = 0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=60, lr0=0.02, beta=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.97
            beta = min(0.97, 0.89 + it * 0.00005)
    return best_h, best_pk * dx

def block_init(N, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(N)
    mid = N // 2
    h[:mid] = 1.0
    holes = rng.choice(mid, size=mid // 6, replace=False)
    h[holes] = 0.0
    bumps = rng.choice(np.arange(mid, N), size=mid // 6, replace=False)
    h[bumps] = 1.0
    return ensure_sum(h, N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    h_bin = ensure_sum(h0, N0)

    d1 = min(deadline - 900, t0 + 25)
    h1, c5_1 = greedy_binary(h_bin, N0, d1, max_sweeps=200, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    d2 = min(deadline - 750, t0 + 140)
    h2, c5_2 = sa_adaptive(best_h, N0, d2, max_sweeps=6000, T0=3.5, Tcool=0.995,
                           T_min=0.03, sf=1.0, pair_prob=0.3, restart_stall=120, kick_size=5)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    N1 = 2 * N0
    if time.time() < deadline - 580:
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d3 = min(deadline - 500, t0 + 210)
        h3, c5_3 = greedy_binary(h_up, N1, d3, max_sweeps=400, sf=0.8)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    if time.time() < deadline - 400:
        h_up2 = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d4 = min(deadline - 350, t0 + 340)
        h4, c5_4 = sa_adaptive(h_up2, N1, d4, max_sweeps=4000, T0=4.0, Tcool=0.995,
                               T_min=0.04, sf=0.6, pair_prob=0.3, restart_stall=150, kick_size=4)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    if time.time() < deadline - 280:
        hb = block_init(N0, seed + 1)
        d5 = min(deadline - 230, t0 + 420)
        h5, c5_5 = sa_adaptive(hb, N0, d5, max_sweeps=4000, T0=4.5, Tcool=0.994,
                               T_min=0.04, sf=1.0, pair_prob=0.25, restart_stall=100, kick_size=6)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    if time.time() < deadline - 180:
        N_fin = len(best_h)
        if N_fin > N0:
            hc = project_sum(upsample(best_h, N_fin, N0) + np.random.randn(N0) * 0.05, N0 / 2.0)
        else:
            hc = project_sum(best_h + np.random.randn(N0) * 0.05, N0 / 2.0)
        d6 = min(deadline - 150, t0 + 520)
        h6, c5_6 = nesterov_opt(hc, N0, d6, k=65, lr0=0.02, beta=0.9)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    if time.time() < deadline - 100:
        N_cur = len(best_h)
        h_sb = ensure_sum(best_h, N_cur)
        for _ in range(8):
            p = np.where(h_sb == 1)[0]
            q = np.where(h_sb == 0)[0]
            if len(p) > 0 and len(q) > 0:
                h_sb[p[np.random.randint(len(p))]], h_sb[q[np.random.randint(len(q))]] = 0.0, 1.0
        d7 = min(deadline - 60, t0 + 600)
        sf7 = 1.0 if N_cur <= N0 else 0.55
        h7, c5_7 = sa_adaptive(h_sb, N_cur, d7, max_sweeps=5000, T0=3.0, Tcool=0.996,
                               T_min=0.03, sf=sf7, pair_prob=0.35, restart_stall=100, kick_size=5)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    if time.time() < deadline - 40:
        if len(best_h) > N0:
            hc2 = project_sum(upsample(best_h, len(best_h), N1) + np.random.randn(N1) * 0.03, N1 / 2.0)
            d8 = min(deadline - 20, t0 + 660)
            h8, c5_8 = nesterov_opt(hc2, N1, d8, k=90, lr0=0.012, beta=0.9)
            if c5_8 < best_c5:
                best_c5, best_h = c5_8, h8.copy()
        else:
            h_res = ensure_sum(upsample(best_h, len(best_h), N1), N1)
            d8 = min(deadline - 20, t0 + 660)
            h8, c5_8 = sa_adaptive(h_res, N1, d8, max_sweeps=2500, T0=3.5, Tcool=0.995,
                                   T_min=0.04, sf=0.5, pair_prob=0.3, restart_stall=120, kick_size=3)
            if c5_8 < best_c5:
                best_c5, best_h = c5_8, h8.copy()

    N_fin = len(best_h)
    if time.time() < deadline - 1:
        hF = ensure_sum(best_h, N_fin)
        sfF = 1.0 if N_fin <= 400 else 0.7
        hF2, c5F = greedy_binary(hF, N_fin, deadline - 0.5, max_sweeps=400, sf=sfF)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
