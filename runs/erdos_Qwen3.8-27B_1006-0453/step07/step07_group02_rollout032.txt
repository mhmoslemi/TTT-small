import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum_binary(h, nh):
    """Fix sum of binary h to exactly nh ones."""
    cur = h.sum()
    if cur > nh:
        diff = int(round(cur - nh))
        ones = np.where(h > 0.5)[0]
        perm = np.random.permutation(ones)
        for i in range(min(diff, len(perm))):
            h[perm[i]] = 0.0
    elif cur < nh:
        diff = int(round(nh - cur))
        zeros = np.where(h < 0.5)[0]
        perm = np.random.permutation(zeros)
        for i in range(min(diff, len(perm))):
            h[perm[i]] = 1.0
    return h

    """Compute change in correlation if we flip position idx (1->0 or 0->1)."""

    i_mk = idx - k_vals
    i_pk = idx + k_vals

    """Binary local search with 1-swap and 2-swap moves."""
    h = h.astype(float).copy()
    fix_sum_binary(h, N // 2)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    
    # Precompute C matrix
    C = np.zeros((N, 2*N-1))
    for i in range(N):
        C[i, :] = compute_C_single(h, N, i)
    
    stuck = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Try 1-swap
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        
        new1 = corr[None, None, :] + C[os_][None, :, :] + C[zs][:, None, :]
        newP1 = new1.max(axis=2)
        mi1 = np.unravel_index(np.argmin(newP1), newP1.shape)
        mv1 = float(newP1[mi1])

            h[zs[mi1[0]]] = 1
            C[os_[mi1[1]], :] = compute_C_single(h, N, os_[mi1[1]])

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum_binary(h, nh):
    cur = int(np.round(h.sum()))
    if cur > nh:
        ones = np.where(h > 0.5)[0]
        pick = np.random.permutation(ones)[:cur - nh]
        h[pick] = 0.0
    elif cur < nh:
        zeros = np.where(h < 0.5)[0]
        pick = np.random.permutation(zeros)[:nh - cur]
        h[pick] = 1.0
    return h

def C_row(h, N, idx):
    k = np.arange(2*N-1) - (N-1)
    im = idx - k
    ip = idx + k
    vm = h[idx]
    s = -1.0 if vm == 1 else 1.0
    A = np.where((im >= 0) & (im < N), s*(1-h[np.clip(im,0,N-1)]), 0.0)
    B = np.where((ip >= 0) & (ip < N), s*h[np.clip(ip,0,N-1)], 0.0)
    return A + B

def project_01(h, target):

        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

            Cm[os_[mi[1]]] = C_row(h, N, os_[mi[1]])
            Cm[zs[mi[0]]] = C_row(h, N, zs[mi[0]])
