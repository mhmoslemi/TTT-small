def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def compute_D(h, N):
    """Compute the D matrix for binary swaps."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum_binary(h, N, seed=42):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h >= 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(seed + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def smooth_cont_opt(h, N, deadline, k_top=100, temp0=0.30, temp_min=0.02, lr0=0.006, beta0=0.88):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m_buf = np.zeros(N)
    lr = lr0
    beta = beta0
    temp = temp0
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m_buf
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = y.copy()
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        top_vals = corr[top_idx]
        mx = top_vals.max()
        w = np.exp((top_vals - mx) / temp)
        sw = w.sum()
        if sw < 1e-30:
            w = np.ones(k_top) / k_top
        else:
            w = w / sw
        grad = np.zeros(N)
        for ai in range(k_top):
            L = int(top_idx[ai]) - center
            wi = w[ai]
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m_buf = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            temp = max(temp_min, temp * 0.997)
            lr *= 0.992
            beta = min(0.96, beta0 + it * 0.00004)
    final_c = float(np.max(fft_corr(best_h))) * dx
    return best_h, final_c

def binary_opt(h, N, deadline, two_swap_p=0.35, three_swap_p=0.10, sample_frac=1.0, seed=99):
    h = h.astype(float).copy()
    rng = np.random.RandomState(seed)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    max_sweeps = 5000
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 5 or len(zeros) < 5:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P_cur
        best_pq = None
        for p in p_list:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:

                best_pq = (int(p), int(zeros[mi]))

                if i1 == i2 or j1 == j2:

                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = (p1, p2, q1, q2)
        best_3p = None
        if rng.random() < three_swap_p and len(ones) >= 6 and len(zeros) >= 6:
            n_try = min(50, max(15, N // 8))
            for _ in range(n_try):
                ia = rng.choice(len(ones), 3, replace=False)
                ja = rng.choice(len(zeros), 3, replace=False)
                ps = (int(ones[ia[0]]), int(ones[ia[1]]), int(ones[ia[2]]))
                qs = (int(zeros[ja[0]]), int(zeros[ja[1]]), int(zeros[ja[2]]))
                new_P = float((c_cur + D[ps[0]] + D[ps[1]] + D[ps[2]] - D[qs[0]] - D[qs[1]] - D[qs[2]]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = None
                    best_3p = (ps, qs)
        applied = False
        if best_3p is not None:
            ps, qs = best_3p
            for p in ps: h[p] = 0.0
            for q in qs: h[q] = 1.0
            applied = True
        elif best_2p is not None:
            p1, p2, q1, q2 = best_2p
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
            applied = True
        elif best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            applied = True
        else:
            if stall > 15:
                n_fl = max(3, N // 40)
                ones_cur = np.where(h == 1)[0]
                zeros_cur = np.where(h == 0)[0]
                for _ in range(n_fl):
                    p = int(ones_cur[rng.randint(len(ones_cur))])
                    q = int(zeros_cur[rng.randint(len(zeros_cur))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 5:
                break
            else:
                stall += 1
                continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-13:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2.0, N_old, endpoint=False)
    x_new = np.linspace(0, 2.0, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
