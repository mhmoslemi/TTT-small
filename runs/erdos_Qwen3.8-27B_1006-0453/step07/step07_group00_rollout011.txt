import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        perm = np.random.permutation(len(idx))[:n1 - target]
        h[idx[perm]] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        perm = np.random.permutation(len(idx))[:target - n1]
        h[idx[perm]] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def sa_optimize(h_bin, deadline, n_peaks=8, n_per_peak=22, cool_rate=0.9991, t_start=0.04, tabu_sz=40, restart_perturb=12000):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    tptr = 0
    tabu_p = np.full(tabu_sz, -1, dtype=int)
    tabu_q = np.full(tabu_sz, -1, dtype=int)
    n_acc = 0
    n_tot = 0
    while time.time() < deadline and it < 300000:
        it += 1
        if it % 400 == 0:
            T = max(T * cool_rate, 0.001)
            if n_tot > 40:
                rate = n_acc / n_tot
                if rate > 0.55:
                    T *= 0.92
                elif rate < 0.06:
                    T = min(T * 1.1, t_start)
                n_acc, n_tot = 0, 0
            if no_impr > restart_perturb:
                h = best_h.copy()
                np.random.shuffle(h)
                for _ in range(max(3, N // 60)):
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                    if len(oi) > 0 and len(zi) > 0:
                        h[np.random.choice(oi)] = 0
                        h[np.random.choice(zi)] = 1
                corr = fft_corr(h)
                s, deltas = build_deltas(h)
                no_impr = 0
                T = t_start * 0.6
                tptr = 0
                tabu_p[:] = -1
                tabu_q[:] = -1
                continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        pk_n = min(n_peaks, len(corr))
        peak_lags = np.argpartition(corr, -pk_n)[-pk_n:]
        cpl, cql = [], []
        for pk in peak_lags:
            ko = int(pk) - (N - 1)
            if ko >= 0:
                vi = np.arange(0, N - ko)
                if len(vi) == 0: continue
                m = (h[vi] == 1) & (h[vi + ko] == 0)
                bp, bq = vi[m], (vi + ko)[m]
            else:
                M = -ko
                vi = np.arange(M, N)
                if len(vi) == 0: continue
                m = (h[vi] == 1) & (h[vi - M] == 0)
                bp, bq = vi[m], (vi - M)[m]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cpl.extend(bp[sel].tolist())
                cql.extend(bq[sel].tolist())
        nr = min(8, len(ones), len(zeros))
        cpl.extend(np.random.choice(ones, nr, replace=False).tolist())
        cql.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cpl)), dtype=int)
        cand_q = np.array(sorted(set(cql)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 10
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        is_tab = False
        for ti in range(tabu_sz):
            if tabu_p[ti] == -1:
                break
            if (tabu_p[ti] == p and tabu_q[ti] == q) or (tabu_p[ti] == q and tabu_q[ti] == p):
                is_tab = True
                break
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and not is_tab and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            corr = corr + deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p + N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q + N] = s
            tabu_p[tptr] = p
            tabu_q[tptr] = q
            tptr = (tptr + 1) % tabu_sz
            n_acc += 1
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
            n_tot += 1
        else:
            no_impr += 1
            n_tot += 1
    return best_h, best_c

def grad_polish(h_in, N, deadline, lr=0.007, k_top=35):
    h = h_in.astype(float).copy()
    m = np.zeros(N)
    beta = 0.8
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    it = 0
    while time.time() < deadline:
        y = h + m
        cy = fft_corr(y)
        pk = float(cy.max())
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        top = np.argpartition(cy, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.96
            beta = min(0.92, 0.78 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):

    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)

    c1 = compute_c5(h1, N0)
    if c1 < best_c5:

    # Phase 2: gradient polish

        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)

        if c2 < best_c5:

    # Phase 3: focused SA (low temp)

        c3 = compute_c5(h3, N0)
