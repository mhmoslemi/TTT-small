    return best

def apply_swap_update(h, corr, o, z, N):
    """Apply swap and update corr. Must be called BEFORE changing h."""
    # For each k where delta is non-zero, update corr
    # Delta is non-zero when o+k in [0,N) or z+k in [0,N) or o-k in [0,N) or z-k in [0,N)
    # k ranges from -(N-1) to N-1
    # o+k in [0,N): k in [-o, N-1-o]
    # z+k in [0,N): k in [-z, N-1-z]
    # o-k in [0,N): k in [o-N+1, o]
    # z-k in [0,N): k in [z-N+1, z]
    k_lo = max(-(N-1), -o, -z, o-N+1, z-N+1)
    k_hi = min(N-1, N-1-o, N-1-z, o, z)
    
    for k in range(k_lo, k_hi + 1):
        delta = 0.0
        # i=o, j=o+k
        if 0 <= o + k < N:
            j = o + k
            if j == z:
                delta -= 1.0
            else:
                delta -= (1.0 - h[j])
        # i=z, j=z+k
        if 0 <= z + k < N:
            j = z + k
            if j == o:
                delta += 1.0
            else:
                delta += (1.0 - h[j])
        # i=o-k, j=o
        if 0 <= o - k < N:
            i2 = o - k
            if i2 != o and i2 != z:
                delta -= h[i2]
        # i=z-k, j=z
        if 0 <= z - k < N:
            i2 = z - k
            if i2 != o and i2 != z:
                delta += h[i2]
        if delta != 0.0:
            corr[N-1+k] += delta
    
    h[o] = 0.0
    h[z] = 1.0
