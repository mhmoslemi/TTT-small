def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)

    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:n_ones]] = 1.0
    return out

        np.random.shuffle(ones)
        h[ones[:n1-target]] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target-n1]] = 1

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum(h, target):
    h = h.copy()
    n1 = int(np.sum(h == 1))
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h

def structured_init(N, variant):

def local_search_exhaustive(h_init, N, deadline):
    """Try all single-swap and pair-swap improvements."""
    h = h_init.astype(float).copy()
    nh = N // 2
    h = fix_sum(h, nh)
    best_c = compute_c5(h, N)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        corr = fft_corr(h)
        cur_max = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Single swaps
        for oi in ones:
            if time.time() > deadline: break
            for zi in zeros:
                if oi == zi: continue
                h[oi] = 0; h[zi] = 1
                nm = float(np.max(fft_corr(h)))
                if nm < cur_max - 1e-14:
                    cur_max = nm
                    best_c = nm * 2.0 / N
                    best_h = h.copy()
                    improved = True
                    break
                h[oi] = 1; h[zi] = 0
            if time.time() > deadline: break
        if time.time() > deadline: break
    return best_h, best_c

def continuous_refine(h_init, N, deadline, lr0=0.004):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    soft_T = 0.04
    it = 0
    while time.time() < deadline - 2:
        it += 1
        corr = fft_corr(h)
        peak = np.max(corr)
        w = np.exp((corr - peak) / soft_T)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        significant = np.where(w > 1e-5)[0]
        for ki in significant:
            L = int(ki - center)
            wt = w[ki]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wt * (1.0 - h[:N-L])
                grad[:N-L] -= wt * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += wt * (1.0 - h[M:])
                    grad[M:] -= wt * h[:N-M]
        h_new = np.clip(h - lr * grad, 0, 1)
        # Enforce sum constraint via bisection on offset
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        c_new = compute_c5(h_new, N)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 0:
            lr *= 0.98
            soft_T *= 0.97
    return best_h, best_c
