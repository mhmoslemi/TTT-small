        # For one at position p, contribution to lag L: h[p+L] + h[p-L] (if valid)

        contrib_zero = np.zeros((nz, n_crit))
        for i, p in enumerate(oi_arr):

                if 0 <= p + L < N:

                if 0 <= p - L < N:

        for i, p in enumerate(zi_arr):

                    v += h[p + L]
                if 0 <= p - L < N:
                    v += h[p - L]
                contrib_zero[i, j] = v

        delta = contrib_one[:, None, :] - contrib_zero[None, :, :]
        new_vals = corr[top_idx][None, None, :] + delta
        new_peaks = new_vals.max(axis=2)
        
        # Find best swap
        flat_idx = np.argmin(new_peaks)
        best_oi = oi_arr[flat_idx // nz]
        best_zi = zi_arr[flat_idx % nz]

            no_improve = 0

        temp *= cool
    
    return best_h, best_c

    vm = (pm >= 0) & (pm < N)
    return h[np.clip(pp, 0, N - 1)] * vp + h[np.clip(pm, 0, N - 1)] * vm

def binary_sa(h_init, deadline, n_crit=35, s_sz=50, t_init=3.0, t_floor=0.005, cool=0.998):
    N = len(h_init)
    center = N - 1
    h = fix_binary(h_init.copy())
    corr = compute_corr(h)
    best_c = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    temp = t_init
    no_imp = 0
    while time.time() < deadline and temp > t_floor:
        top_idx = np.argsort(corr)[-n_crit:]
        top_lags = (top_idx - center).astype(int)
        cur_peak = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        oi_arr = np.random.choice(ones, ns, replace=False)
        zi_arr = np.random.choice(zeros, nz, replace=False)
        c1 = contrib_matrix(h, oi_arr, top_lags, N)
        c0 = contrib_matrix(h, zi_arr, top_lags, N)
        delta = c1[:, None, :] - c0[None, :, :]
        new_vals = corr[top_idx][None, None, :] + delta
        new_peaks = new_vals.max(axis=2)
        fi = int(np.argmin(new_peaks))
        bi, bj = fi // nz, fi % nz
        best_peak = float(new_peaks[bi, bj])
        if best_peak < cur_peak - 1e-10:
            h[oi_arr[bi]] = 0
            h[zi_arr[bj]] = 1
            corr = compute_corr(h)
            c = float(np.max(corr)) * (2.0 / N)
            if c < best_c - 1e-12:
                best_c, best_h = c, h.copy()
            no_imp = 0
        else:
            dE = best_peak - cur_peak
            if dE / temp < 50 and np.random.random() < np.exp(-dE / temp):
                h[oi_arr[bi]] = 0
                h[zi_arr[bj]] = 1
                corr = compute_corr(h)
            no_imp += 1
        if no_imp > 600:
            for _ in range(np.random.randint(3, 7)):
                h[np.random.choice(np.where(h == 1)[0])] = 0
                h[np.random.choice(np.where(h == 0)[0])] = 1
            corr = compute_corr(h)
            no_imp = 0
            temp = min(temp * 1.8, t_init)
        temp *= cool
    return best_h, best_c

def cont_opt(h_init, deadline, beta=14.0, lr=0.007, n_top=35):
    N = len(h_init)
    h = h_init.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c, best_h = c, h.copy()
        top_k = np.argsort(corr)[-n_top:]
        tc = corr[top_k] - tc.min()
        w = np.exp(beta * tc)
        w /= w.sum()
        top_lags = (top_k - center).astype(int)
        grad = np.zeros(N)
        for wi, L in zip(w, top_lags):
            L = int(L)
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L < N:
                grad[L:] += wi * (1.0 - h[:-L])
                grad[:-L] -= wi * h[L:]
            else:
                continue
        h_new = project(h + 0.4 * v - lr * grad, N / 2.0)
        v = 0.5 * v + 0.4 * (h_new - h)
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: Binary SA at initial resolution
    d1 = min(t0 + 220, deadline - 500)
    hb = fix_binary(best_h)
    hb, c1 = binary_sa(hb, d1, n_crit=35, s_sz=50, t_init=3.0, cool=0.9975)
    if c1 < best_c:
        best_c, best_h = c1, hb.copy()
    # Phase 2: Continuous soft-opt
    if time.time() < deadline - 350:
        d2 = min(t0 + 380, deadline - 330)
        hc = project(best_h + np.random.randn(N0) * 0.025, N0 / 2.0)
        hc, c2 = cont_opt(hc, d2, beta=14, lr=0.006, n_top=35)
        if c2 < best_c:
            best_c, best_h = c2, hc.copy()
    # Phase 3: Binary SA round 2
    if time.time() < deadline - 280:
        d3 = min(t0 + 530, deadline - 260)
        hb3 = fix_binary(best_h)
        hb3, c3 = binary_sa(hb3, d3, n_crit=40, s_sz=55, t_init=2.0, cool=0.997)
        if c3 < best_c:
            best_c, best_h = c3, hb3.copy()
    # Phase 4: Try N=400
    if time.time() < deadline - 200:
        N4 = 400
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N4, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb4 = fix_binary(hi)
        d4 = min(t0 + 680, deadline - 140)
        hb4, c4 = binary_sa(hb4, d4, n_crit=30, s_sz=50, t_init=2.0, cool=0.998)
        if time.time() < deadline - 50:
            hc4 = project(hb4 + np.random.randn(N4) * 0.025, N4 / 2.0)
            d5 = min(deadline - 20, time.time() + 45)
            hc4, c5v = cont_opt(hc4, d5, beta=12, lr=0.005, n_top=30)
            if c5v < c4:
                c4, hb4 = c5v, hc4
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()
    # Phase 5: Try N=460 if time
    if time.time() < deadline - 150 and len(best_h) != 460:
        N5 = 460
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N5, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb5 = fix_binary(hi)
        d6 = min(deadline - 80, time.time() + 100)
        hb5, c6 = binary_sa(hb5, d6, n_crit=30, s_sz=50, t_init=2.0, cool=0.998)
        if c6 < best_c:
            best_c, best_h = c6, hb5.copy()
    # Phase 6: Final polish
    if time.time() < deadline - 3:
        d7 = deadline - 2
        hb7 = fix_binary(best_h)
        hb7, c7 = binary_sa(hb7, d7, n_crit=45, s_sz=60, t_init=1.0, cool=0.996)
        if c7 < best_c:
            best_c, best_h = c7, hb7.copy()
    # Final check
    cC = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < cC:
        best_h, best_c = hB, cB
    else:
        best_c = cC
    return best_h, best_c, len(best_h)
