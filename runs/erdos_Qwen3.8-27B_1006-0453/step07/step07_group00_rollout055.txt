import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def make_binary(h):
    N = len(h)
    h = h.astype(float).copy()
    idx = np.argsort(h)
    h[:] = 0.0
    h[idx[N//2:]] = 1.0
    return h
