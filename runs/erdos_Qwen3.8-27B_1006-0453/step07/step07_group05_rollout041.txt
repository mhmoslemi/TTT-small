This is convex in h (since each corr_k is linear in h, and exp of linear is convex). Wait, is corr_k linear in h?

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def fix_sum_binary(h, target):
    h = h.copy().astype(float)
    n1 = int(np.sum(h > 0.5))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:n1-target]] = 0.0
    elif n1 < target:
        idx = np.where(h <= 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:target-n1]] = 1.0
    return h

def enforce_sum(h, target):

    d2 = min(deadline - 350, time.time() + 200)
    hb, c2 = binary_peak_search(h_b, N0, d2, T0=0.04, max_iters=35000)
    c2b = compute_c5(hb, N0)
    if c2b < best_c5:
        best_c5, best_h = c2b, hb.copy()

    # Phase 3: Hybrid continuous+binary refinement from best
    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 180)
        hhy, ch = hybrid_refine(best_h, N0, d3)
        if ch < best_c5:
            best_c5, best_h = ch, hhy.copy()

    # Phase 4: Another round of finer continuous optimization
    if time.time() < deadline - 80:
        d4 = min(deadline - 30, time.time() + 100)
        hc2, c4 = continuous_opt(best_h, N0, d4, lr0=0.0008, temp0=0.015)
        if c4 < best_c5:
            best_c5, best_h = c4, hc2.copy()

    # Phase 5: Try a different N for diversity
    if time.time() < deadline - 40:
        N2 = 384
        x = np.arange(N2)
        freqs = [1.0, 1.414, 1.732, 2.236, 3.162]
        v = np.zeros(N2)
        for f in freqs:
            v += np.sin(2 * np.pi * f * x / N2 + 0.3 * f)
        h2 = (v > np.median(v)).astype(float)
        h2 = fix_sum_binary(h2, N2 // 2)
        d5 = min(deadline - 5, time.time() + 30)
        h2r, c5v = binary_peak_search(h2, N2, d5, T0=0.06, max_iters=10000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2r.copy()

    # Final: verify and return best
    c_final = compute_c5(best_h, len(best_h))
    if c_final < best_c5:
        best_c5 = c_final
    return best_h, best_c5, len(best_h)
