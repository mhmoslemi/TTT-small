def swap_delta(h, p, q, N):
    """Compute delta in correlation for all lags when swapping h[p]:1->0, h[q]:0->1"""
    L = np.arange(-(N-1), N)
    delta = np.zeros(2*N-1)
    
    # Contribution from p (was 1, now 0)
    # Term 1: i=p, need 0<=p+L<N
    valid1 = (p + L >= 0) & (p + L < N)
    idx1 = p + L[valid1]  # p+L values
    delta[valid1] += h[idx1] - 1  # = -(1 - h[p+L])
    
    # Term 2: i=p-L, need 0<=p-L<N
    valid2 = (p - L >= 0) & (p - L < N)
    idx2 = p - L[valid2]
    delta[valid2] += h[idx2]  # = h[p-L]
    
    # Contribution from q (was 0, now 1)
    # Term 3: i=q, need 0<=q+L<N
    valid3 = (q + L >= 0) & (q + L < N)
    idx3 = q + L[valid3]
    delta[valid3] += 1 - h[idx3]  # = 1 - h[q+L]
    
    # Term 4: i=q-L, need 0<=q-L<N
    valid4 = (q - L >= 0) & (q - L < N)
    idx4 = q - L[valid4]
    delta[valid4] -= h[idx4]  # = -h[q-L]
    
    return delta

def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)

    A = np.zeros((N, 2*N-1))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def build_A(h, N):
    A = np.zeros((N, 2 * N - 1))
    for j in range(N):
        lo1 = max(j - N + 1, -(N - 1))
        hi1 = min(j, N - 1)
        if lo1 <= hi1:
            Ls = np.arange(lo1, hi1 + 1)
            A[j, Ls + (N - 1)] += h[j - Ls]
        lo2 = max(-j, -(N - 1))
        hi2 = min(N - 1 - j, N - 1)
        if lo2 <= hi2:
            Ls2 = np.arange(lo2, hi2 + 1)
            A[j, Ls2 + (N - 1)] -= (1.0 - h[j + Ls2])
    return A

def best_swap(c, A, ones, zeros):
    P = float(c.max())
    best_new = P + 1.0
    best_pq = None
    for p in ones:
        d = c + A[p]
        vals = d[None, :] - A[zeros, :]
        maxes = vals.max(axis=1)
        mi = int(np.argmin(maxes))
        if maxes[mi] < best_new:
            best_new = maxes[mi]
            best_pq = (int(p), int(zeros[mi]))
    return best_pq, best_new

def greedy_sa(h_init, N, deadline, T0=3.0, Tcool=0.998, T_min=0.01,
              kick_every=80, kick_n=4, max_sweeps=5000, rng_seed=7):
    h = h_init.astype(float).copy()
    c = fft_corr(h)
    A = build_A(h, N)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    stall = 0
    rng = np.random.RandomState(rng_seed)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if stall >= kick_every:
            for _ in range(kick_n):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
                c = c + A[p] - A[q]
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            A = build_A(h, N)
            stall = 0
            T = T0 * 0.5
            P = float(c.max())
            if P < best_c - 1e-12:
                best_c = P
                best_h = h.copy()
            continue

        best_pq, best_new_P = best_swap(c, A, ones, zeros)
        if best_pq is None:
            break

        delta = best_new_P - P
        accept = delta < -1e-10 or (T > T_min and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))

        if accept:
            p, q = best_pq
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]

        h[p], h[q] = 0.0, 1.0
        c = c + A[p] - A[q]
        A = build_A(h, N)
        T = max(T_min, T * Tcool)

    return best_h, best_c * (2.0 / N)

def two_opt_polish(h, N, deadline, max_rounds=200):
    h = h.astype(float).copy()
    c = fft_corr(h)
    A = build_A(h, N)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        pq1, v1 = best_swap(c, A, ones, zeros)
        if pq1 is None or v1 >= P - 1e-12:
            break
        p1, q1 = pq1
        h[p1], h[q1] = 0.0, 1.0
        c = c + A[p1] - A[q1]
        A = build_A(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        pq2, v2 = best_swap(c, A, ones, zeros)
        if pq2 is not None and v2 < P - 1e-12:
            p2, q2 = pq2
            h[p2], h[q2] = 0.0, 1.0
            c = c + A[p2] - A[q2]
            A = build_A(h, N)
    return best_h, best_c * (2.0 / N)
