    if c5_lp2 < best_c5:

    # Phase 4: Nesterov polish on continuous
    if time.time() < deadline - 200:
        d4 = min(deadline - 180, time.time() + 100)
        h4 = best_h + np.random.randn(N0) * 0.03
        h4 = project_01(h4, N0 / 2.0)
        h_n, c5_n = nesterov_opt(h4, N0, d4, k=40, lr0=0.006)
        if c5_n < best_c5:
            best_c5, best_h = c5_n, h_n.copy()

    # Phase 5: Binary search with annealing from best
    if time.time() < deadline - 150:
        h5 = (best_h > 0.5).astype(float)
        d5 = min(deadline - 130, time.time() + 150)
        h_s5, c5_5 = binary_search(h5, N0, d5, max_iters=10000, sample_sz=130, temp=0.2, kick_thresh=45, kick_sz=5)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h_s5.copy()

    # Phase 6: Try different N values
    for N_try in [360, 400, 420]:
        if time.time() > deadline - 80:
            break
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = project_01(h_int, N_try / 2.0)
        sub_dl = min(deadline - 30, time.time() + 60)
        h_rt, c5_rt = lp_alternate(h_bt, N_try, max_iters=100, deadline=sub_dl)
        h_rb = (h_rt > 0.5).astype(float)
        fix_sum_binary(h_rb, N_try // 2)
        sub_dl2 = min(deadline - 20, time.time() + 40)
        h_rb, c5_rb = binary_search(h_rb, N_try, sub_dl2, max_iters=3000, sample_sz=N_try // 3, kick_thresh=40, kick_sz=4)
        c5_chk = min(c5_rt, c5_rb)
        if c5_rb < c5_rt:
            h_best_try = h_rb
        else:
            h_best_try = h_rt
        if c5_chk < best_c5:
            best_c5, best_h = c5_chk, h_best_try.copy()

    # Phase 7: Final LP polish
    if time.time() < deadline - 20:
        hF = best_h.copy()
        dF = deadline - 5
        hF, c5F = lp_alternate(hF, len(best_h), max_iters=100, deadline=dF)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Final binary cleanup
    if time.time() < deadline - 1:
        hB = (best_h > 0.5).astype(float)
        fix_sum_binary(hB, len(best_h) // 2)
        c5B = compute_c5(hB, len(best_h))
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
