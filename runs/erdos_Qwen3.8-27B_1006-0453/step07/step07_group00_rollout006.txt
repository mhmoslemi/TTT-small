import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        # Flip the smallest values above 0.5
        above = np.where(h > 0.5)[0]
        if len(above) >= n1 - target:
            idx = above[np.argsort(h[above])[:n1-target]]
        else:
            idx = np.random.choice(above, n1 - target, replace=False)
        h[idx] = 0
    elif n1 < target:
        below = np.where(h < 0.5)[0]
        if len(below) >= target - n1:
            idx = below[np.argsort(-h[below])[:target-n1]]
        else:
            idx = np.random.choice(below, target - n1, replace=False)
        h[idx] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_s_and_deltas(h):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        above = np.where(h > 0.5)[0]
        if len(above) >= n1 - target:
            idx = above[np.argsort(h[above])[:n1 - target]]
        else:
            idx = np.random.choice(above, n1 - target, replace=False)
        h[idx] = 0
    elif n1 < target:
        below = np.where(h < 0.5)[0]
        if len(below) >= target - n1:
            idx = below[np.argsort(-h[below])[:target - n1]]
        else:
            idx = np.random.choice(below, target - n1, replace=False)
        h[idx] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def do_flip(h, corr, s, deltas, p, q, N):
    h[p], h[q] = 0.0, 1.0
    corr += deltas[p, :].copy() - deltas[q, :].copy()
    s[p], s[q] = -s[p], -s[q]
    deltas[p, :] = 0.0
    deltas[p, p:p + N] = s
    deltas[q, :] = 0.0
    deltas[q, q:q + N] = s
    return corr, s, deltas

def targeted_sa(h_bin, deadline, n_peaks=6, n_per_peak=20, cool_rate=0.999, t_start=0.05, max_iters=150000):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    stag = 5000

    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.001)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p, cand_q = [], []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp, bq = vi[mask], (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = min(10, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        best_new, bi, bj = 1e10, 0, 0
        chunk = 8
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            corr, s, deltas = do_flip(h, corr, s, deltas, p, q, N)
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        if no_impr > stag:
            h[:] = best_h
            corr = fft_corr(h)
            s, deltas = build_deltas(h)
            for _ in range(4):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]

                    pr = int(np.random.choice(oi))
                    qr = int(np.random.choice(zi))
                    corr, s, deltas = do_flip(h, corr, s, deltas, pr, qr, N)
            no_impr = 0
            T = max(T * 1.5, t_start * 0.4)
    return best_h, best_c

def global_descent(h_bin, deadline):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(np.max(fft_corr(best_h if (best_h := h.copy()) else h)))
    best_h = h.copy()
    no_improve = 0
    L = 2 * N - 1
    while time.time() < deadline and no_improve < 5:
        corr = fft_corr(h)
        c_cur = float(corr.max())
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        s, deltas = build_deltas(h)
        dp_all = deltas[ones]
        dq_all = deltas[zeros]
        best_new, best_ii, best_jj = c_cur, -1, -1
        chunk_p = 15
        for ci in range(0, len(ones), chunk_p):
            ce = min(ci + chunk_p, len(ones))
            vals = corr[np.newaxis, np.newaxis, :] + dp_all[ci:ce, np.newaxis, :] - dq_all[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                best_ii, best_jj = mi[0] + ci, mi[1]
        if best_ii < 0 or best_new >= c_cur - 1e-12:
            no_improve += 1
            break
        p = int(ones[best_ii])
        q = int(zeros[best_jj])
        h[p], h[q] = 0.0, 1.0
        no_improve = 0
    best_c = float(np.max(fft_corr(best_h)))
    return best_h, best_c
