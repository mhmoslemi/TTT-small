def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum_binary(h, target):
    h = np.asarray(h, dtype=float).copy()
    n1 = int(round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0.0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1.0
    return h

def targeted_move(h, N, corr, cur_max):
    center = N - 1
    thresh = cur_max - 1.5
    above = np.where(corr >= thresh)[0]
    if len(above) == 0:
        above = np.array([center + int(np.random.randint(-N + 1, N))])
    pk = int(np.random.choice(above))
    L = pk - center
    if L == 0:
        return None
    if L > 0:
        cands = np.where((h[:N - L] == 1) & (h[L:] == 0))[0]
        if len(cands) == 0:
            return None
        i = int(np.random.choice(cands))
        return (i, i + L)
    else:
        M = -L
        cands = np.where((h[M:] == 1) & (h[:N - M] == 0))[0]
        if len(cands) == 0:
            return None
        j = int(np.random.choice(cands))
        return (j + M, j)

def sa_optimize(h, N, deadline, T0=0.05, p_target=0.45):

    h = fix_sum_binary(h, nh)

    best_h = h.copy()
    T = T0
    it = 0
    stall = 0

        it += 1

                oi = np.random.choice(len(ones), 2, replace=False)

                o1, o2 = int(ones[oi[0]]), int(ones[oi[1]])

def greedy_sweep(h, N, deadline):
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    while time.time() < deadline:
        improved = False
        ones_l = np.where(h == 1)[0]
        zeros_l = np.where(h == 0)[0]
        best_d = 0.0; best_oi = -1; best_zi = -1
        for oi in ones_l:
            for zi in zeros_l:
                if oi == zi:
                    continue
                h[oi] = 0.0; h[zi] = 1.0
                nm = float(np.max(fft_corr(h)))
                d = nm - cur_max
                if d < best_d:
                    best_d = d; best_oi, best_zi = oi, zi
                h[oi] = 1.0; h[zi] = 0.0
        if best_oi >= 0 and best_d < -1e-13:
            h[best_oi] = 0.0; h[best_zi] = 1.0
            corr = fft_corr(h); cur_max = float(np.max(corr))
            if cur_max * 2.0 / N < best_c - 1e-13:
                best_c = cur_max * 2.0 / N; best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c

def make_init(N, v):
    nh = N // 2
    x = np.arange(N)
    if v == 0:
        h = (x % 2).astype(float)
    elif v == 1:
        h = (np.random.rand(N) < 0.5).astype(float)

        h = (s > np.median(s)).astype(float)
    else:
        h = ((x * 11 + 3) % 17 < 8).astype(float)
    return fix_sum_binary(h, nh)
