def fix_sum_h(h, target):
    """Fix the sum of h to be exactly target by flipping bits."""
    s = int(np.sum(h))
    if s > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        for i in range(s - target):
            h[idx[i]] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        for i in range(target - s):
            h[idx[i]] = 1

def compute_C_all(h, N):
    """C[p, k] = change in full_correlation[k] when flipping h[p]."""
    k_vals = np.arange(2*N-1) - (N-1)
    p_idx = np.arange(N)

    i_mk = p_idx[:, None] - k_vals[None, :]

    valid_mk = (i_mk >= 0) & (i_mk < N)

    B = np.where(valid_mk, h[np.clip(i_mk, 0, N-1)], 0.0)
