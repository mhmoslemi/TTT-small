def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    """Convert to binary with exactly N/2 ones."""
    b = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(b.sum())
    if n1 > target:
        ones = np.where(b == 1)[0]
        idx = np.random.choice(ones, n1 - target, replace=False)
        b[idx] = 0
    elif n1 < target:
        zeros = np.where(b == 0)[0]
        idx = np.random.choice(zeros, target - n1, replace=False)
        b[idx] = 1
    return b

                h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
    return bh, bc

# FFT-based correlation
def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0-h), L)))

# Make binary with exact count
def make_bin(h, N, rng):
    b = (h > 0.5).astype(float)
    t = N // 2

    elif s < t:
