**Spectral approach:**
The correlation corr[k] = Σ_i h[i](1-h[i+k]) can be written as:
corr[k] = Σ_i h[i] - Σ_i h[i]h[i+k] = N/2 - R_h(k)

where R_h(k) is the autocorrelation of h at lag k.

The DFT of R_h is |DFT(h)|² (Wiener-Khinchin). So:
R_h(k) = (1/N) Σ_ω |H(ω)|² e^{2πiωk/N}

This means the autocorrelation is determined by the power spectrum of h. To make R_h(k) large for all k, we want |H(ω)|² to be concentrated at ω=0 (DC component) and have small variance.

The DC component: H(0) = Σ h[i] = N/2, so |H(0)|² = N²/4.
This gives R_h(0) = N/2 (which makes sense: Σ h[i]² = N/2 for 0-1 valued h).

For k≠0: R_h(k) = N/4 + (1/N) Σ_{ω≠0} |H(ω)|² e^{2πiωk/N}

So to maximize min_k R_h(k), we want the non-DC spectral components to be small AND to interfere destructively at all non-zero lags.

This is related to "low autocorrelation" sequence design. In the signaling/communications literature, this is the Barker problem (max length 13 for perfect Barker sequences).

For our problem (density 1/2, length 380), the optimal solution likely has a specific spectral structure. But designing it from scratch is hard.

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H * G, L)

def diff_swap(h, i, j, corr):
    """Compute new corr after h[i]:1->0, h[j]:0->1, given current corr."""
    N = len(h)
    # Change in corr due to removing i (1->0):
    # corr[L] -= (1-h[i+L]) if i+L valid; corr[L] += h[i-L] if i-L valid
    # Change in corr due to adding j (0->1):
    # corr[L] += (1-h[j+L]) if j+L valid; corr[L] -= h[j-L] if j-L valid
    # L ranges from -(N-1) to N-1, stored at index L+(N-1) in corr array
    
    new_corr = corr.copy()
    # Remove i
    lo_i = max(0, -i)  # L such that i+L >= 0 → L >= -i
    hi_i = min(N-1, N-1-i)  # L such that i+L < N → L < N-i
    for L in range(lo_i, hi_i + 1):
        idx = L + N - 1
        new_corr[idx] -= (1 - h[i+L])
    lo_i2 = max(-(N-1), i-(N-1))  # L such that i-L >= 0 → L <= i
    hi_i2 = min(N-1, i)  # L such that i-L < N → L > i-N
    for L in range(lo_i2, hi_i2 + 1):
        idx = L + N - 1
        new_corr[idx] += h[i-L]
    # Add j
    lo_j = max(0, -j)
    hi_j = min(N-1, N-1-j)
    for L in range(lo_j, hi_j + 1):
        idx = L + N - 1
        new_corr[idx] += (1 - h[j+L])
    lo_j2 = max(-(N-1), j-(N-1))
    hi_j2 = min(N-1, j)
    for L in range(lo_j2, hi_j2 + 1):
        idx = L + N - 1
        new_corr[idx] -= h[j-L]
    return new_corr

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def c5_of(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * 2.0 / N

def swap_delta(h, i, j):
    N = len(h)
    Ls = np.arange(-(N - 1), N)
    d = np.zeros(2 * N - 1)
    m1 = (i + Ls >= 0) & (i + Ls < N)
    d[m1] -= (1.0 - h[i + Ls[m1]])
    m2 = (i - Ls >= 0) & (i - Ls < N)
    d[m2] += h[i - Ls[m2]]
    m3 = (j + Ls >= 0) & (j + Ls < N)
    d[m3] += (1.0 - h[j + Ls[m3]])
    m4 = (j - Ls >= 0) & (j - Ls < N)
    d[m4] -= h[j - Ls[m4]]
    d[N - 1] -= 2.0
    L1 = j - i
    if 0 < abs(L1) <= N - 1:
        d[L1 + N - 1] += 1.0
        d[-L1 + N - 1] += 1.0
    return d

def fix_sum_bin(h, nh):
    diff = int(h.sum()) - nh
    if diff > 0:
        oidx = np.where(h == 1)[0]
        np.random.shuffle(oidx)
        h[oidx[:diff]] = 0
    elif diff < 0:
        zidx = np.where(h == 0)[0]
        np.random.shuffle(zidx)
        h[zidx[:(-diff)]] = 1
    return h

def local_search(h, N, deadline, sample_sz=120, max_stall=300, T_init=0.0, cool=0.9997):
    h = h.astype(float).copy()
    fix_sum_bin(h, N // 2)
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    T = T_init
    while time.time() < deadline:
        if stall > max_stall and T < 1e-7:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        oi = np.random.choice(ones, ns, replace=False)
        zi = np.random.choice(zeros, nz, replace=False)
        best_delta = -best_c
        best_a, best_b = -1, -1
        for b in range(nz):
            temp_c = corr.copy()
            j = zi[b]
            Ls = np.arange(-(N - 1), N)
            m3 = (j + Ls >= 0) & (j + Ls < N)
            temp_c[m3] += (1.0 - h[j + Ls[m3]])
            m4 = (j - Ls >= 0) & (j - Ls < N)
            temp_c[m4] -= h[j - Ls[m4]]
            base_max = float(np.max(temp_c))
            if base_max < best_c - 0.05:
                for a in range(ns):
                    i = oi[a]
                    d = np.zeros(2 * N - 1)
                    m1 = (i + Ls >= 0) & (i + Ls < N)
                    d[m1] -= (1.0 - h[i + Ls[m1]])
                    m2 = (i - Ls >= 0) & (i - Ls < N)
                    d[m2] += h[i - Ls[m2]]
                    d[N - 1] -= 2.0
                    L1 = j - i
                    if 0 < abs(L1) <= N - 1:
                        d[L1 + N - 1] += 1.0
                        d[-L1 + N - 1] += 1.0
                    nm = float(np.max(temp_c + d))
                    delta = best_c - nm
                    if delta > best_delta:
                        best_delta = delta
                        best_a, best_b = a, b
        if best_a >= 0 and best_delta > 1e-12:
            i, j = oi[best_a], zi[best_b]
            h[i] = 0.0
            h[j] = 1.0
            best_c -= best_delta
            best_h = h.copy()
            stall = 0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            best_c = min(best_c, float(np.max(corr)))
        else:
            stall += 1
            if T_init > 0:
                T *= cool
                if T > 1e-7:
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
                    d = swap_delta(h, i, j)
                    nm = float(np.max(corr + d))
                    if nm < best_c or np.random.random() < np.exp((best_c - nm) / T):

                        h[j] = 1.0

                        zeros = np.where(h == 0)[0]
                        corr = fft_corr(h)
                        cm = float(np.max(corr))
                        if cm < best_c:
                            best_c = cm
                            best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=80, lr0=0.012, beta0=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) > 1e-6:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) / 2
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2, 0, 1)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= k
        h_new = np.clip(y - lr * grad, 0, 1)
        ts = h_new.sum()
        if abs(ts - target) > 1e-6:
            lo2, hi2 = -3.0, 3.0
            for _ in range(40):
                mid = (lo2 + hi2) / 2
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo2 = mid
                else:
                    hi2 = mid
            h_new = np.clip(h_new - (lo2 + hi2) / 2, 0, 1)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.994
            beta = min(0.97, beta0 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
