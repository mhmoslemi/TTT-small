import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum_box(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def softmax_opt(h0, N, deadline, tau0=60.0, tau_end=300.0, lr0=0.015, beta=0.85, top_k_frac=1.0):

    target = N / 2.0
    m = np.zeros(N)
    lh = h.copy()  # for Nesterov y
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    n_iter = 0
    n_corr = 2 * N - 1
    while time.time() < deadline:
        n_iter += 1
        tau = tau0 + (tau_end - tau0) * min(1.0, n_iter / 1200)
        D = compute_D(h, N)
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        # Compute weights (top-k softmax)
        if top_k_frac < 1.0:
            k_sel = max(10, int(n_corr * top_k_frac))
            top_idx = np.argpartition(corr, -k_sel)[-k_sel:]
            w = np.zeros(n_corr)
            ct = corr[top_idx] - corr[top_idx].max()
            w[top_idx] = np.exp(tau * ct)
        else:
            ct = corr - corr.max()
            w = np.exp(tau * ct)
        w_sum = w.sum()
        if w_sum < 1e-30:

        # Nesterov

        m = h_new - h
        h = h_new
        if n_iter % 300 == 0:
            lr0 *= 0.93
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum_box(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def softmax_opt(h0, N, deadline, tau0=50.0, tau_end=300.0, lr0=0.018, beta=0.85):
    h = h0.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    n_iter = 0
    n_corr = 2 * N - 1
    while time.time() < deadline:
        n_iter += 1
        tau = tau0 + (tau_end - tau0) * min(1.0, n_iter / 1000.0)
        D = compute_D(h, N)
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        ct = corr - corr.max()
        w = np.exp(tau * ct)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        grad = D.T @ w
        y = h + beta * m
        h_new = project_sum_box(y + lr0 * grad, target)
        m = h_new - h
        h = h_new
        if n_iter % 250 == 0:
            lr0 *= 0.92
            beta = min(0.95, 0.80 + n_iter * 0.0002)
    return best_h, best_pk * (2.0 / N)

def ensure_binary(h, N):
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - target, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, target - n1, replace=False)] = 1
    return h_bin

def binary_ls(h_bin, N, deadline, max_sweeps=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(7)
    h = h_bin.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        best_new_P = P_cur
        best_pq = None
        best_2p = None
        n_sample = min(len(ones), max(30, int(len(ones) * 0.5)))
        p_list = rng.choice(ones, n_sample, replace=False) if n_sample < len(ones) else ones
        for p in p_list:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-13:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
                best_2p = None
        n_try = min(50, max(15, N // 8))
        for _ in range(n_try):
            i1, i2 = rng.randint(0, len(ones), 2)
            j1, j2 = rng.randint(0, len(zeros), 2)
            if i1 == i2 or j1 == j2:
                continue
            p1, p2 = int(ones[i1]), int(ones[i2])
            q1, q2 = int(zeros[j1]), int(zeros[j2])
            new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
            if new_P < best_new_P - 1e-13:
                best_new_P = new_P
                best_pq = None
                best_2p = (p1, p2, q1, q2)
        if best_2p is not None:
            p1, p2, q1, q2 = best_2p
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
        elif best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            stall += 1
            if stall > 35:
                for _ in range(5):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 15:
                break
            else:
                continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-13:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
    return best_h, best_c * (2.0 / N)

def upsample_fn(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Softmax Nesterov at N0
    d1 = min(deadline - 550, t0 + budget_s * 0.25)
    h1, c5_1 = softmax_opt(h0, N0, d1, tau0=50.0, tau_end=250.0, lr0=0.02, beta=0.85)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Binary local search
    if time.time() < deadline - 400:
        h_bin = ensure_binary(best_h, N0)
        d2 = min(deadline - 350, time.time() + budget_s * 0.16)
        h2, c5_2 = binary_ls(h_bin, N0, d2, max_sweeps=600, rng=rng)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Sharper softmax refinement
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + budget_s * 0.13)
        h3, c5_3 = softmax_opt(best_h, N0, d3, tau0=120.0, tau_end=500.0, lr0=0.01, beta=0.9)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upscale to 2x, optimize
    N1 = 2 * N0
    if time.time() < deadline - 170:
        h_up = project_sum_box(upsample_fn(best_h, N0, N1), N1 / 2.0)
        d4 = min(deadline - 130, time.time() + budget_s * 0.12)
        h4, c5_4 = softmax_opt(h_up, N1, d4, tau0=80.0, tau_end=350.0, lr0=0.012, beta=0.85)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Binary polish at current resolution
    N_fin = len(best_h)
    if time.time() < deadline - 60:
        hF = ensure_binary(best_h, N_fin)
        d5 = min(deadline - 30, time.time() + budget_s * 0.10)
        hF2, c5F = binary_ls(hF, N_fin, d5, max_sweeps=400, rng=rng)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Phase 6: Final targeted softmax with very sharp max
    if time.time() < deadline - 5:
        N_f = len(best_h)
        d6 = deadline - 2
        h6, c5_6 = softmax_opt(best_h, N_f, d6, tau0=300.0, tau_end=800.0, lr0=0.006, beta=0.93)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
