def minimax_opt(h, N, deadline, lr0=0.008, lambda_w=0.5):
    """Exponential weighting minimax optimization."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    
    # Initialize weights uniformly
    shifts = np.arange(-(N-1), N)
    w = np.ones(2*N-1) / (2*N-1)
    
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        
        # Update weights
        w_new = np.exp(lambda_w * corr / (np.max(corr) + 1e-8))
        w_new /= w_new.sum()
        w = 0.5 * w + 0.5 * w_new
        
        # Compute weighted gradient
        grad = np.zeros(N)
        # Top shifts (where w is large)
        top_idx = np.where(w > w.max() * 0.1)[0]
        # Limit to reasonable number
        if len(top_idx) > 50:
            top_idx = top_idx[np.argsort(w[top_idx])[::-1][:50]]
        
        center = N - 1
        for ai in top_idx:
            L = int(ai - center)
            wt = w[ai]
            if L == 0:
                grad += wt * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wt * (1.0 - y[:N-L])
                grad[:N-L] += -wt * y[L:]
            else:
                M = -L
                grad[M:] += wt * (1.0 - y[:N-M])
                grad[:N-M] += -wt * y[M:]
        
        # Nesterov step with projection
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        
        if it % 500 == 0:
            lr *= 0.97
            beta = min(0.97, 0.85 + it * 0.0002)
    
    return best_h, best_pk * dx

    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    """Compute D[i, k] = change in corr[k] when h[i] is toggled."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    # D[i, k] for shift k:
    # For s > 0: (1-h[i-s])*[i>=s] - h[i+s]*[i+s<N]
    # For s < 0 (m=-s): (1-h[i-m])*[i>=m] - h[i+m]*[i+m<N]
    # For s = 0: 1 - 2*h[i]
    D = np.zeros((N, 2*N-1))
    center = N - 1
    h_i = h[:, None]  # (N, 1)
    for s in range(1, N):

        val_right = 1.0 - h[:N-s]  # (N-s,) for i in [s, N-1]
        val_left = -h[s:]  # (N-s,) for i in [0, N-1-s]
        D[s:, center+s] += val_right
        D[:N-s, center+s] += val_left
        # shift -s: same formula
        D[s:, center-s] += val_right
        D[:N-s, center-s] += val_left
