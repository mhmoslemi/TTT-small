import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(999 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    # D[p, k] = change in correlation at shift k if we flip h[p] from 1 to 0
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def exhaustive_local(h, N, deadline, max_sweeps=500):
    """Try all 1-swaps, pick the one that most reduces max correlation."""
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()

            break

        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:

        D = compute_D(h, N)

def nesterov_continuous(h, N, deadline, k_top=30, lr0=0.015, beta0=0.9):
    target = N / 2.0
    dx = 2.0 / N
    h = h.astype(float).copy()
    m = np.zeros(N)
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    beta = beta0
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        # Subgradient: focus on top-k correlations
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        top_max = corr[top_idx].max()
        # Weight: emphasize correlations near the max
        weights = np.zeros(2*N-1)
        weights[top_idx] = 1.0
        # Gradient computation
        center = N - 1
        grad = np.zeros(N)
        for ai in top_idx:
            L = int(ai - center)
            w = weights[ai]
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                # c[L] = sum_{i=0}^{N-L-1} h[i](1-y[i+L])

                    w_neg = np.zeros(N)
                    w_neg[L:] = w * y[:N-L]
                    grad -= w_neg
            else:

                if M < N:

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def local_search(h, N, deadline, sf=1.0, max_sweeps=400):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(10, int(len(ones) * sf))
        plist = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P
        best_pq = None
        for p in plist:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-14:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def sa_binary(h, N, deadline, T0=5.0, Tcool=0.993, Tmin=0.02, sf=1.0, kick=100, ksz=5):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick:
            for _ in range(ksz):
                h[ones[np.random.randint(len(ones))]] = 0.0
                h[zeros[np.random.randint(len(zeros))]] = 1.0
            stall = 0
            T = T0 * 0.7
            c = fft_corr(h)
            P = float(c.max())
            T = max(Tmin, T * Tcool)
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(10, int(len(ones) * sf))
        plist = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in plist:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        accept = delta < -1e-14 or (T > Tmin and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))
        if accept:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            h[ones[np.random.randint(len(ones))]] = 0.0
            h[zeros[np.random.randint(len(zeros))]] = 1.0
        T = max(Tmin, T * Tcool)
    return best_h, best_c * (2.0 / N)

def nesterov_cont(h, N, deadline, ktop=40, lr0=0.018, beta0=0.92, sT=0.4):
    target = N / 2.0
    dx = 2.0 / N
    h = h.astype(float).copy()
    m = np.zeros(N)
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    beta = beta0
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top_k = np.argpartition(corr, -ktop)[-ktop:]
        c_top = corr[top_k]
        wts = np.exp((c_top - c_top.max()) / max(sT, 0.01))
        wts /= wts.sum()
        center = N - 1
        grad = np.zeros(N)
        for idx in range(ktop):
            L = int(top_k[idx] - center)
            w = wts[idx]
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif 0 < L < N:
                a = np.zeros(N)
                a[:N - L] = 1.0 - y[L:]
                b = np.zeros(N)
                b[L:] = y[:N - L]
                grad += w * (a - b)
            elif -N < L < 0:
                M = -L
                a = np.zeros(N)
                a[M:] = 1.0 - y[:N - M]
                b = np.zeros(N)
                b[:N - M] = y[M:]
                grad += w * (a - b)
        grad /= ktop
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.992
            beta = min(0.98, beta0 + it * 0.00004)
    return best_h, best_pk * dx

def perturb_binary(h, n):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n):
        h[ones[np.random.randint(len(ones))]] = 0.0
        h[zeros[np.random.randint(len(zeros))]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    N1 = 2 * N0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick local search at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 700, t0 + 25)
    h1, c5_1 = local_search(h_bin, N0, d1, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: SA at N0 with 2 restarts
    d2a = min(deadline - 580, t0 + 130)
    h2a, c5_2a = sa_binary(h_bin, N0, d2a, T0=6.0, Tcool=0.992, Tmin=0.015, kick=80, ksz=6)
    if c5_2a < best_c5:
        best_c5, best_h = c5_2a, h2a.copy()
    h2b = perturb_binary(ensure_sum(best_h, N0), 15)
    d2b = min(deadline - 500, t0 + 220)
    h2b_out, c5_2b = sa_binary(h2b, N0, d2b, T0=5.0, Tcool=0.993, Tmin=0.02, kick=100, ksz=5)
    if c5_2b < best_c5:
        best_c5, best_h = c5_2b, h2b_out.copy()

    # Phase 3: Upsample to N1, local search
    if time.time() < deadline - 420:
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d3 = min(deadline - 380, t0 + 270)
        h3, c5_3 = local_search(h_up, N1, d3, sf=0.4)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: SA at N1
    if time.time() < deadline - 340:
        h_up2 = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d4 = min(deadline - 290, t0 + 370)
        h4, c5_4 = sa_binary(h_up2, N1, d4, T0=4.5, Tcool=0.994, Tmin=0.03, sf=0.35, kick=150, ksz=4)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Nesterov at N0
    if time.time() < deadline - 200:
        hc = upsample(best_h, len(best_h), N0)
        hc = project_sum(hc + np.random.randn(N0) * 0.06, N0 / 2.0)
        d5 = min(deadline - 170, t0 + 470)
        h5, c5_5 = nesterov_cont(hc, N0, d5, ktop=45, lr0=0.022, beta0=0.91, sT=0.35)
        if c5_5 < best_c5:
            best_c5, best_h = h5.copy(), c5_5

    # Phase 6: Nesterov at N1
    if time.time() < deadline - 110:
        hc2 = upsample(best_h, len(best_h), N1)
        hc2 = project_sum(hc2 + np.random.randn(N1) * 0.035, N1 / 2.0)
        d6 = min(deadline - 80, t0 + 570)
        h6, c5_6 = nesterov_cont(hc2, N1, d6, ktop=65, lr0=0.014, beta0=0.93, sT=0.25)
        if c5_6 < best_c5:
            best_c5, best_h = h6.copy(), c5_6

    # Phase 7: SA restart at N0 with strong perturbation
    if time.time() < deadline - 50:
        h_sa = perturb_binary(ensure_sum(upsample(best_h, len(best_h), N0), N0), 25)
        d7 = min(deadline - 25, t0 + 680)
        h7, c5_7 = sa_binary(h_sa, N0, d7, T0=8.0, Tcool=0.991, Tmin=0.01, sf=1.0, kick=60, ksz=8)
        if c5_7 < best_c5:
            best_c5, best_h = h7.copy(), c5_7

    # Phase 8: Final Nesterov polish
    if time.time() < deadline - 2:
        Nf = len(best_h)
        hf = project_sum(upsample(best_h, Nf, Nf) + np.random.randn(Nf) * 0.015, Nf / 2.0)
        d8 = deadline - 1.0
        h8, c5_8 = nesterov_cont(hf, Nf, d8, ktop=55, lr0=0.01, beta0=0.94, sT=0.15)
        if c5_8 < best_c5:
            best_c5, best_h = h8.copy(), c5_8

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
