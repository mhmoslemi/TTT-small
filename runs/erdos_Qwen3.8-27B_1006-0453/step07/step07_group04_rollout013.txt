def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:L]

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0/len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def compute_D(h, N):
    # Correct D: change in corr[j] when h[p] goes 1→0

def nesterov_opt(h, N, deadline, k_start=20, k_max=200, lr0=0.03):
    h = h.astype(float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk, best_h = pk, h.copy()
        k = min(k_max, k_start + it//20)
        top = np.argpartition(corr, -k)[-k:]
        # Subgradient: weighted by proximity to max
        weights = corr[top] - corr.max() + 1e-6  # small for top, larger for rest
        weights /= weights.sum()
        # Actually use uniform weights over top-k for simplicity
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if abs(L) < N:
                if L == 0:
                    continue
                elif L > 0:
                    grad[L:] += (1.0 - y[:N-L])
                    grad[:N-L] -= y[L:]
                else:
                    M = -L
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= float(k)
        h_new = project_sum(y - lr*grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
            beta = min(0.97, 0.87 + it*0.00005)
    return best_h, best_pk * dx

    return np.interp(x_new, x_old, h_old)

Alternative: Instead of computing the full D matrix, I can compute the change in the max correlation more cheaply. The max is at some lag L*. The change in corr[L*] for a swap (p→0, q→1) is:
Δcorr[L*] = D[p, center+L*] - D[q, center+L*]

This is just two lookups! But the problem is that after the swap, the max might shift to a different lag. So we need to check all lags.

However, a good approximation: if the current max is at lag L*, and we're looking for swaps that reduce the max, we primarily care about lags that are close to the current max. We can use the "top-k lags" approach: only check the top k lags (say k=50) instead of all 2N-1 lags.

This reduces the computation from (n_p * n_zeros * 2N) to (n_p * n_zeros * k) which is much faster.

Let me implement this:

```python
def fast_greedy_step(c, h, N, k_lags=50):
    # Only consider top-k lags
    top_idx = np.argpartition(c, -k_lags)[-k_lags:]
    D = compute_D(h, N)  # still need full D, but only use top_idx columns
    D_top = D[:, top_idx]  # N x k_lags
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    c_top = c[top_idx]
    P = float(c.max())
    best_new_P = P
    best_pq = None
    for p in ones:
        Dz_top = D_top[zeros, :]  # n_zeros x k_lags
        cand = (c_top + D_top[p, :])[None, :] - Dz_top  # n_zeros x k_lags
        npk = cand.max(axis=1)  # n_zeros
        mi = int(np.argmin(npk))
        if npk[mi] < best_new_P:
            best_new_P = float(npk[mi])
            best_pq = (p, zeros[mi])
    return best_pq, best_new_P
```

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:L]

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0/len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    D = h[pm_c].astype(float) * vm - (1.0 - h[pp_c].astype(float)) * vp
    D[:, N-1] = 0.0
    return D

def ensure_sum(h, N):
    nh = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42+N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1-nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh-n1, replace=False)] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def fast_greedy(h, N, deadline, k_lags=50, max_sweeps=2000, sf=1.0):

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c, best_h, stall = P, h.copy(), 0
        else:
            stall += 1
            if stall > 80:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        top_idx = np.argpartition(c, -k_lags)[-k_lags:]
        c_top = c[top_idx]
        D_top = D[:, top_idx]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones)*sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P
        best_pq = None
        for p in p_list:
            cand = (c_top + D_top[p, :])[None, :] - D_top[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-14:

                best_pq = (p, zeros[mi])
        if best_pq is None:

        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0/N)

def fast_sa(h, N, deadline, T0=3.0, Tcool=0.995, T_min=0.03, k_lags=50, sf=1.0, kick_every=100, kick_size=4):

def nesterov_opt(h, N, deadline, k_start=15, k_end=150, lr0=0.025):
    h = h.astype(float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk, best_h = pk, h.copy()
        k = int(k_start + (k_end - k_start) * min(1.0, it / 500.0))
        top = np.argpartition(corr, -k)[-k:] if k < len(corr) - 1 else np.arange(len(corr))
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0 or abs(L) >= N:
                continue
            if L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= float(max(1, k))
        h_new = project_sum(y - lr*grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            lr *= 0.97
            beta = min(0.97, 0.88 + it*0.00004)
    return best_h, best_pk * dx
