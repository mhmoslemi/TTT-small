```python
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T

                g[M:] -= h[:N-M]
    s = h.sum() - target
```

```python
def smooth_obj_grad(h, T, target, lam):
    N = len(h)

    e = np.exp(T*(c-mx))
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            L = i - ctr
            if L == 0:
                g[:] += 1.0 - 2.0 * h
            elif L > 0:
                g[L:] += 1.0 - h[:N-L]
                g[:N-L] -= h[L:]
            else:
                M = -L
                g[:N-M] += 1.0 - h[M:]
                g[M:] -= h[:N-M]
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_run(h, N, deadline, T=30.0, max_outer=3):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_grad(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0,1)]*N,
                     options={'maxiter':100, 'maxfun':300, 'ftol':1e-14})
        h = proj01(res.x, target)
        v = smooth_obj_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.5
    return best_h, best_v * (2.0 / N)

def sa_local_search(h, N, deadline, max_iters=10000, target_frac=0.4, T0=2.0):
    h = h.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    center = N - 1
    T = T0
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 800:
            break
        if T > 0.005:
            T *= 0.999
        corr = fft_corr(h)
        P = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Choose move
        use_target = np.random.rand() < target_frac
        if use_target:
            idx_w = int(np.argmax(corr))
            L = idx_w - center
            i = int(ones[np.random.randint(len(ones))])
            j = i + L
            if not (0 <= j < N) or h[j] != 0 or j == i:
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
        else:
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
        if i == j or h[i] != 1 or h[j] != 0:
            stall += 1
            continue
        h[i], h[j] = 0, 1
        nP = float(np.max(fft_corr(h)))
        dE = nP - P
        if dE < 0 or (T > 0.01 and np.random.rand() < np.exp(-dE / T)):
            if nP < best_c - 1e-14:
                best_c = nP
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1, 0
            stall += 1
    # Final greedy pass
    best_c2 = float(np.max(fft_corr(best_h)))
    if best_c2 > best_c:
        best_c = best_c2
    return best_h, best_c
```

```python
            if L == 0:
                g[:] += 1.0 - 2.0 * h
            elif L > 0:
                g[L:] += 1.0 - h[:N - L]
                g[:N - L] -= h[L:]
            else:
                M = -L
                g[:N - M] += 1.0 - h[M:]
                g[M:] -= h[:N - M]
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_run(h, N, deadline, T=30.0, max_outer=3):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_grad(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 100, 'maxfun': 300, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.5
    return best_h, best_v * (2.0 / N)

def sa_search(h, N, deadline, max_iters, target_frac, T0):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    center = N - 1
    T = T0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 1000:
            break
        if T > 0.005:
            T *= 0.999
        corr = fft_corr(h)
        P = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if np.random.rand() < target_frac:
            idx_w = int(np.argmax(corr))
            L = idx_w - center
            i = int(ones[np.random.randint(len(ones))])
            j = i + L
            if not (0 <= j < N) or j == i or h[j] != 0:
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
        else:
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
        if i == j or h[i] != 1 or h[j] != 0:
            stall += 1
            continue
        h[i], h[j] = 0, 1
        nP = float(np.max(fft_corr(h)))
        dE = nP - P
        if dE < 0 or (T > 0.01 and np.random.rand() < np.exp(-dE / T)):
            if nP < best_c - 1e-14:
                best_c = nP
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1, 0
            stall += 1
    return best_h, best_c

def greedy_repair(h, N, nrounds=50):
    h = h.copy()
    center = N - 1
    for _ in range(nrounds):
        corr = fft_corr(h)
        P = float(np.max(corr))
        idx_w = int(np.argmax(corr))
        L = idx_w - center
        if L == 0:
            break
        ones = np.where(h == 1)[0]
        best_d, best_i, best_j = 0, -1, -1
        for i in ones:
            j = i + L
            if 0 <= j < N and h[j] == 0:
                h[i], h[j] = 0, 1
                nP = float(np.max(fft_corr(h)))
                h[i], h[j] = 1, 0
                d = P - nP
                if d > best_d:
                    best_d, best_i, best_j = d, int(i), int(j)
        if best_i < 0:
            break
        h[best_i], h[best_j] = 0, 1
    return h, float(np.max(fft_corr(h)))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 > 0.5).astype(float).copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Intensive SA on initial
    d1 = min(dl - 250, t0 + 130)
    hb, cb = sa_search(best_h, N0, d1, 20000, 0.35, 2.0)
    if cb < best_c5:
        best_c5, best_h = cb, hb
    hg, cg = greedy_repair(best_h, N0, nrounds=80)
    if cg < best_c5:
        best_c5, best_h = cg, hg

    # Phase 2: Perturbation restarts with SA
    if time.time() < dl - 200:
        d2 = min(dl - 150, time.time() + 160)
        nswaps_list = [12, 25, 6, 35, 18, 8]
        for r in range(len(nswaps_list)):
            if time.time() > d2:
                break
            hp = best_h.copy()
            ones = np.where(hp == 1)[0]
            zeros = np.where(hp == 0)[0]
            ns = min(nswaps_list[r], len(ones), len(zeros))
            si = np.random.choice(ones, ns, replace=False)
            sj = np.random.choice(zeros, ns, replace=False)
            hp[si], hp[sj] = 0, 1
            h2, c2 = sa_search(hp, N0, min(d2, time.time() + 35), 10000, 0.5, 3.5)
            h2g, c2g = greedy_repair(h2, N0, nrounds=40)
            c2f = min(c2, c2g)
            if c2g < c2:
                h2 = h2g
            if c2f < best_c5:
                best_c5, best_h = c2f, h2

    # Phase 3: L-BFGS continuous optimization
    if time.time() < dl - 100:
        d3 = min(dl - 60, time.time() + 90)
        for r in range(3):
            if time.time() > d3:
                break
            hr = best_h + np.random.randn(N0) * (0.12 - 0.03 * r)
            hr = proj01(hr, float(N0) / 2.0)
            hl, cl = lbfgs_run(hr, N0, min(d3, time.time() + 35), T=25.0, max_outer=3)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 4: Multi-resolution search
    if time.time() < dl - 50:
        d4 = min(dl - 25, time.time() + 130)
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        for Nt in [320, 360, 400, 440, 480, 520, 560]:
            if time.time() > d4:
                break
            if Nt % 2 or Nt == len(best_h):
                continue
            xn = np.linspace(0, 2, Nt, endpoint=False)
            hi = np.interp(xn, x_b, best_h)
            ht = (hi > 0.5).astype(float)
            nh_t = Nt // 2
            n1t = int(ht.sum())
            if n1t > nh_t:
                ht[np.random.choice(np.where(ht == 1)[0], n1t - nh_t, replace=False)] = 0
            elif n1t < nh_t:
                ht[np.random.choice(np.where(ht == 0)[0], nh_t - n1t, replace=False)] = 1
            hr, cr = sa_search(ht, Nt, min(d4, time.time() + 25), 8000, 0.4, 2.0)
            hrg, crg = greedy_repair(hr, Nt, nrounds=30)
            c5r = min(compute_c5(hr, Nt), compute_c5(hrg, Nt))
            if compute_c5(hrg, Nt) < c5r:
                hr = hrg
            if c5r < best_c5:
                best_c5, best_h = c5r, hr.copy()
                x_b = np.linspace(0, 2, Nt, endpoint=False)

    # Phase 5: Final intensive SA + greedy
    if time.time() < dl - 8:
        d5 = min(dl - 3, time.time() + 70)
        Nf = len(best_h)
        hf, cf = sa_search(best_h, Nf, d5, 25000, 0.6, 1.0)
        hf2, cf2 = greedy_repair(hf, Nf, nrounds=100)
        cf2 = compute_c5(hf2, Nf)
        cf = compute_c5(hf, Nf)
        if cf2 < cf:
            hf, cf = hf2, cf2
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # Final: continuous vs binary
    fc5 = compute_c5(best_h, len(best_h))
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, len(best_h))
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, len(best_h)
```
