```python
def compute_swap_delta(h, oi, zi, N):
    """Compute change in full correlation when h[oi]: 1→0 and h[zi]: 0→1."""
    L = 2 * N - 1
    dc = np.zeros(L)
    # For each lag k (0 to L-1), shift m = k - (N-1)
    # Δcorr[k] = -(1-h[oi-m])*1_{0<=oi-m<N} + h[oi+m]*1_{0<=oi+m<N}
    #            + (1-h[zi-m])*1_{0<=zi-m<N} - h[zi+m]*1_{0<=zi+m<N}
    
    # This is O(N) - for each k, constant work
    # But we can vectorize:
    m = np.arange(L) - (N - 1)  # shifts, from -(N-1) to N-1
    
    # Term 1: -(1-h[oi-m]) where 0 <= oi-m < N
    idx1_oi = oi - m
    valid1_oi = (idx1_oi >= 0) & (idx1_oi < N)
    dc[valid1_oi] -= (1.0 - h[idx1_oi[valid1_oi]])
    
    # Term 2: +h[oi+m] where 0 <= oi+m < N
    idx2_oi = oi + m
    valid2_oi = (idx2_oi >= 0) & (idx2_oi < N)
    dc[valid2_oi] += h[idx2_oi[valid2_oi]]
    
    # Term 3: +(1-h[zi-m]) where 0 <= zi-m < N
    idx1_zi = zi - m
    valid1_zi = (idx1_zi >= 0) & (idx1_zi < N)
    dc[valid1_zi] += (1.0 - h[idx1_zi[valid1_zi]])
    
    # Term 4: -h[zi+m] where 0 <= zi+m < N
    idx2_zi = zi + m
    valid2_zi = (idx2_zi >= 0) & (idx2_zi < N)
    dc[valid2_zi] -= h[idx2_zi[valid2_zi]]
    
    return dc
```

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5

        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def softmax_gradient(h, corr, beta):
    N = len(h)
    L = 2 * N - 1
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N)
    w /= ws
    grad = np.zeros(N)

        if wk < 1e-12:

        m = k - (N - 1)
        if m >= 0:
            grad[m:N] += wk * (1.0 - h[0:N-m])
            grad[0:N-m] -= wk * h[m:N]
        else:
            mp = -m
            grad[0:N-mp] += wk * (1.0 - h[mp:N])
            grad[mp:N] -= wk * h[0:N-mp]
    return grad

    if time.time() < deadline - 70:
        N_try = 480
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        hb3 = fix_binary(h_int)
        d5 = min(deadline - 50, time.time() + 40)
        hb3, c5 = binary_sa(hb3, d5, T0=0.3, T_end=0.005, max_iters=200000)
        if c5 < best_c:
            best_c, best_h = c5, hb3

    # Phase 4: Final continuous polish
    if time.time() < deadline - 15:
        d6 = min(deadline - 3, time.time() + 30)
        h6, c6 = cont_opt(best_h.copy(), d6, beta0=15, beta1=100, lr0=0.004)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Final binary check
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB

    return best_h, final_c, len(best_h)
