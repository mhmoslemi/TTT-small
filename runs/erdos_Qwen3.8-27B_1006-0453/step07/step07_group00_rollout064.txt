import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

    return float(np.max(fft_corr(h))) * (2.0 / N)

    h = np.round(h).astype(float)
    h = np.clip(h, 0, 1)
    n1 = int(h.sum())
    target = N // 2
    if n1 > target:

        np.random.shuffle(ones)
        h[ones[:n1-target]] = 0
    elif n1 < target:

        np.random.shuffle(zeros)
        h[zeros[:target-n1]] = 1
    return h

def sa_search(h_bin, deadline, n_peaks=8, n_per_peak=20, cool_rate=0.9993, t_start=0.04):
    N = len(h_bin)
    h = h_bin.astype(float).copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i+N] = s
    T = t_start
    no_impr = 0
    it = 0
    
    while time.time() < deadline:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.003)
            
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
            
        # Find top peaks and their "bodies"
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp = vi[mask]
                bq = (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp = vi[mask]
                bq = (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel])
                cand_q.extend(bq[sel])
        
        # Add some random exploration
        nr = min(8, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False))
        cand_q.extend(np.random.choice(zeros, nr, replace=False))
        
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        
        # Evaluate all (p, q) pairs
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        
        # Chunked evaluation
        best_new = 1e10
        bi, bj = 0, 0
        chunk = 4
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        
        if dc < -1e-12 or (T > 0.003 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p+N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q+N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        
        if no_impr > 8000:

            h = best_h.copy() + np.random.randn(N) * 0.02
            h = make_binary(h, N)
            corr = fft_corr(h)

            deltas = np.zeros((N, 2 * N - 1))
            for i in range(N):
                deltas[i, i:i+N] = s
            T = max(T, 0.02)
            no_impr = 0
    
    return best_h, best_c

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = np.round(h).astype(float)
    h = np.clip(h, 0, 1)
    n1 = int(h.sum())
    target = N // 2
    if n1 > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:n1 - target]] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - n1]] = 1
    return h

def project_box_sum(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
    # Phase 5: Greedy hill climb on best binary
    if time.time() < deadline - 120:
        h5b = make_binary(best_h, len(best_h))
        d5 = min(deadline - 80, time.time() + 100)
        h5, _ = greedy_climb(h5b, d5)
        c5 = compute_c5(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()
    # Phase 6: Final continuous polish
    if time.time() < deadline - 10:
        Ny = len(best_h)
        h6 = best_h.astype(float) + np.random.randn(Ny) * 0.015
        h6 = np.clip(h6, 0, 1)
        h6, c6 = cont_opt(h6, Ny, deadline - 2)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()
    # Phase 7: Tiny greedy polish
    if time.time() < deadline - 2:
        h7b = make_binary(best_h, len(best_h))
        h7, _ = greedy_climb(h7b, deadline - 1, n_peaks=20, n_per_peak=40)
        c7 = compute_c5(h7)
        if c7 < best_c5:
            best_c5, best_h = c7, h7
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
