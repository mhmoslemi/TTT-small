import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_binary(h, N, seed=42):
    h = h.astype(float).copy()
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, target - n1, replace=False)] = 1
    return h_bin

def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def sa_binary(h, N, deadline, T0=3.5, Tcool=0.9945, T_min=0.03, sf=1.0,
              kick_every=120, kick_sz=4, max_sweeps=8000, seed=0):

    h = h.astype(float).copy()

    best_h = h.copy()
    T = T0
    stall = 0

        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_sz):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))

    beta = beta0
    k = k0
    stall = 0
    it = 0
    while time.time() < deadline:

            best_h = h.copy()

        else:

        if stall > 200:
            lr *= 0.92
            k = min(k_max, k + 5)

    if time.time() < deadline - 80:
        h_n = upsample(best_h, N0)
        h_n = project_box_sum(h_n + np.random.randn(N0) * 0.05, N0 / 2.0)
        d7 = min(deadline - 40, t0 + 640)
        h7, c7 = nesterov_cont(h_n, N0, d7, k0=30, k_max=60, lr0=0.025, beta0=0.92)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Phase 8: Final greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 2:
        hF = ensure_binary(best_h, N_fin, seed=20)
        sf_fin = 1.0 if N_fin <= 800 else 0.6
        d8 = deadline - 1
        hF2, c8 = greedy_binary(hF, N_fin, d8, sf=sf_fin, max_sweeps=2000, seed=30)
        if c8 < best_c5:
            best_c5, best_h = c8, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
