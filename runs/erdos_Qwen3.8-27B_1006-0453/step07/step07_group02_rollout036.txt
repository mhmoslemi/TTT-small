def corr_delta_swap(h, i, j, N):
    """Compute delta in correlation for swapping h[i]:1->0, h[j]:0->1"""
    # h is 0/1 array, h[i]=1, h[j]=0
    # Returns array of length 2N-1 (deltas for each lag index)
    c = np.zeros(2*N-1)
    for k in range(-N+1, N):
        idx = k + N - 1
        delta = 0
        # Case a: n=i, valid if 0<=i+k<N
        if 0 <= i+k < N:
            # old: h[i+k]*0 = 0, new: h'[i+k]*1
            hp = h[i+k]
            if i+k == i: hp = 0
            elif i+k == j: hp = 1
            delta += hp
        # Case b: n=j, valid if 0<=j+k<N
        if 0 <= j+k < N:
            # old: h[j+k]*1, new: 0
            hj = h[j+k]  # h[j+k] in original
            delta -= hj
        # Case c: n=i-k, valid if 0<=i-k<N, i-k!=i, i-k!=j
        n_c = i - k
        if 0 <= n_c < N and n_c != i and n_c != j:
            delta -= (1 - h[n_c])
        # Case d: n=j-k, valid if 0<=j-k<N, j-k!=i, j-k!=j
        n_d = j - k
        if 0 <= n_d < N and n_d != i and n_d != j:
            delta += (1 - h[n_d])
        c[idx] = delta
    return c

def corr_delta(h, i, j, N):

    h must have h[i]=1, h[j]=0 before the swap."""
    k = np.arange(2*N-1) - (N-1)
    d = np.zeros(2*N-1)
    
    # Case a: n=i contributes h'[i+k]*(1-h'[i]) - h[i+k]*(1-h[i]) = h'[i+k] - 0

    if va.any():
        val = h[ik[va]].astype(float)
        # h'[ik] = 0 where ik==i, 1 where ik==j

        mask_j = (ik[va] == j)
        val[mask_i] = 0.0
