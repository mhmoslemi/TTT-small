import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_obj(h, N, target, lam, K=20):
    """Objective: average of top-K correlations + constraint penalty"""
    c = fft_corr(h)
    # Top-K average
    idx = np.argpartition(c, -K)[-K:]
    val = c[idx].mean()
    g = np.zeros(N)
    # Gradient of top-K avg is average of gradients of those K lags
    ctr = N - 1
    for i in idx:
        L = int(i) - ctr
        g += lag_grad(L, h, N)
    g /= K
    # Add constraint
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lag_grad(L, h, N):
    """Gradient of correlation at lag L with respect to h"""
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0 and L < N:
        # corr[L] = sum_{i=0}^{N-1-L} h[i] * (1-h[i+L])
        # d/dh[i] for i in [0, N-1-L]: (1-h[i+L])
        # d/dh[i] for i in [L, N-1]: -h[i-L]
        g[:N-L] += (1.0 - h[L:])
        g[L:] -= h[:N-L]
    elif L < 0 and abs(L) < N:
        M = -L
        g[M:] += (1.0 - h[:N-M])
        g[:N-M] -= h[M:]
    return g

def smooth_topk(h, N, deadline, T_init=20.0, max_outer=5, K=15):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lam = 1.0
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        # Adaptive K: start small, grow
        K_cur = max(5, K + 3 * outer)
        lam *= 2.0
        res = sp_min(lambda x: topk_obj(x, N, target, lam, K_cur),
                     h, jac=True, method='L-BFGS-B',
                     bounds=[(0, 1)] * N,
                     options={'maxiter': 80, 'maxfun': 200, 'ftol': 1e-15})
        h = proj01(res.x, target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def bin_search(h, N, deadline, max_iters=500, ntri=25, anneal=0.0):

    nh = N // 2

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()

        return h
    lo, hi = -2.0, 2.0

        if np.clip(h - mid, 0, 1).sum() > target:

        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lag_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < L < N:
        g[:N-L] += (1.0 - h[L:])
        g[L:] -= h[:N-L]
    elif -N < L < 0:
        M = -L
        g[M:] += (1.0 - h[:N-M])
        g[:N-M] -= h[M:]
    return g

def topk_obj_grad(h, N, target, lam, K):
    c = fft_corr(h)
    K = min(K, len(c))
    idx = np.argpartition(c, -K)[-K:]
    val = c[idx].mean()
    g = np.zeros(N)
    ctr = N - 1
    for i in idx:
        g += lag_grad(int(i) - ctr, h, N)
    g /= K
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lag_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < L < N:
        g[:N - L] += (1.0 - h[L:])
        g[L:] -= h[:N - L]
    elif -N < L < 0:
        M = -L
        g[M:] += (1.0 - h[:N - M])
        g[:N - M] -= h[M:]
    return g

def topk_obj_grad(h, N, target, lam, K):
    c = fft_corr(h)
    K = min(K, len(c))
    idx = np.argpartition(c, -K)[-K:]
    val = float(c[idx].mean())
    g = np.zeros(N)
    ctr = N - 1
    for i in idx:
        g += lag_grad(int(i) - ctr, h, N)
    g /= K
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def softmax_obj_grad(h, N, target, lam, T):
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = float(mx + np.log(e.sum()) / T)
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lag_grad(int(i) - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, N, deadline, mode, K, T, max_outer):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lam = 1.0
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        lam *= 2.5
        K_cur = min(K + 4 * outer, 50)
        if mode == 'topk':
            res = sp_min(topk_obj_grad, h, args=(N, target, lam, K_cur),
                         jac=True, method='L-BFGS-B',
                         bounds=[(0, 1)] * N,
                         options={'maxiter': 80, 'maxfun': 250, 'ftol': 1e-15})
        else:
            res = sp_min(softmax_obj_grad, h, args=(N, target, lam, T),
                         jac=True, method='L-BFGS-B',
                         bounds=[(0, 1)] * N,
                         options={'maxiter': 80, 'maxfun': 250, 'ftol': 1e-15})
        h = proj01(res.x, target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def bin_search(h, N, deadline, max_iters, ntri, anneal):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    best_c = compute_c5(h, N)
    best_h = h.copy()
    stall = 0
    temp = anneal
    P = best_c
    for it in range(max_iters):
        if time.time() > deadline or stall > 250:
            break
        improved = False
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = -np.inf, -1, -1
        for _ in range(ntri):
            i = np.random.randint(len(ones))
            j = np.random.randint(len(zeros))
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            c_new = float(np.max(fft_corr(h)))
            d = P - c_new
            if d > bd + temp:
                bd, bi, bj = d, oi, zi
            h[oi], h[zi] = 1.0, 0.0
        if bi >= 0 and (bd > 0 or temp > 0):
            h[bi], h[bj] = 0.0, 1.0
            P = float(np.max(fft_corr(h)))
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))
    return best_h, best_c

def bin_search2(h, N, deadline, max_iters):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    best_c = compute_c5(h, N)
    best_h = h.copy()
    stall = 0
    P = best_c
    for it in range(max_iters):
        if time.time() > deadline or stall > 120:
            break
        improved = False
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, ba, bb, bc, bd2 = -np.inf, -1, -1, -1, -1
        for _ in range(12):
            i1, i2 = np.random.choice(len(ones), 2, replace=False)
            j1, j2 = np.random.choice(len(zeros), 2, replace=False)
            a1, a2 = int(ones[i1]), int(ones[i2])
            b1, b2 = int(zeros[j1]), int(zeros[j2])
            h[a1], h[a2], h[b1], h[b2] = 0, 0, 1, 1
            c_new = float(np.max(fft_corr(h)))
            d = P - c_new
            if d > bd:
                bd, ba, bb, bc, bd2 = d, a1, a2, b1, b2
            h[a1], h[a2], h[b1], h[b2] = 1, 1, 0, 0
        if ba >= 0 and bd > 0:
            h[ba], h[bb], h[bc], h[bd2] = 0, 0, 1, 1
            P = float(np.max(fft_corr(h)))
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c

def ils_search(h, N, deadline, max_rounds):
    """Iterated Local Search: perturb + local search"""
    h = (h > 0.5).astype(float)
    nh = N // 2
    best_c = compute_c5(h, N)
    best_h = h.copy()
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        # Perturb: flip k random pairs
        k = np.random.randint(3, 12)
        hp = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(k):
            a = int(np.random.choice(ones))
            b = int(np.random.choice(zeros))
            hp[a], hp[b] = hp[b], hp[a]
        # Quick local search on perturbed
        hd, _ = bin_search(hp, N, min(deadline, time.time() + 20), max_iters=100, ntri=25, anneal=0.0)
        cd = compute_c5(hd, N)
        if cd < best_c:
            best_c = cd
            best_h = hd.copy()
    return best_h, best_c

def multi_res(best_h, best_c5, deadline):
    x_b = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [220, 260, 300, 340, 420, 480, 560]:
        if time.time() > deadline:
            break
        if Nt % 2:
            Nt += 1
        if Nt == len(best_h):
            continue
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_b, best_h)
        ht = (hi > 0.5).astype(float)
        nh = Nt // 2
        n1 = int(ht.sum())
        if n1 > nh:
            ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0.0
        elif n1 < nh:
            ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1.0
        sd = min(deadline, time.time() + 28)
        hr, _ = bin_search(ht, Nt, sd, max_iters=180, ntri=25, anneal=0.0)
        c5r = compute_c5(hr, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_b = np.linspace(0, 2, Nt, endpoint=False)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Initial binary search
    d1 = min(dl - 500, t0 + 55)
    hb, _ = bin_search(best_h, N0, d1, max_iters=200, ntri=30, anneal=0.0)
    c = compute_c5(hb, N0)
    if c < best_c5:
        best_c5, best_h = c, hb.copy()

    # Phase 2: Top-k continuous optimization
    if time.time() < dl - 400:
        d2 = min(dl - 340, time.time() + 140)
        for r in range(5):
            if time.time() > d2:
                break
            hr = best_h + np.random.randn(N0) * (0.14 - 0.02 * r)
            hr = proj01(hr, N0 / 2.0)
            hl, cl = lbfgs_opt(hr, N0, d2, 'topk', K=10, T=30.0, max_outer=3)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 3: Binary + 2-exchange
    if time.time() < dl - 280:
        d3 = min(dl - 220, time.time() + 80)
        hb3, _ = bin_search(best_h, N0, d3, max_iters=600, ntri=35, anneal=0.0)
        c3 = compute_c5(hb3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, hb3.copy()

    if time.time() < dl - 220:
        d4 = min(dl - 170, time.time() + 50)
        h2, c2 = bin_search2(best_h, N0, d4, max_iters=250)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 4: Softmax continuous
    if time.time() < dl - 180:
        d5 = min(dl - 130, time.time() + 55)
        h5 = best_h + np.random.randn(len(best_h)) * 0.06
        h5 = proj01(h5, len(best_h) / 2.0)
        hl5, cl5 = lbfgs_opt(h5, len(best_h), d5, 'softmax', K=0, T=35.0, max_outer=3)
        if cl5 < best_c5:
            best_c5, best_h = cl5, hl5.copy()

    # Phase 5: ILS
    if time.time() < dl - 150:
        d6 = min(dl - 100, time.time() + 60)
        h6, c6 = ils_search(best_h, len(best_h), d6, max_rounds=8)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 6: Multi-resolution
    if time.time() < dl - 80:
        best_h, best_c5 = multi_res(best_h, best_c5, min(dl - 40, time.time() + 100))

    # Phase 7: Final binary polish
    if time.time() < dl - 20:
        d7 = min(dl - 10, time.time() + 40)
        hf, c7 = bin_search(best_h, len(best_h), d7, max_iters=500, ntri=40, anneal=0.0005)
        if c7 < best_c5:
            best_c5, best_h = c7, hf.copy()

    # Phase 8: Final 2-exchange
    if time.time() < dl - 4:
        h8, c8 = bin_search2(best_h, len(best_h), dl - 1, max_iters=80)
        if c8 < best_c5:
            best_c5, best_h = c8, h8.copy()

    # Ensure binary consistency
    fb = (best_h > 0.5).astype(float)
    nb = len(fb)
    fc5a = compute_c5(best_h, nb)
    fc5b = compute_c5(fb, nb)
    if fc5b < fc5a:
        best_h, best_c5 = fb, fc5b
    else:
        best_c5 = fc5a
    return best_h, best_c5, len(best_h)
