        h = (x % 7 < 3).astype(float)
    elif variant == 1:
        h = (np.sin(2 * np.pi * x * 3.0 / N) + 0.6 * np.sin(2 * np.pi * x * 7.0 / N + 1.2) > 0).astype(float)
    elif variant == 2:
        h = ((x * 11) % 19 < 9).astype(float)
    else:
        v = np.sin(2 * np.pi * x * 5.0 / N) + 0.8 * np.cos(2 * np.pi * x * 11.0 / N + 0.7) + 0.5 * np.sin(2 * np.pi * x * 17.0 / N)
        h = (v > np.median(v)).astype(float)
    return ensure_sum(h, N // 2)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_start = (h0 > 0.5).astype(float)
    ensure_sum(h_start, N0 // 2)

    best_h = h_start.copy()
    best_c5 = c5_value(h_start)

    # Phase 1: Focused local search from initial
    d1 = min(deadline - 400, t0 + 250)
    h1, c1 = focused_local_search(h_start, d1, max_rounds=400)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: ILS (Iterated Local Search) with kicks
    h_ils = best_h.copy()
    kick_sizes = [3, 5, 7, 10]
    ki = 0
    while time.time() < deadline - 250:
        # Kick
        ks = kick_sizes[ki % len(kick_sizes)]
        ki += 1
        h_kick = best_h.copy()
        ones_k = np.where(h_kick == 1)[0]
        zeros_k = np.where(h_kick == 0)[0]
        for _ in range(ks):
            p = int(np.random.choice(ones_k))
            q = int(np.random.choice(zeros_k))
            h_kick[p], h_kick[q] = 0, 1
            ones_k = np.where(h_kick == 1)[0]
            zeros_k = np.where(h_kick == 0)[0]

        sub_dl = min(deadline - 250, time.time() + 100)
        h_ls, c_ls = focused_local_search(h_kick, sub_dl, max_rounds=300)
        if c_ls < best_c5:
            best_c5, best_h = c_ls, h_ls.copy()
        elif c_ls < c5_value(h_start) + 0.02:
            h_ils = h_ls.copy()
    # Phase 3: Simulated annealing
    if time.time() < deadline - 120:
        h_sa = best_h.copy()
        sub_dl = min(deadline - 120, time.time() + 180)
        h_sa, c_sa = simulated_annealing(h_sa, sub_dl, T0=0.4, T1=0.005)
        if c_sa < best_c5:
            best_c5, best_h = c_sa, h_sa.copy()
        # Local search after SA
        sub_dl2 = min(deadline - 80, time.time() + 80)
        h_sa2, c_sa2 = focused_local_search(h_sa, sub_dl2, max_rounds=300)
        if c_sa2 < best_c5:
            best_c5, best_h = c_sa2, h_sa2.copy()

    # Phase 4: Try different N and patterns
    for N_try in [200, 300, 500]:
        if time.time() > deadline - 50:
            break
        if N_try == N0:
            continue
        if N_try % 2 != 0:
            N_try += 1
        for vi in range(3):
            if time.time() > deadline - 30:
                break
            hv = structured_init(N_try, vi)
            sub_dl = min(deadline - 25, time.time() + 30)
            hr, cr = focused_local_search(hv, sub_dl, max_rounds=200)
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()
        # Also try Thue-Morse
        if time.time() > deadline - 25:
            break
        htm = thue_morse_init(N_try)
        sub_dl = min(deadline - 20, time.time() + 25)
        hr, cr = focused_local_search(htm, sub_dl, max_rounds=200)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 5: Final polish
    if time.time() < deadline - 3:
        hF, cF = focused_local_search(best_h.copy(), deadline - 1, max_rounds=500)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    # Ensure valid
    best_h = best_h.astype(float)
    best_h[(best_h > 0.5)] = 1.0
    best_h[(best_h <= 0.5)] = 0.0
    ensure_sum(best_h, len(best_h) // 2)
    best_c5 = c5_value(best_h)

    return best_h, best_c5, len(best_h)
