        hr, _ = sa_optimize(hb, d4, n_peaks=5, n_per_peak=14, cool_rate=0.999, t_start=0.04, tabu_len=30)
        cr = compute_c5(hr, N_try)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 5: Final fine SA with very low temperature
    if time.time() < deadline - 10:
        Ny = len(best_h)
        h5 = make_binary(best_h, Ny)
        d5 = deadline - 3
        h5, _ = sa_optimize(h5, d5, n_peaks=10, n_per_peak=25, cool_rate=0.9997, t_start=0.012, tabu_len=50, reheat_T=0.015, kick_size=3)
        c5 = compute_c5(h5, Ny)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
