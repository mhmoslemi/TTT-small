    h = np.clip(h, 0, 1)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def c5_val(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def proj(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def g_lag(y, idx, N):
    s = idx - (N - 1)
    g = np.zeros(N)
    if s >= 0:
        g[:N - s] += (1.0 - y[s:])
        g[s:] -= y[:N - s]
    else:
        m = -s
        g[m:] += (1.0 - y[:-m])
        g[:-m] -= y[m:]
    return g

def nesterov_run(h0, N, dl, kt=30, lr0=0.011, b0=0.92):
    h = proj(h0.astype(float), N / 2.0)
    m = np.zeros(N)
    lr, beta = lr0, b0
    bh, bc = h.copy(), float(np.max(fft_corr(h)))
    st, it = 0, 0
    while time.time() < dl:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < bc - 1e-13:
            bc, bh, st = pk, h.copy(), 0
        else:
            st += 1
        if st > 6000:
            break
        top = np.argpartition(c, -kt)[-kt:]
        mrg = np.maximum(c[top] - pk + 0.002, 0.03)
        w = mrg / mrg.sum()
        gr = np.zeros(N)
        for wi, ix in zip(w, top):
            gr += wi * g_lag(y, ix, N)
        h2 = proj(y - lr * gr, N / 2.0)
        m = beta * (h2 - h)
        h = h2
        it += 1
        if it % 350 == 0:
            lr *= 0.982
            beta = min(0.985, 0.89 + it * 0.000006)
    return bh, bc * (2.0 / N)

def D_mat(h, N):
    sh = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = sh[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pmc = np.clip(pm, 0, N - 1)
    ppc = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pmc], 0.0)
    hp = np.where(vp, h[ppc], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def bin_greedy(h0, N, dl, ms=2000, sf=1.0):
    h = (h0 > 0.5).astype(float)
    t = N // 2
    n1 = int(h.sum())
    if n1 > t:
        ws = np.where(h == 1)[0]
        h[np.random.choice(ws, n1 - t, replace=False)] = 0
    elif n1 < t:
        ws = np.where(h == 0)[0]
        h[np.random.choice(ws, t - n1, replace=False)] = 1
    c = fft_corr(h)
    bh, bc = h.copy(), float(c.max())
    st = 0
    for sw in range(ms):
        if time.time() > dl:
            break
        P = float(c.max())
        if P < bc - 1e-13:
            bc, bh, st = P, h.copy(), 0
        else:
            st += 1
            if st > 80:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = D_mat(h, N)
        n_o = len(ones)
        n_s = n_o if sf >= 1.0 else max(40, int(n_o * sf))
        p_idx = np.random.choice(n_o, n_s, replace=False) if n_s < n_o else range(n_o)
        Do = D[ones[p_idx], :]
        Dz = D[zeros, :]
        best_np, best_pq = P + 1, None
        for pi in range(len(p_idx)):
            base = c + Do[pi, :]
            cand = base[None, :] - Dz
            mx = cand.max(axis=1)
            mi = int(np.argmin(mx))
            if mx[mi] < best_np:
                best_np = float(mx[mi])
                best_pq = (int(ones[p_idx[pi]]), int(zeros[mi]))
        if best_pq is None or best_np >= P - 1e-14:
            break
        h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
        c = fft_corr(h)
    return bh, bc * (2.0 / N)

def bin_sa(h0, N, dl, ms=6000, T0=3.5, Tc=0.994, Tm=0.03, sf=1.0, ke=120, ks=5):
    h = (h0 > 0.5).astype(float)
    t = N // 2
    n1 = int(h.sum())
    if n1 > t:

    c = fft_corr(h)
    bh, bc = h.copy(), float(c.max())
    T, st = T0, 0
    for sw in range(ms):
        if time.time() > dl:
            break
        P = float(c.max())
        if P < bc - 1e-13:
            bc, bh, st = P, h.copy(), 0
        else:
            st += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if st >= ke:
            for _ in range(ks):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            st = 0
            T = T0 * 0.5
            c = fft_corr(h)
            P = float(c.max())
            continue
        D = D_mat(h, N)
        n_o = len(ones)
        n_s = n_o if sf >= 1.0 else max(40, int(n_o * sf))
        p_idx = np.random.choice(n_o, n_s, replace=False) if n_s < n_o else range(n_o)
        Do = D[ones[p_idx], :]
        Dz = D[zeros, :]
        best_np, best_pq = P + 1, None
        for pi in range(len(p_idx)):
            base = c + Do[pi, :]
            cand = base[None, :] - Dz
            mx = cand.max(axis=1)
            mi = int(np.argmin(mx))
            if mx[mi] < best_np:
                best_np = float(mx[mi])
                best_pq = (int(ones[p_idx[pi]]), int(zeros[mi]))
        if best_pq is None:

        delta = best_np - P
        accept = delta < -1e-14 or (T > Tm and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))
        if accept:
            h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
            c = fft_corr(h)
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
        T = max(Tm, T * Tc)
    return bh, bc * (2.0 / N)

def exhaustive_flip(h, N):
    """Try all single swaps, return best."""
    D = D_mat(h, N)
    c = fft_corr(h)
    P = float(c.max())
    bh = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    Do = D[ones, :]
    Dz = D[zeros, :]
    for pi in range(len(ones)):
        base = c + Do[pi, :]
        cand = base[None, :] - Dz
        mx = cand.max(axis=1)
        mi = int(np.argmin(mx))
        if mx[mi] < P - 1e-12:
            P = float(mx[mi])
            bh = h.copy()
            bh[ones[pi]], bh[zeros[mi]] = 0.0, 1.0
    return bh, P * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
