def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    # Gradient of C[h, L] where L is the shift (in sample units)
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(h, T, target, lam=0.0):

    e = np.exp(T * (c - mx))

    val = mx + np.log(e.sum()) / T

        else:

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: L-BFGS polish from initial (90s)
    d1 = min(dl - 500, t0 + 70)
    hl, cl = lbfgs_smooth(best_h, N0, d1, T=25.0, max_outer=3)
    if cl < best_c5:
        best_c5, best_h = cl, hl.copy()

    # Phase 2: FISTA accelerated (150s)
    if time.time() < dl - 300:
        d2 = min(dl - 250, time.time() + 130)
        hf, cf = fista_smooth(best_h, N0, d2, T=35.0, n_max=3000)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Phase 3: Multi-start L-BFGS (120s)
    if time.time() < dl - 200:
        d3 = min(dl - 160, time.time() + 100)
        for i in range(4):
            if time.time() > d3:
                break
            scale = 0.15 - 0.03 * i
            hp = best_h + np.random.randn(N0) * scale
            hp = proj01(hp, N0 / 2.0)
            hl3, cl3 = lbfgs_smooth(hp, N0, d3, T=30.0 + 5 * i, max_outer=2)
            if cl3 < best_c5:
                best_c5, best_h = cl3, hl3.copy()

    # Phase 4: Active-set subgradient (100s)
    if time.time() < dl - 100:
        d4 = min(dl - 60, time.time() + 80)
        hs = best_h + np.random.randn(N0) * 0.05
        hs = proj01(hs, N0 / 2.0)
        hsn, cn = subgrad_active(hs, N0, d4, alpha0=0.025, beta=0.94, atol=0.4)
        if cn < best_c5:
            best_c5, best_h = cn, hsn.copy()

    # Phase 5: FISTA again from current best (80s)
    if time.time() < dl - 80:
        d5 = min(dl - 50, time.time() + 70)
        hf5, cf5 = fista_smooth(best_h, N0, d5, T=50.0, n_max=2500)
        if cf5 < best_c5:
            best_c5, best_h = cf5, hf5.copy()

    # Phase 6: Binary local search (120s)
    if time.time() < dl - 40:
        d6 = min(dl - 20, time.time() + 100)
        hb, _ = bin_search(best_h, N0, d6, ntri=35, max_iters=700, anneal=0.001)
        c5b = compute_c5(hb, N0)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()

    # Phase 7: L-BFGS polish on best binary (50s)
    if time.time() < dl - 20:
        d7 = min(dl - 8, time.time() + 40)
        hl7 = best_h.copy() + np.random.randn(N0) * 0.02
        hl7 = proj01(hl7, N0 / 2.0)
        hl7, cl7 = lbfgs_smooth(hl7, N0, d7, T=40.0, max_outer=2)
        if cl7 < best_c5:
            best_c5, best_h = cl7, hl7.copy()

    # Phase 8: Other resolutions
    if time.time() < dl - 10:
        best_h, best_c5, N0 = try_other_res(best_h, best_c5, N0, min(dl - 5, time.time() + 80), rng)

    # Final: compare continuous vs binary
    fc5 = compute_c5(best_h, len(best_h))
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, len(best_h))
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, len(best_h)
