def compute_corr(h):

    L = 2 * N - 1

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_corr_top(h, corr, k=40):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def cont_opt(h, deadline, k=40, lr0=0.02, mom=0.88):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 50:
            lr0 *= 0.75
            stale = 0
        g = grad_corr_top(h, corr, k)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.75 * v + 0.4 * (h_new - h)
        h = h_new
    return best_h, best_c

def sa_binary(h, deadline, t0=0.08, t_min=0.00005):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    c = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    best_c = c
    t = t0
    no_imp = 0
    scale = 2.0 / N
    it = 0
    while time.time() < deadline:
        oi = int(np.random.choice(np.where(h == 1)[0]))
        zi = int(np.random.choice(np.where(h == 0)[0]))
        h[oi] = 0.0
        h[zi] = 1.0
        new_corr = compute_corr(h)
        new_c = float(np.max(new_corr)) * scale
        if new_c <= c or np.random.random() < np.exp(-(new_c - c) / t):
            c = new_c
            corr = new_corr
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h[oi] = 1.0
            h[zi] = 0.0
            no_imp += 1
        it += 1
        frac = min(it / 400000.0, 1.0)
        t = t0 * (t_min / t0) ** (frac ** 0.6)
        if no_imp > 5000:
            n_shake = 6 + int(8 * np.random.random())

                h[oi2] = 0.0
                h[zi2] = 1.0

            t = max(t, 0.05)
            no_imp = 0

    return best_h, best_c

        pos2 = ones - worst_lag
        valid2 = (pos2 >= 0) & (pos2 < N)
