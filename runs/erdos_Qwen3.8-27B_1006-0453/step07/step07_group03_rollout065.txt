def fft_corr(h):

    H = np.fft.fft(h, L)

def compute_delta(h, N):
    shifts = np.arange(2*N-1) - (N-1)  # shape (2N-1,)
    p = np.arange(N)  # shape (N,)
    
    # Part 1: n = p - shift, valid when 0 <= n < N
    n1 = p[:, None] - shifts[None, :]  # shape (N, 2N-1)
    valid1 = (n1 >= 0) & (n1 < N)
    D = np.where(valid1, -(1.0 - h[np.clip(n1, 0, N-1)]), 0.0)
    
    # Part 2: n2 = p + shift, valid when 0 <= n2 < N
    n2 = p[:, None] + shifts[None, :]  # shape (N, 2N-1)
    valid2 = (n2 >= 0) & (n2 < N)
    D += np.where(valid2, h[np.clip(n2, 0, N-1)], 0.0)
    
    return D

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_bin(h, N, rng=None):
    nh = N // 2
    h = np.asarray(h, dtype=float).copy()
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    if rng is None:
        rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(hb == 1)[0]
        hb[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(hb == 0)[0]
        hb[rng.choice(zeros, nh - n1, replace=False)] = 1
    return hb

def delta_mat(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p = np.arange(N)
    n1 = p[:, None] - shifts[None, :]
    n2 = p[:, None] + shifts[None, :]
    v1 = (n1 >= 0) & (n1 < N)
    v2 = (n2 >= 0) & (n2 < N)
    D = np.where(v1, -(1.0 - h[np.clip(n1, 0, N - 1)]), 0.0)
    D += np.where(v2, h[np.clip(n2, 0, N - 1)], 0.0)
    return D

def upsample(h, N_new):
    N_old = len(h)
    x_old = np.arange(N_old) / float(N_old)
    x_new = np.arange(N_new) / float(N_new)
    return np.interp(x_new, x_old, h)
