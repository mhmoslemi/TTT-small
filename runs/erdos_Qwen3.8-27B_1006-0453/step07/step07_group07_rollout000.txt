import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_top(h, corr, top_k, sharpness=8.0):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-top_k:]
    vals = corr[top_idx]
    w = np.exp(sharpness * (vals - vals.max()))
    w /= w.sum()
    grad = np.zeros(N)
    for i, m in enumerate(top_idx):
        M = abs(int(m) - center)
        if M == 0:
            grad += w[i] * (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += w[i] * (1.0 - h[hi_idx - M])
            grad[lo_idx] -= w[i] * h[lo_idx + M]
    return grad

def cont_opt(h, deadline, lr0=0.02, mom=0.85, top_k=40):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    lr = lr0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 40:
            lr *= 0.88
            stale = 0
            if lr < 0.0005:
                break
        g = grad_top(h, corr, top_k)
        h_new = project(h + mom * v - lr * g, tgt)
        v = mom * v + (1.0 - mom) * (h_new - h)
        h = h_new
    return best_h, best_c

def binary_search(h, deadline, max_iters=5000, n_crit=30, s_sz=45):
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    it = 0
    no_imp = 0
    center = N - 1
    T = 0.005

        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()

        else:
            no_imp += 1
        T = max(0.0005, T * 0.9995)
        if no_imp > 600:

        ones = np.where(h == 1)[0]

        if len(ones) < 2 or len(zeros) < 2:
            break
        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        n_crit_actual = len(top_lags)
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        cur_peak = float(np.max(corr))
        base_vals = corr[center + top_lags].copy()
        best_move = None
        best_new_peak = cur_peak
        # Precompute removal effects for ones

        for i in range(ns):

        it += 1
    return best_h, best_c

    """Create a low-discrepancy initial binary pattern."""
    rng = np.random.RandomState(seed)
    # Use golden-ratio based placement
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.zeros(N)
    for i in range(N // 2):

        pos = min(pos, N - 1)
        if positions[pos] < 1.0:
            positions[pos] = 1.0
    if positions.sum() < N // 2:
        remaining = N // 2 - int(positions.sum())
        empty = np.where(positions == 0)[0]
        chosen = rng.choice(empty, remaining, replace=False)
        positions[chosen] = 1.0
    return positions
