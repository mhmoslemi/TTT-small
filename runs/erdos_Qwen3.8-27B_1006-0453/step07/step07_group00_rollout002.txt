                        no_impr += 1
                else:
                    no_impr += 1
                if no_impr > stagnation_limit:
                    break
                continue
        
        # Standard single swap acceptance
        p, q = int(cand_p[bi]), int(cand_q[bj])
        if dc < -1e-12 or (T > 0.0015 and rng.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0
            deltas[p, p:p+N] = s
            deltas[q, :] = 0.0
            deltas[q, q:q+N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        
        if no_impr > stagnation_limit:
            # Reheat
            T = t_start * 0.3
            no_impr = 0
            # Perturb
            nr = min(3, len(ones), len(zeros))
            if nr >= 2:
                ps = rng.choice(ones, nr, replace=False)
                qs = rng.choice(zeros, nr, replace=False)
                for pp, qq in zip(ps, qs):
                    h[pp], h[qq] = 0.0, 1.0
                corr = fft_corr(h)
                s = 2.0 * h - 1.0
                deltas = build_deltas(s, N)
    
    return best_h, best_c

def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=40):
    h = np.clip(h_in.astype(float), 0, 1)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.82
    it = 0
    while time.time() < deadline:
        y = h + m
        ccy = fft_corr(y)
        pk = float(ccy.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(ccy, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
            beta = min(0.95, 0.80 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
