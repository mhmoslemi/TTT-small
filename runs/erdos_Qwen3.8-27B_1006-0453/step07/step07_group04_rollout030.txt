def relax_2d(h, N, p, q):
    """Optimize h[p] and h[q] (sum=1) to minimize max corr."""
    base_c = fft_corr_with_fractional(h, p, q, [0, 0.5, 1])
    # Actually, compute C(k) as a function of t = h[p] (h[q] = 1-t)
    # For each k, C(k) = a_k * t + b_k (linear)
    # Find t* that minimizes max_k (a_k*t + b_k)
    
    # Compute a_k and b_k:
    t_vals = np.array([0.0, 0.5, 1.0])
    c_vals = []
    for t in t_vals:
        h_try = h.copy()
        h_try[p] = t
        h_try[q] = 1.0 - t
        c_vals.append(fft_corr(h_try))
    c_vals = np.array(c_vals)  # shape (3, 2N-1)
    
    # C(k) = c0[k] + (c2[k] - c0[k]) * t (linear interpolation)
    # Actually with 3 points at t=0, 0.5, 1:
    # a_k = c2[k] - c0[k] (slope)
    # b_k = c0[k] (intercept)
    # Verify: at t=0.5, c0 + 0.5*a = c0 + 0.5*(c2-c0) = (c0+c2)/2
    # Should equal c1[k]. Check: (c0+c2)/2 == c1? Only if linear, which it is.
    
    a = c_vals[2, :] - c_vals[0, :]  # slope
    b = c_vals[0, :]  # intercept
    # C(k; t) = b[k] + a[k]*t for t in [0,1]
    
    # Minimize max_k (b[k] + a[k]*t) over t in [0,1]
    # This is a 1D convex optimization
    # The max of lines is convex, minimum at intersection of top lines
    # Simple approach: golden section or ternary search
    
    def max_corr(t):
        return float(np.max(b + a * t))
    
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(100):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if max_corr(m1) < max_corr(m2):
            hi = m2
        else:
            lo = m1
    t_star = (lo + hi) / 2
    
    h[p] = t_star
    h[q] = 1.0 - t_star
    return h, max_corr(t_star) * (2.0 / N)

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def c5_val(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def greedy_pass(h, N, deadline, max_sweeps=500):
    h = h.astype(float).copy()
    best_p = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_p - 1e-13:
            best_p, best_h = P, h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        found = False
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < P - 1e-14:
                h[p] = 0.0
                h[zeros[mi]] = 1.0
                found = True
                break  # restart the sweep
        if not found:
            break
    return best_h, best_p * (2.0 / N)

def sa_opt(h, N, deadline, max_sweeps=10000, T0=3.0, Tcool=0.995, T_min=0.03,
           sf=1.0, kick_every=120, kick_size=4):
    h = h.astype(float).copy()
    rng = np.random.RandomState(42)
    best_p = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_p - 1e-13:
            best_p, best_h, stall = P, h.copy(), 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.7
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_p * (2.0 / N)

def relax_2d(h, N, p, q, deadline=None):
    """Optimize h[p]=t, h[q]=1-t to minimize max corr."""
    h0 = h.copy()
    h0[p] = 0.0
    h0[q] = 1.0
    c0 = fft_corr(h0)
    h1 = h.copy()
    h1[p] = 1.0
    h1[q] = 0.0
    c1 = fft_corr(h1)
    # C(k; t) = c0[k] + (c1[k] - c0[k]) * t (linear in t since h only appears linearly in C)
    # Wait, is C linear in t? C(k) = sum_n h[n+k]*(1-h[n])
    # If h[p]=t, h[q]=1-t, then:
    # Term with h[p] as first factor: t * (1-h[n]) for n=p-k
    #   Actually: h[n+k]=t means n+k=p, n=p-k. Term: t*(1-h[p-k])
    # Term with h[p] as second factor: h[n+k]*(1-t) for n=p
    #   Term: h[p+k]*(1-t)
    # These are linear in t. ✓
    # Similarly for q with (1-t).
    # So C(k; t) = a_k * t + b_k for some a_k, b_k. ✓
    a = (c1 - c0)  # slope
    b = c0  # intercept
    # Minimize max_k (b[k] + a[k]*t) for t in [0,1]
    # Ternary search
    lo, hi = 0.0, 1.0
    f_lo = float(np.max(b))
    f_hi = float(np.max(b + a))
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    m1 = hi - gr * (hi - lo)
    m2 = lo + gr * (hi - lo)
    f_m1 = float(np.max(b + a * m1))
    f_m2 = float(np.max(b + a * m2))
    for _ in range(80):
        if f_m1 < f_m2:
            hi, f_hi = m2, f_m2
            m2 = m1
            f_m2 = f_m1
            m1 = hi - gr * (hi - lo)
            f_m1 = float(np.max(b + a * m1))
        else:
            lo, f_lo = m1, f_m1
            m1 = m2
            f_m1 = f_m2
            m2 = lo + gr * (hi - lo)
            f_m2 = float(np.max(b + a * m2))
    t_star = (lo + hi) * 0.5
    new_c = float(np.max(b + a * t_star))
    h[p] = t_star
    h[q] = 1.0 - t_star
    return h, new_c * (2.0 / N)

def best_2d_relax(h, N, deadline):
    """Try multiple (p,q) pairs for 2D relaxation."""
    best_h = h.copy()
    best_c = c5_val(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) < 1 or len(zeros) < 1:
        return h, best_c
    # Try top candidates: p from top-20 ones (by contribution to max corr), q from random zeros
    k_star = int(np.argmax(fft_corr(h)))
    center = N - 1
    shift = k_star - center
    # Contribution of h[p] to C(shift): (1-h[p-shift]) if p-shift valid
    contrib = np.zeros(N)
    valid_dn = (np.arange(N) - shift) >= 0
    valid_dn &= (np.arange(N) - shift) < N
    contrib[valid_dn] = (1.0 - h[np.arange(N)[valid_dn] - shift])
    # p's with highest contribution to the worst shift
    top_p = ones[np.argsort(-contrib[ones])[:15]]
    rng = np.random.RandomState(99)
    tried = set()
    for p in top_p:
        if time.time() > deadline:
            break
        for _ in range(5):
            q = zeros[rng.randint(len(zeros))]
            key = (int(p), int(q))
            if key in tried:
                continue
            tried.add(key)
            h_try = h.copy()
            h_try[p] = 0.5  # start from middle
            h_try[q] = 0.5  # ensures sum stays same
            h_try, c_try = relax_2d(h, N, int(p), int(q))
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
    return best_h, best_c

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    nh = N // 2
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(777 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin
