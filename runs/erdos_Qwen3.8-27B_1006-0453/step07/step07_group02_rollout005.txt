The logSumExp trick: max(x) ≈ (1/β) log(sum(exp(β*x)))
For large β, this approaches the max. The gradient is well-defined everywhere.

∂logSumExp/∂x_i = exp(β*x_i) / sum(exp(β*x_j))

Then by chain rule: ∂f/∂h_i = sum_m [∂f/∂corr_m] * [∂corr_m/∂h_i]

The issue is computing ∂corr_m/∂h_i for all m,i. This is an N × (2N-1) Jacobian.

For the cross-correlation:
corr[m] = sum_i h[i] * (1-h[i+m-N+1]) (with boundary handling)

∂corr[m]/∂h[i] = (1-h[i+m-N+1]) - h[i-m+N-1] (with boundary handling, terms are 0 if out of range)

This is a structured matrix (block circulant), so we can apply it to a vector in O(N log N) using FFT!

Specifically:
J^T * v = for each i: sum_m v[m] * ∂corr[m]/∂h[i]
= sum_m v[m] * (1-h[i+m-N+1]) - sum_m v[m] * h[i-m+N-1]

The first sum: sum_m v[m] * (1-h[i+m-N+1])
= sum_m v[m] - sum_m v[m]*h[i+m-N+1]
= sum(v) - (v convolved with h evaluated at i-N+1) [roughly]

The second sum: sum_m v[m] * h[i-m+N-1]

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def c5_of(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * 2.0/N

def fix_bin(h, n1):
    s = int(round(h.sum()))
    if s > n1:
        idx = np.where(h==1)[0]
        h[np.random.choice(idx, s-n1, replace=False)] = 0
    elif s < n1:
        idx = np.where(h==0)[0]
        h[np.random.choice(idx, n1-s, replace=False)] = 1
    return h

def proj(h, tgt):
    h = np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(45):
        mid = (lo+hi)/2
        if np.clip(h-mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def grad_smooth(h, beta):
    N = len(h)
    corr = fft_corr(h)
    c = corr - corr.max()
    w = np.exp(beta * c)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(N), corr
    w = w / ws
    
    m_arr = np.arange(2*N-1)
    j_arr = np.arange(N)
    idx1 = j_arr[:, None] + m_arr[None, :] - N + 1
    idx2 = j_arr[:, None] - m_arr[None, :] + N - 1
    v1 = (idx1 >= 0) & (idx1 < N)
    v2 = (idx2 >= 0) & (idx2 < N)
    c1 = np.clip(idx1, 0, N-1)
    c2 = np.clip(idx2, 0, N-1)
    
    term1 = np.sum(w[None, :] * (1.0 - h[c1]) * v1, axis=1)
    term2 = np.sum(w[None, :] * h[c2] * v2, axis=1)
    grad = term1 - term2
    return grad, corr

def smooth_opt(h, N, dl, beta=80, lr0=0.01, max_it=50000):
    h = h.astype(float).copy()
    tgt = N / 2.0
    h = proj(h, tgt)
    best = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta_cur = beta
    for it in range(max_it):
        if time.time() > dl:
            break
        y = h + m
        y = np.clip(y, 0, 1)
        g, corr = grad_smooth(y, beta_cur)
        pk = float(corr.max())
        if pk < best:
            best = pk
            best_h = h.copy()
        h_new = proj(y - lr * g, tgt)
        m = 0.9 * (h_new - h)
        h = h_new
        if it % 300 == 0:
            lr *= 0.99
            beta_cur = min(200, beta_cur * 1.02)
    return best_h, best * 2.0/N
