```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-10:
        return h
    # Use bisection on shift
    lo, hi = -max(h.max(), 1.0), -min(h.min(), 0.0)
    # Actually simpler: sort and adjust
    # h - shift, clipped to [0,1], should sum to target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lse_subgrad(h, N, deadline, T_init=20.0, T_min=80.0, lr0=0.005, max_iters=5000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)  # momentum
    beta = 0.7
    lr = lr0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        # Log-sum-exp smoothing
        T = T_init + (T_min - T_init) * min(1.0, it / (max_iters * 0.7))

        exp_c = np.exp(T * (corr - max_c))
        weights = exp_c / exp_c.sum()
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def topk_sum(corr, k=5):
    srt = np.sort(corr)[::-1]
    return float(np.sum(srt[:k]))

def project_01(h, target_sum):
    h = np.clip(h, 0, 1)
    if abs(h.sum() - target_sum) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def build_C(h, N):
    """C[i, :] = change in corr[k] if we flip h[i] (1->0 or 0->1)."""
    # If h[i] goes from 0 to 1: corr[k] changes by (1-h[i-k])_valid - h[i+k]_valid * (-1)
    # Actually: if h[i] 0->1:
    #   For corr[k] = sum_j h[j+k]*(1-h[j]), the new h[i] contributes:
    #   +1*(1-h[i-k]) if i-k valid (as h[j+k] with j=i-k)
    #   -h[i+k]*1 if i+k valid (as -h[j]*(h[j+k]) with j=i, h[i] going 0->1)
    #   Wait, it's (1-h[j]) that multiplies h[j+k]. So:
    #   When h[i] changes 0->1:
    #     As h[j+k] (j+k=i, j=i-k): contributes (1-h[i-k]) * 1 if i-k valid
    #     As h[j] (j=i): contributes h[i+k] * (-1) * h[i+k]... no.
    #     
    # Let me re-derive. corr[k] = sum_j h[j+k]*(1-h[j])
    # = sum_j h[j+k] - sum_j h[j+k]*h[j]
    #
    # d corr[k] / d h[i]:
    # Term 1: sum_j h[j+k]. h[i] appears when i = j+k or i = j.
    #   If i = j+k (j = i-k): contributes 1 if i-k in [0,N-1]
    #   If i = j: contributes 1 if i in [0,N-1] (always true) AND i+k in [0,N-1] (so that h[i+k] is in the sum)
    #   
    #   Wait, sum_j h[j+k] where j ranges over [0,N-1] and j+k in [0,N-1].
    #   h[i] appears when j=i (and i+k valid) or j+k=i (i.e., j=i-k valid and i valid).
    #   So contribution: 1[i+k valid] + 1[i-k valid]
    #
    # Term 2: -sum_j h[j+k]*h[j]. h[i] appears in:
    #   - j=i: -h[i+k]*h[i], derivative w.r.t. h[i] is -h[i+k] (if i+k valid)
    #   - j+k=i (j=i-k): -h[i]*h[i-k], derivative w.r.t. h[i] is -h[i-k] (if i-k valid)
    #
    # So d corr[k] / d h[i] = 1[i+k valid] + 1[i-k valid] - h[i+k]*1[i+k valid] - h[i-k]*1[i-k valid]
    # = (1-h[i+k])*1[i+k valid] + (1-h[i-k])*1[i-k valid]
    #
    # Wait that's for the continuous case. For binary, the "change" when flipping h[i] from 0 to 1:
    # Δcorr[k] = (1-h[i+k])*1[i+k valid] + (1-h[i-k])*1[i-k valid]
    # And when flipping from 1 to 0:
    # Δcorr[k] = -(1-h[i+k])*1[i+k valid] - (1-h[i-k])*1[i-k valid]

    C = np.zeros((N, 2*N-1))
    # C[i, :] = Δcorr[k] when h[i] goes 0->1
    for i in range(N):
        for k in range(-(N-1), N):
            ik = i + k
            imk = i - k
            delta = 0.0
            if 0 <= ik < N:
                delta += (1.0 - h[ik])
            if 0 <= imk < N:
                delta += (1.0 - h[imk])
            C[i, k + N - 1] = delta
    return C
```

```python
def build_C(h, N):
    C = np.zeros((N, 2*N-1))
    k = np.arange(-(N-1), N)[None, :]  # (1, 2N-1)
    i = np.arange(N)[:, None]  # (N, 1)
    ik = i + k  # (N, 2N-1)
    imk = i - k
    mask_k = (ik >= 0) & (ik < N)
    mask_mk = (imk >= 0) & (imk < N)
    h_ik = np.where(mask_k, h[np.clip(ik, 0, N-1)], 0)
    h_imk = np.where(mask_mk, h[np.clip(imk, 0, N-1)], 0)
    C = mask_k.astype(float) * (1 - h_ik) + mask_mk.astype(float) * (1 - h_imk)
    return C
```

This is O(N^2) which for N=380 is about 300k entries. Fast.

Now for a swap of h[a]:1->0 and h[b]:0->1:
Δcorr[k] = -C[a, k] + C[b, k] (negative because a goes from 1 to 0, positive because b goes 0 to 1)

Wait, C[i,:] is the change when h[i] goes 0->1. So:
- h[a] goes 1->0: change is -C[a,:]
- h[b] goes 0->1: change is +C[b,:]
- Total: C[b,:] - C[a,:]

New corr[k] = old_corr[k] + C[b,k] - C[a,k]

This is exactly what the previous code does with its C matrix. Good.

Now, for the "top-k sum" objective:
new_topk = sum of top-k values of (corr + C[b] - C[a])

This is harder to optimize than just the max. But we can approximate:
- If the max is at position p, and p is not in the top-k, then the top-k sum ≈ old_topk + (C[b,p]-C[a,p]) if the new value at p enters the top-k.

```python
def corr_grad(h, N, k):
    """Gradient of correlation at lag k."""
    g = np.zeros(N)
    ik = np.arange(N) + k
    imk = np.arange(N) - k
    mask_k = (ik >= 0) & (ik < N)
    mask_mk = (imk >= 0) & (imk < N)
    g[mask_k] += (1 - h[np.clip(ik, 0, N-1)][mask_k])
    g[mask_mk] += (1 - h[np.clip(imk, 0, N-1)][mask_mk])
    return g
```

```python
def hotset_gradient(h, N, top_k=15, lr=0.003, deadline=None, max_iters=3000, target=None):
    if target is None:
        target = N / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        corr = fft_corr(h)
        idx = np.argpartition(corr, -top_k)[-top_k:]
        P = float(np.max(corr))
        if P < best_val - 1e-12:
            best_val = P
            best_h = h.copy()
        grad = np.zeros(N)
        for ki in idx:
            k = int(ki - (N - 1))
            g = h.copy()
            ik = np.arange(N) + k
            imk = np.arange(N) - k
            mk = (ik >= 0) & (ik < N)
            mmk = (imk >= 0) & (imk < N)
            sg = np.zeros(N)
            sg[mk] = 1.0 - h[np.clip(ik[mk], 0, N - 1)]
            sg[mmk] += 1.0 - h[np.clip(imk[mmk], 0, N - 1)]
            grad += sg
        grad /= top_k
        h_new = project_01(h - lr * grad, target)
        h = h_new
        if it % 200 == 0:
            lr *= 0.97
    return best_h, best_val * (2.0 / N)

        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Iterated local search (5 iterations with perturbation)
    for r in range(5):
        if time.time() > deadline - 150:
            break
        hp = perturb_binary(best_h, N0, n_pairs=4 + r * 2)
        dr = min(deadline - 140, time.time() + 60)
        hr, cr = local_search(hp, N0, dr, max_iters=4000, sample_sz=100, kick_thresh=35, kick_sz=4)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 5: Simulated annealing local search
    if time.time() < deadline - 100:
        h5 = (best_h > 0.5).astype(float)
        d5 = min(deadline - 80, time.time() + 80)
        h5, c5v = local_search(h5, N0, d5, max_iters=6000, sample_sz=130, kick_thresh=40, kick_sz=5, temp=0.2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()

    # Phase 6: Multi-resolution exploration
    for N_try in [340, 360, 400, 440, 480]:
        if time.time() > deadline - 60:
            break
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh_t = N_try // 2
        ones_t = np.where(h_bt == 1)[0]
        zeros_t = np.where(h_bt == 0)[0]
        if len(ones_t) > nh_t:
            h_bt[np.random.choice(ones_t, len(ones_t) - nh_t, replace=False)] = 0
        elif len(ones_t) < nh_t:
            h_bt[np.random.choice(zeros_t, nh_t - len(ones_t), replace=False)] = 1
        d6 = min(deadline - 15, time.time() + 35)
        h6, c6 = local_search(h_bt, N_try, d6, max_iters=3000, sample_sz=max(60, N_try // 4), kick_thresh=30, kick_sz=3)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: Final continuous polish
    if time.time() < deadline - 25:
        hF = best_h + np.random.randn(len(best_h)) * 0.05
        hF = project_01(hF, len(best_h) / 2.0)
        hF, cF = hotset_gradient(hF, len(best_h), top_k=20, lr=0.002, deadline=deadline - 10, max_iters=2000)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    # Phase 8: Final binary cleanup
    if time.time() < deadline - 1:
        hB = (best_h > 0.5).astype(float)
        hB, cB = local_search(hB, len(best_h), deadline - 0.5, max_iters=2000, sample_sz=120, kick_thresh=40, kick_sz=3)
        if cB < best_c5:
            best_c5, best_h = cB, hB.copy()

    # Final check: try binary version too
    hB2 = (best_h > 0.5).astype(float)
    cB2 = compute_c5(hB2, len(hB2))
    if cB2 < best_c5:
        best_h = hB2
        best_c5 = cB2

    return best_h, best_c5, len(best_h)
```
