import time

def fft_corr(h):

    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))[:2 * N - 1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, n1):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h, kind='stable')
    out = np.zeros_like(h)

def fix_count(h, n1, rng):

    elif s < n1:

    best_h = h.copy()
    
    n_iters = max(1000, max_iters)

    T = T0

    stall_lim = 4000

def continuous_refine(h, N, deadline, rng):
    """Gradient-based refinement in continuous space"""
    h = h.astype(float).copy()
    target = N / 2.0
    best_c = compute_c5(h)
    best_h = h.copy()
    lr = 0.005
    soft_T = 0.02
    it = 0
    while time.time() < deadline - 2:
        it += 1
        corr = fft_corr(h)
        peak = np.max(corr)
        # Softmax weight on peaks
        w = np.exp((corr - peak) / soft_T)
        w /= w.sum()
        
        # Gradient of the weighted correlation w.r.t. h
        grad = np.zeros(N)
        center = N - 1
        for ki in range(2 * N - 1):
            if w[ki] < 1e-6:
                continue
            L = ki - center
            wt = w[ki]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[L:] += wt * (1.0 - h[:N - L])
                grad[:N - L] -= wt * h[L:]
            elif L < 0 and L > -N:
                M = -L
                grad[M:] -= wt * h[:N - M]
                grad[:N - M] += wt * (1.0 - h[M:])
        
        h_new = np.clip(h - lr * grad, 0, 1)
        # Project to sum constraint
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 0:
            lr *= 0.99
            soft_T *= 0.97
    
    return best_h, best_c

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    V = np.fft.rfft(1.0 - h, L)
    C = H * np.conj(V)
    return np.real(np.fft.irfft(C, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, n1):
    h = np.asarray(h).astype(float)
    idx = np.argsort(-h, kind='stable')
    out = np.zeros(len(h))
    out[idx[:n1]] = 1.0
    return out

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:

    V = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * np.conj(V), L))

def compute_c5(h):

            dE = nm - cur_max
            if dE < -1e-13 or (T > 1e-8 and rng.random() < np.exp(-dE/T)):
                cur_max = nm
                c = cur_max * 2.0 / N
                if c < best_c - 1e-13:
                    best_c, best_h, no_imp = c, h.copy(), 0
                else:
                    no_imp += 1
            else:
                h[oa:oa+Lb] = 1; h[za:za+Lb] = 0
        T *= cool
        if no_imp > stall:
            for _ in range(pp):
                ones = np.flatnonzero(h)
                zeros = np.flatnonzero(1 - h)
                if len(ones)<1 or len(zeros)<1: break
                a = int(ones[rng.integers(len(ones))])
                b = int(zeros[rng.integers(len(zeros))])
                h[a], h[b] = 0, 1
            cur_max = float(np.max(fft_corr(h)))
            T = max(T, T0*0.25)
            no_imp = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    hB = project_binary(h0, N0 // 2)
    best_c5 = compute_c5(hB)
    best_h = hB.copy()
    Nn = N0
    # Phase 1: Multi-start exploration
    inits = [hB.copy()]
    x = np.arange(Nn)
    v1 = 0.9*np.sin(2*np.pi*3*x/Nn) + 0.5*np.sin(2*np.pi*7*x/Nn) + 0.4*np.cos(2*np.pi*11*x/Nn)
    inits.append(project_binary((v1 > np.median(v1)).astype(float), Nn//2))
    v2 = np.sin(2*np.pi*5*x/Nn + 0.7) + 0.6*np.sin(2*np.pi*13*x/Nn + 1.2) + 0.3*np.cos(2*np.pi*2*x/Nn)
    inits.append(project_binary((v2 > np.median(v2)).astype(float), Nn//2))
    hr = np.zeros(Nn)
    hr[rng.choice(Nn, Nn//2, replace=False)] = 1
    inits.append(hr)
    h_alt = project_binary((np.arange(Nn) % 7 < 3).astype(float), Nn//2)
    inits.append(h_alt)
    hp = 397
    h_w = np.zeros(Nn)
    h_w[(np.arange(Nn)**2 % hp) < hp*0.48] = 1
    inits.append(project_binary(h_w, Nn//2))
    for hi in inits:
        if time.time() > deadline - 500:
            break
        d = min(deadline - 480, time.time() + 55)
        hv, cv = anneal(hi, Nn, d, rng, T0=0.06, T1=1e-4, max_iters=120000,
                       sp=0.5, dp=0.2, bp=0.3, bsz=(2,3,4,5,6,8), stall=3500, pp=6)
        if cv < best_c5:
            best_c5, best_h = cv, hv
    # Phase 2: Long intensive anneal
    if time.time() < deadline - 280:
        d2 = min(deadline - 260, time.time() + 320)
        h2, c2 = anneal(best_h, len(best_h), d2, rng, T0=0.03, T1=1e-6,
                       max_iters=600000, sp=0.55, dp=0.18, bp=0.27,
                       bsz=(2,3,4,5,7), stall=6000, pp=10)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    # Phase 3: Try nearby N values
    for dN in [2, -2, 4, -4, 6, -6, 8, -8]:
        if time.time() > deadline - 100:
            break
        Na = len(best_h) + dN
        if Na < 120 or Na % 2 != 0:
            continue
        pos = np.linspace(0, len(best_h), Na)
        idx = np.clip(pos[:-1].astype(int), 0, len(best_h)-1)
        h_a = project_binary(best_h[idx], Na//2)
        d3 = min(deadline - 60, time.time() + 30)
        h3, c3 = anneal(h_a, Na, d3, rng, T0=0.04, T1=1e-5,
                       max_iters=80000, sp=0.5, dp=0.2, bp=0.3, stall=2500, pp=5)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
    # Phase 4: Final fine-grained polish (mostly 1-swaps, low temp)
    NF = len(best_h)
    hF = project_binary(best_h, NF//2)
    d4 = deadline - 1
    hF, c4 = anneal(hF, NF, d4, rng, T0=0.006, T1=1e-8, max_iters=800000,
                   sp=0.82, dp=0.12, bp=0.06, bsz=(2,3), stall=9000, pp=3)
    if c4 < best_c5:
        best_c5, best_h = c4, hF
    hF = project_binary(best_h, len(best_h)//2)
    cF = compute_c5(hF)
    if cF < best_c5:
        best_c5 = cF
    return hF, best_c5, len(hF)
