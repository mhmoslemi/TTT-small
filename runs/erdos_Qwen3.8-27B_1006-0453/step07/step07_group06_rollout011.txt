def lin_grad(L, h, N):

        g[:] = 1.0 - 2.0 * h

        g[:N - L] -= h[L:N]
    else:

        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

    c = fft_corr(h)
    mx = c.max()
    e = np.exp(np.clip(T * (c - mx), -50, 0))
    w = e / (e.sum() + 1e-30)
    val = mx + np.log(e.sum() + 1e-30) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * lin_grad(int(i - ctr), h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

    if time.time() < dl - 90:
        d7 = min(dl - 60, time.time() + 60)
        hb7, c7 = bin_2swap(best_h, N0, d7, max_iters=350, ntri=14, temp0=0.003)
        c7v = c7 * (2.0 / N0)
        if c7v < best_c5:
            best_c5, best_h = c7v, hb7.copy()

    # Phase 8: Multi-start binary (4 random perturbations)
    if time.time() < dl - 60:
        for s in range(4):
            if time.time() > dl - 30:
                break
            hp = best_h + np.random.randn(N0) * 0.04
            hp = make_binary(hp, N0)
            d8 = min(dl - 20, time.time() + 25)
            hpr, cpr = bin_search(hp, N0, d8, max_iters=150, ntri=25, temp0=0.002)
            cprv = cpr * (2.0 / N0)
            if cprv < best_c5:
                best_c5, best_h = cprv, hpr.copy()

    # Phase 9: Final L-BFGS polish
    if time.time() < dl - 30:
        d9 = min(dl - 15, time.time() + 45)
        h9 = best_h + np.random.randn(N0) * 0.02
        h9 = proj01(h9, N0 / 2.0)
        hl9, cl9 = lbfgs_smooth(h9, N0, d9, T=40.0, max_outer=2)
        if cl9 < best_c5:
            best_c5, best_h = cl9, hl9.copy()

    # Phase 10: Final binary polish (long)
    if time.time() < dl - 5:
        d10 = dl - 1.0
        hb10, c10 = bin_search(best_h, N0, d10, max_iters=600, ntri=30, temp0=0.002)
        c10v = c10 * (2.0 / N0)
        if c10v < best_c5:
            best_c5, best_h = c10v, hb10.copy()

    # Final evaluation
    fb = make_binary(best_h, N0)
    fc5b = compute_c5(fb, N0)
    fc5c = compute_c5(best_h, N0)
    if fc5b < fc5c:
        best_h, best_c5 = fb, fc5b
    else:
        best_c5 = fc5c
    return best_h, float(best_c5), len(best_h)
