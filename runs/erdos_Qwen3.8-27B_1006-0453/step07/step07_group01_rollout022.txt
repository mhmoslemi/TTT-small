Let me also think about the gradient more carefully. The objective is:
f(h) = max_k g_k(h) where g_k(h) = sum_i h[i]*(1-h[i+k]) * dx

For the soft-max approximation:
f(h) ≈ (1/T) * log(sum_k exp(T * g_k(h)))

The gradient:
∂f/∂h[j] = sum_k w_k * ∂g_k/∂h[j]

where w_k = exp(T*g_k) / sum_j exp(T*g_j)

And ∂g_k/∂h[j]:
g_k = sum_i h[i]*(1-h[i+k])

For k=0: g_0 = sum_i h[i]*(1-h[i]) = sum_i h[i]*(1-h[i])
∂g_0/∂h[j] = (1-2*h[j])

For k>0: g_k = sum_i h[i]*(1-h[i+k]) where i ranges so that both i and i+k are in [0,N-1]
∂g_k/∂h[j] = sum over all (i,i+k) pairs involving j
= (1-h[j-k]) - h[j+k] (approximately, with boundary handling)

Wait, let me be more careful. g_k = sum_i h[i]*(1-h[i+k]) for i+k < N.

∂g_k/∂h[j] = sum_i [∂h[i]/∂h[j] * (1-h[i+k]) + h[i] * ∂(1-h[i+k])/∂h[j]]
= (1-h[j+k]) * [j+k<N] - h[j-k] * [j-k>=0]

So the gradient is:
∂g_k/∂h[j] = (1-h[j+k])*[j+k<N] - h[j-k]*[j-k>=0]

This is simpler than what the previous code does! Let me use this.

    """Compute new max corr after swapping h[p]=1->0, h[q]=0->1."""
    center = N - 1
    delta = np.zeros(2*N-1)
    
    # p as first: s in [-p, N-1-p], delta = -1 + h[p+s]
    s_lo = max(-N+1, -p)
    s_hi = min(N-1, N-1-p)
    if s_lo <= s_hi:
        s_arr = np.arange(s_lo, s_hi+1)
        idx = center + s_arr
        delta[idx] += -1.0 + h[p + s_arr]
    
    # p as second: s in [p-N+1, p], delta = h'[p-s]
    s_lo = max(-N+1, p-N+1)
    s_hi = min(N-1, p)
    if s_lo <= s_hi:
        s_arr = np.arange(s_lo, s_hi+1)
        idx = center + s_arr
        # h'[p-s] = h[p-s] except when p-s==q
        vals = h[p - s_arr]
        # fix where p-s==q: p-s_arr==q → s_arr==p-q
        mask = (p - s_arr) == q
        vals = vals.copy()
        vals[mask] = 1.0
        delta[idx] += vals

    s_lo = max(-N+1, -q)

Wait, but this is O(N) per swap (due to the vectorized adds). For N=1140, that's about 2000 numpy operations per swap. In practice, numpy will do this in maybe 5-10μs.

Actually, I realize the adds are sparse - each one only affects a contiguous range of length N. So it's 4 array operations of length N. That's maybe 10μs total. So 100 swaps per ms, or 100,000 per second. In 100 seconds, 10 million swaps. That's more than enough.

But wait, for the "first improvement" or "steepest descent" approach, I'd need to evaluate ALL possible swaps, which is O(N²) swap evaluations each O(N), for total O(N³) = 1140³ ≈ 1.5 * 10⁹. That's too slow.

For local search, I'll use a random sampling approach:
- Each iteration, pick a random (p,q) pair
- Evaluate the swap
- Accept if it improves (or with probability for SA)

This is O(1) swaps per iteration, each O(N). In 100 seconds at 10μs per swap, that's 10 million iterations. Should be enough for SA to converge.

Let me now write the full solution:

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, T, k_top):
    idx = np.argpartition(corr, -k_top)[-k_top:]
    c_top = corr[idx]
    cmax = c_top.max()
    w = np.exp(T*(c_top - cmax))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0*h)
        elif s > 0:
            grad[:N-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:N-s]
        else:
            M = -s
            grad[M:] += wi * (1.0 - h[:N-M])
            grad[:N-M] -= wi * h[M:]
    return grad

def continuous_opt(h_init, N, deadline, T=80.0, lr0=0.015, k_top=60):
    h = h_init.copy()
    target = N/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    beta = 0.9
    m = np.zeros(N)
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        c5 = compute_c5(y)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, T, k_top)
        h_new = project_sum(y - lr*grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
        if it % 200 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k_top, len(cs)-1)]
            T = max(20.0, min(250.0, T * (1.08 if gap < 0.02 else 0.93)))
    return best_h, best_c5

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        perm = rng.permutation(ones)
        h_bin[perm[:n1-target]] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        perm = rng.permutation(zeros)
        h_bin[perm[:target-n1]] = 1
    return h_bin

def binary_sa(h_init, N, deadline, T0=3.0, T1=0.05, max_iter=None):
    h = ensure_binary(h_init, N)
    corr = fft_corr(h)
    best_P = float(corr.max())
    best_h = h.copy()
    n_iter = 0
    
    if max_iter is None:
        max_iter = 5000000
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max_iter
        T = T0 * (T1/T0)**frac
        
        # Pick random p (one) and q (zero)
        p = np.random.randint(N)
        while h[p] != 1:
            p = np.random.randint(N)
        q = np.random.randint(N)
        while h[q] != 0:
            q = np.random.randint(N)
        
        # Compute delta
        new_P = eval_swap(corr, h, p, q, N)
        delta = new_P - float(corr.max())
        
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
            curr_P = float(corr.max())
            if curr_P < best_P - 1e-12:
                best_P = curr_P
                best_h = h.copy()
    
    return best_h, best_P * (2.0/N)

def eval_swap(corr, h, p, q, N):

        idx = center + s_arr
        delta[idx] += -1.0 + h[p + s_arr]
    
    # p as second: s in [p-N+1, p]
    s_lo = max(-N+1, p-N+1)
    s_hi = min(N-1, p)

        s_arr = np.arange(s_lo, s_hi+1)

        vals = h[p - s_arr].copy()
        mask = (p - s_arr) == q
        if mask.any():
            vals[mask] = 1.0

        vals = h[q + s_arr].copy()
        mask = (q + s_arr) == p
        if mask.any():
            vals[mask] = 0.0
        delta[idx] += 1.0 - vals
    
    # q as second: s in [q-N+1, q]
    s_lo = max(-N+1, q-N+1)
