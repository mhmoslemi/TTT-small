def compute_flip_deltas(h):
    N = len(h)
    L = 2*N - 1
    s = 2.0*h - 1.0
    fds = np.zeros((N, L))
    for i in range(N):
        # Part 1: from h change, m ∈ [i, i+N-1]
        # fds[i, m] += (-s[i]) * (1-h)[i-m+N-1]
        m_min, m_max = i, i+N-1
        indices = np.arange(m_min, m_max+1)
        g_indices = i - indices + N - 1  # = i-m+N-1
        fds[i, indices] += (-s[i]) * (1.0 - h[g_indices])
        
        # Part 2: from (1-h) change, m ∈ [N-1-i, N-1+i]
        m_min2, m_max2 = N-1-i, N-1+i
        indices2 = np.arange(m_min2, m_max2+1)
        h_indices = i + indices2 - N + 1  # = i+m-N+1
        fds[i, indices2] += s[i] * h[h_indices]
    
    return fds

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def make_binary(h, rng):
    N = len(h)
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[rng.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[rng.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def corr_and_grad(h, k_top=50):
    """Compute correlation and subgradient of max."""
    N = len(h)
    L = 2 * N - 1
    corr = fft_corr(h)
    # Subgradient: average over top-k contributing lags
    thr = corr.max() - 0.5
    active = np.where(corr > thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(N)
    for m in active:
        k = int(m - (N - 1))
        if k == 0:
            grad += 1.0 - 2.0 * h
        elif k > 0:
            # corr[m] = sum_i h[i+k]*(1-h[i]) for valid i
            # d/dh[j]: appears in h[i+k] when j=i+k, i=j-k → (1-h[j-k])
            #           appears in (1-h[i]) when j=i, i=j → -h[j+k]
            grad[k:] += (1.0 - h[:N-k])
            grad[:N-k] -= h[k:]
        else:
            M = -k
            # corr[m] = sum_i h[i]*(1-h[i+M]) for valid i
            grad[:N-M] -= h[M:]
            grad[M:] += (1.0 - h[:N-M])
    grad /= len(active)
    return corr, grad

    N = len(h0)
    h = h0.astype(float).copy()
    m = np.zeros(N)
    beta = 0.84
    best_c = compute_c5(h)
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        y = h + m
        corr, grad = corr_and_grad(y, k_top)
        pk = float(corr.max()) * (2.0 / N)

            best_h = h.copy()
        # Projected gradient step
