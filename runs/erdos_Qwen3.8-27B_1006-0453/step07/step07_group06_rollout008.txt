import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    """Gradient of corr[L] with respect to h."""
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    # Log-sum-exp
    shifted = T * (c - mx)
    e = np.exp(shifted)
    es = e.sum()
    val = mx + np.log(es) / T
    # Gradient
    w = e / es
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * lin_grad(i - ctr, h, N)
    # Constraint penalty
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_round(h, N, deadline, T, lam, maxiter):
    target = float(N) // 2
    res = sp_min(smooth_obj, h, args=(T, target, lam),
                 jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                 options={'maxiter': maxiter, 'maxfun': maxiter * 3, 'ftol': 1e-15})
    return proj01(res.x, target)

def bin_local(h, N, deadline, ntrials=30, max_rounds=2000, anneal_rate=0.998):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_c = np.max(fft_corr(h)) * (2.0 / N)
    best_h = h.copy()
    temp = 0.002
    stall = 0
    for it in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        corr = fft_corr(h)
        P = np.max(corr)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = -np.inf, -1, -1
        for _ in range(ntrials):
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            h[i], h[j] = 0.0, 1.0
            nP = np.max(fft_corr(h))
            d = P - nP
            if d > bd:
                bd, bi, bj = d, i, j
            h[i], h[j] = 1.0, 0.0
        if bi >= 0:
            # Accept with probability
            if bd >= 0 or np.random.random() < np.exp(bd / max(temp, 1e-10)):
                h[bi], h[bj] = 0.0, 1.0
                nc = np.max(fft_corr(h))
                if nc < P:
                    if nc * (2.0/N) < best_c:
                        best_c = nc * (2.0/N)
                        best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        temp *= anneal_rate
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    shifted = T * (c - mx)
    e = np.exp(shifted)
    es = e.sum()
    val = mx + np.log(es) / T
    w = e / es
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def active_set_obj(h, active, N, lam):
    cc = fft_corr(h)
    s = cc[active].sum()
    g = np.zeros(N)
    ctr = N - 1
    for ai in active:
        g += lin_grad(int(ai - ctr), h, N)
    sg = h.sum() - N / 2.0
    s += lam * sg * sg
    g += 2.0 * lam * sg * np.ones(N)
    return s, g

    # Phase 5: Refine continuous from current best (blend)
    if time.time() < dl - 80:
        h5 = best_h + np.random.randn(best_N) * 0.02
        h5 = proj01(h5, best_N / 2.0)
        h5 = lbfgs_step(h5, dl - 70, 4, 20.0, 100)
        h5, c5 = active_refine(h5, best_N, min(dl - 60, time.time() + 80), n_rounds=25, tol=0.4)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Try different N values
    if time.time() < dl - 50:
        x_b = np.linspace(0, 2, best_N, endpoint=False)
        for Nt in [300, 340, 420, 460, 500]:
            if time.time() > dl - 40:
                break
            if Nt % 2:
                Nt += 1
            if Nt == best_N:
                continue
            xn = np.linspace(0, 2, Nt, endpoint=False)
            ht = np.interp(xn, x_b, best_h)
            ht = proj01(ht, Nt / 2.0)
            ht = lbfgs_step(ht, dl - 35, 10, 20.0, 60)
            ht, ct = active_refine(ht, Nt, min(dl - 35, time.time() + 20), n_rounds=8, tol=0.6)
            if ct < best_c5:
                best_c5, best_h, best_N = ct, ht.copy(), Nt

    # Phase 7: Final ultra-fine active set
    if time.time() < dl - 15:
        h7, c7 = active_refine(best_h, best_N, min(dl - 5, time.time() + 60), n_rounds=30, tol=0.3)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Phase 8: Final L-BFGS polish
    if time.time() < dl - 3:
        h8 = best_h + np.random.randn(best_N) * 0.008
        h8 = proj01(h8, best_N / 2.0)
        h8 = lbfgs_step(h8, dl - 2, 3, 25.0, 100)
        c8 = compute_c5(h8, best_N)
        if c8 < best_c5:
            best_c5, best_h = c8, h8.copy()

    # Final evaluation
    fc5 = compute_c5(best_h, best_N)
    fb = (best_h > 0.5).astype(float)
    n1 = int(fb.sum())
    if n1 > best_N // 2:
        idx = np.where(fb == 1)[0]
        fb[np.random.choice(idx, n1 - best_N // 2, replace=False)] = 0
    elif n1 < best_N // 2:
        idx = np.where(fb == 0)[0]
        fb[np.random.choice(idx, best_N // 2 - n1, replace=False)] = 1
    fc5b = compute_c5(fb, best_N)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b

    return best_h, fc5, best_N
