        r = rng.random()
        if r < 0.7:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            new_h = h.copy()
            new_h[p] = 0.0
            new_h[q] = 1.0
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            p1 = int(ones[rng.randint(len(ones))])
            p2 = int(ones[rng.randint(len(ones))])
            mask = np.ones(N, dtype=bool)
            mask[p1] = False
            mask2 = zeros.copy()
            mask2 = mask2[(mask2 != p1) & (mask2 != p2)]
            if len(mask2) < 2:
                p = p1
                q = int(zeros[rng.randint(len(zeros))])
                new_h = h.copy()
                new_h[p] = 0.0
                new_h[q] = 1.0
            else:
                q1 = int(mask2[rng.randint(len(mask2))])
                q2 = int(mask2[rng.randint(len(mask2))])
                new_h = h.copy()
                new_h[p1] = 0.0
                new_h[p2] = 0.0
                new_h[q1] = 1.0
                new_h[q2] = 1.0
        new_P = float(np.max(fft_corr(new_h)))
        c_old = float(np.max(fft_corr(h)))
        delta = new_P - c_old
        if delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h = new_h
            if new_P < best_c - 1e-13:
                best_c = new_P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=50, lr0=0.022):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it * 0.00006)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick greedy at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 650, t0 + 20)
    h1, c5_1 = greedy_binary(h_bin, N0, d1, max_sweeps=150, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Nesterov continuous at N0
    d2 = min(deadline - 550, t0 + 80)
    hc = project_sum(h_bin + rng.randn(N0) * 0.09, N0 / 2.0)
    h2, c5_2 = nesterov_opt(hc, N0, d2, k=50, lr0=0.025)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Fast SA at N0 (long, with 2-bit moves)
    h_sa = ensure_sum(best_h, N0)
    d3 = min(deadline - 400, t0 + 220)
    h3, c5_3 = sa_fast(h_sa, N0, d3, T0=4.0, Tcool=0.9978, T_min=0.012,
                       rng=rng, kick_every=250, kick_size=6)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Greedy polish at N0
    if time.time() < deadline - 320:
        h_g = ensure_sum(best_h, N0)
        d4 = min(deadline - 300, t0 + 270)
        h4, c5_4 = greedy_binary(h_g, N0, d4, max_sweeps=300, sf=0.9)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Second Nesterov run from perturbed init
    if time.time() < deadline - 250:
        hc2 = project_sum(ensure_sum(best_h, N0) + rng.randn(N0) * 0.06, N0 / 2.0)
        d5 = min(deadline - 200, t0 + 350)
        h5, c5_5 = nesterov_opt(hc2, N0, d5, k=65, lr0=0.016)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: SA at 2*N0 (upsampled)
    N1 = 2 * N0
    if time.time() < deadline - 200:
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d6 = min(deadline - 150, t0 + 470)
        h6, c5_6 = sa_fast(h_up, N1, d6, T0=3.5, Tcool=0.9980, T_min=0.02,
                           rng=rng, kick_every=300, kick_size=5)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Greedy at N1
    if time.time() < deadline - 100:
        N_cur = len(best_h)
        h_g2 = ensure_sum(best_h, N_cur)
        d7 = min(deadline - 70, t0 + 530)
        sf7 = 1.0 if N_cur <= 800 else 0.45
        h7, c5_7 = greedy_binary(h_g2, N_cur, d7, max_sweeps=250, sf=sf7)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Nesterov at N1 or N0
    if time.time() < deadline - 40:
        N_fin = len(best_h)
        hc3 = project_sum(best_h + rng.randn(N_fin) * 0.03, N_fin / 2.0)
        d8 = deadline - 5
        k8 = 55 if N_fin <= 800 else 90
        lr8 = 0.018 if N_fin <= 800 else 0.010
        h8, c5_8 = nesterov_opt(hc3, N_fin, d8, k=k8, lr0=lr8)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
