        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_opt(h, N, deadline, T=30.0, max_outer=4):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-15})
        h = proj01(res.x, target)
        v = smooth_obj(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.5
    return best_h, best_v * (2.0 / N)

def bin_search(h, N, deadline, max_iters=600, ntri=30, anneal=0.0, directed=8):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    corr = fft_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stall = 0
    temp = anneal
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bd, bi, bj = 0.0, -1, -1
        # Directed swaps: target worst shifts
        worst_idx = np.argsort(corr)[-directed:]
        for wi in worst_idx:
            s = int(wi - (N - 1))
            # For shift s, c[s] = sum h[n+s]*(1-h[n])
            # Swap i(1->0), j(0->1) changes c[s] by:
            # -g[i-s]+g[j-s]+h[i+s]-h[j+s] (with bounds)
            # We want most negative change
            best_d = 0.0
            best_oi, best_zj = -1, -1
            for _ in range(6):
                oi = int(ones[np.random.randint(len(ones))])
                zj = int(zeros[np.random.randint(len(zeros))])
                d = 0.0
                if 0 <= oi - s < N:
                    d -= (1.0 - h[oi - s])
                if 0 <= zj - s < N:
                    d += (1.0 - h[zj - s])
                if 0 <= oi + s < N:
                    d += h[oi + s]
                if 0 <= zj + s < N:
                    d -= h[zj + s]
                if d < best_d - temp:
                    best_d, best_oi, best_zj = d, oi, zj
            if best_oi >= 0 and best_d < bd:
                bd, bi, bj = best_d, best_oi, best_zj
        # Random swaps
        for _ in range(ntri):
            oi = int(ones[np.random.randint(len(ones))])
            zj = int(zeros[np.random.randint(len(zeros))])
            h[oi], h[zj] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd, bi, bj = d, oi, zj
            h[oi], h[zj] = 1.0, 0.0
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            corr = fft_corr(h)
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
            corr = fft_corr(h)
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))
    return best_h, best_c

def subgrad(h, N, deadline, alpha0=0.04, beta=0.92, atol=2e-5):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < 10000:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            alpha *= 0.95
            beta = min(0.98, 0.90 + 0.0001 * it)
    return best_h, best_pk * (2.0 / N)

def make_periodic(N, period, seed_off=0):
    h = np.zeros(N)
    half = period // 2
    base = np.zeros(period)
    base[:half] = 1.0
    reps = N // period
    for r in range(reps):
        off = (r * seed_off) % period
        seg = np.roll(base, off)
        h[r * period:(r + 1) * period] = seg
    rem = N % period
    if rem > 0:
        h[N - rem:] = np.roll(base, seed_off)[:rem]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Quick binary from initial
    d1 = min(dl - 500, t0 + 40)
    hb, _ = bin_search(best_h, N0, d1, max_iters=200, ntri=25, directed=6)
    c = compute_c5(hb, N0)
    if c < best_c5:
        best_c5, best_h = c, hb.copy()

    # Phase 2: Multi-start L-BFGS
    if time.time() < dl - 350:
        d2 = min(dl - 300, time.time() + 180)
        starts = [best_h.copy()]
        for p in [7, 11, 13, 19, 31]:
            if p < N0:
                starts.append(make_periodic(N0, p, np.random.randint(0, p)))
        starts.append(np.random.randint(0, 2, N0).astype(float))
        for r in range(5):
            starts.append(best_h + np.random.randn(N0) * (0.15 - 0.02 * r))
        for si, hs in enumerate(starts):
            if time.time() > d2:
                break
            if time.time() > dl - 300:
                break
            hs = proj01(hs, N0 / 2.0)
            per_dead = min(d2, time.time() + 30)
            hl, cl = lbfgs_opt(hs, N0, per_dead, T=20.0 + 5 * si, max_outer=3)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 3: Extended binary search with directed swaps
    if time.time() < dl - 250:
        d3 = min(dl - 200, time.time() + 200)
        hb3, _ = bin_search(best_h, N0, d3, max_iters=1500, ntri=20, directed=10, anneal=0.002)
        c3 = compute_c5(hb3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, hb3.copy()

    # Phase 4: Subgradient from perturbed best
    if time.time() < dl - 180:
        d4 = min(dl - 140, time.time() + 100)
        h4 = best_h + np.random.randn(N0) * 0.06
        h4 = proj01(h4, N0 / 2.0)
        hs4, cs4 = subgrad(h4, N0, d4, alpha0=0.03, beta=0.93)
        if cs4 < best_c5:
            best_c5, best_h = cs4, hs4.copy()

    # Phase 5: Multi-resolution
    if time.time() < dl - 100:
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        for Nt in [190, 250, 300, 420, 460, 540, 640, 760]:
            if time.time() > dl - 50:
                break
            if Nt % 2:
                Nt += 1
            xn = np.linspace(0, 2, Nt, endpoint=False)
            hi = np.interp(xn, x_b, best_h)
            ht = (hi > 0.5).astype(float)
            nh = Nt // 2
            n1 = int(ht.sum())
            if n1 > nh:
                ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0.0
            elif n1 < nh:
                ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1.0
            sd = min(dl - 40, time.time() + 30)
            hr, _ = bin_search(ht, Nt, sd, max_iters=300, ntri=20, directed=8)
            c5r = compute_c5(hr, Nt)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr.copy()
                x_b = xn

    # Phase 6: Final L-BFGS polish
    if time.time() < dl - 30:
        d6 = min(dl - 15, time.time() + 60)
        h6 = best_h + np.random.randn(len(best_h)) * 0.04
        h6 = proj01(h6, len(best_h) / 2.0)
        hl6, cl6 = lbfgs_opt(h6, len(best_h), d6, T=40.0, max_outer=3)
        if cl6 < best_c5:
            best_c5, best_h = cl6, hl6.copy()

    # Phase 7: Final binary polish
    if time.time() < dl - 2.0:
        hf = (best_h > 0.5).astype(float)
        hf, _ = bin_search(hf, len(best_h), dl - 0.5, max_iters=300, ntri=25, directed=12)
        cf = compute_c5(hf, len(best_h))
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Final: compare continuous vs binary
    fc5 = compute_c5(best_h, len(best_h))
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, len(best_h))
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, len(best_h)
