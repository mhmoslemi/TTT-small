def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:

    lo, hi = -1.5, 1.5

        else:
            hi = mid

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def precompute_rem(h, N):
    L_full = 2 * N - 1
    ctr = N - 1
    one_minus_h = 1.0 - h
    rem = np.zeros((N, L_full))
    for p in range(N):
        lo1 = max(0, ctr - p)
        hi1 = min(L_full - 1, ctr - p + N - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1) - ctr + p
            rem[p, lo1:hi1+1] -= one_minus_h[idx]
        lo2 = max(0, p)
        hi2 = min(L_full - 1, p + N - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1) - ctr + p
            rem[p, lo2:hi2+1] += h[idx]
    return rem

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(np.clip(T * (c - mx), -50, 50))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    idxs = np.where(w > 1e-7)[0]
    om = 1.0 - h
    for idx in idxs:
        wi = w[idx]
        L = idx - ctr
        if L == 0:
            g += wi * (1.0 - 2.0 * h)
        elif L > 0:
            g[:N-L] += wi * om[L:]
            g[L:] -= wi * h[:N-L]
        else:
            M = -L
            g[M:] += wi * om[:N-M]
            g[:N-M] -= wi * h[M:]
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, N, deadline, T=30.0, max_outer=3, lam0=5.0):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lam = lam0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B',
                     bounds=[(0, 1)] * N,
                     options={'maxiter': 100, 'maxfun': 300, 'ftol': 1e-15})
        h = proj01(res.x, target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lam *= 2.0
    return best_h, best_c

def exact_binary(h, N, deadline):
    """Exact binary local search using precomputed effect vectors."""
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if n1 > nh:
            h[np.random.choice(ones, n1 - nh, replace=False)] = 0
        else:
            h[np.random.choice(zeros, nh - n1, replace=False)] = 1
    
    L_full = 2 * N - 1
    corr = fft_corr(h)
    curr_max = float(np.max(corr))
    best_h = h.copy()
    best_max = curr_max
    
    while time.time() < deadline:
        rem = precompute_rem(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improve = 1e-12
        best_p = -1
        best_q = -1
        
        for p in ones:
            base = corr + rem[p]
            # For each zero q: new_max = max(base - rem[q])
            diffs = base[None, :] - rem[zeros]
            new_maxs = diffs.max(axis=1)
            imp = curr_max - new_maxs
            bi = int(np.argmax(imp))
            if imp[bi] > best_improve:
                best_improve = imp[bi]
                best_p = int(p)
                best_q = int(zeros[bi])
        
        if best_p < 0:
            break  # local optimum
        
        h[best_p] = 0.0
        h[best_q] = 1.0
        corr = fft_corr(h)
        curr_max = float(np.max(corr))
        if curr_max < best_max:
            best_max = curr_max
            best_h = h.copy()
    
    return best_h, best_max * (2.0 / N)
