def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):

    lo, hi = -1.5, 1.5

        if c > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_sa(h, deadline, max_iters=10000, T0=0.04, T1=0.0003, n_crit=25, s_sz=45):
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    center = N - 1
    it = 0
    no_imp = 0
    while time.time() < deadline and it < max_iters:
        T = T0 * (T1 / T0) ** (float(it) / float(max_iters))
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:

                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            no_imp = 0

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        k_c = min(n_crit, max(10, len(corr) // 4))
        top_idx = np.argsort(corr)[-k_c:]

        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        cur_peak = float(np.max(corr))

        best_new = cur_c
        for oi in os_:
            d_oi = np.zeros(2 * N - 1)

                d = 0.0

def cont_opt(h, deadline, lr0=0.015, mom=0.9, k_lags=35):
    N = len(h)

    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)

            stale = 0
        n_lags = min(k_lags, len(corr) - 1)
        thresh = np.percentile(corr, 100.0 - 100.0 * n_lags / len(corr))
        sel = np.where(corr >= thresh)[0]

        grad = np.zeros(N)
        for m in sel:
            M = abs(int(m) - center)

        h = h_new
    return best_h, best_c
