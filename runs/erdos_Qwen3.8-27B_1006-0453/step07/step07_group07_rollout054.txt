def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1

def swap_delta(h, oi, zi, lags, corr):
    # Compute change in corr at given lags when h[oi] 1->0 and h[zi] 0->1
    N = len(h)
    center = N - 1
    deltas = np.zeros(len(lags))
    for i, L in enumerate(lags):
        d = 0.0
        # h[oi] 1->0 effect
        if 0 <= oi + L < N:
            d -= (1.0 - h[oi + L])  # was contributing (1-h[oi+L]), now removed
        if 0 <= oi - L < N:
            d += h[oi - L]  # was contributing h[oi-L]*(1-1)=0, now h[oi-L]*1
        # h[zi] 0->1 effect
        if 0 <= zi + L < N:
            d += (1.0 - h[zi + L])  # was 0, now (1-h[zi+L])
        if 0 <= zi - L < N:
            d -= h[zi - L]  # was h[zi-L]*1, now h[zi-L]*0
        deltas[i] = d
    return deltas

import numpy as np

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    Gh = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * Gh, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_lse(h, corr, beta=50.0, top_k=100):
    N = len(h)
    center = N - 1
    max_c = float(np.max(corr))
    exp_c = np.exp(beta * (corr - max_c))
    total = exp_c.sum()
    if total < 1e-300:
        return np.zeros(N)
    weights = exp_c / total
    order = np.argsort(weights)[-top_k:]
    grad = np.zeros(N)
    for m in order:
        w = weights[int(m)]
        if w < 1e-7:
            continue
        M = int(m) - center
        P = abs(M)
        if P == 0:
            grad += w * (1.0 - 2.0 * h)
        elif P < N:
            if M > 0:
                js = np.arange(0, N - P)
                grad[js] += w * (1.0 - h[js + P])
                js2 = np.arange(P, N)
                grad[js2] -= w * h[js2 - P]
            else:
                js = np.arange(P, N)
                grad[js] += w * (1.0 - h[js - P])
                js2 = np.arange(0, N - P)
                grad[js2] -= w * h[js2 + P]
    return grad

        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.85 * v + 0.15 * (h_new - h)
        h = h_new
    return best_h, best_c

        it += 1
    return best_h, best_c

def de_pop(pop, tgt, F=0.7, CR=0.8, deadline=None):
    npop = len(pop)
    n = len(pop[0])
    fits = [compute_c5(p) for p in pop]
    for _ in range(5000):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(npop):
            cand = np.random.choice([j for j in range(npop) if j != i], 3, replace=False)
            a, b, c2 = cand
            mutant = pop[i] + F * (pop[a] - pop[b]) + 0.12 * np.random.randn(n)
            cross = np.random.rand(n) < CR
            trial = np.where(cross, mutant, pop[i])
            trial = project(trial, tgt)
            f = compute_c5(trial)
            if f < fits[i]:
                pop[i] = trial
                fits[i] = f
        if min(fits) < 0.380:
            break
    bi = int(np.argmin(fits))
    return pop[bi], fits[bi]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: DE exploration
    d1 = min(deadline - 550, t0 + 180)
    pop = [project(h0 + np.random.randn(N0) * 0.18, N0 / 2.0) for _ in range(28)]
    pop[0] = h0.copy()
    de_h, de_c = de_pop(pop, N0 / 2.0, F=0.7, CR=0.8, deadline=d1)
    if de_c < best_c:
        best_c, best_h = de_c, de_h.copy()

    # Phase 2: LSE continuous optimization
    if time.time() < deadline - 400:
        d2 = min(deadline - 350, time.time() + 180)
        h2 = project(best_h + np.random.randn(N0) * 0.04, N0 / 2.0)
        h2, c2 = cont_opt(h2, d2, beta0=35.0, lr0=0.018, mom=0.86)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: Binary IL (main refinement)
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 280)
        hb = fix_binary(best_h)
        hb, c3 = binary_il(hb, d3, max_iters=12000, n_crit=35, s_sz=55)
        if c3 < best_c:
            best_c, best_h = c3, hb.copy()

    # Phase 4: Second continuous pass
    if time.time() < deadline - 180:
        d4 = min(deadline - 130, time.time() + 80)
        h4 = project(best_h + np.random.randn(N0) * 0.025, N0 / 2.0)
        h4, c4 = cont_opt(h4, d4, beta0=80.0, lr0=0.006, mom=0.91)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Binary IL with different params
    if time.time() < deadline - 100:
        d5 = min(deadline - 60, time.time() + 140)
        hb5 = fix_binary(best_h)
        hb5, c5v = binary_il(hb5, d5, max_iters=10000, n_crit=40, s_sz=65)
        if c5v < best_c:
            best_c, best_h = c5v, hb5.copy()

    # Phase 6: Try resolution change
    if time.time() < deadline - 50:
        for N_try in [400, 460]:
            if time.time() > deadline - 25:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, N_try, endpoint=False)
            h_int = np.interp(x_new, x_old, best_h)
            h_bt = fix_binary(h_int)
            d6 = min(deadline - 15, time.time() + 25)
            h_bt, c6 = binary_il(h_bt, d6, max_iters=3000, n_crit=30, s_sz=45)
            if c6 < best_c:
                best_c, best_h = c6, h_bt.copy()

    # Final polish
    if time.time() < deadline - 2:
        hB = fix_binary(best_h)
        hB, cB = binary_il(hB, deadline - 1, max_iters=5000, n_crit=35, s_sz=55)
        if cB < best_c:
            best_h, best_c = hB, cB

    final_c = compute_c5(best_h)
    hB2 = fix_binary(best_h)
    cB2 = compute_c5(hB2)
    if cB2 < final_c:
        best_h, final_c = hB2, cB2
    return best_h, final_c, len(best_h)
