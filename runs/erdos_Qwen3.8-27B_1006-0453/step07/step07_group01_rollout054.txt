def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0001)
        if it % 300 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(20.0, min(250.0, T * (0.9 if gap < 0.01 else 1.08)))
    return best_h, best_pk * (2.0 / N)

def compute_D(h, N):
    """Precompute the D matrix for swap evaluation."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
    return D

def binary_greedy_v2(h, N, deadline, max_sweeps=800, sample_frac=1.0, kick_size=4):
    """Greedy binary local search with adaptive kicks."""
    h = ensure_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(777)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        D = compute_D(h, N)
        Dz = D[zeros, :]
        
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(60, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            if stall > 15:
                # Adaptive kick: larger when more stuck
                n_kick = min(kick_size + stall // 10, 12)
                for _ in range(n_kick):
                    pp = ones[rng.randint(len(ones))]
                    qq = zeros[rng.randint(len(zeros))]
                    h[pp], h[qq] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    
    return best_h, best_c * (2.0 / N)

def binary_sa(h, N, deadline, T0=0.04, T_min=0.002, cool=0.9996):
    """Binary simulated annealing."""
    h = ensure_binary(h, N)
    rng = np.random.RandomState(42)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    curr_c = best_c
    T = T0
    no_improve = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        
        h[p] = 0
        h[q] = 1
        new_c = float(np.max(fft_corr(h)))
        delta = new_c - curr_c
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            curr_c = new_c
            if curr_c < best_c - 1e-13:
                best_c = curr_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[p] = 1
            h[q] = 0
            no_improve += 1
        
        T = max(T_min, T * cool)
        if no_improve > 800:
            T = T0 * 0.4
            no_improve = 0
            for _ in range(4):
                pp = ones[rng.randint(len(ones))]
                qq = zeros[rng.randint(len(zeros))]
                h[pp], h[qq] = 0, 1
    
    return best_h, best_c * (2.0 / N)

def targeted_perturb(h, c, N, rng, n_swaps=5):
    """Perturb based on which positions contribute to the max correlation."""
    max_idx = np.argmax(c)
    shift = int(max_idx - (N - 1))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # Prioritize swaps near the problematic shift
    if shift > 0:
        # Problem is at positive shift: too many 1s followed by 0s at distance shift
        # Try to flip 1s in the high part to 0s, and 0s in the low part to 1s
        p_region = ones[(ones >= N - shift)]
        q_region = zeros[(zeros < shift)]
    else:
        p_region = ones[(ones < -shift)]
        q_region = zeros[(zeros >= N + shift)]
    
    for _ in range(n_swaps):
        if len(p_region) > 0 and len(q_region) > 0:
            h[p_region[rng.randint(len(p_region))]] = 0
            h[q_region[rng.randint(len(q_region))]] = 1
        else:
            h[ones[rng.randint(len(ones))]] = 0
            h[zeros[rng.randint(len(zeros))]] = 1

            for _ in range(5):
                pp = ones[rng.randint(len(ones))]
                qq = zeros[rng.randint(len(zeros))]
                h[pp], h[qq] = 0, 1
    return best_h, best_c * (2.0 / N)

def targeted_perturb(h, c, N, rng, n_swaps=6):
    top_idx = np.argpartition(c, -8)[-8:]
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_swaps):
        ti = top_idx[rng.randint(len(top_idx))]
        shift = int(ti - (N - 1))
        if shift > 0:
            p_reg = ones[(ones >= N - shift) & (ones < N)]
            q_reg = zeros[(zeros >= 0) & (zeros < shift)]
        elif shift < 0:
            p_reg = ones[(ones >= 0) & (ones < -shift)]
            q_reg = zeros[(zeros >= N + shift) & (zeros < N)]
        else:
            p_reg = ones
            q_reg = zeros
        p_r = p_reg if len(p_reg) > 0 else ones
        q_r = q_reg if len(q_reg) > 0 else zeros
        h[p_r[rng.randint(len(p_r))]] = 0
        h[q_r[rng.randint(len(q_r))]] = 1

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Gradient refinement (short, multiple T)
    for T_val in [60.0, 150.0]:
        if time.time() > deadline - 500:
            break
        d1 = min(deadline - 450, time.time() + 35)
        hc = h0.copy() + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        h_r, c5_r = nesterov_opt(hc, N0, d1, k=45, T=T_val, lr0=0.014)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()

    # Phase 2: Main binary greedy from best
    h_bin = ensure_binary(best_h, N0)
    d2 = min(deadline - 350, t0 + 280)
    h_br, c5_br = binary_greedy(h_bin, N0, d2, max_sweeps=1200, sample_frac=1.0, kick_size=5)
    if c5_br < best_c5:
        best_c5, best_h = c5_br, h_br.copy()
        h_bin = ensure_binary(best_h, N0)

    # Phase 3: SA with different schedules
    if time.time() < deadline - 200:
        d3 = min(deadline - 160, time.time() + 100)
        h_sa, c5_sa = binary_sa(h_bin, N0, d3, T0=0.035, T_min=0.003, cool=0.9997, seed=888)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()
            h_bin = ensure_binary(best_h, N0)

    # Phase 4: Targeted perturbation + short greedy (multiple kicks)
    if time.time() < deadline - 100:
        rng_t = np.random.RandomState(500)
        for kick_i in range(6):
            if time.time() > deadline - 40:
                break
            h_t = h_bin.copy()
            c_t = fft_corr(h_t)
            targeted_perturb(h_t, c_t, N0, rng_t, n_swaps=7)
            d4 = min(deadline - 30, time.time() + 22)
            h_t2, c5_t = binary_greedy(h_t, N0, d4, max_sweeps=250, sample_frac=0.85, kick_size=4, seed=200+kick_i)
            if c5_t < best_c5:
                best_c5, best_h = c5_t, h_t2.copy()
                h_bin = ensure_binary(best_h, N0)

    # Phase 5: Second SA from perturbed best
    if time.time() < deadline - 60:
        h_sa2 = ensure_binary(best_h, N0, seed=300)
        rng_p = np.random.RandomState(400)
        c5_c = fft_corr(h_sa2)
        targeted_perturb(h_sa2, c5_c, N0, rng_p, n_swaps=4)
        d5 = min(deadline - 30, time.time() + 40)
        h_sa2r, c5_sa2 = binary_sa(h_sa2, N0, d5, T0=0.025, T_min=0.002, cool=0.9996, seed=999)
        if c5_sa2 < best_c5:
            best_c5, best_h = c5_sa2, h_sa2r.copy()

    # Phase 6: Final greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 15:
        hF = ensure_binary(best_h, N_fin, seed=900)
        d6 = min(deadline - 3, time.time() + 30)
        hF2, c5F = binary_greedy(hF, N_fin, d6, max_sweeps=600, sample_frac=1.0, kick_size=3, seed=1111)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Phase 7: Quick continuous micro-refinement
    if time.time() < deadline - 5:
        hFc = best_h.copy() + np.random.randn(len(best_h)) * 0.015
        hFc = project_sum(hFc, len(best_h) / 2.0)
        d7 = min(deadline - 2, time.time() + 12)
        h7, c5_7 = nesterov_opt(hFc, len(best_h), d7, k=50, T=100.0, lr0=0.006)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
