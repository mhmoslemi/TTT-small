Alternative: compute the change analytically. For a swap (i: 1→0, j: 0→1):

The change in c[m] for the full correlation:
- Removing 1 at i: 
  - In the "h" part: affects c[m] where the h-index is i. h[n+k] = h[i] when n = i-k. Valid for n∈[0,N-1], so k = i-n, n = i-k.

    Wait, (1-h[i]) changes from 0 to 1. So the term becomes h[n+k]*1 instead of h[n+k]*0.
    Change: +h[i+k] for each valid k (0 ≤ i ≤ N-1, 0 ≤ i+k ≤ N-1)

- Adding 1 at j:
  - In the "h" part: h[n+k] = h[j] when n = j-k. 
    Change: +(1-h[j-k]) for valid k
  - In the "(1-h)" part: (1-h[j]) changes from 1 to 0.
    Change: -h[j+k] for valid k

So the total change in c[m] (for lag k = m-(N-1)) is:
Δc[k] = -(1-h[i-k]) + h[i+k] + (1-h[j-k]) - h[j+k] (for valid indices)

Where each term is 0 if the indices are out of range.

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_smooth(h, N, deadline, T=30.0, max_outer=4):
    h = h.astype(float).copy()
    target = float(N) / 2
    best_h = h.copy()
    best_v = smooth_obj_and_grad(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_and_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-15})
        h = proj01(res.x, target)
        v = smooth_obj_and_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 3.0
    return best_h, best_v * (2.0 / N)

def score_swap(h, i, j, corr, N):
    """Compute new max corr after swapping i(1->0) and j(0->1)."""
    if h[i] < 0.5 or h[j] > 0.5:
        return np.inf
    nc = corr.copy()
    ctr = N - 1
    for L in range(-N + 1, N):
        if L == 0:
            nc[ctr] += (1 - h[i]) * -1 + h[j] * 0  # simplified
            continue
        # effect of removing 1 at i
        if 0 <= i + L < N:
            nc[ctr + L] -= (1.0 - h[i + L])
        if 0 <= i - L < N:
            nc[ctr + L] += h[i - L]
        # effect of adding 1 at j
        if 0 <= j + L < N:
            nc[ctr + L] += (1.0 - h[j + L])
        if 0 <= j - L < N:
            nc[ctr + L] -= h[j - L]
    return float(np.max(nc))

        active = np.where(corr >= pk - tol)[0]

        # Build candidate swap list from gradient info

    phi = (1.0 + np.sqrt(5.0)) / 2.0

            h[j] = 1
    return h

        rem -= s
    pos = 0

        gap = rng.randint(2, 8)

    elif n1 > nh:

        c4 = compute_c5(h4, N0)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Multi-resolution (150s)
    if time.time() < dl - 60:
        best_h, best_c5 = multi_res_search(best_h, best_c5, min(dl - 30, time.time() + 140))

    # Phase 6: Final continuous L-BFGS (60s)
    if time.time() < dl - 20:
        Nf = len(best_h)
        d6 = min(dl - 10, time.time() + 55)
        hf = best_h.copy()
        hl6, cl6 = lbfgs_smooth(hf, Nf, d6, T=50.0, max_outer=3)
        if cl6 < best_c5:
            best_c5, best_h = cl6, hl6.copy()

    # Phase 7: Final binary polish
    if time.time() < dl - 5:
        Nf = len(best_h)
        hf = (best_h > 0.5).astype(float)
        hf, _ = bin_search(hf, Nf, dl - 2, max_iters=100, ntri=20)
        cf = compute_c5(hf, Nf)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Final: pick better of continuous vs binary
    Nf = len(best_h)
    fc5 = compute_c5(best_h, Nf)
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, Nf)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, Nf
