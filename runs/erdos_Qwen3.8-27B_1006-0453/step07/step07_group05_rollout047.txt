import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def smooth_obj(corr, tk, p):
    c = len(corr) // 2
    v = corr.copy()
    v[c] = 0.0
    v = np.abs(v)
    idx = np.argpartition(v, -tk)[-tk:]
    return float(np.mean(v[idx] ** p)) ** (1.0 / p)
