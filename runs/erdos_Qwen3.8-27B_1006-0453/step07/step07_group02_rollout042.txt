def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1).copy()

        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final polish binary + nesterov
    if time.time() < deadline - 30:
        d6 = min(deadline - 20, time.time() + 30)
        hf, c5f = binary_ls(best_h, len(best_h), d6, info_per_shift=8, top_k=10, kick=25, kick_n=3)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    if time.time() < deadline - 5:
        hF = best_h + np.random.randn(len(best_h)) * 0.03
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 1, lr=0.015, k_top=70)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final binary check
    hbF = (best_h > 0.5).astype(float)
    c5B = compute_c5(hbF, len(hbF))
    if c5B < best_c5:
        best_h, best_c5 = hbF, c5B

    return best_h, best_c5, len(best_h)
