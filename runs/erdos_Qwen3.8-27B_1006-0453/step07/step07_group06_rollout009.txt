def softmax_obj(h, T, ctr, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    val = mx + np.log(e.sum()) / T
    w = e / e.sum()
    g = np.zeros(N)
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def anneal_lbfgs(h, N, deadline, T0=80.0, T_end=5.0, n_outer=10):
    h = h.astype(float).copy()
    target = N / 2.0
    ctr = N - 1
    best_h = h.copy()
    best_c = compute_c5(h)
    lam = 0.0
    T = T0
    ratio = (T_end / T0) ** (1.0 / max(1, n_outer))
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        T_curr = T
        def obj_g(x):
            return softmax_obj(x, T_curr, ctr, target, lam)
        res = sp_min(obj_g, h, jac=True, method='L-BFGS-B',
                     bounds=[(0, 1)] * N, options={'maxiter': 60, 'ftol': 1e-15, 'gtol': 1e-10})
        h = proj01(res.x, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        T *= ratio
        lam = min(lam * 2.0 + 0.5, 100.0)
    return best_h, best_c

def resample(h_src, Nt):
    x_src = np.linspace(0, 2, len(h_src), endpoint=False)
    x_new = np.linspace(0, 2, Nt, endpoint=False)
    h_new = np.interp(x_new, x_src, h_src)
    h_bin = (h_new > 0.5).astype(float)
    nh = Nt // 2
    n1 = int(h_bin.sum())
    if n1 > nh:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h_new[idx])[:n1-nh]
        h_bin[idx[order]] = 0
    elif n1 < nh:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(h_new[idx])[::-1][:nh-n1]
        h_bin[idx[order]] = 1
    return h_bin

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))
