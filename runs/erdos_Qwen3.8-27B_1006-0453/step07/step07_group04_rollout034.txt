def binary_sa(h0, N, deadline, T0=3.0, Tcool=0.994, T_min=0.05, max_sweeps=5000):
    h = (h0 > 0.5).astype(float).copy()
    n1 = int(h.sum())
    target_n1 = N // 2
    if n1 != target_n1:
        rng = np.random.RandomState(42)
        if n1 > target_n1:
            idx = rng.choice(np.where(h == 1)[0], n1 - target_n1, replace=False)
            h[idx] = 0
        else:
            idx = rng.choice(np.where(h == 0)[0], target_n1 - n1, replace=False)
            h[idx] = 1
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 250:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        p = np.random.choice(ones)
        q = np.random.choice(zeros)
        h[p], h[q] = 0.0, 1.0
        c2 = fft_corr(h)
        P2 = float(c2.max())
        delta = P2 - P
        if delta < 0 or np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)):
            pass
        else:
            h[p], h[q] = 1.0, 0.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def upsample_lin(h_old, N_old, N_new):
    x_old = np.linspace(0.0, 1.0, N_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coarse structure search at N=200
    if time.time() < deadline - 550:
        N_coarse = 200
        h_c = project_sum(upsample_lin(h0, N0, N_coarse), N_coarse)
        d1 = min(deadline - 530, t0 + 50)
        h1, c5_1 = adam_opt(h_c, N_coarse, d1, beta_start=3, beta_end=800, lr=0.012, max_iters=2500)
        h1_up = project_sum(upsample_lin(h1, N_coarse, N0), N0)
        d1b = min(deadline - 490, t0 + 110)
        h1b, c5_1b = topk_opt(h1_up, N0, d1b, top_k=20, lr=0.004, max_iters=2000)
        if c5_1b < best_c5:
            best_c5, best_h = c5_1b, h1b

    # Phase 2: Full Adam at N0
    d2 = min(deadline - 370, t0 + 300)
    h2, c5_2 = adam_opt(best_h, N0, d2, beta_start=8, beta_end=5000, lr=0.004, max_iters=4000)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Top-k focused
    d3 = min(deadline - 230, t0 + 490)
    h3, c5_3 = topk_opt(best_h, N0, d3, top_k=35, lr=0.0025, max_iters=3000)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Adam with tighter params
    d4 = min(deadline - 140, t0 + 620)
    h4, c5_4 = adam_opt(best_h, N0, d4, beta_start=30, beta_end=8000,
                        lr=0.002, b1=0.85, b2=0.95, max_iters=3000)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Binary SA
    d5 = min(deadline - 70, t0 + 820)
    h5, c5_5 = binary_sa(best_h, N0, d5, T0=2.5, Tcool=0.994, T_min=0.04, max_sweeps=4000)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Phase 6: Final topk polish
    if time.time() < deadline - 5:
        h6, c5_6 = topk_opt(best_h, N0, deadline - 1, top_k=15, lr=0.0015, max_iters=2000)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
