δ(q) = +(1-h[1-1])·[1≥1] - h[1+1]·[1≤2] = (1-h[0]) - h[2] = 0 - 1 = -1
Total δ(1) = 0 + (-1) = -1
c_1 was 1, new should be 0. Let me verify: h'=[0,1,1,0], c'_1 = h'[1]*v'[0]+h'[2]*v'[1]+h'[3]*v'[2] = 1*1+1*0+0*0 = 1.

But my formula says c'_1 = 1 + (-1) = 0. Contradiction!

The issue is again the interaction. When p+L = q (0+1=1), the two changes interact.

Specifically, for p change at L=1: the term at n=p=0 is h[0+1]*(1-h[0]) = h[1]*0 = 0 → h'[1]*1 = 1*1 = 1. Delta should be +1.
But my formula gives: h[p+L] = h[1] = 0 (old value). The formula uses the OLD h[1]=0, but after the swap, h[1]=1.

So the formula is correct for non-interacting swaps (where p+L≠q and q+L≠p and p-L≠q and q-L≠p), but fails when there's an interaction.

For the algorithm, I'll handle this by:
1. For single swaps: check if there's an interaction (p±L=q or q±L=p for any critical L). If not, use the formula. If yes, recompute from scratch.
2. For 2-swaps: always recompute from scratch (just compute the affected lags directly).

Actually, for N=380 and 30 critical lags, the probability of interaction for a random swap is about 4*30/380 ≈ 32%. That's too high to ignore.

Better approach: just recompute the max at critical lags directly after each candidate swap. For N=380 and K=30 lags, each recomputation takes about K*N = 11400 multiplications. For 50 candidate swaps per iteration, that's about 570K operations. Very fast.

Let me design the final algorithm:

```python
def max_at_critical(h, lags, N):
    """Compute max correlation at given lags."""
    max_val = 0.0
    for L in lags:
        L = int(L)
        if L > 0:
            if L < N:
                val = np.dot(h[L:], 1.0 - h[:N-L])
            else:
                val = 0.0
        elif L < 0:
            P = -L
            if P < N:
                val = np.dot(h[:N-P], 1.0 - h[P:])
            else:
                val = 0.0
        else:
            val = np.dot(h, 1.0 - h)
        if val > max_val:
            max_val = val
    return max_val
```

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def c5_val(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, rng):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        sel = rng.choice(idx, n1 - nh, replace=False)
        h[sel] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        sel = rng.choice(idx, nh - n1, replace=False)
        h[sel] = 1
    return h

def lag_vals(h, lags, N):
    vals = np.zeros(len(lags))
    for i, L in enumerate(lags):
        Li = int(L)
        if Li > 0 and Li < N:
            vals[i] = np.dot(h[Li:], 1.0 - h[:N - Li])
        elif Li < 0 and -Li < N:
            P = -Li
            vals[i] = np.dot(h[:N - P], 1.0 - h[P:])
        elif Li == 0:
            vals[i] = np.dot(h, 1.0 - h)
    return vals

def get_critical_lags(corr, N, k=40):
    center = N - 1
    idx = np.argsort(corr)[-k:]
    return (idx - center).astype(int)

def binary_search(h, deadline, rng, n_crit=35, max_iters=20000, s_sz1=30, s_sz2=25):
    N = len(h)
    h = to_binary(h.copy(), rng)
    corr = compute_corr(h)
    best_c = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    it = 0
    no_imp = 0
    T = 0.5
    T_min = 0.01
    cool = 0.9995
    while time.time() < deadline and it < max_iters:
        if it > 0 and it % 200 == 0:
            T = max(T * cool, T_min)
            corr = compute_corr(h)
            n_crit_now = min(n_crit, int(30 + 20 * np.exp(-it / 3000.0)))
        if it % 50 == 0:
            lags = get_critical_lags(corr, N, n_crit)
            cur_peak = float(lag_vals(h, lags, N).max())
            cur_c = cur_peak * (2.0 / N)
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        made_move = False
        # 1-swap
        n_try = min(s_sz1, len(ones), len(zeros))
        o_try = rng.choice(ones, n_try, replace=False)
        z_try = rng.choice(zeros, n_try, replace=False)
        best_d = 1e-8
        best_ov, best_zv = None, None
        for oi in o_try:
            for zi in z_try:
                h[oi] = 0.0
                h[zi] = 1.0
                pv = float(lag_vals(h, lags, N).max())
                h[oi] = 1.0
                h[zi] = 0.0
                if pv - cur_peak < -best_d:
                    best_d = cur_peak - pv
                    best_ov, best_zv = oi, zi
        if best_ov is not None:
            h[best_ov] = 0.0
            h[best_zv] = 1.0
            made_move = True
        elif rng.random() < 0.15 and T > T_min * 1.5:
            # random perturbation
            oi = int(rng.choice(ones))
            zi = int(rng.choice(zeros))
            h[oi] = 0.0
            h[zi] = 1.0
            made_move = True
        # 2-swap
        if not made_move and rng.random() < 0.3:
            if len(ones) >= 2 and len(zeros) >= 2:
                o2 = rng.choice(ones, 2, replace=False)
                z2 = rng.choice(zeros, 2, replace=False)
                h[o2[0]] = 0.0; h[o2[1]] = 0.0
                h[z2[0]] = 1.0; h[z2[1]] = 1.0
                pv2 = float(lag_vals(h, lags, N).max())
                if pv2 - cur_peak < -1e-8:
                    made_move = True
                else:
                    h[o2[0]] = 1.0; h[o2[1]] = 1.0
                    h[z2[0]] = 0.0; h[z2[1]] = 0.0
        if not made_move:
            no_imp += 1
            if no_imp > 500:
                if T > T_min:
                    oi = int(rng.choice(ones))
                    zi = int(rng.choice(zeros))
                    h[oi] = 0.0
                    h[zi] = 1.0
                else:
                    break
        it += 1
    return best_h, best_c

def cont_opt(h, deadline, k=40, lr0=0.02, mom=0.88):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = c5_val(h)
    lr = lr0
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 60:
            lr *= 0.8
            stale = 0
            if lr < 0.001:
                break
        lags = get_critical_lags(corr, N, k)
        g = np.zeros(N)
        for L in lags:
            Li = int(L)
            if Li > 0 and Li < N:
                g[Li:] += 1.0 - h[:N - Li]
                g[:N - Li] -= h[Li:]
            elif Li < 0 and -Li < N:
                P = -Li
                g[:N - P] += 1.0 - h[P:]
                g[P:] -= h[:N - P]
            else:
                g += 1.0 - 2.0 * h
        h_new = project_sum(h + mom * v - lr * g / len(lags), tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def interp_res(h, N_new, rng):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return to_binary(h_new, rng)
