```python
    return np.real(np.fft.irfft(H * G, L))

    nh = N // 2
```

```python
    C = np.zeros((N, L))
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h, kind='stable')
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def enforce_sum_binary(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target-n1]] = 1
    return h

def build_C(h, N):
    """C[i, n] = h[i+n] (if valid) + h[i-n] (if valid) using lag k = n-(N-1)"""
    L = 2 * N - 1
    k = np.arange(L) - (N - 1)  # lags: -(N-1), ..., 0, ..., N-1
    i = np.arange(N)
    C = np.zeros((N, L))
    # First term: h[i+k], valid when 0 <= i+k < N
    idx1 = i[:, None] + k[None, :]
    valid1 = (idx1 >= 0) & (idx1 < N)
    C[valid1] += h[idx1[valid1]]
    # Second term: h[i-k], valid when 0 <= i-k < N
    idx2 = i[:, None] - k[None, :]
    valid2 = (idx2 >= 0) & (idx2 < N)
    C[valid2] += h[idx2[valid2]]
    return C

def exhaustive_local_search(h, N, deadline, max_rounds=50):
    """Repeatedly try all single swaps until no improvement."""
    nh = N // 2
    h = project_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    for round_i in range(max_rounds):
        if time.time() > deadline:
            break
        C = build_C(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_dE = 0
        best_oi = None
        best_zi = None
        
        for oi in ones:
            G = corr + C[oi, :]
            # For all zi: new_max = max(G - C[zi, :])
            M = G[None, :] - C  # (N, L)
            new_maxs = M.max(axis=1)  # (N,)
            # Only consider valid zi
            mask = np.zeros(N, dtype=bool)
            mask[zeros] = True
            candidates = new_maxs[mask]
            valid_zeros = zeros
            for zi in valid_zeros:
                dE = new_maxs[zi] - cur_max
                if dE < best_dE - 1e-12:
                    best_dE = dE
                    best_oi = oi
                    best_zi = zi
        
        if best_oi is None:
            break  # Local optimum
        
        h[best_oi] = 0
        h[best_zi] = 1
        corr = fft_corr(h)
        cur_max = float(np.max(corr))
        if cur_max * 2.0 / N < best_c - 1e-14:
            best_c = cur_max * 2.0 / N
            best_h = h.copy()
    
    return best_h, best_c
```
