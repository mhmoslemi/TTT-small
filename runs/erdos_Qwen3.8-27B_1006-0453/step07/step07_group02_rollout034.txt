```python
    # Numerically stable soft max
    m = np.max(corr)
```

```python
import time

def compute_corr(h):
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def compute_C_row(h, N, idx):
    L = 2 * N - 1
    center = N - 1
    k = np.arange(-N+1, N)
    ip = idx + k  # positions for 1-h[idx+k]
    im = idx - k  # positions for h[idx-k]
    P = np.where((ip >= 0) & (ip < N), 1.0 - h[np.clip(ip, 0, N-1)], 0.0)
    M = np.where((im >= 0) & (im < N), h[np.clip(im, 0, N-1)], 0.0)

    elif s < target:

        h[np.random.choice(idx, target - s, replace=False)] = 1
    return h

def project(h, target):

    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
```

Actually, let me compute the gradient more carefully and efficiently.

For corr[ki] where ki corresponds to lag k = ki - (N-1):
corr[ki] = sum_i y[i] * (1-y[i+k]) (with valid indices)

d(corr[ki])/dy[j]:
- y[j] appears in y[i] when i=j: contributes (1-y[j+k]) if 0≤j+k<N
- y[j] appears in (1-y[i+k]) when i+k=j, i=j-k: contributes -y[j-k] if 0≤j-k<N

So: d(corr[ki])/dy[j] = (1-y[j+k])*1_{0≤j+k<N} - y[j-k]*1_{0≤j-k<N}

For a fixed k, this is a shifted version of (1-y) minus a shifted version of y.

grad[j] = sum_{ki in top} wt[ki] * [(1-y[j+k_i])*1_{0≤j+k_i<N} - y[j-k_i]*1_{0≤j-k_i<N}]

I can compute this as:
grad[j] = sum_{ki} wt[ki] * (1-y[j+k_i]) * 1_{valid} - sum_{ki} wt[ki] * y[j-k_i] * 1_{valid}

The first term: for each ki with lag k_i, add wt[ki] * (1-y[k_i : N-k_i]) to grad[0:N-k_i]... wait, the indexing is off.

Let me be very explicit. For lag k (positive):
- (1-y[j+k]) is valid for j = 0 to N-1-k
- So we add wt * (1-y[k:N]) to grad[0:N-k]

For lag k (negative, say k = -m where m > 0):
- (1-y[j-m]) is valid for j = m to N-1
- So we add wt * (1-y[0:N-m]) to grad[m:N]

Similarly for the -y[j-k] term:
For lag k (positive):
- y[j-k] is valid for j = k to N-1
- So we subtract wt * y[0:N-k] from grad[k:N]

For lag k (negative, k = -m):
- y[j+m] is valid for j = 0 to N-1-m
- So we subtract wt * y[m:N] from grad[0:N-m]

This is doable but tedious. For N=380 and ~50 active lags, a simple loop would work:

```python
def compute_softgrad(y, w, N, top_idx):

            grad[k:] -= wt * y[:N-k]
        else:

            # (1-y[j-m]) for j=m..N-1
            grad[m:] += wt * (1.0 - y[:N-m])
            # -y[j+m] for j=0..N-1-m

    return grad
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def compute_C_row(h, N, idx):
    k = np.arange(-N + 1, N)
    ip = idx + k
    im = idx - k
    P = np.where((ip >= 0) & (ip < N), 1.0 - h[np.clip(ip, 0, N - 1)], 0.0)
    M = np.where((im >= 0) & (im < N), h[np.clip(im, 0, N - 1)], 0.0)
    return P - M

def build_C(h, N):
    return np.array([compute_C_row(h, N, i) for i in range(N)])

def fix_sum(h, target):
    s = int(h.sum())
    if s > target:
        idx = np.where(h == 1)[0]
        if s - target <= len(idx):
            h[np.random.choice(idx, s - target, replace=False)] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        if target - s <= len(idx):
            h[np.random.choice(idx, target - s, replace=False)] = 1
    return h

def project(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def binary_search(h, N, deadline, seed, K=12, sample=60, kick_int=80, kick_sz=4):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    L = 2 * N - 1
    C = build_C(h, N)
    corr = compute_corr(h)
    best_c = float(np.max(corr))
    best_h = h.copy()
    stuck = 0

    while time.time() < deadline:
        top_k = np.argpartition(corr, -K)[-K:]
        cur_max = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        ns = min(len(ones), sample)
        nz = min(len(zeros), sample)
        os_ = ones if len(ones) <= sample else rng.choice(ones, sample, replace=False)
        zs = zeros if len(zeros) <= sample else rng.choice(zeros, sample, replace=False)

        Cak = C[os_][:, top_k]
        Cbk = C[zs][:, top_k]
        new_vals = corr[top_k][None, None, :] + Cbk[None, :, :] - Cak[:, None, :]
        new_max = new_vals.max(axis=2)
        mi = np.unravel_index(np.argmin(new_max), new_max.shape)
        a, b = int(os_[mi[0]]), int(zs_[mi[1]])
        test_corr = corr + C[b] - C[a]
        if float(np.max(test_corr)) < cur_max - 1e-10:
            h[a] = 0
            h[b] = 1
            C[a] = compute_C_row(h, N, a)
            C[b] = compute_C_row(h, N, b)
            corr = test_corr
            cm = float(np.max(corr))
            if cm < best_c - 1e-14:
                best_c = cm
                best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck >= kick_int:
                for _ in range(kick_sz):
                    pa = int(rng.choice(ones))
                    pb = int(rng.choice(zeros))
                    h[pa] = 0
                    h[pb] = 1
                    C[pa] = compute_C_row(h, N, pa)
                    C[pb] = compute_C_row(h, N, pb)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = compute_corr(h)
                stuck = 0
            elif stuck > kick_int * 6:
                break
    return best_h, best_c * (2.0 / N)

def softmax_grad(y, w, N, top_idx):
    center = N - 1
    grad = np.zeros(N)
    for idx in top_idx:
        k = int(idx - center)
        wt = w[idx]
        if wt < 1e-9:
            continue
        if k >= 0:
            grad[:N - k] += wt * (1.0 - y[k:])
            grad[k:] -= wt * y[:N - k]
        else:
            m = -k
            grad[m:] += wt * (1.0 - y[:N - m])
            grad[:N - m] -= wt * y[m:]
    return grad

def continuous_opt(h, N, deadline):
    target = N / 2.0
    h = project(h.copy(), target)
    corr = compute_corr(h)
    best_c = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    t_par = 30.0
    lr = 0.03
    beta = 0.9
    m = np.zeros(N)
    n_outer = 0
    while time.time() < deadline:
        y = h + m
        corr = compute_corr(y)
        max_c = float(np.max(corr))
        w = np.exp(t_par * (corr - max_c))
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        top_idx = np.where(w > w.max() * 0.005)[0]
        if len(top_idx) > 60:
            top_idx = np.where(w > 0)[0][np.argsort(w)[-60:]]
        grad = softmax_grad(y, w, N, top_idx)
        h_new = project(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        n_outer += 1
        if n_outer % 200 == 0:
            t_par *= 1.15
            lr *= 0.98
            beta = min(0.97, 0.85 + n_outer * 0.0001)
        if n_outer % 100 == 0:
            c_now = float(np.max(compute_corr(h))) * (2.0 / N)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
    c_now = float(np.max(compute_corr(h))) * (2.0 / N)
    if c_now < best_c:
        best_c, best_h = c_now, h.copy()
    return best_h, best_c

def modular_perturb(h, N, rng):
    p = int(rng.choice([7, 11, 13, 17, 19, 23]))
    mod = np.arange(N) % p
    for val in range(p):
        pos = np.where(mod == val)[0]
        ones_at = pos[h[pos] == 1]
        zeros_at = pos[h[pos] == 0]
        if len(ones_at) > 0 and len(zeros_at) > 0:
            h[int(rng.choice(ones_at))] = 0
            h[int(rng.choice(zeros_at))] = 1
    return h
```
