import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float).copy()
    idx = np.argsort(-h, kind='stable')
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out
