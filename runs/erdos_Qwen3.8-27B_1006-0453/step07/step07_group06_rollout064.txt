    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj(h, target):

    if abs(s - target) < 1e-14:

        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    mx = c.max()

        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        for _ in range(n_tri):
            oi = int(ones[np.random.randint(len(ones))])
            zi = int(zeros[np.random.randint(len(zeros))])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            if P - nP > bd:
                bd, bi, bj = P - nP, oi, zi
            h[oi], h[zi] = 1.0, 0.0
        if bi >= 0 and bd > 0:
            h[bi], h[bj] = 0.0, 1.0
            improved = True
        stall = 0 if improved else stall + 1
    return best_h, best_c

def upsample(h, N_from, N_to):
    x = np.linspace(0, 2, N_from, endpoint=False)
    xt = np.linspace(0, 2, N_to, endpoint=False)
    ht = np.interp(xt, x, h)
    ht = (ht > 0.5).astype(float)
    t = N_to // 2
