def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            d = i - ctr
            if d == 0:
                g[:] += w[i] * (1.0 - 2.0 * h)
            elif d > 0:
                g[d:] += w[i] * (1.0 - h[:N-d])
                g[:N-d] -= w[i] * h[d:]
            else:
                M = -d
                g[:N-M] += w[i] * (1.0 - h[M:])
                g[M:] -= w[i] * h[:N-M]
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

    h = (h > 0.5).astype(float)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        corr = fft_corr(h)
        cur_max = float(corr.max())
        
        best_delta = 1e-12
        best_a, best_b = -1, -1
        
        # Precompute lag array
        lags = np.arange(2*N-1) - (N-1)  # -N+1 to N-1
        
        for a in ones:
            for b in zeros:

                d = lags  # shape (2N-1,)
                # P_old[a] = (1-h[a+d]) where 0<=a+d<N
                a_plus = a + d
                P_old_a = np.where((a_plus >= 0) & (a_plus < N), 1.0 - h[a_plus], 0.0)
                # Q_old[b] = h[b-d] where 0<=b-d<N
                b_minus = b - d
                Q_old_b = np.where((b_minus >= 0) & (b_minus < N), h[b_minus], 0.0)
                # Q_new[a] = h'[a-d] where h' has a->0, b->1
                a_minus = a - d
                Q_new_a = np.where((a_minus >= 0) & (a_minus < N), h[a_minus], 0.0)

                Q_new_a = np.where(a_minus == a, 0.0, Q_new_a)  # d=0
                Q_new_a = np.where(a_minus == b, 1.0, Q_new_a)  # d=a-b
                # P_new[b] = (1-h'[b+d]) where 0<=b+d<N
                b_plus = b + d
                P_new_b = np.where((b_plus >= 0) & (b_plus < N), 1.0 - h[b_plus], 0.0)
                # Special: if b+d == a (d=b-a), h'[a]=0, so 1-h'[a]=1
                P_new_b = np.where(b_plus == a, 1.0, P_new_b)
                # if d=0, b+d=b, h'[b]=1, so 1-h'[b]=0
                P_new_b = np.where(d == 0, 0.0, P_new_b)
                
                delta = -P_old_a - Q_old_b + Q_new_a + P_new_b
                new_val = corr + delta

                best_h = h.copy()
        else:

    return best_h, best_c * (2.0 / N)
