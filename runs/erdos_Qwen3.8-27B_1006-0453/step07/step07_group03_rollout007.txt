    return float(np.max(fft_corr(h))) * (2.0 / N)

            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            nc = float(np.max(fft_corr(h)))
            if nc < P_cur:
                P_cur = nc
                found = True
            else:
                h[p], h[q] = 1.0, 0.0
            if found:
                break
        if not found and rng.random() < 0.35 and len(ones) >= 4 and len(zeros) >= 4:
            p1, p2 = int(ones[rng.randint(len(ones))]), int(ones[rng.randint(len(ones))])
            q1, q2 = int(zeros[rng.randint(len(zeros))]), int(zeros[rng.randint(len(zeros))])
            if p1 != p2 and q1 != q2:
                h[p1], h[p2], h[q1], h[q2] = 0.0, 0.0, 1.0, 1.0
                nc = float(np.max(fft_corr(h)))
                if nc >= P_cur:
                    h[p1], h[p2], h[q1], h[q2] = 1.0, 1.0, 0.0, 0.0
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary local search
    h_bin = ensure_sum(h0, N0)
    d1 = min(t0 + 90, deadline - 750)
    h1, c5_1 = binary_ls(h_bin, N0, d1, rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Main Nesterov (long)
    d2 = min(t0 + 650, deadline - 250)
    hc = best_h + rng.randn(N0) * 0.035
    hc = project_sum(hc, N0 / 2.0)
    h2, c5_2 = nesterov_opt(hc, N0, d2, rng, T0=3.5, lr0=0.016)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Binary snap + local search
    if time.time() < deadline - 130:
        hB = ensure_sum(best_h, N0)
        d3 = min(time.time() + 100, deadline - 100)
        h3, c5_3 = binary_ls(hB, N0, d3, rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Nesterov from binary
    if time.time() < deadline - 80:
        hc2 = best_h + rng.randn(N0) * 0.02
        hc2 = project_sum(hc2, N0 / 2.0)
        d4 = min(time.time() + 130, deadline - 40)
        h4, c5_4 = nesterov_opt(hc2, N0, d4, rng, T0=1.5, lr0=0.010)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Final Nesterov polish
    if time.time() < deadline - 5:
        hc3 = best_h + rng.randn(N0) * 0.008
        hc3 = project_sum(hc3, N0 / 2.0)
        h5, c5_5 = nesterov_opt(hc3, N0, deadline - 1, rng, T0=0.6, lr0=0.005)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, N0 / 2.0)
    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
