    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_smooth(h, N, deadline, T=30.0, max_outer=3):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj(h, T, target, 0.0)
    lam = 4.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 120, 'maxfun': 300, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj(h, T, target, 0.0)
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 3.0
    return best_h, best_v * (2.0 / N)

def freq_perturb(h, N, amt=0.3):
    H = np.fft.rfft(h)
    noise = npfft_noise(len(H), amt)
    H[1:-1] += noise
    h2 = np.fft.irfft(H, N)
    return proj01(h2, N / 2.0)

def npfft_noise(n, amt):
    return (np.random.randn(n) + 1j * np.random.randn(n)) * amt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    best_N = N0

    # Phase 1: SA from initial (fast)
    d1 = min(dl - 500, t0 + 60)
    hb, cb = sa_binary(best_h, N0, d1, max_iters=3000, T0=0.04, T1=0.001, block_prob=0.2)
    cb5 = cb * (2.0 / N0)
    if cb5 < best_c5:
        best_c5, best_h, best_N = cb5, hb.copy(), N0

    # Phase 2: Targeted fix
    d2 = min(dl - 450, time.time() + 80)
    hf, cf = targeted_fix(best_h, N0, d2, n_rounds=300)
    cf5 = cf * (2.0 / N0)
    if cf5 < best_c5:
        best_c5, best_h, best_N = cf5, hf.copy(), N0

    # Phase 3: L-BFGS continuous
    if time.time() < dl - 350:
        d3 = min(dl - 300, time.time() + 120)
        for r in range(5):
            if time.time() > d3:
                break
            hr = best_h + np.random.randn(N0) * (0.2 - 0.03 * r)
            hr = proj01(hr, N0 / 2.0)
            hl, cl = lbfgs_smooth(hr, N0, d3, T=28.0, max_outer=3)
            cl5 = cl
            if cl5 < best_c5:
                best_c5, best_N = cl5, N0
                best_h = (hl > 0.5).astype(float)
                best_c5 = compute_c5(best_h, N0)

    # Phase 4: SA with restarts from best binary
    if time.time() < dl - 250:
        d4 = min(dl - 200, time.time() + 150)
        for restart in range(6):
            if time.time() > d4:
                break
            if restart == 0:
                hstart = best_h.copy()
            else:
                hstart = best_h + np.random.randn(N0) * 0.4
                hstart = (proj01(hstart, N0 / 2.0) > 0.5).astype(float)
            hr, cr = sa_binary(hstart, N0, d4, max_iters=2000,
                                T0=0.03 - 0.003 * restart, T1=0.0005, block_prob=0.3)
            cr5 = cr * (2.0 / N0)
            if cr5 < best_c5:
                best_c5, best_h, best_N = cr5, hr.copy(), N0

    # Phase 5: Targeted fix again
    if time.time() < dl - 180:
        d5 = min(dl - 150, time.time() + 60)
        hf, cf = targeted_fix(best_h, best_N, d5, n_rounds=200)
        cf5 = cf * (2.0 / best_N)
        if cf5 < best_c5:
            best_c5, best_h, best_N = cf5, hf.copy(), best_N

    # Phase 6: Try other N values
    if time.time() < dl - 100:
        d6 = min(dl - 60, time.time() + 120)
        x_b = np.linspace(0, 2, best_N, endpoint=False)
        for Nt in [360, 400, 420, 440, 460, 500]:
            if time.time() > d6:
                break
            if Nt % 2:
                Nt += 1
            if Nt == best_N:
                continue
            xn = np.linspace(0, 2, Nt, endpoint=False)
            hi = np.interp(xn, x_b, best_h)
            ht = ensure_count((hi > 0.5).astype(float), Nt)
            sd = min(d6, time.time() + 30)
            hnp, cnp = sa_binary(ht, Nt, sd, max_iters=1500, T0=0.03, T1=0.001, block_prob=0.25)
            cnp5 = cnp * (2.0 / Nt)
            if cnp5 < best_c5:
                best_c5, best_h, best_N = cnp5, hnp.copy(), Nt

    # Phase 7: Final L-BFGS
    if time.time() < dl - 40:
        d7 = min(dl - 20, time.time() + 60)
        h7 = best_h + np.random.randn(len(best_h)) * 0.08
        h7 = proj01(h7, len(best_h) / 2.0)
        hl7, cl7 = lbfgs_smooth(h7, len(best_h), d7, T=35.0, max_outer=2)
        h7b = (hl7 > 0.5).astype(float)
        h7b = ensure_count(h7b, len(best_h))
        cl7b = compute_c5(h7b, len(best_h))
        if cl7b < best_c5:
            best_c5, best_h, best_N = cl7b, h7b.copy(), len(best_h)

    # Phase 8: Final binary polish
    if time.time() < dl - 5:
        d8 = dl - 2
        hf, cf = sa_binary(best_h, best_N, d8, max_iters=10000, T0=0.01, T1=0.0001, block_prob=0.15)
        cf5 = cf * (2.0 / best_N)
        if cf5 < best_c5:
            best_c5, best_h, best_N = cf5, hf.copy(), best_N

    # Final: compare continuous vs binary
    fc5 = compute_c5(best_h, best_N)
    fb = ensure_count((best_h > 0.5).astype(float), best_N)
    fc5b = compute_c5(fb, best_N)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, len(best_h)
