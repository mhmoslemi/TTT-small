import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=50, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(77)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def remove_delta(p, h, N):
    delta = np.zeros(2 * N - 1)
    i0 = N - 1 - p
    delta[i0:i0 + N] -= (1.0 - h)
    j0 = p
    delta[j0:j0 + N] += h[::-1]
    return delta

def add_delta(q, h, N):
    delta = np.zeros(2 * N - 1)
    i0 = N - 1 - q
    delta[i0:i0 + N] += (1.0 - h)
    j0 = q
    delta[j0:j0 + N] -= h[::-1]
    return delta

    h = ensure_binary(h, N).copy()
    corr = fft_corr(h).copy()
    best_c = float(corr.max())
    best_h = h.copy()
    rng = np.random.RandomState(123)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    stall = 0
    T = sa_temp

            # Find best swap: try random samples of ones, for each try all zeros

            best_new_P = best_c  # don't accept worse in greedy

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=50, T=80.0, lr0=0.015):

    target = N / 2.0
    best_h = h.copy()

    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))

            best_pk = pk
            best_h = h.copy()

    else:

        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)
