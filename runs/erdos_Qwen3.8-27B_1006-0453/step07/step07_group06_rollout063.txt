import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def systematic_swap(h, N, deadline, max_passes=20):
    """Try all single swaps systematically until no improvement."""
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for p in range(max_passes):
        if time.time() > deadline or stall >= 2:

        zeros = np.where(h == 0)[0].tolist()
        improved = False
        corr = fft_corr(h)
        P = float(np.max(corr))

            if time.time() > deadline:
                break

    corr = fft_corr(h)
    P = float(np.max(corr))
    best_P = P

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        nP = float(np.max(fft_corr(h)))

        if d < 0 or np.random.random() < np.exp(-d / T):

                best_P = P
                best_h = h.copy()
        else:

        T *= cooldown
    return best_h, best_P

def thue_morse(N):
    t = np.zeros(N, dtype=int)
    for i in range(N):
        x = i
        c = 0
        while x:
            c ^= 1
            x >>= 1
        t[i] = c
    # Ensure exactly N/2 ones
    n1 = int(t.sum())
    if n1 != N // 2:
        if n1 > N // 2:
            # Remove excess
            excess = np.where(t == 1)[0][:n1 - N//2]
            t[excess] = 0
        else:
            need = N // 2 - n1
            t[np.where(t == 0)[0][:need]] = 1
    return t.astype(float)

def zadoff_chu(N):

        h = np.zeros(N)
        h[idx[:N//2]] = 1.0
    return h

            continue
        si = np.random.choice(len(ones), nf, replace=False)
        sj = np.random.choice(len(zeros), nf, replace=False)
        fl = np.concatenate([ones[si], zeros[sj]])
        h[fl] = 1.0 - h[fl]
        nP = float(np.max(fft_corr(h)))
        d = nP - P
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-10)):
            P = nP
            if P < best_P:
                best_P = P
                best_h = h.copy()
        else:
            h[fl] = 1.0 - h[fl]
        T *= cool
    return best_h, best_P

def block_swap_sa(h, N, deadline, max_iters=3000, T0=0.003, cool=0.98):
    h = h.astype(float).copy()
    P = float(np.max(fft_corr(h)))
    best_P = P
    best_h = h.copy()
    T = T0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        a = np.random.randint(0, N - 4)
        b = np.random.randint(a + 3, N)
        seg = h[a:b + 1].copy()
        h[a:b + 1] = seg[::-1]
        nP = float(np.max(fft_corr(h)))
        d = nP - P
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-10)):
            P = nP
            if P < best_P:
                best_P = P
                best_h = h.copy()
        else:
            h[a:b + 1] = seg
        T *= cool
    return best_h, best_h

def kick_and_search(h, N, deadline, n_kicks=5, kick_size=8):
    h = h.astype(float).copy()
    P = float(np.max(fft_corr(h)))
    best_P = P
    best_h = h.copy()
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nk = min(kick_size, len(ones), len(zeros))
        si = np.random.choice(len(ones), nk, replace=False)
        sj = np.random.choice(len(zeros), nk, replace=False)
        h[ones[si]] = 0.0
        h[zeros[sj]] = 1.0
        dkl = min(deadline, time.time() + 20)
        hk, Pk = systematic_swap(h, N, dkl, max_passes=8)
        if Pk < best_P:
            best_P = Pk
            best_h = hk.copy()
        else:
            h = best_h.copy()
            P = best_P
    return best_h, best_P

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h0 = (h0 > 0.5).astype(float)
    h0 = fix_sum_binary(h0, N0)
    best_P = float(np.max(fft_corr(h0)))
    best_h = h0.copy()

    # Phase 1: Thorough systematic local search
    d1 = min(deadline - 350, t0 + 130)
    h1, P1 = systematic_swap(h0, N0, d1, max_passes=25)
    if P1 < best_P:
        best_P, best_h = P1, h1.copy()

    # Phase 2: Multi-flip SA to escape local minima
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 80)
        h2, P2 = multi_flip_sa(best_h, N0, d2, max_iters=4000, T0=0.006, cool=0.965)
        if P2 < best_P:
            best_P, best_h = P2, h2.copy()
        d2b = min(deadline - 200, time.time() + 50)
        h2b, P2b = systematic_swap(best_h, N0, d2b, max_passes=15)
        if P2b < best_P:
            best_P, best_h = P2b, h2b.copy()

    # Phase 3: Kick-and-search (perturb then exploit)
    if time.time() < deadline - 180:
        d3 = min(deadline - 160, time.time() + 60)
        h3, P3 = kick_and_search(best_h, N0, d3, n_kicks=4, kick_size=10)
        if P3 < best_P:
            best_P, best_h = P3, h3.copy()

    # Phase 4: Alternative structured starts
    alt_starts = [thue_morse_seq(N0), zadoff_chu_seq(N0)]
    for s in alt_starts:
        if time.time() > deadline - 120:
            break
        ds = min(deadline - 100, time.time() + 50)
        hs, Ps = systematic_swap(s, N0, ds, max_passes=12)
        if Ps < best_P:
            best_P, best_h = Ps, hs.copy()
        dsm = min(deadline - 90, time.time() + 30)
        hsm, Psm = multi_flip_sa(hs, N0, dsm, max_iters=2000, T0=0.004, cool=0.97)
        if Psm < best_P:
            best_P, best_h = Psm, hsm.copy()
        dss = min(deadline - 80, time.time() + 25)
        hss, Pss = systematic_swap(best_h, N0, dss, max_passes=8)
        if Pss < best_P:
            best_P, best_h = Pss, hss.copy()

    # Phase 5: Block reversal SA
    if time.time() < deadline - 80:
        d5 = min(deadline - 60, time.time() + 40)
        h5, P5 = block_swap_sa(best_h, N0, d5, max_iters=2000, T0=0.003, cool=0.985)
        if P5 < best_P:
            best_P, best_h = P5, h5.copy()
        d5b = min(deadline - 45, time.time() + 25)
        h5b, P5b = systematic_swap(best_h, N0, d5b, max_passes=8)
        if P5b < best_P:
            best_P, best_h = P5b, h5b.copy()

    # Phase 6: Multi-resolution refinement
    if time.time() < deadline - 30:
        cur_N = len(best_h)
        x_old = np.linspace(0, 2, cur_N, endpoint=False)
        Ntries = [cur_N - 4, cur_N - 2, cur_N + 2, cur_N + 4, cur_N - 6, cur_N + 6]
        for Nt in Ntries:
            if time.time() > deadline - 20:
                break
            if Nt < 100 or Nt % 2 != 0 or Nt == cur_N:
                continue
            x_new = np.linspace(0, 2, Nt, endpoint=False)
            hi = np.interp(x_new, x_old, best_h)
            ht = (hi > 0.5).astype(float)
            ht = fix_sum_binary(ht, Nt)
            dt = min(deadline - 15, time.time() + 30)
            htt, Pt = systematic_swap(ht, Nt, dt, max_passes=8)
            c5_new = Pt * (2.0 / Nt)
            c5_old = best_P * (2.0 / cur_N)
            if c5_new < c5_old:
                best_P = Pt
                best_h = htt.copy()

    # Phase 7: Final intensive polish
    if time.time() < deadline - 2:
        d7 = deadline - 1.0
        h7, P7 = systematic_swap(best_h, len(best_h), d7, max_passes=10)
        if P7 < best_P:
            best_P, best_h = P7, h7.copy()

    c5_final = best_P * (2.0 / len(best_h))
    return best_h, c5_final, len(best_h)
