    """Compute force on each position for 0->1 flip, using softmax of corr."""
    corr = fft_corr(h)  # length 2N-1
    # Softmax weights
    peak = np.max(corr)
    w = np.exp((corr - peak) / T)
    w /= w.sum()
    
    force = np.zeros(N)
    for p in range(N):
        f = 0.0
        for i in range(2*N-1):
            s = i - (N-1)
            term = 0.0
            # (1-h[p+s]) if p+s is valid (0 <= p+s < N)
            if 0 <= p+s < N:

            # -h[p-s] if p-s is valid (0 <= p-s < N)
            if 0 <= p-s < N:
                term -= h[p-s]
            f += w[i] * term
        force[p] = f
    return force, w

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_force(h, N, T):
    corr = fft_corr(h)
    peak = np.max(corr)
    w = np.exp((corr - peak) / T)
    w /= w.sum()
    force = np.zeros(N)
    for p in range(N):
        f = 0.0
        for i in range(2*N-1):
            s = i - (N-1)
            term = 0.0
            if 0 <= p+s < N:
                term += (1.0 - h[p+s])
            if 0 <= p-s < N:
                term -= h[p-s]
            f += w[i] * term
        force[p] = f
    return force, peak * 2.0 / N

def greedy_pair_swap(h, N, T, deadline, max_iters=100000):
    """Local search using force-guided pair swaps."""
    nh = N // 2
    h = h.copy()
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    # Build index lists
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    no_improve = 0
    T_eff = T
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # Compute force
        force, _ = compute_force(h, N, T_eff)
        
        # Find best swap: 1 pos with lowest force, 0 pos with highest force
        one_forces = force[ones]
        zero_forces = force[zeros]
        
        p = ones[np.argmin(one_forces)]  # best 1 to flip to 0
        q = zeros[np.argmax(zero_forces)]  # best 0 to flip to 1
        
        if p == q:
            continue
        
        # Try the swap
        h[p] = 0
        h[q] = 1
        new_corr = fft_corr(h)
        new_max = float(np.max(new_corr))
        dE = new_max - cur_max
        
        if dE < -1e-14 or np.random.random() < np.exp(-dE / max(T_eff, 1e-10)):
            cur_max = new_max
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max * 2.0 / N < best_c - 1e-14:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                no_improve = 0
            else:

import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    return float(np.max(corr_full(h))) * (2.0 / len(h))

def compute_force(h, T):
    N = len(h)
    corr = corr_full(h)
    peak = float(np.max(corr))
    w = np.exp((corr - peak) / T)
    w /= w.sum()
    p_idx = np.arange(N)[:, None]
    s_arr = np.arange(-(N - 1), N)[None, :]
    ps = p_idx + s_arr
    pm = p_idx - s_arr
    m1 = (ps >= 0) & (ps < N)
    m2 = (pm >= 0) & (pm < N)
    hps = np.where(m1, h[np.clip(ps, 0, N - 1)], 0.0)
    hpm = np.where(m2, h[np.clip(pm, 0, N - 1)], 0.0)
    t1 = np.sum(w[None, :] * (1.0 - hps), axis=1)
    t2 = np.sum(w[None, :] * hpm, axis=1)
    return t1 - t2

def proj_binary(h, n_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:n_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy()
    s = int(np.sum(h))
    if s > target:
        idx = np.where(h == 1)[0]
        sel = np.random.choice(idx, s - target, replace=False)
        h[sel] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        sel = np.random.choice(idx, target - s, replace=False)
        h[sel] = 1
    return h

def binary_anneal(h, N, deadline, T0=0.05, max_iters=100000):
    h = fix_sum_binary(h, N // 2)
    corr = corr_full(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = T0
    cool_rate = (0.003 / T0) ** (1.0 / max(1, max_iters))
    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        force = compute_force(h, T)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        r = np.random.random()
        if r < 0.65:
            p = ones[np.argmin(force[ones])]
            q = zeros[np.argmax(force[zeros])]
        elif r < 0.85:
            p = ones[np.argmin(force[ones][:min(10, len(ones))])]
            zl = zeros[np.argsort(force[zeros])[-min(10, len(zeros)):]]
            q = zl[np.random.randint(len(zl))]
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
        if p == q:
            continue
        h[p] = 1 - h[p]
        h[q] = 1 - h[q]
        new_max = float(np.max(corr_full(h)))
        dE = new_max - cur_max
        if dE < -1e-14 or (T > 1e-8 and np.random.random() < np.exp(-dE / T)):
            cur_max = new_max
            c = cur_max * 2.0 / N
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[p] = 1 - h[p]
            h[q] = 1 - h[q]
        T = max(T * cool_rate, 0.003)
        if no_improve > 2500:
            k = np.random.randint(4, 12)
            ones_n = np.where(h == 1)[0]
            zeros_n = np.where(h == 0)[0]
            for _ in range(min(k, len(ones_n) - 1, len(zeros_n) - 1)):
                oi = np.random.choice(ones_n)
                zi = np.random.choice(zeros_n)
                h[oi], h[zi] = 0, 1
                ones_n = np.where(h == 1)[0]
                zeros_n = np.where(h == 0)[0]
            cur_max = float(np.max(corr_full(h)))
            T = max(T, 0.02)
            no_improve = 0
    return best_h, best_c

def continuous_refine(h, N, deadline, lr0=0.004):
    h = h.astype(float).copy()
    target = N / 2.0
    best_c = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    T_soft = 0.04
    it = 0
    while time.time() < deadline - 2:
        it += 1
        force = compute_force(h, T_soft)
        h_new = np.clip(h - lr * force, 0.0, 1.0)
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if it % 800 == 0:
            lr *= 0.985
            T_soft = max(T_soft * 0.95, 0.01)
    return best_h, best_c

def structured_init(N, variant):
    np.random.seed(9999 + variant * 197)
    nh = N // 2
    x = np.arange(N)
    if variant == 0:
        v = np.sin(2 * np.pi * x * 3.0 / N) + 0.6 * np.sin(2 * np.pi * x * 7.0 / N + 0.4) + 0.4 * np.cos(2 * np.pi * x * 13.0 / N + 1.2)
        h = (v > np.median(v)).astype(float)
    elif variant == 1:
        h = ((x * 7 + 3) % 13 < 6).astype(float)
    elif variant == 2:
        phi = 0.5 * (1 + np.sqrt(5))
        h = ((np.arange(N) * phi) % 1.0 < 0.5).astype(float)
    else:
        h = np.zeros(N)
        h[np.random.choice(N, nh, replace=False)] = 1
    return fix_sum_binary(h, nh)
