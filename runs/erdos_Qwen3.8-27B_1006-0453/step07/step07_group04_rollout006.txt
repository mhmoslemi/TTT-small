    p has 2N-1 elements (same as the correlation output)."""
    g = np.zeros(N)
    # For shift s (from -(N-1) to N-1), p_index = s + (N-1)
    for s in range(-(N-1), N):
        pi = p[s + N - 1]
        if pi < 1e-15:
            continue
        if s >= 0:
            # c[s] = sum_{i=0}^{N-1-s} h[i]*(1-h[i+s])
            # ∂/∂h[j]: for j in [0,N-1-s]: (1-h[j+s]); for j in [s,N-1]: -h[j-s]

            valid2 = slice(s, N)
            g[valid1] += pi * (1 - h[s:N])
            g[valid2] -= pi * h[0:N-s]
        else:

            valid2 = slice(0, N-m)
            g[valid1] += pi * (1 - h[0:N-m])
            g[valid2] -= pi * h[m:N]
    return g
