def nesterov_opt(h, N, deadline, k=50, lr0=0.01):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m_vec = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m_vec
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m_vec = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.98
            beta = min(0.97, 0.85 + it * 0.0002)
    return best_h, best_pk * dx

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def corr_max(h):
    return float(np.max(fft_corr(h)))

def compute_c5(h):
    N = len(h)
    return corr_max(h) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    sh = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:,None]
    pm = pos - sh[None,:]
    pp = pos + sh[None,:]
    vm = (pm>=0)&(pm<N)
    vp = (pp>=0)&(pp<N)
    pmc = np.clip(pm,0,N-1)
    ppc = np.clip(pp,0,N-1)
    return -(1.0-h[pmc])*vm + h[ppc]*vp

def ensure_sum(h, N):
    nh = N//2
    h = h.astype(float).copy()
    hb = (h>0.5).astype(float)
    n1 = int(hb.sum())
    if n1==nh: return hb
    r = np.random.RandomState(99999+N)
    if n1>nh:
        idx = np.where(hb==1)[0]
        hb[r.choice(idx, n1-nh, replace=False)] = 0
    else:
        idx = np.where(hb==0)[0]
        hb[r.choice(idx, nh-n1, replace=False)] = 1
    return hb

def bmary_anneal(h, N, deadline, max_sw=500, tf2=0.3, tf3=0.15, seed=7):
    h = h.astype(float).copy()
    if int(h.sum()) != N//2: h = ensure_sum(h, N)
    r = np.random.RandomState(seed)
    cc = fft_corr(h)
    Pc = float(cc.max())
    bc, bh = Pc, h.copy()
    temp = 0.015
    stall = 0
    for sw in range(max_sw):
        if time.time() > deadline: break
        temp = max(0.002, temp*0.994)
        on = np.where(h==1)[0]
        zo = np.where(h==0)[0]
        if len(on)<6 or len(zo)<6: break
        D = compute_D(h, N)
        Dz = D[zo,:]
        bnP = Pc
        bm = None
        # 1-swap: all ones
        for p in on:
            base = cc - D[p,:]
            cand = base[None,:] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < bnP:
                bnP = float(npk[mi])
                bm = (1,p,int(zo[mi]))
        # 2-swap
        if len(on)>=4 and len(zo)>=4 and r.random()<tf2:
            nt = min(60, max(15, N//6))
            for _ in range(nt):
                i1,i2 = r.randint(0,len(on),2)
                j1,j2 = r.randint(0,len(zo),2)
                if i1==i2 or j1==j2: continue
                p1,p2 = int(on[i1]),int(on[i2])
                q1,q2 = int(zo[j1]),int(zo[j2])
                nP = float((cc+D[p1]+D[p2]-D[q1]-D[q2]).max())
                if nP < bnP-1e-12:
                    bnP = nP; bm = (2,p1,p2,q1,q2)
        # 3-swap
        if len(on)>=6 and len(zo)>=6 and r.random()<tf3:
            nt = min(30, max(8, N//12))
            for _ in range(nt):
                ix = r.choice(len(on),3,replace=False)
                jx = r.choice(len(zo),3,replace=False)
                ps = [int(on[i]) for i in ix]
                qs = [int(zo[j]) for j in jx]
                nP = float((cc+D[ps[0]]+D[ps[1]]+D[ps[2]]-D[qs[0]]-D[qs[1]]-D[qs[2]]).max())
                if nP < bnP-1e-12:
                    bnP = nP; bm = (3,ps[0],ps[1],ps[2],qs[0],qs[1],qs[2])
        if bm is not None:
            m = bm
            if m[0]==1: h[m[1]]=0; h[m[2]]=1
            elif m[0]==2: h[m[1]]=h[m[2]]=0; h[m[3]]=h[m[4]]=1
            else: h[m[1]]=h[m[2]]=h[m[3]]=0; h[m[4]]=h[m[5]]=h[m[6]]=1
            cc = fft_corr(h)
            Pc = float(cc.max())
            if Pc < bc-1e-13: bc=Pc; bh=h.copy(); stall=0
            else: stall+=1
        else:
            if stall>20:
                for _ in range(3):
                    p=int(on[r.randint(len(on))]); q=int(zo[r.randint(len(zo))])
                    h[p],h[q]=0,1
                stall=0
            elif stall>10: break
            else: stall+=1
    return bh, bc*(2.0/N)

def nesterov(h, N, deadline, k0=20, k1=80, lr0=0.015):
    h = h.astype(float).copy()
    tgt = N/2.0
    dx = 2.0/N
    bh = h.copy()
    bpk = float(np.max(fft_corr(h)))
    mv = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + mv
        cr = fft_corr(y)
        k = min(int(k0 + (k1-k0)*min(1.0, it/500)), len(cr)-1)
        k = max(k, 5)
        top = np.argpartition(cr, -k)[-k:]
        pk = float(cr[top].max())
        if pk < bpk - 1e-12:
            bpk = pk; bh = h.copy()
        grad = np.zeros(N)
        ct = N-1
        for ai in top:
            L = int(ai-ct)
            if L==0: grad += (1.0-2.0*y)
            elif L>0:
                grad[L:] += (1.0-y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M=-L
                if M<N:
                    grad[:N-M] += (1.0-y[M:])
                    grad[M:] -= y[:N-M]
        grad /= k
        hn = project_sum(y - lr*grad, tgt)
        mv = beta*(hn - h)
        h = hn
        it+=1
        if it%200==0:
            lr *= 0.97
            beta = min(0.96, 0.82+it*0.0002)
    return bh, bpk*dx

def upsample(h, Nn):
    No = len(h)
    xo = np.linspace(0,2,No,endpoint=False)
    xn = np.linspace(0,2,Nn,endpoint=False)
    return np.interp(xn, xo, h)

    return bh, bc5, len(bh)
