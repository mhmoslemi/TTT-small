        #   Terms where h[p] is the 'b' factor: i+j = p, i.e., j = p-i
        #     b[p] = 1-h[p] was 0, now becomes 1
        #     So we ADD h[i] to result[N-1+j] = result[N-1+p-i] for valid i
        #     i ranges from max(0, p-(N-1)) to min(N-1, p)
        #     result index = N-1+p-i ranges from N-1+p-min(N-1,p) to N-1+p-max(0,p-(N-1))
        #     = N-1+p-(N-1) to N-1+p-(p-(N-1)) if p >= N-1
        #     This is getting messy.
        
        # Let me just use a numerical approach: for a small number of candidates, compute the new corr directly.
        # With ~50 candidates from peaks + ~8 random, that's 58*58 = 3364 pairs.
        # For each pair, the change in corr is O(N) to compute directly, but we can vectorize.
        
        # Better approach: precompute the "flip effect" for each position.
        # flip_effect[p] = vector of length 2N-1 showing corr change if h[p] flips
        # For h[p]=1 → 0:
        #   At result index N-1+j (lag j):
        #     - h[p] was contributing (1-h[p+j]) → remove it: change -= (1-h[p+j]) for valid p+j
        #     - (1-h[p]) was 0, now 1: h[i] where i+j=p → adds h[i] for valid i
        #       i = p - j, so change += h[p-j] for valid p-j
        #   Combined: change[N-1+j] = -（1-h[p+j]) + h[p-j] (with 0 for out of bounds)
        #
        # For h[p]=0 → 1:
        #   - h[p] was 0, now 1: adds (1-h[p+j]) for valid p+j
        #   - (1-h[p]) was 1, now 0: removes h[p-j] for valid p-j
        #   Combined: change[N-1+j] = (1-h[p+j]) - h[p-j] (with 0 for out of bounds)
        
        # So for our swap p(1→0) and q(0→1):
        # total_change = flip1_to_0[p] + flip0_to_1[q]
        # where flip1_to_0[p][N-1+j] = -(1-h[p+j]) + h[p-j]
        # and flip0_to_1[q][N-1+j] = (1-h[q+j]) - h[q-j]
        
        # This is O(N) to precompute for all p, q positions.
        # But we only need it for the current h, and h changes each iteration.
        # Precomputing all flip effects takes O(N^2) which is fine for N=380.
        
        # Let me precompute flip effects for all positions each iteration.
        # Actually that's O(N^2) per iteration which might be too slow.
        # Let me compute on the fly for candidate positions only.
        
        # For a candidate p (h[p]=1): 
        #   effect_p[j] = -(1-h[p+j]) + h[p-j] for valid indices (0-indexed, j from -(N-1) to N-1)
        #   In terms of the result array index k = N-1+j:
        #   effect_p[k] = -(1-h[p+k-(N-1)]) + h[p-k+(N-1)] for valid indices
        #
        # For a candidate q (h[q]=0):
        #   effect_q[k] = (1-h[q+k-(N-1)]) - h[q-k+(N-1)] for valid indices
        
        # Let me compute these for the candidate set.
        Nm1 = N - 1
        # For each candidate p, compute the flip effect vector (length 2N-1)
        # Vectorized:
        k_idx = np.arange(2 * N - 1) - Nm1  # k goes from -(N-1) to N-1
        
        # For p in cand_p (h[p]=1): effect at lag k is -(1-h[p+k]) + h[p-k] (with bounds)
        # For q in cand_q (h[q]=0): effect at lag k is (1-h[q+k]) - h[q-k] (with bounds)
        
        # Let me compute the new corr for all (p,q) pairs at once using broadcasting.
        # Effect of flipping p and q = individual_p + individual_q
        # individual_p[k] = -(1-h[p+k]) + h[p-k] (for h[p]=1, both p+k and p-k must be in [0,N-1])
        # individual_q[k] = (1-h[q+k]) - h[q-k] (for h[q]=0, both q+k and q-k must be in [0,N-1])
        
        # For efficiency, let's compute the change in the MAX corr value (not the full vector).
        # new_corr[k] = corr[k] + individual_p[k] + individual_q[k]
        # We want max_k new_corr[k]
        
        # Since individual effects are sparse (nonzero only for |k| in [pos-(N-1), pos+(N-1)]),
        # and N=380, the full vector is 759 elements. Let's just compute the full thing.
        
        # For each p in cand_p:
        #   pks = np.arange(2*N-1)
        #   ks = pks - (N-1)  # actual lag
        #   pk_plus = p + ks  # p+k
        #   pk_minus = p - ks  # p-k
        #   valid_plus = (pk_plus >= 0) & (pk_plus < N)
        #   valid_minus = (pk_minus >= 0) & (pk_minus < N)
        #   effect_p = np.zeros(2*N-1)
        #   effect_p[valid_plus] -= (1 - h[pk_plus[valid_plus]])
        #   effect_p[valid_minus] += h[pk_minus[valid_minus]]
        
        # This is O(N) per candidate. With ~60 candidates, that's 60*759 ≈ 45K ops, fine.
        
        # Let me precompute effects for all candidates:
        ks = np.arange(2 * N - 1) - (N - 1)  # shape (2N-1,)
        
        eff_p = np.zeros((len(cand_p), 2 * N - 1))
        for ip, p in enumerate(cand_p):
            pp = p + ks
            pm = p - ks
            vp = (pp >= 0) & (pp < N)
            vm = (pm >= 0) & (pm < N)
            eff_p[ip, vp] -= (1.0 - h[pp[vp]])
            eff_p[ip, vm] += h[pm[vm]]
        
        eff_q = np.zeros((len(cand_q), 2 * N - 1))
        for iq, q in enumerate(cand_q):
            qq = q + ks
            qm = q - ks
            vq = (qq >= 0) & (qq < N)
            vm = (qm >= 0) & (qm < N)
            eff_q[iq, vq] += (1.0 - h[qq[vq]])
            eff_q[iq, vm] -= h[qm[vm]]
        
        # Now evaluate all pairs: new_corr = corr + eff_p[i] + eff_q[j]
        # We want to find (i,j) minimizing max(new_corr)
        # This is a (len(cand_p) x len(cand_q) x (2N-1)) tensor - too big for broadcasting
        # Use chunked approach:
        chunk = 8
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            # vals[i, j, k] = corr[k] + eff_p[i, k] + eff_q[j, k]
            vals = corr[np.newaxis, np.newaxis, :] + eff_p[ci:ce, np.newaxis, :] + eff_q[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        
        p = int(cand_p[bi])
        q = int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-12 or (T > 0.002 and np.random.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)  # Recompute (simpler and safer)
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
    
    return best_h, best_c


def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=40):
    h = h_in.astype(float).copy()
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.8
    it = 0
    lr_local = lr
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr_local * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr_local *= 0.99
            beta = min(0.93, 0.80 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
