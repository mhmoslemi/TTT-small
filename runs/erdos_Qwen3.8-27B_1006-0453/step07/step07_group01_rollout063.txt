import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def to_binary(h, N, rng):
    order = np.argsort(-h, kind='stable')
    h_bin = np.zeros(N)
    h_bin[order[:N//2]] = 1.0
    return h_bin

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def compute_D(h, N, shifts):
    L = 2*N - 1
    p_col = np.arange(N)[:, None]
    s_row = shifts[None, :]
    pps = p_col + s_row
    pms = p_col - s_row
    vps = (pps >= 0) & (pps < N)
    vms = (pms >= 0) & (pms < N)
    pp = np.clip(pps, 0, N-1)
    pm = np.clip(pms, 0, N-1)
    return -(1.0 - h[pp]).astype(float) * vps + h[pm].astype(float) * vms

def nesterov_refine(h_init, N, deadline, lr0=0.01, k_top=30, T_sm=80.0):
    h = project_sum(h_init.copy(), N/2.0)
    target = N / 2.0
    beta = 0.9
    m = np.zeros(N)
    lr = lr0
    best_h = h.copy()
    best_c = compute_c5(h)
    shifts = np.arange(2*N-1) - (N-1)
    it = 0
    
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        cur_c = compute_c5(y)
        if cur_c < best_c - 1e-14:
            best_c = cur_c
            best_h = h.copy()
        
        # Soft-max gradient
        idx = np.argpartition(c, -k_top)[-k_top:]
        c_top = c[idx]
        w = np.exp(T_sm * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0*h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N-s])
                grad[:N-s] -= wi * h[s:]
            else:
                M = -s
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
        grad /= k_top
        
        h_new = project_sum(y - lr*grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0001)
    
    return best_h, best_c
