import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum_binary(h, nh):
    s = int(np.sum(h))
    if s > nh:
        idx = np.where(h == 1)[0]
        pick = np.random.choice(idx, s - nh, replace=False)
        h[pick] = 0
    elif s < nh:
        idx = np.where(h == 0)[0]
        pick = np.random.choice(idx, nh - s, replace=False)
        h[pick] = 1
    return h

def full_binary_search(h, N, deadline, kick_base=40, kick_var=20, max_kick=12):
    h = h.astype(float).copy()
    fix_sum_binary(h, N // 2)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stuck = 0
    it = 0
    while time.time() < deadline:
        it += 1
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        C = compute_C_mat(h, N)
        Co = C[ones]
        Cz = C[zeros]
        newP = np.empty((len(ones), len(zeros)))
        for oi in range(len(ones)):
            newP[oi] = np.max((corr + Co[oi])[None, :] + Cz, axis=1)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        if newP[mi] < P - 1e-12:
            h[int(ones[mi[0]])] = 0
            h[int(zeros[mi[1]])] = 1
        elif stuck >= kick_base:
            ksz = min(max_kick, kick_base // 8 + int(np.random.random() * kick_var))
            for _ in range(ksz):
                h[int(np.random.choice(ones))] = 0
                h[int(np.random.choice(zeros))] = 1
            stuck = 0
        elif stuck > kick_base * 3:
            break
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=80, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.992
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)
