def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum_binary(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1-target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target-n1, replace=False)] = 1
    return h

def swap_delta(corr, h, a, b, lags):

    if variant == 0:
        h = np.zeros(N)
        h[np.random.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        v = (np.sin(2*np.pi*7*x/N) + 0.8*np.sin(2*np.pi*13*x/N + 0.5) +
             0.6*np.cos(2*np.pi*5*x/N + 1.2))
        h = (v > np.median(v)).astype(float)
    elif variant == 2:
        h = ((x * 7) % 11 < 5).astype(float)
    else:
        v = np.sin(2*np.pi*3*x/N)*1.0 + np.sin(2*np.pi*11*x/N+0.4)*0.7 + \
            np.sin(2*np.pi*19*x/N+1.1)*0.5 + np.cos(2*np.pi*3*x/N+0.9)*0.3
        h = (v > np.median(v)).astype(float)
    return fix_sum_binary(h, nh)

def fast_anneal(h, N, deadline, T0=0.06, T1=1e-6):
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = T0
    it = 0
    no_improve = 0
    stall_limit = 600
    d_arr = np.arange(2 * N - 1) - (N - 1)
    refresh_every = 800

    while time.time() < deadline:
        it += 1
        if it % refresh_every == 0:
            corr = fft_corr(h)
            cur_max = float(np.max(corr))

        roll = np.random.random()
        if roll < 0.7:
            # Single swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            a = int(ones[np.random.randint(len(ones))])
            b = int(zeros[np.random.randint(len(zeros))])
            delta = np.empty(2*N-1)
            j1 = a + d_arr; m = (j1 >= 0) & (j1 < N); delta[:] = 0.0
            delta[m] -= (1.0 - h[j1[m]])
            j2 = b + d_arr; m = (j2 >= 0) & (j2 < N); delta[m] += (1.0 - h[j2[m]])
            j3 = a - d_arr; m = (j3 >= 0) & (j3 < N); delta[m] += h[j3[m]]
            j4 = b - d_arr; m = (j4 >= 0) & (j4 < N); delta[m] -= h[j4[m]]
            new_corr = corr + delta
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < -1e-10 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
                h[a] = 0.0; h[b] = 1.0
                corr = new_corr; cur_max = new_max
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        elif roll < 0.85:
            # 2-swap: two 1s -> 0s, two 0s -> 1s
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            a1, a2 = np.random.choice(ones, 2, replace=False)
            b1, b2 = np.random.choice(zeros, 2, replace=False)
            delta = np.empty(2*N-1); delta[:] = 0.0
            for a, b in [(a1, b1), (a2, b2)]:
                j1 = a + d_arr; m = (j1 >= 0) & (j1 < N); delta[m] -= (1.0 - h[j1[m]])
                j2 = b + d_arr; m = (j2 >= 0) & (j2 < N); delta[m] += (1.0 - h[j2[m]])
                j3 = a - d_arr; m = (j3 >= 0) & (j3 < N); delta[m] += h[j3[m]]
                j4 = b - d_arr; m = (j4 >= 0) & (j4 < N); delta[m] -= h[j4[m]]
            new_corr = corr + delta
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            if dE < -1e-10 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
                h[a1] = 0.0; h[a2] = 0.0; h[b1] = 1.0; h[b2] = 1.0
                corr = new_corr; cur_max = new_max
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            # Run flip: flip a short run
            L_run = int(np.random.choice([2, 3, 4, 5, 7]))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            # Find start of a 1-run of length >= L_run
            valid_starts = []
            i = 0
            while i < N:
                if h[i] == 1:
                    j = i
                    while j < N and h[j] == 1:
                        j += 1
                    if j - i >= L_run:
                        valid_starts.append(i)
                    i = j
                else:
                    i += 1
            if not valid_starts:
                no_improve += 1
                T *= 0.999
                continue
            start = int(valid_starts[np.random.randint(len(valid_starts))])
            # Flip h[start:start+L_run] from 1->0 and h[fl:start+L_run] from 0->1
            # Need matching zero run; just do single flip of run and compensate
            h_old = h[start:start+L_run].copy()
            h[start:start+L_run] = 0.0
            # Find zeros to fill
            zeros_after = np.where(h == 0)[0]
            zeros_after = zeros_after[(zeros_after < start) | (zeros_after >= start + L_run)]
            if len(zeros_after) < L_run:
                h[start:start+L_run] = h_old
                no_improve += 1
                continue
            fill = np.random.choice(zeros_after, L_run, replace=False)
            h[fill] = 1.0
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            if cur_max * 2.0 / N < best_c - 1e-12:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                no_improve = 0
            else:
                # Revert
                h[start:start+L_run] = h_old
                h[fill] = 0.0
                no_improve += 1
                if time.time() < deadline:
                    corr = fft_corr(h)
                    cur_max = float(np.max(corr))

        T = max(T * 0.99997, T1)
        if no_improve > stall_limit:
            for _ in range(6):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                a2 = int(ones[np.random.randint(len(ones))])
                b2 = int(zeros[np.random.randint(len(zeros))])
                h[a2], h[b2] = h[b2], h[a2]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, 0.03)
            no_improve = 0

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 > 0.5).astype(float)
    best_h = fix_sum_binary(best_h, N0 // 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Main anneal on parent (most budget)
    d1 = deadline - 250
    h1, c1 = fast_anneal(best_h, N0, d1, T0=0.08, T1=1e-5)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Multi-N exploration
    N_list = [378, 384, 390, 396, 400, 420, 480, 560]
    remaining = deadline - time.time()
    per_N = max(15, int(remaining * 0.55 / len(N_list)))
    for N_try in N_list:
        if time.time() > deadline - 60:
            break
        best_loc_c = np.inf
        best_loc_h = None
        for vi in range(3):
            if time.time() > deadline - 40:
                break
            hv = structured_init(N_try, vi)
            sub_dl = min(deadline - 30, time.time() + per_N)
            hr, cr = fast_anneal(hv, N_try, sub_dl, T0=0.07, T1=1e-4)
            if cr < best_loc_c:
                best_loc_c, best_loc_h = cr, hr.copy()
        if best_loc_h is not None and best_loc_c < best_c5:
            best_c5, best_h = best_loc_c, best_loc_h

    # Phase 3: Continue annealing best solution
    if time.time() < deadline - 10:
        h3 = fast_anneal(best_h, len(best_h), deadline, T0=0.02, T1=1e-7)
        if h3[1] < best_c5:
            best_c5, best_h = h3[1], h3[0]

    # Final: ensure binary and valid
    hB = (best_h > 0.5).astype(float)
    hB = fix_sum_binary(hB, len(hB) // 2)
    cB = compute_c5(hB)
    if cB <= best_c5 + 1e-9:
        best_h, best_c5 = hB, cB
    else:
        best_c5 = min(best_c5, cB)

    return best_h, best_c5, len(best_h)
