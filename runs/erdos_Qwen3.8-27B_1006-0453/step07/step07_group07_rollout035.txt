    return best_h, best_c

def binary_sa(h, deadline, n_crit=35, s_sz=50, T0=0.04, cool=0.975):
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    center = N - 1
    T = T0
    no_imp = 0
    while time.time() < deadline:
        if no_imp > 500 and T < 0.003:
            break
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 80:
            T *= cool
        if T < 0.0008:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        top_lags = np.argsort(corr)[-n_crit:]
        top_L = (top_lags - center).astype(int)
        cur_peak = float(np.max(corr))
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        best_new = cur_peak
        best_move = None
        for oi in os_:
            doi = np.zeros(2*N-1)
            for L in top_L:
                idx = int(center + L)
                d = 0.0
                if 0 <= oi + L < N:
                    d -= (1.0 - h[oi + L])
                if 0 <= oi - L < N:
                    d -= (1.0 - h[oi - L])
                doi[idx] = d
            for zi in zs:
                new_peak = 0.0
                for idx in top_lags:
                    ii = int(idx)
                    L = int(idx - center)
                    d = 0.0
                    if 0 <= zi + L < N:
                        d += (1.0 - h[zi + L])
                    if 0 <= zi - L < N:
                        d += (1.0 - h[zi - L])
                    val = corr[ii] + doi[ii] + d
                    if val > new_peak:
                        new_peak = val
                if new_peak < cur_peak - 1e-10:
                    accept = True
                elif T > 0.001:
                    accept = np.random.random() < np.exp(-(new_peak - cur_peak) / (T + 1e-10))
                else:
                    accept = False
                if accept and new_peak < best_new:
                    best_new = new_peak
                    best_move = (oi, zi)
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
        elif no_imp > 150 and np.random.random() < 0.25:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
    return best_h, best_c

def continuous_refine(h, deadline, n_perturb=20, lr=0.003):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros(N)
    while time.time() < deadline:
        corr = compute_corr(h)
        max_v = float(np.max(corr))
        c = max_v * (2.0 / N)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
        n_top = max(10, int(0.08 * N))
        thresh = np.sort(corr)[-n_top]
        mask = corr >= thresh
        w = np.where(mask, (corr - corr.min()) / (corr.max() - corr.min() + 1e-10), 0.0)
        wsum = w.sum()
        if wsum > 1e-10:
            w /= wsum
        else:
            w = np.zeros(2*N-1); w[N-1] = 1.0
        center = N - 1
        grad = np.zeros(N)
        for m in np.where(w > 1e-5)[0]:
            M = abs(int(m) - center)
            wi = float(w[int(m)])
            if M == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif M < N:
                grad[np.arange(M, N)] += wi * (1.0 - h[:N-M])
                grad[:N-M] -= wi * h[M:]
        h_new = project(h + 0.85 * v - lr * grad, tgt)
        v = 0.82 * v + 0.25 * (h_new - h)
        h = h_new
        if np.random.random() < 0.01:
            lr *= 0.9
            if lr < 5e-5:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    d1 = min(deadline - 450, t0 + 180)
    h1 = project(h0 + np.random.randn(N0) * 0.05, N0 / 2.0)
    h1, c1 = soft_cont_opt(h1, d1, lr0=0.01, mom=0.88, top_frac=0.08)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    if time.time() < deadline - 300:
        d2 = min(deadline - 280, time.time() + 100)
        hb = fix_binary(best_h)
        hb, c2 = binary_sa(hb, d2, n_crit=40, s_sz=55, T0=0.05, cool=0.97)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()

    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 100)
        h3 = project(best_h + np.random.randn(N0) * 0.03, N0 / 2.0)
        h3, c3 = continuous_refine(h3, d3, n_perturb=20, lr=0.004)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 60)
        hb4 = fix_binary(best_h)
        hb4, c4 = binary_sa(hb4, d4, n_crit=35, s_sz=50, T0=0.03, cool=0.96)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()

    if time.time() < deadline - 60:
        d5 = min(deadline - 45, time.time() + 80)
        h5 = project(best_h + np.random.randn(N0) * 0.02, N0 / 2.0)
        h5, c5 = soft_cont_opt(h5, d5, lr0=0.005, mom=0.92, top_frac=0.06)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    for N_try in [384, 390, 396]:
        if time.time() > deadline - 30:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d6 = min(deadline - 15, time.time() + 35)
        h_bt, c6 = binary_sa(h_bt, d6, n_crit=30, s_sz=45, T0=0.025, cool=0.96)
        hc = project(h_bt + np.random.randn(N_try) * 0.03, N_try / 2.0)
        d6b = min(deadline - 10, time.time() + 20)
        hc, c6b = continuous_refine(hc, d6b, lr=0.003)
        c6f = min(c6, c6b)
        if c6b < c6:
            h_bt = hc
        if c6f < best_c:
            best_c, best_h = c6f, h_bt.copy()

    if time.time() < deadline - 5:
        hB = fix_binary(best_h)
        d7 = min(deadline - 2, time.time() + 15)
        hB, c7 = binary_sa(hB, d7, n_crit=40, s_sz=50, T0=0.015, cool=0.95)
        if c7 < best_c:
            best_h = hB
        hc_final = project(best_h + np.random.randn(len(best_h)) * 0.01, len(best_h) / 2.0)
        d8 = min(deadline - 1, time.time() + 5)
        hc_final, c8 = continuous_refine(hc_final, d8, lr=0.002)
        if c8 < best_c:
            best_h = hc_final

    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
