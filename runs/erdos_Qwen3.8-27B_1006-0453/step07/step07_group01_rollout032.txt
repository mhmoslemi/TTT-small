When A[q] changes from 0 to 1 (and B[q] from 1 to 0):

        pk = float(np.max(corr))

            best_h = h.copy()
        # Gradient of soft max

        c_top = corr[idx]

        grad = np.zeros(N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N, seed=42):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_V(h, N):
    V = np.zeros((N, 2 * N - 1))
    r = np.arange(N)
    for p in range(N):
        ps = p + np.arange(-(N - 1), N)
        pm = p - np.arange(-(N - 1), N)
        mp = (ps >= 0) & (ps < N)
        mm = (pm >= 0) & (pm < N)
        V[p, mp] = h[ps[mp]]
        V[p, mm] += h[pm[mm]]
    return V

def update_V(V, p, q, N):
    r = np.arange(N)
    s1 = p + N - 1 - r
    v1 = (s1 >= 0) & (s1 < 2 * N - 1)
    V[r[v1], s1[v1]] -= 1.0
    s2 = r - p + N - 1
    v2 = (s2 >= 0) & (s2 < 2 * N - 1)
    V[r[v2], s2[v2]] -= 1.0
    s1 = q + N - 1 - r
    v1 = (s1 >= 0) & (s1 < 2 * N - 1)
    V[r[v1], s1[v1]] += 1.0
    s2 = r - q + N - 1
    v2 = (s2 >= 0) & (v2)
    V[r[v2], s2[v2]] += 1.0

def nesterov_opt(h_init, N, deadline, k=40, T=60.0, lr0=0.02):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            sv = int(idx[ii] - center)
            wi = w[ii]
            if sv == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif sv > 0:
                grad[sv:] += wi * (1.0 - h[:N - sv])
                grad[:N - sv] -= wi * h[sv:]
            else:
                M = -sv
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.996
            beta = min(0.98, 0.85 + it * 0.00015)
    return best_h, best_pk * (2.0 / N)

def sa_local_search(h, N, deadline, top_k=60, seed=99):
    h = ensure_binary(h, N, seed).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    V = compute_V(h, N)
    rng = np.random.RandomState(seed)
    stalls = 0
    max_stall = 25
    T_sa = 1.5
    r_arr = np.arange(N)
    it = 0
    while time.time() < deadline:
        it += 1
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            T_sa = min(T_sa * 1.1, 2.0)
        else:
            stalls += 1
            T_sa *= 0.995
            if T_sa < 0.05:
                T_sa = 0.05

        if stalls > max_stall:
            n_pert = min(2 + stalls // 15, 6)
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            for _ in range(n_pert):
                p = int(rng.choice(ones_arr))
                q = int(rng.choice(zeros_arr))
                if p != q:
                    h[p], h[q] = 0.0, 1.0
                    update_V(V, p, q, N)
            c = fft_corr(h)
            stalls = 0
            continue

        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        top_idx = np.argpartition(c, -top_k)[-top_k:]

        found = False
        # Try each 1-position, find best zero-partner
        sub_A = c[top_idx]
        V_zero = V[zeros_arr, top_idx]
        for pi in range(len(ones_arr)):
            p = int(ones_arr[pi])
            A = sub_A + V[p, top_idx]
            B = A[None, :] - V_zero
            nm = B.max(axis=1)
            mi = int(np.argmin(nm))
            if nm[mi] < P - 1e-12:
                q = int(zeros_arr[mi])
                new_c = c + V[p, :] - V[q, :]
                true_max = float(new_c.max())
                if true_max < P - 1e-12:
                    h[p], h[q] = 0.0, 1.0
                    update_V(V, p, q, N)
                    c = new_c
                    found = True
                    stalls = 0
                    break
        if not found:
            if T_sa > 0.02 and rng.random() < 0.4:
                p = int(rng.choice(ones_arr))
                q = int(rng.choice(zeros_arr))
                if p != q:
                    delta = float((c + V[p, :] - V[q, :]).max()) - P
                    if delta < 0 or rng.random() < np.exp(-delta / T_sa):
                        h[p], h[q] = 0.0, 1.0
                        update_V(V, p, q, N)
                        c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def multi_swap_search(h, N, deadline, top_k=50, seed=77):
    h = ensure_binary(h, N, seed).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    V = compute_V(h, N)
    rng = np.random.RandomState(seed)
    stalls = 0
    r_arr = np.arange(N)
    while time.time() < deadline:
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1

        if stalls > 30:
            n_p = rng.randint(2, 4)
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            ps = rng.choice(ones_arr, min(n_p, len(ones_arr)), replace=False)
            qs = rng.choice(zeros_arr, min(n_p, len(zeros_arr)), replace=False)
            for pp, qq in zip(ps, qs):
                h[pp], h[qq] = 0.0, 1.0
            c = fft_corr(h)
            V = compute_V(h, N)
            stalls = 0
            continue

        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        sub_A = c[top_idx]
        V_zero_top = V[zeros_arr, top_idx]

        found = False
        for pi in range(len(ones_arr)):
            p = int(ones_arr[pi])
            A = sub_A + V[p, top_idx]
            B = A[None, :] - V_zero_top
            nm = B.max(axis=1)
            mi = int(np.argmin(nm))
            if nm[mi] < P - 1e-12:
                q = int(zeros_arr[mi])
                new_c = c + V[p, :] - V[q, :]
                if float(new_c.max()) < P - 1e-12:
                    h[p], h[q] = 0.0, 1.0
                    update_V(V, p, q, N)
                    c = new_c
                    found = True
                    stalls = 0
                    break
        if not found and stalls < 10:
            # Try 2-2 swap
            if len(ones_arr) >= 2 and len(zeros_arr) >= 2:
                p1 = int(rng.choice(ones_arr))
                p2 = int(rng.choice(ones_arr))
                if p1 == p2:
                    continue
                q1 = int(rng.choice(zeros_arr))
                q2 = int(rng.choice(zeros_arr))
                if q1 in (p1, p2) or q2 in (p1, p2) or q1 == q2:
                    continue
                new_c = c + V[p1, :] + V[p2, :] - V[q1, :] - V[q2, :]
                if float(new_c.max()) < P - 1e-12:
                    h[p1], h[p2], h[q1], h[q2] = 0.0, 0.0, 1.0, 1.0
                    update_V(V, p1, q1, N)
                    update_V(V, p2, q2, N)
                    c = new_c
                    stalls = 0

    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    h0 = np.array(initial_h_values, dtype=float)

    best_c5 = compute_c5(best_h, N0)

        if c5_n < best_c5:
            best_c5, best_h = c5_n, h_n.copy()

    # Phase 2: Main SA local search on N0 (bulk of time)

    # Phase 3: Second local search with different params
    if time.time() < deadline - 50:

    h_pat = ensure_binary(h_pat, N_alt, seed=55)
    if time.time() < deadline - 40:

        h4, c5_4 = sa_local_search(h_pat, N_alt, d4, top_k=60, seed=seed + 200)

    if time.time() < deadline - 10:

        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
