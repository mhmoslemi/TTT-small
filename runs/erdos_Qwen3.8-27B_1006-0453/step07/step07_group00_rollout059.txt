import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def make_binary(h, N):
    h = np.array(h, dtype=float).copy()
    n1 = int(round(h.sum()))
    tgt = N // 2
    if n1 > tgt:
        idx = np.where(h > 0.5)[0]
        h[np.random.choice(idx, n1 - tgt, replace=False)] = 0
    elif n1 < tgt:
        idx = np.where(h < 0.5)[0]
        h[np.random.choice(idx, tgt - n1, replace=False)] = 1
    return (h > 0.5).astype(float)

def corr_delta(h):
    N = len(h)
    corr = fft_corr(h)
    L = 2 * N - 1
    delta = np.zeros((N, L))
    s = 2.0 * h - 1.0
    for i in range(N):
        delta[i, i:i+N] = s
    return corr, s, delta

def sa(h_in, corr_in, s_in, d_in, deadline, T0=0.04, cool=0.999, npk=8, npp=20, max_ni=4000):
    N = len(h_in)
    h = h_in.copy()
    corr = corr_in.copy()
    s = s_in.copy()
    delta = d_in.copy()
    best = float(corr.max())
    bh = h.copy()
    T = T0
    ni = 0
    it = 0
    while time.time() < deadline and ni < max_ni:
        it += 1
        if it % 500 == 0:
            T = max(T * cool, 0.001)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        pk = np.argpartition(corr, -npk)[-npk:]
        cp, cq = [], []
        for p in pk:
            ko = int(p) - (N - 1)
            if ko >= 0:
                v = np.arange(0, N - ko)
                m = (h[v] == 1) & (h[v + ko] == 0)
                cp.extend(v[m].tolist()); cq.extend((v + ko)[m].tolist())
            else:
                M = -ko
                v = np.arange(M, N)
                m = (h[v] == 1) & (h[v - M] == 0)
                cp.extend(v[m].tolist()); cq.extend((v - M)[m].tolist())
        if cp:
            ns = min(npp * npk, len(cp))
            sel = np.random.choice(len(cp), ns, replace=False)
            cp = [cp[i] for i in sel]; cq = [cq[i] for i in sel]
        nr = min(10, len(ones), len(zeros))
        cp.extend(np.random.choice(ones, nr, replace=False).tolist())
        cq.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cp = np.array(sorted(set(cp)), dtype=int)
        cq = np.array(sorted(set(cq)), dtype=int)
        if not len(cp) or not len(cq):
            break
        bn, bi, bj = best + 1.0, 0, 0
        ch = 6
        for ci in range(0, len(cp), ch):
            ce = min(ci + ch, len(cp))
            d1 = delta[ci:ce, np.newaxis, :]
            for cj in range(0, len(cq), 30):

            delta[q, :] = 0; delta[q, q:q+N] = s
            c = float(corr.max())
            if c < best - 1e-12:
                best = c; bh = h.copy(); ni = 0
            else:

            ni += 1

def cont_opt(h_in, N, deadline, lr=0.005, ktop=50, alpha=10):

    m_v = np.zeros(N); v_v = np.zeros(N)
    b1, b2 = 0.9, 0.999
    t = 0
    best = 1e10
    bh = h.copy()
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        mx = float(corr.max())

        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * (grad * grad)
        m_hat = m_v / (1.0 - b1**t)

        h = h_new
        if t % 400 == 0:

    return bh, best
