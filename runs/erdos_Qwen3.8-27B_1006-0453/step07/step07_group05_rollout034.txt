import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum_binary(h, target):
    h = h.copy().astype(float)
    h[h < 0.5] = 0
    h[h >= 0.5] = 1
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        sel = np.random.permutation(idx)[:n1-target]
        h[sel] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        sel = np.random.permutation(idx)[:target-n1]
        h[sel] = 1
    return h

def defect_score(corr, top_m=10):
    """Score based on top M peaks only (smoother objective)"""
    idx = np.argsort(-corr)[:top_m]
    return np.sum(corr[idx])

def ls_focused(h, N, deadline, max_iters=50000):

def continuous_opt(h, N, deadline):
    """Continuous relaxation with log-sum-exp smoothing"""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = 0.003
    beta = 0.02  # temperature for log-sum-exp
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        peak = np.max(corr)
        # Log-sum-exp gradient weight
        e = (corr - peak) / beta
        e -= e.max()
        w = np.exp(e)
        w /= w.sum()
        
        # Gradient: d(corr[k])/dh[i]
        # corr[k] = sum_j h[j] * (1-h[j+k-N+1]) (for valid range)
        # d(corr[k])/dh[i] = (1 - h[i+k-N+1]) - h[i-(N-1-k)] for valid terms
        # This is complex; use finite difference approximation for top-k terms
        grad = np.zeros(N)
        center = N - 1
        sig = np.where(w > 1e-4)[0]
        for ki in sig:
            L = int(ki - center)
            wt = w[ki]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wt * (1.0 - h[:N-L])
                grad[:N-L] -= wt * h[L:]
            else:
                M = -L
                grad[:N-M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N-M]
        
        h_new = h - lr * grad
        # Project to [0,1] with sum constraint
        h_new = np.clip(h_new, 0, 1)
        # Binary search for offset to fix sum
        lo, hi = -2.0, 2.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        
        c_new = compute_c5(h_new, N)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        it += 1

        grad = np.zeros(N)

            elif d > 0:

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum_binary(h, target):
    h = h.copy().astype(float)
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1.0)[0]
        sel = np.random.permutation(idx)[:n1 - target]
        h[sel] = 0.0
    elif n1 < target:
        idx = np.where(h == 0.0)[0]
        sel = np.random.permutation(idx)[:target - n1]
        h[sel] = 1.0
    return h

def binary_ls(h, N, deadline, ncand=14, kick_n=5, kick_lim=4000):

    cur = float(np.max(fft_corr(h)))

    best_h = h.copy()

    while time.time() < deadline:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        best_mv = None
        for _ in range(ncand):
            r = np.random.random()
            if r < 0.45:
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi] = 0.0; h[zi] = 1.0
                d = float(np.max(fft_corr(h))) - cur
                h[oi] = 1.0; h[zi] = 0.0
                if d < best_d:

                a = int(np.random.randint(0, N - L))
                h2 = h.copy()
                h2[a:a+L] = h[a:a+L][::-1]
                d = float(np.max(fft_corr(h2))) - cur
                if d < best_d:
                    best_d = d; best_mv = (1, a, L, 0)
            else:
                L = int(np.random.choice([1, 2, 3, 4, 5, 7, 10]))
                obs, zbs, i = [], [], 0
                while i < N:

                        j = i

                        if j - i >= L: obs.append(i)
                        i = j
                    else:

                if not obs or not zbs:

                oa = obs[int(np.random.randint(len(obs)))]

def cont_opt(h, N, deadline, lr0=0.005, beta0=0.04):

    target = N / 2.0
    best_h = h.copy()

    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        peak = float(np.max(corr))
        e = (corr - peak) / beta
        e -= e.max()
        w = np.exp(e)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        sig = np.where(w > 3e-4)[0]
        for ki in sig:
            d = int(ki - center)
            wt = w[ki]

                grad += wt * (1.0 - 2.0 * h)

                grad[d:] += wt * (1.0 - h[:N-d])
