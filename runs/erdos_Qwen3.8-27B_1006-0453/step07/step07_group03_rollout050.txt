Actually, let me use the "compute_D" approach from the previous code. The key insight is:

If we flip h[p] from 1 to 0 and h[q] from 0 to 1, the correlation changes predictably. Specifically:

corr_new[k] = corr_old[k] + delta[k]

where delta[k] depends on p, q, and the current h values.

For flipping h[p]: 1→0
- corr[k] loses the contribution h[p]*(1-h[p-shift(k)]) = (1-h[p-shift(k)]) for valid p-shift(k)
- For the (1-h) factor: when i-shift(k) = p, i.e., i = p+shift(k), the term h[i]*(1-h[p]) changes from h[i]*0 to h[i]*1, so corr[k] gains h[p+shift(k)]

For flipping h[q]: 0→1
- corr[k] gains h[q]*(1-h[q-shift(k)]) = (1-h[q-shift(k)]) for valid q-shift(k)  
- For the (1-h) factor: when i-shift(k) = q, i.e., i = q+shift(k), the term h[i]*(1-h[q]) changes from h[i]*1 to h[i]*0, so corr[k] loses h[q+shift(k)]

So the total delta for flipping p (1→0) and q (0→1):
delta[k] = -(1-h[p-shift(k)]) + h[p+shift(k)] + (1-h[q-shift(k)]) - h[q+shift(k)]
           (for valid indices)

This can be computed in O(N) for each (p,q) pair. But we still need to check all (p,q) pairs to find the best one.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_grad(h, N, corr, tau, k):
    center = N - 1
    m = corr.max()
    sh = (corr - m) / tau
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    top_idx = np.argpartition(w, -k)[-k:]
    grad = np.zeros(N)
    for ki in top_idx:
        wi = w[ki]
        if wi < 1e-12:
            continue
        L = int(ki) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad

def compute_D(h, N):
    center = N - 1
    D = np.zeros((N, 2 * N - 1))
    for p in range(N):
        for L in range(-(N - 1), N):
            ki = L + center
            j1 = p - L

            val = 0.0
            if 0 <= j1 < N:
                val += (1.0 - 2.0 * h[j1]) if L == 0 else (1.0 - h[j1])
            if L > 0 and 0 <= j2 < N:
                val -= h[j2]
            elif L < 0 and 0 <= j2 < N:
                val -= h[j2]
            elif L == 0:
                val -= h[p]
            D[p, ki] = val
