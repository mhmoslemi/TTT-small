    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / (e.sum() + 1e-30)

    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def informed_greedy(h, N, deadline, max_iters=5000, ntries=20):

    nh = N // 2
    n1 = int(h.sum())

        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0

        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters and stall < 250:
        corr = fft_corr(h)
        P = float(np.max(corr))
        worst_lags = np.where(corr >= P - 1e-4)[0]
        ctr = N - 1
        candidates = []
        for L in worst_lags[:3]:
            lag = int(L - ctr)
            if lag > 0:
                for i in range(N - lag):
                    if h[i] > 0.5 and h[i + lag] < 0.5:
                        candidates.append((i, i + lag))
                for i in range(lag, N):
                    if h[i] < 0.5 and h[i - lag] > 0.5:
                        candidates.append((i, i - lag))
            elif lag < 0:
                M = -lag
                for i in range(M, N):
                    if h[i] > 0.5 and h[i - M] < 0.5:
                        candidates.append((i, i - M))
                for i in range(N - M):
                    if h[i] < 0.5 and h[i + M] > 0.5:
                        candidates.append((i, i + M))
        if len(candidates) > ntries:
            candidates = [candidates[np.random.randint(len(candidates))] for _ in range(ntries)]
        best_d, best_p, best_q = 0.0, -1, -1

            h[p], h[q] = 1.0 - h[p], 1.0 - h[q]

                continue
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > best_d:

            h[p], h[q] = 1.0 - h[p], 1.0 - h[q]
        if best_p >= 0 and best_d > 1e-14:
            h[best_p], h[best_q] = 1.0 - h[best_p], 1.0 - h[best_q]
            np = P - best_d
            if np < best_c:
                best_c = np
                best_h = h.copy()
            stall = 0
        else:

def sim_anneal(h, N, deadline, max_iters=20000, ntries=12, T0=0.4, cool=0.99985):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    it = 0
    while time.time() < deadline and it < max_iters:
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
        best_d, best_i, best_j = -np.inf, -1, -1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(ntries):
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            h[i], h[j] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > best_d:
                best_d, best_i, best_j = d, i, j
            h[i], h[j] = 1.0, 0.0
        if best_i >= 0:
            if best_d > 0 or (T > 1e-7 and np.random.random() < np.exp(best_d / T)):
                h[best_i], h[best_j] = 0.0, 1.0
        T *= cool
        it += 1
    return best_h, best_c

def softmax_grad(h, T, lam, target):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    ws = e.sum() + 1e-30
    w = e / ws
    obj = mx + np.log(ws) / T + lam * (h.sum() - target) ** 2
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        wi = w[i]
        if wi < 1e-10:
            continue
        L = i - ctr
        if L == 0:
            g += wi * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wi * (1.0 - h[:N - L])
            g[:N - L] -= wi * h[L:]
        else:
            M = -L
            g[:N - M] += wi * (1.0 - h[M:])
            g[M:] -= wi * h[:N - M]
    g += 2.0 * lam * (h.sum() - target)
    return obj, g

def cont_opt(h0, N, deadline):
    from scipy.optimize import minimize
    target = N / 2.0
    best_c = compute_c5(h0, N)
    best_h = h0.copy()
    for T in [60, 40, 25, 15]:
        if time.time() > deadline:
            break
        h = h0 + np.random.randn(N) * 0.1
        h = proj01(h, target)
        for _ in range(2):
            if time.time() > deadline:
                break
            res = minimize(softmax_grad, h, args=(T, 15.0, target),
                          jac=True, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * N,
                          options={'maxiter': 80, 'ftol': 1e-14})
            h = proj01(res.x, target)
            c = compute_c5(h, N)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            h0 = h.copy()
    return best_h, best_c

def multi_res(best_h, best_c5, N, deadline):
