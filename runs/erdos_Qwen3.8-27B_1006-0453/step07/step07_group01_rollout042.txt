import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12: return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target: lo = mid
        else: hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def grad_sm(corr, h, N, k, T):

    ct = corr[idx]
    w = np.exp(T*(ct - ct.max())); w /= w.sum()

        else:

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def grad_sm(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    ct = corr[idx]
    w = np.exp(T*(ct - ct.max()))
    w /= w.sum()
    g = np.zeros(N)

        else:
            M = -s
            g[:N-M] += wi*(1.0-h[M:])

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    p_all = np.arange(N)[:,None]
    s_all = shifts[None,:]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm>=0) & (pm<N)
    vp = (pp>=0) & (pp<N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = -(1.0-hm)*vm.astype(float) + hp*vp.astype(float)
    return D

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_sm(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    ct = corr[idx]
    w = np.exp(T * (ct - ct.max()))
    w /= w.sum()
    g = np.zeros(N)
    ctr = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - ctr)
        wi = w[ii]
        if s == 0:
            g += wi * (1.0 - 2.0 * h)
        elif s > 0:
            g[s:] += wi * (1.0 - h[:N - s])
            g[:N - s] -= wi * h[s:]
        else:
            M = -s
            g[:N - M] += wi * (1.0 - h[M:])
            g[M:] -= wi * h[:N - M]
    return g / k

def nesterov(h0, N, deadline, k=40, T=60.0, lr0=0.015):
    h = h0.astype(float).copy()
    tgt = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        g = grad_sm(corr, y, N, k, T)
        h_new = project_sum(y - lr * g, tgt)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0002)
        if it % 300 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(10.0, min(400.0, T * (0.88 if gap < 0.008 else 1.06)))
    return best_h, best_pk * (2.0 / N)

def ensure_bin(h, N):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    tgt = N // 2
    rng = np.random.RandomState(42)
    if n1 > tgt:
        hx = np.where(hb == 1)[0]
        idx = rng.choice(hx, n1 - tgt, replace=False)
        hb[idx] = 0
    elif n1 < tgt:
        hx = np.where(hb == 0)[0]
        idx = rng.choice(hx, tgt - n1, replace=False)
        hb[idx] = 1
    return hb

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def bin_ls_sa(h, N, deadline, max_sweeps=5000, T0=0.4, Tf=0.01, pert_all=25, sample_n=0):
    h = ensure_bin(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    D = compute_D(h, N)
    stall = 0
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        frac = min(1.0, float(sweep) / (0.8 * max(1, max_sweeps)))
        T_sa = T0 * (Tf / T0) ** frac
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        if sample_n > 0 and len(ones) > sample_n:
            p_list = rng.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        accepted = False
        if best_pq is not None:
            delta = best_new_P - P
            if delta < 0:
                accepted = True
            elif stall > 3 and delta < 4.0 * T_sa:
                if rng.random() < np.exp(-delta / max(T_sa, 1e-10)):
                    accepted = True
            if accepted:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
                D = compute_D(h, N)
                c = fft_corr(h)
        else:
            if stall > pert_all:
                n_r = min(2 + stall // 12, 8)
                for _ in range(n_r):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                D = compute_D(h, N)
                c = fft_corr(h)
                stall = 0
            elif stall > pert_all * 4:
                n_r = min(5 + stall // 8, 14)
                for _ in range(n_r):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                D = compute_D(h, N)
                c = fft_corr(h)
                stall = 0
            else:
                break
    return best_h, best_c * (2.0 / N)

def bin_ls_2swap(h, N, deadline, max_rounds=500, ns=10):
    h = ensure_bin(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    D = compute_D(h, N)
    stall = 0
    rng = np.random.RandomState(777)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 35:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        Dz = D[zeros, :]
        n_s = min(ns, len(ones), len(zeros))
        ps = rng.choice(ones, n_s, replace=False)
        qs = rng.choice(zeros, n_s, replace=False)
        best_new_P = P
        best_quad = None
        Dp = D[ps]
        Dq = Dz[qs]
        for i in range(n_s):
            for j in range(i + 1, n_s):
                dp = Dp[i] + Dp[j]
                for a in range(n_s):
                    for b in range(a + 1, n_s):
                        nc = c + dp - Dq[a] - Dq[b]
                        mval = float(nc.max())
                        if mval < best_new_P:
                            best_new_P = mval
                            best_quad = (int(ps[i]), int(ps[j]), int(qs[a]), int(qs[b]))
        if best_quad is not None:
            p1, p2, q1, q2 = best_quad
            h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
            D = compute_D(h, N)
            c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
