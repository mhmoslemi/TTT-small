def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):

    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):

    if abs(s - target_sum) < 1e-12:

        else:

    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h, target_ones):
    N = len(h)
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > target_ones:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target_ones]] = 0
    elif n1 < target_ones:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target_ones - n1]] = 1
    return h

def grad_top(h, corr, k, center):
    N = len(h)
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            grad[M:] += (1.0 - h[:N-M])
            grad[:N-M] -= h[M:]
    return grad / k

def cont_opt(h, deadline, k_init=30, k_max=100, lr0=0.03, mom=0.85):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    lr = lr0
    k = k_init
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 40:
                lr *= 0.85
                k = min(k + 5, k_max)
                stale = 0
            if stale > 150:
                break
        g = grad_top(h, corr, k, center)
        h_new = project(h + mom * v - lr * g, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def swap_eval(h, corr, center, top_lags, oi, zi, N):
    """Estimate new peak after swapping h[oi]=1 to 0 and h[zi]=0 to 1."""
    new_peak = 0.0
    for L in top_lags:
        cL = corr[center + L]
        d = 0.0
        if 0 <= oi + L < N:
            d += h[oi + L]
        if 0 <= oi - L < N:
            d += h[oi - L]
        if 0 <= zi + L < N:
            d -= h[zi + L]
        if 0 <= zi - L < N:
            d -= h[zi - L]
        val = cL + d
        if val > new_peak:
            new_peak = val
    return new_peak

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_dpeak = 0.0
        best_move = None
        # Evaluate all swaps (or a large random subset if too many)
        n_pairs = len(ones) * len(zeros)

                if dp < best_dpeak:

            break
        h[best_move[0]] = 0
        h[best_move[1]] = 1

        if best_move is None or best_dp >= -1e-9:
            break
        h[best_move[0]] = 0
        h[best_move[1]] = 1
        passes += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Continuous opt (70s)
    if time.time() < deadline - 480:
        d1 = min(deadline - 480, time.time() + 70)
        h1 = project(best_h + np.random.randn(N0) * 0.05, N0 / 2.0)
        h1, c1 = cont_opt(h1, d1, k_init=22, k_max=70, lr0=0.025, mom=0.86)
        if c1 < best_c:
            best_c, best_h = c1, h1.copy()

    # Phase 2: Binary SA (250s)
    if time.time() < deadline - 280:
        d2 = min(deadline - 280, time.time() + 250)
        hb2 = fix_binary(best_h, N0 // 2)
        hb2, c2 = binary_sa(hb2, d2, tabu_len=50, max_iters=5000, n_crit=25)
        if c2 < best_c:
            best_c, best_h = c2, hb2.copy()

    # Phase 3: Exhaustive polish (100s)
    if time.time() < deadline - 140:
        d3 = min(deadline - 140, time.time() + 100)
        h3 = fix_binary(best_h, N0 // 2)
        h3, c3 = binary_polish(h3, d3, n_crit=35)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Perturbed SA (120s)
    if time.time() < deadline - 80:
        d4 = min(deadline - 80, time.time() + 120)
        h4 = best_h.copy()
        n_flip = max(3, N0 // 40)
        idx1 = np.random.choice(np.where(h4 > 0.5)[0], n_flip, replace=False)
        idx0 = np.random.choice(np.where(h4 <= 0.5)[0], n_flip, replace=False)
        h4[idx1] = 0.0
        h4[idx0] = 1.0
        hb4 = fix_binary(h4, N0 // 2)
        hb4, c4 = binary_sa(hb4, d4, tabu_len=80, max_iters=4000, n_crit=30)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()

    # Phase 5: Polish again (60s)
    if time.time() < deadline - 35:
        d5 = min(deadline - 35, time.time() + 55)
        h5 = fix_binary(best_h, N0 // 2)
        h5, c5 = binary_polish(h5, d5, n_crit=50)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Final short SA (25s)
    if time.time() < deadline - 5:
        d6 = min(deadline - 2, time.time() + 25)
        hb6 = fix_binary(best_h, N0 // 2)
        hb6, c6 = binary_sa(hb6, d6, tabu_len=100, max_iters=2000, n_crit=40)
        if c6 < best_c:
            best_c, best_h = c6, hb6.copy()

    final_c = compute_c5(best_h)
    hB = fix_binary(best_h, N0 // 2)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
