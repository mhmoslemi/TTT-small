import time

    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def build_deltas(h):

                use_2swap=True, p_2swap=0.2):

    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    accept_count = 0
    total_count = 0
    
    while time.time() < deadline and it < max_iters:

        if it % 500 == 0:
            T = max(T * cool_rate, 0.001)
            # Adaptive temperature

                rate = accept_count / total_count
                if rate < 0.05:
                    T = min(T * 1.5, t_start)
                accept_count = 0
                total_count = 0

        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:

        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
