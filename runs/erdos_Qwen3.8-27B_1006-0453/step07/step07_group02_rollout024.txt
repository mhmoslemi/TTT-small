import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def binary_search_v2(h, N, deadline, max_iters=10000, sample_sz=None, 
                     kick_thresh=50, kick_sz=5, temp0=0.0, cooling=0.999):
    h = h.astype(float).copy()
    nh = N // 2
    # Fix sum
    while h.sum() > nh:
        h[np.random.choice(np.where(h == 1)[0])] = 0
    while h.sum() < nh:
        h[np.random.choice(np.where(h == 0)[0])] = 1
    
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    C = np.vstack([compute_C_single(h, N, i) for i in range(N)])
    stuck = 0
    temp = temp0
    it = 0
    if sample_sz is None:
        sample_sz = min(N // 2, 200)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        
        # Evaluate all pairs
        Co = C[os_]  # (ns, L)
        Cz = C[zs]  # (nz, L)
        new = corr[None, None, :] + Co[None, :, :] + Cz[:, None, :]
        newP = new.max(axis=2)  # (nz, ns)

        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])

            h[zs[mi[0]]] = 1
            C[os_[mi[1]]] = compute_C_single(h, N, os_[mi[1]])
            C[zs[mi[0]]] = compute_C_single(h, N, zs[mi[0]])
        elif temp > 0 and stuck > 10:

            if np.random.random() < np.exp(-delta / max(temp, 1e-8)):

                q = int(zs[mi[0]])
                h[p] = 0

                C[p] = compute_C_single(h, N, p)

        elif stuck >= kick_thresh:

                h[p] = 0

                C[q] = compute_C_single(h, N, q)
            else:
                break
    
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def build_C(h, N):
    return np.array([compute_C_single(h, N, i) for i in range(N)])

def fix_sum_binary(h, nh):
    while h.sum() > nh:
        h[np.random.choice(np.where(h == 1)[0])] = 0
    while h.sum() < nh:
        h[np.random.choice(np.where(h == 0)[0])] = 1
    return h

            best_h = h.copy()

        else:
            stuck += 1
        if stuck > max_stall:

        ones = np.where(h == 1)[0]
