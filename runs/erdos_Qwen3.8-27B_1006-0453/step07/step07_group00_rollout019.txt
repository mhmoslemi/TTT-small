            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:

            else:

        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - y)
        h = h_new

    x_old2 = x_old * (2.0 / N)
    x_new2 = x_new * (2.0 / n_new)
    return np.interp(x_new2, x_old2, h)

                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0.0
                deltas[p, p:p + N] = s
                deltas[q, :] = 0.0
                deltas[q, q:q + N] = s
                c = float(corr.max())
                if c < raw_best - 1e-12:
                    raw_best = c
                    best_h = h.copy()
                    no_impr = 0
                else:
                    no_impr += 1
            else:
                no_impr += 1
    return best_h, raw_best * (2.0 / N)

def upsample(h, n_new):
    N = len(h)
    if n_new == N:
        return h.copy()
    x_old = np.arange(N) * (2.0 / N)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Primary SA (long run, targeted peaks)
    d1 = min(deadline - 500, t0 + 350)
    h1, c1 = sa_core(h_bin, d1, t_start=0.05, n_peaks=8, n_cand=25, cool_rate=0.9990)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Double-swap SA for structural diversity
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 150)
        h2, c2 = double_swap_sa(best_h, d2, t_start=0.018, mix_ratio=0.35)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Multi-resolution escape
    if time.time() < deadline - 280:
        N_coarse = max(80, (N0 // 2) // 2 * 2)
        hc_in = make_binary(upsample(best_h, N_coarse), N_coarse)
        d3a = min(deadline - 200, time.time() + 90)
        hc, cc = sa_core(hc_in, d3a, t_start=0.04, n_peaks=5, n_cand=15, cool_rate=0.999)
        h_up = make_binary(upsample(hc, N0), N0)
        # Compare with best at same resolution via interp
        h_up2 = make_binary(upsample(best_h, N_coarse), N_coarse)
        # Continue from upsampled coarse
        d3b = min(deadline - 100, time.time() + 140)
        h_up, cu = sa_core(h_up, d3b, t_start=0.03, n_peaks=7, n_cand=22, cool_rate=0.9993)
        cu = compute_c5(h_up, N0)
        if cu < best_c5:
            best_c5, best_h = cu, h_up.copy()

    # Phase 4: ILS - iterated local search with perturbation
    if time.time() < deadline - 150:
        h_ils = best_h.copy()
        for restart in range(3):
            d_ils = min(deadline - 50 - (2 - restart) * 30, time.time() + 50 + restart * 10)
            h_ils, c_ils = sa_core(h_ils, d_ils, t_start=0.012, n_peaks=10, n_cand=30, cool_rate=0.9997)
            if c_ils < best_c5:
                best_c5, best_h = c_ils, h_ils.copy()
            # Perturb for next restart
            n_pert = max(4, N0 // 40)
            ones = np.where(h_ils == 1)[0]
            zeros = np.where(h_ils == 0)[0]
            pf = np.random.choice(ones, min(n_pert // 2, len(ones)), replace=False)
            qf = np.random.choice(zeros, min(n_pert - n_pert // 2, len(zeros)), replace=False)
            h_ils = h_ils.copy()
            h_ils[pf] = 0
            h_ils[qf] = 1

    # Phase 5: Final fine SA polish
    if time.time() < deadline - 8:
        Ny = len(best_h)
        h5 = make_binary(best_h, Ny)
        d5 = deadline - 3
        h5, c5 = sa_core(h5, d5, t_start=0.012, n_peaks=10, n_cand=30, cool_rate=0.9998)
        c5 = compute_c5(h5, Ny)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Try alternate resolutions
    for N_try in [N0 - 20, N0 + 20, N0 // 2]:
        if time.time() > deadline - 25:
            break
        N_try = max(60, N_try)
        if N_try % 2 != 0:
            N_try += 1
        if N_try == len(best_h):
            continue
        hi = make_binary(upsample(best_h, N_try), N_try)
        d_sub = min(deadline - 10, time.time() + 25)
        hr, cr = sa_core(hi, d_sub, t_start=0.025, n_peaks=6, n_cand=18, cool_rate=0.9993)
        cr = compute_c5(hr, N_try)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
