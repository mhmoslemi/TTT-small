def softmax_cont(h_init, N, deadline, beta=80.0, top_k=25, lr0=0.04, mom0=0.82):
    """Continuous optimization using softmax-smoothed max correlation gradient."""
    h = h_init.astype(float).copy()
    h = project_box_sum(h, N / 2.0)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    target = N / 2.0
    mom = np.zeros(N)
    lr = lr0
    mom_a = mom0
    center = N - 1
    
    for it in range(50000):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        # Softmax weights on top correlations
        top_idx = np.argsort(-corr)[:top_k]
        top_vals = corr[top_idx]
        sv = top_vals - top_vals.max()
        w = np.exp(beta * sv)
        w /= w.sum()
        # Gradient
        grad = np.zeros(N)
        for i, ki in enumerate(top_idx):
            L = int(ki) - center
            wi = w[i]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[:N - L] += wi * (1.0 - h[L:])
                grad[L:] -= wi * h[:N - L]
            else:
                M = -L
                grad[M:] += wi * (1.0 - h[:N - M])
                grad[:N - M] -= wi * h[M:]
        # Nesterov momentum
        mom = mom_a * mom + (1 - mom_a) * grad
        h = project_box_sum(h - lr * mom, target)
        # Decay
        if it % 200 == 0:
            lr *= 0.98
            mom_a = min(0.95, mom_a + 0.0002)
        # Periodic rebinarization
        if it % 1000 == 999:
            h = ensure_binary(h, N)
            h = project_box_sum(h + np.random.randn(N) * 0.02, target)
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_binary(h, N):
    nh = N // 2
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    if n1 > nh:
        ones = np.where(hb == 1)[0]
        idx = np.random.choice(ones, n1 - nh, replace=False)
        hb[idx] = 0
    else:
        zeros = np.where(hb == 0)[0]
        idx = np.random.choice(zeros, nh - n1, replace=False)
        hb[idx] = 1
    return hb

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

        if len(ones) < 2 or len(zeros) < 2:

        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_nP = P - 1e-14
        best_pq = None
        for p in p_list:
            cand = (c - D[p, :])[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP:
                best_nP, best_pq = float(npk[mi]), (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
        c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def sa_binary(h, N, deadline, max_sw=10000, T0=4.0, Tcool=0.993, Tmn=0.03,
              sf=1.0, ke=110, ks=5):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c, best_h = float(c.max()), h.copy()
    T, stall = T0, 0
    for _ in range(max_sw):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c, best_h, stall = P, h.copy(), 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= ke:
            am = int(np.argmax(c))
            for _ in range(ks):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall, T = 0, T0 * 0.5
            c = fft_corr(h)
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_nP, best_pq = P + 1.0, None
        for p in p_list:
            cand = (c - D[p, :])[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP:
                best_nP, best_pq = float(npk[mi]), (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        d = best_nP - P
        if d < -1e-14 or (T > Tmn and np.random.random() < np.exp(-max(0.0, d) / max(T, 1e-10))):
            h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        c = fft_corr(h)
        T = max(Tmn, T * Tcool)
    return best_h, best_c * (2.0 / N)

def cont_softmax(h_init, N, deadline, beta=70.0, tk=25, lr0=0.04, ma=0.85):
    h = project_box_sum(h_init.astype(float).copy(), N / 2.0)
    best_h, best_pk = h.copy(), float(np.max(fft_corr(h)))
    target, mom = N / 2.0, np.zeros(N)
    lr, center = lr0, N - 1
    for it in range(80000):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk, best_h = pk, h.copy()
        top = np.argsort(-corr)[:tk]
        tv = corr[top]
        w = np.exp(beta * (tv - tv.max()))
        w /= w.sum()
        grad = np.zeros(N)
        for i, ki in enumerate(top):
            L = int(ki) - center
            wi = w[i]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[:N - L] += wi * (1.0 - h[L:])
                grad[L:] -= wi * h[:N - L]
            else:
                M = -L
                grad[M:] += wi * (1.0 - h[:N - M])
                grad[:N - M] -= wi * h[M:]
        mom = ma * mom + (1.0 - ma) * grad
        h = project_box_sum(h - lr * mom, target)
        if it % 300 == 0:
            lr *= 0.985
            ma = min(0.96, ma + 0.0003)
        if it % 2000 == 1999:
            h = ensure_binary(h, N).astype(float)
            h = project_box_sum(h + np.random.randn(N) * 0.015, target)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    bh, bc = h0.copy(), compute_c5(h0)

    # P1: Quick greedy
    hb = ensure_binary(h0, N0)
    h1, c1 = greedy_binary(hb, N0, min(dl - 950, t0 + 20), 200, 1.0)
    if c1 < bc: bc, bh = c1, h1

    # P2: SA main
    h2, c2 = sa_binary(ensure_binary(bh, N0), N0, min(dl - 780, t0 + 110),
                       12000, 4.5, 0.992, 0.03, 1.0, 100, 5)
    if c2 < bc: bc, bh = c2, h2

    # P3: Continuous softmax + greedy
    hc = project_box_sum(bh[:N0] + np.random.randn(N0) * 0.025, N0 / 2.0)
    h3, c3 = cont_softmax(hc, N0, min(dl - 620, t0 + 250), 70, 22, 0.035, 0.84)
    if c3 < bc: bc, bh = c3, h3
    h3g, c3g = greedy_binary(ensure_binary(bh, N0), N0, min(dl - 580, t0 + 290), 300, 1.0)
    if c3g < bc: bc, bh = c3g, h3g

    # P4: SA #2 (perturbed, high T)
    h4i = ensure_binary(bh, N0)
    for _ in range(10):
        p, q = np.where(h4i == 1)[0], np.where(h4i == 0)[0]
        h4i[p[np.random.randint(len(p))]], h4i[q[np.random.randint(len(q))]] = 0.0, 1.0
    h4, c4 = sa_binary(h4i, N0, min(dl - 440, t0 + 430), 9000, 5.5, 0.994, 0.04, 0.85, 80, 6)
    if c4 < bc: bc, bh = c4, h4

    # P5: Continuous #2 (sharper beta)
    hc5 = project_box_sum(ensure_binary(bh, N0).astype(float) + np.random.randn(N0) * 0.015, N0 / 2.0)
    h5, c5 = cont_softmax(hc5, N0, min(dl - 280, t0 + 590), 120, 15, 0.022, 0.88)
    if c5 < bc: bc, bh = c5, h5
    h5g, c5g = greedy_binary(ensure_binary(bh, N0), N0, min(dl - 240, t0 + 630), 250, 1.0)
    if c5g < bc: bc, bh = c5g, h5g

    # P6: SA #3 (low T, fine-tune)
    h6i = ensure_binary(bh, N0)
    for _ in range(4):
        p, q = np.where(h6i == 1)[0], np.where(h6i == 0)[0]
        h6i[p[np.random.randint(len(p))]], h6i[q[np.random.randint(len(q))]] = 0.0, 1.0
    h6, c6 = sa_binary(h6i, N0, min(dl - 100, t0 + 770), 8000, 2.5, 0.993, 0.015, 1.0, 150, 3)
    if c6 < bc: bc, bh = c6, h6

    # P7: Final continuous + greedy
    hc7 = project_box_sum(ensure_binary(bh, N0).astype(float) + np.random.randn(N0) * 0.01, N0 / 2.0)
    h7, c7 = cont_softmax(hc7, N0, min(dl - 40, t0 + 900), 150, 12, 0.015, 0.90)
    if c7 < bc: bc, bh = c7, h7
    h7g, c7g = greedy_binary(ensure_binary(bh, N0), N0, dl - 1, 500, 1.0)
    if c7g < bc: bc, bh = c7g, h7g

    fc = compute_c5(bh)
    if fc < bc: bc = fc
    bh = np.clip(bh, 0.0, 1.0)
    return bh, bc, len(bh)
