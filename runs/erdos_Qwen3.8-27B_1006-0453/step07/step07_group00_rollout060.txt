import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def cont_polish(h_in, N, deadline, lr, k_top):
    h = h_in.astype(float).copy()
    vx = h.copy()
    corr = fft_corr(vx)

    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        w = np.exp(corr[top] / 8.0)
        w /= w.sum()
        grad = np.zeros(N)
        for idx, t in enumerate(top):
            L = int(t - (N - 1))
            wi = w[idx]
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]

                M = -L
                grad[:N - M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N - M]

        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.965
            beta = min(0.96, 0.84 + it * 0.00015)
    return best_h, best_pk * (2.0 / N)
