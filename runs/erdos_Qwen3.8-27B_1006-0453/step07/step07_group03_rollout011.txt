    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def c5_val(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def ensure_sum(h, N):
    nh = N // 2
    h = np.asarray(h, dtype=float)
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    rng = np.random.RandomState(99 + N)
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        hb[rng.choice(idx, n1 - nh, replace=False)] = 0
    else:
        idx = np.where(hb == 0)[0]
        hb[rng.choice(idx, nh - n1, replace=False)] = 1
    return hb

def comp_D(h, N):
    S = np.arange(2 * N - 1) - (N - 1)
    P = np.arange(N)[:, None]
    sv = S[None, :]
    pm, pp = P - sv, P + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pc, qc = np.clip(pm, 0, N - 1), np.clip(pp, 0, N - 1)
    return h[pc] * vm - (1.0 - h[qc]) * vp

def _proj(h, tgt):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - tgt) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)
