    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_c5 = _eval_c5(h0, N0)
    best_h = h0.copy()

    d1 = t0 + 320
    h1, c1 = _fast_search(h0, N0, d1, T0=0.06, guided_frac=0.3, restart_every=60000)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    if time.time() < deadline - 350:
        d2 = t0 + 550
        h2, c2 = _fast_search(best_h, N0, d2, T0=0.025, guided_frac=0.6, restart_every=45000)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    if time.time() < deadline - 300:
        d3 = t0 + 720
        h3, c3 = _double_search(best_h, N0, d3, T0=0.02, restart_every=35000)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    for N_t in [370, 390, 400, 420]:
        if time.time() > deadline - 120:
            break
        bl_c = np.inf; bl_h = None
        for vi in range(3):
            if time.time() > deadline - 90:
                break
            hv = _init_seq(N_t, vi)
            ds = min(deadline - 80, time.time() + 55)
            hr, cr = _fast_search(hv, N_t, ds, T0=0.05, guided_frac=0.5, restart_every=30000)
            if cr < bl_c:
                bl_c, bl_h = cr, hr
        if bl_h is not None and bl_c < best_c5:
            best_c5, best_h = bl_c, bl_h

    if time.time() < deadline - 15:
        d5 = deadline - 1
        h5, c5 = _fast_search(best_h, len(best_h), d5, T0=0.004, guided_frac=0.8, restart_every=25000)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    bh = (best_h > 0.5).astype(float)
    _fix_sum(bh, len(bh)//2)
    bc = _eval_c5(bh, len(bh))
    if bc < best_c5:
        best_c5, best_h = bc, bh
    return best_h, best_c5, len(best_h)
