        cands = [(ones[bp], zeros[bq])]
        n_rand = 8
        for _ in range(n_rand):
            cands.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        best_val = P
        chosen = None
        for (pp, qq) in cands:
            h[pp], h[qq] = 0.0, 1.0
            cn = fft_corr(h)
            Pn = float(cn.max())
            h[pp], h[qq] = 1.0, 0.0
            if Pn < best_val:
                best_val = Pn
                chosen = (pp, qq)
            elif Pn <= P + 1e-13 and rng.random() < np.exp(-(Pn - P) / T):
                chosen = (pp, qq)
                best_val = Pn
        if chosen is not None:
            h[chosen[0]], h[chosen[1]] = 0.0, 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_P - 1e-14:
                best_P, best_h = P, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T = max(0.001, T * 0.9985)
        if stall > 60:
            nk = min(2 + stall // 20, 7)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(nk):
                pk, qk = ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]
                h[pk], h[qk] = 0.0, 1.0
            c = fft_corr(h)
            P = float(c.max())
            T = max(T, T0 * 0.5)
            stall = 0
    return best_h, best_P * (2.0 / N)

def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h, best_pk = h.copy(), float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr, beta = lr0, 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk, best_h = pk, h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        ct = corr[idx]
        ww = np.exp(T * (ct - ct.max()))
        ww /= ww.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            ss = int(idx[ii] - center)
            wi = ww[ii]
            if ss == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif ss > 0:
                grad[ss:] += wi * (1.0 - h[:N - ss])
                grad[:N - ss] -= wi * h[ss:]
            else:
                M = -ss
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, 0.88 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = ensure_binary(h0, N0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Targeted binary search at N0 with multiple top_k
    for tk, sd in [(8, 10), (15, 20), (25, 30)]:
        if time.time() > deadline - 350:
            break
        d_phase = min(deadline - 300, time.time() + 120)
        h_r, c5_r = targeted_search(best_h, N0, d_phase, top_k=tk, seed=sd)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()

    # Phase 2: Continuous Nesterov refinement
    if time.time() < deadline - 250:
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        d_n = min(deadline - 200, time.time() + 100)
        h_n, c5_n = nesterov_opt(hc, N0, d_n, k=50, T=100.0, lr0=0.012)
        if c5_n < best_c5:
            best_c5, best_h = c5_n, h_n.copy()

    # Phase 3: Second targeted search from best
    if time.time() < deadline - 200:
        h_bin = ensure_binary(best_h, N0)
        d3 = min(deadline - 150, time.time() + 100)
        h_r3, c5_r3 = targeted_search(h_bin, N0, d3, top_k=12, seed=50, T0=0.008)
        if c5_r3 < best_c5:
            best_c5, best_h = c5_r3, h_r3.copy()

    # Phase 4: Try smaller N (50% downsample) for speed
    Ns = N0 // 2 if N0 % 2 == 0 else (N0 // 2 + 1)
    if Ns >= 200 and time.time() < deadline - 150:
        h_small = upsample(best_h, Ns)
        h_small = ensure_binary(h_small, Ns)
        d4 = min(deadline - 100, time.time() + 80)
        h_s4, c5_s4 = targeted_search(h_small, Ns, d4, top_k=10, seed=60, T0=0.01)
        if c5_s4 < best_c5:
            best_c5, best_h = c5_s4, h_s4.copy()

    # Phase 5: Final targeted polish
    N_fin = len(best_h)
    if time.time() < deadline - 60:
        hF = ensure_binary(best_h, N_fin)
        d5 = min(deadline - 30, time.time() + 55)
        hF2, c5F = targeted_search(hF, N_fin, d5, top_k=6, seed=70, T0=0.005)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Phase 6: Final continuous micro-adjust
    if time.time() < deadline - 15:
        hFc = best_h.astype(float) + np.random.randn(N_fin) * 0.015
        hFc = project_sum(hFc, N_fin / 2.0)
        d6 = max(5, deadline - 3)
        h6, c5_6 = nesterov_opt(hFc, N_fin, d6, k=50, T=150.0, lr0=0.006)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
