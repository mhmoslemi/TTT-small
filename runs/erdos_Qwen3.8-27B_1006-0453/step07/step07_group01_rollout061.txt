import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def nesterov_opt(h_init, N, deadline, k=40, T=60.0, lr0=0.02):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = 0.9 * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.996
    return best_h, best_pk * (2.0 / N)

def ensure_binary(h, N, seed=42):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return (1.0 - hm) * vm.astype(float) - hp * vp.astype(float)

def local_search(h, N, deadline, max_sweeps=200, seed=0, full_n_compare=False):
    dx = 2.0 / N
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if (full_n_compare or len(ones) <= 400) else min(len(ones), 250)
        if n_s < len(ones):
            p_idx = rng.choice(len(ones), n_s, replace=False)
        else:
            p_idx = np.arange(len(ones))
        best_new_P = P
        best_pq = None
        for pi in p_idx:
            p = int(ones[pi])
            base = corr - D[p, :]
            cand = base[None, :] + Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * dx

def ilс(h_start, N, deadline, seed=0):
    """Iterated Local Search with random kicks."""
    rng = np.random.RandomState(seed)
    h = h_start.copy()
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    kick_size = 2
    it = 0
    no_improve = 0
    while time.time() < deadline:
        # Local search from current h
        ls_deadline = min(deadline, time.time() + 45)
        h_ls, c_ls = local_search(h, N, ls_deadline, max_sweeps=80, seed=seed + it, full_n_compare=True)
        if c_ls < best_c - 1e-13:
            best_c = c_ls
            best_h = h_ls.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_ls.copy()
        # Kick: random swaps
        n_kick = min(kick_size + no_improve // 5, 8)
        for _ in range(n_kick):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        it += 1
        if no_improve > 30:
            kick_size = max(2, kick_size + 1)
            no_improve = 0
    return best_h, best_c
