def local_search_best(h, N, deadline, rng, n_evals=12):

def block_swaps(h, N, deadline, rng, block_sizes=(2,3,4,5,8)):
    """Try block swap moves."""
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum_binary(h, nh, rng)
    
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    stall = 0
    stall_limit = 2000
    
    while time.time() < deadline:
        best_dE = 0
        best_move = None
        
        for _ in range(8):
            L = int(rng.choice(block_sizes))
            if L >= N - 1:
                continue
            # Find a block of L consecutive 1s and a block of L consecutive 0s
            ones_blocks = []
            zeros_blocks = []
            i = 0
            while i < N:
                if h[i] == 1:
                    j = i
                    while j < N and h[j] == 1:
                        j += 1
                    if j - i >= L:
                        ones_blocks.append(i)
                    i = j
                else:
                    j = i
                    while j < N and h[j] == 0:
                        j += 1
                    if j - i >= L:
                        zeros_blocks.append(i)
                    i = j
            
            if not ones_blocks or not zeros_blocks:
                continue
            ob = int(rng.choice(ones_blocks))
            zb = int(rng.choice(zeros_blocks))
            if abs(ob - zb) < L:
                continue
            
            h[ob:ob+L] = 0
            h[zb:zb+L] = 1
            new_corr = fft_corr(h)
            new_max = float(np.max(new_corr))
            dE = new_max - cur_max
            h[ob:ob+L] = 1
            h[zb:zb+L] = 0
            
            if dE < best_dE:
                best_dE = dE
                best_move = (ob, zb, L)
        
        if best_move is not None:
            ob, zb, L = best_move
            h[ob:ob+L] = 0
            h[zb:zb+L] = 1
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            
            if cur_max * 2.0 / N < best_c - 1e-12:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

            for _ in range(rng.integers(3, 8)):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                h[ones[rng.integers(len(ones))]] = 0
                h[zeros[rng.integers(len(zeros))]] = 1
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            stall = 0
    
    return best_h, best_c

def init_structured(N, variant, rng):
    nh = N // 2
    if variant == 0:
        h = np.zeros(N)
        h[rng.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        # Golden ratio based
        x = np.arange(N)
        vals = (x * 0.618033988749895) % 1.0
        h = (vals > 0.5).astype(float)
        h = fix_sum_binary(h, nh, rng)
    elif variant == 2:
        # Periodic with perturbation
        p = 7
        base = (np.arange(N) % p < p // 2).astype(float)
        h = base.copy()
        # Flip some randomly
        flips = rng.choice(N, N // 10, replace=False)
        h[flips] = 1 - h[flips]
        h = fix_sum_binary(h, nh, rng)
    elif variant == 3:
        # Thue-Morse like
        h = np.zeros(N)
        for i in range(N):
            h[i] = bin(i).count('1') % 2
        h = fix_sum_binary(h, nh, rng)
    else:
        # Multi-frequency
        x = np.arange(N)
        v = np.zeros(N)
        freqs = [3, 7, 13, 21, 37]
        phases = rng.uniform(0, 2*np.pi, len(freqs))
        for f, ph in zip(freqs, phases):
            v += np.sin(2*np.pi*f*x/N + ph)
        h = (v > np.median(v)).astype(float)
        h = fix_sum_binary(h, nh, rng)
    return h

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h, kind='stable')
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target, rng):
    h = h.copy()
    n1 = int(round(h.sum()))
    if n1 > target:
        idx = np.where(h >= 0.5)[0]
        h[rng.choice(idx, min(n1 - target, len(idx)), replace=False)] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        h[rng.choice(idx, min(target - n1, len(idx)), replace=False)] = 1

def compute_grad(corr, h, N, T):

    cmax = np.max(corr)
    w = np.exp((corr - cmax) / T)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(N)

    grad = np.zeros(N)

                best_op, best_zp = op, zp
        if best_op >= 0:
            h[best_op] = 0
            h[best_zp] = 1
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max * 2.0 / N < best_c - 1e-13:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > stall_limit:
            npert = int(rng.integers(5, 15))
            for _ in range(npert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                h[int(ones[rng.integers(len(ones))])] = 0
                h[int(zeros[rng.integers(len(zeros))])] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            stall = 0
    return best_h, best_c

def block_search(h, N, deadline, rng, sizes=(2, 3, 4, 5, 8, 12)):
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum_binary(h, nh, rng)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        best_dE = 0.0
        best_move = None
        for _ in range(6):
            L = int(rng.choice(sizes))
            if L >= N // 2:
                continue
            ob_list, zb_list = [], []
            i = 0
            while i < N:
                if h[i] == 1:
                    j = i
                    while j < N and h[j] == 1:
                        j += 1
                    if j - i >= L:
                        ob_list.append(i)
                    i = j
                else:
                    j = i
                    while j < N and h[j] == 0:
                        j += 1
                    if j - i >= L:
                        zb_list.append(i)
                    i = j
            if not ob_list or not zb_list:
                continue
            ob = int(rng.choice(ob_list))
            zb = int(rng.choice(zb_list))
            if abs(ob - zb) < L:
                continue
            h[ob:ob+L] = 0
            h[zb:zb+L] = 1
            nc = fft_corr(h)
            nm = float(np.max(nc))
            dE = nm - cur_max
            h[ob:ob+L] = 1
            h[zb:zb+L] = 0
            if dE < best_dE:
                best_dE = dE
                best_move = (ob, zb, L)
        if best_move is not None:
            ob, zb, L = best_move
            h[ob:ob+L] = 0
            h[zb:zb+L] = 1
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            if cur_max * 2.0 / N < best_c - 1e-13:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 1500:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(int(rng.integers(3, 8))):
                h[int(ones[rng.integers(len(ones))])] = 0
                h[int(zeros[rng.integers(len(zeros))])] = 1
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            stall = 0
    return best_h, best_c

def init_structured(N, variant, rng):
    nh = N // 2
    x = np.arange(N)
    if variant == 0:
        h = np.zeros(N)
        h[rng.choice(N, nh, replace=False)] = 1
    elif variant == 1:
        vals = (x * 0.6180339887) % 1.0
        h = (vals > 0.5).astype(float)
    elif variant == 2:
        p = 9
        h = ((x % p) < 4).astype(float)
        flips = rng.choice(N, N // 8, replace=False)
        h[flips] = 1 - h[flips]
    elif variant == 3:
        h = np.zeros(N)
        for i in range(N):
            h[i] = bin(i).count('1') % 2
    else:
        v = np.zeros(N)
        freqs = [3, 7, 13, 21, 37]
        phases = rng.uniform(0, 2 * np.pi, len(freqs))
        for f, ph in zip(freqs, phases):
            v += np.sin(2 * np.pi * f * x / N + ph)
        h = (v > np.median(v)).astype(float)
    return fix_sum_binary(h, nh, rng)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Intensive local search on binary (180s)
    h_b = (h0 > 0.5).astype(float)
    d1 = min(deadline - 620, t0 + 180)
    hs, c1 = local_search(h_b, N0, d1, rng, n_evals=12)
    if c1 < best_c5:
        best_c5, best_h = c1, hs.copy()

    # Phase 2: Block search (120s)
    d2 = min(deadline - 500, t0 + 300)
    hs, c2 = block_search(best_h, N0, d2, rng, sizes=(2, 3, 4, 5, 8, 13))
    if c2 < best_c5:
        best_c5, best_h = c2, hs.copy()

    # Phase 3: Continuous softmax optimization (350s)
    if time.time() < deadline - 350:
        hc = best_h.copy()
        hc += rng.standard_normal(N0) * 0.04
        hc = np.clip(hc, 0, 1)
        target = N0 / 2.0
        lo, hi = -5.0, 5.0
        for _ in range(50):
            mid = (lo + hi) / 2
            if np.clip(hc - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        hc = np.clip(hc - (lo + hi) / 2, 0, 1)
        d3 = min(deadline - 300, time.time() + 350)
        hc, c3 = continuous_optimize(hc, N0, d3, rng, T0=0.05, T_end=0.002, lr0=0.007)
        if c3 < best_c5:
            best_c5, best_h = c3, hc.copy()

    # Phase 4: Binary refine from continuous result (120s)
    if time.time() < deadline - 140:
        h4 = project_binary(best_h, N0 // 2)
        d4 = min(deadline - 110, time.time() + 120)
        hs, c4 = local_search(h4, N0, d4, rng, n_evals=14)
        if c4 < best_c5:
            best_c5, best_h = c4, hs.copy()
        d4b = min(deadline - 90, time.time() + 70)
        hs, c4b = block_search(best_h, N0, d4b, rng, sizes=(2, 3, 5, 8, 16))
        if c4b < best_c5:
            best_c5, best_h = c4b, hs.copy()

    # Phase 5: Multi-N exploration
    remaining = deadline - time.time() - 40
    if remaining > 20:
        N_opts = [200, 240, 280, 320, 400, 440]
        N_opts = [n for n in N_opts if n != N0 and n % 2 == 0]
        per_N = remaining / max(1, len(N_opts))
        for Nt in N_opts:
            if time.time() > deadline - 30:
                break
            if per_N < 5:
                break
            sub_dl = time.time() + per_N
            best_loc_c, best_loc_h = np.inf, None
            for vi in range(3):
                if time.time() > sub_dl - 1:
                    break
                hv = init_structured(Nt, vi, rng)
                hr, cr = local_search(hv, Nt, min(sub_dl, time.time() + per_N / 3 - 1), rng, n_evals=8)
                if cr < best_loc_c:
                    best_loc_c, best_loc_h = cr, hr.copy()
            if best_loc_h is not None and best_loc_c < best_c5:
                best_c5, best_h = best_loc_c, best_loc_h

    # Phase 6: Final grind
    if time.time() < deadline - 3:
        hF = project_binary(best_h, len(best_h) // 2)
        NF = len(hF)
        hs, c6 = local_search(hF, NF, deadline - 1, rng, n_evals=12, stall_limit=3000)
        if c6 < best_c5:
            best_c5, best_h = c6, hs

    hB = project_binary(best_h, len(best_h) // 2)
    cB = compute_c5(hB, len(hB))
    if cB < best_c5:
        best_c5, best_h = cB, hB
    return best_h, best_c5, len(best_h)
