    # term1: conv of (1-h) with w[N-1:] (length N)
    w1 = w[N-1:]  # w[N-1], w[N], ..., w[2N-2] (length N)
    term1 = np.convolve(1.0 - h, w1, mode='full')[:N]
    # term2: conv of h with w[:N] (length N)  [w[0], w[1], ..., w[N-1]]
    w2 = w[:N]
    term2 = np.convolve(h, w2, mode='full')[:N]
    grad = term1 - term2
    return grad, c

def continuous_opt(h0, N, deadline, lam_start=3.0, lam_end=50.0, steps=3000, lr0=0.008):
    h = h0.astype(float).copy()
    h = proj_sum(h, N / 2.0)
    m = np.zeros(N)
    best_h = h.copy()
    best_P = float(np.max(fft_corr(h)))
    beta = 0.85
    lr = lr0
    for it in range(steps):
        if time.time() > deadline:
            break
        # Adaptive lambda: start low, increase
        prog = it / max(1, steps - 1)
        lam = lam_start * (lam_end / lam_start) ** prog
        y = h + m
        grad, c = smooth_max_grad(y, N, lam)
        P = float(c.max())
        if P < best_P - 1e-13:
            best_P = P
            best_h = h.copy()
        h_new = proj_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        if it % 200 == 0:
            lr *= 0.97
            beta = min(0.95, 0.8 + prog * 0.2)
    return best_h, best_P * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of(best_h)

    # Phase 1: Binary hill climb at N0 (30% budget)
    d1 = t0 + budget_s * 0.28
    h1, c5_1 = binary_optimize(h0, N0, d1, rng, max_sweeps=3000, p_sub=120, z_sub=120, two_n=200, block_n=50)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Continuous smooth-max at N0 (15% budget)
    if time.time() < deadline - 500:
        d2 = min(deadline - 450, time.time() + budget_s * 0.13)
        hc = best_h + rng.randn(N0) * 0.06
        hc = proj_sum(hc, N0 / 2.0)
        h2, c5_2 = continuous_opt(hc, N0, d2, lam_start=2.0, lam_end=40.0, steps=2000, lr0=0.006)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    # Phase 3: Binarize + refine at N0 (8% budget)
    if time.time() < deadline - 400:
        h3b = to_bin(best_h, N0)
        d3 = min(deadline - 350, time.time() + budget_s * 0.07)
        h3, c5_3 = binary_optimize(h3b, N0, d3, rng, max_sweeps=1500, p_sub=100, z_sub=100, two_n=150, block_n=40)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 4: Upscale to 2*N0, binary optimize (25% budget)
    N1 = 2 * N0
    if time.time() < deadline - 200:
        h_up = to_bin(upsample(best_h, len(best_h), N1), N1)
        d4 = min(deadline - 150, time.time() + budget_s * 0.22)
        h4, c5_4 = binary_optimize(h_up, N1, d4, rng, max_sweeps=1200, p_sub=80, z_sub=80, two_n=120, block_n=30)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Phase 5: Continuous at N1 (10% budget)
    N_cur = len(best_h)
    if time.time() < deadline - 100 and N_cur == N1:
        d5 = min(deadline - 60, time.time() + budget_s * 0.09)
        hc5 = best_h + rng.randn(N_cur) * 0.04
        hc5 = proj_sum(hc5, N_cur / 2.0)
        h5, c5_5 = continuous_opt(hc5, N_cur, d5, lam_start=2.0, lam_end=35.0, steps=1500, lr0=0.005)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5

    # Phase 6: Final binary polish at current N (remaining time)
    N_fin = len(best_h)
    hF = to_bin(best_h, N_fin)
    sf = 80 if N_fin > 500 else 120
    hF2, c5F = binary_optimize(hF, N_fin, deadline - 1, rng, max_sweeps=1000, p_sub=sf, z_sub=sf, two_n=100, block_n=30)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2

    # Final evaluation with np.correlate for exactness
    final_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * (2.0 / len(best_h)))
    return best_h, final_c5, len(best_h)
