                for j in range(nc):
                    L = top_lags[j]
                    d = 0.0
                    for oi in (oi1, oi2):
                        if 0 <= oi + L < N:
                            d -= h[oi + L]
                        if 0 <= oi - L < N:
                            d -= h[oi - L]
                    rem2[j] = d
                for j1 in range(nz1):
                    for j2 in range(j1 + 1, nz1):
                        zi1, zi2 = int(z1s[j1]), int(z1s[j2])
                        add2 = np.zeros(nc)
                        for j in range(nc):
                            L = top_lags[j]
                            d = 0.0
                            for zi in (zi1, zi2):
                                if 0 <= zi + L < N:
                                    d += h[zi + L]
                                if 0 <= zi - L < N:
                                    d += h[zi - L]
                            add2[j] = d
                        nm = float(np.max(corrs + rem2 + add2))
                        if nm < best_new_max - 1e-8:
                            best_new_max = nm
                            best_move = (oi1, oi2, zi1, zi2)
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 0
            h[best_move[2]] = 1
            h[best_move[3]] = 1
            corr = compute_corr(h)
            c = float(np.max(corr)) * (2.0 / N)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: continuous optimization
    d1 = min(deadline - 700, t0 + 80)
    h1 = project(best_h + np.random.randn(N0) * 0.06, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=30, lr0=0.018, mom=0.9)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    # Phase 2: exhaustive binary local search (main work)
    d2 = min(deadline - 350, time.time() + 400)
    hb = fix_binary(best_h)
    hb, c2 = exhaustive_local(hb, d2, n_crit=10, max_rounds=800, perturb_every=35)
    if c2 < best_c:
        best_c, best_h = c2, hb.copy()
    # Phase 3: 2-swap local search
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 120)
        hb3 = fix_binary(best_h)
        hb3, c3 = two_swap_local(hb3, d3, n_crit=8, max_iters=150)
        if c3 < best_c:
            best_c, best_h = c3, hb3.copy()
    # Phase 4: another round of exhaustive search
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 100)
        hb4 = fix_binary(best_h)
        hb4, c4 = exhaustive_local(hb4, d4, n_crit=12, max_rounds=400, perturb_every=30)
        if c4 < best_c:
            best_c, best_h = c4, hb4.copy()
    # Phase 5: try different resolution
    for N_try in [460, 500]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d5 = min(deadline - 40, time.time() + 45)
        h_bt, c5v = exhaustive_local(h_bt, d5, n_crit=10, max_rounds=200, perturb_every=30)
        if c5v < best_c:
            best_c, best_h = c5v, h_bt.copy()
    # Final: quick polish
    if time.time() < deadline - 5:
        d7 = min(deadline - 1, time.time() + 30)
        hB = fix_binary(best_h)
        hB, c7 = exhaustive_local(hB, d7, n_crit=10, max_rounds=100, perturb_every=25)
        if c7 < best_c:
            best_c, best_h = c7, hB.copy()
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
