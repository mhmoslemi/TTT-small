def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_corr(h, corr, k):
    N = len(h)

    grad = np.zeros(N)

        if M == 0:
            grad += (1.0 - 2.0 * h)

            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_corr(h, corr, k):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def cont_opt(h, deadline, k=40, lr0=0.015, mom=0.9):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 40:
            lr0 *= 0.82
            stale = 0
        g = grad_corr(h, corr, k)
        h_new = project(h + lr0 * (mom * v - g), tgt)
        v = mom * v + 0.5 * (h_new - h)
        h = h_new
    return best_h, best_c

def binary_sa(h, deadline, max_iters=50000, n_crit=40, s_sz=50, T0=0.03, T1=0.001):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    c = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    best_c = c
    it = 0
    de = 0
    center = N - 1
    top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
    while time.time() < deadline and it < max_iters:
        T = max(T1, T0 * (T1 / T0) ** (min(it, max_iters) / max(float(max_iters), 1.0)))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if it % 20 == 0:
            top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        corr_top = corr[center + top_lags]
        cur_peak_count = float(np.max(corr))
        D_oi = np.empty((ns, n_crit))
        for j in range(ns):
            oi = os_[j]
            i1 = oi + top_lags
            i2 = oi - top_lags
            v1 = (i1 >= 0) & (i1 < N)
            v2 = (i2 >= 0) & (i2 < N)
            D_oi[j] = np.where(v1, h[np.clip(i1, 0, N-1)], 0.0) + np.where(v2, h[np.clip(i2, 0, N-1)], 0.0)
        D_zi = np.empty((nz, n_crit))
        for j in range(nz):
            zi = zs[j]
            j1 = zi + top_lags
            j2 = zi - top_lags
            v1 = (j1 >= 0) & (j1 < N)
            v2 = (j2 >= 0) & (j2 < N)
            D_zi[j] = np.where(v1, h[np.clip(j1, 0, N-1)], 0.0) + np.where(v2, h[np.clip(j2, 0, N-1)], 0.0)
        new_vals = corr_top[None, None, :] + D_oi[:, None, :] - D_zi[None, :, :]
        new_peaks = np.max(new_vals, axis=2)
        improved = new_peaks < cur_peak_count - 1e-10
        if np.any(improved):
            masked = np.where(improved, new_peaks, np.inf)
            bi = np.unravel_index(np.argmin(masked), masked.shape)
            h[os_[bi[0]]] = 0
            h[zs[bi[1]]] = 1
            corr = compute_corr(h)
            c = float(np.max(corr)) * (2.0 / N)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                de = 0
            else:
                de += 1
        else:
            if T > 3 * T1:
                ri = np.random.randint(0, ns)
                rj = np.random.randint(0, nz)
                npk = float(new_peaks[ri, rj]) * (2.0 / N)
                if np.random.random() < np.exp(-(npk - c) / max(T, 1e-8)):
                    h[os_[ri]] = 0
                    h[zs[rj]] = 1
                    corr = compute_corr(h)
                    c = float(np.max(corr)) * (2.0 / N)
            de += 1
        if de > 50:
            nsw = np.random.randint(3, 10)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            np.random.shuffle(ones)
            np.random.shuffle(zeros)
            for jj in range(min(nsw, len(ones), len(zeros))):
                h[ones[jj]] = 0
                h[zeros[jj]] = 1
            corr = compute_corr(h)
            c = float(np.max(corr)) * (2.0 / N)
            de = 0
        elif de > 300:
            nb = max(N // 10, 10)
            perm = np.random.permutation(N)
            h[perm[:nb]] = 1.0 - h[perm[:nb]]
            h = fix_binary(h)
            corr = compute_corr(h)
            c = float(np.max(corr)) * (2.0 / N)
            de = 0
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: quick continuous opt
    d1 = min(deadline - 550, t0 + 45)
    h1 = project(h0 + np.random.randn(N0) * 0.04, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=35, lr0=0.012, mom=0.9)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    # Phase 2: binary SA at N0
    if time.time() < deadline - 300:
        d2 = min(deadline - 280, time.time() + 220)
        hb = fix_binary(best_h)
        hb, c2 = binary_sa(hb, d2, max_iters=50000, n_crit=40, s_sz=50, T0=0.03, T1=0.001)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()
    # Continuous polish
    if time.time() < deadline - 180:
        dcp = min(deadline - 170, time.time() + 35)
        hcp = project(best_h + np.random.randn(len(best_h)) * 0.01, len(best_h) / 2.0)
        hcp, ccp = cont_opt(hcp, dcp, k=30, lr0=0.005, mom=0.93)
        if ccp < best_c:
            best_c, best_h = ccp, hcp.copy()
    # Phase 3: other resolutions
    for N_try in [400, 382, 420]:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        hb = fix_binary(h_int)
        d3 = min(deadline - 70, time.time() + 65)
        hb, c3 = binary_sa(hb, d3, max_iters=20000, n_crit=30, s_sz=40, T0=0.02, T1=0.001)
        if time.time() < deadline - 35:
            hc = project(hb + np.random.randn(N_try) * 0.02, N_try / 2.0)
            d4 = min(deadline - 25, time.time() + 15)
            hc, c4 = cont_opt(hc, d4, k=25, lr0=0.004, mom=0.93)
            if c4 < c3:
                c3, hb = c4, hc
        if c3 < best_c:
            best_c, best_h = c3, hb.copy()
    # Phase 4: random restart
    if time.time() < deadline - 60:
        Nr = len(best_h)
        hr = np.zeros(Nr)
        idx = np.random.permutation(Nr)
        hr[idx[:Nr // 2]] = 1.0
        hr = fix_binary(hr)
        d5 = min(deadline - 45, time.time() + 50)
        hr, c5v = binary_sa(hr, d5, max_iters=15000, n_crit=30, s_sz=40, T0=0.04, T1=0.001)
        if c5v < best_c:
            best_c, best_h = c5v, hr.copy()
    # Phase 5: final binary polish
    if time.time() < deadline - 15:
        hB = fix_binary(best_h)
        d6 = min(deadline - 5, time.time() + 25)
        hB, c6 = binary_sa(hB, d6, max_iters=10000, n_crit=40, s_sz=50, T0=0.01, T1=0.001)
        if c6 < best_c:
            best_c, best_h = c6, hB.copy()
    # Final micro continuous opt
    if time.time() < deadline - 2:
        hf = project(best_h + np.random.randn(len(best_h)) * 0.004, len(best_h) / 2.0)
        d7 = min(deadline - 1, time.time() + 8)
        hf, c7 = cont_opt(hf, d7, k=20, lr0=0.002, mom=0.94)
        if c7 < best_c:
            best_c, best_h = c7, hf.copy()
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
