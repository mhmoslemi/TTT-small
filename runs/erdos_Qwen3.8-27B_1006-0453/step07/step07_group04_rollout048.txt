def nesterov_run(h_init, N, deadline, k_top=50, lr0=0.015):

    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)

                grad[L:] += (1.0 - y[:N - L])

        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
    return best_h, best_pk * dx

    best_c5 = compute_c5(best_h)
    per_run = (deadline - time.time()) / n_rest
    for r in range(n_rest):
        hd = best_h.copy()
        ones = np.where(hd == 1)[0]
        zeros = np.where(hd == 0)[0]
        for _ in range(kick_each + r):
            hd[ones[np.random.randint(len(ones))]] = 0.0
            hd[zeros[np.random.randint(len(zeros))]] = 1.0
        d_end = min(deadline - 5, time.time() + per_run)
        hr, cr = sa_opt(hd, N, d_end, T0=3.5 + r, Tcool=0.993, T_min=0.02,
                        kick_every=80 + r * 20, kick_size=kick_each + 1)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    hb = ensure_sum_bin(h0, N)

    # Phase 1: Quick binary greedy
    d1 = min(deadline - 750, t0 + 20)
    h1, c1 = greedy_opt(hb, N, d1, sf=1.0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary SA (aggressive)
    d2 = min(deadline - 550, t0 + 120)
    h2, c2 = sa_opt(hb, N, d2, T0=4.5, Tcool=0.992, T_min=0.02,
                    kick_every=70, kick_size=5, sf=1.0)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 3: Low-res explore + upsample
    N_low = max(N // 4, 40)
    x_l = np.linspace(0, 2, N_low, endpoint=False)
    h_l = np.interp(x_l, np.linspace(0, 2, N, endpoint=False), best_h)
    h_l = project_sum(h_l, N_low / 2.0)
    h_lb = ensure_sum_bin(h_l, N_low)
    d3a = min(deadline - 420, t0 + 200)
    hl, cl = sa_opt(h_lb, N_low, d3a, T0=3.0, Tcool=0.994, T_min=0.04,
                    kick_every=50, kick_size=3, sf=1.0)
    h_up = upsample_fn(hl, N_low, N)
    h_up = ensure_sum_bin(h_up, N)
    d3b = min(deadline - 380, t0 + 230)
    h3, c3 = greedy_opt(h_up, N, d3b, sf=1.0)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: Multi-restart SA
    d4 = min(deadline - 280, t0 + 350)
    h4, c4 = multi_restart_sa(ensure_sum_bin(best_h, N), N, d4, n_rest=3, kick_each=4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4.copy()

    # Phase 5: Nesterov continuous (main improvement phase)
    hc = project_sum(best_h + np.random.randn(N) * 0.025, N / 2.0)
    d5 = min(deadline - 130, t0 + 500)
    h5, c5 = nesterov_cont(hc, N, d5, k_top=55, lr0=0.01)
    if c5 < best_c5:
        best_c5, best_h = c5, h5.copy()

    # Phase 6: Binary polish after Nesterov
    h6b = ensure_sum_bin(best_h, N)
    d6 = min(deadline - 80, t0 + 570)
    h6, c6 = greedy_opt(h6b, N, d6, sf=1.0)
    if c6 < best_c5:
        best_c5, best_h = c6, h6.copy()

    # Phase 7: Fine Nesterov from best binary
    hc2 = project_sum(best_h + np.random.randn(N) * 0.015, N / 2.0)
    d7 = min(deadline - 40, t0 + 640)
    h7, c7 = nesterov_cont(hc2, N, d7, k_top=40, lr0=0.006)
    if c7 < best_c5:
        best_c5, best_h = c7, h7.copy()

    # Phase 8: Final SA (low temp, long)
    h8 = ensure_sum_bin(best_h, N)
    ones8 = np.where(h8 == 1)[0]
    zeros8 = np.where(h8 == 0)[0]
    for _ in range(2):
        h8[ones8[np.random.randint(len(ones8))]] = 0.0
        h8[zeros8[np.random.randint(len(zeros8))]] = 1.0
    d8 = min(deadline - 15, t0 + 720)
    h8, c8 = sa_opt(h8, N, d8, T0=2.0, Tcool=0.996, T_min=0.015,
                    kick_every=200, kick_size=3, sf=0.8)
    if c8 < best_c5:
        best_c5, best_h = c8, h8.copy()

    # Phase 9: Final greedy polish
    h9 = ensure_sum_bin(best_h, N)
    d9 = deadline - 0.5
    h9, c9 = greedy_opt(h9, N, d9, sf=1.0)
    if c9 < best_c5:
        best_c5, best_h = c9, h9.copy()

    # Final check
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N
