def adam_polish(h_in, N, deadline, lr=0.02, k_top=40, beta1=0.85, beta2=0.999):

    hx = h.copy()

    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        y = h.copy()
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top)
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        mhat = m / (1 - beta1**it)
        vhat = v / (1 - beta2**it)
        h_new = np.clip(y - lr * mhat / (np.sqrt(vhat) + 1e-8), 0, 1)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0, 1)
        h = h_new
        if it % 500 == 0:
            lr *= 0.97
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def sa_core(h_bin, deadline, k_top=6, t_start=0.05, cool_rate=0.9993, max_iters=300000, do_2swap=True):

        if len(ones) < 2 or len(zeros) < 2:

        cand_p = np.array(sorted(set(cand_p)), dtype=int)

                    best_h = h.copy()
                    no_impr = 0
                else:

def adam_smooth(h_in, N, deadline, lr=0.018, k_top=50):
    h = h_in.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2 = 0.82, 0.999
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * h
            elif L > 0:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
        grad /= len(top)
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        mhat = m / (1 - b1 ** it)
        vhat = v / (1 - b2 ** it)
        h_new = np.clip(h - lr * mhat / (np.sqrt(vhat) + 1e-8), 0, 1)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0, 1)
        h = h_new
        if it % 600 == 0:
            lr *= 0.97
    return best_h, best_pk * (2.0 / N)
