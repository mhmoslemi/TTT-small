def fft_corr(h):
    N = len(h)
    L = 2 * N - 1

    return np.real(np.fft.ifft(H * G))

                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_onto_box_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.992
            beta = min(0.97, 0.86 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_c5 = compute_c5(h0, N0)
    best_h = h0.copy()

    # Phase 1: Binary search, large samples
    d1 = min(deadline - 450, t0 + 300)
    h_bin = (h0 > 0.5).astype(float)
    h_s, c5_s = binary_search(h_bin, N0, d1, sample_sz=300, kick_thresh=60, kick_sz=5, use_2swap=True)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov continuous
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 130)
        hc = best_h + np.random.randn(N0) * 0.06
        hc = project_onto_box_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=50, lr0=0.008)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search with 2-2 swaps from perturbed start
    if time.time() < deadline - 200:
        h3 = (best_h > 0.5).astype(float)
        ones3 = np.where(h3 == 1)[0]
        zeros3 = np.where(h3 == 0)[0]
        for _ in range(6):
            h3[np.random.choice(ones3)] = 0
            h3[np.random.choice(zeros3)] = 1
            ones3 = np.where(h3 == 1)[0]
            zeros3 = np.where(h3 == 0)[0]
        d3 = min(deadline - 150, time.time() + 150)
        h_s3, c5_3 = binary_search(h3, N0, d3, sample_sz=300, kick_thresh=40, kick_sz=6, use_2swap=True)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Try N=400
    if time.time() < deadline - 120:
        N4 = 400
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N4, endpoint=False)
        h4 = (np.interp(x_new, x_old, best_h) > 0.5).astype(float)
        h4 = fix_sum_binary(h4, N4 // 2)
        d4 = min(deadline - 80, time.time() + 90)
        h_s4, c5_4 = binary_search(h4, N4, d4, sample_sz=250, kick_thresh=40, kick_sz=5, use_2swap=True)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_s4.copy()

    # Phase 5: Try N=360
    if time.time() < deadline - 60:
        N5 = 360
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N5, endpoint=False)
        h5 = (np.interp(x_new, x_old, best_h) > 0.5).astype(float)
        h5 = fix_sum_binary(h5, N5 // 2)
        d5 = min(deadline - 30, time.time() + 50)
        h_s5, c5_5 = binary_search(h5, N5, d5, sample_sz=200, kick_thresh=40, kick_sz=4, use_2swap=True)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h_s5.copy()

    # Phase 6: Final binary polish
    if time.time() < deadline - 5:
        hF = (best_h > 0.5).astype(float)
        hF = fix_sum_binary(hF, len(hF) // 2)
        hF, c5F = binary_search(hF, len(hF), deadline - 1, sample_sz=300, kick_thresh=30, kick_sz=3, use_2swap=True)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final: compare binary vs continuous
    final_c5 = compute_c5(best_h, len(best_h))
    hb = (best_h > 0.5).astype(float)
    hb = fix_sum_binary(hb, len(hb) // 2)
    c5b = compute_c5(hb, len(hb))
    if c5b < final_c5:
        best_h, final_c5 = hb, c5b
    return best_h, final_c5, len(best_h)
