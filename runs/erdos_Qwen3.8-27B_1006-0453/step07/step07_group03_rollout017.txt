import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    # D[i, k] = change in corr[k] when h[i] flips from 1 to 0
    # Or we can compute it as the contribution of position i to each shift
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

    h = h.astype(float).copy()

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
