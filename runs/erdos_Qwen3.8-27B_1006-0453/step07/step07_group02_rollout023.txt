def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    """C[i, j] = change in corr[j] if h[i] is flipped"""
    k_vals = np.arange(2*N-1) - (N-1)  # lag values
    pos = np.arange(N)[:, None]

    i_pk = pos + k_vals[None, :]

    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    k_vals = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    i_mk = pos - k_vals[None, :]
    i_pk = pos + k_vals[None, :]
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def fix_sum(h, target):
    curr = int(np.round(h.sum()))
    if curr > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:curr-target]] = 0
    elif curr < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target-curr]] = 1
    return h

            break
