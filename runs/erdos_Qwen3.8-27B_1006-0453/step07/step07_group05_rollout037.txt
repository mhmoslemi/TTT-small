import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1 - h, mode="full")

def _c5(h, N):
    return float(np.max(_corr(h))) * (2.0 / N)

def _proj(h, target):
    idx = np.argsort(-h, kind="stable")
    out = np.zeros(len(h))
    out[idx[:target]] = 1.0
    return out

def _binary_ls(h, N, t_end, iters=500000, bs=(1, 2, 3, 4, 5, 7, 10)):
    h = _proj(h, N // 2).astype(float).copy()
    cur = float(np.max(_corr(h)))
    best_h, best_m = h.copy(), cur
    T = 0.4
    stall = 0
    stall_lim = 4000
    for it in range(iters):
        if time.time() > t_end:
            break
        r = np.random.random()
        if r < 0.45:
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            a = oi[np.random.randint(len(oi))]
            b = zi[np.random.randint(len(zi))]
            h[a], h[b] = 0, 1
            nm = float(np.max(_corr(h)))
            if nm < cur - 1e-12 or (T > 0.005 and np.random.random() < np.exp(-(nm - cur) / T)):
                cur = nm
                if nm < best_m - 1e-12:
                    best_m, best_h, stall = nm, h.copy(), 0
                else:
                    stall += 1
            else:
                h[a], h[b] = 1, 0
        elif r < 0.80:
            L = int(np.random.choice(bs))
            if L > N // 5:
                L = 2

            if found_a < 0:

                s = np.random.randint(0, N - L + 1)
                if np.all(h[s:s + L] == 0) and (s >= found_a + L or s + L <= found_a):
                    found_b = s
                    break
            if found_b < 0:
                continue
            h[found_a:found_a + L] = 0
            h[found_b:found_b + L] = 1
            nm = float(np.max(_corr(h)))
            if nm < cur - 1e-12 or (T > 0.005 and np.random.random() < np.exp(-(nm - cur) / T)):
                cur = nm
                if nm < best_m - 1e-12:

                else:
                    stall += 1

                h[found_b:found_b + L] = 0

            zi = np.where(h == 0)[0]
            if len(oi) < 2 or len(zi) < 2:

            b1, b2 = np.random.choice(zi, 2, replace=False)
            h[a1] = h[a2] = 0
            h[b1] = h[b2] = 1
            nm = float(np.max(_corr(h)))
            if nm < cur - 1e-12 or (T > 0.005 and np.random.random() < np.exp(-(nm - cur) / T)):
                cur = nm

def _cont_refine(h, N, t_end):
    h = h.astype(float).copy()
    tgt = N / 2.0
    best = h.copy()
    best_c = _c5(h, N)
    lr = 0.004
    temp = 0.015
    for it in range(80000):
        if time.time() > t_end:
            break
        corr = _corr(h)
        peak = float(np.max(corr))
        w = np.exp((corr - peak) / temp)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        sig = np.where(w > 2e-4)[0]
        for ki in sig:
            L = int(ki - center)
            wt = w[ki]
            if L > 0:
                grad[L:] += wt * (1.0 - h[:N - L])
                grad[:N - L] -= wt * h[L:]
            elif L < 0:
                M = -L
                grad[:N - M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N - M]
            else:
                grad += wt * (1.0 - 2.0 * h)
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        lo, hi = -5.0, 5.0
        for _ in range(42):
            mid = (lo + hi) / 2.0
            if np.clip(h_new - mid, 0, 1).sum() > tgt:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2.0, 0.0, 1.0)
        nc = _c5(h_new, N)
        if nc < best_c - 1e-11:
            best_c, best = nc, h_new.copy()
        h = h_new
        if it % 400 == 0:
            lr *= 0.992
            temp = max(temp * 0.975, 0.002)
    return best, best_c
