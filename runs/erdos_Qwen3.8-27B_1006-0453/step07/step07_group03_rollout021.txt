def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

        center = N - 1
        for ai in top:
            Ls = int(ai - center)
            if Ls == 0:
                grad += (1.0 - 2.0 * y)
            elif Ls > 0:
                grad[Ls:] += (1.0 - y[:N - Ls])

                    grad[M:] -= y[:N - M]

        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:

            if len(active) < 5:
                active = np.argsort(corr)[-15:]
            ns = min(n_2swap, max(100, N // 2))
            for _ in range(ns):
                i1, i2 = rng.choice(n1, 2, replace=False)
                j1, j2 = rng.choice(n0, 2, replace=False)
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                nc = corr + D[p1] + D[p2] - D[q1] - D[q2]
                nv = float(nc[active].max())
                if nv < best_val:
                    rv = float(nc.max())
                    if rv < best_val:
                        best_val = rv
                        best_2s = (p1, p2, q1, q2)
        if best_2s is not None:
            h[best_2s[0]] = h[best_2s[1]] = 0.0
            h[best_2s[2]] = h[best_2s[3]] = 1.0
        elif best_1s is not None:
            h[best_1s[0]] = 0.0
            h[best_1s[1]] = 1.0
        else:
            nf = 2 + (1 if stall > 10 else 0)
            for _ in range(min(nf, n1)):
                p = int(ones[rng.randint(n1)])
                q = int(zeros[rng.randint(n0)])
                h[p], h[q] = 0.0, 1.0
            corr = fft_corr(h)
            P = float(corr.max())
            if P < best_c:
                best_c = P
                best_h = h.copy()
            stall += 1
            if stall > 80:
                break
            continue
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=80, lr0=0.014, beta0=0.88):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        top = np.argpartition(c, -k)[-k:]
        pk = float(c[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            Ls = int(ai - center)
            if Ls == 0:
                grad += (1.0 - 2.0 * y)
            elif Ls > 0:
                grad[Ls:] += (1.0 - y[:N - Ls])
                grad[:N - Ls] -= y[Ls:]
            else:
                M = -Ls
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.985
            beta = min(0.97, beta0 + it * 0.00008)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Steepest binary at N0 (long)
    d1 = t0 + budget_s * 0.38
    if time.time() < d1:
        h1, c51 = steepest_binary(best_h, N0, d1, max_iters=60000, n_2swap=350, rng=rng)
        if c51 < best_c5:
            best_c5, best_h = c51, h1.copy()

    # Phase 2: Nesterov at N0
    d2 = t0 + budget_s * 0.52
    if time.time() < d2 - 20:
        hc = best_h + rng.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        h2, c52 = nesterov_opt(hc, N0, d2, k=70, lr0=0.013)
        if c52 < best_c5:
            best_c5, best_h = c52, h2.copy()

    # Phase 3: Binary at N1=2*N0
    N1 = 2 * N0
    d3 = t0 + budget_s * 0.74
    if time.time() < d3 - 50:
        h_up = ensure_sum(upsample(ensure_sum(best_h, len(best_h)), len(best_h), N1), N1)
        h3, c53 = steepest_binary(h_up, N1, d3, max_iters=3000, n_2swap=180, rng=rng)
        if c53 < best_c5:
            best_c5, best_h = c53, h3.copy()

    # Phase 4: Nesterov at current best resolution
    d4 = t0 + budget_s * 0.86
    if time.time() < d4 - 20:
        Nc = len(best_h)
        hc4 = best_h + rng.randn(Nc) * 0.03
        hc4 = project_sum(hc4, Nc / 2.0)
        k4 = 90 if Nc <= 400 else 120
        lr4 = 0.010 if Nc <= 400 else 0.006
        h4, c54 = nesterov_opt(hc4, Nc, d4, k=k4, lr0=lr4)
        if c54 < best_c5:
            best_c5, best_h = c54, h4.copy()

    # Phase 5: Another binary pass at N0 (fresh restart from best)
    d5 = t0 + budget_s * 0.93
    if time.time() < d5 - 10:
        Nb = N0
        if len(best_h) != Nb:
            best_re = ensure_sum(upsample(ensure_sum(best_h, len(best_h)), len(best_h), Nb), Nb)
        else:
            best_re = ensure_sum(best_h, Nb)
        rng5 = np.random.RandomState(seed + 77)
        nf = 4
        ones_t = np.where(best_re == 1)[0]
        zeros_t = np.where(best_re == 0)[0]
        for _ in range(nf):
            best_re[ones_t[rng5.randint(len(ones_t))]] = 0.0
            best_re[zeros_t[rng5.randint(len(zeros_t))]] = 1.0
        h5, c55 = steepest_binary(best_re, Nb, d5, max_iters=20000, n_2swap=250, rng=rng5)
        if c55 < best_c5:
            best_c5, best_h = c55, h5.copy()

    # Phase 6: Final polish
    Nf = len(best_h)
    hF = ensure_sum(best_h, Nf)
    ns6 = min(200, max(60, Nf // 3))
    h6, c56 = steepest_binary(hF, Nf, deadline, max_iters=15000, n_2swap=ns6, rng=rng)
    if c56 < best_c5:
        best_c5, best_h = c56, h6.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
