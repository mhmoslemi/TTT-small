    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def fix_sum(h, nh):
    cur = int(np.round(h.sum()))
    if cur > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:cur - nh]] = 0
    elif cur < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - cur]] = 1
    return h

def run_local_search(h, N, deadline, sample_sz=200, top_k=25, kick_after=100, kick_sz=7):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    C = np.array([compute_C_single(h, N, i) for i in range(N)])
    it = 0
    stuck = 0
    while time.time() < deadline:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        top_lags = np.argpartition(corr, -top_k)[-top_k:]
        Co = C[os_][:, top_lags]
        Cz = C[zs][:, top_lags]
        ct = corr[top_lags]
        newP = (ct[None, None, :] + Co[None, :, :] + Cz[:, None, :]).max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        ii = os_[mi[1]]
        jj = zs[mi[0]]
        h[ii] = 0
        h[jj] = 1
        corr_new = fft_corr(h)
        P_new = float(np.max(corr_new))
        if P_new < P - 1e-14:
            C[ii] = compute_C_single(h, N, ii)
            C[jj] = compute_C_single(h, N, jj)
            corr = corr_new
            P = P_new
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
            stuck = 0
        else:
            h[ii] = 1
            h[jj] = 0
            stuck += 1
            if stuck >= kick_after:
                for _ in range(kick_sz):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0
                    h[q] = 1
                    C[p] = compute_C_single(h, N, p)
                    C[q] = compute_C_single(h, N, q)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = fft_corr(h)
                P = float(np.max(corr))
                stuck = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Primary local search (55% budget)
    h_bin = (h0 > 0.5).astype(float)
    fix_sum(h_bin, N0 // 2)
    d1 = t0 + int(0.55 * (budget_s - 5))
    h1, _ = run_local_search(h_bin, N0, d1, sample_sz=200, top_k=25, kick_after=100, kick_sz=7)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: ILS with perturbations (25% budget)
    n_ils = 4
    ils_total = int(0.25 * (budget_s - 5))
    ils_per = max(ils_total // n_ils, 25)
    for r in range(n_ils):
        if time.time() > deadline - 130:
            break
        h_p = best_h.copy()
        n_flip = np.random.randint(6, 16)
        ones_p = np.where(h_p == 1)[0]
        zeros_p = np.where(h_p == 0)[0]
        for _ in range(n_flip):
            p = int(np.random.choice(ones_p))
            q = int(np.random.choice(zeros_p))
            h_p[p] = 0
            h_p[q] = 1
            ones_p = np.where(h_p == 1)[0]
            zeros_p = np.where(h_p == 0)[0]
        d_i = min(deadline - 130 - (n_ils - r - 1) * ils_per, time.time() + ils_per)
        h_i, _ = run_local_search(h_p, N0, d_i, sample_sz=200, top_k=25, kick_after=80, kick_sz=6)
        c_i = compute_c5(h_i, N0)
        if c_i < best_c5:
            best_c5, best_h = c_i, h_i.copy()

    # Phase 3: Try alternate resolutions (10% budget)
    for N_try in [360, 420]:
        if time.time() > deadline - 70:
            break
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = (h_int > 0.5).astype(float)
        fix_sum(h_bt, N_try // 2)
        d_try = min(deadline - 50, time.time() + 35)
        h_rt, _ = run_local_search(h_bt, N_try, d_try, sample_sz=min(180, N_try // 2), top_k=20, kick_after=60, kick_sz=5)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 4: Final intensive polish with larger top_k
    if time.time() < deadline - 2:
        hF = (best_h > 0.5).astype(float)
        fix_sum(hF, len(best_h) // 2)
        hF, _ = run_local_search(hF, len(best_h), deadline - 0.5, sample_sz=250, top_k=35, kick_after=200, kick_sz=5)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
