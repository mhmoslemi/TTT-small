                # Simplify: just do 1-swap if 2-swap is complex
            elif t == 3:
                h[best_move[1]] = 0.0
                h[best_move[2]] = 0.0
                h[best_move[3]] = 0.0
                h[best_move[4]] = 1.0
                h[best_move[5]] = 1.0
                h[best_move[6]] = 1.0
        else:
            if stall > 35:
                n_esc = min(2 + stall // 15, 6)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_esc):
                    p = int(ones_now[rng.randint(len(ones_now))])
                    q = int(zeros_now[rng.randint(len(zeros_now))])
                    h[p], h[q] = 0.0, 1.0
                    zeros_now = np.delete(zeros_now, np.where(zeros_now == q))
                    ones_now = np.append(ones_now, q)
                    ones_now = np.delete(ones_now, np.where(ones_now == p))
                stall = 0
            elif stall > 18:
                break
            else:
                stall += 1
                continue

        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-14:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def binary_anneal_v2(h, N, deadline, max_sweeps=500, sample_frac=1.0, two_swap_prob=0.4, rng=None):
    """Cleaner implementation with correct 2-swap handling."""
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    if rng is None:
        rng = np.random.RandomState(7)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    temp = 0.015
    stall = 0
    c_cur = fft_corr(h)
    P_cur = float(c_cur.max())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        temp = max(0.002, temp * 0.993)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        D_zeros = D[zeros, :]
        best_new_P = P_cur
        best_move = None

        # 1-swap: new C[k] = C[k] + D[p,k] - D[q,k]
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(50, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
            D_p = D[p_list, :]
        else:
            p_list = ones
            D_p = D[ones, :]
        cand1 = (c_cur[None, :] + D_p) - D_zeros
        npk1 = cand1.max(axis=1)
        mi1, mj1 = np.unravel_index(np.argmin(npk1), npk1.shape)
        if npk1[mi1] < best_new_P:
            best_new_P = float(npk1[mi1])
            best_move = ('1s', int(p_list[mi1]), int(zeros[mj1]))

        # 2-swap: new C[k] = C[k] + D[p1,k] + D[p2,k] - D[q1,k] - D[q2,k]
        if len(ones) >= 2 and len(zeros) >= 2 and rng.random() < two_swap_prob:
            n_try = min(200, max(40, N // 3))
            for _ in range(n_try):
                i1 = rng.randint(len(ones))
                i2 = rng.randint(len(ones))
                if i1 == i2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                base2 = c_cur + D[p1] + D[p2]
                # Find best pair of zeros greedily
                rem_D = D_zeros.copy()
                rem_z = zeros.copy()
                bs = base2.copy()
                qs = []
                for _g in range(2):
                    c2 = (bs[None, :] - rem_D).max(axis=1)
                    mg = int(np.argmin(c2))
                    qs.append(int(rem_z[mg]))
                    bs = bs - rem_D[mg]
                    rem_D = np.delete(rem_D, mg, axis=0)
                    rem_z = np.delete(rem_z, mg)
                fp = float((c_cur + D[p1] + D[p2] - D[qs[0]] - D[qs[1]]).max())
                if fp < best_new_P:
                    best_new_P = fp
                    best_move = ('2s', p1, p2, qs[0], qs[1])

        # Apply
        if best_move is not None:
            if best_move[0] == '1s':
                h[best_move[1]] = 0.0
                h[best_move[2]] = 1.0
            else:
                h[best_move[1]] = 0.0
                h[best_move[2]] = 0.0
                h[best_move[3]] = 1.0
                h[best_move[4]] = 1.0
        else:
            if stall > 40:
                n_esc = min(3 + stall // 10, 7)
                ones_n = np.where(h == 1)[0].tolist()
                zeros_n = np.where(h == 0)[0].tolist()
                for _ in range(n_esc):
                    if not ones_n or not zeros_n:
                        break
                    pi = rng.randint(len(ones_n))
                    qi = rng.randint(len(zeros_n))
                    p, q = ones_n[pi], zeros_n[qi]
                    h[p] = 0.0
                    h[q] = 1.0
                    ones_n[pi] = q
                    zeros_n[qi] = p
                    ones_n, zeros_n = sorted(ones_n), sorted(zeros_n)
                stall = 0
            elif stall > 20:
                break
            else:
                stall += 1
                continue

        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-14:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=80, lr0=0.013, rng=None):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    if rng is None:
        rng = np.random.RandomState(99)
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= max(k, 1)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.991
            beta = min(0.97, 0.82 + it * 0.00015)
        if it % 800 == 0 and it > 0:
            h = h + rng.randn(N) * 0.008
            h = project_sum(h, target)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary anneal v2 at N0 (fixed signs, 2-swap)
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 480, t0 + budget_s * 0.28)
    h1, c5_1 = binary_anneal_v2(h_bin, N0, d1, max_sweeps=2000, sample_frac=1.0, two_swap_prob=0.4, rng=rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Nesterov continuous at N0
    if time.time() < deadline - 370:
        hc = best_h + rng.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 340, time.time() + budget_s * 0.11)
        h2, c5_2 = nesterov_opt(hc, N0, d2, k=100, lr0=0.011, rng=rng)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Upscale to 2*N0, binary anneal
    N1 = 2 * N0
    if time.time() < deadline - 230:
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d3 = min(deadline - 190, time.time() + budget_s * 0.16)
        h3, c5_3 = binary_anneal_v2(h_up, N1, d3, max_sweeps=500, sample_frac=0.45, two_swap_prob=0.35, rng=rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Nesterov at current N
    if time.time() < deadline - 140:
        N_cur = len(best_h)
        hc2 = best_h + rng.randn(N_cur) * 0.035
        hc2 = project_sum(hc2, N_cur / 2.0)
        d4 = min(deadline - 110, time.time() + budget_s * 0.08)
        h4, c5_4 = nesterov_opt(hc2, N_cur, d4, k=70, lr0=0.009, rng=rng)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Multi-restart binary at N0
    if time.time() < deadline - 70:
        for rs in range(2):
            if time.time() > deadline - 70:
                break
            rng_r = np.random.RandomState(seed + 500 + rs * 111)
            h_r = ensure_sum(best_h[:N0] if len(best_h) >= N0 else upsample(best_h, len(best_h), N0), N0)
            ones_t = np.where(h_r == 1)[0]
            zeros_t = np.where(h_r == 0)[0]
            n_flips = 6 + rs * 3
            for _ in range(n_flips):
                if len(ones_t) == 0 or len(zeros_t) == 0:
                    break
                pi = rng_r.randint(len(ones_t))
                qi = rng_r.randint(len(zeros_t))
                h_r[ones_t[pi]] = 0.0
                h_r[zeros_t[qi]] = 1.0
                ones_t = np.where(h_r == 1)[0]
                zeros_t = np.where(h_r == 0)[0]
            h_r = ensure_sum(h_r, N0)
            d5 = min(deadline - 60 - rs * 30, time.time() + budget_s * (0.04 + rs * 0.02))
            h5, c5_5 = binary_anneal_v2(h_r, N0, d5, max_sweeps=600, sample_frac=0.85, two_swap_prob=0.4, rng=rng_r)
            if c5_5 < best_c5:
                best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final polish
    N_fin = len(best_h)
    hF = ensure_sum(best_h, N_fin)
    sf = 1.0 if N_fin <= 400 else 0.5
    hF2, c5F = binary_anneal_v2(hF, N_fin, max(deadline - 1, time.time() + 30), max_sweeps=400, sample_frac=sf, two_swap_prob=0.3, rng=rng)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
