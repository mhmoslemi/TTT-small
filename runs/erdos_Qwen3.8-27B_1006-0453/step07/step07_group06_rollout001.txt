import numpy as np
import time
from scipy.optimize import minimize as sp_min

def corr_full(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)

def corr_max(h):
    return float(np.max(corr_full(h)))

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj(h, T, target, lam):
    N = len(h)
    c = corr_full(h)
    mx = c.max()
    c_safe = c - mx
    c_safe = np.clip(c_safe, -100, 0)
    e = np.exp(T * c_safe)
    e_sum = e.sum()
    w = e / e_sum
    val = mx + np.log(e_sum) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(N * 2 - 1):
        if w[i] < 1e-10:
            continue
        k = i - ctr
        if k == 0:
            g += w[i] * (1.0 - 2.0 * h)
        elif k > 0:
            wk = w[i]
            g[k:] += wk * (1.0 - h[:N - k])
            g[:N - k] -= wk * h[k:]
        else:
            M = -k
            wk = w[i]
            g[:N - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:N - M]
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, N, deadline, T_start=15.0, T_end=120.0, n_outer=5):

                     jac=True, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 200, 'maxfun': 600, 'ftol': 1e-15, 'gtol': 1e-12})
        h = proj01(res.x, target)
        c = corr_max(h)

            best_h = h.copy()

def grad_max_corr(h):

    pk = c.max()
    atol = 1e-6
    active = np.where(np.abs(c - pk) < atol)[0]
    if len(active) == 0:

    g = np.zeros(N)
    ctr = N - 1

        else:
            M = -k
            g[:N - M] += (1.0 - h[M:])
            g[M:] -= h[:N - M]
    g /= len(active)
    return g

            h_new = proj01(y - alpha * g, target)
            m = beta * (h_new - h4)
            h4 = h_new
            c4 = corr_max(h4)
            if c4 < b4c - 1e-10:
                b4c = c4
                b4h = h4.copy()
            if it % 300 == 0:
                alpha *= 0.94
                beta = min(0.97, 0.90 + 0.0002 * it)
        if b4c < best_c5:
            best_c5, best_h = b4c, b4h.copy()

    # Phase 5: multi-resolution
    if time.time() < dl - 80:
        best_h, best_c5 = multi_res(best_h, min(dl - 40, time.time() + 120))

    # Phase 6: final L-BFGS
    if time.time() < dl - 35:
        d6 = min(dl - 20, time.time() + 50)
        h6 = best_h + np.random.randn(len(best_h)) * 0.04
        h6 = proj01(h6, len(best_h) / 2.0)
        hl6, cl6 = lbfgs_opt(h6, len(best_h), d6, T_start=25, T_end=100, n_outer=3)
        if cl6 < best_c5:
            best_c5, best_h = cl6, hl6.copy()

    # Phase 7: final binary
    if time.time() < dl - 3.0:
        hf = (best_h > 0.5).astype(float)
        hf, cf = binary_search(hf, len(best_h), dl - 1.0, max_iters=1500, n_trials=30, cool=0.998)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Final: compare continuous vs binary
    fc5 = corr_max(best_h)
    fb = (best_h > 0.5).astype(float)
    fc5b = corr_max(fb)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    # Ensure constraint
    tgt = len(best_h) // 2
    n1 = int(best_h.sum())
    if n1 != tgt:
        if n1 > tgt:
            idx = np.where(best_h > 0.5)[0]
            best_h[idx[np.random.choice(len(idx), n1 - tgt, replace=False)]] = 0.0
        else:
            idx = np.where(best_h < 0.5)[0]
            best_h[idx[np.random.choice(len(idx), tgt - n1, replace=False)]] = 1.0
        fc5 = corr_max(best_h)
    return best_h, fc5, len(best_h)
