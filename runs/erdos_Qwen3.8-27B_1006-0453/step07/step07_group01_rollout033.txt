import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=50, T=80.0, lr0=0.018):

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0002)
    return best_h, best_pk * dx

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        idx = ones[order[:n1 - target]]
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        idx = zeros[order[:target - n1]]
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_search(h, N, deadline, max_sweeps=3000, rng_seed=42):
    h = h.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(rng_seed)
    
    D = compute_D(h, N)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        P = float(corr.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        Dz = D[zeros, :]
        
        # Pruning: compute upper bound for each p
        # U[p] = max(corr + D[p,:])
        ones_D = D[ones, :]  # (n_ones, 2N-1)
        U = (corr[None, :] + ones_D).max(axis=1)  # (n_ones,)
        
        # Filter: only p's with U[p] < P (potential to improve)
        feasible = np.where(U < P - 1e-12)[0]
        if len(feasible) == 0:

        best_new_P = P

        best_qi = None
        
        # Process feasible p's (at most all, usually much fewer)
        for fi in feasible:
            p = ones[fi]
            base = corr + D[p, :]
            npk = (base[None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            val = float(npk[mi])
            if val < best_new_P - 1e-12:
                best_new_P = val
                best_pi = fi
                best_qi = mi
        
        if best_pi is not None:
            p = int(ones[best_pi])
            q = int(zeros[best_qi])
            h[p] = 0.0
            h[q] = 1.0
            D = compute_D(h, N)
            corr = fft_corr(h)
            stall = 0
        else:
            stall += 1
            # 2-for-2 or perturbation
            if stall > 3 and len(ones) >= 2 and len(zeros) >= 2:
                done = False
                n_try = min(20, stall * 5)
                for _ in range(n_try):

                    if nP < P:
