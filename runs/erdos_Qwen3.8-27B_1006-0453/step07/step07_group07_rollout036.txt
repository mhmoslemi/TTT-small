This gives a weighted average of the subgradients, with weights proportional to exp(tau * corr[k]). Large tau → focuses on the peak; small tau → spreads out.

This could help escape local optima where the max function has a "ridge" in the subgradient.

def soft_max_grad(h, tau=5.0):
    N = len(h)
    L = 2*N - 1
    corr = compute_corr(h)
    # Soft max
    max_c = np.max(corr)
    exp_c = np.exp(tau * (corr - max_c))  # shift for numerical stability
    weights = exp_c / exp_c.sum()
    # Gradient is weighted sum of subgradients
    grad = np.zeros(N)
    for k in range(L):
        s = k - (N-1)
        w = weights[k]
        if w < 1e-6:

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:

        np.random.shuffle(idx)

    return h

def soft_max_cont_opt(h, deadline, tau0=3.0, lr0=0.02, mom=0.9):

    N = len(h)
    tgt = N / 2.0
    L = 2 * N - 1
    center = N - 1
    h = h.astype(float).copy()
    v = np.zeros(N)

    best_c = compute_c5(h)
    tau = tau0
    stale = 0

        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 50:
                tau *= 0.9  # sharpen the max
                lr0 *= 0.85
                stale = 0
        
        # Compute soft-max weighted gradient
        max_c = np.max(corr)
        exp_c = np.exp(tau * (corr - max_c))
        weights = exp_c / exp_c.sum()
        
        # Sparse weighting: only use lags with significant weight
        thresh = 1.0 / (len(weights) * 0.3)  # keep top ~30%
        active = weights > thresh
        n_active = int(active.sum())
        if n_active < 3:
            active = np.argsort(weights)[-10:]
            active = np.array([True] * L)
            active[~np.isin(np.arange(L), np.argsort(weights)[-10:])] = False
        
        grad = np.zeros(N)
        for k in np.where(active)[0]:
            s = int(k) - center
            w = weights[k]
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            elif s > 0 and s < N:
                hi_i = np.arange(s, N)
                lo_i = np.arange(0, N - s)
                grad[hi_i] += w * (1.0 - h[lo_i])
                grad[lo_i] -= w * h[hi_i]
            else:
                s = -s
                hi_i = np.arange(s, N)
                lo_i = np.arange(0, N - s)
                grad[hi_i] += w * (1.0 - h[lo_i])
                grad[lo_i] -= w * h[hi_i]

        v = mom * v + (h_new - h)
        h = h_new

        # s < 0: p+s = p-t, p-s = p+t
        # d_rem: -(1-h[p-t])*I(p-t valid) + h[p+t]*I(p+t valid)
        v1o = ones >= t      # p-t >= 0
        v2o = ones < N - t   # p+t < N
        d_rem[v1o, li] -= (1.0 - h[ones[v1o] - t])
        d_rem[v2o, li] += h[ones[v2o] + t]
        # d_add: (1-h[p-t])*I(valid) - h[p+t]*I(valid)
        v1z = zeros >= t

        d_add[v1z, li] += (1.0 - h[zeros[v1z] - t])
        d_add[v2z, li] -= h[zeros[v2z] + t]

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def soft_max_cont_opt(h, deadline, tau0=4.0, lr0=0.018, mom=0.91):
    N = len(h)
    tgt = N / 2.0
    L = 2 * N - 1
    center = N - 1
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    tau = tau0
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 40:
                tau = max(tau * 0.92, 0.5)
                lr0 *= 0.82
                stale = 0
        max_c = np.max(corr)
        exp_c = np.exp(tau * (corr - max_c))
        weights = exp_c / exp_c.sum()
        thresh = 1.0 / (L * 0.35)
        active = np.where(weights > thresh)[0]
        if len(active) < 5:
            active = np.argsort(weights)[-15:]
        grad = np.zeros(N)
        for k in active:
            s = int(k) - center
            w = weights[k]
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            elif abs(s) < N:
                t = abs(s)
                hi_i = np.arange(t, N)
                lo_i = np.arange(0, N - t)
                if s > 0:
                    grad[hi_i] += w * (1.0 - h[lo_i])
                    grad[lo_i] -= w * h[hi_i]
                else:
                    grad[hi_i] += w * (1.0 - h[lo_i])
                    grad[lo_i] -= w * h[hi_i]
        h_new = project(h + mom * v - lr0 * grad, tgt)
        v = mom * v + (h_new - h)
        h = h_new
    return best_h, best_c

def binary_il_v2(h, deadline, max_iters=8000, n_crit=28, top_k=45):
    N = len(h)
    center = N - 1
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    best_c = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    no_imp = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 500:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones, n_zeros = len(ones), len(zeros)
        if n_ones < 2 or n_zeros < 2:
            break
        top_lags = np.argsort(corr)[-n_crit:]
        shifts = (top_lags - center).astype(int)
        cur_max = float(np.max(corr))
        n_lags = len(shifts)
        d_rem = np.zeros((n_ones, n_lags))
        d_add = np.zeros((n_zeros, n_lags))
        for li in range(n_lags):
            s = shifts[li]
            if s == 0:
                continue
            if s > 0:
                v1o = ones < N - s
                v2o = ones >= s
                d_rem[v1o, li] -= (1.0 - h[ones[v1o] + s])
                d_rem[v2o, li] += h[ones[v2o] - s]
                v1z = zeros < N - s
                v2z = zeros >= s
                d_add[v1z, li] += (1.0 - h[zeros[v1z] + s])
                d_add[v2z, li] -= h[zeros[v2z] - s]
            else:
                t = -s
                v1o = ones >= t
                v2o = ones < N - t
                d_rem[v1o, li] -= (1.0 - h[ones[v1o] - t])
                d_rem[v2o, li] += h[ones[v2o] + t]
                v1z = zeros >= t
                v2z = zeros < N - t
                d_add[v1z, li] += (1.0 - h[zeros[v1z] - t])
                d_add[v2z, li] -= h[zeros[v2z] + t]
        ones_score = d_rem.sum(axis=1)
        zeros_score = d_add.sum(axis=1)
        tk = min(top_k, n_ones)
        tkz = min(top_k, n_zeros)
        top_oi = np.argsort(ones_score)[:tk]
        top_zi = np.argsort(zeros_score)[:tkz]
        base = corr[top_lags][None, None, :] + d_rem[top_oi][:, None, :]
        new_vals = base + d_add[top_zi][None, :, :]
        new_max_arr = new_vals.max(axis=2)
        flat_idx = np.argmin(new_max_arr)
        bi, bj = np.unravel_index(flat_idx, new_max_arr.shape)
        npk = new_max_arr[bi, bj]
        if npk < cur_max - 1e-10:
            oi = ones[top_oi[bi]]
            zi = zeros[top_zi[bj]]
            h[oi] = 0
            h[zi] = 1
        elif no_imp > 150 and np.random.random() < 0.35:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        it += 1
    return best_h, best_c
