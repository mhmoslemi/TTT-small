def greedy_local_search(h_init, N, deadline, max_sweeps=300):

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        shifts = np.arange(2*N-1) - (N-1)
        # Compute delta for each (p in ones, s in shifts)
        # This is O(N^2) which is too slow for N=1140
        # Instead, use the previous approach with matrix
        p_all = np.arange(N)[:, None]
        s_all = shifts[None, :]
        pm = p_all - s_all
        pp = p_all + s_all
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N-1)
        pp_c = np.clip(pp, 0, N-1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
        
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        
        # Sample ones for speed
        if len(ones) > 200:
            rng = np.random.RandomState(sweep + N)
            p_list = rng.choice(ones, 200, replace=False)
        else:
            p_list = ones
        
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    
    return best_h, best_c * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(77)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def nesterov_opt(h_init, N, deadline, k=40, lr0=0.02):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        w = np.exp(60.0 * (corr[idx] - corr[idx].max()))
        w /= w.sum()
        grad = np.zeros(N)
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += wi * (1.0 - h[s:])
                grad[s:] -= wi * h[:N-s]
            else:
                M = -s
                grad[M:] += wi * (1.0 - h[:N-M])
                grad[:N-M] -= wi * h[M:]
        h_new = project_sum(y - lr * grad / k, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.88 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def sa_binary(h_init, N, deadline, T0=0.3, T_end=0.005, seed=42, restart_gap=4000):
    h = h_init.astype(float).copy()
    rng = np.random.RandomState(seed)
    cur_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    best_c = cur_c
    no_improve = 0
    start_time = time.time()
    total_time = max(1.0, deadline - start_time)
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / total_time)
        T = T0 * (T_end / T0) ** frac
        valid = False
        for _ in range(25):
            p = rng.randint(N)
            q = rng.randint(N)
            if p != q and h[p] == 1.0 and h[q] == 0.0:
                valid = True
                break
        if not valid:
            continue
        h[p] = 0.0
        h[q] = 1.0
        nc = float(np.max(fft_corr(h)))
        if nc < cur_c - 1e-14:
            cur_c = nc
            if nc < best_c - 1e-14:
                best_c = nc
                best_h = h.copy()
            no_improve = 0

            if rng.random() < 0.4:
                no_improve += 1
            else:
                h[p] = 1.0
                h[q] = 0.0
                no_improve += 1

                cur_c = nc
                no_improve += 1
            else:

                h[q] = 0.0

def greedy_ls(h_init, N, deadline, max_sweeps=300, sample_n=200):
    h = h_init.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        shifts = np.arange(2 * N - 1) - (N - 1)
        p_all = np.arange(N)[:, None]
        s_all = shifts[None, :]
        pm = p_all - s_all
        pp = p_all + s_all
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
        D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
        Dz = D[zeros, :]
        best_new_P = P
        best_pq = None
        if len(ones) > sample_n:
            rs = np.random.RandomState(sweep * 7 + N)
            p_list = rs.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        for p in p_list:
            base = c + D[p, :]
            npk = (base[None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def multi_sa_restart(h_init, N, deadline, n_restarts=3, seed=42):

    cur_best_h = h_init.copy()

    rng = np.random.RandomState(seed)
    for r in range(n_restarts):
        rem = deadline - time.time()
        if rem < 60:
            break
        alloc = min(rem - 30, 200)
        sub_dead = time.time() + alloc
        h_start = cur_best_h.copy()
        n_perturb = rng.randint(2, 5)
        for _ in range(n_perturb):
            for _a in range(30):
                p = rng.randint(N)

                if p != q and h_start[p] == 1 and h_start[q] == 0:
                    h_start[p] = 0.0
                    h_start[q] = 1.0

        T0_val = 0.35 - r * 0.08
        h_r, c5_r = sa_binary(h_start, N, sub_dead, T0=T0_val, T_end=0.003, seed=seed + r * 31)
        if c5_r < cur_best_c:
            cur_best_c = c5_r
            cur_best_h = h_r.copy()
    return cur_best_h, cur_best_c
