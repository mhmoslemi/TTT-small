def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    c = fft_corr(h)
    mx = c.max()

                if worst_new < max_val - 1e-12:
                    delta = max_val - worst_new
                    if delta > best_delta:
                        best_delta = delta
                        best_i, best_j = int(i), int(j)
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            corr = fft_corr(h)
            if np.max(corr) < best_c - 1e-12:
                best_c = float(np.max(corr))
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def random_binary(N, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(N)
    idx = rng.permutation(N)
    h[idx[:N // 2]] = 1.0
    return h

def subgrad_ne(h, N, deadline, alpha0=0.03, beta=0.92, atol=1e-5, max_it=5000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < max_it:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            alpha *= 0.95
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Greedy binary at parent resolution
    d1 = min(dl - 500, t0 + 80)
    hb, cb = greedy_binary(best_h, N0, d1, max_iters=4000, topk=3)
    c5b = compute_c5(hb, N0)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 2: L-BFGS smooth (multiple restarts)
    if time.time() < dl - 400:
        d2 = min(dl - 350, time.time() + 150)
        for r in range(5):
            if time.time() > d2:
                break
            scale = 0.2 - 0.03 * r
            hr = best_h + np.random.randn(N0) * scale
            hr = proj01(hr, N0 / 2.0)
            hl, cl = lbfgs_smooth(hr, N0, d2, T=30.0, max_outer=2)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 3: Greedy binary after L-BFGS
    if time.time() < dl - 300:
        d3 = min(dl - 250, time.time() + 60)
        hb3, cb3 = greedy_binary(best_h, N0, d3, max_iters=3000, topk=4)
        c53 = compute_c5(hb3, N0)
        if c53 < best_c5:
            best_c5, best_h = c53, hb3.copy()

    # Phase 4: Subgradient Nesterov
    if time.time() < dl - 200:
        d4 = min(dl - 160, time.time() + 80)
        hs = best_h + np.random.randn(N0) * 0.06
        hs = proj01(hs, N0 / 2.0)
        hs4, c4 = subgrad_ne(hs, N0, d4, alpha0=0.025, beta=0.94)
        if c4 < best_c5:
            best_c5, best_h = c4, hs4.copy()

    # Phase 5: Multi-resolution greedy
    if time.time() < dl - 120:
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        for Nt in [260, 300, 340, 420, 500, 600, 760, 900]:
            if time.time() > dl - 60:
                break
            if Nt % 2:
                Nt += 1
            if Nt == len(best_h):
                continue
            sd = min(dl - 50, time.time() + 30)
            xn = np.linspace(0, 2, Nt, endpoint=False)
            hi = np.interp(xn, x_b, best_h)
            ht = (hi > 0.5).astype(float)
            nh = Nt // 2
            n1 = int(ht.sum())
            if n1 > nh:
                ht[np.random.choice(np.where(ht == 1)[0], n1 - nh, replace=False)] = 0
            elif n1 < nh:
                ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1
            hr, cr = greedy_binary(ht, Nt, sd, max_iters=2500, topk=3)
            c5r = compute_c5(hr, Nt)
            if c5r < best_c5:
                best_c5, best_h = c5r, hr.copy()
                x_b = np.linspace(0, 2, Nt, endpoint=False)

    # Phase 6: Final L-BFGS polish
    if time.time() < dl - 40:
        d6 = min(dl - 20, time.time() + 50)
        h6 = best_h + np.random.randn(len(best_h)) * 0.04
        h6 = proj01(h6, len(best_h) / 2.0)
        hl6, cl6 = lbfgs_smooth(h6, len(best_h), d6, T=40.0, max_outer=2)
        if cl6 < best_c5:
            best_c5, best_h = cl6, hl6.copy()

    # Phase 7: Final greedy polish
    if time.time() < dl - 5:
        d7 = dl - 1.0
        best_h, c7 = greedy_binary(best_h, len(best_h), d7, max_iters=3000, topk=5)
        c7v = compute_c5(best_h, len(best_h))
        if c7v < best_c5:
            best_c5 = c7v

    # Final: try binary vs continuous
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, len(best_h))
    fc5c = compute_c5(best_h, len(best_h))
    if fc5b < fc5c:
        best_h, best_c5 = fb, fc5b
    else:
        best_c5 = fc5c
    return best_h, best_c5, len(best_h)
