    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0

    it = 0

        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        # Top-k soft max gradient
        idx = np.argpartition(corr, -k)[-k:]

        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(77)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.997
            beta = min(0.98, 0.87 + it * 0.0001)
        if it % 250 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(20.0, min(500.0, T * (0.88 if gap < 0.004 else 1.07)))
    return best_h, best_pk * (2.0 / N)

def fast_binary_ls(h, N, deadline, perturb_every=25):
    h = ensure_binary(h, N).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(99)
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        # Evaluate best single swap using correlation structure
        # For swap p(1→0), q(0→1): Δc[s] = -(1-h[p+s]) + h[p-s] + (1-h[q+s]) - h[q-s]
        # We want to minimize max(c + Δc)
        
        # Fast approach: precompute the "cost" of removing each 1
        # and "gain" of adding each 0, at the peak location
        peak_s = int(np.argmax(c)) - (N - 1)  # shift value
        
        best_new_P = P
        best_pq = None
        # Sample
        ps = ones if len(ones) <= 300 else rng.choice(ones, 300, replace=False)
        qs = zeros if len(zeros) <= 300 else rng.choice(zeros, 300, replace=False)
        
        # For efficiency, compute swap effect at top-20 peak locations
        top_k = np.argpartition(c, -20)[-20:]
        
        for p in ps:
            p_cost = np.zeros(20)
            p_gain = np.zeros(20)
            for ti, t in enumerate(top_k):
                s = int(t) - (N - 1)
                val = 0.0
                if 0 <= p + s < N:
                    val -= (1.0 - h[p + s])
                if 0 <= p - s < N:
                    val += h[p - s]
                p_cost[ti] = val
            for q in qs:
                q_gain = np.zeros(20)
                for ti, t in enumerate(top_k):
                    s = int(t) - (N - 1)
                    val = 0.0
                    if 0 <= q + s < N:
                        val += (1.0 - h[q + s])
                    if 0 <= q - s < N:
                        val -= h[q - s]
                    q_gain[ti] = val
                # Check if max at top-20 would decrease
                new_top = c[top_k] + p_cost + q_gain
                # Also need to check if non-top locations could overtake
                # Conservative: use P + max(p_cost + q_gain) as upper bound for non-top
                nontop_bound = P + (p_cost.max() + q_gain.max())
                cand_max = max(float(new_top.max()), nontop_bound)
                if cand_max < best_new_P:
                    best_new_P = cand_max
                    best_pq = (int(p), int(q))
        
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            if stall > perturb_every:
                n_r = min(3 + stall // 10, 10)
                for _ in range(n_r):
                    p = int(rng.choice(ones))
                    q = int(rng.choice(zeros))
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
    return best_h, best_c * (2.0 / N)

def simulated_anneal(h, N, deadline, T0=0.5, T1=0.005):
    h = ensure_binary(h, N).copy()
    c = fft_corr(h)
    cur_P = float(c.max())
    best_P = cur_P
    best_h = h.copy()
    rng = np.random.RandomState(200)
    n_iter = 0
    n_steps = 8000
    while time.time() < deadline:
        n_iter += 1
        T = max(T1, T0 * np.exp(-3.0 * n_iter / n_steps))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = int(rng.choice(ones))
        q = int(rng.choice(zeros))
        h[p], h[q] = 0.0, 1.0
        c_new = fft_corr(h)
        new_P = float(c_new.max())
        if new_P < cur_P or rng.random() < np.exp(-(new_P - cur_P) / max(T, 1e-12)):
            cur_P = new_P
            c = c_new
            if cur_P < best_P - 1e-13:
                best_P = cur_P
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
    return best_h, best_P * (2.0 / N)
