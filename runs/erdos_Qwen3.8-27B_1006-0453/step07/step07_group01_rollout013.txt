    h = ensure_binary(h, N)
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Find current max shift
        s_idx = int(np.argmax(c))
        s = s_idx - center  # actual shift
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Try flipping each 1 at positions where it contributes to the max shift
        # At shift s: contribution from position i is h[i]*(1-h[i+s])
        # For s > 0: valid i are 0 to N-1-s
        # For s < 0: valid i are -s to N-1
        
        best_delta = 0  # want to minimize P, so want most negative delta in max
        best_pq = None
        
        # Candidate positions to flip (1→0): those contributing to the max shift
        if s >= 0:
            valid_i = np.arange(N - s)
        else:
            valid_i = np.arange(-s, N)
        
        contributing_ones = [i for i in valid_i if h[i] == 1 and h[i+s] == 0]
        
        # Also try other 1-positions (not just contributing ones)
        if len(contributing_ones) < 20:
            contributing_ones = list(ones[:min(50, len(ones))])
        
        # For each candidate 1-flip and a subset of 0-flips
        for p in contributing_ones[:30]:
            # Compute new corr after flipping p (1→0)
            # delta_p[s'] for the correlation
            # Simplified: just compute via the difference formula
            new_c = c.copy()
            # p goes 1→0
            for s2 in range(N - p):
                new_c[s2 + center] -= (1.0 - h[p + s2])
            for s2 in range(p + 1):
                new_c[s2 + center] += h[p - s2]
            
            # Now try flipping each zero
            # For efficiency, just try a few promising zeros
            # Best zero to flip: one where flipping helps the most
            # After flipping p, the new max is at some shift. Flip a zero to help there.
            new_max_idx = int(np.argmax(new_c))
            new_s2 = new_max_idx - center
            
            # Zeros that would help at new_s2:
            if new_s2 >= 0:
                # At shift new_s2, we want to flip h[i+s2] for i where h[i]=1
                # Or flip h[i] where h[i]=0 and h[i+new_s2]=1
                cand_zeros = zeros[zeros >= new_s2][:10]
                # Also try zeros at position i+new_s2 where h[i]=1
                for i in range(N - new_s2):
                    if h[i] == 1 and i + new_s2 < N and h[i + new_s2] == 0:
                        if (i + new_s2) not in cand_zeros:
                            cand_zeros = np.append(cand_zeros, i + new_s2)
                cand_zeros = cand_zeros[:15]
            else:
                cand_zeros = zeros[zeros <= -new_s2][:15]
            
            for q in cand_zeros:
                new_c2 = new_c.copy()
                # q goes 0→1
                for s3 in range(N - int(q)):
                    new_c2[s3 + center] += (1.0 - h[int(q) + s3])
                for s3 in range(int(q) + 1):
                    new_c2[s3 + center] -= h[int(q) - s3]
                
                # Check interaction: if p and q interact
                # (if p+s2 = q or q+s2 = p for some relevant s2)
                # This is a second-order effect, ignore for now
                
                P_try = float(new_c2.max())
                if P_try < P - 1e-10:
                    if P_try < P + best_delta:
                        best_delta = P_try - P
                        best_pq = (p, int(q))
            
            if best_delta < -0.01:  # good improvement found, break early
                break
        
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_c - 1e-13:
                best_c = P
                best_h = h.copy()
        else:
            # Try random perturbations
            n_pert = np.random.randint(1, 4)
            for _ in range(n_pert):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_c - 1e-13:
                best_c = P
                best_h = h.copy()
            if P > best_c + 0.05:
                # Too bad, revert
                h = best_h.copy()
                c = fft_corr(h)
                P = float(c.max())
    
    return best_h, best_c * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def thue_morse(N):
    t = np.zeros(N, dtype=np.float64)
    for i in range(1, N):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def ensure_binary(h, N, seed=0):
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_flip(h, N):
    """Compute F[p] for all p: change in corr when h[p] goes 1->0."""
    L = 2 * N - 1
    center = N - 1
    F = np.zeros((N, L))
    # For each p: F[p][j] = -(1-h[p+j-N+1])*I + h[p-j+N+1-1]*I - I(j==center)
    # Simplify: p+j-N+1 is valid when 0 <= p+j-N+1 < N, i.e., N-1-p <= j <= N-1+p
    # p-j+N-1 is valid when 0 <= p-j+N-1 < N, i.e., N-1-p <= j <= N-1+p
    # Same range! Good.
    for p in range(N):
        j_lo = N - 1 - p
        j_hi = N - 1 + p
        # First term: -(1 - h[p + j - N + 1]) for j in [j_lo, j_hi]
        idx1 = p + np.arange(j_lo, j_hi + 1) - N + 1  # = np.arange(0, 2*p+1)
        F[p, j_lo:j_hi+1] -= (1.0 - h[idx1])
        # Second term: +h[p - j + N - 1] for j in [j_lo, j_hi]
        idx2 = p - np.arange(j_lo, j_hi + 1) + N - 1  # = np.arange(p, -1, -1)
        F[p, j_lo:j_hi+1] += h[idx2]
        # Correction at j=center
        F[p, center] -= 1.0
    return F
