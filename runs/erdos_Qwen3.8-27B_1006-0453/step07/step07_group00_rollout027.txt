    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def compute_grad(h, w, N):
    L = 2 * N - 1
    grad = np.zeros(N)
    for k in range(L):
        if w[k] < 1e-10:
            continue
        lo = max(0, N - 1 - k)
        hi = min(N - 1, 2 * (N - 1) - k)
        j1_lo = max(0, lo + k - N + 1)
        j1_hi = min(N - 1, hi + k - N + 1)
        if j1_lo <= j1_hi:
            idx = np.arange(j1_lo, j1_hi + 1)
            grad[idx] += w[k] * (1.0 - h[idx - k + N - 1])
        j2_lo = max(0, lo)
        j2_hi = min(N - 1, hi)
        if j2_lo <= j2_hi:
            idx = np.arange(j2_lo, j2_hi + 1)
            partner = idx + k - (N - 1)
            valid = (partner >= 0) & (partner < N)
            if np.any(valid):
                grad[idx[valid]] -= w[k] * h[partner[valid]]
    return grad

def sa_optimize(h_bin, deadline, n_peaks=6, n_per_peak=18, cool_rate=0.999, t_start=0.04, use_2swap=True, max_gap=8000):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    while time.time() < deadline and it < 200000:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.0008)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p, cand_q = [], []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp, bq = vi[mask], (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = min(8, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 5
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        accepted = dc < -1e-12 or (T > 0.0008 and np.random.random() < np.exp(-dc / (T + 1e-9)))
        do_2swap = False
        if accepted and use_2swap and len(cand_p) >= 2 and len(cand_q) >= 2 and it % 4 == 0:
            p2_list = [x for x in cand_p if x != p]
            q2_list = [x for x in cand_q if x != q]
            n_2 = min(3, len(p2_list), len(q2_list))
            if n_2 >= 1:
                best_2c = best_new
                bp2, bq2 = -1, -1
                for _ in range(n_2):
                    p2 = p2_list[np.random.randint(len(p2_list))]
                    q2 = q2_list[np.random.randint(len(q2_list))]
                    c2 = corr + deltas[p] + deltas[p2] - deltas[q] - deltas[q2]
                    mc = float(c2.max())
                    if mc < best_2c:
                        best_2c = mc
                        bp2, bq2 = p2, q2
                if bp2 >= 0 and best_2c < best_new:
                    do_2swap = True
                    p2, q2, best_new, dc = bp2, bq2, best_2c, best_2c - best_c
        if accepted:
            if do_2swap:
                h[p] = 0; h[q] = 1; h[p2] = 0; h[q2] = 1
                for pp, qq in [(p, q), (p2, q2)]:
                    s[pp] = 1.0; s[qq] = -1.0
                    deltas[pp, :] = 0; deltas[pp, pp:pp+N] = s
                    deltas[qq, :] = 0; deltas[qq, qq:qq+N] = s
                corr = fft_corr(h)
            else:
                h[p], h[q] = 0.0, 1.0
                corr += deltas[p].copy() - deltas[q].copy()
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0; deltas[p, p:p+N] = s
                deltas[q, :] = 0; deltas[q, q:q+N] = s
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        if no_impr > max_gap:
            break
    return best_h, best_c

def continuous_opt(h_in, N, deadline, lr_init=0.02, T_init=30, T_min=8):
    h = np.clip(h_in.astype(float).copy(), 0, 1)
    adj = (h.sum() - N / 2.0) / N
    h = np.clip(h - adj, 0, 1)
    m = np.zeros(N)
    beta = 0.8
    best_c = np.inf
    best_h = h.copy()
    lr = lr_init
    T = T_init
    it = 0
    while time.time() < deadline:
        y = h + m
        y = np.clip(y, 0, 1)
        adj = (y.sum() - N / 2.0) / N
        y = np.clip(y - adj, 0, 1)
        corr = fft_corr(y)
        pk = float(corr.max())
        w = np.exp(np.clip(T * (corr - pk), -50, 0))
        ws = w.sum()
        if ws > 0:
            w /= ws
        g = compute_grad(y, w, N)
        h_new = np.clip(y - lr * g, 0, 1)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0, 1)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        ac = float(fft_corr(h).max()) * (2.0 / N)
        if ac < best_c:
            best_c = ac
            best_h = h.copy()
        if it % 150 == 0:
            lr = max(0.001, lr * 0.975)
            beta = min(0.93, 0.78 + it * 0.00015)
            T = max(T_min, T * 0.995)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)
    d1 = min(deadline - 320, t0 + 250)
    h1 = sa_optimize(h_bin, d1, n_peaks=7, n_per_peak=20, cool_rate=0.999, t_start=0.045, use_2swap=True)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()
    if time.time() < deadline - 200:
        d2 = min(deadline - 160, time.time() + 140)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hn, c2 = continuous_opt(hc, N0, d2, lr_init=0.018, T_init=25, T_min=6)
        if c2 < best_c5:
            best_c5, best_h = c2, hn.copy()
    if time.time() < deadline - 180:
        h3 = make_binary(best_h, N0)
        d3 = min(deadline - 90, time.time() + 200)
        h3b = sa_optimize(h3, d3, n_peaks=9, n_per_peak=25, cool_rate=0.9994, t_start=0.02, use_2swap=True, max_gap=6000)
        c3 = compute_c5(h3b, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3b.copy()
    if time.time() < deadline - 100:
        d4 = min(deadline - 50, time.time() + 80)
        hc2 = best_h.astype(float) + np.random.randn(N0) * 0.02
        hn2, c4 = continuous_opt(hc2, N0, d4, lr_init=0.01, T_init=15, T_min=4)
        if c4 < best_c5:
            best_c5, best_h = c4, hn2.copy()
    if time.time() < deadline - 60:
        for N_try in [384, 400, 420]:
            if time.time() > deadline - 50:
                break
            if N_try % 2 != 0:
                N_try += 1
            xo = np.linspace(0, 2, len(best_h), endpoint=False)
            xn = np.linspace(0, 2, N_try, endpoint=False)
            hi = np.interp(xn, xo, best_h)
            hb = make_binary(hi, N_try)
            sub_dl = min(deadline - 25, time.time() + 45)
            hr = sa_optimize(hb, sub_dl, n_peaks=6, n_per_peak=15, cool_rate=0.999, t_start=0.035, use_2swap=True)
            cr = compute_c5(hr, N_try)
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()
    if time.time() < deadline - 5:
        Ny = len(best_h)
        h5 = make_binary(best_h, Ny)
        h5b = sa_optimize(h5, deadline - 2, n_peaks=10, n_per_peak=30, cool_rate=0.9997, t_start=0.012, use_2swap=True, max_gap=5000)
        c5f = compute_c5(h5b, Ny)
        if c5f < best_c5:
            best_c5, best_h = c5f, h5b
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
