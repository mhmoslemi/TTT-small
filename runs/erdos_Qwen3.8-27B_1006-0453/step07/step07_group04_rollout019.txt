Let me think about the gradient more carefully.

The objective is: C(h) = max_k corr(h, 1-h)[k] * dx

where corr(h, 1-h)[k] = sum_i h[i] * (1-h[i+k]) for appropriate indices.

Actually, let me re-read the correlation definition. `np.correlate(h, 1-h, mode='full')` computes:
c[k] = sum_i h[i] * (1-h[i - (k - (N-1))])

For the FFT-based version:
- H = fft(h, L)
- G = fft(flip(1-h), L)
- result = ifft(H * G)[:L]

This gives circular correlation, but since we pad to L = 2N-1, it's equivalent to linear correlation.

The gradient of corr[k] with respect to h[j]:
- corr[k] = sum_i h[i] * (1 - h[i + (k - N + 1)])
- For a given lag L = k - (N-1):
  - corr[k] = sum_i h[i] * (1 - h[i+L]) (with appropriate bounds)

The derivative of corr[k] w.r.t. h[j]:
- From the h[i] term: if i = j, contributes (1 - h[j+L])
- From the (1-h[i+L]) term: if i+L = j, i.e., i = j-L, contributes -h[j-L]

So grad_j = (1 - h[j+L]) * 1[j+L in range] - h[j-L] * 1[j-L in range]

For the max objective, we need gradient only from active (top) lags.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box(h, target):
    # Project onto {0<=h<=1} and sum(h)=target
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_max_corr(h, corr, k_top=30):
    """Gradient of max_k corr[k] * dx w.r.t. h, using top-k lags."""
    N = len(h)
    dx = 2.0 / N
    top = np.argpartition(corr, -k_top)[-k_top:]
    center = N - 1
    grad = np.zeros(N)
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            # corr[center+L] = sum_{i=0}^{N-1-L} h[i]*(1-h[i+L])
            # d/dh[j]: from h[j]*(1-h[j+L]) -> (1-h[j+L]) for j < N-L
            #          from h[j-L]*(1-h[j]) -> -h[j-L] for j >= L
            if L < N:
                grad[:N-L] += (1.0 - h[L:])
                grad[L:] -= h[:N-L]
        else:
            M = -L  # M > 0
            # corr[center-M] = sum_{i=M}^{N-1} h[i]*(1-h[i-M])
            # d/dh[j]: from h[j]*(1-h[j-M]) -> (1-h[j-M]) for j >= M
            #          from h[j+M]*(1-h[j]) -> -h[j+M] for j < N-M
            if M < N:
                grad[M:] += (1.0 - h[:N-M])
                grad[:N-M] -= h[M:]
    return grad * dx / k_top

def adam_continuous(h0, N, deadline, k_top=40, lr=0.05, bs=1.0, b2=0.999, eps=1e-8, smooth_T=8.0):
    """Adam optimizer with soft-max smoothing for the max-correlation objective."""
    h = h0.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    plateau = 0
    # Adaptive k_top
    k_cur = k_top
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        c_val = float(np.max(corr))
        if c_val < best_c - 1e-13:
            best_c = c_val
            best_h = h.copy()
            plateau = 0
        else:
            plateau += 1
            if plateau > 500:
                # Increase k_top to smooth out
                k_cur = min(100, k_cur + 5)
                plateau = 0
                if plateau > 2000:
                    break
        
        # Use soft-max weights for smooth gradient
        # Or just use top-k
        top = np.argpartition(corr, -k_cur)[-k_cur:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                if L < N:
                    grad[:N-L] += (1.0 - h[L:])
                    grad[L:] -= h[:N-L]
            else:
                M = -L
                if M < N:
                    grad[M:] += (1.0 - h[:N-M])
                    grad[:N-M] -= h[M:]
        
        # Adam update
        alpha = lr / (1.0 + t * 0.001)  # Slight decay
        m = bs * m + (1 - bs) * grad
        v = b2 * v + (1 - b2) * grad**2
        m_hat = m / (1 - bs**t)
        v_hat = v / (1 - b2**t)
        h_new = project_box(h - alpha * m_hat / (np.sqrt(v_hat) + eps), target)
        h = h_new
    return best_h, best_c * dx

def greedy_binary_fast(h, N, deadline, max_sweeps=2000, sf=1.0):

        zeros = np.where(h < 0.5)[0]

            break

        N2 = 2 * N - 1
        # D[p, l] = delta of corr[l] when moving 1 from position p
        # For lag L (where l = L + N - 1):
        #   If L == 0: delta = -(1 - 2*h[p]) ... actually let's compute directly
        # Moving h[p] from 1 to 0 and h[q] from 0 to 1:
        #   delta_corr[l] = -D[p,l] + D_neg[q,l]
        # where D[p,l] is contribution of h[p]=1 to corr at lag l
        # and D_neg[q,l] is contribution of h[q]=1 to corr at lag l

        # corr_new[l] = corr[l] - (contribution of h[p]) + (contribution of h[q])

        #   If s > 0: corr includes h[p]*(1-h[p+s]) -> removing h[p] changes corr by -(1-h[p+s]) if p+s<N, and -h[p-s]*(1-h[p]) ... 
        # This is getting complex. Let me use the D matrix approach from before but more carefully.
        
        # For shift s (can be negative, zero, or positive):
        # corr_c[s] = sum_i h[i] * (1 - h[i+s]) where i and i+s must be in [0, N-1]
        # 
        # If we change h[p]: 1->0, the change in corr_c[s] is:
        #   - (1-h[p+s]) * 1[p+s in range] + h[p-s] * 1[p-s in range] * (-1)
        #   Wait, let me be more careful.
        #   corr_c[s] = sum_i h_i (1-h_{i+s})
        #   = sum_i h_i - sum_i h_i h_{i+s}
        #   d(corr_c[s])/d(h_p) = 1 - h_{p+s} - h_{p-s} (with boundary conditions)
        #   Actually: sum_i h_i (1-h_{i+s})
        #   = sum_i h_i - sum_i h_i * h_{i+s}
        #   d/dh_p of sum_i h_i = 1
        #   d/dh_p of sum_i h_i * h_{i+s} = h_{p+s} (from i=p term) + h_{p-s} (from i=p-s term, if in range)
        #   So d(corr_c[s])/d(h_p) = 1 - h_{p+s}*1[p+s in [0,N-1]] - h_{p-s}*1[p-s in [0,N-1]]
        #   (with h_{p+s} = 0 if out of range)
        #
        # So if we set h[p]: 1->0, delta_corr[s] = -d(corr_c[s])/d(h_p) = -(1 - h_{p+s} - h_{p-s})
        # If we set h[q]: 0->1, delta_corr[s] = +d(corr_c[s])/d(h_q) = +(1 - h_{q+s} - h_{q-s})
        #
        # Combined: new_corr[s] = corr[s] - (1 - h[p+s] - h[p-s]) + (1 - h[q+s] - h[q-s])
        #   = corr[s] - 2 + h[p+s] + h[p-s] - h[q+s] - h[q-s]
        #   (with out-of-range terms being 0)
        
        # For efficiency, precompute for each p in ones and q in zeros:
        # delta[p,q,s] = -(1-h[p+s]-h[p-s]) + (1-h[q+s]-h[q-s])
        #   = h[p+s]+h[p-s]-h[q+s]-h[q-s]
        #   (out of range = 0)
        
        # new_corr[s] = corr[s] + h[p+s]+h[p-s]-h[q+s]-h[q-s]
        # new_max = max_s (corr[s] + h[p+s]+h[p-s]-h[q+s]-h[q-s])
        
        # For a fixed p, define A_p[s] = h[p+s] + h[p-s] (out of range = 0)
        # For a fixed q, define B_q[s] = h[q+s] + h[q-s] (out of range = 0)
        # new_corr[s] = corr[s] + A_p[s] - B_q[s]
        # new_max = max_s (corr[s] + A_p[s] - B_q[s])
        
        # To find best (p,q): for each p, find q that minimizes max_s(corr[s] + A_p[s] - B_q[s])
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Sample subset if too large
        if sf < 1.0:
            n_s = max(20, int(n_ones * sf))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
            
        best_new_P = P + 1.0
        best_pq = None
        
        # Precompute B for all zeros: B_q[s] = h[q+s] + h[q-s]
        # This is an N_zeros x N2 matrix - might be too large
        # For N=760, N2=1519, N_zeros≈380: 380*1519 ≈ 577K - OK

        shifts = np.arange(N2) - (N - 1)  # -N+1, ..., N-1
        # B[q, s_idx] = h[q + shifts[s_idx]] + h[q - shifts[s_idx]] (out of range = 0)
        
        q_idx = zeros  # indices
        # B matrix
        pos1 = q_idx[:, None] + shifts[None, :]
        pos2 = q_idx[:, None] - shifts[None, :]
        B = np.where((pos1 >= 0) & (pos1 < N), h[pos1.clip(0, N-1)], 0.0) + \
            np.where((pos2 >= 0) & (pos2 < N), h[pos2.clip(0, N-1)], 0.0)

            posA2 = p - shifts
            A = np.where((posA1 >= 0) & (posA1 < N), h[np.clip(posA1, 0, N-1)], 0.0) + \
                np.where((posA2 >= 0) & (posA2 < N), h[np.clip(posA2, 0, N-1)], 0.0)
            base = corr + A  # (N2,)
            # new_corr for each q: base - B[q, :]
            # new_max for each q: max(base - B[q, :])

def sa_binary(h, N, deadline, T0=4.0, Tcool=0.994, T_min=0.03, sf=1.0, kick_every=200, kick_size=5):
    """Simulated annealing on binary solutions."""
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    shifts = np.arange(2*N-1) - (N-1)
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            continue
        
        # Best move
        n_ones = len(ones)
        if sf < 1.0:
            n_s = max(20, int(n_ones * sf))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        
        q_idx = zeros
        pos1 = q_idx[:, None] + shifts[None, :]
        pos2 = q_idx[:, None] - shifts[None, :]
        B = np.where((pos1 >= 0) & (pos1 < N), h[pos1.clip(0, N-1)], 0.0) + \
            np.where((pos2 >= 0) & (pos2 < N), h[pos2.clip(0, N-1)], 0.0)
        
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            posA1 = p + shifts
            posA2 = p - shifts
            A = np.where((posA1 >= 0) & (posA1 < N), h[np.clip(posA1, 0, N-1)], 0.0) + \
                np.where((posA2 >= 0) & (posA2 < N), h[np.clip(posA2, 0, N-1)], 0.0)
            base = corr + A
            cand_max = (base[None, :] - B).max(axis=1)
            mi = int(np.argmin(cand_max))
            if cand_max[mi] < best_new_P:
                best_new_P = float(cand_max[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            break
        
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    
    return best_h, best_c * (2.0/N)

def ensure_binary(h, N):
    """Round to binary with correct sum."""
    nh = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, corr, k_top):
    N = len(h)
    top = np.argpartition(corr, -k_top)[-k_top:]
    center = N - 1
    grad = np.zeros(N)
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[:N - L] += (1.0 - h[L:])
            grad[L:] -= h[:N - L]
        else:
            M = -L
            if M < N:
                grad[M:] += (1.0 - h[:N - M])
                grad[:N - M] -= h[M:]
    return grad / k_top

def adam_cont(h0, N, deadline, k_top=50, lr=0.04, b1=0.9, b2=0.999, eps=1e-8):
    h = h0.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    plat = 0
    kc = k_top
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        cv = float(np.max(corr))
        if cv < best_c - 1e-13:
            best_c = cv
            best_h = h.copy()
            plat = 0
        else:
            plat += 1
            if plat == 200:
                kc = min(150, kc + 10)
                plat = 0
            if plat > 500:
                h += np.random.randn(N) * 0.02
                h = project_box(h, target)
                plat = 0
        grad = compute_grad(h, corr, kc)
        a = lr * (1.0 / (1.0 + 0.002 * t))
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = project_box(h - a * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c * (2.0 / N)

def build_B(h, zeros, shifts, N):
    q = zeros[:, None]
    p1 = q + shifts[None, :]
    p2 = q - shifts[None, :]
    B = np.where((p1 >= 0) & (p1 < N), h[p1.clip(0, N-1)], 0.0) + \
        np.where((p2 >= 0) & (p2 < N), h[p2.clip(0, N-1)], 0.0)
    return B

def build_A(h, p, shifts, N):
    p1 = p + shifts
    p2 = p - shifts
    return np.where((p1 >= 0) & (p1 < N), h[p1.clip(0, N-1)], 0.0) + \
        np.where((p2 >= 0) & (p2 < N), h[p2.clip(0, N-1)], 0.0)

def greedy_bin(h, N, deadline, sf=1.0, max_stall=80):
    h = h.astype(float).copy()
    shifts = np.arange(2*N-1) - (N-1)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P; best_h = h.copy(); stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        B = build_B(h, zeros, shifts, N)
        if sf < 1.0:
            ns = max(20, int(len(ones) * sf))
            p_list = np.random.choice(ones, ns, replace=False)
        else:
            p_list = ones
        best_np = P + 1.0
        best_pq = None
        for p in p_list:
            A = build_A(h, p, shifts, N)
            base = c + A
            cm = (base[None, :] - B).max(axis=1)
            mi = int(np.argmin(cm))
            if cm[mi] < best_np:
                best_np = float(cm[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def ensure_bin(h, N):
    nh = N // 2
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    rng = np.random.RandomState(99)
    if n1 > nh:
        idx = rng.choice(np.where(hb == 1)[0], n1 - nh, replace=False)
        hb[idx] = 0
    else:
        idx = rng.choice(np.where(hb == 0)[0], nh - n1, replace=False)
        hb[idx] = 1
    return hb

def to_N(h, N):
    if len(h) == N:
        return h.copy()
    x = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, N, endpoint=False)
    return project_box(np.interp(xn, x, h), N / 2.0)
