import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_bb(h, N, deadline, alpha0=0.05, beta=0.85, atol=1e-5, max_iter=6000):
    """Nesterov-accelerated subgradient with Barzilai-Bordeaux step size."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    it = 0
    last_h = h.copy()
    last_g = None
    alpha = alpha0
    y = h.copy()
    
    while time.time() < deadline and it < max_iter:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        
        ctr = N - 1
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        
        # BB step size
        if last_g is not None:
            s = h - last_h
            g_diff = grad - last_g
            ss = np.dot(s, s)
            sg = np.dot(s, g_diff)
            gg = np.dot(g_diff, g_diff)
            if ss > 1e-12 and gg > 1e-12:
                if abs(sg) > 1e-12:
                    alpha_bb1 = ss / abs(sg)
                else:
                    alpha_bb1 = alpha
                alpha_bb2 = sg / gg if gg > 1e-12 else alpha
                alpha = 0.5 * (alpha_bb1 + alpha_bb2)

        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h_old)
        h = h_new

        last_g = grad.copy()
        
        it += 1
        if it % 300 == 0:
            beta = min(0.98, 0.85 + 0.0003 * it)
            if it % 1000 == 0:
                m = np.zeros(N)  # restart momentum
    
    return best_h, best_c * (2.0 / N)

def smooth_lbfgs(h, N, deadline, T=20.0, max_outer=3):

def bin_search_hill(h, N, deadline, max_iters=3000, ntri=30, block_sizes=None):

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    corr = fft_corr(h)
    P = float(np.max(corr))
    
    if block_sizes is None:
        block_sizes = [1, 1, 1, 1, 2, 2, 3]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        for _ in range(ntri):
            # Pick a move type
            move_type = np.random.choice(block_sizes)
            if move_type == 1:
                i = int(np.random.randint(len(ones)))
                j = int(np.random.randint(len(zeros)))
                oi, zi = int(ones[i]), int(zeros[j])
                h[oi], h[zi] = 0.0, 1.0
                nP = float(np.max(fft_corr(h)))
                h[oi], h[zi] = 1.0, 0.0
                d = P - nP
                if d > bd:
                    bd = d
                    best_move = (oi, zi, 1)
            else:
                # Block swap: swap a block of 'move_type' ones with same number of zeros
                if len(ones) < move_type or len(zeros) < move_type:
                    continue
                block_idx = np.random.choice(len(ones) - move_type + 1, 1)[0]
                ones_block = np.sort(np.random.choice(ones, move_type, replace=False))
                zeros_block = np.sort(np.random.choice(zeros, move_type, replace=False))
                # Check blocks are contiguous? No, just swap the selected positions
                h[ones_block] = 0.0
                h[zeros_block] = 1.0
                nP = float(np.max(fft_corr(h)))
                h[ones_block] = 1.0
                h[zeros_block] = 0.0
                d = P - nP
                if d > bd:
                    bd = d
                    best_move = (ones_block, zeros_block, move_type)
        
        if best_move is not None:
            oi, zi, mt = best_move
            if mt == 1:
                h[oi], h[zi] = 0.0, 1.0

                h[oi] = 0.0
                h[zi] = 1.0
            stall = 0
        else:
            stall += 1
            # Reload correlation
            corr = fft_corr(h)
            P = float(np.max(corr))
        if best_move is None:
            corr = fft_corr(h)
            P = float(np.max(corr))
    
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:N - L] = 1.0 - h[L:N]
        g[L:] -= h[:N - L]
    else:
        M = -L
        g[M:] = 1.0 - h[:N - M]
        g[:N - M] -= h[M:]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_bb(h, N, deadline, alpha0=0.04, beta=0.88, atol=1e-5, max_iter=6000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    it = 0
    prev_h = h.copy()
    prev_g = None
    alpha = alpha0
    while time.time() < deadline and it < max_iter:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_c - 1e-12:
            best_c = pk
            best_h = h.copy()
        ctr = N - 1
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        if prev_g is not None:
            s = h - prev_h
            gd = grad - prev_g
            ss = np.dot(s, s)
            sg = np.dot(s, gd)
            gg = np.dot(gd, gd)
            if ss > 1e-12 and gg > 1e-12 and abs(sg) > 1e-12:
                alpha = 0.5 * (ss / abs(sg) + sg / gg)
                alpha = np.clip(alpha, 1e-4, 0.4)
        h_old = h.copy()
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h_old)
        h = h_new
        prev_h = h_old
        prev_g = grad.copy()
        it += 1
        if it % 400 == 0:
            beta = min(0.99, 0.88 + 0.00025 * it)
        if it % 1200 == 0:
            m = np.zeros(N)
    return best_h, best_c * (2.0 / N)

def smooth_lbfgs(h, N, deadline, T=22.0, max_outer=3):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_c = compute_c5(h, N)
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        def _og(x):
            x = np.clip(x, 0, 1)
            c = fft_corr(x)
            mx = c.max()
            e = np.exp(T * (c - mx))
            w = e / e.sum()
            val = mx + np.log(e.sum()) / T
            g = np.zeros(N)
            ctr = N - 1
            for i in range(len(c)):
                if w[i] > 1e-9:
                    g += w[i] * lin_grad(int(i - ctr), x, N)
            s = x.sum() - target
            val += 0.5 * s * s
            g += s * np.ones(N)
            return val, g
        res = sp_min(_og, h, jac=True, method='L-BFGS-B',
                     bounds=[(0, 1)] * N,
                     options={'maxiter': 100, 'maxfun': 300, 'ftol': 1e-14})
        h = proj01(res.x, target)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def bin_opt(h, N, deadline, max_iters=4000, ntri=28, use_blocks=True):

    nh = N // 2
    n1 = int(h.sum())

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()

    P = best_c
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bd, move = 0.0, None
        for _ in range(ntri):

                if len(ones) < sz or len(zeros) < sz:

                zb = np.random.choice(zeros, sz, replace=False)
                h[ob] = 0.0
                h[zb] = 1.0
                nP = float(np.max(fft_corr(h)))
                h[ob] = 1.0
                h[zb] = 0.0
                d = P - nP
                if d > bd:
                    bd = d
                    move = (ob, zb)
            else:
                i = int(np.random.randint(len(ones)))
                j = int(np.random.randint(len(zeros)))
                oi, zi = int(ones[i]), int(zeros[j])
                h[oi], h[zi] = 0.0, 1.0
                nP = float(np.max(fft_corr(h)))
                h[oi], h[zi] = 1.0, 0.0
                d = P - nP
                if d > bd:
                    bd = d
                    move = ((oi,), (zi,))
        if move is not None:
            h[move[0]] = 0.0
            h[move[1]] = 1.0
            stall = 0
        else:
            stall += 1
        P = float(np.max(fft_corr(h)))
    return best_h, best_c

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    P = best_c
    decay = (t1 / t0) ** (1.0 / max(1, max_iters))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        temp = t0 * (decay ** it)
        if temp < 1e-7:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(ones[np.random.randint(len(ones))])
        zi = int(zeros[np.random.randint(len(zeros))])
        h[oi], h[zi] = 0.0, 1.0
        nP = float(np.max(fft_corr(h)))
        d = P - nP
        if d > 0 or np.random.rand() < np.exp(d / temp):

            if nP < best_c - 1e-12:
