    best_pk = float(np.max(c))
    m = np.zeros(N)

    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()

        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx_top)):
            s = int(idx_top[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]

            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0001)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_idx = np.arange(N)[:, None]
    s_idx = shifts[None, :]
    pm = p_idx - s_idx
    pp = p_idx + s_idx
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def update_D_rows(D, h, N, positions, shifts):
    for pos in positions:
        pm_row = pos - shifts
        pp_row = pos + shifts
        vm_row = (pm_row >= 0) & (pm_row < N)
        vp_row = (pp_row >= 0) & (pp_row < N)
        pm_c_row = np.clip(pm_row, 0, N - 1)
        pp_c_row = np.clip(pp_row, 0, N - 1)
        hm_row = np.where(vm_row, h[pm_c_row], 0.0)
        hp_row = np.where(vp_row, h[pp_c_row], 0.0)
        D[pos, :] = -(1.0 - hm_row) * vm_row.astype(float) + hp_row * vp_row.astype(float)

def sa_multiswap(h_init, N, deadline, t0=1.0, t_end=0.001, max_swaps=2):
    h = h_init.copy()
    c = fft_corr(h)
    P = float(c.max())
    best_h = h.copy()
    best_P = P
    D = compute_D(h, N)
    shifts = np.arange(2 * N - 1) - (N - 1)
    rng = np.random.RandomState(7)
    t_start = time.time()
    n_iters = max(1, int((deadline - t_start) / 0.0003))
    cool = (t_end / t0) ** (1.0 / n_iters)
    t = t0
    it = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < max_swaps or len(zeros) < max_swaps:
            break
        n_s = rng.randint(1, max_swaps + 1)
        pp = rng.choice(ones, n_s, replace=False)
        qq = rng.choice(zeros, n_s, replace=False)
        new_c = c.copy()
        for p in pp:
            new_c -= D[p, :]
        for q in qq:
            new_c += D[q, :]
        new_P = float(new_c.max())
        delta = new_P - P
        accept = delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t))
        if accept:
            for p in pp:
                h[p] = 0.0
            for q in qq:
                h[q] = 1.0
            update_D_rows(D, h, N, list(pp) + list(qq), shifts)
            c = new_c
            P = new_P
            if P < best_P - 1e-14:
                best_P = P
                best_h = h.copy()
        t *= cool
        it += 1
    return best_h, best_P * (2.0 / N)

def nesterov_opt(h_init, N, deadline, k=50, T=100.0, lr0=0.012):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    c = fft_corr(h)
    best_pk = float(np.max(c))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx_top = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx_top]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx_top)):
            s = int(idx_top[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.994
            beta = min(0.98, 0.85 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)
