import numpy as np
import time

def fft_corr_full(h):
    """Full correlation of h with (1-h) using FFT."""
    N = len(h)
    L = 2 * N - 1
    Lp = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, Lp)
    Rb = np.fft.rfft(1.0 - h[::-1], Lp)
    return np.fft.irfft(H * Rb, Lp)[:L]

def compute_c5(h):
    N = len(h)
    corr = fft_corr_full(h)
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_gradient(h, eps, N):
    """Compute gradient of smooth max of correlation values."""
    corr = fft_corr_full(h)  # length 2N-1
    # Softmax weights
    corr_max = corr.max()
    w = np.exp((corr - corr_max) / eps)
    w /= w.sum()
    
    # Compute gradient via O(N^2) broadcasting
    j = np.arange(N)[:, None]  # (N, 1)
    s = np.arange(-(N-1), N)[None, :]  # (1, 2N-1) shifts
    W_idx = s + (N - 1)  # index into corr/w
    
    # g_plus[j] = sum_s w[s+N-1] * h[j+s] * I(0<=j+s<N)
    valid_plus = (j + s >= 0) & (j + s < N)
    h_plus = np.where(valid_plus, h[np.clip(j + s, 0, N-1)], 0.0)
    g_plus = (w[W_idx] * h_plus).sum(axis=1)
    
    # g_minus[j] = sum_s w[s+N-1] * h[j-s] * I(0<=j-s<N)
    valid_minus = (j - s >= 0) & (j - s < N)
    h_minus = np.where(valid_minus, h[np.clip(j - s, 0, N-1)], 0.0)
    g_minus = (w[W_idx] * h_minus).sum(axis=1)
    
    # w_sum[j] = sum_s w[s+N-1] * I(0<=j+s<N)
    w_sum = (w[W_idx] * valid_plus).sum(axis=1)
    
    grad = w_sum - g_plus - g_minus
    return grad, corr

def fista_optimize(h0, N, deadline, eps0=10.0, eps_min=0.5, n_iters=5000):
    """FISTA with smoothed max for continuous optimization."""
    h = h0.astype(float).copy()
    prev_h = h.copy()
    y = h.copy()
    t = 1.0
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    eps = eps0
    eps_decay = (eps_min / eps0) ** (1.0 / max(n_iters, 1))
    alpha = 2.0  # initial step size
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Update epsilon (anneal)
        if it % 50 == 0:
            eps = max(eps_min, eps * eps_decay ** 50)
        
        # Compute gradient
        grad, corr = smooth_max_gradient(y, eps, N)
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            alpha = 2.0
        else:
            alpha = min(alpha * 1.05, 4.0 / (gnorm + 1e-8))
        
        # FISTA update
        h_new = project_sum(y - alpha * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        h, prev_h, t = h_new, h_new, t_new
        
        # Track best
        c5 = float(np.max(fft_corr_full(h))) * (2.0 / N)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_sa(h, N, deadline, max_sweeps=3000, T0=3.0, Tcool=0.994, T_min=0.05, sf=1.0, kick_every=150, kick_size=4):
    """Binary simulated annealing with swap moves."""
    h = h.astype(float).copy()
    # Ensure binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Ensure sum
    target = N // 2
    s = int(h.sum())
    if s != target:
        if s > target:
            ones = np.where(h == 1)[0]
            np.random.shuffle(ones)
            h[ones[:s-target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            np.random.shuffle(zeros)
            h[zeros[:target-s]] = 1
    
    best_c = float(np.max(fft_corr_full(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr_full(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            stall = 0
            T = max(T0 * 0.5, T_min * 3)
            c = fft_corr_full(h)
            P = float(c.max())
            continue
        
        # Compute swap gradient efficiently
        # For each one p and zero q, the change in corr is:
        # delta_c[k] = -D_p[k] + D_q[k] where D_p[k] is the contribution of position p
        # Actually: if h[p] goes 1->0, corr changes by -(1-h[p+s]) at shift s
        # and -h[p-s] at shift -s (for the (1-h) factor)
        # If h[q] goes 0->1, corr changes by +(1-h[q+s]) at shift s
        # and +h[q-s] at shift -s
        
        # Simplified: compute the full delta for the best swap
        # Use the same O(N^2) approach but for finding best pair
        
        D = np.zeros((N, 2*N-1))
        # For each position i, D[i, k] = change in corr[k] if h[i] flips from 1 to 0
        # = -(1-h[i+s_k])*I - h[i-s_k]*I where s_k = k-(N-1)
        j = np.arange(N)[:, None]
        s = np.arange(-(N-1), N)[None, :]
        W_idx = s + (N - 1)
        
        valid_plus = (j + s >= 0) & (j + s < N)
        h_plus = np.where(valid_plus, h[np.clip(j + s, 0, N-1)], 0.0)
        D_term1 = (1.0 - h_plus) * valid_plus  # (N, 2N-1)
        
        valid_minus = (j - s >= 0) & (j - s < N)
        h_minus = np.where(valid_minus, h[np.clip(j - s, 0, N-1)], 0.0)
        D_term2 = h_minus * valid_minus  # (N, 2N-1)
        
        D = -(D_term1 + D_term2)  # change in corr when h[i] goes 1->0
        # When h[i] goes 0->1: change is (1-h[i+s])*I + h[i-s]*I = -D
        
        Dz = -D[zeros, :]  # change in corr when h[q] goes 0->1
        # n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, max(30, int(len(ones)*sf)), replace=False)
        
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            # New corr = c + D[p] + Dz[q] for various q
            # D[p] is the change from p going 1->0
            # Dz[q] is the change from q going 0->1
            base = c + D[p, :]  # corr after p->0
            cand = base[None, :] + Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            break
        delta = best_new_P - P
        accept = delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))
        if accept:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    
    return best_h, best_c * (2.0 / N)

import numpy as np
import time

def fft_corr_full(h):
    N = len(h)
    L = 2 * N - 1
    Lp = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, Lp)
    Rb = np.fft.rfft(1.0 - h[::-1], Lp)
    return np.fft.irfft(H * Rb, Lp)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr_full(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, eps, N):
    corr = fft_corr_full(h)
    cm = corr.max()
    w = np.exp((corr - cm) / eps)
    w /= w.sum()
    j = np.arange(N)[:, None]
    s = np.arange(-(N-1), N)[None, :]
    Wi = s + (N - 1)
    vp = (j + s >= 0) & (j + s < N)
    hm = (j - s >= 0) & (j - s < N)
    hp = np.where(vp, h[np.clip(j + s, 0, N-1)], 0.0)
    hmn = np.where(hm, h[np.clip(j - s, 0, N-1)], 0.0)
    wv = w[Wi]
    gp = (wv * hp).sum(axis=1)
    gm = (wv * hmn).sum(axis=1)
    wsum = (wv * vp).sum(axis=1)
    return wsum - gp - gm, corr

def compute_D(h, N):
    j = np.arange(N)[:, None]
    s = np.arange(-(N-1), N)[None, :]
    vp = (j + s >= 0) & (j + s < N)
    hm = (j - s >= 0) & (j - s < N)
    hp = np.where(vp, h[np.clip(j + s, 0, N-1)], 0.0)
    hmn = np.where(hm, h[np.clip(j - s, 0, N-1)], 0.0)
    return -((1.0 - hp) * vp + hmn * hm)

        d4 = min(deadline - 120, t0 + 640)
        h4, c4 = binary_sa(best_h, N0, d4, max_sweeps=3000, T0=2.5, Tcool=0.993,
                           T_min=0.03, sf=0.8, kick_every=100, kick_size=4)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Try N=480
    N_small = 480
    if time.time() < deadline - 120:
        hs = project_sum(upsample(best_h, len(best_h), N_small) + np.random.randn(N_small)*0.02, N_small/2.0)
        d5 = min(deadline - 80, t0 + 800)
        hs_r, c5 = fista_opt(hs, N_small, d5, eps0=8.0, eps_min=0.5, maxit=3000)
        if c5 < best_c5:
            best_c5, best_h = c5, hs_r.copy()

    # Phase 6: Final binary polish at best N
    Nfin = len(best_h)
    if time.time() < deadline - 5:
        hf = ensure_bin(best_h, Nfin)
        d6 = deadline - 2
        sf6 = 1.0 if Nfin <= 500 else 0.6
        hf_r, c6 = binary_sa(hf, Nfin, d6, max_sweeps=2000, T0=2.0, Tcool=0.992,
                             T_min=0.03, sf=sf6, kick_every=100, kick_size=4)
        if c6 < best_c5:
            best_c5, best_h = c6, hf_r.copy()

    # Phase 7: Quick FISTA polish
    if time.time() < deadline - 1:
        hp = project_sum(best_h + np.random.randn(len(best_h))*0.015, len(best_h)/2.0)
        hp_r, c7 = fista_opt(hp, len(best_h), deadline - 0.5, eps0=3.0, eps_min=0.2, maxit=500)
        if c7 < best_c5:
            best_c5, best_h = c7, hp_r.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
