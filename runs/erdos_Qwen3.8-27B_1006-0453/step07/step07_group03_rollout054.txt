    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final Nesterov refinement (8% budget)
    N_fin = len(best_h)
    d6 = deadline - 1
    h6_init = best_h + rng.randn(N_fin) * 0.025
    h6_init = project_sum(h6_init, N_fin / 2.0)
    h6, c5_6 = nesterov_opt(h6_init, N_fin, d6, k_init=30, k_max=70, lr0=0.007, beta0=0.93)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
