def fft_corr(h):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, N, top_k):
    """Compute subgradient using top_k largest correlations."""
    c = fft_corr(h)
    top = np.argpartition(c, -top_k)[-top_k:]
    weights = c[top] - c.min()
    weights = np.maximum(weights, 0)
    if weights.sum() < 1e-10:
        weights = np.ones(top_k) / top_k
    else:
        weights /= weights.sum()
    center = N - 1
    grad = np.zeros(N)
    for idx, ai in enumerate(top):
        L = int(ai - center)
        w = weights[idx]
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += w * (1.0 - h[M:])
                grad[M:] -= w * h[:N - M]
    return grad, c

def adam_continuous(h0, N, deadline, k=60, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    patience = 0
    while time.time() < deadline:
        t += 1
        grad, c = compute_grad(h, N, k)
        pk = float(np.max(c))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        mhat = m / (1 - beta1**t)
        vhat = v / (1 - beta2**t)
        h_new = project_sum(h - lr * mhat / (np.sqrt(vhat) + eps), target)
        h = h_new
        if patience > 500 and t > 1000:
            break
    return best_h, best_pk * dx

np.correlate(a, v, mode="full"):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    h = h.astype(float).copy()
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(777 + N)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def build_A(h, N):
    pos = np.arange(N)[:, None]
    shifts_arr = np.arange(2 * N - 1) - (N - 1)
    sv = shifts_arr[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    A = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    A += np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    A[:, N - 1] = 0.0
    return A

def greedy_binary(h0, N, deadline, max_sweeps=500, sf=1.0):
    h = h0.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(31337)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        A = build_A(h, N)
        n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
        best_new_P = P - 1e-14
        best_pq = None
        for p in p_list:
            B = c + A[p, :]
            cand = B[None, :] - A[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def binary_sa(h0, N, deadline, max_sweeps=8000, T0=4.0, Tcool=0.994, T_min=0.03,
              sf=1.0, kick_every=120, kick_size=5, tabu_len=40):
    h = h0.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    tabu = []
    tabu_set = set()
    rng = np.random.RandomState(42)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            tabu = []
            tabu_set = set()
            continue
        A = build_A(h, N)
        n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            B = c + A[p, :]
            cand = B[None, :] - A[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        p_use, q_use = best_pq
        mk = (min(p_use, q_use), max(p_use, q_use))
        if mk in tabu_set:
            p_use = int(ones[rng.randint(len(ones))])
            q_use = int(zeros[rng.randint(len(zeros))])
        accept = delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))
        if accept:
            h[p_use] = 0.0
            h[q_use] = 1.0
            tabu.append(mk)
            if len(tabu) > tabu_len:
                tabu_set.discard(tabu.pop(0))
            nmk = (min(p_use, q_use), max(p_use, q_use))
            tabu_set.add(nmk)
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def compute_subgrad(h, N, top_k):
    c = fft_corr(h)
    idx = np.argpartition(c, -top_k)[-top_k:]
    w = c[idx] - c.min()
    w = np.maximum(w, 0.0)
    ws = w.sum()
    if ws < 1e-12:
        w = np.ones(top_k) / top_k
    else:
        w /= ws
    center = N - 1
    grad = np.zeros(N)
    for i, ai in enumerate(idx):
        L = int(ai - center)
        wi = w[i]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad, c

def adam_cont(h0, N, deadline, k=50, lr=0.012, b1=0.85, b2=0.999, eps=1e-8, reseed_every=400):
    h = h0.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    patience = 0
    while time.time() < deadline:
        t += 1
        grad, c = compute_subgrad(h, N, k)
        pk = float(np.max(c))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if patience > 800:
            break
        if t % reseed_every == 0:
            v *= 0.5
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
