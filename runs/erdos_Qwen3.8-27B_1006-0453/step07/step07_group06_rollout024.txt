import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def block_local(h, N, deadline, max_iters=500, bsizes=(2,3,4,5,6), ntr=25, anneal=0.0):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h==1)[0], n1-nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h==0)[0], nh-n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    temp = anneal
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c, best_h, improved = P, h.copy(), True
        bd, bi, bj, bs = 0.0, -1, -1, 1
        for _ in range(ntr):
            bs = int(bsizes[np.random.randint(len(bsizes))])
            i = np.random.randint(0, N - bs + 1)
            j = np.random.randint(0, N - bs + 1)
            if abs(i - j) < bs:
                continue
            seg = h[i:i+bs].copy()
            h[i:i+bs] = h[j:j+bs]
            h[j:j+bs] = seg
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd, bi, bj, bs = d, i, j, bs
            h[j:j+bs] = h[i:i+bs]
            h[i:i+bs] = seg
        if bi >= 0:
            seg = h[bi:bi+bs].copy()
            h[bi:bi+bs] = h[bj:bj+bs]
            h[bj:bj+bs] = seg
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / max(it + 1, max_iters)))
    return best_h, best_c

def single_local(h, N, deadline, max_iters=3000, ntr=50, anneal=0.0):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())

    temp = anneal

        ones = np.where(h == 1)[0]

        bd, bi, bj = 0.0, -1, -1

            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            h[i], h[j] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd, bi, bj = d, i, j
            h[i], h[j] = 1.0, 0.0
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / max(it + 1, max_iters)))
    return best_h, best_c

def lbfgs_opt(h, N, deadline, T=30.0, nouter=4, lam0=5.0):
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_and_grad(h, T, target, 0.0)[0]
    lam = lam0
    for _ in range(nouter):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_and_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0,1)]*N,
                     options={'maxiter':150,'maxfun':400,'ftol':1e-14})
        h = proj01(res.x, target)
        v = smooth_obj_and_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v, best_h = v, h.copy()
        lam *= 2.5
    return best_h, best_v * (2.0 / N)

def structured_start(N, rng, kind=0):
    h = np.zeros(N)
    if kind == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        x = np.arange(N) * phi
        h[(x - np.floor(x)) > 0.5] = 1
    elif kind == 1:
        i = 0
        while i < N:
            b1, b0 = int(rng.integers(4,10)), int(rng.integers(4,10))
            h[i:min(i+b1, N)] = 1
            i += b1
            h[i:min(i+b0, N)] = 0
            i += b0
    else:
        period = int(N / (1 + rng.uniform(1.0, 2.5)))
        for p in range(period):
            pass
        pattern = (np.arange(N) % (2*period)) < period
        h[pattern] = 1
        h = rng.permutation(h) * 0 + h
        h[rng.permutation(N)] = pattern[rng.permutation(N)]
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[rng.choice(np.where(h==1)[0], n1-nh, replace=False)] = 0
    elif n1 < nh:
        h[rng.choice(np.where(h==0)[0], nh-n1, replace=False)] = 1
    return h

def upsample(hc, Nf, rng):
    Nc = len(hc)
    xc = np.linspace(0, 2, Nc, endpoint=False)
    xf = np.linspace(0, 2, Nf, endpoint=False)
    hi = np.interp(xf, xc, hc)
    hb = (hi > 0.5).astype(float)
    nh = Nf // 2
    n1 = int(hb.sum())
    if n1 > nh:
        hb[rng.choice(np.where(hb==1)[0], n1-nh, replace=False)] = 0
    elif n1 < nh:
        hb[rng.choice(np.where(hb==0)[0], nh-n1, replace=False)] = 1
    return hb
