def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)

def compute_D(h, N):
    """Compute derivative matrix for binary swap optimization."""
    shifts = np.arange(2 * N - 1) - (N - 1)

    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)

    pp_c = np.clip(pp, 0, N - 1)

def ensure_binary(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(777 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def interp_h(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def greedy_binary(h, N, deadline, max_sweeps=2000, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_nP = P + 1.0
        best_pq = None
        for p in p_list:
            cand = (c[None, :] - D[p, :])[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP:
                best_nP = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def binary_sa(h, N, deadline, max_sweeps=5000, T0=3.0, Tcool=0.994, T_min=0.04, sf=0.8):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    kicks = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= 100:
            nk = min(5, len(ones), len(zeros))
            for j in range(nk):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            T = T0 * 0.4
            stall = 0
            kicks += 1
            if kicks > 18:
                break
            continue
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(40, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_nP = P + 1.0
        best_pq = None
        for p in p_list:
            cand = (c[None, :] - D[p, :])[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP:
                best_nP = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_nP - P
        if delta < 1e-13 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / T)):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def nesterov_cont(h, N, deadline, k_top=55, lr0=0.022, alpha0=0.90):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    alpha = alpha0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif 0 < L < N:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        h_new = project_sum(y - lr * grad, target)
        m = alpha * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            lr *= 0.985
            alpha = min(0.97, 0.88 + it * 0.00004)
    return best_h, best_pk * dx

def cont_sa(h, N, deadline, T0=0.008, Tcool=0.9995, n_pert=4, step=0.06):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    while time.time() < deadline:
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800:
            break
        idx = np.random.choice(N, min(n_pert, N), replace=False)
        h_t = h.copy()
        h_t[idx] = h_t[idx] + np.random.randn(len(idx)) * step
        h_t = project_sum(h_t, target)
        pk_t = float(np.max(fft_corr(h_t)))
        d = pk_t - pk
        if d < 0 or (T > 1e-8 and np.random.random() < np.exp(-d / T)):
            h = h_t
        T = max(1e-8, T * Tcool)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Greedy binary N0 (25s)
    h_bin = ensure_binary(h0, N0)
    d1 = min(deadline - 900, t0 + 25)
    h1, c51 = greedy_binary(h_bin, N0, d1, max_sweeps=350, sf=1.0)
    if c51 < best_c5:
        best_c5, best_h = c51, h1.copy()

    # Phase 2: SA N0 (90s)
    d2 = min(deadline - 780, t0 + 115)
    h2, c52 = binary_sa(h_bin, N0, d2, max_sweeps=7000, T0=3.5, Tcool=0.994, T_min=0.04, sf=0.85)
    if c52 < best_c5:
        best_c5, best_h = c52, h2.copy()

    # Phase 3: Nesterov at N=400 (160s)
    Ns = 400
    if time.time() < deadline - 520:
        hs = project_sum(interp_h(best_h, Ns), Ns / 2.0)
        d3 = min(deadline - 500, t0 + 275)
        h3, c53 = nesterov_cont(hs, Ns, d3, k_top=50, lr0=0.028, alpha0=0.90)
        if c53 < best_c5:
            best_c5, best_h = c53, h3.copy()

    # Phase 4: Cont SA at N=400 (80s)
    if time.time() < deadline - 400:
        hs2 = project_sum(interp_h(best_h, Ns) + np.random.randn(Ns) * 0.04, Ns / 2.0)
        d4 = min(deadline - 400, t0 + 355)
        h4, c54 = cont_sa(hs2, Ns, d4, T0=0.006, Tcool=0.9995, n_pert=4, step=0.05)
        if c54 < best_c5:
            best_c5, best_h = c54, h4.copy()

    # Phase 5: Nesterov restart N=400 (80s)
    if time.time() < deadline - 320:
        hs3 = project_sum(interp_h(best_h, Ns) + np.random.randn(Ns) * 0.03, Ns / 2.0)
        d5 = min(deadline - 320, t0 + 435)
        h5, c55 = nesterov_cont(hs3, Ns, d5, k_top=55, lr0=0.02, alpha0=0.92)
        if c55 < best_c5:
            best_c5, best_h = c55, h5.copy()

    # Phase 6: Upsample to N0, greedy+SA (130s)
    if time.time() < deadline - 180:
        h_up = ensure_binary(interp_h(best_h, N0), N0)
        d6a = min(deadline - 140, t0 + 500)
        h6, c56 = greedy_binary(h_up, N0, d6a, max_sweeps=400, sf=0.7)
        if c56 < best_c5:
            best_c5, best_h = c56, h6.copy()

    # Phase 7: SA N0 perturbed (110s)
    if time.time() < deadline - 80:
        h_sa = ensure_binary(interp_h(best_h, N0), N0)
        for _ in range(12):
            p = np.where(h_sa == 1)[0]
            q = np.where(h_sa == 0)[0]
            if len(p) > 0 and len(q) > 0:
                h_sa[p[np.random.randint(len(p))]], h_sa[q[np.random.randint(len(q))]] = 0.0, 1.0
        d7 = min(deadline - 80, t0 + 610)
        h7, c57 = binary_sa(h_sa, N0, d7, max_sweeps=5000, T0=4.5, Tcool=0.993, T_min=0.03, sf=0.75)
        if c57 < best_c5:
            best_c5, best_h = c57, h7.copy()

    # Phase 8: Nesterov at N0 (90s)
    if time.time() < deadline - 40:
        hc = project_sum(interp_h(best_h, N0) + np.random.randn(N0) * 0.025, N0 / 2.0)
        d8 = min(deadline - 25, t0 + 700)
        h8, c58 = nesterov_cont(hc, N0, d8, k_top=65, lr0=0.014, alpha0=0.93)
        if c58 < best_c5:
            best_c5, best_h = c58, h8.copy()

    # Phase 9: Cont SA at N0 (40s)
    if time.time() < deadline - 15:
        hc2 = project_sum(interp_h(best_h, N0), N0 / 2.0)
        d9 = min(deadline - 5, t0 + 740)
        h9, c59 = cont_sa(hc2, N0, d9, T0=0.005, Tcool=0.9996, n_pert=3, step=0.04)
        if c59 < best_c5:
            best_c5, best_h = c59, h9.copy()

    # Phase 10: Final greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 1:
        hF = ensure_binary(best_h, N_fin)
        sf_f = 1.0 if N_fin <= 400 else 0.65
        hF2, c5F = greedy_binary(hF, N_fin, deadline - 0.5, max_sweeps=300, sf=sf_f)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
