    tgt = N / 2.0

    best_h = h.copy()

    v = np.zeros(N)  # Adam second moment
    t = 0
    
    while time.time() < deadline:
        # Animate alpha
        frac = min(1.0, (time.time() - t0) / duration)
        alpha = alpha0 + (alpha1 - alpha0) * frac
        
        c = fft_corr(h)
        cur = float(np.max(c))
        if cur * (2.0/N) < best_c - 1e-13:
            best_c = cur * (2.0/N)
            best_h = h.copy()
        
        # Compute softmax weights
        ac = alpha * c
        ac_max = ac.max()
        w = np.exp(ac - ac_max)
        w /= w.sum()
        
        # Compute gradient: sum_k w[k] * dcorr_k/dh
        # dcorr_k/dh[i] = (1-h[i+k]) for i+k < N - (h[i-k]) for i-k >= 0
        # where k = shift index mapped to actual shift
        # In our fft_corr, index k corresponds to shift L = k - (N-1)
        
        grad = np.zeros(N)
        center = N - 1
        for k in range(2 * N - 1):
            wk = w[k]
            if wk < 1e-8:
                continue
            L = k - center
            if L >= 0:
                # corr[k] = sum_{i=0}^{N-1-L} h[i] * (1-h[i+L])
                # d/dh[i] = (1-h[i+L]) for i < N-L, and -h[i-L] for i >= L (with i-L < N-L-i... )

                # = (1-h[i+L]) if i+L < N (i appears as first factor)
                #   - h[i-L] if i-L >= 0 (i appears as second factor, at position j=i-L)
                if L < N:
                    grad[:N-L] += wk * (1.0 - h[L:N])  # first factor contribution
                    grad[L:N] -= wk * h[:N-L]  # second factor contribution
            else:
                M = -L
                if M < N:
                    grad[M:N] += wk * (1.0 - h[:N-M])
                    grad[:N-M] -= wk * h[M:N]
        
        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        h_new = proj_box_sum(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), tgt)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj_box_sum(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - tgt) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def to_binary(h, N, rng):
    thr = np.partition(h, N - 1 - N // 2)[N - 1 - N // 2]
    out = (h >= thr).astype(float)
    n1 = int(out.sum())
    if n1 != N // 2:
        idx_eq = np.where(h == thr)[0]
        if n1 < N // 2:
            out[idx_eq[:N // 2 - n1]] = 1.0
        else:
            sel = rng.choice(idx_eq, n1 - N // 2, replace=False)
            out[sel] = 0.0
    return out

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def greedy_bin(h, N, deadline, ms=300, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(42)
    for _ in range(ms):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c, best_h, stall = P, h.copy(), 0
        else:
            stall += 1
            if stall > 60:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        pl = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
        best_nP, best_pq = P, None
        for p in pl:
            npk = (c[None, :] - D[p, :][None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP - 1e-14:
                best_nP, best_pq = float(npk[mi]), (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
    return best_h, best_c * (2.0 / N)

def sa_bin(h, N, deadline, T0=5.0, tc=0.993, Tmin=0.02, sf=1.0, kr=150, ks=6):
    h = h.astype(float).copy()
    best_c
