    N = len(h)

        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_b, best_h)
        ht = make_binary(hi, Nt)
        hr, _ = bin_anneal(ht, Nt, min(d4, time.time() + 35), 5000, 28, 2, 0.003, 0.0, stall_limit=400)
        c5r = compute_c5(hr, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()

    # Phase 5: 3-swap from current best
    if time.time() < dl - 60:
        Nb = len(best_h)
        d5 = min(dl - 35, time.time() + 40)
        h3, _ = bin_anneal(best_h, N
b, d5, 4000, 25, 3, 0.004, 0.0, stall_limit=350)
        c5 = compute_c5(h3, N
b)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 6: Spectral restart + quick local search
    if time.time() < dl - 40:
        N
b = len(best_h)
        d6 = min(dl
 - 20, time.time() + 30)
        for k in range(4):
            if time.time() > d6:
                break
            hs = spectral_init(N
b, np.random)
            hd, _ = bin_anneal(hs, N
b, min(d6, time.time() + 7), 500, 15, 2, 0.002, 0.0, stall_limit=80)
            cd = compute_c5(hd, N
b)
            if cd < best_c5:
                best_c5, best_h = cd, hd.copy()

    # Phase 7: Final 1-swap polish
    if time.time() < dl - 3:
        N
f = len(best_h)
        hf = (best_h > 0.5).astype(float)
        hf, _ = bin_anneal(hf, N
f, dl - 1.0, 6000, 18, 1, 0.0, 0.0, stall_limit=200)
        cf = compute_c5(hf, N
f)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Final: compare continuous vs binary
    N
fin = len(best_h)
    fc5 = compute_c5(best_h, N
fin)
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, N
fin)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, Nfin
