    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(np.conj(H) * G, L)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

        elif n1 < nh:
            ht[np.random.choice(np.where(ht == 0)[0], nh - n1, replace=False)] = 1.0
        sd = min(deadline, time.time() + 50)
        hr, _ = bin_search(ht, Nt, sd, max_iters=250, ntri=20)
        c5r = compute_c5(hr, Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Quick binary polish
    d1 = min(dl - 600, t0 + 80)
    hb, _ = bin_search(best_h, N0, d1, max_iters=200, ntri=25, anneal=0.002)
    c = compute_c5(hb, N0)
    if c < best_c5:
        best_c5, best_h = c, hb.copy()

    # Phase 2: Continuous optimization with FISTA (multiple params)
    if time.time() < dl - 400:
        d2 = min(dl - 300, time.time() + 180)
        for r in range(5):
            if time.time() > d2:
                break
            sigma = 0.25 - 0.05 * r
            hr = best_h + np.random.randn(N0) * sigma
            hr = proj01(hr, N0 / 2.0)
            for T_val in [30.0, 50.0, 80.0]:
                if time.time() > d2:
                    break
                hf, cf = fista_smooth(hr, N0, d2, T=T_val, max_iter=120, lam=8.0)
                if cf < best_c5:
                    best_c5, best_h = cf, hf.copy()

    # Phase 3: Shift-based search
    if time.time() < dl - 250:
        d3 = min(dl - 150, time.time() + 120)
        hs, cs = shift_and_swap(best_h, N0, d3, n_shifts=5)
        if cs < best_c5:
            best_c5, best_h = cs, hs.copy()

    # Phase 4: Extended binary search with double swaps
    if time.time() < dl - 150:
        d4 = min(dl - 80, time.time() + 150)
        hb4, _ = bin_search(best_h, N0, d4, max_iters=700, ntri=30, anneal=0.001)
        c4 = compute_c5(hb4, N0)
        if c4 < best_c5:
            best_c5, best_h = c4, hb4.copy()

    # Phase 5: More continuous with different init
    if time.time() < dl - 100:
        d5 = min(dl - 50, time.time() + 100)
        for r in range(3):
            if time.time() > d5:
                break
            hr = best_h + np.random.randn(N0) * (0.15 - 0.04 * r)
            hr = proj01(hr, N0 / 2.0)
            hf, cf = fista_smooth(hr, N0, d5, T=60.0, max_iter=100, lam=5.0)
            if cf < best_c5:
                best_c5, best_h = cf, hf.copy()

    # Phase 6: Multi-resolution
    if time.time() < dl - 60:
        best_h, best_c5 = try_resolutions(best_h, best_c5, min(dl - 30, time.time() + 150))

    # Phase 7: Final binary polish on best
    if time.time() < dl - 10:
        d7 = dl - 2.0
        hb7, _ = bin_search(best_h, len(best_h), d7, max_iters=400, ntri=35, anneal=0.001)
        c7 = compute_c5(hb7, len(best_h))
        if c7 < best_c5:
            best_c5, best_h = c7, hb7.copy()

    # Final: compare continuous vs binary
    fc5 = compute_c5(best_h, len(best_h))
    fb = (best_h > 0.5).astype(float)
    fc5b = compute_c5(fb, len(best_h))
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b
    return best_h, fc5, len(best_h)
