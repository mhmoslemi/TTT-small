def fft_corr(h):

    L = 2*N - 1
    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def nesterov_sub(h, N, deadline, k=50, lr0=0.005):
    h = h.astype(float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    for it in range(20000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        # Subgradient: average of gradients for active shifts
        grad = np.zeros(N)
        center = N-1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0*y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]

        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it*0.00005)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def find_opt_t(c_base, hv, s):
    a = c_base - hv * s
    b = s
    def sg_range(t):
        vals = a + t * b
        mx = vals.max()
        act = vals >= mx - 1e-9
        if not act.any():
            return 0.0, 0.0
        return b[act].min(), b[act].max()
    sm0, _ = sg_range(0.0)
    if sm0 >= -1e-11:
        return 0.0
    _, sx1 = sg_range(1.0)
    if sx1 <= 1e-11:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        sm, sx = sg_range(mid)
        if sm >= -1e-11:
            hi = mid
        elif sx <= 1e-11:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def coord_descent(h, N, deadline, max_sweeps=500, rng=None):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(fft_corr(h).max())
    if rng is None:
        rng = np.random.RandomState(42)
    lags = np.arange(2 * N - 1) - (N - 1)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_base = fft_corr(h)
        order = rng.permutation(N)
        for i in order:
            hv = h[i]
            fv = (i + lags >= 0) & (i + lags < N)
            sv = (i - lags >= 0) & (i - lags < N)
            s = np.zeros(2 * N - 1)
            fi = np.clip(i + lags, 0, N - 1)
            si = np.clip(i - lags, 0, N - 1)
            s[fv] = 1.0 - h[fi[fv]]
            s[sv] -= h[si[sv]]
            t_opt = find_opt_t(c_base, hv, s)
            if abs(t_opt - hv) > 1e-14:
                c_base += (t_opt - hv) * s
                h[i] = t_opt
    h = project_sum(h, target)
    cm = float(fft_corr(h).max())
    if cm < best_c:
        best_c = cm
        best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def nesterov_sub(h, N, deadline, k=50, lr0=0.004):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    for it in range(50000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 200 == 0:
            lr *= 0.992
            beta = min(0.97, 0.85 + it * 0.00004)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    return np.interp(np.linspace(0, 2, N_new, endpoint=False),
                     np.linspace(0, 2, N_old, endpoint=False), h_old)

def binary_polish(h, N, deadline, max_sweeps=300, rng=None):
    h = h.astype(float).copy()
    target = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 != target:
        r = np.random.RandomState(77)
        if n1 > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[r.choice(idx, n1 - target, replace=False)] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[r.choice(idx, target - n1, replace=False)] = 1
    if rng is None:
        rng = np.random.RandomState(13)
    best_c = float(fft_corr(h_bin).max())
    best_h = h_bin.copy()
    lags = np.arange(2 * N - 1) - (N - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        c_cur = fft_corr(h_bin)
        P = float(c_cur.max())
        best_delta = 0.0
        best_pq = None
        n_try_p = min(len(ones), 50)
        p_list = rng.choice(ones, n_try_p, replace=False) if len(ones) > 50 else ones
        for p in p_list:
            fv = (p + lags >= 0) & (p + lags < N)
            sv = (p - lags >= 0) & (p - lags < N)
            sp = np.zeros(2 * N - 1)
            fi = np.clip(p + lags, 0, N - 1)
            si = np.clip(p - lags, 0, N - 1)
            sp[fv] = 1.0 - h_bin[fi[fv]]
            sp[sv] -= h_bin[si[sv]]
            n_try_q = min(len(zeros), 20)
            q_list = rng.choice(zeros, n_try_q, replace=False) if len(zeros) > 20 else zeros
            for q in q_list:
                fvq = (q + lags >= 0) & (q + lags < N)
                svq = (q - lags >= 0) & (q - lags < N)
                sq = np.zeros(2 * N - 1)
                fiq = np.clip(q + lags, 0, N - 1)
                siq = np.clip(q - lags, 0, N - 1)
                sq[fvq] = 1.0 - h_bin[fiq[fvq]]
                sq[svq] -= h_bin[siq[svq]]
                new_P = float((c_cur - sp + sq).max())
                if new_P < P + best_delta:
                    best_delta = new_P - P
                    best_pq = (int(p), int(q))
        if best_pq is not None and best_delta < -1e-12:
            h_bin[best_pq[0]] = 0.0
            h_bin[best_pq[1]] = 1.0
            P_new = float(fft_corr(h_bin).max())
            if P_new < best_c - 1e-14:
                best_c = P_new
                best_h = h_bin.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    return best_h, best_c * (2.0 / N)
