    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max_grad(corr, T=50):
    """Compute gradient of log-sum-exp smoothing of max(corr)."""
    # Returns softmax weights and the smooth objective value
    shifted = corr - np.max(corr)
    weights = np.exp(T * shifted)
    weights /= np.sum(weights)
    return weights, np.max(corr) + np.log(np.sum(weights)) / T

def grad_of_corr(h, corr_weights):
    """Compute weighted gradient of correlations."""
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    for k_idx, w in enumerate(corr_weights):
        if w < 1e-10:
            continue
        L = k_idx - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[L:] += w * (1.0 - h[:N-L])
            grad[:N-L] -= w * h[L:]
        elif L < 0 and -L < N:
            M = -L
            grad[:N-M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:N-M]
    return grad

def adam_smooth(h, N, deadline, T=50, lr=0.005, b1=0.9, b2=0.999, eps=1e-8):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    T_cur = T
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        c5_cur = float(np.max(corr)) * (2.0 / N)
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h.copy()
        weights, _ = smooth_max_grad(corr, T_cur)
        grad = grad_of_corr(h, weights)
        # Adam
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h = project_01_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        # Anneal temperature
        if t % 300 == 0:
            T_cur = max(5, T_cur * 0.95)
            lr *= 0.985
    return best_h, best_c5

                zeros = np.where(h_b == 0)[0]

                h_b[q] = 0

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_weighted(h, weights):
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    for k_idx in range(len(weights)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        L = k_idx - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:N - M]
    return grad

    e = np.exp(T * np.maximum(shifted, -20.0))
    s = e.sum()
    if s < 1e-30:

    return e / s

            best_c5 = c5_cur
            best_h = h.copy()

            T = max(10.0, T * 0.94)
            lr *= 0.982
    return best_h, best_c5

def penalty_adam(h, N, deadline, T_target=None, lr_init=0.006):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h, N)
    if T_target is None:
        T_target = float(np.max(fft_corr(h))) - 0.5
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    lr = lr_init
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        c5_cur = pk * 2.0 / N
        if c5_cur < best_c5 - 1e-11:
            best_c5 = c5_cur
            best_h = h.copy()
        excess = np.maximum(corr - T_target, 0.0)
        w = 2.0 * excess
        sw = w.sum()
        if sw < 1e-12:
            T_target = pk - 0.3
        else:
            w = w / sw
        grad = grad_weighted(h, w)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        h = project_01_sum(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), target)
        if t % 300 == 0:
            lr *= 0.988
    return best_h, best_c5

def binary_refine(h, N, deadline, trials_per_iter=12, kick_thresh=60, kick_sz=3):
    h = h.astype(float).copy()
    n_ones = int(np.round(h.sum()))
    if n_ones != N // 2:
        idx = np.argsort(h)
        h_b = np.zeros(N)
        h_b[idx[N // 2:]] = 1.0
    else:
        h_b = (h >= np.sort(h)[N // 2]).astype(float)
        while h_b.sum() > N / 2:
            h_b[np.argmax(h_b * (1 - h_b))] = 0
        while h_b.sum() < N / 2:
            h_b[np.argmin(h_b * h_b)] = 1
    best_c5 = compute_c5(h_b, N)
    best_h = h_b.copy()
    corr = fft_corr(h_b)
    P = float(np.max(corr))
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    stuck = 0
    while time.time() < deadline:
        if len(ones) < 2 or len(zeros) < 2:
            break
        improved = False
        for _ in range(trials_per_iter):
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h_b[p] = 0.0
            h_b[q] = 1.0
            c_new = float(np.max(fft_corr(h_b)))
            if c_new < P - 1e-13:
                P = c_new
                c5_new = P * 2.0 / N
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_b.copy()
                ones = np.where(h_b == 1)[0]
                zeros = np.where(h_b == 0)[0]
                improved = True
            else:
                h_b[p] = 1.0
                h_b[q] = 0.0
        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck >= kick_thresh:
                for _ in range(kick_sz):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h_b[p] = 0.0
                    h_b[q] = 1.0
                    ones = np.where(h_b == 1)[0]
                    zeros = np.where(h_b == 0)[0]
                P = float(np.max(fft_corr(h_b)))
                stuck = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Smooth max Adam, moderate temperature
    if time.time() < deadline - 380:
        d1 = min(deadline - 360, t0 + 200)
        h1 = best_h + np.random.randn(N0) * 0.15
        h1 = project_01_sum(h1, N0 / 2.0)
        h1, c5_1 = adam_smooth(h1, N0, d1, T_init=80, lr_init=0.004)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Penalty-based Adam (target-driven)
    if time.time() < deadline - 300:
        d2 = min(deadline - 280, t0 + 380)
        h2 = best_h + np.random.randn(N0) * 0.08
        h2 = project_01_sum(h2, N0 / 2.0)
        h2, c5_2 = penalty_adam(h2, N0, d2, lr_init=0.005)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Low-temperature smooth Adam (fine tuning)
    if time.time() < deadline - 240:
        d3 = min(deadline - 220, t0 + 520)
        h3 = best_h + np.random.randn(N0) * 0.04
        h3 = project_01_sum(h3, N0 / 2.0)
        h3, c5_3 = adam_smooth(h3, N0, d3, T_init=200, lr_init=0.002)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary refinement
    if time.time() < deadline - 180:
        d4 = min(deadline - 160, t0 + 650)
        h4, c5_4 = binary_refine(best_h, N0, d4, trials_per_iter=15, kick_thresh=50, kick_sz=3)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Multi-resolution exploration
    for N_try in [300, 360, 420]:
        if time.time() > deadline - 120:
            break
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = project_01_sum(h_int, N_try / 2.0)
        sub_dl = min(deadline - 50, time.time() + 50)
        h_rt, c5_rt = adam_smooth(h_bt, N_try, sub_dl, T_init=50, lr_init=0.005)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final continuous polish (very fine)
    if time.time() < deadline - 15:
        hF = best_h + np.random.randn(len(best_h)) * 0.02
        hF = project_01_sum(hF, len(best_h) / 2.0)
        hF, c5F = adam_smooth(hF, len(best_h), deadline - 3.0, T_init=300, lr_init=0.001)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 7: Quick binary sweep
    if time.time() < deadline - 1.0:
        hB, c5B = binary_refine(best_h, len(best_h), deadline - 0.3,
                                trials_per_iter=8, kick_thresh=30, kick_sz=2)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    return best_h, best_c5, len(best_h)
