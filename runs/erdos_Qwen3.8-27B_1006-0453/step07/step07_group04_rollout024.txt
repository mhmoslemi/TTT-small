import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def greedy_ls(h, N, deadline, max_sweeps=5000, sf=1.0):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def ilt_search(h, N, deadline, n_iter=150, pert_lo=3, pert_hi=10):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 3:
            break
        sub_dl = time.time() + min(40, remaining * 0.4)
        h_ls, c_ls = greedy_ls(h, N, sub_dl, max_sweeps=3000, sf=1.0)
        if c_ls < best_c - 1e-14:
            best_c = c_ls
            best_h = h_ls.copy()
            stall = 0
        else:
            stall += 1
        h = h_ls.copy()
        n_pert = np.random.randint(pert_lo, pert_hi + 1)
        if stall > 10:
            n_pert = int(n_pert * 1.5) + 2
            stall = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(min(n_pert, 20)):
            if len(ones) == 0 or len(zeros) == 0:
                break
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=70, lr0=0.015, beta0=0.91):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        top_idx = np.argpartition(corr, -k)[-k:]
        top_vals = corr[top_idx].copy()
        top_vals = top_vals - top_vals.max()
        weights = np.exp(top_vals / 0.8)
        wsum = weights.sum()
        if wsum < 1e-10:
            weights = np.ones(k) / k
        else:
            weights /= wsum
        grad = np.zeros(N)
        center = N - 1
        for i in range(k):
            w = weights[i]
            L = int(top_idx[i] - center)
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w * (1.0 - y[:N - L])
                grad[:N - L] -= w * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - y[M:])
                    grad[M:] -= w * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.97
            beta = min(0.995, beta0 + it * 0.00004)
        if it % 2000 == 0 and lr < 0.0005:
            break
    return best_h, best_pk * dx
