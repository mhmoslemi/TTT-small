import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    """Project h to [0,1]^N with sum = target."""
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Binary search for shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient_steps(h, N, deadline, lr0=0.1, n_iters=5000):
    """Subgradient descent on continuous relaxation with max-correlation objective."""
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        corr = fft_corr(h)
        peak = float(np.max(corr))
        if peak * 2.0 / N < best_c - 1e-12:
            best_c = peak * 2.0 / N
            best_h = h.copy()
        
        # Subgradient: f_k*(p) = h[p-k*+(N-1)] - (1-h[p+k*-(N-1)])
        k_star = int(np.argmax(corr))
        shift = k_star - (N - 1)
        
        f = np.zeros(N)
        if shift >= 0:
            f[shift:] += h[:N - shift]       # h[p-shift]
            e = max(0, N - shift)
            f[:e] -= (1 - h[shift:])         # (1-h[p+shift])
            f[e:] -= 1.0
        else:
            s = -shift
            f[:N-s] += h[s:]                 # h[p+s]
            f[s:] -= (1 - h[:N-s])           # (1-h[p-s])
            f[:s] -= 1.0
        
        h = h - lr * f
        h = project_sum(h, target)
        
        if it % 100 == 0:
            lr *= 0.98
    
    return best_h, best_c

def targeted_bin_search(h, N, deadline, max_iters=30000, n_cands=4):
    """Targeted binary local search focusing on peak correlation."""
    h = h.astype(float).copy()
    nh = N // 2
    if h.sum() != nh:
        ones = np.where(h > 0.5)[0]
        if len(ones) > nh:
            h[np.random.choice(ones, len(ones) - nh, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[np.random.choice(zeros, nh - len(ones), replace=False)] = 1
    
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        k_star = int(np.argmax(corr))
        shift = k_star - (N - 1)
        
        # f(p) = h[p-shift] - (1-h[p+shift])  (OOB: h=0, 1-h=1)
        f = np.zeros(N)
        if shift >= 0:
            f[shift:] += h[:N - shift]
            e = max(0, N - shift)
            f[:e] -= (1 - h[shift:])
            f[e:] -= 1.0
        else:
            s = -shift
            f[:N-s] += h[s:]
            f[s:] -= (1 - h[:N-s])
            f[:s] -= 1.0
        
        f_ones = f[ones]
        f_zeros = f[zeros]
        
        # Pick candidates: min f among ones, max f among zeros
        i_cands = ones[np.argsort(f_ones)[:n_cands]]
        j_cands = zeros[np.argsort(-f_zeros)[:n_cands]]
        
        best_nm = np.inf
        best_pair = None
        for i in i_cands:
            for j in j_cands:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nm = float(np.max(fft_corr(h)))
                h[i] = 1
                h[j] = 0
                if nm < best_nm:
                    best_nm = nm
                    best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            corr = fft_corr(h)
            cur_max = best_nm
            if cur_max * 2.0 / N < best_c - 1e-12:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Random swap
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
            h[i] = 0
            h[j] = 1
            nm = float(np.max(fft_corr(h)))
            if nm > cur_max + 3.0/N:
                h[i] = 1
                h[j] = 0
            else:
                corr = fft_corr(h)
                cur_max = nm
            no_improve += 1
        
        if no_improve > 8000:
            for _ in range(5):
                ii = int(np.random.choice(np.where(h == 1)[0]))
                jj = int(np.random.choice(np.where(h == 0)[0]))
                h[ii], h[jj] = h[jj], h[ii]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            no_improve = 0
    
    return best_h, best_c

def block_swap_search(h, N, deadline, block_sizes=(2, 3, 5, 8), max_iters=20000):
    """Block swap local search."""
    h = h.astype(float).copy()
    nh = N // 2
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        L = int(np.random.choice(block_sizes))
        if L >= N // 2:
            continue
        
        # Find one-blocks and zero-blocks of length >= L
        one_blocks = []
        zero_blocks = []
        i = 0
        while i < N:
            if h[i] == 1:
                j = i
                while j < N and h[j] == 1:
                    j += 1
                if j - i >= L:
                    one_blocks.append(i)
                i = j
            else:
                j = i
                while j < N and h[j] == 0:
                    j += 1
                if j - i >= L:
                    zero_blocks.append(i)
                i = j
        
        if not one_blocks or not zero_blocks:
            continue
        
        ob = int(np.random.choice(one_blocks))
        zb = int(np.random.choice(zero_blocks))
        if abs(ob - zb) < L:
            continue
        
        h[ob:ob+L] = 0
        h[zb:zb+L] = 1
        nm = float(np.max(fft_corr(h)))
        
        if nm <= cur_max:
            corr = fft_corr(h)
            cur_max = nm
            if cur_max * 2.0 / N < best_c - 1e-12:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
        else:
            h[ob:ob+L] = 1
            h[zb:zb+L] = 0
    
    return best_h, best_c

    t0 = time.time()

    np.random.seed(seed)

    for N_try in [200, 260, 300, 340, 420, 480]:
        if time.time() > deadline - 60:

        if N_try % 2 != 0:

        if N_try == N0:
            continue
        # Structured init
        x = np.arange(N_try)
        v = np.sin(2*np.pi*x*3.0/N_try) + 0.7*np.sin(2*np.pi*x*7.0/N_try) + 0.5*np.cos(2*np.pi*x*11.0/N_try)
        hv = (v > np.median(v)).astype(float)
        nh = N_try // 2
        if hv.sum() != nh:
            vs = np.argsort(-v)
            hv = np.zeros(N_try)
            hv[vs[:nh]] = 1
        
        sub_dl = min(deadline - 30, time.time() + 30)
        hv, cv = targeted_bin_search(hv, N_try, sub_dl, max_iters=10000, n_cands=3)
        if cv < best_c5:
            best_c5, best_h = cv, hv
    
    # Phase 5: Final targeted search on best
    hF = (best_h > 0.5).astype(float)
    dF = deadline - 1
    hF, cF = targeted_bin_search(hF, len(hF), dF, max_iters=30000, n_cands=5)
    cF = compute_c5(hF, len(hF))
    if cF < best_c5:
        best_h, best_c5 = hF, cF
    
    # Final validation
    hF = (best_h > 0.5).astype(float)
    cF = compute_c5(hF, len(hF))
    
    return hF, cF, len(hF)

def compute_delta_c(h, N, i, j):
    L = 2*N - 1
    # g_i[shift] for shift = -(N-1)...(N-1)
    # In numpy, shift index 0 corresponds to shift=-(N-1), index N-1 to shift=0, index 2N-2 to shift=N-1
    # g[shift_idx] = h[i - shift] - (1-h[i+shift]) with OOB
    
    # Method: use the fact that g is a "cross-correlation" of a spike
    # g_i = corr(spike_i, h) - corr(spike_i, 1-h) + correction
    
    # Actually, let me just compute it directly using array slicing
    g_i = np.zeros(L)
    g_j = np.zeros(L)
    
    for p, g in [(i, g_i), (j, g_j)]:
        # h[p-shift] for shift in [-(N-1), N-1]
        # shift = k - (N-1), so p-shift = p-k+(N-1)
        # Valid when 0 <= p-k+(N-1) <= N-1, i.e., k in [p, p+N-1]
        lo = max(0, p)
        hi = min(L-1, p+N-1)
        if lo <= hi:
            # g[lo..hi] += h[p-(lo-(N-1))..p-(hi-(N-1))]
            # = h[p-N+1+lo .. p-N+1+hi]
            idx_lo = p - (lo - (N-1))
            idx_hi = p - (hi - (N-1)) - 1  # inclusive
            g[lo:hi+1] += h[idx_lo:idx_hi+1]
        
        # (1-h[p+shift]): p+shift = p+k-(N-1)
        # Valid when 0 <= p+k-(N-1) <= N-1, i.e., k in [N-1-p, N-1+p]
        lo2 = max(0, N-1-p)
        hi2 = min(L-1, N-1+p)
        if lo2 <= hi2:
            idx_lo2 = p + (lo2 - (N-1))
            idx_hi2 = p + (hi2 - (N-1))
            g[lo2:hi2+1] -= (1 - h[idx_lo2:idx_hi2+1])
        
        # OOB: 1-h[OOB] = 1, so we subtract 1 where p+k-(N-1) is out of [0,N-1]
        # p+k-(N-1) < 0: k < N-1-p
        # p+k-(N-1) > N-1: k > N-1+p
        oob_lo = max(0, min(L, N-1-p))  # k from 0 to N-2-p
        oob_hi = max(0, N-1+p+1)        # k from N+p to 2N-2
        if oob_lo < L:
            g[:min(L, N-1-p)] -= 1
        if oob_hi < L:
            g[oob_hi:] -= 1
    
    delta = g_i - g_j
    delta[N-1] -= 2  # δ correction
    return delta

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_binary(h, nh):
    h = h.copy()
    s = int(h.sum())
    if s > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, s - nh, replace=False)] = 0
    elif s < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - s, replace=False)] = 1
    return h

def peak_grad(h, N, corr):
    """Compute subgradient f(p) = h[p-shift] - (1-h[p+shift]) at peak lag."""
    k_star = int(np.argmax(corr))
    shift = k_star - (N - 1)
    f = np.zeros(N)
    if shift >= 0:
        f[shift:] += h[:N - shift]
        e = max(0, N - shift)
        f[:e] -= (1 - h[shift:])
        f[e:] -= 1.0
    else:
        s = -shift
        f[:N - s] += h[s:]
        f[s:] -= (1 - h[:N - s])
        f[:s] -= 1.0
    return f

def targeted_search(h, N, deadline, max_iters=60000, n_cands=4, block_prob=0.15):
    h = h.astype(float).copy()
    nh = N // 2
    fix_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        if np.random.random() < block_prob:
            # Block swap
            L = int(np.random.choice([2, 3, 5, 7]))
            ob, zb = [], []
            i = 0
            while i < N:
                if h[i] == 1:
                    j = i
                    while j < N and h[j] == 1:
                        j += 1
                    if j - i >= L:
                        ob.append(i)
                    i = j
                else:
                    j = i
                    while j < N and h[j] == 0:
                        j += 1
                    if j - i >= L:
                        zb.append(i)
                    i = j
            if not ob or not zb:
                no_improve += 1
                continue
            bi, bz = int(np.random.choice(ob)), int(np.random.choice(zb))
            if abs(bi - bz) < L:
                no_improve += 1
                continue
            h[bi:bi+L] = 0
            h[bz:bz+L] = 1
            nm = float(np.max(fft_corr(h)))
            if nm <= cur_max:
                cur_max = nm
                corr = fft_corr(h)
                if cur_max * 2.0 / N < best_c - 1e-12:
                    best_c = cur_max * 2.0 / N
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[bi:bi+L] = 1
                h[bz:bz+L] = 0
                no_improve += 1
            continue

        # Targeted single swap
        f = peak_grad(h, N, corr)
        f1 = f[ones]
        f0 = f[zeros]
        i_c = ones[np.argsort(f1)[:n_cands]]
        j_c = zeros[np.argsort(-f0)[:n_cands]]

        best_nm = np.inf
        best_pair = None
        for ii in i_c:
            for jj in j_c:
                if ii == jj:
                    continue
                h[ii] = 0
                h[jj] = 1
                nm = float(np.max(fft_corr(h)))
                h[ii] = 1
                h[jj] = 0
                if nm < best_nm:
                    best_nm = nm
                    best_pair = (ii, jj)

        if best_pair is not None:
            ii, jj = best_pair
            h[ii] = 0
            h[jj] = 1
            corr = fft_corr(h)
            cur_max = best_nm
            if cur_max * 2.0 / N < best_c - 1e-12:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            ii = int(np.random.choice(ones))
            jj = int(np.random.choice(zeros))
            h[ii] = 0
            h[jj] = 1
            nm = float(np.max(fft_corr(h)))
            if nm > cur_max + 4.0 / N:
                h[ii] = 1
                h[jj] = 0
            else:
                corr = fft_corr(h)
                cur_max = nm
            no_improve += 1

        if no_improve > 6000:
            for _ in range(4):
                a = int(np.random.choice(np.where(h == 1)[0]))
                b = int(np.random.choice(np.where(h == 0)[0]))
                h[a], h[b] = h[b], h[a]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            no_improve = 0

    return best_h, best_c

def subgrad_refine(h, N, deadline, lr0=0.08):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    it = 0
    while time.time() < deadline - 1:
        it += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk * 2.0 / N < best_c - 1e-10:
            best_c = pk * 2.0 / N
            best_h = h.copy()
        f = peak_grad(h, N, corr)
        h = h - lr * f
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= 0.97
    return best_h, best_c

def struct_init(N, v):
    x = np.arange(N)
    vals = np.sin(2*np.pi*x*(v+2.0)/N) + 0.6*np.sin(2*np.pi*x*(2*v+3.0)/N) + 0.4*np.cos(2*np.pi*x*(3*v+1.0)/N)
    h = (vals > np.median(vals)).astype(float)
    nh = N // 2
    ps = np.argsort(-vals)
    h = np.zeros(N)
    h[ps[:nh]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    h0 = np.array(initial_h_values, dtype=float)
