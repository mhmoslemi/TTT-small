import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())

    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)

    return h

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_ls(h, deadline, K=10, max_noimp=300):
    """Fast binary local search using vectorized swap evaluation."""
    N = len(h)
    h = h.copy()
    center = N - 1
    best_h = h.copy()
    best_c = compute_c5(h)
    no_imp = 0
    
    while time.time() < deadline and no_imp < max_noimp:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        c = peak * 2.0 / N
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        top_lags = np.argsort(corr)[-K:]
        top_m = (top_lags - center).astype(int)
        top_R = corr[top_lags].copy()
        
        # Compute G[k, p] = h[p-m_k] - h[p+m_k]
        G = np.zeros((K, N))
        p = np.arange(N)
        for k in range(K):
            m = top_m[k]
            lo_idx = p - m
            hi_idx = p + m
            valid_lo = (lo_idx >= 0) & (lo_idx < N)
            valid_hi = (hi_idx >= 0) & (hi_idx < N)
            G[k, valid_lo] += h[lo_idx[valid_lo]]
            G[k, valid_hi] -= h[hi_idx[valid_hi]]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        Ge = G[:, ones]   # (K, n1)
        Gz = G[:, zeros]  # (K, n0)
        
        # scores[i,j] = max_k (top_R[k] + Ge[k,i] - Gz[k,j])
        # Vectorize: (n1, n0, K)
        scores = top_R[None, None, :] + Ge[:, None, :] - Gz[None, :, :]
        max_s = scores.max(axis=2)
        
        # Also track current peak (might be at a non-top lag)
        cur_peak = peak
        best_ai, best_bi = np.unravel_index(np.argmin(max_s), max_s.shape)
        est_peak = max_s[best_ai, best_bi]
        
        if est_peak < cur_peak - 1e-10:
            a = ones[best_ai]
            b = zeros[best_bi]
            h[a], h[b] = 0.0, 1.0
        else:
            no_imp += 2  # Penalize more for no improvement
    
    return best_h, best_c

def sa_binary(h, deadline, T0=None, cooling=0.999):
    """Simulated annealing on binary h for exploration."""
    N = len(h)
    h = fix_binary(h.copy())
    center = N - 1
    best_h = h.copy()
    best_c = compute_c5(h)
    
    corr = compute_corr(h)
    peak = float(np.max(corr))
    if T0 is None:
        T0 = peak * 0.05
    
    it = 0
    while time.time() < deadline:
        it += 1
        corr = compute_corr(h)
        peak = float(np.max(corr))
        c = peak * 2.0 / N
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        
        h[a], h[b] = 0.0, 1.0
        corr_new = compute_corr(h)
        peak_new = float(np.max(corr_new))
        
        if peak_new > peak:
            if np.random.random() > np.exp(-(peak_new - peak) / max(T0, 1e-8)):
                h[a], h[b] = 1.0, 0.0  # Reject
                peak_new = peak
        peak = peak_new
        T0 *= cooling
    
    return best_h, best_c

def cont_opt(h, deadline, k=20, lr0=0.02, mom=0.85, n_lags=10):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    center = N - 1
    
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 60:
            lr0 *= 0.8
            stale = 0
            if lr0 < 1e-5:
                break
        
        # Gradient wrt top n_lags
        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            M = abs(int(m) - center)
            if M == 0:
                grad += (1.0 - 2.0 * h)
            elif M < N:
                hi_idx = np.arange(M, N)
                lo_idx = np.arange(0, N - M)
                grad[hi_idx] += (1.0 - h[hi_idx - M])
                grad[lo_idx] -= h[lo_idx + M]
        grad /= k
        
        h_new = project(h + mom * v - lr0 * grad, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    
    return best_h, best_c

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_ls(h, deadline, K=12, max_noimp=400):
    N = len(h)
    h = h.copy()
    center = N - 1
    best_h = h.copy()
    best_c = compute_c5(h)
    no_imp = 0
    p = np.arange(N)
    while time.time() < deadline and no_imp < max_noimp:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        c = peak * 2.0 / N
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        top_lags = np.argsort(corr)[-K:]
        top_m = (top_lags - center).astype(int)
        top_R = corr[top_lags].copy()
        G = np.zeros((K, N))
        for k in range(K):
            m = top_m[k]
            lo_idx = p - m
            hi_idx = p + m
            vl = (lo_idx >= 0) & (lo_idx < N)
            vh = (hi_idx >= 0) & (hi_idx < N)
            G[k, vl] += h[lo_idx[vl]]
            G[k, vh] -= h[hi_idx[vh]]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        Ge = G[:, ones]
        Gz = G[:, zeros]
        scores = top_R[None, None, :] + Ge[:, None, :] - Gz[None, :, :]
        max_s = scores.max(axis=2)
        best_ai, best_bi = np.unravel_index(np.argmin(max_s), max_s.shape)
        if max_s[best_ai, best_bi] < peak - 1e-10:
            a = ones[best_ai]
            b = zeros[best_bi]
            h[a], h[b] = 0.0, 1.0
        else:
            no_imp += 1
            if no_imp > max_noimp:
                break
    return best_h, best_c

def sa_binary(h, deadline, K=8, T0_factor=0.04, cooling=0.9999):
    N = len(h)
    h = fix_binary(h.copy())
    center = N - 1
    best_h = h.copy()
    best_c = compute_c5(h)
    corr = compute_corr(h)
    peak = float(np.max(corr))
    T = T0_factor * peak
    top_lags = np.argsort(corr)[-K:]
    top_m = (top_lags - center).astype(int)
    top_R = corr[top_lags].copy()
    it = 0
    while time.time() < deadline and T > 1e-8:
        it += 1
        if it % 200 == 0:
            corr = compute_corr(h)
            peak = float(np.max(corr))
            c = peak * 2.0 / N
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
            top_lags = np.argsort(corr)[-K:]
            top_m = (top_lags - center).astype(int)
            top_R = corr[top_lags].copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        if a == b:
            continue
        est_new_peak = 0.0
        for k in range(K):
            m = top_m[k]
            d = 0.0
            if 0 <= a - m < N:
                d += h[a - m]
            if 0 <= a + m < N:
                d -= h[a + m]
            if 0 <= b - m < N:
                d -= h[b - m]
            if 0 <= b + m < N:
                d += h[b + m]
            v = top_R[k] + d
            if v > est_new_peak:
                est_new_peak = v
        delta = est_new_peak - peak
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
            h[a], h[b] = 0.0, 1.0
        T *= cooling
    corr = compute_corr(h)
    peak = float(np.max(corr))
    c = peak * 2.0 / N
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def cont_opt(h, deadline, k=25, lr0=0.015, mom=0.88):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 50:
            lr0 *= 0.8
            stale = 0
            if lr0 < 5e-6:
                break
        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            M = abs(int(m) - center)
            if M == 0:
                grad += (1.0 - 2.0 * h)
            elif M < N:
                hi_idx = np.arange(M, N)
                lo_idx = np.arange(0, N - M)
                grad[hi_idx] += (1.0 - h[hi_idx - M])
                grad[lo_idx] -= h[lo_idx + M]
        grad /= k
        h_new = project(h + mom * v - lr0 * grad, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c
