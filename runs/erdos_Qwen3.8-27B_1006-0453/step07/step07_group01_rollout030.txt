import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N, seed=77):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def downsample_binary(h, N_old, N_new, seed=42):
    h_bin = ensure_binary(h, N_old, seed)
    x_old = (np.arange(N_old) + 0.5) * (N_new / N_old)
    x_new = np.arange(N_new) + 0.5
    vals = np.interp(x_new, x_old + (N_new / N_old - 1) * np.arange(0, N_old, N_new / N_new if N_new > N_old else 1), h_bin)
    h_interp = np.interp(np.linspace(0, N_old - 1, N_new), np.linspace(0, N_old - 1, N_old), h_bin)
    return ensure_binary(h_interp, N_new, seed)

def thue_morse_binary(N, offset=0):
    t = np.zeros(N)
    for i in range(N):
        c = 0
        v = i + offset
        while v:
            c += v & 1
            v >>= 1
        t[i] = c & 1
    return ensure_binary(t, N, seed=1)

    t = np.zeros(N)
    for i in range(N):
        t[i] = 1.0 if ((i * a + b) % N) < N * frac else 0.0

    vm_base = (pm_base >= 0) & (pm_base < N)

    pm_c_base = np.clip(pm_base, 0, N - 1)
