    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    if N0 % 2 != 0:
        N0 -= 1
        h0 = h0[:N0]
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: short continuous optimization
    d1 = min(deadline - 300, t0 + 60)
    h1 = project(best_h + np.random.randn(N0) * 0.05, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    # Phase 2: main binary SA
    d2 = min(deadline - 180, time.time() + 280)
    hb = fix_binary(best_h)
    hb, c2 = binary_sa(hb, d2, 6000, 25, 50, T0=0.45, T_end=0.004)
    if c2 < best_c:
        best_c, best_h = c2, hb.copy()
    # Phase 3: multi-resolution
    for N_try in [440, 520]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d3 = min(deadline - 50, time.time() + 65)
        h_bt, c3 = binary_sa(h_bt, d3, 2500, 22, 40, T0=0.35, T_end=0.008)
        if c3 < best_c:
            best_c, best_h = c3, h_bt.copy()
    # Phase 4: final polish at best resolution
    if time.time() < deadline - 15:
        if len(best_h) != N0:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, N0, endpoint=False)
            best_h = fix_binary(np.interp(x_new, x_old, best_h))
            best_c = compute_c5(best_h)
        d4 = min(deadline - 3, time.time() + 55)
        hB = fix_binary(best_h)
        hB, c4 = binary_sa(hB, d4, 3500, 28, 55, T0=0.2, T_end=0.003)
        if c4 < best_c:
            best_c, best_h = c4, hB.copy()
    # Ensure binary
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < best_c:
        best_h, best_c = hB, cB
    return best_h, best_c, len(best_h)
