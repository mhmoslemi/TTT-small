    v = np.zeros(N)
    b1, b2 = 0.85, 0.999
    b1c = b2c = 1.0
    it = 0
    lr = lr0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        if it % 100 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k+10, len(cs)-1)]
            if gap < 0.003:
                T = min(400.0, T * 1.1)
            elif gap > 0.04:
                T = max(10.0, T * 0.93)
        grad = soft_grad(corr, h, N, k, T)
        it += 1
        b1c = 1.0 - (1.0-b1)**it
        b2c = 1.0 - (1.0-b2)**it
        m = b1*m + (1.0-b1)*grad
        v = b2*v + (1.0-b2)*grad*grad
        h = h - lr * (m/b1c) / (np.sqrt(v/b2c) + 1e-8)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= 0.997
    return best_h, best_pk * (2.0/N)

def ensure_binary(h, N):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    tgt = N // 2
    if n1 == tgt:
        return hb
    rng = np.random.RandomState(77)
    if n1 > tgt:
        idx = rng.choice(np.where(hb==1)[0], n1-tgt, replace=False)
        hb[idx] = 0
    else:
        idx = rng.choice(np.where(hb==0)[0], tgt-n1, replace=False)
        hb[idx] = 1
    return hb

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return hp * vp.astype(float) - (1.0 - hm) * vm.astype(float)

def binary_SA(h_init, N, deadline, T0=0.5, Tcool=0.9997, max_steps=50000):
    h = ensure_binary(h_init, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    rng = np.random.RandomState(42)
    D = compute_D(h, N)
    best_h_bin = h.copy()
    best_c_val = best_c
    stall = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = int(ones[rng.randint(len(ones))])
        q = int(zeros[rng.randint(len(zeros))])
        new_c = c + D[p, :] - D[q, :]
        new_P = float(new_c.max())
        delta = new_P - float(c.max())
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-8)):
            h[p], h[q] = 0.0, 1.0
            c = new_c
            D = compute_D(h, N)  # This is O(N*2N) which is expensive
            if new_P < best_c_val - 1e-14:
                best_c_val = new_P
                best_h_bin = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T *= Tcool
        if stall > 200:
            # Perturb
            n_pert = min(3, len(ones)//10)
            for _ in range(n_pert):
                p2 = int(ones[rng.randint(len(ones))])
                q2 = int(zeros[rng.randint(len(zeros))])
                h[p2], h[q2] = 0.0, 1.0
            c = fft_corr(h)
            D = compute_D(h, N)
            stall = 0
    return best_h_bin, best_c_val * (2.0/N)
