    B = np.where((i_pk>=0) & (i_pk<N), h[np.clip(i_pk,0,N-1)], 0.0)
    return A - B

def evaluate_swaps(h, N, ones, zeros, corr, C_cache, sample_sz=200):

def golden_seq(N):

    h = np.zeros(N)
    t = 0

        c = 0
        while t:

            t >>= 1
        h[i] = 1 if c == 0 else 0

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def C_single(h, N, idx):
    kv = np.arange(2 * N - 1) - (N - 1)
    im = idx - kv
    ip = idx + kv
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    return A - B

def C_mat(h, N):
    return np.array([C_single(h, N, i) for i in range(N)])

def fix_binary(h, N):
    h = (h > 0.5).astype(float).copy()
    target = N // 2
    s = int(h.sum())
    if s > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, s - target, replace=False)] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - s, replace=False)] = 1
    return h

def local_search(h, N, deadline, sz=140, kick_t=55, kick_s=4, max_it=50000, tabu_sz=0):

    stuck = 0
    it = 0
    tabu = set()
    while it < max_it and time.time() < deadline:
        it += 1
        corr = fft_corr(h)
        P = float(corr.max()) * 2.0 / N
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sz)

        if tabu_sz > 0:
            for i in range(ns):

                    k = os_[i] * N + zs[j]
                    if k in tabu:
                        os_[i] = -1
                    if os_[i] >= 0:

            valid_o = np.where(os_ >= 0)[0]

                ns, nz = len(os_), len(zs)
        new = corr[None, None, :] + Cc[os_][None, :, :] + Cc[zs][:, None, :]
        nP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(nP), nP.shape)

def golden_init(N):
    h = np.zeros(N)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(N // 2):
        h[int((i * phi) % N)] = 1
    return h

def thue_morse_init(N):
    h = np.zeros(N)
    for i in range(N):
        t, c = i, 0
        while t:
            c ^= t & 1
            t >>= 1
        h[i] = 1.0 if c == 0 else 0.0
    return fix_binary(h, N)

def block_init(N):

        s = 2 * i * bl
        h[s:s + bl] = 1.0
    return fix_binary(h, N)

def nesterov_round(h, N, deadline, iters=300, lr=0.008):
    h = h.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    beta = 0.9
    for _ in range(iters):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -40)[-40:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= 40.0
        hn = np.clip(y - lr * grad, 0, 1)
        lo, hi = -2.0, 2.0
        for _ in range(30):
            mid = (lo + hi) / 2
            if np.clip(hn - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        hn = np.clip(hn - (lo + hi) / 2, 0, 1)
        m = beta * (hn - h)
        h = hn
    h_bin = fix_binary(h, N)
    return h_bin, compute_c5(h_bin, N)
