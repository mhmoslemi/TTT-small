import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N-1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:,None]
    k_vals = np.arange(2*N-1)[None,:] - (N-1)
    im = pos - k_vals
    ip = pos + k_vals
    A = np.where((im>=0)&(im<N), 1.0-h[np.clip(im,0,N-1)], 0.0)
    B = np.where((ip>=0)&(ip<N), h[np.clip(ip,0,N-1)], 0.0)
    return A - B

def compute_C_single(h, N, idx):
    k_vals = np.arange(2*N-1) - (N-1)
    im = idx - k_vals
    ip = idx + k_vals
    A = np.where((im>=0)&(im<N), 1.0-h[np.clip(im,0,N-1)], 0.0)
    B = np.where((ip>=0)&(ip<N), h[np.clip(ip,0,N-1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1.0)[0]
        if len(idx) >= n1-nh:
            h[np.random.choice(idx, n1-nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0.0)[0]
        if len(idx) >= nh-n1:
            h[np.random.choice(idx, nh-n1, replace=False)] = 1
    return h

def tabu_search(h, N, deadline, max_iters=10000, sample_sz=200, tabu_sz=100, kick_thresh=80, kick_sz=5):
    h = h.astype(float).copy()
    fix_sum(h, N//2)
    best_c = np.inf
    best_h = h.copy()
    C = compute_C_mat(h, N)
    stuck = 0
    tabu = {}  # (p,q) -> iteration
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]  # (ns, 2N-1)
        Cz = C[zs]  # (nz, 2N-1)
        new = corr[None,None,:] + Co[None,:,:] + Cz[:,None,:]  # (nz, ns, 2N-1)
        newP = new.max(axis=2)  # (nz, ns)
        # Apply tabu: set tabu moves to inf
        for iq in range(nz):
            for io in range(ns):
                if (os_[io], zs[iq]) in tabu:
                    newP[iq, io] = np.inf
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-14 and np.isfinite(mv):
            p, q = int(os_[mi[1]]), int(zs[mi[0]])
            h[p] = 0; h[q] = 1
            C[p,:] = compute_C_single(h, N, p)
            C[q,:] = compute_C_single(h, N, q)
            tabu[(p,q)] = it
            tabu[(q,p)] = it  # can't immediately reverse
            if len(tabu) > tabu_sz:
                # Remove oldest entries
                keys = sorted(tabu.keys(), key=lambda x: tabu[x])[:tabu_sz//2]
                for k in keys:
                    del tabu[k]
        elif stuck >= kick_thresh:
            for _ in range(kick_sz):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0; h[q] = 1
                C[p,:] = compute_C_single(h, N, p)
                C[q,:] = compute_C_single(h, N, q)
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            stuck = 0
        else:
            break
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=80, lr0=0.015):
    h = h.astype(float).copy()
    target = N/2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N-1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0*y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= k
        h_new = project_01(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it*0.0003)
    return best_h, best_pk*(2.0/N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    im = pos - k_vals
    ip = pos + k_vals
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    return A - B

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    im = idx - k_vals
    ip = idx + k_vals
    A = np.where((im >= 0) & (im < N), 1.0 - h[np.clip(im, 0, N - 1)], 0.0)
    B = np.where((ip >= 0) & (ip < N), h[np.clip(ip, 0, N - 1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        if len(idx) >= nh - n1:
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def tabu_search(h, N, deadline, max_iters=10000, sample_sz=150, tabu_len=80, kick_thresh=70, kick_sz=5):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    best_c = np.inf
    best_h = h.copy()
    C = compute_C_mat(h, N)
    stuck = 0
    tabu_arr = np.full((N, N), -100000, dtype=np.int32)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        new = corr[None, None, :] + Co[None, :, :] + Cz[:, None, :]
        newP = new.max(axis=2)
        tb = tabu_arr[np.ix_(os_, zs)].T
        newP[tb > it - tabu_len] = np.inf
        if not np.any(np.isfinite(newP)):
            newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-14 and np.isfinite(mv):
            p, q = int(os_[mi[1]]), int(zs[mi[0]])
            h[p] = 0
            h[q] = 1
            C[p, :] = compute_C_single(h, N, p)
            C[q, :] = compute_C_single(h, N, q)
            tabu_arr[p, q] = it
            tabu_arr[q, p] = it + 1
        elif stuck >= kick_thresh:
            for _ in range(kick_sz):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1
                C[p, :] = compute_C_single(h, N, p)
                C[q, :] = compute_C_single(h, N, q)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stuck = 0
        else:
            break
    return best_h, best_c

def block_perturb(h, N, nblocks=3, blen=4):
    h = h.copy()
    for _ in range(nblocks):
        start = np.random.randint(0, N - blen)
        seg = h[start:start+blen]
        n_ones_in_seg = int(seg.sum())
        h[start:start+blen] = 1.0 - seg
        delta = blen - 2 * n_ones_in_seg
        if delta > 0:
            idx = np.where(h == 0)[0]
            idx = idx[idx >= 0]
            pick = np.random.choice(idx, min(delta, len(idx)), replace=False)
            h[pick] = 1
        elif delta < 0:
            idx = np.where(h == 1)[0]
            pick = np.random.choice(idx, min(-delta, len(idx)), replace=False)
            h[pick] = 0
    fix_sum(h, N // 2)
    return h

def nesterov_opt(h, N, deadline, k=80, lr0=0.012, n_restarts=2):
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        hc = h + np.random.randn(N) * (0.06 + 0.04 * r)
        hc = project_01(hc, N * 0.5)
        m = np.zeros(N)
        lr = lr0 * (0.8 ** r)
        beta = 0.88
        sub_dl = deadline - max(5, (deadline - time.time()) / (n_restarts - r))
        it = 0
        while time.time() < sub_dl:
            y = h + m
            corr = fft_corr(y)
            pk = float(np.max(corr))
            if pk < best_pk:
                best_pk = pk
                best_h = h.copy()
            top = np.argpartition(corr, -k)[-k:]
            grad = np.zeros(N)
            center = N - 1
            for ai in top:
                L = int(ai - center)
                if L == 0:
                    grad += (1.0 - 2.0 * y)
                elif L > 0:
                    grad[L:] += (1.0 - y[:N - L])
                    grad[:N - L] -= y[L:]
                else:
                    M = -L
                    if M < N:
                        grad[:N - M] += (1.0 - y[M:])
                        grad[M:] -= y[:N - M]
            grad /= k
            h = project_01(y - lr * grad, N * 0.5)
            m = beta * (h - (y - m))
            it += 1
            if it % 150 == 0:
                lr *= 0.992
                beta = min(0.96, 0.85 + it * 0.0003)
        h = best_h.copy()
    return best_h, best_pk * (2.0 / N)

def full_1opt(h, N):
    C = compute_C_mat(h, N)
    corr = fft_corr(h)
    P = float(np.max(corr))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    improved = False
    for o in ones:
        best_mv = P
        best_q = -1
        for z in zeros:
            nv = float(np.max(corr + C[o] + C[z]))
            if nv < best_mv:
                best_mv = nv
                best_q = z
        if best_q >= 0 and best_mv < P - 1e-14:
            h[int(ones[int(np.where(ones == o)[0][0])])] = 0
            h[best_q] = 1
            C = compute_C_mat(h, N)
            corr = fft_corr(h)
            P = best_mv
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            improved = True
            break
    return h, float(np.max(fft_corr(h)))
