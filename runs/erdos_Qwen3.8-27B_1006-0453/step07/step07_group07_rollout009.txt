import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_top(h, corr, k=50):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def cont_opt(h, deadline, k=50, lr0=0.02, mom=0.90):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 50:
            lr0 *= 0.75
            if lr0 < 5e-5:
                break
            stale = 0
        g = grad_top(h, corr, k)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.85 * v + 0.25 * (h_new - h)
        h = h_new
    return best_h, best_c

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.random.permutation(np.where(h == 1)[0])
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.random.permutation(np.where(h == 0)[0])
        h[idx[:nh - n1]] = 1
    return h

def binary_sa(h, deadline, n_crit=35, s_sz=45, t_init=0.015, t_min=0.002, cool=0.997):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    c = float(np.max(corr)) * (2.0 / N)
    best_c = c
    best_h = h.copy()
    center = N - 1
    T = t_init
    no_imp = 0
    it = 0
    while time.time() < deadline:
        if no_imp > 600:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        top_lags = top_lags[top_lags != 0]
        if len(top_lags) == 0:
            break
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        cur_peak = float(np.max(corr))
        corr_vals = corr[center + top_lags]
        dl_ones = np.zeros((ns, len(top_lags)))
        dl_zeros = np.zeros((nz, len(top_lags)))
        for xi, oi in enumerate(os_):
            for li, L in enumerate(top_lags):
                d = 0.0
                if 0 <= oi + L < N:
                    d -= h[oi + L]
                if 0 <= oi - L < N:
                    d += (1.0 - h[oi - L])
                dl_ones[xi, li] = d
        for xj, zi in enumerate(zs):
            for li, L in enumerate(top_lags):
                d = 0.0
                if 0 <= zi + L < N:
                    d -= h[zi + L]
                if 0 <= zi - L < N:
                    d += (1.0 - h[zi - L])
                dl_zeros[xj, li] = d
        new_vals = corr_vals[None, None, :] + dl_ones[:, None, :] + dl_zeros[None, :, :]
        new_peaks = new_vals.max(axis=2)
        idx_best = np.unravel_index(np.argmin(new_peaks), new_peaks.shape)
        min_peak = new_peaks[idx_best]
        accept = False
        if min_peak < cur_peak - 1e-12:
            accept = True
        elif T > t_min:
            if np.random.random() < np.exp(-(min_peak - cur_peak) / T):
                accept = True
        if accept:
            h[os_[idx_best[0]]] = 0
            h[zs[idx_best[1]]] = 1
            corr = compute_corr(h)
            c = float(np.max(corr)) * (2.0 / N)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cool
        it += 1
    return best_h, best_c

def binary_sa_restart(h, deadline, n_restarts=3, n_crit=35, s_sz=45):
    best_h = h.copy()
    best_c = compute_c5(h)
    per = max(20, (deadline - time.time()) / n_restarts - 5)
    for r in range(n_restarts):

        if r == 0:
            h_init = best_h.copy()
        else:
            h_init = best_h.copy()
            n1 = int(h_init.sum())

            idx1 = np.random.choice(np.where(h_init == 1)[0], swap_n, replace=False)
            idx0 = np.random.choice(np.where(h_init == 0)[0], swap_n, replace=False)
            h_init[idx1] = 0
            h_init[idx0] = 1
        d = time.time() + per
        h_r, c_r = binary_sa(h_init, d, n_crit=n_crit, s_sz=s_sz,
                             t_init=0.02, t_min=0.003, cool=0.996)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
    return best_h, best_c
