```python
def adam_opt(h, N, deadline, k_start=30, k_max=120, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        t += 1
        # Adaptive k: increase when stuck
        k = min(k_max, k_start + t // 200)
        grad = grad_topk(h, corr, k)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        h = h_new
    return best_h, best_pk * (2.0 / N)
```

Actually, let me try a different idea: **subgradient method with exponential weighting of top shifts**. Instead of using a fixed set of top-k shifts, use ALL shifts but weight them exponentially by their correlation value. This creates a smooth approximation of the max function.

f(h) = max_k corr_k(h)

Smooth approximation: f_eps(h) = eps * log(sum_k exp(corr_k / eps))

The gradient of this is: sum_k (corr_k' * exp(corr_k/eps)) / sum_k exp(corr_k/eps)

For small eps, this focuses on the top shifts. For larger eps, it's more balanced.

This could be more effective than the top-k approach because it naturally weights all near-max shifts.

But computing this requires the full correlation vector and exponentials, which is O(N) per step - same as top-k.

```python
def grad_smooth(h, corr, eps=0.5):
    N = len(h)
    center = N - 1
    # Softmax weights
    shifted = corr / eps
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    # Now compute weighted gradient
    grad = np.zeros(N)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-10:
            continue
        s = int(k_idx - center)
        wj = w[k_idx]
        if s == 0:
            grad += wj * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wj * (1.0 - h[:N - s])
            grad[:N - s] -= wj * h[s:]
        else:
            M = -s
            if M < N:
                grad[:N - M] += wj * (1.0 - h[M:])
                grad[M:] -= wj * h[:N - M]
    return grad
```

corr_k(h) = sum_i h[i] * (1-h[i+k])

This is NOT linear in h. It's bilinear (product of two linear terms). So the objective is NOT convex.

However, if we fix some variables and linearize... hmm.

Actually, for the binary case (h ∈ {0,1}), the objective is:
max_k sum_i h[i](1-h[i+k]) = max_k (sum_i h[i] - sum_i h[i]h[i+k])

Since sum_i h[i] = N/2 is fixed, minimizing max_k sum_i h[i](1-h[i+k]) is equivalent to maximizing min_k sum_i h[i]h[i+k].

sum_i h[i]h[i+k] is the autocorrelation of h at lag k. So we want to maximize the minimum autocorrelation!

This is an interesting reformulation. We want a binary sequence with N/2 ones that has a flat (high) autocorrelation.

For the continuous case, it's different because h[i](1-h[i+k]) is not linear.

- Binary sequence of length N with exactly N/2 ones
- Maximize the minimum autocorrelation (over all lags)

This is related to optimal binary sequences / Barker sequences / low autocorrelation sequences. The dual problem (minimizing the maximum autocorrelation) is well-studied in signal processing.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, corr, k):
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    top = np.argpartition(corr, -k)[-k:]
    w = np.maximum(corr[top], 0)
    wn = w.sum()
    if wn < 1e-15:
        w = np.ones(k) / k
    else:
        w = w / wn
    for j in range(k):
        s = int(top[j] - center)
        wj = w[j]
        if wj < 1e-12:
            continue
        if s == 0:
            grad += wj * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wj * (1.0 - h[:N - s])
            grad[:N - s] -= wj * h[s:]
        else:
            M = -s
            if M < N:
                grad[:N - M] += wj * (1.0 - h[M:])
                grad[M:] -= wj * h[:N - M]
    return grad

def ensure_sum_binary(h, N):
    nh = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(777 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp
