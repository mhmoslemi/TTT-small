```python
def fast_greedy(h, N, deadline, top_M=30, max_iters=5000):
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    target_shifts = np.where(c > P - 0.1 * P)[0]  # shifts within 10% of max
    if len(target_shifts) > top_M:
        top_vals = np.sort(c)[-top_M:]
        threshold = top_vals[0]
        target_shifts = np.where(c >= threshold - 1e-10)[0]
        if len(target_shifts) > top_M:
            target_shifts = np.argpartition(c, -top_M)[-top_M:]
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Recompute D periodically
        if it % 50 == 0:
            D = compute_D(h, N)
            # Update target shifts
            top_vals = np.sort(c)[-top_M:]
            target_shifts = np.where(c >= top_vals[0] - 1e-10)[0]
        
        Dl = D[:, target_shifts]  # (N, M)
        cl = c[target_shifts]  # (M,)
        
        # Best swap: minimize max over target shifts of (cl + D[q,t] - D[p,t])
        best_P = P + 1
        best_pq = None
        for p in ones:
            base = cl - Dl[p]  # (M,)
            # For each q: new_cl = base + Dl[q], new_P = max(new_cl)
            # Vectorize: (|zeros|, M)
            cand = base[None, :] + Dl[zeros]  # (|zeros|, M)
            npk = cand.max(axis=1)  # (|zeros|,)
            mi = int(np.argmin(npk))
            if npk[mi] < best_P:
                best_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None or best_P >= P - 1e-12:
            # Stuck, do a random swap
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = fft_corr(h)
            P = float(c.max())
            continue
        
        # Apply swap
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        # Update
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
    
    return best_h, best_c * (2.0/N)
```

Let me also add the "continuous" Nesterov phase which the previous code had. The idea is:
- Allow h to take values in [0,1] (not just 0/1)
- Use gradient descent on the smooth approximation of max(c)
- Project onto [0,1]^N and sum = N/2 at each step
- This can find solutions with non-binary values that have lower max correlation

The gradient of c with respect to h:
c[k] = Σ_i h[i] · (1 - h[i+s_k])
∂c[k]/∂h[j] = (1-h[j+s_k]) - h[j-s_k] (for the j appearing as the first factor)
            + (from the (1-h[i+s_k]) term when j = i+s_k, i.e., i = j-s_k: -h[j-s_k])

Wait, let me be more careful:
c[k] = Σ_i h[i] - Σ_i h[i]·h[i+s_k]
∂c[k]/∂h[j] = 1 - h[j+s_k] - h[j-s_k] (where h[·] = 0 out of range)

So the gradient of max_k c[k] at h is:
If k* = argmax: ∇max = ∇c[k*] = 1 - h·[s*+1] - h·[-s*+1] (appropriately shifted)

For the top-K softmax: S_β = (1/β)log(Σ exp(β c[k]))
∇S_β = Σ_k softmax(β c)[k] · ∇c[k]

This gives a weighted average of the gradients of the top shifts.

```python
def nesterov_smooth(h, N, deadline, k=40, lr0=0.01, beta_sm=5.0):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_P = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        P = float(np.max(c))
        if P < best_P - 1e-14:
            best_P = P
            best_h = h.copy()
        # Softmax weights
        bc = beta_sm * c
        bc -= bc.max()
        w = np.exp(bc)
        w /= w.sum()
        # Top-k selection
        top = np.argpartition(c, -k)[-k:]
        wt = w[top]
        wt /= wt.sum()
        # Gradient: for each top shift, compute ∂c[k]/∂h
        # ∂c[k]/∂h[j] = 1 - h[j+s_k] - h[j-s_k]
        shifts = np.arange(2*N-1)
        top_shifts = (top - (N-1)).astype(int)
        grad = np.zeros(N)
        for w_i, s in zip(wt, top_shifts):
            # h[j+s] for j=0..N-1: valid for j=0..N-1-s if s>=0
            if s >= 0:
                h_plus = np.zeros(N)
                h_plus[:N-s] = y[s:]  # h[j+s] for j=0..N-1-s
                h_minus = np.zeros(N)
                h_minus[s:] = y[:N-s]  # h[j-s] for j=s..N-1
            else:
                s2 = -s
                h_plus = np.zeros(N)
                h_plus[s2:] = y[:N-s2]  # h[j+s] = h[j-s2] for j=s2..N-1
                h_minus = np.zeros(N)
                h_minus[:N-s2] = y[s2:]  # h[j-s] = h[j+s2] for j=0..N-1-s2
            grad += w_i * (1.0 - h_plus - h_minus)
        # Project gradient descent step
        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        s_val = h_new.sum()
        if abs(s_val - target) > 1e-10:
            # Bisection to fix sum
            lo, hi = -1.0, 1.0
            for _ in range(30):
                mid = (lo+hi)*0.5
                if np.clip(h_new-mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo+hi)*0.5, 0, 1)
        m = 0.9 * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.98
    return best_h, best_P * (2.0/N)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_bin(h, N, seed=42):
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    rng = np.random.RandomState(seed)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    elif n1 < target:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    ip = pos + sv
    im = pos - sv
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    ip_c = np.clip(ip, 0, N - 1)
    im_c = np.clip(im, 0, N - 1)
    h_ip = np.where(vp, h[ip_c], 0.0)
    h_im = np.where(vm, h[im_c], 0.0)
    return (1.0 - h_ip) - h_im

    target = N / 2.0
    best_h = h.copy()
    best_P = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        bc = beta_sm * c
        bc -= bc.max()
        w = np.exp(bc)
        w /= w.sum()
        top = np.argpartition(c, -k)[-k:]
        wt = w[top].copy()
        wt /= wt.sum()
        top_shifts = (top - (N - 1)).astype(int)
        grad = np.zeros(N)
        for w_i, s in zip(wt, top_shifts):
            if s >= 0:
                hp = np.zeros(N)
                if s < N:
                    hp[:N - s] = y[s:]
                hm = np.zeros(N)
                if s < N:
                    hm[s:] = y[:N - s]
            else:

                if s2 < N:
                    hp[s2:] = y[:N - s2]

                    hm[:N - s2] = y[s2:]

        P = float(np.max(fft_corr(h)))
        if P < best_P - 1e-14:

            for _ in range(40):
                mid = (lo+hi)*0.5
                if np.clip(hr-mid,0,1).sum() > N1/2:
                    lo = mid
                else:
                    hi = mid
            hr = np.clip(hr - (lo+hi)*0.5, 0, 1)
        d6 = min(deadline - 100, t0 + 570)
        h6, c5_6 = nesterov_smooth(hr, N1, d6, k=60, lr0=0.01, beta_sm=5.0)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Back to N0, diverse SA
    if time.time() < deadline - 80:
        h_s = ensure_sum_bin(upsample(best_h, len(best_h), N0), N0, seed=99)
        for _ in range(8):
            p = np.where(h_s == 1)[0]
            q = np.where(h_s == 0)[0]
            if len(p) > 0 and len(q) > 0:
                h_s[p[np.random.randint(len(p))]], h_s[q[np.random.randint(len(q))]] = 0.0, 1.0
        d7 = min(deadline - 50, t0 + 650)
        h7, c5_7 = sa_topk(h_s, N0, d7, top_M=50, T0=3.0, Tcool=0.994,
                           T_min=0.01, sf=1.0, kick_every=100, kick_size=5)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final fast greedy polish
    N_fin = len(best_h)
    if time.time() < deadline - 2:
        hF = ensure_sum_bin(best_h, N_fin, seed=55)
        sf_fin = 1.0 if N_fin <= N0 else 0.5
        hF2, c5F = fast_greedy(hF, N_fin, deadline, top_M=50, sf=sf_fin)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
