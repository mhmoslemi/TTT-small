def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.astype(float).copy()

    return h

def continuous_opt(h, N, deadline, lr0=0.006, temp0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    temp = temp0
    while time.time() < deadline - 2:
        corr = fft_corr(h)
        peak = np.max(corr)
        w = np.exp((corr - peak) / temp)
        w /= w.sum()
        center = N - 1
        grad = np.zeros(N)
        sig = np.where(w > 5e-5)[0]
        for k in sig:
            L = int(k - center)
            wt = w[k]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[L:] += wt * (1.0 - h[:N-L])
                grad[:N-L] -= wt * h[L:]
            elif -N < L < 0:
                M = -L
                grad[:N-M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N-M]
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) > 0.05:
            lo, hi = -2.0, 2.0
            for _ in range(45):
                mid = (lo + hi) / 2
                if np.clip(h - mid, 0, 1).sum() > target: lo = mid
                else: hi = mid
            h = np.clip(h - (lo + hi) / 2, 0, 1)
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c; best_h = h.copy()
        lr *= 0.9997
        temp *= 0.999
    return best_h, best_c
