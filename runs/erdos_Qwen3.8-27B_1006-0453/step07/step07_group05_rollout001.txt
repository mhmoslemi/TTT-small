import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def c5_of(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * 2.0 / N

def binarize(h, target):
    h = h.copy()
    n1 = int(np.round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h <= 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    h[h > 0.5] = 1
    h[h <= 0.5] = 0
    return h

def sa_search(h, N, deadline, T0=0.06, T1=1e-7, max_iters=500000):
    h = binarize(h, N // 2)
    corr = fft_corr(h)
    cur = float(np.max(corr))
    best_c = cur * 2.0 / N
    best_h = h.copy()
    T = T0
    total = min(max_iters, max(1, int((deadline - time.time()) * 12000)))
    cool = (T1 / T0) ** (1.0 / total)
    stall = 0
    stall_limit = 5000
    it = 0
    while time.time() < deadline and it < total:
        it += 1
        r = np.random.random()
        if r < 0.55:
            oi = np.random.randint(N)
            zi = np.random.randint(N)
            if h[oi] == h[zi] or oi == zi:
                T *= cool
                continue
            h[oi], h[zi] = h[zi], h[oi]
            nc = float(np.max(fft_corr(h)))
            dE = nc - cur
            if dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
                cur = nc
                h[oi], h[zi] = h[zi], h[oi]
                if cur * 2.0 / N < best_c - 1e-13:
                    best_c = cur * 2.0 / N
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[oi], h[zi] = h[zi], h[oi]
        elif r < 0.82:
            o1 = np.random.randint(N)
            o2 = np.random.randint(N)
            z1 = np.random.randint(N)
            z2 = np.random.randint(N)
            if h[o1] != 1 or h[o2] != 1 or h[z1] != 0 or h[z2] != 0:
                T *= cool
                continue
            if o1 == o2 or z1 == z2 or o1 == z1 or o1 == z2 or o2 == z1 or o2 == z2:
                T *= cool
                continue
            h[o1], h[o2], h[z1], h[z2] = h[z1], h[z2], h[o1], h[o2]
            nc = float(np.max(fft_corr(h)))
            dE = nc - cur
            if dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T)):
                cur = nc
                h[o1], h[o2], h[z1], h[z2] = h[z1], h[z2], h[o1], h[o2]
                if cur * 2.0 / N < best_c - 1e-13:

                    stall = 0
                else:
                    stall += 1
            else:

        else:

            if h[oi:oi+Lb].sum() != Lb or h[zi:zi+Lb].sum() != 0:
                T *= cool
                continue
            blk1 = h[oi:oi+Lb].copy()
            blk2 = h[zi:zi+Lb].copy()
            h[oi:oi+Lb] = blk2

            nc = float(np.max(fft_corr(h)))

                b = np.random.randint(N)

def ils_search(h, N, deadline):
    h = binarize(h, N // 2)
    best_c = c5_of(h)
    best_h = h.copy()
    perturb_r = 3
    while time.time() < deadline:
        sub_dl = min(deadline, time.time() + 0.05)
        h2 = h.copy()
        n_flip = perturb_r
        for _ in range(n_flip):
            a = np.random.randint(N)
            b = np.random.randint(N)
            if h2[a] != h2[b] and a != b:
                h2[a], h2[b] = h2[b], h2[a]
        h2, c2 = sa_search(h2, N, sub_dl, T0=0.03, T1=1e-6, max_iters=20000)
        if c2 < best_c - 1e-13:
            best_c = c2
            best_h = h2.copy()
            h = h2.copy()
            perturb_r = max(2, perturb_r - 1)
        else:
            perturb_r = min(15, perturb_r + 1)
    return best_h, best_c

def make_structured_init(N, variant):
    np.random.seed(9999 + variant * 7919)
    x = np.arange(N)
    if variant == 0:
        h = np.random.randint(0, 2, N).astype(float)
    elif variant == 1:
        h = (np.sin(2*np.pi*x*7.0/N) + 0.6*np.sin(2*np.pi*x*13.0/N) > 0).astype(float)
    elif variant == 2:

    elif variant == 3:
        tc = 0
        for bit in range(16):
            tc ^= ((x >> bit) * (x >> (bit+1))) & 1
        h = (tc & 1).astype(float)
    else:
        h = (np.cos(2*np.pi*x*3/N + 0.5) * np.cos(2*np.pi*x*8/N) > -0.1).astype(float)
    return binarize(h, N // 2)
