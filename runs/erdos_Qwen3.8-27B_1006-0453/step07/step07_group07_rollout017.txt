import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def weighted_subgrad(h, corr, k=60, beta=4.0):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    vals = corr[top_idx]
    w = np.exp(beta * (vals - vals.max()))
    w /= w.sum()
    grad = np.zeros(N)
    for idx, wi in zip(top_idx, w):
        L = int(idx) - center
        if abs(L) >= N:
            continue
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            Lp = -L
            grad[:N - Lp] += wi * (1.0 - h[Lp:])
            grad[Lp:] -= wi * h[:N - Lp]
    return grad

def cont_opt(h, deadline, k=60, lr0=0.018, mom=0.88, beta=4.0):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr0
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 50:
            lr *= 0.75
            stale = 0
            if lr < 0.0008:
                break
        grad = weighted_subgrad(h, corr, k, beta)
        h_new = project(h + mom * v - lr * grad, tgt)
        v = 0.9 * v + 0.5 * (h_new - h)
        h = h_new
    return best_h, best_c
