import numpy as np
import time
from scipy.optimize import minimize

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_row(h, N, idx):
    """Compute C[idx] - the change in correlation for flipping h[idx] from 0 to 1."""
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def compute_C_mat(h, N):
    C = np.zeros((N, 2*N-1))
    for i in range(N):
        C[i] = compute_C_row(h, N, i)
    return C

def fix_sum(h, nh):
    """Fix sum of h to be exactly nh."""
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1
    return h

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_row(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def compute_C_mat(h, N):
    C = np.zeros((N, 2*N-1))
    for i in range(N):
        C[i] = compute_C_row(h, N, i)
    return C

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1
    return h

def greedy_search(h, N, deadline, max_kick=8, kick_thresh=25, batch=80):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    C = compute_C_mat(h, N)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    stuck = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        Co = C[ones]
        Cz = C[zeros]
        best_newP = P + 1e-10
        best_oi = -1
        best_zi = -1
        n0 = len(zeros)
        
        for zs in range(0, n0, batch):
            ze = min(zs + batch, n0)
            nv = corr[None,None,:] + Co[None,:,:] + Cz[zs:ze][None,:,:]
            nm = nv.max(axis=2)
            fi = np.argmin(nm)
            oi, zi = np.unravel_index(fi, nm.shape)
            v = float(nm[oi, zi])
            if v < best_newP:
                best_newP = v
                best_oi = oi
                best_zi = zs + zi
        
        if best_oi >= 0 and best_newP < P - 1e-12:
            p = ones[best_oi]
            q = zeros[best_zi]
            h[p] = 0
            h[q] = 1
            C[p] = compute_C_row(h, N, p)
            C[q] = compute_C_row(h, N, q)
            corr = fft_corr(h)
            P = best_newP
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck >= kick_thresh:
                ks = min(max_kick, 4 + stuck // 20)
                for _ in range(ks):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0
                    h[q] = 1
                    C[p] = compute_C_row(h, N, p)
                    C[q] = compute_C_row(h, N, q)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
