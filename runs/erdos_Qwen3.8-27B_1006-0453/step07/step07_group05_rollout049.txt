import numpy as np
import time

def fft_corr(h):
    """Compute linear cross-correlation of h with (1-h) using FFT."""
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_binary(h, target):
    h = h.copy()
    n1 = int(np.sum(h > 0.5))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h <= 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:target-n1]] = 1
    return h

        # Gradient: p * corr^{p-1} * grad(corr) / K / f_p^{p-1}

            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient

        h_new = h - lr * grad
        # Project to [0,1] and fix sum

        if c_new < best_c:

        # Adapt parameters

                else:

    return best_h, best_c

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    idx = np.argsort(-h)

    out[idx[:target_ones]] = 1.0
    return out

def fix_sum(h, target):
