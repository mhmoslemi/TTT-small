import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N, seed=77):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > target:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def make_M(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    i = np.arange(N)[:, None]
    s = shifts[None, :]
    im = i - s
    ip = i + s
    vm = (im >= 0) & (im < N)
    vp = (ip >= 0) & (ip < N)
    im_c = np.clip(im, 0, N - 1)
    ip_c = np.clip(ip, 0, N - 1)
    hm = np.where(vm, h[im_c], 0.0)
    hp = np.where(vp, h[ip_c], 0.0)
    return (1.0 - hm) * vm.astype(float) - hp * vp.astype(float)

def binary_ls(h, N, deadline, rng, max_sweeps=5000, sample_n=200):

    M = make_M(h, N)
    best_c = float(corr.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(corr.max())
        if P < best_c - 1e-13:

            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        if n1 < 2 or n0 < 2:
            break
        Dz = M[zeros, :]
        best_new = P + 1.0
        best_pq = None
        if n1 > sample_n:
            p_idx = rng.choice(n1, sample_n, replace=False)
            p_list = ones[p_idx]
        else:
            p_list = ones
        for pi in range(len(p_list)):
            p = p_list[pi]
            base = corr - M[p, :]
            nm = (base[None, :] + Dz).max(axis=1)
            mi = int(np.argmin(nm))
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None and best_new < P - 1e-12:

def binary_sa(h, N, deadline, rng, T0=3.0, T1=0.02, max_iter=100000):
    h = h.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    cur_c = best_c
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max_iter)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        h[p], h[q] = 0.0, 1.0
        corr = fft_corr(h)
        new_c = float(corr.max())
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-9)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
            corr = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.014):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = project_sum(h, target)
    best_pk = float(np.max(fft_corr(best_h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = project_sum(y, target)
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M2 = -s
                grad[:N - M2] += wi * (1.0 - h[M2:])
                grad[M2:] -= wi * h[:N - M2]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.995
            beta = min(0.97, 0.85 + it * 0.0002)
        if it % 300 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(20.0, min(400.0, T * (0.88 if gap < 0.003 else 1.06)))
    return best_h
