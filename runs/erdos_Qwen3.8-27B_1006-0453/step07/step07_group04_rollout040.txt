```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    Lp = int(2 ** np.ceil(np.log2(L)))
    H = np.fft.rfft(h, Lp)
    G = np.fft.rfft(np.flip(1.0 - h), Lp)
    return np.real(np.fft.irfft(H * G, Lp))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c_k))) w.r.t. h"""
    N = len(h)
    L = 2 * N - 1
    c = fft_corr(h)
    c_shifted = c - c.max()
    exp_bc = np.exp(beta * c_shifted)
    w = exp_bc / exp_bc.sum()
    
    L2 = int(2 ** np.ceil(np.log2(3 * N)))
    f1mh = np.fft.rfft(1.0 - h, L2)
    fh = np.fft.rfft(h, L2)
    fw = np.fft.rfft(w, L2)
    
    term1 = np.real(np.fft.irfft(f1mh * np.conj(fw), L2))[:N]
    term2 = np.real(np.fft.irfft(fh * fw, L2))[:N]
    return (term1 - term2) / beta

def nesterov_smooth(h, N, deadline, beta0=20.0, lr0=0.03, max_iter=5000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx_val = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline and it < max_iter:
        y = h + m
        grad = smooth_grad(y, beta)
        h_new = project_sum(y - lr * grad, target)
        m = 0.9 * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            c = float(np.max(fft_corr(h)))
            if c < best_c:
                best_c = c
                best_h = h.copy()
            beta = min(200.0, beta * 1.1)
            lr *= 0.97
    c = float(np.max(fft_corr(h)))
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c * dx_val
```

```python
    N = len(h)
    L = 2*N - 1
    c = fft_corr(h)
    c_shift = c - c.max()
    w = np.exp(beta * c_shift)
    w /= w.sum()
    
    L2 = int(2 ** np.ceil(np.log2(N + L - 1)))  # ≥ 3N-2
    fw = np.fft.rfft(w, L2)
    f1h = np.fft.rfft(1.0 - h, L2)
    fh = np.fft.rfft(h, L2)
    
    g1 = np.real(np.fft.irfft(fw * np.conj(f1h), L2))[N-1:N-1+N]
    g2 = np.real(np.fft.irfft(fw * np.conj(fh), L2))[N-1:N-1+N]
    
    return (g1 - g2) / beta
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    Lp = int(2 ** np.ceil(np.log2(L)))
    H = np.fft.rfft(h, Lp)
    G = np.fft.rfft(np.flip(1.0 - h), Lp)
    return np.real(np.fft.irfft(H * G, Lp))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, beta):
    N = len(h)
    L = 2 * N - 1
    c = fft_corr(h)
    c_shift = c - c.max()
    w = np.exp(beta * c_shift)
    w /= w.sum()
    L2 = int(2 ** np.ceil(np.log2(3 * N)))
    fw = np.fft.rfft(w, L2)
    f1h = np.fft.rfft(1.0 - h, L2)
    fh = np.fft.rfft(h, L2)
    g1 = np.real(np.fft.irfft(f1h * np.conj(fw), L2))[N-1:2*N-1]
    g2 = np.real(np.fft.irfft(fh * fw, L2))[N-1:2*N-1]
    return (g1 - g2) / beta

def nesterov_smooth(h, N, deadline, beta0=15.0, lr0=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    dx_val = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        grad = smooth_grad(y, beta)
        h_new = project_sum(y - lr * grad, target)
        m = 0.9 * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            c = float(np.max(fft_corr(h)))
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
            beta = min(300.0, beta * 1.08)
            lr = max(0.001, lr * 0.995)
    c = float(np.max(fft_corr(h)))
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c * dx_val

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_sa(h, N, deadline, T0=3.0, Tcool=0.995, T_min=0.03, max_kicks=50):

    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    P = best_c
    T = T0
    stall = 0
    kick_count = 0
    while time.time() < deadline and kick_count < max_kicks:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        best_new_P = P + 1e-10
        best_pq = None
        # Sample a subset of ones for speed
        n_sample = min(len(ones), max(50, int(len(ones) * 0.8)))
        p_list = ones if n_sample >= len(ones) else np.random.choice(ones, n_sample, replace=False)
        for p in p_list:
            base = c - D[p, :]
            cand_max = base[None, :] - Dz
            npk = cand_max.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-12 or (T > T_min and np.random.random() < np.exp(-max(0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_c - 1e-13:
                best_c = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Random kick

            if stall > 80:

                    qq = zeros[np.random.randint(len(zeros))]
                    if pp != qq:
                        h[pp], h[qq] = 0.0, 1.0
                c = fft_corr(h)
                P = float(c.max())
                kick_count += 1
                stall = 0
                T = T0 * 0.7
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

    x_new = np.linspace(0, 2, N_new, endpoint=False)
```
