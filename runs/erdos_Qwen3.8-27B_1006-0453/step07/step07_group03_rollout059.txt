```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_shift(h, L, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:N-L])
        g[:N-L] -= h[L:]
    else:
        M = -L
        g[:N-M] += (1.0 - h[M:])
        g[M:] -= h[:N-M]
    return g

def adam_opt(h, N, deadline, tau=10.0, lr0=0.006, k_focus=40):
    h = h.astype(float).copy()
    target = N / 2.0

    m = np.zeros(N)
    v = np.zeros(N)

    b1, b2, eps = 0.8, 0.999, 1e-8
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    lr = lr0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        cmax = float(corr.max())
        if cmax < best_c - 1e-14:
            best_c = cmax
            best_h = h.copy()
        # Softmax weights on top k shifts
        top_k = np.argpartition(corr, -k_focus)[-k_focus:]
        ct = corr[top_k]

        v = b2 * v + (1 - b2) * grad * grad

        if t % 300 == 0:

            tau = min(25.0, tau * 1.03)
    return best_h, best_c * dx

            ones = np.where(h == 1)[0]
            h[rng.choice(ones, n1 - target_n, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, target_n - n1, replace=False)] = 1
    
    c_cur = fft_corr(h)
    best_c = float(c_cur.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Precompute D matrix (change for 1->0 at each position)
    def make_D(h):
        D = np.zeros((N, 2*N-1))
        for p in range(N):
            for L in range(-(N-1), N):
                if L == 0:
                    continue  # no change for binary flip at L=0
                k = L + N - 1
                val = 0.0
                if 0 <= p + L < N:
                    val += h[p + L]
                if 0 <= p - L < N:
                    val -= (1.0 - h[p - L])
                D[p, k] = val
        return D
```

Wait, this D computation is O(N^2 * N) = O(N^3) which for N=380 is about 55 million ops. That's too slow to recompute every iteration.

Let me use the vectorized version from the previous code:

```python
def make_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv  # p + L
    pm = pos - sv  # p - L
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N-1)
    pm_c = np.clip(pm, 0, N-1)
    hp = np.where(vp, h[pp_c], 0.0)
    hm = np.where(vm, h[pm_c], 0.0)
    D = hp - (1.0 - hm) * vm
    # Set L=0 column to 0 (no change for binary flip)
    D[:, N-1] = 0.0
    return D
```

This is O(N^2) memory and computation. For N=380, that's 380*759 ≈ 289K elements. Fine.

For a swap (p: 1→0, q: 0→1), the change in c[k] is:
- From p going 1→0: +D[p,k] (as computed above)
- From q going 0→1: -D[q,k] (negated)
- Total: D[p,k] - D[q,k]

So c_new[k] = c_cur[k] + D[p,k] - D[q,k]

We want to minimize max(c_new).

For 1-swap: try all (p, q) pairs, find the one that minimizes max(c_cur + D[p] - D[q]).

This is O(N_ones * N_zeros * (2N-1)) which for N=380 is about 190*190*759 ≈ 27M ops. That's doable per sweep but maybe slow for many sweeps.

The previous code uses a smarter approach: for each p, find the best q using vectorized ops. That's O(N_ones * N_zeros * (2N-1)) but with numpy vectorization.

Let me use a similar approach but with the correct sign:

```python
            h[p] = 0.0
            h[q] = 1.0

        elif best_type == 2:

            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
            c_cur = fft_corr(h)
        else:
            break  # local optimum

        if P_new < best_c - 1e-14:
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_shift(h, L, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:N - L])
        g[:N - L] -= h[L:]
    else:
        M = -L
        g[:N - M] += (1.0 - h[M:])
        g[M:] -= h[:N - M]
    return g

def make_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    hp = np.where(vp, h[pp_c], 0.0)
    hm = np.where(vm, h[pm_c], 0.0)
    D = hp - (1.0 - hm) * vm
    D[:, N - 1] = 0.0
    return D

def adam_opt(h, N, deadline, tau=10.0, lr0=0.006, k_focus=40):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    lr = lr0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        cmax = float(corr.max())
        if cmax < best_c - 1e-14:
            best_c = cmax
            best_h = h.copy()
        top_k = np.argpartition(corr, -k_focus)[-k_focus:]
        ct = corr[top_k]
        w = np.exp(tau * (ct - ct.max()))
        w /= w.sum()
        grad = np.zeros(N)
        for wi, idx in zip(w, top_k):
            L = int(idx - (N - 1))
            grad += wi * grad_shift(h, L, N)
        grad /= k_focus
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if t % 400 == 0:
            lr *= 0.96
            tau = min(28.0, tau * 1.03)
    return best_h, best_c * dx

def binary_ls(h, N, deadline, rng, max_iter=2000):
    h = (h >= 0.5).astype(float)
    target_n = N // 2
    n1 = int(h.sum())
    if n1 != target_n:
        if n1 > target_n:
            idx = np.where(h == 1)[0]
            h[rng.choice(idx, n1 - target_n, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[rng.choice(idx, target_n - n1, replace=False)] = 1
    c_cur = fft_corr(h)
    best_c = float(c_cur.max())
    best_h = h.copy()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = make_D(h, N)
        P_cur = float(c_cur.max())
        best_new_P = P_cur
        best_move = None
        best_type = 0
        for p in ones:
            base = c_cur + D[p]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-15:
                best_new_P = float(npk[mi])
                best_move = (int(p), int(zeros[mi]))
                best_type = 1
        if len(ones) >= 2 and len(zeros) >= 2:
            n_try = min(300, N)
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
                if new_P < best_new_P - 1e-15:
                    best_new_P = new_P
                    best_move = (p1, p2, q1, q2)
                    best_type = 2
        if best_type == 1:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            c_cur = fft_corr(h)
        elif best_type == 2:
            p1, p2, q1, q2 = best_move
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
            c_cur = fft_corr(h)
        else:
            break
        P_new = float(c_cur.max())
        if P_new < best_c - 1e-14:
            best_c = P_new
            best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def ils_binary(h, N, deadline, rng, n_rounds=6, pert_n=3):
    best_h, best_c = h.copy(), compute_c5(h, N)
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        cur = best_h.copy()
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        for _ in range(pert_n):
            if len(ones) < 2 or len(zeros) < 2:
                break
            cur[ones[rng.randint(len(ones))]] = 0.0
            cur[zeros[rng.randint(len(zeros))]] = 1.0
        hh, cc = binary_ls(cur, N, deadline, rng, max_iter=800)
        if cc < best_c:
            best_c, best_h = cc, hh
    return best_h, best_c

def upsample(h_old, N_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

        if c1 < best_c5:

    # Phase 2: ILS binary at N0

    # Phase 3: Adam at N0 again (finer)

        if c3 < best_c5:

    # Phase 4: Upsample to 2*N0, Adam + ILS

        d4a = t0 + budget_s * 0.62

    # Final evaluation
```
