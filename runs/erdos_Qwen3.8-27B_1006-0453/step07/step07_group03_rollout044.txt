        if len(ones) < 3 or len(zeros) < 3:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 30:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P_cur
        best_pq = None
        for p in p_list:
            base = c_cur - D[p, :]
            cand = base[None, :] - D_z
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        best_2p = None
        if len(ones) >= 4 and len(zeros) >= 4 and rng.random() < two_swap:
            n_try = min(120, max(40, N // 3))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_P = float((c_cur + D[p1] + D[p2] - D[q1] - D[q2]).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = None
                    best_2p = (p1, p2, q1, q2)
        if best_2p is not None:
            p1, p2, q1, q2 = best_2p
            h[p1] = h[p2] = 0.0
            h[q1] = h[q2] = 1.0
        elif best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            if stall > 20:
                for _ in range(4):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                stall = 0
            elif stall > 6:
                break
            else:
                stall += 1
                continue
        c_cur = fft_corr(h)
        P_cur = float(c_cur.max())
        if P_cur < best_c - 1e-13:
            best_c = P_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c * (2.0 / N)

def nesterov_sm(h, N, deadline, k=60, lr0=0.012, tau_mult=2.5, mom_decay=0.98):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        tau = tau_mult * dx * 2.0
        cc = corr[top] - corr[top].max()
        w = np.exp(cc / tau)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            wi = float(w[ai - top.min() if ai in top else np.where(top == ai)[0][0]])
            if wi < 1e-9:
                continue
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0 and L < N:
                g = np.zeros(N)
                g[L:] = 1.0 - y[:N - L]
                g[:N - L] -= y[L:]
                grad += wi * g
            else:
                M = -L
                if 0 < M < N:
                    g = np.zeros(N)
                    g[:N - M] = 1.0 - y[M:]
                    g[M:] -= y[:N - M]
                    grad += wi * g
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.982
            beta *= mom_decay
            beta = max(0.8, min(0.97, beta))
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary annealing at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 400, t0 + budget_s * 0.28)
    h1, c5_1 = binary_anneal(h_bin, N0, d1, max_sweeps=800, sample_frac=1.0, two_swap=0.35, rng=rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Nesterov continuous with softmax weights
    if time.time() < deadline - 300:
        hc = best_h + rng.randn(N0) * 0.035
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 260, time.time() + budget_s * 0.16)
        h2, c5_2 = nesterov_sm(hc, N0, d2, k=55, lr0=0.013, tau_mult=3.0, mom_decay=0.985)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Binary anneal at slightly larger N
    N1 = N0 + 20
    if time.time() < deadline - 220:
        h_up = ensure_sum(upsample(best_h, N0, N1), N1)
        d3 = min(deadline - 180, time.time() + budget_s * 0.14)
        h3, c5_3 = binary_anneal(h_up, N1, d3, max_sweeps=300, sample_frac=0.6, two_swap=0.4, rng=rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Nesterov at N0 with different perturbation
    if time.time() < deadline - 140:
        h_res = upsample(best_h, len(best_h), N0)
        hc2 = h_res + rng.randn(N0) * 0.06
        hc2 = project_sum(hc2, N0 / 2.0)
        d4 = min(deadline - 100, time.time() + budget_s * 0.12)
        h4, c5_4 = nesterov_sm(hc2, N0, d4, k=45, lr0=0.018, tau_mult=2.0, mom_decay=0.98)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Binary restart with perturbation
    if time.time() < deadline - 70:
        rng2 = np.random.RandomState(seed + 77)
        h_b2 = ensure_sum(best_h, len(best_h))
        Nb = len(h_b2)
        n_flip = 7
        ones_t = np.where(h_b2 == 1)[0]
        zeros_t = np.where(h_b2 == 0)[0]
        for _ in range(n_flip):
            h_b2[ones_t[rng2.randint(len(ones_t))]] = 0.0
            h_b2[zeros_t[rng2.randint(len(zeros_t))]] = 1.0
        h_b2 = ensure_sum(h_b2, Nb)
        d5 = min(deadline - 40, time.time() + budget_s * 0.08)
        sf5 = 1.0 if Nb <= 400 else 0.5
        h5, c5_5 = binary_anneal(h_b2, Nb, d5, max_sweeps=350, sample_frac=sf5, two_swap=0.35, rng=rng2)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 5:
        h_res = upsample(best_h, len(best_h), len(best_h))
        hc3 = h_res + rng.randn(len(h_res)) * 0.02
        hc3 = project_sum(hc3, len(h_res) / 2.0)
        d6 = deadline - 1
        h6, c5_6 = nesterov_sm(hc3, len(h_res), d6, k=40, lr0=0.008, tau_mult=4.0, mom_decay=0.99)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
