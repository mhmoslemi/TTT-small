def sa_opt(h, N, deadline, T0=4.0, Tcool=0.994, T_min=0.03, sf=1.0,
           kick_every=120, kick_size=5, max_sweeps=10000, block_prob=0.1):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    sweep = 0
    while sweep < max_sweeps:
        if time.time() > deadline:
            break
        sweep += 1
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= kick_every:
            # Reheat and kick
            for _ in range(kick_size):
                p = np.where(h > 0.5)[0]
                q = np.where(h < 0.5)[0]
                if len(p) > 0 and len(q) > 0:
                    h[p[np.random.randint(len(p))]], h[q[np.random.randint(len(q))]] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            continue
        
        h_bin = (h > 0.5).astype(float)
        D = compute_D(h_bin, N)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Try block move with some probability
        if np.random.random() < block_prob and len(ones) > 3:
            # Find consecutive blocks of 1s
            diffs = np.diff(ones)
            breaks = np.where(diffs > 1)[0] + 1
            blocks = np.split(ones, breaks)
            blocks = [b for b in blocks if 2 <= len(b) <= 10]
            if blocks:
                block = blocks[np.random.randint(len(blocks))]
                s = 1 if np.random.random() < 0.5 else -1
                new_start = block[0] + s
                new_end = block[-1] + s + 1
                if new_start >= 0 and new_end <= N and all(h_bin[new_start:new_end] == 0):
                    # Try the block shift
                    h_test = h_bin.copy()
                    h_test[block] = 0
                    h_test[new_start:new_end] = 1
                    c_test = c - D[block, :].sum(axis=0) + D[new_start:new_end, :].sum(axis=0)
                    P_test = float(c_test.max())
                    delta = P_test - P
                    if delta < 0 or np.random.random() < np.exp(-max(0, delta)/max(T, 1e-10)):
                        h_bin = h_test
                        h = h_bin
                    T = max(T_min, T * Tcool)
                    continue
        
        # Single swap
        n_s = len(ones) if sf >= 1.0 else max(50, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        Dz = D[zeros, :]
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            break
        
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta)/max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0/N)

def softmax_continuous(h, N, deadline, T0=1.0, Tcool=0.997, T_min=0.05, lr0=0.03, k=30):
    """Continuous optimization using softmax-regularized max gradient"""
    h = project_sum(h.copy(), N/2.0)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    T = T0
    lr = lr0
    beta = 0.9
    m = np.zeros(N)
    it = 0
    stall = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        c5_val = pk * (2.0/N)
        if c5_val < best_pk * (2.0/N) - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                T = max(T0*0.5, T*2.0)
                lr = lr0 * 0.8
                stall = 0
        
        # Softmax weights
        c_shift = c - c.max()
        w = np.exp(c_shift / T)
        w /= w.sum()
        
        # Gradient: sum_k w[k] * dc[k]/dh[j]
        # Use top-k approximation for efficiency
        top_idx = np.argpartition(c, -k)[-k:]
        w_top = w[top_idx]
        w_top /= w_top.sum()
        
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            shift = int(ai - center)
            wt = w_top[np.where(top_idx == ai)[0][0]]
            if shift == 0:
                grad += wt * (1.0 - 2.0*y)
            elif shift > 0:
                # c = sum_{i=0}^{N-1-shift} h[i]*(1-h[i+shift])
                # dc/dh[j] = (1-h[j+shift]) for j in [0, N-1-shift]
                #           + h[j-shift] for j in [shift, N-1]
                #           - 1 for j in [0, N-1-shift]
                idx_valid = np.arange(0, N-shift)
                grad[idx_valid] += wt * (1.0 - y[shift:N])
                if shift < N:
                    idx_cross = np.arange(shift, N)
                    grad[idx_cross] += wt * y[:N-shift]
                grad[idx_valid] -= wt
            else:
                s = -shift
                # c = sum_{i=s}^{N-1} h[i]*(1-h[i-s])
                idx_valid = np.arange(s, N)
                grad[idx_valid] += wt * (1.0 - y[:N-s])
                idx_cross = np.arange(0, N-s)
                grad[idx_cross] += wt * y[s:]
                grad[idx_valid] -= wt
        
        grad /= k
        
        h_new = project_sum(y - lr * grad, N/2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            T = max(T_min, T * Tcool)
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.0001)
    return best_h, best_pk * (2.0/N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)

    valid = np.zeros((N, 2 * N - 1), dtype=bool)
    for ki, s in enumerate(shifts):
        if s >= 0:
            valid[:N - s, ki] = True
        else:
            valid[-s:, ki] = True
    ip = i_r + s_r
    im = i_r - s_r

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum(h, N, seed=42):
    nh = N // 2
    h = np.array(h, dtype=float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    i_r = np.arange(N)[:, None]
    s_r = shifts[None, :]
    valid = np.zeros((N, 2 * N - 1), dtype=bool)
    for ki, s in enumerate(shifts):
        if s >= 0:
            valid[:N - s, ki] = True
        else:
            valid[-s:, ki] = True
    ip = i_r + s_r
    im = i_r - s_r
    hp = np.where((ip >= 0) & (ip < N) & valid, h[np.clip(ip, 0, N - 1)], 0.0)
    hm = np.where((im >= 0) & (im < N) & valid, h[np.clip(im, 0, N - 1)], 0.0)
    return np.where(valid, 1.0 - hp - hm, 0.0)

def upsample_fn(h_old, N_old, N_new):
    return np.interp(np.linspace(0, 2, N_new, endpoint=False),
                     np.linspace(0, 2, N_old, endpoint=False), h_old)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()

        P = float(c.max())

            best_c, best_h, stall = P, h.copy(), 0
        else:
            stall += 1
            if stall > 60:
                break
        ones, zeros = np.where(h > 0.5)[0], np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]

def softmax_cont(h, N, deadline, T0=1.2, Tcool=0.996, T_min=0.03, lr0=0.02, k=40):

    best_h, best_pk = h.copy(), float(np.max(fft_corr(h)))
    T, lr, beta, m, it, stall = T0, lr0, 0.9, np.zeros(N), 0, 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max()) * (2.0 / N)
        if pk < best_pk * (2.0 / N) - 1e-14:
            best_pk, best_h, stall = pk, h.copy(), 0
        else:
            stall += 1
            if stall > 200:
                T, lr, stall = max(T0 * 0.4, T * 1.5), lr0 * 0.8, 0
        cs = c - c.max()
        w = np.exp(cs / T)
        w /= w.sum()
        top = np.argpartition(c, -k)[-k:]
        wt = w[top].copy()
        wt /= wt.sum()
        grad = np.zeros(N)

            lr *= 0.995
