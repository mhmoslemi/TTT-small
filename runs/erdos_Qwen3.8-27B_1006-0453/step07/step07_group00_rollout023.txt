def subgrad_polish(h_in, deadline, k_top=30, lr=0.015):
    """Subgradient method with reweighting"""
    N = len(h_in)
    h = h_in.astype(float).copy()
    best_c = float(fft_corr(h).max()) * (2.0/N)
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max()) * (2.0/N)
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        # Weighted subgradient
        weights = np.zeros(len(top))
        c_vals = corr[top]
        w_sum = c_vals - c_vals.mean()
        w_sum = np.maximum(w_sum, 0.1)
        weights = w_sum / w_sum.sum()
        grad = np.zeros(N)
        for wi, t in zip(weights, top):
            L = int(t - (N - 1))
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N-L])
                grad[:N-L] -= wi * y[L:]
            else:
                M = -L
                grad[:N-M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N-M]
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N/2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
            beta = min(0.93, 0.80 + it * 0.0003)
    return best_h, best_c

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):

        if len(ones) < 2 or len(zeros) < 2:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp, bq = vi[mask], (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = min(6, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 10
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[None, None, :] + dp[ci:ce, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi]); bi, bj = mi[0]+ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        if best_new < best_c - 1e-12:
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0; deltas[p, p:p+N] = s
            deltas[q, :] = 0.0; deltas[q, q:q+N] = s
            best_c = float(corr.max()); best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
    return best_h, best_c

def subgrad_polish(h_in, deadline, k_top=35, lr=0.012):
    N = len(h_in)
    h = h_in.astype(float).copy()
    best_c = float(fft_corr(h).max()) * (2.0/N)
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.83
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max()) * (2.0/N)
        if pk < best_c:
            best_c = pk; best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        c_vals = corr[top]
        excess = np.maximum(c_vals - c_vals.mean() + 1e-6, 0.05)
        weights = excess / excess.sum()
        grad = np.zeros(N)
        for wi, t in zip(weights, top):
            L = int(t - (N - 1))
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N-L])
                grad[:N-L] -= wi * y[L:]
            else:
                M = -L
                grad[:N-M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N-M]
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N/2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.985
            beta = min(0.92, 0.80 + it * 0.0004)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    d1 = min(deadline - 450, t0 + 320)
    h1, _ = sa_optimize(h_bin, d1, n_peaks=7, n_per_peak=22, n_random=12,
                        cool_rate=0.999, t_start=0.055, no_impr_limit=6000)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    if time.time() < deadline - 350:
        hc = make_binary(best_h, N0)
        d2 = min(deadline - 300, time.time() + 90)
        h2, _ = hill_climb(hc, d2, n_peaks=12, n_per_peak=45)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 130)
        hc3 = best_h.astype(float) + np.random.randn(N0) * 0.035
        hc3 = np.clip(hc3, 0, 1)
        hc3 = np.clip(hc3 - (hc3.sum() - N0/2.0)/N0, 0, 1)
        h3, c3 = subgrad_polish(hc3, d3, k_top=40, lr=0.011)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    if time.time() < deadline - 180:
        h4 = make_binary(best_h, N0)
        d4 = min(deadline - 100, time.time() + 220)
        h4, _ = sa_optimize(h4, d4, n_peaks=9, n_per_peak=28, n_random=14,
                            cool_rate=0.9994, t_start=0.028, no_impr_limit=7000)
        c4 = compute_c5(h4)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    if time.time() < deadline - 60:
        h5 = make_binary(best_h, N0)
        d5 = min(deadline - 25, time.time() + 80)
        h5, _ = hill_climb(h5, d5, n_peaks=14, n_per_peak=55)
        c5 = compute_c5(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    for N_try in [400, 440, 500, 560]:
        if time.time() > deadline - 40:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, N_try)
        sub_dl = min(deadline - 10, time.time() + 40)
        hr, _ = sa_optimize(hb, sub_dl, n_peaks=6, n_per_peak=16, n_random=8,
                            cool_rate=0.999, t_start=0.04, no_impr_limit=4000)
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
