import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = np.array(h, dtype=float).copy()
    target = N // 2
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > target:
        # Remove excess ones: pick ones with lowest impact
        ones = np.where(h == 1)[0]
        # Sort by some heuristic - we'll just pick randomly for now
        to_remove = np.random.choice(ones, n1 - target, replace=False)
        h[to_remove] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        to_add = np.random.choice(zeros, target - n1, replace=False)
        h[to_add] = 1.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def targeted_sa(h_bin, deadline, n_peaks=6, n_per_peak=15, cool_rate=0.999, t_start=0.04, max_iters=200000):

    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s, deltas = build_deltas(h)
    T = t_start
    it = 0
    no_impr = 0
    total_iters_cap = max_iters

    while time.time() < deadline and it < total_iters_cap:
        it += 1
        if it % 600 == 0:
            T = max(T * cool_rate, 0.001)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        # Target the peaks
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp = vi[mask]
                bq = (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp = vi[mask]
                bq = (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())

        # Add random candidates
        nr = min(8, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())

        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break

        # Find best swap
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 8
        best_new = 1e10
        bi, bj = 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]

        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c

            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()

            deltas[p, :] = 0.0

            deltas[q, :] = 0.0

            if c < best_c - 1e-14:

            break

def nesterov_subgrad(h_in, N, deadline, lr=0.008, k_top=50, max_iters=5000):
    h = np.array(h_in, dtype=float).copy()
    m = np.zeros(N)
    beta = 0.85
    it = 0
    best_h = h.copy()
    best_c = np.inf
    lr_decay = 0.9995
    
    while time.time() < deadline and it < max_iters:
        y = h + m
        y = np.clip(y, 0, 1)
        corr = fft_corr(y)
        c = float(corr.max()) * (2.0 / N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Subgradient of max: focus on top peaks
        top = np.argpartition(corr, -k_top)[-k_top:]
        # Weight by how close to max
        max_val = corr[top].max()
        weights = np.exp((corr[top] - max_val) / 0.01)
        weights /= weights.sum()
        
        grad = np.zeros(N)
        for wi, t in zip(weights, top):
            L_off = int(t - (N - 1))
            if L_off == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L_off > 0:
                grad[L_off:] += wi * (1.0 - y[:N - L_off])
                grad[:N - L_off] -= wi * y[L_off:]
            else:
                M = -L_off
                grad[:N - M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N - M]
        
        # Nesterov step
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        # Project to mean constraint
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= lr_decay
        
    return best_h, best_c

def greedy_local_search(h_bin, max_rounds=5):

    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    s, deltas = build_deltas(h)

        improved = False
        ones = np.where(h == 1)[0]

        for p in ones:

                new_c = corr.max() + deltas[p] - deltas[q]
                nc = float(new_c.max())
                if nc < best_c - 1e-14:

                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, best_c * (2.0 / N)

                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp, bq = vi[mask], (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = np.random.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())
        nr = min(10, len(ones), len(zeros))
        cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
        cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        dp, dq = deltas[cand_p], deltas[cand_q]
        chunk = 8
        best_new, bi, bj = 1e10, 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new, bi, bj = float(nm[mi]), mi[0] + ci, mi[1]
        p, q = int(cand_p[bi]), int(cand_q[bj])
        dc = best_new - best_c
        if dc < -1e-14 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-12))):
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p].copy() - deltas[q].copy()
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0; deltas[p, p:p + N] = s
            deltas[q, :] = 0.0; deltas[q, q:q + N] = s
            c = float(corr.max())
            if c < best_c - 1e-14:
                best_c, best_h, no_impr = c, h.copy(), 0
            else:
                no_impr += 1
        else:
            no_impr += 1
        if no_impr > 6000 and T < 0.002:
            break
        if no_impr > 15000:
            break
    return best_h, best_c * (2.0 / N)

def greedy_ls(h_bin, max_rounds=20):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    s, deltas = build_deltas(h)
    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        dp, dq = deltas[ones], deltas[zeros]
        chunk = 25
        best_new, bi, bj = best_c, -1, -1
        for ci in range(0, len(ones), chunk):
            ce = min(ci + chunk, len(ones))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] - dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new, bi, bj = float(nm[mi]), mi[0] + ci, mi[1]
        if bi < 0 or best_new >= best_c - 1e-14:
            break
        p, q = int(ones[bi]), int(zeros[bj])
        h[p], h[q] = 0.0, 1.0
        corr += deltas[p].copy() - deltas[q].copy()
        s[p], s[q] = -s[p], -s[q]
        deltas[p, :] = 0.0; deltas[p, p:p + N] = s
        deltas[q, :] = 0.0; deltas[q, q:q + N] = s
        best_c = best_new
    return h, best_c * (2.0 / N)

def nesterov_sg(h_in, N, deadline, lr=0.006, k_top=60, max_iters=8000):
    h = np.array(h_in, dtype=float).copy()
    m = np.zeros(N)
    beta = 0.85
    it = 0
    best_h, best_c = h.copy(), np.inf
    lr_decay = 0.9997
    while time.time() < deadline and it < max_iters:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        c = float(corr.max()) * (2.0 / N)
        if c < best_c:
            best_c, best_h = c, h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        mv = corr[top].max()
        w = np.exp((corr[top] - mv) / 0.008)
        w /= w.sum()
        grad = np.zeros(N)
        for wi, t in zip(w, top):
            Lo = int(t - (N - 1))
            if Lo == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif Lo > 0:
                grad[Lo:] += wi * (1.0 - y[:N - Lo])
                grad[:N - Lo] -= wi * y[Lo:]
            else:
                M = -Lo
                grad[:N - M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N - M]
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= lr_decay
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0)
    best_h, best_c5 = h_bin.copy(), compute_c5(h_bin, N0)

    # Phase 1: Main SA
    d1 = min(deadline - 400, t0 + 220)
    h1, c1 = targeted_sa(h_bin, d1, n_peaks=7, n_per_peak=20, cool_rate=0.9988, t_start=0.05)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Greedy refinement
    if time.time() < deadline - 250:
        hg, cg = greedy_ls(best_h, max_rounds=30)
        if cg < best_c5:
            best_c5, best_h = cg, hg

    # Phase 3: Nesterov subgradient on continuous
    if time.time() < deadline - 200:
        d3 = min(deadline - 160, time.time() + 100)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, cn = nesterov_sg(hc, N0, d3, lr=0.005, k_top=50)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 4: SA from Nesterov result
    if time.time() < deadline - 120:
        h4 = make_binary(best_h, N0)
        d4 = min(deadline - 80, time.time() + 120)
        h4, c4 = targeted_sa(h4, d4, n_peaks=9, n_per_peak=22, cool_rate=0.9995, t_start=0.02)
        if c4 < best_c5:
            best_c5, best_h = c4, h4
        h4g, c4g = greedy_ls(h4, max_rounds=30)
        if c4g < best_c5:
            best_c5, best_h = c4g, h4g

    # Phase 5: Multi-resolution exploration
    for N_try in [340, 380, 400, 420, 460]:
        if time.time() > deadline - 50:
            break
        if N_try % 2 == 0 and N_try != N0:
            xo = np.linspace(0, 2, len(best_h), endpoint=False)
            xn = np.linspace(0, 2, N_try, endpoint=False)
            hi = np.interp(xn, xo, best_h)
            hb = make_binary(hi, N_try)
            sub_dl = min(deadline - 30, time.time() + 50)
            hr, cr = targeted_sa(hb, sub_dl, n_peaks=6, n_per_peak=15, cool_rate=0.999, t_start=0.035)
            hr, cr = greedy_ls(hr, max_rounds=20)
            if cr < best_c5:
                best_c5, best_h = cr, hr

    # Phase 6: Final SA + greedy on best
    Ny = len(best_h)
    if time.time() < deadline - 10:
        h6 = make_binary(best_h, Ny)
        d6 = min(deadline - 5, time.time() + 80)
        h6, c6 = targeted_sa(h6, d6, n_peaks=10, n_per_peak=25, cool_rate=0.9997, t_start=0.012)
        h6, c6 = greedy_ls(h6, max_rounds=40)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
