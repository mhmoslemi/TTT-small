    D = np.zeros((N, 2 * N - 1))

def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=40, beta_init=0.8):

    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = beta_init
    it = 0
    while time.time() < deadline:
        y = h + m
        yc = np.clip(y, 0, 1)
        corr = fft_corr(yc)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * yc
            elif L > 0:
                grad[L:] += (1.0 - yc[:N - L])
                grad[:N - L] -= yc[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - yc[M:])
                grad[M:] -= yc[:N - M]
        grad /= max(1, len(top))
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
            beta = min(0.95, beta_init + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
