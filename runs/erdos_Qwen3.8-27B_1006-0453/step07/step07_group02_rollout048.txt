def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    N = len(h)
    return float(np.max(get_corr(h))) * (2.0 / N)

def fix_sum(h, target_ones):
    n1 = int(h.sum())
    if n1 > target_ones:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target_ones, replace=False)] = 0
    elif n1 < target_ones:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target_ones - n1, replace=False)] = 1

def local_search(h, deadline, max_iters=30000, sample_sz=120, temp=0.0, kick_thresh=60, kick_sz=4):

    N = len(h)
    fix_sum(h, N // 2)
    corr = get_corr(h)
    L = len(corr)
    best_c5 = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()

        if time.time() > deadline:
            break

        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        nz = min(len(zeros), sample_sz)
        oi = np.random.choice(ones, ns, replace=False)
        zi = np.random.choice(zeros, nz, replace=False)
        # A[p,k] = h[p-k] - h[p+k] (effect of removing 1 at p)
        a_mk = oi[:, None] - lags[None, :]
        a_pk = oi[:, None] + lags[None, :]
        A = np.zeros((ns, L))
        vm = (a_mk >= 0) & (a_mk < N)
        A[vm] += h[np.clip(a_mk, 0, N - 1)[vm]]
        vp = (a_pk >= 0) & (a_pk < N)
        A[vp] -= h[np.clip(a_pk, 0, N - 1)[vp]]

def subgradient_opt(h, deadline, k_top=80, lr0=0.02, max_iters=5000):
    N = len(h)
    h = h.astype(np.float64).copy()
    target = N / 2.0
    best_c5 = c5_of(h)
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    for it in range(max_iters):
        if time.time() > deadline:
            break
        y = h + m
        corr = get_corr(y)
        pk = float(np.max(corr)) * (2.0 / N)
        if pk < best_c5 - 1e-14:
            best_c5 = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        s = h_new.sum()
        if abs(s - target) > 1e-10:
            lo, hi = -2.0, 2.0

        h = h_new

            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0003)
