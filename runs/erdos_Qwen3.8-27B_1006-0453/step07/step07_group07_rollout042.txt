Let me denote lag L = m - (N-1), so L ranges from -(N-1) to N-1.
corr[N-1+L] = Σ_i h[i] * (1-h[i+L]) where both indices are in [0,N-1].

If we swap: h[i] goes 1→0, h[j] goes 0→1:
- For h[i] changing from 1 to 0:
  - Term h[i]*(1-h[i+L]) changes: was 1*(1-h[i+L]), now 0 → change = -(1-h[i+L]) = h[i+L]-1
  - Term h[i-L]*(1-h[i]) changes: was h[i-L]*(1-1)=0, now h[i-L]*(1-0)=h[i-L] → change = +h[i-L]
  - Net for i: h[i+L] - 1 + h[i-L] (for valid indices)

- For h[j] changing from 0 to 1:

  - Term h[j-L]*(1-h[j]) changes: was h[j-L]*(1-0)=h[j-L], now h[j-L]*(1-1)=0 → change = -h[j-L]

Total change at lag L: (h[i+L]-1+h[i-L]) + (1-h[j+L]-h[j-L])
= h[i+L] + h[i-L] - h[j+L] - h[j-L]

(with appropriate zero-padding for out-of-range indices)

```python
def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5

        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]

        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_corr_top(h, corr, k=40):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def swap_delta(h, i, j, corr, top_lags, center):
    """Compute change in correlation at top_lags when swapping h[i] 1->0, h[j] 0->1"""
    N = len(h)
    hi, hj = int(h[i]), int(h[j])
    # For position i (going 1->0):
    # delta_i[L] = h[i+L] - 1 + h[i-L] (where valid)
    # For position j (going 0->1):
    # delta_j[L] = 1 - h[j+L] - h[j-L]
    # Total delta[L] = h[i+L] + h[i-L] - h[j+L] - h[j-L]
    new_corr = np.zeros(len(top_lags))
    for idx, L in enumerate(top_lags):
        d = 0.0
        ipL = i + L
        imL = i - L
        jpL = j + L
        jmL = j - L
        if 0 <= ipL < N: d += h[ipL]
        if 0 <= imL < N: d += h[imL]
        if 0 <= jpL < N: d -= h[jpL]
        if 0 <= jmL < N: d -= h[jmL]
        new_corr[idx] = corr[center + L] + d
    return np.max(new_corr)
```

```python
import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def grad_corr_top(h, corr, k=40):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def cont_opt(h, deadline, k=40, lr0=0.02, mom=0.88):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 60:
            lr0 *= 0.88
            stale = 0
            if lr0 < 1e-5:
                break
        g = grad_corr_top(h, corr, k)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def single_swap_peak(h, oi, ji, corr, top_lags, center):
    N = len(h)
    d_oi_ji = abs(oi - ji)
    peak = 0.0
    for L in top_lags:
        d = 0.0
        ipL, imL = oi + L, oi - L
        jpL, jmL = ji + L, ji - L
        if 0 <= ipL < N: d += h[ipL]
        if 0 <= imL < N: d += h[imL]
        if 0 <= jpL < N: d -= h[jpL]
        if 0 <= jmL < N: d -= h[jmL]
        if d_oi_ji == abs(int(L)):
            d += 1.0
        v = corr[center + L] + d
        if v > peak:
            peak = v
    return peak

def double_swap_peak(h, oi1, oi2, zi1, zi2, corr, top_lags, center):
    N = len(h)
    changed = {int(oi1): 0.0, int(oi2): 0.0, int(zi1): 1.0, int(zi2): 1.0}
    peak = 0.0
    for L in top_lags:
        d = 0.0
        affected = set()
        for p in changed:
            if 0 <= p < N:
                affected.add(p)
            ip = p - int(L)
            if 0 <= ip < N:
                affected.add(ip)
        for i in affected:
            ipL = i + int(L)
            if not (0 <= i < N and 0 <= ipL < N):
                continue
            old_i = h[i] if i not in changed else h[i]
            old_p = h[ipL] if ipL not in changed else h[ipL]
            new_i = changed.get(i, h[i])
            new_p = changed.get(ipL, h[ipL])
            d += new_i * (1.0 - new_p) - old_i * (1.0 - old_p)
        v = corr[center + L] + d
        if v > peak:
            peak = v
    return peak

def binary_sa(h, deadline, T0=0.05, T_min=0.0005, n_crit=30, max_iters=10000, use_double=True):
    N = len(h)
    h = fix_binary(h.copy())
    corr = compute_corr(h)
    cur_peak = float(np.max(corr))
    best_c = cur_peak * (2.0 / N)
    best_h = h.copy()
    center = N - 1
    it = 0
    no_imp = 0
    
    while time.time() < deadline and it < max_iters:
        frac = min(it / max_iters, 1.0)
        T = T0 * (T_min / T0) ** frac
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        
        # Decide single or double swap
        do_double = use_double and (len(ones) >= 2 and len(zeros) >= 2 and np.random.random() < 0.3)
        
        if do_double:
            o1, o2 = np.random.choice(ones, 2, replace=False)
            z1, z2 = np.random.choice(zeros, 2, replace=False)
            new_peak = double_swap_peak(h, o1, o2, z1, z2, corr, top_lags, center)
            delta = new_peak - cur_peak
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
                h[o1] = 0; h[o2] = 0
                h[z1] = 1; h[z2] = 1
        else:
            oi = int(np.random.choice(ones))
            ji = int(np.random.choice(zeros))
            new_peak = single_swap_peak(h, oi, ji, corr, top_lags, center)
            delta = new_peak - cur_peak
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
                h[oi] = 0; h[ji] = 1
        
        corr = compute_corr(h)
        cur_peak = float(np.max(corr))
        c = cur_peak * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        elif it < max_iters and no_imp > 400 and np.random.random() < 0.1:

            no_imp = 0
```

```python
def swap_change(h, oi, ji, L):
    """Compute change in C(L) when h[oi] goes 1->0 and h[ji] goes 0->1."""
    N = len(h)
    # Affected terms: i values where i or i+L is in {oi, ji}
    affected_i = set()
    for p in (oi, ji):
        if 0 <= p < N and 0 <= p + L < N:
            affected_i.add(p)  # p is the first index
        if 0 <= p - L < N and 0 <= p < N:
            affected_i.add(p - L)  # p-L is the first index, p is the second
    
    delta = 0.0
    for i in affected_i:
        ipL = i + L
        if not (0 <= i < N and 0 <= ipL < N):
            continue
        # Old values
        old_h_i = 1.0 if i == oi else (0.0 if i == ji else h[i])
        old_h_ipL = 1.0 if ipL == oi else (0.0 if ipL == ji else h[ipL])
        # New values
        new_h_i = 0.0 if i == oi else (1.0 if i == ji else h[i])
        new_h_ipL = 0.0 if ipL == oi else (1.0 if ipL == ji else h[ipL])
        delta += new_h_i * (1.0 - new_h_ipL) - old_h_i * (1.0 - old_h_ipL)
    return delta
```

This is O(1) per lag (at most 4 unique i values) and handles all interactions correctly! Let me use this approach.

```python
import numpy as np

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())

def grad_corr_top(h, corr, k=40):

            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

                peak_L = int(top_lags[0])
                cand = []
                for oi in ones:
                    t = oi + peak_L
                    if 0 <= t < N and h[t] == 0:
                        cand.append(oi)
                if len(cand) > 0:
                    oi = int(np.random.choice(cand))
                else:
                    oi = int(np.random.choice(ones))
            else:
                oi = int(np.random.choice(ones))
            ji = int(np.random.choice(zeros))
            new_peak = swap_peak_single(h, oi, ji, corr, top_lags, center)
            delta = new_peak - cur_peak
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-9)):
                h[oi] = 0; h[ji] = 1
        corr = compute_corr(h)
        cur_peak = float(np.max(corr))
        c = cur_peak * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 400 and np.random.random() < 0.15:
                T0 = max(T0 * 3.0, 0.08)
                no_imp = 0
        if no_imp > 1000:
            break
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # Phase 1: continuous optimization from perturbed initial
    d1 = min(deadline - 520, t0 + 85)
    h1 = project(h0 + np.random.randn(N0) * 0.12, N0 / 2.0)
    h1, c1 = cont_opt(h1, d1, k=35, lr0=0.022, mom=0.9)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    # Phase 2: binary SA with double swaps and guided moves
    if time.time() < deadline - 400:
        d2 = min(deadline - 390, time.time() + 140)
        hb = fix_binary(best_h)
        hb, c2 = binary_sa(hb, d2, T0=0.07, T_min=0.001, n_crit=40, max_iters=15000, use_double=True, guided_frac=0.35)
        if c2 < best_c:
            best_c, best_h = c2, hb.copy()
    # Phase 3: second continuous pass + binary
    if time.time() < deadline - 280:
        d3 = min(deadline - 270, time.time() + 55)
        h3 = project(best_h + np.random.randn(len(best_h)) * 0.05, len(best_h) / 2.0)
        h3, c3 = cont_opt(h3, d3, k=30, lr0=0.014, mom=0.92)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()
        if time.time() < deadline - 220:
            d4 = min(deadline - 210, time.time() + 75)
            hb4 = fix_binary(best_h)
            hb4, c4 = binary_sa(hb4, d4, T0=0.05, T_min=0.0006, n_crit=45, max_iters=12000, use_double=True, guided_frac=0.4)
            if c4 < best_c:
                best_c, best_h = c4, hb4.copy()
    # Phase 4: try alternate resolution
    for N_try in [400, 420]:
        if time.time() > deadline - 110:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_int)
        d5 = min(deadline - 90, time.time() + 40)
        h_bt, c5 = binary_sa(h_bt, d5, T0=0.05, T_min=0.001, n_crit=30, max_iters=5000, use_double=True, guided_frac=0.3)
        if c5 < best_c:
            best_c, best_h = c5, h_bt.copy()
    # Phase 5: final low-T polish
    if time.time() < deadline - 12:
        hB = fix_binary(best_h)
        d6 = min(deadline - 2, time.time() + 35)
        hB, c6 = binary_sa(hB, d6, T0=0.018, T_min=0.0001, n_crit=50, max_iters=8000, use_double=False, guided_frac=0.5)
        if c6 < best_c:
            best_c, best_h = c6, hB.copy()
    final_c = compute_c5(best_h)
    hB = fix_binary(best_h)
    cB = compute_c5(hB)
    if cB < final_c:
        best_h, final_c = hB, cB
    return best_h, final_c, len(best_h)
```
