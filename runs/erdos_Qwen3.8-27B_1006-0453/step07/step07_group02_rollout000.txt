def project_01(h, target):

        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):

    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]

            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)

def init_C(h, N):

    for i in range(N):
        C[i] = compute_C_single(h, N, i)

def local_search(h, N, deadline, max_iters, sample_sz, kick_thresh, kick_sz, use_tabu=True):

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)

        Cz = C[zs]

def guided_perturb(h, N, n_flips):

    corr = fft_corr(h)
    # Find worst shifts
    top_k = np.argpartition(corr, -10)[-10:]
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_flips):
        # Bias toward positions involved in worst shifts
        r = np.random.random()
        if r < 0.5 and len(ones) > 0:
            # Pick a 1-position near a worst shift
            k = int(np.random.choice(top_k))
            L = k - (N - 1)
            if L >= 0:
                candidates = [ones[(i + L) % N] for i in range(min(5, len(ones)))]
            else:
                candidates = [ones[(i - L) % N] for i in range(min(5, len(ones)))]
            if candidates:
                p = int(np.random.choice(candidates))
            else:

        h[p] = 0
        h[q] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

def nesterov_opt(h, N, deadline, k=80, lr0=0.01):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(np.max(corr))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, nh):
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        if len(idx) >= n1 - nh:
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        if len(idx) >= nh - n1:
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    return h

def compute_C_single(h, N, idx):
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def init_C(h, N):
    C = np.zeros((N, 2 * N - 1))
    for i in range(N):
        C[i] = compute_C_single(h, N, i)
    return C

def local_search(h, N, deadline, max_iters, sample_sz, kick_thresh, kick_sz):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    C = init_C(h, N)
    corr = fft_corr(h)
    P = float(np.max(corr))
    best_c = P
    best_h = h.copy()
    stuck = 0
    tabu_set = set()
    tabu_q = []
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        newP = fft_corr(h)[None, None, :] + C[os_][None, :, :] + C[zs][:, None, :]
        newP = newP.max(axis=2)
        flat = newP.ravel()
        order = np.argsort(flat)[:40]
        best_mi = None
        for oi in order:
            mi = np.unravel_index(oi, (ns, nz))
            key = (int(os_[mi[1]]), int(zs_[mi[0]]))
            if key not in tabu_set:
                best_mi = mi
                break
        if best_mi is None:
            best_mi = np.unravel_index(order[0], (ns, nz))
        mv = float(flat[order[0]])
        if mv < P - 1e-14 or (stuck > 100 and mv < P + 2.0):
            a = int(os_[best_mi[1]])
            b = int(zs_[best_mi[0]])
            h[a] = 0
            h[b] = 1
            C[a] = compute_C_single(h, N, a)
            C[b] = compute_C_single(h, N, b)
            tabu_q.append((a, b))
            tabu_set.add((a, b))
            if len(tabu_q) > 40:
                tabu_set.discard(tabu_q.pop(0))
            corr = fft_corr(h)
            P = float(np.max(corr))
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        elif stuck >= kick_thresh:
            for _ in range(kick_sz):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1
                C[p] = compute_C_single(h, N, p)
                C[q] = compute_C_single(h, N, q)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            P = float(np.max(corr))
            stuck = 0
            tabu_set.clear()
            tabu_q.clear()
        else:
            break
    return best_h, best_c

def guided_perturb(h, N, n_flips):
    h = h.copy()
    corr = fft_corr(h)
    top_k = np.argpartition(corr, -8)[-8:]
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_flips):
        p = int(np.random.choice(ones))
        q = int(np.random.choice(zeros))
        h[p] = 0
        h[q] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h

def nesterov_opt(h, N, deadline, k=60, lr0=0.009):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(np.max(corr))
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.992
            beta = min(0.96, 0.86 + it * 0.00015)
    return best_h, best_pk * (2.0 / N)
