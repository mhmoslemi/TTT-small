import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_top(h, corr, k=70):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)

            best_h = h.copy()
            stale = 0
        else:

        h = h_new
    return best_h, best_c

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def peak_contrib(h, corr, n_crit=50):
    """Compute per-position contribution to top peak lags."""
    N = len(h)
    center = N - 1
    top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
    top_lags = top_lags[top_lags != 0]
    contrib = np.zeros(N)
    for L in top_lags:
        La = abs(L)
        if La >= N:
            continue
        if L > 0:
            idx = np.arange(0, N - La)
            contrib[idx] += h[idx] * (1.0 - h[idx + La])
            idx2 = np.arange(La, N)
            contrib[idx2] += h[idx2] * (1.0 - h[idx2 - La])
        else:
            idx = np.arange(La, N)
            contrib[idx] += h[idx] * (1.0 - h[idx - La])
            idx2 = np.arange(0, N - La)
            contrib[idx2] += h[idx2] * (1.0 - h[idx2 + La])
    return contrib
