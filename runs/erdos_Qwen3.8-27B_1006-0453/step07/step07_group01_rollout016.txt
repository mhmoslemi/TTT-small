    P = float(c.max())
    best_P = P
    T = T0
    it = 0
    stall = 0
    L = 2*N - 1
    while time.time() < deadline:
        D1, D0 = compute_D(h, N)
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)<1 or len(zeros)<1:
            break
        best_dP = 0.0
        best_pq = None
        for p in ones:
            base = c + D1[p]
            cands = base[None,:] + D0[zeros]
            npk = cands.max(axis=1)
            mi = int(np.argmin(npk))
            dP = float(npk[mi]) - P
            if dP < best_dP:
                best_dP = dP
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            stall += 1
            if stall > 200:
                break
            continue
        accept = best_dP < 0 or (T > T_min and rng.random() < np.exp(-best_dP/T))
        if accept:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_P - 1e-14:
                best_P = P
                best_h = h.copy()
                stall = 0
        T *= cool
        it += 1
    return best_h, best_P*(2.0/N)

def cont_opt(h_init, N, deadline, T=80.0, lr=0.015):
    h = h_init.astype(float).copy()
    target = N/2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        # top-k gradient
        k = min(30, N)
        idx = np.argpartition(corr, -k)[-k:]
        ct = corr[idx]
        w = np.exp(T*(ct - ct.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N-1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi*(1.0-2.0*y)
            elif s > 0:
                grad[s:] += wi*(1.0-y[:N-s])
                grad[:N-s] -= wi*y[s:]
            else:
                M = -s
                grad[:N-M] += wi*(1.0-y[M:])
                grad[M:] -= wi*y[:N-M]
        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.997
            beta = min(0.98, 0.85+it*0.0001)
    return best_h, best_pk*(2.0/N)

def upsample(h, N_old, N_new):
    xo = np.linspace(0, 2, N_old, endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, xo, h)

def thue_morse_bin(N, rng):
    t = np.zeros(N, dtype=np.int32)
    for i in range(1, N):
        t[i] = t[i>>1] ^ (i & 1)
    h = t.astype(float)
    return ensure_binary(h, N, rng)
