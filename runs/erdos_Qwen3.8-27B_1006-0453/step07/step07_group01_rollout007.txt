import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(77)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def build_deltas(h, N):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    t_all = np.arange(2 * N - 1)[None, :]
    s_all = t_all - (N - 1)
    p_all = ones[:, None]
    pm = p_all - s_all
    pp = p_all + s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    D1 = -((1.0 - h[pm_c]) * vm.astype(np.float64)) + h[pp_c] * vp.astype(np.float64)
    q_all = zeros[:, None]
    qm = q_all - s_all
    qp = q_all + s_all
    vm2 = (qm >= 0) & (qm < N)
    vp2 = (qp >= 0) & (qp < N)
    qm_c = np.clip(qm, 0, N - 1)
    qp_c = np.clip(qp, 0, N - 1)
    Dz = ((1.0 - h[qm_c]) * vm2.astype(np.float64)) - h[qp_c] * vp2.astype(np.float64)
    return D1, Dz

def local_search_1swap(h, N, deadline, max_sweeps=2000):

    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            break
        D1, Dz = build_deltas(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_new_P = P
        best_pq = None
        for pi in range(len(ones)):

            cand = base[None, :] + Dz

    best_h = h.copy()

    n_zeros = len(zeros)

        batch = min(200, max_iters - it)
        for _ in range(batch):
            i1, i2 = np.random.choice(n_ones, 2, replace=False)
            j1, j2 = np.random.choice(n_zeros, 2, replace=False)
            new_c = c + D1[i1, :] + D1[i2, :] + Dz[j1, :] + Dz[j2, :]
            npk = float(new_c.max())
            if npk < P - 1e-14:
                p1, p2 = ones[i1], ones[i2]
                q1, q2 = zeros[j1], zeros[j2]
                h[p1] = h[p2] = 0.0
                h[q1] = h[q2] = 1.0
                c = new_c
                P = npk
                ones[~np.isin(ones, [p1, p2])] = ones[~np.isin(ones, [p1, p2])]
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                n_ones = len(ones)
                n_zeros = len(zeros)
                D1, Dz = build_deltas(h, N)

                    best_h = h.copy()

def simulated_anneal(h, N, deadline, t0=0.3, t1=0.01, n_iter=20000):
    h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    c = fft_corr(h)
    P = float(c.max())
    T = t0
    cool = (t1 / t0) ** (1.0 / n_iter)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cool
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        s_vals = np.arange(2 * N - 1) - (N - 1)
        pm = p - s_vals
        pp = p + s_vals
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        d1 = -((1.0 - h[pm_c]) * vm) + h[pp_c] * vp
        qm = q - s_vals
        qp = q + s_vals
        vm2 = (qm >= 0) & (qm < N)
        vp2 = (qp >= 0) & (qp < N)
        qm_c = np.clip(qm, 0, N - 1)
        qp_c = np.clip(qp, 0, N - 1)
        dz = ((1.0 - h[qm_c]) * vm2) - h[qp_c] * vp2
        new_c = c + d1 + dz
        npk = float(new_c.max())
        delta = npk - P
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
            h[p] = 0.0
            h[q] = 1.0
            c = new_c
            P = npk
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h_init, N, deadline, k=40, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.998
            beta = min(0.97, 0.85 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)
