import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_grad(h, corr, T, thresh=1e-13):
    N = len(h)
    ex = np.exp((corr - corr.max()) / T)
    tw = ex.sum()
    if tw < 1e-30:
        return np.zeros(N)
    w = ex / tw
    grad = np.zeros(N)
    omm = 1.0 - h
    for s in range(-(N - 1), N):
        ws = w[N - 1 + s]
        if ws < thresh:
            continue
        if s >= 0:
            nv = N - s
            if nv > 0:
                grad[:nv] += ws * omm[s:]
                grad[s:s + nv] -= ws * h[:nv]
        else:
            nv = N + s
            if nv > 0:
                grad[-nv:] += ws * omm[:nv]
                grad[:nv] -= ws * h[-nv:]
    return grad

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
