            qmk = q - k_vals
            Eq[i, (qk >= 0) & (qk < N)] += (1 - h[qk[(qk >= 0) & (qk < N)]])
            Eq[i, (qmk >= 0) & (qmk < N)] -= h[qmk[(qmk >= 0) & (qmk < N)]]
        new_max = (corr[None, None, :] + Ep[None, :, :] + Eq[:, None, :]).max(axis=2)
        mi = np.unravel_index(new_max.argmin(), new_max.shape)
        mv = float(new_max[mi])
        temp = temp0 * max(0.0, 1.0 - it / 2000.0)
        if mv < P - 1e-10:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
        elif temp > 0 and np.random.random() < temp:
            h[os_[np.random.randint(ns)]] = 0
            h[zs[np.random.randint(nz)]] = 1
        elif stuck >= kick_thresh:
            if use_freq_kick:
                h = freq_kick(h, N, kick_sz, 0.12)
            else:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(kick_sz):
                    h[np.random.choice(ones)] = 0
                    h[np.random.choice(zeros)] = 1
            stuck = 0
            corr = fft_corr(h)
            P = float(np.max(corr))
        else:
            break
    return best_h, best_c * (2.0 / N)

def swap2_search(h, N, deadline, kick_thresh=50, kick_sz=4):
    h = h.astype(float).copy()
    fix_sum(h, N // 2)
    P = float(np.max(fft_corr(h)))
    best_c = P
    best_h = h.copy()
    stuck = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        for attempt in range(6):
            p1, p2 = np.random.choice(ones, 2, replace=False)
            q1, q2 = np.random.choice(zeros, 2, replace=False)
            h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
            new_P = float(np.max(fft_corr(h)))
            if new_P < P - 1e-12:
                P = new_P
                improved = True
                break
            h[p1] = 1; h[p2] = 1; h[q1] = 0; h[q2] = 0
        if improved:
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
            stuck = 0
        else:
            stuck += 6
            if stuck >= kick_thresh:
                h = freq_kick(h, N, kick_sz, 0.10)
                stuck = 0
            else:
                break
    return best_h, best_c * (2.0 / N)

def nesterov_cont(h, N, deadline, k=50, lr0=0.008):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0 and L < N:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            elif L < 0:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.994
            beta = min(0.97, 0.88 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    d1 = min(deadline - 420, t0 + 280)
    h1 = (h0 > 0.5).astype(float)
    h1, c1 = swap1_search(h1, N0, d1, sample_sz=220, kick_thresh=60, kick_sz=5, temp0=0.15)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    if time.time() < deadline - 320:
        d2 = min(deadline - 300, time.time() + 140)
        h2 = (best_h > 0.5).astype(float)
        h2, c2 = swap2_search(h2, N0, d2, kick_thresh=45, kick_sz=4)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    if time.time() < deadline - 220:
        d3 = min(deadline - 200, time.time() + 100)
        hc = best_h + np.random.randn(N0) * 0.06
        hc = project_01(hc, N0 / 2.0)
        hc, c3 = nesterov_cont(hc, N0, d3, k=50, lr0=0.009)
        if c3 < best_c5:
            best_c5, best_h = c3, hc.copy()

    if time.time() < deadline - 170:
        h4 = (best_h > 0.5).astype(float)
        d4 = min(deadline - 150, time.time() + 130)
        h4, c4 = swap1_search(h4, N0, d4, sample_sz=250, kick_thresh=50, kick_sz=5, temp0=0.2, use_freq_kick=True)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    if time.time() < deadline - 120:
        d5 = min(deadline - 100, time.time() + 100)
        h5 = (best_h > 0.5).astype(float)
        h5, c5 = swap2_search(h5, N0, d5, kick_thresh=40, kick_sz=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    for N_try in [378, 382, 384, 376]:
        if time.time() > deadline - 60:
            break
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_b, best_h)
        h_bt = (h_int > 0.5).astype(float)
        fix_sum(h_bt, N_try // 2)
        d6 = min(deadline - 15, time.time() + 35)
        h6, c6 = swap1_search(h_bt, N_try, d6, sample_sz=N_try // 2, kick_thresh=35, kick_sz=4, temp0=0.1)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    if time.time() < deadline - 12:
        hF = (best_h > 0.5).astype(float)
        hF, cF = swap1_search(hF, len(best_h), deadline - 2, sample_sz=200, kick_thresh=30, kick_sz=3, temp0=0.05)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    hb = (best_h > 0.5).astype(float)
    fix_sum(hb, len(hb) // 2)
    cb = compute_c5(hb, len(hb))
    cc = compute_c5(best_h, len(best_h))
    if cb < cc:
        best_h, best_c5 = hb, cb
    return best_h, best_c5, len(best_h)
