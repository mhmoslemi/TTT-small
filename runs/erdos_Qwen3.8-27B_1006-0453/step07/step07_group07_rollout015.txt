import numpy as np
from scipy.optimize import linprog
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_lp_given_complement(g, N):
    """Solve min t s.t. corr_m(h, g) <= t for all m, 0<=h<=1, sum(h)=N/2"""
    center = N - 1
    n_vars = N + 1
    c_obj = np.zeros(n_vars)
    c_obj[N] = 1.0
    
    # Build A_ub: (2N-1) x (N+1)
    # For row m (0-indexed), l = m - center
    # Coefficient of h[i]: g[i-l] if 0<=i-l<N else 0
    A_ub = np.zeros((2*N-1, n_vars))
    for m in range(2*N - 1):
        l = m - center
        for i in range(N):
            j = i - l
            if 0 <= j < N:
                A_ub[m, i] = g[j]
        A_ub[m, N] = -1.0
    b_ub = np.ones(2*N - 1)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.array([N / 2.0])
    
    bounds = [(0, 1)] * N + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:N]
    return None

def alternating_lp(h_init, deadline, max_iters=100):
    N = len(h_init)
    h = project_box_sum(h_init.copy(), N / 2.0)
    g = 1.0 - h
    best_c = compute_c5(h)
    best_h = h.copy()
    prev_c = best_c
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = solve_lp_given_complement(g, N)
        if h_new is not None:
            h = h_new
        g_new = solve_lp_given_complement(h, N)
        if g_new is not None:
            g = g_new
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        if abs(prev_c - c) < 1e-10:
            break
        prev_c = c
    return best_h, best_c

def soft_max_gradient_step(h, tau=50.0, lr=0.01, n_steps=200, deadline=None):
    N = len(h)
    tgt = N / 2.0
    h = project_box_sum(h.copy(), tgt)
    best_c = compute_c5(h)
    best_h = h.copy()
    for step in range(n_steps):
        if deadline and time.time() > deadline:
            break
        corr = compute_corr(h)
        scaled = tau * corr
        mx = scaled.max()
        exp_s = np.exp(scaled - mx)
        p = exp_s / exp_s.sum()
        
        # Compute gradient
        center = N - 1
        grad = np.zeros(N)
        for i in range(N):
            # S_A: sum p[m]*(1-h[i-l_m]) where 0<=i-l_m<N
            # l_m = m - center
            # i - m + center in [0, N) => center+1-i <= m <= center+i
            m_lo = max(0, center + 1 - i)
            m_hi = min(2*N-2, center + i)
            if m_lo <= m_hi:
                ms = np.arange(m_lo, m_hi + 1)
                js = i - (ms - center)
                grad[i] += np.dot(p[m_lo:m_hi+1], 1.0 - h[js])
            # S_B: sum p[m]*h[i+l_m] where 0<=i+l_m<N
            # i + m - center in [0, N) => center-i <= m <= center+N-1-i
            m_lo2 = max(0, center - i)
            m_hi2 = min(2*N-2, center + N - 1 - i)
            if m_lo2 <= m_hi2:
                ms2 = np.arange(m_lo2, m_hi2 + 1)
                js2 = i + (ms2 - center)
                grad[i] -= np.dot(p[m_lo2:m_hi2+1], h[js2])
        grad *= (2.0 / N)
        
        h = project_box_sum(h - lr * grad, tgt)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lr = max(0.001, lr * 0.995)
    return best_h, best_c
