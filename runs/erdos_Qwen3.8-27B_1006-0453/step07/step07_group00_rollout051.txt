def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def nesterov_polish(h_in, N, deadline, lr=0.01, k_top=50):
    h = h_in.astype(float).copy()
    m = np.zeros(N)
    beta = 0.82
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    it = 0
    while time.time() < deadline and it < 5000:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.97
            beta = min(0.95, 0.80 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    idx = h.argsort()
    h[idx[:target]] = 1.0
    h[idx[target:]] = 0.0
    return h

def build_del(s, N):
    d = np.zeros((N, 2 * N - 1))
    for i in range(N):
        d[i, i:i + N] = s
    return d

def full_scan_ls(h, corr, del_, N, max_pass=30):
    best_c = float(corr.max())
    for _ in range(max_pass):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        best_dc = 1e10
        bp, bq = -1, -1
        for p in ones:

            nm = vals.max(axis=1)
            idx = int(np.argmin(nm))
            if nm[idx] < best_dc:

                bp, bq = p, zeros[idx]
        if best_dc >= best_c - 1e-12:

        h[bp], h[bq] = 0.0, 1.0

def grad_polish(h_in, N, deadline, lr=0.01, k_top=50):
    h = h_in.astype(float).copy()
    m = np.zeros(N)
    beta = 0.82
    bpk = float(fft_corr(h).max())
    bh = h.copy()
    it = 0
    while time.time() < deadline and it < 4000:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < bpk:
            bpk = pk; bh = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L]); grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:]); grad[M:] -= y[:N-M]
        grad /= len(top)
        hn = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (hn.sum() - N / 2.0) / N
        hn = np.clip(hn - adj, 0.0, 1.0)
        m = beta * (hn - h)
        h = hn
        it += 1
        if it % 300 == 0:
            lr *= 0.97; beta = min(0.95, 0.80 + it * 0.0003)
    return bh, bpk * (2.0 / N)
