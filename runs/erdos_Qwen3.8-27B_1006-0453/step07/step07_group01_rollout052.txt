    # left_contrib[p,s] = (1-h[p+s]) if 0≤p+s<N else 0  [this is what we ADD when 0→1, SUBTRACT when 1→0]
    # right_contrib[p,s] = h[p-s] if 0≤p-s<N else 0    [this is what we ADD when 1→0, SUBTRACT when 0→1]
    
    # For h[p]=1 (flip to 0): D = right - left
    # For h[p]=0 (flip to 1): D = left - right

    s_idx = shifts[None, :]

    ps = p_idx + s_idx

    left = np.where(left_valid, 1.0 - h_f[np.clip(ps, 0, N-1)], 0.0)
    
    # right: valid when 0 ≤ p-s < N, value = h[p-s]
    ps2 = p_idx - s_idx
    right_valid = (ps2 >= 0) & (ps2 < N)
    right = np.where(right_valid, h_f[np.clip(ps2, 0, N-1)], 0.0)
    
    sign = 1.0 - 2.0 * h_f[:, None]
    D = sign * (left - right)
    return D

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(999)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0.0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1.0
    return h_bin

def thue_morse_seq(N):
    n = np.arange(N, dtype=np.int64)
    t = np.zeros(N, dtype=np.int64)
    m = n.copy()
    for _ in range(12):
        t += m & 1
        m >>= 1
    return (t % 2).astype(float)

    h = h_init.astype(float).copy()
    target = N / 2.0
    m_arr = np.zeros(N)
    v_arr = np.zeros(N)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        m_arr = beta1 * m_arr + (1.0 - beta1) * grad
        v_arr = beta2 * v_arr + (1.0 - beta2) * (grad * grad)
        m_hat = m_arr / (1.0 - beta1 ** t)
        v_hat = v_arr / (1.0 - beta2 ** t)
        lr_t = lr * (1.0 if t < 200 else 0.998 ** (t - 200))
        h = project_sum(h - lr_t * m_hat / (np.sqrt(v_hat) + eps), target)

            hTM = ensure_binary(hTM, N0)
        d3 = min(deadline - 280, time.time() + 60)
        h_a3, c5_a3 = adam_opt(hTM + np.random.randn(N0) * 0.03, N0, d3, k=40, T=50.0, lr=0.008)
        if c5_a3 < best_c5:
            best_c5, best_h = c5_a3, h_a3.copy()
        d3b = min(deadline - 250, time.time() + 100)
        hB3 = ensure_binary(best_h if c5_a3 < best_c5 else h_a3, N0)
        h_br3, c5_br3 = binary_sa_search(hB3, N0, d3b, sa_T0=0.04, sample_n=55)
        if c5_br3 < best_c5:
            best_c5, best_h = c5_br3, h_br3.copy()

    # Phase 4: Second binary SA from best with different params
    if time.time() < deadline - 150:
        d4 = min(deadline - 120, time.time() + 150)
        hB4 = ensure_binary(best_h, N0)
        h_br4, c5_br4 = binary_sa_search(hB4, N0, d4, sa_T0=0.015, sample_n=40)
        if c5_br4 < best_c5:
            best_c5, best_h = c5_br4, h_br4.copy()

    # Phase 5: Try N1 = next power-of-2-friendly size
    N1 = 1024
    if N1 <= N0 + 100 and time.time() < deadline - 120:
        h_up = upsample_h(best_h, len(best_h), N1)
        h_up = project_sum(h_up, N1 / 2.0)
        d5 = min(deadline - 100, time.time() + 40)
        h_a5, c5_a5 = adam_opt(h_up, N1, d5, k=45, T=70.0, lr=0.007)
        c5_norm = c5_a5
        if c5_norm < best_c5:
            best_c5, best_h = c5_norm, h_a5.copy()
        d5b = min(deadline - 80, time.time() + 50)
        hB5 = ensure_binary(best_h, len(best_h))
        h_br5, c5_br5 = binary_sa_search(hB5, len(best_h), d5b, sa_T0=0.02, sample_n=45)
        if c5_br5 < best_c5:
            best_c5, best_h = c5_br5, h_br5.copy()

    # Phase 6: Final binary polish on best
    N_fin = len(best_h)
    if time.time() < deadline - 20:
        hF = ensure_binary(best_h, N_fin)
        sf = 35 if N_fin > 800 else 50
        d6 = min(deadline - 3, time.time() + 50)
        hF2, c5F = binary_sa_search(hF, N_fin, d6, sa_T0=0.01, sample_n=sf)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Phase 7: Final micro continuous adjustment
    if time.time() < deadline - 2:
        hFc = best_h + np.random.randn(len(best_h)) * 0.015
        hFc = project_sum(hFc, len(best_h) / 2.0)
        d7 = min(deadline - 1, time.time() + 20)
        h7, c5_7 = adam_opt(hFc, len(best_h), d7, k=50, T=120.0, lr=0.005)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
