        # Adapt k

For the coordinate descent on continuous h:

- Compute the current corr values
- The derivative of corr[k] with respect to h[i] is:

  (where the signs depend on the lag convention)

- This is: find h* = argmin_{h in [0,1]} max_k (corr[k] + h * d_k)

- We can find the minimum by checking the breakpoints where the max changes

    return grad

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def continuous_opt(h_init, N, deadline, k0=80, lr0=0.04, seed=0):
    rng = np.random.RandomState(seed)
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.90
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        k = max(3, int(k0 * max(0.15, 1.0 - 0.75 * min(1.0, it / 4000.0))))
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 150:
            lr = max(0.004, lr * 0.8)
            beta = min(0.96, beta + 0.03)
            m *= 0.2
            stall = 0
            h = project_sum(h + rng.randn(N) * 0.01 * (lr / lr0 + 0.3), target)
        grad = subgrad_topk(h, corr, k)
        y = h + m
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 500 == 0:
            lr *= 0.992
            beta = min(0.96, 0.89 + it * 0.00004)
    return best_h, best_pk * dx

def binary_opt_fast(h_init, N, deadline, seed=0, sf=1.0):
    rng = np.random.RandomState(seed)
    h = h_init.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    elif n1 < target:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    best_c = float(np.max(fft_corr(h_bin)))
    best_h = h_bin.copy()
    stall = 0
    while time.time() < deadline:
        c = fft_corr(h_bin)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h_bin.copy()
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h_bin, N)
        Dz = D[zeros, :]
        n_s = min(len(ones), max(40, int(len(ones) * sf)))
        p_list = ones if n_s >= len(ones) else rng.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:
            h_bin[best_pq[0]] = 0.0
            h_bin[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def ensure_binary_sum(h, N):
    h_bin = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h_bin.sum())
    rng = np.random.RandomState(999)
    if n1 != target:
        if n1 > target:
            ones = np.where(h_bin == 1)[0]
            idx = rng.choice(ones, n1 - target, replace=False)
            h_bin[idx] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            idx = rng.choice(zeros, target - n1, replace=False)
            h_bin[idx] = 1
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick continuous at N0
    d1 = min(deadline - 750, t0 + 35)
    h1, c1 = continuous_opt(h0, N0, d1, k0=70, lr0=0.05, seed=seed)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary at N0
    d2 = min(deadline - 650, t0 + 90)
    h2, c2 = binary_opt_fast(h0, N0, d2, seed=seed+1, sf=1.0)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 3: Long continuous at N0 (main workhorse)
    d3 = min(deadline - 380, t0 + 380)
    h3, c3 = continuous_opt(best_h, N0, d3, k0=100, lr0=0.045, seed=seed+2)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: Binary refinement from continuous
    d4 = min(deadline - 300, t0 + 470)
    h4, c4 = binary_opt_fast(best_h, N0, d4, seed=seed+3, sf=0.8)
    if c4 < best_c5:
        best_c5, best_h = c4, h4.copy()

    # Phase 5: Continuous at 2*N0
    N1 = 2 * N0
    if time.time() < deadline - 250:
        h_up = project_sum(upsample(best_h, len(best_h), N1), N1 / 2.0)
        d5 = min(deadline - 180, t0 + 640)
        h5, c5v = continuous_opt(h_up, N1, d5, k0=130, lr0=0.022, seed=seed+4)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()

    # Phase 6: Continuous at N0 with diverse perturbation
    if time.time() < deadline - 140:
        h_pert = project_sum(best_h[:N0] + np.random.randn(N0) * 0.06, N0 / 2.0)
        d6 = min(deadline - 100, t0 + 790)
        h6, c6 = continuous_opt(h_pert, N0, d6, k0=50, lr0=0.038, seed=seed+5)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: Binary at N1
    N_fin7 = len(best_h)
    if N_fin7 == N1 and time.time() < deadline - 80:
        d7 = min(deadline - 50, t0 + 900)
        h7, c7 = binary_opt_fast(best_h, N1, d7, seed=seed+6, sf=0.5)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Phase 8: Final continuous polish
    N_fin = len(best_h)
    d8 = deadline - 1
    h8, c8 = continuous_opt(best_h, N_fin, d8, k0=35, lr0=0.018, seed=seed+7)
    if c8 < best_c5:
        best_c5, best_h = c8, h8.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
