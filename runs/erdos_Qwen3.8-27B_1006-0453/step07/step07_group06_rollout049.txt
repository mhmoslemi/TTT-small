def softmax_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            L = i - ctr
            if L == 0:
                g[:] += w[i] * (1.0 - 2.0 * h)
            elif L > 0:
                g[L:] += w[i] * (1.0 - h[:N - L])
                g[:N - L] -= w[i] * h[L:N]
            else:
                M = -L
                g[:N - M] += w[i] * (1.0 - h[M:])
                g[M:] -= w[i] * h[:N - M]
    g += lam * s * np.ones(N)
    return val, g

            best_c5, best_h = c2b, hd2.copy()

    # Phase 3: More N=384 restarts
    if time.time() < dl - 200:
        d3 = min(dl - 150, time.time() + 150)
        for r in range(3):
            if time.time() > d3:
                break
            h384r = interp_to_N(best_h, len(best_h), N2)
            h384r += np.random.randn(N2) * 0.15
            h384r = make_binary(h384r, N2)
            hr, cr = bin_search_focused(h384r, N2, min(d3, time.time() + 50), max_iters=80000, ntrials=18)
            cr_c5 = compute_c5(hr)
            if cr_c5 < best_c5:
                best_c5, best_h = cr_c5, hr.copy()

    # Phase 4: L-BFGS continuous relaxation (100s)
    if time.time() < dl - 100:
        d4 = min(dl - 60, time.time() + 100)
        h4 = best_h + np.random.randn(len(best_h)) * 0.06
        h4 = proj01(h4, len(best_h) / 2.0)
        hl4, cl4 = lbfgs_smooth(h4, len(best_h), d4, T=35.0, max_outer=4)
        c4 = compute_c5(hl4)
        if c4 < best_c5:
            best_c5, best_h = c4, hl4.copy()

    # Phase 5: Final binary polish (80s)
    if time.time() < dl - 30:
        d5 = min(dl - 10, time.time() + 80)
        hb5, cb5 = bin_search_focused(best_h, len(best_h), d5, max_iters=100000, ntrials=18)
        c5f = compute_c5(hb5)
        if c5f < best_c5:
            best_c5, best_h = c5f, hb5.copy()

    # Phase 6: Try N=382 binary
    if time.time() < dl - 15:
        N3 = 382
        h382 = make_binary(interp_to_N(best_h, len(best_h), N3), N3)
        hb3, cb3 = bin_search_focused(h382, N3, dl - 2, max_iters=50000, ntrials=18)
        c3 = compute_c5(hb3)
        if c3 < best_c5:
            best_c5, best_h = c3, hb3.copy()

    # Final: compare continuous vs binary
    fc5 = compute_c5(best_h)
    fb = (best_h > 0.5).astype(np.float64)
    fc5b = compute_c5(fb)
    if fc5b < fc5:
        best_h, fc5 = fb, fc5b

    return best_h, fc5, len(best_h)
