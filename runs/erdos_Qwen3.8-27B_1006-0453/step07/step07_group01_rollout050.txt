            q = int(zeros[rng3.randint(len(zeros))])
            h_pert[p], h_pert[q] = 0.0, 1.0
        d3 = min(deadline - 120, time.time() + 120)
        h_b2, c5_b2 = binary_search(h_pert, N0, d3, seed=123)
        if c5_b2 < best_c5:
            best_c5, best_h = c5_b2, h_b2.copy()

    # Phase 4: Continuous refinement from best binary
    if time.time() < deadline - 100:
        hc2 = best_h.copy() + np.random.randn(N0) * 0.03
        hc2 = project_sum(hc2, N0 / 2.0)
        d4 = min(deadline - 60, time.time() + 50)
        h_c2, c5_c2 = continuous_opt(hc2, N0, d4, k=55, T=180.0, lr0=0.01)
        if c5_c2 < best_c5:
            best_c5, best_h = c5_c2, h_c2.copy()

    # Phase 5: Final binary polish (multiple seeds)
    if time.time() < deadline - 25:
        d5a = min(deadline - 15, time.time() + 30)
        h_f, c5_f = binary_search(best_h, N0, d5a, seed=2024)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Phase 6: Very short final nudge
    if time.time() < deadline - 3:
        d6 = min(deadline - 2, time.time() + 15)
        h_g, c5_g = binary_search(best_h, N0, d6, seed=55)
        if c5_g < best_c5:
            best_c5, best_h = c5_g, h_g.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N0
