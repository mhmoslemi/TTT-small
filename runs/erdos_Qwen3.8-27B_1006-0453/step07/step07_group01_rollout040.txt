import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def ensure_binary(h, N, rng):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    """Compute D matrix: D[p,s] = change in corr[s] when h[p] goes 1->0."""
    shifts = np.arange(2*N-1) - (N-1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all + s_all
    pp = p_all - s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    # D10: flip 1->0
    D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
    return D

def binary_sa(h_init, N, deadline, T0=0.2, T_min=0.0005, max_sweeps=3000, rng_seed=42):
    rng = np.random.RandomState(rng_seed)
    h = ensure_binary(h_init, N, rng).copy()
    N = len(h)
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    T = T0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    total_swaps = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep > 0 and sweep % 80 == 0:
            T = max(T_min, T * 0.65)
        
        if sweep % 20 == 0:
            D = compute_D(h, N)
            c = fft_corr(h)
            P = float(c.max())
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        
        D = compute_D(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        
        # Try multiple random 1-swaps
        n_try = max(5, min(15, N // 40))
        for _ in range(n_try):
            pi = int(ones[rng.randint(n1)])
            qi = int(zeros[rng.randint(n0)])
            new_P = float((c + D[pi] - D[qi]).max())
            delta = new_P - P
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                h[pi] = 0.0
                h[qi] = 1.0
                # Update c
                c = c + D[pi] - D[qi]
                P = new_P
                total_swaps += 1
                break  # One swap per trial set
    
    return best_h, best_c * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

        return h
    lo, hi = -1.0, 1.0

def ensure_binary(h, N, rng):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    p_all = np.arange(N)[:, None]
    s_all = shifts[None, :]
    pm = p_all + s_all
    pp = p_all - s_all
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
    D[:, N - 1] = 0.0
    return D

def binary_sa(h_init, N, deadline, T0=0.18, T_min=0.0003, max_sweeps=5000, rng_seed=42):

    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += wi * (1.0 - h[:N - s])
                grad[:N - s] -= wi * h[s:]
            else:
                M = -s
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        grad /= k
        h_new = _project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * 0.00015)
    return best_h, best_pk * (2.0 / N)

def _project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def thue_morse(N, offset=0):
    h = np.zeros(N)
    for i in range(N):
        v = i + offset
        cnt = 0
        while v > 0:
            cnt += v & 1
            v >>= 1
        h[i] = cnt % 2
    return h

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov warm-up at N0
    d1 = min(deadline - 600, t0 + 100)
    hc = h0.copy() + np.random.randn(N0) * 0.05
    hc = _project_sum(hc, N0 / 2.0)
    h_n, c5_n = nesterov_opt(hc, N0, d1, k=40, T=80.0, lr0=0.02)
    if c5_n < best_c5:
        best_c5, best_h = c5_n, h_n.copy()

    # Phase 2: Binary SA at N0 (main phase)
    h_bin = ensure_binary(best_h, N0, np.random.RandomState(1))
    d2 = min(deadline - 350, t0 + 500)
    h_br, c5_br = binary_sa(h_bin, N0, d2, T0=0.2, T_min=0.0002, max_sweeps=8000, rng_seed=seed)
    if c5_br < best_c5:
        best_c5, best_h = c5_br, h_br.copy()

    # Phase 3: Thue-Morse init at N0, binary SA
    d3_start = time.time()
    if d3_start < deadline - 300:
        tm = thue_morse(N0, offset=0)
        tm = _project_sum(tm, N0 / 2.0)
        d3 = min(deadline - 250, d3_start + 180)
        h_tm, c5_tm = binary_sa(tm, N0, d3, T0=0.25, T_min=0.0003, max_sweeps=4000, rng_seed=seed + 1)
        if c5_tm < best_c5:
            best_c5, best_h = c5_tm, h_tm.copy()

    # Phase 4: N=512 from Thue-Morse
    N2 = 512
    d4_start = time.time()
    if d4_start < deadline - 200:
        tm2 = thue_morse(N2, offset=3)
        d4 = min(deadline - 160, d4_start + 120)
        h_t2, c5_t2 = binary_sa(tm2, N2, d4, T0=0.2, T_min=0.0003, max_sweeps=3000, rng_seed=seed + 2)
        if c5_t2 < best_c5:
            best_c5, best_h = c5_t2, h_t2.copy()

    # Phase 5: N=1024 from best upsampled
    N3 = 1024
    d5_start = time.time()
    if d5_start < deadline - 150 and len(best_h) != N3:
        h_up = upsample(best_h, len(best_h), N3)
        h_up = ensure_binary(h_up, N3, np.random.RandomState(5))
        d5 = min(deadline - 120, d5_start + 100)
        h_u3, c5_u3 = binary_sa(h_up, N3, d5, T0=0.15, T_min=0.0004, max_sweeps=2500, rng_seed=seed + 3)
        if c5_u3 < best_c5:
            best_c5, best_h = c5_u3, h_u3.copy()

    # Phase 6: Second binary SA from best with different seed
    d6_start = time.time()
    if d6_start < deadline - 60:
        Nf = len(best_h)
        h_f = ensure_binary(best_h, Nf, np.random.RandomState(99))
        d6 = min(deadline - 30, d6_start + 80)
        h_f2, c5_f2 = binary_sa(h_f, Nf, d6, T0=0.12, T_min=0.0001, max_sweeps=4000, rng_seed=seed + 10)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2.copy()

    # Phase 7: Final continuous micro-pass
    d7_start = time.time()
    if d7_start < deadline - 10:
        Nf = len(best_h)
        hFc = best_h.copy() + np.random.randn(Nf) * 0.015
        hFc = _project_sum(hFc, Nf / 2.0)
        d7 = min(deadline - 2, d7_start + 35)
        h7, c5_7 = nesterov_opt(hFc, Nf, d7, k=50, T=120.0, lr0=0.008)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
