def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(42)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def quick_greedy(h, N, n_rounds=50):

    best_c = float(np.max(fft_corr(h)))
    for _ in range(n_rounds):
        imp = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

                # Worst: h[i]=1, h[i+s]=0

def nesterov_continuous(h_init, N, deadline, k=30, T=80.0, lr0=0.015):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N

    m = np.zeros(N)

    beta = 0.9
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(len(idx)):
            s = int(idx[ii] - center)
            wi = w[ii]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:

                grad[:N - s] -= wi * h[s:]

                grad[M:] -= wi * h[:N - M]

        h_new = np.clip(h_new, 0.0, 1.0)

        d6 = min(deadline - 60, time.time() + 70)
        h6, c5_6 = nesterov_continuous(h_cont, N0, d6, k=40, T=100.0, lr0=0.012)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6
    
    # Phase 7: Second Nesterov pass from best
    if time.time() < deadline - 50:
        h_c2 = best_h + np.random.randn(len(best_h)) * 0.02
        h_c2 = np.clip(h_c2, 0, 1)
        h_c2 -= (h_c2.sum() - len(best_h) / 2.0) / len(best_h)
        h_c2 = np.clip(h_c2, 0, 1)
        d7 = min(deadline - 20, time.time() + 35)
        h7, c5_7 = nesterov_continuous(h_c2, len(best_h), d7, k=50, T=150.0, lr0=0.008)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7
    
    # Phase 8: Final binary polish (if best is close to binary)
    N_fin = len(best_h)
    h_bin_fin = to_binary(best_h, N_fin)
    c5_bin = compute_c5(h_bin_fin)
    if c5_bin <= best_c5 and time.time() < deadline - 20:
        hF, c5F = directed_greedy(h_bin_fin, N_fin, n_rounds=100)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    
    # Phase 9: Final micro-SA
    if time.time() < deadline - 5:
        N_f = len(best_h)
        if best_h == (best_h > 0.5).astype(float):
            d9 = deadline - 3
            h9, c5_9 = sa_binary(best_h, N_f, d9, T0=2.0, T_end=0.1, reheat_T=3.0)
            if c5_9 < best_c5:
                best_c5, best_h = c5_9, h9
        else:
            h_c9 = best_h.copy()
            h_c9 += np.random.randn(N_f) * 0.015
            h_c9 = np.clip(h_c9, 0, 1)
            h_c9 -= (h_c9.sum() - N_f / 2.0) / N_f
            h_c9 = np.clip(h_c9, 0, 1)
            d9c = deadline - 3
            h9c, c5_9c = nesterov_continuous(h_c9, N_f, d9c, k=50, T=200.0, lr0=0.006)
            if c5_9c < best_c5:
                best_c5, best_h = c5_9c, h9c
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
