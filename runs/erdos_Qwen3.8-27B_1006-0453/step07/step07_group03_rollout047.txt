    """D[p, k] = change in corr[k] when h[p] goes 1->0"""
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv  # p + s
    pp = pos - sv  # p - s
    vp = (pm >= 0) & (pm < N)
    vm = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    # change for h[p]: 1->0 in the h[i]*(1-h[i+s]) term: -(1-h[p+s])
    term1 = -(1.0 - h[pm_c]) * vp
    # change in the h[i+s]*(1-h[i]) term where i+s=p, i=p-s: +h[p-s]
    term2 = h[pp_c] * vm
    return term1 + term2

def compute_Dp(h, N):
    """D[q, k] = change in corr[k] when h[q] goes 0->1"""
    # This is the negative of compute_Dm evaluated at q (where h[q]=0)
    # But we need to be careful: when h[q]=0, the terms are different
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv  # q + s
    pp = pos - sv  # q - s
    vp = (pm >= 0) & (pm < N)
    vm = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    # change for h[q]: 0->1 in the h[i]*(1-h[i+s]) term: +(1-h[q+s])
    term1 = (1.0 - h[pm_c]) * vp
    # change in the h[i+s]*(1-h[i]) term where i+s=q, i=q-s: -h[q-s]
    term2 = -h[pp_c] * vm
    return term1 + term2

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L <<= 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 5e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k=50, lr0=0.01):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = y.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.98
            beta = min(0.97, 0.85 + 0.0002 * it)
    return best_h, best_pk * dx
