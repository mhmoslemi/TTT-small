import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    nh = N // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(N)
    b[idx[:nh]] = 1.0
    return b

def grad_c5(h, top_indices, N):
    """Compute gradient of max correlation w.r.t. h, using given top indices."""
    grad = np.zeros(N)
    center = N - 1
    for ai in top_indices:
        sh = int(ai - center)
        if sh == 0:
            grad += (1.0 - 2.0 * h)
        elif sh > 0:
            grad[:N - sh] += (1.0 - h[sh:])
            grad[sh:] -= h[:N - sh]
        else:
            M = -sh
            grad[M:] += (1.0 - h[:N - M])
            grad[:N - M] -= h[M:]
    grad /= len(top_indices)
    return grad

def adam_opt(h, N, deadline, k_frac=0.2, lr=0.004):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    L_tot = 2 * N - 1
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        k = max(16, int(k_frac * L_tot))
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
        g = grad_c5(h, top, N)

        h = project_sum(h - lr * mhat / (np.sqrt(vhat) + 1e-8), target)
        if t % 2000 == 0:

    return best_h, best_pk * dx

def block_swap(h, N, rng, max_block=3):
    """Try a block swap: find two runs of same length and swap."""
    h = h.copy()
    # Find all runs
    runs = []  # (start, length, value)
    i = 0
    while i < N:
        j = i
        while j < N and h[j] == h[i]:
            j += 1
        runs.append((i, j - i, h[i]))
        i = j
    ones_runs = [(s, l) for s, l, v in runs if v == 1]
    zeros_runs = [(s, l) for s, l, v in runs if v == 0]
    if not ones_runs or not zeros_runs:
        return h, False
    # Try to find matching length runs (up to max_block)
    for blen in range(1, max_block + 1):
        ones_match = [(s, l) for s, l in ones_runs if l >= blen]
        zeros_match = [(s, l) for s, l in zeros_runs if l >= blen]
        if ones_match and zeros_match:
            s1, _ = ones_match[rng.randint(len(ones_match))]
            s2, _ = zeros_match[rng.randint(len(zeros_match))]
            # Swap blen elements from run 1 with blen from run 2
            h[s1:s1+blen] = 0.0
            h[s2:s2+blen] = 1.0
            return h, True
    return h, False

def binary_anneal(h, N, deadline, rng, temp0=0.008, cool=0.995, max_swaps=50000):
    """Binary annealing with single and double swaps, with SA acceptance."""
    h = h.astype(float).copy()
    nh = N // 2
    # Ensure binary
    idx = np.argsort(-h)
    b = np.zeros(N)
    b[idx[:nh]] = 1.0
    h = b
    best_h = h.copy()
    best_c = fft_corr(h)
    best_pk = float(best_c.max())
    temp = temp0
    swaps = 0
    while time.time() < deadline and swaps < max_swaps:
        swaps += 1
        temp *= cool
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        corr = fft_corr(h)
        pk = float(corr.max())
        # Try single swap
        p = int(ones[rng.randint(len(ones))])
        q = int(zeros[rng.randint(len(zeros))])
        h[p], h[q] = 0.0, 1.0
        new_c = float(fft_corr(h).max())
        delta = new_c - pk
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-6)):
            if new_c < best_pk - 1e-13:
                best_pk = new_c
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
    return best_h, best_pk * (2.0 / N)

    """Compute log-sum-exp of correlations with temperature 1/K."""

    max_c = corr.max()
    # log-sum-exp trick

    log_sum = max_c + np.log(np.sum(np.exp(shifted))) / K

def soft_max_grad(h, N, top_idx, K=50):

    corr = fft_corr(h)

        if float(pk_cand[mi]) < float(c_cur.max()) - 1e-12:
            best_delta = float(pk_cand[mi]) - float(c_cur.max())
            best_move = (int(ones[pi]), int(zeros[mi]), 1)
        # Random 2-swap
        if rng.random() < two_frac:
            n_try = min(60, max(20, N // 6))
            for _ in range(n_try):
                i1, i2 = rng.randint(0, len(ones), 2)
                j1, j2 = rng.randint(0, len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                p1, p2 = int(ones[i1]), int(ones[i2])
                q1, q2 = int(zeros[j1]), int(zeros[j2])
                new_m = float((c_cur - D[p1] - D[p2] + D[q1] + D[q2]).max())
                delta = new_m - float(c_cur.max())
                if delta < best_delta:
                    best_delta = delta
                    best_move = (p1, p2, q1, q2, 2)
        if best_move is not None:
            if best_move[-1] == 1:
                p, q = best_move[0], best_move[1]
            else:
                p, q, q2, _ = best_move[0], best_move[1], best_move[2], best_move[3]
            accept = best_delta < 0 or rng.random() < np.exp(-best_delta / max(temp, 1e-7))
            if accept:
                if best_move[-1] == 1:
                    h[p], h[q] = 0.0, 1.0
                else:
                    h[p] = h[q] = 0.0
                    h[q2] = 1.0
                    qq = best_move[3] if best_move[-1] == 2 else q
                    # For 2-swap: (p1, p2, q1, q2)
                    h[best_move[0]] = h[best_move[1]] = 0.0
                    h[best_move[2]] = h[best_move[3]] = 1.0
                c_cur = fft_corr(h)
                pk = float(c_cur.max())
                if pk < best_pk - 1e-13:
                    best_pk = pk
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
            if stall > 30:
                nf = rng.randint(1, 4)
                for _ in range(nf):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                c_cur = fft_corr(h)
                stall = 0
        if it % 5000 == 0:
            temp = max(0.002, temp * 0.85)
    return best_h, best_pk * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    def elapsed():
        return time.time() - t0

    # Phase 1: Adam soft-max at N0 (multiple K values)
    d1 = min(deadline - 450, t0 + budget_s * 0.30)
    h1, c5_1 = adam_softmax(h0.copy(), N0, d1, K0=20.0, lr0=0.0025)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Binary annealing at N0
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + budget_s * 0.18)
        h2, c5_2 = binary_anneal(best_h, N0, d2, rng, temp0=0.008, cool=0.998, max_iter=60000, two_frac=0.3)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Adam at N0 with higher K (sharper)
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + budget_s * 0.12)
        hc = best_h + rng.randn(N0) * 0.02
        hc = project_sum(hc, N0 / 2.0)
        h3, c5_3 = adam_softmax(hc, N0, d3, K0=50.0, lr0=0.0015)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upscale to N1=2*N0, Adam refine
    N1 = 2 * N0
    if time.time() < deadline - 180:
        h_up = upsample(best_h, len(best_h), N1)
        h_up = project_sum(h_up, N1 / 2.0)
        d4 = min(deadline - 130, time.time() + budget_s * 0.15)
        h4, c5_4 = adam_softmax(h_up, N1, d4, K0=30.0, lr0=0.002)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Binary polish at best resolution
    N_fin = len(best_h)
    if time.time() < deadline - 40:
        hF = to_binary(best_h, N_fin)
        d5 = deadline - 5
        sf = binary_anneal(hF, N_fin, d5, rng, temp0=0.005, cool=0.999, max_iter=40000, two_frac=0.25)
        if sf[1] < best_c5:
            best_c5, best_h = sf[1], sf[0].copy()

    # Final: also try keeping fractional if better
    final_c5 = compute_c5(best_h, len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
