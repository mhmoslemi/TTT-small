        h5s = perturb_binary(ensure_sum(best_h, N0), N0, 12, rng5)
        d5 = min(deadline - 200, t0 + 550)
        h5, c5_5 = binary_sa(h5s, N0, d5, T0=5.0, Tcool=0.992, T_min=0.015, sf=1.0, kick_every=80, kick_sz=6, seed=seed+200)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Greedy + SA again from best
    if time.time() < deadline - 120:
        h6g = ensure_sum(best_h[:N0] if len(best_h) == N0 else upsample(best_h, len(best_h), N0), N0)
        d6a = min(deadline - 100, t0 + 640)
        h6a, c5_6a = greedy_binary(h6g, N0, d6a, max_sweeps=300)
        if c5_6a < best_c5:
            best_c5, best_h = c5_6a, h6a.copy()
        if time.time() < deadline - 60:
            rng6 = np.random.RandomState(seed + 300)
            h6s = perturb_binary(ensure_sum(best_h, N0), N0, 8, rng6)
            d6b = deadline - 40
            h6b, c5_6b = binary_sa(h6s, N0, d6b, T0=2.5, Tcool=0.995, T_min=0.03, sf=1.0, kick_every=130, kick_sz=3, seed=seed+400)
            if c5_6b < best_c5:
                best_c5, best_h = c5_6b, h6b.copy()

    # Phase 7: Final greedy polish
    if time.time() < deadline - 1:
        hF = ensure_sum(best_h[:N0] if len(best_h) == N0 else upsample(best_h, len(best_h), N0), N0)
        d7 = deadline - 0.5
        hF2, c5F = greedy_binary(hF, N0, d7, max_sweeps=200, sf=1.0)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    best_h = best_h[:N0] if len(best_h) != N0 else best_h
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N0
