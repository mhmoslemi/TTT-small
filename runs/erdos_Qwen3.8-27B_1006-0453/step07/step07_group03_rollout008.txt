import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, n1 - nh, replace=False)] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h_bin

def compute_perturbation(h, N):
    """Compute how much each position contributes to the max correlation."""
    # For 1-swap: moving a 1 from p to q changes c by -D[p] + D[q]
    # where D[j] captures the change
    c = fft_corr(h)

    # Instead, use the analytical formula
    # ∂c[k]/∂h[j] = 1 - h[j+k'] - h[j] (when shifted properly)
    # This is expensive to compute fully. Use direct formula for swap evaluation.
    return c
