    n_s = 2 * N - 1
    shifts = np.arange(n_s) - (N - 1)
    p = np.arange(N)[:, None]
    s = shifts[None, :]
    ps = p + s
    pm = p - s
    v1 = (ps >= 0) & (ps < N)
    v2 = (pm >= 0) & (pm < N)
    B = np.where(v1, 1.0 - h[np.clip(ps, 0, N - 1)], 0.0)
    B += np.where(v2, -h[np.clip(pm, 0, N - 1)], 0.0)
    return B

        pk = float(corr.max())

            best_pk = pk
            best_h = h.copy()

            old = h[p]

            f0 = A.max()
            f1 = (A + Bp).max()
            fm = (A + x * Bp).max()
            if f0 <= fm and f0 <= f1:
                x = 0.0
            elif f1 <= fm and f1 <= f0:
                x = 1.0

                h[p] = x

                B[:, :].fill(0.0)  # Invalid now, but we recompute at start of sweep
                changed += 1

                best_pk = pk
                best_h = h.copy()
            corr = new_corr
        else:
            break

    pk = float(corr.max())
    if pk < best_pk:
        best_h = h.copy()
        best_pk = pk
    return best_h, best_pk * (2.0 / N)

Wait, there's a problem with the coordinate descent. When I update h[p] incrementally, the B matrix becomes stale (since B depends on h). I was recomputing B at the start of each sweep, but after changing h[p] in the middle of a sweep, B for other positions p' is slightly stale.

This is actually OK for coordinate descent - it's like Gauss-Seidel. The staleness is small (only one position changed) and the method should still converge. But the incremental corr update is exact, so the convergence criterion is valid.

However, I should NOT recompute B inside the sweep (it's O(N²) which is expensive). Instead, I'll recompute it at the start of each sweep.

But wait, I also invalidate B when I do the projection at the end of the sweep. So I need to recompute corr from scratch after projection.
