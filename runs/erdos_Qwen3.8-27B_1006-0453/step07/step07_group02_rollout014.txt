import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h.astype(float), L)
    G = np.fft.fft(np.flip(1.0 - h.astype(float)), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_delta_corr(h, N, p, q):
    """Compute delta in correlation when swapping h[p]=1->0 and h[q]=0->1"""
    # corr[k] = sum_i h[i] * (1-h[i+k]) where k ranges from -(N-1) to N-1
    # Output indexed as corr_full[m] where m = k + (N-1)
    delta = np.zeros(2*N-1)
    center = N - 1
    
    # Term 1: -(1-h[p+k]) for k where p+k in [0,N-1]
    # k ranges from -p to N-1-p
    for k in range(-p, N-p):
        pk = p + k
        delta[center + k] -= (1.0 - h[pk])
    
    # Term 2: +(1-h[q+k]) for k where q+k in [0,N-1]
    for k in range(-q, N-q):
        qk = q + k
        delta[center + k] += (1.0 - h[qk])
    
    # Term 3: +h[p-k] for k where p-k in [0,N-1]
    for k in range(p-(N-1), p+1):
        pk = p - k
        delta[center + k] += h[pk]
    
    # Term 4: -h[q-k] for k where q-k in [0,N-1]
    for k in range(q-(N-1), q+1):
        qk = q - k
        delta[center + k] -= h[qk]
    
    return delta

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h.astype(float), L)
    G = np.fft.fft(np.flip(1.0 - h.astype(float)), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def aps_solve(h, N, deadline, max_iters=50000, T0=0.3, T_end=0.01, cool=0.99997):
    """Annealed pair swap search"""
    h = h.astype(float).copy()
    target = N // 2
    # Ensure exact sum
    s = h.sum()
    if s > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:int(s - target)]] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:int(target - s)]] = 1
    
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    T = T0
    stuck = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        p = int(np.random.choice(ones))
        q = int(np.random.choice(zeros))
        
        # Perform swap
        h[p] = 0
        h[q] = 1
        new_corr = fft_corr(h)
        new_max = float(np.max(new_corr))
        
        delta = new_max - cur_max
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Undo
            h[p] = 1
            h[q] = 0
        
        T = max(T * cool, T_end)
        if stuck > 2000:
            stuck = 0
            # Random kick
            for _ in range(5):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    
    return best_h, best_max * (2.0 / N)

            # Need h[i]=1, h[i+k]=0 and h[j]=0, h[j+k]=1
            # Try random ones and zeros
            candidates_p = ones
        else:
            candidates_p = ones
        
        p = int(np.random.choice(candidates_p))
        q = int(np.random.choice(zeros))
        
        # Swap
        h[p], h[q] = h[q], h[p]
        new_corr = fft_corr(h)
        new_max = float(np.max(new_corr))
        
        if new_max < cur_max or np.random.random() < np.exp((cur_max - new_max) / max(T, 1e-10)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[p], h[q] = h[q], h[p]  # undo
        
        if stuck > 5000:
            # Kick
            for _ in range(3):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p], h[q] = h[q], h[p]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stuck = 0
    
    return best_h, best_max * (2.0 / N)

This is clean but might be slow in Python due to the per-iteration overhead. Let me estimate:
- np.random.choice: ~1μs each (2 calls = 2μs)
- Array swap: ~0.1μs
- fft_corr: ~10μs (two FFTs of size 759 + one IFFT + real)
- np.max: ~0.5μs
- Comparison and logic: ~1μs
- Total: ~15μs per iteration

In 1000 seconds: ~66 million iterations. That should be more than enough.

But wait, I need to be smarter. Pure random swaps in a space of C(380,190) ≈ 10^113 possibilities are useless. Even with 66 million iterations, we're exploring a tiny fraction.

The key is to make the search **directed**. Let me think about a smarter approach:

**Idea: Greedy + SA hybrid**

Phase 1 (Greedy): 
- Repeatedly find the best swap (using the C-matrix trick) and perform it
- Stop when no improvement found
- This quickly finds a local optimum

Phase 2 (SA):
- Starting from the local optimum, do SA with random swaps
- Use very low temperature (just above 0) to occasionally escape
- This explores the neighborhood of the local optimum

Phase 3 (Restart):
- Perturb the best solution (swap 5-20 random pairs)
- Go back to Phase 1
- Repeat

Phase 4 (Multi-N):
- Try the best solution at different N values by interpolation + re-optimization
