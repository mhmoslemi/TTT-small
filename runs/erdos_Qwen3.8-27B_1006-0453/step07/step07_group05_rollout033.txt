            wt = w[k]
            if m == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif m > 0:
                g2 = N - m
                if g2 > 0:
                    grad[:g2] += wt * (1.0 - h[m:])
                    if m < N:
                        grad[m:] -= wt * h[:N - m]
            else:
                M = -m
                if M < N:
                    g2 = N - M
                    grad[M:] += wt * (1.0 - h[:g2])
                    grad[:g2] -= wt * h[M:]
        h = h - lr * grad
        h = project_sum(h, N / 2.0)
        c = c5_of(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 200 == 0:
            lr *= 0.98
        if lr < 1e-7:
            lr = 1e-7
    return best_h, best_c

def topk_lag_aware(h, deadline, K=15):
    h = h.copy().astype(float)
    N = len(h)
    corr = corr_of(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    T = 0.1
    for it in range(300000):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.95)[0]
        zeros = np.where(h < 0.05)[0]
        if len(ones) == 0 or len(zeros) === 0:
            break
        a = int(np.random.choice(ones))
        c = int(np.random.choice(zeros))
        h[a], h[c] = 0.0, 1.0
        nm = float(np.max(corr_of(h)))
        dE = nm - cur_max
        if dE < -1e-13 or (T > 1e-10 and np.random.random() < np.exp(-dE / (T + 1e-15))):
            cur_max = nm
            if cur_max < best_c * N / 2.0:
                best_c = cur_max * 2.0 / N
                best_h = h.copy()
        else:
            h[a], h[c] = 1.0, 0.0
        T *= 0.9998
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_c = c5_of(h0)
    best_h = h0.copy()

    # Phase 1: Greedy binary from parent
    d1 = min(deadline - 500, t0 + 60)
    hb, cb = greedy_binary(h0, d1)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Continuous optimization
    if time.time() < deadline - 300:
        hc = best_h.astype(float).copy()
        d2 = min(deadline - 250, time.time() + 120)
        hc, cc = cont_opt(hc, d2, p=6, lr0=0.003)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 3: ILS - perturb + SA + greedy
    n_rest = 20
    if time.time() < deadline - 200:
        rem = deadline - 200 - time.time()
        per_restart = max(2, int(rem / n_rest))
        for r in range(n_rest):
            if time.time() > deadline - 200:
                break
            K = 2 + r
            hp = best_h.copy()
            ones = np.where(hp > 0.5)[0]
            zeros = np.where(hp < 0.5)[0]
            for _ in range(K):
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                hp[oi], hp[zi] = 1 - hp[zi], hp[oi]
            sub = min(deadline - 180, time.time() + max(3, per_restart // 2))
            hr, cr = sa_binary(hp, sub, T0=0.08 + 0.02 * r, p2=0.3)
            if time.time() < deadline - 170:
                sub2 = min(deadline - 160, time.time() + max(2, per_restart // 4))
                hr2, cr2 = greedy_binary(hr, sub2)
                if cr2 < cr:
                    hr, cr = hr2, cr2
            if cr < best_c:
                best_c, best_h = cr, hr.copy()

    # Phase 4: Continuous refine on best
    if time.time() < deadline - 80:
        hc = best_h.astype(float).copy()
        d4 = min(deadline - 60, time.time() + 80)
        hc, cc = cont_opt(hc, d4, p=10, lr0=0.002)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 5: SA from continuous solution (round logic)
    if time.time() < deadline - 30:
        hr = best_h.copy()
        thr = np.median(hr[hr > 0.1] + hr[hr < 0.9]) if len(hr[(hr > 0.1) & (hr < 0.9)]) > 0 else 0.5
        hr_bin = (hr > thr).astype(float)
        hr_bin = np.clip(hr_bin, 0, 1)
        n1 = int(hr_bin.sum())
        tgt = N0 // 2
        if n1 != tgt:
            if n1 > tgt:
                idx = np.where(hr_bin == 1)[0]
                hr_bin[np.random.choice(idx, n1 - tgt, replace=False)] = 0
            else:
                idx = np.where(hr_bin == 0)[0]
                hr_bin[np.random.choice(idx, tgt - n1, replace=False)] = 1
        d5 = min(deadline - 10, time.time() + 25)
        hr2, cr2 = sa_binary(hr_bin, d5, T0=0.05, p2=0.3)
        if time.time() < deadline - 5:
            hr2, cr2 = greedy_binary(hr2, min(deadline - 2, time.time() + 10))
        if cr2 < best_c:
            best_c, best_h = cr2, hr2.copy()

    # Final: ensure valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, N0 / 2.0)
    final_c = c5_of(best_h)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c, N0
