    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5

def fix_binary(h):

    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:

        perm = np.random.permutation(len(idx))

def binary_sa(h_init, deadline, max_iters=50000, n_crit=25, T0=0.003, T_decay=0.99995, double_prob=0.1):

    h = fix_binary(h_init.copy())
    corr = compute_corr(h)
    cur_c = float(np.max(corr))
    best_c = cur_c
    best_h = h.copy()
    it = 0
    no_imp = 0
    T = T0
    
    while time.time() < deadline and it < max_iters:
        if no_imp > 400:
            # Diversify: random restart from perturbed best
            if np.random.random() < 0.3:
                h = best_h.copy()
                for _ in range(5):
                    oi = int(np.random.choice(np.where(h==1)[0]))
                    zi = int(np.random.choice(np.where(h==0)[0]))
                    h[oi] = 0; h[zi] = 1
                corr = compute_corr(h)
                cur_c = float(np.max(corr))
            no_imp = 0
            T = T0 * 0.5
        
        ones = np.where(h == 1).astype(int)
        zeros = np.where(h == 0).astype(int)
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        n_tries = 30
        for _ in range(n_tries):
            if np.random.random() < double_prob:
                # Double swap
                if len(ones) < 2 or len(zeros) < 2:
                    continue
                o1, o2 = np.random.choice(ones, 2, replace=False)
                z1, z2 = np.random.choice(zeros, 2, replace=False)
                h[o1]=0; h[o2]=0; h[z1]=1; h[z2]=1
                new_corr = compute_corr(h)
                new_c = float(np.max(new_corr))
                if new_c < cur_c or np.random.random() < np.exp(-(new_c-cur_c)/max(T,1e-10)):
                    corr = new_corr
                    cur_c = new_c
                    if cur_c < best_c - 1e-12:
                        best_c = cur_c; best_h = h.copy(); no_imp = 0
                    else: no_imp += 1
                    break
                else:
                    h[o1]=1; h[o2]=1; h[z1]=0; h[z2]=0
            else:
                # Single swap
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                # Quick check: only look at top lags
                top_idx = np.argpartition(corr, -n_crit)[-n_crit:]
                k_vals = (top_idx - (N-1)).astype(float)
                # Delta computation
                om = oi - k_vals; op = oi + k_vals
                zm = zi - k_vals; zp = zi + k_vals
                d = np.zeros(n_crit)
                m1 = (om>=0)&(om<N); d[m1] -= (1.0 - h[om[m1]])
                m2 = (op>=0)&(op<N); d[m2] += h[op[m2]]
                m3 = (zm>=0)&(zm<N); d[m3] += (1.0 - h[zm[m3]])
                m4 = (zp>=0)&(zp<N); d[m4] -= h[zp[m4]]
                new_top = corr[top_idx] + d
                new_max_top = float(np.max(new_top))
                if new_max_top < cur_c - 1e-12 or np.random.random() < np.exp(-(new_max_top-cur_c)/max(T,1e-10)):
                    h[oi]=0; h[zi]=1
                    corr = compute_corr(h)
                    cur_c = float(np.max(corr))
                    ones = np.where(h==1).astype(int)
                    zeros = np.where(h==0).astype(int)
                    if cur_c < best_c - 1e-12:
                        best_c = cur_c; best_h = h.copy(); no_imp = 0
                    else: no_imp += 1
                    break
            no_imp += 1
        T *= T_decay
        it += 1
    return best_h, best_c * 2.0 / N

def cont_opt(h_init, deadline, k=50, lr0=0.02, mom=0.85):

        else:
            stale += 1
        if stale > 80:
            lr0 *= 0.85; stale = 0
        top_idx = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for m in top_idx:
            L = int(m - center)
            if L == 0:
                grad += (1.0 - 2.0*h)
            elif L > 0 and L < N:
                grad[:N-L] += (1.0 - h[L:N])
                grad[L:] -= h[:N-L]
            elif L < 0 and -L < N:
                M = -L
                grad[M:] += (1.0 - h[:N-M])
                grad[:M] -= h[M:]
        g = grad / k
        h_new = project(h + mom*v - lr0*g, tgt)
        v = mom*v + (h_new - h)
        h = h_new
    return best_h, best_c
