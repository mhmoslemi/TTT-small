    return best_h, best_c * (2.0 / N)

def adam_opt(h, N, deadline, k=50, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    beta1, beta2 = 0.8, 0.999
    t = 0
    lr = lr0
    stall = 0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                h = best_h + np.random.randn(N) * 0.03
                h = project_sum(h, target)
                stall = 0
                lr = max(0.005, lr * 1.5)
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (2.0 * h - 1.0)
            elif L > 0:
                grad[:N - L] += h[L:]
                grad[L:] -= (1.0 - h[:N - L])
            else:
                M = -L
                if M < N:
                    grad[M:] += h[:N - M]
                    grad[:N - M] -= (1.0 - h[M:])
        grad /= k
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * grad * grad
        m1h = m1 / bc1
        m2h = m2 / bc2
        h_new = project_sum(h - lr * m1h / (np.sqrt(m2h) + 1e-8), target)
        h = h_new
        if t % 300 == 0:
            lr *= 0.97
    return best_h, best_pk * dx

def upsample_fn(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Intensive binary at N0
    hB = ensure_binary(h0, N0)
    d1 = min(deadline - 500, t0 + budget_s * 0.35)
    h1, c5_1 = binary_opt(hB, N0, d1, rng, max_sweeps=8000, twop=0.5, threep=0.1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Adam continuous at N0
    if time.time() < deadline - 350:
        hc = best_h + rng.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 300, time.time() + budget_s * 0.15)
        h2, c5_2 = adam_opt(hc, N0, d2, k=60, lr0=0.04)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Second binary at N0 from perturbed best
    if time.time() < deadline - 200:
        rng2 = np.random.RandomState(seed + 200)
        hB2 = ensure_binary(best_h, N0)
        ones_t = np.where(hB2 == 1)[0]
        zeros_t = np.where(hB2 == 0)[0]
        for _ in range(8):
            hB2[ones_t[rng2.randint(len(ones_t))]] = 0.0
            hB2[zeros_t[rng2.randint(len(zeros_t))]] = 1.0
        d3 = min(deadline - 150, time.time() + budget_s * 0.15)
        h3, c5_3 = binary_opt(hB2, N0, d3, rng2, max_sweeps=5000, twop=0.55, threep=0.12)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upscale to 2*N0
    if time.time() < deadline - 120:
        N1 = 2 * N0
        h_up = ensure_binary(upsample_fn(best_h, N0, N1), N1)
        d4 = min(deadline - 80, time.time() + budget_s * 0.18)
        rng3 = np.random.RandomState(seed + 500)
        h4, c5_4 = binary_opt(h_up, N1, d4, rng3, max_sweeps=2000, twop=0.4, threep=0.05)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Adam on best (possibly at 2N)
    if time.time() < deadline - 60:
        Nf = len(best_h)
        hf = best_h + rng.randn(Nf) * 0.03
        hf = project_sum(hf, Nf / 2.0)
        d5 = min(deadline - 30, time.time() + budget_s * 0.10)
        h5, c5_5 = adam_opt(hf, Nf, d5, k=40, lr0=0.025)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final binary polish
    N_fin = len(best_h)
    hF = ensure_binary(best_h, N_fin)
    rngF = np.random.RandomState(seed + 999)
    sf = 1.0 if N_fin <= 500 else 0.6
    d6 = deadline - 1
    hF2, c5F = binary_opt(hF, N_fin, d6, rngF, max_sweeps=3000, twop=0.5, threep=0.08)
    if c5F < best_c5:
        best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
