```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def corr_full(h):
    """Compute full correlation h * (1-h) efficiently"""
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def lin_grad(L, h, N):
    """Gradient of correlation at lag L with respect to h"""
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    """Project h to [0,1]^N with sum = target"""
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_smooth(h, N, deadline, T=30.0, max_outer=3):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_and_grad(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_and_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 120, 'maxfun': 300, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj_and_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.5
    return best_h, best_v * (2.0 / N)

def gradient_guided_binary(h, N, deadline, max_iters=3000):
    """Binary optimization guided by continuous gradient"""
    h = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    
    best_h = h.copy()
    best_c = np.max(fft_corr(h))
    stall = 0

            best_h = h.copy()
            stall = 0
        else:

        # Find best gradient-guided swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        g_ones = grad[ones]
        g_zeros = grad[zeros]

        best_delta = 0.0
        best_oi, best_zi = -1, -1
        n_try = min(15, len(ones))

        top_zeros = zeros[np.argsort(g_zeros)][:n_try]

        rand_zeros = np.random.choice(zeros, min(5, len(zeros)), replace=False)
        candidates_o = np.unique(np.concatenate([top_ones, rand_ones]))
        candidates_z = np.unique(np.concatenate([top_zeros, rand_zeros]))

            for zi in candidates_z:
                h[oi], h[zi] = 0.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = pk - nP
                if d > best_delta:
                    best_delta = d
                    best_oi, best_zi = oi, zi
                h[oi], h[zi] = 1.0, 0.0

            h[best_oi], h[best_zi] = 0.0, 1.0

            oi = np.random.choice(ones)
            zi = np.random.choice(zeros)
            h[oi], h[zi] = 0.0, 1.0
        
        if it % 500 == 0:
            alpha *= 0.95
    
    return best_h, best_c
```

```python
    return np.correlate(h, 1.0 - h, mode='full')

def linear_grad_at_lag(lag, h, N):
    """Gradient of corr_full[h, 1-h] at given lag w.r.t. h"""
    g = np.zeros(N)
    if lag == 0:
        g[:] = 2.0 * h - 1.0  # d/dh of h*(1-h) = 1-2h, but we have sum, so...
        # Actually corr[lag=0] = sum h[i]*(1-h[i])
        # d/dh[j] = (1-h[j]) - h[j] = 1-2h[j]
    elif lag > 0:
        # corr = sum_{i=0}^{N-1-lag} h[i]*(1-h[i+lag])
        # d/dh[j]: 
        #   if j in [0, N-1-lag]: term h[j]*(1-h[j+lag]) gives (1-h[j+lag])
        #   if j in [lag, N-1]: term h[j-lag]*(1-h[j]) gives -h[j-lag]
        g[:N-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:N-lag]
    else:
        M = -lag
        # corr = sum_{i=M}^{N-1} h[i]*(1-h[i-M])
        # d/dh[j]:
        #   if j in [M, N-1]: h[j]*(1-h[j-M]) gives (1-h[j-M])
        #   if j in [0, N-1-M]: h[j+M]*(1-h[j]) gives -h[j+M]
        g[M:] += (1.0 - h[:N-M])
        g[:N-M] -= h[M:]
    return g

def grad_sum_active(corr, h, N, tol=0.01):
    """Sum of gradients over active lags (near-max)"""
    pk = corr.max()
    ctr = N - 1
    active = np.where(corr >= pk - tol)[0]
    g = np.zeros(N)
    for ai in active:
        lag = int(ai - ctr)
        g += linear_grad_at_lag(lag, h, N)
    g /= len(active)
    return g

    if abs(s - target) < 1e-12:
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(corr_full(h))) * (2.0 / N)

def grad_at_lag(lag, h, N):
    g = np.zeros(N)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:N-lag] += 1.0 - h[lag:]
        g[lag:] -= h[:N-lag]
    else:
        M = -lag
        g[M:] += 1.0 - h[:N-M]
        g[:N-M] -= h[M:]
    return g

def total_grad(h, N, tol=0.02):
    corr = corr_full(h)
    pk = corr.max()
    ctr = N - 1
    active = np.where(corr >= pk - tol)[0]
    g = np.zeros(N)
    for ai in active:
        g += grad_at_lag(int(ai - ctr), h, N)
    g /= len(active)
    return g, corr, pk

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    corr = np.real(np.fft.ifft(np.fft.fft(h, 2*N) * np.conj(np.fft.fft(1-h, 2*N))))
    corr = np.concatenate([corr[:N], corr[N-1:0:-1]])  # hmm this is getting complicated
    # Let me just use direct
    c = corr_full(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * grad_at_lag(int(i - ctr), h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_opt(h, N, deadline, T=30.0, max_outer=2):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        try:
            res = sp_min(smooth_obj, h, args=(T, target, lam),
                         jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                         options={'maxiter': 100, 'maxfun': 250, 'ftol': 1e-14})
            h = proj01(res.x, target)
            v = smooth_obj(h, T, target, 0.0)[0]
            if v < best_v:
                best_v = v
                best_h = h.copy()
            lam *= 2.5
        except:
            break
    return best_h, best_v * (2.0 / N)

def grad_binary_search(h, N, deadline, max_iters=3000):
    """Binary search with gradient-guided move selection"""
    h = (h > 0.5).astype(float)
    target = N // 2
    h = fix_count(h, target)
    best_h = h.copy()
    best_pk = float(np.max(corr_full(h)))
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        
        corr = corr_full(h)
        pk = float(corr.max())
        
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Get gradient
        g, _, _ = total_grad(h, N, tol=0.02)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Gradient-guided candidates
        # Want to flip 1→0 where g is most positive, 0→1 where g is most negative
        go = g[ones]
        gz = g[zeros]
        
        # Top gradient candidates + random
        n_top = 10
        top_o = ones[np.argsort(go)][-n_top:]  # highest g at ones
        top_z = zeros[np.argsort(gz)][:n_top]   # lowest g at zeros
        rand_o = np.random.choice(ones, min(5, len(ones)), replace=False)
        rand_z = np.random.choice(zeros, min(5, len(zeros)), replace=False)
        cand_o = np.unique(np.concatenate([top_o, rand_o]))
        cand_z = np.unique(np.concatenate([top_z, rand_z]))
        
        best_d = -np.inf
        best_oi, best_zi = -1, -1
        
        for oi in cand_o:
            for zi in cand_z:
                h[oi], h[zi] = 0.0, 1.0
                nP = float(np.max(corr_full(h)))
                d = pk - nP
                if d > best_d:
                    best_d = d
                    best_oi, best_zi = oi, zi
                h[oi], h[zi] = 1.0, 0.0
        
        if best_oi >= 0 and (best_d > 0 or np.random.random() < 0.1):
            h[best_oi], h[best_zi] = 0.0, 1.0
    
    return best_h, best_pk

def block_binary_search(h, N, deadline, max_iters=500):

    h = (h > 0.5).astype(float)
    target = N // 2

    best_h = h.copy()
    best_pk = float(np.max(corr_full(h)))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        pk = float(np.max(corr_full(h)))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        
        r = np.random.random()
        if r < 0.4:
            # Single swap (gradient-guided would be better but keep simple)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            oi = np.random.choice(ones)
            zi = np.random.choice(zeros)
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(corr_full(h)))
            if nP > pk + 0.3:
                h[oi], h[zi] = 1.0, 0.0  # revert if too bad
        elif r < 0.7:

                                                        h[s1:s1+blen].copy())
                    found = True

                zeros = np.where(h == 0)[0]

                    h[oi], h[zi] = 1.0, 0.0

            if nP > pk + 1.0:
                h[a:b+1] = h[a:b+1][::-1].copy()  # revert
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_min

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(corr_full(h))) * (2.0 / N)

def grad_at_lag(lag, h, N):
    g = np.zeros(N)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:N-lag] += 1.0 - h[lag:]
        g[lag:] -= h[:N-lag]
    else:
        M = -lag
        g[M:] += 1.0 - h[:N-M]
        g[:N-M] -= h[M:]
    return g

def total_grad(h, N, tol=0.02):
    corr = corr_full(h)
    pk = float(corr.max())
    ctr = N - 1
    active = np.where(corr >= pk - tol)[0]
    g = np.zeros(N)
    for ai in active:
        g += grad_at_lag(int(ai - ctr), h, N)
    g /= len(active)
    return g, corr, pk

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fix_count(h, target):
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    return h

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = corr_full(h)
    mx = float(c.max())
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * grad_at_lag(int(i - ctr), h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_opt(h, N, deadline, T=30.0, max_outer=2):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        try:
            res = sp_min(smooth_obj, h, args=(T, target, lam),
                         jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                         options={'maxiter': 100, 'maxfun': 250, 'ftol': 1e-14})
            h = proj01(res.x, target)
            v = smooth_obj(h, T, target, 0.0)[0]
            if v < best_v:
                best_v = v
                best_h = h.copy()
            lam *= 2.5
        except Exception:
            break
    return best_h, best_v * (2.0 / N)

def grad_binary_search(h, N, deadline, max_iters=3000, n_cand=12):
    h = fix_count(h, N // 2)
    target = N // 2
    best_h = h.copy()
    best_pk = float(np.max(corr_full(h)))
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        corr = corr_full(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        g, _, _ = total_grad(h, N, tol=0.02)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        go = g[ones]
        gz = g[zeros]
        nt = min(n_cand, len(ones))
        top_o = ones[np.argsort(go)][-nt:]
        top_z = zeros[np.argsort(gz)][:nt]
        rn = min(4, len(ones))
        rand_o = np.random.choice(ones, rn, replace=False)
        rand_z = np.random.choice(zeros, rn, replace=False)
        cand_o = np.unique(np.concatenate([top_o, rand_o]))
        cand_z = np.unique(np.concatenate([top_z, rand_z]))
        best_d = -np.inf
        best_oi, best_zi = -1, -1
        for oi in cand_o:
            for zi in cand_z:
                h[oi], h[zi] = 0.0, 1.0
                nP = float(np.max(corr_full(h)))
                d = pk - nP
                if d > best_d:
                    best_d = d
                    best_oi, best_zi = oi, zi
                h[oi], h[zi] = 1.0, 0.0
        if best_oi >= 0 and (best_d > 0 or np.random.random() < 0.08):
            h[best_oi], h[best_zi] = 0.0, 1.0
    return best_h, best_pk

def block_binary_search(h, N, deadline, max_iters=400):
    h = fix_count(h, N // 2)
    best_h = h.copy()
    best_pk = float(np.max(corr_full(h)))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        pk = float(np.max(corr_full(h)))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        r = np.random.random()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if r < 0.35:
            oi = np.random.choice(ones)
            zi = np.random.choice(zeros)
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(corr_full(h)))
            if nP > pk + 0.3:
                h[oi], h[zi] = 1.0, 0.0
        elif r < 0.7:
            blen = np.random.randint(2, 5)
            done = False
            for _ in range(12):
                s1 = np.random.randint(0, N - blen + 1)
                s2 = np.random.randint(0, N - blen + 1)
                if abs(s1 - s2) < blen:
                    continue
                if np.all(h[s1:s1+blen] == 1) and np.all(h[s2:s2+blen] == 0):
                    a = h[s1:s1+blen].copy()
                    h[s1:s1+blen] = 0.0
                    h[s2:s2+blen] = 1.0
                    nP = float(np.max(corr_full(h)))
                    if nP > pk + 0.5:
                        h[s1:s1+blen] = a
                        h[s2:s2+blen] = 0.0
                    done = True
                    break
            if not done:
                oi = np.random.choice(ones)
                zi = np.random.choice(zeros)
                h[oi], h[zi] = 0.0, 1.0
                nP = float(np.max(corr_full(h)))
                if nP > pk + 0.3:
                    h[oi], h[zi] = 1.0, 0.0
        else:
            a = np.random.randint(0, N - 3)
            b = np.random.randint(a + 3, N)
            seg = h[a:b+1].copy()
            h[a:b+1] = seg[::-1]
            nP = float(np.max(corr_full(h)))
            if nP > pk + 1.0:
                h[a:b+1] = seg
    return best_h, best_pk

def make_diverse(h_base, N, n=5):
    results = [h_base.copy()]
    target = N // 2
    for i in range(n):
        h = h_base.copy()
        n_flip = max(5, N // (12 + 4 * i))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if n_flip < len(ones) and n_flip < len(zeros):
            h[np.random.choice(ones, n_flip, replace=False)] = 0
            h[np.random.choice(zeros, n_flip, replace=False)] = 1
        results.append(fix_count(h, target))
    return results
```
