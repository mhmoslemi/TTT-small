import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    # D[p, k] = change in correlation[k] when h[p] goes from 1 to 0
    # For shift s (0-indexed, s = k - (N-1)):
    #   h contributes at position p as h[n+k] when n = p - s, i.e., term h[p]*(1-h[p-s])
    #   h contributes at position p as h[n] when n = p, i.e., term h[p+s]*(1-h[p])
    # Change when h[p]: 1->0:
    #   Lose: h[p]*(1-h[p-s]) = 1*(1-h[p-s]) for valid p-s
    #   Gain: h[p+s]*(1-0) - h[p+s]*(1-1) = h[p+s] for valid p+s
    # So D[p, s+N-1] = -(1-h[p-s]) + h[p+s] (with out-of-range = 0)
    
    shifts = np.arange(2 * N - 1) - (N - 1)  # shift values: -(N-1) to (N-1)
    pos = np.arange(N)[:, None]  # (N, 1)
    sv = shifts[None, :]  # (1, 2N-1)
    pm = pos - sv  # p - s: position for "as h[n+k]"
    pp = pos + sv  # p + s: position for "as h[n]"
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def best_swap_search(c, D, ones, zeros, sf=1.0, seed=0):
    """Find the swap (p from ones, q from zeros) that minimizes max(c + D[p]-D[q])."""
    rng = np.random.RandomState(seed)
    n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
    p_list = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
    P = float(c.max())
    Dz = D[zeros, :]  # (n_zeros, 2N-1)
    best_new_P = P + 1.0
    best_pq = None
    for p in p_list:
        base = c - D[p, :]
        cand = base[None, :] - Dz
        npk = cand.max(axis=1)
        mi = int(np.argmin(npk))
        if npk[mi] < best_new_P:
            best_new_P = float(npk[mi])
            best_pq = (int(p), int(zeros[mi]))
    return best_pq, best_new_P, P

def sa_optimize(h, N, deadline, max_sweeps=10000, T0=3.0, Tcool=0.995, T_min=0.03,
                sf=1.0, kick_every=100, kick_size=4, seed=0):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    sweep = 0
    while sweep < max_sweeps:
        if time.time() > deadline:
            break
        sweep += 1
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            continue
        D = compute_D(h, N)
        pq, new_P, P = best_swap_search(c, D, ones, zeros, sf=sf, seed=sweep + seed)
        if pq is None:
            break
        delta = new_P - P
        accept = delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0.0, delta) / max(T, 1e-10)))
        if accept:
            h[pq[0]] = 0.0
            h[pq[1]] = 1.0
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def softmax_continuous(h, N, deadline, k=100, lr=0.03, beta=50.0, seed=0):
    """Optimize using log-sum-exp approximation of max."""
    rng = np.random.RandomState(seed)

        # Log-sum-exp for smooth max

        logsumexp = cmax + np.log(np.sum(np.exp(corr - cmax)))
        # Gradient of log-sum-exp w.r.t. corr

        # Gradient of corr w.r.t. h: for shift s, d(corr[s])/dh[i] = (1-h[i-s]) (if i-s valid) + ...

        # d/dh[p] corr[k]:

        #   if p = n: -h[n+s] = -h[p+s]
        # So grad[p] = sum_k weights[k] * [(1-h[max(0,p-s)] if valid) - h[min(N-1,p+s)] if valid]
        
        # Build gradient

        # D[p,k] depends on h[p-s] and h[p+s]

        # (1-h[p-s]) for valid p-s

            pass
        
        # Simpler: just compute D and do matrix-vector product
        D_full = compute_D(h, N)  # (N, 2N-1)
        grad = -(D_full.T @ weights)  # (N,)
        
        # Nesterov step
        h_new = project_sum(h - lr * grad, target)
        m = 0.8 * (h_new - h)
        h = h_new
        it += 1

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def find_best_swap(c, D, ones, zeros, sf, rng):
    n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
    p_list = ones if sf >= 1.0 else rng.choice(ones, n_s, replace=False)
    Dz = D[zeros, :]
    best_new_P = 1e9
    best_pq = None
    for p in p_list:
        base = c - D[p, :]
        cand = base[None, :] - Dz
        npk = cand.max(axis=1)
        mi = int(np.argmin(npk))
        if npk[mi] < best_new_P:
            best_new_P = float(npk[mi])
            best_pq = (int(p), int(zeros[mi]))
    return best_pq, best_new_P

def sa_run(h, N, deadline, max_sweeps, T0, Tcool, T_min, sf, kick_every, kick_size, seed):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            n_kick = kick_size
            for _ in range(n_kick):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                if p != q:
                    h[p], h[q] = 0.0, 1.0
            stall = 0
            T = max(T0 * 0.4, T_min)
            continue
        D = compute_D(h, N)
        pq, new_P = find_best_swap(c, D, ones, zeros, sf, rng)
        if pq is None:
            break
        delta = new_P - P
        accept = delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(0.0, -delta) / max(T, 1e-10)))
        if accept:
            h[pq[0]] = 0.0
            h[pq[1]] = 1.0
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def greedy_run(h, N, deadline, max_sweeps, sf, seed):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        pq, new_P = find_best_swap(c, D, ones, zeros, sf, rng)
        if pq is None or new_P >= P - 1e-14:
            if pq is not None and new_P < P - 1e-14:
                h[pq[0]] = 0.0
                h[pq[1]] = 1.0
                continue
            break
        h[pq[0]] = 0.0
        h[pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def continuous_opt(h, N, deadline, lr0, seed):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        cmax = corr.max()
        logsumexp = cmax + np.log(np.sum(np.exp(corr - cmax + 1)))
        weights = np.exp(corr - logsumexp + 1)
        D = compute_D(y, N)
        grad = D.T @ weights
        h_new = project_sum(y - lr * grad, target)
        m = 0.85 * (h_new - h)
        h = h_new
        it += 1
        if it % 30 == 0:
            pk = float(np.max(fft_corr(h)))
            if pk < best_pk - 1e-14:
                best_pk = pk
                best_h = h.copy()
            lr *= 0.995
    pk = float(np.max(fft_corr(h)))
    if pk < best_pk:
        best_pk = pk
        best_h = h.copy()
    return best_h, best_pk * dx

def perturb_binary(h, N, n_perturb, seed):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_perturb):
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        if p != q:
            h[p], h[q] = 0.0, 1.0
    return ensure_sum(h, N)
