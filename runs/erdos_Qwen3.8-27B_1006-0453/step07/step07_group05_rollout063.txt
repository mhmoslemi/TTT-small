Let j = i-k, so k = i-j:
B[i] = Σ_{j: j∈[0,N), i-j∈[-(N-1),N-1]} w[i-j] * h[j]
This is the convolution of w and h evaluated at i.

import numpy as np
import time

def c5_val(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, 'full'))) * (2.0 / N)

def bin_proj(h, target):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h, kind='stable')
    out = np.zeros_like(h)
    out[idx[:target]] = 1.0
    return out

def sa_binary(h, N, deadline, T0=0.055, stall=2500):
    target = N // 2
    h = bin_proj(h, target)
    cur = float(np.max(np.correlate(h, 1 - h, 'full')))
    best, best_h = cur, h.copy()
    T, no_imp = T0, 0
    while time.time() < deadline:
        r = np.random.random()
        if r < 0.5:
            oi = int(np.random.choice(np.where(h == 1)[0]))
            zi = int(np.random.choice(np.where(h == 0)[0]))
            if oi == zi:
                continue
            h[oi], h[zi] = 0, 1
            new = float(np.max(np.correlate(h, 1 - h, 'full')))
            dE = new - cur
            if dE < -1e-15 or (T > 1e-10 and np.random.random() < np.exp(-dE / T)):
                cur = new
                if cur < best:
                    best, best_h, no_imp = cur, h.copy(), 0
                else:
                    no_imp += 1
            else:
                h[oi], h[zi] = 1, 0
        elif r < 0.82:
            ones, zeros = np.where(h == 1)[0], np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            o1, o2 = np.random.choice(ones, 2, replace=False)
            z1, z2 = np.random.choice(zeros, 2, replace=False)
            h[o1], h[o2], h[z1], h[z2] = 0, 0, 1, 1
            new = float(np.max(np.correlate(h, 1 - h, 'full')))

                    no_imp += 1
            else:

        else:

    L = N - 1
    while time.time() < deadline - 1:

            else:
                hi2 = mid

        if c_new < best_c:

        h = h_new
        lr = max(lr * 0.9992, 1e-5)
        soft = max(soft * 0.999, 0.005)
    return best_h, best_c
