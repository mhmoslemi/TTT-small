    """D[p, lag] = change in correlation[lag] when h[p] goes 1->0"""
    center = N - 1
    # lag in [0, 2N-2], actual shift = lag - center
    lags = np.arange(2 * N - 1) - center  # shape (2N-1,)
    # For position p and shift s: (1-h)[p+s] if 0<=p+s<N, else 0
    # The contribution of h[p]=1 to corr at lag corresponding to shift s is (1-h[p+s])
    # So removing h[p] changes corr[lag] by -(1-h[p+s]) = h[p+s]-1
    # where s = lag - center

    h_shifted = np.where(valid, h[idx_c], 0.0)

        # D[p, :] = change in corr when removing h[p]
        # Adding h[q] changes corr by -(D[q, :]) (since going 0->1 is opposite)
        # Wait: D[p,:] is the change when h[p] goes 1->0
        # When h[q] goes 0->1, the change is the negative of D[q,:]

def fista_continuous(h, N, deadline, k_top=60, lr0=0.03, beta0=0.7, max_iter=20000):
    """FISTA-style proximal gradient for continuous relaxation"""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    t_k = 1.0
    lr = lr0
    beta = beta0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        
        # Subgradient: top-k lags
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        
        # FISTA step
        h_new = y - lr * grad
        # Project onto {0 <= h <= 1, sum(h) = target}
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = h_new.sum() - target
        if abs(diff) > 1e-10:
            lo, hi = -1.0, 1.0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                if float(np.clip(h_new - mid, 0, 1).sum()) > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        m = ((t_k - 1) / t_new) * (h_new - y)
        h = h_new
        t_k = t_new
        
        if it % 500 == 0:
            lr *= 0.95
            beta = min(0.97, beta0 + it * 0.00001)
    
    return best_h, best_pk * dx

        if time.time() > deadline:

        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_nP = P + 1.0
        best_pq = None
        Dz = D[zeros, :]
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP:
                best_nP = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def sa_1swap(h, N, deadline, T0=5.0, Tcool=0.992, T_min=0.02, max_iter=20000, sf=0.8, kick_every=180, kick_size=4):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stall = 0
            T = min(T0, T * 2.5)
            c = fft_corr(h)
            P = float(c.max())
            continue
        D = compute_D(h, N)
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_nP = P + 1.0
        best_pq = None
        Dz = D[zeros, :]
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_nP:
                best_nP = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_nP - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta) / max(T, 1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0 / N)

def block_kick(h, N, n_blocks=3, max_block_size=8):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) < n_blocks or len(zeros) < n_blocks:
        return h
    for _ in range(n_blocks):
        start = np.random.randint(N)
        bsz = np.random.randint(2, max_block_size + 1)
        bsz = min(bsz, N - start)
        block = slice(start, start + bsz)
        n_one_in_block = int(h[block].sum())
        n_zero_in_block = bsz - n_one_in_block
        if n_one_in_block > 0 and n_zero_in_block > 0:
            h[block] = 1.0 - h[block]
            # Fix count
            target = N // 2
            cur = int(h.sum())
            if cur > target:
                idx = np.where(h == 1)[0]
                rem = idx[np.random.choice(len(idx), cur - target, replace=False)]
                h[rem] = 0.0
            elif cur < target:
                idx = np.where(h == 0)[0]
                add = idx[np.random.choice(len(idx), target - cur, replace=False)]
                h[add] = 1.0
    return h

def fista_cont(h, N, deadline, k_top=55, lr0=0.025):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    t_k = 1.0
    lr = lr0
    for it in range(50000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        hn = y - lr * grad
        hn = np.clip(hn, 0.0, 1.0)
        diff = hn.sum() - target
        if abs(diff) > 1e-10:
            lo, hi = -1.0, 1.0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                if float(np.clip(hn - mid, 0, 1).sum()) > target:
                    lo = mid
                else:
                    hi = mid
            hn = np.clip(hn - (lo + hi) * 0.5, 0, 1)
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        m = ((t_k - 1) / t_new) * (hn - y)
        h = hn
        t_k = t_new
        if it % 800 == 0:
            lr *= 0.94
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    tgt = N0 // 2
    best_h = h0.copy()
    best_c5 = c5_of(best_h)

    # Phase 1: Quick greedy
    h_b = to_binary(h0, tgt)
    t_dead = min(deadline - 800, t0 + 30)
    h1, c51 = greedy_1swap(h_b, N0, t_dead, max_iter=400, sf=1.0)
    if c51 < best_c5:
        best_c5, best_h = c51, h1.copy()

    # Phase 2: SA main
    t_dead = min(deadline - 600, t0 + 180)
    h2, c52 = sa_1swap(h_b, N0, t_dead, T0=7.0, Tcool=0.991, T_min=0.02, max_iter=20000, sf=0.7, kick_every=160, kick_size=5)
    if c52 < best_c5:
        best_c5, best_h = c52, h2.copy()

    # Phase 3: SA with block kicks
    h3 = to_binary(best_h, tgt)
    h3 = block_kick(h3, N0, n_blocks=4, max_block_size=10)
    t_dead = min(deadline - 450, t0 + 320)
    h3b, c53 = sa_1swap(h3, N0, t_dead, T0=9.0, Tcool=0.990, T_min=0.015, max_iter=15000, sf=1.0, kick_every=120, kick_size=6)
    if c53 < best_c5:
        best_c5, best_h = c53, h3b.copy()

    # Phase 4: FISTA continuous
    if time.time() < deadline - 250:
        hc = to_binary(best_h, tgt) + np.random.randn(N0) * 0.04
        hc = np.clip(hc, 0.05, 0.95)
        t_dead = min(deadline - 200, t0 + 440)
        h4, c54 = fista_cont(hc, N0, t_dead, k_top=50, lr0=0.022)
        if c54 < best_c5:
            best_c5, best_h = c54, h4.copy()

    # Phase 5: Binary round + greedy
    if time.time() < deadline - 120:
        h5b = to_binary(best_h, tgt)
        t_dead = min(deadline - 80, t0 + 510)
        h5, c55 = greedy_1swap(h5b, N0, t_dead, max_iter=1000, sf=1.0)
        if c55 < best_c5:
            best_c5, best_h = c55, h5.copy()

    # Phase 6: Multiple short SA restarts
    if time.time() < deadline - 60:
        for restart in range(3):
            if time.time() > deadline - 20:
                break
            hr = to_binary(best_h, tgt)
            nr = 10 + restart * 5
            for _ in range(nr):
                po = np.where(hr == 1)[0]
                pq = np.where(hr == 0)[0]
                if len(po) > 0 and len(pq) > 0:
                    hr[po[np.random.randint(len(po))]] = 0.0
                    hr[pq[np.random.randint(len(pq))]] = 1.0
            t_dead = min(deadline - 30 - restart * 15, time.time() + 60)
            hr2, c5r = sa_1swap(hr, N0, t_dead, T0=5.0 + restart * 2, Tcool=0.992, T_min=0.02, max_iter=8000, sf=0.9, kick_every=140, kick_size=4)
            if c5r < best_c5:
                best_c5, best_h = c5r, hr2.copy()

    # Phase 7: Final greedy polish
    if time.time() < deadline - 2:
        hF = to_binary(best_h, tgt)
        hF2, c5F = greedy_1swap(hF, N0, deadline - 1, max_iter=2000, sf=0.7)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Final evaluation
    h_out = best_h
    if np.allclose(h_out, np.round(h_out), atol=0.01):
        h_out = to_binary(h_out, tgt)
    final_c5 = c5_of(h_out)
    if final_c5 > best_c5 + 1e-6:
        h_out = np.clip(best_h, 0, 1)
        final_c5 = c5_of(h_out)
    return h_out, final_c5, len(h_out)
