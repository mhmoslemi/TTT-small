import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target_ones):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target_ones]] = 1.0
    return out

def fix_sum_bin(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target-n1]] = 1
    return h

def binary_search(h, N, deadline, max_iters, T0, T1, use_2pair=True, block_sizes=(1,1,1,2,2,3,5)):

    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    total_iters = max(1, max_iters)
    cool = (T1 / T0) ** (1.0 / total_iters)
    T = T0
    no_improve = 0
    stall_limit = 2000
    it = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline and it < max_iters:

        if use_2pair and r < 0.25:

            i = int(ones[np.random.randint(len(ones))])

            k = int(zeros[np.random.randint(len(zeros))])
            l = int(zeros[np.random.randint(len(zeros))])
            if i == j or k == l or i in (k, l) or j in (k, l):
                continue
            h[i], h[j], h[k], h[l] = 0, 0, 1, 1

            new_max = float(np.max(new_corr))

                cur_max = new_max
                corr = new_corr

                    best_h = h.copy()
                    no_improve = 0
                else:

            else:
                h[i], h[j], h[k], h[l] = 1, 1, 0, 0
        elif r < 0.60:

            if len(ones) < 1 or len(zeros) < 1:

            i = int(ones[np.random.randint(len(ones))])

            new_max = float(np.max(new_corr))

            i = 0

                    if j - i >= L:

                    while j < N and h[j] == 0:

                    if j - i >= L:

            if not one_starts or not zero_starts:

            ob = one_starts[int(np.random.randint(len(one_starts)))]

            if abs(ob - zb) < L:

            h_ob = h[ob:ob+L].copy()

                    no_improve = 0

            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, 0.02)
            no_improve = 0
    
    return best_h, best_c

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target]] = 1.0
    return out

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, target):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros_like(h)
    out[idx[:target]] = 1.0
    return out

def fix_sum_bin(h, target):
    h = h.copy()
    n1 = int(round(h.sum()))
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h

def init_random(N, seed):
    np.random.seed(seed)
    h = np.zeros(N)
    h[np.random.choice(N, N // 2, replace=False)] = 1
    return h

def init_thue(N, seed):
    np.random.seed(seed)
    pat = np.zeros(1)
    while len(pat) < N:
        pat = np.concatenate([pat, 1.0 - pat])
    h = pat[:N].copy()
    return fix_sum_bin(h, N // 2)

def init_quasi(N, seed):
    np.random.seed(seed)
    x = np.arange(N)
    v = np.zeros(N)
    for f in [3, 7, 13, 21, 37]:
        v += np.sin(2.0 * np.pi * f * x / N + 0.3 * f)
    h = (v > np.median(v)).astype(float)
    return fix_sum_bin(h, N // 2)

def init_jitter(N, seed, base):

    h = base.astype(float).copy()

    zeros = np.where(h == 0)[0]

        zi = int(zeros[np.random.randint(len(zeros))])
