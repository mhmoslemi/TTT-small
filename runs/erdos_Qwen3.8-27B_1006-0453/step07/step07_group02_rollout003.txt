def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_row(h, N, i):
    L = 2 * N - 1
    center = N - 1
    k = np.arange(L) - center
    i_mk = i - k
    i_pk = i + k
    valid_mk = (i_mk >= 0) & (i_mk < N)
    valid_pk = (i_pk >= 0) & (i_pk < N)
    A = np.where(valid_mk, 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where(valid_pk, h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def compute_C_full(h, N):
    L = 2 * N - 1
    center = N - 1
    pos = np.arange(N)[:, None]
    k_vals = np.arange(L)[None, :] - center
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    valid_mk = (i_mk >= 0) & (i_mk < N)
    valid_pk = (i_pk >= 0) & (i_pk < N)
    A = np.where(valid_mk, 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where(valid_pk, h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def fix_sum_binary(h, target):

        else:
            hi = mid

    beta = 0.9

        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:

            if Lk == 0:
                grad += (1.0 - 2.0 * y)
            elif Lk > 0:
                grad[Lk:] += (1.0 - y[:N-Lk])
                grad[:N-Lk] -= y[Lk:]
            else:

                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= k_top
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.992
            beta = min(0.97, 0.85 + it * 0.0004)

            delta = C[oc][:, None, :] + C[zeros][None, :, :]
            newP = (corr[None, None, :] + delta).max(axis=2)
            lm = float(newP.min())
            if lm < best_val:
                best_val = lm
                loc = np.unravel_index(newP.argmin(), newP.shape)
                best_ai = int(oc[loc[0]])
                best_bj = int(zeros[loc[1]])
        if best_val < P - 1e-12:
            h[best_ai] = 0
            h[best_bj] = 1
            corr = corr + C[best_ai] + C[best_bj]
            C[best_ai] = compute_C_row(h, N, best_ai)
            C[best_bj] = compute_C_row(h, N, best_bj)
            since_rc += 1
            if since_rc >= recompute_every:
                C = compute_C_full(h, N)
                since_rc = 0
            nm = float(np.max(corr))
            if nm < best_c - 1e-14:
                best_c = nm
                best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck >= kick_at:
                for _ in range(kick_n):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0
                    h[q] = 1
                corr = fft_corr(h)
                C = compute_C_full(h, N)
                since_rc = 0
                stuck = 0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                break
    return best_h, best_c * (2.0 / N)

def project_01(h, target_sum):
    h = np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def nesterov_opt(h, N, deadline, k_top=40, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            Lk = int(ai - center)
            if Lk == 0:
                grad += (1.0 - 2.0 * y)
            elif Lk > 0:
                grad[Lk:] += (1.0 - y[:N - Lk])
                grad[:N - Lk] -= y[Lk:]
            else:
                M = -Lk
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.992
            beta = min(0.97, 0.85 + it * 0.0004)
    return best_h, best_pk * (2.0 / N)

def simulated_anneal(h, N, deadline, T0=4.0, T1=0.02, max_iters=80000):
    h = h.astype(float).copy()
    fix_sum_binary(h, N // 2)
    corr = fft_corr(h)
    cur_c = float(np.max(corr))
    best_c = cur_c
    best_h = h.copy()
    for n in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (n / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = int(np.random.choice(ones))
        q = int(np.random.choice(zeros))
        h[p], h[q] = 0.0, 1.0
        new_corr = fft_corr(h)
        new_c = float(np.max(new_corr))
        if new_c < cur_c or np.random.random() < np.exp(-((new_c - cur_c) / max(T, 1e-8))):
            cur_c = new_c
            corr = new_corr
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
    return best_h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    fix_sum_binary(h_bin, N0 // 2)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Full-pair greedy local search (primary)
    d1 = t0 + 380
    hs, c5s = greedy_search(h_bin.copy(), N0, d1, max_iters=50000, kick_at=100, kick_n=5, recompute_every=35)
    if c5s < best_c5:
        best_c5, best_h = c5s, hs.copy()

    # Phase 2: Nesterov continuous relaxation
    if time.time() < deadline - 220:
        d2 = min(deadline - 180, time.time() + 110)
        hc = best_h + np.random.randn(N0) * 0.12
        hc = project_01(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k_top=35, lr0=0.01)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Simulated annealing (exact FFT)
    if time.time() < deadline - 150:
        d3 = min(deadline - 120, time.time() + 90)
        hsa = (best_h > 0.5).astype(float)
        fix_sum_binary(hsa, N0 // 2)
        o1 = np.where(hsa == 1)[0]
        z1 = np.where(hsa == 0)[0]
        for _ in range(8):
            hsa[np.random.choice(o1)] = 0
            hsa[np.random.choice(z1)] = 1
            o1 = np.where(hsa == 1)[0]
            z1 = np.where(hsa == 0)[0]
        hsa2, c5sa = simulated_anneal(hsa, N0, d3, T0=5.0, T1=0.01, max_iters=60000)
        if c5sa < best_c5:
            best_c5, best_h = c5sa, hsa2.copy()

    # Phase 4: Second greedy from perturbed start
    if time.time() < deadline - 120:
        d4 = min(deadline - 90, time.time() + 100)
        h4 = (best_h > 0.5).astype(float)
        fix_sum_binary(h4, N0 // 2)
        o4 = np.where(h4 == 1)[0]
        z4 = np.where(h4 == 0)[0]
        for _ in range(12):
            h4[np.random.choice(o4)] = 0
            h4[np.random.choice(z4)] = 1
            o4 = np.where(h4 == 1)[0]
            z4 = np.where(h4 == 0)[0]
        h4s, c54 = greedy_search(h4, N0, d4, max_iters=30000, kick_at=80, kick_n=5, recompute_every=30)
        if c54 < best_c5:
            best_c5, best_h = c54, h4s.copy()

    # Phase 5: Multi-resolution exploration
    for Ntry in [340, 400, 440]:
        if time.time() > deadline - 50:
            break
        if abs(Ntry - N0) < 5:
            continue
        x_b = np.linspace(0, 2, len(best_h), endpoint=False)
        x_n = np.linspace(0, 2, Ntry, endpoint=False)
        hi = np.interp(x_n, x_b, best_h)
        hb = (hi > 0.5).astype(float)
        fix_sum_binary(hb, Ntry // 2)
        sd = min(deadline - 15, time.time() + 40)
        hr, c5r = greedy_search(hb, Ntry, sd, max_iters=15000, kick_at=50, kick_n=4, recompute_every=30)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()

    # Phase 6: Final greedy polish
    if time.time() < deadline - 8:
        hF = (best_h > 0.5).astype(float)
        fix_sum_binary(hF, len(best_h) // 2)
        hF, c5F = greedy_search(hF, len(best_h), deadline - 1, max_iters=15000, kick_at=50, kick_n=3, recompute_every=25)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    final_h = (best_h > 0.5).astype(float)
    fix_sum_binary(final_h, len(best_h) // 2)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
