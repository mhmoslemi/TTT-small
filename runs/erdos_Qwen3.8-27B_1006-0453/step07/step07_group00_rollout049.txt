import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N, seed=0):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[rng.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[rng.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(h):
    N = len(h)
    s = 2.0 * h - 1.0
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return s, deltas

def update_delta(s, deltas, i, N):
    s[i] = -s[i]
    deltas[i, :] = 0.0
    deltas[i, i:i + N] = s

def sa_search(h_bin, deadline, seed=42):

        if len(ones) < 2 or len(zeros) < 2:

            nk = min(nk, len(ones), len(zeros))

            for idx in range(nk):

            corr = fft_corr(h)

        threshold = best_c * 0.9  # Consider lags within 90% of best

def continuous_polish(h_in, deadline, seed=42):
    """Projected gradient with momentum on continuous relaxation."""
    rng = np.random.RandomState(seed)
    N = len(h_in)
    h = h_in.astype(float).copy()
    # Perturb slightly
    h = h + rng.randn(N) * 0.03
    h = np.clip(h, 0, 1)
    h = np.clip(h - (h.sum() - N/2) / N, 0, 1)
    
    m = np.zeros(N)
    beta = 0.85
    lr = 0.008
    best_c = compute_c5(h, N)
    best_h = h.copy()
    
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_c:
            best_c = pk
            best_h = y.copy()
        
        # Gradient using top peaks
        k_top = 30
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        
        h_new = np.clip(y - lr * grad, 0, 1)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0, 1)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            lr *= 0.98
            beta = min(0.93, 0.80 + it * 0.0001)
    
    return best_h, best_c
