Actually, for the binary case, there's a neat trick: the change in correlation at lag k when we swap position i (from 1 to 0) with position j (from 0 to 1) is:
- At lag k: the contribution changes based on h(j) vs h(i) at positions shifted by k

For binary h, flipping h[i] from 1 to 0 and h[j] from 0 to 1:
- At lag k: old contribution had h[i]*(1-h[i+k]) + h[j]*(1-h[j+k])
- New contribution has 0*(1-h[i+k]) + 1*(1-h[j+k]) = 1-h[j+k]
- Change = (1-h[j+k]) - h[i]*(1-h[i+k])

This is computable in O(1) per swap per lag, but we need to check all lags to find the new max, which is O(N) where N is the number of lags (2N-1).

The FFT approach computes all lags in O(N log N), which is better for checking after each swap.

import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def smooth_obj(h, T, target, lam, N):
    c = fft_corr(h)
    cx = c - c.max()
    mx = c.max()
    e = np.exp(T * cx)
    es = e.sum()
    val = mx + np.log(es) / T
    g = np.zeros(N)
    ctr = N - 1
    w = e / es
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, N, deadline, T=30.0, max_outer=5, max_iter=300):
    h = h.astype(float).copy()
    target = float(N) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    lam = 1.0
    for outer in range(max_outer):
        if time.time() > deadline - 5:
            break
        res = sp_min(smooth_obj, h, args=(T, target, lam, N),
                     jac=True, method='L-BFGS-B', bounds=[(0,1)]*N,
                     options={'maxiter': max_iter, 'maxfun': max_iter*2, 'ftol': 1e-15})
        h = proj01(res.x, target)
        c5 = compute_c5(h, N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lam *= 3.0
    return best_h, best_c5

            xo = np.linspace(0, 2, Nt, endpoint=False)
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    bh = h0.copy()
    bc5 = compute_c5(bh)

    # Phase 1: quick binary
    d1 = min(dl - 700, t0 + 35)
    hb, cb = binopt(bh, N0, d1, 150, 18, 0.001, 0.995)
    if cb * 2.0 / N0 < bc5:
        bc5, bh = cb * 2.0 / N0, hb.copy()

    # Phase 2: L-BFGS main optimization (280s, diverse starts)
    if time.time() < dl - 480:
        d2 = min(dl - 430, time.time() + 280)
        # 4 runs from perturbations of best
        for r in range(4):
            if time.time() > d2:
                break
            hr = bh + np.random.randn(N0) * (0.14 - 0.015 * r)
            hr = proj01(hr, N0 / 2.0)
            hl, cl = lbfgs(hr, N0, d2, 20.0 + 5 * r, 5, 250)
            if cl < bc5:
                bc5, bh = cl, hl.copy()
        # 3 runs from diverse patterns
        if time.time() < d2 - 30:
            pats = []
            pr = (np.random.random(N0) > 0.5).astype(float)
            pats.append(pr)
            pp = (np.arange(N0) % 2 == 0).astype(float)
            pats.append(pp)
            p4 = (np.arange(N0) % 4 < 2).astype(float)
            pats.append(p4)
            for p in pats:
                if time.time() > d2 - 35:
                    break
                p = proj01(p, N0 / 2.0)
                hl, cl = lbfgs(p, N0, min(d2, time.time() + 35), 20.0, 3, 200)
                if cl < bc5:
                    bc5, bh = cl, hl.copy()

    # Phase 3: intensive binary with block moves (200s)
    if time.time() < dl - 260:
        d3 = min(dl - 220, time.time() + 200)
        hb3, cb3 = binopt(bh, N0, d3, 10000, 32, 0.0004, 0.9985)
        if cb3 * 2.0 / N0 < bc5:
            bc5, bh = cb3 * 2.0 / N0, hb3.copy()

    # Phase 4: multi-resolution (170s)
    if time.time() < dl - 110:
        d4 = min(dl - 90, time.time() + 170)
        bh, bc5 = res_search(bh, bc5, d4, [200, 240, 280, 320, 360, 400, 460, 520, 600, 700, 800])

    # Phase 5: final L-BFGS refinement (80s)
    if time.time() < dl - 35:
        d5 = min(dl - 18, time.time() + 80)
        for r in range(3):
            if time.time() > d5:
                break
            h5 = bh + np.random.randn(len(bh)) * (0.04 - 0.01 * r)
            h5 = proj01(h5, len(bh) / 2.0)
            hl5, cl5 = lbfgs(h5, len(bh), min(d5, time.time() + 30), 35.0 + 10 * r, 3, 300)
            if cl5 < bc5:
                bc5, bh = cl5, hl5.copy()

    # Phase 6: final binary polish (25s)
    if time.time() < dl - 3:
        hf, cbf = binopt(bh, len(bh), dl - 1, 3000, 28, 0.0003, 0.999)
        if cbf * 2.0 / len(bh) < bc5:
            bc5, bh = cbf * 2.0 / len(bh), hf.copy()

    Nf = len(bh)
    c5c = compute_c5(bh)
    fb = (bh > 0.5).astype(float)
    c5b = compute_c5(fb)
    if c5b < c5c:
        bh, bc5 = fb, c5b
    else:
        bc5 = c5c
    return bh, bc5, Nf
