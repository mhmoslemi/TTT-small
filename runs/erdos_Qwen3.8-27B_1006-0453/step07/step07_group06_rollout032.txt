2. Move δ from h[j] to h[j'] (or vice versa)
3. Choose δ to minimize the max correlation
4. Accept if improved

This is like a continuous version of the binary swap.

For efficiency, I can approximate the effect of a small δ:
corr_k changes by δ * (∂corr_k/∂h[j'] - ∂corr_k/∂h[j])

So the change in the max is approximately:
δ * max_k w_k * (∂corr_k/∂h[j'] - ∂corr_k/∂h[j])

where w_k are the weights from the log-sum-exp.

This gives a direction for the swap. I can then do a line search to find the optimal δ.

This is essentially a projected gradient step! The gradient at h is g, and the projected gradient step in the "swap" direction (e_{j'} - e_j) is:
δ* = -<g, e_{j'} - e_j> / ||e_{j'} - e_j||² = -(g[j'] - g[j])

So the optimal δ for the swap is -(g[j'] - g[j]), clipped to maintain [0,1] constraints.

This is just coordinate descent on the pair (j, j')! And the best pair is the one with the largest |g[j'] - g[j]|.

So the algorithm is:
1. Compute gradient g
2. Find the pair (j, j') that maximizes |g[j'] - g[j]| (with appropriate sign)
3. Move mass from the higher-gradient coordinate to the lower
4. Project to [0,1] and sum constraint
5. Repeat

This is essentially a "top-2 coordinate descent" which is very efficient!

    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_grad(h, T, target, lam):
    N = len(h)
    dx = 2.0 / N
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    es = e.sum()
    val = (mx + np.log(es + 1e-30) / T) * dx
    w = e / es

    g = np.zeros(N)

        if wi < 1e-10:
            continue
        Ls = i - ctr
        if Ls == 0:
            g += wi * (1.0 - 2.0 * h)
        elif Ls > 0:
            g[Ls:] += wi * (1.0 - h[:N - Ls])
            g[:N - Ls] -= wi * h[Ls:]

            M = -Ls
            g[:N - M] += wi * (1.0 - h[M:])
            g[M:] -= wi * h[:N - M]

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_grad(h, T, target, lam):
    N = len(h)
    dx = 2.0 / N
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    es = e.sum()
    val = (mx + np.log(es + 1e-30) / T) * dx
    w = e / es
    ctr = N - 1
    g = np.zeros(N)
    for i in np.where(w > 1e-10)[0]:
        wi = w[i]
        Ls = i - ctr
        if Ls == 0:
            g += wi * (1.0 - 2.0 * h)
        elif Ls > 0:
            g[Ls:] += wi * (1.0 - h[:N - Ls])
            g[:N - Ls] -= wi * h[Ls:]
        else:
            M = -Ls
            g[:N - M] += wi * (1.0 - h[M:])
            g[M:] -= wi * h[:N - M]
    g *= dx
    if lam > 0:
        s = h.sum() - target
        val += 0.5 * lam * s * s
        g = g + lam * s
    return val, g

def lbfgs_run(h, N, deadline, Ts=(20, 50, 100), max_it=120):

    tgt = N / 2.0
    bh, bc = h.copy(), compute_c5(h, N)

        if time.time() > deadline:
            break

        c = compute_c5(h, N)
        if c < bc:
