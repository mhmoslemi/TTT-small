```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    out = np.real(np.fft.irfft(H * G, L))
    return out

def c5_of(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def C_single(h, N, idx):
    k = np.arange(-(N-1), N)
    im = idx - k
    ip = idx + k
    A = np.zeros(2*N-1)
    B = np.zeros(2*N-1)
    m_ok = (im >= 0) & (im < N)
    p_ok = (ip >= 0) & (ip < N)
    A[m_ok] = 1.0 - h[im[m_ok]]
    B[p_ok] = h[ip[p_ok]]
    return A - B

def C_mat(h, N):
    C = np.zeros((N, 2*N-1))
    for i in range(N):
        C[i] = C_single(h, N, i)
```

```python
    corr = fft_corr(h)
    # Avoid overflow
    m = np.max(corr)
    exp_corr = np.exp(alpha * (corr - m))
    weights = exp_corr / np.sum(exp_corr)
    soft_max = m + np.log(np.sum(exp_corr)) / alpha
    return soft_max, weights, corr
```

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0-h, L)
    return np.real(np.fft.irfft(H*G, L))

def c5_of(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def C_single(vec, N, idx):
    k = np.arange(-(N-1), N)
    im = idx - k
    ip = idx + k
    A = np.where((im>=0)&(im<N), 1.0-vec[np.clip(im,0,N-1)], 0.0)
    B = np.where((ip>=0)&(ip<N), vec[np.clip(ip,0,N-1)], 0.0)
    return A - B

def C_mat_full(vec, N):
    pos = np.arange(N)[:,None]
    k = np.arange(-(N-1),N)[None,:]
    im = pos - k
    ip = pos + k
    A = np.where((im>=0)&(im<N), 1.0-vec[np.clip(im,0,N-1)], 0.0)
    B = np.where((ip>=0)&(ip<N), vec[np.clip(ip,0,N-1)], 0.0)
    return A - B

def project_01(h, target_sum):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        s_mid = np.clip(h-mid, 0, 1).sum()
        if s_mid > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def fix_binary_sum(h, target_ones):
    h = (h > 0.5).astype(float)
    cur = int(h.sum())
    if cur > target_ones:
        rem = cur - target_ones
        ones = np.where(h==1)[0]
        h[np.random.choice(ones, rem, replace=False)] = 0
    elif cur < target_ones:
        rem = target_ones - cur
        zeros = np.where(h==0)[0]
        h[np.random.choice(zeros, rem, replace=False)] = 1
    return h

def greedy_ls(h, N, deadline, max_rounds=5000):
    h = h.copy()
    C = C_mat_full(h, N)
    corr = fft_corr(h)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        P = float(np.max(corr))
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        # Full or sampled enumeration
        if len(ones)*len(zeros) <= 40000:
            os_, zs = ones, zeros
        else:
            ns = min(len(ones), 150)
            nz = min(len(zeros), 150)
            os_ = np.random.choice(ones, ns, replace=False)
            zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]
        Cz = C[zs]
        newP = np.max(corr[None,None,:] + Co[None,:,:] + Cz[:,None,:], axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv >= P - 1e-13:
            break
        p = int(os_[mi[1]])
        q = int(zs[mi[0]])
        h[p] = 0
        h[q] = 1
        corr = corr + Co[mi[1]] + Cz[mi[0]]
        C[p] = C_single(h, N, p)
        C[q] = C_single(h, N, q)
    return h, float(np.max(fft_corr(h)))*(2.0/N)
