import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(G * np.conj(H)))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.copy()
    s = h.sum()
    target = N // 2
    if not np.allclose(h, np.round(h)):
        # Round using threshold with correct sum
        sorted_h = np.sort(h)[::-1]
        thr = sorted_h[min(target, N-1)]
        h = (h >= thr).astype(float)
        # Fix if needed
        diff = target - h.sum()
        if diff > 0:
            margins = np.where(h == 0, h, -1)
            h[np.argsort(margins)[::-1][:diff]] = 1
        elif diff < 0:
            margins = np.where(h == 1, -h, N)
            h[np.argsort(margins)[:(-diff)]] = 0
    return h

def get_C_row(h, N, idx):
    """Get C[idx] = A - B where A[L]=(1-h[idx-L]) and B[L]=h[idx+L]"""
    k_vals = np.arange(2 * N - 1) - (N - 1)
    i_mk = idx - k_vals
    i_pk = idx + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def binary_anneal(h, N, deadline, max_iters=20000, T0=2.0, T1=0.01, sample_sz=100):
    """Simulated annealing with swap moves."""
    h = h.astype(float).copy()
    target = N // 2
    # Ensure binary with correct sum
    if h.sum() != target:
        s = int(h.sum())
        if s > target:
            ones = np.where(h == 1)[0]
            h[np.random.choice(ones, s - target, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[np.random.choice(zeros, target - s, replace=False)] = 1
    
    C = np.stack([get_C_row(h, N, i) for i in range(N)])
    corr = fft_corr(h).copy()
    best_c = np.max(corr)
    best_h = h.copy()
    cur_c = best_c
    
    n_total = 2 * N - 1
    center = N - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(1, max_iters))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Sample candidate moves

def evaluate_swap_topk(corr, h, N, o_s, z_s, topk=30):
    """Evaluate max correlation after swapping each (o,z) pair, using only top-k peaks."""
    N = len(h)
    L = 2*N - 1
    center = N - 1
    k_vals = np.arange(L) - center
    
    # Get top-k indices
    top_idx = np.argpartition(corr, -topk)[-topk:]
    
    # For each o in o_s: A_o[top_idx] = (1-h[o - k_vals[top_idx]]) for valid
    A = np.zeros((len(o_s), topk))
    for i, o in enumerate(o_s):
        idx = o - k_vals[top_idx]
        valid = (idx >= 0) & (idx < N)
        A[i, valid] = 1.0 - h[idx[valid]]
    
    # For each z in z_s: D_z[top_idx] = (1-h[z-k_vals[top_idx]])[valid_a] - h[z+k_vals[top_idx]][valid_b]
    D = np.zeros((len(z_s), topk))
    for i, z in enumerate(z_s):
        idx_a = z - k_vals[top_idx]
        va = (idx_a >= 0) & (idx_a < N)
        idx_b = z + k_vals[top_idx]
        vb = (idx_b >= 0) & (idx_b < N)
        D[i, va] += 1.0 - h[idx_a[va]]
        D[i, vb] -= h[idx_b[vb]]

    # max over top_idx
    new_vals = corr[top_idx][None, None, :] - A[:, None, :] + D[None, :, :]

    return new_max

    """Compute max of new correlation after swap (o:1→0, z:0→1) without modifying h."""
    L = 2*N - 1
    center = N - 1
    k_vals = np.arange(L) - center

    idx_om = o - k_vals
    idx_om_valid = (idx_om >= 0) & (idx_om < N)
    idx_zm = z - k_vals
    idx_zm_valid = (idx_zm >= 0) & (idx_zm < N)
    idx_zp = z + k_vals
    idx_zp_valid = (idx_zp >= 0) & (idx_zp < N)

    delta[idx_zm_valid] += (1.0 - h[idx_zm[idx_zm_valid]])
    delta[idx_zp_valid] -= h[idx_zp[idx_zp_valid]]
    
    new_corr = corr + delta
    return float(np.max(new_corr))

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.conj(np.fft.fft(1.0-h, L))))

def c5_val(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * 2.0 / N

def make_bin(h, N):
    """Round h to binary with exactly N//2 ones."""
    target = N // 2
    s = int(round(h.sum()))
    h = h.copy()
    if s != target:
        # Use thresholding
        thr = np.sort(h)[::-1][target]
        h = (h >= thr).astype(float)
        if h.sum() > target:
            h[np.where(h==1)[0][np.random.choice(np.where(h==1)[0].size, int(h.sum()-target), replace=False)]] = 0
        elif h.sum() < target:
            h[np.where(h==0)[0][np.random.choice(np.where(h==0)[0].size, int(target-h.sum()), replace=False)]] = 1
    return h

def delta_for_swap(h, N, o, z):
    """Compute delta c for swapping h[o]=1→0, h[z]=0→1."""
    L_arr = 2*N - 1
    center = N - 1
    k = np.arange(L_arr) - center
    io = o - k
    iz_m = z - k
    iz_p = z + k
    d = np.zeros(L_arr)
    vo = (io >= 0) & (io < N)
    vm = (iz_m >= 0) & (iz_m < N)
    vp = (iz_p >= 0) & (iz_p < N)
    d[vo] -= (1.0 - h[np.clip(io[vo], 0, N-1)])
    d[vm] += (1.0 - h[np.clip(iz_m[vm], 0, N-1)])
    d[vp] -= h[np.clip(iz_p[vp], 0, N-1)]
    return d

def local_search(h, N, deadline, T0=1.5, T1=0.05, sample_sz=80, topk=30):
    """Simulated annealing local search for binary h."""
    h = h.astype(float).copy()
    corr = fft_corr(h).copy()
    best_c = float(np.max(corr))
    best_h = h.copy()
    cur_c = best_c
    L_arr = 2*N - 1
    center = N - 1
    k = np.arange(L_arr) - center
    
    # Precompute index arrays for fast evaluation
    max_it = 30000
    it = 0
    stuck = 0
    
    while time.time() < deadline and it < max_it:
        it += 1
        T = T0 * (T1/T0) ** (it / max_it)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        o_s = np.random.choice(ones, ns, replace=False)
        z_s = np.random.choice(zeros, nz, replace=False)
        
        # Compute delta for all pairs using top-k screening
        # First, get the top-k shift indices
        top_idx = np.argpartition(corr, -topk)[-topk:]
        
        # For each o: A_o[top_idx] = (1-h[o-k[top_idx]])[valid]
        o_k = o_s[:, None] - k[top_idx][None, :]  # (ns, topk)
        o_v = (o_k >= 0) & (o_k < N)
        A_o = np.where(o_v, 1.0 - h[np.clip(o_k, 0, N-1)], 0.0)
        
        # For each z: D_z[top_idx] = (1-h[z-k[top_idx]])[va] - h[z+k[top_idx]][vb]
        z_km = z_s[:, None] - k[top_idx][None, :]
        z_vam = (z_km >= 0) & (z_km < N)
        z_kp = z_s[:, None] + k[top_idx][None, :]
        z_vap = (z_kp >= 0) & (z_kp < N)
        D_z = np.where(z_vam, 1.0 - h[np.clip(z_km, 0, N-1)], 0.0) - np.where(z_vap, h[np.clip(z_kp
