import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def fix_sum_binary(h, target):
    h = np.round(h).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h

def anneal(h, N, deadline, T0=0.05, T1=1e-7, max_iters=80000, use_2swap=True):
    h = h.astype(float).copy()
    nh = N // 2
    h = fix_sum_binary(h, nh)
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * 2.0 / N
    best_h = h.copy()
    
    T = T0
    n_iters = max(1, int((deadline - time.time()) / 0.001)) # rough
    cool = (T1 / T0) ** (1.0 / max(100, max_iters))
    no_improve = 0
    it = 0
    stall_limit = 6000
    
    while time.time() < deadline and it < max_iters:
        it += 1
        r = np.random.random()
        accepted = False
        
        if r < 0.55 or not use_2swap:
            # Single swap with worst-lag targeting
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            peak_idx = np.argmax(corr)
            center = N - 1
            lag = peak_idx - center
            
            if np.random.random() < 0.5:
                # Targeted: pick a 1 that contributes to peak lag
                if lag >= 0:
                    # h[i]=1, h[i+lag]=0 are bad pairs for positive lag
                    candidates = ones[ones + lag < N]
                    if len(candidates) > 0:
                        valid = [i for i in candidates if h[i + lag] == 0]
                        if valid:
                            oi = int(np.random.choice(valid))
                        else:
                            oi = int(np.random.choice(ones))
                    else:
                        oi = int(np.random.choice(ones))
                else:
                    candidates = ones[ones - lag < N]
                    if len(candidates) > 0:
                        valid = [i for i in candidates if h[i - lag] == 0]
                        if valid:
                            oi = int(np.random.choice(valid))
                        else:
                            oi = int(np.random.choice(ones))
                    else:
                        oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
            else:
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
            
            if oi == zi:
                continue
            h[oi] = 0
            h[zi] = 1
        elif r < 0.85 and use_2swap:
            # 2-swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            idx_o = np.random.choice(len(ones), 2, replace=False)
            idx_z = np.random.choice(len(zeros), 2, replace=False)
            i1, i2 = ones[idx_o]
            j1, j2 = zeros[idx_z]
            h[i1] = h[i2] = 0
            h[j1] = h[j2] = 1
        else:

        new_corr = fft_corr(h)

        dE = new_max - cur_max

            corr = new_corr
            if cur_max * 2.0 / N < best_c - 1e-12:

        else:

                h[oi] = 1
                h[zi] = 0
            elif r < 0.85:

                h[j1] = h[j2] = 0

                oi = int(np.random.choice(ones))

            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, T0 * 0.4)
            no_improve = 0
    
    return best_h, best_c

**Idea: Frequency-domain optimization**
The correlation is the inverse FFT of the product of FFTs. So H * G where G = FFT(1-h). We want to minimize max of IFFT(H*G).

**Idea: Greedy pair elimination**
For the current worst lag k*, the "bad pairs" are (i, i+k*) where h[i]=1, h[i+k*]=0. If we could flip h[i+k*] to 1 (and h[i] to 0), we'd reduce c[k*] by 1. But this increases c[k*-k*] = c[0] by 0 (for binary) and affects other lags.

The net effect of swapping h[i]:1→0 and h[j]:0→1 (where j = i+k*):
- c[k*] decreases by 1 (lost one bad pair)
- c[-k*] decreases by 1 (by symmetry of the problem... actually not exactly)
- Other lags change by ±1

So the max could go down by 1 (if k* was the unique max) or stay the same (if another lag was also at the max).

This suggests a greedy approach: always swap to reduce the worst lag, and accept as long as the new max ≤ old max.

**Algorithm: Greedy + SA hybrid**
1. Compute correlation, find max lag
2. Try ALL swaps that reduce the max lag by 1 (or don't increase the global max)
3. Among those, pick the one that most reduces the 2nd/3rd worst lags
4. If no such swap exists, use SA with small temperature

For lag k* > 0: bad pairs are (i, i+k*) where h[i]=1, h[i+k*]=0.
To reduce c[k*], we want to:
- Set h[i+k*] = 1 (and h[i] = 0): reduces c[k*] by 1
- Set h[i] = 0 (and h[i+k*] = 1): same thing

So the swap is: h[i] 1→0, h[i+k*] 0→1. This is a swap at distance k*.

Number of such swaps: O(N) for a given k*. We can try all of them in O(N * N log N) time using FFT, which is about 380 * 3400 = 1.3M ops. That's fast!

def greedy_reduce_max(h, N, corr, max_iters=1000):
    """Try to reduce the max correlation by finding a swap at the worst lag."""
    h = h.copy()
    nh = N // 2
    it = 0
    while it < max_iters and time.time() < ...:
        peak = np.max(corr)
        peak_idx = np.argmax(corr)
        center = N - 1
        lag = peak_idx - center
        
        if lag == 0:
            # Zero lag: need to make more binary
            break
        
        # For positive lag: bad pairs are (i, i+lag) with h[i]=1, h[i+lag]=0
        # For negative lag: bad pairs are (i+|lag|, i) with h[i+|lag|]=1, h[i]=0
        # Which is the same as: h[i]=1, h[i+lag]=0 (with modular wrapping? No, full correlation)
        
        # Actually for full correlation (not circular):
        # corr[N-1+k] = sum_i h[i+k] * (1-h[i]) for i where both are in range
        # For k > 0: sum_{i=0}^{N-k-1} h[i+k](1-h[i])
        # For k < 0: sum_{i=-k}^{N-1} h[i+k](1-h[i]) = sum_{i=0}^{N+ k-1} h[i](1-h[i-k])
        
        # Let me just work with the lag value
        k = lag
        found_improvement = False
        
        if k > 0:
            # bad pairs: h[i+k]=1, h[i]=0 for i in [0, N-k-1]
            # To fix: swap h[i+k]:1→0, h[i]:0→1
            for i in range(N - k):
                if h[i+k] == 1 and h[i] == 0:
                    h[i+k] = 0
                    h[i] = 1
                    new_corr = fft_corr(h)
                    new_peak = np.max(new_corr)
                    if new_peak < peak:
                        corr = new_corr
                        found_improvement = True
                        break
                    else:
                        h[i+k] = 1
                        h[i] = 0
        elif k < 0:
            kk = -k
            # bad pairs: h[i]=1, h[i+kk]=0 for i in [kk, N-1]
            # Wait, for negative lag: corr[N-1-kk] = sum h[i](1-h[i-kk]) for i in [kk, N-1]
            # bad pairs: h[i]=1, h[i-kk]=0 for i in [kk, N-1]
            # To fix: swap h[i]:1→0, h[i-kk]:0→1
            for i in range(kk, N):
                if h[i] == 1 and h[i-kk] == 0:
                    h[i] = 0
                    h[i-kk] = 1
                    new_corr = fft_corr(h)
                    new_peak = np.max(new_corr)
                    if new_peak < peak:
                        corr = new_corr
                        found_improvement = True
                        break
                    else:
                        h[i] = 1
                        h[i-kk] = 0
        
        if not found_improvement:
            break
        it += 1
    
    return h, corr
