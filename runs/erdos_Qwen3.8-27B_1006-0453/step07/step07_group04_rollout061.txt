import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_max_shifts(h, top_k=30):
    N = len(h)
    corr = fft_corr(h)
    top = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        s = int(ai - center)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0 and s < N:
            grad[s:] += (1.0 - h[:N-s])
            grad[:N-s] -= h[s:]
        elif s < 0 and -s < N:
            m = -s
            grad[:N-m] += (1.0 - h[m:])
            grad[m:] -= h[:N-m]
    return grad / top_k, corr

def nesterov_cont(h, N, deadline, k=40, lr0=0.015, beta_init=0.9, verbose=False):
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(np.max(corr))
    m = np.zeros(N)
    lr = lr0
    beta = beta_init
    stall = 0
    patience = 500
    it = 0
    while time.time() < deadline:
        y = h + m
        g, corr = grad_max_shifts(y, top_k=k)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Backtracking-ish: if not improving for a while, reduce lr
        if stall > patience:
            lr *= 0.7
            m *= 0.3
            # Perturb to escape
            idx = np.random.choice(N, size=N//50, replace=False)
            y[idx] += np.random.randn(len(idx)) * 0.1
            stall = 0
            patience = int(patience * 0.8)
        
        h_new = project_sum(y - lr * g, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.97
            beta = min(0.98, beta_init + it * 0.00005)
    return best_h, best_pk * dx

def greedy_swap(h, N, deadline, max_sweeps=3000, sf=1.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        # For continuous, use threshold
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Compute change in correlation for swap
        # Swapping h[p] from 1 to 0 and h[q] from 0 to 1:
        # Delta corr[j] = -h[q]_contrib + ... 
        # Easier: just compute the new corr top value
        n_s = len(ones) if sf >= 1.0 else max(20, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1e-9
        best_pq = None
        # For binary case, the delta is simpler
        # corr_new[j] = corr[j] + delta[j] where delta comes from the two position changes
        # h[p]: 1->0, h[q]: 0->1
        # Effect on corr: 
        #   For shift s: corr_s = sum_i h[i+s](1-h[i])
        #   Removing h[p]=1: removes (1-h[p+s]) from corr_{s} and h[p-s]*(...) 
        #   This is getting complex, let's just recompute for candidate pairs
        # Actually for binary, we can compute delta efficiently
        # h changes at p (1->0) and q (0->1)
        # corr[j] = sum_i h[i]*(1-h[i+j-N+1])
        # Let's use the D matrix approach
        center = N - 1
        # Delta from h[p]: 1->0
        # corr[j] includes h[p]*(1-h[p+s]) where s=j-center, valid if p+s in [0,N-1]
        # and h[p+s]*(1-h[p]) where p+s in [0,N-1], i.e., s in [-(p), N-1-p] -> j in [center-p, center+N-1-p]
        # So removing h[p]: 
        #   For j where p+s valid (s=j-center, p+s in [0,N-1]): -1*(1-h[p+s])
        #   For j where p-s valid (s=j-center, h[p-s] term): -(1-h[p])=0 since h[p]=1
        #     wait: h[p-s]*(1-h[p]) - this term is h[p-s]*0 = 0, so removing h[p] from (1-h[p]) contributes 
        #     Actually (1-h[p]) goes from 0 to 1, so the term h[p-s]*(1-h[p]) goes from 0 to h[p-s]
        #     That's a +h[p-s] change
        # Hmm, let me just do it numerically for small sets
        
        # Efficient approach: compute correlation delta for one position change
        # h[p]: 1->0 means new_h = h with h[p]=0
        # delta_p[j] = new_corr[j] - corr[j]
        # = (contribution of h[p] to corr[j] with old value 1) subtracted
        # + (change in (1-h[p]) terms)
        
        # Actually, the simplest efficient way for binary:
        # corr[j] for shift s=j-center:
        #   Term from h[p] as first factor: (1-h[p+s]) if 0<=p+s<N
        #   Term from h[p] as second factor in (1-h): h[p-s]*(1-h[p]=0) = 0 ... wait
        #   Actually (1-h[p]) = 0, so h[p] doesn't contribute to the (1-h) term
        #   But (1-h[p]) = 1 after change, so NEW contribution from (1-h) term: h[p-s]*1
        #   And OLD contribution from first factor: 1*(1-h[p+s]) = 1-h[p+s]
        #   So delta_p[j] = -(1-h[p+s]) + h[p-s] for valid indices
        # Similarly for h[q]: 0->1
        #   OLD: h[q]=0, so first factor contributes 0, (1-h[q])=1 so h[q-s]*1 contributes
        #   NEW: h[q]=1, first factor contributes (1-h[q+s]), (1-h[q])=0
        #   delta_q[j] = (1-h[q+s]) - h[q-s]
        
        # For the pair (p,q): delta[j] = delta_p[j] + delta_q[j] (approximately, ignoring interaction)
        # Actually there IS interaction: if p+s = q or q-s = p, the terms interact
        # But for large N this is negligible. Let's ignore it for speed.
        
        # Compute delta for each p efficiently
        # delta_p for shift s: -(1-h[p+s]) + h[p-s] (where valid)
        # This is a vector of length 2N-1
        
        # Let me precompute for all candidate p:
        # Actually this is O(N) per p, and we have O(N) p's, so O(N^2) per sweep
        # That's fine for N=760: 760*760 = 577K per sweep
        
        # Let me just do it directly
        for p in p_list:
            # delta_p: effect of h[p] going 1->0
            dp = np.zeros(2*N-1)
            # First factor: h[p+s] for s in range, p+s in [0,N-1]
            s_min = -p
            s_max = N-1-p
            s_range = np.arange(s_min, s_max+1)
            idx = s_range + center
            dp[idx] -= (1.0 - h[p+s_range])
            # Second factor: (1-h[p]) goes 0->1, so h[p-s]*1
            s_min2 = p-N+1
            s_max2 = p
            s_range2 = np.arange(s_min2, s_max2+1)
            idx2 = s_range2 + center
            dp[idx2] += h[p-s_range2]
            
            # Now for each q in zeros, total delta = dp + dq
            # dq for h[q] going 0->1:
            # First factor: h[q+s] goes 0->1, so +(1-h[q+s]) 
            # Second factor: (1-h[q]) goes 1->0, so -h[q-s]
            # dq = +(1-h[q+s]) - h[q-s]
            
            # new_corr[j] = corr[j] + dp[j] + dq[j]
            # We want min over (p,q) of max_j new_corr[j]
            
            # For each p, the best q minimizes max(corr + dp + dq)
            # dq depends on q in a structured way
            # Let's compute for all q:
            base = corr + dp  # shape (2N-1,)
            # For each q: new_max = max(base + dq_q)
            # dq_q[j] = (1-h[q+s]) for q+s valid - h[q-s] for q-s valid
            
            # This is O(N_zeros * N) per p, which is O(N^2) per p, total O(N^3) - too slow
            # For N=760, that's 440M - too slow
            
            # Alternative: for a given p, compute base = corr + dp
            # Then for each q, newcorr = base + dq_q
            # newcorr max = max over j of (base[j] + dq_q[j])
            # = max over j of (base[j]) + max over j where dq_q[j]!=0 of (dq_q[j]) if they don't overlap with max of base
            # This is complex. Let's just limit: try top-50 q's for each p
            
            n_q_try = min(50, len(zeros))
            q_try = np.random.choice(zeros, n_q_try, replace=False) if n_q_try < len(zeros) else zeros
            for q in q_try:
                dq = np.zeros(2*N-1)
                s_min = -q
                s_max = N-1-q
                sr = np.arange(s_min, s_max+1)
                dq[sr + center] += (1.0 - h[q+sr])
                s_min2 = q-N+1
                s_max2 = q
                sr2 = np.arange(s_min2, s_max2+1)
                dq[sr2 + center] -= h[q-sr2]
                
                new_corr = base + dq
                new_P = float(new_corr.max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = (int(p), int(q))
        
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def sa_binary(h, N, deadline, max_sweeps=5000, T0=3.0, Tcool=0.995, T_min=0.03, sf=1.0, kick_every=100, kick_size=4):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if stall >= kick_every:
            for _ in range(kick_size):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            stall = 0
            T = T0 * 0.5
            continue
        # Greedy best move
        n_s = len(ones) if sf >= 1.0 else max(30, int(len(ones) * sf))
        p_list = ones if sf >= 1.0 else np.random.choice(ones, n_s, replace=False)
        best_new_P = P + 1.0
        best_pq = None
        center = N - 1
        for p in p_list:
            dp = np.zeros(2*N-1)
            s_min = -p; s_max = N-1-p
            sr = np.arange(s_min, s_max+1)
            dp[sr+center] -= (1.0 - h[p+sr])
            s_min2 = p-N+1; s_max2 = p
            sr2 = np.arange(s_min2, s_max2+1)
            dp[sr2+center] += h[p-sr2]
            base = c + dp
            n_q_try = min(30, len(zeros))
            q_try = np.random.choice(zeros, n_q_try, replace=False) if n_q_try < len(zeros) else zeros
            for q in q_try:
                dq = np.zeros(2*N-1)
                sr = np.arange(-q, N-q)
                dq[sr+center] += (1.0 - h[q+sr])
                sr2 = np.arange(q-N+1, q+1)
                dq[sr2+center] -= h[q-sr2]
                new_P = float((base+dq).max())
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = (int(p), int(q))
        if best_pq is None:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            continue
        delta = best_new_P - P
        if delta < -1e-14 or (T > T_min and np.random.random() < np.exp(-max(0.0, delta)/max(T,1e-10))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
        T = max(T_min, T * Tcool)
    return best_h, best_c * (2.0/N)

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(42 + N)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
