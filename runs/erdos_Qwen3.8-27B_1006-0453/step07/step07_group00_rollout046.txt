Actually, for the max function, the subgradient is the indicator of the argmax (if unique) or the convex hull of indicators (if multiple). The top-k approach is a smooth approximation. Let me try a different smooth approximation: the log-sum-exp.

log_max_k corr_k ≈ (1/β) * log(sum_k exp(β * corr_k))

The gradient of this is:
d/dh[j] = (1/β) * sum_k [exp(β*corr_k) * dcorr_k/dh[j] / sum_exp]
         = sum_k [softmax_k * dcorr_k/dh[j]]

This is a smooth surrogate that converges to the max as β→∞.

For the dcorr_k/dh[j] computation:
For lag with offset L = k - (N-1):
If L >= 0: corr_k = sum_{i=0}^{N-1-L} h[i]*(1-h[i+L])
d/dh[j]: 
  - j appears as h[i] when i=j (if j <= N-1-L): contributes (1-h[j+L])
  - j appears as h[i+L] when i+L=j, i=j-L (if j-L >= 0 and j-L <= N-1-L): contributes -h[j-L]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2 * N - 1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h):
    N = len(h)
    h = np.asarray(h, dtype=float)
    N2 = N // 2
    idx = np.argsort(-h)
    out = np.zeros(N)
    out[idx[:N2]] = 1.0
    return out

def compute_deltas(h, corr):
    N = len(h)
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        h[i] = 1.0 - h[i]
        deltas[i, :] = fft_corr(h) - corr
        h[i] = 1.0 - h[i]
    return deltas

def full_grad(h, beta=8.0):
    N = len(h)
    corr = fft_corr(h)
    # soft-max weights
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        if w[k] < 1e-6:
            continue
        L = k - (N - 1)
        if L == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w[k] * (1.0 - h[:N - L])
            grad[:N - L] -= w[k] * h[L:]
        else:
            M = -L
            grad[:N - M] += w[k] * (1.0 - h[M:])
            grad[M:] -= w[k] * h[:N - M]
    return grad, corr

def sa_search(h, deadline, n_peaks=8, n_per_peak=20, t_start=0.045,
              cool_rate=0.9994, max_iter=150000, reheat_every=15000):
    N = len(h)
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    T = t_start
    it = 0
    stall = 0
    deltas = compute_deltas(h, corr)
    
    while time.time() < deadline and it < max_iter:
        it += 1
        if it % 400 == 0:
            T = max(T * cool_rate, 0.0008)
        if it % reheat_every == 0 and stall > 3000:
            T = max(T * 2.5, 0.02)
            stall = 0
        
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p, cand_q = [], []
        
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                if mask.any():
                    bp, bq = vi[mask], (vi + k_off)[mask]
                    ns = min(n_per_peak, len(bp))
                    sel = np.random.choice(len(bp), ns, replace=False)
                    cand_p.extend(bp[sel].tolist())
                    cand_q.extend(bq[sel].tolist())
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                if mask.any():
                    bp, bq = vi[mask], (vi - M)[mask]
                    ns = min(n_per_peak, len(bp))
                    sel = np.random.choice(len(bp), ns, replace=False)
                    cand_p.extend(bp[sel].tolist())
                    cand_q.extend(bq[sel].tolist())
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nr = min(10, len(ones), len(zeros))
        if nr > 0:
            cand_p.extend(np.random.choice(ones, nr, replace=False).tolist())
            cand_q.extend(np.random.choice(zeros, nr, replace=False).tolist())
        
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        if len(cand_p) == 0 or len(cand_q) == 0:
            break
        
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        chunk = 30
        best_new, bi, bj = np.inf, 0, 0
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[None, None, :] + dp[ci:ce, None, :] + dq[None, :, :]
            nm = vals.max(axis=2)
            fi = int(nm.argmin())
            mi, mj = fi // len(cand_q), fi % len(cand_q)
            if nm[mi, mj] < best_new:
                best_new, bi, bj = float(nm[mi, mj]), mi + ci, mj
        
        p, q = int(cand_p[bi]), int(cand_q[bj])
        cur_c = float(corr.max())
        dc = best_new - cur_c
        accept = dc < -1e-12 or (T > 0.0005 and np.random.random() < np.exp(-dc / (T + 1e-9)))
        
        if accept:
            h[p], h[q] = 0.0, 1.0
            corr = corr + deltas[p] + deltas[q]
            h[p] = 1.0
            deltas[p, :] = fft_corr(h) - corr - deltas[q]
            h[p] = 0.0
            h[q] = 0.0
            deltas[q, :] = fft_corr(h) - corr
            h[q] = 1.0
            c_cur = float(corr.max())
            stall = 0 if c_cur < best_c - 1e-12 else stall + 1
            if c_cur < best_c - 1e-12:
                best_c, best_h = c_cur, h.copy()
        else:
            stall += 1
        
        if stall > 12000:
            break
    return best_h, best_c * (2.0 / N)

def nesterov_poly(h_in, N, deadline, lr=0.015, beta=10.0):
    h = np.clip(h_in.astype(float), 0, 1)
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    bet_m = 0.80
    it = 0
    while time.time() < deadline:
        y = h + m
        y = np.clip(y, 0, 1)
        g, corr_y = full_grad(y, beta)
        pk = float(corr_y.max())
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        h_new = np.clip(y - lr * g, 0, 1)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0, 1)
        m = bet_m * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.988
            bet_m = min(0.92, 0.78 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
