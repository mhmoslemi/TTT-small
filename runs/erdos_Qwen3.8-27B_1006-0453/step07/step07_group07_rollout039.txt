def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0].copy()
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0].copy()
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def compute_deltas_one(h, oi, top_lags):
    """Precompute delta for flipping h[oi] from 1 to 0 at each top lag."""
    N = len(h)
    d = np.zeros(len(top_lags))
    for li, L in enumerate(top_lags):
        aL = abs(L)
        if aL == 0:
            d[li] = 0.0
        elif aL < N:
            v = 0.0
            if L > 0:
                if L <= oi < N:
                    v -= (1.0 - h[oi - L])
                if 0 <= oi <= N - 1 - L:
                    v += h[oi + L]
            else:
                if oi >= aL:
                    v -= (1.0 - h[oi - aL])
                if 0 <= oi + aL < N:
                    v += h[oi + aL]
            d[li] = v
    return d

def compute_deltas_zero(h, zi, top_lags):
    """Precompute delta for flipping h[zi] from 0 to 1 at each top lag."""
    N = len(h)
    d = np.zeros(len(top_lags))
    for li, L in enumerate(top_lags):
        aL = abs(L)
        if aL == 0:
            d[li] = 0.0
        elif aL < N:
            v = 0.0
            if L > 0:
                if L <= zi < N:
                    v += (1.0 - h[zi - L])
                if 0 <= zi <= N - 1 - L:
                    v -= h[zi + L]
            else:
                if zi >= aL:
                    v += (1.0 - h[zi - aL])
                if 0 <= zi + aL < N:
                    v -= h[zi + aL]
            d[li] = v
    return d

def binary_il(h, deadline, max_iters=5000, n_crit=25, s_sz=45):
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    it = 0
    no_imp = 0
    center = N - 1
    while time.time() < deadline and it < max_iters:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 400:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        # Precompute all deltas
        d_o = np.array([compute_deltas_one(h, oi, top_lags) for oi in os_])
        d_z = np.array([compute_deltas_zero(h, zi, top_lags) for zi in zs])
        # corr values at top lags
        corr_top = corr[center + top_lags]
        cur_peak = float(np.max(corr))
        best_move = None
        best_new = c
        # Vectorized: new_values[si, zi, li] = corr_top[li] + d_o[si,li] + d_z[zi,li]
        # new_max[si,zi] = max over li
        new_values = corr_top[None, None, :] + d_o[:, None, :] + d_z[None, :, :]
        new_max = new_values.max(axis=2)
        # Find best move where new_max < cur_peak
        mask = new_max < cur_peak - 1e-10
        if mask.any():
            # Among improving moves, find the one with smallest new_max
            flat_idx = np.argmin(np.where(mask, new_max, np.inf))
            si, zi = divmod(flat_idx, nz)
            best_move = (os_[si], zs[zi])
            best_new = new_max[si, zi] * (2.0 / N)
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
        elif no_imp > 100 and np.random.random() < 0.25:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        it += 1
    return best_h, best_c
