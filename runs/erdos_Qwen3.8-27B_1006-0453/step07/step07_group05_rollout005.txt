        corr = fft_corr(h)
        peak = np.max(corr)
        # Softmax-like weighting
        w = np.exp((corr - peak) / soft_temp)
        w /= w.sum()

        center = N - 1

                grad[L:] += wt * (1.0 - h[:N-L])
                grad[:N-L] -= wt * h[L:]

                    grad[:N-M] += wt * (1.0 - h[M:])
                    grad[M:] -= wt * h[:N-M]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

                h[o1], h[o2], h[z1], h[z2] = 1, 1, 0, 0
        elif r < 0.78:
            # Targeted: fix peak lag conflict
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            pk = int(np.argmax(corr))
            pk_lag = pk - center
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            if abs(oi - zi) == abs(pk_lag) or (oi + pk_lag >= 0 and oi + pk_lag < N and h[oi + pk_lag] == 0):
                zi = (oi + abs(pk_lag)) % N
                if h[zi] == 1:
                    zi = int(np.random.choice(zeros))
            h[oi], h[zi] = 0, 1
            nc = fft_corr(h)
            nm = float(np.max(nc))
            dE = nm - cur_max
            acc = dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T))
            if acc:
                cur_max, corr = nm, nc
                n_accept += 1
            else:
                h[oi], h[zi] = 1, 0
        else:
            # Block swap
            bl = int(np.random.choice([3, 5, 7]))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < bl or len(zeros) < bl:
                continue
            oi = np.random.choice(ones, bl, replace=False)
            zi = np.random.choice(zeros, bl, replace=False)
            saved_o, saved_z = h[oi].copy(), h[zi].copy()
            h[oi], h[zi] = 0, 1
            nc = fft_corr(h)
            nm = float(np.max(nc))
            dE = nm - cur_max
            acc = dE < -1e-14 or (T > 1e-9 and np.random.random() < np.exp(-dE / T))
            if acc:
                cur_max, corr = nm, nc
                n_accept += 1
            else:
                h[oi], h[zi] = saved_o, saved_z

        c = cur_max * 2.0 / N
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if n_total % 4000 == 0:
            rate = n_accept / max(1, n_total)
            n_accept, n_total = 0, 0
            if rate < 0.05:
                T = min(T * 2.0, T0)
            elif rate > 0.4:
                T *= 0.6

        T = max(T * cool_rate, T1)

        if no_improve > 4000:
            for _ in range(10):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) < 1 or len(zeros) < 1:
                    break
                a, b = int(np.random.choice(ones)), int(np.random.choice(zeros))
                h[a], h[b] = h[b], h[a]
            corr = fft_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, 0.015)
            no_improve = 0
    return best_h, best_c

def continuous_refine(h, N, deadline, lr0=0.0025):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    lr = lr0
    bt1, bt2 = 0.9, 0.999
    m = np.zeros(N)
    v = np.zeros(N)
    t = 1
    soft_temp = 0.025
    while time.time() < deadline - 3:
        corr = fft_corr(h)
        peak = np.max(corr)
        w = np.exp((corr - peak) / soft_temp)
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        sig = np.where(w > 0.001)[0]
        for ki in sig:
            L = int(ki - center)
            wt = w[ki]
            if L == 0:
                grad += wt * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wt * (1.0 - h[:N - L])
                grad[:N - L] -= wt * h[L:]
            else:
                M = -L
                grad[:N - M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N - M]
        m = bt1 * m + (1 - bt1) * grad
        v = bt2 * v + (1 - bt2) * grad * grad
        mh = m / (1 - bt1 ** t)
        vh = v / (1 - bt2 ** t)
        h_new = np.clip(h - lr * mh / (np.sqrt(vh) + 1e-8), 0, 1)
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        c_new = compute_c5(h_new, N)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        t += 1
        if t % 400 == 0:
            lr *= 0.98
            soft_temp = max(soft_temp * 0.94, 0.008)
    return best_h, best_c

def structured_init(N, seed_val):
    np.random.seed(seed_val)
    nh = N // 2
    x = np.arange(N)
    v = (np.sin(2 * np.pi * 3 * x / N) + 0.6 * np.sin(2 * np.pi * 7 * x / N)
         + 0.4 * np.sin(2 * np.pi * 13 * x / N) + 0.3 * np.cos(2 * np.pi * 5 * x / N))
    h = (v > np.median(v)).astype(float)
    h = fix_sum(h, nh)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    nh0 = N0 // 2
    h_b = project_binary(h0, nh0)
    best_h = h_b.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Smart annealing on initial binary (35%)
    d1 = t0 + budget_s * 0.35
    h1, c1 = smart_anneal(h_b, N0, d1, max_iters=100000, T0=0.05, T1=0.001)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Continuous Adam refinement (25%)
    if time.time() < deadline - budget_s * 0.22:
        hc = best_h.astype(float) + np.random.randn(N0) * 0.02
        hc = np.clip(hc, 0, 1)
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(hc - mid, 0, 1).sum() > nh0:
                lo = mid
            else:
                hi = mid
        hc = np.clip(hc - (lo + hi) / 2, 0, 1)
        d2 = min(deadline - budget_s * 0.18, time.time() + budget_s * 0.22)
        hc, c2 = continuous_refine(hc, N0, d2, lr0=0.002)
        if c2 < best_c5:
            best_c5, best_h = c2, hc

    # Phase 3: Binary re-projection + deeper anneal (25%)
    if time.time() < deadline - budget_s * 0.15:
        h3 = project_binary(best_h, nh0)
        d3 = min(deadline - budget_s * 0.10, time.time() + budget_s * 0.22)
        h3, c3 = smart_anneal(h3, N0, d3, max_iters=80000, T0=0.025, T1=0.0001)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Multi-scale at different N (10%)
    if time.time() < deadline - budget_s * 0.06:
        for N_try in [200, 300, 460]:
            if time.time() > deadline - budget_s * 0.04:
                break
            if N_try == N0:
                continue
            hv = structured_init(N_try, seed + N_try)
            sub_dl = min(deadline - budget_s * 0.02, time.time() + budget_s * 0.05)
            hr, cr = smart_anneal(hv, N_try, sub_dl, max_iters=20000, T0=0.08, T1=0.001)
            if cr < best_c5:
                best_c5, best_h = cr, hr

    # Phase 5: Final polish
    if time.time() < deadline - 4:
        hF = project_binary(best_h, len(best_h) // 2)
        hF, cF = smart_anneal(hF, len(hF), deadline - 2, max_iters=60000, T0=0.012, T1=0.00001)
        cF = compute_c5(hF, len(hF))
        if cF < best_c5:
            best_c5, best_h = cF, hF

    hB = project_binary(best_h, len(best_h) // 2)
    cB = compute_c5(hB, len(hB))
    if cB < best_c5:
        best_c5, best_h = cB, hB

    return best_h, best_c5, len(best_h)
