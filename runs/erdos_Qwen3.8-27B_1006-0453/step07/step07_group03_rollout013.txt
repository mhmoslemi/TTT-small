def softmax_corr_grad(h, N, beta):
    corr = fft_corr(h)
    max_c = corr.max()
    shifted = corr - max_c
    exps = np.exp(beta * shifted)
    w = exps / exps.sum()
    val = float(max_c + np.log(exps.sum()) / beta)
    
    # grad[j] = sum_s w[s+N-1] * (1-h[j-s])*[0<=j-s<N] - sum_s w[s+N-1] * h[j+s]*[0<=j+s<N]
    # Vectorize using matrix operations
    # For each j, s ranges over [max(-(N-1), j-N+1), min(N-1, j)] for first term
    # and [max(-(N-1), -j), min(N-1, N-1-j)] for second term
    
    # Create indexing arrays
    j_idx = np.arange(N)[:, None]  # (N, 1)
    s_idx = np.arange(-(N-1), N)   # (2N-1,)
    
    # First term: valid where 0 <= j-s < N
    valid1 = (j_idx - s_idx[None, :] >= 0) & (j_idx - s_idx[None, :] < N)  # (N, 2N-1)
    h_idx1 = np.clip(j_idx - s_idx[None, :], 0, N-1)  # (N, 2N-1)
    w_idx_s = (s_idx + N - 1)  # (2N-1,): index into w
    
    term1 = valid1 * w[w_idx_s][None, :] * (1.0 - h[h_idx1])
    A = term1.sum(axis=1)  # (N,)
    
    # Second term: valid where 0 <= j+s < N
    valid2 = (j_idx + s_idx[None, :] >= 0) & (j_idx + s_idx[None, :] < N)  # (N, 2N-1)
    h_idx2 = np.clip(j_idx + s_idx[None, :], 0, N-1)  # (N, 2N-1)
    
    term2 = valid2 * w[w_idx_s][None, :] * h[h_idx2]
    B = term2.sum(axis=1)  # (N,)
    
    grad = A - B
    return val, grad
