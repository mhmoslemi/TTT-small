import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_smooth(h, N, deadline, T=30.0, max_outer=5):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj(h, T, target, 0.0)[0]
    lam = 5.0
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        T_cur = T * (1.0 + 0.5 * outer)  # increase temperature
        res = sp_min(smooth_obj, h, args=(T_cur, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj(h, T_cur, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.0
    return best_h, best_v * (2.0 / N)

    NH = N // 2

        P = float(np.max(corr))
        improved = False

            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        # Single swap
        for _ in range(ntri):
            i = np.random.randint(len(ones))
            j = np.random.randint(len(zeros))
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd, bi, bj = d, oi, zi
            h[oi], h[zi] = 1.0, 0.0
        # Paired swap (move two 1s to two 0 positions)
        if len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(ntri // 2):
                i1, i2 = np.random.choice(len(ones), 2, replace=False)
                j1, j2 = np.random.choice(len(zeros), 2, replace=False)
                o1, o2 = int(ones[i1]), int(ones[i2])
                z1, z2 = int(zeros[j1]), int(zeros[j2])
                h[o1], h[o2], h[z1], h[z2] = 0.0, 0.0, 1.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > bd + temp:
                    bd, bi, bj = d, -2, -2  # marker for paired
                    # Need to store the actual indices
                    bi = (o1, o2, z1, z2)
                h[o1], h[o2], h[z1], h[z2] = 1.0, 1.0, 0.0, 0.0
        if bi >= 0:

            improved = True
        elif isinstance(bi, tuple):
            o1, o2, z1, z2 = bi
            h[o1], h[o2], h[z1], h[z2] = 0.0, 0.0, 1.0, 1.0
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(np.clip(T * (c - mx), -500, 500))
    w = e / e.sum()
    val = mx + np.log(e.sum() + 1e-30) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-12:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_smooth(h, N, deadline, T=30.0, max_outer=4):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj(h, T, target, 0.0)[0]
    lam = 5.0
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        T_cur = T * (1.0 + 0.3 * outer)
        res = sp_min(smooth_obj, h, args=(T_cur, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 400, 'maxfun': 1200, 'ftol': 1e-16,
                              'gtol': 1e-12, 'maxls': 50})
        h = proj01(res.x, target)
        v = smooth_obj(h, T_cur, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.0
    return best_h, best_v * (2.0 / N)

def bin_search(h, N, deadline, max_iters=400, ntri=22, anneal=0.0):
    h = (h > 0.5).astype(float)
    NH = N // 2
    n1 = int(h.sum())
    if n1 > NH:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - NH]] = 0.0
    elif n1 < NH:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:NH - n1]] = 1.0
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    temp = anneal
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 180:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        for _ in range(ntri):
            i = np.random.randint(len(ones))
            j = np.random.randint(len(zeros))
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:
                bd, bi, bj = d, oi, zi
            h[oi], h[zi] = 1.0, 0.0
        # Paired swap
        if len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(max(2, ntri // 3)):
                i1, i2 = np.random.choice(len(ones), 2, replace=False)
                j1, j2 = np.random.choice(len(zeros), 2, replace=False)
                o1, o2 = int(ones[i1]), int(ones[i2])
                z1, z2 = int(zeros[j1]), int(zeros[j2])
                h[o1], h[o2], h[z1], h[z2] = 0.0, 0.0, 1.0, 1.0
                nP = float(np.max(fft_corr(h)))
                d = P - nP
                if d > bd + temp:
                    bd = d
                    bi, bj = -1, -1
                    h[o1], h[o2], h[z1], h[z2] = 1.0, 1.0, 0.0, 0.0
                    h[o1], h[o2], h[z1], h[z2] = 0.0, 0.0, 1.0, 1.0
                    improved = True
                    bd = -np.inf
                    break
                h[o1], h[o2], h[z1], h[z2] = 1.0, 1.0, 0.0, 0.0
        if not improved and bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))
    return best_h, best_c

def subgrad_ne(h, N, deadline, alpha0=0.03, beta=0.9, atol=1e-5):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < 6000:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            alpha *= 0.95
            beta = min(0.98, 0.88 + 0.0003 * it)
    return best_h, best_pk * (2.0 / N)
