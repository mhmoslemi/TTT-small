```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(s, N):
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i+N] = s
    return deltas

def exhaustive_local_search(h, N, deadline):
    """Try all single swaps to find best improvement"""
    h = h.copy()
    s = 2.0 * h - 1.0
    deltas = build_deltas(s, N)
    corr = fft_corr(h)
    best_c = float(corr.max())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for p in ones:
            # Vectorized: for all q in zeros, compute max(corr + deltas[p] - deltas[q])
            vals = corr[None, :] + deltas[p][None, :] - deltas[zeros][:, None]
            new_maxes = vals.max(axis=1)
            bi = np.argmin(new_maxes)
            if new_maxes[bi] < best_c - 1e-12:
                q = zeros[bi]
                h[p], h[q] = 0.0, 1.0
                corr += deltas[p] - deltas[q]
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0; deltas[p, p:p+N] = s
                deltas[q, :] = 0; deltas[q, q:q+N] = s
                best_c = float(corr.max())
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
                break  # restart from beginning
    return h, best_c

    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s = 2.0 * h - 1.0
    deltas = build_deltas(s, N)
    T = t_start
    it = 0
    no_impr = 0
    stall = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.001)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)

            else:

        cand_q.extend(np.random.choice(zeros, nr, replace=False))
        
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)
        
        use_2swap = np.random.random() < p_2swap and len(cand_p) >= 2 and len(cand_q) >= 2

            j1 = np.random.randint(len(cand_q))

            if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):

                h[q1], h[q2] = 1.0, 1.0

                s[p1], s[p2] = -s[p1], -s[p2]
                s[q1], s[q2] = -s[q1], -s[q2]
                for idx in [p1, p2, q1, q2]:
                    deltas[idx, :] = 0
                    deltas[idx, idx:idx+N] = s
                c = float(corr.max())
                if c < best_c - 1e-12:
                    best_c, best_h = c, h.copy()
                    no_impr, stall = 0, 0
                else:
                    no_impr += 1; stall += 1
            else:
                no_impr += 1; stall += 1
        else:
            if len(cand_p) == 0 or len(cand_q) == 0:
                break
            dp = deltas[cand_p]
            dq = deltas[cand_q]
            chunk = 8
            best_new, bi, bj = 1e10, 0, 0
            for ci in range(0, len(cand_p), chunk):
                ce = min(ci + chunk, len(cand_p))
                vals = corr[None, None, :] + dp[ci:ce, None, :] - dq[None, :, :]
                nm = vals.max(axis=2)
                mi = np.unravel_index(int(nm.argmin()), nm.shape)
                if nm[mi] < best_new:
                    best_new, bi, bj = float(nm[mi]), mi[0]+ci, mi[1]
            
            p, q = int(cand_p[bi]), int(cand_q[bj])
            dc = best_new - float(corr.max())
            if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):
                h[p], h[q] = 0.0, 1.0
                corr += deltas[p] - deltas[q]
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0; deltas[p, p:p+N] = s
                deltas[q, :] = 0; deltas[q, q:q+N] = s
                c = float(corr.max())
                if c < best_c - 1e-12:
                    best_c, best_h = c, h.copy()
                    no_impr, stall = 0, 0
                else:
                    no_impr += 1; stall += 1
            else:
                no_impr += 1; stall += 1
        
        if stall > 3000:
            T = max(t_start * 0.5, T * 2.0)
            stall = 0
        if no_impr > 10000:
            break
    return best_h, best_c
```

```python
            # = max_k(corr[k] + deltas[p,k] - D[q,k]) where D = deltas[zeros]
            # We want to minimize this over q
            D = deltas[zeros]  # shape (n_zeros, 2N-1)
            base = corr + deltas[p]  # shape (2N-1,)
            # new_max for each q: max_k(base[k] - D[q,k])
            new_maxes = (base[None,:] - D).max(axis=1)  # shape (n_zeros,)
            bi = np.argmin(new_maxes)
            if new_maxes[bi] < best_c - 1e-12:
                q = zeros[bi]
                h[p], h[q] = 0.0, 1.0
                corr = base - deltas[q]  # Wait, this isn't right
                # Actually: new_corr = corr + deltas[p] - deltas[q]
                # And we know base = corr + deltas[p]
                # So new_corr = base - deltas[q]
                corr = base - deltas[q]
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0; deltas[p, p:p+N] = s
                deltas[q, :] = 0; deltas[q, q:q+N] = s
                best_c = float(corr.max())
                improved = True
                break  # restart
    return h, best_c
```

The key vectorization: `(base[None,:] - D).max(axis=1)` computes for all zeros q simultaneously. Shape is (n_zeros, 2N-1) = (190, 759) ≈ 144K elements per p. Memory is fine.

Total time for one full pass: 190 iterations × 144K ops ≈ 27M ops. In numpy this is very fast.

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    target = N // 2
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h < 0.5)[0]
        h[np.random.choice(idx, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(s, N):
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i+N] = s
    return deltas

def exhaustive_local_search(h, N, deadline):
    h = h.copy()
    s = 2.0 * h - 1.0
    deltas = build_deltas(s, N)
    corr = fft_corr(h)
    best_c = float(corr.max())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_dc = 1e10
        best_pi, best_qi = -1, -1
        chunk = 10
        for ci in range(0, len(ones), chunk):
            ce = min(ci + chunk, len(ones))
            dp_c = deltas[ones[ci:ce]]
            vals = corr[None, None, :] + dp_c[:, None, :] - deltas[zeros][None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_dc:
                best_dc = float(nm[mi])
                best_pi, best_qi = ci + mi[0], mi[1]
        if best_dc < best_c - 1e-12 and best_pi >= 0:
            p, q = int(ones[best_pi]), int(zeros[best_qi])
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p] - deltas[q]
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0; deltas[p, p:p+N] = s
            deltas[q, :] = 0; deltas[q, q:q+N] = s
            best_c = float(corr.max())
            improved = True
    return h, best_c

def targeted_sa(h_bin, deadline, n_peaks=6, n_per_peak=20, cool_rate=0.9993, t_start=0.04, max_iters=120000, p_2swap=0.25):
    N = len(h_bin)
    h = h_bin.copy()
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    s = 2.0 * h - 1.0
    deltas = build_deltas(s, N)
    T = t_start
    it = 0
    no_impr = 0
    stall = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % 500 == 0:
            T = max(T * cool_rate, 0.001)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]
        cand_p, cand_q = [], []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp, bq = vi[mask], (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)

            if len(bp) > 0:

        cand_q.extend(np.random.choice(zeros, nr, replace=False))
        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)

                h[p1], h[p2] = 0.0, 0.0

                s[p1], s[p2] = -s[p1], -s[p2]
                s[q1], s[q2] = -s[q1], -s[q2]
                for idx in (p1, p2, q1, q2):
                    deltas[idx, :] = 0; deltas[idx, idx:idx+N] = s
                c = float(corr.max())
                if c < best_c - 1e-12:
                    best_c, best_h = c, h.copy(); no_impr = stall = 0
                else:
                    no_impr += 1; stall += 1
            else:
                no_impr += 1; stall += 1
        else:
            if len(cand_p) == 0 or len(cand_q) == 0:
                break
            dp, dq = deltas[cand_p], deltas[cand_q]
            best_new, bi, bj = 1e10, 0, 0
            for ci in range(0, len(cand_p), 8):
                ce = min(ci + 8, len(cand_p))
                vals = corr[None, None, :] + dp[ci:ce, None, :] - dq[None, :, :]
                nm = vals.max(axis=2)

            p, q = int(cand_p[bi]), int(cand_q[bj])
            dc = best_new - float(corr.max())
            if dc < -1e-12 or (T > 0.001 and np.random.random() < np.exp(-dc / (T + 1e-9))):
                h[p], h[q] = 0.0, 1.0
                corr += deltas[p] - deltas[q]
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0; deltas[p, p:p+N] = s
                deltas[q, :] = 0; deltas[q, q:q+N] = s
                c = float(corr.max())
                if c < best_c - 1e-12:
                    best_c, best_h = c, h.copy(); no_impr = stall = 0
                else:
                    no_impr += 1; stall += 1
            else:
                no_impr += 1; stall += 1
        if stall > 3000:
            T = max(t_start * 0.6, T * 1.8); stall = 0
        if no_impr > 10000:
            break
    return best_h, best_c
```
