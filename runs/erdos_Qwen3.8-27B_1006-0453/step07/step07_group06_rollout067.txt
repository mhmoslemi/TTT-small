        g[:N - L] -= h[L:N]

    val += 0.5 * lam * s * s

    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def raw_max(h):
    return float(np.max(fft_corr(h)))

def compute_c5(h, N):
    return raw_max(h) * (2.0 / N)

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def smooth_f(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(min(T, 50) * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    thresh = 1e-10 / len(c)
    for i in range(len(c)):
        if w[i] > thresh:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g = g + lam * s
    return val, g

def lbfgs_phase(h, N, deadline, T=30.0, max_outer=5, lam0=5.0):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    lam = lam0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_f, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 200, 'maxfun': 600, 'ftol': 1e-15})
        h = proj01(res.x, target)
        c5 = compute_c5(h, N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lam *= 3.0
    return best_h, best_c5

def bin_local(h, N, deadline, n_trials=40, max_rounds=5000):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        w1 = np.where(h == 1)[0]
        np.random.shuffle(w1)
        h[w1[:n1 - nh]] = 0
    elif n1 < nh:
        w0 = np.where(h == 0)[0]
        np.random.shuffle(w0)
        h[w0[:nh - n1]] = 1
    best_h = h.copy()
    best = raw_max(h)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        corr = fft_corr(h)
        mx = float(np.max(corr))
        peak_k = int(np.argmax(corr)) - (N - 1)
        if mx < best - 1e-14:
            best = mx
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        ntry = n_trials if stall < 50 else 20
        for _ in range(ntry):
            # Targeted: prefer 1s near the peak lag structure
            i_idx = np.random.randint(len(ones))
            j_idx = np.random.randint(len(zeros))
            i, j = int(ones[i_idx]), int(zeros[j_idx])
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            nm = float(np.max(fft_corr(h)))
            d = mx - nm
            if d > bd:
                bd, bi, bj = d, i, j
            h[i], h[j] = 1.0, 0.0
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
        else:
            stall += 1
    return best_h, best

def iters_local_search(h, N, deadline, n_cycles=5):
    """Iterated local search: optimize, perturb, re-optimize."""
    best_h = h.copy()
    best = compute_c5(h, N)
    per_cycle = max(10, int((deadline - time.time()) / max(n_cycles, 1)))
    for cyc in range(n_cycles):
        if time.time() > deadline - 5:
            break
        t_end = min(deadline - 3, time.time() + per_cycle * 0.6)
        h1, c1 = lbfgs_phase(best_h, N, t_end, T=25.0, max_outer=3, lam0=5.0)
        if c1 < best:
            best, best_h = c1, h1.copy()
        # Perturb
        t_end2 = min(deadline - 2, time.time() + per_cycle * 0.4)
        npk = 5 + 3 * cyc  # increasing perturbation
        h2 = best_h.copy()
        w1 = np.where(h2 > 0.5)[0]
        w0 = np.where(h2 < 0.5)[0]
        for _ in range(npk):
            if len(w1) < 1 or len(w0) < 1:
                break
            a, b = int(w1[np.random.randint(len(w1))]), int(w0[np.random.randint(len(w0))])
            h2[a], h2[b] = 0.0, 1.0
            w1 = np.where(h2 > 0.5)[0]
            w0 = np.where(h2 < 0.5)[0]
        h3, c3 = bin_local(h2, N, t_end2, n_trials=35, max_rounds=2000)
        c5_3 = c3 * (2.0 / N)
        if c5_3 < best:
            best, best_h = c5_3, h3.copy()
    return best_h, best

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def rawmax(h):
    return float(np.max(fft_corr(h)))

def c5val(h):
    return rawmax(h) * 2.0 / len(h)

def proj01(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - tgt) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def lgrad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        return 1.0 - 2.0 * h
    if L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def sm_f(h, T, tgt, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    Tc = min(T, 60.0)
    e = np.exp(Tc * (c - mx))
    es = e.sum()
    w = e / es
    val = mx + np.log(es) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-11:
            g += w[i] * lgrad(i - ctr, h, N)
    s = h.sum() - tgt
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs(h, N, dl, T=30, mout=4, lam0=5):
    h = h.astype(float).copy()
    tgt = float(N) // 2
    bh, bc = h.copy(), c5val(h)
    lam = lam0
    for _ in range(mout):
        if time.time() > dl:
            break
        r = sp_min(sm_f, h, args=(T, tgt, lam), jac=True,
                   method='L-BFGS-B', bounds=[(0, 1)] * N,
                   options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-15})
        h = proj01(r.x, tgt)
        cv = c5val(h)
        if cv < bc:
            bc, bh = cv, h.copy()
        lam *= 3
    return bh, bc

    bh, bm = h.copy(), rawmax(h)
    st = 0

            if Nt % 2 or Nt < 100 or time.time() > d: continue
            xn = np.linspace(0, 2, Nt, endpoint=False)
            hi = np.interp(xn, x0, bh)
            ht = (hi > 0.5).astype(float)
            nh = Nt // 2
            n1 = int(ht.sum())
            if n1 > nh:
                w = np.where(ht == 1)[0]; np.random.shuffle(w); ht[w[:n1-nh]] = 0
            elif n1 < nh:
                w = np.where(ht == 0)[0]; np.random.shuffle(w); ht[w[:nh-n1]] = 1
            hb, rm = binsearch(ht, Nt, min(d, time.time() + 20), nt=30, mr=2000)
            hr = proj01(hb + np.random.randn(Nt) * 0.05, Nt / 2.0)
            hl, cl = lbfgs(hr, Nt, min(d, time.time() + 15), T=30, mout=2, lam0=5)
            cv = cl
            if cv < bc:
                bc, bh = cv, hl
                x0 = np.linspace(0, 2, Nt, endpoint=False)
    
    # Final polish
    Nf = len(bh)
    if time.time() < dl - 10:
        d = min(dl - 5, time.time() + 30)
        h_fin = bh + np.random.randn(Nf) * 0.02
        h_fin = proj01(h_fin, Nf / 2.0)
