import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_binary(h, N, seed=42):
    h = h.astype(float).copy()
    nh = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(seed + N)
    if n1 > nh:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - nh, replace=False)
        h_bin[idx] = 0.0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], nh - n1, replace=False)
        h_bin[idx] = 1.0
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def add_interactions(cand, zeros, p, c_len, center):
    Lp = zeros - p
    idx = center + Lp
    valid = (idx >= 0) & (idx < c_len) & (Lp != 0)
    if valid.any():
        rows = np.where(valid)[0]
        cand[rows, idx[valid]] += np.sign(Lp[valid])

            stall += 1
            if np.random.random() < 0.15:
                pr = int(np.where(h == 1)[0][np.random.randint(N // 2)])
                qr = int(np.where(h == 0)[0][np.random.randint(N // 2)])
                if pr != qr:
                    h[pr], h[qr] = 0.0, 1.0
                    c = fft_corr(h)
                    P = float(c.max())
        dcount += 1
        if dcount >= drefresh:
            D = compute_D(h, N)
            c = fft_corr(h)
            P = float(c.max())
            dcount = 0
        if stall >= kick_every:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(kick_size):
                pr = int(ones[np.random.randint(len(ones))])
                qr = int(zeros[np.random.randint(len(zeros))])
                h[pr], h[qr] = 0.0, 1.0
            c = fft_corr(h)
            P = float(c.max())
            D = compute_D(h, N)
            stall = 0
            T = T0 * 0.6
        T = max(T_min, T * Tcool)
        step += 1
    return best_h, best_c * (2.0 / N)

def nesterov_cont(h, N, deadline, k=50, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    center = N - 1
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        s = float(h_new.sum())
        if abs(s - target) > 1e-10:
            lo, hi = -1.0, 1.0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                if float(np.clip(h_new - mid, 0.0, 1.0).sum()) > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo + hi) * 0.5, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.985
            beta = min(0.97, 0.88 + it * 0.00005)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    h_bin = ensure_sum_binary(h0, N0)

    # Phase 1: Fixed greedy (corrected D sign + interactions)
    d1 = min(deadline - 850, t0 + 50)
    h1, c5_1 = greedy_fixed(h_bin, N0, d1, max_sweeps=400, sf=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Guided SA at N0 (main search)
    d2 = min(deadline - 600, t0 + 250)
    h2, c5_2 = sa_guided(ensure_sum_binary(best_h, N0), N0, d2, T0=3.5, Tcool=0.9985, T_min=0.02, drefresh=25, kick_every=100, kick_size=5)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Second greedy pass
    d3 = min(deadline - 400, t0 + 380)
    h3, c5_3 = greedy_fixed(ensure_sum_binary(best_h, N0), N0, d3, max_sweeps=350, sf=0.8)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: SA with perturbation (escape local optima)
    h_pert = ensure_sum_binary(best_h, N0)
    for _ in range(12):
        p_ = np.where(h_pert == 1)[0]
        q_ = np.where(h_pert == 0)[0]
        if len(p_) > 0 and len(q_) > 0:
            h_pert[p_[np.random.randint(len(p_))]] = 0.0
            h_pert[q_[np.random.randint(len(q_))]] = 1.0
    d4 = min(deadline - 250, t0 + 620)
    h4, c5_4 = sa_guided(h_pert, N0, d4, T0=5.0, Tcool=0.997, T_min=0.015, drefresh=20, kick_every=80, kick_size=7)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Nesterov continuous to find new basin
    if time.time() < deadline - 200:
        hc = ensure_sum_binary(best_h, N0).astype(float)
        hc = np.clip(hc + np.random.randn(N0) * 0.08, 0.0, 1.0)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0.0, 1.0)
        d5 = min(deadline - 150, t0 + 800)
        h5, c5_5 = nesterov_cont(hc, N0, d5, k=55, lr0=0.018)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: SA on binarized Nesterov result
    if time.time() < deadline - 120:
        h_b6 = ensure_sum_binary(best_h, N0)
        d6 = min(deadline - 80, t0 + 950)
        h6, c5_6 = sa_guided(h_b6, N0, d6, T0=3.0, Tcool=0.998, T_min=0.02, drefresh=30, kick_every=100, kick_size=4)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final greedy polish
    if time.time() < deadline - 2:
        hF = ensure_sum_binary(best_h, N0)
        d7 = deadline - 1
        hF2, c5F = greedy_fixed(hF, N0, d7, max_sweeps=300, sf=0.9)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
