import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_bin(h, N):
    nh = N // 2
    h = np.clip(h, 0, 1)
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    if n1 == nh:
        return h_b
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        idx = np.where(h_b == 1)[0]
        h_b[rng.choice(idx, n1 - nh, replace=False)] = 0
    else:
        idx = np.where(h_b == 0)[0]
        h_b[rng.choice(idx, nh - n1, replace=False)] = 1
    return h_b

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    hp = np.where(vp, h[pp_c], 0.0)
    hm = np.where(vm, h[pm_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def exhaustive_single_swap(h, N, deadline):
    """Find and apply best single swap until no improvement."""
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    D = compute_D(h, N)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        D_o = D[ones, :]  # (n1, 2N-1)
        D_z = D[zeros, :]  # (n0, 2N-1)
        best_val = P
        best_pq = None

            mi = int(np.argmin(cand_max))

                best_val = float(cand_max[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:

            h[p] = 0.0
            h[q] = 1.0
            c = fft_corr(h)
            D = compute_D(h, N)
            P = c.max()
            improved = True
    return h, P * (2.0 / N)

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0 - h), L)))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_bin(h, N):
    nh = N // 2
    h = np.clip(h.astype(float), 0, 1)
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    rng = np.random.RandomState(7 + N)
    if n1 > nh:
        hb[rng.choice(np.where(hb == 1)[0], n1 - nh, replace=False)] = 0
    else:
        hb[rng.choice(np.where(hb == 0)[0], nh - n1, replace=False)] = 1
    return hb

def compute_D(h, N):
    sh = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = sh[None, :]
    pp, pm = pos + sv, pos - sv
    vp, vm = (pp >= 0) & (pp < N), (pm >= 0) & (pm < N)
    hp = np.where(vp, h[np.clip(pp, 0, N-1)], 0.0)
    hm = np.where(vm, h[np.clip(pm, 0, N-1)], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def swap_search(h, N, deadline, max_iters=200):
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    D = compute_D(h, N)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        D_z = D[zeros, :]
        best_P = P
        best_pq = None
        for p in ones:
            val = (c + D[p, None, :] - D_z[None, :, :]).max(axis=2)
            mi = int(np.argmin(val))
            if val[mi] < best_P - 1e-12:
                best_P = float(val[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
        D = compute_D(h, N)
        P = float(c.max())
    return h, P * (2.0 / N)

def block_search(h, N, deadline, max_block=4, max_iters=50):
    """Swap blocks of consecutive 1s with consecutive 0s."""
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    for it in range(max_iters):
        if time.time() > deadline:
            break
        improved = False
        # Find runs of 1s and 0s
        runs_1, runs_0 = [], []
        i = 0
        while i < N:
            if h[i] == 1:
                j = i
                while j < N and h[j] == 1:
                    j += 1
                if j - i <= max_block:
                    runs_1.append((i, j))
                i = j
            else:
                j = i
                while j < N and h[j] == 0:
                    j += 1
                if j - i <= max_block:
                    runs_0.append((i, j))
                i = j
        best_P = P
        best_move = None
        for (i1, i2) in runs_1:
            for (j1, j2) in runs_0:
                if i2 - i1 != j2 - j1:
                    continue
                # Try shifting the block
                new_h = h.copy()
                new_h[i1:i2] = 0
                new_h[j1:j2] = 1
                np_ = float(np.max(fft_corr(new_h)))
                if np_ < best_P - 1e-12:
                    best_P = np_
                    best_move = (i1, i2, j1, j2)
        if best_move is not None:
            i1, i2, j1, j2 = best_move
            h[i1:i2] = 0
            h[j1:j2] = 1
            c = fft_corr(h)
            P = float(c.max())
            improved = True
        if not improved:
            break
    return h, P * (2.0 / N)

def nesterov_opt(h, N, deadline, k=60, lr0=0.01):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr, beta, it = lr0, 0.9, 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        h_new = project_sum(y - lr * grad / k, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.97
            beta = min(0.97, 0.85 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

    x = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, x, h)

        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_bin(h, N):
    nh = N // 2
    h = np.clip(h.astype(float), 0, 1)
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    rng = np.random.RandomState(7 + N)
    if n1 > nh:
        hb[rng.choice(np.where(hb == 1)[0], n1 - nh, replace=False)] = 0
    else:
        hb[rng.choice(np.where(hb == 0)[0], nh - n1, replace=False)] = 1
    return hb

def compute_D(h, N):
    sh = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = sh[None, :]
    pp, pm = pos + sv, pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    hp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    hm = np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def swap_search(h, N, deadline, max_iters=300, temp0=0.002):
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    D = compute_D(h, N)
    best_h, best_P = h.copy(), P
    temp = temp0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        D_z = D[zeros, :]
        best_move_P = P
        best_pq = None
        for p in ones:
            val = (c + D[p, None, :] - D_z[None, :, :]).max(axis=2)
            mi = int(np.argmin(val))
            if val[mi] < best_move_P - 1e-12:
                best_move_P = float(val[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if stall > 10:
                break
            if stall > 4 and temp > 1e-5:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                c = fft_corr(h)
                D = compute_D(h, N)
                P = float(c.max())
            temp *= 0.95
            stall += 1
            continue
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
        D = compute_D(h, N)
        P = float(c.max())
        if P < best_P - 1e-13:
            best_P = P
            best_h = h.copy()
            stall = 0
            temp = max(temp * 0.98, 1e-5)
        else:
            stall += 1
    return best_h, best_P * (2.0 / N)

def block_search(h, N, deadline, max_blk=4):
    h = h.astype(float).copy()
    P = float(np.max(fft_corr(h)))
    for _ in range(40):
        if time.time() > deadline:
            break
        improved = False
        r1, r0 = [], []
        i = 0
        while i < N:
            j = i + 1
            while j < N and h[j] == h[i]:
                j += 1
            blen = j - i
            if blen <= max_blk:
                (r1 if h[i] == 1 else r0).append((i, j))
            i = j
        best_P = P
        best_mv = None
        for (a, b) in r1:
            for (c2, d2) in r0:
                if b - a != d2 - c2:
                    continue
                h[a:b] = 0.0
                h[c2:d2] = 1.0
                np_ = float(np.max(fft_corr(h)))
                h[a:b] = 1.0
                h[c2:d2] = 0.0
                if np_ < best_P - 1e-12:
                    best_P = np_
                    best_mv = (a, b, c2, d2)
        if best_mv:
            a, b, c2, d2 = best_mv
            h[a:b] = 0.0
            h[c2:d2] = 1.0
            P = best_P
            improved = True
        if not improved:
            break
    return h, P * (2.0 / N)

def nesterov_opt(h, N, deadline, k=60, lr0=0.008):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr, beta, it = lr0, 0.9, 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k)[-k:]
        pk = float(corr[top].max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        h_new = project_sum(y - lr * grad / k, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.96
            beta = min(0.97, 0.85 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def upsample_fn(h, N_new):
    x = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, x, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Fixed-sign exhaustive swap search at N0
    h1 = ensure_bin(h0, N0)
    d1 = min(dl - 450, t0 + budget_s * 0.28)
    h1, c5_1 = swap_search(h1, N0, d1, max_iters=500, temp0=0.003)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Block moves then swap again
    if time.time() < dl - 350:
        d2 = min(dl - 300, time.time() + budget_s * 0.10)
        h2, c5_2 = block_search(best_h, N0, d2, max_blk=4)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
        d2b = min(dl - 250, time.time() + budget_s * 0.08)
        h2b, c5_2b = swap_search(ensure_bin(best_h, N0), N0, d2b, max_iters=300, temp0=0.002)
        if c5_2b < best_c5:
            best_c5, best_h = c5_2b, h2b.copy()

    # Phase 3: Continuous Nesterov at N0
    if time.time() < dl - 220:
        hc = best_h + rng.randn(N0) * 0.025
        hc = project_sum(hc, N0 / 2.0)
        d3 = min(dl - 180, time.time() + budget_s * 0.13)
        h3, c5_3 = nesterov_opt(hc, N0, d3, k=65, lr0=0.006)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upscale to 2*N0, swap search
    N1 = 2 * N0
    if time.time() < dl - 130:
        h_up = ensure_bin(upsample_fn(best_h, N1), N1)
        d4 = min(dl - 80, time.time() + budget_s * 0.18)
        h4, c5_4 = swap_search(h_up, N1, d4, max_iters=200, temp0=0.002)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Second Nesterov (different perturbation)
    if time.time() < dl - 50:
        h5c = best_h[:N0] + rng.randn(N0) * 0.04
        h5c = project_sum(h5c, N0 / 2.0)
        d5 = min(dl - 30, time.time() + budget_s * 0.07)
        h5, c5_5 = nesterov_opt(h5c, N0, d5, k=50, lr0=0.010)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final swap polish at best resolution
    Nf = len(best_h)
    hf = ensure_bin(best_h, Nf)
    hf, c5f = swap_search(hf, Nf, dl - 1, max_iters=80, temp0=0.001)
    if c5f < best_c5:
        best_c5, best_h = c5f, hf.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
