    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def fix_sum(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:

        np.random.shuffle(idx)

def continuous_gradient_step(h, tau, top_k=200):
    """One gradient step on the softmax-max objective."""
    N = len(h)
    c = fft_corr(h)
    c_max = np.max(c)
    logs = (c - c_max) / tau
    w = np.exp(logs - np.max(logs))
    w_sum = w.sum()
    w = w / w_sum
    
    # Get top-k significant lags
    top_idx = np.argsort(-w)[:top_k]
    
    grad = np.zeros(N)
    center = N - 1
    for ki in top_idx:
        if w[ki] < 1e-8:
            break
        lag = ki - center
        wt = w[ki]
        if lag == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[lag:] += wt * (1.0 - h[:N-lag])
                grad[:N-lag] -= wt * h[lag:]
        else:
            M = -lag
            if M < N:
                grad[:N-M] += wt * (1.0 - h[M:])
                grad[M:] -= wt * h[:N-M]
    
    return grad, c_max + tau * np.log(w_sum + 1e-300)

    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    tau = tau0
    step = 0
    
    while time.time() < deadline - 5:
        step += 1
        grad, obj = continuous_gradient_step(h, tau, top_k)
        h_new = h - lr * grad
        # Project to [0,1] and fix sum
        h_new = np.clip(h_new, 0, 1)
        # Binary search for the shift that makes sum = target
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = (lo + hi) / 2
            if np.clip(h_new - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h_new = np.clip(h_new - (lo + hi) / 2, 0, 1)
        
        c_new = compute_c5(h_new, N)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        tau = max(tau_min, tau * 0.995)
        if step % 100 == 0:
            lr *= 0.95
    
    return best_h, best_c
