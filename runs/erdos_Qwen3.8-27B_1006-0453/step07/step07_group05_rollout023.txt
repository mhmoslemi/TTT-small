def fft_corr(h):

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def sum_constraint_project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} by shifting."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    s = exps.sum()
    w = exps / s
    f = (max_c + np.log(s)) / beta
    return f, w

    e = np.exp(beta*(corr-mc))

    w = e/s
    val = (mc + np.log(s))/beta
    g = np.zeros(N)

            g[:N-m] -= wt*h[m:]
        else:
            mm = -m
            g[:N-mm] += wt*(1-h[mm:])
            g[mm:] -= wt*h[:N-mm]
    return val, g

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def proj_sum(h, target):
