    elif n1 < target:
        zeros = np.where(h < 0.5)[0]
        h[rng.choice(zeros, target - n1, replace=False)] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def compute_deltas(h):
    N = len(h)
    L = 2 * N - 1
    deltas = np.zeros((N, L), dtype=np.float64)
    for i in range(N):
        s_val = 2.0 * h[i] - 1.0
        d = np.zeros(L)
        # First factor: j=i, term h[i]*(1-h[i+k]), change = -s_val*(1-h[i+k])/2 * 2 = -s_val*(1-h[i+k]) relative... 
        # Actually flipping i: h[i] -> 1-h[i]. Delta of h[i]*(1-h[i+k]) = (1-2h[i])*(1-h[i+k]) = -s_val*(1-h[i+k])... 
        # Wait: new_h[i] = 1-h[i], old = h[i]. Delta = (1-h[i]-h[i])*(1-h[i+k]) = -s_val*(1-h[i+k])
        # Second factor: j+k=i means j=i-k. Delta = h[j]*(1-new_h[i]) - h[j]*(1-h[i]) = h[j]*s_val
        for k in range(1 - N, N):
            idx = k + N - 1
            ipk = i + k
            ik = i - k
            val = 0.0
            # First factor contribution (j=i): delta = -s_val * (1-h[i+k]) if i+k in range
            if 0 <= ipk < N:
                val += -s_val * (1.0 - h[ipk])
            # Second factor contribution (j=i-k): delta = h[i-k] * s_val if i-k in range
            if 0 <= ik < N:
                val += h[ik] * s_val
            d[idx] = val
        deltas[i] = d
    return deltas

def tabu_sa(h_bin, deadline, rng, n_peaks=6, n_per_peak=15, tabu_len=80,
            cool_rate=0.9993, t_start=0.035, max_iters=150000):
    N = len(h_bin)
    L = 2 * N - 1
    h = h_bin.copy().astype(np.float64)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    T = t_start
    T_min = 0.001
    it = 0
    no_impr = 0
    tabu_until = np.full(N, -1, dtype=np.int64)
    deltas = compute_deltas(h)

    while time.time() < deadline and it < max_iters:
        it += 1
        T = max(t_start * (cool_rate ** it), T_min)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        # Target high-peak lags
        thresh = best_c * 0.82
        peak_lags = np.where(corr > thresh)[0]
        if len(peak_lags) > n_peaks * 2:
            peak_lags = np.argpartition(corr, -n_peaks * 2)[-n_peaks * 2:]
        elif len(peak_lags) == 0:
            peak_lags = np.argpartition(corr, -n_peaks)[-n_peaks:]

        cand_p = []
        cand_q = []
        for pk in peak_lags:
            k_off = int(pk) - (N - 1)
            if k_off >= 0:
                vi = np.arange(0, N - k_off)
                mask = (h[vi] == 1) & (h[vi + k_off] == 0)
                bp = vi[mask]
                bq = (vi + k_off)[mask]
            else:
                M = -k_off
                vi = np.arange(M, N)
                mask = (h[vi] == 1) & (h[vi - M] == 0)
                bp = vi[mask]
                bq = (vi - M)[mask]
            if len(bp) > 0:
                ns = min(n_per_peak, len(bp))
                sel = rng.choice(len(bp), ns, replace=False)
                cand_p.extend(bp[sel].tolist())
                cand_q.extend(bq[sel].tolist())

        # Random diversification
        nr = min(8, len(ones), len(zeros))
        if nr > 0:
            cand_p.extend(rng.choice(ones, nr, replace=False).tolist())
            cand_q.extend(rng.choice(zeros, nr, replace=False).tolist())

        cand_p = np.array(sorted(set(cand_p)), dtype=int)
        cand_q = np.array(sorted(set(cand_q)), dtype=int)

        # Tabu filter
        tp = cand_p[tabu_until[cand_p] < it]
        tq = cand_q[tabu_until[cand_q] < it]
        if len(tp) > 0 and len(tq) > 0:
            cand_p, cand_q = tp, tq

        if len(cand_p) == 0 or len(cand_q) == 0:
            break

        # Evaluate swaps in chunks
        chunk = 5
        best_new = 1e10
        bi, bj = 0, 0
        dp = deltas[cand_p]
        dq = deltas[cand_q]
        for ci in range(0, len(cand_p), chunk):
            ce = min(ci + chunk, len(cand_p))
            vals = corr[np.newaxis, np.newaxis, :] + dp[ci:ce, np.newaxis, :] + dq[np.newaxis, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new:
                best_new = float(nm[mi])
                bi, bj = mi[0] + ci, mi[1]

        p = int(cand_p[bi])
        q = int(cand_q[bj])
        dc = best_new - best_c

        if dc < -1e-12 or (T > T_min and rng.random() < np.exp(-dc / (T + 1e-9))):
            h[p], h[q] = 0.0, 1.0
            corr += deltas[p] + deltas[q]
            tabu_until[p] = it + tabu_len
            tabu_until[q] = it + tabu_len
            c = float(corr.max())
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            no_impr += 1

        if no_impr > 6000:
            break
        if it % 2000 == 0:
            deltas = compute_deltas(h)

    return best_h, best_c

def continuous_polish(h_in, N, deadline, lr=0.008, k_top=30):
    h = h_in.astype(float).copy()
    hx = h.copy()
    m = np.zeros(N)
    beta = 0.80
    best_pk = float('inf')
    best_h = h.copy()
    it = 0
    while time.time() < deadline and it < 5000:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        adj = (h_new.sum() - N / 2.0) / N
        h_new = np.clip(h_new - adj, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.98
            beta = min(0.93, 0.78 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = make_binary(h0, N0, rng)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Main SA at native resolution
    d1 = min(deadline - 400, t0 + 300)
    h1, _ = tabu_sa(h_bin, d1, rng, n_peaks=6, n_per_peak=18, tabu_len=100,
                    cool_rate=0.9992, t_start=0.04)
    c1 = compute_c5(h1, N0)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Continuous polish then rebinarize
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 100)
        hc = best_h.astype(float) + rng.randn(N0) * 0.03
        hc = np.clip(hc, 0, 1)
        hc = np.clip(hc - (hc.sum() - N0 / 2.0) / N0, 0, 1)
        hn, _ = continuous_polish(hc, N0, d2, lr=0.008, k_top=40)
        hb = make_binary(hn, N0, rng)
        c2 = compute_c5(hb, N0)
        if c2 < best_c5:
            best_c5, best_h = c2, hb.copy()

    # Phase 3: Second SA pass (lower temp, more peaks)
    if time.time() < deadline - 200:
        h3 = make_binary(best_h, N0, rng)
        d3 = min(deadline - 120, time.time() + 200)
        h3, _ = tabu_sa(h3, d3, rng, n_peaks=8, n_per_peak=25, tabu_len=60,
                        cool_rate=0.9996, t_start=0.02)
        c3 = compute_c5(h3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Try other resolutions
    for N_try in [320, 280, 240, 360, 400]:
        if time.time() > deadline - 80:
            break
        if N_try % 2 != 0:
            N_try += 1
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hb = make_binary(hi, N_try, rng)
        # Also try random perturbation at this resolution
        sub_dl = min(deadline - 40, time.time() + 60)
        hr, _ = tabu_sa(hb, sub_dl, rng, n_peaks=5, n_per_peak=12, tabu_len=50,
                        cool_rate=0.999, t_start=0.04)
        cr = compute_c5(hr, N_try)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 5: Final fine-tune
    if time.time() < deadline - 5:
        Ny = len(best_h)
        h5 = make_binary(best_h, Ny, rng)
        h5, _ = tabu_sa(h5, deadline - 2, rng, n_peaks=10, n_per_peak=30,
                        tabu_len=40, cool_rate=0.9998, t_start=0.012)
        c5 = compute_c5(h5, Ny)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
