                best_h = h.copy()

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(lag, h, N):
    g = np.zeros(N)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:N - lag]
        g[:N - lag] -= h[lag:N]
    else:
        M = -lag
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(np.sum(e)) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-8:
            g += w[i] * lin_grad(int(i - ctr), h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s
    return val, g

def lbfgs_opt(h, deadline, T_list, lam0=5.0, inner_iter=3):
    N = len(h)
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for T in T_list:
        if time.time() > deadline:
            break
        lam = lam0
        for _ in range(inner_iter):
            if time.time() > deadline:
                break
            res = sp_min(smooth_obj_grad, h, args=(T, target, lam),
                        jac=True, method='L-BFGS-B',
                        bounds=[(0, 1)] * N,
                        options={'maxiter': 120, 'maxfun': 350, 'ftol': 1e-14})
            h = proj01(res.x, target)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lam *= 3.0
    return best_h, best_c

def binary_block_search(h, N, deadline, max_iters=500, ntrials=18, block_sizes=(1, 2, 3)):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.random.choice(np.where(h == 1)[0], n1 - nh, replace=False)] = 0
    elif n1 < nh:
        h[np.random.choice(np.where(h == 0)[0], nh - n1, replace=False)] = 1
    best_c = np.max(fft_corr(h)) * (2.0 / N)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        c_val = P * (2.0 / N)
        improved = False
        if c_val < best_c - 1e-14:
            best_c = c_val
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bd, best_move = 0.0, None
        for _ in range(ntrials):
            bsz = int(np.random.choice(block_sizes))
            if bsz > len(ones) or bsz > len(zeros):
                bsz = 1
            si = np.random.randint(0, len(ones) - bsz + 1)
            zi = np.random.randint(0, len(zeros) - bsz + 1)
            oi = ones[si:si + bsz]
            zi2 = zeros[zi:zi + bsz]
            if len(set(oi.tolist()) & set(zi2.tolist())) > 0:
                continue
            h[oi] = 0.0
            h[zi2] = 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            h[oi] = 1.0
            h[zi2] = 0.0
            if d > bd:
                bd = d
                best_move = (oi.copy(), zi2.copy())
        if best_move is not None and bd > 0:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c

def subgrad_ne(h, N, deadline, alpha0=0.03, beta=0.92, atol=1e-5, max_iter=6000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < max_iter:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            alpha *= 0.95
            beta = min(0.97, 0.88 + 0.0002 * it)
    return best_h, best_pk * (2.0 / N)
