import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_w(corr, T):
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    return e / np.sum(e)

def proj_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s < 1e-14:
            return h
        h = h + room * (diff / s)
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_smoothed(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    w = soft_max_w(corr, T)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += wk * (1.0 - h[:n - k])
            g[:n - k] += wk * (-h[k:])
        else:
            m = -k
            g[:n - m] += wk * (1.0 - h[m:])
            g[m:] += wk * (-h[:n - m])
    return g * dx

def momentum_desc(h0, T, n, max_iter, alpha, gamma, t_end=None):
    h = h0.copy()
    v = np.zeros(n)
    for i in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        g = grad_smoothed(h, T)
        v = gamma * v + alpha * (-g)
        h = proj_h(h + v)
    return h

def coord_descent(h, n, t_end, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lag_arr = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_h[i] = 0.0
            corr_base = get_corr(best_h)
            best_h[i] = h_save
            j1 = i + lag_arr
            j2 = i - lag_arr
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] = 1.0 - best_h[np.clip(j1[v1], 0, n - 1)]
            coefs[v2] += -best_h[np.clip(j2[v2], 0, n - 1)]
            dx = 2.0 / n
            res = minimize_scalar(
                lambda v: np.max(corr_base + v * coefs) * dx,
                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            hi_new = float(res.x)
            if abs(hi_new - h_save) > 1e-13:
                h_try = best_h.copy()
                h_try[i] = hi_new
                h_try = proj_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5
