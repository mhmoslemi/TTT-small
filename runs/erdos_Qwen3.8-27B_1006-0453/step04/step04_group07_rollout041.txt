```python
def fast_coord_descent(h_in, n, t_end, max_sweeps=5000, tol=1e-13):
    dx = 2.0 / n
    best_h = h_in.copy()
    one_minus_h = 1.0 - best_h
    corr = np.correlate(best_h, one_minus_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0 = n - 1
    
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            # Build delta_coeff of length 2n-1
            dc = np.zeros(2*n-1)
            # Term 1: delta*(1-h[i-k]) for valid i-k
            # k = m - c0, need 0 <= i-k < n → c0-i < m <= c0+i, i.e., n-1-i < m <= n-1+i
            # m ranges from max(0, n-i) to min(2n-2, n+i-1)
            lo1 = max(0, n - i)
            hi1 = min(2*n-2, n+i-1)
            for m in range(lo1, hi1+1):
                k = m - c0
                dc[m] += (1.0 - best_h[i-k])
            # Term 2: -delta*h[i+k] for valid i+k
            # k = m - c0, need 0 <= i+k < n → -i <= k < n-i → c0-i <= m < c0+n-i
            # m ranges from max(0, c0-i) to min(2n-2, c0+n-i-1)
            lo2 = max(0, c0 - i)
            hi2 = min(2*n-2, c0 + n - i - 1)
            for m in range(lo2, hi2+1):
                k = m - c0
                dc[m] -= best_h[i+k]
            
            # For m=c0 (k=0): the linear part is (1-old) + (-old) = 1-2old ✓
            # Need to subtract d² at m=c0
            
            def obj1d(v):
                d = v - old
                mx = np.max(corr + d*dc)
                # Correct the k=0 entry
                c0_val = corr[c0] + d*dc[c0] - d*d
                if c0_val > mx:
                    mx = c0_val
                return mx
            
            f0 = obj1d(0.0)
            f1 = obj1d(1.0)
            fcur = best_c5 / dx
            
            if f0 <= f1 and f0 <= fcur:
                bv, bf = 0.0, f0
            elif f1 <= f0 and f1 <= fcur:
                bv, bf = 1.0, f1
            else:
                res = minimize_scalar(obj1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9, 'maxiter': 30})
                bv, bf = res.x, res.fun
                if f0 < bf: bv, bf = 0.0, f0
                if f1 < bf: bv, bf = 1.0, f1
            
            if bf * dx < best_c5 - tol:
                d = bv - old
                best_h[i] = bv
                one_minus_h[i] = 1.0 - bv
                corr += d * dc
                corr[c0] -= d * d
                best_c5 = bf * dx
                improved = True
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
```

One more thought: the "binary local search" idea. For binary h (values exactly 0 or 1), the problem is:
- Find subset S of size n/2 minimizing max_k |{i : i∈S, i+k∉S}| (with boundary handling)

This can be attacked with:
1. Start from a binary approximation of the current h
2. Try flipping individual bits (0→1 and 1→0, must maintain |S|=n/2)
3. Accept if it reduces the max

Since we need to maintain |S|=n/2, we do "swap" moves: flip a 0 to 1 and a 1 to 0 simultaneously.

The effect of flipping h[a] from 0 to 1 and h[b] from 1 to 0:
- For each shift k: corr[k] changes by:
  - From h[a]: +(1-h[a+k]) - h[a-k] (linear, since it goes 0→1)
  - From h[b]: -(1-h[b+k]) + h[b-k] (linear, since it goes 1→0)

Actually for binary: h[a] goes from 0 to 1 (d=+1), h[b] goes from 1 to 0 (d=-1).
The change in corr[m] is: (+1)*dc_a[m] + (-1)*dc_b[m] where dc_a and dc_b are the coefficient vectors for positions a and b.

This is a 1D problem (which pair to swap), and we can just try all pairs (O(n²) per iteration).

```python
    c0v = corr[c0] + d * dc[c0] - d * d
    return max(mx, c0v)

def fast_cd(h_in, n, t_end, max_sweeps=5000, tol=1e-13):
    h = h_in.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    c0 = n - 1
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        if sw > 0 and sw % 300 == 0:
            corr[:] = np.correlate(h, 1.0 - h, mode='full')
            best_c5 = np.max(corr) * (2.0 / n)
        improved = False
        for i in range(n):
            old = h[i]
            dc = np.zeros(2 * n - 1)
            lo = max(0, i - n + 1)
            js = np.arange(lo, n)
            dc[n - 1 - i + js] += 1.0 - h[js]
            dc[n - 1 + i - js] -= h[js]
            f0 = max(np.max(corr - old * dc), corr[c0] - old * dc[c0] - old * old)
            f1 = max(np.max(corr + (1.0 - old) * dc), corr[c0] + (1.0 - old) * dc[c0] - (1.0 - old) ** 2)
            fc = best_c5 / (2.0 / n)
            if f0 <= f1 and f0 <= fc:
                bv, bf = 0.0, f0
            elif f1 <= f0 and f1 <= fc:
                bv, bf = 1.0, f1
            else:
                res = minimize_scalar(_obj_cd, args=(old, corr, dc, c0),
                                      bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 30})
                bv, bf = res.x, res.fun
                if f0 < bf:
                    bv, bf = 0.0, f0
                if f1 < bf:
                    bv, bf = 1.0, f1
            if bf * (2.0 / n) < best_c5 - tol:
                d = bv - old
                h[i] = bv
                corr += d * dc
                corr[c0] -= d * d
                best_c5 = bf * (2.0 / n)
                improved = True
        if not improved:
            break
    return h, best_c5

def _smooth.obj(v, T, n):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def slsqp_opt(h0, T, n, maxiter=400):
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)
    def _con(x):
        return x.sum() - n * 0.5
    res = minimize(_obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _con}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def _slsqp_obj(x, dy_ref):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(500.0 * (corr - m)))) / 500.0) * (2.0 / n)

def _slsqp_con(x):
    return x.sum() - len(x) * 0.5

def slsqp_light(h0, T, n, maxiter=200):
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        ss = np.sum(np.exp(T * (corr - m)))
        return (m + np.log(ss) / T) * (2.0 / n)
    res = minimize(_obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def basin_hop(h_in, c5_in, n, t_end, max_iters=300, seed_rng=None):
    rng = seed_rng if seed_rng is not None else np.random
    best_h, best_c5 = h_in.copy(), c5_in
    work_h, work_c5 = h_in.copy(), c5_in
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - 0.9 * it / max_iters)
        ptype = rng.randint(0, 4)
        if ptype == 0:
            step = 0.03 + 0.2 * cool
            h_p = work_h + step * rng.randn(n)
        elif ptype == 1:
            h_p = work_h.copy()
            bl = rng.randint(10, n // 2)
            st = rng.randint(0, n - bl)
            h_p[st:st + bl] = 1.0 - h_p[st:st + bl]
        elif ptype == 2:
            h_p = (work_h > 0.5).astype(float)
            h_p = project(h_p, n)
            h_p += 0.1 * rng.randn(n)
        else:
            h_p = work_h.copy()
            idx = rng.choice(n, size=n // 3, replace=False)
            h_p[idx] = rng.uniform(0, 1, size=len(idx))
        h_p = project(h_p, n)
        hc, c5v = fast_cd(h_p, n, min(time.time() + 8.0, t_end), max_sweeps=300)
        temp = 0.003 * cool + 0.0002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def binary_swap(h_in, n, t_end, max_iters=200):
    h = (h_in > 0.5).astype(float)
    h = project(h, n)
    c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        improved = False
        zeros = np.where(h <= 0.5)[0]
        for ai in zeros:
            dc_a = np.zeros(2 * n - 1)
            lo = max(0, ai - n + 1)
            js = np.arange(lo, n)
            dc_a[n - 1 - ai + js] += 1.0 - h[js]
            dc_a[n - 1 + ai + 0 - js] += 0
            dc_a[n - 1 + ai - js] -= h[js]
            for bi in ones:
                if bi == ai:
                    continue
                new_corr = corr + dc_a
                new_corr[n - 1 + bi - js] -= (1.0 - h[js])
                new_corr[n - 1 - bi + js] += h[js]
                nm = np.max(new_corr)
                if nm < mx - 1e-12:
                    h[ai] = 1.0
                    h[bi] = 0.0
                    corr = new_corr
                    mx = nm
                    c5 = mx * (2.0 / n)
                    ones = np.where(h > 0.5)[0]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.ones(n_init) * 0.5
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent (30%)
    t1 = t0 + budget_s * 0.30
    best_h, best_c5 = fast_cd(best_h, n, t1, max_sweeps=10000)

    # Phase 2: SLSQP (10%)
    t2 = t0 + budget_s * 0.40
    for T in [100, 500, 2000]:
        if time.time() > t2:
            break
        try:
            hc = slsqp_light(best_h, T, n, maxiter=300)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 3: More coordinate descent (10%)
    t3 = t0 + budget_s * 0.50
    best_h, best_c5 = fast_cd(best_h, n, t3, max_sweeps=10000)

    # Phase 4: Basin hopping (25%)
    t4 = t0 + budget_s * 0.75
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t4, max_iters=500, seed_rng=rng)

    # Phase 5: Binary swap search (10%)
    t5 = t0 + budget_s * 0.85
    h_bin, c5_bin = binary_swap(best_h, n, t5, max_iters=100)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin

    # Phase 6: Different resolutions (8%)
    t6 = t0 + budget_s * 0.93
    for n_try in [80, 100, 120, 200, 240]:
        if time.time() > t6:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = fast_cd(h_tr, n_try, min(time.time() + 15, t6), max_sweeps=2000)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 7: Final polish (7%)
    t7 = t0 + budget_s * 0.97
    best_h, best_c5 = fast_cd(best_h, n, t7, max_sweeps=10000)
    if time.time() < t0 + budget_s * 0.99:
        try:
            hc = slsqp_light(best_h, 5000, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
