import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    v = h[i]
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    coef = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        c = 0.0
        if 0 <= i+lag < n:
            c += (1.0 - hz[i+lag])
        if 0 <= i-lag < n:
            c += (-hz[i-lag])
        coef[k] = c
    k0 = n-1
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] += t*(1.0-t) - v*(1.0-v)
        return float(np.max(cv))
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=1000, reverse_every=10):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    sweep = 0
    while time.time() < t_end and sweep < max_sweeps:
        improved = False
        order = np.arange(n) if (sweep // reverse_every) % 2 == 0 else np.arange(n-1, -1, -1)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        sweep += 1
        if not improved and sweep > 3:
            break
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n-1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n-lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n-lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n-nl])
                grad[:n-nl] += pk * (-h[nl:])
    return grad * dx

def adam_opt(h, n, t_end, lr=0.01, T_init=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    T = T_init
    t = 0
    while time.time() < t_end:
        t += 1
        g = smooth_grad(best_h, T)
        m = beta1*m + (1-beta1)*g
        v = beta1*v + (1-beta1)*g*g
        mhat = m / (1 - beta1**t)
        vhat = v / (1 - beta2**t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        trial = proj(best_h - step, n)
        c5_try = compute_c5(trial)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = trial
            T = min(T * 1.05, 20000)
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
    return best_h, best_c5

    t_phase4 = t0 + budget_s * 0.70

    n_particles = 4

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    v = h[i]
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        if 0 <= i + lag < n:
            c += (1.0 - hz[i + lag])
        if 0 <= i - lag < n:
            c += (-hz[i - lag])
        coef[k] = c
    k0 = n - 1
    quad_term = v * (1.0 - v)
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] += t * (1.0 - t) - quad_term
        return float(np.max(cv))
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.arange(n) if (sweep % 2 == 0) else np.arange(n - 1, -1, -1)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved and sweep > 2:
            break
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def nesterov_opt(h, n, t_end, lr_init=0.005, T_init=400, beta=0.9):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    x = best_h.copy()
    lr = lr_init
    T = T_init
    for it in range(200000):
        if time.time() > t_end:
            break
        look = x - lr * v
        look = proj(look, n)
        g = smooth_grad(look, T)
        v = beta * v - lr * g
        x_new = proj(x + v, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new.copy()
            T = min(T * 1.03, 30000)
            if it % 500 == 0:
                lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.85
            v *= 0.5
            x = best_h.copy()
            if lr < 1e-13:
                break
    return best_h, best_c5

def structured_perturb(h, n, cool, rng):
    r = rng.rand()
    x = np.linspace(0, 2, n, endpoint=False)
    if r < 0.35:
        m_mode = rng.randint(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.08 + 0.22 * cool
        pert = amp * np.cos(np.pi * m_mode * x + phase)
    elif r < 0.6:
        center = rng.uniform(0, 2)
        width = rng.uniform(0.08, 0.4)
        amp = 0.1 + 0.25 * cool
        sign = 1.0 if rng.rand() > 0.5 else -1.0
        pert = sign * amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    elif r < 0.8:
        m1, m2 = rng.randint(1, 6), rng.randint(2, 10)
        amp1, amp2 = 0.05 * cool, 0.04 * cool
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        pert = amp1 * np.sin(np.pi * m1 * x + p1) + amp2 * np.sin(np.pi * m2 * x + p2)
    else:
        pert = (0.04 + 0.1 * cool) * rng.standard_normal(n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 180
        x = np.linspace(0, 2, n_init, endpoint=False)
        best_h = 0.5 + 0.42 * np.cos(np.pi * x) + 0.06 * np.cos(2 * np.pi * x)
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent
    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=3000)

    # Phase 2: Nesterov
    t2 = t0 + budget_s * 0.32
    if time.time() < t2:
        hn, c5n = nesterov_opt(best_h, n, t2, lr_init=0.006, T_init=500)
        if c5n < best_c5:
            best_h, best_c5 = hn, c5n
            best_h, best_c5 = coord_descent(best_h, len(best_h), t0 + budget_s * 0.40, max_sweeps=2000)

    # Phase 3: More coordinate descent
    t3 = t0 + budget_s * 0.48
    best_h, best_c5 = coord_descent(best_h, len(best_h), t3, max_sweeps=3000)

    # Phase 4: Basin hopping with structured perturbations
    t4 = t0 + budget_s * 0.72
    work_h = best_h.copy()
    work_c5 = best_c5
    n = len(best_h)
    for it in range(80):
        if time.time() > t4:
            break
        cool = max(0.15, 1.0 - 0.85 * it / 80.0)
        h_pert = proj(structured_perturb(work_h, n, cool, rng), n)
        h_pol, c5_pol = coord_descent(h_pert, n, min(time.time() + 6.0, t4), max_sweeps=80)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol.copy(), c5_pol
            work_h, work_c5 = h_pol.copy(), c5_pol
        else:
            temp = 0.003 * cool + 0.0004
            if c5_pol < work_c5 or rng.rand() < np.exp(-(c5_pol - work_c5) / max(temp, 1e-10)):
                work_h, work_c5 = h_pol.copy(), c5_pol

    # Phase 5: Multi-resolution
    t5 = t0 + budget_s * 0.88
    for n_try in [144, 160, 200, 220, 240]:
        if time.time() > t5:
            break
        n_cur = len(best_h)
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = proj(h_interp, n_try)
        t_sub = min(time.time() + budget_s * 0.025, t5)
        h_r, c5_r = coord_descent(h_interp, n_try, t_sub, max_sweeps=600)
        if time.time() < t5:
            h_r2, c5_r2 = nesterov_opt(h_r, n_try, min(time.time() + budget_s * 0.015, t5), lr_init=0.004, T_init=600)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r

    # Phase 6: Final coordinate descent
    n_fin = len(best_h)
    t6 = t0 + budget_s * 0.98
    best_h, best_c5 = coord_descent(best_h, n_fin, t6, max_sweeps=5000)

    # Phase 7: Micro-polish
    for _ in range(2):
        if time.time() > t0 + budget_s * 0.995:
            break
        for i in range(n_fin):
            if time.time() > t0 + budget_s * 0.995:
                break
            hi = coord_opt(best_h, n_fin, i)
            if abs(hi - best_h[i]) > 1e-14:
                best_h[i] = hi
                best_h = proj(best_h, n_fin)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
