```python
        vals = corr0 + t * s
        vals[center] = corr0[center] + t - t*t
        return float(np.max(vals))
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-12:
            h = h + room * (diff / tot)
        else:
            h = np.clip(h * (target / max(s, 1e-12)), 0, 1)
    else:
        h = h * (target / max(s, 1e-12))
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    center = n - 1
    h0 = h.copy()
    h0[i] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    s = np.zeros(2 * n - 1)
    for d in range(1, n):
        ip = i - d
        ip2 = i + d
        a = 1.0 - (h[ip] if 0 <= ip < n else 0.0)
        b = h[ip2] if 0 <= ip2 < n else 0.0
        s[center - d] = a - b
        a2 = 1.0 - (h[ip2] if 0 <= ip2 < n else 0.0)
        b2 = h[ip] if 0 <= ip < n else 0.0
        s[center + d] = a2 - b2
    def f1d(t):
        vals = corr0 + t * s
        vals[center] = corr0[center] + t - t * t
        return float(np.max(vals))
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500, jitter=0.0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved and jitter == 0:
            break
        if jitter > 0 and sweep % 20 == 19:
            h_p = best_h + jitter * np.random.randn(n)
            h_p = project(h_p, n)
            c5p = compute_c5(h_p)
            if c5p > best_c5 * 1.002:
                h_p = best_h
            for j in range(n):
                h_p[j] = coord_opt(h_p, n, j)
                h_p = project(h_p, n)
            c5p = compute_c5(h_p)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = h_p
    return best_h, best_c5

def subgrad_momentum(h, n, t_end, iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    alpha = 0.1
    center = n - 1
    for it in range(iters):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = corr.max()
        thresh = m - 0.002 * n
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for idx in active:
            d = center - idx
            for i in range(n):
                ip = i - d
                ip2 = i + d
                if d == 0:
                    grad[i] += 1.0 - 2.0 * best_h[i]
                else:
                    g = 0.0
                    if 0 <= ip < n:
                        g += 1.0 - best_h[ip]
                    if 0 <= ip2 < n:
                        g -= best_h[ip2]
                    grad[i] += g
        grad /= len(active)
        v = 0.9 * v - alpha * grad
        h_new = project(best_h + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            alpha = min(alpha * 1.02, 0.5)
        else:
            alpha *= 0.9
            v *= 0.5
    return best_h, best_c5

def interpolate_to(h_src, n_src, n_dst):
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    h_new = np.interp(x_new, x_old, h_src, period=2.0)
    return project(h_new, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = 0.5 + 0.2*np.sin(2*np.pi*x) + 0.1*np.sin(4*np.pi*x)
        best_h = np.clip(best_h, 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=100)

    t2 = t0 + budget_s * 0.50
    ns = [80, 100, 120, 160, 200, 240, 320]
    for n_try in ns:
        if time.time() > t2:
            break
        if n_try == n:
            h_cur = best_h.copy()
        else:
            h_cur = interpolate_to(best_h, n, n_try)
        h_cur, c5_cur = coord_descent(h_cur, n_try, t2, max_sweeps=60)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur
            n = n_try

    t3 = t0 + budget_s * 0.72
    for it in range(50):
        if time.time() > t3:
            break
        h_sg, c5_sg = subgrad_momentum(best_h, n, min(time.time() + 15, t3), iters=200)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
        h_cd, c5_cd = coord_descent(best_h, n, min(time.time() + 10, t3), max_sweeps=30)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    t4 = t0 + budget_s * 0.82
    for n_try in [max(64, n//2), n, min(384, n*2)]:
        if time.time() > t4:
            break
        if n_try == n:
            h_cur = best_h.copy()
        else:
            h_cur = interpolate_to(best_h, n, n_try)
        h_cur, c5_cur = coord_descent(h_cur, n_try, t4, max_sweeps=40, jitter=0.03)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur
            n = n_try

    t5 = t0 + budget_s * 0.92
    h_cur, c5_cur = coord_descent(best_h, n, t5, max_sweeps=300)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur

    t6 = t_end - 2.0
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=500)

    for it in range(10):
        if time.time() > t6:
            break
        h_jit = best_h + 0.02 * np.random.randn(n)
        h_jit = project(h_jit, n)

        c5_j = compute_c5(h_jit)
        if c5_j < best_c5:
            best_c5 = c5_j
            best_h = h_jit

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
