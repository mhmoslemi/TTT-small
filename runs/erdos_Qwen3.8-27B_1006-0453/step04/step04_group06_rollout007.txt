import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    m = corr.max()
    exp_t = np.exp(T * (corr - m))
    s_exp = exp_t.sum()
    obj = (m + np.log(s_exp) / T) * dx
    w = exp_t / s_exp
    grad = np.zeros(n)
    grad += w[center] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[center + k]
        if wk > 1e-14:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        wkm = w[center - k]
        if wkm > 1e-14:
            grad[:n - k] += wkm * (1.0 - h[k:])
            grad[k:] -= wkm * h[:n - k]
    grad *= dx
    return obj, grad

def coord_opt_fast(h, n, p):
    h0 = h.copy()
    h0[p] = 0.0
    corr_base = np.correlate(h0, 1.0 - h0, mode='full')
    center = n - 1
    dx = 2.0 / n
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        c = 0.0
        if p >= k:
            c += 1.0 - h[p - k]
        if p < n - k:
            c -= h[p + k]
        a[center + k] = c
        c = 0.0
        if p < n - k:
            c += 1.0 - h[p + k]
        if p >= k:
            c -= h[p - k]
        a[center - k] = c
    ct = corr_base[center]
    cb = corr_base
    aa = a
    def f1d(x):
        vals = cb + x * aa
        vals[center] = ct + x - x * x
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coord_descent_fast(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end - 0.5:
                return best_h, best_c5
            hi_new = coord_opt_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-9:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_onto_sum(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def fista_smooth(h, n, T, max_iter=200):
    x = h.copy()
    z = h.copy()
    t_mom = 1.0
    obj, grad = smooth_obj_grad(z, T)
    L = 80.0
    for i in range(max_iter):
        x = z - grad / L
        x = project_onto_sum(x, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z_new = x + ((t_mom - 1.0) / t_new) * (x - z)
        z_new = np.clip(z_new, 0.0, 1.0)
        obj_new, grad_new = smooth_obj_grad(z_new, T)
        if obj_new > obj + np.dot(grad, z_new - z) + L * np.sum((z_new - z) ** 2):
            L *= 2.5
            x = z - grad / L
            x = project_onto_sum(x, n)
            z_new = x + ((t_mom - 1.0) / t_new) * (x - z)
            z_new = np.clip(z_new, 0.0, 1.0)
            obj_new, grad_new = smooth_obj_grad(z_new, T)
        z, t_mom = z_new, t_new
        obj, grad = obj_new, grad_new
    return project_onto_sum(z, n)

def make_fourier_start(n, a1, a2, a3):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + a1 * np.cos(np.pi * x) + a2 * np.cos(2 * np.pi * x) + a3 * np.cos(3 * np.pi * x)
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_onto_sum(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = make_fourier_start(160, 0.28, 0.12, 0.05)
    n = len(h_init)
    best_h = project_onto_sum(h_init, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent
    t1 = t0 + budget_s * 0.30
    best_h, best_c5 = coord_descent_fast(best_h, n, t1)

    # Phase 2: FISTA with increasing T + coord polish
    for T in [100, 500, 2000, 10000]:
        if time.time() > t0 + budget_s * 0.50:
            break
        h_f = fista_smooth(best_h, n, T, max_iter=200)
        c5f = compute_c5(h_f)
        if c5f < best_c5:
            best_h, best_c5 = h_f, c5f
        h_cd, c5_cd = coord_descent_fast(best_h, n, min(time.time() + 8, t0 + budget_s * 0.50))
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 3: Multi-perturbation restart
    for attempt in range(6):
        if time.time() > t0 + budget_s * 0.72:
            break
        scale = 0.03 + 0.02 * attempt
        h_p = best_h + scale * np.random.randn(n)
        h_p = project_onto_sum(h_p, n)
        h_p = fista_smooth(h_p, n, 1000, max_iter=80)
        h_p, c5p = coord_descent_fast(h_p, n, min(time.time() + 15, t0 + budget_s * 0.72))
        if c5p < best_c5:
            best_h, best_c5 = h_p, c5p

    # Phase 4: Multi-resolution refinement
    for n_try in [120, 200, 240]:
        if time.time() > t_end - 60:
            break
        h_int = interpolate_to(best_h, n, n_try)
        h_int = fista_smooth(h_int, n_try, 500, max_iter=100)
        h_int, c5i = coord_descent_fast(h_int, n_try, min(time.time() + 30, t_end - 60))
        if c5i < best_c5:
            best_h, best_c5, n = h_int, c5i, n_try

    # Phase 5: Final long coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t_end - 3)

    # Final: one more FISTA at very high T + coord
    if time.time() < t_end - 5:
        h_f = fista_smooth(best_h, n, 50000, max_iter=50)
        c5f = compute_c5(h_f)
        if c5f < best_c5:
            best_h = h_f
            best_c5 = c5f
        best_h, best_c5 = coord_descent_fast(best_h, n, t_end - 1)

    best_h = project_onto_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
