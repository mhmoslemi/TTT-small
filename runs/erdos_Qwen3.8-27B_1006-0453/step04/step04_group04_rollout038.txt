```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / T

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if p[k] < 1e-15:
            continue
        lag = k - (n-1)
        if lag >= 0:
            L = n - lag
            grad[:L] += p[k] * (1.0 - h[lag:])
            grad[lag:] += p[k] * (-h[:L])
        else:
            L = n + lag
            grad[-lag:] += p[k] * (1.0 - h[:L])
            grad[:L] += p[k] * (-h[-lag:])
    return grad * dx

def proj(h, n):
    h = np.clip(h, 0, 1).copy()
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = (1 - h).sum()
        if r > 0: h += (1-h) * (d/r)
    else:
        s = h.sum()
        if s > 0: h *= -d/s
    return np.clip(h, 0, 1)

def lbfgs_opt(h, T, n, maxiter=2000):
    def f(x):
        return soft_max(x, T) * (2.0/n)
    def g(x):
        return soft_max_grad(x, T)

        return f(x) + 1e4 * (x.sum() - n/2.0)**2

        return g(x) + 2e4 * (x.sum() - n/2.0) * np.ones(n)
    res = minimize(f_pen, h, jac=g_pen, method='L-BFGS-B',
```

where μ is the Lagrange multiplier and ν is the penalty parameter. At each outer iteration, I solve the inner L-BFGS-B problem (with the box constraints), then update μ.

1. Start with a feasible h (sum = n/2)
2. At each step, compute the gradient of the soft-max objective
3. Subtract the mean of the gradient (to enforce the sum constraint in the gradient direction)
4. Take a step, project onto [0,1]
5. Adjust to maintain sum = n/2

This is essentially projected gradient descent with the sum constraint handled by centering the gradient.

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.exp(np.clip(T*(corr - m), -500, 500))
    return m + np.log(s.sum()) / T

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(np.clip(T*(corr - m), -500, 500))
    p = e / e.sum()
    # Vectorized gradient
    # For each lag l = k-(n-1), the contribution to h[j] is:
    #   p[k] * (1-h[j-l]) for j in [max(n-l-1,0), min(n-1, n+l-1)] (when l>=0: j in [l, n-1])
    #   p[k] * (-h[j+l]) for j in [max(0,-l), min(n-1, n-1-l)] (when l>=0: j in [0, n-1-l])
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-15: continue
        lag = k - (n-1)
        if lag >= 0:
            L = n - lag
            grad[:L] += p[k] * (-h[lag:])
            grad[lag:] += p[k] * (1.0 - h[:L])
        else:
            nl = -lag
            L = n - nl
            grad[nl:] += p[k] * (1.0 - h[:L])
            grad[:L] += p[k] * (-h[nl:])
    return grad * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    d = tgt - h.sum()
    if abs(d) < 1e-14: return h
    if d > 0:
        r = (1.0 - h).sum()
        if r > 0: h += (1.0 - h) * (d / r)
    else:
        s = h.sum()
        if s > 0: h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def pgd(h, n, T, t_end, lr0=0.005):
    """Projected gradient descent with centered gradient."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr0
    for it in range(20000):
        if time.time() > t_end: break
        g = soft_max_grad(best_h, T)
        g -= g.mean()  # maintain sum constraint
        step = lr
        for _ in range(30):
            trial = proj(best_h - step * g, n)
            c5 = compute_c5(trial)
            if c5 < best_c5:
                best_h, best_c5 = trial, c5
                lr = min(step * 1.05, 0.05)
                break
            step *= 0.5
        else:
            T = min(T * 1.5, 50000)
            lr = max(lr * 0.5, 1e-10)
            if lr < 1e-12: break
    return best_h, best_c5

def coord_pair(h, n, t_end, max_sweeps=100):
    """Optimize pairs of adjacent coordinates."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end: break
        improved = False
        for i in range(n-1):
            if time.time() > t_end: return best_h, best_c5
            a, b = best_h[i], best_h[i+1]
            # Optimize (h[i], h[i+1]) jointly
            def f2(ab):
                x = best_h.copy()
                x[i] = ab[0]
                x[i+1] = ab[1]
                x = proj(x, n)
                return compute_c5(x)
            # Use 2D bounded minimization
            def f1d(t):
                # Parameterize: h[i] = a + t1, h[i+1] = b + t2
                # with constraint a+t1 + b+t2 = a+b (maintain sum locally)
                # So t2 = -t1
                # h[i] = a + t, h[i+1] = b - t
                x = best_h.copy()
                x[i] = np.clip(a + t, 0, 1)
                x[i+1] = np.clip(b - t, 0, 1)
                x = proj(x, n)
                return compute_c5(x)
            res = minimize_scalar(f1d, bounds=(-min(a, 1-b), min(a, 1-b)), 
                                  method='bounded', options={'xatol': 1e-10})
            if res.fun < best_c5 - 1e-15:
                t = res.x
                best_h[i] = np.clip(a + t, 0, 1)
                best_h[i+1] = np.clip(b - t, 0, 1)
                best_h = proj(best_h, n)
                best_c5 = res.fun
                improved = True
        if not improved: break
    return best_h, best_c5
```

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    m = np.max(corr)
    s = np.exp(np.clip(T * (corr - m), -500, 500))
    return m + np.log(s.sum()) / T

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

            L = n - lag
            if L > 0:

                grad[lag:] += p[k] * (1.0 - h[:L])

```python
def compute_c5(h):

def sm_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(np.clip(T * (corr - m), -500, 500))
    p = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        lag = k - (n - 1)

            g[:L] += p[k] * (-h[lag:])
            g[lag:] += p[k] * (1.0 - h[:L])
        else:

            L = n - nl
            g[nl:] += p[k] * (1.0 - h[:L])
            g[:L] += p[k] * (-h[nl:])
    return g * dx
```
