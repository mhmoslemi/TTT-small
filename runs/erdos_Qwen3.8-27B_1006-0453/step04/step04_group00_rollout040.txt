    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > target:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    elif s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * ((target - s) / tot)
    if abs(h.sum() - target) > 1e-10:
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_idx(n):
    m = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    p1, p2 = [], []
    for i in range(n):
        j1 = i - lags
        v1 = (j1 >= 0) & (j1 < n)
        j2 = i + lags
        v2 = (j2 >= 0) & (j2 < n)
        p1.append((v1, j1[v1]))
        p2.append((v2, j2[v2]))
    return p1, p2

def coef_vec(i, h, n, p1, p2, buf):
    buf[:] = 0.0
    v1, j1v = p1[i]
    v2, j2v = p2[i]
    buf[v1] += 1.0 - h[j1v]
    buf[v2] -= h[j2v]
    return buf

def min_bw(corr, coef, ov, tmp):
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        np.multiply(coef, mid - ov, out=tmp)
        tmp += corr
        km = np.argmax(tmp)
        if coef[km] < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) * 0.5

def coord_desc(h, n, t_end, max_sweeps=200000):
    m = 2 * n - 1
    p1, p2 = precompute_idx(n)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    buf = np.zeros(m)
    tmp = np.zeros(m)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        order = np.random.permutation(n)
        for i in range(n):
            ii = order[i]
            ov = h[ii]
            coef_vec(ii, h, n, p1, p2, buf)
            t = min_bw(corr, buf, ov, tmp)
            dt = t - ov
            if abs(dt) > 1e-14:
                corr += dt * buf
                h[ii] = t
                imp += 1
        if imp == 0:
            break
    corr = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(corr) * 2.0 / n)

def softmax_grad(h, n, t_end, T0=200, n_it=300):
    m = 2 * n - 1
    p1, p2 = precompute_idx(n)
    coefs = np.zeros((n, m))
    for i in range(n):
        coefs[i] = coef_vec(i, h, n, p1, p2, coefs[i])
    w = np.zeros(m)
    bh = h.copy()
    bc5 = compute_c5(bh)
    dx = 2.0 / n
    for it in range(n_it):
        if time.time() > t_end:
            break
        corr_vals = np.correlate(bh, 1.0 - bh, mode='full') * dx
        fm = np.max(corr_vals)
        T = T0 * (1.0 - 0.7 * it / n_it)
        lw = T * (corr_vals - fm)
        lw -= np.max(lw)
        w[:] = np.exp(lw)
        w /= w.sum()
        grad = coefs @ w
        lr = (0.008 * (1.0 - 0.8 * it / n_it) + 0.001) * dx
        hn = project_h(bh - lr * grad)
        c5n = compute_c5(hn)
        if c5n < bc5:
            bc5 = c5n
            bh = hn
            for i in range(n):
                coefs[i] = coef_vec(i, bh, n, p1, p2, coefs[i])
    return bh, bc5

def basin_hop(h, n, t_end, n_it=80):
    bh = h.copy()
    bc5 = compute_c5(bh)
    for it in range(n_it):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.95 * (it / n_it)
        m = it % 5
        if m == 0:
            p = (0.02 + 0.1 * cool) * np.random.randn(n)
        elif m == 1:
            f = np.random.randint(1, 25)
            ph = np.random.uniform(0, 6.2832)
            p = (0.04 + 0.08 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif m == 2:
            p = np.zeros(n)
            ns = int(4 + 14 * cool)
            for _ in range(ns):
                a, b = np.random.randint(0, n, 2)
                p[a] += 0.5 * cool
                p[b] -= 0.5 * cool
        elif m == 3:
            sh = np.random.randint(1, n // 3)
            p = 0.25 * cool * (np.roll(bh, sh) - bh)
        else:
            p = np.zeros(n)
            st = np.random.randint(0, n - 10)
            ln = np.random.randint(5, 25)
            p[st:st + ln] = np.random.uniform(-1, 1, ln) * 0.3 * cool
        hp = project_h(bh + p)
        hp, c5p = coord_desc(hp, n, min(time.time() + 2.5, t_end), max_sweeps=30)
        temp = 0.004 * cool + 0.00005
        if c5p < bc5 or (c5p > bc5 and np.random.rand() < np.exp(-(c5p - bc5) / temp)):
            bh, bc5 = hp, c5p
    return bh, bc5

    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    bh, bc5 = coord_desc(bh, n, t0 + budget_s * 0.28, max_sweeps=80000)
    bh, bc5 = basin_hop(bh, n, t0 + budget_s * 0.45, n_it=70)
    bh, bc5 = coord_desc(bh, n, t0 + budget_s * 0.52, max_sweeps=80000)
    bh, bc5 = softmax_grad(bh, n, t0 + budget_s * 0.58, T0=500, n_it
