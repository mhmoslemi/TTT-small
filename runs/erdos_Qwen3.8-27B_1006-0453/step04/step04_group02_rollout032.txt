import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return np.clip(h, 0, 1)
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-12))
        h = np.clip(h, 0.0, 1.0)
        if h.sum() > target + 1e-12:
            h = np.clip(h * (target / h.sum()), 0, 1)
    return h

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = exps.sum()
    p = exps / Z  # softmax weights
    val = (m + np.log(Z) / T) * dx
    
    # Gradient: dx * sum_k p_k * d(corr_k)/dh
    grad = np.zeros(n)
    # Precompute: for each i, sum over k of p_k * (1-h[i-k])*I(k<=i) etc.
    # c[0] (idx n-1): d = 1-2h
    grad += p[n-1] * (1.0 - 2.0 * h)
    # For k=1..n-1:
    for k in range(1, n):
        pk_plus = p[n-1+k] * dx  # weight for lag +k
        pk_minus = p[n-1-k] * dx  # weight for lag -k
        if pk_plus > 1e-14 or pk_minus > 1e-14:
            for i in range(n):
                # c[k] (lag +k): d/dh[i] = (1-h[i-k])*I(k<=i) - h[i+k]*I(i<=n-1-k)
                cp = 0.0
                if i >= k:
                    cp += (1.0 - h[i-k])
                if i <= n-1-k:
                    cp -= h[i+k]
                # c[-k] (lag -k): d/dh[i] = (1-h[i+k])*I(i<=n-1-k) - h[i-k]*I(k<=i)
                cm = 0.0
                if i <= n-1-k:
                    cm += (1.0 - h[i+k])
                if i >= k:
                    cm -= h[i-k]
                grad[i] += pk_plus * cp + pk_minus * cm
    return val, grad
