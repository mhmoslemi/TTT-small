import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def coord_descent(h, n, t_end, n_sweeps=200, top_k=3):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        # Find active lags
        idx = np.argsort(corr)[-top_k:]
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            # Compute how each correlation changes with h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            
            def f1d(s):
                return np.max(corr + s * coef)
            
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-12})
            s_new = res.x
            if abs(s_new) > 1e-13:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

    n = len(h)

            c5_try = compute_c5(h_try)
            if c5_try < best_c5:

    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def subgradient_step(h, target, lr, t_end, n_iter=500):
    """Subgradient method with Nesterov momentum."""
    n = len(h)
    x = h.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_c5 = compute_c5(x)
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = compute_corr(x)
        # Find top lags
        top = np.argsort(corr)[-3:]
        # Average gradient over top lags
        grad = np.zeros(n)
        for lag_idx in top:
            m = lag_idx - (n - 1)  # actual lag
            for i in range(n):
                c = 0.0
                j1 = i - m
                j2 = i + m
                if 0 <= j1 < n:
                    c += (1.0 - x[j1])
                if 0 <= j2 < n:
                    c -= x[j2]
                grad[i] += c
        grad /= len(top)
        
        # Nesterov step
        v = 0.9 * v - lr * grad
        x_new = x + v
        x_new = project_to_sum(x_new, target)
        
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_x, best_c5 = x_new.copy(), c5_new
        
        x = x_new
        # Decay learning rate
        lr *= 0.999
    
    return best_x, best_c5

    n = len(h)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    with np.errstate(over='ignore'):
        return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

    """Find delta in [lo, hi] that minimizes max(corr + delta*coef)."""
    n_c = len(corr)
    # Find the delta where the max is minimized
    # The upper envelope of lines corr[m] + delta*coef[m]
    # changes at breakpoints
    
    # Simple approach: ternary search on the piecewise-linear function
    # (it's convex in delta since it's max of linear functions)
    
    # Actually, max of linear functions is convex, so we can use
    # the fact that the minimizer is where the subgradient contains 0
    
    # subgradient at delta: {mean of coef[m] for m in argmax(corr + delta*coef)}
    
    # Binary search on subgradient
    lo_s = lo
    hi_s = hi
    
    def subgrad_contains_zero(d):
        vals = corr + d * coef
        mx = np.max(vals)
        active = np.where(np.abs(vals - mx) < 1e-10)[0]
        min_sg = np.min(coef[active])
        max_sg = np.max(coef[active])
        return min_sg <= 0 <= max_sg
    
    # If subgradient changes sign, binary search
    # Otherwise, optimum is at boundary
    vals_lo = corr + lo * coef
    vals_hi = corr + hi * coef
    
    # Check if optimum is at lo
    if np.min(coef[np.where(np.abs(vals_lo - np.max(vals_lo)) < 1e-10)[0]]) > 0:
        return lo
    # Check if optimum is at hi
    if np.max(coef[np.where(np.abs(vals_hi - np.max(vals_hi)) < 1e-10)[0]]) < 0:
        return hi

    for _ in range(50):

        vals = corr + mid * coef
        mx = np.max(vals)
        active = np.where(np.abs(vals - mx) < 1e-10)[0]
        if np.min(coef[active]) <= 0:
            hi_s = mid
        else:
            lo_s = mid
    
    return (lo_s + hi_s) / 2

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
