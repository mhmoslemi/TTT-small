import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_single_lag(h, N, k):
    """Gradient of corr[k] w.r.t. h"""
    d = k - (N - 1)
    grad = np.zeros(N)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0 and d < N:
        grad[:N-d] = (1.0 - h[d:])
        grad[d:] = -h[:N-d]
    elif d < 0 and -d < N:
        m = -d
        grad[m:] = (1.0 - h[:N-m])
        grad[:N-m] = -h[m:]
    return grad

def grad_top_k(h, N, corr, k_count=10):
    """Weighted gradient from top-k lags"""
    top = np.argsort(corr)[-k_count:]
    weights = corr[top] - corr[top]
    # Use softmax weights
    cs = corr[top] / 0.5
    cs -= cs.max()
    ws = np.exp(cs)
    ws /= ws.sum()
    
    grad = np.zeros(N)
    for i, k in enumerate(top):
        grad += ws[i] * grad_single_lag(h, N, k)
    return grad

    """LSE-based Adam optimization"""
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_peak = float(compute_corr(h).max())
    
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    t_start = time.time()
    
    while time.time() < deadline:
        elapsed_frac = min(1.0, (time.time() - t_start) / max(1, deadline - t_start))
        tau = tau0 * (tau1/tau0) ** elapsed_frac
        
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        # LSE gradient
        cs = corr / tau
        cs -= cs.max()
        ec = np.exp(cs)
        w = ec / ec.sum()
        
        grad = np.zeros(N)
        center = N - 1
        for k in range(2*N-1):
            wk = w[k]
            if wk < 1e-7:
                continue
            d = k - center
            if d == 0:
                grad += wk * (1.0 - 2.0*h)
            elif d > 0 and d < N:
                grad[:N-d] += wk * (1.0 - h[d:])
                grad[d:] -= wk * h[:N-d]
            elif d < 0 and -d < N:
                m2 = -d
                grad[m2:] += wk * (1.0 - h[:N-m2])
                grad[:N-m2] -= wk * h[m2:]
        
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1-b1**it)
        vh = v / (1-b2**it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        
        if it % 200 == 0:
            lr *= 0.97
    
    return best_h, best_peak * (2.0/N)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lse_grad(h, N, tau, corr=None):
    if corr is None:
        corr = compute_corr(h)
    cs = corr / tau
    cs -= cs.max()
    ec = np.exp(cs)
    w = ec / ec.sum()
    grad = np.zeros(N)
    center = N - 1
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-7:
            continue
        d = k - center
        if d == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < d < N:
            grad[:N - d] += wk * (1.0 - h[d:])
            grad[d:] -= wk * h[:N - d]
        elif 0 < -d < N:
            m2 = -d
            grad[m2:] += wk * (1.0 - h[:N - m2])
            grad[:N - m2] -= wk * h[m2:]
    return grad

def lag_grad(h, N, k):
    d = k - (N - 1)
    grad = np.zeros(N)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif 0 < d < N:
        grad[:N - d] = 1.0 - h[d:]
        grad[d:] = -h[:N - d]
    elif 0 < -d < N:
        m2 = -d
        grad[m2:] = 1.0 - h[:N - m2]
        grad[:N - m2] = -h[m2:]
    return grad

def binary_ls(h, N, deadline, rng):
    h = h.astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    corr = compute_corr(h)
    best_peak = float(corr.max())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 15000 < 1500 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                corr = compute_corr(h)
                npk = float(corr.max())
                if npk < best_peak:
                    best_peak = npk
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[i], h[j] = h[j], h[i]
            else:
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2] = h[j2], h[i2]
                corr = compute_corr(h)
                if float(corr.max()) < best_peak:
                    best_peak = float(corr.max())
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[i1], h[j1] = h[j1], h[i1]
                    h[i2], h[j2] = h[j2], h[i2]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            corr = compute_corr(h)
            npk = float(corr.max())
            if npk < best_peak:
                best_peak = npk
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = h[j], h[i]
    return h, best_peak * (2.0 / N)

def lse_adam(h, N, target, deadline, rng, tau0=10.0, tau1=0.2, lr0=0.015):
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_peak = float(compute_corr(h).max())
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    t_start = time.time()
    total = max(1.0, deadline - t_start)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total)
        tau = tau0 * (tau1 / tau0) ** frac
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        grad = lse_grad(h, N, tau, corr)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if it % 250 == 0:
            lr *= 0.98
    return best_h, best_peak * (2.0 / N)

def peak_penalty(h, N, target, deadline, rng):
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_peak = float(compute_corr(h).max())
    lr = 0.004
    for _ in range(30):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        k_top = int(np.argmax(corr))
        if peak <= 0:
            break
        for _ in range(80):
            if time.time() > deadline:
                break
            grad = lag_grad(h, N, k_top)
            h = project_sum(h - lr * grad, target)
            corr2 = compute_corr(h)
            npk = float(corr2.max())
            if npk < best_peak:
                best_peak = npk
                best_h = h.copy()
            k_new = int(np.argmax(corr2))
            if k_new != k_top:
                k_top = k_new
        lr *= 0.75
    return best_h, best_peak * (2.0 / N)

    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h1, c1 = binary_ls(hb, N, d1, rng)

    if no > N // 2:

        dist = np.abs(best_h[idx] - 0.5)

    elif no < N // 2:
