def compute_overlaps(h, N):
    """Compute overlap[k] for k=-(N-1), ..., N-1"""
    # overlap[k] = #{i: valid, h[i]=1, h[i+k]=0}
    # For k>=0: overlap[k] = sum_{i=0}^{N-k-1} h[i]*(1-h[i+k])
    # For k<0: overlap[k] = sum_{i=|k|}^{N-1} h[i]*(1-h[i+|k|]) [with appropriate indexing]
    ov = np.zeros(2*N-1)
    center = N-1
    for k in range(N):
        if k == 0:
            ov[center] = 0  # for binary
        else:
            # positive lag k
            ov[center+k] = np.sum(h[:N-k] * (1-h[k:]))
            # negative lag -k
            ov[center-k] = np.sum(h[k:] * (1-h[:N-k]))
    return ov
