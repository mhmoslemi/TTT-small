        pi, qj = int(ones[i]), int(zeros[j])
        if pi == qj:
            continue
        h[pi], h[qj] = h[qj], h[pi]
        corr = compute_corr(h)
        new_pk = float(corr.max())
        T = max(0.01, T * 0.999)
        if new_pk < cur_pk or rng.random() < np.exp(-(new_pk - cur_pk) / T):
            cur_pk = new_pk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
        else:
            h[pi], h[qj] = h[qj], h[pi]
    return best, best_pk * (2.0 / N)

def directed_swap(h, N, deadline, rng):
    """Directed local search: try swaps that reduce the peak"""
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    
    corr = compute_corr(h)
    peak = float(corr.max())
    
    for it in range(5000):
        if time.time() > deadline:
            break
        # Find peak lag
        pk_idx = int(np.argmax(corr))
        pk_L = pk_idx - (N - 1)
        
        # Find positions contributing most to this peak
        if pk_L >= 0:
            # corr[N-1+L] = sum_{n=L}^{N-1} h[n]*(1-h[n-L])
            contribs = h[pk_L:] * (1 - h[:N-pk_L])
        else:
            # corr[N-1-L] = sum_{n=0}^{N-1+L} h[n]*(1-h[n-L]) where L<0
            M = -pk_L
            contribs = h[:N-M] * (1 - h[M:])
        
        # Try flipping a contributing 1 to 0, or a non-contributing 0 to 1
        # Positions contributing (h=1 where it pairs with h=0):
        if pk_L >= 0:
            ones_in = np.where((h[pk_L:] == 1) & (h[:N-pk_L] == 0))[0] + pk_L
            zeros_in = np.where((h[pk_L:] == 0) & (h[:N-pk_L] == 1))[0] + pk_L
        else:
            M = -pk_L
            ones_in = np.where((h[:N-M] == 1) & (h[M:] == 0))[0]
            zeros_in = np.where((h[:N-M] == 0) & (h[M:] == 1))[0]
        
        if len(ones_in) == 0:
            break
            
        # Pick a random one from the contributing set
        p = int(ones_in[rng.randint(len(ones_in))])
        zeros_all = np.where(h == 0)[0]
        if len(zeros_all) == 0:
            break
        q = int(zeros_all[rng.randint(len(zeros_all))])
        
        h[p], h[q] = h[q], h[p]
        new_corr = compute_corr(h)
        new_peak = float(new_corr.max())
        if new_peak <= peak:
            peak = new_peak
            corr = new_corr
        else:
            h[p], h[q] = h[q], h[p]
    
    return h, peak * (2.0 / N)
