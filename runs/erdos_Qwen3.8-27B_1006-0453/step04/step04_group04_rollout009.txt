        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - h[:n - nl])
            grad[:n - nl] -= pk * h[nl:]
    return grad * dx

def nesterov_gd(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    lr = 0.005
    gamma = 0.9
    T = 200
    stall = 0
    for it in range(5000):
        if time.time() > t_end:
            break
        T = 200 + it * 2
        g = grad_smooth(best_h, T)
        v = gamma * v - lr * g
        trial = proj(best_h + v, n)
        c5_t = compute_c5(trial)
        if c5_t < best_c5 - 1e-14:
            best_h = trial
            best_c5 = c5_t
            stall = 0
            lr = min(lr * 1.02, 0.03)
        else:
            stall += 1
            lr *= 0.85
            v *= 0.3
            if stall > 100:
                lr = 0.005
                v = np.zeros(n)
                stall = 0
    return best_h, best_c5

def coord_opt(h, n, i):
    h_old = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    k0 = n - 1
    coeffs = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag == 0:
            continue
        c = 0.0
        if i + lag < n:
            c += (1.0 - h[i + lag])
        if i - lag >= 0:
            c -= h[i - lag]
        coeffs[k] = c
    base = corr.copy()
    base0 = base[k0]
    
    def f1d(t):
        val = (t - h_old) * coeffs
        val[k0] = (t - h_old) * (1.0 - t - h_old)
        return float(np.max(base + val))
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = f1d(c), f1d(d)
    for _ in range(40):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a); fc = f1d(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a); fd = f1d(d)
    return 0.5 * (a + b)

def coord_desc(h, n, t_end, nsweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(nsweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_opt(best_h, n, i)
            if abs(t_new - best_h[i]) > 1e-10:
                best_h[i] = t_new
                best_h = proj(best_h, n)
                c5_t = compute_c5(best_h)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def structured_perturb(h, n, rng):
    """Apply a structured perturbation"""
    h_new = h.copy()
    mode = rng.randint(4)
    if mode == 0:
        # Swap two values
        i, j = rng.choice(n, 2, replace=False)
        h_new[i], h_new[j] = h[j], h[i]
    elif mode == 1:
        # Invert a sub-interval
        i = rng.randint(0, n - 5)
        j = min(i + rng.randint(3, 15), n)
        h_new[i:j] = 1.0 - h[i:j]
    elif mode == 2:
        # Circular shift of a sub-interval
        i = rng.randint(0, n - 10)
        j = min(i + rng.randint(5, 20), n)
        shift = rng.randint(1, 4)
        h_new[i:j] = np.roll(h[i:j], shift)
    else:
        # Random noise on a sub-interval
        i = rng.randint(0, max(1, n - 10))
        j = min(i + rng.randint(3, 15), n)
        h_new[i:j] += 0.1 * rng.randn(j - i)
    return proj(h_new, n)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += pk * (1.0 - h[lag:n])
            grad[lag:] -= pk * h[:n - lag]
        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - h[:n - nl])
            grad[:n - nl] -= pk * h[nl:]
    return grad * dx
