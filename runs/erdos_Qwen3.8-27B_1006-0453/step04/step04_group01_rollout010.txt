        return smooth_obj_grad(x, T)[1]
    def ceq(x):
        return x.sum() - n / 2.0
    def ceq_j(x):
        return np.ones(n)
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': ceq, 'jac': ceq_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-start (8% budget)
    t1 = t0 + budget_s * 0.08
    for T in [50, 200, 800, 3000]:
        if time.time() > t1:
            break
        hc = slsqp_solve(best_h.copy(), T, n, 300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Intensive fast CD (42% budget)
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = fast_cd(best_h, n, t2, max_sweeps=20000)

    # Phase 3: Diversification (22% budget)
    t3 = t0 + budget_s * 0.72
    work_h = best_h.copy()
    work_c5 = best_c5
    n_div = 25
    for it in range(n_div):
        if time.time() > t3:
            break
        cool = 1.0 - it / float(n_div)
        if it % 3 == 0:
            h_pert = work_h + (0.02 + 0.07 * cool) * np.random.randn(n)
        elif it % 3 == 1:
            _, g = smooth_obj_grad(work_h, 300)
            h_pert = work_h - (0.003 + 0.015 * cool) * g
        else:
            mask = np.random.random(n) < 0.3
            h_pert = work_h.copy()
            h_pert[mask] = np.clip(h_pert[mask] + np.random.randn(mask.sum()) * 0.15, 0, 1)
        h_pert = project(h_pert, n)
        T_s = 300 + 4000 * cool
        hc = slsqp_solve(h_pert, T_s, n, 200)
        hc, c5v = fast_cd(hc, n, min(t3, time.time() + budget_s * 0.008), max_sweeps=200)
        temp = 0.0015 * cool + 0.0002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Final intensive CD (18% budget)
    t4 = t0 + budget_s * 0.90
    best_h, best_c5 = fast_cd(best_h, n, t4, max_sweeps=20000)

    # Phase 5: Final SLSQP + CD polish (8% budget)
    t5 = t0 + budget_s * 0.95
    for T in [8000, 30000]:
        if time.time() > t5:
            break
        hc = slsqp_solve(best_h.copy(), T, n, 500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    if time.time() < t0 + budget_s * 0.985:
        best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.995, max_sweeps=500)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
