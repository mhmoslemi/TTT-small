def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    m = np.max(corr)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()

        return h

        tot = room.sum()
        if tot < 1e-13:
            return h
