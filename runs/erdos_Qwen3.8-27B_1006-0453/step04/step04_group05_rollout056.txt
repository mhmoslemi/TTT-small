def compute_c5(h, N):

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    target = N / 2.0
    h_bin = make_binary(h0, N)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, N)

    # Phase 1: Binary steepest descent from initial
    h1 = binary_sd(h_bin, N, max_sweeps=400, deadline=t0 + 120)
    c5_1 = compute_c5(h1, N)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Continuous Adam soft-max
    h_cont = best_h.astype(float) + rng.randn(N) * 0.1
    h_cont = project_sum(h_cont, target)
    h2, c5_2 = adam_softmax(h_cont, N, target, t0 + 350, tau0=30.0, tau1=500.0, lr0=0.03)
    h2_bin = make_binary(h2, N)
    h2_sd = binary_sd(h2_bin, N, max_sweeps=300, deadline=t0 + 420)
    c5_2b = compute_c5(h2_sd, N)
    if c5_2b < best_c5:
        best_c5, best_h = c5_2b, h2_sd.copy()
    if c5_2 < best_c5:
        best_c5 = c5_2

    # Phase 3: Multiple perturb-and-refine cycles
    for cycle in range(4):
        if time.time() > t0 + 650:
            break
        n_pert = 3 + rng.randint(4)
        h_pert = perturb_binary(best_h, N, n_pert, rng)
        h_bs = binary_sd(h_pert, N, max_sweeps=250, deadline=t0 + 660)
        c5_b = compute_c5(h_bs, N)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_bs.copy()
        # Continuous polish from this
        if time.time() < t0 + 648:
            h_cp = best_h.astype(float) + rng.randn(N) * 0.06
            h_cp = project_sum(h_cp, target)
            h_c, c5_c = adam_softmax(h_cp, N, target, t0 + 648, tau0=60.0, tau1=800.0, lr0=0.015)
            h_cb = make_binary(h_c, N)
            h_csd = binary_sd(h_cb, N, max_sweeps=200, deadline=t0 + 700 - cycle * 10)
            c5_cd = compute_c5(h_csd, N)
            if c5_cd < best_c5:
                best_c5, best_h = c5_cd, h_csd.copy()

    # Phase 4: Try nearby N values
    for dN in [2, -2, 4, -4, 6]:
        Nt = N + dN
        if Nt < 200 or Nt % 2 != 0:
            continue
        if time.time() > t0 + 950:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_t = make_binary(h_interp, Nt)
        h_ts = binary_sd(h_t, Nt, max_sweeps=200, deadline=t0 + 980)
        c5_t = compute_c5(h_ts, Nt)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_ts.copy()

    # Phase 5: Final Adam polish on absolute best
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hF = best_h.astype(float) + rng.randn(Nf) * 0.04
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = adam_softmax(hF, Nf, Nf / 2.0, deadline - 2, tau0=100.0, tau1=1000.0, lr0=0.008)
        hFb = make_binary(hF, Nf)
        c5Fb = compute_c5(hFb, Nf)
        if c5Fb < best_c5:
            best_c5, best_h = c5Fb, hFb.copy()
        elif c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Final binary SD
    if time.time() < deadline - 1:
        h_bin_final = make_binary(best_h, len(best_h))
        h_final = binary_sd(h_bin_final, len(best_h), max_sweeps=100, deadline=deadline)
        c5_final = compute_c5(h_final, len(best_h))
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
