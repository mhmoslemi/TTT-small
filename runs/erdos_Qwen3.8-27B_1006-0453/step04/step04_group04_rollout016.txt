        full[indices] = x
        return compute_c5(full)
    
    def con_fun(x):
        return x.sum() - (n/2.0 - h0_fixed)
    
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * len(indices)
    h_sub = h[indices].copy()
    res = minimize(obj, h_sub, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_new = h.copy()
    h_new[indices] = np.clip(res.x, 0, 1)
    return h_new
