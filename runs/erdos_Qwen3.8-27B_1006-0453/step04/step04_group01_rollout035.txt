    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

    target = n / 2.0
    vec = np.asarray(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vec) - 1.0
    hi = 1.0 - np.min(vec) + 1.0

        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_smooth(h, n, T, maxiter=500):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(res.x, n)
    except Exception:
        return project(h, n)

def coord_opt_one(h, n, i, t_end):
    """Optimize single coordinate h[i]."""
    def f1d(hi):
        h_try = h.copy()
        h_try[i] = hi
        h_try = project(h_try, n)
        return compute_c5(h_try)
    try:
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-10})
        if res.success:
            h_new = h.copy()
            h_new[i] = res.x
            h_new = project(h_new, n)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h, compute_c5(h)

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_new, c5_new = coord_opt_one(best_h, n, i, t_end)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
                improved = True
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 4:

    return best_h, best_c5

def block_swap_search(h, n, t_end, n_iterations=500000):
    """Non-local search via random swaps and reversals."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.default_rng(12345)
    
    for it in range(n_iterations):
        if time.time() > t_end:
            break
        if it % 500 == 0:
            # Check time more frequently at high iteration counts
            pass
        
        move_type = rng.random()
        h_try = best_h.copy()
        
        if move_type < 0.5:
            # Random pair swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            h_try[i], h_try[j] = h_try[j], h_try[i]
        elif move_type < 0.8:

            a = rng.integers(0, n - 1)
            b = rng.integers(a + 1, n)
            h_try[a:b+1] = h_try[a:b+1][::-1]
        else:

            k = rng.integers(3, 8)
            idx = rng.choice(n, size=k, replace=False)
            vals = h_try[idx].copy()
            rng.shuffle(vals)
            h_try[idx] = vals
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    vec = np.asarray(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vec) - 1.0
    hi = 1.0 - np.min(vec) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -700, 700)))) / T
    return val * dx

        return smooth_obj(x, T)

        return x.sum() - n / 2.0

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(res.x, n)
    except Exception:
        return project(h.copy(), n)

def subgrad_step(h, n, top_k, T_w, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    top_vals = corr[top_idx]
    m_top = np.max(top_vals)
    w = np.exp(np.clip(T_w * (top_vals - m_top), -700, 700))
    w /= w.sum()
    grad = np.zeros(n)
    for idx_i, k_idx in enumerate(top_idx):
        wv = w[idx_i]
        if wv < 1e-14:

        lag = k_idx - (n - 1)
        if lag >= 0:

            if L > 0:

            if lag > 0:
                grad[lag:] += wv * (-h[:L])

            L = n - ll
            if L > 0:
                grad[ll:] += wv * (1.0 - h[:L])
                grad[:L] += wv * (-h[ll:])
    grad *= dx
    h_new = h - alpha * grad
    return project(h_new, n)

        if time.time() > sg_end:
            break
        cool = 1.0 - it / float(n_sg)
        alpha = 0.005 * cool + 0.001
        h_new = subgrad_step(work_h, n, top_k=8, T_w=300.0, alpha=alpha)
        c5_new = compute_c5(h_new)
        if c5_new < work_c5 + 0.001:
            work_h = h_new
            if c5_new < work_c5:
                work_c5 = c5_new
        else:
            h_new = work_h + 0.05 * cool * np.random.randn(n)
            h_new = project(h_new, n)
    # SLSQP polish on subgrad result
    hc = slsqp_smooth(work_h, n, 5000, maxiter=400)
    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    # Phase 5: Second swap search (15%)
    best_h, best_c5 = swap_search(best_h, n, t0 + budget_s * 0.68, seed=314)

    # Phase 6: Coordinate descent (15%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.83, max_sweeps=800)

    # Phase 7: Final SLSQP with high T (7%)
    for T in [20000, 80000]:
        if time.time() - t0 > budget_s * 0.90:
            break
        hc = slsqp_smooth(best_h.copy(), n, T, maxiter=1200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 8: Final swap + CD (10%)
    best_h, best_c5 = swap_search(best_h, n, t0 + budget_s * 0.95, seed=777)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=300)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
