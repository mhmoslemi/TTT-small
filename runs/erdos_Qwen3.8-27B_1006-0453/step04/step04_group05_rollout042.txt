import numpy as np

def compute_corr(h):

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_grad(h, N, beta):

def binary_local_search(h, N, max_sweeps=200, deadline=None, use_2swap=False):
    """Binary steepest descent with 1-swap and optional 2-swap."""
    h = h.astype(float).copy()
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Precompute D: D[p, L] = effect on corr of removing h[p]=1
        # D[p, L] = -(1-h[p+L])*valid - h[p-L])*valid
        idx = np.arange(2*N - 1)
        L_arr = idx - (N - 1)
        p_mat = np.arange(N)[:, None]
        L_mat = L_arr[None, :]
        
        pp = p_mat + L_mat
        pm = p_mat - L_mat
        vp_p = (pp >= 0) & (pp < N)
        vp_m = (pm >= 0) & (pm < N)
        
        h_pp = np.where(vp_p, h[np.clip(pp, 0, N-1)], 0.0)
        h_pm = np.where(vp_m, h[np.clip(pm, 0, N-1)], 0.0)
        
        D = -(1.0 - h_pp) * vp_p.astype(float) - h_pm * vp_m.astype(float)
        # D shape: (N, 2N-1)
        # new_corr after swap p(1->0), q(0->1) = c + D[p,:] - D[q,:]
        # (because D[p,:] is the change from removing, and -D[q,:] is the change from adding)

        best_P = P + 1e-10
        best_swap = None

            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_P:
                best_P = float(new_peaks[mi])
                best_swap = (p, int(zeros[mi]))
        
        # Try 2-swap if available (optional, more expensive)
        if use_2swap and len(ones) >= 2 and len(zeros) >= 2:
            # For 2-swap: remove p1, p2 and add q1, q2
            # new_corr = c + D[p1,:] + D[p2,:] - D[q1,:] - D[q2,:]
            # This is O(N^2 * Z^2) which is too expensive.
            # Instead, try: for the best 1-swap found, see if a 2-swap is better.
            # Or: for each pair of ones (p1, p2), find best pair of zeros (q1, q2).
            # Still too expensive. Skip 2-swap for now.
            pass

        h[best_swap[0]] = 0.0
        h[best_swap[1]] = 1.0

def continuous_opt(h, N, deadline, beta=100.0, lr0=0.05, k_batch=50):
    """Adam optimization with soft-max gradient."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    lr = lr0
    it = 0
    patience = 0
    
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        peak = float(c.max())
        if peak < best_peak - 1e-10:
            best_peak = peak
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Compute soft-max gradient
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()
        
        # Gradient computation (O(N^2))
        grad = np.zeros(N)
        for i in range(N):
            m1 = N - i
            grad[i] += float(np.dot(w[N-1:N-1+m1], 1.0 - h[i:i+m1]))
            m2 = i + 1
            grad[i] -= float(np.dot(w[N-1-i:N], h[0:i+1]))
        
        # Subtract mean to help with sum constraint
        grad -= grad.mean()
        
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        if it % 100 == 0:
            lr *= 0.98
            if patience > 300:
                break
    
    return best_h, best_peak * dx

def make_binary(h, N):
    """Convert continuous h to binary with exactly N/2 ones."""
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    nh = N // 2
    if n_ones > nh:
        # Remove excess ones (randomized by sorting)
        diff = np.abs(h - 1.0)
        idx = np.where(h_b == 1)[0]
        order = idx[np.argsort(diff[idx])]
        h_b[order[nh:]] = 0
    elif n_ones < nh:
        diff = np.abs(h)
        idx = np.where(h_b == 0)[0]
        order = idx[np.argsort(-diff[idx])]
        h_b[order[:nh - n_ones]] = 1
    return h_b

def downsample(h, N_old, N_new):
    """Downsample from N_old to N_new using averaging."""
    if N_new >= N_old:
        return h
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def upsample(h, N_old, N_new, rng):

    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smax_grad(h, N, w):
    """Compute gradient using Toeplitz matrix approach. w has shape (2N-1)."""
    d1 = np.arange(N)[None, :] - np.arange(N)[:, None]  # m - j
    d2 = np.arange(N)[:, None] - np.arange(N)[None, :]  # j - m
    T1 = w[N-1 + d1]
    T2 = w[N-1 + d2]
    return T1 @ (1.0 - h) - T2 @ h

def bin_search(h, N, max_sweeps, deadline=None):
    """Binary steepest descent."""
    h = h.astype(float).copy()
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Compute D
        idx = np.arange(2*N-1)
        L = idx - (N-1)
        pm = np.arange(N)[:, None]
        Lm = L[None, :]
        pp = pm + Lm
        pmi = pm - Lm
        vp = (pp >= 0) & (pp < N)
        vm = (pmi >= 0) & (pmi < N)
        hpp = np.where(vp, h[np.clip(pp, 0, N-1)], 0.0)
        hpm = np.where(vm, h[np.clip(pmi, 0, N-1)], 0.0)
        D = hpm * vm.astype(float) - (1.0 - hpp) * vp.astype(float)
        # D[p, :] = change in corr when h[p] goes 1->0
        # new_corr = c + D[p,:] - D[q,:]
        best_P = P + 1e-10
        best_swap = None
        for p in ones:
            base = c + D[p, :]
            new_peaks = (base[None, :] - D[zeros, :]).max(axis=1)
            mi = np.argmin(new_peaks)
            if float(new_peaks[mi]) < best_P:
                best_P = float(new_peaks[mi])
                best_swap = (p, int(zeros[mi]))
        if best_swap is None:
            break
        h[best_swap[0]] = 0.0
        h[best_swap[1]] = 1.0
    return h

def cont_opt(h, N, deadline, beta=100.0, lr0=0.05):
    """Adam with soft-max gradient."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    lr = lr0
    it = 0
    no_improve = 0
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        pk = float(c.max())
        if pk < best_pk - 1e-9:
            best_pk = pk
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        w = exp_c / exp_c.sum()
        g = smax_grad(h, N, w)
        g -= g.mean()
        it += 1
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g*g
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 50 == 0:
            lr *= 0.99
            if no_improve > 200:
                break
    return best_h, best_pk * dx

def to_binary(h, N):
    h_b = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h_b.sum())
    if n1 > nh:
        idx = np.where(h_b == 1)[0]
        order = idx[np.argsort(np.abs(h - 1.0)[idx])]
        h_b[order[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h_b == 0)[0]
        order = idx[np.argsort(-np.abs(h)[idx])]
        h_b[order[:nh-n1]] = 1
    return h_b

def interp_h(h, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
