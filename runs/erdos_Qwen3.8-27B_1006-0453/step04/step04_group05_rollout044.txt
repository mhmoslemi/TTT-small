def grad_corr_top(h, N, top_idx, weights):
    """Compute gradient of weighted sum of top correlation entries."""
    N = len(h)
    center = N - 1
    grad = np.zeros(N)
    for w, m in zip(weights, top_idx):
        d = int(m) - center
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            if d < N:
                grad[:N - d] += w * (1.0 - h[d:N])
                grad[d:] -= w * h[0:N - d]
        else:
            e = -d
            if e < N:
                grad[e:] += w * (1.0 - h[0:N - e])
                grad[:N - e] -= w * h[e:N]
    return grad

    k = 100
    for it in range(iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        k_eff = min(k, 2 * N - 1)
        top_idx = np.argpartition(corr, -k_eff)[-k_eff:]
        tv = corr[top_idx]
        w = np.exp(0.8 * (tv - tv.max()))
        w /= w.sum()
        grad = grad_corr_top(h, N, top_idx, w)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        h = h - lr * (m / (1.0 - b1**(it+1))) / (np.sqrt(v / (1.0 - b2**(it+1))) + eps)
        h = project_sum(h, target)
        if it % 1000 == 0:
            lr *= 0.95
    return best_h, best_peak * (2.0 / N)

def resample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    h_new = np.clip(h_new, 0, 1)
    return project_sum(h_new, N_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_N = N0
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Continuous Adam on parent (50% of budget)
    d1 = t0 + budget_s * 0.45
    for lr_try in [0.05, 0.03]:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + (d1 - time.time()) * 0.6)
        h_c, c5_c = adam_continuous(h0, N0, N0 / 2.0, sub_dl, lr0=lr_try, k=100,
                                     decay_every=250, decay=0.96)
        if c5_c < best_c5:
            best_c5, best_h, best_N = c5_c, h_c.copy(), N0

    # Phase 2: Binary local search (25% of budget)
    d2 = t0 + budget_s * 0.70
    if time.time() < d2 - 30:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = binary_refine(h_bin, N0, d2 - 10, max_sweeps=600)
        c5_b = compute_c5(h_bin, N0)
        if c5_b < best_c5:
            best_c5, best_h, best_N = c5_b, h_bin.copy(), N0
        # Un-binarize and polish
        if time.time() < d2 - 5:
            h_pol = best_h.astype(float) + rng.randn(N0) * 0.05
            h_pol = project_sum(h_pol, N0 / 2.0)
            h_p, c5_p = adam_continuous(h_pol, N0, N0 / 2.0, d2 - 1, lr0=0.02, k=80,
                                         decay_every=200, decay=0.97)
            if c5_p < best_c5:
                best_c5, best_h, best_N = c5_p, h_p.copy(), N0

    # Phase 3: Try different N values (15% of budget)
    d3 = t0 + budget_s * 0.85
    for N_try in [N0 + 20, N0 + 60, N0 - 20, N0 + 100]:
        if time.time() > d3 - 60 or N_try < 200 or N_try % 2 != 0:
            if N_try % 2 != 0:
                N_try -= 1
            continue
        if time.time() > d3 - 60:
            break
        h_r = resample(best_h, best_N, N_try)
        sub_dl = min(d3 - 10, time.time() + 40)
        h_c, c5_c = adam_continuous(h_r, N_try, N_try / 2.0, sub_dl, lr0=0.04, k=80,
                                     decay_every=200, decay=0.97)
        c5_c = compute_c5(h_c, N_try)
        if c5_c < best_c5:
            best_c5, best_h, best_N = c5_c, h_c.copy(), N_try
        # Quick binary refinement
        if c5_c >= best_c5 and time.time() < d3 - 15:
            h_bin = (h_c > 0.5).astype(float)
            h_bin = binary_refine(h_bin, N_try, d3 - 5, max_sweeps=200)
            c5_b = compute_c5(h_bin, N_try)
            if c5_b < best_c5:
                best_c5, best_h, best_N = c5_b, h_bin.copy(), N_try

    # Phase 4: Final polish (remaining time)
    if time.time() < deadline - 2:
        hF = best_h.astype(float) + rng.randn(best_N) * 0.03
        hF = project_sum(hF, best_N / 2.0)
        hF, c5F = softmax_warm(hF, best_N, best_N / 2.0, deadline - 1, iters=8000, lr=0.015)
        if c5F < best_c5:
            best_c5, best_h, best_N = c5F, hF.copy(), best_N

    final_c5 = compute_c5(best_h, best_N)
    return best_h, final_c5, best_N
