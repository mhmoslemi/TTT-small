        best_cor = np.correlate(best_h, 1.0 - best_h, mode='full')
        best_c5 = np.max(best_cor) * dx
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            if np.any(v1):
                coeff[v1] += (1.0 - best_h[idx1[v1]])
            if np.any(v2):
                coeff[v2] -= best_h[idx2[v2]]
            coeff[c0_idx] = 0.0
            c0_base = best_cor[c0_idx]
            v0_val = _cd_eval(0.0, old, best_cor, coeff, c0_base)
            v1_val = _cd_eval(1.0, old, best_cor, coeff, c0_base)
            res = minimize_scalar(_cd_eval, args=(old, best_cor, coeff, c0_base),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            best_v, best_f = res.x, res.fun
            if v0_val < best_f:
                best_v, best_f = 0.0, v0_val
            if v1_val < best_f:
                best_v, best_f = 1.0, v1_val
            if best_f * dx < best_c5 - 1e-14:
                delta = best_v - old
                best_h[i] = best_v
                best_c5 = best_f * dx
                best_cor += delta * coeff
                best_cor[c0_idx] = c0_base + best_v * (1.0 - best_v) - old * (1.0 - old)
                improved = True
        if not improved:
            break
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def basin_hop_v2(work_h, work_c5, n, t_end, max_iters=200, rng=None):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - 0.9 * it / max_iters)
        r = rng.random()
        if r < 0.35:
            step = 0.02 + 0.15 * cool
            h_p = work_h + step * rng.standard_normal(n)
        elif r < 0.65:
            blen = int(rng.integers(max(2, n // 20), max(3, n // 5)))
            bstart = int(rng.integers(0, n - blen))
            h_p = work_h.copy()
            h_p[bstart:bstart + blen] = 1.0 - h_p[bstart:bstart + blen]
        else:
            freq = rng.uniform(1.0, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.1 * cool + 0.03
            h_p = work_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        h_p = project(h_p, n)
        T_use = int(50 + 5000 * cool)
        try:
            hc = slsqp_opt(h_p, T_use, n, maxiter=300)
            c5v = compute_c5(hc)
        except Exception:
            continue
        temp = 0.003 * cool + 0.0003
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def make_start(n, rng):
    r = rng.random()
    if r < 0.3:
        h = np.ones(n) * 0.5
        for i in range(n // 2):
            v = max(0.0, min(1.0, 0.5 + 0.3 * (rng.random() - 0.5)))
            h[i] = v
            h[(i + n // 2) % n] = 1.0 - v
    elif r < 0.6:
        h = np.ones(n) * 0.5
        for i in range(0, n, 2):
            h[i] = 1.0
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(2, 6) * np.arange(n) / n)
        h = project(h, n)
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy fast coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.22, max_sweeps=2000)

    # Phase 2: SLSQP with multiple T
    for T in [20, 80, 300, 1200, 5000]:
        if time.time() > t0 + budget_s * 0.33:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=800)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 3: More coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.45, max_sweeps=2000)

    # Phase 4: Basin hopping (main global search)
    best_h, best_c5 = basin_hop_v2(best_h, best_c5, n, t0 + budget_s * 0.70,
                                   max_iters=400, rng=rng)

    # Phase 5: Post-hop refinement
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.80, max_sweeps=1500)

    # Phase 6: Try alternate resolutions
    for n_try in [140, 180, 200, 240]:
        if time.time() > t0 + budget_s * 0.90:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent_fast(h_tr, n_try, t0 + budget_s * 0.915, max_sweeps=300)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 7: Final SLSQP + coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.95, max_sweeps=500)
    if time.time() < t0 + budget_s * 0.97:
        for T in [8000, 30000]:
            if time.time() > t0 + budget_s * 0.975:
                break
            try:
                hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    # Final polish coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.995, max_sweeps=200)
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
