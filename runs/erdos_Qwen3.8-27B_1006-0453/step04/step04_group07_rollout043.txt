```
def binary_sa(h, n, t_end, ...):
    # h is binary, sum = n/2
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    corr = compute_corr(h)  # full correlation
    best_c5 = np.max(corr) * (2.0/n)
    
    for iteration in range(max_iters):
        if time.time() > t_end: break
        # Pick random i from ones, j from zeros
        i = random choice from ones
        j = random choice from zeros
        # Compute delta
        delta = compute_delta(h, i, j)  # O(n)
        new_max = max(corr + delta)  # O(n)
        new_c5 = new_max * (2.0/n)
        # Accept/reject
        T = temperature based on iteration
        if new_c5 < current_c5 or random() < exp((current_c5 - new_c5)/T):
            # Apply swap
            h[i], h[j] = h[j], h[i] = 0, 1
            corr += delta
            update ones, zeros sets
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    return best_h, best_c5
```

```python
    ks = np.arange(-(n-1), n)
    g = np.zeros((n, 2*n-1))
```

```python
        idx1 = i + ks
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - ks

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, n, maxiter=500):
    dx = 2.0 / n

        c = np.correlate(x, 1.0 - x, mode='full')
```

```python
def compute_g(h, n):
    ks = np.arange(-(n - 1), n)
    g = np.zeros((n, 2 * n - 1), dtype=np.float64)
    for i in range(n):
        idx1 = i + ks

    offset = n - 1

    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    h = h.copy()
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def binary_hill_climb(h, n, t_end, max_iters=500):
    dx = 2.0 / n
    corr = compute_corr_full(h.copy())
    cur_max = np.max(corr)
    best_h = h.copy()
    best_c5 = cur_max * dx
    g = compute_g(h, n)
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_imp = 0.0
        bi, bj = -1, -1
        for i in ones:
            nc = corr[None, :] + g[i][None, :] - g[zeros]
            nm = np.max(nc, axis=1)
            bm = np.argmin(nm)
            if nm[bm] < cur_max - best_imp:
                best_imp = cur_max - nm[bm]
                bi, bj = int(i), int(zeros[bm])
        if bi < 0 or best_imp < 1e-12:
            break
        h[bi], h[bj] = 0.0, 1.0
        corr = corr + g[bi] - g[bj]
        update_g_after_swap(g, bi, bj, n)
        cur_max = np.max(corr)
        if cur_max * dx < best_c5:
            best_c5 = cur_max * dx
            best_h = h.copy()
    return best_h, best_c5

def binary_sa(h, n, t_end, max_iters=500000, seed_offset=0):
    rng = np.random.RandomState(42 + seed_offset)
    dx = 2.0 / n
    corr = compute_corr_full(h.copy())
    cur_max = np.max(corr)
    cur_c5 = cur_max * dx
    best_h = h.copy()
    best_c5 = cur_c5
    g = compute_g(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    for it in range(max_iters):
        if time.time() > t_end:
            break
        temp = 0.3 * (1.0 - it / max_iters) + 0.03
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = int(ones[ii])
        j = int(zeros[jj])
        d = g[i] - g[j]
        nm = np.max(corr + d)
        nc5 = nm * dx
        if nc5 < cur_c5 or rng.rand() < np.exp(-(nc5 - cur_c5) / (temp * dx)):
            h[i], h[j] = 0.0, 1.0
            corr = corr + d
            update_g_after_swap(g, i, j, n)
            ones[ii] = j
            zeros[jj] = i
            cur_max = nm
            cur_c5 = nc5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def slsqp_refine(h0, n, maxiter=400):
    dx = 2.0 / n
    T_arr = [50.0, 200.0, 1000.0, 5000.0]
    h = h0.copy()
    for T in T_arr:
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(c)
            return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
        def con(x):
            return x.sum() - n / 2.0
        try:
            res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': con}],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h = project(np.clip(res.x, 0, 1), n)
        except Exception:
            pass
    return h

def interp_to(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xw = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(xw, xo, h), n_new)

def kronecker_binary(n):
    k = n // 2
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    pos = np.floor(np.arange(k) * phi * (n / k)).astype(int) % n
    h = np.zeros(n)
    h[np.unique(pos)] = 1.0
    # fix count
    cnt = h.sum()
    if cnt != k:
        idx = np.argsort(-h)
        h = np.zeros(n)
        h[idx[:k]] = 1.0
    return h
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_g(h, n):
    ks = np.arange(-(n - 1), n)
    g = np.zeros((n, 2 * n - 1), dtype=np.float64)
    for i in range(n):
        idx1 = i + ks
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - ks
        v2 = (idx2 >= 0) & (idx2 < n)
        g[i, v1] += h[idx1[v1]]
        g[i, v2] += h[idx2[v2]]
    return g

def update_g_swap(g, i, j, n):
    off = n - 1
    lim = 2 * n - 1
    for m in range(n):
        k = i - m
        if 0 <= k + off < lim:
            g[m, k + off] -= 1.0
        k = m - i
        if 0 <= k + off < lim:
            g[m, k + off] -= 1.0
        k = j - m
        if 0 <= k + off < lim:
            g[m, k + off] += 1.0
        k = m - j
        if 0 <= k + off < lim:
            g[m, k + off] += 1.0

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s < 1e-14:
            return h
        h = h + room * (diff / s)
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
```
