def subgrad_step(h, n, lr, top_m=3):
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    # Find top_m lags
    idx = np.argpartition(-corr, top_m)[:top_m]
    g = np.zeros(n)
    for c in idx:
        k = c - (n-1)
        # dg/dh[j] = (1-h[j+k])*I(j+k∈[0,n)) - h[j-k]*I(j-k∈[0,n), k≠0)
        # For k=0: dg/dh[j] = 1-2h[j]
        if k == 0:
            g += (1.0 - 2.0*h)
        else:
            # Term 1: j+k in [0,n)
            j_lo = max(0, -k)
            j_hi = min(n-1, n-1-k)
            g[j_lo:j_hi+1] += (1.0 - h[j_lo+k:j_hi+1+k])
            # Term 2: j-k in [0,n)
            j_lo2 = max(0, k)
            j_hi2 = min(n-1, n-1+k)
            g[j_lo2:j_hi2+1] -= h[j_lo2-k:j_hi2+1-k]
    g *= dx / top_m
    return h - lr * g

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        return np.clip(h + room * (diff / tot), 0.0, 1.0)
    h = np.clip(h + diff / n, 0.0, 1.0)
    s = h.sum()
    if abs(target - s) > 1e-12 and s > 1e-12:
        h = np.clip(h * (target / s), 0.0, 1.0)
    return h

def _cd_obj(corr, A, c0, old, v):
    d = v - old
    cn = corr + d * A
    cn[c0] = corr[c0] + d * (1.0 - v - old)
    return float(cn.max())

def cd_opt(h, n, t_end, max_sweeps=3000):
    dx = 2.0 / n
    bh = h.copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full').copy()
    best_c5 = float(corr.max()) * dx
    c0 = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = bh[i]
            A = np.zeros(2 * n - 1)
            k_lo = max(-i, -(n - 1))
            k_hi = min(n - 1 - i, n - 1)
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi + 1)
                A[n - 1 + ks] += (1.0 - bh[i + ks])
            k_lo2 = max(-(n - 1), i - n + 1)
            k_hi2 = min(i, n - 1)
            if k_lo2 <= k_hi2:
                ks2 = np.arange(k_lo2, k_hi2 + 1)
                m = ks2 != 0
                if m.any():
                    ks2m = ks2[m]
                    A[n - 1 + ks2m] -= bh[i - ks2m]
            best_f = float(corr.max())
            best_v = old
            for vc in (0.0, 1.0):
                fv = _cd_obj(corr, A, c0, old, vc)
                if fv < best_f - 1e-13:
                    best_f = fv
                    best_v = vc
            res = minimize_scalar(lambda v: _cd_obj(corr, A, c0, old, v),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 5e-8})
            if res.fun < best_f - 1e-13:
                best_f = res.fun
                best_v = res.x
            if best_v != old and best_f * dx < best_c5 - 1e-14:
                d = best_v - old
                corr = corr + d * A
                corr[c0] += d * (1.0 - best_v - old) - d * A[c0]
                bh[i] = best_v
                best_c5 = best_f * dx
                improved = True
        if not improved:
            break
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    return bh, float(corr.max()) * dx

def subgrad_descent(h, n, t_end, max_iters=5000):
    dx = 2.0 / n
    bh = h.copy()
    best_c5 = compute_c5(bh)
    best_h = bh.copy()
    lr0 = 0.02
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(bh, 1.0 - bh, mode='full')
        top_m = 3
        idx = np.argpartition(-corr, top_m)[:top_m]
        g = np.zeros(n)
        for c in idx:
            k = c - (n - 1)
            if k == 0:
                g += (1.0 - 2.0 * bh)
            else:
                j_lo = max(0, -k)
                j_hi = min(n - 1, n - 1 - k)
                g[j_lo:j_hi + 1] += (1.0 - bh[j_lo + k:j_hi + 1 + k])
                j_lo2 = max(0, k)
                j_hi2 = min(n - 1, n - 1 + k)
                g[j_lo2:j_hi2 + 1] -= bh[j_lo2 - k:j_hi2 + 1 - k]
        g *= dx / top_m
        lr = lr0 / (1.0 + it * 0.01)
        bh = project(bh - lr * g, n)
        c5v = compute_c5(bh)
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            best_h = bh.copy()
    return best_h, best_c5

def structured_perturb(h, n, rng):
    h = h.copy()
    mode = rng.randint(4)
    if mode == 0:
        # circular shift
        s = rng.randint(1, max(2, n // 10))
        h = np.roll(h, s)
    elif mode == 1:
        # flip a contiguous block
        i = rng.randint(0, n - 5)
        L = rng.randint(3, max(4, n // 5))
        h[i:i + L] = 1.0 - h[i:i + L]
    elif mode == 2:
        # sinusoidal perturbation
        freq = rng.choice([2, 3, 5, 7, 11])
        amp = 0.1 + 0.2 * rng.random()
        x = np.linspace(0, 2, n, endpoint=False)
        h += amp * np.sin(2 * np.pi * freq * x / 2.0 + rng.random() * 2 * np.pi)
    else:
        # swap pairs
        for _ in range(rng.randint(2, 8)):
            i, j = rng.choice(n, 2, replace=False)
            h[i], h[j] = h[j], h[i]
    return project(h, n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:

        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    best_n = n

    def elapsed():
        return time.time() - t0

    # Phase 1: CD
    best_h, best_c5 = cd_opt(best_h, n, t0 + budget_s * 0.20, max_sweeps=4000)

    # Phase 2: Subgradient descent
    sg_h, sg_c5 = subgrad_descent(best_h, n, t0 + budget_s * 0.30, max_iters=6000)
    if sg_c5 < best_c5:
        best_c5, best_h = sg_c5, sg_h
    best_h, best_c5 = cd_opt(best_h, n, t0 + budget_s * 0.38, max_sweeps=3000)

    # Phase 3: Basin hopping with structured perturbations
    work_h, work_c5 = best_h.copy(), best_c5
    max_hops = int(budget_s * 0.18 / 1.2)
    for it in range(max_hops):
        if time.time() > t0 + budget_s * 0.62:
            break
        cool = max(0.1, 1.0 - 0.7 * it / max_hops)
        h_p = structured_perturb(work_h, n, rng)
        h_c, c5v = cd_opt(h_p, n, t0 + budget_s * 0.62, max_sweeps=800)
        temp = 0.003 * cool + 0.0003
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = h_c, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, h_c.copy()
        # Periodic CD on working solution
        work_h, work_c5 = cd_opt(work_h, n, min(time.time() + 2.0, t0 + budget_s * 0.62), max_sweeps=100)

    # Phase 4: Multi-resolution exploration
    for n_try in [80, 100, 120, 200, 240]:
        if elapsed() > budget_s * 0.80:

        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = cd_opt(h_tr, n_try, t0 + budget_s * 0.82, max_sweeps=1500)
        if c5_tr < best_c5:
            best_c5, best_h, best_n = c5_tr, h_tr, n_try
            h_tr, c5_tr2 = subgrad_descent(h_tr, n_try, t0 + budget_s * 0.84, max_iters=2000)
            if c5_tr2 < best_c5:

            h_tr, c5_tr3 = cd_opt(h_tr, n_try, t0 + budget_s * 0.85, max_sweeps=2000)
            if c5_tr3 < best_c5:

    # Phase 5: Intensive final CD

    best_c5 = compute_c5(best_h)
