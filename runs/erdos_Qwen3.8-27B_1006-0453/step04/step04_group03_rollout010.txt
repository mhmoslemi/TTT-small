def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_desc(h, n, t_end, max_sweeps=500, rand_order=False):
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    idx = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if rand_order:
            np.random.shuffle(idx)
        improved = False
        for i in idx:
            if time.time() > t_end:
                break
            old = h[i]
            lo, hi = -old, 1.0 - old
            if hi - lo < 1e-13:
                continue
            # Compute coefficient vector (change in corr when h[i] changes by 1)
            coef = np.zeros(2 * n - 1)
            # h[i] part: +（1-h[j]) for j = 0..n-1, at lag positions n-1-i+j
            coef[n-1-i : 2*n-1-i] += (1.0 - h)
            # (1-h[i]) part: -h[j] for j = n-1..0, at lag positions i + (n-1-j)
            # j goes n-1 to 0, position goes i to i+n-1
            coef[i : n+i] -= h[n-1::-1]
            
            # Minimize max(corr + d*coef) for d in [lo, hi]
            # The function is piecewise linear, convex in d
            # Use ternary search or just sample + golden section
            # For speed, try: 0, lo, hi, and the unconstrained minimum
            # The unconstrained min of max of lines is where the top two lines cross.
            # For speed, just use minimize_scalar with bounded method.
            def f1d(d):
                return np.max(corr + d * coef)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 100})
            d = res.x
            if abs(d) > 1e-11:
                h[i] = old + d
                corr += d * coef
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    improved = True
        if not improved:
            break
    return h, compute_c5(h)

def min_max_1d(corr, coef, lo, hi):
    # Golden section search for min of max(corr + d*coef)
    gr = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(corr + c * coef)
    fd = np.max(corr + d * coef)
    for _ in range(60):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(corr + c * coef)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(corr + d * coef)
    return 0.5 * (a + b)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def golden_minmax(corr, coef, lo, hi, iters=45):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(corr + c * coef)
    fd = np.max(corr + d * coef)
    for _ in range(iters):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = np.max(corr + c * coef)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = np.max(corr + d * coef)
    return 0.5 * (a + b)

def coord_desc(h, n, t_end, max_sweeps=500, randomize=False):
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    idx = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if randomize:
            np.random.shuffle(idx)
        improved = False
        for i in idx:
            if time.time() > t_end:
                break
            old = h[i]
            lo, hi = -old, 1.0 - old
            if hi - lo < 1e-13:
                continue
            coef = np.zeros(2 * n - 1)
            coef[n-1-i : 2*n-i] += (1.0 - h)
            coef[i : n+i] -= h[n-1::-1]
            d = golden_minmax(corr, coef, lo, hi)
            if abs(d) > 1e-10:
                h[i] = old + d
                corr += d * coef
                nc = np.max(corr) * dx
                if nc < best_c5 - 1e-15:
                    best_c5 = nc
                    improved = True
        if not improved:
            break
    return h, compute_c5(h)

def pair_move(h, n, n_pairs, t_end):
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = np.random.choice(n, 2, replace=False)
        for _ in range(6):
            for idx in (i, j):
                old = h[idx]
                lo, hi = -old, 1.0 - old
                if hi - lo < 1e-13:
                    continue
                coef = np.zeros(2 * n - 1)
                coef[n-1-idx : 2*n-idx] += (1.0 - h)
                coef[idx : n+idx] -= h[n-1::-1]
                d = golden_minmax(corr, coef, lo, hi, iters=35)
                if abs(d) > 1e-10:
                    h[idx] = old + d
                    corr += d * coef
        nc = np.max(corr) * dx
        if nc < best_c5 - 1e-14:
            best_c5 = nc
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent (12% budget)
    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = coord_desc(best_h, n, t1, max_sweeps=300)

    # Phase 2: Basin hopping with diverse perturbations (53% budget)
    t2 = t0 + budget_s * 0.65
    work_h = best_h.copy()
    work_c5 = best_c5
    x_grid = np.arange(n) * 2.0 / n
    max_iters = 100
    it = 0
    while time.time() < t2 and it < max_iters:
        it += 1
        cool = max(1.0 - it / max_iters, 0.02)
        r = np.random.rand()
        if r < 0.15:
            step = 0.01 + 0.07 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.30:
            h_p = work_h.copy()
            i, j = np.random.choice(n, 2, replace=False)
            h_p[i], h_p[j] = h_p[j], h_p[i]
        elif r < 0.45:
            h_p = work_h.copy()
            b0 = np.random.randint(0, max(1, n - 12))
            bl = np.random.randint(4, min(18, n - b0))
            step = 0.02 + 0.09 * cool
            h_p[b0:b0+bl] += step * np.random.randn(bl)
        elif r < 0.60:
            k = np.random.randint(1, 8)
            ph = np.random.uniform(0, 2.0 * np.pi)
            amp = 0.01 + 0.06 * cool
            h_p = work_h + amp * np.sin(np.pi * k * x_grid + ph)
        elif r < 0.75:
            alpha = 0.05 + 0.30 * cool
            h_p = (1.0 - alpha) * work_h + alpha * (1.0 - work_h[::-1].copy())
        else:
            alpha = 0.3 + 0.3 * cool
            h_rand = np.clip(0.5 + 0.35 * np.random.randn(n), 0, 1)
            h_p = (1.0 - alpha) * work_h + alpha * h_rand
        h_p = project_to_sum(h_p, n / 2)
        t_sub = time.time() + 2.0
        h_c, c5v = coord_desc(h_p, n, t_sub, max_sweeps=25)
        temp = 0.001 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h, work_c5 = h_c, c5v
            if c5v < best_c5:
                best_h, best_c5 = h_c.copy(), c5v

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 3: Pair moves at the best point (7% budget)
    t3 = t0 + budget_s * 0.72
    best_h, best_c5 = pair_move(best_h, n, 60, t3)

    # Phase 4: Multi-resolution exploration (13% budget)
    t4 = t0 + budget_s * 0.85
    for n_try in [96, 128, 160, 192, 256]:
        if time.time() > t4:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        h_i, c5v = coord_desc(h_i, n_try, time.time() + 3.0, max_sweeps=40)
        if c5v < best_c5:
            best_h, best_c5, n = h_i, c5v, n_try

    # Phase 5: Final intensive polishing (13% budget)
    t_end = t0 + budget_s * 0.97
    best_h, best_c5 = coord_desc(best_h, len(best_h), t_end, max_sweeps=400, randomize=True)
    if time.time() < t_end:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t_end, max_sweeps=300)
    n = len(best_h)
    best_h, best_c5 = pair_move(best_h, n, 30, t0 + budget_s * 0.985)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
