def two_flip_search(h, N, n_tries, rng, deadline):

def anneal(h, N, n_iters, rng, deadline, T0=5.0, T1=0.01):

    nh = N // 2
    best = h.copy()
    best_pk = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        # Random 1-flip
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_pk = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
