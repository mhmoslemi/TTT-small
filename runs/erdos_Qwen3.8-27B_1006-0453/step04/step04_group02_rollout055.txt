def coord_descent(h, n, t_end, max_sweeps=300, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    dx = 2.0/n
    best = h.copy()
    corr = np.correlate(best, 1.0-best, mode='full')
    best_c5 = np.max(corr)*dx
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            d = np.zeros(2*n-1)
            for k in range(1, n):
                v = 0.0
                if 0 <= i-k < n: v += 1.0-best[i-k]
                if 0 <= i+k < n: v -= best[i+k]
                d[n-1+k] = v
                v2 = 0.0
                if 0 <= i+k < n: v2 += 1.0-best[i+k]
                if 0 <= i-k < n: v2 -= best[i-k]
                d[n-1-k] = v2
            nt = solve_1d(corr, d, corr[n-1], old*(1.0-old), old, n-1)
            if abs(nt-old) < 1e-12:
                continue
            best[i] = nt
            best = project(best, n)
            corr = np.correlate(best, 1.0-best, mode='full')
            c5 = np.max(corr)*dx
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                imp += 1
        if imp > 0:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best, best_c5

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0/n
    def _obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = np.max(c)
        return (m + np.logsumexp(T*(c-m)+1e-30))/T * dx
    def _con(x):
        return x.sum() - n*0.5
    res = minimize(_obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':_con}],
                   options={'maxiter':maxiter,'ftol':1e-15})
    return project(np.clip(res.x, 0, 1), n)

    best = h.copy()
    best_c5 = compute_c5(best)

        lr = 0.1

        if time.time() > t_end:
            break
        # Compute smooth objective gradient
        corr = np.correlate(best, 1.0-best, mode='full')
        m = np.max(corr)
        # Smooth gradient (softmax of the top values)
        weights = np.exp(T*(corr-m)+1e-30)
        weights /= weights.sum()
        # Gradient of corr[m] w.r.t. h[i]
        # corr at position p = sum_j h[j]*(1-h[j-(p-(n-1))])
        # d(corr[p])/d(h[i]) = (1-h[i-l]) if i-l valid, where l = p-(n-1)  [h[i] as first factor]
        #                     + h[i+l]*(1) * (-1) if i+l valid  [h[i] in (1-h) factor]
        # = (1-h[i-l]) - h[i+l]
