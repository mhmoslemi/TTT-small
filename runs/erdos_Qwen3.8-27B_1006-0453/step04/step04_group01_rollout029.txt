    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    vec = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)

def compute_delta_base(h, n, i):

    delta = np.zeros(2*n-1)
    h1 = 1.0 - h
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        d = 0.0
        if 0 <= i+k <= n-1:
            d += h1[i+k]
        if 0 <= i-k <= n-1:
            d -= h[i-k]
        delta[k_idx] = d
    return delta

    corr = np.correlate(h, 1.0 - h, mode='full')
    db = compute_delta_base(h, n, i)
    lo_d = -h[i]
    hi_d = 1.0 - h[i]
    
    def obj(d):
        c = corr + d * db
        m = np.max(c)
        val = m + np.log(np.sum(np.exp(T * (c - m)))) / T
        return val * dx
    
    res = minimize_scalar(obj, bounds=(lo_d, hi_d), method='bounded', options={'xatol': 1e-12})
    return h[i] + res.x

    no_improve = 0

            new_val = coord_opt_fast(best_h, n, i, T)

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    target = n / 2.0
    vec = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-12:
        return vec
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def delta_for_coord(h, n, i):
    delta = np.zeros(2 * n - 1)
    h1 = 1.0 - h
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        d = 0.0
        if 0 <= i + k <= n - 1:
            d += h1[i + k]
        if 0 <= i - k <= n - 1:
            d -= h[i - k]
        delta[k_idx] = d
    return delta

def coord_step(h, n, i, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    db = delta_for_coord(h, n, i)
    lo_d, hi_d = -h[i], 1.0 - h[i]
    if lo_d > hi_d:
        return h[i]
    dx = 2.0 / n

    def obj(d):
        c = corr + d * db
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx

    res = minimize_scalar(obj, bounds=(lo_d, hi_d), method='bounded',
                          options={'xatol': 1e-12})
    return h[i] + res.x

def fast_cd(h, n, t_end, T, max_sweeps=10000, patience=3):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_imp = 0
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        np.random.shuffle(order)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            nv = coord_step(best_h, n, i, T)
            if abs(nv - best_h[i]) > 1e-12:
                best_h[i] = nv
                best_h = project(best_h, n)
                c5v = compute_c5(best_h)
                if c5v < best_c5 - 1e-14:
                    best_c5 = c5v
                    improved = True
        no_imp = 0 if improved else no_imp + 1
        if no_imp >= patience:
            break
    return best_h, best_c5

def block_slsqp(h, n, block, T, maxiter=200):
    bsz = len(block)
    def obj(x):
        h2 = h.copy()
        h2[block] = x
        return smooth_c5(h2, T)
    x0 = h[block].copy()
    bnds = [(0.0, 1.0)] * bsz
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h2 = h.copy()
        h2[block] = np.clip(res.x, 0.0, 1.0)
        return project(h2, n)
    except Exception:
        return h.copy()

def slsqp_full(h, n, T, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with moderate T
    t_p1 = t0 + budget_s * 0.10
    for T in [200, 1000, 5000]:
        if elapsed() > t_p1:
            break
        hc = slsqp_full(best_h, n, T, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Fast coordinate descent (40% budget)
    t_p2 = t0 + budget_s * 0.50
    T_schedule = [500, 2000, 10000, 50000]
    chunk = (t_p2 - t0) / len(T_schedule)
    for T in T_schedule:
        t_sub = t0 + chunk
        chunk += (t_p2 - t0) / len(T_schedule)
        best_h, best_c5 = fast_cd(best_h, n, min(t_sub, t_p2), T, max_sweeps=50000, patience=2)

    # Phase 3: Block SLSQP diversification (20% budget)
    t_p3 = t0 + budget_s * 0.70
    n_blk = 0
    while elapsed() < t_p3:
        n_blk += 1
        bsz = int(np.random.choice([2, 3, 4, 5, 6]))
        block = np.random.choice(n, size=bsz, replace=False)
        T_blk = int(np.random.choice([1000, 5000, 20000]))
        hc = block_slsqp(best_h, n, block, T_blk, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
            if n_blk % 10 == 0:
                t_cd = min(t_p3, time.time() + budget_s * 0.008)
                best_h, best_c5 = fast_cd(best_h, n, t_cd, 10000, max_sweeps=2000, patience=2)

    # Phase 4: Perturbation + re-optimize (15% budget)
    t_p4 = t0 + budget_s * 0.85
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 0
    while elapsed() < t_p4:
        n_iter += 1
        frac = n_iter / 200.0
        sigma = 0.02 + 0.08 * (1.0 - min(frac, 1.0))
        if n_iter % 5 == 0:
            idx = np.random.choice(n, size=n // 8, replace=False)
            work_h[idx] = np.random.uniform(0, 1, size=len(idx))
            work_h = project(work_h, n)
        else:
            work_h = project(work_h + sigma * np.random.randn(n), n)
        t_sub = min(t_p4, time.time() + budget_s * 0.012)
        hc, c5v = fast_cd(work_h, n, t_sub, 20000, max_sweeps=3000, patience=2)
        temp = 0.002 * (1.0 - min(frac, 1.0)) + 0.0002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h, work_c5 = hc, c5v
        if work_c5 < best_c5:
            best_c5, best_h = work_c5, work_h.copy()

    # Phase 5: Final intensive CD (10% budget)
    t_p5 = t0 + budget_s * 0.97
    for T in [50000, 200000, 1000000]:
        if elapsed() > t_p5:
            break
        best_h, best_c5 = fast_cd(best_h, n, min(t_p5, time.time() + budget_s * 0.015),
                                  T, max_sweeps=20000, patience=2)

    # Final SLSQP polish
    if elapsed() < budget_s * 0.99:
        hc = slsqp_full(best_h, n, 50000, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
