        h_p = project_to_sum(h_p, n / 2.0)
        h_p, c5v = coord_descent(h_p, time.time() + 1.5, max_sweeps=300, ti=35)
        temp = 0.0015 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = h_p, c5v
            if c5v < best_c5:
                best_h, best_c5 = h_p.copy(), c5v

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 4: Multi-resolution search (12% budget)
    t4_end = t0 + budget_s * 0.87
    for n_try in [128, 160, 192, 256, 288]:
        if n_try == n or time.time() > t4_end:
            continue
        try:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_i = np.interp(x_new, x_old, best_h)
            h_i = project_to_sum(h_i, n_try / 2.0)
            h_i, c5v = coord_descent(h_i, min(time.time() + 6.0, t4_end), max_sweeps=8000, ti=40)
            if c5v < best_c5:
                best_c5, best_h, n = c5v, h_i, n_try
        except Exception:
            pass

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 5: Final intensive coordinate descent (12% budget)
    t5_end = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, t5_end, max_sweeps=100000, ti=45)

    # Final micro-polish with high-T SLSQP
    if time.time() < t0 + budget_s * 0.99:
        try:
            hc = slsqp_polish(best_h, 80000, len(best_h), maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
