    if abs(s - target) < 1e-12:
        return h

        if trial.sum() > target:

        else:

        if k <= i <= n-1:
            slopes[center+k] += (1.0 - h_base[i-k])
        if 0 <= i <= n-1-k:
            slopes[center+k] -= h_base[i+k]
        
        # Lag -k (index center-k): c_-k = sum h[j](1-h[j+k])
        # d/dh[i] = (1-h[i+k])*[i<=n-1-k] - h[i-k]*[k<=i]
        if 0 <= i <= n-1-k:
            slopes[center-k] += (1.0 - h_base[i+k])
        if k <= i <= n-1:
            slopes[center-k] -= h_base[i-k]
    
    # Lag 0 is special: c_0 = sum h[j](1-h[j]), term at i is h[i](1-h[i]) = x - x^2
    # So corr[center] already has h[i]=0 contribution = 0
    # New: corr[center] + x - x^2
    
    def f1d(x):
        vals = corr + slopes * x
        vals[center] += x - x * x  # wait, corr[center] already has 0 contribution from h[i]
        # Actually corr[center] with h[i]=0: the term h[i](1-h[i]) = 0
        # New term: x(1-x) = x - x^2
        # So vals[center] should be corr[center] + x - x^2
        # But I already did vals = corr + slopes*x, and slopes[center] was not set
        # (I only looped k=1..n-1)
        # So vals[center] = corr[center] + 0 = corr[center]
        # Need to add x - x^2
        vals[center] = corr[center] + x - x*x
        return np.max(vals) * dx
    
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    s = exps.sum()
    w = exps / s
    obj = (m + np.log(s) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    sig = w > 1e-10
    idxs = np.where(sig)[0]
    for idx in idxs:
        k = idx - center
        widx = w[idx]
        ak = abs(k)
        if ak == 0:
            grad += widx * (1.0 - 2.0 * h)
        elif ak < n:
            if k > 0:
                grad[k:] += widx * (1.0 - h[:n - k])
                grad[:n - k] -= widx * h[k:]
            else:
                a = ak
                grad[:n - a] += widx * (1.0 - h[a:])
                grad[a:] -= widx * h[:n - a]
    grad *= dx
    return obj, grad

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        trial = np.clip(h - mid, 0.0, 1.0)
        if trial.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def nesterov_opt(h, t_end, T_init=30.0, T_mult=1.003, lr_init=0.05, beta_init=0.85):
    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    T = T_init
    lr = lr_init
    beta = beta_init
    no_improve = 0
    while time.time() < t_end:
        h_look = h_curr + beta * (h_curr - h_prev)
        h_look = project_sum(h_look, n)
        obj_s, grad = smooth_obj_grad(h_look, T)
        T = min(T * T_mult, 50000.0)
        h_new = h_curr - lr * grad
        h_new = project_sum(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.995
        h_prev = h_curr
        h_curr = h_new
        if no_improve > 200:
            break
    return best_h, best_c5

def coord_opt_fast(h, n, i):
    dx = 2.0 / n
    center = n - 1
    h_base = h.copy()
    h_base[i] = 0.0
    corr = np.correlate(h_base, 1.0 - h_base, mode='full')
    slopes = np.zeros(2 * n - 1)
    for k in range(1, n):
        if k <= i <= n - 1:
            slopes[center + k] += (1.0 - h_base[i - k])
        if 0 <= i <= n - 1 - k:
            slopes[center + k] -= h_base[i + k]
        if 0 <= i <= n - 1 - k:
            slopes[center - k] += (1.0 - h_base[i + k])
        if k <= i <= n - 1:
            slopes[center - k] -= h_base[i - k]
    c0 = corr[center]
    def f1d(x):
        vals = corr + slopes * x
        vals[center] = c0 + x - x * x
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
    return res.x

def coord_descent_fast(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_fast(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_sum(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, maxiter=800):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def obj_grad(x):
        _, g = smooth_obj_grad(x, T)
        return g
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   jac=obj_grad, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(np.clip(res.x, 0, 1), n)
