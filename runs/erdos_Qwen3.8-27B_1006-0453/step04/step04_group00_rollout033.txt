            break

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    corr = compute_corr(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m))) + 1e-300) / T

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    p = np.exp(T * (corr - m))
    s = p.sum()
    if s < 1e-300:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = p / s
    i_arr = np.arange(n)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        delta = k - n + 1
        c = np.ones(n)
        idx1 = i_arr + delta
        m1 = (idx1 >= 0) & (idx1 < n)
        if np.any(m1):
            c[m1] -= h[idx1[m1]]
        idx2 = i_arr - delta
        m2 = (idx2 >= 0) & (idx2 < n)
        if np.any(m2):
            c[m2] -= h[idx2[m2]]
        grad += pk * c
    return grad * dx

def lbfgs_opt(h, T, maxiter=400):
    n = len(h)
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(T,),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 3000})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def fast_coord_descent(h, t_end, max_sweeps=200):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        full_corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                delta = k - n + 1
                c = 1.0
                if 0 <= i + delta < n:
                    c -= best_h[i + delta]
                if 0 <= i - delta < n:
                    c -= best_h[i - delta]
                coefs[k] = c * dx
            corr_base = full_corr - best_h[i] * coefs
            def f1d(hi_val, cb=corr_base, cf=coefs):
                return np.max(cb + hi_val * cf)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                full_corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
                improved = True
        if not improved:
            break
    return best_h, best_c5

def swap_local_search(h, t_end, max_iters=50000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5
