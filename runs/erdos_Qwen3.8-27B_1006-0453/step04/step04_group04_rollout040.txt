def fast_coord_opt(h, n, i):
    """Optimize coordinate i, returning the optimal value."""
    dx = 2.0 / n
    a = h[i]
    
    # Compute corr and the coefficient vector d for coordinate i
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    
    # d[k] = change in corr[k] per unit change in h[i]
    d = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        if lag >= 0:
            if i + lag < n:
                d[k] += (1.0 - h[i+lag])  # h[i]*(1-h[i+lag]) → (1-h[i+lag])
            if i >= lag and lag > 0:
                d[k] -= h[i-lag]  # h[i-lag]*(1-h[i]) → -h[i-lag]
            # Special case lag=0: h[i]*(1-h[i]), derivative w.r.t h[i] is 1-2*h[i]
            if lag == 0:
                d[k] = 1.0 - 2.0*h[i]  # d/dt [t*(1-t)] = 1-2t
        else:
            nl = -lag
            if i >= nl:
                d[k] += (1.0 - h[i-nl])
            if i + nl < n and nl > 0:
                d[k] -= h[i+nl]
    
    # Now minimize max_k (corr[k] - a*d[k] + t*d[k]) over t in [0,1]
    # = max_k (corr[k] + (t-a)*d[k]) + a*... 
    # Wait, let me redo.

    # Actually: corr[k] includes the contribution of h[i]=a.

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        scale = target / s
        h *= scale
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            h *= target / s if s > 0 else 1.0
        else:
            h += room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def slsqp_step(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return soft_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return proj(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coeff_for_i(h, n, i):
    c = np.zeros(2 * n - 1)
    for ell in range(n):
        k = n - 1 + ell
        if ell == 0:
            c[k] = 1.0 - 2.0 * h[i]
        else:
            v = 0.0
            if i + ell < n:
                v += 1.0 - h[i + ell]
            if i >= ell:
                v -= h[i - ell]
            c[k] = v
    for m in range(1, n):
        k = n - 1 - m
        v = 0.0
        if i >= m:
            v += 1.0 - h[i - m]
        if i + m < n:
            v -= h[i + m]
        c[k] = v
    return c

def coord_opt_1d(base, c, lo_bound=0.0, hi_bound=1.0):
    # Minimize max_k (base[k] + t*c[k]) for t in [lo_bound, hi_bound]
    # Binary search on subgradient
    lo, hi = lo_bound, hi_bound
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = base + mid * c
        m = np.max(vals)
        active = vals >= m - 1e-14
        d_act = c[active]
        if np.all(d_act >= -1e-14):
            hi = mid
        elif np.all(d_act <= 1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_sweeps=1000, time_limit=None):
    best_h = h.copy()
    best_c5 = float('inf')
    for sweep in range(max_sweeps):
        if time_limit and time.time() > time_limit:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_c5 = np.max(corr) * (2.0 / n)
        if max_c5 < best_c5:
            best_c5 = max_c5
        improved = False
        order = np.arange(n)
        if sweep % 2 == 1:
            order = order[::-1]
        for i in order:
            if time_limit and time.time() > time_limit:
                return best_h, best_c5
            c = coeff_for_i(best_h, n, i)
            a = best_h[i]
            base = corr - a * c

            if abs(t_opt - a) > 1e-12:

                best_h[i] = t_opt
                improved = True
        # Project at end of sweep
        best_h = proj(best_h, n)
        if not improved:
            break
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

2. If sum > n/2: need to reduce. Find smallest λ such that sum(min(h, 1+λ)) = n/2... 

Actually, the exact projection onto {h : 0≤h≤1, sum(h)=s} is:
- If sum(clip(h,0,1)) ∈ [target, target] (i.e., already feasible): done
- If sum > target: reduce. The amount to reduce from each element is proportional to... actually it's:
  Find τ such that sum(min(1, h + τ)) = target where τ < 0.
  This means: for h[i] + τ < 1, the element becomes h[i]+τ. For h[i]+τ ≥ 1, it stays at 1.
  
  Let's say the elements with h[i] + τ < 1 form set A, and h[i]+τ ≥ 1 form set B.
  sum_A h[i] + |A|*τ + sum_B 1 = target
  τ = (target - sum_B 1 - sum_A h[i]) / |A|
  
  We need to find the right τ such that the set A is consistent.
  
  This is a "water-filling" algorithm. Sort h in decreasing order. Start with τ = 0 (all at h[i]). Decrease τ until sum = target. At each step, the element that hits 0 is removed from set A.

This is O(n log n). For n=180, trivial.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > 1e-14:
            h *= target / s
        else:
            h[:] = target / n
    return np.clip(h, 0.0, 1.0)

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def slsqp_step(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return soft_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return proj(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coeff_for_i(h, n, i):
    c = np.zeros(2 * n - 1)
    for ell in range(n):
        k = n - 1 + ell
        if ell == 0:
            c[k] = 1.0 - 2.0 * h[i]
        else:
            v = 0.0
            if i + ell < n:
                v += 1.0 - h[i + ell]
            if i >= ell:
                v -= h[i - ell]
            c[k] = v
    for m in range(1, n):
        k = n - 1 - m
        v = 0.0
        if i >= m:
            v += 1.0 - h[i - m]
        if i + m < n:
            v -= h[i + m]
        c[k] = v
    return c

def coord_opt_1d(base, c):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = base + mid * c
        m = np.max(vals)
        active = vals >= m - 1e-14
        d_act = c[active]
        if np.all(d_act >= -1e-14):
            hi = mid
        elif np.all(d_act <= 1e-14):
            lo = mid
        else:
            break
        if hi - lo < 1e-15:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_sweeps=5000, time_limit=None, reverse_alternate=True):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        order = np.arange(n) if (sweep % 2 == 0 or not reverse_alternate) else np.arange(n - 1, -1, -1)
        for i in order:
            if time_limit is not None and time.time() > time_limit:
                return best_h, best_c5
            c = coeff_for_i(best_h, n, i)
            a = best_h[i]
            base = corr - a * c
            t_opt = coord_opt_1d(base, c)
            if abs(t_opt - a) > 1e-13:
                delta = t_opt - a
                corr += delta * c
                best_h[i] = t_opt
                improved = True
        s = best_h.sum()
        if abs(s - n / 2.0) > 1e-10:
            best_h = proj(best_h, n)
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c5_curr = np.max(corr) * dx
        if c5_curr < best_c5 - 1e-15:
            best_c5 = c5_curr
        if not improved:
            break
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def pair_swap_opt(h, n, i, j):
    dx = 2.0 / n
    ci = coeff_for_i(h, n, i)
    cj = coeff_for_i(h, n, j)
    d = ci - cj
    a_i = h[i]
    a_j = h[j]
    s = a_i + a_j
    base = np.correlate(h, 1.0 - h, mode='full') - a_i * ci + a_j * cj
    lo_b = max(0.0, s - 1.0)
    hi_b = min(1.0, s)
    if lo_b >= hi_b:
        return a_i, a_j
    lo, hi = lo_b, hi_b
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = base + (mid - a_i) * d
        m = np.max(vals)
        active = vals >= m - 1e-14
        dd = d[active]
        if np.all(dd >= -1e-14):
            hi = mid
        elif np.all(dd <= 1e-14):
            lo = mid
        else:
            break
    t_i = 0.5 * (lo + hi)
    t_j = s - t_i
    return t_i, t_j
