    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, T):
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(sh))) / T

def slsqp_opt(h, n, T, maxiter=600):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr, T) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return proj(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt_1d(h, n, i):
    dx = 2.0 / n
    hi = h[i]
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full').copy()
    coefs = np.zeros(2 * n - 1)
    quad_flag = np.zeros(2 * n - 1)
    for s in range(-(n - 1), n):
        k = s + n - 1
        c = 0.0
        if 0 <= i - s < n and s != 0:
            c += (1.0 - h[i - s])
        if 0 <= i + s < n and s != 0:
            c += (-h[i + s])
        coefs[k] = c
        if s == 0:
            quad_flag[k] = 1.0
    def f1d(t):
        cv = corr_base + (t - hi) * coefs
        k0 = n - 1
        cv[k0] = corr_base[k0] + (t - hi) * (1.0 - t + hi)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return float(res.x)

        else:

        h_interp = proj(h_interp, n_try)
        h_interp, c5_interp = coord_descent(h_interp, n_try,
                                            min(time.time() + 8.0, t0 + budget_s * 0.96), max_sweeps=3000)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp
            n = n_try

    # Phase 7: Final coordinate descent + SLSQP
    if time.time() < t_end - 5:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 10, max_sweeps=5000)
    if time.time() < t_end - 3:
        hc = slsqp_opt(best_h, n, 100000, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
