from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def proj(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5_val, h, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',

    return float(res.x)

        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.72, max_sweeps=600)

    # Phase 5: Try alternate resolutions
    for n_try in [140, 160, 120]:
        if time.time() > t_end * 0.82:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n_try)
        for T in [200, 2000]:
            if time.time() > t_end * 0.84:
                break
            try:
                hc = slsqp_opt(h_interp, T, n_try, maxiter=600, ftol=1e-14)
                c5v = compute_c5(hc)
                hc = quick_coord(hc, n_try, sweeps=3)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n = n_try
            except Exception:
                pass

    # Phase 6: Final coordination descent
    if time.time() < t_end * 0.92:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.92, max_sweeps=400)

    # Phase 7: High-T SLSQP polish
    for T in [10000, 50000]:
        if time.time() > t_end * 0.98:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=1000, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
