def project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:

    if s > target:

        positive = h > 1e-10
        if positive.sum() == 0:

        # Reduce proportionally among positive elements

        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * deficit / tot
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * diff / tot
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= h * (-diff) / tot
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_optimize(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    alpha = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        a_p = ((1.0 - h[im]) if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
        a_m = ((1.0 - h[ip]) if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        alpha[center + k] = a_p
        alpha[center - k] = a_m
    dx = 2.0 / n
    ct = corr_base[center]
    def f1d(x):
        v = ct + x - x * x
        m = np.max(corr_base + x * alpha)
        return max(v, m) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
    return res.x

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

        return smooth_obj(x, T)

        return x.sum() - n / 2.0

    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=50000):

    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_optimize(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        no_improve = no_improve + 1 if not improved else 0
        if no_improve >= 3:
            break
    return best_h, best_c5

def block_swap(h, n, rng):
    b1 = rng.randint(0, n - 8)
    b2 = rng.randint(0, n - 8)
