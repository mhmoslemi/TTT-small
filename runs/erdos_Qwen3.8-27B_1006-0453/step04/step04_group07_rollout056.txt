    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    return np.correlate(h, 1.0 - h, mode='full')

def project_with_sum(h, n):
    """Project h onto [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)

            h = np.clip(h, 0.0, 1.0)

            h *= (1.0 + diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_hot(h, n, t_end, max_sweeps=500, K=12):
    dx = 2.0 / n
    best_h = h.copy()
    corr = full_corr(best_h)
    best_c5 = np.max(corr) * dx
    n_corr = 2 * n - 1
    c0 = n - 1  # index of lag 0
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        # Identify hot lags
        hot = np.argsort(corr)[::-1][:K]
        improved = False
        
        # Randomize order
        order = np.random.permutation(n)
        
        for i in order:
            old = best_h[i]
            if old <= 1e-12 or old >= 1.0 - 1e-12:
                # Check if moving to boundary helps
                pass
            
            # Coefficient for each hot lag
            coeffs = np.zeros(K)
            quad_lag = -1
            for mi, ki in enumerate(hot):
                lag = ki - (n - 1)
                c = 0.0
                j1 = i + lag
                if 0 <= j1 < n:
                    c += 1.0 - best_h[j1]
                j2 = i - lag
                if 0 <= j2 < n:
                    c -= best_h[j2]
                if lag == 0:
                    quad_lag = mi
                coeffs[mi] = c
            
            # Evaluate
            base = corr[hot]
            if quad_lag >= 0:
                base_q = base[quad_lag]
            
            best_f = np.max(base)
            best_v = old
            
            for v in [0.0, 1.0, 0.25, 0.5, 0.75]:
                if v == old:
                    continue
                new_vals = base + (v - old) * coeffs
                if quad_lag >= 0:
                    new_vals[quad_lag] = base[quad_lag] + (v - old) * (1.0 - v - old)
                fv = np.max(new_vals)
                if fv < best_f - 1e-14:
                    best_f, best_v = fv, v
            
            # Fine search around best
            if 0.01 < best_v < 0.99:
                lo = max(0.0, best_v - 0.15)
                hi = min(1.0, best_v + 0.15)
                best_local = best_f
                for _ in range(8):
                    mid = (lo + hi) / 2
                    new_vals = base + (mid - old) * coeffs
                    if quad_lag >= 0:
                        new_vals[quad_lag] = base[quad_lag] + (mid - old) * (1.0 - mid - old)
                    fv = np.max(new_vals)
                    if fv < best_local:
                        best_local = fv
                        best_v = mid
                        hi = mid
                    else:
                        lo = mid
                if best_local < best_f - 1e-14:
                    best_f, best_v = best_local, best_v
            
            if best_f * dx < best_c5 - 1e-14:
                best_h[i] = best_v
                best_c5 = best_f * dx
                improved = True
        
        if improved or sweep % 20 == 0:
            corr = full_corr(best_h)
            actual = np.max(corr) * dx
            if actual < best_c5:
                best_c5 = actual
            elif actual > best_c5 + 1e-10:
                best_c5 = actual
    
    return best_h, best_c5

def proj_grad_nesterov(h0, n, t_end, T_seq=None, max_iter=5000):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    c0 = n - 1
    if T_seq is None:
        T_seq = np.linspace(50, 5000, max_iter)
    
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.5 / (T_seq[0] * dx)  # initial learning rate
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        T = T_seq[min(it, len(T_seq)-1)]
        
        corr = full_corr(h)
        # Softmax weights
        cm = np.max(corr)
        exps = np.exp(T * (corr - cm))
        w = exps / exps.sum()
        
        # Gradient: grad[j] = dx * sum_k w[k] * dcorr[k]/dh[j]
        # Vectorized: for each j, sum over k of w[k] * (1-h[j+k]) - w[k] * h[j-k]
        # This is a weighted correlation!
        # grad[j] = dx * (corr_w_plus[j] - corr_w_minus[j]) + quad correction
        # where corr_w_plus[j] = sum_k w[k] * (1-h[j+k]) for valid j+k
        #       corr_w_minus[j] = sum_k w[k] * h[j-k] for valid j-k
        
        # Actually: dcorr[k]/dh[j] = (1-h[j+k])*I[j+k valid] - h[j-k]*I[j-k valid]
        #          + for k=0: -2h[j]+1 instead of (1-h[j])-h[j] = 1-2h[j] ✓
        
        # grad[j] = dx * sum_{k≠0} w[k] * [(1-h[j+k])I - h[j-k]I] + dx * w[0] * (1-2h[j])
        
        # Let's compute: A[j] = sum_{k} w[k] * (1-h[j+k]) for j+k in [0,n), k>0
        #                 B[j] = sum_{k} w[k] * h[j-k] for j-k in [0,n), k>0
        # grad[j] = dx * (A[j] - B[j] + sum_{k<0} w[k]*(1-h[j+k]) - sum_{k<0} w[k]*h[j-k]) + dx*w[0]*(1-2h[j])
        
        # Hmm this is getting complex. Let me just do it the O(n²) way but with numpy.
        
        # dcorr[k]/dh[j] for all j:
        # For lag k (index ki = k + n - 1):
        #   dcorr/dh[j] = (1-h[j+k]) if 0<=j+k<n, else 0
        #              - h[j-k] if 0<=j-k<n, else 0
        # For k=0: dcorr/dh[j] = 1-2h[j]
        
        grad = np.zeros(n)
        # Non-zero lags
        for ki in range(n_corr):
            if ki == c0:
                continue
            wk = w[ki]
            if wk < 1e-10:
                continue
            lag = ki - (n - 1)
            if lag > 0:
                # (1-h[j+lag]) for j in [0, n-lag)
                grad[:n-lag] += wk * (1.0 - h[lag:])
                # -h[j-lag] for j in [lag, n)
                grad[lag:] -= wk * h[:n-lag]
            else:
                lag_abs = -lag
                # (1-h[j-lag_abs]) for j in [lag_abs, n)
                grad[lag_abs:] += wk * (1.0 - h[:n-lag_abs])
                # -h[j+lag_abs] for j in [0, n-lag_abs)
                grad[:n-lag_abs] -= wk * h[lag_abs:]
        # Zero lag
        grad += w[c0] * (1.0 - 2.0 * h)
        grad *= dx
        
        # Nesterov momentum
        mu = (t - 1) / (t + 1)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        # Gradient at y
        # For simplicity, use gradient at h (not y) - this is "heavy ball" style
        # Or recompute at y for true Nesterov
        grad_y = None  # skip for speed, use grad at h
        
        step = lr * t  # growing step for Nesterov
        h_new = h - step * grad
        h_new = project_h(h_new, n)
        
        h_prev = h
        h = h_new
        t += 1.0
        lr = 0.5 / (T * dx * (1 + 0.01 * it))
        
        # Track best
        if it % 100 == 0 or it == max_iter - 1:
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
        
        # Periodic coordinate descent refinement
        if it % 500 == 0:
            h, c5_cd = coord_descent_hot(h, n, t_end - 5, max_sweeps=20, K=8)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h.copy()
    
    return best_h, best_c5

def perturb(h, n, scale, rng):
    h_p = h + scale * rng.standard_normal(n)
    # Also try random block flips
    if rng.random() < 0.3:
        for _ in range(3):
            start = rng.randint(0, n - 5)
            width = rng.randint(2, 8)
            h_p[start:start+width] = 1.0 - h_p[start:start+width]
    return project_h(h_p, n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_h(np.interp(x_new, x_old, h), n_new)

            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
            else:
                la = -lag
                grad[:n - la] += wk * (1.0 - h[la:])
                grad[la:] -= wk * h[:n - la]
        grad *= dx

        mu = (t - 1.0) / (t + 1.0)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        lr = 1.0 / (max(np.max(np.abs(grad)), 1e-10) * (1.0 + 0.5 * t))
        h_new = y - lr * grad
        h_new = project_h(h_new, n)
        h_prev = h
        h = h_new
        t += 1.0

        if it % 200 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()

        if it % 800 == 0:
            h, c5_cd = coord_descent_hot(h, n, min(time.time() + 3.0, t_end - 2), max_sweeps=30, K=10)
            c5_cd = compute_c5(h)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h.copy()

    return best_h, best_c5

def perturb_h(h, n, scale, rng):
    h_p = h + scale * rng.standard_normal(n)
    r = rng.random()
    if r < 0.3:
        for _ in range(4):
            s = rng.integers(0, max(1, n - 6))
            w = rng.integers(2, 7)
            h_p[s:s + w] = 1.0 - h_p[s:s + w]
    elif r < 0.5:
        idx = rng.integers(0, n, size=5)
        h_p[idx] = rng.random(5)
    return project_h(h_p, n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_h(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Intensive coordinate descent (28%)
    t1 = t0 + budget_s * 0.28
    best_h, best_c5 = coord_descent_hot(best_h, n, t1, max_sweeps=3000, K=15)

    # Phase 2: Projected gradient Nesterov (22%)
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = proj_grad_nesterov(best_h, n, t2, K_grad=25, max_iter=12000)

    # Phase 3: Coord descent after gradient (10%)
    t3 = t0 + budget_s * 0.60
    best_h, best_c5 = coord_descent_hot(best_h, n, t3, max_sweeps=1500, K=12)

    # Phase 4: Multi-start local search (25%)
    t4 = t0 + budget_s * 0.85
    n_restarts = 0
    while time.time() < t4:
        n_restarts += 1
        cool = max(0.1, 1.0 - n_restarts / 80.0)
        scale = 0.03 + 0.12 * cool
        h_p = perturb_h(best_h, n, scale, rng)
        t_inner = min(time.time() + 6.0, t4 - 1.0)
        h_p, c5_p = coord_descent_hot(h_p, n, t_inner, max_sweeps=150, K=10)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 5: Resolution adaptation (7%)
    t5 = t0 + budget_s * 0.92
    for n_try in (128, 200, 192, 240):
        if time.time() > t5:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent_hot(h_tr, n_try, t5 - 1, max_sweeps=200, K=10)
        c5_tr = compute_c5(h_tr)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr
            n = n_try

    # Phase 6: Final polish (8%)
    t6 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent_hot(best_h, n, t6, max_sweeps=800, K=15)
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
