from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

where a[k] is the coefficient we already computed. So we don't need to recompute the full correlation! We just update the relevant entries.

    """Coordinate descent with incremental correlation update."""
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5, corr
            h_save = best_h[i]
            
            # Compute coefficients
            s_arr = np.arange(2 * n - 1) - n + 1
            idx1 = i + s_arr
            idx2 = i - s_arr
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            a = np.zeros(2 * n - 1)
            a[v1] += (1.0 - best_h[np.clip(idx1[v1], 0, n-1)])
            a[v2] += (-best_h[np.clip(idx2[v2], 0, n-1)])
            
            # Remove h[i] from correlation
            corr_no_i = corr - h_save * a
            
            # Find best t
            max_val = np.max(corr_no_i + h_save * a)
            active = np.where(corr_no_i + h_save * a >= max_val - 0.05)[0]
            
            candidates = [0.0, 1.0, h_save]
            alen = len(active)
            if alen <= 40:
                for ii in range(alen):
                    for jj in range(ii+1, alen):
                        k1, k2 = active[ii], active[jj]
                        da = a[k1] - a[k2]
                        if abs(da) < 1e-14:
                            continue
                        t = (corr_no_i[k2] - corr_no_i[k1]) / da
                        if 0.0 <= t <= 1.0:
                            candidates.append(t)

            best_f = np.max(corr_no_i + h_save * a)
            for t in candidates:
                f = np.max(corr_no_i + t * a)
                if f < best_f:
                    best_f = f
                    best_t = t
            
            if abs(best_t - h_save) < 1e-12:
                continue
            
            # Check with projection (sum constraint)
            # For now, just update and check
            best_h[i] = best_t
            corr = corr_no_i + best_t * a  # incremental update
            new_c5 = np.max(corr) * dx
            
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                improved = True
            else:

                best_h[i] = h_save
                corr = corr_no_i + h_save * a

    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp(h, T, n, maxiter=800):
    def obj(x):
        return smooth_obj(x, T, n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h

    idx1 = i + s_arr
    idx2 = i - s_arr
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    a[v1] += (1.0 - h[np.clip(idx1[v1], 0, n - 1)])
    a[v2] += (-h[np.clip(idx2[v2], 0, n - 1)])
    corr0 = corr - hs * a
    cur = corr0 + hs * a
    mx = np.max(cur)

            for jj in range(ii + 1, len(act)):
                k1, k2 = act[ii], act[jj]
                da = a[k1] - a[k2]
                if abs(da) > 1e-14:
                    t = (corr0[k2] - corr0[k1]) / da
                    if 0.0 <= t <= 1.0:
                        cands.append(t)
    best_t, best_f = hs, np.max(corr)
    for t in cands:
        f = np.max(corr0 + t * a)
        if f < best_f:
            best_f, best_t = f, t
    return best_t

    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - n + 1
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')

        if not imp:

    return best_h, best_c5

    hs = h[i]
    i1 = i + s_arr
    i2 = i - s_arr
    v1 = (i1 >= 0) & (i1 < n)
    v2 = (i2 >= 0) & (i2 < n)
    a = np.zeros(2 * n - 1)
    a[v1] += (1.0 - h[np.clip(i1[v1], 0, n - 1)])
    a[v2] += (-h[np.clip(i2[v2], 0, n - 1)])
    c0 = corr - hs * a
    cur = c0 + hs * a
    mx = np.max(cur)
    act = np.where(cur >= mx - 0.08)[0]
    cands = [0.0, 1.0, hs]
    if len(act) <= 50:
        for p in range(len(act)):
            for q in range(p + 1, len(act)):
                k1, k2 = act[p], act[q]
                da = a[k1] - a[k2]
                if abs(da) > 1e-14:
                    t = (c0[k2] - c0[k1]) / da
                    if 0.0 <= t <= 1.0:
                        cands.append(t)
    best_t, best_f = hs, mx
    for t in cands:
        f = np.max(c0 + t * a)
        if f < best_f:
            best_f, best_t = f, t
    return best_t


def cd(h, n, t_end, sweeps=400):
    bh = h.copy()
    bc = compute_c5(bh)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - n + 1
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    for _ in range(sweeps):
        if time.time() > t_end:

                imp = True
            else:
                bh = sv
        if not imp:
            break
    return bh, bc

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:

                break
