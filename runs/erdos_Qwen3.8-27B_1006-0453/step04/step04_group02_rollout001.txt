import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    # Proportional projection
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        s2 = max(s, 1e-14)
        h = h * (target / s2)
        h = np.clip(h, 0.0, 1.0)
        s3 = h.sum()
        diff2 = target - s3
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
    return h

def coord_opt(h, n, i):
    """Optimize coordinate i given all others fixed."""
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        dk2 = 0.0
        if 0 <= i + k < n:
            dk2 += (1.0 - h[i + k])
        if 0 <= i - k < n:
            dk2 += (-h[i - k])
        d[n - 1 - k] = dk2
    
    c0 = corr_base[n - 1]
    # For t in [0,1]:
    # corr_m(t) = corr_base[m] + t * d[m] for m != n-1
    # corr_{n-1}(t) = c0 + t * (1 - t) = c0 + t - t^2
    
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 200})

        _, corr = compute_c5(best_h)

        imp = np.zeros(n)

                    imp[i] += h[i] * (1 - h[i+k])
                    imp[i] += (1 - h[i]) * h[i-k] if 0 <= i-k < n else 0
        # Sort by descending importance
        order = np.argsort(-imp)
        
        for i in order:

            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-11:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5, _ = compute_c5(best_h)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        s2 = max(s, 1e-14)
        h *= (target / s2)
        h = np.clip(h, 0.0, 1.0)
        s3 = h.sum()
        diff2 = target - s3
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * (diff2 / tot)
        elif diff2 < -1e-14:
            h = np.clip(h * (target / max(s3, 1e-14)), 0, 1)
    return h

def opt_coord(h, n, i):
    dx = 2.0 / n
    hs = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hs
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        a = 1.0 - h[i - k] if 0 <= i - k < n else 0.0
        b = -h[i + k] if 0 <= i + k < n else 0.0
        d[n - 1 + k] = a + b
        c = 1.0 - h[i + k] if 0 <= i + k < n else 0.0
        e = -h[i - k] if 0 <= i - k < n else 0.0
        d[n - 1 - k] = c + e
    c0 = cb[n - 1]
    def f1(t):
        v = cb + t * d
        v[n - 1] = c0 + t - t * t
        return np.max(v) * dx
    return minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',

def slsqp(h, T, n, maxit=500):

        def obj(x):
            return smooth_c5(x, T)
        def con(x):
            return x.sum() - n / 2.0
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': maxit, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)

**Key idea: "Pressure equalization" heuristic**

After coordinate descent converges, the max correlation is achieved at some set of lags S = {m : corr[m] ≈ max}. To reduce the max, I need to reduce corr[m] for m ∈ S.

For each m ∈ S, corr[m] = sum_j h[j] * (1-h[j+m-n+1]). To reduce this:
- Reduce h[j] for j where (1-h[j+m-n+1]) is large
- Or increase h[j+m-n+1] for j where h[j] is large

But these adjustments affect other correlation values. The "pressure" approach:
1. Compute a "pressure" field: for each position i, sum the contributions to max correlations
2. Move h in the direction that reduces the max pressure

Specifically:
- For each max lag m*, compute the gradient of corr[m*] with respect to each h[i]
- Sum these gradients over all m* at the max
- Step h down in that direction (and compensate elsewhere to maintain the constraint)

This is essentially a subgradient step for the max function, applied only to the active (max) constraints.

    """Subgradient step targeting max correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    # Active set: correlations within 1% of max
    active = corr > max_c * 0.995
    # Compute subgradient for active constraints
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        shift = m - (n - 1)
        # corr[m] = sum_j h[j] * (1-h[j+shift]) for valid j
        for j in range(n):
            j2 = j + shift
            if 0 <= j2 < n:
                grad[j] += (1 - h[j2])  # d(corr[m])/d(h[j])
                grad[j2] += -h[j]       # d(corr[m])/d(h[j2])
    # Normalize
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h
    # Step in negative gradient direction
    h_new = h - step_size * grad / gn
    # Also add small random perturbation
    h_new += step_size * 0.1 * rng.standard_normal(n)
    return project(h_new, n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        s2 = max(s, 1e-14)
        h *= target / s2
        h = np.clip(h, 0.0, 1.0)
        s3 = h.sum()
        d2 = target - s3
        if d2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * (d2 / tot)
        elif d2 < -1e-14:
            h = np.clip(h * (target / max(s3, 1e-14)), 0, 1)
    return h

def opt_coord(h, n, i):
    dx = 2.0 / n
    hs = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hs
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        a = (1.0 - h[i + k]) if 0 <= i + k < n else 0.0
        b = -h[i - k] if 0 <= i - k < n else 0.0
        d[n - 1 + k] = a + b
        c = (1.0 - h[i - k]) if 0 <= i - k < n else 0.0
        e = -h[i + k] if 0 <= i + k < n else 0.0
        d[n - 1 - k] = c + e
    c0 = cb[n - 1]
    def f1(t):
        v = cb + t * d
        v[n - 1] = c0 + t - t * t
        return np.max(v) * dx
    return minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 150}).x

def opt_pair(h, n, i, j):
    dx = 2.0 / n
    hi0, hj0 = h[i], h[j]
    h[i] = h[j] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hi0
    h[j] = hj0
    def f2(v):
        t1, t2 = v
        h[i], h[j] = t1, t2
        corr = np.correlate(h, 1.0 - h, mode='full')
        h[i], h[j] = hi0, hj0
        return np.max(corr) * dx
    res = minimize(f2, [hi0, hj0], method='L-BFGS-B',
                   bounds=[(0,1),(0,1)], options={'maxiter': 100, 'ftol': 1e-14})
    return res.x

def cd_pass(h, n, t_end, rng, n_sweeps=80, pair_prob=0.05):
    best = h.copy()
    best_c5 = compute_c5(best)
    stall = 0
    for sw in range(n_sweeps):
        if time.time() > t_end:
            break
        changed = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            if rng.random() < pair_prob and i + 1 < n:
                v = opt_pair(best, n, i, i + 1)
                best[i], best[i+1] = v[0], v[1]
            else:
                best[i] = opt_coord(best, n, i)
            best = project(best, n)
            c5 = compute_c5(best)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                changed = True
        if not changed:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0
    return best, best_c5

def slsqp(h, T, n, maxit=500):
    try:
        def obj(x):
            return smooth_c5(x, T)
        def con(x):
            return x.sum() - n / 2.0
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': maxit, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def pressure_perturb(h, n, step, rng):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = np.where(corr > mx - 0.02 * mx)[0]
    grad = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        lo = max(0, -shift)
        hi = min(n, n - shift)
        for j in range(lo, hi):
            j2 = j + shift
            grad[j] += (1.0 - h[j2])
            if j2 != j:
                grad[j2] += -h[j]
    gn = np.linalg.norm(grad)
    if gn > 1e-14:
        h = h - step * grad / gn
    h += step * 0.3 * rng.standard_normal(n)
    return project(h, n)

def interp(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(xn, xo, h), n_new)

def basin_hop(h, n, t_end, rng, n_it=60, step0=0.06, T_sm=500):
    best = h.copy()
    best_c5 = compute_c5(best)
    cur, cur_c5 = best.copy(), best_c5
    for it in range(n_it):
        if time.time() > t_end:
            break
        cool = max(0.15, 1.0 - 0.85 * it / n_it)
        step = step0 * cool
        pert = project(cur + step * rng.standard_normal(n), n)
        pert = pressure_perturb(pert, n, 0.01, rng)
        hc = slsqp(pert, T_sm, n, maxit=200)
        cv = compute_c5(hc)
        temp = 0.003 * cool + 0.0002
        if cv < cur_c5 or rng.random() < np.exp(-(cv - cur_c5) / temp):
            cur, cur_c5 = hc, cv
        if cur_c5 < best_c5:
            best, best_c5 = cur.copy(), cur_c5
    return best, best_c5

        cv = compute_c5(hc)
        if cv < best_c5 - 1e-8:
            best_h, best_c5, n = hc, cv, nt
            hi = interp(best_h, n)
            hc = slsqp(hi, 8000, n, maxit=700)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
            best_h, best_c5 = cd_pass(best_h, n, min(time.time() + budget_s*0.06, t4), rng, n_sweeps=25)

    # Phase 5: Second CD round + final SLSQP
    t5 = t0 + budget_s * 0.95
    best_h, best_c5 = cd_pass(best_h, n, t5, rng, n_sweeps=50, pair_prob=0.04)
    if time.time() < t0 + budget_s * 0.98:
        hc = slsqp(best_h, 20000, n, maxit=800)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 6: Final CD
    best_h, best_c5 = cd_pass(best_h, n, t0 + budget_s * 0.995, rng, n_sweeps=20)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
