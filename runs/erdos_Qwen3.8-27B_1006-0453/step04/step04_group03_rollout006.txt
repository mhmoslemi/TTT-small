def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    # Distribute proportionally to available room
    if diff > 0:
        room = np.maximum(1.0 - h, 1e-10)
        h = h + room * (diff / room.sum())
    else:
        h = np.maximum(h, 1e-10)
        h = h * (1.0 + diff / h.sum())
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def solve_1d_max(corr, coef, lo, hi):
    """Find s in [lo,hi] minimizing max(corr + s*coef)"""
    if hi - lo < 1e-14:

        vals = corr + s * coef

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = np.maximum(1.0 - h, 1e-12)
        h = h + room * (diff / room.sum())
    else:
        h = np.maximum(h, 1e-12)
        h = h * (1.0 + diff / h.sum())
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def solve_1d_convex_plinear(corr, coef, lo, hi):
    """Minimize max(corr + s*coef) over s in [lo,hi] using subgradient bisection."""
    if hi - lo < 1e-14:
        return 0.0
    def subgrad_range(s):
        vals = corr + s * coef
        mx = np.max(vals)
        active = vals >= mx - 1e-9
        return coef[active].min(), coef[active].max()
    _, hi_lo = subgrad_range(lo)
    if hi_lo >= -1e-12:
        return lo
    lo_hi, _ = subgrad_range(hi)
    if lo_hi <= 1e-12:
        return hi
    a, b = lo, hi
    for _ in range(55):
        mid = 0.5 * (a + b)
        sg_lo, sg_hi = subgrad_range(mid)
        if sg_lo <= 1e-10 and sg_hi >= -1e-10:
            return mid
        if sg_hi < 0:
            a = mid
        else:
            b = mid
    return 0.5 * (a + b)

def coord_descent(h, n, t_end, max_sweeps=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    shifts = np.arange(-(n-1), n)
    npairs = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(npairs)
            idx_d = i + shifts
            vd = (idx_d >= 0) & (idx_d < n)
            coef[vd] += (1.0 - best_h[idx_d[vd]])
            idx_i = i - shifts
            vi = (idx_i >= 0) & (idx_i < n)
            coef[vi] += (-best_h[idx_i[vi]])
            s_new = solve_1d_convex_plinear(corr, coef, -h_old, 1.0 - h_old)
            if abs(s_new) > 1e-13:
                best_h[i] += s_new
                corr += s_new * coef
                c5v = np.max(corr) * dx
                if c5v < best_c5:
                    best_c5 = c5v
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def fourier_perturb(h, n, mag):
    hf = np.fft.rfft(h)
    nf = len(hf)
    mask = np.zeros(nf, dtype=bool)
    mask[1:min(nf, 8)] = True
    hf[mask] += mag * (np.random.randn(nf[mask]) + 1j * np.random.randn(nf[mask]))
    h_new = np.fft.irfft(hf, n=n)
    return project_to_sum(h_new, n / 2)

def shift_blend_perturb(h, n, frac):
    sh = int(np.random.randint(-int(frac * n), max(1, int(frac * n))))
    hs = np.roll(h, sh)
    a = np.random.uniform(0.4, 0.9)
    return project_to_sum(a * hs + (1 - a) * h, n / 2)

def sparse_perturb(h, n, mag, density):
    h2 = h.copy()
    n_s = max(2, int(n * density))
    idx = np.random.choice(n, n_s, replace=False)
    h2[idx] += mag * np.random.randn(n_s)
    return project_to_sum(h2, n / 2)

def gaussian_perturb(h, n, mag):
    return project_to_sum(h + mag * np.random.randn(n), n / 2)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:

        best_h[:112] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD polish (12% budget)
    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=800)

    # Phase 2: Basin hopping with diverse perturbations (38% budget)
    t2 = t0 + budget_s * 0.50
    work_h = best_h.copy()
    work_c5 = best_c5
    it_count = 0
    total_iters = max(40, int(budget_s * 0.35 / 5.0))

    while time.time() < t2:
        it_count += 1
        cool = max(1.0 - it_count / float(total_iters), 0.03)
        r = np.random.rand()
        if r < 0.30:
            h_p = fourier_perturb(work_h, n, 0.04 + 0.12 * cool)
        elif r < 0.55:
            h_p = shift_blend_perturb(work_h, n, 0.01 + 0.07 * cool)
        elif r < 0.75:
            h_p = gaussian_perturb(work_h, n, 0.02 + 0.10 * cool)
        elif r < 0.90:
            h_p = sparse_perturb(work_h, n, 0.08 + 0.15 * cool, 0.15)
        else:
            # Double Fourier: perturb more freqs
            h_f = np.fft.rfft(work_h)
            nf = len(h_f)
            h_f[1:min(nf, 15)] += 0.06 * cool * (np.random.randn(nf) + 1j * np.random.randn(nf))
            h_p = project_to_sum(np.fft.irfft(h_f, n=n), n / 2)

        cd_budget = min(4.0, max(0.8, (t2 - time.time()) / 40.0))
        hc, c5v = coord_descent(h_p, n, time.time() + cd_budget, max_sweeps=150)

        temp = 0.002 * cool + 0.00005
        accept = c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-11))
        if accept:
            work_h, work_c5 = hc, c5v
        if c5v < best_c5:
            best_c5, best_h = c5v, hc.copy()

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 3: Try alternate grid sizes (20% budget)
    t3 = t0 + budget_s * 0.72
    for n_try in [192, 256, 288, 320]:
        if time.time() > t3 or n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        hc, c5v = coord_descent(h_i, n_try, time.time() + min(10.0, budget_s * 0.04), max_sweeps=400)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 4: Final intensive CD (28% budget)
    t4 = t0 + budget_s * 0.96
    best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=5000)

    # Tiny final pass
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=300)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
