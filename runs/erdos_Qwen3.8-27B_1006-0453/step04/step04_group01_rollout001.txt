import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    diff = target - vc.sum()
    if abs(diff) < 1e-14:
        return vc
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(vc + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vc + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    ov = get_overlaps(h)
    mx = np.max(ov)
    shifted = T * (ov - mx)
    emax = np.max(shifted)
    S = np.sum(np.exp(shifted - emax))
    f = mx + (emax + np.log(S)) / T
    w = np.exp(shifted - emax) / S
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        k = k_idx - (n - 1)
        wkd = wk * dx
        grad += wkd
        if k >= 0:
            grad[:n - k] -= wkd * h[k:]
            if k > 0:
                grad[k:] -= wkd * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= wkd * h[:n - kk]
            grad[:n - kk] -= wkd * h[kk:]
    return f, grad

def exact_subgrad(h, n, eps=1e-10):
    dx = 2.0 / n
    ov = get_overlaps(h)
    mx = np.max(ov)
    active = ov >= mx - eps
    n_act = np.sum(active)
    if n_act == 0:
        active = ov == mx
        n_act = 1
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        k = k_idx - (n - 1)
        gd = dx / n_act
        grad += gd
        if k >= 0:
            grad[:n - k] -= gd * h[k:]
            if k > 0:
                grad[k:] -= gd * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= gd * h[:n - kk]
            grad[:n - kk] -= gd * h[kk:]
    return grad, mx

def nesterov_phase(h, n, t_end, T_init=100, T_max=50000):

    best_c5 = compute_c5(h)
    x = h.copy()
    v = np.zeros(n)
    alpha = 0.5 / (n ** 0.5)
    mu = 0.85
    T = T_init
    stalled = 0
    while time.time() < t_end:
        T = min(T * 1.05, T_max)
        f, g = smooth_obj_grad(x, T)
        x_new = x - alpha * g
        v = mu * v + (1 - mu) * (x_new - x)
        x = x - alpha * g
        x = project(x, n)
        c5 = compute_c5(x)
        if c5 < best_c5 - 1e-12:

            best_h = x.copy()
            stalled = 0

    return best_h, best_c5

def backtracking_subgrad(h, n, t_end, max_iter=2000):

def coord_descent(h, n, t_end, max_sweeps=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            ov = get_overlaps(best_h)
            mx = np.max(ov)
            a_i = np.zeros(2 * n - 1)
            for k_idx in range(2 * n - 1):
                k = k_idx - (n - 1)
                a_val = dx
                if k >= 0:
                    if k < n:
                        a_val -= best_h[(i + k) % n] if (i + k) < n else 0.0
                    if k > 0 and (i - k) >= 0:
                        a_val -= best_h[i - k]
                else:
                    kk = -k
                    if (i + kk) < n:
                        a_val -= best_h[i + kk]
                    if (i - kk) >= 0:
                        a_val -= best_h[i - kk]
                a_i[k_idx] = a_val
            b_i = ov - a_i * best_h[i]
            def f1d(xv):
                return np.max(a_i * xv + b_i)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',

                h_try[i] = res.x
                h_try = project(h_try, n)

                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        no_imp = 0 if improved else no_imp + 1
        if no_imp >= 5:
            break
    return best_h, best_c5
