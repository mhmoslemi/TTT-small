from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-13:
        return vc
    def s_at(c):
        return np.clip(h + c, 0, 1).sum()
    lo, hi = -h.max() - 1, 2 - h.min()
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if s_at(mid) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def make_grad_idx(n):
    W1 = (n - 1) - np.arange(n)[:, None] + np.arange(n)[None, :]
    W2 = np.arange(n)[:, None] + (n - 1) - np.arange(n)[None, :]
    return W1, W2

def smooth_grad(h, T, W1, W2):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    lw = T * (corr - m)
    lw -= lw.max()
    w = np.exp(lw)
    w /= w.sum()
    t1 = w[W1] @ h
    t2 = w[W2] @ h
    return dx * (1.0 - t1 - t2)

def smooth_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def nesterov_cd(h, n, W1, W2, t_end, max_iters):
    """Nesterov accelerated gradient on smoothed objective + projection."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x = h.copy()
    y = x.copy()
    beta = 0.0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        T = 200 + 500 * (it / max_iters)
        g = smooth_grad(y, T, W1, W2)
        lr = 0.02 / (1.0 + 0.01 * it)
        x_new = y - lr * g
        beta = (1.0 - np.sqrt(1.0 + 2.0 / (it + 2))) / 2.0
        y = x_new + beta * (x_new - x)
        x = project(x_new, n)
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
    return best_h, best_c5

def coord_opt_single(h, n, i, t_end):
    """Optimize single coordinate h[i] via golden section on the piecewise-linear max."""
    # corr changes linearly with h[i], so max of linear functions
    # Just do golden section on [0,1]
    def f1d(vi):
        h[i] = vi
        c = np.correlate(h, 1.0 - h, mode='full')
        val = np.max(c) * (2.0 / n)
        return val
    old = h[i]
    res = minimize_scalar(f1d, bounds=(0, 1), method='bounded', options={'xatol': 1e-10})
    h[i] = res.x
    return res.x

def coord_descent(h, n, t_end, max_sweeps=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old_val = best_h[i]
            # Find optimal h[i] given all others fixed
            # The objective is max_k corr_k * dx where corr_k is linear in h[i]
            # For each k: corr_k(h[i]) = base_k + h[i] * slope_k
            # We want to minimize max_k (base_k + h[i]*slope_k) * dx over h[i] in [0,1]
            # subject to sum constraint (handled by projection)

                c = np.correlate(h_test, 1.0 - h_test, mode='full')
                return np.max(c) * dx

                                  options={'xatol': 1e-10})

                    best_h = project(best_h, n)
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 5:
            break
    return best_h, best_c5

                best_h = hc
        except Exception:
            pass

def fast_coord_opt(h, n, i, corr_current):
    """Optimize h[i] given current correlation values. Returns new h[i] and new corr."""
    dx = 2.0 / n
    # corr[k_idx] = sum_j h[j]*(1-h[j+k]) where k = k_idx - (n-1)
    # When h[i] changes from old to v:
    # For k_idx where i is in range: corr changes by (v - old) * (1-h[i+k])
    # For k_idx where i+k is in range: corr changes by (v - old) * (-h[i-k])  [h[i-k] term]
    # Net: corr[k_idx] += (v - old) * ((1-h[i+k]) - h[i-k]) for relevant k_idx
    
    old = h[i]
    # Precompute slopes
    n_corr = 2 * n - 1
    slopes = np.zeros(n_corr)
    for k_idx in range(n_corr):
        k = k_idx - (n - 1)
        slope = 0.0
        # h[i] appears as h[j] when j=i, so we need i+k valid
        if 0 <= i + k < n:
            slope += (1.0 - h[i + k])
        # h[i] appears in (1-h[j+k]) when j+k=i, so j=i-k, need valid
        if 0 <= i - k < n:
            slope += (-h[i - k])
        slopes[k_idx] = slope
    
    # Now corr_new[k_idx] = corr_current[k_idx] + (v - old) * slopes[k_idx]
    # We want to minimize max_k corr_new[k_idx] * dx over v in [0,1]
    # = minimize max_k (corr_current[k_idx] + (v-old)*slopes[k_idx]) * dx
    
    # Ternary search
    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        f1 = np.max(corr_current + (m1 - old) * slopes)
        f2 = np.max(corr_current + (m2 - old) * slopes)
        if f1 < f2:
            b = m2
        else:
            a = m1
    v_opt = (a + b) / 2
    # Update correlation
    corr_new = corr_current + (v_opt - old) * slopes
    return v_opt, corr_new

        if 0 <= i + k < n:
            s += 1.0 - h[i + k]
        if 0 <= i - k < n:
            s -= h[i - k]
        slopes[k_idx] = s
    return slopes

def ternary_max(corr, slopes, old, iters=60):
    """Find v in [0,1] minimizing max(corr + (v-old)*slopes)."""
    a, b = 0.0, 1.0
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + (m1 - old) * slopes)
        f2 = np.max(corr + (m2 - old) * slopes)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def intensive_cd(h, n, t_end, max_sweeps=10000):
    """Fast coordinate descent maintaining correlation array."""
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            slopes = fast_slopes(best_h, n, i)
            v_opt = ternary_max(corr, slopes, best_h[i], iters=40)
            delta = v_opt - best_h[i]
            if abs(delta) < 1e-10:
                continue
            # Apply change
            best_h[i] = v_opt
            corr += delta * slopes
            # Project
            best_h = project(best_h, n)
            # Recompute corr (since projection changed other elements)
            corr_new = np.correlate(best_h, 1.0 - best_h, mode='full')
            c5_try = np.max(corr_new) * dx
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                corr = corr_new
                improved = True
            else:
                # Undo
                best_h[i] = best_h[i] - delta  # approximate undo
                best_h = project(best_h, n)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 4:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-13:
        return vc
    def s_at(c):
        return np.clip(h + c, 0, 1).sum()
    lo, hi = -h.max() - 1.0, 2.0 - h.min()
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if s_at(mid) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def make_grad_idx(n):
    W1 = (n - 1) - np.arange(n)[:, None] + np.arange(n)[None, :]
    W2 = np.arange(n)[:, None] + (n - 1) - np.arange(n)[None, :]
    return W1, W2

def smooth_grad_fast(h, T, W1, W2):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    lw = T * (corr - m)
    lw -= lw.max()
    w = np.exp(lw)
    w /= w.sum()
    t1 = w[W1] @ h
    t2 = w[W2] @ h
    return dx * (1.0 - t1 - t2)

def smooth_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def nesterov_phase(h, n, W1, W2, t_end, max_iters=80000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x = h.copy()
    y = x.copy()
    for it in range(max_iters):
        if time.time() > t_end:
            break
        T = 150 + 600.0 * it / max_iters
        g = smooth_grad_fast(y, T, W1, W2)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        lr = 0.05 / (1.0 + 0.05 * it)
        x_new = y - lr * g
        mu = 2.0 / (it + 2)
        beta = (1.0 - np.sqrt(1.0 + mu)) / 2.0
        y = x_new + beta * (x_new - x)
        x = project(x_new, n)
        if it % 50 == 0:
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
        if it % 200 == 0:
            best_h = project(best_h, n)
    c5_final = compute_c5(x)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = x.copy()
    return best_h, best_c5

def fast_slopes(h, n, i):
    slopes = np.zeros(2 * n - 1)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        s = 0.0
        if 0 <= i + k < n:
            s += 1.0 - h[i + k]
        if 0 <= i - k < n:
            s -= h[i - k]
        slopes[k_idx] = s
    return slopes

def ternary_opt(corr, slopes, old, iters=45):
    a, b = 0.0, 1.0
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + (m1 - old) * slopes)
        f2 = np.max(corr + (m2 - old) * slopes)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) * 0.5

def intensive_cd(h, n, t_end, max_sweeps=8000):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h_save = best_h.copy()
            slopes = fast_slopes(best_h, n, i)
            v_opt = ternary_opt(corr, slopes, best_h[i], iters=40)
            delta = v_opt - best_h[i]
            if abs(delta) < 1e-10:
                continue
            best_h[i] = v_opt
            best_h = project(best_h, n)
            corr_new = np.correlate(best_h, 1.0 - best_h, mode='full')
            c5_try = np.max(corr_new) * dx
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                corr = corr_new
                improved = True
            else:
                best_h = h_save
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 4:
            break
    return best_h, best_c5

def slsqp_step(h, n, T, maxiter=800, ftol=1e-13):
    def obj(x):
        return smooth_val(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con_eq}],
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    W1, W2 = make_grad_idx(n)

    # Phase 1: SLSQP warmup (8%)
    t1 = t0 + budget_s * 0.08
    for T in [100, 500, 2000]:
        if elapsed() > t1:
            break
        hc = slsqp_step(best_h.copy(), n, T, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Nesterov accelerated gradient (22%)
    t2 = t0 + budget_s * 0.30
    best_h, best_c5 = nesterov_phase(best_h, n, W1, W2, t2, max_iters=100000)

    # Phase 3: Intensive coordinate descent (48%)
    t3 = t0 + budget_s * 0.78
    best_h, best_c5 = intensive_cd(best_h, n, t3, max_sweeps=10000)

    # Phase 4: Diversification - multiple restarts with SLSQP + CD (15%)
    t4 = t0 + budget_s * 0.93
    for r in range(15):
        if elapsed() > t4:
            break
        if r % 3 == 0:
            h_pert = best_h + 0.06 * np.random.randn(n)
        elif r % 3 == 1:
            idx = np.random.choice(n, n // 3, replace=False)
            h_pert = best_h.copy()
            h_pert[idx] = np.clip(h_pert[idx] + 0.15 * np.random.randn(len(idx)), 0, 1)
        else:
            h_pert = best_h.copy()
            h_pert = np.tanh(3.0 * (h_pert - 0.5)) * 0.3 + 0.5
            h_pert += 0.03 * np.random.randn(n)
        h_pert = project(h_pert, n)
        T_s = 1000 + 3000 * (1.0 - r / 15.0)
        h_pert = slsqp_step(h_pert, n, T_s, maxiter=300)
        t_sub = min(t4, time.time() + budget_s * 0.018)
        h_pert, c5v = intensive_cd(h_pert, n, t_sub, max_sweeps=200)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_pert

    # Phase 5: Final SLSQP + CD polish (7%)
    t5 = t0 + budget_s * 0.99
    for T in [5000, 20000]:
        if elapsed() > t5:
            break
        hc = slsqp_step(best_h.copy(), n, T, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if elapsed() < t5:
        best_h, best_c5 = intensive_cd(best_h, n, t5, max_sweeps=2000)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
