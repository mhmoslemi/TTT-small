def project(h, n):

            if not mask.any():
                break

def coord_sweep(h, n, order=None, t_end=None):

from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    target = 0.5 * n
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sweep(h, n, order, t_end=None):
    center = n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for i in order:
        if t_end is not None and time.time() > t_end:
            break
        old = h[i]
        if old < 1e-15:
            continue
        
        # Compute coefficient vector a
        a = np.zeros(2 * n - 1)
        for k in range(-n + 1, n):
            ik = i + k
            ik_fin = i - k
            v_ik = (1.0 - h[ik]) if 0 <= ik < n else 0.0
            v_fin = h[ik_fin] if 0 <= ik_fin < n else 0.0
            a[center + k] = v_ik - v_fin

        corr_0 = corr - old * a
        corr_0[center] = corr[center] - old + old * old

        a_center_val = a[center]  # save = 1-2*old
        b[center] = 1.0
        
        # Optimize
        def f1d(x):
            mx = corr_0.max()
            # Quick check: if x=0 is already optimal area
            # f(x) = max(corr_0 + x*b, center gets -x²)
            v = corr_0 + x * b
            v[center] = corr_0[center] + x - x * x
