def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * ((target - s) / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def to_binary_cont(h, n):
    """Convert continuous h to binary, maintaining sum = n/2"""
    h_sorted = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[h_sorted[:n//2]] = 1.0
    return b

def binary_delta_corr(h, i, n):
    """Compute the change in corr vector when h[i] flips (1<->0)"""
    # h[i] changes by -1 (if h[i]=1) or +1 (if h[i]=0)

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h.copy()
        h = h + room * ((target - s) / tot)
    else:
        if s < 1e-12:
            return h.copy()
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def corr_grad_k(h, k, n):
    """Gradient of corr[n-1+k] with respect to h"""
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    if k > 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        g[:n+k] += 1.0 - h[-k:]
        g[-k:] -= h[:n+k]
    return g

def cd_continuous(h, n, t_end, max_sweeps=500, tol=1e-12):
    """Coordinate descent for continuous h"""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    off = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Precompute coeff for each k
            # corr change for k≠0: (v-old) * [(1-h[i-k])*valid - h[i+k]*valid]
            # corr change for k=0: v*(1-v) - old*(1-old)
            coeff = np.zeros(2*n-1)
            # k = idx - off
            for idx in range(2*n-1):
                k = idx - off
                c = 0.0
                if k == 0:
                    c = None  # special
                else:
                    if 0 <= i-k < n:
                        c += 1.0 - h[i-k]
                    if 0 <= i+k < n:
                        c -= h[i+k]
                    coeff[idx] = c
            
            c0_old = corr[off]
            
            # Objective: max of (corr + (v-old)*coeff) for k≠0, and (c0_old + v*(1-v)-old*(1-old)) for k=0
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[off] = c0_old + v*(1.0-v) - old*(1.0-old)
                return np.max(cn)
            
            # Find minimum: check boundaries and use golden section / bounded
            v0 = obj(0.0)
            v1 = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            bv, bf = res.x, res.fun
            if v0 < bf: bv, bf = 0.0, v0
            if v1 < bf: bv, bf = 1.0, v1
            
            if bf * dx < best_c5 - tol:
                h[i] = bv
                corr = np.correlate(h, 1.0 - h, mode='full')
                best_c5 = bf * dx
                improved = True
        if not improved:
            break
    return h, best_c5

def binary_sa(h0, n, t_end, T_start=2.0, T_end=0.01, cool_per_iter=0.99995):
    # Start from binary approximation
    h = np.zeros(n)
    sorted_idx = np.argsort(h0)[::-1]
    h[sorted_idx[:n//2]] = 1.0
    
    corr = np.correlate(h, 1-h, mode='full')
    curr_max = np.max(corr)
    best_h = h.copy()
    best_max = curr_max
    
    ones = sorted_idx[:n//2].copy()
    zeros = sorted_idx[n//2:].copy()
    one_set = set(ones.tolist())
    zero_set = set(zeros.tolist())
    n_ones = n//2
    
    T = T_start
    rng = np.random
    stalling = 0
    
    while time.time() < t_end:
        # Pick random 1 and 0
        i = ones[rng.randint(n_ones)]
        j = zeros[rng.randint(n_ones)]
        
        # Compute delta corr
        d = np.zeros(2*n-1)
        off = n-1
        # h[i]: 1->0
        d[off:off+i+1] += h[i::-1] - 1.0
        if n-1-i > 0:
            d[off:off+n-i] += h[i:n]
        # h[j]: 0->1
        d[off:off+j+1] += 1.0 - h[j::-1]
        if n-1-j > 0:
            d[off:off+n-j] -= h[j:n]
        
        new_corr = corr + d
        new_max = np.max(new_corr)
        
        delta = new_max - curr_max
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            corr = new_corr
            curr_max = new_max
            h[i] = 0.0
            h[j] = 1.0
            # Update lookup arrays
            pos_i = np.where(ones == i)[0][0]
            pos_j = np.where(zeros == j)[0][0]
            ones[pos_i] = j
            zeros[pos_j] = i
        
        if new_max < best_max:
            best_max = new_max
            best_h = h.copy()
            stalling = 0
        else:
            stalling += 1
            if stalling > 200000:
                # Reheat
                T = max(T, 0.5)
                stalling = 0
        
        T = max(T_end, T * cool_per_iter)
    
    return best_h, best_max * (2.0/n)
