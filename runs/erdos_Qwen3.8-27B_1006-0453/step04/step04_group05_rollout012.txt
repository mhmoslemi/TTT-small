def adam_smoothed_max(h, N, deadline, tau=20.0, K=80, lr=0.01):
    dx = 2.0 / N
    best_h = h.copy()
    best_val = np.max(compute_corr(h)) * dx
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0
    
    while time.time() < deadline:
        c = compute_corr(h)  # length 2N-1
        # Smoothed max
        # Scale for numerical stability
        c_scaled = tau * c * dx  # these are the f_k values scaled by tau
        c_max = c_scaled.max()
        exps = np.exp(c_scaled - c_max)
        weights = exps / exps.sum()
        
        # Gradient: weighted sum of individual gradients
        # Only compute for top K lags for efficiency
        top_k = np.argpartition(c, -K)[-K:]
        grad = np.zeros(N)
        center = N - 1
        for k in top_k:
            w = weights[k]
            d = k - center
            if d == 0:
                grad += w * (1 - 2*h) * dx
            elif d > 0:
                # c[center+d] = sum_{n=0}^{N-1-d} h[n+d] * (1-h[n])

                if d < N:
                    grad[d:] += w * (1 - h[:N-d]) * dx
                    grad[:N-d] -= w * h[d:] * dx
            else:
                m_ = -d
                # c[center-m_] = sum_{n=0}^{N-1-m_} h[n] * (1-h[n+m_])
                # d/dh[j]: (1-h[j+m_]) for j in [0, N-1-m_] - h[j-m_] for j in [m_, N-1]
                if m_ < N:
                    grad[:N-m_] += w * (1 - h[m_:]) * dx
                    grad[m_:] -= w * h[:N-m_] * dx
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2

        frac = min(1.0, step / max(1, n_total))

        vh = v / (1 - b2 ** step)

        out[idx[:nh - n1]] = 1
    return out

def grad_smoothed_max(h, N, K=60, tau=30.0):
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = tau * c
    cmax = cs.max()
    exps = np.exp(cs - cmax)
    w = exps / exps.sum()
    top_k = np.argsort(c)[-K:]
    w_top = w[top_k]
    w_top = w_top / w_top.sum()
    grad = np.zeros(N)
    center = N - 1
    for i in range(len(top_k)):
        k = top_k[i]
        wt = w_top[i] * dx
        d = k - center
        if d == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif 0 < d < N:
            grad[d:] += wt * (1.0 - h[:N - d])
            grad[:N - d] -= wt * h[d:]
        elif -N < d < 0:
            m = -d
            grad[m:] += wt * (1.0 - h[:N - m])
            grad[:N - m] -= wt * h[m:]
    return grad, float(c.max()) * dx

def adam_opt(h, N, deadline, tau0=40.0, tau1=8.0, K=60, lr=0.025):
    target = N / 2.0
    best_h = h.copy()
    best_val = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    step = 0
    while time.time() < deadline:
        step += 1
        frac = min(1.0, step / 8000.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad, val = grad_smoothed_max(h, N, K, tau)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1.0 - b1 ** step)
        vh = v / (1.0 - b2 ** step)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_val

def binary_anneal(h, N, n_iters, rng, T0=4.0, T1=0.05):
    h = h.copy()
    dx = 2.0 / N
    best = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_raw = float(c.max())
    cur_raw = best_raw
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / n_iters)
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        h[p], h[q] = 0.0, 1.0
        new_raw = float(np.correlate(h, 1.0 - h, mode='full').max())
        delta = new_raw - cur_raw
        if delta < 0 or (T > 0.01 and rng.random() < np.exp(-delta / (T * max(cur_raw, 1.0)))):
            cur_raw = new_raw
            if new_raw < best_raw:
                best_raw = new_raw
                best = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
    return best, best_raw * dx

def binary_greedy(h, N, n_sweeps, rng):
    h = h.copy()
    best = h.copy()
    best_raw = float(np.correlate(h, 1.0 - h, mode='full').max())
    for _ in range(n_sweeps):
        improved = False
        n1 = int(h.sum())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(N):
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            new_raw = float(np.correlate(h, 1.0 - h, mode='full').max())
            if new_raw < best_raw:
                best_raw = new_raw
                best = h.copy()
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[p], h[q] = 1.0, 0.0
        if not improved:
            break
    return best, best_raw * (2.0 / N)

def interp_to(h_old, N_new, N_old):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary-ize and quick greedy
    h_b = make_binary(h0, N0, rng)
    h_b, c5_b = binary_greedy(h_b, N0, 8, rng)
    if c5_b < best_c5:
        best_c5, best_h = c5_b, h_b.copy()

    # Phase 2: Binary anneal (multiple restarts)
    d2 = deadline - 400
    h_cur = best_h.copy()
    for restart in range(15):
        if time.time() > d2:
            break
        n_ann = 8000
        h_a, c5_a = binary_anneal(h_cur, N0, n_ann, rng, T0=3.0, T1=0.03)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a.copy()
            h_cur = h_a.copy()
        # Perturb and re-anneal
        h_pert = h_a.copy()
        n_flip = max(3, N0 // 80)
        flips = rng.choice(N0, size=n_flip, replace=False)
        h_pert[flips] = 1.0 - h_pert[flips]
        h_pert = make_binary(h_pert, N0, rng)
        h_a2, c5_a2 = binary_anneal(h_pert, N0, 4000, rng, T0=2.0, T1=0.02)
        if c5_a2 < best_c5:
            best_c5, best_h = c5_a2, h_a2.copy()
            h_cur = h_a2.copy()

    # Phase 3: Continuous Adam optimization
    if time.time() < deadline - 150:
        h_c = best_h.astype(float) + rng.randn(N0) * 0.1
        h_c = project_sum(h_c, N0 / 2.0)
        d3 = deadline - 120
        h_c, c5_c = adam_opt(h_c, N0, d3, tau0=50.0, tau1=10.0, K=80, lr=0.02)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()

    # Phase 4: Binary-ize continuous result and refine
    if time.time() < deadline - 80:
        h_b2 = make_binary(best_h, N0, rng)
        h_b2, c5_b2 = binary_greedy(h_b2, N0, 10, rng)
        if c5_b2 < best_c5:
            best_c5, best_h = c5_b2, h_b2.copy()
        h_b2, c5_b2 = binary_anneal(h_b2, N0, 6000, rng, T0=2.0, T1=0.02)
        if c5_b2 < best_c5:
            best_c5, best_h = c5_b2, h_b2.copy()

    # Phase 5: Second continuous pass with different params
    if time.time() < deadline - 50:
        h_c2 = best_h.astype(float) + rng.randn(N0) * 0.05
        h_c2 = project_sum(h_c2, N0 / 2.0)
        h_c2, c5_c2 = adam_opt(h_c2, N0, deadline - 30, tau0=80.0, tau1=20.0, K=100, lr=0.012)
        if c5_c2 < best_c5:
            best_c5, best_h = c5_c2, h_c2.copy()

    # Phase 6: Try different N
    for dN in [40, -20, 60]:
        if time.time() > deadline - 30:
            break
        N_try = N0 + dN
        if N_try < 200 or N_try % 2 != 0:
            N_try -= 1
            if N_try < 200:
                continue
        h_interp = interp_to(best_h, N_try, N0)
        h_bin_new = make_binary(h_interp, N_try, rng)
        h_bin_new, c5_new = binary_greedy(h_bin_new, N_try, 5, rng)
        if time.time() < deadline - 15:
            h_bin_new, c5_new2 = binary_anneal(h_bin_new, N_try, 4000, rng, T0=2.0, T1=0.03)
            if c5_new2 < c5_new:
                c5_new = c5_new2
                h_bin_new = h_bin_new
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_bin_new.copy()
        # Quick continuous on this N
        if time.time() < deadline - 10 and c5_new <= best_c5:
            h_cn = h_bin_new.astype(float) + rng.randn(N_try) * 0.06
            h_cn = project_sum(h_cn, N_try / 2.0)
            h_cn, c5_cn = adam_opt(h_cn, N_try, min(deadline - 5, time.time() + 40),
                                     tau0=60.0, tau1=15.0, K=60, lr=0.015)
            if c5_cn < best_c5:
                best_c5, best_h = c5_cn, h_cn.copy()

    # Final: last polish
    if time.time() < deadline - 2:
        hF = best_h.astype(float) + rng.randn(len(best_h)) * 0.03
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = adam_opt(hF, len(best_h), deadline - 1, tau0=100.0, tau1=30.0, K=80, lr=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
