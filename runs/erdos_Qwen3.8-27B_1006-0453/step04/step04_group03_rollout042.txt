def project_feasible(g, target_sum):
    """Project g onto {0<=h<=1, sum(h)=target_sum}"""

def subgradient_step(h, corr, alpha, n_active=3):
    """Take one subgradient step using top n_active correlations."""
    n = len(h)
    # Find top active constraints
    idx = np.argsort(corr)[-n_active:]
    grad = np.zeros(n)
    for k_idx in idx:

    return np.correlate(h.astype(np.float64), 1.0 - h, mode='full')

    n = len(g)
    lo, hi = g.min() - 2.0, g.max() + 2.0

        s = np.clip(g - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def grad_single_k(h, k, n):
    """Gradient of f_k(h) = sum_i h[i](1-h[i+k]) w.r.t. h."""
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        ka = -k
        if ka < n:
            g[ka:] += (1.0 - h[:-ka] if ka < n else np.array([]))
        if ka > 0:
            g[:n-ka] -= h[ka:]
    return g

36. Now for the 1D solver. Let me implement it carefully:

```python
def solve_1d_convex(corr, coef, lo, hi):
    """Minimize max_m(corr[m] + s*coef[m]) for s in [lo, hi]."""
    # Bisection on subdifferential
    for _ in range(60):
        mid = 0.5*(lo+hi)
        vals = corr + mid*coef
        mx = vals.max()
        # Active set: indices where vals[m] is within tolerance of max
        active = coef[vals >= mx - 1e-10]
        if len(active) == 0:
            break
        if active.min() > 1e-12:
            hi = mid
        elif active.max() < -1e-12:
            lo = mid
        else:
            break
    return 0.5*(lo+hi)
```

This should converge in ~20-30 iterations for our precision needs. Each iteration is O(n) (n=2n-1=447 for n=224).
