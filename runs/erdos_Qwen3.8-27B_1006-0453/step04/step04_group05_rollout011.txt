def delta_corr(h, N, p, q):
    """Delta in full correlation when h[p]:1->0, h[q]:0->1."""
    Nc = 2 * N - 1
    center = N - 1
    dc = np.zeros(Nc)
    # Remove h[p] (=1 becomes 0)
    # -g[p+l] for l=0..N-1-p (m=center+l)
    sl = slice(center, center + N - p)
    dc[sl] -= (1.0 - h[p:p + N - p])
    # +h[p-l] for l=0..p (m=center-l)
