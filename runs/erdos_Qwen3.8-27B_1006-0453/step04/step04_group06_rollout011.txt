import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_opt_fast(h, corr, n, i):
    center = n - 1
    hi = h[i]
    alpha = np.zeros(2 * n - 1)
    for l in range(1, n):
        ap = 0.0
        if i + l < n:
            ap += (1.0 - h[i + l])
        if i - l >= 0:
            ap -= h[i - l]
        alpha[center + l] = ap
        an = 0.0
        if i - l >= 0:
            an += (1.0 - h[i - l])
        if i + l < n:
            an -= h[i + l]
        alpha[center - l] = an

    def f(x):
        vals = corr + (x - hi) * alpha
        vals[center] = corr[center] + x - x * x - hi + hi * hi
        return np.max(vals)

    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def smooth_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    ex = np.exp(T * (corr - m) * dx)
    return (m + np.log(ex.sum()) / T) * dx

def smooth_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    ex = np.exp(T * (corr - m) * dx)
    w = ex / ex.sum()
    g = np.zeros(n)
    center = n - 1
    for j in range(n):
        gj = 0.0
        for l in range(-n + 1, n):
            k = center + l
            if l == 0:
                al = 1.0 - 2.0 * h[j]
            else:
                al = 0.0
                if 0 <= j + l < n:
                    al += (1.0 - h[j + l])
                if 0 <= j - l < n:
                    al -= h[j - l]
            gj += w[k] * al
        g[j] = gj * dx
    return g

def slsqp_opt(h, T, n, maxiter=3000):
    def obj(x):
        return smooth_obj(x, T, n)
    def objg(x):
        return smooth_grad(x, T, n)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(len(x))}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(objg, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project(np.clip(res.x, 0, 1), n)

def proj_grad_step(h, T, n, t_max):
    h = h.copy()
    alpha = 0.05
    stale = 0
    for it in range(20000):
        if time.time() > t_max:
            break
        obj = smooth_obj(h, T, n)
        g = smooth_grad(h, T, n)
        h_try = project(h - alpha * g, n)
        obj_try = smooth_obj(h_try, T, n)
        if obj_try < obj - 1e-15:
            h = h_try
            alpha = min(alpha * 1.05, 0.3)
            stale = 0
        else:
            alpha *= 0.7
            stale += 1
            if stale > 20 or alpha < 1e-9:
                break
    return h

def cd_sweep(h, corr, n, t_end, max_sweeps=5000):
    best_c5 = np.max(corr) * (2.0 / n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            hi_new = coord_opt_fast(h, corr, n, i)
            if abs(hi_new - h[i]) > 1e-14:
                h[i] = hi_new
                corr = np.correlate(h, 1.0 - h, mode='full')
                improved = True
        c5_now = np.max(corr) * (2.0 / n)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not improved:
            break
    return h, best_c5, corr

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.28 * np.cos(np.pi * x) + 0.12 * np.cos(2 * np.pi * x)
                         - 0.04 * np.sin(np.pi * x) + 0.03 * np.sin(2 * np.pi * x), 0, 1)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5, best_corr = compute_c5(best_h)

    # Phase 1: Coordinate descent (25%)
    h = best_h.copy()
    corr = best_corr.copy()
    h, c5v, corr = cd_sweep(h, corr, n, t0 + budget_s * 0.25)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()

    # Phase 2: SLSQP with increasing T (20%)
    for T in [30, 100, 400, 1500, 6000, 25000]:
        if time.time() > t0 + budget_s * 0.45:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=3000)
            cv, _ = compute_c5(hc)
            if cv < best_c5:
                best_c5, best_h = cv, hc
        except Exception:
            pass

    # Phase 3: Projected gradient + CD (15%)
    pg_end = t0 + budget_s * 0.60
    for T in [2000, 8000, 30000]:
        if time.time() > pg_end:
            break
        hp = proj_grad_step(best_h, T, n, pg_end)
        cp, _ = compute_c5(hp)
        if cp < best_c5:
            best_c5, best_h = cp, hp
        h2, c2, _ = cd_sweep(best_h.copy(), np.correlate(best_h, 1-best_h, 'full'), n,
                             min(time.time() + 5, pg_end), max_sweeps=100)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 4: Diverse restarts (15%)
    restart_end = t0 + budget_s * 0.75
    for r in range(20):
        if time.time() > restart_end:
            break
        h_try = best_h.copy()
        if r % 4 == 0:
            h_try += 0.04 * np.random.randn(n)
        elif r % 4 == 1:
            bs = np.random.randint(3, 15)
            st = np.random.randint(0, max(1, n - bs))
            h_try[st:st+bs] += 0.08 * np.random.randn(bs)
        elif r % 4 == 2:
            idx = np.random.choice(n, size=min(6, n), replace=False)
            signs = np.random.choice([-1, 1], size=len(idx))
            h_try[idx] += 0.06 * signs
        else:
            # Swap pairs
            idx = np.random.choice(n, size=(n // 5, 2), replace=False).flatten()
            for a in range(0, len(idx), 2):
                h_try[idx[a]], h_try[idx[a+1]] = h_try[idx[a+1]], h_try[idx[a]]
        h_try = project(h_try, n)
        corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
        h_try, c_try, _ = cd_sweep(h_try, corr_try, n,
                                    min(time.time() + 4, restart_end), max_sweeps=60)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 5: Extended CD + SLSQP polish (20%)
    cd_end = t0 + budget_s * 0.90
    h = best_h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    h, c5v, corr = cd_sweep(h, corr, n, cd_end, max_sweeps=10000)
    if c5v < best_c5:
        best_c5, best_h = c5v, h

    for T in [50000, 200000, 800000]:
        if time.time() > t_end - 10:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=5000)
            cv, _ = compute_c5(hc)
            if cv < best_c5:
                best_c5, best_h = cv, hc
        except Exception:
            pass
        # CD after SLSQP
        h2, c2, _ = cd_sweep(best_h.copy(), np.correlate(best_h, 1-best_h, 'full'), n,
                             t_end - 5, max_sweeps=2000)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Final tighten
    if time.time() < t_end - 2:
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        for _ in range(200):
            if time.time() > t_end - 1:
                break
            improved = False
            for i in range(n):
                hi_new = coord_opt_fast(h, corr, n, i)
                if abs(hi_new - h[i]) > 1e-15:
                    h[i] = hi_new
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    improved = True
            if not improved:
                break
        c5v = np.max(corr) * (2.0 / n)
        if c5v < best_c5:
            best_c5, best_h = c5v, h

    best_h = project(best_h, n)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, n
