    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        h = np.full(n, 0.5)
        return h
    if s > target:

        h = np.clip(h, 0, 1)

        if tot < 1e-12:
            h = h * (target / s)
            h = np.clip(h, 0, 1)
        else:

    if abs(diff) > 1e-10:

    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 3
        if mod == 0:
            pert = (0.02 + 0.1 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.08 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        else:
            mask = np.random.random(n) < (0.2 + 0.3 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.15 * cool) * np.random.randn(mask.sum())
        hp = proj_h(work + pert, n)
        T_val = 300.0 + 5000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=250)
        c5t = compute_c5(ht)
        temp = 0.0008 * cool + 0.00005
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing T
    for T in [50, 200, 800, 3000, 12000, 50000]:
        if time.time() - t0 > budget_s * 0.08:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.18, sweeps=80)

    # Phase 3: Binary SA
    h_bin, c5_bin = bin_sa(best_h, n, t0 + budget_s * 0.35)
    if c5_bin < best_c5:
        best_h = lbfgs_opt(h_bin, 2000.0, n, iters=400)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.42, sweeps=40)

    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.62, n_iters=120)

    # Phase 5: Coordinate descent after basin hopping
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.72, sweeps=60)

    # Phase 6: Try higher resolution
    for n_hi in [200, 240]:
        if time.time() - t0 > budget_s * 0.84:
            break
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0 / n_hi
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_hi)
            h_hi = lbfgs_opt(h_hi, 1500.0, n_hi, iters=350)
            c5_hi = compute_c5(h_hi)
            h_hi, c5_hi = coord_desc(h_hi, n_hi, t0 + budget_s * 0.90, sweeps=30)
            if c5_hi < best_c5:
                best_h = h_hi
                best_c5 = c5_hi
        except Exception:
            pass

    # Phase 7: Final refinement
    if time.time() - t0 < budget_s * 0.93:
        best_h = lbfgs_opt(best_h, 80000.0, len(best_h), iters=800)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.98, sweeps=30)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
