def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, n):
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        L = k - (n - 1)
        if L == 0:
            grads[k] = 1.0 - 2.0 * h
        elif L > 0:
            if n - L > 0:
                grads[k, :n-L] += (1.0 - h[L:])
            if L < n:
                grads[k, L:] += (-h[:n-L])
        else:
            M = -L
            if n - M > 0:
                grads[k, M:] += (1.0 - h[:n-M])
            if M < n:
                grads[k, :n-M] += (-h[M:])
    return grads

def lp_step(h, n, delta=None):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = corr_gradient(h, n)
    Gdx = G * dx
    rhs = Gdx @ h - corr * dx
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    cons.append(Gdx @ h_var - t_var <= rhs)
    if delta is not None:
        cons.append(h_var - h <= delta)
        cons.append(h - h_var <= delta)
    prob = cp.Problem(cp.Minimize(t_var + 1e-8 * cp.sum_squares(h_var - h)), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, None
    if h_var.value is None or t_var.value is None:
        return None, None
    return h_var.value, t_var.value

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0

    bnds = [(0.0, 1.0)] * n

                   options={'maxiter': maxiter, 'ftol': ftol})

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_grad_all(h, n):
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        L = idx - (n - 1)
        if L == 0:
            G[idx] = 1.0 - 2.0 * h
        elif L > 0:
            G[idx, :n - L] += 1.0 - h[L:]
            G[idx, L:] += -h[:n - L]
        else:
            M = -L
            G[idx, M:] += 1.0 - h[:n - M]
            G[idx, :n - M] += -h[M:]
    return G

def lp_solve(h, n, delta=0.15):
    dx = 2.0 / n
    hv = cp.Variable(n)
    tv = cp.Variable()
    corr = np.correlate(h, 1.0 - h, mode='full')
    Gdx = corr_grad_all(h, n) * dx
    rhs = Gdx @ h - corr * dx
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0, Gdx @ hv - tv <= rhs]
    if delta is not None:
        cons.append(hv <= h + delta)
        cons.append(hv >= h - delta)
    prob = cp.Problem(cp.Minimize(tv + 1e-10 * cp.sum_squares(hv - h)), cons)
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if hv.value is not None:
                return np.clip(hv.value, 0, 1)
        except Exception:
            continue
    return None

def _smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)

                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

    cv[k0] = cb[k0] + t * (1.0 - t)
    return np.max(cv) * dx

            d = a + gr * (b - a)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_grad_all(h, n):
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        L = idx - (n - 1)
        if L == 0:
            G[idx] = 1.0 - 2.0 * h
        elif L > 0:
            G[idx, :n - L] += 1.0 - h[L:]
            G[idx, L:] += -h[:n - L]
        else:
            M = -L
            G[idx, M:] += 1.0 - h[:n - M]
            G[idx, :n - M] += -h[M:]
    return G

def lp_solve(h, n, delta=0.15):
    dx = 2.0 / n
    hv = cp.Variable(n)
    tv = cp.Variable()
    corr = np.correlate(h, 1.0 - h, mode='full')
    Gdx = corr_grad_all(h, n) * dx
    rhs = Gdx @ h - corr * dx
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0, Gdx @ hv - tv <= rhs]
    if delta is not None:
        cons.append(hv <= h + delta)
        cons.append(hv >= h - delta)
    prob = cp.Problem(cp.Minimize(tv + 1e-10 * cp.sum_squares(hv - h)), cons)
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if hv.value is not None:
                return np.clip(hv.value, 0, 1)
        except Exception:
            continue
    return None

def _smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def _eq_con(x, n):
    return x.sum() - n / 2.0

def slsqp_opt(h0, T, n, maxiter=500):
    res = minimize(_smooth_obj, h0, args=(T,), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def _fval1d(t, cb, coef, k0, dx):
    cv = cb + t * coef
    cv[k0] = cb[k0] + t * (1.0 - t)
    return np.max(cv) * dx

def _min1d_golden(cb, coef, k0, dx, iters=60):
    a, b = 0.0, 1.0
    gr = 0.618033988749895
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _fval1d(c, cb, coef, k0, dx)
    fd = _fval1d(d, cb, coef, k0, dx)
    for _ in range(iters):
        if b - a < 1e-13:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = _fval1d(c, cb, coef, k0, dx)
        else:
            a, c, fc = c, d, fc
            d = a + gr * (b - a)
            fd = _fval1d(d, cb, coef, k0, dx)
    return 0.5 * (a + b)

def coord_h(i, h, n):
    dx = 2.0 / n
    k0 = n - 1
    hz = h.copy()
    hz[i] = 0.0
    cb = np.correlate(hz, 1.0 - hz, mode='full')
    L = np.arange(-(n - 1), n)
    coef = np.zeros(len(L))
    pos = L > 0
    neg = L < 0
    m1 = pos & (i + L < n)
    coef[m1] += 1.0 - hz[i + L[m1]]
    m2 = pos & (i - L >= 0)
    coef[m2] += -hz[i - L[m2]]
    m3 = neg & (i + L >= 0)
    coef[m3] += 1.0 - hz[i + L[m3]]
    m4 = neg & (i - L < n)
    coef[m4] += -hz[i - L[m4]]
    return _min1d_golden(cb, coef, k0, dx)

def coord_descent(h, n, t_end, sweeps=300):
    bh = h.copy()
    bc = compute_c5(bh)
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return bh, bc
            tn = coord_h(i, bh, n)
            ht = bh.copy()
            ht[i] = tn
            ht = proj(ht, n)
            cn = compute_c5(ht)
            if cn < bc - 1e-14:
                bh, bc, imp = ht, cn, True
        if not imp:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(180)
        h[:90] = 1.0
    n = len(h)
    h = proj(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Sequential LP with shrinking trust region
    end1 = t0 + budget_s * 0.30
    for it in range(400):
        if time.time() > end1:
            break
        delta = max(0.015, 0.20 * (0.955 ** it))
        hn = lp_solve(h, n, delta)
        if hn is None:
            break
        hn = proj(hn, n)
        cn = compute_c5(hn)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn.copy()
        if np.max(np.abs(hn - h)) < 1e-9:
            break
        h = hn

    # Phase 2: Coordinate descent polish
    if time.time() < t0 + budget_s * 0.48:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.48)

    # Phase 3: Refine with tight LP
    if time.time() < t0 + budget_s * 0.63:
        h = best_h.copy()
        for it in range(300):
            if time.time() > t0 + budget_s * 0.63:
                break
            hn = lp_solve(h, n, 0.04)

            hn = proj(hn, n)
            cn = compute_c5(hn)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
            if np.max(np.abs(hn - h)) < 1e-9:
                break
            h = hn

    # Phase 4: Basin hopping with LP polish
    if time.time() < t0 + budget_s * 0.80:
        for it in range(150):
            if time.time() > t0 + budget_s * 0.80:
                break
            cool = 1.0 - 0.7 * (it / 150.0)
            step = (0.03 + 0.08 * cool) * np.random.randn(n)

            hn = lp_solve(hp, n, 0.10)

        for T in [5000, 50000]:
            if time.time() > t0 + budget_s * 0.98:

            try:

                    break
                hi = proj(hn, n_try)
            hi = coord_descent(hi, n_try, t0 + budget_s * 0.99, sweeps=50)[0]
            cn = compute_c5(hi)
            if cn < best_c5:
                best_h, best_c5, n = hi, cn, n_try

    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
