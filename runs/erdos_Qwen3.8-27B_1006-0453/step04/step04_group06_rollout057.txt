```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            h = h * (target / max(s, 1e-15))
        else:
            h = h + room * (diff / tot)
    else:
        h = h * (target / max(s, 1e-15))
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, T):
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def smooth_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = T * (corr - m)
    emax = e.max()
    w = np.exp(e - emax)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    center = n - 1
    for j in range(2 * n - 1):
        if w[j] < 1e-14:
            continue
        k = j - center
        if k == 0:
            grad += w[j] * (1.0 - 2.0 * h)

            grad[:n-k] += w[j] * (1.0 - h[k:])
            grad[k:] -= w[j] * h[:n-k]
        else:
            k2 = -k
            grad[k2:] += w[j] * (1.0 - h[:n-k2])
            grad[:n-k2] -= w[j] * h[k2:]
    val = m + np.log(wsum) / T
    return val, grad

        v, _ = smooth_obj_grad(x, T)
        return v
    def objg(x):
        _, g = smooth_obj_grad(x, T)
        return g
    res = minimize(objg, h, method='L-BFGS-B', jac=True,

    return project_h(np.clip(res.x, 0, 1), n)

    h_base = h.copy()
    h_base[i] = 0.0

        a[center + k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[center - k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
```

```python
def smooth_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = T * (corr - m)
    emax = e.max()
    w = np.exp(e - emax)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
        wsum = 1.0
    else:
        w /= wsum
    u = 1.0 - 2.0 * h  # length n
    # grad[m] = dot(w[m:m+n], u) for m = 0..n-1
    grad = np.array([np.dot(w[m:m+n], u) for m in range(n)])
    val = m + np.log(wsum) / T
    return val, grad
```

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            h = h * (target / max(s, 1e-15))
        else:
            h = h + room * (diff / tot)
    else:
        h = h * (target / max(s, 1e-15))
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = T * (corr - m)
    emax = e.max()
    w = np.exp(e - emax)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
        wsum = 1.0
    else:
        w /= wsum
    u = 1.0 - 2.0 * h
    grad = np.empty(n)
    for m_idx in range(n):
        grad[m_idx] = np.dot(w[m_idx:m_idx + n], u)
    val = m + np.log(wsum) / T
    return val, grad
