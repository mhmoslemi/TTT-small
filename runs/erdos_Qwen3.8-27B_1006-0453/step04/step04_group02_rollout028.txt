                    if c5_up < best_c5:
                        best_h, best_c5 = h_up, c5_up

    # Phase 4: Coordinate descent
    if time.time() < t0 + budget_s * 0.78:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.78, max_sweeps=45, rng=rng)

    # Phase 5: Subgradient with very high T (sharp max)
    if time.time() < t0 + budget_s * 0.87:
        h_sg, c5_sg = subgrad_descent(best_h, n, 2500, 500.0, 50.0, 0.008)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 6: Final coordinate descent polish
    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=30, rng=rng)

    # Phase 7: Final subgradient micro-adjustment
    if time.time() < t0 + budget_s * 0.99:
        h_sg, c5_sg = subgrad_descent(best_h, n, 1000, 2000.0, 5.0, 0.003)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
