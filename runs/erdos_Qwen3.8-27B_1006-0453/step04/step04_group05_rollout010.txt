  D[1] (h[q]:1→0): h[q]*(1-h[p]) goes from 1*(1-h[p])=1*0=0 to 0. Delta: -1*(1-h[p]) = -0 = 0.
  For 0→1: delta is 0.
  Sum: 0+0=0. Actual: +1. Cross term: +1.

So for a swap, the total change is:
delta_total = D[p] - D[q] + cross_term

where cross_term is non-zero only at specific lags k = p-q+(N-1) and k = q-p+(N-1).

This makes the D-matrix approach for swaps more complex. The previous code might have had this bug, or it might have been working "in spite of" the bug because the cross term is small relative to the overall correlation magnitude.

def compute_corr(h):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        lo, hi = -2.0, 2.0
        for _ in range(55):

                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_gradient(corr, h, N, temp):
    center = N - 1
    log_w = -corr / temp
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    
    grad = np.zeros(N)
    for ai in range(len(corr)):
        wi = w[ai]
        if wi < 1e-8:
            continue
        L = ai - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        elif L < 0 and abs(L) < N:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    temp_start = max(0.1, best_peak * 0.3)
    
    while time.time() < deadline and it < n_total:
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()

        vh = v / (1.0 - b2 ** it)

        h = project_sum(h, target)

            temp_start *= 0.5

            h[p], h[q] = 1.0, 0.0
            if nv < best_new:
                best_new = nv
                best_pq = (p, q)
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            # Random perturbation to escape
            for _ in range(5):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                h[p], h[q] = 0.0, 1.0
            break
    return h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Local search on initial (binary)
    d1 = min(deadline - 300, t0 + 150)
    h_b, c5_b = local_search(best_h, N0, d1, rng, max_cand=500)
    if c5_b < best_c5:
        best_c5, best_h = c5_b, h_b.copy()

    # Phase 2: p-norm Adam
    if time.time() < deadline - 150:
        h_c = binarize(best_h, N0).astype(float) + rng.randn(N0) * 0.04
        h_c = project_sum(h_c, N0 / 2.0)
        h_a, c5_a = adam_opt(h_c, N0, N0 / 2.0, min(deadline - 80, time.time() + 250),
                              mode='pnorm', p0=6, p_end=100, early_t=0.4, lr0=0.018, n_total=10000)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a.copy()

    # Phase 3: Binarize + local search
    if time.time() < deadline - 80:
        h_bin = binarize(best_h, N0)
        h_b2, c5_b2 = local_search(h_bin, N0, min(deadline - 40, time.time() + 150), rng, max_cand=400)
        if c5_b2 < best_c5:
            best_c5, best_h = c5_b2, h_b2.copy()

    # Phase 4: Softmax Adam on best
    if time.time() < deadline - 60:
        h_c2 = best_h.astype(float) + rng.randn(len(best_h)) * 0.03
        h_c2 = project_sum(h_c2, len(best_h) / 2.0)
        h_a2, c5_a2 = adam_opt(h_c2, len(best_h), len(best_h) / 2.0,
                                min(deadline - 30, time.time() + 100),
                                mode='softmax', lr0=0.012, n_total=5000)
        if c5_a2 < best_c5:
            best_c5, best_h = c5_a2, h_a2.copy()

    # Phase 5: Different N values
    for N_try in [N0 + 40, N0 - 40, N0 + 80, N0 + 160, N0 - 80]:
        if time.time() > deadline - 25 or N_try < 200:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = binarize(h_new, N_try)
        h_b3, c5_b3 = local_search(h_new, N_try, min(deadline - 12, time.time() + 50), rng, max_cand=200)
        if c5_b3 < best_c5:
            best_c5, best_h = c5_b3, h_b3.copy()
        if time.time() < deadline - 8:
            h_c3 = h_b3.astype(float) + rng.randn(N_try) * 0.03
            h_c3 = project_sum(h_c3, N_try / 2.0)
            h_a3, c5_a3 = adam_opt(h_c3, N_try, N_try / 2.0, min(deadline - 4, time.time() + 25),
                                    mode='pnorm', p0=10, p_end=60, lr0=0.012, n_total=2000)
            if c5_a3 < best_c5:
                best_c5, best_h = c5_a3, h_a3.copy()

    # Final: aggressive local search
    if time.time() < deadline - 2:
        h_f = binarize(best_h, len(best_h))
        h_f2, c5_f2 = local_search(h_f, len(best_h), deadline - 1, rng, max_cand=600)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
