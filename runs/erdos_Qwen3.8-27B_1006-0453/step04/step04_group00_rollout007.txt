import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def logsumexp_obj(h, T, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def logsumexp_grad(h, T, target):
    """Compute gradient of log-sum-smooth max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    corr_shifted = T * (corr - np.max(corr))
    w = np.exp(corr_shifted)
    w /= w.sum()
    # grad[k] = dx * sum_m w[m] * d(corr_m)/d(h[k])
    # d(corr_m)/d(h[k]) = (1-h[k-m+n-1])*valid1 - h[k+m-n+1]*valid2
    grad = np.zeros(n)
    for k in range(n):
        # m range where k-m+n-1 in [0,n-1]: m in [k+1, k+n-1]
        # m range where k+m-n+1 in [0,n-1]: m in [n-1-k, n+1-k]
        # Using the correlation indexing: corr[m] involves pairs (i, i-m+n-1)
        # d(corr_m)/d(h[k]): k appears as first factor when i=k, i.e., m = i - j + n - 1 for some j
        # Actually let me just use the formula directly
        m1_lo = max(0, k + 1)  # m such that k-m+n-1 >= 0 => m <= k+n-1; and m >= 0
        m1_hi = min(2*n-2, k + n - 1)
        # k-m+n-1 in [0,n-1]: 0 <= k-m+n-1 <= n-1 => k+1 <= m <= k+n-1
        # k+m-n+1 in [0,n-1]: 0 <= k+m-n+1 <= n-1 => n-1-k <= m <= n-1+k
        m1_lo = max(0, k + 1)
        m1_hi = min(2*n-2, k + n - 1)
        m2_lo = max(0, n - 1 - k)
        m2_hi = min(2*n-2, n - 1 + k)
        
        # First term: (1-h[k-m+n-1]) for m in [m1_lo, m1_hi]
        if m1_hi >= m1_lo:
            m_vals = np.arange(m1_lo, m1_hi + 1)
            j_vals = k - m_vals + n - 1  # index into h
            # j_vals should be in [0, n-1] by construction
            grad[k] += np.sum(w[m_vals] * (1.0 - h[j_vals]))
        
        # Second term: -h[k+m-n+1] for m in [m2_lo, m2_hi]
        if m2_hi >= m2_lo:
            m_vals = np.arange(m2_lo, m2_hi + 1)
            j_vals = k + m_vals - n + 1
            grad[k] -= np.sum(w[m_vals] * h[j_vals])
    
    return grad * dx

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    target = n / 2.0
    def obj(x):
        return logsumexp_obj(x, T, target)
    def con(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), target)
    except Exception:
        return h

def grad_descent(h, T, target, n, lr, steps, t_end):
    """Plain gradient descent with the smooth objective."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for s in range(steps):
        if time.time() > t_end:
            break
        g = logsumexp_grad(best_h, T, target)
        # Project gradient onto constraint (make it orthogonal to all-ones)
        g_mean = g.mean()
        g = g - g_mean
        h_new = best_h - lr * g
        h_new = project_h(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=200):
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Optimize h[i] with sum constraint: fix others, adjust h[i]
            # sum(h) = target => h[i] = target - sum(h[j] for j≠i)
            # But h[i] must be in [0,1], so this is constrained
            # Actually with the sum constraint, if we change h[i], we must adjust another variable
            # Simpler: just optimize h[i] freely and then project
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            k_range = np.arange(-(n - 1), n)
            j1 = i + k_range
            j2 = i - k_range
            valid1 = (j1 >= 0) & (j1 < n)
            valid2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[valid1] += (1.0 - best_h[np.clip(j1[valid1], 0, n - 1)])
            coefs[valid2] += (-best_h[np.clip(j2[valid2], 0, n - 1)])
            def f1d(hi):
                return np.max(corr_base + hi * coefs) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = T * (corr - np.max(corr))
    w = np.exp(cs)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(n):
        m1lo = max(0, k + 1)
        m1hi = min(2 * n - 2, k + n - 1)
        if m1hi >= m1lo:
            idx = k - np.arange(m1lo, m1hi + 1) + n - 1
            grad[k] += np.dot(w[m1lo:m1hi+1], 1.0 - h[idx])
        m2lo = max(0, n - 1 - k)
        m2hi = min(2 * n - 2, n - 1 + k)
        if m2hi >= m2lo:
            idx = k + np.arange(m2lo, m2hi + 1) - n + 1
            grad[k] -= np.dot(w[m2lo:m2hi+1], h[idx])
    return grad * dx

def slsqp_opt(h, T, n, maxiter=1500, ftol=1e-15):
    target = n / 2.0
    def obj(x):
        return lse_obj(x, T)
    def con(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), target)
    except Exception:
        return h

def grad_step(h, T, target, lr, n):
    g = lse_grad(h, T)
    g = g - g.mean()
    h_new = project_h(h - lr * g, target)
    return h_new

def power_mean_c5(h, p=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (np.sum(np.abs(corr) ** p)) ** (1.0 / p) * dx

def coord_opt_point(h, i, n):
    dx = 2.0 / n
    target = n / 2.0
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    def f1d(hi):
        return np.max(corr_base + hi * coefs) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=300):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_point(best_h, i, n)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved += 1
        if improved == 0:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c5

        elif mod == 2:

        elif mod == 3:

                d = 0.08 * cool * (work_h[a] - work_h[b])

        else:

        c5v = compute_c5(hc)

            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
