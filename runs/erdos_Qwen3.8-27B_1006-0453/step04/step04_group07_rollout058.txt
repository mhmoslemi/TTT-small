import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_weighted(h, w):
    n = len(h)
    grad = np.zeros(n)
    for idx_p in range(2 * n - 1):
        p = idx_p - (n - 1)
        wp = w[idx_p]
        if abs(wp) < 1e-15:
            continue
        a1 = max(0, p)
        b1 = min(n, p + n)
        if a1 < b1:
            grad[a1:b1] += wp * (1.0 - h[a1 - p:b1 - p])
        a2 = max(0, -p)
        b2 = min(n, n - p)
        if a2 < b2:
            grad[a2:b2] -= wp * h[a2 + p:b2 + p]
    return grad

def multiplicative_weights(h0, n, t_end, max_rounds=40):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        frac = r / max(1, max_rounds - 1)
        beta = 20.0 * (1.0 + 200.0 * frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_max = np.max(corr)
        w = np.exp(beta * (corr - corr_max))
        w /= w.sum()
        
        # Projected gradient with backtracking
        lr = 0.5
        for sub in range(100):
            if time.time() > t_end:
                break
            g = grad_weighted(h, w) * dx
            h_new = h - lr * g
            h_new = project(h_new, n)
            # Check improvement on weighted objective
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            F_new = np.dot(w, corr_new) * dx
            F_old = np.dot(w, corr) * dx
            if F_new < F_old - 1e-16:
                h = h_new
                corr = corr_new
            else:
                lr *= 0.5
            if lr < 1e-10:
                break
        
        c5_v = compute_c5(h)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h.copy()
    
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            
            best_v = old
            best_f = np.max(corr)
            
            # Try v=0 and v=1
            for v_try in [0.0, 1.0]:
                cn = corr + (v_try - old) * coeff
                cn[c0_idx] = c0_base + (v_try - old) * (1.0 - v_try - old)
                f_try = np.max(cn)
                if f_try < best_f:
                    best_f = f_try
                    best_v = v_try
            
            # Interior optimum via minimize_scalar
            if 0.0 < old < 1.0:
                def obj_f(v):
                    cn = corr + (v - old) * coeff
                    cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                    return np.max(cn)
                try:
                    res = minimize_scalar(obj_f, bounds=(0.0, 1.0), method='bounded',
                                          options={'xatol': 1e-8})
                    if res.fun < best_f:
                        best_f = res.fun
                        best_v = res.x
                except Exception:
                    pass
            
            if best_f * dx < best_c5 - 1e-12:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        
        if not improved and sweep > 5:
            break
    
    return best_h, best_c5

def slsqp_weighted(h0, w, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.dot(w, corr) * dx
    def obj_grad(x):
        return grad_weighted(x, w) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

def basin_hop(work_h, work_c5, n, t_end, max_iters=150):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.9 * it / max_iters)
        step = 0.03 + 0.15 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project(h_p, n)
        # Quick coordinate descent on perturbed
        hc, c5v = coord_descent(h_p, n, min(time.time() + 2, t_end), max_sweeps=30)
        temp = 0.003 * cool + 0.0003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

    if 'initial_h_values' in globals() and initial_h_values is not None:

        return time.time() - t0

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.12, max_sweeps=300)

    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        frac = (r + 1) / max(1, max_rounds)
        beta = beta_min * (beta_max / beta_min) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        cm = np.max(corr)
        w = np.exp(beta * (corr - cm))
        w /= w.sum()
        lr = 0.3 / (1.0 + 0.1 * beta)
        for sub in range(80):
            if time.time() > t_end:
                break
            g = grad_weighted(h, w) * dx
            h_new = project(h - lr * g, n)
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            F_new = np.dot(w, corr_new) * dx
            F_old = np.dot(w, corr) * dx
            if F_new < F_old - 1e-17:
                h = h_new
                corr = corr_new
            else:
                lr *= 0.5
            if lr < 1e-12:
                break
        c5_v = compute_c5(h)
        if c5_v < best_c5 - 1e-14:
            best_c5 = c5_v
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0b = corr[c0_idx]
            best_v = old
            best_f = np.max(corr)
            for vt in (0.0, 1.0):
                cn = corr + (vt - old) * coeff
                cn[c0_idx] = c0b + (vt - old) * (1.0 - vt - old)
                ft = np.max(cn)
                if ft < best_f:
                    best_f = ft
                    best_v = vt
            if 0.01 < old < 0.99:
                def obj_f(v):
                    cn = corr + (v - old) * coeff
                    cn[c0_idx] = c0b + (v - old) * (1.0 - v - old)
                    return np.max(cn)
                try:
                    res = minimize_scalar(obj_f, bounds=(0.0, 1.0), method='bounded',
                                          options={'xatol': 1e-9})
                    if res.fun < best_f:
                        best_f = res.fun
                        best_v = res.x
                except Exception:
                    pass
            if best_f < best_c5 / dx - 1e-13:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved and sweep > 3:
            break
    return best_h, best_c5

def slsqp_weighted(h0, w, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        return np.dot(w, np.correlate(x, 1.0 - x, mode='full')) * dx
    def jac(x):
        return grad_weighted(x, w) * dx
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

def basin_hop(work_h, work_c5, n, t_end, max_iters=120):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.9 * it / max_iters)
        step = 0.02 + 0.12 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project(h_p, n)
        t_sub = min(time.time() + 3.0, t_end)
        hc, c5v = coord_descent(h_p, n, t_sub, max_sweeps=40)
        temp = 0.004 * cool + 0.0003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        ni = 160
        best_h = np.ones(ni) * 0.5
        for i in range(0, ni, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Coordinate descent warm start
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.10, max_sweeps=400)

    # Phase 2: Multiplicative weights (key new method)
    best_h, best_c5 = multiplicative_weights(best_h, n, t0 + budget_s * 0.28, max_rounds=50)

    # Phase 3: Coordinate descent polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=300)

    # Phase 4: Second MW round with different schedule
    best_h, best_c5 = multiplicative_weights(best_h, n, t0 + budget_s * 0.52, max_rounds=35, beta_min=30, beta_max=8000)

    # Phase 5: SLSQP on concentrated weights
    if elapsed() < budget_s * 0.62:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        cm = np.max(corr)
        w = np.exp(80.0 * (corr - cm))
        w /= w.sum()
        try:
            hc = slsqp_weighted(best_h, w, n, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 6: Basin hopping with random restarts
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.76, max_iters=100)

    # Phase 7: Another MW + CD cycle
    if elapsed() < budget_s * 0.88:
        best_h, best_c5 = multiplicative_weights(best_h, n, t0 + budget_s * 0.90, max_rounds=20)
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.93, max_sweeps=300)

    # Phase 8: Try higher resolutions
    for n_try in [200, 240, 320]:
        if elapsed() > budget_s * 0.94:
            break
        t_rem = t0 + budget_s * 0.99 - time.time()
        if t_rem < 5:
            break
        h_tr = interp_h(best_h, n_try)
        t_sub = t0 + budget_s * 0.96
        h_tr, c5_tr = coord_descent(h_tr, n_try, t_sub, max_sweeps=80)
        if elapsed() < budget_s * 0.98:
            h_tr, c5_tr = multiplicative_weights(h_tr, n_try, t0 + budget_s * 0.995, max_rounds=12)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
