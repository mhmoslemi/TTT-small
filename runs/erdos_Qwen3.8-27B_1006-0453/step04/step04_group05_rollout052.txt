For binary h (h ∈ {0,1}), the correlation at lag k is:
c(k) = Σ_i h[i] * (1-h[i+k]) for valid i

For the continuous case, the gradient of c(k) with respect to h[j] is:
- For lag k > 0: dc(k)/dh[j] = (1-h[j-k]) - h[j+k] (when those indices are valid)

- For lag k < 0: similar

    """Project h onto {x : 0<=x<=1, sum(x)=target} using binary search."""
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-10:

            lo = mid
        else:

def adam_smooth(h, N, target, deadline, k=30, lr0=0.03, decay=0.993, nesterov=True):
    """Adam with top-k gradient, projected constraint."""
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad = grad_top_k(h, N, k)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        if nesterov:

        peak = float(np.max(compute_corr(h)))
        if peak < best_peak:

            best_h = h.copy()
        if it % 300 == 0:

def anneal_binary(h, N, n_iters, rng, T0=2.0, T1=0.001, deadline=None):
    h = h.astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    
    best = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        T = T0 * (T1/T0) ** (it / max(n_iters, 1))
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h[i], h[j] = h[j], h[i]
        new_pk = float(np.max(compute_corr(h)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.001 and rng.random() < np.exp(-delta/T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_pk * (2.0/N)

        h[idx[nh:]] = 0

        h[idx[:nh - n_ones]] = 1

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    L_vals = np.arange(2 * N - 1) - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = L_vals[None, :]
    pd = p_idx + d_vals
    pmd = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (pmd >= 0) & (pmd < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N - 1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(pmd, 0, N - 1)], 0.0)
    return -valid_pd.astype(float) * (1.0 - h_pd) + valid_pmd.astype(float) * h_pmd

def steepest_binary(h, N, max_sweeps=400, deadline=None):
    h = h.astype(float).copy()
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_corr(h)
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D = compute_D(h, N)
        best_peak = P
        best_pq = None
        for p in ones:
            base = c - D[p]
            candidates = base[None, :] + D[zeros, :]
            new_peaks = candidates.max(axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_peak:
                best_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def grad_powersum(h, N, p=40):
    c = compute_corr(h)
    center = N - 1
    cp = np.power(np.maximum(c, 1e-15), p)
    cp_sum = cp.sum()
    if cp_sum < 1e-30:
        w = np.ones(2 * N - 1) / (2 * N - 1)
    else:
        w = p * np.maximum(c, 0) * cp / (cp_sum * p)
        w = w / max(w.sum(), 1e-30)
    L_vals = np.arange(2 * N - 1) - center
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        wk = w[idx]
        if wk < 1e-12:
            continue
        L = L_vals[idx]
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad

def adam_powersum(h, N, target, deadline, p=40, lr0=0.03, decay=0.993):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad = grad_powersum(h, N, p)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h_new, target)
        peak = float(np.max(compute_corr(h)))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        if it % 200 == 0:
            lr *= decay
    return best_h, best_peak * dx

def anneal_binary(h, N, n_iters, rng, T0=2.5, T1=0.002, deadline=None):

    best_pk = float(np.max(compute_corr(h)))
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        ifdeadline = deadline
        if deadline is not None and time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        if rng.random() < 0.05 and len(ones) > 5 and len(zeros) > 5:
            blen = rng.randint(2, 5)
            i0 = int(ones[rng.randint(max(1, len(ones) - blen))])
            j0 = int(zeros[rng.randint(max(1, len(zeros) - blen))])
            i0 = min(i0, N - blen - 1)
            j0 = min(j0, N - blen - 1)
            h[i0:i0+blen], h[j0:j0+blen] = h[j0:j0+blen], h[i0:i0+blen]
        else:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i], h[j] = h[j], h[i]
        new_pk = float(np.max(compute_corr(h)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.001 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            if rng.random() < 0.05:
                h[j0:j0+blen], h[i0:i0+blen] = h[i0:i0+blen], h[j0:j0+blen]
            else:
                h[i], h[j] = h[j], h[i]
    return best, best_pk * (2.0 / N)
