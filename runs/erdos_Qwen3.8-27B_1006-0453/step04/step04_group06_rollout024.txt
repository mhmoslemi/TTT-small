    #   - For k where i appears as j+k (j=i-k): corr[n-1+k] += val*(1-h[i-k]), k=0..i

        if k <= n1 - i:
            a_arr[idx] += -h[i+k]  # h[i+k]*(1-val) = h[i+k] - val*h[i+k]
            b_arr[idx] += h[i+k]  # constant part
    
    # Negative lags k = 1..n-1: corr[n1-k]
    for k in range(1, n):

        # h[i] as h[j] with j=i: need i+k <= n-1

    # So a_arr[n1] = (1-h[i]) - h[i] + h[i] = 1-h[i]...

    # Wait no. corr_base[n1] = sum_j 0*(1-h[j]) + ... hmm.

Let me also consider: the evaluate function computes C₅ = max(np.correlate(h, 1-h, mode="full") * dx). So it's the max over ALL lags, including the partial lags near the boundaries. This is important - it's not just the circular autocorrelation.

For the boundary lags (small overlap), the correlation is naturally smaller because fewer terms contribute. So the max is likely achieved at intermediate lags.

import numpy as np
from scipy.optimize import minimize_scalar, minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    if abs(s2 - target) > 1e-10 and s2 > 0:
        # Iterative projection
        for _ in range(5):
            diff = target - h.sum()
            if diff > 0:
                mask = h < 1.0
                if mask.sum() > 0:
                    h[mask] += diff / mask.sum()
            else:
                mask = h > 0
                if mask.sum() > 0:
                    h[mask] *= (target / h.sum())
            h = np.clip(h, 0, 1)
            if abs(h.sum() - target) < 1e-12:
                break
    return h

def corr_and_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, corr.max()

def coord_opt(h, n, i):
    """Minimize max correlation by adjusting h[i]."""
    h_save = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    n1 = n - 1
    # Build linear coefficients
    a = np.zeros(2*n-1)
    b = np.zeros(2*n-1)
    for k in range(n):
        if k <= i:
            a[n1+k] += (1.0 - h[i-k])
        if k + i < n:
            a[n1+k] -= h[i+k]
            b[n1+k] += h[i+k]
    for k in range(1, n):
        if k + i < n:
            a[n1-k] += (1.0 - h[i+k])
        if k <= i:
            a[n1-k] -= h[i-k]
            b[n1-k] += h[i-k]
    # Self-term: val*(1-val) at n1, we have a[n1]*val + b[n1] = (1-0)*val + 0 = val
    # Need val - val^2, so subtract val^2 in objective
    def f(v):
        c = corr0 + a*v + b
        c[n1] -= v*v
        return c.max()
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol':1e-10})
    h[i] = res.x
    return res.x

            break
        imp = False

            if time.time() > t_end:
                return h, best_c5
            old = h[i]
            coord_opt(h, n, i)
            if abs(h[i] - old) > 1e-9:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    imp = True
                else:
                    h[i] = old
        if not imp:
            break
    return h, best_c5

    target = n // 2
    b = np.zeros(n)
    idx = np.argsort(-h)
    b[idx[:target]] = 1.0

                    zeros = np.where(b == 0)[0]

            if improved:

        if not improved:

```python
import cvxpy as cp

def solve_lp_fix_other(h_fixed, n):
    """Solve: min t s.t. sum_j h[j]*(1-h_fixed[j+k]) <= t for all k, sum h = n/2, 0<=h<=1"""
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h.sum() == n/2, h >= 0, h <= 1]
    # For each lag k in -(n-1)..(n-1):
    one_minus = 1.0 - h_fixed
    for k in range(-(n-1), n):
        # I_k = sum_j h[j] * (1-h_fixed[j+k])
        # For k >= 0: sum_{j=0}^{n-1-k} h[j] * (1-h_fixed[j+k])
        # For k < 0 (m=-k): sum_{j=m}^{n-1} h[j] * (1-h_fixed[j-m])
        if k >= 0:
            coeffs = one_minus[k:]  # shape (n-k,)
            # I_k = h[:n-k] @ coeffs
            constraints.append(h[:n-k] @ coeffs <= t)
        else:
            m = -k
            coeffs = one_minus[:n-m]  # shape (n-m,)
            # I_m (negative lag) = h[m:] @ coeffs
            constraints.append(h[m:] @ coeffs <= t)
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)  # or GLPK, or CBC
    except:
        prob.solve()
    if h.value is None:
        return h_fixed.copy(), t.value if t.value is not None else compute_c5(h_fixed)
    return np.clip(h.value, 0, 1), t.value * (2.0/n)

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] *= (target / h[mask].sum())
        h = np.clip(h, 0, 1)
    return h

def coord_opt(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    n1 = n - 1
    a = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    for k in range(n):
        if k <= i:
            a[n1 + k] += (1.0 - h[i - k])
        if k + i < n:
            a[n1 + k] -= h[i + k]
            b[n1 + k] += h[i + k]
    for k in range(1, n):
        if k + i < n:
            a[n1 - k] += (1.0 - h[i + k])
        if k <= i:
            a[n1 - k] -= h[i - k]
            b[n1 - k] += h[i - k]
    def f(v):
        c = corr0 + a * v + b
        c[n1] -= v * v
        return c.max()
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    h[i] = res.x
    return res.x

def cd_sweep(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
