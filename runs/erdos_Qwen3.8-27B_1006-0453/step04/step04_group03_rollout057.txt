def compute_c5(h):
    n = len(h)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        # Need to reduce
        excess = s - target
        # Scale down values proportional to their current values
        mask = h > 0
        h[mask] -= h[mask] * (excess / s)
    else:
        # Need to increase
        deficit = target - s
        room = 1.0 - h
        tot_room = room.sum()
        if tot_room > 1e-14:
            h += room * (deficit / tot_room)
    return np.clip(h, 0.0, 1.0)

            lo = mid
        else:

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s > target:
        excess = s - target
        pos_mask = h > 1e-15
        if pos_mask.sum() == 0:
            return h
        # Water-filling: reduce from larger values first
        order = np.argsort(h[pos_mask])[::-1]
        indices = np.where(pos_mask)[0][order]
        remaining = excess
        for idx in indices:
            reduce = min(h[idx], remaining)
            h[idx] -= reduce
            remaining -= reduce
            if remaining < 1e-15:
                break
    else:
        deficit = target - s
        room_mask = h < 1.0 - 1e-15
        if room_mask.sum() == 0:
            return h
        order = np.argsort(1.0 - h[room_mask])[::-1]
        indices = np.where(room_mask)[0][order]
        remaining = deficit
        for idx in indices:
            add = min(1.0 - h[idx], remaining)
            h[idx] += add
            remaining -= add
            if remaining < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(20):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            excess = s - target
            # Proportionally reduce non-zero entries
            mask = h > 1e-15
            h[mask] -= h[mask] * (excess / s)
            h = np.clip(h, 0, 1)
        else:
            deficit = target - s
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += room * (deficit / tot)
            h = np.clip(h, 0, 1)
    return h

def solve_1d(a, b, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = a + mid * b
        mx = vals.max()
        active = vals >= mx - 1e-13
        bm = b[active]
        if len(bm) == 0:
            break
        if bm.min() > 1e-11:
            hi = mid
        elif bm.max() < -1e-11:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def cd_pairs(h, n, t_end, offset=0):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = corr.max() * dx
    idx_all = np.arange(-(n-1), n)
    
    for sweep in range(100000):
        if time.time() > t_end:
            break
        for j in range(n):
            k = (j + offset) % n
            if k == j:
                continue
            hj, hk = best_h[j], best_h[k]
            lo_t = -hj
            hi_t = 1.0 - hk
            if hi_t - lo_t < 1e-15:
                continue
            # Compute combined slope: diff[ℓ] = slope_j(ℓ) - slope_k(ℓ)
            # slope_j(ℓ) = (1-h[j+ℓ])_valid - h[j-ℓ]_valid
            # slope_k(ℓ) = (1-h[k+ℓ])_valid - h[k-ℓ]_valid
            diff = np.zeros(2*n-1)
            v1 = (j + idx_all >= 0) & (j + idx_all < n)
            diff[v1] += (1.0 - best_h[j + idx_all[v1]])
            v2 = (j - idx_all >= 0) & (j - idx_all < n)
            diff[v2] -= best_h[j - idx_all[v2]]
            v1 = (k + idx_all >= 0) & (k + idx_all < n)
            diff[v1] -= (1.0 - best_h[k + idx_all[v1]])
            v2 = (k - idx_all >= 0) & (k - idx_all < n)
            diff[v2] += best_h[k - idx_all[v2]]
            
            t = solve_1d(corr, diff, lo_t, hi_t)
            if abs(t) > 1e-14:
                best_h[j] += t
                best_h[k] -= t
                corr += t * diff
    
    best_c5 = corr.max() * dx
    return best_h, best_c5

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqq_polish(h, n, T, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0-x, mode='full'), T) * dx
    res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0}],
                   options={'maxiter':maxiter, 'ftol':1e-14})
    return project_to_sum(np.clip(res.x, 0, 1), n/2.0)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            excess = s - target
            mask = h > 1e-15
            if mask.sum() == 0:
                break
            h[mask] -= h[mask] * (excess / s)
        else:
            deficit = target - s
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += room * (deficit / tot)
    return np.clip(h, 0.0, 1.0)

def solve_1d(a, b, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) * 0.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a + mid * b
        mx = vals.max()
        active = vals >= mx - 1e-12
        bm = b[active]
        if bm.size == 0:
            break
        if bm.min() > 1e-10:
            hi = mid
        elif bm.max() < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def cd_pairs(h, n, t_end, offset=1):
    bh = h.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(bh, 1.0 - bh, mode='full').astype(float)
    best_c5 = corr.max() * dx
    idx_all = np.arange(-(n - 1), n)
    best_c5_initial = best_c5
    sweeps = 0
    while time.time() < t_end:
        sweeps += 1
        for j in range(n):
            k = (j + offset) % n
            if k == j:
                continue
            lo_t = -bh[j]
            hi_t = 1.0 - bh[k]
            if hi_t - lo_t < 1e-15:
                continue
            diff = np.zeros(2 * n - 1)
            v1 = (j + idx_all >= 0) & (j + idx_all < n)
            diff[v1] += (1.0 - bh[j + idx_all[v1]])
            v2 = (j - idx_all >= 0) & (j - idx_all < n)
            diff[v2] -= bh[j - idx_all[v2]]
            v1 = (k + idx_all >= 0) & (k + idx_all < n)
            diff[v1] -= (1.0 - bh[k + idx_all[v1]])
            v2 = (k - idx_all >= 0) & (k - idx_all < n)
            diff[v2] += bh[k - idx_all[v2]]
            t_opt = solve_1d(corr, diff, lo_t, hi_t)
            if abs(t_opt) > 1e-14:
                bh[j] += t_opt
                bh[k] -= t_opt
                corr += t_opt * diff
    best_c5 = corr.max() * dx
    return bh, best_c5

def smooth_max_adjoint(vals, T):
    m = np.max(vals)
    weights = np.exp(T * (vals - m))
    weights /= weights.sum()
    return m + np.log(weights.sum()) / T, vals, weights

def slsqp_polish(h, n, T, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        xc = np.clip(x, 0, 1)
        return smooth_max_adjoint(np.correlate(xc, 1.0 - xc, mode='full'), T)[0] * dx
    def obj_grad(x):
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode='full')
        m = corr.max()
        w = np.exp(T * (corr - m))
        w /= w.sum()
        grad = np.zeros(n)
        for lag in range(2 * n - 1):
            if w[lag] < 1e-10:
                continue
            k = lag - (n - 1)
            for i in range(n):
                j1 = i + k
                j2 = i - k
                if 0 <= j1 < n:
                    grad[j1] -= w[lag] * xc[j1]  # d/dx_i of x_i*(1-x_j) at i
                if 0 <= j2 < n:
                    grad[j2] -= w[lag] * xc[j2]
        return -dx * T * (wa_correction(corr, w, T))  # simplified
    # Use numeric gradient for reliability

                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
