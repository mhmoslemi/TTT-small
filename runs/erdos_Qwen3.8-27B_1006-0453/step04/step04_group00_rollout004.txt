import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.clip(h, 0, 1)
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def subgrad_step(h, n, alpha, topK):
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-topK:]
    grad = np.zeros(n)
    for m in top_idx:
        s = m - (n - 1)
        if s == 0:
            j = np.arange(n)
            grad[j] += (1.0 - 2.0 * h[j])
        else:
            j1_lo = max(0, -s)
            j1_hi = min(n - 1, n - 1 - s)
            if j1_hi >= j1_lo:
                js = np.arange(j1_lo, j1_hi + 1) + s
                grad[j1_lo:j1_hi + 1] += (1.0 - h[js])
            j2_lo = max(0, s)
            j2_hi = min(n - 1, n - 1 + s)
            if j2_hi >= j2_lo:
                js = np.arange(j2_lo, j2_hi + 1) - s
                grad[j2_lo:j2_hi + 1] -= h[js]
    grad *= dx / topK
    return grad

def momentum_opt(h, n, t_end, n_iter=2000):
    h_curr = h.copy()
    vel = np.zeros(n)
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    beta = 0.85
    base_alpha = 0.08
    for it in range(n_iter):
        if time.time() > t_end:
            break
        alpha = base_alpha / (1.0 + 0.005 * it)
        topK = max(2, min(15, 12 - it // 100))
        g = subgrad_step(h_curr, n, alpha, topK)
        vel = beta * vel - alpha * g
        h_new = project_h(h_curr + vel, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c5

def coord_opt_scalar(h, n, i):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coefs = np.zeros(n_corr)
    quad = 0.0
    for m in range(n_corr):
        s = m - (n - 1)
        if s == 0:
            coefs[m] = 1.0
            quad = -1.0
            continue
        c = 0.0
        if 0 <= i - s < n:
            c += (1.0 - h_base[i - s])
        if 0 <= i + s < n:
            c += (-h_base[i + s])
        coefs[m] = c
    def f1d(a):
        vals = corr_base + a * coefs
        if quad != 0.0:
            vals[n - 1] += quad * a * a
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = compute_corr(best_h)
        top_shifts = np.argsort(corr)[-6:]
        active = np.zeros(n, dtype=bool)
        for m in top_shifts:
            s = m - (n - 1)
            lo = max(0, -s)
            hi = min(n - 1, n - 1 - s)
            active[lo:hi + 1] = True
            lo2 = max(0, s)
            hi2 = min(n - 1, n - 1 + s)
            active[lo2:hi2 + 1] = True
        active_idx = np.where(active)[0]
        np.random.shuffle(active_idx)
        improved = False
        for i in active_idx:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_scalar(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-9:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_trials=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5
    for trial in range(n_trials):
        if time.time() > t_end:
            break
        cool = 1.0 - trial / n_trials
        choice = trial % 5
        if choice == 0:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif choice == 1:
            freq = np.random.randint(1, 10)
            phase = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif choice == 2:
            bs = np.random.randint(n // 10, n // 3)
            st = np.random.randint(0, n - bs)
            pert = np.zeros(n)
            pert[st:st + bs] = (0.05 + 0.15 * cool) * np.random.randn(bs)
        elif choice == 3:
            pert = np.zeros(n)
            idx = np.random.choice(n, size=max(4, n // 8), replace=False)
            pert[idx] = (0.08 + 0.12 * cool) * np.random.randn(len(idx))
        else:
            f1 = np.random.randint(2, 8)
            f2 = np.random.randint(10, 20)
            p1 = np.random.uniform(0, 2 * np.pi)
            p2 = np.random.uniform(0, 2 * np.pi)
            pert = 0.04 * np.sin(2 * np.pi * f1 * np.arange(n) / n + p1) + \
                   0.03 * np.cos(2 * np.pi * f2 * np.arange(n) / n + p2)
            pert += 0.02 * np.random.randn(n)
        h_pert = project_h(work_h + pert, n)
        sub_t = t_end
        h_opt, c5_opt = momentum_opt(h_pert, n, sub_t, n_iter=80)
        temp = 0.004 * cool + 0.0003
        if c5_opt < work_c5 or np.random.rand() < np.exp(-(c5_opt - work_c5) / temp):
            work_h = h_opt
            work_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    if work_c5 < best_c5:
        best_h, best_c5 = work_h, work_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
