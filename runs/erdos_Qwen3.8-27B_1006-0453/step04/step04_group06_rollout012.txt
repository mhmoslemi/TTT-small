from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s >= target:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            h = np.full(n, target / n)
        else:
            h = h + room * ((target - s) / tot)
            h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # Spread correction over all elements proportional to room
        room = 1.0 - h
        if diff > 0:
            tot = room.sum()
            if tot > 0:
                h += room * (diff / tot)
        else:
            h *= (target / h.sum()) if h.sum() > 0 else 1
        h = np.clip(h, 0.0, 1.0)
    return h
