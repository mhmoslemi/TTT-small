def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, n_sweeps=500, xatol=1e-13):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    k_range = np.arange(-(n-1), n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            j1 = i - k_range
            j2 = i + k_range
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2*n-1)
            coef[v1] += 1.0 - best_h[j1[v1]]
            coef[v2] += -best_h[j2[v2]]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            def f1d(s):
                return np.max(corr + s * coef)
            f0 = f1d(0.0)
            if f1d(lo) >= f0 and f1d(hi) >= f0:
                # Check if interior minimum exists
                # Gradient at 0: sign of coef at argmax
                am = np.argmax(corr)
                if coef[am] == 0:
                    continue
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': xatol})
            s_new = res.x
            if abs(s_new) > 1e-14:
                best_h[i] = h_old + s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_dir(h, corr, n, temperature=5.0):
    max_val = np.max(corr)
    w = np.exp(temperature * (corr - max_val))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    js = np.arange(n)
    for m in range(2*n-1):
        if w[m] < 1e-5:
            continue
        k = m - (n-1)
        term = np.zeros(n)
        jp = js - k
        jm = js + k
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        term[vp] = 1.0 - h[jp[vp]]
        term[vm] -= h[jm[vm]]
        grad += w[m] * term
    return grad

def slsqp_polish(h, T, n, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0-x, mode='full')
        m = np.max(c)
        return m + np.log(np.sum(np.exp(T*(c-m))))/T * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n/2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

        best_h[:64] = 1.0
    n = len(best_h)
    best_h = project_to_sum(best_h, n/2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive coordinate descent (45% budget)
    t1 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, t1, n_sweeps=1000, xatol=1e-13)

    # Phase 2: Basin hopping with subgradient-guided perturbations (30% budget)
    t2 = t0 + budget_s * 0.75
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iters = 50
    for it in range(n_iters):
        if elapsed() > budget_s * 0.73:
            break
        cool = max(1.0 - 0.95*(it/float(n_iters)), 0.03)
        r = np.random.rand()
        if r < 0.25:
            # Subgradient-guided step
            corr_w = compute_corr(work_h)
            sg = subgrad_dir(work_h, corr_w, n, temperature=3.0+5.0*cool)
            step_sz = 0.01 + 0.04*cool
            h_p = work_h - step_sz * sg
        elif r < 0.45:
            # Random smooth perturbation
            freqs = np.random.uniform(1, 6, 4)
            phases = np.random.uniform(0, 2*np.pi, 4)
            amp = 0.02 + 0.06*cool
            pert = np.zeros(n)
            for f, p in zip(freqs, phases):
                pert += amp * np.sin(2*np.pi*f*np.arange(n)/n + p)
            h_p = work_h + pert
        elif r < 0.65:
            # Localized block perturbation
            bs = np.random.randint(8, 25)
            st = np.random.randint(0, max(1, n-bs))
            h_p = work_h.copy()
            h_p[st:st+bs] += np.random.uniform(-0.08, 0.08, bs)
        elif r < 0.80:
            # Cyclic shift blend
            sh = np.random.randint(-2, 3)
            h_p = 0.75*np.roll(work_h, sh) + 0.25*work_h
        else:
            # Global random
            step = 0.02 + 0.08*cool
            h_p = work_h + step * np.random.randn(n)
        h_p = project_to_sum(h_p, n/2)
        hc, c5v = coord_descent(h_p, n, time.time()+1.5, n_sweeps=25, xatol=1e-11)
        temp = 0.002*cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v-work_c5)/max(temp,1e-12)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:

    for n_try in [192, 256, 200]:
        if n_try == n or elapsed() > budget_s * 0.85:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try/2)
        hc, c5v = coord_descent(h_i, n_try, time.time()+6.0, n_sweeps=80, xatol=1e-12)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 4: SLSQP polish (5% budget)

            if elapsed() > budget_s * 0.95:

                if c5v < best_c5:

    # Phase 5: Final coordinate descent (remaining budget)

    best_h, best_c5 = coord_descent(best_h, n, t_end, n_sweeps=500, xatol=1e-14)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
