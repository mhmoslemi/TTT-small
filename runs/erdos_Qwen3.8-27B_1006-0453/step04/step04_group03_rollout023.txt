```python
    m = corr.max()
```

```python
import cvxpy as cp

def lp_subgradient_step(h, n, M=10):
    """One cutting-plane iteration."""
    dx = 2.0 / n
    corr = np.array(np.correlate(h, 1.0 - h, mode='full'))
    
    # Select top M lags
    top_idx = np.argsort(corr)[-M:].astype(int)
    
    h_var = cp.Variable(n)
    z = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    for idx in top_idx:
        d = idx - (n - 1)
        g_val = corr[idx] * dx
        # Gradient of g_d at h
        grad = np.zeros(n)
        js = max(0, -d)
        je = min(n, n - d)
        if js < je:
            grad[js:je] += (1.0 - h[js+d:je+d])
        js2 = max(0, d)
        je2 = min(n, n + d)
        if js2 < je2:
            grad[js2:je2] -= h[js2-d:je2-d]
        # Linearized: g_val + dx * grad @ (h_var - h) ≤ z
        constraints.append(g_val + dx * (grad @ (h_var - h)) <= z)
    
    obj = cp.Minimize(z)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or CPLEX, GUROBI, etc.
    
    if h_var.value is not None:
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_to_sum(h_new, n / 2)
        return h_new
    return h
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def vec_coef(h, i, n):
    d = np.arange(-(n-1), n)
    j1 = i + d
    j2 = i - d
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coef = np.zeros(2*n - 1)
    coef[v1] += (1.0 - h[j1[v1]])
    coef[v2] -= h[j2[v2]]
    return coef

def min_max_bisect(corr, coef, lo, hi, n_iter=25):
    for _ in range(n_iter):
        mid = (lo + hi) * 0.5
        vals = corr + mid * coef
        slope = coef[vals.argmax()]
        if slope > 0:
            hi = mid
        elif slope < 0:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def coord_descent(h, t_end, n_sweeps=1000):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.array(np.correlate(best_h, 1.0 - best_h, mode='full'), dtype=float)
    best_c5 = corr.max() * dx
    d_arr = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h_old = best_h[i]
            j1 = i + d_arr
            j2 = i - d_arr
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2*n - 1)
            coef[v1] += (1.0 - best_h[j1[v1]])
            coef[v2] -= best_h[j2[v2]]
            
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-12:
                continue
            s_new = min_max_bisect(corr, coef, lo, hi)
            new_val = h_old + s_new
            new_val = np.clip(new_val, 0.0, 1.0)
            s_actual = new_val - h_old
            if abs(s_actual) > 1e-13:
                corr += s_actual * coef
                c5_new = corr.max() * dx
                if c5_new < best_c5:
                    best_h[i] = new_val
                    best_c5 = c5_new
                    improved = True
                else:
                    corr -= s_actual * coef
        if not improved:
            break
    
    return best_h, best_c5

def soft_grad(h, T):
    n = len(h)
    corr = np.array(np.correlate(h, 1.0 - h, mode='full'), dtype=float)
    m = corr.max()
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        d = idx - (n-1)
        js = max(0, -d)
        je = min(n, n - d)
        if js < je:
            grad[js:je] += wi * (1.0 - h[js+d:je+d])
        js2 = max(0, d)
        je2 = min(n, n + d)
        if js2 < je2:
            grad[js2:je2] -= wi * h[js2-d:je2-d]
    return grad, corr

def subgrad_loop(h, t_end, n_steps=500):
    n = len(h)
    target = n / 2.0
    hw = h.copy()
    v = np.zeros(n)
    for T in [20, 50, 150, 500, 1500]:
        if time.time() > t_end:
            break
        alpha = 0.05
        for step in range(n_steps // 5):
            if time.time() > t_end:
                return hw
            grad, _ = soft_grad(hw, T)
            v = 0.7 * v + 0.3 * grad
            hw = project_to_sum(hw - alpha * v, target)
            alpha *= 0.995
    return hw

    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0
    
    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)
    t_end = t0 + budget_s
    
    # Phase 1: Subgradient (15%)
    t1 = t0 + budget_s * 0.15
    h_sg = subgrad_loop(best_h, t1, n_steps=500)
    h_sg, c5v = coord_descent(h_sg, t0 + 3, n_sweeps=20)
    if c5v < best_c5:
        best_c5, best_h = c5v, h_sg
    
    # Phase 2: Heavy coordinate descent (35%)
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, t2, n_sweeps=2000)
    
    # Phase 3: Basin hopping (30%)
    t3 = t0 + budget_s * 0.80
    h_bh = best_h.copy()
    c5_bh = best_c5
    n_iter_bh = 200
    for it in range(n_iter_bh):
        if time.time() > t3:
            break
        cool = max(1.0 - 0.95 * (it / float(n_iter_bh)), 0.05)
        r = np.random.rand()
        if r < 0.25:
            step = 0.01 + 0.08 * cool
            h_p = h_bh + step * np.random.randn(n)
        elif r < 0.50:
            # Sparse perturbation
            k = max(1, int(n * 0.1 * cool))
            mask = np.zeros(n, dtype=bool)
            mask[np.random.choice(n, k, replace=False)] = True
            step = 0.02 + 0.1 * cool
            h_p = h_bh.copy()
            h_p[mask] += step * np.random.randn(k)
        elif r < 0.70:
            # Shift perturbation
            s = int(np.random.randint(-5, 6))
            h_p = np.roll(h_bh, s)
            h_p = 0.8 * h_p + 0.2 * h_bh
        elif r < 0.85:
            # Binary push
            h_p = h_bh.copy()
            mask = np.random.rand(n) < 0.05
            for idx in np.where(mask)[0]:
                if h_p[idx] > 0.5:
                    h_p[idx] = min(1.0, h_p[idx] + 0.2)
                else:
                    h_p[idx] = max(0.0, h_p[idx] - 0.2)
        else:
            # Antisymmetric perturbation
            h_p = h_bh.copy()
            mask = np.random.rand(n) < 0.1
            h_p[mask] = 1.0 - h_p[mask]

        # Quick polish
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def min_max_bisect(corr, coef, lo, hi, n_iter=25):
    for _ in range(n_iter):
        mid = (lo + hi) * 0.5
        vals = corr + mid * coef
        slope = coef[vals.argmax()]
        if slope > 0:
            hi = mid
        elif slope < 0:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, t_end, n_sweeps=1000):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.array(np.correlate(best_h, 1.0 - best_h, mode='full'), dtype=float)
    best_c5 = corr.max() * dx
    d_arr = np.arange(-(n-1), n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h_old = best_h[i]
            j1 = i + d_arr
            j2 = i - d_arr
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2*n - 1)
            coef[v1] += (1.0 - best_h[j1[v1]])
            coef[v2] -= best_h[j2[v2]]
            lo, hi = -h_old, 1.0 - h_old
            if hi - lo < 1e-12:
                continue
            s_new = min_max_bisect(corr, coef, lo, hi)
            nv = np.clip(h_old + s_new, 0.0, 1.0)
            sa = nv - h_old
            if abs(sa) > 1e-13:
                corr += sa * coef
                if corr.max() * dx < best_c5:
                    best_h[i] = nv
                    best_c5 = corr.max() * dx
                    improved = True
                else:
                    corr -= sa * coef
        if not improved:
            break
    return best_h, best_c5

def soft_grad(h, T):
    n = len(h)
    corr = np.array(np.correlate(h, 1.0 - h, mode='full'), dtype=float)
    m = corr.max()
    exps = np.exp(np.clip(T * (corr - m), -50, 50))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        d = idx - (n-1)
        js, je = max(0, -d), min(n, n - d)
        if js < je:
            grad[js:je] += wi * (1.0 - h[js+d:je+d])
        js2, je2 = max(0, d), min(n, n + d)
        if js2 < je2:
            grad[js2:je2] -= wi * h[js2-d:je2-d]
    return grad, corr

def subgrad_loop(h, t_end, n_total=600):
    n = len(h)
    target = n / 2.0
    hw = h.copy()
    v = np.zeros(n)
    best_hw_c5 = compute_c5(hw)
    T_schedule = [20, 50, 150, 500, 1500]
    steps_per_T = n_total // len(T_schedule)
    for T in T_schedule:
        if time.time() > t_end:
            break
        alpha = 0.04
        for _ in range(steps_per_T):

                return hw, best_hw_c5
            grad, _ = soft_grad(hw, T)
            v = 0.75 * v + 0.25 * grad
            hw = project_to_sum(hw - alpha * v, target)
            alpha *= 0.995
            c5v = compute_c5(hw)
            if c5v < best_hw_c5:
                best_hw_c5 = c5v
    return hw, best_hw_c5

    return best_h, best_c5, len(best_h)
```
