import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.sum(np.exp(np.clip(T*(corr - m), -500, 500)))
    return (m + np.log(s)/T) * dx

def soft_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = np.exp(np.clip(T*(corr - m), -500, 500))
    p = e / e.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        pk = p[k]
        if pk < 1e-13:
            continue
        lag = k - (n-1)
        if lag >= 0:
            if n > lag:
                g[:n-lag] += pk * (1.0 - h[lag:n])
                g[lag:] += pk * (-h[:n-lag])
        else:
            nl = -lag
            if n > nl:
                g[nl:] += pk * (1.0 - h[:n-nl])
                g[:n-nl] += pk * (-h[nl:])
    return g * dx

def slsqp(h0, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return soft_max_c5(x, T)
    def con(x):
        return x.sum() - n/2.0
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':con}],
                   options={'maxiter':maxiter, 'ftol':ftol})
    return proj(np.clip(res.x, 0, 1), n)
