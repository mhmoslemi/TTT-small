import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14:
            h = np.clip(h * (target / max(s2, 1e-14)), 0.0, 1.0)
    return h

def _eval1d(t, corr_base, d, c0, dx):
    mid = len(corr_base) // 2
    vals = corr_base + t * d
    vals[mid] = c0 + t * (1.0 - t)
    return np.max(vals) * dx

def coord_opt_fixed(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 + k] += (1.0 - h[i - k])
            d[n - 1 - k] += (-h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
            d[n - 1 - k] += (1.0 - h[i + k])
    c0 = corr_base[n - 1]
    dx = 2.0 / n
    res = minimize_scalar(_eval1d, bounds=(0.0, 1.0), method='bounded',

    return res.x

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):

        order = rng.permutation(n)
        for i in order:

            hi = coord_opt_fixed(best_h, n, i)

    best_c5 = compute_c5(best_h)
