    """Minimize convex f on [lo, hi] using golden section."""
    invphi = (np.sqrt(5) - 1) / 2
    invphi2 = invphi * invphi
    a, b = lo, hi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    while b - a > tol:
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)
    return (a + b) / 2, f((a+b)/2)
