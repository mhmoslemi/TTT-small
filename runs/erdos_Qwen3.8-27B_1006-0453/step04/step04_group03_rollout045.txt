        h_cur = h_cur - step * grad
        h_cur = project_to_sum(h_cur, n/2)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqp_step(h, T, n, maxiter=600, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def coord_descent(h, t_end, n_sweeps=8000):
    n = len(h)
    best_h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = np.max(corr) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        if sweep > 0 and sweep % 200 == 0:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            best_c5 = np.max(corr) * dx

        start = sweep % n
        for si in range(n):
            i = (start + si) % n
            h_old = best_h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue

            coef = np.zeros(2 * n - 1)
            klo = max(0, -(n - 1))
            khi = min(n - 1, n - 1)
            for k in range(max(-i, -(n-1)), min(n-1-i, n-1) + 1):
                coef[n - 1 + k] += (1.0 - best_h[i + k])
            for k in range(max(i - n + 1, -(n-1)), min(i, n-1) + 1):
                coef[n - 1 + k] -= best_h[i - k]

            a, b = lo, hi
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = np.max(corr + c * coef)
            fd = np.max(corr + d * coef)
            for _ in range(35):

                    c = b - gr * (b - a)
                    fc = np.max(corr + c * coef)
                else:
                    a, c, fc = c, d, fd
                    d = a + gr * (b - a)
                    fd = np.max(corr + d * coef)
            s_new = (a + b) / 2.0
            if abs(s_new) > 1e-14:
                best_h[i] += s_new
                corr += s_new * coef

    return best_h, np.max(corr) * dx
