    # Phase 2: Long coordinate descent (33% budget)
    t_cd2 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, t_cd2, max_sweeps=8000, seed_cd=42)

    # Phase 3: Diversification with pairwise (28% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    t_phase3 = t0 + budget_s * 0.73
    n_div = 25
    for it in range(n_div):
        if time.time() > t_phase3:
            break
        cool = 1.0 - it / float(n_div)
        # Diversify strategies
        if it % 5 == 0:
            k = max(4, n // 15)
            idx = np.random.choice(n, k, replace=False)
            work_h[idx] += np.random.uniform(-0.08, 0.08, k) * (0.3 + 0.7 * cool)
        elif it % 5 == 1:
            # Mass redistribution
            hi = np.argsort(work_h)[-6:]
            lo = np.argsort(work_h)[:6]
            for a, b in zip(hi, lo):
                amt = min(0.04 * cool, (work_h[a] - work_h[b]) * 0.25)
                if amt > 0:
                    work_h[a] -= amt
                    work_h[b] += amt
        elif it % 5 == 2:
            # Flip a small block
            blen = max(2, n // 40)
            start = np.random.randint(0, n - blen)
            work_h[start:start+blen] = 1.0 - work_h[start:start+blen]
        else:
            work_h += np.random.randn(n) * 0.008 * cool

        work_h = project(work_h, n)

        # SLSQP refine
        try:
            T_s = int(300 + 8000 * cool)
            hc = slsqp_opt(work_h.copy(), T_s, n, maxiter=500)
        except Exception:
            hc = work_h
        # CD polish
        hc, c5v = coord_descent(hc, n, min(t_phase3, time.time() + budget_s * 0.018),
                                max_sweeps=150, use_pairs=(it % 2 == 0), seed_cd=100+it)

        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-max(0.0, c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Pairwise-heavy CD (10% budget)
    t_cd4 = t0 + budget_s * 0.83
    best_h, best_c5 = coord_descent(best_h, n, t_cd4, max_sweeps=3000, use_pairs=True, seed_cd=99)

    # Phase 5: Final SLSQP + CD (17% budget)
    for T in [30000, 150000]:
        if elapsed() > budget_s * 0.94:
            break
        try:
            hc = slsqp_opt(best_h.copy(), T, n, maxiter=2500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=3000, seed_cd=777)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
