import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def cd_fast(h, n, t_end, n_sweeps=200):
    best_h = h.copy()
    dx = 2.0 / n
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(lambda s: np.max(corr + s * coef),
                                  bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-13})
            s_new = res.x
            if abs(s_new) > 1e-14:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def slsqp_level(h, n, active_idx, t_end):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        ac = c[active_idx]
        mean_a = np.mean(ac)
        return mean_a + 0.5 * np.sum((ac - mean_a) ** 2) * len(ac) + 0.005 * (np.max(c) - mean_a) ** 2
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n, constraints=cons,
                   options={'maxiter': 800, 'ftol': 1e-15})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def interp_h(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)
