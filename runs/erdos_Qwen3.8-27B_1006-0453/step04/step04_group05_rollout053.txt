```python
    # For L = 0: sum_n h[n]*(1-h[n])
    
    # ∂c/∂h[p]:
    # L > 0: (1-h[p-L]) if p>=L, minus h[p-L]... wait
    # c = sum_{n=L}^{N-1} h[n]*(1-h[n-L])
    # ∂/∂h[p]: 
    #   as h[n]: (1-h[p-L]) if L<=p<=N-1
    #   as h[n-L]: -h[p+L] if 0<=p+L- L... n-L=p means n=p+L, valid if L<=p+L<=N-1 i.e. 0<=p<=N-1-L
    #   So: (1-h[p-L])*[p>=L] - h[p+L]*[p<=N-1-L]
    
    # L < 0, M = -L:
    # c = sum_{n=0}^{N-1-M} h[n]*(1-h[n+M])
    # ∂/∂h[p]:
    #   as h[n]: (1-h[p+M]) if 0<=p<=N-1-M
    #   as h[n+M]: -h[p-M] if p-M>=0 and p-M<=N-1-M i.e. M<=p<=N-1
    #   So: (1-h[p+M])*[p<=N-1-M] - h[p-M]*[p>=M]
    
    # L = 0: 1-2h[p]
    
    D = np.zeros((N, 2*N-1))
    center = N - 1
    for L in range(-N+1, N):
        k = center + L
        if L == 0:
            D[:, k] = 1.0 - 2.0 * h
        elif L > 0:
            D[L:, k] += (1.0 - h[:N-L])  # h[p-L] for p in [L, N-1]
            D[:N-L, k] -= h[L:]  # h[p+L] for p in [0, N-1-L]
        else:
            M = -L
            D[:N-M, k] += (1.0 - h[M:])  # h[p+M] for p in [0, N-1-M]
            D[M:, k] -= h[:N-M]  # h[p-M] for p in [M, N-1]
    return D
```

```python
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):

def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

def grad_single(h, L, N):
    """Gradient of correlation at lag L w.r.t. h"""
    center = N - 1
    g = np.zeros(N)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:N-L])
        g[:N-L] -= h[L:]
    else:
        M = -L
        if M < N:
            g[:N-M] += (1.0 - h[M:])
            g[M:] -= h[:N-M]
    return g

def grad_topk(h, corr, k_top, N):
    center = N - 1
    total = 2 * N - 1
    if k_top >= total:
        top = np.arange(total)
    else:
        top = np.argpartition(corr, total - k_top)[-k_top:]
    grad = np.zeros(N)
    for ai in top:
        L = int(ai) - center
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N-L])
            grad[:N-L] -= h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += (1.0 - h[M:])
                grad[M:] -= h[:N-M]
    return grad / k_top

def binary_steepest(h, N, deadline, max_sweeps=300):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh-n_ones]] = 1
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Compute D for all ones and zeros positions
        # D[p,k] = grad of corr[k] w.r.t. h[p]
        center = N - 1
        best_pk = P
        best_pq = None
        for p in ones:
            g_p = np.zeros(2*N-1)
            for k in range(2*N-1):
                L = k - center
                if L == 0:
                    val = 1.0 - 2.0*h[p]
                elif L > 0:
                    val = 0.0
                    if p >= L: val += 1.0 - h[p-L]
                    if p <= N-1-L: val -= h[p+L]
                else:
                    M = -L
                    val = 0.0
                    if p <= N-1-M: val += 1.0 - h[p+M]
                    if p >= M: val -= h[p-M]
                g_p[k] = val
            base = corr - g_p
            for q in zeros:
                g_q = np.zeros(2*N-1)
                for k in range(2*N-1):
                    L = k - center
                    if L == 0:
                        val = 1.0 - 2.0*h[q]
                    elif L > 0:
                        val = 0.0
                        if q >= L: val += 1.0 - h[q-L]
                        if q <= N-1-L: val -= h[q+L]
                    else:
                        M = -L
                        val = 0.0
                        if q <= N-1-M: val += 1.0 - h[q+M]
                        if q >= M: val -= h[q-M]
                    g_q[k] = val
                new_pk = float(np.max(base + g_q))
                if new_pk < best_pk:
                    best_pk = new_pk
                    best_pq = (p, q)
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h
```

Wait, this is way too slow - O(n_ones * n_zeros * N) per sweep with Python loops. The previous code vectorized this with numpy. Let me go back to the D-matrix approach.

Let me precompute the D matrix more efficiently. Actually, the D matrix can be computed as:

For each position p, D[p, k] is the gradient of corr[k] w.r.t. h[p]. This is the same as: if we add a small ε to h[p], how does corr[k] change?

Actually, I realize the D matrix computation can be done very efficiently:

D[p, :] for a fixed p:
- D[p, N-1] = 1-2h[p] (lag 0)
- For L > 0 (index k = N-1+L): D[p, k] = (1-h[p-L])*[p>=L] - h[p+L]*[p<=N-1-L]
- For L < 0 (index k = N-1+L, M=-L): D[p, k] = (1-h[p+M])*[p<=N-1-M] - h[p-M]*[p>=M]

This is O(N) per p, so O(N^2) for all p. For N=380 that's 144K - very fast.

Let me vectorize:

```python
def compute_D(h, N):
    D = np.zeros((N, 2*N-1))
    center = N - 1
    D[:, center] = 1.0 - 2.0 * h
    # For L > 0: k = center + L, L in [1, N-1]
    for L in range(1, N):
        k = center + L
        # D[p, k] = (1-h[p-L]) for p in [L, N-1], minus h[p+L] for p in [0, N-1-L]
        D[L:, k] += (1.0 - h[:N-L])
        D[:N-L, k] -= h[L:]
        # For L (negative): k = center - L
        k2 = center - L
        D[:N-L, k2] += (1.0 - h[L:])
        D[L:, k2] -= h[:N-L]
    return D
```

This is O(N^2) total, fast.

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    D = np.zeros((N, 2 * N - 1))
    center = N - 1
    D[:, center] = 1.0 - 2.0 * h
    for L in range(1, N):
        k1 = center + L
        D[L:, k1] += (1.0 - h[:N - L])
        D[:N - L, k1] -= h[L:]
        k2 = center - L
        D[:N - L, k2] += (1.0 - h[L:])
        D[L:, k2] -= h[:N - L]
    return D

def grad_topk(h, corr, k_top, N):
    center = N - 1
    total = 2 * N - 1
    if k_top >= total:
        top = np.arange(total)
    else:
        top = np.argpartition(corr, total - k_top)[-k_top:]
    grad = np.zeros(N)
    for ai in top:
        L = int(ai) - center
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
    return grad / float(k_top)

def binary_steepest(h_in, N, deadline, max_sweeps=400):
    h = (h_in > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n1]] = 1
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D = compute_D(h, N)
        D_z = D[zeros, :]
        best_pk = P
        best_pq = None
        for p in ones:
            base = corr - D[p, :]
            np_peaks = (base[None, :] + D_z).max(axis=1)
            mi = int(np.argmin(np_peaks))
            if np_peaks[mi] < best_pk:
                best_pk = float(np_peaks[mi])
                best_pq = (p, zeros[mi])
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def two_swap_binary(h, N, deadline, n_iters=2000, rng=None):
    h = h.astype(float).copy()
    nh = N // 2
    if rng is None:
        rng = np.random.RandomState(0)
    corr = compute_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        p1, p2 = ones[rng.randint(len(ones), size=2)]
        q1, q2 = zeros[rng.randint(len(zeros), size=2)]
        if p1 == p2 or q1 == q2:
            continue
        h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
        new_pk = float(np.max(compute_corr(h)))
        if new_pk < best_pk:
            best_pk = new_pk
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[p1] = 1; h[p2] = 1; h[q1] = 0; h[q2] = 0
    return best_h, best_pk

def continuous_adam(h0, N, target, deadline, k_start=60, k_end=1, lr0=0.035, lr_end=0.003):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    t_start = time.time()
    total_time = max(1.0, deadline - t_start)
    while time.time() < deadline - 0.5:
        frac = min(1.0, (time.time() - t_start) / total_time)
        k_top = max(k_end, int(k_start * (1 - frac) + k_end * frac))
        lr = lr0 * (1 - frac) + lr_end * frac
        corr = compute_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = grad_topk(h, corr, k_top, N)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_pk * (2.0 / N)

def make_binary(N, pattern, rng):
    if pattern == 'alt':
        h = np.zeros(N)
        h[::2] = 1.0
    elif pattern == 'blocks':
        h = np.zeros(N)
        h[:N // 2] = 1.0
    elif pattern == 'trip':
        h = np.zeros(N)
        per = N // 3
        h[0:per] = 1.0
        h[2 * per:] = 1.0
    else:
        h = (rng.random(N) > 0.5).astype(float)
    n1 = int(h.sum())
    nh = N // 2
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n1]] = 1
    return h

    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
```
