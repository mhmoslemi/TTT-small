def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full').astype(np.float64)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def _1d_min(corr, coef, lo, hi):
    """Minimize max_j(corr[j] + s*coef[j]) over s in [lo, hi]"""
    def fv(s):
        return np.max(corr + s * coef)
    best_s, best_v = lo, fv(lo)
    vh = fv(hi)
    if vh < best_v:
        best_s, best_v = hi, vh
    a, b = lo, hi
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if fv(m1) < fv(m2):
            b = m2
        else:
            a = m1
    s_mid = 0.5 * (a + b)
    vm = fv(s_mid)
    if vm < best_v:
        best_s, best_v = s_mid, vm
    return best_s

def coord_descent(h, n, t_end, n_sweeps=500):
    bh = h.copy()
    bc5 = compute_c5(bh)
    dx = 2.0 / n
    corr = compute_corr(bh)
    for _ in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return bh, bc5
            ho = bh[i]
            # Compute coefficient vector
            m = np.arange(2 * n - 1)
            k = m - (n - 1)
            j1 = i - k
            j2 = i + k
            coef = np.zeros(2 * n - 1)
            mk1 = (j1 >= 0) & (j1 < n)
            mk2 = (j2 >= 0) & (j2 < n)
            coef[mk1] += 1.0 - bh[j1[mk1]]
            coef[mk2] -= bh[j2[mk2]]
            lo, hi = -ho, 1.0 - ho
            if hi - lo < 1e-14:
                continue
            sn = _1d_min(corr, coef, lo, hi)
            if abs(sn) > 1e-13:
                bh[i] += sn
                corr += sn * coef
                cn = np.max(corr) * dx
                if cn < bc5 - 1e-15:
                    bc5 = cn
                    improved = True
        if not improved:
            break
    return bh, bc5

def slsqp(h, T, n, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)

from scipy.optimize import minimize
import time

class _SC:
    def __init__(self, n):
        self.t = n / 2.0
    def __call__(self, x):
        return x.sum() - self.t

class _Obj:
    def __init__(self, T, dx):
        self.T = T
        self.dx = dx
    def __call__(self, x):
        x = np.clip(x, 0.0, 1.0)
        return smax(np.correlate(x, 1.0 - x, mode='full'), self.T) * self.dx

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-15:
        return h
    if d > 0:
        r = (1.0 - h).sum()
        if r < 1e-15:
            return h
        h = h + (1.0 - h) * (d / r)
    else:
        s = h.sum()
        if s < 1e-15:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def smax(v, T):
    m = v.max()
    return m + np.log(np.sum(np.exp(T * (v - m)))) / T
