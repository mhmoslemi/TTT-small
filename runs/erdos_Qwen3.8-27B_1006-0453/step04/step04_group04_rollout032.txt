            #   contribution: (1 - h[j-nl])
            # Second term: j-(-nl) = j+nl in [0,n) and j in [0,n) → j in [0, n-1-nl]

            i2_start, i2_end = 0, n - nl
            if i2_end > 0:
                grad[i2_start:i2_end] += pk * (-h[nl:])
    
    return grad * dx

```python
def fast_coord_opt(h, n, i):
    """Optimal value for h[i] given all other h fixed."""
    dx = 2.0 / n
    # Compute current correlations
    curr = np.correlate(h, 1.0 - h, mode='full')
    # Save h[i]
    t = h[i]
    # Compute coefficients c_k
    c = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        val = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        if 0 <= j2 < n:
            val -= h[j2]
        c[k] = val
    # corr[k] changes from curr[k] to curr[k] + (t'-t)*c[k]
    # Minimize max_k (curr[k] - t*c[k] + t'*c[k]) over t' in [0,1]
    # = min_{t' in [0,1]} max_k (A[k] + t'*c[k])
    # where A[k] = curr[k] - t*c[k]
    A = curr - t * c
    # Find optimal t'
    # The function f(t') = max_k (A[k] + t'*c[k]) is convex in t'
    # Minimum is where the derivative changes sign
    # f'(t') = c[k*] where k* = argmax (A[k] + t'*c[k])
    # At optimum: min_k c[k] for active k <= 0 <= max_k c[k] for active k
    # Or at boundary t'=0 or t'=1
    
    # Simple approach: ternary search on [0,1]
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(A + m1 * c) * dx
        f2 = np.max(A + m2 * c) * dx
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2
```

def compute_c(i, h, n):
    """Compute the coefficient vector c for changing h[i]."""
    c = np.zeros(2 * n - 1)
    lags = np.arange(-(n-1), n)
    # j1 = i + lag, valid if 0 <= j1 < n
    j1 = i + lags
    v1 = (j1 >= 0) & (j1 < n)
    c[v1] += (1.0 - h[j1[v1]])
    # j2 = i - lag, valid if 0 <= j2 < n
    j2 = i - lags
    v2 = (j2 >= 0) & (j2 < n)
    c[v2] -= h[j2[v2]]
    return c

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_set(x, n):
    target = n / 2.0
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c_single(i, h, n):
    lags = np.arange(-(n - 1), n)
    c = np.zeros(2 * n - 1)
    j1 = i + lags
    v1 = (j1 >= 0) & (j1 < n)
    c[v1] += (1.0 - h[j1[v1]])
    j2 = i - lags
    v2 = (j2 >= 0) & (j2 < n)
    c[v2] -= h[j2[v2]]
    return c

def compute_all_c(h, n):
    c = np.zeros((n, 2 * n - 1))
    lags = np.arange(-(n - 1), n)
    for i in range(n):
        j1 = i + lags
        v1 = (j1 >= 0) & (j1 < n)
        c[i, v1] += (1.0 - h[j1[v1]])
        j2 = i - lags
        v2 = (j2 >= 0) & (j2 < n)
        c[i, v2] -= h[j2[v2]]
    return c

def coord_descent_fast(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for idx, i in enumerate(order):
            t = best_h[i]
            c = compute_c_single(i, best_h, n)
            A = corr - t * c
            lo, hi = 0.0, 1.0
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(A + m1 * c) < np.max(A + m2 * c):
                    hi = m2
                else:
                    lo = m1
            t_new = (lo + hi) / 2.0
            if abs(t_new - t) > 1e-12:
                corr += (t_new - t) * c
                best_h[i] = t_new
                if idx % 30 == 29:
                    best_h = project_set(best_h, n)
                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                nc5 = np.max(corr) * dx
                if nc5 < best_c5:
                    best_c5 = nc5
                    improved = True
        best_h = project_set(best_h, n)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        nc5 = np.max(corr) * dx
        if nc5 < best_c5:
            best_c5 = nc5
        if not improved and sweep > 5:
            break
    return best_h, best_c5

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500.0, 500.0)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    obj = (m + np.log(exps.sum()) / T) * dx
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = lags[k]
        if lag >= 0:
            s, e = 0, n - lag
            if e > 0:
                grad[:e] += pk * (1.0 - h[lag:n])
            grad[lag:] += pk * (-h[:e])
        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - h[:n - nl])
            grad[:n - nl] += pk * (-h[nl:])
    return obj, grad * dx

def adam_phase(h, n, t_end, iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.005
    T = 200
    for it in range(iters):
        if time.time() > t_end:
            break
        obj, g = smooth_obj_grad(best_h, T)
        m_s = beta1 * m_s + (1 - beta1) * g
        v_s = beta2 * v_s + (1 - beta2) * g * g
        b1c = 1.0 - beta1 ** (it + 1)
        b2c = 1.0 - beta2 ** (it + 1)
        m_hat = m_s / b1c
        v_hat = v_s / b2c
        trial = best_h - lr * m_hat / (np.sqrt(v_hat) + eps)
        trial = project_set(trial, n)
        c5_t = compute_c5(trial)
        if c5_t < best_c5:
            best_h = trial
            best_c5 = c5_t
            T = min(T * 1.02, 20000.0)
        else:
            lr *= 0.95
    return best_h, best_c5

def binary_local_search(h_frac, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    h = np.zeros(n)
    order = np.argsort(h_frac)
    for idx in range(n // 2):
        h[order[n - 1 - idx]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    C = compute_all_c(h, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_delta = 0
        best_pair = None
        Cz = C[zeros]
        Co = C[ones]
        deltas = Cz[:, None, :] - Co[None, :, :]
        new_maxes = np.max(corr[None, None, :] + deltas, axis=2)
        flat = new_maxes.ravel()
        fi = np.argmin(flat)
        ii = fi // len(ones)
        jj = fi % len(ones)
        if flat[fi] * dx < best_c5 - 1e-14:
            i, j = zeros[ii], ones[jj]
            h[i], h[j] = 1.0, 0.0
            corr += C[i] - C[j]
            for k_idx in range(n):
                d_ii = i + np.arange(n) - (n - 1)
                d_jj = j + np.arange(n) - (n - 1)
                v_ii = (d_ii >= 0) & (d_ii < n)
                v_jj = (d_jj >= 0) & (d_jj < n)
                C[k_idx, v_ii] -= (1.0 - h[d_ii[v_ii]])
                C[k_idx, v_ii] += h[d_ii[v_ii]]
                C[k_idx, v_jj] += (1.0 - h[d_jj[v_jj]])
                C[k_idx, v_jj] -= h[d_jj[v_jj]]
            C = compute_all_c(h, n)
            nc5 = np.max(corr) * dx
            if nc5 < best_c5:
                best_c5 = nc5
                improved = True
        if not improved:
            break
    return h, best_c5

def flatten_phase(h, n, t_end, max_iters=100):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        maxv = np.max(corr)
        active = np.where(corr >= maxv * 0.999)[0]
        n_active = len(active)
        if n_active <= 1:
            break
        contrib = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            c_k = compute_c_single_for_lag(best_h, n, lag)
            contrib += c_k * (corr[k] - maxv * 0.99 + 0.01)
        step = 0.002
        trial = project_set(best_h - step * contrib, n)
        c5_t = compute_c5(trial)
        if c5_t < best_c5:
            best_h = trial
            best_c5 = c5_t
        else:
            step *= 0.5
            trial = project_set(best_h - step * contrib, n)
            c5_t = compute_c5(trial)
            if c5_t < best_c5:
                best_h = trial
                best_c5 = c5_t
    return best_h, best_c5

def compute_c_single_for_lag(h, n, lag):
    c = np.zeros(n)
    if lag >= 0:
        e = n - lag
        if e > 0:
            c[:e] += (1.0 - h[lag:n])
        c[lag:] += (-h[:e])
    else:
        nl = -lag
        c[nl:] += (1.0 - h[:n - nl])
        c[:n - nl] += (-h[nl:])
    return c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 180
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0
    n = len(best_h)
    best_h = project_set(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Fast coordinate descent
    if elapsed() < budget_s * 0.25:
        best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.25, max_sweeps=300)

    # Phase 2: Adam with smooth objective
    if elapsed() < budget_s * 0.45:
        best_h, best_c5 = adam_phase(best_h, n, t0 + budget_s * 0.45, iters=3000)

    # Phase 3: Flatten active constraints
    if elapsed() < budget_s * 0.55:
        best_h, best_c5 = flatten_phase(best_h, n, t0 + budget_s * 0.55, max_iters=200)

    # Phase 4: More coordinate descent
    if elapsed() < budget_s * 0.70:
        best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.70, max_sweeps=400)

    # Phase 5: Binary local search
    if elapsed() < budget_s * 0.80:
        h_bin, c5_bin = binary_local_search(best_h, n, t0 + budget_s * 0.80, max_sweeps=300)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
            # Polish the binary solution fractionally
            best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.85, max_sweeps=100)

    # Phase 6: Try different resolutions
    for n_try in [160, 200, 240]:
        if elapsed() > budget_s * 0.90:

        x_old = np.linspace(0, 2, n, endpoint=False)

    best_h = project_set(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
