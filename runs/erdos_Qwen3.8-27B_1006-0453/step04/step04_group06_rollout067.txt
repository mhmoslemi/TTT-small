import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    t = n * 0.5
    if abs(s - t) < 1e-14:
        return h
    if s < 1e-14:
        return np.full(n, 0.5)
    return np.clip(h * (t / s), 0.0, 1.0)

def spec_flatten(h, n, iters=20):
    h = h.copy()
    for _ in range(iters):
        H = np.fft.rfft(h)
        mg = np.abs(H)
        mg[mg < 1e-15] = 1e-15
        lm = np.log(mg)
        nm = np.exp(0.7 * lm + 0.3 * lm.mean())
        h = project_h(np.fft.irfft(nm * (H / mg), n=n), n)
    return h

def solve1d(base, C, ctr):
    g = np.linspace(0.0, 1.0, 33)
    v = base[None, :] + g[:, None] * C[None, :]
    v[:, ctr] = base[ctr] + g - g * g
    i = v.max(1).argmin()
    lo = g[max(0, i - 1)]
    hi = g[min(32, i + 1)]
    g2 = np.linspace(lo, hi, 129)
    v2 = base[None, :] + g2[:, None] * C[None, :]
    v2[:, ctr] = base[ctr] + g2 - g2 * g2
    return g2[v2.max(1).argmin()]

def coord_desc(h, n, t_end, max_sw=99999):
    ctr = n - 1
    ki = np.arange(2 * n - 1)
    dx = 2.0 / n
    bh = h.copy()
    cr = np.correlate(bh, 1.0 - bh, mode='full')
    bc = cr.max() * dx
    for _ in range(max_sw):
        if time.time() > t_end - 1.0:
            return bh, bc
        for i in np.random.permutation(n):
            ov = bh[i]
            a = i + ki - ctr
            b = i - ki + ctr
            va = (a >= 0) & (a < n)
            vb = (b >= 0) & (b < n)
            hA = np.where(va, bh[np.clip(a, 0, n - 1)], 0.0)
            hB = np.where(vb, bh[np.clip(b, 0, n - 1)], 0.0)
            C = 1.0 - hA - hB
            base = cr - ov * C - hB
            base[ctr] = cr[ctr] - ov + ov * ov
            nv = solve1d(base, C, ctr)
            if abs(nv - ov) > 1e-9:
                bh[i] = nv
                cr = base + nv * C
                cr[ctr] = base[ctr] + nv - nv * nv
        s = bh.sum()
        if abs(s - n * 0.5) > 1e-10:
            bh = project_h(bh, n)
            cr = np.correlate(bh, 1.0 - bh, mode='full')
        c = cr.max() * dx
        if c < bc:
            bc = c
    return bh, bc
