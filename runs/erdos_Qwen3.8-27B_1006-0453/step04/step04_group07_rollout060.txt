def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def _f1d(v, corr, coeff, c0, c0b, old):
    d = v - old
    cn = corr + d * coeff
    cn[c0] = c0b + d * (1.0 - v - old)
    return float(cn.max())

    lo, hi = 0.0, 1.0
    gr = 0.6180339887
    x1 = lo + (1.0 - gr) * (hi - lo)
    x2 = lo + gr * (hi - lo)

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def _f1d(v, corr, coeff, c0, c0b, old):
    d = v - old
    cn = corr + d * coeff
    cn[c0] = c0b + d * (1.0 - v - old)
    return float(cn.max())

def _golden1d(corr, coeff, c0, c0b, old):
    lo, hi = 0.0, 1.0
    gr = 0.6180339887
    x1 = lo + (1.0 - gr) * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = _f1d(x1, corr, coeff, c0, c0b, old)
    f2 = _f1d(x2, corr, coeff, c0, c0b, old)
    for _ in range(30):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + (1.0 - gr) * (hi - lo)
            f1 = _f1d(x1, corr, coeff, c0, c0b, old)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = _f1d(x2, corr, coeff, c0, c0b, old)
    v = (lo + hi) * 0.5
    f = _f1d(v, corr, coeff, c0, c0b, old)
    f0 = _f1d(0.0, corr, coeff, c0, c0b, old)
    f1b = _f1d(1.0, corr, coeff, c0, c0b, old)
    if f0 < f:
        v, f = 0.0, f0
    if f1b < f:
        v, f = 1.0, f1b
    return v, f

def coord_desc(h_in, t_end, max_sweeps=5000):
    n = len(h_in)
    dx = 2.0 / n
    h = h_in.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    bc5 = float(corr.max()) * dx
    c0 = n - 1
    sr = np.arange(-(n - 1), n)
    rem = max_sweeps
    while rem > 0 and time.time() < t_end:
        rem -= 1
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return h, float(corr.max()) * dx
            old = h[i]
            ip = i + sr
            im = i - sr
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            cf = np.zeros(2 * n - 1)
            cf[vp] += 1.0 - h[ip[vp]]
            cf[vm] -= h[im[vm]]
            c0b = corr[c0]
            bv, bf = _golden1d(corr, cf, c0, c0b, old)
            if bf * dx < bc5 - 1e-15:
                d = bv - old
                corr += d * cf
                corr[c0] = c0b + d * (1.0 - bv - old)
                h[i] = bv
                bc5 = bf * dx
                imp = True
        if not imp:
            break
    return h, float(corr.max()) * dx

def smooth_grad_step(h, n, T, learn_rate, c5_target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    g = np.zeros(n)
    sr = np.arange(-(n - 1), n)
    for i in range(n):
        ip = i + sr
        im = i - sr
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        cf = np.zeros(2 * n - 1)
        cf[vp] += 1.0 - h[ip[vp]]
        cf[vm] -= h[im[vm]]
        g[i] = np.dot(w, cf)
    lr = learn_rate * (bc5 - c5_target)
    h_new = h - lr * g
    h_new = project_feasible(h_new, n)
    return h_new

def interp_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
