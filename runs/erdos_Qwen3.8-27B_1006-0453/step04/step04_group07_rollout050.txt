import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fft_grad(h, w, n):
    N = 1
    while N < 4 * n:
        N <<= 1
    h_pad = np.zeros(N)
    h_pad[:n] = h
    w_pad = np.zeros(N)
    for k in range(-(n - 1), n):

    B = np.fft.irfft(W * H, N)[:n]
    return 1.0 - A - B

def grad_descent(h0, n, t_end, T_start=20, T_end=5000, max_iters=5000):

    best_h = h0.copy()

    t0 = time.time()

        if time.time() > t_end:
            break
        frac = min(1.0, it / max(1, max_iters))
        T = T_start + (T_end - T_start) * (frac ** 2)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        actual_c5 = m * dx
        if actual_c5 < best_c5:

        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[n - 1] = 1.0
        else:
            w /= w_sum
        g = fft_grad(h, w, n)
        step = 0.5 / (1.0 + T * 0.1)
        h = project(h - step * g, target)
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')

    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    t0 = time.time()
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            # Find best v for coordinate i
            # corr_new[m] = corr[m] + (v - old) * coeff[m]

            v0_sc[c0_idx] = c0_base + (0.0 - old) * (1.0 - 0.0 - old)

            best_f = np.max(v0_sc)
            f0 = np.max(v0_sc)
            f1 = np.max(v1_sc)
            if f0 < best_f:

            if f1 < best_f:
                best_v, best_f = 1.0, f1

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = 0.0, 1.0
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            for _ in range(40):
