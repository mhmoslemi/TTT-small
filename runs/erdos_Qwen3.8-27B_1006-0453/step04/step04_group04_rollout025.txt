def proj(h, n):

    diff = target - s

    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        # Proportional distribution
        alloc = room * (diff / tot)
        h += np.minimum(alloc, room)
        # Fix any remaining

        diff2 = target - s2
        if diff2 > 1e-12:

            tot2 = room2.sum()
            if tot2 > 1e-13:

            return h
        # Proportional reduction
        h *= (target / s)

    corr = np.correlate(h, 1.0 - h, mode='full')

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov with moderate T
    if elapsed() < budget_s * 0.12:
        h1, c1 = nesterov_opt(best_h, n, t0 + budget_s * 0.06, T_init=30, T_mult=1.015, T_max=8000, lr_init=0.04)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    if elapsed() < budget_s * 0.20:
        h1, c1 = nesterov_opt(best_h, n, t0 + budget_s * 0.06, T_init=800, T_mult=1.01, T_max=80000, lr_init=0.015)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Coordinate descent
    if elapsed() < budget_s * 0.32:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.32, max_sweeps=400)

    # Phase 3: Active-lag optimization + more Nesterov at high T
    if elapsed() < budget_s * 0.38:
        h1, c1 = active_lag_opt(best_h, n, t0 + budget_s * 0.04, tol=0.999)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    if elapsed() < budget_s * 0.46:
        h1, c1 = nesterov_opt(best_h, n, t0 + budget_s * 0.06, T_init=5000, T_mult=1.008, T_max=200000, lr_init=0.008)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 4: Coordinate descent again
    if elapsed() < budget_s * 0.55:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.55, max_sweeps=500)

    # Phase 5: Basin hopping with structured perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 200
    for it in range(n_bh):
        if elapsed() > budget_s * 0.78:
            break
        cool = 1.0 - 0.88 * (it / float(n_bh))
        h_pert = structured_perturb(work_h, n, rng)
        t_local = min(elapsed() + 6.0, budget_s * 0.78)
        h_opt, c5_opt = nesterov_opt(h_pert, n, t_local, T_init=300, T_mult=1.02, T_max=40000, lr_init=0.02)
        t_cd = min(elapsed() + 3.0, budget_s * 0.78)
        h_opt, c5_opt = coord_descent(h_opt, n, t_cd, max_sweeps=40)
        temp = 0.0025 * cool + 0.0001
        if c5_opt < work_c5 or rng.random() < np.exp(-(c5_opt - work_c5) / max(temp, 1e-10)):
            work_h = h_opt
            work_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    if work_c5 < best_c5:
        best_h, best_c5 = work_h, work_c5

    # Phase 6: Try different resolutions
    for n_try in [160, 200, 216]:
        if elapsed() > budget_s * 0.88:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n_try)
        h_opt, c5_opt = nesterov_opt(h_interp, n_try, t0 + min(elapsed() + 20, budget_s * 0.88),
                                     T_init=500, T_mult=1.015, T_max=50000, lr_init=0.02)
        h_opt, c5_opt = coord_descent(h_opt, n_try, t0 + min(elapsed() + 10, budget_s * 0.88), max_sweeps=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n = n_try

    # Phase 7: Final coordinate descent + active lag + Nesterov polish
    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.93, max_sweeps=600)
    if elapsed() < budget_s * 0.96:
        h1, c1 = active_lag_opt(best_h, n, t0 + budget_s * 0.02, tol=0.9995)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    if elapsed() < budget_s * 0.99:
        h1, c1 = nesterov_opt(best_h, n, t0 + budget_s * 0.015, T_init=50000, T_mult=1.003, T_max=500000, lr_init=0.003)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    if elapsed() < budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=200)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
