    c = linearized_coeffs(h, n, active)
    
    # LP: min c·h s.t. lb ≤ h ≤ ub, sum(h) = n/2
    lb = np.maximum(0, h - epsilon)
    ub = np.minimum(1, h + epsilon)
    # Ensure lb ≤ ub is feasible with sum = n/2
    if lb.sum() > n/2 + 1e-10 or ub.sum() < n/2 - 1e-10:

    b_eq = np.array([n/2.0])

    if res.success:

    return h

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / s)
    else:
        diff = target - s
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * (target / max(s, 1e-14)), 0, 1)
        h = h + room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def lin_coeffs(h, n, lags):
    c = np.zeros(n)
    center = n - 1

        k = li - center

                c[i - m] += (1.0 - h[i])
                c[i] -= h[i - m]
    return c

import numpy as np
from scipy.optimize import minimize_scalar, minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / s)
    else:
        diff = target - s
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * (target / max(s, 1e-14)), 0, 1)
        h = h + room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def lin_coeffs(h, n, lags):
    c = np.zeros(n)
    center = n - 1
    for li in lags:
        k = li - center
        if k == 0:
            c[:] += (1.0 - 2.0 * h)
        elif k > 0:
            for i in range(n - k):
                c[i + k] += (1.0 - h[i])
                c[i] -= h[i + k]
        else:
            m = -k
            for i in range(m, n):
                c[i - m] += (1.0 - h[i])
                c[i] -= h[i - m]
    return c

import numpy as np
from scipy.optimize import minimize_scalar, minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / max(s, 1e-14))
    else:
        diff = target - s
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * (target / max(s, 1e-14)), 0, 1)
        h = h + room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def lin_coeffs(h, n, lags):
    c = np.zeros(n)
    center = n - 1
    for li in lags:
        k = li - center
        if k == 0:
            c[:] += (1.0 - 2.0 * h)
        elif k > 0:
            for i in range(n - k):
                c[i + k] += (1.0 - h[i])
                c[i] -= h[i + k]
        else:
            m = -k
            for i in range(m, n):
                c[i - m] += (1.0 - h[i])
                c[i] -= h[i - m]
    return c

def lp_step(h, n, eps, n_active=5):
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    thresh = m - 0.06 * n
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([center])
    active = active[np.argsort(corr[active])[::-1][:n_active]]
    c = lin_coeffs(h, n, active)
    lb = np.maximum(0.0, h - eps)
    ub = np.minimum(1.0, h + eps)
    if lb.sum() > n / 2.0 + 1e-9 or ub.sum() < n / 2.0 - 1e-9:
        return h
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    try:
        res = linprog(c, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lb, ub)), method='highs')
        if res.success:
            return project(res.x, n)
    except Exception:
        pass
    return h

def lp_iterate(h, n, t_end, eps0=0.3):
    best_c5 = compute_c5(h)
    eps = eps0
    stall = 0
    while time.time() < t_end:
        h_new = lp_step(h, n, eps, n_active=5)
        if np.allclose(h_new, h, atol=1e-14):
            h_new = lp_step(h, n, eps * 1.5, n_active=12)
        c5v = compute_c5(h_new)
        if c5v < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5v
            eps = min(eps * 1.03, 0.5)
            stall = 0
        else:
            eps *= 0.82
            stall += 1
            if stall > 25 and eps < 0.008:
                break
    return h, best_c5

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        a[center + k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[center - k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)

    def f1d(x):

def coord_descent(h, n, t_end, max_sweeps=200):
    best_c5 = compute_c5(h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        for i in range(n):
            if time.time() > t_end:
                return h, best_c5
            x_new = coord_opt(h, n, i)
            h[i] = x_new
            h = project(h, n)
            c5v = compute_c5(h)
            if c5v < best_c5 - 1e-13:
                best_c5 = c5v
    return h, best_c5

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    dx = 2.0 / len(h)
    return (mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T) * dx

def corr_grad_smooth(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    w = np.exp(T * (corr - mx))
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(len(corr)):
        wi = w[idx] * dx
        if wi < 1e-15:
            continue
        k = idx - center
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
        else:
            m = -k
            grad[m:] += wi * (1.0 - h[:n - m])
            grad[:n - m] -= wi * h[m:]
    return grad

def lgbfgs_opt(h, T, n, maxiter=400):
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_c5(x, T) + 500.0 * (x.sum() - n / 2.0) ** 2
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        g = corr_grad_smooth(x, T) + 1000.0 * (x.sum() - n / 2.0)
        return g
    res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return project(np.clip(res.x, 0, 1), n)

def fourier_h(coeffs, n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    harmonic_freqs = [np.pi, 2 * np.pi, 3 * np.pi, 4 * np.pi]
    for idx, f in enumerate(harmonic_freqs):
        if 2 * idx < len(coeffs):
            h += coeffs[2 * idx] * np.cos(f * x) + coeffs[2 * idx + 1] * np.sin(f * x)
    return np.clip(h, 0, 1)

def fourier_opt(n, t_end):

    best_h = None
